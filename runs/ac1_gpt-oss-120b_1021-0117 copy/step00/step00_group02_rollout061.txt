        else:
            # Pareto(a) + 1 has mean a/(a-1) for a>1; we use a=2.5
            base = rng.pareto(a=2.5, size=n) + 1.0

        # descending order – large values will be placed far apart later
        base = np.sort(base)[::-1]

        # ----------- choose a step that is coprime to n (odd works) -----
        step = int(rng.integers(1, n))
        if step % 2 == 0:
            step += 1
        if step >= n:
            step = n - 1 if (n - 1) % 2 == 1 else n - 2
        # ensure coprime (for even n any odd step works)
        while math.gcd(step, n) != 1:
            step = (step + 2) % n
            if step == 0:
                step = 1

        # ----------- build the anti‑aligned permutation ----------------
        idx = (np.arange(n, dtype=np.int64) * step) % n
        anti = base[idx]

        # Rescale anti so that its L1‑norm equals n (average value 1)
        total = np.sum(anti)
        if total == 0.0:
            continue
        anti = anti * (n / total)

        # ----------- evaluate pure anti vector ------------------------
        score_anti = _metric(anti, n)
        if score_anti < best_score:
            best = anti.copy()
            best_score = score_anti
            if best_score < target_score:
                break

        # ----------- blend a few random alphas with the uniform vector
        for _ in range(3):
            alpha = rng.random()          # 0 … 1
            # candidate = (1‑α)·uniform + α·anti   (average stay = 1)
            cand = uniform + alpha * (anti - uniform)
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _metric(cand, n)
            if sc < best_score:
                best = cand.copy()
                best_score = sc
                if best_score < target_score:
                    break

        if best_score < target_score:
            break

        # ----------- occasional cheap local perturbation of the best ----
        # (same idea as the reference implementation)
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        cand = best.copy()
        cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _metric(cand, n)
        if sc < best_score:
            best = cand.copy()
            best_score = sc
            if best_score < target_score:
                break

    # ------------------------------------------------------------------
    #  Return the best sequence found (as a plain Python list of floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
```
