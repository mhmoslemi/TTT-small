    while time left > small:

            coeffs = f[k - js]

import time
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Helper: convolution of a real non‑negative vector with itself
# (fast via FFT, O(n log n) memory‑free)
# ------------------------------------------------------------
def _self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 2 * n - 1
    fft_len = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, fft_len)
    full = np.fft.irfft(A * A, fft_len)[:size]
    return full


# ------------------------------------------------------------
# Helper: evaluate a candidate using the authoritative evaluator
# ------------------------------------------------------------
def _score_of(a: np.ndarray) -> float:
    # a is already clipped to [0,1000] and finite
    return float(evaluate_sequence(a.tolist()))    # the supplied evaluator


# ------------------------------------------------------------
# Linear‑programming step:
# maximise   sum(g)
# subject to   f * g (at selected lags) ≤ M
#              0 ≤ g ≤ 1000
# ------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float, active_lags, max_val: float = 1000.0):
    n = f.size
    rows = []
    b = []
    for k in active_lags:
        # coefficients for this lag: g_j appears with weight f_{k-j}
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo > hi:                     # empty interval, skip
            continue
        coeff = f[k - np.arange(lo, hi + 1, dtype=int)]
        if not np.any(coeff):           # all zero – constraint is void
            continue
        row = np.zeros(n, dtype=np.float64)
        row[lo:hi + 1] = coeff
        rows.append(row)
        b.append(M)
    if not rows:                       # no effective constraints – we can fill everything
        g = np.full(n, max_val, dtype=np.float64)
        return g

    A = np.vstack(rows)
    b = np.asarray(b, dtype=np.float64)

    # maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, max_val)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if not res.success:
        # fallback – return zeros so the caller will just keep the old sequence
        return np.zeros_like(f)
    return res.x


# ------------------------------------------------------------
# Simple block‑perturbation used as a cheap fallback
# ------------------------------------------------------------
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= np.exp(rng.normal(0.0, 0.12))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

        g = _solve_lp(f, M, active, max_val=1000.0)

        # if the LP gave nothing useful, fall back to a cheap random tweak
        if g.sum() <= f.sum() * (1.0 + 1e-12):
            cand = _make_block_perturbation(f, rng)
            sc = _score_of(cand)
            if sc < best_score:
                best, best_score = cand.copy(), sc
                f = cand
            continue

        # 4) blend a fraction of g into the current sequence.
        #    Several step sizes are tried; the best improvement is kept.
        improved = False
        for t in (0.15, 0.30, 0.45, 0.60):
            new_f = (1.0 - t) * f + t * g
            np.clip(new_f, 0.0, 1000.0, out=new_f)
            sc = _score_of(new_f)
            if np.isfinite(sc) and sc < best_score:
                best, best_score = new_f.copy(), sc
                f = new_f
                improved = True
                break          # accept the first successful step size

        if not improved:
            # no step size helped – try a cheap random perturbation instead
            cand = _make_block_perturbation(f, rng)
            sc = _score_of(cand)
            if sc < best_score:
                best, best_score = cand.copy(), sc
                f = cand

        # early exit if we already beat the target bound
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
