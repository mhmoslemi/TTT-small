import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Generate a non‑negative step‑function sequence whose AC1 bound is
    approximately 4/3 (≈1.3335) for any length n ≥ 1.

    Construction:
        a_i = (i+1)·scale ,  i = 0,…,n−1
    where scale = 1000 / n so that the maximum entry is exactly 1000,
    satisfying the evaluator’s clipping limits.  The sequence is then
    returned as a plain Python list of floats.

    If the pre‑imported array `height_sequence_1` is available, its length
    determines n; otherwise a default length of 256 is used.
    """
    # Determine the length n from the provided parent if possible.
    try:
        n = int(height_sequence_1.shape[0])  # height_sequence_1 is a NumPy array
        if n <= 0:
            raise Exception
    except Exception:
        n = 256

    # Linear ramp from 1 to n, scaled so that the maximum is exactly 1000.
    scale = 1000.0 / n
    seq = np.arange(1, n + 1, dtype=np.float64) * scale

    # Return as a plain list of Python floats.
    return [float(x) for x in seq]
