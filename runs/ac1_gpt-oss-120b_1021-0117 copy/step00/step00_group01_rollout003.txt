import numpy as np, time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)  # some buffer
    
    # Determine base length
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise Exception
    except Exception:
        base = np.ones(256, dtype=np.float64)
    n = base.size
    
    # Start with uniform candidate
    uniform = np.full(n, 1.0, dtype=np.float64)
    best = uniform
    best_score = float(evaluate_sequence(best.tolist()))
    
    # Evaluate parent as fallback
    parent = np.clip(base, 0.0, 1000.0)
    parent_score = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(parent_score) and parent_score < best_score:
        best = parent
        best_score = parent_score
    
    # Simple random perturbations within remaining budget
    while time.monotonic() < deadline:
        # generate small random perturbation noise on uniform
        noise = rng.normal(loc=0.0, scale=0.1, size=n)
        cand = uniform * np.exp(noise)  # multiplicative perturb
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            uniform = cand  # update base for future perturbations
    # Return as python list
    return [float(x) for x in best]

    # Determine the length to use
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
    except Exception:
        # Fallback when no parent is provided
        base = np.ones(256, dtype=np.float64)

    n = base.size

    # Uniform sequence (any positive constant works, 1.0 is convenient)
    candidate = np.full(n, 1.0, dtype=np.float64)

    # Clip to the evaluator's allowed range (redundant for 1.0)
    candidate = np.clip(candidate, 0.0, 1000.0)

    # Return as a plain Python list of floats
    return [float(x) for x in candidate]
