def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.time() + max(0.0, float(budget_s) - 0.5)
    # get parent if defined
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim !=1 or parent.size==0 or not np.isfinite(parent).all():
            raise Exception
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = None
    if parent is None:
        n = 3809
        # start uniform
        a = np.full(n, 1.0, dtype=np.float64)
    else:
        a = parent.copy()
    # ensure sum >0.01
    if a.sum() < 0.01:
        a = np.ones_like(a) * (0.01 / a.size)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # define small functions
    n = best.size
    # main loop
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # Pick method probabilistically
        r = rng.random()
        if r < 0.4:
            # block scaling similar to previous but adjust scale factor
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand = best.copy()
            scale = float(np.exp(rng.normal(0.0, 0.1)))
            cand[start:stop] *= scale
        elif r < 0.7:
            # swap two entries
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer mass
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            cand = best.copy()
            amount = cand[i] * rng.random() * 0.5  # move up to 50% of source
            cand[i] -= amount
            cand[j] += amount
        # clip and ensure bounds
        np.clip(cand, 0.0, 1000.0, out=cand)
        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
    return [float(x) for x in best]

import numpy as np
import time

def _eval(arr):
    """wrapper around the provided evaluator that returns a float."""
    return float(evaluate_sequence(arr.tolist()))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step function minimizing the AC1 bound.

    The algorithm builds a variety of parametric candidates (exponential,
    power‑law, linear, sparse random) and keeps the best one seen.
    Afterwards a simple hill‑climbing / simulated‑annealing loop
    explores the neighbourhood of the current best while staying within
    O(n) memory.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # initialise length and a fallback starting vector
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        parent = np.clip(parent, 0.0, 1000.0)
        n = parent.size
    except Exception:
        n = 3809
        parent = np.ones(n, dtype=np.float64)

    # Normalise only to avoid a zero sum; scaling does not affect the score.
    if parent.sum() < 0.01:
        parent = np.ones_like(parent) * (0.01 / parent.size)
    base = parent / parent.sum()                # sum = 1

    best = base.copy()
    best_score = _eval(best)

    # ------------------------------------------------------------------
    # helper to try a candidate and update the best one
    # ------------------------------------------------------------------
    def _try(x):
        nonlocal best, best_score
        if x.shape != (n,):
            return
        np.clip(x, 0.0, 1000.0, out=x)
        if x.sum() < 0.01:       # keep feasible
            return
        sc = _eval(x)
        if np.isfinite(sc) and sc < best_score:
            best = x.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # generate a diverse set of parametric candidates
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # exponential decays
    for alpha in np.geomspace(1e-5, 0.02, num=30):
        a = np.exp(-alpha * idx)
        a = a / a.sum()
        _try(a)
        # a couple of random shuffles of the same multiset
        for _ in range(2):
            perm = rng.permutation(n)
            _try(a[perm])

    # power‑law decays
    for p in np.linspace(0.0, 5.0, num=30):
        a = (idx + 1.0) ** (-p)
        s = a.sum()
        if s == 0:
            continue
        a = a / s
        _try(a)
        perm = rng.permutation(n)
        _try(a[perm])

    # linear taper (including the triangular case)
    a = np.linspace(1.0, 0.0, n)
    a = a / a.sum()
    _try(a)
    for _ in range(3):
        _try(a[rng.permutation(n)])

    # sparse random supports – equal weights
    for m in (50, 70, 90, 110, 130):
        if m >= n:
            continue
        sel = rng.choice(n, size=m, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[sel] = 1.0 / m
        _try(a)

    # sparse random supports – random positive weights
    for m in (70, 100, 120):
        if m >= n:
            continue
        sel = rng.choice(n, size=m, replace=False)
        vals = rng.random(m)
        vals /= vals.sum()
        a = np.zeros(n, dtype=np.float64)
        a[sel] = vals
        _try(a)

    # Dirichlet‑distributed full vectors (few draws)
    for _ in range(5):
        a = rng.random(n)
        a = a / a.sum()
        _try(a)

    # ------------------------------------------------------------------
    # hill‑climbing / simulated‑annealing phase
    # ------------------------------------------------------------------
    deadline = time.time() + max(0.0, float(budget_s) - 0.5)
    no_improve = 0
    temperature = best_score if best_score > 0 else 1.0

    while time.time() < deadline:
        # choose a random perturbation type
        r = rng.random()
        cand = best.copy()

        if r < 0.40:                      # block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            scale = float(np.exp(rng.normal(0.0, 0.09)))
            cand[start:stop] *= scale

        elif r < 0.70:                    # transfer a fraction of one entry to another
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and cand[i] > 0.0:
                delta = cand[i] * rng.random() * 0.4
                cand[i] -= delta
                cand[j] += delta

        else:                             # swap two entries
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            cand[i], cand[j] = cand[j], cand[i]

        # occasional global scramble
        if rng.random() < 0.01:
            cand = cand[rng.permutation(n)]

        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _eval(cand)
        if np.isfinite(sc):
            accept = False
            if sc < best_score:
                accept = True
                no_improve = 0
            else:
                # simulated‑annealing acceptance for occasional uphill moves
                prob = np.exp((best_score - sc) / max(temperature, 1e-12))
                if rng.random() < prob:
                    accept = True
                    no_improve += 1
                else:
                    no_improve += 1
            if accept:
                best = cand
                best_score = sc
                temperature *= 0.9999  # slowly cool

        # restart if we have been stuck for a while
        if no_improve > 800:
            # fresh random exponential candidate (shuffled)
            alpha = rng.uniform(1e-5, 0.02)
            a = np.exp(-alpha * idx)
            a = a / a.sum()
            a = a[rng.permutation(n)]
            _try(a)
            no_improve = 0
            temperature = best_score

    # ------------------------------------------------------------------
    # return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
