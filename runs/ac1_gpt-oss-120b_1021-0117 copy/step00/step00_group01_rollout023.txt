import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Find a non‑negative step function with a low AC1 upper bound.

    The search builds random sparse 0/1 vectors of the same length as
    `height_sequence_1` (or a default length) and keeps the best one
    according to `evaluate_sequence`.  The density of ones is varied
    randomly between 20 % and 80 % to explore a wide range of
    cardinalities.  The algorithm stops early if the target bound
    (1.50274349) is reached.
    """
    rng = np.random.default_rng(seed)
    # reserve a few seconds for returning the result cleanly
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # initialise fallback (the supplied parent, if any)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
        )
    else:
        # sensible default when the parent is missing
        parent = np.ones(1024, dtype=np.float64)

    n = parent.size
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # random sparse search
    # ------------------------------------------------------------------
    target_score = 1.50274349
    MIN_DENS = 0.20   # at least 20 % of the entries are 1
    MAX_DENS = 0.80   # at most 80 % are 1

    # pre‑allocate a zero vector to avoid repeated allocation of the same shape
    zero_vec = np.zeros(n, dtype=np.float64)

    while time.monotonic() < deadline:
        # pick a random density and build a 0/1 mask
        dens = rng.random() * (MAX_DENS - MIN_DENS) + MIN_DENS
        mask = rng.random(n) < dens
        if not mask.any():
            continue   # avoid the degenerate all‑zero case

        # candidate = mask.astype(float)
        candidate = zero_vec.copy()
        candidate[mask] = 1.0

        # evaluate and possibly accept
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            if best_score < target_score:
                break    # good enough – stop early

    # ------------------------------------------------------------------
    # return as a plain Python list of floats
    # ------------------------------------------------------------------
    return [float(x) for x in best]
