def project_simplex(v):
    # Projects onto simplex {x >=0, sum=1}
    # Using algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def subgradient_step(a, idx):
    n = len(a)
    g = np.zeros_like(a)
    # compute overlap range
    lo = max(0, idx - (n-1))
    hi = min(n-1, idx)
    # i from lo..hi
    # j = idx - i
    # Use vectorized:
    i = np.arange(lo, hi+1)
    j = idx - i
    g[i] = 2 * a[j]
    return g

import numpy as np, time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    # Determine length
    try:
        n = int(len(height_sequence_1))
        use_parent = True
    except NameError:
        # fallback to a moderate length if parent not provided
        n = 2000
        use_parent = False
    # arcsine distribution
    i = np.arange(n, dtype=np.float64)
    # compute weights, avoid division by zero: (i+0.5)*(n-0.5-i) > 0 for all i
    w = 1.0 / np.sqrt( (i + 0.5) * (n - 0.5 - i) )
    a = w / w.sum()
    # ensure sum slightly > 0.01 (it will be 1)
    best = a.copy()
    # compute internal objective (score)
    conv = np.convolve(a, a)
    best_score = 2.0 * n * conv.max() / (a.sum()**2)

    iteration = 0
    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        max_val = conv.max()
        max_idx = conv.argmax()
        # gradient for this max index
        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        j = max_idx - idx_range
        g = np.zeros_like(a)
        g[idx_range] = 2.0 * a[j]

        # step size schedule
        step = 1.0 / np.sqrt(iteration + 1.0)
        # mirror descent update (positive weights)
        a = a * np.exp(-step * g)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = rng.random(n)
            a_sum = a.sum()
        a = a / a_sum

        # evaluate new score
        conv_new = np.convolve(a, a)
        score = 2.0 * n * conv_new.max() / (a.sum()**2)

        if score < best_score:
            best_score = score
            best = a.copy()

        # occasional random jitter to escape flat regions
        if iteration % 200 == 0 and iteration > 0:
            jitter = rng.random(n) * 0.001
            a = (a + jitter)
            a = a / a.sum()

        iteration += 1

    # produce final list, ensure bounds
    out = np.clip(best, 0.0, 1000.0)
    # Guarantee sum >0.01
    if out.sum() < 0.01:
        out = out + (0.01 / out.size)
        out = out / out.sum() * 0.01
    return [float(x) for x in out]

import numpy as np
import time

def _project_simplex(v):
    """Project a vector onto the probability simplex (non‑negative, sum=1)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a non‑negative step‑function that (approximately) minimises
    2·n·max(conv(a,a))/sum(a)^2  (the AC1 upper bound).

    The construction is a discretised arcsine law, whose self‑convolution
    tends to a uniform distribution.  A few mirror‑descent steps are added
    to further flatten the convolution peak.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # length of the problem – use the supplied parent if it exists

    # ----- 1.  Initialise with a discretised arcsine density -----
    i = np.arange(n, dtype=np.float64)

    w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = w / w.sum()                 # a is a probability vector (sum≈1)

    # keep the best solution found
    conv = np.convolve(a, a)
    best = a.copy()
    best_score = 2.0 * n * conv.max() / (a.sum() ** 2)

    iteration = 0
    # ----- 2.  Mirror‑descent loop (time‑bounded) -----
    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        max_val = conv.max()
        max_idx = conv.argmax()

        # sub‑gradient of the active convolution entry
        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        j = max_idx - idx_range
        g = np.zeros_like(a)
        g[idx_range] = 2.0 * a[j]

        # step size decays like 1/√t
        step = 1.0 / np.sqrt(iteration + 1.0)

        a = a * np.exp(-step * g)
