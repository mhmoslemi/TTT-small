def compute_score(a):
    sum_a = a.sum()
    # compute convolution using FFT
    L = 2*n - 1
    # Choose next pow2 for efficient FFT
    m = int(2**np.ceil(np.log2(L)))
    fft_a = np.fft.rfft(a, m)
    conv = np.fft.irfft(fft_a*fft_a, m)[:L]
    max_b = conv.max()
    return 2 * n * max_b / (sum_a**2)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    # get n
    n = int(np.asarray(height_sequence_1).shape[0]) if 'height_sequence_1' in globals() else 256
    # base families:
    best = None
    best_score = math.inf
    # Helper: evaluate
    def eval_seq(arr):
        s = arr.sum()
        if s == 0:
            return math.inf
        L = arr.shape[0]*2 -1
        m = 1 << (int(math.ceil(math.log2(L))))
        fft = np.fft.rfft(arr, m)
        conv = np.fft.irfft(fft*fft, m)[:L]
        maxb = conv.max()
        return 2 * n * maxb / (s*s)
    # Baseline linear decreasing:
    lin = np.linspace(1.0, 0.0, n, dtype=np.float64)
    score = eval_seq(lin)
    if score < best_score:
        best, best_score = lin, score
    # Add exponential candidates
    lams = np.logspace(-5, -2.5, num=30) # from 1e-5 to ~0.003
    for lam in lams:
        if time.monotonic() > deadline:
            break
        arr = np.exp(-lam * np.arange(n, dtype=np.float64))
        score = eval_seq(arr)
        if score < best_score:
            best, best_score = arr, score
    # Power law candidates (1 - i/(n-1))**p
    p_vals = np.linspace(0.2, 8.0, 30)
    idx = np.arange(n, dtype=np.float64)
    base = 1.0 - idx/(n-1)
    for p in p_vals:
        if time.monotonic() > deadline:
            break
        arr = np.power(base, p)
        score = eval_seq(arr)
        if score < best_score:
            best, best_score = arr, score
    # Local random perturbations of best (few)
    while time.monotonic() < deadline:
        # generate perturbation: multiply a random block by random factor
        candidate = best.copy()
        width = int(rng.integers(max(1, n // 500), max(2, n // 20)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        factor = float(np.exp(rng.normal(0.0, 0.08)))
        candidate[start:start+width] *= factor
        # optionally smooth a bit by averaging with reversed copy
        if rng.random() < 0.2:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = eval_seq(candidate)
        if score < best_score:
            best, best_score = candidate, score
    # Return best scaled as needed (any scaling okay)
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

import numpy as np
import time
import math

def _conv_score(arr: np.ndarray, n: int) -> float:
    """Return the exact AC1 bound for a non‑negative array, using FFT."""
    s = arr.sum()
    if s == 0.0:
        return math.inf
    L = arr.shape[0] * 2 - 1
    # power‑of‑two length for a fast FFT
    m = 1 << (int(math.ceil(math.log2(L))))
    fft = np.fft.rfft(arr, m)
    conv = np.fft.irfft(fft * fft, m)[:L]
    maxb = conv.max()
    return 2.0 * n * maxb / (s * s)
