import numpy as np
import time

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step function that minimizes the AC1 bound.
    The search explores simple parametric families (exponential and power‑law
    decays) and applies a few cheap local perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Base sequence (fallback)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
    except Exception:
        # no parent supplied – use a modest default
        parent = np.ones(256, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    n = int(parent.shape[0])

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate (fast wrapper around the authoritative one)
    # ------------------------------------------------------------------
    def eval_score(arr: np.ndarray) -> float:
        """Return the official score for a 1‑D numpy array."""
        # The evaluator expects a Python list, but passing a list of floats is cheap.
        return float(evaluate_sequence(arr.tolist()))

    best = parent.copy()
    best_score = eval_score(best)

    # ------------------------------------------------------------------
    # Parametric families ------------------------------------------------
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # 1) Exponential decay: a_i = exp(-beta * i)
    for beta in np.linspace(0.0, 0.0005, 51):          # 0 also gives the uniform case
        if beta == 0.0:
            cand = np.ones(n, dtype=np.float64)
        else:
            cand = np.exp(-beta * idx)
        score = eval_score(cand)
        if score < best_score:
            best = cand.copy()
            best_score = score

    # 2) Power‑law decay: a_i = 1 / (i+1)^alpha
    for alpha in np.linspace(0.0, 2.5, 51):
        if alpha == 0.0:
            cand = np.ones(n, dtype=np.float64)
        else:
            cand = 1.0 / np.power(idx + 1.0, alpha)
        score = eval_score(cand)
        if score < best_score:
            best = cand.copy()
            best_score = score

    # 3) Symmetric exponential (peak at the centre, decay outwards)
    centre = (n - 1) / 2.0
    for beta in np.linspace(0.0, 0.0005, 51):
        if beta == 0.0:
            cand = np.ones(n, dtype=np.float64)
        else:
            cand = np.exp(-beta * np.abs(idx - centre))
        score = eval_score(cand)
        if score < best_score:
            best = cand.copy()
            best_score = score

    # ------------------------------------------------------------------
    # Light local search (block perturbations) while time permits
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # choose a random block and scale it a little
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        cand = best.copy()
        # multiplicative jitter (log‑normal)
        factor = float(np.exp(rng.normal(0.0, 0.07)))
        cand[start:start + width] *= factor
        # occasional mirroring to explore symmetric solutions
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])
        cand = np.clip(cand, 0.0, 1000.0)
        score = eval_score(cand)
        if score < best_score:
            best = cand
            best_score = score

    # Return as a plain Python list of floats
    return [float(v) for v in best]
