    # We repeatedly pick the k where the convolution attains its maximum,
    # locate the pair (i, k-i) that contributes the most to this value,
    # and move a fraction of mass from i to another index j that does not
    # participate in this sum (or participates only weakly).  The convolution
    # is updated incrementally in O(n) time.
    # ------------------------------------------------------------------
    max_iters = 200_000          # absolute safety cap
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        k_max = int(np.argmax(b))

        # range of i that actually contributes to b[k_max]
        start = max(0, k_max - (n - 1))
        end = min(k_max, n - 1)

        # product contribution a[i] * a[k_max-i] for each i in the range
        idxs = np.arange(start, end + 1, dtype=np.int64)
        partners = k_max - idxs
        prod = a[idxs] * a[partners]

        # i with the strongest contribution
        i = int(start + np.argmax(prod))

        # choose a partner j that does NOT affect this k_max if possible
        if start > 0:
            j = int(rng.integers(0, start))
        elif end < n - 1:
            j = int(rng.integers(end + 1, n))
        else:
            # full overlap – pick index with minimal contribution
            all_idx = np.arange(n, dtype=np.int64)
            all_partner = k_max - all_idx
            mask = (all_partner >= 0) & (all_partner < n)
            prod_all = np.zeros(n, dtype=np.float64)
            prod_all[mask] = a[all_idx[mask]] * a[all_partner[mask]]
            prod_all[i] = np.inf                     # exclude i itself
            j = int(np.argmin(prod_all))

        if j == i:
            continue

        # move a random fraction (up to 50%) of the mass from i to j
        mass_i = a[i]
        if mass_i <= 0.0:
            continue
        fraction = rng.random() * 0.5                # in (0,0.5)
        delta = -mass_i * fraction                   # change for a[i] (negative)

        # incremental update of the convolution vector
        b_new = b.copy()
        # contribution of the change at i
        b_new[i:i + n] += 2.0 * delta * a
        # contribution of the opposite change at j
        b_new[j:j + n] -= 2.0 * delta * a
        # diagonal terms
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        # cross term (i,j)
        cross_idx = i + j
        b_new[cross_idx] += -2.0 * delta * delta

        # compute new bound (total sum stays 1 up to round‑off)
        new_max = b_new.max()
        new_score = 2.0 * n * new_max / (total * total)

        if new_score < best_score:
            # accept move
            a[i] += delta
            a[j] -= delta
            b = b_new
            best_score = new_score
            best_seq = a.copy()
            # optional early stop if already very good
            if best_score < 1.5:
                # keep searching a little longer for possible further gain
                pass

    # --------------------------------------------------------------
    # Final safety adjustments and return
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    tot = best_seq.sum()
    if tot < 0.01:
        # enforce the minimal l1‑norm required by the evaluator
        best_seq = best_seq * (0.01 / tot)

    return [float(x) for x in best_seq]
