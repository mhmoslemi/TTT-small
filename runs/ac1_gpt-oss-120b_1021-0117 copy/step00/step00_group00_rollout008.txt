def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)
    # get parent length
    try:
        n = len(height_sequence_1)
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # ensure best_score finite else set high.
    best_score = best_score if math.isfinite(best_score) else float('inf')

    # We'll attempt many random masks.

import time
import math
import numpy as np

def _make_candidate(n, rng):
    """
    Build a random non‑negative step function:
      - pick a random inclusion probability p,
      - choose indices independently with that probability,
      - give each chosen index the same weight so the total sum is 1.
    Occasionally enforce symmetry to add diversity.
    Returns a NumPy array of length n (or None if the mask is too small).
    """
    # probability of keeping an index; vary to explore different sparsities
    p = rng.uniform(0.02, 0.35)
    mask = rng.random(n) < p
    k = int(mask.sum())
    if k < 2:                         # need at least two non‑zero entries
        return None
    weight = 1.0 / k
    cand = np.zeros(n, dtype=np.float64)
    cand[mask] = weight

    # optional symmetrisation (keeps total sum = 1)
    if rng.random() < 0.2:
        cand = 0.5 * (cand + cand[::-1])

    # clipping is cheap and guarantees the evaluator's bounds
    np.clip(cand, 0.0, 1000.0, out=cand)

    # renormalise to avoid tiny round‑off drift
    s = cand.sum()
    if s == 0.0:
        return None
    cand *= (1.0 / s)
    return cand
