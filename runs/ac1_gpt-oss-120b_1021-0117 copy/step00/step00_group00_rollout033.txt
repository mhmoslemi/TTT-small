def compute_score(arr):
    arr = np.asarray(arr, dtype=np.float64)
    S = arr.sum()
    if S < 0.01:
        return np.inf

def gradient_step(a, step=0.05):
    n = len(a)
    # compute convolution
    conv = np.convolve(a, a)
    max_b = conv.max()
    k = conv.argmax()
    S = a.sum()
    # compute offset values: for each i, partner = k - i if in range else 0
    # partner indices may be out-of-range so use 0.
    # We'll compute partner array using vectorized cross.
    partner_idx = k - np.arange(n)
    # mask for valid partners
    valid = (partner_idx >= 0) & (partner_idx < n)
    partner_vals = np.zeros_like(a)
    partner_vals[valid] = a[partner_idx[valid]]
    # gradient of B w.r.t a_i is 2 * partner_vals
    # compute gradient of objective: g = (4n/S**3) * (partner_vals * S - max_b)
    g = partner_vals * S - max_b
    # compute step factor using exponent
    # typical scale: we want a small modification; use factor exp(-step * g / np.max(np.abs(g) + 1e-12))
    scale = np.exp(-step * g / (np.max(np.abs(g)) + 1e-12))
    a_new = a * scale
    # ensure nonnegative
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new

def random_block_move(a, rng):
    n = len(a)
    width = rng.integers(1, max(2, n//10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))  # scaling factor
    a_new = a.copy()
    a_new[start:start+width] *= factor
    a_new = np.clip(a_new, 0.0, 1000.0)
    # renormalize to sum 1
    s = a_new.sum()
    if s > 0:
        a_new /= s
    return a_new

import time
import numpy as np

def _score(arr):
    """Return (objective, max_conv, argmax) for a non‑negative array."""
    s = arr.sum()
    if s < 0.01:
        return np.inf, None, None
    conv = np.convolve(arr, arr)
    maxb = conv.max()
    k = conv.argmax()
    n = arr.size
    obj = 2.0 * n * maxb / (s * s)
    return obj, maxb, k


def _geometric(N, c):
    """Geometric decreasing a_i = r**i with r = 1 - c/N, normalized to sum 1."""
    r = 1.0 - c / float(N)
    r = max(0.0, r)
    idx = np.arange(N, dtype=np.float64)
    a = np.exp(idx * np.log(r)) if r > 0 else np.zeros(N)
    total = a.sum()
    if total == 0.0:
        a = np.ones(N, dtype=np.float64) / N
    else:
        a /= total
    return a


def _random_block(a, rng):
    """Multiply a random contiguous block by a log‑normal factor."""
    n = a.size
    max_w = max(1, n // 10)
    w = int(rng.integers(1, max_w + 1))
    start = int(rng.integers(0, n - w + 1))
    factor = float(np.exp(rng.normal(0.0, 0.3)))
    b = a.copy()
    b[start:start + w] *= factor
    b = np.clip(b, 0.0, 1000.0)
    s = b.sum()
    if s > 0.0:
        b /= s
    return b


def _gradient_step(a, lr):
    """One projected gradient step for the AC1 objective."""
    f, maxb, k = _score(a)
    if not np.isfinite(f):
        return a
    s = a.sum()
    idx = np.arange(a.size, dtype=np.int64)
    partner = k - idx
    ok = (partner >= 0) & (partner < a.size)
    partner_vals = np.zeros_like(a)
    partner_vals[ok] = a[partner[ok]]
    # sub‑gradient of the objective (up to a positive factor)
    g = partner_vals * s - maxb
    # normalise step size
    scale = lr / (np.max(np.abs(g)) + 1e-12)
    b = a - scale * g
    b = np.clip(b, 0.0, 1000.0)
    sb = b.sum()
    if sb > 0.0:
        b /= sb
    return b
