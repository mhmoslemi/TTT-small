import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step function that minimizes the AC1 bound.
    The search explores a parametrised family a_i = n**p / (i+offset)**p
    (with optional clipping at 1000) and then performs cheap local tweaks.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # Determine length and a fallback parent sequence
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
        n = int(parent_arr.size)
        best = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        # fallback: a modest uniform vector
        n = 256
        best = np.full(n, 1.0, dtype=np.float64)

    # ------------------------------------------------------------------
    # Helper to test a candidate
    # ------------------------------------------------------------------
    best_score = float(evaluate_sequence(best.tolist()))

    def _try(cand):
        nonlocal best, best_score
        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------------
    # Parametric grid search (p, offset) – the main source of improvement
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # classic exponent around the analytically optimal 0.5
    p_vals = np.array([0.48, 0.49, 0.5, 0.51, 0.52], dtype=np.float64)

    # offsets ranging from very small (allowing a tiny singularity) to
    # modest values; geometric spacing captures the relevant scale.
    offset_vals = np.geomspace(0.001, 10.0, num=40, dtype=np.float64)

    idx = np.arange(n, dtype=np.float64)  # reuse for speed

    for p in p_vals:
        # scaling factor n**p makes the sum ≈ 2·n (good baseline)
        scale = n ** p
        for off in offset_vals:
            if time.monotonic() >= deadline:
                break
            # a_i = scale / (i+off)**p
            a = scale * np.power(idx + off, -p)
            # clipping respects the hard 0‑1000 bound
            a = np.clip(a, 0.0, 1000.0)
            _try(a)

    # ------------------------------------------------------------------
    # Light weight local search around the best found candidate.
    # ------------------------------------------------------------------
    # we perform a few hundred cheap moves; each needs a convolution.
    while time.monotonic() < deadline:
        # compute convolution and locate the current worst index
        conv = np.convolve(best, best)
        s = int(conv.argmax())

        # contribution of each coordinate to the peak at index s
        i = np.arange(n, dtype=np.int64)
        j = s - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[valid] = best[i[valid]] * best[j[valid]]

        # concentrate on the most offending coordinates
        top_k = max(1, n // 300)                     # ~1 % of entries
        idx_top = np.argpartition(contrib, -top_k)[-top_k:]

        # propose a modest reduction of those entries and redistribute mass
        factor = 1.0 - rng.uniform(0.01, 0.04)        # 1 %–4 % shrinkage
        cand = best.copy()
        removed = np.sum(cand[idx_top] * (1.0 - factor))
        cand[idx_top] *= factor

        # redistribute the removed mass uniformly (preserves total sum)
        cand += removed / n

        # keep values in range and test
        cand = np.clip(cand, 0.0, 1000.0)
        _try(cand)

        # occasional random jitter to avoid stagnation
        if rng.random() < 0.03:
            jitter = np.exp(rng.normal(0.0, 0.03, size=n))
            cand = best * jitter
            cand = np.clip(cand, 0.0, 1000.0)
            _try(cand)

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
