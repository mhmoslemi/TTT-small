    best_S = None

        S = []
        sum_counts = {}

            i = rng.integers(0,n)
            ok=True
            # self sum

                ok=False

                for j in S:
                    s = i + j
                    if sum_counts.get(s,0) >= max_collisions:
                        ok=False
                        break
            if ok:
                # add
                for j in S:
                    s = i + j
                    sum_counts[s] = sum_counts.get(s,0)+1

                best_score = score
                best_S = list(S)

import time
import numpy as np

# the evaluator and (optionally) height_sequence_1 are provided by the harness


def _uniform_candidate(n: int) -> np.ndarray:
    """Uniform step heights whose total sum equals √(2 n)."""
    total = np.sqrt(2.0 * n)
    return np.full(n, total / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random positive vector with the same ℓ₁‑norm as the uniform candidate."""
    # Exponential draws give a Dirichlet(1,…,1) distribution.
    x = rng.exponential(scale=1.0, size=n)
    total = np.sqrt(2.0 * n)          # any positive total works – scaling is irrelevant
    x = x / x.sum() * total
    return np.clip(x, 0.0, 1000.0)


def _antisymmetric_pair_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    For each symmetric pair (i, n‑1‑i) draw a random split of the total mass.
    This keeps the total ℓ₁‑norm unchanged but perturbs the central convolution term.
    """
    total = np.sqrt(2.0 * n)
    base = total / n                 # uniform height
    a = np.empty(n, dtype=np.float64)

    half = n // 2
    for i in range(half):
        alpha = rng.random()
        a[i] = 2.0 * base * alpha
        a[n - 1 - i] = 2.0 * base * (1.0 - alpha)

    if n % 2:                         # middle element for odd length
        a[half] = base

    return np.clip(a, 0.0, 1000.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative heights.
    The routine tries a few cheap stochastic constructions and keeps the best
    one seen before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Length of the problem (fallback to 256 if the parent is unavailable)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256

    # Start with the uniform construction – this yields the known bound 2.0
    best_seq = _uniform_candidate(n)
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # If a parent sequence is supplied, consider it as a second baseline
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size and np.isfinite(parent).all():
            parent_score = float(evaluate_sequence(parent.tolist()))
            if parent_score < best_score:
                best_seq = parent
                best_score = parent_score

    # Light‑weight stochastic search – only a few dozen evaluations fit easily
    # inside the time budget.  The constructions are O(n) memory and cheap.
    while time.monotonic() < deadline:
        # Choose one of the cheap generators at random
        choice = rng.integers(0, 3)
        if choice == 0:
            cand = _random_dirichlet_candidate(n, rng)
        elif choice == 1:
            cand = _antisymmetric_pair_candidate(n, rng)
        else:
            # small perturbation of the current best (multiplicative block)
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand = best_seq.copy()
            cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
            if rng.random() < 0.35:
                cand = 0.5 * (cand + cand[::-1])
            cand = np.clip(cand, 0.0, 1000.0)

        # Exact evaluation (authoritative)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score

    # Return plain Python floats – the harness expects a list.
    return [float(v) for v in best_seq]
