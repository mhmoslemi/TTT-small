import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a uniform non‑negative step‑function of length matching the provided

    # Determine target length: use the parent length if available and valid.
    default_len = 256

            parent = np.asarray(height_sequence_1, dtype=np.float64)
