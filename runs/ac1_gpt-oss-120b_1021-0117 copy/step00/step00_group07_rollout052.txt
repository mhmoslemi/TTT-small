def compute_score(arr):
    # arr already numpy array, assumed nonnegative and <=1000
    n = arr.size
    # convolution
    b = np.convolve(arr, arr)
    max_b = b.max()
    sum_a = arr.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np
import time


def _score(seq: np.ndarray) -> float:
    """
    Compute the AC1 upper bound used by ``evaluate_sequence``.
    ``seq`` is assumed to be a 1‑D NumPy array of non‑negative values already
    clipped to the [0,1000] range.
    """
    n = seq.size
    # linear convolution (full) – NumPy uses an efficient C implementation
    b = np.convolve(seq, seq)
    max_b = b.max()
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Create a cheap O(n) random perturbation of ``seq``."""
    n = seq.size
    cand = seq.copy()
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:          # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a list of non‑negative heights with a small AC1 upper bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.4)

    # --------------------------------------------------------------------

    #    is unavailable or invalid)

    try:

            raise ValueError
        parent = np.clip(base, 0.0, 1000.0)
    except Exception:                     # parent not present or broken
        parent = np.ones(256, dtype=np.float64)

    n = parent.size

    #    (including the parent) and keep the best among them.
    # --------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # try the supplied parent first
    val = _score(parent)
    if val < best_val:
        best_val = val
        best_seq = parent.copy()

    # scan a small grid of offsets d (0 … 10) – only ~20 candidates
    for d in np.linspace(0.0, 10.0, 21):
        a = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 1.0 + d)
        np.clip(a, 0.0, 1000.0, out=a)
        val = _score(a)
        if val < best_val:

    # guarantee a starting point

    # 3. Local optimisation loop.

    #    coefficient occurs, pull a little mass away from the pair that

    # --------------------------------------------------------------------
    it = 0

        # full convolution of the current sequence
        conv = np.convolve(cur_seq, cur_seq)
        max_idx = int(np.argmax(conv))

        # indices (i,j) with i+j == max_idx
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = max_idx - i_vals

                if cand_val < best_val:

                    best_seq = cand.copy()

    # --------------------------------------------------------------------
    return [float(x) for x in best_seq]
