def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Acquire parent if exists
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim !=1 or parent.size ==0 or not np.isfinite(parent).all():
            raise Exception
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # Time check
    n = best.size

    # Evaluate p-parameterized families
    # we include uniform (p=0)
    p_vals = np.linspace(0.0, 1.2, 61)  # tune bound
    # compute index arrays once
    idx1 = np.arange(1, n+1, dtype=np.float64)  # from 1 to n
    idx2 = np.arange(n, 0, -1, dtype=np.float64)  # from n to 1

    # Precompute (n-i) array shape basically idx2

    for p in p_vals:
        # early exit if time near deadline
        if time.monotonic() > deadline:
            break
        # shape
        seq = np.power(idx1, -p) + np.power(idx2, -p)
        # no scaling needed; values small (<2)
        # Compute score efficiently
        # Use FFT
        m = 2*n - 1
        fa = np.fft.rfft(seq, n=m)
        conv = np.fft.irfft(fa * fa, n=m)
        max_b = conv.max()
        sum_a = seq.sum()
        if sum_a < 0.01:
            continue
        score = 2.0 * n * max_b / (sum_a * sum_a)
        if score < best_score:
            best = seq
            best_score = float(score)

    # Optional random local adjustments using small perturbations if time left
    while time.monotonic() < deadline:
        # random perturbation: choose a segment and multiply by random factor
        width = rng.integers(max(1, n // 200), max(2, n // 12))
        start = rng.integers(0, n - width + 1)
        factor = np.exp(rng.normal(0.0, 0.05))
        cand = best.copy()
        cand[start:start+width] *= factor
        # optional symmetry averaging
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])
        # Clip and compute score
        cand = np.clip(cand, 0.0, 1000.0)
        # compute score
        fa = np.fft.rfft(cand, n=2*n-1)
        conv = np.fft.irfft(fa * fa, n=2*n-1)
        max_b = conv.max()
        sum_a = cand.sum()
        if sum_a < 0.01:
            continue
        score = 2.0 * n * max_b / (sum_a * sum_a)
        if score < best_score:
            best = cand
            best_score = float(score)

    # Return best candidate as list
    return [float(x) for x in best]

import time
import numpy as np


def _score(seq: np.ndarray) -> float:
    """Fast evaluation of the AC1 upper bound for a given sequence."""
    n = seq.size
    s = seq.sum()
    if s < 0.01:               # safeguard against division by zero
        return float("inf")
    m = 2 * n - 1
    # linear convolution via FFT (no extra memory beyond O(n))
    fa = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step‑function that beats the target AC1 bound.
    The routine first tries a one‑parameter family a_i = (i+1)^(-p) +
    (n‑i)^(-p) (symmetric power‑law).  It then performs a few cheap
    stochastic refinements while respecting the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ----- initialise with the provided parent if it exists -----

    except Exception:
        # fallback – a short flat sequence
        parent = np.ones(256, dtype=np.float64)

    best_score = float(evaluate_sequence(best_seq.tolist()))
    n = best_seq.size

    # ----- grid search over the power‑law exponent p -----
    idx1 = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    idx2 = np.arange(n, 0, -1, dtype=np.float64)         # n … 1

        sc = _score(seq)
