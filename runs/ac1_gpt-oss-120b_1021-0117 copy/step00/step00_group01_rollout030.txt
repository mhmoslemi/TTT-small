def generate_sidon_set(n, max_size):
    sums = set()
    S = []
    for x in range(n):
        # quick check: can we add? compute sum with each s in S, plus x+x
        conflict = False
        if x*2 in sums:
            conflict = True
        else:
            for s in S:
                if x+s in sums:
                    conflict = True
                    break
        if not conflict:
            S.append(x)
            # add sums
            sums.add(x*2)
            for s in S[:-1]:
                sums.add(x+s)
            if len(S) >= max_size:
                break
    return S
```
