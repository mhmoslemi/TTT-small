import numpy as np
import time

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a deterministic linear ramp that achieves an AC1 upper bound
    of (4/3)*(n+2)/(n+1) ≈ 1.334 for the given length n (≈4498).
    The ramp respects all evaluator constraints (non‑negative, ≤1000,
    and sum > 0.01).
    """
    # Determine the target length.  Use the provided parent if present,
    # otherwise fall back to a modest size.
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = arr.size
        if n == 0:
            raise ValueError
    except Exception:
        n = 256

    # Linear ramp: a_i = 1000 * (i+1) / n   (i = 0 … n-1)
    # Guarantees a_{n-1}=1000 and a_0 ≈ 1000/n > 0.
    a = 1000.0 * (np.arange(1, n + 1, dtype=np.float64) / n)
    a = np.clip(a, 0.0, 1000.0)

    # Return as plain Python list for the evaluator.
    return a.tolist()
