import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a simple constant-step sequence which attains the theoretical
    lower bound 2.0 for the AC1 upper‑bound.
    The length matches the provided `height_sequence_1` if available,
    otherwise a modest fallback length is used.
    """
    # Determine the problem size (n)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        # No parent supplied – fall back to a modest size
        n = 256
    else:
        # Validate parent; if invalid, fall back as above
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():

        else:

    # we choose 1.0 for simplicity and to stay safely inside the [0,1000] range.
    candidate = np.full(n, 1.0, dtype=np.float64)

    # Ensure permissible range (clipping is cheap and guarantees compliance)
    candidate = np.clip(candidate, 0.0, 1000.0)

    # Return as a plain Python list of floats, as required by the evaluator.
