import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) that aims to minimise
    the AC1 upper bound defined by `evaluate_sequence`.

    Strategy:
    * Use the length of the supplied `height_sequence_1` when available,
      otherwise fall back to a modest default length.
    * Create many random candidates (uniform[0,1]) – the shape of a random
      distribution typically yields a convolution peak close to the average
      value, giving a ratio near 1, well below the baseline 2.0.
    * Keep the best‑scoring candidate seen before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    # reserve a few seconds for cleanup / returning the answer
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # Determine the problem size (length of the sequence)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(base.shape[0]) if base.ndim == 1 and base.size > 0 else 0
    except Exception:
        n = 0
    if n <= 0:
        n = 1024                               # a safe default if nothing is given

    # ------------------------------------------------------------------
    # Fallback candidate (guarantees a finite answer)
    # ------------------------------------------------------------------
    best = np.ones(n, dtype=np.float64)       # uniform → ratio exactly 2
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Random search – O(n) memory, each evaluation O(n log n) via np.convolve
    # ------------------------------------------------------------------
    # The ratio is homogeneous: scaling the whole vector does not change it,
    # therefore a simple uniform(0,1) distribution already explores the
    # essential shape.
    while time.monotonic() < deadline:
        cand = rng.random(n, dtype=np.float64)   # values in [0,1)
        # Evaluate the candidate; conversion to list is required by the evaluator
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # Early exit if we already beat an ambitious target
            if best_score < 1.5:
                break

    # Return as plain Python list of floats
    return [float(x) for x in best]
