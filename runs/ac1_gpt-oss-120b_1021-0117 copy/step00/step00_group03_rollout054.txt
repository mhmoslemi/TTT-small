def generate_sidon_set(n, target_m=None, rng=None):
    # target_m optional; if None compute floor(sqrt(2*n))
    max_m = int(np.sqrt(2*n))
    if target_m is None or target_m > max_m:
        target_m = max_m
    # we will attempt random greedy until we reach target_m or time runs out
    chosen = []
    sums = set()
    attempts = 0
    # use while len(chosen) < target_m:
    while len(chosen) < target_m:
        # generate random candidate uniformly from [0, n-1]
        cand = int(rng.integers(n))
        # add if not already chosen:
        if cand in chosen:
            continue
        # Compute new sums: cand + c for each c in chosen (including self)
        ok = True
        new_sums = []
        for c in chosen:
            s = cand + c
            if s in sums:
                ok = False
                break
            new_sums.append(s)
        # also self-sum:
        self_sum = cand + cand
        if self_sum in sums:
            ok = False
        else:
            new_sums.append(self_sum)
        if ok:
            chosen.append(cand)
            for s in new_sums:
                sums.add(s)
        attempts += 1
        if attempts > 10 * target_m * n: # fallback too many attempts, break
            break
    return chosen

def greedy_sidon(n, rng):
    # random permutation of indices
    perm = rng.permutation(n)
    chosen = []
    sums = set()
    for idx in perm:
        # test collisions
        collision = False
        # compute new sums with existing chosen
        for j in chosen:
            s = idx + j
            if s in sums:
                collision = True
                break
        if collision:
            continue
        self_s = idx + idx
        if self_s in sums:
            continue
        # ok, add it
        for j in chosen:
            sums.add(idx + j)
        sums.add(self_s)
        chosen.append(idx)
    return chosen

def simplex_projection(v, s):
    # project onto {x≥0, sum x = s}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v, s=1.0):
    """Project v onto the simplex {x>=0, sum(x)=s}."""
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

def _project_onto_simplex(v, s):
    """Project vector v onto the simplex {x >= 0, sum(x) = s}."""
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic projected‑gradient search for a non‑negative step function
    minimising the AC1 bound.  The algorithm works entirely with O(n)
    memory and respects the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ---- initialise ---------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)

    n = base.size
    target_sum = np.sqrt(2.0 * n)               # normalising sum (scale invariant)
    a = np.clip(base, 0.0, 1000.0)
    a = a * (target_sum / np.sum(a))            # project onto the simplex once

    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))

    # ---- hyper‑parameters ----------------------------------------------------
    alpha = 20.0          # softness of the soft‑max (larger = closer to true max)
    lr = 0.6              # initial step size
    lr_decay = 0.9995     # exponential decay per iteration
    max_no_improve = 400  # restart after this many non‑improving steps
    no_improve = 0

    # ---- main optimisation loop ---------------------------------------------
    while time.monotonic() < deadline:
        # compute convolution and a soft‑max weighting over its entries
        conv = np.convolve(a, a)                # length 2*n-1
        max_val = conv.max()
        # soft‑max on the convolution (stable by subtracting the max)
        exp_arg = alpha * (conv - max_val)
        weights = np.exp(exp_arg, dtype=np.float64)
        w_sum = weights.sum()
        if w_sum == 0.0:
            weights = np.full_like(weights, 1.0 / weights.size)
        else:
            weights /= w_sum                       # now a probability distribution

        # gradient of the smooth approximation of max(conv)
        a_rev = a[::-1]
        grad_full = np.convolve(a_rev, weights)   # length 3*n-2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]    # slice back to length n

        # gradient step and projection onto the simplex (sum = target_sum)
        a_new = a - lr * grad
        a_new = _project_onto_simplex(a_new, target_sum)

        # evaluate only when we have a strict improvement
        score = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(score) and score < best_score:

        lr *= lr_decay

    return [float(x) for x in best_a]
