        # compute gradient g
        i_start = max(0, k_max - (n-1))
        i_end = min(n-1, k_max)
        # i indices inclusive
        i = np.arange(i_start, i_end+1, dtype=np.int64)
        j = k_max - i
        g = np.zeros_like(a)
        g[i] += a[j]
        g[j] += a[i]

        # apply update
        new_a = a - step * g
        # clip nonnegative
        new_a = np.clip(new_a, 0.0, None)
        # renormalize to sum 1 (avoid division by zero)
        total = new_a.sum()
        if total < 1e-12:
            # fallback: reset to uniform
            new_a = np.ones_like(a) / n
            total = 1.0
        new_a = new_a / total

        # evaluate score
        score = float(evaluate_sequence(new_a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = new_a.copy()
            best_score = score

        # accept new_a if it improves or if we use simulated annealing?
        if score <= best_score + 1e-12: # keep improvement; else maybe revert
            a = new_a
        else:
            # occasionally accept a random perturbation to escape
            if rng.random() < 0.01:
                a = new_a
        # adjust step size adaptively?
        # maybe decay step over time
        step = max(1e-4, step * 0.9999)  # slowly decay

        # also occasional random perturbation to explore
        if rng.random() < 0.02:
            # random noise
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = np.clip(a * noise, 0.0, None)
            a = a / a.sum()
        max_iter += 1

    return [float(x) for x in best]
