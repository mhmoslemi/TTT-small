    rng.shuffle(indices)

        new_sums = [idx + j for j in best_set] + [2*idx]
        # check for collisions

        if not conflict:

            con_sums.update(new_sums)
    return best_set

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: exact score (same formula as evaluate_sequence)
def _score(arr: np.ndarray) -> float:
    """Return the AC1 upper‑bound for a non‑negative vector."""
    n = arr.size
    conv = np.convolve(arr, arr)          # linear convolution, length 2n‑1
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Helper: greedy construction of a Sidon (B2) set
def _greedy_sidon(n: int, rng: np.random.Generator,
                  inner_trials: int = 5) -> list[int]:
    """
    Return a list of indices with pairwise distinct sums.
    Multiple inner_trials are performed and the longest set is kept.
    """
    best_set: list[int] = []
    for _ in range(inner_trials):
        cur_set: list[int] = []
        used_sums: set[int] = set()
        order = np.arange(n, dtype=np.int64)
        rng.shuffle(order)
        for idx in order:
            idx = int(idx)
            # check self‑sum
            if 2 * idx in used_sums:
                continue
            # check sums with already chosen indices
            conflict = False
            for j in cur_set:
                if idx + j in used_sums:
                    conflict = True
                    break
            if conflict:
                continue
            # accept idx
            for j in cur_set:
                used_sums.add(idx + j)
            used_sums.add(2 * idx)
            cur_set.append(idx)
        if len(cur_set) > len(best_set):
            best_set = cur_set
    return best_set
