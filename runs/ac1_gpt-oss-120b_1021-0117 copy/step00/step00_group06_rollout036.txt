    best_score = float('inf')

    # try several exponents; larger p gives smaller theoretical bound
    for p in (1, 2, 3, 4, 5, 6, 7):
        if time.monotonic() > deadline:
            break

        # increasing power‑law
        seq = 1000.0 * np.power(idx, p)
        np.clip(seq, 0.0, 1000.0, out=seq)               # safety clamp
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()

        # decreasing version (mirror)
        seq_rev = seq[::-1]
        score_rev = float(evaluate_sequence(seq_rev.tolist()))
        if np.isfinite(score_rev) and score_rev < best_score:
            best_score = score_rev
            best_seq = seq_rev.copy()

    # fallback – just in case none of the candidates were accepted
    if best_seq is None:
        if isinstance(parent, np.ndarray):
            best_seq = np.clip(parent.astype(np.float64), 0.0, 1000.0)
        else:
            best_seq = np.full(n, 1.0, dtype=np.float64)

    # return a plain Python list of floats
    return [float(v) for v in best_seq]
