def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # Determine length n
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    # Initialize best sequence
    if 'height_sequence_1' in globals():
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if base.ndim != 1 or base.size != n or not np.isfinite(base).all():
            base = None
    else:
        base = None

    if base is not None and np.sum(base) > 0.01:
        # Use normalized version of base (preserve shape but scale to sum 1)
        init = base / np.sum(base)
    else:
        # random Dirichlet
        init = rng.random(n)
        init = init / init.sum()
    # ensure values within [0,1000] (already <1)
    current = init.copy()
    best = current.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur_score = best_score

    # Greedy gradient descent loop
    no_improve = 0
    while time.monotonic() < deadline:
        # compute convolution and its max index
        conv = np.convolve(current, current)
        M = conv.max()
        idx_max = int(conv.argmax())
        # compute overlapping indices
        start = max(0, idx_max - (n - 1))
        end = min(idx_max, n - 1)
        # if no overlap (should not happen) 
        if start > end:
            # random perturbation fallback
            delta_vec = np.exp(rng.normal(0.0, 0.02, size=n))
            candidate = np.clip(current * delta_vec, 0.0, 1000.0)
        else:
            i_range = np.arange(start, end + 1)
            j_range = idx_max - i_range  # partner indices
            a_j = current[j_range]
            # select indices
            i_high = i_range[np.argmax(a_j)]
            i_low = i_range[np.argmin(a_j)]
            # if can't move (i_high == i_low or small mass), do random small perturb
            if i_high == i_low or current[i_high] <= 1e-12:
                # random perturb small
                delta_vec = np.exp(rng.normal(0.0, 0.02, size=n))
                candidate = np.clip(current * delta_vec, 0.0, 1000.0)
            else:
                # transfer small fraction of mass from high to low
                frac = rng.uniform(0.01, 0.05)  # 1%..5%
                delta = min(current[i_high] * frac, current[i_high])
                candidate = current.copy()
                candidate[i_high] -= delta
                candidate[i_low] += delta
                # optionally add small random noise to avoid stagnation
                if rng.random() < 0.02:
                    noise = rng.normal(0.0, 1e-4, size=n)
                    candidate = np.clip(candidate + noise, 0.0, 1000.0)
        # evaluate
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < cur_score - 1e-12:
            current = candidate
            cur_score = score
            no_improve = 0
            if score < best_score:
                best = candidate
                best_score = score
        else:
            no_improve += 1
        # occasional random restart if stuck
        if no_improve > 2000:
            # reinitialize to best with small random perturbation
            perturb = np.exp(rng.normal(0.0, 0.01, size=n))
            current = np.clip(best * perturb, 0.0, 1000.0)
            cur_score = float(evaluate_sequence(current.tolist()))
            no_improve = 0
        # early exit if we already beat target
        if best_score < 1.50274349:
            break
    return [float(x) for x in best]

def _gradient_move(a: np.ndarray, idx_max: int, rng: np.random.Generator) -> np.ndarray:
    """Move a small amount of mass from a high‑gradient index to a low‑gradient index."""
    n = a.shape[0]
    # overlapping region for the shift that attains the maximum
    start = max(0, idx_max - (n - 1))
    end = min(idx_max, n - 1)
    if start > end:                     # should never happen, but guard anyway
        return a * np.exp(rng.normal(0.0, 0.02, size=n))

    i_range = np.arange(start, end + 1)
    j_range = idx_max - i_range          # partner indices (i + j = idx_max)

    # values of the partners controlling the sub‑gradient
    partner_vals = a[j_range]

    # choose the most and the least influential positions
    i_high = i_range[np.argmax(partner_vals)]
    i_low = i_range[np.argmin(partner_vals)]

    if i_high == i_low or a[i_high] <= 0.0:
        # fall back to a tiny random multiplicative jitter
        mult = np.exp(rng.normal(0.0, 0.02, size=n))
        return np.clip(a * mult, 0.0, 1000.0)

    # transfer a modest fraction of the mass from the high‑gradient to the low‑gradient entry
    frac = rng.uniform(0.01, 0.05)               # 1‑5 % of the source entry
    delta = min(a[i_high] * frac, a[i_high])
    b = a.copy()
    b[i_high] -= delta
    b[i_low] += delta
    # occasional tiny noise to keep the dynamics alive
    if rng.random() < 0.02:
        b = np.clip(b + rng.normal(0.0, 1e-5, size=n), 0.0, 1000.0)
    return b


def _random_jitter(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Small random multiplicative perturbation (keeps the sum unchanged)."""
    mult = np.exp(rng.normal(0.0, 0.03, size=a.shape))
    return np.clip(a * mult, 0.0, 1000.0)
