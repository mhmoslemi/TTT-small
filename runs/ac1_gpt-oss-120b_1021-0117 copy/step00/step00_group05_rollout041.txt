def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # length
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256
    # base inverse sqrt
    idx = np.arange(n, dtype=np.float64) + 1.0
    base = 1.0 / np.sqrt(idx)
    # scale to max 1000
    base = base * (1000.0 / base.max())
    a = base.copy()
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # initial sigma as relative variance
    sigma = 0.1
    # precompute maybe clipping after each step.
    while time.monotonic() < deadline:
        # propose perturbation
        # multiplicative log-normal perturbation
        perturb = np.exp(rng.normal(0.0, sigma, size=n))
        cand = a * perturb
        # enforce bounds
        np.clip(cand, 0.0, 1000.0, out=cand)
        # optional small probability to symmetrize
        if rng.random() < 0.2:
            cand = 0.5 * (cand + cand[::-1])
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            a = cand  # continue from improved candidate
            # gradually reduce sigma to focus search
            sigma *= 0.95
        else:
            # maybe occasional reset sigma if no improvement for while
            sigma = min(0.2, sigma * 1.01)

    return [float(x) for x in best]

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Hill‑climbing search for a non‑negative step function that minimizes
    the AC1 upper bound used by `evaluate_sequence`.
    The search starts from the analytically motivated inverse‑square‑root
    shape a_i = 1/√(i+1) (scaled to the allowed max 1000) and then
    performs a mixture of global log‑normal perturbations and local
    block rescalings while keeping the best candidate found.
    """
    rng = np.random.default_rng(seed)
    # leave a few seconds margin for the return
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # length of the problem (use the provided parent if available)
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:               # fallback length if the parent is missing
        n = 256

    # ---- initial candidate : inverse square‑root shape -----------------
    # a_i = 1/√(i+1)  –  this shape gives a near‑flat autocorrelation
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = 1.0 / np.sqrt(idx)
    # scale so that the largest entry touches the allowed upper bound
    a *= 1000.0 / a.max()
    a = np.clip(a, 0.0, 1000.0)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # parameters for the stochastic moves
    sigma = 0.12               # log‑normal noise magnitude
    no_improve = 0
    while time.monotonic() < deadline:
        # ----- propose a new candidate ---------------------------------
        # two complementary perturbation families
        if rng.random() < 0.55:
            # global multiplicative noise (log‑normal)
            factor = np.exp(rng.normal(0.0, sigma, size=n))
            cand = a * factor
        else:
            # local block scaling
            cand = a.copy()
            start = int(rng.integers(0, n))
            # width at most 1/10 of the sequence, at least 1 element
            width = int(rng.integers(1, max(2, n // 10)))
            stop = min(n, start + width)
            scale = float(np.exp(rng.normal(0.0, 0.25)))
            cand[start:stop] *= scale

        # occasional symmetric averaging (helps with the mirror symmetry)
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])

        np.clip(cand, 0.0, 1000.0, out=cand)

        # ----- evaluate -------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            a = cand                     # continue search from the improved point
            sigma *= 0.95                # narrow the search when progress is made
            no_improve = 0
        else:
            no_improve += 1
            # slowly increase the step size if we have been stuck
            if no_improve > 200:
                sigma = min(0.5, sigma * 1.07)
                no_improve = 0

    # return a plain Python list as required by the evaluator
    return [float(v) for v in best]
