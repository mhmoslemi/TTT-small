import time
import math
import numpy as np

def compute_score(seq: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative sequence."""
    # Convolution of seq with itself (full mode)
    b = np.convolve(seq, seq)
    max_b = b.max()
    s = seq.sum()
    n = seq.size
    if s < 0.01:
        return math.inf
    return 2.0 * n * max_b / (s * s)

    Uses a simple sub‑gradient descent on the maximal convolution entry,
    combined with occasional block perturbations.  The routine respects the
    time budget and always returns a valid fallback candidate.

    # ------------------------------------------------------------------
