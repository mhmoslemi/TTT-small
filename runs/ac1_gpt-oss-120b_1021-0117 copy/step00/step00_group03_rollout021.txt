def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
    # fallback base on height_sequence_1 if exists
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0) if 'height_sequence_1' in globals() else np.ones(256)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # attempt random sparse binary vectors
    # Determine a target k range based on n to get expected score below target
    # target_score = 1.5027, we can choose k = int(np.sqrt(2*n) * 1.2) for safety
    k_target = int(np.sqrt(2 * n) * 1.5)  # maybe
    k_min = max(1, k_target // 2)
    k_max = max(k_target * 2, k_target + 20)
    # ensure within n
    k_max = min(k_max, n)
    while time.monotonic() < deadline:
        # choose random k in range
        k = rng.integers(k_min, k_max+1)
        idx = rng.choice(n, size=k, replace=False)
        candidate = np.zeros(n, dtype=np.float64)
        candidate[idx] = 1.0
        score = float(evaluate_sequence(candidate.tolist()))
        # Update best
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
    return [float(x) for x in best]

import time
import math
import numpy as np

def _random_binary_vector(n, k, rng):
    """Return a length‑n float array with exactly k entries equal to 1.0."""
    vec = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    vec[idx] = 1.0
    return vec

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step function whose AC1 upper bound
    (as defined by evaluate_sequence) is as small as possible.
    The routine stays within the time budget and always returns a
    valid fallback (the provided parent sequence).
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin before the sandbox kills the process
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # base (fallback) sequence – use the supplied parent if it exists
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = parent.size
    else:
        # very rare case: no parent, create a trivial one
        n = 256
        parent = np.ones(n, dtype=np.float64)

    best_vec = parent.copy()
    best_score = float(evaluate_sequence(best_vec.tolist()))

    # ------------------------------------------------------------------
    # parameters for the random sparse binary search
    # we target a density that gives an expected score ≈ 1
    #   k ≈ 2·√(2·n)  (empirically good for n≈2‑3k)
    #   allow some variation around that target
    # ------------------------------------------------------------------
    k_target = int(2 * math.sqrt(2 * n))
    k_min = max(1, int(0.7 * k_target))
    k_max = min(n, int(1.5 * k_target))

    # ------------------------------------------------------------------
    # main search loop – generate many random binary vectors and keep
    # the one with the smallest AC1 bound.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        k = rng.integers(k_min, k_max + 1)
        cand = _random_binary_vector(n, k, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_vec = cand
            best_score = score

        # occasional local refinement of the current best:
        # swap one present index with a missing one
        if rng.random() < 0.15 and best_score < 2.0:
            present = np.flatnonzero(best_vec > 0)
            if present.size:
                out_idx = rng.choice(present)
                # choose a new index not already used
                # (setdiff1d is cheap for this modest n)
                missing = np.setdiff1d(np.arange(n), present, assume_unique=True)
                if missing.size:
                    in_idx = rng.choice(missing)
                    cand2 = best_vec.copy()
                    cand2[out_idx] = 0.0
                    cand2[in_idx] = 1.0
                    score2 = float(evaluate_sequence(cand2.tolist()))
                    if np.isfinite(score2) and score2 < best_score:
                        best_vec = cand2
                        best_score = score2

    # ------------------------------------------------------------------
    # Return the best vector found (as a plain Python list of floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
