function step_once(a):
    c = np.convolve(a, a)
    kmax = np.argmax(c)
    # compute partner grad vector
    i = np.arange(len(a))
    j = kmax - i
    valid = (j >= 0) & (j < len(a))
    grad = np.zeros_like(a)
    grad[valid] = a[j[valid]]
    # source: index with max grad* a maybe? We'll just max grad
    src_candidates = np.where(a > 1e-12)[0]
    if src_candidates.size == 0: return a
    src_grad = grad[src_candidates]
    src = src_candidates[np.argmax(src_grad)]
    # destination: index with min grad among those not src
    dst_candidates = np.where(a < 1000-1e-12)[0]
    # optionally exclude src
    dst_candidates = dst_candidates[dst_candidates != src]
    if dst_candidates.size == 0:
        dst = src # no move
        return a
    dst_grad = grad[dst_candidates]
    dst = dst_candidates[np.argmin(dst_grad)]
    # delta: small fraction of source amount
    delta = min(a[src] * 0.1, (1000 - a[dst]) * 0.1, 0.1)
    # actual delta = min(delta, something)
    delta = min(delta, a[src])
    # apply move
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    return a_new

import time
import numpy as np

# ------------------------------------------------------------
# Helper to build a simple piecewise-constant sequence.
# ------------------------------------------------------------
def _build_sequence(n: int, w: int, r: float, symmetric: bool = False) -> np.ndarray:
    """
    Build a length-n non‑negative sequence.
    Prefix of length w (and suffix if symmetric) has value r*v,
    the remaining entries have value v, where v normalises the sum to 1.
    """
    if w < 0:
        w = 0
    if symmetric:
        w = min(w, n // 2)          # must fit both ends
        denom = (n - 2 * w) + 2 * w * r
        v = 1.0 / denom
        u = r * v
        a = np.full(n, v, dtype=np.float64)
        a[:w] = u
        a[-w:] = u
    else:
        w = min(w, n - 1)           # at least one interior element
        denom = (n - w) + w * r
        v = 1.0 / denom
        u = r * v
        a = np.full(n, v, dtype=np.float64)
        a[:w] = u
    return a


# ------------------------------------------------------------
# Main entry point required by the harness.
# ------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step‑function height vector that minimises the
    AC1 upper bound used in `evaluate_sequence`.
    The algorithm scans a few piecewise‑constant families (prefix
    and symmetric prefix+suffix) with varying width and ratio,
    then performs cheap hill‑climbing moves.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # -----------------------------------------------------------------
    # Base sequence: use the supplied height_sequence_1 when present.
    # -----------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or not np.isfinite(base).all():
            raise Exception
        base = np.clip(base, 0.0, 1000.0)
        n = base.size
        best = base.copy()
    except Exception:
        # fallback – uniform length 256 vector
        n = 256
        best = np.full(n, 1.0 / n, dtype=np.float64)

    # initial score (authoritative)
    best_score = float(evaluate_sequence(best.tolist()))

    # -----------------------------------------------------------------
    # 1) Coarse grid search over simple piecewise families.
    # -----------------------------------------------------------------
    # fractions of the length used for the high‑value prefix
    fractions = [0.01, 0.02, 0.05, 0.10, 0.15, 0.20, 0.25, 0.30, 0.35, 0.40]
    # amplification ratios (edge value = r * interior value)
    ratios = [1.5, 2.0, 3.0, 5.0, 10.0]

    for f in fractions:
        w = max(1, int(n * f))
        if w >= n // 2:
            w = n // 2 - 1   # keep at least one interior element
        for r in ratios:
            # ----- prefix only -------------------------------------------------
            cand = _build_sequence(n, w, r, symmetric=False)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best, best_score = cand, score

            # ----- symmetric (both ends) ---------------------------------------
            cand = _build_sequence(n, w, r, symmetric=True)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best, best_score = cand, score

            # early exit if time is up
            if time.monotonic() >= deadline:
                break
        if time.monotonic() >= deadline:
            break

    # -----------------------------------------------------------------
    # 2) Light hill‑climbing using random mass transfers.
    # -----------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if best[i] <= 0.0 or best[j] >= 1000.0:
            continue
        # move a random fraction (≤10%) of mass from i to j
        max_move = min(best[i], 1000.0 - best[j])
        delta = max_move * rng.random() * 0.1
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = cand, score

    # Return as plain Python list of floats
    return [float(x) for x in best]
