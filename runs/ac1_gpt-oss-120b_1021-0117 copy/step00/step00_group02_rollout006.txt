def sidon_set(max_n):
    S = []
    sums = set()
    for x in range(max_n):
        ok = True
        # k = 2*x
        if (2*x) in sums:
            ok = False
        else:
            for y in S:
                if (x + y) in sums:
                    ok = False
                    break
        if ok:
            S.append(x)
            sums.add(2*x)
            for y in S[:-1]: # previous elements only
                sums.add(x + y)
    return S

import numpy as np
import time
import math

def _sidon_set(limit: int):
    """Greedy construction of a Sidon (B2) set in {0,…,limit‑1}."""
    s = []
    sums = set()
    for x in range(limit):
        # check involvement of 2*x (pair with itself)
        if 2 * x in sums:
            continue
        conflict = False
        for y in s:
            if x + y in sums:
                conflict = True
                break
        if conflict:
            continue
        # accept x
        for y in s:
            sums.add(x + y)
        sums.add(2 * x)
        s.append(x)
    return s

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Construct a low‑AC1 sequence using a Sidon set and a uniform background."""
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)

    # length of the problem (fallback to 256 if parent missing)
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            n = 256
        else:
            n = int(base.size)
    except Exception:
        n = 256

    # ---- 1. Sidon set (almost √n elements) -------------------------
    sid = _sidon_set(n)
    B = len(sid)

    # ---- 2. Choose a small uniform background ε --------------------
    eps = 0.1                     # background height (any positive value works)
    eps = float(eps)

    # ---- 3. Height for the special indices -------------------------
    # constraints:
    #   2 * h^2   <= n * eps^2          (big–big pair must stay below the background max)
    #   B * h * eps <= n * eps^2       (big–small pairs must stay below the background max)
    # The first gives h <= eps * sqrt(n/2)
    # The second gives h <= n * eps / B
    max_h1 = eps * math.sqrt(n / 2.0)
    max_h2 = (n * eps) / B if B > 0 else max_h1
    max_h = min(max_h1, max_h2) * 0.9   # safety margin
    if max_h <= 0.0:
        max_h = eps * 0.01

    # ---- 4. Build the candidate vector --------------------------------
    a = np.full(n, eps, dtype=np.float64)
    if B > 0:
        a[sid] = max_h
    # enforce the evaluator's hard bounds
    np.clip(a, 0.0, 1000.0, out=a)

    # ---- 5. Validate (fallback to the provided parent if needed) -----
    # evaluate our candidate
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        score = float('inf')

    # evaluate fallback parent (if available) and keep the better one
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                        0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent.tolist()))
        if parent_score < score:
            a = parent
    except Exception:
        pass  # if parent missing or invalid we keep our candidate

    # ensure we finish before the deadline
    if time.monotonic() > deadline:
        # nothing special needed – we already have a valid vector
        pass

    return [float(x) for x in a]
