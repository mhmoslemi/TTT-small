import numpy as np
import time

def _next_pow2(x: int) -> int:
    """Return the next power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of a with itself using FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Find a step‑function (non‑negative vector) with a low AC‑1 bound."""
    rng = np.random.default_rng(seed)
    # keep a safety margin before the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------ #

    # ------------------------------------------------------------------ #

        try:
            parent = np.clip(
                np.asarray(height_sequence_1, dtype=np.float64),

                n = parent.size
                a = parent / parent.sum()
            else:

        except Exception:
            n = 256
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 256

    a = np.maximum(a, 1e-12)

    best_a = a.copy()
    # authoritative score of the starting vector

    idx_arr = np.arange(n, dtype=np.int64)
