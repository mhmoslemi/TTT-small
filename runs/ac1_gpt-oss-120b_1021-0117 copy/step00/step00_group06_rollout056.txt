def proj_simplex(v, s=1.0):
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # index count
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
