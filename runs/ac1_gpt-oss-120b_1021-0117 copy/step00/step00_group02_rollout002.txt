    # uniform candidate
    uniform = np.full(n, 1.0, dtype=np.float64)
    # evaluate both
    best = uniform
    best_score = float(evaluate_sequence(best.tolist()))
    # compare with base
    score_base = float(evaluate_sequence(base.tolist()))
    if score_base < best_score:
        best = base
        best_score = score_base
    # simple random search for a few seconds
    while time.monotonic() < deadline:
        # generate random perturbation around uniform within small sigma
        cand = uniform * np.exp(rng.normal(0.0, 0.1, size=n))

        s = np.sum(cand)
        if s < 0.01:

        # compute score
        sc = float(evaluate_sequence(cand.tolist()))
        if math.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
    return [float(x) for x in best]

import time
import math
import numpy as np

def _as_array(x):
    """Safely turn a possibly‑missing global into a 1‑D NumPy array."""
    try:
        arr = np.asarray(x, dtype=np.float64)
        if arr.ndim != 1 or arr.size == 0 or not np.isfinite(arr).all():
            raise ValueError
        return arr
    except Exception:
        # fallback length if no parent is provided
        return np.ones(256, dtype=np.float64)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound.

    The theoretical optimum is attained by a uniform sequence, which gives
    a score of exactly 2.  We compare the supplied parent (if any) with the
    uniform candidate and perform a few cheap random perturbations as a
    safety net, always keeping the best valid solution seen before the deadline.
    """
    rng = np.random.default_rng(seed)
    # Leave a small safety margin before the sandbox kills us.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Parent sequence (if present) → fallback to a constant vector.
    parent = _as_array(globals().get("height_sequence_1", None))
    n = parent.size

    # Uniform candidate – the proven optimum of the bound.
    uniform = np.full(n, 1.0, dtype=np.float64)

    # Evaluate both and keep the better one.
    best = uniform
    best_score = float(evaluate_sequence(best.tolist()))
    score_parent = float(evaluate_sequence(parent.tolist()))
    if math.isfinite(score_parent) and score_parent < best_score:
        best = parent
        best_score = score_parent

    # Very cheap stochastic refinement: a handful of log‑normal perturbations.
    # Each evaluation costs O(n log n) via NumPy's convolution; we stay well
    # within the time budget by limiting the loop.
    while time.monotonic() < deadline:

        if math.isfinite(score) and score < best_score:
