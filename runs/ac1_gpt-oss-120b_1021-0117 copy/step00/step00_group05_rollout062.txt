def reduce_peak(a, k, gamma):
    n = len(a)
    indices = np.arange(n)
    partners = k - indices
    mask = (partners >=0) & (partners < n)
    factor = np.ones_like(a)
    factor[mask] = 1 - gamma * a[partners[mask]]
    # To maintain positivity we can clip after.
    a_new = a * factor
    # ensure not negative
    a_new = np.clip(a_new, 0.0, 1000.0)
    # renorm
    s = a_new.sum()
    if s>0: a_new /= s
    return a_new

import numpy as np
import time

def _fft_max_convolution(seq, L):
    """Fast max of full convolution seq*seq using FFT of length L."""
    A = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * seq.size - 1]
    # Numerical noise can produce tiny negative values – ignore them.
    return float(conv.max() if conv.size else 0.0)


def _fast_score(seq, L):
    """Scale‑invariant score used for internal decisions."""
    n = seq.size
    max_b = _fft_max_convolution(seq, L)
    s = float(seq.sum())
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic search for a step‑function that lowers the AC1 upper bound.
    Returns a list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initialise from the supplied parent if available, otherwise use a
    # simple uniform start.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
        )
    else:
        base = np.ones(256, dtype=np.float64)

    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    seq = base.copy()
    total = seq.sum()
    if total <= 0.0:
        seq[:] = 1.0
        total = seq.sum()
    seq /= total                     # normalise to L1 = 1 (scale‑invariant)

    n = seq.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # ------------------------------------------------------------------
    # Evaluate the starting point exactly – this will be the fallback.
    # ------------------------------------------------------------------
    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    best_fast = _fast_score(best_seq, L)

    cur_seq = best_seq.copy()
    cur_fast = best_fast

    # SA parameters
    temperature = max(best_score * 0.05, 1e-3)
    cooling = 0.9997
    sigma = 0.15          # magnitude of multiplicative perturbations
    gamma = 0.06          # strength of peak‑reduction step

    # ------------------------------------------------------------------
    # Main loop – keep an eye on the deadline.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- choose a move -------------------------------------------------
        move = rng.integers(0, 5)
        cand = cur_seq.copy()

        if move == 0:
            # Random block multiplicative perturbation
            w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - w + 1)))
            block = slice(start, start + w)
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[block] *= factor

        elif move == 1:
            # Symmetrisation (averaging with its reversal)
            cand = 0.5 * (cand + cand[::-1])

        elif move == 2:
            # Gradient‑like reduction of the current maximal convolution entry
            A = np.fft.rfft(cand, n=L)
            conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
            k = int(conv.argmax())
            idx = np.arange(n)
            partner = k - idx
            mask = (partner >= 0) & (partner < n)
            # Reduce entries that directly feed the peak
            cand[mask] *= 1.0 - gamma * cand[partner[mask]]

        elif move == 3:
            # Random Dirichlet‑style restart (full redraw)
            cand = rng.random(n)
        else:
            # Tiny Gaussian jitter of the whole vector
            cand *= np.exp(rng.normal(0.0, sigma * 0.3, size=n))

        # Enforce bounds, renormalise to keep sum = 1 (scale‑invariant)
        np.clip(cand, 0.0, 1000.0, out=cand)
        s = float(cand.sum())
        if s < 1e-12:
            continue
        cand /= s

        # ------------------------------------------------------------------
        # Fast score for the Metropolis criterion
        # ------------------------------------------------------------------
        cand_fast = _fast_score(cand, L)

        # Metropolis acceptance
        improve = cand_fast < cur_fast
        if not improve:
            prob = np.exp(-(cand_fast - cur_fast) / max(temperature, 1e-12))
            improve = rng.random() < prob

        if improve:
            cur_seq = cand
            cur_fast = cand_fast

            # If this candidate looks promising, verify with the authoritative scorer
            if cand_fast < best_fast:

                if np.isfinite(exact) and exact < best_score:

                    best_score = exact
                    best_fast = cand_fast
                    # Early exit if we already beat the target bound
                    if best_score < 1.50274349:
                        return [float(x) for x in best_seq]

        # Temperature schedule
        temperature *= cooling

    # ------------------------------------------------------------------
    # Return the best sequence found (always a non‑empty list of floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
