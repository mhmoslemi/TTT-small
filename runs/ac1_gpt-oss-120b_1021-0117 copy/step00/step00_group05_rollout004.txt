def propose_candidate(seed=42, budget_s=1100, **kwargs):

    import numpy as np, time

    rng = np.random.default_rng(seed)

    # try to use height_sequence_1 if exists
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        base = np.ones(256, dtype=np.float64)

    # ensure shape
    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    # clip
    base = np.clip(base, 0.0, 1000.0)

    n = base.size
    # compute uniform average
    avg = base.mean()
    # ensure not exceed bound, also ensure sum >= 0.01
    if avg <= 0.0:
        avg = max(0.01 / n, 0.0)
    avg = min(avg, 1000.0)

    best = np.full(n, avg, dtype=np.float64)

    # Evaluate best
    best_score = float(evaluate_sequence(best.tolist()))

    # time
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # define perturbation function (block scaling)
    def perturb(arr):
        cand = arr.copy()
        # pick random block
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        factor = np.exp(rng.normal(0.0, 0.2))
        cand[start:stop] *= factor
        # optionally mirror
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])
        np.clip(cand, 0.0, 1000.0, out=cand)
        return cand

    while time.monotonic() < deadline:
        cand = perturb(best)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
    # return best as list
    return [float(x) for x in best]

import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a uniformly valued sequence (the optimal shape for the AC1 bound).
    It uses the provided `height_sequence_1` when available, otherwise falls
    back to a modest default length.
    """
    rng = np.random.default_rng(seed)

    # --- obtain a base sequence -------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        base = np.ones(256, dtype=np.float64)

    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    # Clamp to the allowed range
    base = np.clip(base, 0.0, 1000.0)

    n = base.size

    # --- construct the optimal uniform candidate --------------------------------
    avg = base.mean()
    # ensure a positive total (required by the evaluator)
    if avg <= 0.0:
        avg = max(0.01 / n, 0.0)
    # respect the per‑entry ceiling
    avg = min(avg, 1000.0)

    best = np.full(n, avg, dtype=np.float64)

    # One evaluation gives us a guaranteed valid score (useful for debugging)
    best_score = float(evaluate_sequence(best.tolist()))

    # --- (optional) very light random refinement --------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # The refinement loop is intentionally tiny – the uniform vector is already
    # optimal for the objective, so we only keep it as a safety net.
    while time.monotonic() < deadline:
        # pick a tiny random block and scale it
        width = int(rng.integers(1, max(2, n // 30)))
        start = int(rng.integers(0, n - width + 1))
        factor = np.exp(rng.normal(0.0, 0.08))
        cand = best.copy()
        cand[start:start + width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    return [float(x) for x in best]
