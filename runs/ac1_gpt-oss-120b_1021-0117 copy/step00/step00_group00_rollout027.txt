def solve_lp(f, M, idxs):   
    n = len(f)
    K = len(idxs)
    A = np.zeros((K, n), dtype=np.float64)
    for row_idx, k in enumerate(idxs):
        # Fill row
        j_start = max(0, k - n + 1)
        j_end = min(k, n-1)
        # for j in j_start..j_end inclusive:
        # set A[row_idx, j] = f[k-j]
        # can assign via slice: A[row_idx, j_start:j_end+1] = f[k - np.arange(j_start, j_end+1)]
        js = np.arange(j_start, j_end+1)
        A[row_idx, js] = f[k - js]
    b = np.full(K, M, dtype=np.float64)
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        return None
    g = res.x
    return g

import numpy as np
import time
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: a simple local perturbation (used when LP fails or for exploration)
def make_block_perturbation(base, rng):
    """Return a non‑negative O(n) perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # random block width (5‑30% of n)
    w = int(rng.integers(max(1, n // 30), max(2, n // 3)))
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = min(n, start + w)
    # multiplicative tweak of the block
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional mirror‑symmetrisation
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Linear‑programming step (sparse constraints)
def solve_lp_for_g(f_vec, max_conv, active_idx):
    """Solve  max Σg  s.t.  f*g ≤ max_conv on the selected shifts.
       Returns None on failure."""
    n = f_vec.size
    k_cnt = len(active_idx)
    # Build constraint matrix A (k_cnt × n)
    A = np.zeros((k_cnt, n), dtype=np.float64)
    for r, k in enumerate(active_idx):
        # valid j satisfy 0 ≤ j ≤ n‑1 and 0 ≤ k‑j ≤ n‑1
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1, dtype=np.intp)
        A[r, js] = f_vec[k - js]          # f_{k‑j}
    # Normalise to improve numerical stability
    if max_conv == 0:
        return None
    A_norm = A / max_conv
    b_norm = np.ones(k_cnt, dtype=np.float64)

    # Objective: maximise Σ g  →  minimise −Σ g
    c = -np.ones(n, dtype=np.float64)

    # Variable bounds: 0 ≤ g_i ≤ 1000  (same limits as the evaluator)
    bounds = [(0.0, 1000.0) for _ in range(n)]

    # Solve with HiGHS (fast for many variables, few constraints)
    try:
        res = linprog(c, A_ub=A_norm, b_ub=b_norm,
                      bounds=bounds, method='highs', options={'presolve': True})
    except Exception:                               # pragma: no cover
        return None
    if not res.success:
        return None
    return res.x
