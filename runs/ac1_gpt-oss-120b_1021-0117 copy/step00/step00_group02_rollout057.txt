        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        n = a.size

        shift = np.zeros_like(a)

        # update
        cand = a - eta * grad
        cand = np.clip(cand, 0.0, None)  # nonnegative

def _conv_max(a, L):
    A = np.fft.rfft(a, n=L)
    B = A * A
    conv = np.fft.irfft(B, n=L)[:2*a.shape[0]-1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx, conv

    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # initial sequence
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except NameError:

    init = np.clip(init, 0.0, 1000.0)
    n = init.size
    target_sum = math.sqrt(2.0 * n)
    total = init.sum()
    if total == 0.0:
        a = np.full(n, target_sum / n, dtype=np.float64)
    else:
        a = init * (target_sum / total)

    L = 1 << ((2 * n - 1).bit_length())

    # best so far
    best_val, _, _ = _conv_max(a, L)
    best_a = a.copy()

    step_factor = 0.2

        # find the pair (i,j) that contributes most to this max
        i_idx = np.arange(n)

        i_valid = i_idx[mask]
        j_valid = j_idx[mask]
        if i_valid.size == 0:

import numpy as np
import time
import math


def _max_convolution(a: np.ndarray, L: int):
    """Return (max value, index of max, full convolution) of a*a using FFT."""
    A = np.fft.rfft(a, n=L)
    B = A * A
    conv = np.fft.irfft(B, n=L)[: 2 * a.shape[0] - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx, conv


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a coefficient vector with a low AC‑1 upper bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ---- initial vector -------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except NameError:
        init = np.ones(256, dtype=np.float64)

    init = np.clip(init, 0.0, 1000.0)
    n = init.size
    target_sum = math.sqrt(2.0 * n)

    s = init.sum()
    if s == 0.0:
        a = np.full(n, target_sum / n, dtype=np.float64)
    else:
        a = init * (target_sum / s)

    # FFT length (next power of two >= 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ---- keep the best solution found -----------------------------------
    best_val, _, _ = _max_convolution(a, L)
    best_a = a.copy()

    step_factor = 0.2          # size of each mass transfer
    min_step = 1e-12

    while time.monotonic() < deadline:
        cur_max, k_max, _ = _max_convolution(a, L)

        # ---- find the pair (i,j) that contributes most to the current max ----
        i_all = np.arange(n)
        j_all = k_max - i_all
        mask = (j_all >= 0) & (j_all < n)
        i_valid = i_all[mask]
        j_valid = j_all[mask]
        if i_valid.size == 0:
            break

        contrib = a[i_valid] * a[j_valid]
        best_pair = int(np.argmax(contrib))
        i_sel = int(i_valid[best_pair])
        j_sel = int(j_valid[best_pair])

        delta = step_factor * min(a[i_sel], a[j_sel])
        if delta < min_step:
            step_factor *= 0.5
            if step_factor < 1e-6:
                break
            continue

        # ---- move a little mass from the heavy pair towards the centre ----
        cand = a.copy()
        cand[i_sel] -= delta
        cand[j_sel] -= delta

        # recipient: roughly halfway between the two indices
        if i_sel != j_sel:
            rec = (i_sel + j_sel) // 2
        else:
            rec = rng.integers(0, n)
        if rec == i_sel or rec == j_sel:
            rec = (rec + 1) % n
        cand[rec] += 2.0 * delta

        cand = np.clip(cand, 0.0, 1000.0)

        new_max, _, _ = _max_convolution(cand, L)

        if new_max + 1e-12 < cur_max:
            a = cand
            step_factor *= 1.1

                best_val = new_max
                best_a = cand.copy()
        else:
