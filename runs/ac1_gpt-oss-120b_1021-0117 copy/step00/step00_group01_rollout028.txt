    """Fast wrapper around the authoritative evaluator for a NumPy array."""

def _random_sparse_candidate(rng: np.random.Generator, n: int) -> np.ndarray:
    """Create a random sparse non‑negative vector of length n."""
    # Choose a sparsity level between 1 % and 20 % of the entries.
    frac = rng.uniform(0.01, 0.20)
    m = max(2, int(frac * n))
    # Random distinct indices.
    idx = rng.choice(n, size=m, replace=False)
    # Positive values (exponential is heavy‑tailed, good for exploration).
    vals = rng.exponential(scale=1.0, size=m) + 1e-6
    cand = np.zeros(n, dtype=np.float64)
    cand[idx] = vals
    # Clip to the permitted range.
    return np.clip(cand, 0.0, 1000.0)


def _block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random block of the base vector."""
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12 (as in the reference solution)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional mirroring – cheap symmetry operation
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _swap_two(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap two entries in a copy of base."""
    cand = base.copy()
    n = cand.size
    i, j = rng.choice(n, size=2, replace=False)
    cand[i], cand[j] = cand[j], cand[i]
    return cand


def _transfer_mass(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a tiny amount of mass from one index to another."""
    cand = base.copy()
    n = cand.size
    i, j = rng.choice(n, size=2, replace=False)
    if cand[i] <= 0.0:
        return cand
    # move at most 10 % of the source entry
    delta = rng.random() * cand[i] * 0.10
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)

    initialisations.

    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    parent = np.clip(parent, 0.0, 1000.0)
    best = parent.copy()
