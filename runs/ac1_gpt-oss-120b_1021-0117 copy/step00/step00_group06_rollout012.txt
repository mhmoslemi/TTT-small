            si = i + i

                sums = i + np.array(S, dtype=int)
                if used[sums].any():
                    continue

            # update used for sums with new i
            # includes self sum as part of sums list
            sums_new = i + np.array(S, dtype=int)  # includes i itself
            used[sums_new] = True
        return np.array(S, dtype=int)

        m = S.size
        if m < 2:  # need at least 2 maybe ratio not good

def compute_max_b(seq):
    n = len(seq)
    # zero pad to length next pow of 2 >= 2n-1
    size = 1 << (2*n-1).bit_length()
    # compute FFT
    S = np.fft.rfft(seq, n=size)   # we can use rfft for real.
    conv = np.fft.irfft(S * S, n=size)
    conv = conv[:2*n-1]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # define length n from height_sequence_1 if exists else default
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 256

    # holds best candidate and best score
    best_seq = np.ones(n, dtype=np.float64)  # fallback uniform
    best_seq = best_seq / best_seq.sum()   # normalise to sum 1
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # optionally start with parent if exists
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.size == n and np.isfinite(parent).all():
            parent = parent / parent.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            if parent_score < best_score:
                best_seq = parent.copy()
                best_score = parent_score

    # function to compute ratio (AC1) from sequence
    def compute_score(a):
        n = a.shape[0]
        # zero pad to next power of two for FFT (speed)
        size = 1 << ((2*n - 1).bit_length())
        F = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(F * F, n=size)[:2*n-1]
        max_b = conv.max()
        sum_a = a.sum()
        if sum_a < 1e-12:
            return float('inf')
        return float(2.0 * n * max_b / (sum_a * sum_a))

    # fast compute conv and max bucket
    def conv_max_and_k(a):
        n = a.shape[0]
        size = 1 << ((2*n - 1).bit_length())
        F = np.fft.rfft(a, n=size)
        c = np.fft.irfft(F * F, n=size)[:2*n-1]
        maxb = c.max()
        k = int(c.argmax())
        return maxb, k, c

    # main hillclimbing loop
    a = best_seq.copy()

    # we might run multiple restarts
    while time.monotonic() < deadline:
        # run local optimization from current a for a block of steps
        # set a local copy for internal steps
        a_local = a.copy()
        score_local = compute_score(a_local)
        # For each step, try a random move improving score
        steps = 0
        # set step limit per restart based on remaining time
        max_steps = 2000
        while steps < max_steps and time.monotonic() < deadline:
            maxb, k_max, conv_arr = conv_max_and_k(a_local)
            # identify indices contributing to k_max
            i_start = max(0, k_max - (n-1))
            i_end = min(n-1, k_max)
            # potential i indices with nonzero a
            # compute mask of nonzero a
            # we can randomly choose i among range but ensure a[i]>0 and a[j]>0
            # we attempt limited tries
            found = False
            for _ in range(5):
                i = rng.integers(i_start, i_end+1)
                j = k_max - i
                if a_local[i] > 1e-12 and a_local[j] > 1e-12:
                    found = True
                    break
            if not found:
                # no contributing pair found, break out
                break
            # choose a target index t not equal to i
            # Prefer indices where a[t] is relatively small
            # Sample until we find t where a[t] < a_local[i] maybe
            t = rng.integers(0, n)
            # avoid picking i itself
            while t == i:
                t = rng.integers(0, n)
            # move small portion
            eps_factor = rng.uniform(0.01, 0.1)
            delta = eps_factor * a_local[i]
            new_val_i = a_local[i] - delta
            new_val_t = a_local[t] + delta
            if new_val_i < 0:
                delta = a_local[i]  # move whole
                new_val_i = 0.0
                new_val_t = a_local[t] + delta
            a_candidate = a_local.copy()
            a_candidate[i] = new_val_i
            a_candidate[t] = new_val_t
            # recompute score
            cand_score = compute_score(a_candidate)
            if cand_score < score_local - 1e-12:
                a_local = a_candidate
                score_local = cand_score
                # update global best if improved
                if cand_score < best_score:
                    best_seq = a_local.copy()
                    best_score = cand_score
            steps += 1
            # optional early break if score low enough
            if best_score <= 1.5028:
                # good enough, break everything
                return [float(x) for x in best_seq]
        # after inner loop, decide whether to restart with random perturbation
        # random perturbation: perturb a by small random multiplicative factor
        # keep sum normalized
        jitter = np.exp(rng.normal(0.0, 0.05, size=n))
        a = a_local * jitter
        # clip to [0,1000] (not needed)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / s

    return [float(x) for x in best_seq]

import numpy as np
import time

def _conv_max_and_idx(a):
    """Return max convolution value, its index, and the full convolution."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = conv.max()
    idx = int(conv.argmax())
    return max_val, idx, conv


def _ac1_score(a, max_conv=None):
    """Exact AC1 score using the same formula as evaluate_sequence."""
    n = a.shape[0]
    if max_conv is None:
        max_conv, _, _ = _conv_max_and_idx(a)
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a sequence with a low AC1 bound.

    The algorithm starts from a reference (the imported height_sequence_1 when
    available) and performs a simple projected sub‑gradient descent that
    repeatedly moves a small amount of mass away from the convolution bucket
    attaining the current maximum.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # initialise a base sequence -------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            base = np.ones(256, dtype=np.float64)
    else:
        base = np.ones(256, dtype=np.float64)

    # normalise to unit L1 norm – the AC1 bound is scale invariant
    a = base.copy()
    s = a.sum()
    a = a / s if s > 0 else a

    n = a.shape[0]

    # ------------------------------------------------------------------
    # best‑so‑far bookkeeping ----------------------------------------------------
    best_seq = a.copy()
    best_score = _ac1_score(best_seq)

    # ------------------------------------------------------------------
    # main optimisation loop ----------------------------------------------------
    while time.monotonic() < deadline:
        # compute current maximum bucket
        maxb, k_max, _ = _conv_max_and_idx(a)

        # indices that contribute to the bucket k_max
        i0 = max(0, k_max - (n - 1))
        i1 = min(n - 1, k_max)
        i_candidates = np.arange(i0, i1 + 1, dtype=np.int64)
        j_candidates = k_max - i_candidates

        # keep only pairs where both entries are positive
        mask = (a[i_candidates] > 1e-12) & (a[j_candidates] > 1e-12)
        if not np.any(mask):
            # nothing to move – break out of the inner optimisation
            break

        i_c = i_candidates[mask]
        j_c = j_candidates[mask]

        # product contributions of each ordered pair (i,j)
        prod = a[i_c] * a[j_c]
        # pick the largest product – reducing it tends to lower the max bucket
        idx = int(np.argmax(prod))
        i = int(i_c[idx])
        j = int(j_c[idx])

        # choose two destination slots (could be the same)
        t1 = int(rng.integers(0, n))
        t2 = int(rng.integers(0, n))
        while t1 == i:
            t1 = int(rng.integers(0, n))
        while t2 == j:
            t2 = int(rng.integers(0, n))

        # amount to move – a small fraction of the source entries
        eps_i = float(rng.uniform(0.02, 0.09))
        eps_j = float(rng.uniform(0.02, 0.09))
        delta_i = eps_i * a[i]
        delta_j = eps_j * a[j]

        a_new = a.copy()
        a_new[i] -= delta_i
        a_new[t1] += delta_i
        a_new[j] -= delta_j
        a_new[t2] += delta_j

        # enforce non‑negativity (tiny numerical issues)
        a_new = np.clip(a_new, 0.0, None)

        # renormalise to keep the L1 norm fixed (scale does not affect the score)
        s_new = a_new.sum()
        if s_new > 0:
            a_new /= s_new

        # evaluate the candidate – use the fast routine (no extra clipping needed)
        new_score = _ac1_score(a_new)

        # accept only if we improve
        if new_score < best_score - 1e-12:
            a = a_new
            best_seq = a_new.copy()
            best_score = new_score
            # stop early if we have already passed the target bound
            if best_score <= 1.50275:
                break
        else:
            # occasional random jitter to escape flat regions
            if rng.random() < 0.03:
                jitter = np.exp(rng.normal(0.0, 0.04, size=n))
                a = a * jitter
                a /= a.sum()
        # safety break if time runs out
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # final safety: evaluate with the authoritative evaluator
    # (this also guarantees clipping to the allowed range)
    final = best_seq.copy()
    final = np.clip(final, 0.0, 1000.0)
    # avoid a zero‑sum edge case
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
