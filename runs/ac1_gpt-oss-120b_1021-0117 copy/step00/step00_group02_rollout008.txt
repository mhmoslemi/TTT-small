import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Generate a monotone power‑law sequence and keep the best by exact score."""
    # time budget (leave a small safety margin)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # length of the problem – fall back to a modest size if the parent is missing
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise Exception
        n = base.size
    except Exception:
        n = 256

    i = np.arange(n, dtype=np.float64)                # indices 0 … n‑1
    t = i / (n - 1) if n > 1 else np.zeros_like(i)    # normalized to [0,1]

    best_seq = None
    best_score = float('inf')

    # try several power‑law shapes; exponent 0 gives the uniform (worst) case,
    # higher exponents concentrate mass at the right end and dramatically
    # lower the AC1 bound.
    for p in (0, 1, 2, 3, 4, 5, 6):
        if time.monotonic() >= deadline:
            break
        seq = 1000.0 * np.power(t, p)                # monotone, max = 1000
        # safety clamp (not really needed, but follows the evaluator's limits)
        seq = np.clip(seq, 0.0, 1000.0)
        # evaluate with the authoritative scorer
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()

    # If for any reason no sequence was accepted (should not happen), fall back
    # to a simple uniform candidate.
    if best_seq is None:
        best_seq = np.full(n, 500.0, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Return a plain Python list of floats as required by the harness
    return [float(x) for x in best_seq]
