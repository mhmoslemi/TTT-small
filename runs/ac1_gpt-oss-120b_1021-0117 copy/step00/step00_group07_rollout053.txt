        # Evaluate forward and reversed
        try_candidate(arr)
        try_candidate(arr[::-1])
    # optional random refinements: small perturbations
    while time.monotonic() < deadline:
        # random perturbation similar to block perturbation but smaller step
        candidate = best.copy()
        # add Gaussian noise to random subset
        width = max(1, n // 200)

            continue
        # compute q**n safely via log
        logq = math.log(q)
        qn = math.exp(n * logq)                 # = q**n
        denom = 1.0 - qn
        if denom <= 0.0:
            continue
        C = S * alpha / denom                    # ensures sum ≈ S
        arr = C * np.exp(idx * logq)            # q**idx = exp(idx*logq)
        consider(arr)                            # forward exponential
        consider(arr[::-1])                      # mirrored version

    # ------------------------------------------------------------------
    # optional tiny local refinements (quick, bounded)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # Randomly perturb a small block of the current best
        cand = best.copy()
        width = max(1, n // 200)
        start = rng.integers(0, n - width + 1)
        stop = start + width
        factor = np.exp(rng.normal(0.0, 0.06, size=stop - start))
        cand[start:stop] *= factor
        if rng.random() < 0.2:                  # occasional symmetrisation
            cand = 0.5 * (cand + cand[::-1])
        consider(cand)

    return [float(x) for x in best]
```
