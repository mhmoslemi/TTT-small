def propose_candidate(seed=42, budget_s=1100, **kwargs):

    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as evaluated by ``evaluate_sequence``.
    The search stays O(n) in memory and respects the time budget.
    """
    rng = np.random.default_rng(seed)
    # Reserve a small safety margin before the hard timeout.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Length of the problem (use the provided parent if it exists).
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    # ------------------------------------------------------------------
    # Initial candidate – a perfectly uniform vector (theoretical optimum).
    # ------------------------------------------------------------------
    best_seq = np.full(n, 1.0, dtype=np.float64)          # any constant works
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # Simple stochastic search that stays linear in memory.
    # ----------------------------------------------------
    # The objective for non‑negative vectors attains its lower bound 2 at a
    # uniform distribution, but floating‑point rounding can give a tiny
    # improvement.  We therefore probe a few nearby families.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # Choose a perturbation style.
        mode = rng.integers(3)

        if mode == 0:
            # Uniform constant with a random magnitude.
            c = rng.uniform(0.001, 1000.0)
            cand = np.full(n, c, dtype=np.float64)

        elif mode == 1:
            # Uniform plus a low‑amplitude sinusoid.
            base = rng.uniform(0.1, 1000.0)
            amp  = rng.uniform(0.0, base * 0.15)
            freq = rng.integers(1, max(2, n // 8))
            x = np.arange(n, dtype=np.float64)
            sin_term = amp * np.sin(2.0 * np.pi * freq * x / n)
            cand = np.full(n, base, dtype=np.float64) + sin_term

        else:
            # Localised multiplicative perturbation of the current best.
            width = rng.integers(max(1, n // 200), max(2, n // 12))
            start = rng.integers(0, max(1, n - width + 1))
