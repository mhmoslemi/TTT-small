import numpy as np
import time

def make_block_perturbation(base, rng):
    """
    Produce a cheap O(n) perturbation of `base`.
    The perturbation multiplies a random contiguous block by a log‑normal factor
    and optionally symmetrises the whole array.
    """
    cand = np.array(base, copy=True)
    n = cand.shape[0]
    # Choose a block size between n/200 and n/12 (min 1, max n)
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # Apply a log‑normal scaling to the block
    scale = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= scale
    # With some probability mirror the sequence (helps escape local minima)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
