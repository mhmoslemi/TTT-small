def targeted_move(seq, rng):
    n = seq.size
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (all top level, no closures)
# ------------------------------------------------------------
def _initial_sqrt_shape(n: int) -> np.ndarray:
    """
    Return a probability vector a of length n with a_i ∝ 1/√i (i=1..n)
    and L1 norm equal to 1.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 .. n
    a = 1.0 / np.sqrt(i)
    a /= a.sum()
    return a


def _perturb_block(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative random perturbation of a random contiguous block."""
    n = seq.size
    # width between n/200 and n/12 (at least 1)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.08)))       # moderate log‑normal tweak
    cand = seq.copy()
    cand[start:start + width] *= factor
    cand = np.clip(cand, 0.0, None)                     # keep non‑negative
    # renormalise to keep L1 = 1 (scale‑invariant objective)
    s = cand.sum()
    if s > 0:
        cand /= s
    return cand


def _transfer_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = seq.size
    i, j = rng.choice(n, size=2, replace=False)
    if seq[i] <= 0.0:
        return seq.copy()
    # up to 30 % of the source mass is transferred
    frac = rng.random() * 0.30
    delta = seq[i] * frac
    cand = seq.copy()
    cand[i] -= delta
    cand[j] += delta
    # L1 is unchanged, but guard against tiny negative rounding
    cand = np.clip(cand, 0.0, None)
    return cand


def _targeted_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Reduce the peak of the convolution.
    Pick an index participating in the current maximal convolution entry
    (probability proportional to its contribution) and shift a fraction of its
    mass to a random other location.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))               # location of the current maximum
    idx = np.arange(n, dtype=np.int64)
    rev = k - idx
    mask = (rev >= 0) & (rev < n)

    # contribution = a_i * a_{k-i}
    contrib = np.zeros_like(seq)
    contrib[mask] = seq[idx[mask]] * seq[rev[mask]]
    tot = contrib.sum()
    if tot == 0.0:
        return seq.copy()

    probs = contrib / tot
    i = rng.choice(n, p=probs)             # index weighted by its contribution
    if seq[i] <= 0.0:
        return seq.copy()
    # choose a different destination uniformly
    j = rng.integers(0, n - 1)
    while j == i:
        j = rng.integers(0, n - 1)

    # move at most 25 % of the source weight
    frac = rng.random() * 0.25
    delta = seq[i] * frac
    cand = seq.copy()
    cand[i] -= delta
    cand[j] += delta
    cand = np.clip(cand, 0.0, None)        # keep non‑negative (sum unchanged)
    return cand

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Produce a non‑negative sequence of heights that (hopefully) drives the
    AC1 upper bound below the historic 2.0 benchmark.
    The routine keeps the best candidate seen before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    # Give a tiny safety margin for the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # -----------------------------------------------------------------
    # Determine problem size and a reasonable starting point
    # -----------------------------------------------------------------
    try:

        base = np.clip(base, 0.0, 1000.0)

        if base.sum() > 0.0:
            base = base / base.sum()
    except Exception:
        # Fallback to a moderate default size if the parent is missing/invalid
        n = 256
        base = np.ones(n, dtype=np.float64) / n

    # Primary seed: smooth 1/√ shape (known to give ≈π/2 ≈1.57)

    best = a.copy()

    # Stochastic local search with several move types
    # -----------------------------------------------------------------
    while time.monotonic() < deadline:

        if move_type < 0.45:

        elif move_type < 0.80:
