        shift[valid] = a[idx[valid]]
        # compute direction
        direction = max_val - shift * S
        # If direction is all zeros (stalled), add random jitter and continue
        if np.allclose(direction, 0, atol=1e-12):
            # small random perturbation
            a = a + 0.01 * (rng.random(n) - 0.5)
            a = np.clip(a, 0, None)
            if a.sum() == 0:
                a[0] = 1.0
            a = a / a.sum()
            continue
        # step candidate
        a_candidate = a + eta * direction
        a_candidate = np.clip(a_candidate, 0.0, None) # no upper bound (not needed)
        # renorm
        if a_candidate.sum() == 0:
            a_candidate = a
        else:
            a_candidate = a_candidate / a_candidate.sum()
        # compute new max after candidate
        conv_new = np.convolve(a_candidate, a_candidate)
        max_val_new = np.max(conv_new)
        S_new = np.sum(a_candidate)  # should be 1
        score_new = 2 * n * max_val_new / (S_new * S_new)
        if score_new < score:
            a = a_candidate
            # encourage larger step if improvement
            eta = min(eta * 1.2, 5.0)
        else:
            # reduce step size and try again maybe
            eta = eta * 0.5
            # optional small random perturb
            if eta < 1e-6:
                # random restart
                a = np.maximum(a * np.exp(rng.normal(0, 0.1, size=n)), 0.0)
                a = a / a.sum()
                eta = 0.5
                continue
        # occasionally randomize (prob 1%)
        if rng.random() < 0.01:
            # small random multiplicative noise
            a = a * np.exp(rng.normal(0, 0.05, size=n))
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
    # final ensure all entries <= 1000
    best = np.clip(best, 0.0, 1000.0)
    # return list
    return [float(x) for x in best]

    deadline = time.monotonic() + max(0.0, budget_s - 5.0)
    # Determine n

    # Initialize a (random Dirichlet)
    a = rng.exponential(scale=1.0, size=n)
    a = a / a.sum()

import numpy as np
import time

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative; return uniform distribution
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimizes the AC1 bound evaluated by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    # time budget, leave a small safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Determine problem size and an initial feasible vector (sum = 1)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            n = base.size
            a0 = np.clip(base, 0.0, 1000.0)
        else:
            n = 256
            a0 = np.ones(n, dtype=np.float64)
    else:
        n = 256
        a0 = np.ones(n, dtype=np.float64)

    # normalize to a probability vector (sum = 1)
    a0 = a0 / a0.sum()
    # ------------------------------------------------------------------
    # Helper to evaluate the objective (sum=1 => denominator = 1)
    # ------------------------------------------------------------------
    def _score(vec):
        c = np.convolve(vec, vec)
        return float(2 * n * np.max(c) / (np.sum(vec) ** 2))

    # start with a few sensible candidates
    best = a0.copy()
    best_score = _score(best)

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = _score(uniform)
    if uniform_score < best_score:
        best, best_score = uniform.copy(), uniform_score

    # ------------------------------------------------------------------
    # Main stochastic projected‑gradient loop
    # ------------------------------------------------------------------
    step_initial = 0.5          # initial step length for the gradient
    eta = step_initial
    no_improve = 0
    max_no_improve = 150       # after this many useless steps we restart

    # start from a random Dirichlet point (exponential distribution)
    cur = rng.exponential(scale=1.0, size=n)
    cur = cur / cur.sum()

    while time.monotonic() < deadline:
        # compute convolution and the active index (where the max occurs)
        conv = np.convolve(cur, cur)
        max_val = float(conv.max())
        idx = int(conv.argmax())               # index of the maximal entry

        # gradient of the active quadratic term D_idx w.r.t a
        # (∂D_idx/∂a_i) = 2 * a_{idx-i}   (zero if out of range)
        ids = idx - np.arange(n)
        mask = (ids >= 0) & (ids < n)
        grad = np.zeros_like(cur)
        grad[mask] = cur[ids[mask]]            # we drop the factor 2 (scale absorbed in η)

        # proposed update: projected gradient step
        cand = cur - eta * grad
        cand = _project_simplex(cand)           # enforce a_i >= 0 and sum = 1

        # evaluate the candidate

        if cand_score < best_score:

    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
