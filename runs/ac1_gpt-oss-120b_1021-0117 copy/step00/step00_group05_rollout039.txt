def project_simplex(v, s=1.0):
    """Project onto simplex {x >=0, sum x = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time

def _project_onto_simplex(v):
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a step‑function that lowers the AC1 upper bound.

    The routine starts from the provided ``height_sequence_1`` (if present),
    normalises it to unit sum and then performs a simple stochastic
    projected‑gradient / hill‑climbing search while respecting the non‑negative
    simplex constraint.  The best sequence found before the time budget expires
    is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    # reserve a few seconds for graceful termination
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)

    # --------------------------------------------------------------
    # initialise from the parent if possible, otherwise fall back
    # --------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)  # generic fallback size
    base = np.clip(base, 0.0, 1000.0)
    if base.sum() < 1e-12:
        base[:] = 1.0
    a = base / base.sum()                     # guarantee unit sum
    n = a.size

    # keep the best solution seen so far
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------------
    # a cheap deterministic baseline (uniform) – might already be better
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # --------------------------------------------------------------
    # main stochastic optimisation loop
    # --------------------------------------------------------------
    # we work directly with the unit‑sum simplex, so the objective is
    # scale‑invariant.
    while time.monotonic() < deadline:
        # ---- propose a small transfer between two coordinates ----------
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] <= 0.0:
            continue
        # move a random fraction (1–20 %) of a[i] to a[j]
        delta = a[i] * rng.uniform(0.01, 0.20)
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta                       # sum stays 1

        # ---- evaluate --------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            a = cand.copy()
            continue   # keep the improvement

        # ---- occasional global perturbation to escape local minima -----
        if rng.random() < 0.003:
            # multiplicative noise on the whole vector, then re‑project
            noisy = a * np.exp(rng.normal(0.0, 0.07, size=n))
            noisy = np.clip(noisy, 0.0, None)
            noisy_sum = noisy.sum()
            if noisy_sum > 0.0:
                noisy = noisy / noisy_sum
                score_noisy = float(evaluate_sequence(noisy.tolist()))
                if np.isfinite(score_noisy) and score_noisy < best_score:
                    best = noisy.copy()
                    best_score = score_noisy
                    a = noisy.copy()
                    continue

        # ---- random restart with a Dirichlet draw (tiny probability) ---
        if rng.random() < 0.001:
            restart = rng.dirichlet(np.ones(n))
            score_restart = float(evaluate_sequence(restart.tolist()))
            if np.isfinite(score_restart) and score_restart < best_score:
                best = restart.copy()
                best_score = score_restart
                a = restart.copy()

    # --------------------------------------------------------------
    # convert to plain list of Python floats for the evaluator
    # --------------------------------------------------------------
    return [float(x) for x in best]
