import numpy as np, time

def evaluate(seq): return evaluate_sequence(seq.tolist())

def generate_linear(n):
    # descending from max to 0
    return np.linspace(1000.0, 0.0, n, dtype=np.float64)

def generate_exponential(n, r):
    # a_i = 1000 * r**i
    # use np.power for speed
    return 1000.0 * np.power(r, np.arange(n))

def make_block_perturbation(base, rng):
    # same as earlier but maybe smaller sigma
    candidate = base.copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))  # same
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.08)))  # small sigma
    if rng.random() < 0.2:
        candidate = 0.5 * (candidate + candidate[::-1])
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    # Determine n
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent_arr = np.ones(256, dtype=np.float64)
    n = parent_arr.size if parent_arr.size>0 else 256
    # Start candidates list
    best = None
    best_score = np.inf

    # evaluate parent if valid
    if parent_arr.size > 0:
        parent = np.clip(parent_arr, 0.0, 1000.0)
        score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(score):
            best = parent.copy()
            best_score = score

    # generate linear decreasing
    linear = generate_linear(n)
    linear = np.clip(linear, 0.0, 1000.0)
    score = float(evaluate_sequence(linear.tolist()))
    if score < best_score:
        best = linear
        best_score = score

    # try several exponential rates
    for r in [0.9999, 0.9995, 0.9990, 0.998, 0.997]:
        exp_seq = generate_exponential(n, r)
        exp_seq = np.clip(exp_seq, 0.0, 1000.0)
        s = float(evaluate_sequence(exp_seq.tolist()))
        if s < best_score:
            best = exp_seq
            best_score = s

    # simple random start maybe uniform
    # skip if time low

    # local search loop
    while time.monotonic() < deadline:
        if best is None:
            # start from linear fallback
            base = generate_linear(n)
        else:
            base = best
        cand = make_block_perturbation(base, rng)
        # optionally add random reordering to explore shape
        if rng.random() < 0.15:
            idx = rng.permutation(n)
            cand = cand[idx]
        # evaluate
        s = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(s) and s < best_score:
            best = cand
            best_score = s
    # Return list
    if best is None:
        # fallback uniform
        best = np.full(n, 1.0)
    return [float(x) for x in best]

def _linear_ramp(n: int) -> np.ndarray:
    """Monotonically decreasing ramp from 1000 down to 0 (inclusive)."""
    return np.linspace(1000.0, 0.0, n, dtype=np.float64)


def _exponential_ramp(n: int, r: float) -> np.ndarray:
    """A decreasing exponential a_i = 1000 * r**i , 0 < r < 1."""
    return 1000.0 * np.power(r, np.arange(n, dtype=np.float64))


def _perturb(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Small block‑wise multiplicative perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # block width between 0.5% and 5% of n
    lo = max(1, n // 200)
    hi = max(2, n // 20)
