def shape_sequence(n, p=0.5, offset=0.0, cutoff_frac=0.0):
    # index array
    i = np.arange(n, dtype=np.float64)
    # compute base values:
    vals = (i + 1 + offset) ** (-p)
    if cutoff_frac > 0:
        cutoff = int(cutoff_frac * n)
        vals[:cutoff] = 0.0
    # clip
    return np.clip(vals, 0.0, 1000.0)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
    if n == 0:
        n = 256
    # define function to evaluate
    def eval_seq(arr):
        return float(evaluate_sequence(arr.tolist()))
    # initialize best using shape
    best = np.array((np.arange(n, dtype=float) + 1) ** -0.5, dtype=float)
    best = np.clip(best, 0.0, 1000.0)
    best_score = eval_seq(best)
    # also consider a few parameterized shapes
    for _ in range(30):
        p = rng.uniform(0.45, 0.55)
        off = rng.uniform(0, 5.0)
        cand = np.clip((np.arange(n, dtype=float) + 1 + off) ** (-p), 0.0, 1000.0)
        score = eval_seq(cand)
        if score < best_score:
            best = cand
            best_score = score
    # hill climbing
    # keep a stable sum of best for scaling
    target_sum = best.sum()
    while time.monotonic() < deadline:
        cand = best.copy()
        # random block perturbation
        width = int(rng.integers(1, max(2, n // 100)))  # small block
        start = int(rng.integers(0, n - width + 1))
        cand[start:start+width] *= np.exp(rng.normal(0.0, 0.1, size=width))
        # random perturb some random indices
        k = int(rng.integers(1, max(2, n // 200)))
        idx = rng.choice(n, size=k, replace=False)
        cand[idx] *= np.exp(rng.normal(0.0, 0.2, size=k))
        # clip and renormalize sum to target_sum
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s > 0:
            cand *= (target_sum / s)
        # evaluate
        score = eval_seq(cand)
        if score < best_score:
            best = cand
            best_score = score
            target_sum = best.sum()   # keep updated
    # final fallback: ensure valid
    if not np.isfinite(best_score):
        # fallback to uniform
        best = np.ones(n, dtype=float)
        best_score = eval_seq(best)
    return [float(x) for x in best]

import numpy as np
import time

def _make_power_shape(n, p=0.5, offset=0.0):
    """a_i = (i+1+offset) ** (-p)  , non‑negative."""
    idx = np.arange(n, dtype=np.float64)
    a = (idx + 1.0 + offset) ** (-p)
    return np.clip(a, 0.0, 1000.0)


def _apply_center_modulation(a, lam, rng):
    """Reduce central values (where the convolution peak lives)."""
    n = a.size
    # quadratic bowl: 1 - ( (i-mid)/mid )**2  in [0,1]
    mid = (n - 1) / 2.0
    t = (np.arange(n, dtype=np.float64) - mid) / mid          # in [-1,1]
    factor = 1.0 - lam * (1.0 - t * t)                         # >=0  if lam<=1
    cand = a * factor
    cand = np.clip(cand, 0.0, 1000.0)
    # renormalise to keep the total mass unchanged (scaling does not affect the ratio)
    s = cand.sum()
    if s > 0.0:
        cand *= (a.sum() / s)
    return cand


def _block_perturb(a, rng):
    """A cheap O(n) multiplicative perturbation of a random block."""
    n = a.size
    cand = a.copy()
    # random block size (≈ 0.5‑2 % of n)
    width = int(rng.integers(max(1, n // 200), max(2, n // 50)))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    # log‑normal scaling of the whole block
    cand[start:stop] *= np.exp(rng.normal(0.0, 0.12, size=width))
    # occasional mirroring (helps symmetry breaking)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    cand = np.clip(cand, 0.0, 1000.0)
    # keep the total sum the same (scaling does not change the objective)
    s = cand.sum()
    if s > 0.0:
        cand *= (a.sum() / s)
    return cand


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a step‑function that minimises the AC‑1 upper bound.

    The search starts from the analytically motivated shape a_i = 1/√(i+1),
    then explores nearby shapes via simple stochastic moves while respecting
    the evaluator's constraints.
    """
    rng = np.random.default_rng(seed)
    # total available time – a small safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # length of the problem (height_sequence_1 is supplied by the harness)

    best_score = float(evaluate_sequence(best.tolist()))
