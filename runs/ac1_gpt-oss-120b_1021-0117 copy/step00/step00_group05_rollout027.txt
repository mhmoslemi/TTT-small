    def compute_score(vec):
        # vec sum may not be 1 but we compute using evaluator for accuracy
        return float(evaluate_sequence(vec.tolist()))

    # Candidate generation (scan powers)
    best_vec = None

        score = compute_score(a.tolist())
        if score < best_score:

            best_vec = a.copy()

    # Scan reversing (increasing shape)

    # Add random Dirichlet candidates
    for _ in range(20):

    return [float(x) for x in best_vec]

import time
import numpy as np

def _fft_len(n: int) -> int:
    """next power of two >= 2*n-1, used for linear convolution via FFT."""
    return 1 << (int(np.ceil(np.log2(2 * n - 1))))

def _conv_full(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (zero padded) using real FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]

def _conv_max(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of a*a."""
    return _conv_full(a, fft_len).max()

def _fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score 2*n*max(a*a)/(sum a)^2 – fast version (sum(a) may differ from 1)."""
    n = a.shape[0]
    mx = _conv_max(a, fft_len)
    s = a.sum()
    if s == 0:
        return np.inf
    return 2.0 * n * mx / (s * s)

    """Search for a step‑function minimizing the AC1 upper bound.

    The routine works entirely in O(n) memory, uses only FFTs for convolution,
    and respects the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Determine length n and a fallback uniform vector
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise Exception
        n = base.shape[0]
    except Exception:
        n = 256
        base = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len = _fft_len(n)

    # uniform (already normalised)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Initialise best candidate from a few simple families
    # ------------------------------------------------------------------
    best_vec = uniform.copy()
    best_val = _fast_score(best_vec, fft_len)

    # parent (if present) – normalise to sum 1
    parent = np.clip(base, 0.0, 1000.0)
    if parent.sum() > 0:
        parent = parent / parent.sum()
        val = _fast_score(parent, fft_len)
        if val < best_val:
            best_val = val
            best_vec = parent.copy()

    # power‑law family a_i ∝ (i+1)^p, p in [-2,0]
    idx = np.arange(n, dtype=np.float64) + 1.0
    best_p = None
    for p in np.linspace(-2.0, 0.0, 41):
        a = idx ** p
        a /= a.sum()
        val = _fast_score(a, fft_len)
        if val < best_val:
            best_val = val
            best_vec = a.copy()
            best_p = p

    # refined search around the best exponent (if any)
    if best_p is not None:
        for p in np.linspace(best_p - 0.2, best_p + 0.2, 41):
            a = idx ** p
            a /= a.sum()
            val = _fast_score(a, fft_len)
            if val < best_val:
                best_val = val
                best_vec = a.copy()
                best_p = p

    # a few exponential shapes a_i ∝ exp(p*i), p negative and tiny
    for p in np.linspace(-0.02, -0.001, 21):
        a = np.exp(p * np.arange(n, dtype=np.float64))
        a /= a.sum()
        val = _fast_score(a, fft_len)
        if val < best_val:
            best_val = val
            best_vec = a.copy()

    # ------------------------------------------------------------------
    # Gradient‑descent like refinement using the current worst shift
    # ------------------------------------------------------------------
    a = best_vec.copy()
    cur_val = best_val
    grad = np.empty(n, dtype=np.float64)
    it = 0
    while time.monotonic() < deadline and it < 20000:
        conv_full = _conv_full(a, fft_len)
        M = conv_full.max()
        k = int(conv_full.argmax())
        # gradient of the worst convolution entry
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        grad.fill(0.0)
        i_range = np.arange(start, end + 1, dtype=np.int64)
        grad[i_range] = a[k - i_range]

        avg = M / n
        d = avg - grad                     # sum(d) == 0, preserves total mass
        eta = 0.2 / (1.0 + it * 1e-3)       # slowly decaying step size
        a_new = a + eta * d
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0.0:                       # safety fall‑back (should never happen)
            a_new = uniform.copy()
            s = a_new.sum()
        a_new /= s

        new_val = _fast_score(a_new, fft_len)
        if new_val < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_vec = a_new.copy()
        else:
            # occasional random jitter to escape tiny plateaus
            if it % 500 == 0:
                jitter = rng.normal(0.0, 0.01, size=n) * a
                a = np.clip(a + jitter, 0.0, None)
                a /= a.sum()
                cur_val = _fast_score(a, fft_len)

        it += 1

    # ------------------------------------------------------------------
    # Final validation with the authoritative evaluator
    # ------------------------------------------------------------------
    final_vec = best_vec
    final_score = float(evaluate_sequence(final_vec.tolist()))
    if not np.isfinite(final_score):
        # safest fallback – uniform vector
        final_vec = uniform

    # Return as a plain Python list of floats
    return [float(x) for x in final_vec]
