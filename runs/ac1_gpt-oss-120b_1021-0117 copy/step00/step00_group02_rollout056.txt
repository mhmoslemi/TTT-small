import time
import math
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Construct a sparse non‑negative step function whose autocorrelation
    (full convolution with itself) has a small maximum value.
    The construction uses a random greedy Sidon set: a set of indices
    whose pairwise sums are all distinct.  Assigning equal weight to those
    indices gives max_{k}(a*a)[k] = 1/m^2 where m = |set|, and the
    evaluator score becomes 2·n/m².  By making m as large as possible
    (≈√(2n)) we obtain scores well below 2.0.
    """
    rng = np.random.default_rng(seed)
    # Reserve a safety margin so we return before the sandbox kills us.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Length of the problem instance (may be supplied as height_sequence_1)
    try:
        n = int(height_sequence_1.shape[0])      # pre‑imported if present
    except Exception:
        n = 256                                   # fallback length

    # Uniform vector (score = 2) – kept as a safe fallback.
    best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = 2.0

    # Helper: compute the exact score of a candidate (used only on improvements)
    def exact_score(arr):
        # arr is a NumPy 1‑D array of floats
        return float(evaluate_sequence(arr.tolist()))

    # Main stochastic search – build many random Sidon sets.
    while time.monotonic() < deadline:
        # Random order of indices; bias toward early acceptance.
        perm = rng.permutation(n)
        selected = []                 # chosen indices
        used_sums = set()             # sums i+j that are already taken

        for idx in perm:
            s_self = idx + idx
            if s_self in used_sums:
                continue
            conflict = False
            for j in selected:
                if idx + j in used_sums:
                    conflict = True
                    break
            if conflict:
                continue

            # Accept this index.
            selected.append(idx)
            used_sums.add(s_self)
            for j in selected[:-1]:          # add sums with previously chosen indices
                used_sums.add(idx + j)

            # Early exit if we already cannot beat the current best theoretical score.
            # The theoretical score for a set of size m is 2*n/m^2.
            if len(selected) > 0:
                theor = 2.0 * n / (len(selected) * len(selected))
                if theor >= best_score:
                    # Even a larger set will not improve the current best.
                    # Continue building to maybe enlarge the set later.
                    pass

        m = len(selected)
        if m < 2:                      # need at least two points to improve over uniform
            continue

        # Theoretical score for this construction.
        theor_score = 2.0 * n / (m * m)

        # Accept if the theoretical bound improves.
        if theor_score < best_score - 1e-12:
            # Build the actual sequence (equal weight on the Sidon positions).
            cand = np.zeros(n, dtype=np.float64)
            val = 1.0 / m
            cand[selected] = val

            # Verify with the authoritative evaluator (rarely needed, but safe).
            sc = exact_score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                # Stop early if we already beat the target bound.
                if best_score <= 1.50274349:
                    break

    # Final safety: ensure we have a valid output.
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Return as plain Python list of floats.
    return [float(x) for x in best_seq]
