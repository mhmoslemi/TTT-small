def project_simplex(v):
    """Project onto simplex {x: x >= 0, sum x = 1}."""
    n = v.shape[0]
    # shift to guarantee positivity?
    # Using algorithm by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # Find rho = max { j | u_j - (cssv_j / j) > 0 }
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[rho - 1] / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np, time

def _project_positive_normalize(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform
        return np.ones_like(v) / v.shape[0]
    return v / s

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Determine length n
    try:
        n = int(height_sequence_1.shape[0])
        has_parent = True
    except Exception:
        # No parent; default n = 256
        n = 256
        has_parent = False

    # start from uniform
    a = np.ones(n, dtype=np.float64) / n

    # If a parent exists, try to use its shape (normalized)
    if has_parent:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size == n and np.isfinite(parent).all():
            s = parent.sum()
            if s > 0.0:
                cand = parent / s
                # evaluate both candidates via the official evaluator
                parent_score = float(evaluate_sequence(parent.tolist()))
                uniform_score = float(evaluate_sequence((np.ones(n)/n).tolist()))
                if np.isfinite(parent_score) and parent_score < uniform_score:
                    a = cand

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

def _positive_normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1 (simple simplex projection)."""
