def project_simplex(v):
    """Projects vector v onto the probability simplex (nonnegative sum=1)."""
    # Using algorithm from Duchi et al. (O(n log n))
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def conv_self(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, size)
    C = A * A
    conv = np.fft.irfft(C, size)[:2*n-1]
    # Clip tiny negative values
    conv = np.maximum(conv, 0.0)
    return conv

Now implement cross-correlation gradient:

def grad_weighted(w, a):
    # w length 2n-1, a length n.
    rev_a = a[::-1]
    size = 1 << ((len(w) + len(a) - 1).bit_length())
    W = np.fft.rfft(w, size)
    A_rev = np.fft.rfft(rev_a, size)
    conv = np.fft.irfft(W * A_rev, size)
    # Slice
    offset = len(a) - 1
    g = conv[offset: offset + len(a)]
    g = np.maximum(g, 0.0)  # gradient components should be nonnegative but may have small negative due to numeric noise
    return 2 * g

import time
import numpy as np

# ---------- helper functions (top level) ----------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Fast convolution a*a using real FFT. Returns length 2*len(a)-1 array."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, size)
    C = A * A
    conv = np.fft.irfft(C, size)[: 2 * n - 1]
    # eliminate tiny negatives from numerical error
    conv[conv < 0] = 0.0
    return conv


def _grad_max_diag(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of the max‑diagonal objective for index k_max."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    i = np.arange(i_start, i_end + 1, dtype=int)
    # contribution 2 * a_{k_max-i} for each i that participates
    g[i] = 2.0 * a[k_max - i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    if v.ndim != 1:
        raise ValueError("vector required")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _score_from_vec(v: np.ndarray) -> float:
    """Exact score for a (non‑negative) vector using the evaluator's formula."""
    n = v.shape[0]
    conv = _conv_self(v)
    max_b = conv.max()
    sum_a = v.sum()
    # sum_a is guaranteed >0 in our usage
    return float(2.0 * n * max_b / (sum_a * sum_a))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of heights that (hopefully) improves on the
    baseline score 2.0.  The algorithm performs a fast sub‑gradient descent
    on the max‑diagonal objective, starting from random Dirichlet vectors
    and keeping the best point seen before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Determine the length n (use height_sequence_1 if present, else 256)
    # ------------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        if base_seq.ndim != 1 or base_seq.size == 0:
            raise ValueError
        n = base_seq.size
    except Exception:
        n = 256
        base_seq = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Fallback: uniform distribution (known score = 2.0)
    # ------------------------------------------------------------------
    best = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = 2.0  # uniform yields exactly 2.0

    # ------------------------------------------------------------------
    # Optionally use the provided parent (height_sequence_1)
    # ------------------------------------------------------------------
    parent = np.clip(base_seq, 0.0, 1000.0)
    if parent.size == n:
        parent_norm = parent / parent.sum()
        parent_score = _score_from_vec(parent_norm)
        if parent_score < best_score:
            best = parent_norm.copy()
            best_score = parent_score

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    # Hyper‑parameters (tuned heuristically)
    base_eta = 0.5          # initial step size
    eta_min = 1e-4
    decay = 1e-4            # linear decay of step size
    restart_interval = 2500  # re‑initialise from scratch periodically

    # current iterate
    a = rng.dirichlet(np.full(n, 0.5)).astype(np.float64)   # random start, sum = 1

    it = 0
    while time.monotonic() < deadline:
        # occasional restart to avoid getting stuck
        if it % restart_interval == 0 and it > 0:
            a = rng.dirichlet(np.full(n, 0.5)).astype(np.float64)

        # convolution and index of the current maximal diagonal
        conv = _conv_self(a)
        k_max = int(conv.argmax())

        # sub‑gradient for that diagonal
        g = _grad_max_diag(a, k_max)

        # step size that slowly decays
        eta = max(base_eta / (1.0 + decay * it), eta_min)

        # gradient step, projection onto the simplex, and a tiny random jitter
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        if rng.random() < 0.02:                      # occasional exploration
            noise = rng.normal(scale=0.02, size=n)
            a_new = np.maximum(a_new + noise, 0.0)
            a_new = _project_simplex(a_new)

        # evaluate the candidate (fast, no call to evaluate_sequence)
        cur_score = _score_from_vec(a_new)

        if cur_score < best_score:
            best = a_new.copy()
            best_score = cur_score
            # if we made progress, allow a slightly larger step next time
            base_eta = min(base_eta * 1.1, 5.0)

        a = a_new
        it += 1

    # ------------------------------------------------------------------
    # Return the best vector found, scaled to stay comfortably inside
    # the evaluator's clipping range (0 … 1000).  Scaling does not affect
    # the score because the objective is homogeneous of degree zero.
    # ------------------------------------------------------------------
    out = np.clip(best * 1000.0, 0.0, 1000.0)
    return [float(x) for x in out.tolist()]
