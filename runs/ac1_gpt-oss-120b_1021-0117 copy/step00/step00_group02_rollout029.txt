import numpy as np
import time

# ----------------------------------------------------------------------
# fast upper‑bound computation using FFT (identical to the exact formula
# except that it avoids the O(n²) direct convolution).
# ----------------------------------------------------------------------
def fast_ac1_score(arr: np.ndarray) -> float:
    n = arr.shape[0]
    # convolution via FFT on a zero‑padded length 2*n array
    fa = np.fft.rfft(arr, 2 * n)
    conv = np.fft.irfft(fa * fa, 2 * n)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = arr.sum()
    if sum_a == 0.0:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)

    if r == 1.0:
        exp_part = 1.0
    else:
        exp_part = np.power(r, idx, dtype=np.float64)
    return (i1 ** (-p)) * exp_part
