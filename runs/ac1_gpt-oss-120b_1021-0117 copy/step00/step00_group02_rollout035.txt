    best_score = float(evaluate_sequence(best.tolist()))

    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)
    rng = np.random.default_rng(seed)

    while time.monotonic() < deadline:
        # Randomly pick a contiguous block and perturb its amplitude.
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        cand = best.copy()
        factor = float(np.exp(rng.normal(0.0, 0.06)))
        cand[start:start + width] *= factor

        # Occasionally symmetrize to keep the shape balanced.
        if rng.random() < 0.25:
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    return [float(v) for v in best]
