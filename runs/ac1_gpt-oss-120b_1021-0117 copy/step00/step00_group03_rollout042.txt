import time
import numpy as np

def _gradient_descent(a, n, deadline):
    """Attempt to flatten the convolution of a using simple gradient steps."""
    best = a.copy()
    best_score = float('inf')
    it = 0
    while time.monotonic() < deadline:
        # convolution and objective (sum(a) is kept = 1)
        b = np.convolve(a, a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b          # because sum(a) == 1
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # stop if already very flat
        if cur_score <= 1.6:
            break
        # target constant value of the convolution
        mu = 1.0 / (2 * n - 1)               # average of b (since sum(b)=1)
        diff = b - mu
        # gradient of sum_k (b_k-mu)^2 w.r.t a (valid correlation)
        grad = 4.0 * np.correlate(diff, a, mode='valid')
        max_abs = np.max(np.abs(grad))
        if max_abs == 0:
            break
        # step size, scaled down over time
        eta = 0.1 / (max_abs + 1e-12)
        eta *= 0.99 ** it
        a_new = a - eta * grad
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s <= 0.0:
            # re‑initialise if we collapsed to zero
            a_new = rng.dirichlet(np.full(n, 0.5))
            s = 1.0
        a = a_new / s
        it += 1
    return best

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Generate a non‑negative step‑function minimizing the AC1 bound.

    The routine builds a random Dirichlet start vector (sum = 1) and then
    performs a simple gradient‑descent that tries to make the convolution
    as flat as possible.  The best vector seen before the time budget
    expires is returned.
    """
    global rng
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Determine length from the supplied parent if possible
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = int(parent.size)
    except Exception:
        n = 256                                   # fallback length

    # Start from a Dirichlet distribution (α=0.5 gives a sparseish seed)
    a = rng.dirichlet(np.full(n, 0.5)).astype(np.float64)

    # Normalise to unit sum – the evaluator is scale invariant
    a /= a.sum()

    # Run the optimisation loop within the time budget
    best = _gradient_descent(a, n, deadline)

    # Return as plain Python list of floats
    return [float(v) for v in best]
