def _optimal_epsilon(n):
    """Solve e^x = 1 + 2x for x, then return eps = x / n."""
    x = 1.5  # start near the root
    for _ in range(20):
        fx = np.exp(x) - 1.0 - 2.0 * x
        dfx = np.exp(x) - 2.0
        x -= fx / dfx
    return x / float(n)


def _conv_max(arr, size):
    """Return (max value of a*a, index where it occurs) using an FFT of length size."""
    fft = np.fft.rfft(arr, size)
    conv = np.fft.irfft(fft * fft, size)[: 2 * len(arr) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx
