Pseudo:

```
import cvxpy as cp

def compute_g(f, M, active_ks):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_ks:
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # indices i = i_start..i_end inclusive
        # corresponding g indices j = k - i
        # we can use cp.sum(cp.multiply(f[i_start:i_end+1], g[k - np.arange(i_start, i_end+1)]))
        i_range = np.arange(i_start, i_end+1)
        g_indices = k - i_range
        expr = cp.sum(cp.multiply(f[i_range], g[g_indices]))
        constraints.append(expr <= M)
    # also sum(g) maximize; objective:
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, eps_abs=1e-8, eps_rel=1e-8, warm_start=True)
    if prob.status in ['optimal', 'optimal_inaccurate']:
        g_val = g.value
        if g_val is None:
            return None
        else:
            return g_val
    else:
        return None
```

def build_sparse_set(n, target_L, max_g, rng):
    A = []
    sum_counts = {}
    attempts = 0
    max_attempts = 1000000
    while len(A) < target_L and attempts < max_attempts:
        i = int(rng.integers(0, n))
        if i in A:
            attempts += 1
            continue
        ok = True
        # check sums with existing elements
        for j in A:
            s = i + j
            cnt = sum_counts.get(s, 0)
            if cnt >= max_g:
                ok = False
                break
        # check self sum
        if ok:
            s = i + i
            cnt = sum_counts.get(s, 0)
            if cnt >= max_g:
                ok = False
        if ok:
            # add element
            for j in A:
                s = i + j
                sum_counts[s] = sum_counts.get(s, 0) + 1
            # self
            s = i + i
            sum_counts[s] = sum_counts.get(s, 0) + 1
            A.append(i)
        attempts += 1
    return A

import time
import numpy as np


def _construct_set(n, max_g, rng):
    """Greedy construction of a set of indices with at most max_g unordered
    representations for every possible sum i+j. Returns the list of chosen
    indices and the actual maximal count that occurred."""
    order = rng.permutation(n)
    chosen = []
    counts = {}               # sum -> unordered pair count
    for i in order:
        # test whether adding i would violate the bound max_g
        ok = True
        for j in chosen:
            s = i + j
            if counts.get(s, 0) >= max_g:
                ok = False
                break
        if not ok:
            continue
        s_self = i + i
        if counts.get(s_self, 0) >= max_g:
            continue
        # commit i
        for j in chosen:
            s = i + j
            counts[s] = counts.get(s, 0) + 1
        counts[s_self] = counts.get(s_self, 0) + 1
        chosen.append(i)
    maxcnt = max(counts.values()) if counts else 0
    return chosen, maxcnt


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a sparse 0‑1 pattern that lowers the autocorrelation bound.
    The algorithm builds a set of indices where each sum i+j appears at most
    a few times, which keeps the peak of the convolution small while the
    ℓ¹‑norm stays large."""
    rng = np.random.default_rng(seed)
    # keep a small safety margin so we can return before the sandbox kills us
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # length of the sequence – use the provided parent if it exists
    try:
        n = len(height_sequence_1)          # provided by the harness
    except Exception:                     # fall back to a modest size
        n = 256

    best_score = float('inf')
    best_seq = None

    # start with a fairly strict bound on repetitions per sum
    max_g = 6

    while time.monotonic() < deadline:
        # build a candidate set
        indices, observed_max = _construct_set(n, max_g, rng)

        # if the set is too tiny it cannot beat the baseline
        if len(indices) < 80:
            continue

        # create the 0‑1 sequence (height 1 on selected positions)
        seq = np.zeros(n, dtype=np.float64)
        seq[indices] = 1.0

        # evaluate the objective (the evaluator enforces clipping and sum>0.01)
        score = float(evaluate_sequence(seq.tolist()))

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
            # stop early if we are already well below the target bound
            if best_score < 1.6:
                break

        # occasional chance to relax the repetition bound – this may allow a
        # larger set at the cost of a slightly larger peak.
        if rng.random() < 0.15:
            max_g = int(rng.integers(6, 10))

    # fallback: uniform sequence (guaranteed to be valid)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # final safety clipping (the evaluator also does this)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return [float(x) for x in best_seq]
