def compute_score(a):
    n = a.shape[0]
    # compute convolution via FFT
    N = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, N)
    b = np.fft.irfft(a_fft * a_fft, N)[:2*n-1]
    S = a.sum()
    if S < 0.01:
        return np.inf, b, S
    M = b.max()
    score = 2.0 * n * M / (S * S)

import numpy as np
import time

def _score_and_conv(a):
    """Return (score, convolution array) for a non‑negative vector a."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    af = np.fft.rfft(a, N)
    conv = np.fft.irfft(af * af, N)[: 2 * n - 1]
    s = a.sum()
    if s < 0.01:
        return np.inf, conv
    return float(2.0 * n * conv.max() / (s * s)), conv


def _flatten_one_step(a, step, eps):
    """One Newton‑like step trying to make the convolution flat."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, N)
    conv = np.fft.irfft(af * af, N)[: 2 * n - 1]

    s = a.sum()
    avg = (s * s) / (2 * n - 1)
    err = conv - avg

    ef = np.fft.rfft(err, N)
    den = 2.0 * af
    den = np.where(np.abs(den) < eps, eps, den)          # avoid divide‑by‑zero
    delta_f = -step * ef / den
    delta = np.fft.irfft(delta_f, N)[:n]

    # limit how far we move in one iteration
    limit = np.maximum(0.5, 0.2 * a)
    delta = np.clip(delta, -limit, limit)

    a_new = a + delta
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a list of non‑negative coefficients with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ----- initialise -------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.clip(parent, 0.0, 1000.0)
        if a0.ndim != 1 or a0.size == 0 or not np.isfinite(a0).all():
            raise ValueError
    except Exception:
        a0 = np.ones(256, dtype=np.float64)

    n = a0.size
    if a0.sum() < 0.01:                      # never start with an empty mass
        a0 = np.abs(rng.standard_normal(n)) + 0.01

    # a small random perturbation of the parent gives a non‑trivial start
    a = a0 * (1.0 + 0.05 * rng.standard_normal(n))
    a = np.clip(a, 0.0, 1000.0)

    best = a.copy()
    best_score, _ = _score_and_conv(best)

    # ----- optimisation parameters -----------------------------------------
    eps = 1e-12
    step = 1.0

    # ----- main loop --------------------------------------------------------
    while time.monotonic() < deadline:
        # inner Newton‑like iterations; break early if time runs out
        for _ in range(200):
            a_new = _flatten_one_step(a, step, eps)
            cur_score, cur_conv = _score_and_conv(a_new)

            # keep the globally best vector
            if np.isfinite(cur_score) and cur_score < best_score:
                best = a_new.copy()
                best_score = cur_score
                # stop early if we already beat the known target
                if best_score < 1.50274349:
                    return [float(x) for x in best]

            # accept improvement, otherwise shrink step size
            if cur_score < best_score:
                a = a_new
                step = min(step * 1.05, 2.0)
            else:
                step *= 0.7
                if step < 1e-4:                           # occasional restart
                    a = best * (1.0 + 0.01 * rng.standard_normal(n))
                    a = np.clip(a, 0.0, 1000.0)
                    step = 1.0

            if time.monotonic() >= deadline:
                break

        # occasional random re‑initialisation to escape possible stagnation
        if rng.random() < 0.03 and time.monotonic() < deadline:
            a = np.abs(rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)

    # ----- return -----------------------------------------------------------
    return [float(x) for x in best]
