```python
import numpy as np, time, math

def compute_ac1(seq):
    n = len(seq)
    a = np.asarray(seq, dtype=np.float64)
    M = 1 << ((2*n-2).bit_length()) # next power of two ≥ 2n-1
    A = np.fft.rfft(a, n=M)
    b = np.fft.irfft(A * A, n=M)[:2*n-1]
    max_b = np.max(b)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (sum_a**2)
```

```python
def greedy_sidon(n, max_size=80):
    sums = set()
    S = []
    for i in range(n):
        ok = True
        for s in S:
            if i + s in sums:
                ok = False
                break
        if not ok:
            continue
        if i + i in sums:
            continue
        # add
        for s in S:
            sums.add(i + s)
        sums.add(i + i)
        S.append(i)
        if len(S) >= max_size:
            break
    return S
```

def _conv_max(seq):
    """Return max of linear convolution of seq with itself (full length)."""
    n = seq.shape[0]
    # length for FFT (next power of two >= 2n-1)
    m = 1 << ((2 * n - 2).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    return float(conv.max())


def _ac1_score(seq):
    """Exact AC1 value (fast via FFT)."""
    n = seq.shape[0]
    max_b = _conv_max(seq)
    s = float(seq.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)
