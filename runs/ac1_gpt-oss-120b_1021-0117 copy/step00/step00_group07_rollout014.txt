def ac1_score(a):
    # a: np.array floats nonnegative.
    n = len(a)
    # compute convolution using FFT.
    N = 1 << ((2*n-1).bit_length()) # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A # note: a*a (convolution) corresponds to 'linear' conv via zero padding.
    conv = np.fft.irfft(B, n=N)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2 * n * max_b / (sum_a**2)

        A = np.fft.rfft(a, n=L)

        start = n - 1

        a_new = a - step * grad

        a = a_new

        score = float(evaluate_sequence(a.tolist()))

        # ensure step not too small or too large

    # Return the best candidate found, re‑scaled to the original L1 norm
    # ------------------------------------------------------------------
    final_seq = (best_a * sum_parent).tolist()
    return [float(x) for x in final_seq]
