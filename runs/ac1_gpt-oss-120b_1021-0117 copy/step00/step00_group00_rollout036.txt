    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256

        i = np.arange(n, dtype=np.float64)
        seq = ( (i + offset) ** (-p) + ( (n - 1 - i) + offset) ** (-p) )
        # clip to limits (0,1000)
        np.clip(seq, 0.0, 1000.0, out=seq)
        return seq

    # Initialize best using height_sequence_1 if present
    base = np.asarray(height_sequence_1, dtype=np.float64) if 'height_sequence_1' in globals() else np.ones(n, dtype=np.float64)
    base = np.clip(base, 0.0, 1000.0)
    best_seq = base.copy()

    # Evaluate initial param candidates
    for _ in range(20):
        p = rng.uniform(0.1, 0.9)
        offset = rng.uniform(0.1, 5.0)
        cand = make_seq(p, offset)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score

    # Main loop: random search around best p/offset
    # Keep track of current best params
    best_p = None
    best_offset = None
    if best_score < float(evaluate_sequence(base.tolist())):
        # we have a better seq; we can approximate its parameters using fitting? not needed
        pass
    # Instead continue random sampling
    while time.monotonic() < deadline:
        p = rng.uniform(0.1, 0.9)
        offset = rng.uniform(0.1, 5.0)
        cand = make_seq(p, offset)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
            # Optionally adjust search ranges based on best values
        # occasional local perturbation of best sequence
        if rng.random() < 0.1:
            # small multiplicative perturbation on a random block
            cand2 = best_seq.copy()
            width = rng.integers(1, max(2, n//50))

            cand2[start:start+width] *= np.exp(rng.normal(0, 0.05))

            score2 = float(evaluate_sequence(cand2.tolist()))

import time
import numpy as np

def _make_power_sequence(n, p, delta):
    """Generate a symmetric power-law sequence:
       a_i = (i+delta)^(-p) + (n-1-i+delta)^(-p)
    """
    idx = np.arange(n, dtype=np.float64)
    # avoid divide‑by‑zero – delta is always >0
    seq = np.power(idx + delta, -p) + np.power(n - 1 - idx + delta, -p)
    # keep within the evaluator limits
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step function that minimises the AC1 bound.

    The algorithm starts from a symmetric power‑law shape (the continuous
    analogue of the arcsine distribution) and then performs many cheap
    perturbations, keeping the best candidate seen before the deadline.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Length of the sequence (use the provided parent if present)
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = parent.size
    else:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Initial candidates
    # ------------------------------------------------------------------
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # a simple symmetric power‑law (p=0.5, delta=0.5) is a strong start
    seq0 = _make_power_sequence(n, p=0.5, delta=0.5)
    score0 = float(evaluate_sequence(seq0.tolist()))
    if score0 < best_score:
        best_seq, best_score = seq0, score0

    # ------------------------------------------------------------------
    # Random exploration in the (p, delta) space
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # draw random parameters
        p = rng.uniform(0.05, 0.49)          # p must stay < 0.5 for the theory
        delta = rng.uniform(0.05, 10.0)      # positive offset
        cand = _make_power_sequence(n, p, delta)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_seq, best_score = cand, sc

        # Occasionally perform a cheap local tweak on the current best
        if rng.random() < 0.12:
            # block perturbation (multiplicative Gaussian noise)
            start = rng.integers(0, n)
            width = rng.integers(1, max(2, n // 40))
            stop = min(n, start + width)
            pert = best_seq.copy()
            factor = np.exp(rng.normal(0.0, 0.06, width))
            pert[start:stop] *= factor
            np.clip(pert, 0.0, 1000.0, out=pert)
            scp = float(evaluate_sequence(pert.tolist()))
            if np.isfinite(scp) and scp < best_score:
                best_seq, best_score = pert, scp

        # keep the loop cheap – the heavy work is inside evaluate_sequence
        # which uses an efficient FFT based convolution.

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
