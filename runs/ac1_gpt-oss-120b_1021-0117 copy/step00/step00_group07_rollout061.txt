import numpy as np
import time

# ---------- helper functions -------------------------------------------------
def fast_conv(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑2 FFT (real input)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    # numerical noise may give tiny negatives – clamp to zero
    return np.maximum(conv, 0.0)


def score_from_array(a: np.ndarray) -> float:
    """Upper bound 2 * n * max(conv) for a already normalized (sum = 1)."""
    n = a.shape[0]
    conv = fast_conv(a)
    return float(2.0 * n * conv.max())


def init_geometric(n: int) -> np.ndarray:
    """
    Geometric start with the analytically optimal decay
    r = 1 - c_opt / n, c_opt ≈ 1.256 (solution of e^c = 1 + 2c).
    The vector is normalised to sum 1.
    """
    c_opt = 1.256
    alpha = c_opt / n
    r = 1.0 - alpha
    # raw geometric weights (may be very small for large indices)
    raw = alpha * (r ** np.arange(n, dtype=np.float64))
    s = raw.sum()
    if s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return raw / s


def random_perturb(a: np.ndarray, rng: np.random.Generator) -> None:
    """Apply a tiny random scaling to a random block (keeps sum unchanged)."""
    n = a.size
    width = int(rng.integers(1, max(2, n // 200)))
    start = int(rng.integers(0, max(1, n - width)))
    factor = 1.0 + rng.normal(0.0, 0.02)          # small gaussian factor
    a[start:start + width] *= factor
    a.clip(min=0.0, out=a)                       # enforce non‑negativity
    s = a.sum()
    if s > 0.0:
        a /= s


# ---------- main search routine ---------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the AC1 bound.
    Returns a list of floats; the sum of the list is normalised to 1
    (any positive scaling would give the same bound, but the evaluator
    enforces non‑negativity and clipping at 1000).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ----- initialise -------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = parent.size
    else:
        n = 256                     # fallback length
    # start from the analytic geometric shape
    a = init_geometric(n)

    best_a = a.copy()
    best_score = score_from_array(best_a)

    itr = 0
    while time.monotonic() < deadline:
        itr += 1
        # occasional random shake‑up to avoid stagnation
        if itr % 500 == 0:
            random_perturb(a, rng)
            cur_score = score_from_array(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
            continue

        conv = fast_conv(a)
        k_max = int(np.argmax(conv))

        # contributions a_i * a_{k_max-i} (zero where index out of range)
        idx = np.arange(n, dtype=np.int64)
        j = k_max - idx
        valid = (j >= 0) & (j < n)
        prod = np.zeros_like(a)
        prod[valid] = a[valid] * a[j[valid]]

        # choose a high‑contribution index (i_max) and a low‑contribution index (i_min)
        if prod.max() - prod.min() < 1e-14:                # all contributions equal (e.g. uniform)
            i_max = int(rng.integers(0, n))
            i_min = int(rng.integers(0, n - 1))
            if i_min >= i_max:
                i_min += 1
        else:

            top_k = max(1, int(0.02 * n))
            bot_k = max(1, int(0.02 * n))
            top_idxs = np.argpartition(-prod, top_k - 1)[:top_k]
            bot_idxs = np.argpartition(prod, bot_k - 1)[:bot_k]
