def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
    # ensure we have correct type:
    n = int(n)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    best_seq = None
    best_score = float('inf')
    # first fallback: uniform sequence (all ones) maybe
    uniform = np.ones(n, dtype=np.float64)
    score = float(evaluate_sequence(uniform.tolist()))
    best_seq = uniform
    best_score = score

    # parameter sweep for v-shape
    for alpha in np.linspace(0.5, 5.0, 46):  # 46 steps
        if time.monotonic() > deadline:
            break
        r = math.exp(-alpha / n)
        idx = np.arange(n)
        m = np.minimum(idx, n-1-idx)
        a = r ** m
        # convert to list
        seq = a.tolist()
        # evaluate
        sc = float(evaluate_sequence(seq))
        if sc < best_score:
            best_seq = a
            best_score = sc

    # also try exponential left
    for alpha in np.linspace(0.5,5.0,46):
        if time.monotonic()>deadline: break
        r = math.exp(-alpha / n)
        a = r ** np.arange(n)
        sc = float(evaluate_sequence(a.tolist()))
        if sc < best_score:
            best_seq = a
            best_score = sc

    # also try linear decreasing normalized to max 1
    linear = (np.arange(n)[::-1] + 1) / n  # decreasing from 1 to 1/n
    sc = float(evaluate_sequence(linear.tolist()))
    if sc < best_score:
        best_seq = linear
        best_score = sc

    # Also try simple random local search starting from best_seq
    rng = np.random.default_rng(seed)
    current = best_seq.copy()
    current_score = best_score
    while time.monotonic() < deadline:
        # perturb few entries
        cand = current.copy()
        # choose random block width
        width = int(rng.integers(1, max(2, n//100)))
        start = int(rng.integers(0, n - width + 1))
        # apply multiplicative noise
        cand[start:start+width] *= np.exp(rng.normal(0.0, 0.05))
        # ensure nonneg and capped
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = float(evaluate_sequence(cand.tolist()))
        if sc < current_score:
            current = cand
            current_score = sc
            if sc < best_score:
                best_seq = cand
                best_score = sc

    # final answer (ensure list, float)
    return [float(x) for x in best_seq]

import numpy as np
import time
import math

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step sequence that minimizes the AC1 bound.
    Starts from a simple exponential shape (which already beats the uniform
    2.0 bound) and performs a mixture of random block perturbations and
    targeted reductions of the dominant convolution term.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Determine problem size and base data (parent may be provided)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = int(len(height_sequence_1))
    else:
        # fallback length – the exact value does not matter much
        n = 256

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 2.  Helper to evaluate a candidate (list → float)
    # ------------------------------------------------------------------
    def score_of(arr):
        """Return the authoritative score for a numpy 1‑D array."""
        # evaluator works on Python list – conversion is cheap
        return float(evaluate_sequence(arr.tolist()))

    # ------------------------------------------------------------------
    # 3.  Simple exponential families (a_i = r^i) – cheap grid search
    # ------------------------------------------------------------------
    best_seq = np.ones(n, dtype=np.float64)      # uniform fallback
    best_score = score_of(best_seq)

    def exponential(alpha):
        """Exponential decay a_i = r**i with r = exp(-alpha/n)."""
        r = math.exp(-alpha / n)
        return r ** np.arange(n, dtype=np.float64)

    # try a handful of alphas; the sweet spot is around 1.0–2.0
    for alpha in np.linspace(0.5, 5.0, 46):
        if time.monotonic() > deadline:
            break
        cand = exponential(alpha)
        s = score_of(cand)
        if s < best_score:
            best_score = s
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # 4.  Additional parametric families (smooth power‑exponential)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64) / (n - 1.0)
    for power in (0.5, 1.0, 1.5, 2.0, 3.0):
        if time.monotonic() > deadline:
            break
        # a_i = exp(-alpha * (i/(n-1))**power)  with a modest alpha
        for alpha in np.linspace(0.5, 5.0, 13):
            cand = np.exp(-alpha * (idx ** power))
            s = score_of(cand)
            if s < best_score:
                best_score = s
                best_seq = cand.copy()

    # ------------------------------------------------------------------
    # 5.  Local‑search loop (mix of random blocks and targeted updates)
    # ------------------------------------------------------------------
    current = best_seq.copy()
    # keep a small epsilon to avoid exact zeros (evaluate_sequence would
    # reject sum < 0.01, but scaling invariance means we can stay safe)
    eps = 1e-12
    current = np.maximum(current, eps)

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 5a.  Random block perturbation (similar to the original baseline)
        # --------------------------------------------------------------
        if rng.random() < 0.65:
            width = int(rng.integers(1, max(2, n // 30)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.07)))  # modest log‑normal
            cand = current.copy()
            cand[start:start + width] *= factor
        else:
            # ----------------------------------------------------------
            # 5b.  Targeted reduction of the dominant convolution pair
            # ----------------------------------------------------------
            b = np.convolve(current, current)
            k = int(np.argmax(b))                     # index of current max
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_vals = np.arange(i_start, i_end + 1)
            j_vals = k - i_vals
            contrib = current[i_vals] * current[j_vals]
            idx_max = int(np.argmax(contrib))
            i_max = i_vals[idx_max]
            j_max = j_vals[idx_max]

            # shrink both members of the maximal product a little
            shrink = 0.85 + 0.15 * rng.random()       # between 0.85 and 1.0
            cand = current.copy()
            cand[i_max] *= shrink
            cand[j_max] *= shrink

        # keep values in the allowed interval and avoid exact zeros
        np.clip(cand, eps, 1000.0, out=cand)

        # evaluate the proposal
        cand_score = score_of(cand)

        # accept if it improves the best-so‑far (always keep the best)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            current = cand.copy()
            continue

        # otherwise occasionally accept a worsening move to escape plateaus
        # (simulated‑annealing‑style acceptance)
        if rng.random() < 0.02:
            current = cand.copy()

    # ------------------------------------------------------------------
    # 6.  Return the best sequence discovered (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
