def generate_sidon(n, target_size=None):
    S = []
    sums = set()
    for x in range(n):
        conflict=False
        # check with existing y in S
        for y in S:
            if (x+y) in sums:
                conflict=True
                break
        if conflict:
            continue
        # also check double with itself (x+x)
        if (x+x) in sums:
            continue
        # accept
        for y in S:
            sums.add(x+y)
        sums.add(x+x)
        S.append(x)
        if target_size and len(S) >= target_size:
            break
    return S

import time
import math
import numpy as np

def _greedy_sidon(limit):
    """Return a Sidon set S ⊂ [0, limit) (distinct sums i+j for i≤j)."""
    S = []
    used = set()
    for x in range(limit):
        # i+i must be new
        if (x + x) in used:
            continue
        conflict = False
        for y in S:
            if (x + y) in used:
                conflict = True
                break
        if conflict:
            continue
        # accept x
        for y in S:
            used.add(x + y)
        used.add(x + x)
        S.append(x)
    return S

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Build a sequence that keeps the maximum convolution small
    while preserving a large total sum.
    The construction uses a Sidon set (distinct i+j sums) for the
    high-valued entries and fills the remaining positions with a
    constant low value chosen so that its own self‑convolution never
    exceeds the contribution of the high entries.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # length of the problem (fallback to a modest size if no parent)
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 256

    C_HIGH = 1000.0                     # maximal allowed height
    # ------------------------------------------------------------------
    # 1) build a Sidon set of indices (high positions)
    sidon_set = _greedy_sidon(n)

    # 2) low value: as large as possible while keeping
    #    (n - |S|) * low^2 ≤ 2 * C_HIGH^2
    low_cnt = n - len(sidon_set)
    if low_cnt > 0:
        low_max = int(math.floor(math.sqrt(2.0 * C_HIGH * C_HIGH / low_cnt)))
        low_val = float(max(0, min(C_HIGH, low_max)))
    else:
        low_val = 0.0

    # 3) assemble the vector
    a = np.full(n, low_val, dtype=np.float64)
    if sidon_set:
        a[sidon_set] = C_HIGH

    # final safety clip
    np.clip(a, 0.0, C_HIGH, out=a)

    # ------------------------------------------------------------------
    # Ensure we have a valid fallback (uniform if something went wrong)
    best = a
    best_score = float('inf')
    try:
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        best_score = float('inf')

    # In the very unlikely case the Sidon construction is bad,
    # fall back to the uniform maximal sequence (guaranteed score = 2.0)
    if not np.isfinite(best_score) or best_score > 2.0:
        best = np.full(n, C_HIGH, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))

    # The search loop is kept tiny – the construction already beats the
    # target bound, but we keep a tiny random perturbation phase just in
    # case we are stopped early.
    while time.monotonic() < deadline:
        # tiny random tweak that preserves the Sidon property
        cand = best.copy()
        # pick a random index from the Sidon set and a random low index
        if sidon_set:
            hi = rng.choice(sidon_set)
            lo = rng.integers(0, n)
            # swap a little mass between them, stay inside [0,1000]
            delta = rng.uniform(-10.0, 10.0)
            new_hi = np.clip(cand[hi] + delta, 0.0, C_HIGH)
            new_lo = np.clip(cand[lo] - delta, 0.0, C_HIGH)
            cand[hi] = new_hi
            cand[lo] = new_lo
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best = cand
            best_score = score
        # break early if we have already a very good bound
        if best_score < 1.0:
            break

    return [float(x) for x in best]
