import numpy as np
import time

# ------------------------------------------------------------
# Greedy construction of a sparse index set with a bounded
# number of ordered representations of each sum (i+j).
# ------------------------------------------------------------
def greedy_index_set(length: int, max_rep: int, rng) -> list:
    """
    Return a list of indices S ⊂ {0,…,length-1} such that for every
    possible sum s=i+j the number of ordered pairs (i,j) with i,j∈S
    and i+j=s does not exceed max_rep.
    The construction is randomized; repeat it many times and keep the
    largest set found.
    """
    # counts[s] = current ordered representation count for sum s
    counts = np.zeros(2 * length - 1, dtype=np.int16)
    chosen = []                     # the index set being built
    order = np.arange(length, dtype=np.int32)
    rng.shuffle(order)              # random order for a greedy trial

    for i in order:
        # check the self‑pair (i,i)
        self_sum = 2 * i
        if counts[self_sum] + 1 > max_rep:
            continue

        if chosen:
            # vectorised check for all existing indices
            s = i + np.array(chosen, dtype=np.int32)
            if np.any(counts[s] + 2 > max_rep):
                continue

        # accept i
        chosen.append(int(i))
        counts[self_sum] += 1

        if chosen[:-1]:   # update the pair‑sums with the older elements
            old = np.array(chosen[:-1], dtype=np.int32)
            s = i + old
            counts[s] += 2

    return chosen


# ------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Build a binary (0/1) sequence whose autocorrelation infinity norm
    is kept low by controlling the number of ordered pair collisions.
    The construction runs many greedy attempts within the time budget
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # -----------------------------------------------------------------
    # Base fallback: use the provided parent (if any) or a uniform vector.
    # -----------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
        )
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            base = np.ones(256, dtype=np.float64)
    else:
        base = np.ones(256, dtype=np.float64)

    best_seq = base.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    n = best_seq.size

    # -----------------------------------------------------------------
    # Greedy search for a low‑collision index set.
    # -----------------------------------------------------------------
    TARGET = 1.50274349  # the target bound (lower is better)

    def evaluate_set(indices):
        """Create a 0/1 vector from indices and compute its score."""
        cand = np.zeros(n, dtype=np.float64)
        if indices:
            cand[indices] = 1.0
        return float(evaluate_sequence(cand.tolist())), cand

    # try several repetitions with different max_rep values
    while time.monotonic() < deadline:
        # choose a max representation budget (3, 4 or 5) – lower is stricter
        max_rep = int(rng.integers(3, 6))
        idx_set = greedy_index_set(n, max_rep, rng)

        # keep only large sets; tiny sets cannot beat the target
        if len(idx_set) < 10:
            continue

        score, cand = evaluate_set(idx_set)
        if score < best_score:
            best_seq = cand
            best_score = score
        if best_score <= TARGET:
            break   # good enough, stop early

    # -----------------------------------------------------------------
    # Simple local refinement: occasional block perturbations of the best.
    # -----------------------------------------------------------------
    def block_perturbation(arr, rng):
        """Apply a cheap O(n) perturbation to keep exploring."""
        a = arr.copy()
        n = a.size
        w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - w + 1)))
        stop = min(n, start + w)
        a[start:stop] *= np.exp(rng.normal(0.0, 0.12))
        if rng.random() < 0.35:
            a = 0.5 * (a + a[::-1])
        return np.clip(a, 0.0, 1000.0)

    while time.monotonic() < deadline:
        cand = block_perturbation(best_seq, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
            if best_score <= TARGET:
                break

    return [float(v) for v in best_seq]
