def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0) # leave 1 sec for return

    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
    if 'height_sequence_1' in globals():
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    else:
        base = np.full(n, 1.0, dtype=np.float64)
    # ensure sum >0
    if np.sum(base) < 0.01:
        base = np.ones(n, dtype=np.float64)

    # normalize to sum=1
    a = base.copy()
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.ones_like(a) / n
    else:
        a = a / a_sum

    best = a.copy()
    # compute best score
    conv = np.convolve(best, best)
    best_score = float(2 * n * conv.max() / (best.sum()**2))

    # parameters
    eta = 0.1
    iter_no_improve = 0
    max_no_improve = 200
    # loop
    while time.monotonic() < deadline:
        # compute convolution
        conv = np.convolve(a, a)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]
        # compute gradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n, max_idx + 1)  # exclusive
        for i in range(i_start, i_end):
            j = max_idx - i
            g[i] = 2.0 * a[j]
        # gradient step
        step = eta * g
        a_new = a - step
        # projection: nonnegative
        a_new = np.clip(a_new, 0.0, None)
        # renormalize
        s = a_new.sum()
        if s == 0.0:
            a_new = np.ones_like(a_new) / n
        else:
            a_new = a_new / s
        # evaluate
        conv_new = np.convolve(a_new, a_new)
        score_new = float(2 * n * conv_new.max() / (a_new.sum()**2))
        # acceptance
        if score_new < best_score:
            best = a_new.copy()
            best_score = score_new
            iter_no_improve = 0
        else:
            iter_no_improve += 1
        # update a for next iteration: keep new a or maybe keep best? We'll continue with a_new.
        a = a_new

        # adaptive eta: if no improvement for many iter, reduce eta.
        if iter_no_improve > max_no_improve:
            eta *= 0.5
            iter_no_improve = 0
            # random restart sometimes
            if rng.random() < 0.5:
                # random exponential
                a = rng.exponential(scale=1.0, size=n)
                a = a / a.sum()
        # occasional random perturb

            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))

            a_pert[start:start+width] *= factor
            a_pert = np.clip(a_pert, 0.0, None)
            a_pert = a_pert / a_pert.sum()
            # evaluate
            conv_pert = np.convolve(a_pert, a_pert)
            score_pert = float(2 * n * conv_pert.max() / (a_pert.sum()**2))
            if score_pert < best_score:
                best = a_pert.copy()
                best_score = score_pert
                a = a_pert
        # enforce some occasional symmetrization - maybe improve

            a_sym = 0.5 * (a + a[::-1])

            a = a_sym

    # ensure within [0,1000] and nonnegative already

    # but maintain sum >0.01 -> already 1.
    result_list = [float(x) for x in result]
    return result_list

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
    # base from height_sequence_1 but maybe not good; we will randomize
    base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0) if 'height_sequence_1' in globals() else np.ones(n, dtype=np.float64)
    if base.ndim != 1 or base.size != n or not np.isfinite(base).all():
        base = np.ones(n, dtype=np.float64)

    # function to normalize
    def normalize(x):
        s = x.sum()
        if s == 0.0:
            return np.full_like(x, 1.0 / x.size)
        return x / s

    # initial candidate: use random exponential distribution
    a = normalize(rng.exponential(scale=1.0, size=n))
    # also keep best as initial base normalized
    best = normalize(base)
    best_score = float('inf')
    # evaluate best and initial a
    for cand in (best, a):
        conv = np.convolve(cand, cand)
        score = float(2 * n * conv.max() / (cand.sum()**2))
        if score < best_score:
            best = cand.copy()
            best_score = score

    # parameters
    eta0 = 0.15
    decay = 0.9995
    max_iters_per_restart = 500
    restarts = 0

    while time.monotonic() < deadline:
        # start a new restart with random perturb from best or random
        if restarts == 0 or rng.random() < 0.3:
            a = normalize(rng.exponential(scale=1.0, size=n))
        else:
            # perturb best slightly
            a = best.copy()
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.1))
            a[start:start+width] *= factor
            a = np.clip(a, 0.0, None)
            a = normalize(a)

        eta = eta0
        prev_score = float('inf')
        for it in range(max_iters_per_restart):
            if time.monotonic() >= deadline:
                break
            conv = np.convolve(a, a)
            max_idx = int(conv.argmax())
            max_val = conv[max_idx]

            # compute gradient for max index
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n, max_idx + 1)
            idx = np.arange(i_start, i_end)
            g = np.zeros_like(a)
            g[idx] = 2.0 * a[max_idx - idx]

            # step
            a = a - eta * g
            a = np.clip(a, 0.0, None)
            a = normalize(a)

            # evaluate score
            conv_new = np.convolve(a, a)
            score = float(2 * n * conv_new.max() / (a.sum()**2))

            # keep best
            if score < best_score:
                best = a.copy()
                best_score = score

            # break if improvement stalls
            if abs(prev_score - score) < 1e-12:
                break
            prev_score = score
            eta *= decay

        restarts += 1

    # ensure values within [0,1000];
    best = np.clip(best, 0.0, 1000.0)
    # adjust scale to make sum at least 0.01 (already 1)
    return [float(x) for x in best]

    """Return a non‑negative vector with L1‑norm 1."""
    s = v.sum()
    if s == 0.0:

def _max_convolution(a: np.ndarray):
    """Full linear convolution of a with itself and the index of its maximum."""
    c = np.convolve(a, a)
    idx = int(c.argmax())
    return c, idx, c[idx]


def _gradient_for_max(a: np.ndarray, k: int) -> np.ndarray:

    while time.monotonic() < deadline:

        # start from a fresh random point (Dirichlet‑like) or a perturbed best
        if restarts == 0 or rng.random() < 0.35:
            a = _normalize(rng.exponential(scale=1.0, size=n))
        else:
            a = best.copy()
            width = int(rng.integers(max(1, n // 250), max(2, n // 15)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.08))
            a[start:start + width] *= factor
            a = _normalize(a)

        eta = eta_initial
        last_score = float('inf')
        stagnant = 0

        for _ in range(inner_iters):
            if time.monotonic() >= deadline:
                break

            conv, idx_max, _ = _max_convolution(a)
            grad = _gradient_for_max(a, idx_max)

            # simple projected gradient step
            a = a - eta * grad
            a = np.clip(a, 0.0, None)
            a = _normalize(a)

            # evaluate with cheap estimator first
            conv_new, _, _ = _max_convolution(a)
            cand_score = float(2 * n * conv_new.max() / (a.sum() ** 2))

            # occasional exact check (cheap for small n, safe here)
            if cand_score < best_score:
                try:
                    exact = float(evaluate_sequence(a.tolist()))
                except Exception:
                    exact = cand_score
                if math.isfinite(exact) and exact < best_score:
                    best = a.copy()
                    best_score = exact

            # early‑stop if progress stalls
            if abs(last_score - cand_score) < 1e-12:
                stagnant += 1
                if stagnant >= max_no_improve:
                    break
            else:
                stagnant = 0
            last_score = cand_score
            eta *= decay

        restarts += 1

        # occasional global perturbation to escape possible plateaus
        if rng.random() < 0.07:
            pert = best.copy()
            width = int(rng.integers(max(1, n // 300), max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.15))
            pert[start:start + width] *= factor
            pert = _normalize(pert)

            try:
                sc = float(evaluate_sequence(pert.tolist()))
                if sc < best_score:
                    best = pert.copy()
                    best_score = sc
            except Exception:
                pass

    # ------------------------------------------------------------------
    # final safety: make sure the output respects the evaluator's bounds
    best = np.clip(best, 0.0, 1000.0)
    # sum is already 1 (>0.01), so the evaluator will not reject it
    return [float(x) for x in best]
