import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 bound evaluated by the pre‑imported `evaluate_sequence`.
    The simple strategy builds a uniform candidate (which attains the
    theoretical optimum bound of 2) and, if available, compares it with the
    provided parent construction `height_sequence_1`.  A brief random search
    with very low overhead is performed until the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ---- Determine problem size ------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
            if parent_arr.ndim != 1 or parent_arr.size == 0:
                raise ValueError
        except Exception:
            parent_arr = None
    else:
        parent_arr = None

    n = parent_arr.size if parent_arr is not None else 256

    # ---- Uniform baseline (theoretical optimum) -------------------------------
    uniform = np.full(n, 1.0, dtype=np.float64)          # any positive constant works
    uniform = np.clip(uniform, 0.0, 1000.0)

    # ---- Evaluate candidates ---------------------------------------------------
    def score(arr):
        return float(evaluate_sequence(arr.tolist()))

    best = uniform
    best_score = score(best)

    # Compare with the supplied parent, if it exists
    if parent_arr is not None:
        parent = np.clip(parent_arr, 0.0, 1000.0)
        parent_score = score(parent)
        if parent_score < best_score:
            best = parent
            best_score = parent_score

    # ---- Tiny random refinement (optional) ------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.05)
    # The refinement loop is deliberately lightweight: O(n) work per iteration.
    while time.monotonic() < deadline:
        # Random scaling of the uniform vector (tiny perturbation)
        scale = 0.95 + 0.10 * rng.random()   # between 0.95 and 1.05
        cand = uniform * scale

        # Occasionally zero a small random fraction of entries
        if rng.random() < 0.07:
            mask = rng.random(cand.shape) < 0.03
            cand[mask] = 0.0

        cand = np.clip(cand, 0.0, 1000.0)
        s = score(cand)
        if s < best_score:
            best = cand
            best_score = s

    # ---- Return result ---------------------------------------------------------
    return [float(x) for x in best]
