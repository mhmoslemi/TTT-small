def generate_uniform(n):

def generate_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 0.0)
    a /= a_sum  # normalize sum to 1

import time
import math
import numpy as np

def _uniform_candidate(n: int) -> np.ndarray:
    """Return a simple uniform non‑negative sequence."""
    # any positive constant works; 1.0 is convenient.
    return np.full(n, 1.0, dtype=np.float64)


def _gaussian_candidate(n: int, sigma: float) -> np.ndarray:
    """Return a symmetric Gaussian‑shaped candidate, L1‑normalised."""
    idx = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-((idx - mu) ** 2) / (2.0 * sigma ** 2))
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total                       # normalise L1‑norm to 1
    return a


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑empty list of non‑negative step heights that (approximately)
    minimise the AC1 upper bound as measured by `evaluate_sequence`.
    The routine runs a tiny time‑bounded search, starting from a uniform
    sequence and probing a few Gaussian shapes and random local perturbations.
    """
    rng = np.random.default_rng(seed)
    # safeguard against overshooting the sandbox limit
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # length of the problem – use the supplied parent if present
    try:
        n = len(height_sequence_1)          # pre‑imported by the harness
    except Exception:
        n = 256

    # ------- initial candidate ------------------------------------------------
    best = _uniform_candidate(n)
    best_score = float(evaluate_sequence(best.tolist()))

    # ------- sweep a few Gaussian widths --------------------------------------
    # sigma range: from very narrow to almost flat
    for sigma in np.linspace(n / 50.0, n / 2.0, num=24):
        cand = _gaussian_candidate(n, sigma)
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best, best_score = cand, score
        if time.monotonic() >= deadline:
            return [float(x) for x in best]

    # ------- inexpensive random block perturbations ---------------------------
    while time.monotonic() < deadline:
        # pick a random contiguous block
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        cand = best.copy()
        # multiplicative perturbation with a modest log‑normal factor
        factor = math.exp(rng.normal(0.0, 0.07))
        cand[start:stop] *= factor

        # occasional symmetric averaging – keeps the shape balanced
        if rng.random() < 0.2:
            cand = 0.5 * (cand + cand[::-1])

        # re‑normalise L1‑norm to avoid drift (the objective is scale‑invariant)
        s = cand.sum()
        if s > 0.0:
            cand *= (1.0 / s)

        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best, best_score = cand, score

    # Return the best sequence discovered (as plain Python floats)
    return [float(x) for x in best]
