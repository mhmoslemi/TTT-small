def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-3.0)
    # Get parent if present
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(256, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        n = 256
    else:
        n = parent.size
    # Compute prime p <= sqrt(n)
    # However need p^2 <= n
    sqrt_n = int(np.sqrt(n))
    def is_prime(num):
        if num < 2: return False
        if num % 2 == 0:
            return num == 2
        r = int(np.sqrt(num))
        for d in range(3, r+1, 2):
            if num % d == 0:
                return False
        return True
    p = sqrt_n
    while p > 1:
        if p*p <= n and is_prime(p):
            break
        p -= 1
    if p < 2:
        # fallback to uniform
        seq = np.full(n, 1.0/n, dtype=np.float64)
    else:
        m = p
        # construct Sidon set
        idx = np.array([a * p + (a * a) % p for a in range(p)], dtype=np.int64)
        # ensure idx < n
        idx = idx[idx < n]
        m = idx.size
        N = n - m
        # compute optimal heavy mass
        A = 2.0 / (m * m)
        B = float(n) / (N * N)
        M_opt = B / (A + B) if (A + B) != 0 else 0.0
        if M_opt > 1.0:
            M_opt = 1.0
        if M_opt < 0.0:
            M_opt = 0.0
        alpha = M_opt / m if m > 0 else 0.0
        epsilon = (1.0 - M_opt) / N if N > 0 else 0.0
        # Build base sequence
        seq = np.full(n, epsilon, dtype=np.float64)
        seq[idx] = alpha
        # renormalize to guard rounding
        total = seq.sum()
        if total <= 0:
            seq = np.full(n, 1.0/n)
        else:
            seq *= 1.0 / total
    # Ensure values within [0,1000]
    seq = np.clip(seq, 0.0, 1000.0)
    best = seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # Local search: random multiplicative perturbations on random subset
    # We'll do simple loops for duration
    while time.monotonic() < deadline:
        # Choose random indices to perturb
        cand = best.copy()
        # number of perturbed indices randomly between 1 and 5% of n
        num = int(rng.integers(1, max(2, n // 200)))  # up to 1% maybe
        ids = rng.choice(n, size=num, replace=False)
        # Apply multiplicative perturbation
        factor = np.exp(rng.normal(0.0, 0.2, size=num))
        cand[ids] *= factor
        # keep nonnegative
        cand = np.clip(cand, 0.0, 1000.0)
        # renormalize sum to 1 (scale invariant)
        s = cand.sum()
        if s == 0:
            continue
        cand /= s
        # Evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
    # Return best as list of floats
    return [float(x) for x in best]

import numpy as np
import time

def _is_prime(x: int) -> bool:
    """Simple primality test for small integers."""
    if x < 2:
        return False
    if x % 2 == 0:
        return x == 2
    r = int(x**0.5)
    for d in range(3, r + 1, 2):
        if x % d == 0:
            return False
    return True


def _sidon_set(p: int, n: int) -> np.ndarray:
    """
    Construct the classic Sidon set S = { a*p + a^2 (mod p) : a=0..p-1 }.
    All elements are < p^2 ≤ n, so they fit into the sequence length.
    """
    arr = np.empty(p, dtype=np.int64)
    for a in range(p):
        arr[a] = a * p + (a * a) % p
    # keep only those lying inside the allowed range
    return arr[arr < n]
