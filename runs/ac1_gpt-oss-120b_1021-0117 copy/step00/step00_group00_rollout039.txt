def swap_move(a, rng):
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    cand = a.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand

def random_move(a, rng):

    cand = a + delta
    cand = np.clip(cand, 0.0, 1000.0)
    # ensure sum >0.01, scale to original sum
    s = np.sum(cand)
    if s < 0.01:
        cand = a
    else:
        cand *= np.sum(a) / s
    return cand

def conv_fft(a):
    n = a.size

    A = np.fft.rfft(a, L)
    conv = np.fft.irfft(A * A, L)

import numpy as np
import time
import math

# ---------- helper utilities (top level, no closures) ----------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Convolution of a with itself using an FFT (full length)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())          # next power of two
    A = np.fft.rfft(a, L)
    conv_full = np.fft.irfft(A * A, L)[: 2 * n - 1]
    return conv_full


def _fast_score(a: np.ndarray) -> float:
    """Quick estimate of the AC1 bound (uses FFT, no clipping)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = a.sum()
    if s <= 0.0:
        return np.inf
    conv = _conv_fft(a)
    return 2.0 * n * conv.max() / (s * s)


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """O(n) non‑negative block perturbation of ``base``."""
    cand = np.array(base, copy=True)
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(math.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional mirroring
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _exponential_candidate(n: int, c: float) -> np.ndarray:
    """Decreasing geometric shape a_i = exp(-c * i / n)."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-c * i / n)


def _power_candidate(n: int, p: float) -> np.ndarray:
    """Power‑law shape a_i = (i+1)^p   (p < 0 gives a decreasing sequence)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return np.power(i, p)


def _dirichlet_candidate(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Symmetric Dirichlet sample (Gamma‑based)."""
    g = rng.gamma(alpha, 1.0, size=n)
    a = g / g.sum()
    return a


def _transfer_move(a: np.ndarray, rng: np.random.Generator, max_frac: float = 0.1) -> np.ndarray:
    """Move a random fraction from one entry to another (keeps sum constant)."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    delta = max_frac * rng.random() * a[i]
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    return np.clip(b, 0.0, 1000.0)

        rev = cand[::-1]
        if _fast_score(rev) < best_score * 1.1:
            score = float(evaluate_sequence(rev.tolist()))
            if score < best_score:
                best = rev.copy()
                best_score = score

    # ------------------------------------------------------------------------
    # 2) stochastic exploration (Dirichlet draws + local moves)
    # ------------------------------------------------------------------------
    while time.monotonic() < deadline:
        # Dirichlet sample with a random concentration parameter
        alpha = rng.uniform(0.2, 5.0)
        cand = _dirichlet_candidate(n, rng, alpha)

        # occasional small block perturbation to add diversity
        if rng.random() < 0.2:
            cand = _make_block_perturbation(cand, rng)

        if _fast_score(cand) < best_score * 1.05:
            score = float(evaluate_sequence(cand.tolist()))
            if score < best_score:
                best = cand.copy()
                best_score = score

        # small mass‑transfer moves from the current best
        if rng.random() < 0.3:
            cand = _transfer_move(best, rng, max_frac=0.05)
            if _fast_score(cand) < best_score * 1.02:
                score = float(evaluate_sequence(cand.tolist()))
                if score < best_score:
                    best = cand.copy()
                    best_score = score

        # block perturbation of the current best
        if rng.random() < 0.15:
            cand = _make_block_perturbation(best, rng)
            if _fast_score(cand) < best_score * 1.03:
                score = float(evaluate_sequence(cand.tolist()))
                if score < best_score:
                    best = cand.copy()
                    best_score = score

        # stop early if we have already beaten a strong target
        if best_score < 1.55:
            break

    # ------------------------------------------------------------------------
    # final safety fallback
    # ------------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = np.clip(parent, 0.0, 1000.0)

    return [float(x) for x in best]
