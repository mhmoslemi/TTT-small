def reduce_max_step(a, rng):
    n = a.size
    b = np.convolve(a, a)
    k = int(b.argmax())
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    partner = a[k - idx]
    prod = a[idx] * partner
    # find index with max product
    t = int(idx[np.argmax(prod)])
    u = k - t
    # find low index
    # choose a low-weight index not equal t or u
    low_candidates = np.where(~np.isin(np.arange(n), [t, u]))[0]
    if low_candidates.size == 0:
        return a
    low = int(rng.choice(low_candidates))
    delta = min(a[t], a[u]) * 0.2  # move 20% of smaller
    if delta < 1e-12: # too small
        return a
    a2 = a.copy()
    a2[t] -= delta
    a2[u] -= delta
    a2[low] += 2 * delta
    np.clip(a2, 0.0, 1000.0, out=a2)
    return a2

# Helper: evaluate the exact score (identical to the provided evaluator)
def _score(seq: np.ndarray) -> float:
    """Return the AC1 upper‑bound score for a non‑negative sequence."""
    s = seq.sum()
    if s < 0.01:
        return np.inf
    # full linear convolution (np.convolve is O(n²) but fast in C)
    b = np.convolve(seq, seq)
    return float(2.0 * seq.size * b.max() / (s * s))

    best = a.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # We repeatedly identify the lag with maximal convolution value,
    # then move a small amount of mass from the pair contributing the
    # most to that lag towards the pair contributing the least.
    while time.monotonic() < deadline:
        # compute full convolution and the worst lag
        conv = np.convolve(a, a)
        k = int(conv.argmax())                     # lag with maximal value

        # indices that take part in this lag
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        idx = np.arange(i0, i1 + 1, dtype=np.int64)   # all i with 0 ≤ i ≤ n‑1 and 0 ≤ k‑i ≤ n‑1
        partner = a[k - idx]                         # a_{k‑i}
        prod = a[idx] * partner                       # contribution of each pair

        # pair with the largest contribution
        imax = int(idx[np.argmax(prod)])
        jmax = k - imax

        # pair with the smallest (non‑zero) contribution – the place we will
        # move mass to.  If all contributions are equal we pick a random
        # index outside the current pair.
        if prod.size > 1:
            imin = int(idx[np.argmin(prod)])
            jmin = k - imin
            if imin in (imax, jmax) or jmin in (imax, jmax):
                # pick a truly different low‑contribution index
                candidates = np.setdiff1d(np.arange(n), [imax, jmax])
                imin = int(rng.choice(candidates))
                jmin = int(rng.choice(candidates))
        else:
            candidates = np.setdiff1d(np.arange(n), [imax, jmax])
            imin = int(rng.choice(candidates))
            jmin = int(rng.choice(candidates))

        # move a modest fraction of mass from the high pair to the low pair
        delta_i = 0.2 * a[imax]                # 20 % of the larger entry
        delta_j = 0.2 * a[jmax]                # 20 % of its partner
        delta = min(delta_i, delta_j, a[imax], a[jmax])

        if delta <= 0.0:
            # no mass left to move – perform a tiny random perturbation to stay active
            width = max(1, n // 500)
            start = rng.integers(0, n - width + 1)
            a[start:start + width] *= np.exp(rng.normal(0.0, 0.05))
            a = np.clip(a, 0.0, 1000.0)
            a /= a.sum()
            continue

        new_a = a.copy()
        new_a[imax] -= delta
        new_a[jmax] -= delta
        new_a[imin] += delta
        new_a[jmin] += delta

        # keep the vector feasible
        np.clip(new_a, 0.0, 1000.0, out=new_a)
        new_a_sum = new_a.sum()
        if new_a_sum < 0.01:                # safeguard against accidental zero‑sum
            new_a = np.full_like(new_a, 1.0 / n)
        else:
            new_a /= new_a_sum               # re‑normalise to L¹ = 1

        cur_score = _score(new_a)

        if cur_score < best_score:
            best = new_a.copy()
            best_score = cur_score
            a = new_a
            # stop early if we have already beaten the target
            if best_score < 1.50274349:
                break
        else:
            # occasional uphill move (tiny probability) to escape plateaus
            if rng.random() < 0.001:
                a = new_a

        # every few hundred iterations perform a broader random shake‑up
        if rng.random() < 0.004:
            width = max(1, n // 200)
            start = rng.integers(0, n - width + 1)
            a[start:start + width] *= np.exp(rng.normal(0.0, 0.12))
            a = np.clip(a, 0.0, 1000.0)
            a /= a.sum()

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list of floats).
    return [float(x) for x in best]
