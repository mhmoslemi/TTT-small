def try_step(a, max_idx, min_idx, rng):
    n = len(a)
    i = np.arange(n)
    # compute mask for max_idx
    j = max_idx - i
    mask_max = (j >= 0) & (j < n)
    # contributions product
    prod_max = np.zeros_like(a)
    prod_max[mask_max] = a[i[mask_max]] * a[j[mask_max]]
    # find i_max = argmax prod_max
    i_max = int(np.argmax(prod_max))
    # similarly for min_idx
    j2 = min_idx - i
    mask_min = (j2 >= 0) & (j2 < n)
    prod_min = np.zeros_like(a)
    prod_min[mask_min] = a[i[mask_min]] * a[j2[mask_min]]

    zero_mask = (prod_min == 0) & (a < 1000.0)

import numpy as np, time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # get parent if available
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise Exception
        base = np.clip(base, 0.0, 1000.0)
    except Exception:
        base = np.ones(256, dtype=np.float64)
    n = base.size
    # Normalise to sum 1 (scale invariant)
    total = base.sum()
    if total <= 0.0:
        base = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        base = base / total
    # start from a slightly perturbed version if base is perfectly uniform
    if np.allclose(base, np.full_like(base, 1.0 / n)):
        # add tiny random noise
        base = base * (1.0 + 0.01 * rng.normal(size=n))
        base = np.clip(base, 0.0, None)
        base = base / base.sum()
    cur = base.copy()
    best = cur.copy()
    # evaluate using authoritative evaluator for robustness
    best_score = float(evaluate_sequence(best.tolist()))
    cur_score = best_score

    # Parameters for the directed moves
    step_factor = 0.2          # fraction of mass moved each step
    iter_no_improve = 0
    iter_total = 0

    while time.monotonic() < deadline:
        iter_total += 1
        # compute convolution and locate the peak
        conv = np.convolve(cur, cur)
        max_idx = int(np.argmax(conv))

        # contributions of each position to that peak (ordered pairs)
        i = np.arange(n)
        partner = max_idx - i
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(cur)
        contrib[mask] = cur[i[mask]] * cur[partner[mask]]

        # indices with largest / smallest contribution
        i_max = int(np.argmax(contrib))
        # choose a target index whose contribution to the same peak is minimal
        # prefer entries that are not saturated at 1000 (here values are tiny)
        candidates = np.where((contrib == contrib.min()))[0]
        if candidates.size == 0:
            # fallback: any index different from i_max
            candidates = np.arange(n)
            candidates = candidates[candidates != i_max]
        i_min = int(rng.choice(candidates))

        # move a fraction of mass from i_max to i_min
        delta = cur[i_max] * step_factor
        # keep within bounds
        delta = min(delta, 1000.0 - cur[i_min])
        if delta <= 0.0:
            # try a smaller step
            delta = cur[i_max] * (step_factor * 0.1)
            if delta <= 0.0:
                continue
        new = cur.copy()
        new[i_max] -= delta
        new[i_min] += delta

        # keep total mass unchanged (numerical drift possible)
        # normalise just in case
        s = new.sum()
        if s == 0.0:
            continue
        new /= s

        new_score = float(evaluate_sequence(new.tolist()))
        if np.isfinite(new_score) and new_score < cur_score:
            cur = new
            cur_score = new_score
            iter_no_improve = 0
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
        else:
            iter_no_improve += 1

        # occasionally shrink step size to allow fine tuning
        if iter_total % 200 == 0:
            step_factor = max(0.01, step_factor * 0.9)

        # occasional random shake to escape stagnation
        if iter_no_improve > 500:
            # random block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.3)))
            cur[start:start+width] *= factor
            np.clip(cur, 0.0, 1000.0, out=cur)
            cur /= cur.sum()
            cur_score = float(evaluate_sequence(cur.tolist()))
            iter_no_improve = 0

    # final safeguard: ensure sum > 0.01
    if best.sum() < 0.01:
        best = np.full_like(best, 0.01 / n)

    # Return as plain Python list
    return [float(x) for x in best]

import time
import numpy as np

def _eval(seq):
    """Fast wrapper around the authoritative evaluator."""
    return float(evaluate_sequence(seq.tolist()))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic that tries to flatten the self‑convolution of a non‑negative
    sequence.  It works with the parent sequence ``height_sequence_1`` when
    available, otherwise falls back to a uniform vector.
    """
    rng = np.random.default_rng(seed)
    # leave a small margin for the return path
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # initialise --------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise Exception
        base = np.clip(base, 0.0, 1000.0)
    except Exception:
        base = np.ones(256, dtype=np.float64)

    n = base.size
    s = base.sum()
    if s <= 0.0:
        cur = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        cur = base / s                     # scale to sum‑one (scale‑invariant)

    # break perfect symmetry if the parent happens to be exactly uniform
    if np.allclose(cur, np.full_like(cur, 1.0 / n)):
        cur *= 1.0 + 0.01 * rng.normal(size=n)
        cur = np.clip(cur, 0.0, None)
        cur /= cur.sum()

    best = cur.copy()
    best_score = _eval(best)
    cur_score = best_score

    # ------------------------------------------------------------------
    # main loop ---------------------------------------------------------
    step = 0.20                # fraction of mass moved each directed step
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # convolve and locate the current peak
        conv = np.convolve(cur, cur)
        k_max = int(np.argmax(conv))

        # contribution of each index to that peak (ordered pairs)
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(cur)
        contrib[mask] = cur[idx[mask]] * cur[partner[mask]]

        # choose a source (large contribution) and a sink (small contribution)
        i_src = int(np.argmax(contrib))
        # prefer a sink that actually contributes zero – this cannot increase
        # the current peak
        zero_sink = np.where((contrib == 0.0) & (cur < 1000.0))[0]
        if zero_sink.size:
            i_sink = int(rng.choice(zero_sink))
        else:
            # otherwise take one of the smallest contributors with room to grow
            cand = np.argsort(contrib)[:min(20, n)]
            cand = cand[cur[cand] < 1000.0]
            if cand.size == 0:
                i_sink = int(rng.choice(np.arange(n)))
            else:
                i_sink = int(rng.choice(cand))

        # move a fraction of the mass
        delta = cur[i_src] * step
        delta = min(delta, 1000.0 - cur[i_sink])          # respect the upper bound
        if delta <= 0.0:                                 # nothing to move
            continue
        new = cur.copy()
        new[i_src] -= delta
        new[i_sink] += delta
        # renormalise to avoid drift caused by rounding
        tot = new.sum()
        if tot == 0.0:
            continue
        new /= tot

        new_score = _eval(new)

        if np.isfinite(new_score) and new_score < cur_score:
            cur = new
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
        else:
            no_improve += 1

        # gradually shrink the step size for fine tuning
        if it % 200 == 0:
            step = max(0.01, step * 0.9)

        # occasional random shake‑up if we are stuck
        if no_improve > 800:
            w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            s0 = int(rng.integers(0, max(1, n - w + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.35)))
            cur[s0:s0 + w] *= factor
            np.clip(cur, 0.0, 1000.0, out=cur)
            cur /= cur.sum()
            cur_score = _eval(cur)
            no_improve = 0

    # ------------------------------------------------------------------
    # final safety checks ------------------------------------------------
    if best.sum() < 0.01:                     # never return an invalid sum
        best = np.full_like(best, 0.01 / n)

    return [float(x) for x in best]
