import time
import math
import numpy as np

def _build_shape(alpha: float, n: int) -> np.ndarray:
    """Return a non‑negative step function of length ``n``.
    a_i = (n-i)^{alpha}, scaled so that the maximum entry equals 1000.
    """
    # descending integers n, n-1, ..., 1 as float
    arr = np.arange(n, 0, -1, dtype=np.float64)
    # raise to the chosen power (alpha is an integer in our usage)
    vals = np.power(arr, alpha)
    # scale to respect the 0‑1000 bound
    scale = 1000.0 / vals[0]
    a = vals * scale
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a sequence that minimizes the AC1 upper bound.

    The construction uses the deterministic shape a_i = (n‑i)^α,
    trying a few small integer exponents and keeping the best one.
    """
    rng = np.random.default_rng(seed)  # kept for possible future extensions
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Determine the length to work with – use the provided parent if sensible.
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent_arr.ndim != 1
            or parent_arr.size == 0
            or not np.isfinite(parent_arr).all()

            n = 256
        else:
            n = int(parent_arr.size)
    except Exception:  # pragma: no cover
        n = 256

    # Candidate exponents – α = 5 gives a very low bound for typical n.
    exponents = [2, 3, 4, 5]

    best_seq = None
    best_score = float("inf")

    for a in exponents:
        cand = _build_shape(float(a), n)
        # authoritative evaluator (authoritative, no shortcuts)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:

        best_seq = np.ones(n, dtype=np.float64)

    if time.monotonic() > deadline:

        return [float(x) for x in best_seq]
