def build_sidon_set(n):
    # Use base set from quadratic construction with prime p.
    p = max prime <= int(np.sqrt(n))
    base = [i * p + (i * i) % p for i in range(p) if i * p + (i * i) % p < n]
    # Already distinct values.
    # Make set (list) sorted.
    applied = set(base)
    used_sums = np.zeros(2*n, dtype=bool)
    # mark sums for base.
    for i in base:
        for j in base:
            if j >= i:  # only need half
                used_sums[i + j] = True
    # then iterate over remaining indices.
    for i in range(n):
        if i in applied:
            continue
        # check if i adds no new sum collisions
        conflict=False
        for j in applied:
            if used_sums[i + j]:
                conflict=True
                break
        if not conflict and not used_sums[i + i]:
            # add i
            for j in applied:
                used_sums[i + j]=True
                used_sums[j + i]=True  # symmetrical but same
            used_sums[i + i]=True
            applied.add(i)
    return sorted(applied)

import numpy as np
import time
import math

def _is_prime(x: int) -> bool:
    if x < 2:
        return False
    if x % 2 == 0:
        return x == 2
    r = int(math.isqrt(x))
    for d in range(3, r + 1, 2):
        if x % d == 0:
            return False
    return True


def _largest_prime_le(limit: int) -> int:
    for p in range(limit, 1, -1):
        if _is_prime(p):
            return p
    return 2


def _sidon_set(n: int) -> np.ndarray:
    """
    Return a Sidon (B2) set of indices in [0, n).
    Starts from the classic quadratic construction and then greedily
    adds more indices while preserving the Sidon property.
    """
    p = _largest_prime_le(int(math.isqrt(n)))
    # quadratic construction: i·p + (i² mod p)
    base = [i * p + (i * i) % p for i in range(p) if i * p + (i * i) % p < n]

    idx_set = set(base)
    # used[t] == True  <=>  there already exists a pair (i,j) with i<=j, i+j == t
    used = np.zeros(2 * n, dtype=bool)

    # mark sums from the base set
    base_list = list(idx_set)
    Lb = len(base_list)
    for i in range(Lb):
        a = base_list[i]
        for j in range(i, Lb):
            b = base_list[j]
            used[a + b] = True

    # greedy augmentation
    for cand in range(n):
        if cand in idx_set:
            continue
        # check self‑sum
        if used[cand + cand]:
            continue
        # check sums with existing members
        conflict = False
        for e in idx_set:
            if used[cand + e]:
                conflict = True
                break
        if conflict:
            continue
        # safe: add it and record new sums
        for e in idx_set:
            used[cand + e] = True
            used[e + cand] = True
        used[cand + cand] = True
        idx_set.add(cand)

    return np.array(sorted(idx_set), dtype=np.int64)
