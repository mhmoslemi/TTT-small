import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step sequence with a low AC1 bound.

    The routine works on the normalized problem (∑a = √(2 n)) and
    minimises the maximal coefficient of the self‑convolution using
    cheap FFT‑based updates.  It never calls the slow authoritative
    evaluator inside the main loop – only at the very end would the
    harness do so.
    """
    rng = np.random.default_rng(seed)
    # leave a few seconds margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # initialise the sequence ------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception()
        a = np.clip(base, 0.0, 1000.0)
    except Exception:
        # very small fallback if the parent is unavailable
        n = 256
        a = np.full(n, 1.0, dtype=np.float64)

    n = a.size
    target_sum = np.sqrt(2.0 * n)
    cur = a.sum()
    if cur <= 0.0:
        cur = 1.0
    a = a * (target_sum / cur)                     # normalise ∑a = √(2 n)

    # ------------------------------------------------------------------
    # helper for FFT based convolution --------------------------------------
    m = 1 << ((2 * n - 1).bit_length())            # power‑of‑two ≥ 2n‑1
    a_pad = np.zeros(m, dtype=np.float64)

    def max_convolution(x):
        """return (max coefficient, index) of x*x using FFT."""
        a_pad[:n] = x
        fx = np.fft.rfft(a_pad)
        conv = np.fft.irfft(fx * fx)[: 2 * n - 1]
        idx = int(conv.argmax())
        return float(conv.max()), idx

    # current best ---------------------------------------------------------
    best = a.copy()
    best_max, _ = max_convolution(best)
    best_score = 2.0 * n * best_max / (target_sum ** 2)

    # ------------------------------------------------------------------
    # main optimisation loop ------------------------------------------------
    eps = 0.02                     # relative amount taken from a “large” entry
    while time.monotonic() < deadline:
        cur_max, kmax = max_convolution(a)

        # partner values for the dominant sum kmax
        idx = kmax - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        partner = np.zeros_like(a)
        partner[mask] = a[idx[mask]]

        # index whose product a_i * a_{kmax-i} is largest
        i_max = int((a * partner).argmax())
        j = kmax - i_max                     # the paired index (may equal i_max)

        # --------------------------------------------------------------
        # propose a local change that reduces the dominant term
        # --------------------------------------------------------------
        if i_max == j:                       # square term a_i^2
            max_delta = a[i_max] / 2.0
            delta = min(eps * a[i_max], max_delta)
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[i_max] -= 2.0 * delta
            # choose a location that does NOT influence the current kmax
            zero_places = np.where(~mask)[0]
            if zero_places.size:
                r = int(rng.choice(zero_places))
            else:
                r = int(rng.integers(0, n - 1))
                while r == i_max:
                    r = int(rng.integers(0, n - 1))
            a_new[r] += 2.0 * delta
        else:
            max_delta = min(a[i_max], a[j])
            delta = min(eps * a[i_max], max_delta)
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[i_max] -= delta
            a_new[j] -= delta
            # pick a receiver that is invisible to the current kmax
            zero_places = np.where(~mask)[0]
            cand = zero_places[(zero_places != i_max) & (zero_places != j)]
            if cand.size:
                r = int(rng.choice(cand))
            else:
                r = int(rng.integers(0, n - 1))
                while r == i_max or r == j:
                    r = int(rng.integers(0, n - 1))
            a_new[r] += 2.0 * delta

        # enforce bounds and exact sum (renormalise to avoid drift)
        np.clip(a_new, 0.0, 1000.0, out=a_new)
        s = a_new.sum()
        if s <= 0.0:
            continue
        a_new *= (target_sum / s)

        # evaluate the proposal (fast, no np.convolve)
        new_max, _ = max_convolution(a_new)
        new_score = 2.0 * n * new_max / (target_sum ** 2)

        if new_score < best_score:
            best = a_new
            best_score = new_score
            a = a_new                       # continue from the improvement
        else:
            # occasional random acceptance to avoid stalling
            if rng.random() < 0.001:
                a = a_new

    # ------------------------------------------------------------------
    # return the best sequence found (as a plain Python list)
    return [float(v) for v in best]
