        width = int(rng.integers(w_low, w_high + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))
        # multiplicative log‑normal perturbation
        factor = float(np.exp(rng.normal(0.0, 0.07)))
        cand[start:start + width] *= factor

        # occasional mild smoothing – cheap, O(n)
        if rng.random() < 0.12:
            # replace the block by its average to keep the shape simple
            avg = float(cand[start:start + width].mean())
            cand[start:start + width] = avg

        # enforce the hard bounds required by the evaluator
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate exactly (authoritative)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = cand, score

    # ------------------------------------------------------------------
    # Return a plain Python list of floats (the evaluator accepts it).
    # ------------------------------------------------------------------
    return [float(v) for v in best]
