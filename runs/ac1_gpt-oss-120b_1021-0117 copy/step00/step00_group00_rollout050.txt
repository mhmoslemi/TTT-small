import numpy as np
import time

def _conv_max_fft(a: np.ndarray, n_fft: int) -> float:
    """Return max of a*a using a power-of-two FFT (linear convolution)."""
    A = np.fft.rfft(a, n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[: 2 * a.size - 1]
    return float(conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a non‑negative step sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ----------------------------------------------------------------------
    # initialise a starting sequence ------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
            raise ValueError
        a = np.clip(init, 0.0, 1000.0)
    except Exception:
        a = np.full(4806, 1.0, dtype=np.float64)          # fallback length

    n = a.size
    # work with a normalised vector; scaling does not affect the score
    a = a / a.sum()
    a = a.astype(np.float64, copy=False)

    # FFT size – next power of two >= 2n‑1
    n_fft = 1 << ((2 * n - 1).bit_length())

    # ----------------------------------------------------------------------
    # helper to compute the surrogate score (identical to the official one)
    def _score(arr: np.ndarray):
        max_b = _conv_max_fft(arr, n_fft)
        s = arr.sum()
        return 2.0 * n * max_b / (s * s), max_b

    cur_score, cur_max = _score(a)
    best_a = a.copy()
    best_score = cur_score

    # store the best official score (might be different due to tiny numerical noise)
    best_official = float(evaluate_sequence(best_a.tolist()))

    # ----------------------------------------------------------------------
    # main hill‑climbing loop ----------------------------------------------------
    # We repeatedly damp the index that contributes most to the current maximum
    # convolution value and redistribute a small fraction of its mass elsewhere.
    # Occasionally we accept a worsening move (simulated‑annealing style) to

    while time.monotonic() < deadline:

        A = np.fft.rfft(a, n_fft)
        C = A * A

        max_b = float(conv.max())

        # contributions of each entry to the peak at kmax
        idx = np.arange(n)
