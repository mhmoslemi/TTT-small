import time
import numpy as np

def _initial_shape(length: int) -> np.ndarray:
    """Return a 1/√(i+1) shaped non‑negative vector of the given length."""
    idx = np.arange(length, dtype=np.float64)
    # 1/√(i+1) : avoids division by zero and keeps values O(1)
    return 1.0 / np.sqrt(idx + 1.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function minimizing the AC1 upper bound.
    Uses a simple gradient‑like hill‑climber starting from the 1/√i shape.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Starting point – use the theoretical 1/√i shape.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = int(height_sequence_1.shape[0])
    else:
        # fallback to a moderate length if the parent is missing
        n = 256
    cur = _initial_shape(n)

    # ensure values respect the evaluator's clipping limits
    cur = np.clip(cur, 0.0, 1000.0)

    # keep the best sequence seen so far
    best = cur.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 2. Main loop – repeatedly try to flatten the peak of the convolution.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # small helper arrays allocated once
    idx = np.arange(n, dtype=np.int64)

    while time.monotonic() < deadline:
        # full linear convolution (np.convolve uses an FFT internally for this size)
        conv = np.convolve(cur, cur)
        kmax = int(np.argmax(conv))
        M = conv[kmax]
        if M <= 0.0:
            break

        S = np.sum(cur)
        # threshold a_{k-i} = M / S  (see gradient analysis)
        thresh = M / S

        # partner values a_{kmax-i} for each i where the index is valid
        k_minus_i = kmax - idx

        partners.fill(0.0)
        partners[valid] = cur[k_minus_i[valid]]

        # indices where the partner is larger or smaller than the threshold
        high_idx = np.where(partners > thresh)[0]

        # If we cannot find a direction of improvement, break out.
        if high_idx.size == 0 or low_idx.size == 0:

        # Choose a random high‑contributing and a low‑contributing index.
        i_high = int(rng.choice(high_idx))

        max_reduce = cur[i_high]                 # cannot go below 0

            continue
