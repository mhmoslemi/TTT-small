def fast_score(seq, n, m):
    seq = np.asarray(seq, dtype=np.float64)
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)[:2*n-1]
    maxb = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * maxb / (sum_a ** 2))

# up to floating‑point rounding).  Uses an FFT of size m = next power of 2
# ≥ 2*n to keep each call O(n log n) and O(m) memory.
def _fast_ac1_score(seq, n, m):
    """Return 2*n*max(conv(seq,seq))/sum(seq)**2 using FFT."""
    # Ensure a NumPy float64 array
    a = np.asarray(seq, dtype=np.float64)
    # FFT of the zero‑padded sequence
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    # Numerical noise may produce tiny negatives – clip safely
    max_b = float(conv.max())
    sum_a = float(a.sum())
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Helper: generate a power‑law sequence a_i = (i+offset)^{-alpha}
def _power_law(alpha, offset, n):
    """Return a non‑negative step sequence of length n."""
    idx = np.arange(n, dtype=np.float64) + offset
    # exponent may be zero -> all ones, which is fine
    seq = np.power(idx, -alpha, where=idx > 0, out=np.ones_like(idx))
    return seq


# ----------------------------------------------------------------------
# Helper: a cheap random perturbation that respects the O(n) memory bound.
def _make_block_perturbation(base, rng):
    """Multiplicative Gaussian perturbation of a random contiguous block."""
    cand = np.array(base, copy=True)
    n = cand.size
    # block width ~ uniform in [n/200, n/12]
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    # multiplicative log‑normal factor
    factor = np.exp(rng.normal(0.0, 0.12))
    cand[start:stop] *= factor
    # occasional symmetrisation (helps for some shapes)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # keep inside the allowed numeric bounds
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound.

    The routine builds a few analytic families (power‑law shapes),
    picks the best by a fast FFT‑based bound, then runs a short
    stochastic local search to improve it further.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Base sequence – either the provided parent or a simple uniform fallback
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    # FFT length (next power of two ≥ 2*n)
    m = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 1) Analytic candidates: power‑law families (alpha in (0.2,0.8))
    best_seq = parent.copy()
    best_score = _fast_ac1_score(best_seq, n, m)

    alphas = np.linspace(0.2, 0.78, 30)   # 30 values gives good coverage
    offsets = np.linspace(0.1, 4.0, 12)   # small shifts avoid singularity at 0

    for alpha in alphas:
        for off in offsets:
            cand = _power_law(alpha, off, n)
            # Clip (not strictly necessary – values already < 1000)
            np.clip(cand, 0.0, 1000.0, out=cand)
            score = _fast_ac1_score(cand, n, m)
            if score < best_score:
                best_seq = cand
                best_score = score

    # ------------------------------------------------------------------
    # 2) Stochastic refinement – cheap block perturbations
    #    (the same strategy as the original example, but seeded from the
    #     analytically‑found best).
    while time.monotonic() < deadline:
        cand = _make_block_perturbation(best_seq, rng)
        score = _fast_ac1_score(cand, n, m)
        if score < best_score:
            best_seq = cand
            best_score = score

    # ------------------------------------------------------------------
    # Final validation with the authoritative evaluator (cheap, runs once)
    # Convert to plain Python list of floats as required by the harness.
    final_score = float(evaluate_sequence(best_seq.tolist()))
    # If for any unlikely reason the fast estimate mis‑led us, fallback
    # to the plain parent sequence.
    if not np.isfinite(final_score) or final_score > best_score:
        best_seq = parent
    return [float(x) for x in best_seq]
