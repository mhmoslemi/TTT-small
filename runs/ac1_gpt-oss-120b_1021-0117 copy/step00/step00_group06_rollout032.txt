def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)
    if 'height_sequence_1' in globals():
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size if parent_arr.ndim == 1 and parent_arr.size > 0 else 256
    else:
        n = 256
    # Ensure n > 0
    # baseline uniform
    best = np.full(n, 1.0/ n)  # sum=1
    best_score = evaluate_sequence(best.tolist())
    # exponential search
    # choose list of epsilons: geometric from 2e-4 to 0.2? Actually we need epsilon around 0.001 - 0.1 maybe.
    epsilons = np.logspace(np.log10(5e-4), np.log10(0.1), num=30)  # adjust
    for eps in epsilons:
        r = 1.0 - eps
        if r <= 0.0:
            continue
        # compute a = r**np.arange(n)
        idx = np.arange(n, dtype=np.float64)
        a = np.exp(idx * math.log(r))  # stable
        total = a.sum()
        if total == 0.0:
            continue
        a = a / total
        # ensure not exceed 1000
        a = np.clip(a, 0.0, 1000.0)
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < best_score:
            best = a
            best_score = score
    # local random block perturbations
    # simple block perturbation as earlier
    def make_perturb(base):
        cand = base.copy()
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])
        return np.clip(cand, 0.0, 1000.0)

    while time.monotonic() < deadline:
        candidate = make_perturb(best)
        score = evaluate_sequence(candidate.tolist())
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
    return best.tolist()

import numpy as np, time, math

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return the uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
