    n = a.shape[0]
    # compute convolution via FFT
    m = 1 << ((2*n - 1).bit_length())

    A = np.fft.rfft(a, n=m)

        # accept

        # mark sums used
        if selected:
            # mark with others:
            pair_used[sums] = True
        pair_used[2*idx] = True

def propose_candidate(seed=42,budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0
    # get parent if available

        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),0.0,1000.0)
        if parent.ndim !=1 or parent.size==0 or not np.isfinite(parent).all():
            raise Exception()
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    n = parent.size

    # baseline best (using parent)
    best = parent.copy()
    best = np.maximum(best, 1e-12)  # avoid zeros
    best = best / best.sum()
    best_score = float(evaluate_sequence(best.tolist()))
    best_raw = best

    # utility functions
    def fast_score(arr):
        a = np.asarray(arr, dtype=np.float64)
        m = 1 << ((2*n-1).bit_length())
        A = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(A * A, n=m)[:2*n-1]
        maxb = conv.max()
        sum_a = a.sum()
        if sum_a < 0.01:
            return np.inf
        return 2 * n * maxb / (sum_a * sum_a)

    # generate exponential candidates
    eps_vals = np.logspace(np.log10(1.0/(n*10.0)), np.log10(0.01), num=25)
    for eps in eps_vals:
        a = np.exp(-eps * np.arange(n, dtype=np.float64))
        a /= a.sum()
        score = fast_score(a)
        if score < best_score:
            best = a
            best_score = score

    # symmetric exponential
    for eps in eps_vals:
        d = np.minimum(np.arange(n), n-1 - np.arange(n))
        a = np.exp(-eps * d.astype(np.float64))
        a /= a.sum()
        score = fast_score(a)
        if score < best_score:
            best = a
            best_score = score

    # random sparse sets
    for m in (int(n*0.02), int(n*0.04), int(n*0.06), int(n*0.08), int(n*0.1)):
        if m<2:
            continue
        for _ in range(5):
            a = np.zeros(n, dtype=np.float64)
            idx = rng.choice(n, size=m, replace=False)
            a[idx] = rng.random(m) + 1e-2  # ensure positive
            a /= a.sum()
            score = fast_score(a)
            if score < best_score:
                best = a
                best_score = score

    # gradient descent iterations
    while time.monotonic() < deadline:
        # compute convolution via fft
        m_fft = 1 << ((2*n-1).bit_length())
        A = np.fft.rfft(best, n=m_fft)
        conv = np.fft.irfft(A * A, n=m_fft)[:2*n-1]
        maxb = conv.max()
        # compute gradient
        tol = 1e-12 * maxb if maxb>0 else 0.0
        ks = np.where(conv >= maxb - tol)[0]
        grad = np.zeros_like(best)
        for k in ks:
            start = max(0, k - (n-1))
            end = min(k, n-1)
            i = np.arange(start, end+1)
            j = k - i
            grad[i] += 2.0 * best[j]
        # step size
        eta = 0.1
        # optional adapt: if any negative after step, shrink eta
        step = best - eta * grad
        if (step < 0).any():
            # reduce eta until nonnegative (simple halving)
            while (step < 0).any() and eta > 1e-8:
                eta *= 0.5
                step = best - eta * grad
        new = step
        new_sum = new.sum()
        if new_sum <= 0:
            new = best.copy()
        else:
            new = np.maximum(new, 0.0)
            new = new / new_sum  # renorm to sum=1
        new_score = fast_score(new)
        if new_score < best_score:
            best = new
            best_score = new_score
        else:
            # occasional random perturbation to escape local minima
            if rng.random() < 0.05:
                # block perturb as earlier
                width = int(rng.integers(max(1, n // 200), max(2, n // 12)))

                stop = min(n, start + width)
                factor = np.exp(rng.normal(0.0, 0.12))
                cand = best.copy()
                cand[start:stop] *= factor
                cand = np.maximum(cand, 0.0)
                cand /= cand.sum()
                cand_score = fast_score(cand)
                if cand_score < best_score:
                    best = cand
                    best_score = cand_score

    # final evaluate with authoritative evaluator to ensure validity and update best
    final_score = float(evaluate_sequence(best.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
        # best already in variable

    return [float(x) for x in best]

import numpy as np
import time

# ------------------------------------------------------------
# Helper: fast evaluation of the objective (no scaling)
# ------------------------------------------------------------
def _fast_score(a, n):
    """Return 2*n*max(conv(a,a))/sum(a)^2 using FFT (O(n log n))."""
    a = np.asarray(a, dtype=np.float64)
    if a.sum() < 0.01:
        return np.inf
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    maxb = conv.max()
    return 2.0 * n * maxb / (a.sum() ** 2)


# ------------------------------------------------------------
# Helper: gradient of the max‑convolution entry
# ------------------------------------------------------------
def _max_conv_gradient(a, conv, n):
    """Gradient of the current maximal convolution value w.r.t. a."""
    maxb = conv.max()
    # indices where the max is attained (tolerance avoids missing ties)
    tol = maxb * 1e-12
    ks = np.where(conv >= maxb - tol)[0]

    grad = np.zeros_like(a)
    for k in ks:
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        i = np.arange(start, end + 1, dtype=np.int64)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad
