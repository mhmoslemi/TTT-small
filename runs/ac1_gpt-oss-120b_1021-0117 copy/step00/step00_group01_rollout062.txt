import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Construct a sequence that keeps the convolution peak very low
    by placing a few maximal entries on a Sidon set and filling the
    remaining positions with a small constant.  The Sidon property
    guarantees that each sum i+j occurs at most once, so the
    convolution cannot accumulate many large products at any lag.
    """
    # ------------------------------------------------------------------
    # 1.  Baseline (fallback) – use the provided parent if it exists.
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is supplied by the harness
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = int(parent.size)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:  # pragma: no cover
        # If for any reason the parent is unavailable, start from a uniform vector.
        n = 256
        best_seq = np.full(n, 1000.0, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # 2.  Build a Sidon set (B2 sequence) – indices whose pairwise sums
    #     are all distinct.  A greedy construction suffices for n≈6000.
    # ------------------------------------------------------------------
    max_val = 1000.0
    max_sq = max_val * max_val
    used_sums = set()
    sidon = []                     # list of indices belonging to the Sidon set
    for i in range(n):
        # self‑sum must be unused
        if i + i in used_sums:
            continue
        # check sums with all already chosen indices
        conflict = False
        for j in sidon:
            if i + j in used_sums:
                conflict = True
                break
        if conflict:
            continue
        # accept i
        sidon.append(i)
        # store new sums (i+i and i+j for all j∈sidon)
        used_sums.add(i + i)
        for j in sidon:
            used_sums.add(i + j)

    m = len(sidon)                 # number of maximal entries

    # ------------------------------------------------------------------
    # 3.  Choose a small filler value eps so that the contribution of
    #     the many small entries never exceeds the peak caused by a
    #     single maximal entry (max_sq).  We enforce
    #            (n‑m) * eps² ≤ max_sq
    # ------------------------------------------------------------------
    if n - m > 0:
        eps = int(np.floor(np.sqrt(max_sq / (n - m))))
    else:                           # all positions are maximal (unlikely)
        eps = 1
    eps = max(1, min(int(eps), int(max_val)))   # keep in [1,1000]

    # ------------------------------------------------------------------

    #     eps elsewhere.
    # ------------------------------------------------------------------
    seq = np.full(n, float(eps), dtype=np.float64)
    if m:

    # ------------------------------------------------------------------
    # 5.  Evaluate the candidate; keep it if it improves over the parent.
    # ------------------------------------------------------------------
    score = float(evaluate_sequence(seq.tolist()))
    if np.isfinite(score) and score < best_score:
        best_seq = seq
        best_score = score

    # ------------------------------------------------------------------
    # 6.  Return the best sequence found (as plain Python list).
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
