import numpy as np, time

def gradient_move(a, eta, rng):
    n = a.shape[0]
    # compute convolution
    d = np.convolve(a, a)
    k = int(np.argmax(d))
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi+1)
    g = np.zeros_like(a)
    g[idx] = a[k - idx]
    # compute step
    step = eta * g
    # ensure step not larger than current a (to avoid negative)
    # we will clip after subtraction
    cand = a - step
    # clip negative to zero
    cand = np.clip(cand, 0, None)
    # compute removed mass
    removed = a.sum() - cand.sum()
    if removed > 0:
        j = rng.integers(0, n)
        cand[j] += removed
    # final clip high bound
    cand = np.clip(cand, 0.0, 1000.0)
    return cand

def random_transfer(a, rng):
    n = a.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    val_i = a[i]
    if val_i <= 0:
        return a.copy()
    delta = val_i * rng.random() * 0.5   # up to half of a_i
    cand = a.copy()
    cand[i] -= delta
    cand[j] += delta
    return cand

def random_swap(a, rng):
    n = a.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    cand = a.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand

def random_block_scale(a, rng):
    n = a.shape[0]
    cand = a.copy()
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = rng.integers(0, max(1, n - width + 1))
    stop = start + width
    factor = np.exp(rng.normal(0.0, 0.12))
    cand[start:stop] *= factor
    # optional mirroring occasionally
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    if 'height_sequence_1' in globals():
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            base = np.ones(256, dtype=np.float64)
    else:
        base = np.ones(256, dtype=np.float64)

    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    current = best.copy()
    current_score = best_score

    n = best.size
    eta = 0.05

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.5:
            cand = gradient_move(current, eta, rng)
        elif r < 0.75:
            cand = random_transfer(current, rng)
        elif r < 0.9:
            cand = random_swap(current, rng)
        else:
            cand = random_block_perturbation(current, rng)  # we can reuse earlier function
        # Evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best = cand
                best_score = score
                current = cand
                current_score = score
            elif score < current_score:
                current = cand
                current_score = score
        # adapt eta occasionally
        if rng.random() < 0.01:
            eta *= 0.9
    # Return best
    return [float(x) for x in best]

def _make_block_perturbation(base, rng):
    """Scale a random contiguous block and optionally mirror."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _random_transfer(a, rng):
    """Move a random fraction of mass from one index to another."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    if a[i] == 0.0:
        return a.copy()
    delta = a[i] * rng.random() * 0.5
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    return np.clip(b, 0.0, 1000.0)


def _random_swap(a, rng):
    """Swap two entries."""
    n = a.size
    i, j = rng.choice(n, size=2, replace=False)
    b = a.copy()
    b[i], b[j] = b[j], b[i]
    return b


def _mass_shift_reduce_peak(a, rng):
    """Identify the peak of a*a and shift mass away from the biggest contributor."""
    n = a.size
    d = np.convolve(a, a)
    k = int(np.argmax(d))

    hi = min(n - 1, k)
