import time
import math
import numpy as np

def _sidon_positions(n: int, rng: np.random.Generator, needed: int) -> list:
    """Greedy construction of a Sidon set (all i+j distinct) up to `needed` size."""
    positions = []
    used_sums = set()
    order = rng.permutation(n)
    for idx in order:
        s2 = idx + idx
        if s2 in used_sums:
            continue
        conflict = False
        for p in positions:
            if idx + p in used_sums:
                conflict = True
                break
        if conflict:
            continue
        positions.append(int(idx))
        used_sums.add(s2)
        for p in positions[:-1]:
            used_sums.add(idx + p)
        if len(positions) >= needed:
            break
    return positions

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """Return a non‑negative step‑function that minimizes the AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Determine problem size (use the provided parent if possible)
    _parent = globals().get("height_sequence_1", None)
    try:
        base = np.asarray(_parent, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)

    # Simple uniform fallback (score = 2.0)
    best_seq = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # Theoretical maximum size of a Sidon set in [0,n-1]
    max_k = ( -1 + math.isqrt(1 + 8 * (2 * n - 1)) ) // 2
    # We need at least 101 elements to beat the target bound ~1.5
    need_k = 101 if max_k >= 101 else max_k

    while time.monotonic() < deadline:
        # Build a Sidon set; stop once we have enough positions.
        positions = _sidon_positions(n, rng, need_k)
        k = len(positions)
        if k < 2:                     # cannot improve the bound
            continue

        # Create the candidate sequence (weight = 1 on each chosen position)
        cand = np.zeros(n, dtype=np.float64)
