def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0,float(budget_s)-1.0)
    # get parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise Exception()
        parent_arr = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        parent_arr = np.ones(256, dtype=np.float64)
    n = parent_arr.size
    best = parent_arr.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # try deterministic uniform baseline: constant vector
    const = np.full(n, 1.0, dtype=np.float64)
    const = const / const.sum() * best.sum()   # scale so total sum same as best (scale irrelevant)
    const = np.clip(const, 0.0, 1000.0)
    score_const = float(evaluate_sequence(const.tolist()))
    if score_const < best_score:
        best = const
        best_score = score_const
    # Precompute descending indices
    desc = np.arange(n, 0, -1, dtype=np.float64)  # n..1
    # Search loop
    while time.monotonic() < deadline:
        # generate candidate shape
        # random exponent between -1 and 2?
        q = rng.uniform(0.0, 2.0)   # exponent
        raw = desc ** q
        # add some random noise variant
        if rng.random() < 0.3:
            noise = rng.normal(loc=1.0, scale=0.1, size=n)
            raw = raw * noise
        # Ensure nonnegative
        raw = np.clip(raw, 0.0, None)
        # Normalize to sum same as best sum (or 1)
        raw_sum = raw.sum()
        if raw_sum == 0:
            continue
        # scaling not required, but keep sum similar to best to avoid near-zero sum
        raw = raw * (best.sum() / raw_sum)
        # Clip per limits
        raw = np.clip(raw, 0.0, 1000.0)
        s = float(evaluate_sequence(raw.tolist()))
        if np.isfinite(s) and s < best_score:
            best = raw
            best_score = s
        # also test random exponential vectors sorted descending
        exp_vec = rng.exponential(scale=1.0, size=n)
        exp_vec = np.sort(exp_vec)[::-1]   # descending
        exp_vec = exp_vec * (best.sum() / exp_vec.sum())
        exp_vec = np.clip(exp_vec,0.0,1000.0)
        s2 = float(evaluate_sequence(exp_vec.tolist()))
        if np.isfinite(s2) and s2 < best_score:
            best = exp_vec
            best_score = s2
    return [float(x) for x in best]

import numpy as np
import time

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step function that minimizes the AC1 bound:
        score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

    The routine combines a few inexpensive families of candidates
    (sorted versions, exponential/γ‑shaped vectors, geometric decay)
    with a light gradient‑like refinement on the current best.
    It respects the time budget and always returns a feasible fallback.
    """
    rng = np.random.default_rng(seed)
    # time limit (leave a small safety margin)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- base sequence -------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception()
        base = np.clip(base, 0.0, 1000.0)
    except Exception:
        base = np.ones(256, dtype=np.float64)   # safe fallback
    n = base.size

    # ----- utility --------------------------------------------------------
    def score_of(arr):
        """evaluate and return the score (float)."""
        return float(evaluate_sequence(arr.tolist()))

    # ----- initial candidates ---------------------------------------------
    best_seq = base.copy()
    best_score = score_of(best_seq)

    # uniform vector (same sum as base)
    uniform = np.full(n, 1.0, dtype=np.float64)
    uniform *= best_seq.sum() / uniform.sum()
    uniform = np.clip(uniform, 0.0, 1000.0)
    s = score_of(uniform)
    if s < best_score:
        best_seq, best_score = uniform, s

    # sorted descending version of the parent (often improves the peak)
    sorted_desc = np.sort(base)[::-1]
    s = score_of(sorted_desc)
    if s < best_score:
        best_seq, best_score = sorted_desc, s

    # ----- search loop ----------------------------------------------------
    # pre‑allocate an index array for the gradient step
    idx = np.arange(n, dtype=np.int64)

    while time.monotonic() < deadline:
        # pick a generation method at random
        method = rng.integers(0, 5)

        if method == 0:
            # exponential random numbers, sorted descending
            cand = rng.exponential(scale=1.0, size=n)
            cand = np.sort(cand)[::-1]

        elif method == 1:
            # gamma distribution (shape < 1 gives heavier tails)
            cand = rng.gamma(shape=0.8, scale=1.0, size=n)
            cand = np.sort(cand)[::-1]

        elif method == 2:
            # geometric decay a_i = exp(-t * i)  (t drawn from an exponential)
            t = rng.exponential(scale=0.5) + 1e-12   # avoid t=0
            cand = np.exp(-t * idx)

        elif method == 3:
            # mixture: start from best and perturb a random block
            cand = best_seq.copy()
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            factor = np.exp(rng.normal(0.0, 0.12))
            cand[start:stop] *= factor
            if rng.random() < 0.35:   # occasional symmetrisation
                cand = 0.5 * (cand + cand[::-1])

        else:
            # smooth version: convolve a random vector with a short box kernel
            raw = rng.random(n)
            k = rng.integers(3, 15)
            kernel = np.ones(k) / k
            cand = np.convolve(raw, kernel, mode='same')

        # normalise to the same total mass (scale invariance)
        total = cand.sum()
        if total == 0.0:
            continue
        cand *= (best_seq.sum() / total)
        cand = np.clip(cand, 0.0, 1000.0)

        s = score_of(cand)
        if np.isfinite(s) and s < best_score:
            best_seq, best_score = cand, s

        # ----- occasional cheap gradient‑like refinement ------------------
        if rng.random() < 0.12:
            # compute convolution once
            conv = np.convolve(best_seq, best_seq)
            k_max = int(np.argmax(conv))

            # gradient of the active quadratic form b_{k_max}
            partner = k_max - idx
            mask = (partner >= 0) & (partner < n)
            grad = np.zeros_like(best_seq)
            grad[mask] = 2.0 * best_seq[partner[mask]]

            # direction that tries to reduce the active peak while keeping Σa constant
            direction = -(grad - grad.mean())
            # step size proportional to typical magnitude, but not too large
            step = 0.05 * best_seq.mean()
            cand2 = best_seq + step * direction
            cand2 = np.clip(cand2, 0.0, None)
            if cand2.sum() == 0.0:
                continue
            cand2 *= (best_seq.sum() / cand2.sum())
            cand2 = np.clip(cand2, 0.0, 1000.0)

            s2 = score_of(cand2)
            if np.isfinite(s2) and s2 < best_score:
                best_seq, best_score = cand2, s2

    # return a plain Python list of floats
    return [float(v) for v in best_seq]
