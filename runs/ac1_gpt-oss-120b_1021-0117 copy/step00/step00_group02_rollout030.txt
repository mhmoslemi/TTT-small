import numpy as np
import time

def _evaluate(seq):
    """Wrapper for the provided evaluator that works with NumPy arrays."""
    return float(evaluate_sequence(seq.tolist()))

def _make_candidate(base, rng):
    """Small local perturbation (O(n) memory)."""
    n = base.size
    cand = base.copy()
    # random block width and position
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.2)))
    cand[start:stop] *= factor
    # occasional mirroring (helps symmetry)
    if rng.random() < 0.2:
        cand[:] = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) beats the target bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ----- fallback parent -------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise ValueError
        parent = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    best_seq = parent.copy()
    best_score = _evaluate(best_seq)

    n = best_seq.size
    idx = np.arange(n, dtype=np.float64)

    # ----- analytic family a_i = (i + α + 1)^(-p) -------------------------
    # search over exponents p near 0.5 and over shift α (as a fraction of n)
    p_candidates = (0.45, 0.5, 0.55)
    # coarse search: α ∈ [0, 0.6 n] with 31 points
    for p in p_candidates:
        best_frac = None
        for frac in np.linspace(0.0, 0.6, 31):
            if time.monotonic() > deadline:
                break
            alpha = frac * n
            seq = np.power(idx + alpha + 1.0, -p)
            # seq already non‑negative and far below 1000, no clipping needed
            score = _evaluate(seq)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_frac = frac

        # ----- fine refinement around best coarse fraction ----------------
        if best_frac is not None:
            lo = max(0.0, best_frac - 0.05)
            hi = min(0.6, best_frac + 0.05)

                if time.monotonic() > deadline:

                    best_score = score
                    best_seq = seq.copy()
                    best_frac = frac
