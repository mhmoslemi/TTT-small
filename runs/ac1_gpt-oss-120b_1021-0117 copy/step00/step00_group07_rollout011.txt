def fast_score(seq):
    n = seq.shape[0]
    size = 1 << ((2*n-1).bit_length())
    f = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f*f, size)[:2*n-1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (seq.sum()**2)

def fast_score(seq):
    n = seq.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    freq = np.fft.rfft(seq, size)
    conv = np.fft.irfft(freq * freq, size)
    max_b = conv[:2*n-1].max()
    return 2.0 * n * max_b / (seq.sum()**2)
```

    # Ensure nonnegative
    a = np.clip(a, 0.0, None)
    # Renormalize
    total = a.sum()
    if total == 0: return seq
