def propose_swap(a):
    B = np.convolve(a,a)
    kmax = np.argmax(B)
    partner = np.zeros_like(a)
    # compute partner for each i
    # vectorized:
    idx = np.arange(n)
    partner_indices = kmax - idx
    mask = (partner_indices >= 0) & (partner_indices < n)
    partner[mask] = a[partner_indices[mask]]
    contrib = a * partner  # product
    # compute probabilities for selecting i to decrease: proportion to contrib
    prob_high = contrib / contrib.sum()
    # for j to increase: choose weight based on inverse of partner maybe?
    weight_low = 1.0 / (partner + 1e-12)  # high when partner small
    prob_low = weight_low / weight_low.sum()
    i = rng.choice(n, p=prob_high)
    j = rng.choice(n, p=prob_low)
    # transfer delta = min(a[i], 1000-a[j]) * step
    if i == j:
        return None
    max_transfer = min(a[i], 1000 - a[j])
    if max_transfer <= 1e-12:
        return None
    delta = max_transfer * rng.random() * 0.1  # up to 10%
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return a_new
```

# Helper utilities (fast, O(n) memory)

def _score(seq: np.ndarray) -> float:
    """Exact objective used by the benchmark (no clipping needed)."""
    b = np.convolve(seq, seq)
    max_b = float(b.max())
    s = float(seq.sum())
    if s < 0.01:
        return float('inf')
    return 2.0 * seq.size * max_b / (s * s)


def _make_power_law(n: int, beta: float = 0.5, offs: float = 0.0) -> np.ndarray:
    """Generate a_i = (i+1+offs)^{-beta}."""
    idx = np.arange(n, dtype=np.float64)
    return (idx + 1.0 + offs) ** (-beta)


def _targeted_transfer(cur: np.ndarray, kmax: int, rng: np.random.Generator) -> np.ndarray | None:
    """Move a small amount of mass from a high‑contributing index to a low‑partner one."""
    n = cur.size
    idx = np.arange(n, dtype=np.int64)
    partner_idx = kmax - idx
    valid = (partner_idx >= 0) & (partner_idx < n)

    partner = np.zeros_like(cur)
    partner[valid] = cur[partner_idx[valid]]

    contrib = cur * partner                     # contribution of each i to B_{kmax}
    tot = contrib.sum()
    if tot == 0.0:
        return None

    # choose i proportional to its contribution (large contribution → likely source)
    prob_i = contrib / tot
    i = rng.choice(n, p=prob_i)

    # choose j where the partner value is small (so its product to the centre is tiny)
    inv_w = 1.0 / (partner + 1e-12)
    prob_j = inv_w / inv_w.sum()
    j = rng.choice(n, p=prob_j)

    if i == j:
        return None

    # amount to move – keep it modest and respect bounds
    max_transfer = min(cur[i], 1000.0 - cur[j])
    if max_transfer <= 1e-12:
        return None
    delta = max_transfer * rng.random() * 0.2   # up to 20 % of the possible transfer
    new = cur.copy()
    new[i] -= delta
    new[j] += delta
    return new


def _block_perturb(cur: np.ndarray, rng: np.random.Generator) -> np.ndarray:

    return cand

            raise ValueError
        n = parent.size
        fallback = np.clip(parent, 0.0, 1000.0)
    except Exception:
        # no parent supplied – use a modest default length
        n = 256
        fallback = np.ones(n, dtype=np.float64)

    best = fallback.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # 1️⃣  Quick exploration of a few analytic families (power‑law shapes)
    # ------------------------------------------------------------------
    for beta in (0.5, 0.48, 0.45, 0.55):
        for offs in (0.0, 1.0, 5.0, 10.0, 20.0):
            cand = _make_power_law(n, beta=beta, offs=offs)
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_score:
                best = cand
                best_score = sc

    # Work from the current best; keep a separate mutable copy.
    cur = best.copy()

    # ------------------------------------------------------------------
    # 2️⃣  Guided local search – repeatedly try cheap moves.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # current convolution and its dominant index
        conv_cur = np.convolve(cur, cur)
        kmax = int(np.argmax(conv_cur))

        # decide which move to attempt
        if rng.random() < 0.7:                         # 70 % targeted transfer
            cand = _targeted_transfer(cur, kmax, rng)
            if cand is None:
                continue
        else:                                          # 30 % random block perturbation
            cand = _block_perturb(cur, rng)

        sc = _score(cand)
        # accept only improving moves (greedy descent)
        if sc < best_score:
            best = cand
            best_score = sc
            cur = cand
        # else keep `cur` unchanged – we keep searching from the best seen

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
