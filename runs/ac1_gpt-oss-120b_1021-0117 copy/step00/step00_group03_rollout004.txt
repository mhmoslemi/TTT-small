def max_convolution(a):
    n = a.shape[0]
    size = 2 * n - 1
    # compute via FFT with size next_pow2?
    m = 1 << (size - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:size]
    return conv.max()
