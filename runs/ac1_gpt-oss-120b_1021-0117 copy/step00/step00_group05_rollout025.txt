def build_sparse_positions(n, target_m, max_collisions=2, rng=None):
    pos=[]
    sum_counts = {}
    # Optionally random shuffle candidate indices to avoid deterministic early blockage.
    candidates = list(range(n))
    if rng:
        rng.shuffle(candidates)
    for x in candidates:
        # compute potential sums
        ok=True
        temp=[]
        for y in pos:
            s = x + y
            cnt = sum_counts.get(s, 0) + 1
            if cnt > max_collisions:
                ok=False
                break
            temp.append(s)
        # also self sum
        s_self = x + x
        cnt_self = sum_counts.get(s_self, 0) + 1
        if cnt_self > max_collisions:
            ok=False
        if ok:
            # accept x
            pos.append(x)
            for s in temp:
                sum_counts[s]=sum_counts.get(s,0)+1
            sum_counts[s_self]=sum_counts.get(s_self,0)+1
            if len(pos) >= target_m:
                break
    return pos, sum_counts

import numpy as np
import time

def _build_positions(n: int, max_per_sum: int, rng: np.random.Generator):
    """Greedy construction of a set of indices S ⊂ [0,n) such that each
    unordered pair (i,j) with i≤j contributes at most ``max_per_sum`` to the
    sum i+j.  Returns the list of positions and the array of pair‑counts."""
    # count of unordered pairs (including i=j) that have a given sum
    sum_cnt = np.zeros(2 * n - 1, dtype=np.int16)
    positions = []

    # random order of candidates improves chance of reaching the capacity bound
    cand = np.arange(n, dtype=np.int32)
    rng.shuffle(cand)

    for x in cand:
        # check self‑pair
        s_self = int(x + x)
        if sum_cnt[s_self] + 1 > max_per_sum:
            continue

        # check pairs with already chosen positions
        ok = True
        for y in positions:
            s = int(x + y)
            if sum_cnt[s] + 1 > max_per_sum:
                ok = False
                break
        if not ok:
            continue

        # accept x
        positions.append(int(x))
        sum_cnt[s_self] += 1
        for y in positions[:-1]:          # previous elements only
            sum_cnt[int(x + y)] += 1

    return positions, sum_cnt


def propose_candidate(seed=42, budget_s=1100, **kwargs):

    The routine tries several random greedy constructions and keeps the best.
    """

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Length of the problem – use the provided parent if present.
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).size)
    else:
        n = 256

    best_seq = None
    best_score = float('inf')

    # Try a few different random seeds / parameters within the budget.
    attempts = 0
    while time.monotonic() < deadline and attempts < 30:
        attempts += 1

        # we allow at most 3 unordered pairs per sum (thus ordered count ≤6)
        max_per_sum = 3

        # build a candidate support set
        pos, sum_cnt = _build_positions(n, max_per_sum, rng)

        if len(pos) < 2:            # need at least two points to be meaningful
            continue
