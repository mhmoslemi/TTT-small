import time
import numpy as np

def _geometric_sequence(n: int, delta: float) -> np.ndarray:
    """
    Produce a decreasing geometric sequence a_i = (1 - delta/n)^i  (i=0..n-1).
    The parameter delta>0 controls the decay; delta≈1 gives a good
    trade‑off for the AC1 bound.
    """
    eps = delta / float(n)
    r = 1.0 - eps
    a = np.empty(n, dtype=np.float64)
    cur = 1.0
    for i in range(n):
        a[i] = cur
        cur *= r
    return a

                if np.isfinite(sc2) and sc2 < best_score:
                    best = cand
                    best_score = sc2

    # ------------------------------------------------------------------
    # 3.  Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    return [float(x) for x in best]
```
