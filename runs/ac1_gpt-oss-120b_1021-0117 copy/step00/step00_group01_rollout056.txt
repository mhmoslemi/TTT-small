        n = 256

    if best.sum() < 0.01:
        best = np.ones_like(best)

    # ------------------------------------------------------------------
    # Keep the current best sequence and its score
    # ------------------------------------------------------------------
    best_score = float(evaluate_sequence(best.tolist()))

    rng = np.random.default_rng(seed)
