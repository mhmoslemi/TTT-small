import time
import numpy as np

def _conv_score(seq: np.ndarray, n: int) -> float:
    """Return the exact AC1 upper bound for a non‑negative sequence."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    # convolution (full) and max
    max_b = np.convolve(seq, seq).max()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _candidate_from_params(n: int, idx: np.ndarray,
                          kind: str, p1: float, p2: float = 0.0) -> np.ndarray:
    """Generate a candidate array from a few simple parametric families."""
    if kind == "exp":
        a = np.exp(-p1 * idx / n)                     # p1 = c
    elif kind == "exp_off":
        a = p2 + (1.0 - p2) * np.exp(-p1 * idx / n)   # p1 = c, p2 = w
    elif kind == "pow":
        a = ((n - idx) / n) ** p1                     # p1 = p
    elif kind == "pow_off":
        a = p2 + (1.0 - p2) * ((n - idx) / n) ** p1   # p1 = p, p2 = w
    else:
        a = np.ones(n, dtype=np.float64)
    # keep in allowed range (values are ≤1, so only lower‑bound needed)
    np.clip(a, 0.0, 1000.0, out=a)
    return a

def _block_perturb(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Small O(n) random perturbation used for local refinement."""
    cand = base.copy()
    n = cand.size
    # pick a random block width (1‑%‑10 % of n)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional smoothing by mirroring
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # global jitter
    if rng.random() < 0.2:
        cand *= float(np.exp(rng.normal(0.0, 0.02)))
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a low‑score step function without allocating dense matrices."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- initial parent (if supplied) ------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = int(parent.size)
    idx = np.arange(n, dtype=np.float64)

    best_seq = parent.copy()
    best_score = _conv_score(best_seq, n)

    # ----- parametric sweep -------------------------------------------------
    # exponential family
    for c in np.linspace(0.0, 5.0, 51):
        if time.monotonic() >= deadline:
            break
        cand = _candidate_from_params(n, idx, "exp", c)
        sc = _conv_score(cand, n)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # exponential + offset
    for c in np.linspace(0.2, 4.0, 31):
        for w in np.linspace(0.0, 0.5, 6):
            if time.monotonic() >= deadline:
                break
            cand = _candidate_from_params(n, idx, "exp_off", c, w)
            sc = _conv_score(cand, n)
            if sc < best_score:
                best_seq, best_score = cand, sc

    # power‑law decreasing
    for p in np.linspace(0.0, 5.0, 51):
        if time.monotonic() >= deadline:
            break
        cand = _candidate_from_params(n, idx, "pow", p)
        sc = _conv_score(cand, n)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # power‑law with offset
    for p in np.linspace(0.2, 4.0, 31):
        for w in np.linspace(0.0, 0.5, 6):
            if time.monotonic() >= deadline:
                break
            cand = _candidate_from_params(n, idx, "pow_off", p, w)
            sc = _conv_score(cand, n)
            if sc < best_score:
                best_seq, best_score = cand, sc

    # ----- greedy local refinement -----------------------------------------
    while time.monotonic() < deadline:
        cand = _block_perturb(best_seq, rng)
        sc = _conv_score(cand, n)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # Return a plain list of Python floats
    return [float(x) for x in best_seq]
