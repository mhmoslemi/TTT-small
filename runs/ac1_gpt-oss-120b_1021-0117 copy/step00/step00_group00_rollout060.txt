def _build_sidon_even(n, target_k, rng):
    """
    Build a Sidon set of indices (all even) inside [0, n).
    The returned indices have the property that all pairwise sums are distinct.
    """
    # Work in the reduced even lattice: positions i -> actual 2*i.
    n_even = (n + 1) // 2        # number of even slots (0,2,4,...)
    max_sum = 2 * n_even - 2     # max possible sum on the reduced lattice
    used = np.zeros(max_sum + 1, dtype=bool)
    half_positions = []

    # random order to increase chance of hitting the target size quickly
    order = np.arange(n_even, dtype=np.int64)
    rng.shuffle(order)

    for i in order:
        if used[2 * i]:
            continue
        conflict = False
        for p in half_positions:
            if used[i + p]:
                conflict = True
                break
        if conflict:
            continue

            used[i + p] = True
        if len(half_positions) >= target_k:

    return [2 * i for i in half_positions]
