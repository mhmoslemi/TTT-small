    import numpy as np, time, math, random
    deadline = time.monotonic() + max(0., float(budget_s) - 1.0)
    rng = np.random.default_rng(seed)
    # base length n
    try:
        n = int(len(height_sequence_1))
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if base.ndim != 1 or n == 0:
            raise Exception
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # test exponential family
    for alpha in np.linspace(0.2, 2.5, 24):  # 24 steps
        cand = np.exp(-alpha * np.arange(n) / n)
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best = cand
            best_score = score
    # test power law family
    for p in np.linspace(0.1, 3.0, 30):
        idx = np.arange(1, n+1)
        cand = idx.astype(np.float64) ** (-p)
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best = cand
            best_score = score
    # test linear ramp
    linear = np.linspace(0, 1, n)  # start 0, end 1
    cand = linear
    score = float(evaluate_sequence(cand.tolist()))
    if score < best_score:
        best = cand
        best_score = score

    # Now perform simple stochastic local search (perturbation) to refine
    # Use make_block_perturbation similar to earlier but can also random multiplicative noise.
    while time.monotonic() < deadline:
        # propose perturbation of best
        width = int(rng.integers(max(1, n // 500), max(2, n // 20)))
        start = int(rng.integers(0, max(0, n - width + 1)))
        candidate = best.copy()
        factor = float(math.exp(rng.normal(0.0, 0.05)))  # small relative change
        candidate[start:start+width] *= factor
        # optionally apply small global random scaling? But scaling changes sum, ratio unaffected? Actually scaling changes both sum and max quadratically; ratio unchanged if scaling factor multiplies all entries; but we only multiply part.
        # clip
        np.clip(candidate, 0.0, 1000.0, out=candidate)
        # optionally symmetry
        if rng.random() < 0.3:
            candidate = 0.5*(candidate + candidate[::-1])
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
    return [float(x) for x in best]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0., float(budget_s) - 0.5)
    try:
        n = len(height_sequence_1)
        base_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        n = 256
        base_seq = np.ones(n, dtype=np.float64)

    best = None
    best_score = float('inf')
    # candidates: linear increasing and decreasing
    scale = 1000.0 / n
    idx = np.arange(1, n+1, dtype=np.float64)  # 1..n
    inc = idx * scale
    dec = inc[::-1]

    for cand in (inc, dec, base_seq):
        # ensure within range
        c = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(c.tolist()))
        if np.isfinite(score) and score < best_score:
            best = c
            best_score = score

    # simple local search using block perturbations
    while time.monotonic() < deadline:
        # random block width
        width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        cand = best.copy()
        factor = float(math.exp(rng.normal(0.0, 0.05)))   # near 1 scaling
        cand[start:start+width] *= factor
        # optional symmetrize randomly
        if rng.random() < 0.2:
            cand = 0.5 * (cand + cand[::-1])
        np.clip(cand, 0.0, 1000.0, out=cand)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    return [float(x) for x in best]

import time
import math
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a step‑function minimizing the AC1 upper bound."""
    rng = np.random.default_rng(seed)
    # allow a small safety margin before the sandbox kills the process
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # length of the problem (height_sequence_1 is supplied by the harness)
    try:
        n = int(len(height_sequence_1))
        base = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
        )
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)

    best = None
    best_score = float("inf")

    # ---------- deterministic constructions ----------
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1 … n

    # linear ramp (the champion)
    linear = idx * (1000.0 / n)
    candidates = [linear]

    # decreasing ramp (mirror of linear)
    candidates.append(linear[::-1])

    # sqrt ramp
    sqrt_ramp = np.sqrt(idx) * (1000.0 / np.sqrt(n))
    candidates.append(sqrt_ramp)

    # quadratic ramp
    quad_ramp = (idx ** 2) * (1000.0 / (n ** 2))
    candidates.append(quad_ramp)

    # cubic ramp (optional, may worsen but cheap to try)
    cubic_ramp = (idx ** 3) * (1000.0 / (n ** 3))
    candidates.append(cubic_ramp)

    # original parent if it looks promising
    candidates.append(base)

    for cand in candidates:
        seq = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best = seq.copy()
            best_score = score

    # ---------- cheap stochastic refinement ----------
    while time.monotonic() < deadline:
        # choose a random block and scale it slightly
        width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        cand = best.copy()
        factor = math.exp(rng.normal(0.0, 0.06))   # modest multiplicative tweak
        cand[start:start + width] *= factor

        # occasional mild symmetrisation (helps keep shape balanced)
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])

        np.clip(cand, 0.0, 1000.0, out=cand)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # fallback – if nothing succeeded, return the linear ramp
    if best is None:
        best = linear

    return [float(x) for x in best]
