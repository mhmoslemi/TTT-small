    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    # Determine length
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size

        n = 256
        base = np.ones(n, dtype=np.float64)
    # Start from uniform sequence (all ones)
    best = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    # fallback to parent if better initially
    if np.isfinite(best_score):
        # also test parent
        parent = np.clip(base, 0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent.tolist()))
        if parent_score < best_score:
            best = parent
            best_score = parent_score
    # Simple local moves
    while time.monotonic() < deadline:
        # random move: transfer mass between two positions
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        cur_i = best[i]
        if cur_i <= 0.0:
            continue
        delta = float(rng.random() * cur_i)  # move up to all mass at i
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # clip to allowed range
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
    return [float(x) for x in best]

import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound.

    The routine starts from a uniform sequence (which achieves the
    theoretical bound of 2) and performs cheap random mass‑transfer moves
    while time permits, keeping the best candidate seen.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the sandbox termination
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Length of the problem – use the provided parent if present
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:               # no parent supplied
        n = 256
        base = np.ones(n, dtype=np.float64)

    # Start from a uniform vector (score = 2) – this is optimal for
    # non‑negative sequences under the given metric.
    best = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # If the supplied parent is already better, adopt it.
    parent = np.clip(base, 0.0, 1000.0)
    parent_score = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(parent_score) and parent_score < best_score:
        best = parent
        best_score = parent_score

    # Simple stochastic local search: move a random amount from one index
    # to another while preserving total sum.  The move is accepted only
    # if it strictly improves the AC1 bound.
    while time.monotonic() < deadline:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        cur_i = best[i]
        if cur_i <= 0.0:
            continue
        # transfer a random fraction of the mass at i to j
        delta = float(rng.random() * cur_i)
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # enforce evaluator’s hard clipping
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # Return a plain Python list of floats as required by the harness.
    return [float(x) for x in best]
