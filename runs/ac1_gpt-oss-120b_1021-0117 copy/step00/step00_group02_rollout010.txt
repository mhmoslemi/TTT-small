import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step‑function that yields a very low AC1 upper bound.
    The shape a_i = 1000 * (i/(n-1))^p with p=6 concentrates mass near the end,
    driving max_{k} (a*a)[k] down while keeping the total sum large.
    """
    # Length of the problem (use the provided parent if present)
    try:
        n = int(len(height_sequence_1))
        if n <= 0:
            raise ValueError
    except Exception:
        n = 256

    # Exponent – larger values give smaller bounds (p=6 is ample)
    p = 6

    # Normalised positions in [0,1]
    if n == 1:
        t = np.array([1.0], dtype=np.float64)
    else:
        t = np.arange(n, dtype=np.float64) / (n - 1)

    # Generate the heights, enforce the allowed range
    a = 1000.0 * np.power(t, p)
    a = np.clip(a, 0.0, 1000.0)

    # Safety: guarantee a non‑zero sum (the evaluator requires sum >= 0.01)
    if a.sum() < 0.01:
        a = np.full(n, 0.01 / n, dtype=np.float64)

    # Return as plain Python list of floats
    return [float(v) for v in a]
