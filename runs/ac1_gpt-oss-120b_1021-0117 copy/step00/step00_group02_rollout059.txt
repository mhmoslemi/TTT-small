def fast_score(a):
    n = len(a)
    # Compute convolution using FFT:
    nfft = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, nfft)
    fb = fa * fa
    conv = np.fft.irfft(fb, nfft)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2 * n * max_b / (sum_a**2)

import numpy as np
import time

def _fast_score(seq: np.ndarray) -> float:
    """
    Compute the AC1 upper–bound score using an FFT‑based convolution.
    This mirrors evaluate_sequence but avoids the O(n²) np.convolve.
    """
    n = seq.size
    # next power of two for a linear convolution
    nfft = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, nfft)
    conv = np.fft.irfft(f * f, nfft)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _build_sidon_set(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Greedy construction of a Sidon set (all sums i+j are distinct).
    Returns the indices of the selected positions.
    """
    # random order of candidate indices
    order = rng.permutation(n)
    # boolean array of length 2n-1 recording which sums have appeared
    sums_seen = np.zeros(2 * n - 1, dtype=np.bool_)
    chosen = []
    # iterate over the random permutation
    for i in order:
        # quick reject if diagonal sum already used
        if sums_seen[2 * i]:
            continue
        # test all existing sums i + j
        ok = True
        for j in chosen:
            s = i + j
            if sums_seen[s]:
                ok = False
                break
        if not ok:
            continue
        # accept i
        for j in chosen:
            sums_seen[i + j] = True
        sums_seen[2 * i] = True
        chosen.append(i)
    return np.asarray(chosen, dtype=np.int32)
