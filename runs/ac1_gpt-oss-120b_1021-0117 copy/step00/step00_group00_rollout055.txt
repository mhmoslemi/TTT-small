            if (i + i) in used:

import numpy as np
import time

def _greedy_sidon(n, order):
    """Return a Sidon set (distinct pairwise sums) constructed in the given order."""
    used = np.zeros(2 * n, dtype=bool)          # sums 0 .. 2n-2
    S = []
    for i in order:
        # test collisions with already chosen elements
        bad = used[i + i]
        if not bad:
            for j in S:
                if used[i + j]:
                    bad = True
                    break
        if bad:
            continue
        # accept i and mark its sums
        for j in S:
            used[i + j] = True
        used[i + i] = True
        S.append(i)
    return S

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Construct a non‑negative step function whose AC1 upper bound is
    better than the trivial uniform bound (score < 2.0).
    The method builds a (large) Sidon set – a set of indices whose pairwise
    sums are all distinct – and puts equal mass on those positions.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Length of the problem – use the provided parent if possible
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            n = 256
        else:
            n = int(base.size)
    except NameError:
        n = 256

    # Uniform distribution (score = 2.0) as a safe fallback
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best = uniform
    best_score = float(evaluate_sequence(best.tolist()))

    # Try a few deterministic orders first (forward and reverse)
    for deterministic_order in (np.arange(n), np.arange(n - 1, -1, -1)):
        S = _greedy_sidon(n, deterministic_order)
        if len(S) < 2:
            continue
        w = 1.0 / len(S)
        cand = np.zeros(n, dtype=np.float64)
        cand[S] = w
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc

    # Randomised greedy attempts while time permits
    attempts = 0
    while time.monotonic() < deadline and attempts < 500:
        attempts += 1
        order = rng.permutation(n)
        S = _greedy_sidon(n, order)
