# ---------- basic helpers -------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex (non‑negative, sum = 1)."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (scale‑invariant)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ---------- families for a quick warm‑start -------------------------------
def _family_exponential(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def _family_powerlaw(n, p):
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    return a / a.sum()


def _family_linear(n):
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _family_quadratic(n):
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()


def _family_cosine(n, p):
    x = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64)) / (n + 1))
    a = np.power(x, p)
    return a / a.sum()


def _family_double_end(n, p):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** -p) + ((n - i + 1) ** -p)
    return a / a.sum()


# ---------- main search ---------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ----- initial vector (parent if present, otherwise uniform) ----------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    else:
        n = 256
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur.size
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # ----- explore a few compact families ----------------------------------
    for r in np.linspace(0.90, 0.9999, 25):
        cand = _family_exponential(n, r)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    for p in np.linspace(0.2, 5.0, 30):
        cand = _family_powerlaw(n, p)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    cand = _family_linear(n)
    sc = _score(cand)
    if sc < best_sc:
        best_sc, best_vec = sc, cand.copy()
    cand = _family_quadratic(n)
    sc = _score(cand)
    if sc < best_sc:
        best_sc, best_vec = sc, cand.copy()
    for p in np.linspace(0.5, 4.0, 20):
        cand = _family_cosine(n, p)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    for p in np.linspace(0.2, 4.0, 20):
        cand = _family_double_end(n, p)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    # ----- optimisation loop ---------------------------------------------
    it = 0
    eta0 = 0.8                      # exponentiated‑gradient base step
    alpha0 = 0.6                    # Euclidean‑projection base step
    target = 1.50274349

    # temperature for occasional uphill moves (simulated‑annealing style)
    T0 = 0.1 * cur_sc

    while time.monotonic() < deadline:
        it += 1
        # current full convolution and the worst lag
        conv = _conv_fft(cur)
        k = int(np.argmax(conv))

        # sub‑gradient for the active lag
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]

        # ----- candidate 1 : exponentiated gradient (mirror descent) -----
        eta = eta0 / math.sqrt(it)
        cand1 = cur * np.exp(-eta * g)
        cs = cand1.sum()
        if cs > 0:
            cand1 /= cs
        else:
            cand1 = np.full_like(cur, 1.0 / n)

        sc1 = _score(cand1)

        # ----- candidate 2 : Euclidean projection sub‑gradient -------------
        alpha = alpha0 / math.sqrt(it)
        cand2 = _project_simplex(cur - alpha * g)
        sc2 = _score(cand2)

        # ----- decide which candidate to keep ----------------------------
        # Prefer the better of the two; allow occasional uphill step
        if sc1 < cur_sc or (sc2 < cur_sc):
            # take the better improving candidate
            if sc1 <= sc2:
                cur, cur_sc = cand1, sc1
            else:
                cur, cur_sc = cand2, sc2
        else:
            # no improvement – probabilistic uphill move (annealing)
            T = T0 * (1.0 - (time.monotonic() - start) / float(budget_s))
            if T > 0 and rng.random() < math.exp(-(min(sc1, sc2) - cur_sc) / T):
                if sc1 <= sc2:
                    cur, cur_sc = cand1, sc1
                else:
                    cur, cur_sc = cand2, sc2

        # update global best
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            if best_sc < target:
                break

        # ----- occasional random perturbations ----------------------------
        if it % 250 == 0:
            # tiny Gaussian jitter
            jitter = rng.normal(scale=5e-5, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            s = tmp.sum()
            if s > 0:
                tmp /= s
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                        if best_sc < target:
                            break

            # directed mass‑transfer move
            nz = np.where(cur > 1e-12)[0]
            if nz.size:
                i = int(rng.choice(nz))
                j = int(rng.integers(0, n - 1))
                if j >= i:
                    j += 1
                max_delta = cur[i] * 0.5
                if max_delta > 1e-12:
                    delta = max_delta * rng.random()
                    cand = cur.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand_sc = _score(cand)
                    if cand_sc < cur_sc:
                        cur, cur_sc = cand, cand_sc
                        if cand_sc < best_sc:
                            best_sc, best_vec = cand_sc, cand.copy()
                            if best_sc < target:
                                break

    # ----- final safeguard -------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
