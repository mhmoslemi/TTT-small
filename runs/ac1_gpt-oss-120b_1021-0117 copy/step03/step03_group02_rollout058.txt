import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x (for FFT padding)."""
    return 1 << ((x - 1).bit_length()) if x > 1 else 1


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x >= 0, sum(x)=1}.
    Returns a new array (the original is not modified).
    """
    n = v.size
    if n == 0:
        return v
    # sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # due to numerical errors, renormalise exactly
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w[:] = 1.0 / n
    return w


def _sqrt_series(n: int) -> np.ndarray:
    """Fallback analytic initialization (slowly decreasing positive series)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """
    Compute the linear convolution of real vectors x and y using FFT of length L.
    Returns an array of length L (the result is zero‑padded to L).
    """
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    return np.fft.irfft(fx * fy, n=L)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function heights vector that
    minimizes 2 * n * max(conv(a,a)) / (sum(a)^2).
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # 5 s safety margin

    # ------------ initialise -------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = _sqrt_series(256)

    a = _project_simplex(a)          # normalise to sum = 1
    n = a.size

    # FFT lengths (fixed for the whole run)
    L_conv = _next_pow2(2 * n - 1)                 # for a*a
    L_grad = _next_pow2(3 * n - 2)                 # for w*a (worst case)

    # ------------ evaluation utilities ---------------------------------------
    def _score(vec: np.ndarray) -> float:
        """Direct score (sum(vec) is 1 after projection)."""
        conv_full = _conv_fft(vec, vec, L_conv)[: 2 * n - 1]
        max_val = conv_full.max()
        return 2.0 * n * max_val

    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score

    # ------------ optimisation parameters ------------------------------------
    step = 0.25                     # initial learning rate
    min_step = 1e-6
    decay = 0.9995
    alpha = 12.0                    # soft‑max sharpness
    momentum_coeff = 0.9
    mom = np.zeros_like(a)          # momentum buffer
    no_improve = 0
    max_no_improve = 3000
    prob_perturb = 0.02

    # ------------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current convolution and soft‑max weights -----------------------
        conv_full = _conv_fft(a, a, L_conv)[: 2 * n - 1]
        max_val = conv_full.max()
        # soft‑max over *all* lags (more accurate sub‑gradient)
        diff = conv_full - max_val                # ≤ 0
        w_raw = np.exp(alpha * diff)              # in (0, 1]
        w_sum = w_raw.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # ----- gradient of the weighted max (2 * w * a) --------------------
        grad_full = _conv_fft(w, a, L_grad)[: 3 * n - 2]
        # gradient on the first n entries (see derivation)
        grad = 2.0 * grad_full[:n]

        # ----- momentum step + projection -----------------------------------
        mom = momentum_coeff * mom - step * grad
        cand = a + mom
        cand = _project_simplex(cand)

        # ----- evaluate candidate -------------------------------------------
        cand_score = _score(cand)

        improved = False
        if cand_score < cur_score - 1e-14:
            a = cand
            cur_score = cand_score
            mom = np.zeros_like(a)       # reset momentum after a successful move
            improved = True
        else:
            # occasional uphill move (simulated annealing flavour)
            temp = max(step, 1e-4)
            if rng.random() < np.exp(-(cand_score - cur_score) / temp):
                a = cand
                cur_score = cand_score
                improved = True

        # ----- keep global best ---------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            no_improve = 0
        else:
            if not improved:
                no_improve += 1
            else:
                no_improve = 0

        # ----- random mass‑transfer perturbation -----------------------------
        if rng.random() < prob_perturb:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.3
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                cand2_score = _score(cand2)
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()
                        no_improve = 0

        # ----- step‑size decay -----------------------------------------------
        step = max(min_step, step * decay)

        # ----- restart if stuck -----------------------------------------------
        if no_improve >= max_no_improve:
            a = rng.random(n)
            a = _project_simplex(a)
            mom = np.zeros_like(a)
            cur_score = _score(a)
            no_improve = 0

        # ----- early‑exit safeguard -------------------------------------------
        if time.monotonic() + 0.5 > deadline:
            break

    # Return the best found vector as plain Python list
    return [float(v) for v in best]
