import time, math, numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a * a using FFT length L."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    return conv


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    rng = np.random.default_rng(seed)
