    n = a.shape[0]

    # define conv function using preallocated buffer? simply compute each time

    def conv_fft(v):
        fv = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2*n - 1]
        return conv

        conv = conv_fft(v)
        return 2.0 * n * conv.max()

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum x = 1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]

                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cand[i] * rng.random() * 0.4
                cand[i] -= delta
                cand[j] += delta

        # --------------------------------------------------------------
        # Project back onto the simplex (keep sum = 1, non‑negative)
        # --------------------------------------------------------------
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s <= 0.0:
            continue
        cand /= s

        cand_sc = _score(cand)

        # --------------------------------------------------------------
        # Acceptance rule (greedy + occasional SA step)
        # --------------------------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            cur_conv = _conv_fft(cur)
            cur_kmax = int(cur_conv.argmax())
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            # simulated‑annealing style acceptance
            elapsed = time.time() - start
            frac = elapsed / (deadline - start) if deadline > start else 1.0
            T = max(1e-7, (1.0 - frac) * 0.01)   # temperature decreasing linearly
            if rng.random() < math.exp(-(cand_sc - cur_sc) / T):
                cur, cur_sc = cand, cand_sc
                cur_conv = _conv_fft(cur)
                cur_kmax = int(cur_conv.argmax())

        # --------------------------------------------------------------
        # Periodic projected‑gradient refinement (sub‑gradient of max‑conv)
        # --------------------------------------------------------------
        if it % grad_interval == 0:
            K = 30                                          # size of active set
            if K >= cur_conv.size:
                active = np.arange(cur_conv.size)
            else:
                active = np.argpartition(-cur_conv, K - 1)[:K]

            g = np.zeros_like(cur)
            for k in active:
                lo = max(0, k - (n - 1))
                hi = min(k, n - 1)
                idx = np.arange(lo, hi + 1, dtype=np.int64)
                g[idx] += 2.0 * cur[k - idx]               # sub‑gradient of (a*a)_k

            alpha = step0 / np.sqrt(it)
            cand_g = cur - alpha * g
            cand_g = np.maximum(cand_g, 0.0)
            s_g = cand_g.sum()
            if s_g > 0.0:
                cand_g /= s_g
                cand_g_sc = _score(cand_g)
                if cand_g_sc < cur_sc:
                    cur, cur_sc = cand_g, cand_g_sc
                    cur_conv = _conv_fft(cur)
                    cur_kmax = int(cur_conv.argmax())
                    if cand_g_sc < best_sc:
                        best_sc, best_vec = cand_g_sc, cand_g.copy()

        # --------------------------------------------------------------
        # Early exit if target already reached
        # --------------------------------------------------------------
        if best_sc < target_score:
            break

    # ------------------------------------------------------------------
    # Final quick local refinement (tiny random swaps) while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * rng.random() * 0.3
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = np.maximum(tmp, 0.0)
        tmp_sum = tmp.sum()
        if tmp_sum == 0.0:
            continue
        tmp /= tmp_sum
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            cur_conv = _conv_fft(cur)
            cur_kmax = int(cur_conv.argmax())
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
