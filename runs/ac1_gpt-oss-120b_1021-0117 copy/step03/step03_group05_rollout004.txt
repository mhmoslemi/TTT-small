import time

def _conv_fft(v: np.ndarray) -> np.ndarray:
    """Linear convolution of v with itself using FFT."""
    n = v.size
    L = 1 << ((2 * n - 1).bit_length())      # next power of 2
    fv = np.fft.rfft(v, n=L)
    conv_full = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv_full


def _score(v: np.ndarray) -> float:
    """AC1 upper bound for a non‑negative vector (sum may be ≠1)."""
    conv = _conv_fft(v)
    n = v.size
    return 2.0 * n * conv.max() / (v.sum() ** 2)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1}."""
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, u.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
