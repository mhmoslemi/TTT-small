Implementation of simplex projection:

```
def project_simplex(v):
    # based on Duchi et al.
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u > css / np.arange(1, len(v)+1))[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w
```

import time
import numpy as np

def _linear_conv(v: np.ndarray) -> np.ndarray:
    """Full linear convolution v*v using a power‑of‑two FFT."""
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


def _score(v: np.ndarray) -> float:
    """Exact AC1 bound for a non‑negative vector."""
    n = v.shape[0]
    conv = _linear_conv(v)
    return 2.0 * n * conv.max() / (v.sum() ** 2)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u > css / np.arange(1, v.size + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_from_active(v: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Sub‑gradient built from the active convolution indices."""
    n = v.shape[0]
    g = np.zeros_like(v)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * v[k - idx]
    return g

    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The routine respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)

    # keep a small safety margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------ #

    # ------------------------------------------------------------------ #
    try:

        init = np.full(n0, 1.0 / n0, dtype=np.float64)

    init = np.maximum(init, 0.0)
    s = init.sum()
    if s <= 0.0:
        cur = np.full_like(init, 1.0 / init.size)
    else:
        cur = init / s                     # enforce sum = 1

    n = cur.size
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # ------------------------------------------------------------------ #
    # optimisation parameters
    # ------------------------------------------------------------------ #
    it = 0
    no_improve = 0
    K = 12                     # active convolution entries

    target = 1.50274349
