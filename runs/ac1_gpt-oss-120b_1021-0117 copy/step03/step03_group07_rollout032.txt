        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]                 # sub‑gradient of active lag

        eta = 0.6 / np.sqrt(it)                     # diminishing step size
        cand = cur * np.exp(-eta * g)               # exponentiated‑gradient step
        s = cand.sum()
        cand = cand / s if s > 0 else np.full_like(cand, 1.0 / n)

        cand_sc = _score_vec(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        # early exit when the bound is already well below the target
        if best_sc < 1.34:
            break

    # -------------------- final safety normalisation -------------------------
    best_vec = np.maximum(best_vec, 0.0)
    tot = best_vec.sum()
    if tot < 0.01:                                 # avoid the evaluator's inf case
        best_vec = np.full_like(best_vec, 1.0 / best_vec.size)
    else:
        best_vec = best_vec / tot

    return [float(v) for v in best_vec]
