import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Heuristic search for a sequence that minimizes the AC1 upper bound."""
    rng = np.random.default_rng(seed)

    # ---------- time management ----------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ---------- initial sequence ----------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = parent.copy()
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s  # normalise to unit sum (simplex)

    # ---------- best-so-far ----------
    best = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(best, best))

    # uniform fallback (often worse but cheap)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = 2.0 * n * np.max(np.convolve(uniform, uniform))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # ---------- FFT preparation ----------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ---------- optimisation parameters ----------
    eta0 = 0.8          # initial step size
    beta = 0.9          # momentum coefficient
    jitter_every = 150  # steps
    jitter_std = 0.02
    target_score = 1.50274349
    max_stale = 2500    # steps without improvement before restart

    v = np.zeros_like(a)      # momentum buffer
    it = 0
    stale = 0

    while time.monotonic() < deadline:
        # ----- convolution (FFT) -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # ----- update best -----
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            stale = 0
            if best_score <= target_score:
                break
        else:
            stale += 1

        # ----- exact sub‑gradient at all maximal points -----
        # indices where convolution attains its maximum
        idx_max = np.flatnonzero(conv == max_val)
        g = np.zeros_like(a)
        for k in idx_max:
            start = max(0, k - (n - 1))
            end = min(n - 1, k)
            i = np.arange(start, end + 1, dtype=np.int64)
            j = k - i
            # each contribution is 2 * a_j
            g[i] += 2.0 * a[j]
        # average if multiple maximisers
        if len(idx_max) > 1:
            g /= float(len(idx_max))

        # ----- momentum step & projection -----
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g
        a = _proj_simplex(a - eta * v)

        # ----- occasional jitter to escape plateaus -----
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v.fill(0.0)
            stale = 0

        # ----- restart after prolonged stagnation -----
        if stale >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.03, size=n))
            a = _proj_simplex(a * perturb)
            v.fill(0.0)
            stale = 0

    # ---------- return best found ----------
    return [float(x) for x in best]
