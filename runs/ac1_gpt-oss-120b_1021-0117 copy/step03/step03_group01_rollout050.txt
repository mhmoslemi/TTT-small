import time
import numpy as np

# --------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, ≥0)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # safeguard against all‑zero (should not happen)
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w

# --------------------------------------------------------------
# Analytic square‑root series (a*a = 1 + … + x^{2n‑2})
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = _sqrt_series(256)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)          # enforce sum = 1, non‑negative
    n = a.size

    # ----- FFT size (must accommodate 3n‑2) -------------------------
    L = 1 << ((3 * n - 2).bit_length())
    slice_start = n - 1               # start index for gradient extraction
    corr_len = 3 * n - 2               # full length of w * a_rev convolution

    # ----- optimisation hyper‑parameters -----------------------------
    alpha = 5.0                        # smoothing for max‑approximation
    lr = 0.12
    decay_lr = 0.9995
    min_lr = 1e-5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    prob_random = 0.007                # occasional random mass transfer

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    no_imp = 0

    # ----- evaluate initial candidate (exact) -----------------------
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())

    # ----- main optimisation loop ------------------------------------
    while time.monotonic() < deadline:
        t += 1
        # gradually tighten the smooth max
        if t % 250 == 0 and alpha < 30.0:
            alpha *= 1.2

        # ----- forward FFT of current a -----------------------------
        fa = np.fft.rfft(a, n=L)

        # ----- full auto‑convolution (a*a) -------------------------
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]

        # ----- smooth max (log‑sum‑exp) ----------------------------
        maxv = conv.max()
        exp_vals = np.exp(alpha * (conv - maxv))
        w = exp_vals / exp_vals.sum()
        S = np.dot(w, conv)                     # = log‑sum‑exp / α

        # ----- gradient of the smooth max ---------------------------
        # reverse a for correlation
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, n=L)
        fw = np.fft.rfft(w, n=L)

        corr = np.fft.irfft(fw * np.conj(fa_rev), n=L)[:corr_len]
        grad_S = 2.0 * corr[slice_start : slice_start + n]   # dS/da

        # ----- gradient of the (scale‑invariant) objective -----------
        # with sum(a)=1 the objective is 2*n*S
        grad = 2.0 * n * grad_S

        # ----- Adam update -----------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        # ----- occasional random mass transfer ----------------------
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a_candidate[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.3
                delta = a_candidate[i] * frac
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = _project_simplex(a_candidate)

        # ----- compute exact score for the candidate -----------------
        fa_c = np.fft.rfft(a_candidate, n=L)
        conv_c = np.fft.irfft(fa_c * fa_c, n=L)[: 2 * n - 1]
        score_c = 2.0 * n * conv_c.max()   # sum(a_candidate)=1 by projection

        # ----- accept if improvement --------------------------------
        if score_c < best_score - 1e-12:
            best = a_candidate.copy()
            best_score = score_c
            no_imp = 0
        else:

        a = a_candidate

        # ----- learning‑rate decay ----------------------------------
        lr = max(min_lr, lr * decay_lr)

        # ----- early‑exit guard ------------------------------------
        if time.monotonic() + 0.6 > deadline:
            break

    # final verification using the official evaluator (ensures exactness)
    final_score = evaluate_sequence(best.tolist())
    # keep the fallback if the evaluator reports inf (should not happen)
    if final_score == float('inf') or final_score > best_score:
        best = a.copy()
    return [float(x) for x in best]
