def _conv_fft(a, L):
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[:2*a.size-1]

def _score(a, conv, n):
    return 2.0 * n * conv.max() / (a.sum()**2)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    # enforce nonneg and normalize sum=1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0 / a.size)
        s = a.sum()
    a = a / s

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    conv_a = _conv_fft(a, L)
    cur_sc = _score(a, conv_a, n)
    best = a.copy()

    random_eps = 0.25

            no_improve = 0

        if best_sc < 1.50274349:
            break

    # return list
    return [float(x) for x in best]

import time, math, numpy as np

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of x with itself using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)
    return conv[: 2 * x.size - 1]

def _score_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Exact same formula as evaluate_sequence."""
    return 2.0 * n * conv.max() / (sum_a * sum_a)
