import time
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # enforce sum exactly 1 (numerical safety)
    if w.sum() == 0:
        return np.full_like(v, 1.0 / n)
    return w / w.sum()


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Full (linear) convolution a*a using FFT of length L."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return conv


def _subgradient_max(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """Subgradient of max_k (a * a)[k] w.r.t. a (exact, using all active k)."""
    n = a.size
    g = np.zeros_like(a)
    for idx in max_idxs:
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        i_arr = np.arange(lo, hi + 1, dtype=np.int64)
        # each contribution appears twice in the convolution sum
        g[i_arr] += 2.0 * a[idx - i_arr]
    return g
