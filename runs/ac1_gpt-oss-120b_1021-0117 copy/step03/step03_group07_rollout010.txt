def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (size O(L log L))."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (sum not forced to 1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _make_sqrt_series(n: int) -> np.ndarray:
    """Recursive construction from the original baseline."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / s
    return a


def _make_edge_exponential(n: int, r: float) -> np.ndarray:
    """Symmetric exponential decay from the edges (r<1)."""
    d = np.arange(n)
    dist = np.minimum(d, n - 1 - d)
    a = np.power(r, dist, dtype=np.float64)
    a /= a.sum()
    return a

                else:
                    theta = 0.0
                proj_up = np.maximum(v - theta, 0.0)
                cand_up_sc = _score(proj_up)
                if cand_up_sc < cur_sc:
                    cur, cur_sc = proj_up, cand_up_sc
                    if cand_up_sc < best_sc:
                        best_sc, best_vec = cand_up_sc, proj_up.copy()
                        if best_sc < target:
                            break

        if it2 % 2500 == 0 and rng.random() < 0.12:
            # random mass‑transfer move
            nz = np.where(cur > 1e-14)[0]
            if nz.size:
                i = int(rng.choice(nz))
                j = int(rng.integers(0, n))
                if j == i:
                    j = (j + 1) % n
                d = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= d
                tmp[j] += d
                tmp = np.maximum(tmp, 0.0)
                tmp_sum = tmp.sum()
                if tmp_sum > 0:
                    tmp /= tmp_sum
                    sc_tmp = _score(tmp)
                    if sc_tmp < cur_sc:
                        cur, cur_sc = tmp, sc_tmp
                        if sc_tmp < best_sc:
                            best_sc, best_vec = sc_tmp, tmp.copy()
                            if best_sc < target:
                                break

        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 3 – targeted random mass‑transfer moves (exploratory)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cur_conv = _conv_fft(cur)
        k = int(np.argmax(cur_conv))
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = cur[idx] * cur[k - idx]
        if prod.size:
            i = int(idx[np.argmax(prod)])
        else:
            i = int(rng.integers(0, n))
        j = int(np.argmin(cur))
        if j == i:
            j = (j + 1) % n

        max_delta = cur[i] * 0.5
        if max_delta < 1e-12:
            continue
        delta = max_delta * rng.random()
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break

        # rare tiny jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=5e-5, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp_sum = tmp.sum()
            if tmp_sum > 0:
                tmp /= tmp_sum
                sc_tmp = _score(tmp)
                if sc_tmp < cur_sc:
                    cur, cur_sc = tmp, sc_tmp
                    if sc_tmp < best_sc:
                        best_sc, best_vec = sc_tmp, tmp.copy()
                        if best_sc < target:
                            break

    # ------------------------------------------------------------------
    # Fallback sanity check
    # ------------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
