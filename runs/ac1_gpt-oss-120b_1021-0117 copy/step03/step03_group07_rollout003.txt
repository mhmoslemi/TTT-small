import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a single FFT (real input)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (no scaling of the vector)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small margin for the return path
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # -------------------------------------------------------------
    # initialise – use the parent if it exists, otherwise a uniform start
    # -------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.clip(cur, 0.0, None)
        s = cur.sum()
        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    else:
        n = 256
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur.size
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # -------------------------------------------------------------
    # a few analytic families – cheap way to get a good start
    # -------------------------------------------------------------
    # exponential decay
    for r in np.linspace(0.90, 0.9999, 18):
        a = np.power(r, np.arange(n, dtype=np.float64))
        a /= a.sum()
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # power‑law
    for p in np.linspace(0.2, 4.0, 18):
        a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        a /= a.sum()
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # linear and quadratic ramps
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    sc = _score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    a /= a.sum()
    sc = _score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    # -------------------------------------------------------------
    # simple projected sub‑gradient descent on the true objective
    # -------------------------------------------------------------
    step0 = 0.5
    it = 0
    while time.monotonic() < deadline and it < 20000:
        it += 1
        conv = _conv_fft(cur)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution term
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]

        # diminishing step size, Euclidean projection back to simplex
        step = step0 / np.sqrt(it)
        cand = cur - step * g
        cand = _project_simplex(cand)

        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        # occasional random jitter to escape tiny plateaus
        if it % 500 == 0 and rng.random() < 0.15:
            jitter = rng.normal(scale=1e-4, size=n)
            cur = np.clip(cur + jitter, 0.0, None)
            cur /= cur.sum()
            cur_sc = _score(cur)
            if cur_sc < best_sc:
                best_sc, best_vec = cur_sc, cur.copy()

    # -------------------------------------------------------------
    # fallback – if everything failed, return a uniform vector
    # -------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
