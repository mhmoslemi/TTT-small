import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1, dtype=np.float64)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: enforce exact sum==1
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w[0] = 1.0
    return w


# ------------------------------------------------------------
# Main entry point

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ---- initialise -------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: uniform distribution
        a = np.ones(1, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)               # sum(a) == 1

    n = a.size
    # length sufficient for both 2n-1 (auto‑conv) and 3n-2 (grad‑conv)
    L = 1 << ((3 * n - 2).bit_length())

    # ---- evaluate initial candidate ----------------------------------
    A = np.fft.rfft(a, n=L)

    max_val = conv_full.max()
    cur_score = 2.0 * n * max_val
    best = a.copy()

    # ---- optimisation parameters ------------------------------------
