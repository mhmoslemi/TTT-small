import cvxpy as cp
def max_sum_under_conv_bound(f, M):
    n = f.size
    g = cp.Variable(n, nonneg=True)
    conv_fg = cp.conv(f, g)  # returns length 2*n-1
    constraints = [conv_fg <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None:
        return None
    return np.array(g.value).reshape(-1)

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper: simple sqrt series (fallback if no parent sequence is supplied)
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ----------------------------------------------------------------------
# Helper: max of full autocorrelation via FFT (fast, O(n log n))
# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max_{k} (a*a)[k] using FFT of length L (power of two)."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    return float(conv.max())

        total = a.size
    a /= total                       # now sum(a) == 1

    n = a.size
    # FFT length – next power of two for safe convolution
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Keep the best discovered sequence.
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = 2.0 * n * _conv_max(a, L)

    # ------------------------------------------------------------------
    # Phase 1: alternating linear‑programming refinement.
    # ------------------------------------------------------------------
    mix_factor = 0.5                 # initial mixing weight
    while time.monotonic() < deadline:
        # current autocorrelation peak M
        M = _conv_max(a, L)

        # Linear program: maximise sum(g) s.t. a * g ≤ M (component‑wise)
        g = cp.Variable(n, nonneg=True)
        f_const = cp.Parameter(n, value=a, constant=True)
        conv_expr = cp.conv(f_const, g)            # length 2n‑1
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv_expr <= M])
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
        except Exception:
            break
        if g.value is None:
            break
        g_vec = np.array(g.value, dtype=np.float64)
        s_g = g_vec.sum()
        if s_g == 0.0:
            break
        # Rescale to the same simplex (sum = 1)
        g_vec /= s_g

        # Mix the current candidate with the LP direction
        a = (1.0 - mix_factor) * a + mix_factor * g_vec
        a = np.maximum(a, 0.0)
        a /= a.sum()                # renormalise (preserves sum = 1)

        # Evaluate
        cur_score = 2.0 * n * _conv_max(a, L)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        # Decay mixing factor slowly
        mix_factor = max(0.01, mix_factor * 0.95)

        # Time guard
        if time.monotonic() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2: lightweight projected gradient descent on the max‑lag.
    # ------------------------------------------------------------------
    a = best.copy()
    cur_score = best_score
    step = 0.2
    min_step = 1e-6
    decay = 0.9995
    K = 8                     # number of lags to focus on
    alpha = 6.0               # soft‑max sharpness

    while time.monotonic() < deadline:
        # Full autocorrelation via FFT (reuse L)
        Fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # Select K largest lags (or all if fewer)
        if K >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            part = np.argpartition(conv, -K)
            top_idx = part[-K:]

        # Soft‑max weighting over chosen lags
        w_raw = np.exp(alpha * (conv[top_idx] - max_val))
        w_sum = w_raw.sum()
        w = w_raw / w_sum if w_sum > 0 else np.full_like(w_raw, 1.0 / w_raw.size)

        # Gradient of the max‑lag surrogate
        grad = np.zeros_like(a)
        for idx, wi in zip(top_idx, w):
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            if lo <= hi:
                j = np.arange(lo, hi + 1)
                grad[j] += wi * 2.0 * a[idx - j]

        # Take a projected step
        cand = a - step * grad
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0:
            cand /= s

        cand_score = 2.0 * n * _conv_max(cand, L)
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()

        step = max(min_step, step * decay)

        if time.monotonic() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # Return the best sequence discovered as a plain Python list.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
```
