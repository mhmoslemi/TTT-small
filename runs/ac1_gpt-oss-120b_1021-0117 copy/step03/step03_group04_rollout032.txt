import time
import math
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

            best = cand.copy()
    a = best.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # 3. Optimisation parameters
    # ------------------------------------------------------------------
    eta0 = 3.0          # initial step magnitude (larger for broader search)
    mu = 0.95           # momentum decay
    K = 20              # number of top shifts to build a sub‑gradient
    v = np.zeros_like(a)   # momentum buffer
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # 4. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- recompute convolution and current score -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            if best_score <= target:
                break

        # ----- sub‑gradient from the top K convolution peaks -----
        if K < conv_len:
            top_idx = np.argpartition(-conv, K)[:K]
        else:
            top_idx = np.arange(conv_len)
        top_vals = conv[top_idx]
        wsum = top_vals.sum()
        if wsum > 0:
            weights = top_vals / wsum
        else:
            weights = np.full_like(top_vals, 1.0 / top_vals.size)

        g = np.zeros_like(a)
        for shift, w in zip(top_idx, weights):
            offset = shift - (n - 1)
            if offset >= 0:
                idx_range = np.arange(offset, n, dtype=np.int64)
                partner = shift - idx_range
            else:
                idx_range = np.arange(0, n + offset, dtype=np.int64)
                partner = shift - idx_range
            g[idx_range] += w * 2.0 * a[partner]

        # ----- momentum update -----
        v = mu * v + (1.0 - mu) * g

        # ----- diminishing step size -----
        eta = eta0 / np.sqrt(it)

        # ----- projected gradient step -----
        a = _proj_simplex(a - eta * v)

        # ----- occasional stochastic jitter -----
        if it % 500 == 0:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * jitter)

        # ----- occasional reset around the current best -----
        if it % 2000 == 0:
            a = best * np.exp(rng.normal(0.0, 0.01, size=n))
            a = _proj_simplex(a)

    # ------------------------------------------------------------------
    # 5. Return the best sequence as plain Python floats
    # ------------------------------------------------------------------
    return [float(v) for v in best]
