def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Produce a non‑negative step‑function height sequence with a very low
    AC1 upper bound.  The construction uses a simple power‑law shape
    a_i ∝ (i+1)^p with p=5, normalised to unit ℓ¹‑norm.

    # Length of the problem – use the parent if it is present.
