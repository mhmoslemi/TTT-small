import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if ind.size == 0:
        theta = 0.0
    else:
        rho = ind[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(v, 1.0 / v.size)
    else:
        w = w / s
    return w

    """
    Search for a non‑negative step‑function minimizing the AC1 bound.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
