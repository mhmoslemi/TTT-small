# ---------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# ---------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:                # numerical safeguard
        w[:] = 1.0 / n
    return w

# ---------------------------------------------------------
# Analytic sqrt‑series (fallback initializer)
# ---------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ---------------------------------------------------------
# Full convolution via FFT, also returns indices with near‑max value
# ---------------------------------------------------------
def _conv_max_and_indices(vec: np.ndarray, L: int, eps: float = 1e-12):
    """Return (max_value, array_of_indices) of the self‑convolution."""
    fv = np.fft.rfft(vec, n=L)
    conv_full = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    max_val = conv_full.max()
    idxs = np.where(conv_full >= max_val - eps)[0]
    return float(max_val), idxs

# ---------------------------------------------------------
# Gradient of max (or plateau) entries of the convolution
# ---------------------------------------------------------
def _gradient_multi(vec: np.ndarray, idxs):
    n = vec.size
    g = np.zeros_like(vec)
    for idx in idxs:
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        j = np.arange(lo, hi + 1)
        g[j] += 2.0 * vec[idx - j]
    return g

# ---------------------------------------------------------
# Core optimiser – hybrid gradient / stochastic local search
# ---------------------------------------------------------
def _optimize(start_vec: np.ndarray, deadline: float, rng: np.random.Generator):
    a = start_vec.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # initial score
    max_val, idxs = _conv_max_and_indices(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # hyper‑parameters
    step = 0.5
    min_step = 1e-7
    decay = 0.9995
    beta = 0.9
    momentum = np.zeros_like(a)

    prob_random = 0.12      # mass‑transfer moves
    prob_smooth = 0.02      # local averaging
    prob_perturb = 0.02     # global Gaussian jitter

    while True:
        now = time.monotonic()
        if now + 0.2 > deadline:        # reserve a tiny margin
            break

        # ---- gradient descent on current worst autocorrelation ----
        g = _gradient_multi(a, idxs)
        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        cand = _project_simplex(cand)

        max_val_c, idxs_c = _conv_max_and_indices(cand, L)
        cand_score = 2.0 * n * max_val_c

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            max_val, idxs = max_val_c, idxs_c
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            # occasional uphill acceptance (simulated annealing style)
            T = step
            if rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                a = cand
                cur_score = cand_score
                max_val, idxs = max_val_c, idxs_c

        # ---- random mass transfer between two coordinates ----
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 1e-14:
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                frac = rng.random() * 0.5
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                max_val2, idxs2 = _conv_max_and_indices(cand2, L)
                score2 = 2.0 * n * max_val2
                if score2 < cur_score:
                    a = cand2
                    cur_score = score2
                    max_val, idxs = max_val2, idxs2
                    if score2 < best_score:
                        best_score = score2
                        best = cand2.copy()

        # ---- smoothing with a 3‑point kernel ----
        if rng.random() < prob_smooth:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)
            max_val_s, idxs_s = _conv_max_and_indices(smooth, L)
            score_s = 2.0 * n * max_val_s
            if score_s < cur_score:
                a = smooth
                cur_score = score_s
                max_val, idxs = max_val_s, idxs_s
                if score_s < best_score:
                    best_score = score_s
                    best = smooth.copy()

        # ---- global Gaussian perturbation ----
        if rng.random() < prob_perturb:
            noise = rng.normal(scale=0.02, size=n)
            cand3 = a + noise
            cand3 = _project_simplex(cand3)
            max_val3, idxs3 = _conv_max_and_indices(cand3, L)
            score3 = 2.0 * n * max_val3
            if score3 < cur_score:
                a = cand3
                cur_score = score3
                max_val, idxs = max_val3, idxs3
                if score3 < best_score:
                    best_score = score3
                    best = cand3.copy()

        # ---- step size decay ----
        step = max(min_step, step * decay)

    return best, best_score

# ---------------------------------------------------------
# Public entry point
# ---------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ---- initialise ----------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a0 = _sqrt_series(1024)                # fallback length if parent missing

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    # ---- main optimisation ---------------------------------------------
    best, best_score = _optimize(a0, deadline, rng)

    # ---- second start if time permits ----------------------------------
    if time.monotonic() + 10.0 < deadline:
        rand_vec = rng.random(a0.shape)
        rand_vec = _project_simplex(rand_vec)
        alt_best, alt_score = _optimize(rand_vec, deadline, rng)
        if alt_score < best_score:
            best = alt_best
            best_score = alt_score

    # Ensure we always return a valid (non‑empty) list
    if not np.isfinite(best_score) or best.size == 0:
        best = a0

    return [float(x) for x in best]
