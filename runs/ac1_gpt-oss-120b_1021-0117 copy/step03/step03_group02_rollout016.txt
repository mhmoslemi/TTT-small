import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    n = v.size
    if n == 0:
        return v
    # sort in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    theta = 0.0
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Full convolution via FFT (returns max and the full array)
# ------------------------------------------------------------
def _conv_full_max(a: np.ndarray, L: int):
    """Convolve a with itself (full) using FFT of length L, return max and array."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------

    else:

    # initial score

    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # -------------------- optimisation parameters --------------------
    step_init = 0.8

    decay = 0.9999
    momentum = np.zeros_like(a)
    beta = 0.9                     # momentum factor
