import time, numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.5)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        a = rng.random(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.size
    _L = 1 << ((2 * n - 1).bit_length())  # FFT length

    def conv_full(v):
        fv = np.fft.rfft(v, n=_L)
        c = np.fft.irfft(fv * fv, n=_L)[: 2 * n - 1]
        return c

    def score(v, conv=None):
        if conv is None:
            conv = conv_full(v)
        return float(2 * n * conv.max())

    def proj_simplex(v):
        """Euclidean projection onto the simplex {x>=0, sum x = 1}."""
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    # initial scores
    conv = conv_full(a)
    cur_sc = score(a, conv)
    best_a = a.copy()
    best_sc = cur_sc

    # parameters
    K = 20
    add_alpha0 = 1.0
    mul_eta0 = 1.0
    target = 1.50274349
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # compute active set
        if K >= conv.size:
            active = np.arange(conv.size)
        else:
            # use argpartition for performance
            active = np.argpartition(-conv, K - 1)[:K]

        # subgradient (sum of gradients of selected terms)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * a[k - idx]

        # additive update
        alpha = add_alpha0 / np.sqrt(it)
        cand_add = a - alpha * g
        cand_add = proj_simplex(cand_add)
        conv_add = conv_full(cand_add)
        cand_add_sc = score(cand_add, conv_add)

        # multiplicative update
        eta = mul_eta0 / np.sqrt(it)
        exp_factor = np.exp(-eta * g)
        cand_mul = a * exp_factor
        s_mul = cand_mul.sum()
        if s_mul == 0:
            cand_mul = np.full_like(a, 1.0 / n)
        else:
            cand_mul = cand_mul / s_mul
        conv_mul = conv_full(cand_mul)
        cand_mul_sc = score(cand_mul, conv_mul)

        # select best among candidates
        # also keep current as option
        if cand_add_sc < cand_mul_sc and cand_add_sc < cur_sc:
            a, conv, cur_sc = cand_add, conv_add, cand_add_sc
        elif cand_mul_sc < cur_sc:
            a, conv, cur_sc = cand_mul, conv_mul, cand_mul_sc
        else:
            # occasional Metropolis acceptance
            if rng.random() < 0.01:
                # pick one of the two candidates at random
                if cand_add_sc < cand_mul_sc:
                    a, conv, cur_sc = cand_add, conv_add, cand_add_sc
                else:
                    a, conv, cur_sc = cand_mul, conv_mul, cand_mul_sc

        # record best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # random mass transfer if stuck
        if no_improve > 1000 and rng.random() < 0.7:

            if a[i] > 0:
                j = rng.integers(0, n - 1)
                if j >= i:
