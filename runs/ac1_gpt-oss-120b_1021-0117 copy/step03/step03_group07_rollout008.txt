import numpy as np, time, math
# define conv_fft, score, apply_mass_move, etc.
def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
    return conv

def _score_from_conv(conv, sum_a):
    n = (conv.size + 1)//2
    return float(2.0 * n * conv.max() / (sum_a * sum_a))

def _score(a):
    conv = _conv_fft(a)
    return _score_from_conv(conv, a.sum())

def _apply_mass_move(a, conv, i, j, delta):
    n = a.shape[0]
    new_a = a.copy()
    new_a[i] -= delta
    new_a[j] += delta
    # ensure non-negative within floating errors
    # linear updates
    new_conv = conv.copy()

    new_conv[j:j+n] += 2.0 * delta * a

    new_conv[idxij] -= 2.0 * delta*delta
    # Clip small negatives

    return new_a, new_conv

import time

import numpy as np

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via real FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]

def _score(a: np.ndarray) -> float:
    """Exact AC1 upper bound for a (sum not forced to 1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _apply_mass_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """Transfer *delta* mass from i to j and update the full convolution."""
    n = a.shape[0]
    new_a = a.copy()
    new_a[i] -= delta
    new_a[j] += delta
    new_conv = conv.copy()
    # linear part of the update (uses original a)
    new_conv[i:i + n] -= 2.0 * delta * a
    new_conv[j:j + n] += 2.0 * delta * a
    # quadratic (d*d) part
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    new_conv[i + j] -= 2.0 * delta * delta
    return new_a, new_conv
