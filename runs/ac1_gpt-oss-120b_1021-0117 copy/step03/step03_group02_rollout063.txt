def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = np.ones(1, dtype=np.float64)  # fallback
    # normalize to simplex
    s = a.sum()
    if s == 0:
        a = np.ones_like(a)
        s = a.sum()
    a = _project_simplex(a / s)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    def compute_score(vec):
        max_val, _ = _conv_full_max(vec, L)
        return 2.0 * n * max_val

    cur_score = compute_score(a)
    best = a.copy()
    best_score = cur_score

    step = 0.5
    min_step = 1e-8
    decay = 0.99995
    alpha = 30.0
    prob_random = 0.01
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() < deadline:
        max_val, conv = _conv_full_max(a, L)
        # softmax weights over all lags
        w_raw = np.exp(alpha * (conv - max_val))
        w_sum = w_raw.sum()
        if w_sum == 0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # gradient
        g = np.zeros_like(a)
        # compute contributions using vectorised slices
        # for each lag k we add 2 * w_k * a_{k-j}
        for k, wk in enumerate(w):
            if wk == 0.0:
                continue
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo > hi:
                continue
            # indices j = lo..hi
            # corresponding a indices = k - j
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * wk * a[k - idx]

        cand = a - step * g
        cand = _project_simplex(cand)

        cand_score = compute_score(cand)

        improved = False
        if cand_score < cur_score - 1e-15:
            a = cand
            cur_score = cand_score
            improved = True
        else:
            # occasional uphill accept
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-12)):
                a = cand
                cur_score = cand_score
                improved = True

        if cand_score < best_score - 1e-14:
            best = cand.copy()
            best_score = cand_score
            no_improve = 0
        else:
            if not improved:
                no_improve += 1
            else:
                no_improve = 0

        # random perturb
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n)
                if j == i:
                    j = (j + 1) % n
                frac = rng.random() * 0.3
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                cand2_score = compute_score(cand2)
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best = cand2.copy()
                        best_score = cand2_score
                        no_improve = 0

        step = max(min_step, step * decay)

        if no_improve >= max_no_improve:
            # random restart but keep best
            a = rng.random(n)
            a = _project_simplex(a)
            step = 0.5
            no_improve = 0

        if time.monotonic() + 0.5 > deadline:
            break

    return [float(x) for x in best]

import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
# Full self‑convolution via FFT (returns full array and its max)
# ------------------------------------------------------------
def _conv_self(a: np.ndarray, L: int):
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return conv, float(conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------
    # initialise
    # --------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = np.ones(1, dtype=np.float64)  # fallback, never used
    # normalise to the simplex
    a = _project_simplex(a / (a.sum() if a.sum() != 0 else 1.0))

    n = a.size
    # FFT length large enough for both a*a (2n‑1) and a*w_rev (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------
    # helpers
    # --------------------------------------------------------
    def _score(vec):
        _, mx = _conv_self(vec, L)
        return 2.0 * n * mx

    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score

    # --------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------
    step = 0.5
    min_step = 1e-8
    decay = 0.99995
    alpha = 30.0           # soft‑max sharpness
    prob_random = 0.01     # occasional random mass transfer
    no_improve = 0
    max_no_improve = 200

    # --------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- self‑convolution, max and soft‑max weights -----
        conv, max_val = _conv_self(a, L)
        w_raw = np.exp(alpha * (conv - max_val))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # ----- gradient = 2 * (a * w) cross‑correlation -----
        # cross‑correlation g_j = 2 * sum_i a_i * w_{i+j}
        w_rev = w[::-1]
        fa = np.fft.rfft(a, n=L)

        corr = np.fft.irfft(fa * fw, n=L)[: n + w.size - 1]       # length 3n‑2
        g = 2.0 * corr[(w.size - 1) - np.arange(n)]

        cand = a - step * g
