import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a non‑negative vector onto the probability simplex."""
    n = v.size
    # make a copy to avoid side effects
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # guard against all‑zero output
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w

# ----------------------------------------------------------------------
# Convolution (FFT) – return max value and its index
# ----------------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    """Return (max_value, argmax) of the full linear convolution vec*vec."""
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * vec.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

# ----------------------------------------------------------------------
# Core optimizer – projected Adam on the current peak of the convolution
# ----------------------------------------------------------------------
def _optimize(start_vec: np.ndarray, deadline: float, rng: np.random.Generator):
    a = start_vec.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # helper to evaluate the objective (sum is 1 after projection)
    def _score(vec):
        m, _ = _conv_max_and_index(vec, L)
        return 2.0 * n * m

    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # probabilities for occasional perturbations
    prob_random = 0.02   # mass transfer
    prob_noise   = 0.01  # add tiny Gaussian noise
    prob_smooth  = 0.005 # three‑point smoothing

    while time.time() < deadline:
        t += 1

        # ------------------------------------------------------------------
        # compute gradient of the currently maximal convolution entry
        # ------------------------------------------------------------------
        max_val, max_idx = _conv_max_and_index(a, L)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            g[j] = 2.0 * a[max_idx - j]   # sub‑gradient of the max entry

        # ------------------------------------------------------------------
        # Adam update
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.4 / (np.sqrt(v_hat) + eps)   # base learning rate = 0.4

        cand = a - step * m_hat
        cand = _project_simplex(cand)

        cand_score = _score(cand)

        # ------------------------------------------------------------------
        # accept if improvement, else occasional uphill acceptance
        # ------------------------------------------------------------------
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            # simulated‑annealing style occasional acceptance
            T = max(0.01, step.max())
            if rng.random() < math.exp(-(cand_score - cur_score) / T):
                a = cand
                cur_score = cand_score

        # ------------------------------------------------------------------
        # occasional random mass transfer (keep sum = 1 via simplex proj.)
        # ------------------------------------------------------------------
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.4
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                sc2 = _score(cand2)
                if sc2 < cur_score:
                    a = cand2
                    cur_score = sc2
                if sc2 < best_score:
                    best_score = sc2
                    best = cand2.copy()

        # ------------------------------------------------------------------
        # occasional Gaussian noise perturbation (helps escape flat plateaus)
        # ------------------------------------------------------------------
        if rng.random() < prob_noise:
            noise = rng.normal(scale=0.01, size=n)
            cand3 = a + noise
            cand3 = np.maximum(cand3, 0.0)
            cand3 = _project_simplex(cand3)
            sc3 = _score(cand3)
            if sc3 < cur_score:
                a = cand3
                cur_score = sc3
            if sc3 < best_score:
                best_score = sc3
                best = cand3.copy()

        # ------------------------------------------------------------------
        # occasional smoothing (3‑point kernel) – useful for spiky patterns
        # ------------------------------------------------------------------
        if rng.random() < prob_smooth:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)
            scs = _score(smooth)
            if scs < cur_score:
                a = smooth
                cur_score = scs
            if scs < best_score:
                best_score = scs
                best = smooth.copy()

        # ------------------------------------------------------------------
        # stagnation handling – if no improvement for a while, jitter a bit
        # ------------------------------------------------------------------
        if t % 5000 == 0:
            # if we haven't improved recently, inject small random jitter
            jitter = rng.normal(scale=0.005, size=n)
            a = np.clip(a + jitter, 0.0, None)
            a = _project_simplex(a)

        # ------------------------------------------------------------------
        # early‑exit guard
        # ------------------------------------------------------------------
        if time.time() + 0.5 > deadline:
            break

    return best, best_score
