def propose_candidate(seed=42, budget_s=1100):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 256
        a = np.random.rand(n)
    # project to simplex sum=1
    a = _project_simplex(a / np.sum(a))

    n = a.size
    # compute L for FFT.
    L = 1 << ((2 * n - 1).bit_length())
    # initial best
    max_val, _ = _conv_full_max(a, L)
    best = a.copy()
    best_score = 2.0 * n * max_val

    # parameters
    step = 0.5
    min_step = 1e-6
    decay = 0.9995
    no_improve = 0
    max_no_improve = 200
    eps_active = 1e-12
    perturb_prob = 0.05

    while time.monotonic() < deadline:
        max_val, conv = _conv_full_max(a, L)
        cur_score = 2.0 * n * max_val
        # if improved best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # get active indices (within tolerance)
        active_idx = np.where(conv >= max_val - 1e-15 * max_val)[0] # relative tolerance?

        # compute subgradient
        g = np.zeros_like(a)
        for j in active_idx:
            lo = max(0, j - (n - 1))
            hi = min(n - 1, j)
            i_arr = np.arange(lo, hi+1)
            g[i_arr] += 2.0 * a[j - i_arr]

        # step update
        cand = a - step * g
        cand = _project_simplex(cand)
        # evaluate
        max_val_c, _ = _conv_full_max(cand, L)
        cand_score = 2.0 * n * max_val_c

        # accept if better, else maybe occasional uphill
        if cand_score < cur_score - 1e-15:
            a = cand
            cur_score = cand_score
        else:
            # occasional uphill using simulated annealing
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-12)):
                a = cand
                cur_score = cand_score

            src = rng.integers(n)

                    dst += 1

                cand2 = _project_simplex(cand2)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # initial sequence
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 256
        a = np.random.rand(n)

    # ensure nonnegative, normalize sum to 1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.ones_like(a) / a.size
    else:
        a = a / s

    n = a.size
    L = 1 << ((2*n - 1).bit_length())

    # function to compute max conv
    def conv_max(v):
        fv = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
        max_val = float(conv.max())
        return max_val, conv

    max_val, conv = conv_max(a)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # params
    step = 0.5
    min_step = 1e-7
    decay = 0.9995
    alpha = 12.0   # softmax sharpness
    prob_random = 0.04
    no_improve = 0
    max_no_improve = 250

    while time.monotonic() < deadline:
        max_val, conv = conv_max(a)
        cur_score = 2.0 * n * max_val
        # update best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # compute weight vector
        # avoid overflow: subtract max_val for numeric stability inside exponent already zero at max
        w_raw = np.exp(alpha * (conv - max_val))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # gradient via correlation
        g = 2.0 * np.correlate(w, a, mode='valid')
        # additive step and projection onto simplex
        cand = a - step * g
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand /= cand_sum
        else:
            cand = np.ones_like(a) / n

        # evaluate candidate
        max_val_c, _ = conv_max(cand)
        cand_score = 2.0 * n * max_val_c

        # accept if improvement
        accepted = False
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            accepted = True
        else:
            # occasional uphill
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-12)):
                a = cand
                cur_score = cand_score
                accepted = True

        # random perturbation
        if rng.random() < prob_random:
            src = rng.integers(n)
            if a[src] > 1e-12:
                dst = rng.integers(n-1)
                if dst >= src:
                    dst += 1
                frac = rng.random() * 0.2
                delta = a[src] * frac
                cand2 = a.copy()
                cand2[src] -= delta
                cand2[dst] += delta
                cand2 = np.maximum(cand2, 0.0)
                cand2_sum = cand2.sum()
                if cand2_sum > 0:
                    cand2 /= cand2_sum
                else:
                    cand2 = np.ones_like(a) / n
                max_val2, _ = conv_max(cand2)
                cand2_score = 2.0 * n * max_val2
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score

        # decay step

            a = a / a.sum()
            step = 0.5
            no_improve = 0

        # early exit
        if time.monotonic() + 0.5 > deadline:
            break

def _conv_full_max(a: np.ndarray, L: int):
    """Return (max convolution value, full convolution array) for a*a."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv


def _normalize_to_simplex(v: np.ndarray):
    """Project non‑negative vector onto the probability simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    # fallback to uniform
    return np.full_like(v, 1.0 / v.size)

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------ initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: modest random start
        a = rng.random(256, dtype=np.float64)

    a = _normalize_to_simplex(a)
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two length for FFT

    max_val, _ = _conv_full_max(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # ------ optimisation parameters ------------------------------------
    step = 0.5               # learning rate
    min_step = 1e-7
    decay = 0.9995
    alpha = 12.0             # soft‑max sharpness for weighting lags
    prob_random = 0.04       # occasional random mass‑transfer
    no_improve = 0
    max_no_improve = 250

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- evaluate current state -----------------------------------
        max_val, conv = _conv_full_max(a, L)
        cur_score = 2.0 * n * max_val

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- compute smooth weights over all lags --------------------
        # w_i ∝ exp(alpha * (conv_i - max_val))
        w_raw = np.exp(alpha * (conv - max_val))
        w_sum = w_raw.sum()
        w = w_raw / w_sum if w_sum != 0.0 else np.full_like(w_raw, 1.0 / w_raw.size)

        # ----- gradient of the smooth objective -----------------------
        # g = 2 * correlate(w, a)   (valid mode gives length n)
        g = 2.0 * np.correlate(w, a, mode="valid")

        # ----- additive step + projection onto simplex -----------------
        cand = a - step * g
        cand = _normalize_to_simplex(cand)

        # ----- evaluate candidate ---------------------------------------
        max_val_c, _ = _conv_full_max(cand, L)
        cand_score = 2.0 * n * max_val_c

        # ----- accept if better (or occasional uphill) -----------------
        improved = False
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            improved = True
        else:
            # simulated‑annealing style uphill move
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-12)):
                a = cand
                cur_score = cand_score
                improved = True

        # ----- enforce symmetry (optional, helps stability) ----------
        a = (a + a[::-1]) * 0.5

        # ----- random mass‑transfer perturbation -----------------------
        if rng.random() < prob_random:
            src = rng.integers(0, n)
            if a[src] > 1e-12:
                dst = rng.integers(0, n - 1)
                if dst >= src:
                    dst += 1
                frac = rng.random() * 0.2
                delta = a[src] * frac
                cand2 = a.copy()
                cand2[src] -= delta
                cand2[dst] += delta
                cand2 = _normalize_to_simplex(cand2)
                max_val2, _ = _conv_full_max(cand2, L)
                cand2_score = 2.0 * n * max_val2
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    improved = True

        # ----- decay step size -----------------------------------------
        step = max(min_step, step * decay)

        # ----- restart if stagnated ------------------------------------
        if no_improve >= max_no_improve:
            a = rng.random(n)
            a = _normalize_to_simplex(a)
            step = 0.5
            no_improve = 0

        # ----- early‑exit safeguard ------------------------------------
        if time.monotonic() + 0.5 > deadline:
            break

    # return best candidate as plain Python list
    return [float(x) for x in best]
