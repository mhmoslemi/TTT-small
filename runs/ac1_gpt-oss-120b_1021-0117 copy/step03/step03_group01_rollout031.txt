# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (L1 = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # safeguard against all‑zero output
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w

        else:

            if rng.random() < 0.02:
                a = cand
                cur_score = cand_score

        # update schedules
        step = max(min_step, step * decay)
        alpha = min(alpha_max, alpha * 1.01)
        if step <= min_step and alpha >= alpha_max:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
```
