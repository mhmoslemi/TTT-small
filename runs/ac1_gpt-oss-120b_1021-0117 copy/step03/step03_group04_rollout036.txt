    # Random Dirichlet‑like start
    rand_vec = rng.random(n)
    rand_vec = np.maximum(rand_vec, 0.0)
    rand_vec /= rand_vec.sum()

    candidates = [uniform, tent, rand_vec]
    if parent is not None:
        candidates.append(parent)

    # ------------------------------------------------------------------
    # 3. Helper – fast score (sum is always 1 in our iterates)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((3 * n - 2 - 1).bit_length())  # enough for a*w convolution

    def fast_score(a: np.ndarray) -> float:
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()  # denominator = 1 because sum(a)=1

    # ------------------------------------------------------------------
    # 4. Pick the best initial candidate
    # ------------------------------------------------------------------
    best = None
    best_val = float("inf")
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_val:
            best_val = sc
            best = cand.copy()

    a = best.copy()

    # ------------------------------------------------------------------
    # 5. Optimisation parameters
    # ------------------------------------------------------------------
    it = 0
    eta0 = 0.7          # initial step magnitude
    mu = 0.85           # momentum factor
    momentum = np.zeros_like(a)
    alpha = 0.6         # soft‑max temperature
    target = 1.50274349

    # ------------------------------------------------------------------
    # 6. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # --- full convolution (a*a) ---
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if best_val <= target:
                break

        # --- soft‑max weights for all shifts ---
        # stability: shift by max before exponentiation
        max_c = conv.max()
        w_raw = np.exp(alpha * (conv - max_c))
        w = w_raw / w_raw.sum()          # sum to 1, symmetric automatically

        # --- gradient of smooth max (2 * a * w) ---
        fw = np.fft.rfft(w, n=fft_len)
        grad_full = np.fft.irfft(fa * fw, n=fft_len)[: 3 * n - 2]
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice length n

        # --- momentum update ---
        momentum = mu * momentum + (1.0 - mu) * grad

        # --- step size schedule (diminishing) ---
        eta = eta0 / np.sqrt(it)

        # --- exponentiated‑gradient update on the simplex ---
        a = a * np.exp(-eta * momentum)
        s = a.sum()
        if s == 0.0:                     # numerical safeguard
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # --- occasional re‑temperature increase & jitter ---
        if it % 300 == 0:
            alpha = min(alpha * 1.15, 5.0)          # sharpen soft‑max slowly
        if it % 500 == 0:
            jitter = np.exp(rng.normal(0.0, 0.015, size=n))
            a = a * jitter
            a = a / a.sum()

    # ------------------------------------------------------------------
    # 7. Return the best found sequence as plain Python floats
    # ------------------------------------------------------------------
    return [float(v) for v in best]
```
