import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)  # safety margin

    # ----- initialise sequence -----
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = rng.random(256, dtype=np.float64)
    # normalise to the simplex (sum = 1, non‑negative)
    a = _project_simplex(a / (a.sum() if a.sum() != 0 else 1.0))

    n = a.size
    factor = 2.0 * n                     # score = factor * max(conv) because sum=1
    L = 1 << ((3 * n - 2).bit_length())  # FFT length sufficient for all convolutions

    # pre‑compute an index array for gradient extraction
    idx_arr = np.arange(n, dtype=np.int64)
    # ------------------------------------------------------------
    # initial evaluation
    # ------------------------------------------------------------
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    conv = conv_full[: 2 * n - 1]
    cur_max = conv.max()
    cur_score = factor * cur_max
    best = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------
    # optimisation hyper‑parameters
    # ------------------------------------------------------------
    step = 0.5
    min_step = 1e-8
    decay = 0.9997
    alpha = 5.0                     # soft‑max sharpness for weighting lags
    prob_random = 0.025
    no_improve = 0
    max_no_improve = 500
    max_iters = 10 ** 7   # just a huge cap, loop controlled by time

    # ------------------------------------------------------------
    while time.monotonic() < deadline and no_improve < max_iters:
        # ---- convolution of current a (already have fft_a) ----
        conv_full = np.fft.irfft(fft_a * fft_a, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_score = factor * max_val

        # ---- update global best ----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ---- compute soft‑max weights over all lags ----
        # (exp of negative numbers, max gets weight 1)
        w = np.exp(alpha * (conv - max_val))

        # ---- gradient via convolution with reversed weights ----
        w_rev = w[::-1]
        fft_wrev = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(fft_a * fft_wrev, n=L)
        mid = w_rev.size - 1                     # = 2*n-2
        g = 2.0 * grad_full[mid - idx_arr]       # gradient of the weighted max

        # ---- candidate update (projected sub‑gradient step) ----
        cand = a - step * g
        cand = _project_simplex(cand)

        # ---- evaluate candidate ----
        fft_cand = np.fft.rfft(cand, n=L)
        conv_c_full = np.fft.irfft(fft_cand * fft_cand, n=L)
        max_c = conv_c_full[: 2 * n - 1].max()
        cand_score = factor * max_c

        # ---- accept if improvement or occasional uphill move ----
        improved = False
        if cand_score < cur_score - 1e-15:
            a = cand
            fft_a = fft_cand
            cur_score = cand_score
            improved = True
            no_improve = 0
        else:
            # probabilistic uphill move (simulated annealing flavour)
            prob_up = np.exp(-(cand_score - cur_score) / max(step, 1e-12))
            if rng.random() < prob_up:
                a = cand
                fft_a = fft_cand
                cur_score = cand_score
                improved = True

        # ---- occasional random mass‑transfer perturbation ----
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.4
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                fft_cand2 = np.fft.rfft(cand2, n=L)
                conv2_full = np.fft.irfft(fft_cand2 * fft_cand2, n=L)
                max2 = conv2_full[: 2 * n - 1].max()
                cand2_score = factor * max2
                if cand2_score < cur_score:
                    a = cand2
                    fft_a = fft_cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()
                    no_improve = 0

        # ---- step size decay ----
        step = max(min_step, step * decay)

        # ---- restart if stuck for long ----
        if no_improve >= max_no_improve:
            a = rng.random(n, dtype=np.float64)
            a = _project_simplex(a)
            fft_a = np.fft.rfft(a, n=L)
            step = 0.5
            no_improve = 0

        # ---- early exit safeguard ----
        if time.monotonic() + 0.5 > deadline:
            break

    # return best found sequence as a plain Python list
    return [float(x) for x in best]
