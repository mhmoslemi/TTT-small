def propose_candidate(...):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 2.0)  # keep margin
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent,0,1000)
        n = parent.size
        a = np.maximum(parent,0)
        a = a / a.sum()  # normalize to sum1
    except:
        n = 256
        a = np.full(n, 1.0/n)
    best = a.copy()
    best_score = 2*n*np.max(np.convolve(best,best))

    # Preprocess fft length
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len-1).bit_length())
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2 # initial learning rate
    it = 0
    no_improve = 0
    stagnation_limit = 2000
    # jitter frequency
    jitter_period = 100
    jitter_std = 0.02

    while time.monotonic() < deadline:
        # compute conv via FFT
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa*fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        cur_score = 2*n*max_val
        # update best
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= 1.50274349:
                break
        else:
            no_improve += 1

        # compute gradient as mean of active peaks
        # tolerance for near max
        tol = max_val * 1e-8 + 1e-12
        active = np.where(conv >= max_val - tol)[0]
        # optionally limit number of active for speed
        max_active = 30
        if active.size > max_active:
            # pick top max_active by conv value (they are near max anyway)
            # active already near max; we can randomly sample
            rng.shuffle(active)
            active = active[:max_active]
        g = np.zeros_like(a)
        for k in active:
            # compute contributions
            start_i = max(0, k - (n - 1))
            end_i = min(n-1, k)  # inclusive
            # vectorized:
            i = np.arange(start_i, end_i+1)
            partner = k - i
            g[i] += 2.0 * a[partner]
        # optional scaling: use average
        if active.size > 0:
            g /= active.size

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # enforce nonnegativity and sum=1 via simplex projection
        a = _proj_simplex(a)

        # jitter periodically
        if it % jitter_period == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            m.fill(0)
            v.fill(0)
            no_improve = 0

        # restart from best if stagnates
        if no_improve >= stagnation_limit:
            a = best.copy()
            pert = np.exp(rng.normal(0.0, 0.01, size=n))
            a = _proj_simplex(a * pert)
            m.fill(0)
            v.fill(0)
            no_improve = 0

    # final check: maybe evaluate best using evaluator; but evaluator normalizes sum again (and clips) and returns objective.
    return [float(x) for x in best]

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_conv(a: np.ndarray):
    """Return max of full linear convolution a*a and the full convolution array."""
    n = a.size
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv.max(), conv


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step–function that minimizes the AC1 upper bound.
    Exploits the LP formulation described in the problem statement and a
    simple momentum‑free update scheme.  Returns the best sequence found before
    the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1.  Initialise a (fallback to uniform if parent missing)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        a = np.maximum(parent, 0.0)
    except Exception:
        a = None

    if a is None or a.size == 0:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = int(a.size)
        a = a / a.sum()
    best = a.copy()
    best_score = 2.0 * n * _max_conv(best)[0]

    # ------------------------------------------------------------------
    # 2.  Pre‑compute sparse index pattern for the convolution constraints
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    total_nnz = n * n
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = col_idx + np.tile(np.arange(n, dtype=np.int32), n)

    # ------------------------------------------------------------------
    # 3.  Optimisation loop (LP + convex update)
    # ------------------------------------------------------------------
    t = 0.5                     # update mixing proportion
    it = 0
    max_iters = 2000
    jitter_every = 60
    jitter_std = 0.02
    target_score = 1.50274349

    while time.monotonic() < deadline and it < max_iters:
        # ----- evaluate current sequence --------------------------------
        max_val, _ = _max_conv(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target_score:
                break

        # ----- build LP (maximise sum(g) s.t. conv(a,g) <= max_val) ----
        M = max_val
        data = np.tile(a, n)                     # each column gets full a vector
        A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(conv_len, n))
        b_ub = np.full(conv_len, M, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)    # maximise sum(g)

        # solve
        res = opt.linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs', options={'presolve': True})
        if not res.success or res.x is None:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break
        # normalise g to have the same unit sum as a
        g = g / sum_g

        # ----- convex combination update ---------------------------------
        a = (1.0 - t) * a + t * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        it += 1

        # ----- occasional jitter to avoid stagnation --------------------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = a * jitter
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 4.  Return the best sequence found (plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
