import time

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

    # -------------------- optimisation parameters --------------------
    step = 0.5
    min_step = 1e-5
    decay = 0.9992
    beta = 0.9                     # momentum coefficient
    momentum = np.zeros_like(a)

    K = 20                         # top‑lag count
    alpha = 20.0                   # soft‑max sharpness
    prob_random = 0.15            # chance of random transfer
    max_no_improve = 500
    no_improve = 0

    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution & top‑lag selection -----
        max_val, conv = _conv_full_max(a, L)

        # pick K largest lags (or all if fewer)
        if K >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            part = np.argpartition(conv, -K)
            top_idx = part[-K:]

        # soft‑max weights for these lags
        w_raw = np.exp(alpha * (conv[top_idx] - max_val))
        w_sum = w_raw.sum()
        w = w_raw / w_sum if w_sum != 0.0 else np.full_like(w_raw, 1.0 / w_raw.size)

        # ----- gradient from top lags (with momentum) -----
        g = np.zeros_like(a)
        for idx, wi in zip(top_idx, w):
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            if lo <= hi:
                j = np.arange(lo, hi + 1)
                g[j] += wi * 2.0 * a[idx - j]

        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        max_val_c, _ = _conv_full_max(cand, L)
        cand_score = 2.0 * n * max_val_c

        improved = False
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            improved = True
        else:
            # occasional uphill acceptance
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-12)):
                a = cand
                cur_score = cand_score
                improved = True

        # ----- keep global best -----
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            no_improve = 0
        else:
            if not improved:
                no_improve += 1
            else:
                no_improve = 0

        # ----- random mass‑transfer perturbation -----
        if rng.random() < prob_random:
            src_candidates = np.flatnonzero(a > 1e-12)
            if src_candidates.size:
                src = rng.choice(src_candidates)
                dst = rng.choice(np.delete(np.arange(n), src))
                frac = rng.random() * 0.2
                delta = a[src] * frac
                cand2 = a.copy()
                cand2[src] -= delta
                cand2[dst] += delta
                cand2 = _project_simplex(cand2)
                max_val2, _ = _conv_full_max(cand2, L)
                cand2_score = 2.0 * n * max_val2
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()
                        no_improve = 0

        # ----- step decay -----
        step = max(min_step, step * decay)

        # ----- restart if stuck -----
        if no_improve >= max_no_improve:
            a = rng.random(n)
            a = _project_simplex(a)
            momentum = np.zeros_like(a)
            step = 0.5
            no_improve = 0

        # ----- early‑exit guard -----
        if time.monotonic() + 0.2 > deadline:
            break

    # return best candidate as plain Python list
    return [float(x) for x in best]
