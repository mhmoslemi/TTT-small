import time
import math
import numpy as np

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of x with itself using a power‑of‑two FFT."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]
    return conv

def _score_from_conv(conv: np.ndarray, n: int, s: float) -> float:
    """Score exactly as evaluate_sequence."""
    return 2.0 * n * conv.max() / (s * s)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # safety margin so we have time to return a result
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------------
    # 1. Initialisation
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = np.full(2000, 1.0 / 2000, dtype=np.float64)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
        total = 1.0
    a /= total                               # probability simplex, sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())      # power‑of‑two length for FFT

    cur = a.copy()
    cur_conv = _conv_fft(cur, L)
    cur_sc = _score_from_conv(cur_conv, n, cur.sum())

    best = cur.copy()
    best_sc = cur_sc

    # ----------------------------------------------------------------------
    # 2. Search parameters
    # ----------------------------------------------------------------------
    it = 0
    alpha0 = 0.7            # initial mirror‑step size
    eps_move = 0.20         # max fraction of a weight that can be moved
    jitter_scale = 5e-5    # Gaussian jitter scale
    no_improve = 0
    temp = 0.12             # temperature for Metropolis acceptances

    # ----------------------------------------------------------------------
    # 3. Main optimisation loop
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- 3.1 compute convolution and active index ----
        conv = _conv_fft(cur, L)
        k = int(conv.argmax())                     # index of maximal entry
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # ---- 3.2 sub‑gradient (mirror descent) ----
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]                # ∂max / ∂a_i at the active spot
        eta = alpha0 / math.sqrt(it + 1.0)
        mul = np.exp(-eta * g)
        cand = cur * mul
        cand_sum = cand.sum()
        if cand_sum > 0.0 and np.isfinite(cand_sum):
            cand /= cand_sum
        else:
            cand = np.full_like(cur, 1.0 / n)

        cand_conv = _conv_fft(cand, L)
        cand_sc = _score_from_conv(cand_conv, n, cand.sum())

        # ---- 3.3 acceptance ----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # Metropolis‑type acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        if cand_sc < best_sc:
            best_sc, best = cand_sc, cand.copy()

        # ---- 3.4 random pairwise mass transfer ----
        if rng.random() < 0.09:
            i = rng.integers(0, n)
