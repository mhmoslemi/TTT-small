import time
import math
import numpy as np
import cvxpy as cp

# --------------------------------------------------------------
# Projection onto the probability simplex (L1 = 1, non‑negative)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1, dtype=v.dtype)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:          # safeguard – should not happen
        w[:] = 1.0 / n
    return w

# --------------------------------------------------------------
# Full convolution via FFT (returns the whole array)
# --------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv

# --------------------------------------------------------------
# Score (exactly the evaluator’s formula) assuming non‑negative a
# --------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.size
    conv = _full_convolution(a)
    max_b = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:               # should never happen after projection
        return math.inf
    return 2.0 * n * max_b / (s * s)

# --------------------------------------------------------------
# Build the linear constraints matrix for a set of convolution
# indices. For each chosen index k we have   sum_i f_i * g_{k-i} <= M .
# --------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    n = f.size
    m = idxs.size
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(idxs):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        j = np.arange(lo, hi + 1, dtype=np.int64)
        # coefficients are f_{k-j}
        A[row, lo:hi + 1] = f[k - j]
    return A

# --------------------------------------------------------------
# Core iterative LP improvement
# --------------------------------------------------------------
def _lp_improve(start: np.ndarray, deadline: float, rng: np.random.Generator):
    n = start.size
    f = start.copy()
    best = f.copy()
    best_score = _score(best)

    # parameters
    top_k = 30                # number of indices taken as active constraints
    max_no_improve = 5        # stop if no improvement for this many rounds
    no_improve = 0

    while time.monotonic() < deadline - 0.5:
        conv_full = _full_convolution(f)
        cur_score = 2.0 * n * float(conv_full.max())
        if cur_score < best_score:
            best_score = cur_score
            best = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        # pick active indices (largest convolution values)
        k_act = min(top_k, conv_full.size)
        idxs = np.argpartition(-conv_full, k_act - 1)[:k_act]

        # build linear constraints matrix
        A = _build_constraint_matrix(f, idxs)

        # linear program: minimise M subject to A g <= M, g >= 0, sum(g) = 1
        g = cp.Variable(n)
        M = cp.Variable()
        constraints = [
            g >= 0,
            cp.sum(g) == 1,
            A @ g <= M,
            M >= 0
        ]
        prob = cp.Problem(cp.Minimize(M), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
        except Exception:
            break

        if g.value is None:
            break

        g_new = np.asarray(g.value, dtype=np.float64)
        # numerical clean‑up
        g_new = np.maximum(g_new, 0.0)
        s = g_new.sum()
        if s <= 0.0:
            break
        g_new /= s                     # enforce exact sum = 1

        f = g_new                       # proceed to next iteration

    return best, best_score

# --------------------------------------------------------------
# Public entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # initialise – use the provided parent if present, otherwise a simple
    # sqrt‑series of comparable length.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: generate a short sqrt‑series, then tile
        n_fallback = 256
        a0 = np.empty(n_fallback, dtype=np.float64)
        a0[0] = 1.0
        for k in range(1, n_fallback):
            if k == 1:
                s = 0.0
            else:
                s = float(np.dot(a0[1:k], a0[k - 1:0:-1]))
            a0[k] = (1.0 - s) / 2.0
        # tile to a larger length for a fair comparison
        repeat = max(1, 2535 // n_fallback + 1)
        a0 = np.tile(a0, repeat)[:2535]

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    # ------------------------------------------------------------------
    # primary optimisation run
    # ------------------------------------------------------------------
    best_vec, best_score = _lp_improve(a0, deadline, rng)

    # ------------------------------------------------------------------
    # second random start if time permits (adds robustness)
    # ------------------------------------------------------------------
    if time.monotonic() + 8.0 < deadline:
        rnd = rng.random(a0.shape, dtype=np.float64)
        rnd = _project_simplex(rnd)
        alt_vec, alt_score = _lp_improve(rnd, deadline, rng)
        if alt_score < best_score:
            best_vec, best_score = alt_vec, alt_score

    # ------------------------------------------------------------------
    # final safety: if everything failed, fall back to the simple
    # projection of the parent (guaranteed feasible)
    # ------------------------------------------------------------------
    if best_vec is None or best_vec.size == 0:
        best_vec = a0

    return [float(x) for x in best_vec]
