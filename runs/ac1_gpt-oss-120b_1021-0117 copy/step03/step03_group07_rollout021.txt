# ----------------------------------------------------------------------
# Basic convolution and scoring utilities (full convolution via FFT)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (returns length 2n-1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fv = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fv * fv, n=L)
    return conv_full[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (sum is not forced to 1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    maxb = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * maxb / (s * s))

def _make_exponential(n: int, r: float) -> np.ndarray:

    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _make_polynomial(n: int, q: float) -> np.ndarray:
    a = (1.0 - np.arange(n, dtype=np.float64) / (n - 1)) ** q
