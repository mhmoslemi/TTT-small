def simplex_projection(v):
    # https://arxiv.org/pdf/1309.1541.pdf (Michelot)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0
    # ensure sum to 1 (maybe due rounding)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def simplex_projection(v):
    # v is numpy array
    # from https://github.com/indigo-dnv/pyLagrange/blob/master/pyLagrange/simplex.py
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    # find rho
    rho_arr = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    # Ensure sum to 1 due to potential floating errors
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / n)
    return w

import time
import numpy as np

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    cond = u - css / np.arange(1, n + 1)
    rho = np.where(cond > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_fft(v: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution v * v using FFT of length L."""
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * v.size - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """Linear correlation (w ⋆ a) for lag 0..len(a)-1 using FFT length L."""
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(wf * np.conj(af), n=L)[: a.size]
    return corr
