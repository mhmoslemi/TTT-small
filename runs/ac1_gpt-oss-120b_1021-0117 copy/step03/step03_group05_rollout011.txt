def _project_simplex(v):
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
```

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    # ------------------------------------------------------------------
    # main sub‑gradient descent + occasional random perturbations
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        conv = _conv(cur)
        maxc = conv.max()
        eps = max(rel_eps * maxc, 1e-14)
        active = np.where(conv >= maxc - eps)[0]

        # sub‑gradient of the max‑convolution term(s)
        grad = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * cur[k - idx]

        # step size decays like 1/sqrt(iter)
        step = step0 / np.sqrt(it)

        # ----- additive (projected‑gradient) candidate -----
        cand_add = cur - step * grad
        cand_add = _project_simplex(cand_add)
        cand_add_sc = _score(cand_add)

        # ----- multiplicative (exponential‑weights) candidate -----
        exp_factor = np.exp(-step * grad)
        cand_mul = cur * exp_factor
        cand_mul = _project_simplex(cand_mul)
        cand_mul_sc = _score(cand_mul)

        # pick the best of the three (current, add, mul)
        if cand_add_sc <= cand_mul_sc and cand_add_sc < cur_sc:
            cur, cur_sc = cand_add, cand_add_sc
        elif cand_mul_sc < cur_sc:
            cur, cur_sc = cand_mul, cand_mul_sc
        else:
            # occasional uphill acceptance to escape flat plateaus
            delta = min(cand_add_sc, cand_mul_sc) - cur_sc
            if delta < 0:
                # shouldn't happen, but keep safety
                cur, cur_sc = (cand_add if cand_add_sc < cand_mul_sc else cand_mul), \
                               (cand_add_sc if cand_add_sc < cand_mul_sc else cand_mul_sc)
            else:
                temp = 0.05 / np.sqrt(it)   # small temperature
                if rng.random() < np.exp(-delta / (temp + 1e-12)):
                    cur, cur_sc = (cand_add if cand_add_sc < cand_mul_sc else cand_mul), \
                                   (cand_add_sc if cand_add_sc < cand_mul_sc else cand_mul_sc)

        # update global best
        if cur_sc < best_sc:
            best, best_sc = cur.copy(), cur_sc
            if best_sc < target:
                break

        # ------------------------------------------------------------------
        # occasional random mass‑transfer move (helps to escape stagnation)
        # ------------------------------------------------------------------
        if it % 2000 == 0 and rng.random() < 0.3:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc

        # ------------------------------------------------------------------
        # jittered restart toward the current best (once in a while)
        # ------------------------------------------------------------------
        if it % 8000 == 0 and rng.random() < 0.25:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < best_sc:
                best, best_sc = tmp.copy(), tmp_sc
            if rng.random() < 0.5:
                cur, cur_sc = tmp.copy(), tmp_sc

    # ----------------------------------------------------------------------
    # final short local refinement (random pairwise transfers)
    # ----------------------------------------------------------------------
    refin_end = time.monotonic() + min(0.4, deadline - time.monotonic())
    while time.monotonic() < refin_end:
        i = rng.integers(0, n)
        if best[i] == 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best[i] * rng.random() * 0.3
        tmp = best.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < best_sc:
            best, best_sc = tmp, tmp_sc

    # return plain Python list
    return [float(v) for v in best]
