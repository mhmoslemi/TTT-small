import time, numpy as np, math

def _project_simplex(v):
    # same as before

def _conv_full(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    return conv

def _conv_max_and_index(a):
    conv = _conv_full(a)
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv

def _optimize(start, deadline, rng):
    a = start.copy()
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    # initial convolution
    max_val, max_idx, conv = _conv_max_and_index(a)
    cur_score = 2 * n * max_val
    best = a.copy()
    best_score = cur_score
    # momentum
    momentum = np.zeros_like(a)
    # step size
    step = 0.4
    min_step = 1e-6
    decay = 0.9995
    # probabilities
    prob_move = 0.1
    prob_smooth = 0.02
    # main loop
    while time.monotonic() < deadline:
        # compute gradient for max index only
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:
            j = np.arange(lo, hi+1)
            g[j] = 2.0 * a[max_idx - j]
        # momentum update
        momentum = 0.9 * momentum + 0.1 * g
        cand = a - step * momentum
        cand = _project_simplex(cand)
        max_val_c, max_idx_c, _ = _conv_max_and_index(cand)
        cand_score = 2 * n * max_val_c
        # acceptance
        if cand_score < cur_score:
            a, cur_score, max_idx = cand, cand_score, max_idx_c
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            # simulated annealing using step as T
            if rng.random() < math.exp(-(cand_score - cur_score) / (step + 1e-12)):
                a, cur_score, max_idx = cand, cand_score, max_idx_c
        # random mass transfer
        if rng.random() < prob_move:
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                # transfer fraction up to 30%
                frac = rng.random() * 0.3
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                max_val2, max_idx2, _ = _conv_max_and_index(cand2)
                score2 = 2*n*max_val2
                if score2 < cur_score:
                    a, cur_score, max_idx = cand2, score2, max_idx2
                    if score2 < best_score:
                        best, best_score = cand2.copy(), score2
        # smoothing
        if rng.random() < prob_smooth:
            kernel = np.array([0.25, 0.5, 0.25])
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)
            max_vals, max_idx_s, _ = _conv_max_and_index(smooth)
            score_s = 2*n*max_vals
            if score_s < cur_score:
                a, cur_score, max_idx = smooth, score_s, max_idx_s
                if score_s < best_score:
                    best, best_score = smooth.copy(), score_s
        # step adjustment
        step = max(min_step, step * decay)
        # early termination check
        if time.monotonic() + 0.5 > deadline:
            break
    return best, best_score

import time
import math
import numpy as np

# -------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# -------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:                     # guard against pathological case
        w[:] = 1.0 / n
    return w

# -------------------------------------------------------------
# Analytic sqrt‑series (a*a = 1+…+x^{2n‑2})
# -------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k-1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a

# -------------------------------------------------------------
# Full convolution via FFT, returning max value, its index, and the array
# -------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray):
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv

# -------------------------------------------------------------
# Local optimisation (projected sub‑gradient with momentum)
# -------------------------------------------------------------
def _optimize(start_vec: np.ndarray, deadline: float, rng: np.random.Generator,
              max_iters: int = None):
    a = start_vec.copy()
    n = a.size
    max_val, max_idx, _ = _conv_max_and_index(a)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    momentum = np.zeros_like(a)
    step = 0.6
    min_step = 1e-6
    decay = 0.9995
    prob_move = 0.07
    prob_smooth = 0.015
    beta = 0.9
    it = 0

    while time.monotonic() < deadline:
        if max_iters is not None and it >= max_iters:
            break
        it += 1

        # ---- sub‑gradient of the current maximal entry ----
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            g[j] = 2.0 * a[max_idx - j]

        # ---- momentum step & projection ----
        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_max, cand_idx, _ = _conv_max_and_index(cand)
        cand_score = 2.0 * n * cand_max

        # ---- acceptance (greedy + occasional uphill) ----
        if cand_score < cur_score:
            a, cur_score, max_idx = cand, cand_score, cand_idx
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            # simulated‑annealing style acceptance
            if rng.random() < math.exp(-(cand_score - cur_score) / (step + 1e-12)):
                a, cur_score, max_idx = cand, cand_score, cand_idx

        # ---- random mass transfer ----
        if rng.random() < prob_move:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.3
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                max2, idx2, _ = _conv_max_and_index(cand2)
                score2 = 2.0 * n * max2
                if score2 < cur_score:
                    a, cur_score, max_idx = cand2, score2, idx2
                    if score2 < best_score:
                        best, best_score = cand2.copy(), score2

        # ---- occasional smoothing (3‑point kernel) ----
        if rng.random() < prob_smooth:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)
            max_s, idx_s, _ = _conv_max_and_index(smooth)
            score_s = 2.0 * n * max_s
            if score_s < cur_score:
                a, cur_score, max_idx = smooth, score_s, idx_s
                if score_s < best_score:
                    best, best_score = smooth.copy(), score_s

        # ---- step‑size update ----
        step = max(min_step, step * decay)

        # ---- early exit guard ----
        if time.monotonic() + 0.5 > deadline:
            break

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ---- initialise base vector -------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.maximum(base, 0.0)
    else:
        # fall back to a deterministic construction if the parent is absent
        n = 2000
        base = _sqrt_series(n)
    base = _project_simplex(base)

    # ---- keep a safe fallback ----------------------------------------------------
    best_global = base.copy()
    # (evaluate to guard against accidental INF – should be finite)
    try:
        sc = evaluate_sequence([float(x) for x in best_global])
        if math.isfinite(sc):
            best_score = sc
        else:
            best_score = float('inf')
    except Exception:
        best_score = float('inf')

    # ---- first long optimisation (≈ half of the budget) -------------------------
    mid_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.55
    cand, cand_score = _optimize(base, mid_deadline, rng)
    if cand_score < best_score:
        best_global, best_score = cand, cand_score

    # ---- secondary searches with perturbations ---------------------------------
    while time.monotonic() + 0.5 < deadline:
        # generate a new start: perturb current best or random
        if rng.random() < 0.6:
            # perturb around the current best
            noise = rng.normal(scale=0.02, size=best_global.shape)
            start = best_global + noise
            start = np.clip(start, 0.0, None)
        else:
            # completely random start
            start = rng.random(best_global.shape)
        start = _project_simplex(start)

        # run a modest optimisation (fixed iteration budget)
        it_budget = 15000
        cand, cand_score = _optimize(start, deadline, rng, max_iters=it_budget)
        if cand_score < best_score:
            best_global, best_score = cand, cand_score

        # occasional early exit if we are very close to theoretical limit
        if best_score < 1.50274:
            break

    return [float(x) for x in best_global]
