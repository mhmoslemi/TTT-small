import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a sequence normalized to sum = 1
# ------------------------------------------------------------
def _score(a: np.ndarray, conv: np.ndarray = None) -> float:
    """Return 2*n*max(conv) assuming sum(a)=1."""
    n = a.shape[0]
    if conv is None:
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
