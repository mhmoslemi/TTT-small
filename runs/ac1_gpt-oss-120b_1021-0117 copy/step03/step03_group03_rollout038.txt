def _proj_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Uses a stochastic gradient procedure with simplex projection,
    occasional jitter and occasional random restarts.
    Returns the best sequence discovered before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # keep a safety margin

    # ------------------------------------------------------------------
    # 2.  Determine length and initialise a starting point
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        start_vec = parent.copy()
    except Exception:                         # fallback to uniform
        n = 256
        start_vec = np.ones(n, dtype=np.float64)

    start_vec = np.maximum(start_vec, 0.0)
    s = start_vec.sum()
    if s < 1e-12:
        start_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        start_vec = start_vec / s                     # normalise to sum = 1

    # ------------------------------------------------------------------
    # 3.  Prepare utilities (FFT size, convolution helper)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())    # next power of two

    def _conv_max(arr: np.ndarray) -> float:
        """Return max of the linear self‑convolution of arr (sum(arr)==1)."""
        fa = np.fft.rfft(arr, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return float(conv.max())

    # ------------------------------------------------------------------
    # 4.  Initialise best candidate (uniform vs parent)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    best = uniform.copy()
    best_score = 2.0 * n * _conv_max(uniform)

    # try parent if it exists and is better than uniform
    parent_score = 2.0 * n * _conv_max(start_vec)
    if parent_score < best_score:
        best = start_vec.copy()
        best_score = parent_score

    # current working point
    a = best.copy()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        cand /= cand.sum()

        if score < best_score:

    # ------------------------------------------------------------------
    eta0 = 0.6                # initial step size (will decay)
