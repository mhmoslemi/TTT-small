def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1,n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x >= 0, sum x = 1}."""
    # works for 1-D arrays
    w = np.sort(v)[::-1]
    css = np.cumsum(w) - 1
    ind = np.arange(1, len(v) + 1)
    cond = w - css / ind > 0
    if not np.any(cond):
        # all zero? return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x>=0, sum(x)=1}. """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    u = np.sort(v)[::-1]                     # descending sort
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        # all entries would become zero; return uniform distribution
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(cond)[0][-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(v: np.ndarray) -> np.ndarray:
    """Full linear convolution of v with itself using a power‑of‑2 FFT."""
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # smallest power of two >= 2n-1
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv

def _score(v: np.ndarray) -> float:
    """Fast AC1 upper‑bound score (identical to evaluate_sequence for non‑negative vectors)."""
    n = v.shape[0]
    s = v.sum()
    if s <= 0.0:
        return float('inf')
    conv = _conv_fft(v)
    max_conv = conv.max()
    if max_conv < 0.0:
        max_conv = 0.0
    return 2.0 * n * max_conv / (s * s)
