import time

# ----------------------------------------------------------------------
# Simple non‑negative normalization (projects onto the simplex)
# ----------------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Clamp to >=0 and renormalize so sum == 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v[:] = 1.0 / v.size
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Score used by the official evaluator (exact same formula)
# ----------------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = vec.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (s * s))


# ----------------------------------------------------------------------
# Analytic square‑root series (fallback initializer)
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ----------------------------------------------------------------------
# Core optimiser – soft‑max weighted gradient descent + occasional moves
# ----------------------------------------------------------------------
def _optimize(start_vec: np.ndarray, deadline: float, rng: np.random.Generator):
    a = start_vec.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score

    # hyper‑parameters
    step = 0.30          # initial step size
    min_step = 1e-7
    decay = 0.9995
    prob_random = 0.03   # random mass transfer
    prob_smooth = 0.005  # small smoothing
    momentum = np.zeros_like(a)
    mom_decay = 0.9
    beta = 5.0           # soft‑max temperature
    beta_inc = 1.001
    beta_max = 200.0

    while time.monotonic() < deadline:
        # ---------- convolution of a with itself ----------
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ---------- soft‑max weights ----------
        # w_i ∝ exp(beta * (conv_i - max_val))
        exps = np.exp(beta * (conv - max_val))
        w = exps / exps.sum()

        # ---------- gradient of weighted sum ----------
        fw = np.fft.rfft(w, n=L)
        # correlation = ifft(conj(fa) * fw)
        corr_full = np.fft.irfft(np.conj(fa) * fw, n=L)
        grad = 2.0 * corr_full[:n]   # only first n entries are needed

        # ---------- momentum update ----------
        momentum = mom_decay * momentum + (1.0 - mom_decay) * grad

        # ---------- candidate & projection ----------
        cand = a - step * momentum
        cand = _normalize(cand)

        cand_score = _score(cand)

        # ---------- accept if better ----------
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            # occasional uphill move (tiny probability)
            if rng.random() < 0.02:
                a = cand
                cur_score = cand_score

        # ---------- random mass transfer ----------
        if rng.random() < prob_random:

            if a[i] > 0.0:
                j = rng.integers(0, n - 1)

                a = smooth
                cur_score = smooth_score

                    best_score = smooth_score

        # ---------- schedule updates ----------

        beta = min(beta_max, beta * beta_inc)

        # guard against overrunning the deadline
        if time.monotonic() + 0.5 > deadline:
