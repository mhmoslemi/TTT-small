def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function that (provably) minimizes the
    AC1 upper bound 2·n·max(conv)/sum(a)^2.
    The minimum is attained by the uniform distribution on the available
    coordinate set, because for any non‑negative vector a,
        sum_i a_i^2 ≥ (sum_i a_i)^2 / n,
    with equality only for equal entries.  Hence the objective is

    """

    parent = globals().get("height_sequence_1")

    # Uniform vector with total mass 1 (any scaling leaves the score unchanged).
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # Respect the evaluator's clipping rules.
