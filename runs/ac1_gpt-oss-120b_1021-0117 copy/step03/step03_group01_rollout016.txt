import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:          # extremely unlikely, but guard against it
        w[:] = 1.0 / n
    return w

# ----------------------------------------------------------------------
# Softmax weights for a given temperature (beta)
# ----------------------------------------------------------------------
def _softmax_weights(x: np.ndarray, beta: float) -> np.ndarray:
    y = beta * x
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

# ----------------------------------------------------------------------
# Core optimisation loop (matrix‑free, FFT based)
# ----------------------------------------------------------------------
def _optimize(start: np.ndarray, deadline: float, rng: np.random.Generator):
    a = start.copy()
    n = a.size
    # FFT length large enough for a (n) * w (2n‑1) convolution -> 3n‑2
    L = 1 << ((3 * n - 2).bit_length())

    # hyper‑parameters
    step = 0.4
    step_min = 1e-5
    step_decay = 0.9996
    momentum_coeff = 0.9
    momentum = np.zeros_like(a)

    beta = 0.5
    beta_max = 20.0
    beta_inc = (beta_max - beta) / 2000.0  # gradual increase

    # current best
    Fa = np.fft.rfft(a, L)
    conv_full = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
    cur_score = 2.0 * n * conv_full.max()
