import time
import math
import numpy as np

def _conv_fft(v: np.ndarray, L: int) -> np.ndarray:
    """Convolution of v with itself using FFT of length L."""
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * v.size - 1]
    return conv

def _score(v: np.ndarray, n: int, L: int) -> float:
    """Compute the AC1 bound for a non‑negative vector v."""
    conv = _conv_fft(v, L)
    max_b = conv.max()
    s = v.sum()
    if s <= 0.0:
        return math.inf
    return 2.0 * n * max_b / (s * s)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    # based on the algorithm of Wang & Carreira‑Perpiñán (2013)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # ensure exact sum=1 (within floating tolerance)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w.fill(1.0 / w.size)
    return w

def _gradient_from_active(cur: np.ndarray, active: np.ndarray, n: int) -> np.ndarray:
    """Gradient of the max‑convolution term w.r.t. cur, summed over active indices."""
    g = np.zeros_like(cur)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * cur[k - idx]
    return g

    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The routine respects the given time budget and keeps the best solution found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # small safety margin

    # ------------------------------------------------------------------
    # Initial vector (use parent if present, otherwise a simple analytic shape)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: a smooth sine‑window shape
        n0 = 256
        i = np.arange(n0, dtype=np.float64)
        a0 = np.sin(math.pi * (i + 0.5) / n0) ** 2
    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 <= 0.0:
        a0.fill(1.0 / a0.size)
    else:
        a0 /= s0

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())     # power‑of‑two FFT length

    # ------------------------------------------------------------------
    # Optimisation state
    # ------------------------------------------------------------------
    cur = a0.copy()
    cur_sc = _score(cur, n, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    eps = 1e-8
    lr0 = 0.5
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1_pow = 1.0
    beta2_pow = 1.0

    it = 0
    no_improve = 0
    K = 30                     # number of active convolution indices
    target = 1.50274349

        # --------------------------------------------------------------

        # --------------------------------------------------------------
