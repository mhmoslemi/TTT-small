import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound using a soft‑max weighted
    gradient and Adam updates, with simplex projection to enforce non‑negativity
    and unit sum. Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Time management
    # ------------------------------------------------------------------
    margin = 5.0                       # safety margin before hard termination
    deadline = time.monotonic() + max(0.0, float(budget_s) - margin)

    # ------------------------------------------------------------------
    # 2. Initialise sequence (fallback if parent unavailable)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = parent.copy()
    except Exception:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                      # normalise to sum = 1

    # ------------------------------------------------------------------
    # 3. Compute common FFT sizes
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1                     # length of full self‑convolution
    conv2_len = n + conv_len - 1             # length needed for a * w_rev
    fft_len2 = 1 << ((conv2_len - 1).bit_length())   # power‑of‑2 for all FFTs

    # ------------------------------------------------------------------
    # 4. Helper to evaluate current score (sum = 1 ⇒ 2·n·max_conv)
    # ------------------------------------------------------------------
    def _current_score(x):
        fa = np.fft.rfft(x, n=fft_len2)
        conv_full = np.fft.irfft(fa * fa, n=fft_len2)[:conv_len]
        return 2.0 * n * conv_full.max(), conv_full, fa

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best = a.copy()
    best_score, conv_best, _ = _current_score(best)

    # compare with uniform distribution as a cheap fallback
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_score, _, _ = _current_score(uniform)
    if uni_score < best_score:
        best = uniform.copy()
        best_score = uni_score
        a = uniform.copy()

    # ------------------------------------------------------------------
    # 6. Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    target_score = 1.50274349
    lr = 0.05                     # base learning rate for Adam
    beta1 = 0.9                   # Adam first‑moment decay
    beta2 = 0.999                 # Adam second‑moment decay
    eps = 1e-8                    # Adam numerical stability

    # ------------------------------------------------------------------
    return [float(v) for v in best]
