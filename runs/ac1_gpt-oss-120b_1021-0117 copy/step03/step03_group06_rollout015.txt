def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np

    # random generator
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # length
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    # FFT length
    L = 1 << ((2 * n - 1).bit_length())

    # helper conv function
    def conv_fft(a):
        fx = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fx * fx, n=L)[: 2 * n - 1]
        return conv

    # score using sum=1 normalization
    def score_of(a, conv=None):
        # a assumed normalized to sum=1
        if conv is None:
            conv = conv_fft(a)
        return 2.0 * n * conv.max()

    # initial base sequence
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = np.full(n, 1.0 / n, dtype=np.float64)
    # enforce nonneg
    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s <= 0:
        cur = np.full_like(cur, 1.0 / n)
    else:
        cur = cur / s

    # candidates from param families
    best = cur.copy()
    best_sc = score_of(best, conv_fft(best))

    # exponential candidates
    lam_vals = np.linspace(0.0001, 0.01, 30)  # 30 candidates
    for lam in lam_vals:
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        sc = score_of(a, conv_fft(a))
        if sc < best_sc:
            best_sc = sc
            best = a.copy()

    # power-law candidates
    alpha_vals = np.linspace(0.1, 2.0, 30)
    for alpha in alpha_vals:
        a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), alpha)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        sc = score_of(a, conv_fft(a))
        if sc < best_sc:
            best_sc = sc
            best = a.copy()

    # maybe triangular shape
    tri = np.minimum(np.arange(1, n+1), np.arange(n, 0, -1))
    a = tri.astype(np.float64)
    a /= a.sum()
    sc = score_of(a, conv_fft(a))
    if sc < best_sc:
        best_sc = sc
        best = a.copy()

    # set cur to best
    cur = best.copy()
    cur_sc = best_sc

    # optimization parameters
    it = 0
    alpha0 = 0.6

import time, math, numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution of a using an FFT of length L."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    return conv

def _score(a: np.ndarray, conv: np.ndarray = None) -> float:
    """Score = 2·n·max(conv)  (a is assumed to sum to 1)."""
    n = a.size
    if conv is None:
        raise ValueError
    return 2.0 * n * conv.max()

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a non‑negative step‑function with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a short safety margin
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # Determine length and initialise a feasible vector.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    cur = cur / s if s > 0 else np.full_like(cur, 1.0 / n)

    # FFT length (next power of two)
    L = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Explore a few parametric families (exponential, power‑law,
    # triangular) and keep the best as a starting point.
    # --------------------------------------------------------------
    best = cur.copy()
    best_sc = _score(best, _conv_fft(best, L))

    # exponential decay a_i = exp(-lam * i)
    for lam in np.linspace(0.0001, 0.0100, 30):
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a /= a.sum()
        sc = _score(a, _conv_fft(a, L))
        if sc < best_sc:
            best_sc, best = sc, a.copy()

    # power‑law decay a_i = (i+1)^(-alpha)
    for alpha in np.linspace(0.1, 2.0, 30):
        a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), alpha)
        a /= a.sum()
        sc = _score(a, _conv_fft(a, L))
        if sc < best_sc:
            best_sc, best = sc, a.copy()

    # simple triangular profile
    tri = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1))
    a = tri.astype(np.float64)
    a /= a.sum()
    sc = _score(a, _conv_fft(a, L))
    if sc < best_sc:
        best_sc, best = sc, a.copy()

    # initialise the iterative optimiser
    cur = best.copy()
    cur_sc = best_sc
    it = 0
    alpha0 = 0.6          # step‑size prefactor for mirror descent
    eps_move = 0.25       # max fraction of mass moved in a random transfer
    jitter_scale = 1e-4
    no_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop (sub‑gradient + random exploration).
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and active index -----
        conv = _conv_fft(cur, L)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # ----- sub‑gradient of the active convolution entry -----
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]

        # ----- mirror‑descent (exponentiated) step -----
        eta = alpha0 / math.sqrt(it)
        mul = np.exp(-eta * g)
        cand = cur * mul
        cs = cand.sum()
        cand = cand / cs if cs > 0 else np.full_like(cand, 1.0 / n)

        cand_conv = _conv_fft(cand, L)
        cand_sc = _score(cand, cand_conv)

        # ----- acceptance (deterministic + occasional uphill) -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (eta + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        if cand_sc < best_sc:
            best_sc, best = cand_sc, cand.copy()

        # ----- random mass‑transfer moves (two tries per iteration) -----
        for _ in range(2):
            if rng.random() < 0.12:
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    delta = cur[i] * rng.random() * eps_move
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = np.maximum(tmp, 0.0)
                    ts = tmp.sum()
                    if ts > 0.0:
                        tmp = tmp / ts
                        tmp_conv = _conv_fft(tmp, L)
                        tmp_sc = _score(tmp, tmp_conv)
                        if tmp_sc < cur_sc:
                            cur, cur_sc = tmp, tmp_sc
                            no_improve = 0
                        if tmp_sc < best_sc:
                            best_sc, best = tmp_sc, tmp.copy()

        # ----- jittered restart from the current best point -----
        if it % 2500 == 0:
            jitter = rng.normal(scale=jitter_scale, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)

                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best = tmp_sc, tmp.copy()
                no_improve = 0

            ts = tmp.sum()
            if ts > 0.0:
                tmp = tmp / ts
                cur = tmp
                cur_sc = _score(cur, _conv_fft(cur, L))
                no_improve = 0

        # ----- early exit if target achieved -----
        if best_sc <= 1.50274349:
            break

    # Return the best sequence as a plain Python list
    return [float(x) for x in best]
