import time
import numpy as np

def _proj_simplex(v):
    # Euclidean projection onto probability simplex.
    n = v.shape[0]
    # Small optimization: if sum already 1 and all >=0, return copy.
    if v.min() >= 0 and abs(v.sum() - 1.0) <= 1e-12:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    n = v.shape[0]
    if v.min() >= 0 and abs(v.sum() - 1.0) <= 1e-12:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Returns a non‑negative step‑function (simplex‑normalised) that minimizes
    the evaluator's score, staying within the allocated time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # -------------------------------------------------------------
    # 1.  Acquire a baseline sequence (parent if available)
    # -------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = parent.copy()
    except Exception:
        n = 256
        a = rng.exponential(1.0, size=n)

    # Normalise to the probability simplex (sum = 1, non‑negative)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s

    # -------------------------------------------------------------
    # 2.  Pre‑compute FFT size for linear convolution
    # -------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # -------------------------------------------------------------
    # 3.  Initialise bookkeeping
    # -------------------------------------------------------------
    best = a.copy()
    fa = np.fft.rfft(best, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    best_score = float(2.0 * n * conv.max())

    # uniform fallback (sometimes better than the raw parent)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    fu = np.fft.rfft(uniform, n=fft_len)
    conv_u = np.fft.irfft(fu * fu, n=fft_len)[:conv_len]
    uniform_score = float(2.0 * n * conv_u.max())
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # -------------------------------------------------------------
    # 4.  Optimisation parameters
    # -------------------------------------------------------------
    eta0 = 0.20          # initial step magnitude
    beta = 0.90          # momentum decay
    jitter_every = 300   # iteration interval for multiplicative jitter
    jitter_std = 0.02    # logarithmic jitter std‑dev
    max_stale = 2500     # iterations without improvement before restart
    target_score = 1.50274349
    top_k = 3            # number of top convolution peaks used for gradient

    v = np.zeros_like(a)   # momentum buffer
    it = 0
    stale = 0

    # -------------------------------------------------------------
    # 5.  Helper to accumulate gradient from a set of peak indices
    # -------------------------------------------------------------
    def accumulate_gradient(peaks):
        """Return averaged gradient contributed by the peaks."""
        g = np.zeros_like(a)
        for m in peaks:
            low = max(0, m - (n - 1))
            high = min(n - 1, m)
            if low > high:
                continue
            idx = np.arange(low, high + 1, dtype=np.int64)
            partner = m - idx
            g[idx] += 2.0 * a[partner]

    while time.monotonic() < deadline:

            best = a.copy()

        if conv_len > top_k:
            peak_idxs = np.argpartition(conv, -top_k)[-top_k:]
        else:

        g = accumulate_gradient(peak_idxs)

    # -------------------------------------------------------------

    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:                # safety fallback

    return [float(x) for x in best]
