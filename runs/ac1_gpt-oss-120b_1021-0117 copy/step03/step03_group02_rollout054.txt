def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    # initialization
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = np.ones(256, dtype=np.float64)
    # project onto simplex
    a = _project_simplex(a / max(a.sum(), 1e-12))
    n = a.size
    # FFT length
    L = 1 << ((2*n - 1).bit_length())
    
    # compute initial score
    max_val, _ = _conv_full_max(a, L)
    best = a.copy()
    best_score = float(2 * n * max_val / (a.sum() ** 2))
    cur_score = best_score
    
    # parameters
    step = 0.2
    decay = 0.9995
    min_step = 1e-6
    beta = 0.9   # momentum
    vel = np.zeros_like(a)
    alpha = 12.0  # softmax sharpness
    K = max(20, int(0.02 * (2*n-1)))  # top lags count
    prob_rand = 0.025
    no_improve = 0
    max_no_improve = 250
    
    # main loop
    while time.monotonic() < deadline:
        max_val, conv = _conv_full_max(a, L)
        # Determine top indices
        if K >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            # get indices of top K values
            part = np.argpartition(conv, -K)
            top_idx = part[-K:]
        # compute softmax weights
        # subtract max to avoid overflow
        w_raw = np.exp(alpha * (conv[top_idx] - max_val))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum
        # compute gradient
        g = np.zeros_like(a)
        for idx, wi in zip(top_idx, w):
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            if lo <= hi:
                # indices j = lo..hi
                # correspond to a[idx - j] values
                j_slice = slice(lo, hi+1)
                a_slice = a[idx - np.arange(lo, hi+1)]
                g[j_slice] += wi * 2.0 * a_slice
        # momentum
        vel = beta * vel - step * g
        cand = a + vel
        cand = _project_simplex(cand)
        # evaluate candidate
        max_val_c, _ = _conv_full_max(cand, L)
        cand_score = float(2 * n * max_val_c / (cand.sum() ** 2))
        improved = False
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            improved = True
        else:
            # occasional uphill
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-12)):
                a = cand
                cur_score = cand_score
                improved = True
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            no_improve = 0
        else:
            if not improved:
                no_improve += 1
        # random perturb
        if rng.random() < prob_rand:
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.5
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                max_val2, _ = _conv_full_max(cand2, L)
                cand2_score = float(2 * n * max_val2 / (cand2.sum() ** 2))
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()
                        no_improve = 0
        # decay step
        step = max(min_step, step * decay)
        # restart if stuck
        if no_improve >= max_no_improve:
            # restart from random simplex
            a = rng.random(n)
            a = _project_simplex(a)
            vel = np.zeros_like(a)
            step = 0.2
            no_improve = 0
        # early exit if near deadline
        if time.monotonic() + 0.5 > deadline:
            break
    
    return [float(x) for x in best]

def _conv_full_max(a: np.ndarray, L: int):
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2*a.size - 1]
    return float(conv.max()), conv

import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection only defined for 1‑D vectors")
    n = v.shape[0]
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Full linear convolution via FFT (returns max value and the full array)
# ------------------------------------------------------------
def _conv_full_max(a: np.ndarray, L: int):
    """Convolution a * a (full) using an FFT of length L.
    Returns (max value, full convolution array)."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv

# ------------------------------------------------------------
# Main optimisation entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ---- initialise -------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:

        a = np.ones(128, dtype=np.float64)
    # project onto the simplex (sum = 1, non‑negative)
    a = _project_simplex(a / max(a.sum(), 1e-12))
    n = a.size
    # power‑of‑two FFT length for the convolution
    L = 1 << ((2 * n - 1).bit_length())

    # ---- evaluate initial candidate ---------------------------------
    max_val, _ = _conv_full_max(a, L)
    cur_score = 2.0 * n * max_val          # sum(a) == 1 after projection
    best = a.copy()
    best_score = cur_score

    # ---- hyper‑parameters -------------------------------------------
    step = 0.30                 # initial step length

    min_step = 1e-6
    momentum = 0.9              # exponential moving average coefficient
    vel = np.zeros_like(a)      # momentum buffer
    alpha = 10.0                # soft‑max sharpness for the sub‑gradient
    prob_rand = 0.025           # chance of a random redistribution move
