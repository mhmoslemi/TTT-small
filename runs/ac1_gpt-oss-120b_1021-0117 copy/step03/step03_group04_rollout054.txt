import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (sum = 1, x >= 0)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    if v.ndim != 1:
        raise ValueError
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative coefficients (a step‑function) that
    attempts to minimise the AC1 upper bound.
    The routine is deliberately lightweight: it starts from a uniform
    distribution (or the supplied parent), evaluates a fast FFT‑based
    surrogate, and then makes a few cheap stochastic tweaks while keeping
    the best solution found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Time budget – leave a small safety margin.
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 2. Determine problem size and a reasonable start vector.
    # --------------------------------------------------------------
    n = 256                         # fallback length
    parent = None

    # The harness may provide `height_sequence_1`; use it if present.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            parent = None
        else:
            parent = np.clip(parent, 0.0, 1000.0)
            n = int(parent.size)
    except Exception:
        parent = None

    # Uniform simplex vector (sum = 1)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # 3. Fast AC1 surrogate (FFT) – score = 2 * n * max(conv)
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def _fast_score(x: np.ndarray) -> float:
        """FFT‑based evaluation of 2·n·max(a∗a)."""
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * float(conv.max())

    # --------------------------------------------------------------
    # 4. Initialise candidate set and pick the best start.
    # --------------------------------------------------------------
    candidates = [uniform]
    if parent is not None:
        # Project the parent onto the simplex – this keeps sum = 1.
        cand_parent = _proj_simplex(parent)
        candidates.append(cand_parent)

    best = None
    best_score = float("inf")
    for cand in candidates:
        sc = _fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()

    # Working copy for stochastic refinement.
    a = best.copy()

    # 5. Very cheap stochastic refinement.
    #    – Transfer a tiny amount of mass between two random indices.
    #    – Keep the best ever seen.
    # --------------------------------------------------------------
    while time.monotonic() < deadline:

        i, j = rng.choice(n, size=2, replace=False)

        max_transfer = min(a[i], 1000.0 - a[j])

            best = a_new.copy()
            a = a_new
            # optional early exit if we cross the historical target
            if best_score <= 1.50274349:

            # occasional acceptance to escape tiny plateaux
            if rng.random() < 0.001:
                a = a_new

    # --------------------------------------------------------------
    # 6. Return the best vector as plain Python floats.
    # --------------------------------------------------------------
    return [float(v) for v in best]
