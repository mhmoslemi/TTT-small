import time, numpy as np

def _next_pow2(x):
    return 1 << (x-1).bit_length()

def _conv_fft(v):
    n = v.size
    L = _next_pow2(2*n - 1)
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[:2*n-1]

def _score(v):
    conv = _conv_fft(v)
    maxc = conv.max()
    sumv = v.sum()
    if sumv < 0.01:
        return float('inf')
    return 2.0 * v.size * maxc / (sumv**2)

def _project_simplex(v):
    # v may be any shape, return projected to sum=1, nonnegative
    # Implementation of standard algorithm
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    rng = np.random.default_rng(seed)

    # initialization
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 256
        a = rng.random(n0)
    # project onto simplex sum=1
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a_sum

    n = a.size
    best_vec = a.copy()
    best_sc = _score(best_vec)

    cur = a.copy()
    cur_sc = best_sc

    # parameters
    add_alpha0 = 0.8
    mul_eta0 = 0.8
    target = 1.50274349
    no_improve = 0
    it = 0

    # Pre-allocate arrays maybe not needed
    while time.time() < deadline:
        it += 1
        # compute convolution
        conv = _conv_fft(cur)
        max_conv = conv.max()
        # early exit
        if best_sc < target:

        # active set: all indices within eps of max, or top K
        eps = max_conv * 0.001  # 0.1% tolerance

import time
import numpy as np

def _conv_fft(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[: 2 * n - 1]

def _score(v: np.ndarray) -> float:
    conv = _conv_fft(v)
    maxc = conv.max()
    s = v.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * v.size * maxc / (s * s)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x>=0, sum x = 1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w
