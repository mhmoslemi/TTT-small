def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of x with itself using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]
    return conv

def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray):
    """
    Sub‑gradient of max(conv) w.r.t. a.
    Returns (g, k) where k is the index attaining the maximum.
    """
    n = a.size
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    j = k - idx
    g = np.zeros_like(a)
    # first term: a_{k-i}
    g[idx] = a[j]
    # second term: a_i when i != k-i
    mask = (2 * idx != k)
    g[idx[mask]] += a[idx[mask]]
    return g, k

            if tmp_sc < best_sc:
                best_sc, best = tmp_sc, tmp.copy()

        # ----- stagnation restart -------------------------------------------
        if no_improve > 8000:
            cur = best + rng.normal(scale=jitter_scale * 5, size=n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur_sc = 2.0 * n * _conv_fft(cur, L).max()
            no_improve = 0

        # ----- LP‑based refinement ------------------------------------------
        if it % lp_interval == 0 and time.monotonic() + lp_time_per_call < deadline:
            g0 = _lp_refine(cur, max_conv, lp_time_per_call)
            if g0 is not None:
                for t in (0.2, 0.5, 0.8):
                    tmp = (1.0 - t) * cur + t * g0
                    tmp = np.maximum(tmp, 0.0)
                    tmp = tmp / tmp.sum()
                    tmp_sc = 2.0 * n * _conv_fft(tmp, L).max()
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best = tmp_sc, tmp.copy()

        # ----- early exit ----------------------------------------------------
        if best_sc < 1.50274349:
            break

    return [float(x) for x in best]
