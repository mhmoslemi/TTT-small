def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.5)

    # initialize a
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = np.full(2000, 1.0/2000, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full_like(a, 1.0/a.size)
    else:
        a = a / s

    n = a.size
    L_self = 1 << ((2*n-1).bit_length())
    L_corr = 1 << ((3*n-2).bit_length()) # for w_norm (2n-1) + a_rev (n) -1 = 3n-2

    best = a.copy()
    # compute initial score
    conv = _conv_two_fft(a, a, L_self)
    cur_sc = 2.0 * n * conv.max() / (a.sum()**2)
    best_sc = cur_sc

    it = 0
    eta0 = 0.5
    beta0 = 0.5
    beta_max = 10.0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # compute conv and score
        conv = _conv_two_fft(a, a, L_self)
        cur_max = conv.max()
        cur_sc = 2.0 * n * cur_max / (a.sum()**2)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Early exit if good enough
        if best_sc < 1.50274349:
            break

        # softmax weight
        beta = beta0 + (beta_max - beta0) * min(1.0, it / 5000.0)  # ramp up
        # shift for numerical stability
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # compute gradient via correlation
        a_rev = a[::-1]
        corr_full = _conv_two_fft(w_norm, a_rev, L_corr)
        g = 2.0 * corr_full[n-1:n-1+n]   # length n

        # step size
        eta = eta0 / math.sqrt(it)
        # update multiplicatively
        mul = np.exp(-eta * g)
        a = a * mul
        a_sum = a.sum()
        if a_sum == 0 or not np.isfinite(a_sum):
            a = np.full_like(a, 1.0/a.size)
        else:
            a = a / a_sum

        # occasional random transfer
        if rng.random() < 0.05:
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.3
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()

        # occasional jitter restart if stagnated
        if no_improve > 5000:
            a = best + rng.normal(scale=1e-4 * 5, size=n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

        # check time
        if time.monotonic() >= deadline:
            break

    return [float(x) for x in best]

def _conv_two_fft(x, y, L):
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)
    return conv[:len(x)+len(y)-1]

def _proj_simplex(v):
    # ensure v sum to 1, nonnegative

import time, math, numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_two_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of x and y using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)
    return conv[: len(x) + len(y) - 1]

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (non‑negative, sum = 1)."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero or negative – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------------
    # Initial vector (probability simplex)
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / total

    n = a.size
    # FFT sizes (powers of two)
    L_self = _next_pow2(2 * n - 1)          # for a*a
    L_corr = _next_pow2(3 * n - 2)          # for residual * a_rev

    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------
    conv = _conv_two_fft(a, a, L_self)

    # ----------------------------------------------------------------------

        # ----- current convolution and score -----
        conv = _conv_two_fft(a, a, L_self)
        cur_sc = 2.0 * n * conv.max() / (a.sum() ** 2)

        target = (a.sum() ** 2) / (2 * n - 1)          # uniform sum value
        residual = conv - target                       # shape (2n‑1,)
