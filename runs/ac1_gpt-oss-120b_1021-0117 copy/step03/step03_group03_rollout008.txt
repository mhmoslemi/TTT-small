                v[:] = 0.0
                it = 0
                no_improve = 0
    # Return best found
    if best_seq is None:
        # fallback uniform
        best_seq = uniform.copy()
    return [float(v) for v in best_seq]

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny sum errors
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / v.size)
    return w

def _conv_max(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Return full linear convolution a*a (via FFT) limited to length conv_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv

def _gradient_from_peaks(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (exact, using all maximal indices)."""
    n = a.shape[0]
    # tolerate numerical noise
    tol = max_val * 1e-12
    mask = np.abs(conv - max_val) <= tol
    peaks = np.where(mask)[0]
    g = np.zeros_like(a)
    for p in peaks:
        start = max(0, p - (n - 1))
        end = min(n - 1, p)
        idx = np.arange(start, end + 1, dtype=int)
        partner = p - idx
        g[idx] += a[partner]
    g *= 2.0
    return g

def _score(a: np.ndarray, max_conv: float) -> float:
    n = a.shape[0]
    s = a.sum()
    return 2.0 * n * max_conv / (s * s)
