def compute_multi_subgradient(a, conv, top_m):
    n = a.shape[0]
    # find indices with largest conv values
    idxs = np.argpartition(conv, -top_m)[-top_m:]
    g = np.zeros_like(a)

        idx = np.arange(lo, hi + 1)
        g[idx] += 2.0 * a[k - idx]   # g_i = sum over k weight * 2*a_{k-i}
    g /= top_m  # average
    return g

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real‑FFT (output length 2n‑1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective of the evaluator for a vector with sum(a)=1."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())


def _make_exponential(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)
