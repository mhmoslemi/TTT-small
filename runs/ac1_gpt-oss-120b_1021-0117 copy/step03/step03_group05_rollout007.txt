import time

def _full_convolve(v: np.ndarray) -> np.ndarray:
    """Linear convolution v*v using FFT (output length 2*len(v)-1)."""
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


def _score_of(v: np.ndarray) -> float:
    """Compute the AC1 upper bound for vector v."""
    conv = _full_convolve(v)
    n = v.shape[0]
    maxc = conv.max()
    s = v.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, len(v) + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_from_active(v: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Return ∑_{k in active} ∇(v*v)_k (sub‑gradient of the max)."""
    n = v.shape[0]

        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # derivative 2*v_{k-idx}
        g[idx] += 2.0 * v[k - idx]
    return g

        if cur[i] == 0.0:

        j = rng.integers(0, n)
        if i == j:
            continue
        delta = cur[i] * rng.random() * 0.3
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _proj_simplex(tmp)
        tmp_score = _score_of(tmp)
        if tmp_score < cur_score:
            cur, cur_score = tmp, tmp_score
            if tmp_score < best_score:
                best_vec, best_score = tmp.copy(), tmp_score

    # return the best vector encountered as a plain Python list
    return [float(x) for x in best_vec]
