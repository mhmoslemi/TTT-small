                    a, cur_sc = a_tmp, sc_tmp
                    if sc_tmp < best_sc:
                        best_sc, best_vec = sc_tmp, a_tmp.copy()
                        no_improve = 0
                else:
                    # occasional acceptance of worse move
                    if rng.random() < 0.05:
                        a, cur_sc = a_tmp, sc_tmp
            no_improve = 0

        # ------------------------------------------------------------------
        # tiny jittered restart towards the best known solution
        # ------------------------------------------------------------------
        if it % 5000 == 0 and rng.random() < 0.35:
            jitter = rng.normal(scale=1e-4, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = project_simplex(tmp)
            sc_tmp = score_from_conv(conv_full(tmp))
            if sc_tmp < cur_sc:
                a, cur_sc = tmp, sc_tmp
                if sc_tmp < best_sc:
                    best_sc, best_vec = sc_tmp, tmp.copy()

    # ----------------------------------------------------------------------
    # final quick refinement (random transfers) with any remaining time
    # ----------------------------------------------------------------------
    refin_end = time.monotonic() + min(0.3, deadline - time.monotonic())
    while time.monotonic() < refin_end:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = a[i] * rng.random() * 0.25
        a_tmp = a.copy()
        a_tmp[i] -= delta
        a_tmp[j] += delta
        a_tmp = project_simplex(a_tmp)
        sc_tmp = score_from_conv(conv_full(a_tmp))
        if sc_tmp < cur_sc:
            a, cur_sc = a_tmp, sc_tmp
            if sc_tmp < best_sc:
                best_sc, best_vec = sc_tmp, a_tmp.copy()

    # return the best vector discovered
    return [float(x) for x in best_vec]
