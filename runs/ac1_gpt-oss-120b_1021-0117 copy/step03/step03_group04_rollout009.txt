        eta = eta0 / np.sqrt(it)

        a = a - eta * v
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0:
            a = uniform.copy()
        else:
            a = a / s

            a = a / a.sum()
            v.fill(0.0)
