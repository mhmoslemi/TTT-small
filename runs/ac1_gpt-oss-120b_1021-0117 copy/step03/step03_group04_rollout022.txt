import time
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1, dtype=np.int64)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # degenerate – return uniform
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Gradient contribution from a single convolution shift
# ------------------------------------------------------------
def _add_grad_shift(g: np.ndarray, a: np.ndarray, offset: int, weight: float) -> None:
    """Add weighted sub‑gradient of b_{offset + (n‑1)} to g (in‑place)."""
    n = a.shape[0]
    if offset >= 0:
        L = n - offset
        if L <= 0:
            return
        # g[i+offset] += a[i]   for i=0..L‑1
        g[offset:] += weight * a[:L]
        # g[i]       += a[i+offset] for i=0..L‑1
        g[:L] += weight * a[offset:]
    else:
        d = -offset
        L = n - d
        if L <= 0:
            return
        # g[i]       += a[i+d]   for i=0..L‑1
        g[:L] += weight * a[d:]
        # g[i+d]     += a[i]     for i=0..L‑1
        g[d:] += weight * a[:L]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Gradient‑based optimizer for the AC1 upper bound.
    Returns the best coefficient list found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    stop_time = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_raw.ndim != 1 or parent_raw.size == 0:
            raise ValueError
        parent_raw = np.clip(parent_raw, 0.0, 1000.0)
    except Exception:
        parent_raw = None

    if parent_raw is not None:
        n = int(parent_raw.size)
        a_parent = np.maximum(parent_raw, 0.0)
        s = a_parent.sum()
        a_parent = a_parent / s if s >= 0.01 else np.full(n, 1.0 / n)
    else:
        n = 256
        a_parent = None

    # Uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # Tent (triangular) shape, edges zero
    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1.0, n - idx) - 1.0
    tent = np.maximum(tent, 0.0)
    if tent.sum() > 0:
        tent = tent / tent.sum()
    else:
        tent = uniform.copy()

    # Raised cosine (half‑wave)
    cosine = 0.5 * (1.0 + np.cos(2.0 * np.pi * (idx - (n - 1) / 2.0) / n))
    cosine = np.maximum(cosine, 0.0)
    cosine = cosine / cosine.sum()

    # Random Dirichlet‑like start

    best = None

            v.fill(0.0)
            no_improve = 0

    # If the loop ends without finding a better candidate, fall back to `best`
    final = best if best is not None else uniform
    return [float(v) for v in final]
