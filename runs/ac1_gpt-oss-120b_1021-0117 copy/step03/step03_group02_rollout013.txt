import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

        a = np.asarray(height_sequence_1, dtype=np.float64)

            a[k] = (1.0 - np.dot(a[1:k], a[k - 1 : 0 : -1])) / 2.0

    # normalise to the simplex (sum = 1, all non‑negative)

    if a_sum == 0:
        a = np.ones_like(a) / a.size
    else:
        a = a / a_sum
    a = _project_simplex(a)

    n = a.size
