import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    # based on the algorithm of Wang & Carreira-Perpinán (2013)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # all entries are equal
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation that reduces the AC1 upper bound.
    Uses a sub‑gradient step on the peak of the convolution,
    an exponential‑moving momentum term and periodic stochastic
    perturbations.  Returns the best coefficient vector discovered
    before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2.  Initialise a starting sequence (fallback to uniform)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = parent.copy()
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)

    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                         # normalise to sum = 1

    # ------------------------------------------------------------------
    # 3.  Fallback uniform sequence and baseline score
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two

    def _score(vec: np.ndarray) -> float:
        fa = np.fft.rfft(vec, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return float(2.0 * n * conv.max())

    best = a.copy()
    best_score = _score(best)
    uniform_score = _score(uniform)
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # ------------------------------------------------------------------
    # 4.  Optimisation parameters
    # ------------------------------------------------------------------
    eta0 = 0.5            # initial step size
    beta = 0.9            # momentum decay
    jitter_every = 150   # steps between multiplicative jitter
    jitter_std = 0.02    # log‑normal jitter strength
    max_stale = 3000      # steps without improvement before restart
    target_score = 1.50274349

    v = np.zeros_like(a)  # momentum buffer
    it = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # 5.  Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ------- convolution (FFT) ------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # ------- record best -----------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ------- sub‑gradient from the maximal convolution index -----
        k = int(conv.argmax())                     # index of the peak
        idx = np.arange(n)
        shift = k - idx
        valid = (shift >= 0) & (shift < n)
        # gradient = a_i + a_{k-i} (when defined)
        g = a.copy()
        g[valid] += a[shift[valid]]

        # ------- momentum --------------------------------------------
        v = beta * v + (1.0 - beta) * g
        grad = v

        # ------- step size (sqrt decay) -------------------------------
        it += 1
        eta = eta0 / np.sqrt(it)

        # ------- additive update + simplex projection ------------------
        a = _proj_simplex(a - eta * grad)

        # ------- occasional stochastic jitter -------------------------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v.fill(0.0)
            no_improve = 0

        # ------- stagnation handling (restart or random perturb) -----
        if no_improve >= max_stale:

                perturb = np.exp(rng.normal(0.0, 0.05, size=n))
                a = _proj_simplex(a * perturb)
            v.fill(0.0)
            no_improve = 0

    # ------------------------------------------------------------------
    # 6.  Return the best sequence (plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
