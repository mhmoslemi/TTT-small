   L = 1 << ((2*n - 1).bit_length())
   fv = np.fft.rfft(a, n=L)
   # conv = irfft(fv*fv)

def _score(a):
   n = len(a)
   conv = _conv_fft(a)
   return float(2.0 * n * conv.max() / (a.sum()**2))

def _make_exponential(n, r):
   a = r ** np.arange(n, dtype=np.float64)
   s = a.sum()
   if s == 0:

   else:

   return a

   if s == 0:

    """Full linear convolution a*a using a power‑of‑two FFT."""
