import time, numpy as np

def _conv_fft(v, fft_len):
    fv = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(fv * fv, n=fft_len)[: 2 * v.size - 1]
    return conv

def _score(v):
    n = v.size
    fft_len = 1 << ((2 * n - 1).bit_length())
    conv = _conv_fft(v, fft_len)
    max_b = conv.max()
    sum_a = v.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def _project_simplex(v):
    """ Euclidean projection onto simplex {x>=0, sum=1} """
    # implementation from Duchi et al.
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _gradient(v, active, fft_len):
    n = v.size
    # mask of active indices
    mask = np.zeros(2*n - 1, dtype=np.float64)
    mask[active] = 1.0

    # slice from n-1 to n-1+n

import time

def _conv_fft(v: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution v * v using FFT of fixed length."""
    fv = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(fv * fv, n=fft_len)
    return conv[: 2 * v.size - 1]

def _score(v: np.ndarray) -> float:
    """AC1 upper‑bound for a non‑negative vector."""
    n = v.size
    if n == 0:
        return float("inf")
    fft_len = 1 << ((2 * n - 1).bit_length())
    conv = _conv_fft(v, fft_len)
    max_b = conv.max()
    s = v.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]

    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]

        theta = 0.0

        # early exit if we have already beaten the target
        if best_sc < target:
            break

    # ----------------------------------------------------------------------
    # final cheap refinement (still respecting the overall deadline)
    # ----------------------------------------------------------------------
    refin_end = time.monotonic() + min(0.4, deadline - time.monotonic())
    while time.monotonic() < refin_end:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * rng.random() * 0.3
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc

    # return the best vector seen as a plain Python list
    return [float(x) for x in best_vec]
```
