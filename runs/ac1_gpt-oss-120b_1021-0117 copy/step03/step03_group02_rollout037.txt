import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    if v.ndim != 1:
        v = v.ravel()
    n = v.size
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

# ------------------------------------------------------------
# Full convolution a*a via FFT, returning maximum and whole array
# ------------------------------------------------------------
def _conv_full_max(a: np.ndarray, L: int):
    """Return (max(b), b) where b = a convolve a (full, length 2*n-1)."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # initialise a (use parent if available)
    # --------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # simple exponential decay as a fallback
        n = 256
        idx = np.arange(n)
        decay = 0.999
        a = np.exp(-0.001 * idx) * (1 - decay) / (1 - decay ** n)

    n = a.size
    a = _project_simplex(a / (a.sum() if a.sum() != 0 else 1.0))

    # --------------------------------------------------------
    # pre‑compute FFT length
    # --------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())   # next power of two

    # --------------------------------------------------------
    # evaluate initial candidate
    # --------------------------------------------------------
    max_val, _ = _conv_full_max(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # --------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------
    step0 = 0.5                # initial (unscaled) step size
    min_step = 1e-8
    K = 10                     # number of top lags used for gradient
    alpha = 12.0               # soft‑max sharpness
    prob_random = 0.03         # chance of a random mass‑transfer move
    max_no_improve = 400
    no_improve = 0
    it = 0

    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        # ---- current convolution & top‑lag selection ----
        max_val, conv = _conv_full_max(a, L)
        if K >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            part = np.argpartition(conv, -K)
            top_idx = part[-K:]

        # soft‑max weights over the selected lags
        w_raw = np.exp(alpha * (conv[top_idx] - max_val))
        w = w_raw / w_raw.sum()

        # ---- (sub)gradient of the max (softened) ----
        g = np.zeros_like(a)
        for idx, wi in zip(top_idx, w):
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            j = np.arange(lo, hi + 1)
            partner = idx - j
            g[j] += wi * 2.0 * a[partner]
            if idx % 2 == 0:
                mid = idx // 2
                if lo <= mid <= hi:
                    g[mid] -= wi * a[mid]          # correct diagonal term

        # ---- back‑tracking line search ----
        step = step0 / np.sqrt(it)
        cand_score = float('inf')
        for _ in range(12):           # at most 12 reductions of step
            if step < min_step:
                break
            cand = a - step * g
            cand = _project_simplex(cand)
            max_c, _ = _conv_full_max(cand, L)
            cand_score = 2.0 * n * max_c
            if cand_score < cur_score - 1e-14:
                break
            step *= 0.5

        improved = False
        if cand_score < cur_score - 1e-14:
            a = cand
            cur_score = cand_score
            improved = True
        else:
            # occasional uphill move (simulated‑annealing flavour)
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-12)):
                a = cand
                cur_score = cand_score
                improved = True

        # ---- keep global best ----
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            no_improve = 0
        else:
            if not improved:
                no_improve += 1
            else:
                no_improve = 0

        # ---- random mass‑transfer perturbation ----
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.4
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                max2, _ = _conv_full_max(cand2, L)
                cand2_score = 2.0 * n * max2
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best = cand2.copy()
                        best_score = cand2_score
                        no_improve = 0

        # ---- restart if stagnated ----
        if no_improve >= max_no_improve:
            a = rng.random(n)
            a = _project_simplex(a)
            it = 0
            no_improve = 0

        # ---- early exit safeguard ----
        if time.monotonic() + 0.2 > deadline:
            break

    # --------------------------------------------------------
    # final fallback (should never be needed)
    # --------------------------------------------------------
    if best is None or best_score == float('inf'):
        best = np.full(n, 1.0 / n)
    return [float(x) for x in best]
