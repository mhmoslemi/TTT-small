def simplex_proj(v):
    # Euclidean projection onto simplex (sum=1, v>=0)
    # Implementation known: sort decreasing; compute cumulative sum; find rho; compute lambda; subtract.
    # Works for any v (including negative). returns vector sum=1, all >=0.
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = v - theta
    w[w < 0] = 0
    # due to numerical might not sum exactly to 1, renormalize:
    s = w.sum()
    if s > 0: w = w / s
    else: w = np.full_like(v, 1.0/len(v))
    return w
