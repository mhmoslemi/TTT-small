```
def proj_simplex(v):
    # v: np.ndarray
    # returns projection onto simplex {x >=0, sum x = 1}
    # if sum(v)<=0, set uniform
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def proj_simplex(v):
    # Return projection of v onto simplex {x >= 0, sum x = 1}
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive? Then return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

import time, math, numpy as np

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    fx = np.fft.rfft(x, n=L)
    return np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]

def _score_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    return 2.0 * n * conv.max() / (sum_a * sum_a)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initialise
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = np.full(2000, 1.0 / 2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    cur_conv = _conv_fft(cur, L)
    cur_sc = _score_from_conv(cur_conv, n, cur.sum())
    best, best_sc = cur.copy(), cur_sc

    it = 0
    step0 = 0.8          # initial additive step size
    eps_move = 0.5       # fraction of mass moved in random transfer
    jitter_scale = 1e-5
    jitter_int = 2500
    move_prob = 0.15
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # gradient of the active maximal convolution entry
        conv = _conv_fft(cur, L)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]

        # additive sub‑gradient step with simplex projection (clip‑renorm)
        eta = step0 / math.sqrt(it)
        cand = cur - eta * g
        cand = np.maximum(cand, 0.0)
        cs = cand.sum()
        cand = cand / cs if cs > 0 else np.full_like(cand, 1.0 / n)

        cand_conv = _conv_fft(cand, L)
        cand_sc = _score_from_conv(cand_conv, n, cand.sum())

        # accept move
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (eta + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        if cand_sc < best_sc:
            best_sc, best = cand_sc, cand.copy()

        # targeted random mass transfer (active -> inactive)
        if rng.random() < move_prob and lo <= hi:
            active = np.where(cur[lo:hi + 1] > 0)[0]
            if active.size:
                i = lo + rng.choice(active)
                # pick j outside active region
                left = lo
                right = n - hi - 1
                if left + right == 0:
                    j = i
                elif left and (right == 0 or rng.random() < left / (left + right)):
                    j = rng.integers(0, left)
                else:
                    j = rng.integers(hi + 1, n)
                delta = cur[i] * rng.random() * eps_move
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                tmp = tmp / tmp.sum()
                tmp_sc = _score_from_conv(_conv_fft(tmp, L), n, tmp.sum())
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best = tmp_sc, tmp.copy()

        # jittered restart from current best
        if it % jitter_int == 0:
            jitter = rng.normal(scale=jitter_scale, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = tmp / tmp.sum()
            tmp_sc = _score_from_conv(_conv_fft(tmp, L), n, tmp.sum())
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best = tmp_sc, tmp.copy()
            no_improve = 0

        # restart after long stagnation
        if no_improve > 8000:
            jitter = rng.normal(scale=jitter_scale * 5, size=n)
            cur = best + jitter
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur_sc = _score_from_conv(_conv_fft(cur, L), n, cur.sum())
            no_improve = 0

        if best_sc < 1.50274349:
            break

    return [float(x) for x in best]
