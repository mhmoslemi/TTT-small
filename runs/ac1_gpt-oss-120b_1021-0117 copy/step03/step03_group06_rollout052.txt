def compute_gradient(a, conv, M):
    eps = 1e-12
    active = np.where(conv >= M - eps)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi+1)
        g[idx] += 2.0 * a[k - idx]
    return g

def propose_candidate(seed=42,budget_s=1100,**kwargs):
    rng = np.random.default_rng(seed)
    start=time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5) # safety margin
    # init a
    if "height_sequence_1" in globals():
         a = np.array(height_sequence_1, dtype=np.float64).copy()
    else:

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
         a = np.full_like(a, 1.0/a.size)
    else:

    n = a.size
    L = 1 << ((2*n-1).bit_length())
    # functions for conv & score
    def conv_of(v):
         fv = np.fft.rfft(v, n=L)
         conv_full = np.fft.irfft(fv * fv, n=L)[:2*n-1]
         return conv_full
    def compute_score(v, conv=None):
         if conv is None:
             conv = conv_of(v)
         return 2.0 * n * conv.max() / (v.sum()**2)
    # initial evaluation
    cur = a.copy()
    cur_conv = conv_of(cur)
    cur_score = compute_score(cur, cur_conv)
    best = cur.copy()
    best_score = cur_score
    # parameters
    iter_num = 0
    # learning schedule parameters
    alpha = 0.8
    # tolerance for gradient
    # main loop
    while time.monotonic() < deadline:
        iter_num += 1
        # compute active shift(s)
        M = cur_conv.max()
        # gradient
        g = np.zeros_like(cur)

import time, math, numpy as np

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of x with itself using an FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    return np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]

def _score_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Exact score used by the evaluator."""
    return 2.0 * n * conv.max() / (sum_a * sum_a)

def _gradient_active(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """Sub‑gradient of the max‑convolution at the active shift(s)."""
    n = a.shape[0]
    # indices whose convolution value is within a tiny absolute tolerance of the maximum
    tol = max(1e-12, max_val * 1e-12)
    active = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = k - (n - 1)
        if lo < 0:
            lo = 0
        hi = k
        if hi >= n:
            hi = n - 1
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    return g
