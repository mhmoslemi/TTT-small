import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    """
    Heuristic optimizer for the AC1 bound.
    Returns a list of non‑negative coefficients (sum ≈ 1) that

    def fast_score(x: np.ndarray) -> float:
