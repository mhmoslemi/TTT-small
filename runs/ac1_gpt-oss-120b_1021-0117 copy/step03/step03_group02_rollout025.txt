    deadline = time.monotonic() + max(0, budget_s-2)

    if height_sequence_1 exists:

    # Normalize to sum 1
    a = np.maximum(a, 0)
    a_sum = a.sum()
    if a_sum ==0: a = np.ones_like(a) / a.size

    n = a.size

    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    # compute initial epsilon etc.

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    t = 0

    # schedule gamma
    gamma = 8.0
    gamma_inc = 0.02  # increase gradually each iteration up to maybe 200

    while time.monotonic() < deadline:
        # compute conv for scoring, also for gradient
        # Use FFT length L = next power of two >=2n-1

        fv = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]

        max_c = conv.max()
        # current score = 2*n*max_c
        cur_score = 2.0 * n * max_c

        # update best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # compute gradient using softmax weighting across all lags
        # compute w: shape len conv
        # Use gamma (maybe increase)
        # To avoid overflow, subtract max:
        scaled = gamma * (conv - max_c)
        # Clip extreme to avoid overflow maybe

        w_exp = np.exp(scaled - np.max(scaled)) # shift for numeric stability
        w = w_exp / w_exp.sum()

        # compute gradient via convolution of a with reverse w
        # Determine FFT length L2 = next power of two >= n + (2n-1) -1 = 3n-2

        a_f = np.fft.rfft(a, n=L2)
        w_rev_f = np.fft.rfft(w[::-1], n=L2)
        grad_full = np.fft.irfft(a_f * w_rev_f, n=L2)
        # Extract slice
        start = (2*n - 1) - 1  # len_w -1
        grad = 2.0 * grad_full[start : start + n]

        # Adam update
        t += 1
        lr = 0.1 / (1.0 + 0.01 * t)  # decreasing
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)

def _project_simplex(v):
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size > 0:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.35
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)

        # ----- increase smoothing parameter (sharpen) -----
        gamma = min(250.0, gamma + gamma_inc)

        # ----- restart if we are stuck for a long time -----
        if no_improve >= max_no_improve:
            a = rng.random(n)
            a = _project_simplex(a)
            m.fill(0.0)
            v_vec.fill(0.0)
            t = 0
            gamma = 5.0
            no_improve = 0

        # ----- deadline guard -----
        if time.monotonic() + 0.5 > deadline:
            break

    # return the best sequence found as plain Python floats
    return [float(x) for x in best]
