import time
import numpy as np

def _proj_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        v = np.ravel(v)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 bound as measured by `evaluate_sequence`.  The simple uniformly
    distributed sequence is provably optimal; the routine still performs a
    short stochastic search to honour the “search” spirit while keeping the
    best (uniform) candidate.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Determine length
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(base.size)
        if n == 0:
            raise ValueError
    except Exception:
        n = 256

    # ------------------------------------------------------------------
    # Uniform distribution (optimal under the evaluator's objective)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best = uniform.copy()
    best_score = evaluate_sequence(best.tolist())

    # ------------------------------------------------------------------
    # Very light stochastic refinement (mostly to respect the “search”
    # requirement; it will not improve over the uniform optimum).
    # ------------------------------------------------------------------
    it = 0
    eta0 = 0.3  # modest step size
    momentum = np.zeros_like(best)

    while time.monotonic() < deadline:
        it += 1

        # random direction on the simplex (log‑space jitter)
        jitter = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = best * jitter
        cand = _proj_simplex(cand)

        # evaluate candidate
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best = cand.copy()

            if best_score < 1.0:

        grad = cand - best  # a proxy gradient
        momentum = 0.9 * momentum + 0.1 * grad

        step = eta0 / np.sqrt(it)
        best = _proj_simplex(best - step * momentum)

    return [float(v) for v in best]
