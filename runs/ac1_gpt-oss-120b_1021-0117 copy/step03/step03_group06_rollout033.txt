import time, math, numpy as np

def _next_pow2(min_len: int) -> int:
    """Return the smallest power of two >= min_len."""
    return 1 << ((min_len - 1).bit_length())

def _self_conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of x with itself (full length) via FFT of size L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]
    return conv

def _corr_fft(x: np.ndarray, y_rev: np.ndarray, L: int) -> np.ndarray:
    """Full linear correlation of x with y (using y reversed) via FFT of size L."""
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y_rev, n=L)
    corr = np.fft.irfft(fx * fy, n=L)[: x.size + y_rev.size - 1]
    return corr

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function that lowers the AC1 bound.
    Returns a plain Python list of coefficients.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    # ----------------------------------------------------------------------
    # Initial vector (use parent if available)
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    cur = np.maximum(cur, 0.0)                # enforce non‑negativity
    s = cur.sum()
    if s < 0.01:                              # avoid the infinitesimal case
        cur = np.full_like(cur, 1.0 / cur.size)
    else:
        cur /= s                               # normalise to a probability vector

    n = cur.size
    L = _next_pow2(3 * n - 2)                 # length sufficient for all FFTs

    # initial convolution and score (sum is 1, so score = 2*n*max_conv)
    conv_cur = _self_conv_fft(cur, L)
    cur_score = 2.0 * n * conv_cur.max()
    best = cur.copy()
    best_score = cur_score

    # algorithmic parameters
    eta0 = 0.5               # initial step size factor
    beta0 = 0.5              # initial soft‑max sharpness
    beta_max = 20.0
    max_iters_est = 6000     # rough number of iterations we expect to afford
    beta_inc = (beta_max - beta0) / max_iters_est
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # --------------------------------------------------------------
        # Smooth max weight (soft‑max) over the current convolution
        # --------------------------------------------------------------
        beta = beta0 + beta_inc * it
        if beta > beta_max:
            beta = beta_max

        max_c = conv_cur.max()
        # stabilise exponential: subtract max_c
        exp_vals = np.exp(beta * (conv_cur - max_c))
        p = exp_vals / exp_vals.sum()                # soft‑max probabilities

        # --------------------------------------------------------------
        # Gradient of the smooth max (correlation step)
        # --------------------------------------------------------------
        rev_p = p[::-1]                              # reversed for correlation
        corr_full = _corr_fft(cur, rev_p, L)         # length = 3n-2
        # slice that corresponds to offsets 0..n‑1 (see analysis)
        slice_corr = corr_full[(n - 1):(2 * n - 1)]
        g = 2.0 * slice_corr[::-1]                   # sub‑gradient vector

        # --------------------------------------------------------------
        # Mirror‑descent update (exponentiated gradient)
        # --------------------------------------------------------------
        eta = eta0 / math.sqrt(it)
        # Clip gradient to avoid overflow in the exponent
        g_clip = np.clip(g, -10.0, 10.0)
        # Keep exponent in a safe range

        cand = cur * exp_factor
        cand_sum = cand.sum()
        if cand_sum <= 0.0 or not np.isfinite(cand_sum):
            cand = np.full_like(cur, 1.0 / n)
        else:
            cand /= cand_sum

        # --------------------------------------------------------------
        # Score of the candidate (fast via FFT)
        # --------------------------------------------------------------
        conv_cand = _self_conv_fft(cand, L)
        cand_score = 2.0 * n * conv_cand.max()

        # --------------------------------------------------------------
        # Acceptance
        # --------------------------------------------------------------
        if cand_score < cur_score:
            cur = cand
            conv_cur = conv_cand
            cur_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

        if cand_score < best_score:

            best_score = cand_score
            no_improve = 0

        # --------------------------------------------------------------
        if rng.random() < 0.015:

            cur_score = 2.0 * n * conv_cur.max()
            no_improve = 0

        # --------------------------------------------------------------
        # Early exit if target achieved
        # --------------------------------------------------------------
        if best_score <= 1.50274349:
            break

    # Return as plain Python list (the evaluator will re‑check)
    return [float(v) for v in best]
