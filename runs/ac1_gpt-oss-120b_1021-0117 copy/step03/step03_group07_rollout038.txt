def evaluate_sequence(sequence: list) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using real FFT (linear, not circular)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]

def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _make_exponential(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a

def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    a /= a.sum()
    return a

def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a

def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    a /= a.sum()
    return a

def _make_sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / s
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # -----------------------------------------------------------------
    # Initial vector (parent if present, otherwise a simple fallback)
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    else:
        n = 256
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur.size
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # -----------------------------------------------------------------
    # Try a handful of analytical families for a quick improvement
    # -----------------------------------------------------------------
    for r in np.linspace(0.90, 0.9999, 25):
        cand = _make_exponential(n, r)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    for p in np.linspace(0.2, 4.0, 25):
        cand = _make_powerlaw(n, p)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    for f in (_make_linear, _make_quadratic, _make_sqrt_series):
        cand = f(n)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()

    cur = best_vec.copy()
    cur_sc = best_sc
    cur_conv = _conv_fft(cur)

    target = 1.50274349   # stop early if we beat it

    # -----------------------------------------------------------------
    # Phase‑1 : cheap sub‑gradient (multiplicative) steps on the active lag
    # -----------------------------------------------------------------
    it = 0
    eta0 = 0.6
    while time.monotonic() < deadline:
        it += 1
        k = int(np.argmax(cur_conv))

        # sub‑gradient of the active convolution coefficient
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]

        eta = eta0 / np.sqrt(it)
        cand = cur * np.exp(-eta * g)
        cand_sum = cand.sum()
        cand = cand / cand_sum if cand_sum > 0 else np.full_like(cur, 1.0 / n)

        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc, cur_conv = cand, cand_sc, _conv_fft(cand)
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break
        else:
            # occasional noisy acceptance to escape plateaus
            if rng.random() < 0.003:
                cur, cur_sc, cur_conv = cand, cand_sc, _conv_fft(cand)

        # every few hundred iterations perform a smoothed (log‑sum‑exp) step
        if it % 400 == 0:
            beta = 0.07 * np.sqrt(it / 400.0) + 0.02
            shift = cur_conv - cur_conv.max()
            expw = np.exp(beta * shift)
            w = expw / expw.sum()                     # weights over all lags
            # gradient of the smoothed objective
            a_rev = cur[::-1]
            L2 = 1 << ((3 * n - 2).bit_length())
            fw = np.fft.rfft(w, n=L2)
            fa = np.fft.rfft(a_rev, n=L2)
            conv2 = np.fft.irfft(fw * fa, n=L2)
            grad = 2.0 * conv2[n - 1 : n - 1 + n]

            eta2 = 0.4 / np.sqrt(it)
            cand2 = cur - eta2 * grad
            cand2 = _project_simplex(cand2)
            cand2_sc = _score(cand2)
            if cand2_sc < cur_sc:
                cur, cur_sc, cur_conv = cand2, cand2_sc, _conv_fft(cand2)
                if cand2_sc < best_sc:
                    best_sc, best_vec = cand2_sc, cand2.copy()
                    if best_sc < target:
                        break

        # tiny random jitter every 1500 iterations
        if it % 1500 == 0 and rng.random() < 0.25:
            jitter = rng.normal(scale=8e-5, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp /= tmp.sum()
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc, cur_conv = tmp, tmp_sc, _conv_fft(tmp)
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    if best_sc < target:
                        break

    # -----------------------------------------------------------------
    # Phase‑2 : occasional random mass‑transfer moves (exploratory)
    # -----------------------------------------------------------------
    while time.monotonic() < deadline:
        # active lag again to guide where to move mass from
        cur_conv = _conv_fft(cur)
        k = int(np.argmax(cur_conv))
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = cur[idx] * cur[k - idx]
        i = int(idx[np.argmax(prod)]) if prod.size else int(rng.integers(0, n))
        j = int(np.argmin(cur))
        if j == i:
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1
        max_delta = cur[i] * 0.45
        if max_delta < 1e-13:
            break
        delta = max_delta * rng.random()
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break

        # rare global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=2e-5, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp /= tmp.sum()
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    if best_sc < target:
                        break

    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
