def _conv_fft(a, b):
    n = a.shape[0]
    m = b.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)

import time
import numpy as np

# ----------------------------------------------------------------------
# basic utilities (no heavy memory)
# ----------------------------------------------------------------------
def _positive_normalize(v: np.ndarray) -> np.ndarray:
    """Force non‑negative entries and renormalise to sum 1 (or uniform if zero)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v /= s
    else:
        v = np.full_like(v, 1.0 / v.size)
    return v


def _conv_fft(a: np.ndarray, b: np.ndarray = None) -> np.ndarray:
    """Full linear convolution using real FFT (matrix‑free)."""
    if b is None:
        b = a
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (sum not forced to 1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# families of analytic initialisations (used for a quick good start)
# ----------------------------------------------------------------------
def _make_exponential(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    return a / a.sum()


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    return a / a.sum()


def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()


def _make_sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # margin for safety

    # ------------------------------------------------------------------
    # initialise from parent if present, otherwise uniform
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
    else:
        cur = np.full(256, 1.0 / 256, dtype=np.float64)

    cur = _positive_normalize(cur)
    n = cur.shape[0]

    # best solution seen so far
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # ------------------------------------------------------------------
    # explore a few analytic families (cheap, deterministic)
    # ------------------------------------------------------------------
    for r in np.linspace(0.90, 0.9999, 30):
        cand = _make_exponential(n, r)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    for p in np.linspace(0.2, 4.0, 30):
        cand = _make_powerlaw(n, p)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    for maker in (_make_linear, _make_quadratic, _make_sqrt_series):
        cand = maker(n)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()

    # ------------------------------------------------------------------
    # a few random simple perturbations (uniform Dirichlet style)
    # ------------------------------------------------------------------
    for _ in range(12):
        cand = _positive_normalize(rng.random(n))
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()

    # ------------------------------------------------------------------
    # Phase 1 – smooth soft‑max gradient descent (matrix‑free)
    # ------------------------------------------------------------------
    target = 1.50274349
    a = best_vec.copy()
    it = 0
    while time.monotonic() < deadline:
        it += 1
        conv = _conv_fft(a)                         # a * a
        max_c = conv.max()
        # soft‑max over convolution values
        alpha = 5.0 + 0.02 * it                     # slowly increasing
        exp_vals = np.exp(alpha * (conv - max_c))   # safe: max term = 1
        w = exp_vals / exp_vals.sum()               # weights, sum=1

        # gradient of the smoothed objective:
        a_rev = a[::-1]
        conv2 = _conv_fft(w, a_rev)                 # length 3n‑2
        grad = 2.0 * conv2[n - 1 : n - 1 + n]        # slice to length n

        eta = 0.5 / np.sqrt(it)                     # diminishing step
        a = _positive_normalize(a - eta * grad)

        sc = _score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
            if best_sc < target:
                break

        # occasional jitter to escape plateaus
        if it % 800 == 0 and rng.random() < 0.2:
            jitter = rng.normal(scale=1e-4, size=n)
            cand = _positive_normalize(a + jitter)
            scj = _score(cand)
            if scj < best_sc:
                best_sc, best_vec = scj, cand.copy()
                a = cand

        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – focused active‑lag sub‑gradient steps (tiny fine‑tuning)
    # ------------------------------------------------------------------
    it2 = 0
    while time.monotonic() < deadline:
        it2 += 1
        conv = _conv_fft(best_vec)
        k = int(np.argmax(conv))
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(best_vec)
        g[idx] = 2.0 * best_vec[k - idx]

        eta2 = 0.12 / np.sqrt(it2)
        cand = _positive_normalize(best_vec - eta2 * g)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
            if best_sc < target:
                break

        # rare random mass transfer to explore neighbourhood
        if it2 % 2500 == 0 and rng.random() < 0.15:
            nz = np.where(best_vec > 1e-12)[0]
            if nz.size:
                i = int(rng.choice(nz))
                j = int(rng.integers(0, n - 1))
                if j == i:
                    j = (j + 1) % n
                delta = best_vec[i] * rng.random() * 0.4
                tmp = best_vec.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _positive_normalize(tmp)
                sc2 = _score(tmp)
                if sc2 < best_sc:
                    best_sc, best_vec = sc2, tmp.copy()

        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # final safety net – if everything failed fall back to uniform
    # ------------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
