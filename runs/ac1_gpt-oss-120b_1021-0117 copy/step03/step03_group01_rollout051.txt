def propose_candidate(seed=42, budget_s=1100):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256 # maybe fallback
    # initial vector
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:

        a0 = _sqrt_series(n) # maybe implement sqrt series else random.
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    best = a0.copy()
    best_score = 2.0 * n * _conv_max(best) # compute conv max

    # preallocate arrays
    momentum = np.zeros_like(a0)
    step = 0.2
    min_step = 1e-5
    decay = 0.9995
    beta = 0.9

    L = 1 << ((2*n-1).bit_length())  # for conv and gradient conv correlation
    # We'll also use same L for correlation (since w length <= 2n-1 < L)
    while time.monotonic() < deadline:
        # compute conv and max
        arr_fft = np.fft.rfft(a0, n=L)
        conv_full = np.fft.irfft(arr_fft * arr_fft, n=L)[:2*n-1]  # real
        max_val = conv_full.max()
        # compute gradient for top entries near max
        eps = max_val * 0.001  # 0.1% tolerance
        # Might adapt eps gradually: start large then shrink.
        mask = conv_full >= (max_val - eps)
        w = mask.astype(np.float64) # shape (2*n-1,)
        # compute FFT of w (padded to L)
        w_fft = np.fft.rfft(w, n=L)
        # gradient = 2 * correlation
        grad_full = np.fft.irfft(arr_fft * np.conj(w_fft), n=L)[:len(a0)]  # slice first n entries
        grad = 2.0 * grad_full
        # momentum
        momentum = beta * momentum + (1 - beta) * grad
        # step update
        cand = a0 - step * momentum
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[:2*n-1]
        cand_max = cand_conv.max()
        cand_score = 2.0 * n * cand_max  # sum=1 after projection (should be ~1)
        cur_score = 2.0 * n * max_val

        if cand_score < cur_score:
            a0 = cand
            max_val = cand_max
            # update best if improved
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            # maybe accept uphill with small probability
            if rng.random() < 0.001:
                a0 = cand
                max_val = cand_max

            if a0[i] > 0:

                    j += 1

                cand2 = a0.copy()
                cand2[i] -= delta

import time

# ----------------------------------------------------------------------
# Simplex projection (onto the probability simplex, sum = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    rho = rho[-1] if rho.size else 0
    theta = css[rho] / (rho + 1) if rho >= 0 else 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:                     # safeguard (should not happen)
        w[:] = 1.0 / n
    return w


# ----------------------------------------------------------------------
# Analytic sqrt‑series (fallback initializer)
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a
