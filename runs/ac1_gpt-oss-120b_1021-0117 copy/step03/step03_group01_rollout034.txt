def _adam_step(a, g, m, v, t, lr, beta1, beta2, eps):
    m = beta1 * m + (1 - beta1) * g
    v = beta2 * v + (1 - beta2) * (g * g)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)

    new_a = a - step

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    best = a.copy()
    best_max = _conv_max_and_index(a, L)[0]
    best_score = 2.0 * n * best_max

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    t = 0

    while time.monotonic() < deadline:
        # compute conv and max
        conv = _conv_full(a, L)
        max_val = float(conv.max())
        max_idx = int(conv.argmax())
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # gradient
        g = np.zeros_like(a)
        # compute gradient for all max indices within tolerance
        # tolerance small relative to max_val
        tol = max_val * 1e-12  # maybe adjust absolute
        idxs = np.where(conv >= max_val - tol)[0]
        if idxs.size:
            for k in idxs:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                j = np.arange(lo, hi+1)
                g[j] += 2.0 * a[k - j]
        else:
            # fallback single idx
            lo = max(0, max_idx - (n - 1))
            hi = min(n - 1, max_idx)
            if lo <= hi:
                j = np.arange(lo, hi+1)
                g[j] = 2.0 * a[max_idx - j]

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - step
        cand = _project_simplex(cand)
        # evaluate candidate
        cand_conv = _conv_full(cand, L)
        cand_max = float(cand_conv.max())
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score:
            a = cand
            # update best if needed
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            # sometimes accept uphill (simulated annealing)
            if rng.random() < 0.01 * math.exp(-(cand_score - cur_score) / max(lr, 1e-12)):
                a = cand

        # random mass transfer
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n - 1)
                if j >= i:

                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                cand2_conv = _conv_full(cand2, L)
                cand2_max = float(cand2_conv.max())
                cand2_score = 2.0 * n * cand2_max
                if cand2_score < best_score:
                    best_score = cand2_score
                    best = cand2.copy()
                    a = cand2
                elif cand2_score < cur_score and rng.random() < 0.5:
                    a = cand2

        # occasional smoothing

            kernel = np.array([0.25, 0.5, 0.25])

                a = smooth

        lr = max(0.01, lr * 0.9995)

        if time.monotonic() + 0.5 > deadline:

def _project_simplex(v):
    # Euclidean projection onto simplex sum=1
    # reference: algorithm from papers.
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho_arr = u - css / np.arange(1, len(v)+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0)
    # If all zero due to numerical issues, set uniform
    if w.sum() == 0:
        w[:] = 1.0 / len(v)
    return w
```

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:        # numerical safety
        w[:] = 1.0 / n
    return w


# ----------------------------------------------------------------------
# Full convolution via FFT (real‑valued, returns the first 2*n‑1 entries)
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Optimiser: Adam on the gradient of the current maximal convolution entry
# ----------------------------------------------------------------------
def _optimize(start_vec: np.ndarray, deadline: float, rng: np.random.Generator):
    a = start_vec.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())          # next power of two

    # best‑ever tracking
    conv = _conv_full(a, L)
    best_max = float(conv.max())
    best_score = 2.0 * n * best_max
    best = a.copy()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    t = 0

    # occasional‑move probabilities
    prob_random = 0.015
    prob_smooth = 0.006

    while time.monotonic() < deadline:
        # ----- current convolution & gradient (all maximal entries) -----
        conv = _conv_full(a, L)
        max_val = float(conv.max())
        cur_score = 2.0 * n * max_val
        max_idxs = np.where(conv >= max_val - 1e-12)[0]

        # gradient of max (subgradient)
        g = np.zeros_like(a)
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            j = np.arange(lo, hi + 1)
            g[j] += 2.0 * a[k - j]

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_conv = _conv_full(cand, L)
        cand_max = float(cand_conv.max())
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            # occasional uphill acceptance (simulated annealing style)
            if rng.random() < math.exp(-(cand_score - cur_score) / max(lr, 1e-12)):
                a = cand

        # ----- random mass transfer (exploration) -----
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.5
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)

                conv2 = _conv_full(cand2, L)
                max2 = float(conv2.max())
                score2 = 2.0 * n * max2
                if score2 < best_score:

                    best = cand2.copy()
                if score2 < cur_score:
                    a = cand2

        # ----- smoothing (3‑point kernel) -----
        if rng.random() < prob_smooth:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)

            conv_s = _conv_full(smooth, L)
            max_s = float(conv_s.max())
            score_s = 2.0 * n * max_s
            if score_s < best_score:
                best_score = score_s
                best = smooth.copy()
            if score_s < cur_score:
                a = smooth

        # ----- step‑size decay -----
        lr = max(0.01, lr * 0.9995)

        # ----- time guard -----
        if time.monotonic() + 0.5 > deadline:
            break

    return best, best_score
