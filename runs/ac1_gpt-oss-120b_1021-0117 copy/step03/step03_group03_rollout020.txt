def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)  # leave safety margin
    # initial a:
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(parent, 0.0, 1000.0)
        n = a.size
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)
    # ensure nonnegative, avoid zeros:
    a = np.maximum(a, 1e-12)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0/n)
    else:
        a = a / s
    # maintain best:
    # compute initial score
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    # compute conv
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    best = a.copy()
    best_score = float(2 * n * conv.max())
    # Also keep a fallback uniform:
    uniform = np.full(n, 1.0/n)
    fa_u = np.fft.rfft(uniform, n=fft_len)
    conv_u = np.fft.irfft(fa_u * fa_u, n=fft_len)[:conv_len]
    uniform_score = float(2 * n * conv_u.max())
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()
    # parameters
    eta0 = 5.0
    beta = 0.9
    jitter_every = 200
    jitter_std = 0.02
    max_stale = 3000
    no_improve = 0
    momentum = np.zeros_like(a)
    it = 0
    target_score = 1.50274349
    # main loop
    while time.monotonic() < deadline:
        # compute convection and max
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1
        # gradient from max indices
        # find all indices where conv equals max_val within tolerance
        # Use absolute tolerance relative to max_val:
        tol = 1e-12
        max_idxs = np.flatnonzero(np.abs(conv - max_val) <= tol)
        # compute gradient as sum contributions, scaled by 1/len
        g = np.zeros_like(a)
        for k in max_idxs:
            offset = k - (n - 1)
            if offset >= 0:
                # indices offset..n-1 get contributions from a[0:n-offset]
                g[offset:] += 2.0 * a[:n - offset]
            else:
                # indices 0..n+offset-1 get contributions from a[-offset:]
                g[:n + offset] += 2.0 * a[-offset:]
        # average gradient among multiple peaks
        if max_idxs.size > 1:
            g /= max_idxs.size
        # momentum update
        it += 1
        eta = eta0 / np.sqrt(it)
        momentum = beta * momentum + (1.0 - beta) * g
        # exponentiated gradient update
        # ensure positivity: add epsilon before exponent
        eps = 1e-30
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, eps)
        a = a / a.sum()
        # jitter
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = a * jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            momentum.fill(0.0)
            no_improve = 0
        # stagnation restart
        if no_improve >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = a * perturb
            a = np.maximum(a, eps)
            a = a / a.sum()
            momentum.fill(0.0)
            no_improve = 0
    # final return best as python list of floats
    return [float(v) for v in best]

import time
import numpy as np

def _proj_simplex(v):
    """ Euclidean projection onto the probability simplex. """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Subgradient‑based optimisation of the AC1 bound.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise sequence (fallback to uniform if parent missing)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(parent, 0.0, 1000.0)
        n = a.size
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 1e-12)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # normalise to sum = 1

    # ------------------------------------------------------------------
    # 2. Pre‑compute FFT size
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 3. Evaluate initial candidate and store the best so far
    # ------------------------------------------------------------------
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    best = a.copy()
    best_score = float(2 * n * conv.max())

    # also check a simple uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    fu = np.fft.rfft(uniform, n=fft_len)
    cu = np.fft.irfft(fu * fu, n=fft_len)[:conv_len]
    uniform_score = float(2 * n * cu.max())
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # ------------------------------------------------------------------
    # 4. Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    eta0 = 5.0            # initial step magnitude
    beta = 0.9            # momentum decay
    top_k = 5             # number of active convolution peaks
    jitter_every = 180    # steps between random perturbations
    jitter_std = 0.025    # log‑normal jitter strength
    max_stale = 3000      # restart after this many non‑improving steps
    target_score = 1.50274349

    momentum = np.zeros_like(a)
    it = 0
    stale = 0

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- convolution (FFT) ------------------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ---- keep the best ----------------------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            stale = 0
            if best_score <= target_score:
                break
        else:
            stale += 1

        # ---- sub‑gradient from the top‑k convolution peaks --------------------
        if conv.size > top_k:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        else:
            idxs = np.arange(conv.size)

        g = np.zeros_like(a)
        for k in idxs:
            offset = k - (n - 1)
            if offset >= 0:
                # g[offset:] += 2 * a[:n-offset]
                g[offset:] += 2.0 * a[:n - offset]
            else:
                # g[:n+offset] += 2 * a[-offset:]
                g[:n + offset] += 2.0 * a[-offset:]

        if idxs.size > 1:
            g /= idxs.size

        # ---- momentum update --------------------------------------------------
        it += 1
        eta = eta0 / np.sqrt(it)
        momentum = beta * momentum + (1.0 - beta) * g

        # ---- exponentiated‑gradient step (keeps positivity) -------------------
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-30)      # avoid exact zeros
        a /= a.sum()                  # project onto simplex (renormalise)

        # ---- occasional jitter to escape plateaus -----------------------------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = a * jitter
            a = np.maximum(a, 1e-30)
            a /= a.sum()
            momentum.fill(0.0)
            stale = 0

        # ---- restart if stagnant ---------------------------------------------
        if stale >= max_stale:
            a = best.copy()
            pert = np.exp(rng.normal(0.0, 0.02, size=n))
            a = a * pert
            a = np.maximum(a, 1e-30)
            a /= a.sum()
            momentum.fill(0.0)
            stale = 0

    # ------------------------------------------------------------------
    # 6. Return the best sequence as plain Python floats
    # ------------------------------------------------------------------
    return [float(v) for v in best]
