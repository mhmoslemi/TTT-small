def propose_candidate(seed=42,budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.5)
    
    # init
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate some plausible initial vector: e.g., sqrt series length 2000
        n = 2000
        a = np.ones(n, dtype=np.float64)  # placeholder
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a[:] = 1.0/a.size
    else:
        a = a / s
    n = a.size
    
    # choose FFT length
    L = 1 << ((3*n-2).bit_length())
    
    # projector function (reuse)
    def project_simplex(v):
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho_arr = np.where(u - css/np.arange(1, len(v)+1) > 0)[0]
        if rho_arr.size == 0:
            theta = 0.0
        else:
            rho = rho_arr[-1]
            theta = css[rho] / (rho+1)
        w = np.maximum(v - theta, 0.0)
        return w
    
    # scoring (fast)
    def compute_score(a_vec, conv_full=None):
        n_local = a_vec.shape[0]
        if conv_full is None:
            # conv via FFT
            a_fft = np.fft.rfft(a_vec, n=L)
            conv_full = np.fft.irfft(a_fft*a_fft, n=L)[:2*n_local-1]
        max_val = conv_full.max()
        return float(2.0*n_local*max_val)

    best = a.copy()
    best_score = compute_score(best)
    cur = a.copy()
    cur_score = best_score

    # optimizer parameters
    eps0 = 0.1
    eps_decay = 0.9997
    eps_min = 1e-8
    base_lr = 0.2
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        it += 1
        # compute convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft*a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # smoothing weight
        eps = max(eps_min, eps0 * (eps_decay ** it))
        # either softmax or active set:
        # Use top-k within eps tolerance
        # Use at least 1 index
        mask = conv >= max_val - eps
        w = np.zeros_like(conv)
        if np.any(mask):
            # uniform weight among active indices
            w[mask] = 1.0 / mask.sum()
        else:  # fallback
            w[np.argmax(conv)] = 1.0
        # compute gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)  # length >= 3n-2
        idx = (2*n-2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # adam update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g*g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = (base_lr / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + adam_eps)
        cand = cur - step
        cand = project_simplex(cand)
        cand_score = compute_score(cand)
        # accept if improvement, else maybe accept with small prob
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                no_improve = 0
        else:
            # occasional acceptance
            if rng.random() < 0.001:
                cur, cur_score = cand, cand_score
            no_improve += 1
        
        # periodic symmetrization
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = project_simplex(sym)
            sym_score = compute_score(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score
                    no_improve = 0
        
        # random perturbation if stuck
        if no_improve > 3000:
            # random mass transfer
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n-1)
                if j >= i: j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = project_simplex(tmp)
                tmp_score = compute_score(tmp)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score
                    no_improve = 0
                else:
                    no_improve = 0
        
        if best_score < 1.50274349:
            break
    
    return [float(x) for x in best]

    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size+1) > 0)[0]

# Euclidean projection onto the simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

                    no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score, _ = _score_fft(tmp, L)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- early exit if we already beat the target -----
        if best_score < 1.50274349:
            break

    # ----- final optional symmetrisation (may further lower the bound) -----
    sym = (best + best[::-1]) * 0.5
    sym = _project_simplex(sym)
    sym_score, _ = _score_fft(sym, L)
    if sym_score < best_score:
        best, best_score = sym, sym_score

    # --------------------------------------------------------
    # Return as a plain Python list of floats
    # --------------------------------------------------------
    return [float(x) for x in best]
