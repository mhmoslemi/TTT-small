import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_candidates = np.where(rho_arr > 0)[0]
    if rho_candidates.size:
        rho = rho_candidates[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # ensure exact sum = 1 (numerical errors)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    return w


# ------------------------------------------------------------
# Simple fallback initializer (sqrt series) – rarely used
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # reserve a little buffer for cleanup
    deadline = start + max(0.0, float(budget_s) - 2.0)

    # -------------------- initialise --------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = _sqrt_series(256)

    if a.size == 0:
        a = np.ones(1, dtype=np.float64)

    # normalize to the simplex (sum == 1, non‑negative)
    a = _project_simplex(a / (a.sum() if a.sum() != 0 else 1.0))

    n = a.size
    # FFT lengths (powers of two)
    L_conv = 1 << ((2 * n - 1).bit_length())          # for a*a
    L_grad = 1 << ((3 * n - 2).bit_length())          # for gradient convolution

    # initial score
    fv = np.fft.rfft(a, n=L_conv)
    conv = np.fft.irfft(fv * fv, n=L_conv)[: 2 * n - 1]
    max_conv = float(conv.max())
    cur_score = 2.0 * n * max_conv
    best = a.copy()
    best_score = cur_score

    # -------------------- optimisation parameters --------------------
    step = 0.5                # initial step size
    min_step = 1e-7
    decay = 0.9995
    alpha = 8.0               # soft‑max sharpness (starts moderate)
    alpha_target = 14.0       # final sharpness
    prob_random = 0.015       # chance for a random mass‑transfer perturbation
    no_improve_iters = 0
    max_no_improve = 500      # restart after this many non‑improving steps

    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- compute soft‑max weights for the current convolution -----
        # conv already holds a*a from previous iteration
        w_raw = np.exp(alpha * (conv - max_conv))          # max_conv = max(conv)
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # ----- gradient via convolution (full weight vector) -----
        a_rev = a[::-1]                                    # reversed a
        w_f = np.fft.rfft(w, n=L_grad)
        a_rev_f = np.fft.rfft(a_rev, n=L_grad)
        g_full = np.fft.irfft(w_f * a_rev_f, n=L_grad)
        # gradient length n starts at offset (n‑1)
        g = 2.0 * g_full[n - 1 : n - 1 + n]                # factor 2 as per sub‑gradient

        # ----- take a step and project back onto the simplex -----
        cand = a - step * g
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        fv_c = np.fft.rfft(cand, n=L_conv)
        conv_c = np.fft.irfft(fv_c * fv_c, n=L_conv)[: 2 * n - 1]
        max_conv_c = float(conv_c.max())
        cand_score = 2.0 * n * max_conv_c

        # ----- accept / reject logic -----
        improved = False
        if cand_score < cur_score:
            a, conv, max_conv, cur_score = cand, conv_c, max_conv_c, cand_score
            improved = True
        else:
            # occasional uphill acceptance (simulated‑annealing flavour)
            temp = max(step, 1e-9) * 10.0
            if rng.random() < math.exp(-(cand_score - cur_score) / temp):
                a, conv, max_conv, cur_score = cand, conv_c, max_conv_c, cand_score
                improved = True

        # ----- keep global best -----
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            no_improve_iters = 0
        else:
            if not improved:
                no_improve_iters += 1
            else:
                no_improve_iters = 0

        # ----- occasional random mass‑transfer move -----
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.3
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                fv2 = np.fft.rfft(cand2, n=L_conv)
                conv2 = np.fft.irfft(fv2 * fv2, n=L_conv)[: 2 * n - 1]
                cand2_score = 2.0 * n * float(conv2.max())
                if cand2_score < cur_score:
                    a, conv, max_conv, cur_score = cand2, conv2, float(conv2.max()), cand2_score
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()
                        no_improve_iters = 0

        # ----- step decay and soft‑max sharpening -----
        step = max(min_step, step * decay)
        if alpha < alpha_target:
            alpha = min(alpha_target, alpha + (alpha_target - alpha) * 0.001)

        # ----- restart if stuck -----
        if no_improve_iters >= max_no_improve:
            a = rng.random(n)
            a = _project_simplex(a)
            fv = np.fft.rfft(a, n=L_conv)
            conv = np.fft.irfft(fv * fv, n=L_conv)[: 2 * n - 1]
            max_conv = float(conv.max())
            cur_score = 2.0 * n * max_conv
            step = 0.5
            no_improve_iters = 0

        # ----- safeguard against overrunning the deadline -----
        if time.monotonic() + 0.5 > deadline:
            break

    # Return the best sequence found as a plain Python list
    return [float(x) for x in best]
