import time

# ----------------------------------------------------------------------
# Helper utilities (simple, no large literals)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a using a pre‑chosen FFT length."""
    fa = np.fft.rfft(a, n=L)
    full = np.fft.irfft(fa * fa, n=L)
    return full[: 2 * a.size - 1]


def _score(a: np.ndarray, L: int) -> float:
    """Fast evaluation of the objective (sum is assumed 1)."""
    n = a.size
    conv = _conv_fft(a, L)
    return float(2.0 * n * conv.max())  # sum(a)=1, so denominator is 1


def _make_exponential(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    return a / a.sum()


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    return a / a.sum()


def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()


def _make_sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a.fill(1.0 / n)
    else:
        a /= s
    return a

            else:
                tmp = np.full_like(cur, 1.0 / n)
            tmp_sc = _score(tmp, L)
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc

    # --------------------------------------------------------------
    # Fallback safety
    # --------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
