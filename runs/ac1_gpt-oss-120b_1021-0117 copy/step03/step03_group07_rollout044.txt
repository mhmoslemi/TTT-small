import time
import numpy as np

# ---------- helper utilities ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via real FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())       # next power of two
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (sum is not forced to 1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    else:

    n = cur.size
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # ---- evaluate a few analytic families for a quick boost -------------
    for r in np.linspace(0.90, 0.9999, 20):
        a = np.power(r, np.arange(n, dtype=np.float64))
        a /= a.sum()
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
    for p in np.linspace(0.2, 4.0, 20):

    target = 1.50274349
