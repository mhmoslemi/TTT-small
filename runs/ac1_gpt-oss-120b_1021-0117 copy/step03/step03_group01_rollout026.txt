    max_val, max_idx = _conv_max_and_index(a, L)

    best_local_score = cur_score

import time, math, numpy as np, scipy.sparse as sp, scipy.optimize as opt

# ----------------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w

# ----------------------------------------------------------------------
# Convolution via FFT – returns max value and its index
# ----------------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * vec.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

# ----------------------------------------------------------------------
# Score = 2 * n * max(conv)  (sum is normalised to 1)
# ----------------------------------------------------------------------
def _score(vec: np.ndarray, L: int) -> float:
    max_val, _ = _conv_max_and_index(vec, L)
    return 2.0 * vec.size * max_val

# ----------------------------------------------------------------------
# Local optimiser – projected gradient with a few stochastic tricks
# ----------------------------------------------------------------------
def _local_optimize(start: np.ndarray, deadline: float, rng: np.random.Generator):
    a = start.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    max_val, max_idx = _conv_max_and_index(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    step = 0.5
    min_step = 1e-7
    decay = 0.9995
    beta = 0.9
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            g[j] = 2.0 * a[max_idx - j]

        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        cand = _project_simplex(cand)

        cand_max, cand_idx = _conv_max_and_index(cand, L)
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            max_idx = cand_idx
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            if rng.random() < math.exp(-(cand_score - cur_score) / max(step, 1e-12)):
                a = cand
                cur_score = cand_score
                max_idx = cand_idx

        # random mass transfer
        if rng.random() < 0.05:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.4
                a2 = a.copy()
                a2[i] -= delta
                a2[j] += delta
                a2 = _project_simplex(a2)
                max2, _ = _conv_max_and_index(a2, L)
                score2 = 2.0 * n * max2
                if score2 < cur_score:
                    a = a2
                    cur_score = score2
                    if score2 < best_score:
                        best = a2.copy()
                        best_score = score2

        # occasional smoothing
        if rng.random() < 0.02:
            kern = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kern, mode='same')
            smooth = _project_simplex(smooth)
            max_s, _ = _conv_max_and_index(smooth, L)
            score_s = 2.0 * n * max_s
            if score_s < cur_score:
                a = smooth
                cur_score = score_s
                if score_s < best_score:
                    best = smooth.copy()
                    best_score = score_s

        step = max(min_step, step * decay)

        if time.time() + 0.5 > deadline:
            break

    return best, best_score

# ----------------------------------------------------------------------
# Cached convolution‑matrix index construction (rows/cols for sparse A)
# ----------------------------------------------------------------------
def _conv_matrix_indices(n: int):
    cache = getattr(_conv_matrix_indices, "_cache", {})
    if n not in cache:
        rows = np.arange(n)[:, None] + np.arange(n)[None, :]
        rows = rows.ravel().astype(np.int32)
        cols = np.tile(np.arange(n), n).astype(np.int32)
        cache[n] = (rows, cols)
        _conv_matrix_indices._cache = cache
    return cache[n]

# ----------------------------------------------------------------------
# One LP‑based improvement step (mixes the solved g into f)
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, M: float, deadline: float, t: float = 0.5, iters: int = 3):
    n = f.size
    rows, cols = _conv_matrix_indices(n)
    L = 1 << ((2 * n - 1).bit_length())
    for _ in range(iters):
        if time.time() + 5.0 > deadline:
            break
        data = np.repeat(f, n)
        A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # minimise -sum(g)  => maximise sum(g)
        res = opt.linprog(
            c, A_ub=A, b_ub=b,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True}
        )
        if not res.success:
            break
        g = res.x
        S = g.sum()
        if S <= 1.0 + 1e-12:
            break
        g_scaled = g / S
        f = (1.0 - t) * f + t * g_scaled
        M, _ = _conv_max_and_index(f, L)
    return f, M

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.time() + max(0.0, float(budget_s) - 1.0)

    # initialise ---------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: simple decreasing exponential if no parent provided
        n = 256
        a0 = np.exp(-np.arange(n) / 20.0)
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    best_vec = a0.copy()
    best_score = _score(best_vec, 1 << ((2 * a0.size - 1).bit_length()))

    # multiple local searches --------------------------------------------
    restarts = 12
    for r in range(restarts):
        if time.time() + 5.0 > deadline:
            break
        if r == 0:
            start = best_vec.copy()
        else:
            start = rng.random(a0.size)
            start = _project_simplex(start)
        cand, cand_score = _local_optimize(start, deadline, rng)
        if cand_score < best_score:
            best_vec = cand.copy()
            best_score = cand_score

    # final LP‑based refinement -------------------------------------------
    if time.time() + 5.0 < deadline:
        L = 1 << ((2 * best_vec.size - 1).bit_length())
        M, _ = _conv_max_and_index(best_vec, L)
        refined, _ = _lp_refine(best_vec, M, deadline, t=0.5, iters=3)
        refined_score = _score(refined, L)
        if refined_score < best_score:
            best_vec = refined
            best_score = refined_score

    # return plain Python list -------------------------------------------
    return [float(x) for x in best_vec]
