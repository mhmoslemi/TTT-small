def full_convolution_fft(x, y):
    n = len(x) + len(y) - 1
    L = 1 << (n - 1).bit_length() # next power of 2 >= n
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[:n]
    return conv

def _compute_gradient(a, beta):
    n = a.shape[0]
    # compute self-convolution using FFT
    L_a = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=L_a)
    b = np.fft.irfft(fa * fa, n=L_a)[:2*n-1]
    # compute softmax weights
    maxb = b.max()
    # avoid overflow:
    exp_arg = beta * (b - maxb)
    w = np.exp(exp_arg, dtype=np.float64)
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
        w_sum = 1.0
    w /= w_sum
    # compute gradient via convolution w * a_rev
    a_rev = a[::-1]
    # conv of w (len 2n-1) and a_rev (len n) => length 3n-2
    L_w = 1 << ((len(w) + n - 1).bit_length())
    fw = np.fft.rfft(w, n=L_w)
    far = np.fft.rfft(a_rev, n=L_w)
    conv = np.fft.irfft(fw * far, n=L_w)[:len(w) + n - 1]
    grad = 2.0 * conv[n-1 : n-1 + n]   # slice length n
    return grad, b

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum=1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:                     # safety fallback
        w[:] = 1.0 / n
    return w

# ------------------------------------------------------------
# Full convolution via FFT (real inputs)
# ------------------------------------------------------------
def _self_conv_fft(a: np.ndarray, L_self: int) -> np.ndarray:
    """Return full convolution a*a using FFT, length 2*len(a)-1."""
    fa = np.fft.rfft(a, n=L_self)
    conv = np.fft.irfft(fa * fa, n=L_self)
    return conv[: 2 * a.shape[0] - 1]

def _cross_conv_fft(w: np.ndarray, a_rev: np.ndarray, L_cross: int) -> np.ndarray:
    """Return full convolution w * a_rev using FFT."""
    fw = np.fft.rfft(w, n=L_cross)
    far = np.fft.rfft(a_rev, n=L_cross)
    conv = np.fft.irfft(fw * far, n=L_cross)
    # length = len(w) + len(a_rev) - 1  (which equals 3*n-2)
    return conv[: len(w) + len(a_rev) - 1]

# ------------------------------------------------------------
# Gradient of a smooth approximation to max(b) where b = a*a
# ------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float,
                     L_self: int, L_cross: int) -> tuple:
    """
    Compute gradient of (softmax over convolution) and return it together with b.
    a is assumed to sum to 1 (simplex), beta > 0.
    """
    n = a.shape[0]

    # a*a  (full convolution)
    b = _self_conv_fft(a, L_self)                     # length 2n-1

    # soft‑max weights
    maxb = b.max()
    exp_arg = beta * (b - maxb)                       # safely shifted
    w = np.exp(exp_arg, dtype=np.float64)
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
        w_sum = 1.0
    w /= w_sum

    # gradient = 2 * (w * a_rev) sliced
    a_rev = a[::-1]
    conv = _cross_conv_fft(w, a_rev, L_cross)         # length 3n-2
    grad = 2.0 * conv[n - 1 : n - 1 + n]              # slice of length n
    return grad, b

# ------------------------------------------------------------
