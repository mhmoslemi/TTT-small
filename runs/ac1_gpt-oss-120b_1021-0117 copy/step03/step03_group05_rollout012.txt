import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if ind.size == 0:
        theta = 0.0
    else:
        rho = ind[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2048
        cur = np.ones(n0, dtype=np.float64) / n0
    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s <= 0.0:
        cur = np.full_like(cur, 1.0 / cur.size)
    else:
        cur = cur / s

    n = cur.size
    # FFT length for full convolution (2n-1)
    _L = 1 << ((2 * n - 2).bit_length())

    def _conv_fft(v: np.ndarray) -> np.ndarray:
        fv = np.fft.rfft(v, n=_L)
        c = np.fft.irfft(fv * fv, n=_L)
        return c[: 2 * n - 1]

    def _score(v: np.ndarray) -> float:
        c = _conv_fft(v)
        return 2.0 * n * c.max() / (v.sum() ** 2)

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    best = cur.copy()
    best_sc = _score(best)

    # optimisation parameters
    it = 0
    momentum_vec = np.zeros_like(cur)
    base_alpha = 2.0          # additive step scale
    base_eta = 1.0            # multiplicative step scale
    momentum = 0.9
    K = 80                    # active convolution indices
    target = 1.50274349

    # ------------------------------------------------------------------
    # main loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(cur)
        # active set: K largest convolution entries
        if K < conv.size:
            active = np.argpartition(-conv, K - 1)[:K]
        else:
            active = np.arange(conv.size)

        # sub‑gradient (sum of gradients of selected terms)
        g = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]

        # momentum
        momentum_vec = momentum * momentum_vec + (1.0 - momentum) * g

        # step sizes
        alpha = base_alpha / np.sqrt(it)
        eta = base_eta / np.sqrt(it)

        # additive candidate
        cand_add = cur - alpha * momentum_vec
        cand_add = _project_simplex(cand_add)
        sc_add = _score(cand_add)

        # multiplicative candidate
        exp_factor = np.exp(-eta * momentum_vec)
        cand_mul = cur * exp_factor
        cand_mul = _project_simplex(cand_mul)
        sc_mul = _score(cand_mul)

        # choose best among current, additive, multiplicative
        cur_sc = _score(cur)
        if sc_add < cur_sc and sc_add <= sc_mul:
            cur, cur_sc = cand_add, sc_add
        elif sc_mul < cur_sc:
            cur, cur_sc = cand_mul, sc_mul
        else:
            # occasional slight uphill move to escape plateaus
            if rng.random() < 0.02:
                if sc_add < sc_mul:
                    cur, cur_sc = cand_add, sc_add
                else:
                    cur, cur_sc = cand_mul, sc_mul

        # update global best
        if cur_sc < best_sc:
            best = cur.copy()
            best_sc = cur_sc
            # early exit
            if best_sc < target:
                break

        # occasional random mass‑transfer to diversify
        if it % 1000 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best = tmp.copy()
                        best_sc = tmp_sc

        # jittered restart towards best solution
        if it % 5000 == 0 and rng.random() < 0.25:
            jitter = rng.normal(scale=5e-4, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best = tmp.copy()
                    best_sc = tmp_sc

    # ------------------------------------------------------------------
    # final short random refinement (use remaining time)
    # ------------------------------------------------------------------
    refin_end = time.time() + min(0.3, deadline - time.time())
    while time.time() < refin_end:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * rng.random() * 0.3
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best = tmp.copy()
                best_sc = tmp_sc

    # return as plain Python list
    return [float(x) for x in best]
