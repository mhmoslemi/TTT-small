def run_descent(a_init, max_iter):
    a = a_init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_local = a.copy()
    best_local_score = float('inf')
    for t in range(1, max_iter+1):
        # time check
        if time.monotonic() >= deadline:
            break
        # compute subgradient and max
        g, cur_max = subgrad_max(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local = a.copy()
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = (lr0 / np.sqrt(t)) * m_hat / (np.sqrt(v_hat) + eps)
        a = _proj_simplex(a - step)
        # occasional jitter
        if t % 500 == 0:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * jitter)
    return best_local, best_local_score

import time
import numpy as np

def _normalize_nonneg(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1 (simple simplex projection)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v = v / s
    else:
        v = np.full_like(v, 1.0 / v.size)
    return v


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic optimiser for the AC1 bound.
    Returns the best non‑negative step–function (list of floats) found
    before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Timing
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2. Length and basic seeds
    # ------------------------------------------------------------------
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_raw.size
        parent = np.clip(parent_raw, 0.0, 1000.0)
        parent = _normalize_nonneg(parent)
    except Exception:
        n = 256
        parent = None

    # uniform seed
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # triangular (tent) seed
    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1.0, n - idx) - 1.0
    tent = np.clip(tent, 0.0, None)
    tent = _normalize_nonneg(tent)

    # Gaussian‑shaped seed
    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((idx - n / 2) / sigma) ** 2)
    gauss = _normalize_nonneg(gauss)

    # Random Dirichlet‑like seed
    rand = rng.random(n)
    rand = _normalize_nonneg(rand)

    seeds = [uniform, tent, gauss, rand]
    if parent is not None:
        seeds.append(parent)

    # ------------------------------------------------------------------
    # 3. FFT helpers (fast evaluation of max convolution)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def _max_conv(a: np.ndarray) -> float:
        """Return max of a*a convolution (full)."""
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return conv.max()

    # ------------------------------------------------------------------
    # 4. Choose initial best candidate
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for cand in seeds:
        cur = _max_conv(cand) * 2.0 * n
        if cur < best_score:
            best_score = cur
            best_seq = cand.copy()
    # fallback safety
    if best_seq is None:
        best_seq = uniform.copy()
        best_score = _max_conv(best_seq) * 2.0 * n

    # ------------------------------------------------------------------
    # 5. Optimisation primitives
    # ------------------------------------------------------------------
    # Adam parameters
    lr0 = 1.0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # ------------------------------------------------------------------
    # 6. Core optimisation routine (run many short restarts)
    # ------------------------------------------------------------------
    overall_best = best_seq.copy()
    overall_best_score = best_score

    # limits for restarts
    max_iters_per_restart = 4000
    restart_counter = 0

    while time.monotonic() < deadline:
        # pick a fresh start: jittered version of current overall best
        start = overall_best.copy()
        jitter = np.exp(rng.normal(0.0, 0.02, size=n))
        start = _normalize_nonneg(start * jitter)

        # Adam state
        m = np.zeros_like(start)
        v = np.zeros_like(start)

        # inner optimisation loop
        for it in range(1, max_iters_per_restart + 1):
            if time.monotonic() >= deadline:
                break

            # ----- compute max‑convolution and its index -----
            fa = np.fft.rfft(start, n=fft_len)
            conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
            max_val = conv.max()
            max_idx = int(np.argmax(conv))

            # ----- sub‑gradient for the active shift -----
            # indices i that participate in this shift
            i_min = max(0, max_idx - (n - 1))
            i_max = min(n - 1, max_idx)
            i_range = np.arange(i_min, i_max + 1, dtype=np.int64)
            g = np.zeros_like(start)
            g[i_range] = 2.0 * start[max_idx - i_range]

            # ----- keep track of best inner solution -----
            cur_score = 2.0 * n * max_val
            if cur_score < overall_best_score:
                overall_best_score = cur_score
                overall_best = start.copy()
                # early exit if we beat the target
                if overall_best_score <= 1.50274349:
                    break

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)
            start = _normalize_nonneg(start - step)

            # ----- occasional jitter to escape plateaus -----
            if it % 500 == 0:

                start = _normalize_nonneg(start * jitter)

        restart_counter += 1
        # quick check for time; if insufficient for another restart we exit
        if time.monotonic() + 1.0 >= deadline:
            break

    # ------------------------------------------------------------------
    # 7. Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in overall_best]
