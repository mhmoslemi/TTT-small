import time, math, numpy as np
from scipy.optimize import linprog

# ---------- convolution helpers ----------
def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution x*x using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)
    return conv[: 2 * x.size - 1]


def _score_vec(v: np.ndarray, L: int) -> float:
    """Score = 2*n*max(conv)/sum(v)^2."""
    n = v.size
    conv = _conv_fft(v, L)
    return 2.0 * n * conv.max() / (v.sum() ** 2)


# ---------- constraint matrix for LP ----------
def _build_A(f: np.ndarray) -> np.ndarray:
    """Build matrix A (shape (2n-1, n)) such that (f*g)[k] = (A@g)[k]."""
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    # first n rows
    for k in range(n):
        A[k, : k + 1] = f[k::-1]               # f[k], f[k-1], ..., f[0]

    # remaining rows
    for k in range(n, rows):
        length = rows - k                      # decreasing length
        start = k - n + 1
        A[k, start:] = f[-length:][::-1]       # tail of f reversed
    return A


# ---------- simple simplex projection (non‑negative, sum=1) ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s
