    n = len(a)
    # Get top K indices
    if K >= conv.size:

        ks = np.argpartition(-conv, K-1)[:K]
    g = np.zeros_like(a)

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi+1)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)
    # length n
    if 'height_sequence_1' in globals():

        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        if s > 0:
            cur /= s
        else:
            cur = np.full_like(cur, 1.0 / cur.size)
    else:
        n = 256
        cur = np.full(n, 1.0 / n, dtype=np.float64)
    n = cur.size
    # conv function
    def conv_fft(a):
        L = 1 << ((2 * n - 1).bit_length())
        fv = np.fft.rfft(a, n=L)
        return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    def score(a):
        conv = conv_fft(a)
        return float(2.0 * n * conv.max() / (a.sum() ** 2))
    def project_simplex(v):
        # Euclidean projection onto simplex sum 1, nonnegative
        # Using standard algorithm
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)
    # function to compute gradient based on top k conv values
    def active_gradient(a, conv_arr, K=5):
        # K active lags
        K = min(K, conv_arr.size)
        # Get indices of top K conv values
        if K <= 1:
            ks = [int(np.argmax(conv_arr))]
        else:
            # use argpartition for efficiency
            ks = np.argpartition(-conv_arr, K-1)[:K]
        g = np.zeros_like(a)
        # accumulate contributions
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * a[k - idx]
        # average
        g /= K
        return g
    # Evaluate families to get good start
    best_vec = cur.copy()
    best_sc = score(best_vec)
    # helper to try candidate
    def consider(vec):
        nonlocal best_vec, best_sc
        sc = score(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
        return sc
    # families

    for r in np.linspace(0.90, 0.9999, 25):

        a = a / a.sum()

        a = a / a.sum()
        consider(a)

    a = np.arange(n, 0, -1, dtype=np.float64)
    a = a / a.sum()
    consider(a)

    consider(a)

        a = a / a.sum()
        consider(a)

        consider(a)

import time
import math
import numpy as np

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a via FFT (real input)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]

def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (scale‑invariant)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _active_gradient(a: np.ndarray, conv_arr: np.ndarray, K: int = 5) -> np.ndarray:
    """Sparse sub‑gradient built from the K largest convolution entries."""
    n = a.shape[0]
    K = min(K, conv_arr.size)
    if K <= 1:
        ks = [int(np.argmax(conv_arr))]
    else:
        ks = np.argpartition(-conv_arr, K - 1)[:K]
    g = np.zeros_like(a)
    for k in ks:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    g /= K
    return g

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # -----------------------------------------------------------------
    # initialise from parent if present, otherwise uniform
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    else:
        n0 = 256
        cur = np.full(n0, 1.0 / n0, dtype=np.float64)

    n = cur.size
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # -----------------------------------------------------------------
    # try a collection of analytic families for a good start
    # -----------------------------------------------------------------
    def _consider(vec):
        nonlocal best_sc, best_vec
        sc = _score(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
        return sc

    # exponential decays
    for r in np.linspace(0.90, 0.9999, 25):
        a = np.power(r, np.arange(n, dtype=np.float64))
        _consider(a / a.sum())

    # power‑law tails
    for p in np.linspace(0.2, 4.0, 25):
        a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        _consider(a / a.sum())

    # simple polynomials
    a = np.arange(n, 0, -1, dtype=np.float64)
    _consider(a / a.sum())
    a = np.square(a)
    _consider(a / a.sum())

    # monotone Gaussians
    for sigma in (10, 30, 80, 200, 500):
        x = np.arange(n, dtype=np.float64)
        a = np.exp(- (x ** 2) / (2.0 * sigma ** 2))
        _consider(a / a.sum())

    # symmetric (bell‑shaped) Gaussians
    mid = (n - 1) / 2.0
    for sigma in (20, 50, 120, 300, 800):
        x = np.arange(n, dtype=np.float64)

        _consider(a / a.sum())

    cur = best_vec.copy()
    cur_sc = best_sc
    cur_conv = _conv_fft(cur)

    target = 1.50274349

                    if best_sc < target:
                        break

            # transfer a chunk of mass from a relatively large to a small entry
            big_idxs = np.where(cur > cur.mean())[0]
            small_idxs = np.where(cur < cur.mean())[0]
            if big_idxs.size and small_idxs.size:
                i = int(rng.choice(big_idxs))
                j = int(rng.choice(small_idxs))
                delta = cur[i] * 0.2 * rng.random()
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc, cur_conv = tmp, tmp_sc, _conv_fft(tmp)

        # generous early‑exit if time is exhausted
        if time.monotonic() >= deadline:

    # -----------------------------------------------------------------
    # final safety net
    # -----------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
