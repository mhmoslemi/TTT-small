import time, math, numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the simplex (L1=1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:               # guard against all‑zero output
        w[:] = 1.0 / n
    return w

# ----------------------------------------------------------------------
# Simple analytic starting series (optional)
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        s = float(np.dot(a[1:k], a[k-1:0:-1])) if k > 1 else 0.0
        a[k] = (1.0 - s) / 2.0
    return a

# ----------------------------------------------------------------------
# Next power of two (for FFT padding)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

# ----------------------------------------------------------------------
# Core optimisation – Adam + soft‑max gradient on the autocorrelation max
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 2.0)

    # ---- initial vector -------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)
    else:
        a = _sqrt_series(256)
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

    n = a.size
    L_self = _next_pow2(2 * n - 1)          # length for self‑convolution
    L_grad = _next_pow2(3 * n - 2)          # length for gradient convolution

    # ---- Adam state -----------------------------------------------------
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # ---- schedule parameters --------------------------------------------
    lam_start, lam_end = 0.5, 12.0          # soft‑max temperature
    lr_start, lr_min = 0.15, 1e-5

    # ---- initial scores -------------------------------------------------
    fv = np.fft.rfft(a, n=L_self)
    conv_a = np.fft.irfft(fv * fv, n=L_self)[: 2 * n - 1]
    cur_max = float(conv_a.max())
    cur_score = 2.0 * n * cur_max
    best_vec = a.copy()
    best_score = cur_score

    t = 1
    prob_random, prob_smooth = 0.025, 0.008
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)

    while time.monotonic() < deadline:
        # ---- soft‑max weights from current autocorrelation ----------------
        max_c = conv_a.max()
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / (budget_s - 2.0))
        lam = lam_start + (lam_end - lam_start) * (frac ** 1.5)
        w = np.exp(lam * (conv_a - max_c))    # already shifted by max_c
        w /= w.sum()

        # ---- gradient (cross‑correlation w * a_rev) --------------------
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=L_grad)
        fa = np.fft.rfft(a_rev, n=L_grad)
        corr = np.fft.irfft(fw * fa, n=L_grad)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 :: -1]        # length n

        # ---- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = max(lr_min, lr_start * (1.0 - frac))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - step
        cand = _project_simplex(cand)

        # ---- evaluate candidate (FFT, no dense conv) -------------------
        fv_c = np.fft.rfft(cand, n=L_self)
        conv_c = np.fft.irfft(fv_c * fv_c, n=L_self)[: 2 * n - 1]
        cand_max = float(conv_c.max())
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score:
            a = cand
            conv_a = conv_c
            cur_score = cand_score
            cur_max = cand_max
        else:
            # tiny chance to accept worse candidate (helps escape)
            if rng.random() < 0.001:
                a = cand
                conv_a = conv_c
                cur_score = cand_score
                cur_max = cand_max

        if cand_score < best_score:
            best_score = cand_score
            best_vec = cand.copy()

        # ---- occasional random mass transfer ----------------------------
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.5
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                fv2 = np.fft.rfft(cand2, n=L_self)
                conv2 = np.fft.irfft(fv2 * fv2, n=L_self)[: 2 * n - 1]
                score2 = 2.0 * n * float(conv2.max())
                if score2 < best_score:
                    best_score = score2
                    best_vec = cand2.copy()
                if score2 < cur_score:
                    a = cand2
                    conv_a = conv2
                    cur_score = score2

        # ---- occasional smoothing -----------------------------------------
        if rng.random() < prob_smooth:
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)
            fv_s = np.fft.rfft(smooth, n=L_self)
            conv_s = np.fft.irfft(fv_s * fv_s, n=L_self)[: 2 * n - 1]
            score_s = 2.0 * n * float(conv_s.max())
            if score_s < best_score:
                best_score = score_s
                best_vec = smooth.copy()
            if score_s < cur_score:
                a = smooth
                conv_a = conv_s
                cur_score = score_s

        t += 1
        # ---- early exit guard --------------------------------------------
        if time.monotonic() + 0.5 > deadline:
            break

    return [float(x) for x in best_vec]
