import numpy as np

def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a*a using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ---- initialise -------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: simple exponential decay
        n0 = 256
        a = np.exp(-np.arange(n0) / 30.0)
    a = _project_onto_simplex(a)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two

    # ---- Adam optimiser state ---------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12                     # base learning rate
    t_iter = 0

    # soft‑max sharpness schedule
    alpha = 5.0
    alpha_max = 30.0
    alpha_inc = 1.001

    # best‑so‑far bookkeeping
    conv = _full_conv(a, L)
    best = a.copy()
    best_score = 2.0 * n * conv.max()
    no_improve = 0
    max_no_improve = 5000

    # ---- main loop -------------------------------------------------
    while time.monotonic() < deadline:
        # convolution of current iterate
        conv = _full_conv(a, L)
        max_conv = conv.max()

        # soft‑max weights over all lags
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient via convolution (w * a_rev)
        a_rev = a[::-1]
        w_f = np.fft.rfft(w, n=L)
        a_rev_f = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(w_f * a_rev_f, n=L)[: 2 * n - 1]
        g = 2.0 * grad_full[n - 1:n - 1 + n]   # length‑n gradient

        # Adam update
        t_iter += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - step
        cand = _project_onto_simplex(cand)

        # evaluate candidate (fast using FFT)
        conv_c = _full_conv(cand, L)
        cand_score = 2.0 * n * conv_c.max()

        # keep global best
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        # accept if improved or with tiny probability
        if cand_score < 2.0 * n * max_conv or rng.random() < 0.0005:
            a = cand

        # gradual sharpening of soft‑max
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_inc)

        # occasional random mass‑transfer perturbation
        if rng.random() < 0.01:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                frac = rng.random() * 0.5
                delta = a[i] * frac
                pert = a.copy()
                pert[i] -= delta
                pert[j] += delta
                pert = _project_onto_simplex(pert)
                conv_p = _full_conv(pert, L)
                pert_score = 2.0 * n * conv_p.max()
                if pert_score < best_score:
                    best_score, best = pert_score, pert.copy()
                    no_improve = 0
                if pert_score < 2.0 * n * max_conv:
                    a = pert

        # restart if stuck for a long time
        if no_improve >= max_no_improve:
            a = rng.random(n)
            a = _project_onto_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            no_improve = 0

        # safeguard against overshooting deadline
        if time.monotonic() + 0.2 > deadline:
            break

    # return the best found sequence as plain Python list
    return [float(x) for x in best]
