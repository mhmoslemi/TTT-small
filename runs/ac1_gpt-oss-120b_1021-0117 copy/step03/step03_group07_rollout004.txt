def _conv_fft_two(x, y):
    n = len(x) + len(y) - 1
    L = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    return np.fft.irfft(fx * fy, n=L)[:n]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (output length 2n‑1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def _conv_fft_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors (output length len(x)+len(y)-1)."""
    m = len(x) + len(y) - 1
    L = 1 << ((m - 1).bit_length())
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)
    return conv[:m]


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (sum is forced to 1)."""
    n = a.shape[0]
    conv = _conv_fft_self(a)
    return float(2.0 * n * conv.max())


def _make_exponential(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    s = a.sum()
    return a / s


def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    s = a.sum()
    return a / s


def _make_sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s
    return a

        # occasional bigger reshuffle
        if rng.random() < 0.001:
            perm = rng.permutation(n)
            cand = (cur[perm] + cur) * 0.5
            cand = _project_simplex(cand)
            cand_sc = _score(cand)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
                    if best_sc < target:
                        break

        if it2 % 2500 == 0:
            # small random redistribution of a few mass units
            nz = np.where(cur > 1e-12)[0]
            if nz.size:
                i = int(rng.choice(nz))
                j = int(rng.integers(0, n - 1))
                if j == i:
                    j = (j + 1) % n
                d = cur[i] * rng.random() * 0.3
                cand = cur.copy()
                cand[i] -= d
                cand[j] += d
                cand = _project_simplex(cand)
                cand_sc = _score(cand)
                if cand_sc < cur_sc:
                    cur, cur_sc = cand, cand_sc
                    if cand_sc < best_sc:
                        best_sc, best_vec = cand_sc, cand.copy()
                        if best_sc < target:
                            break

        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    #   5) final fallback safety
    # ------------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
