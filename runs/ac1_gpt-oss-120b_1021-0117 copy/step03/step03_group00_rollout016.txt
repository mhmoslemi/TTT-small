import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _max_conv(a: np.ndarray, L: int) -> float:
    """Maximum of linear convolution a*a using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) whose AC1 bound is low."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random positive vector (length ≈2500)
        n_fallback = 2500
        a0 = rng.random(n_fallback, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 < 1e-12:
        a0.fill(1.0 / a0.size)
    else:
        a0 /= s0                     # simplex normalisation

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())   # FFT length ≥ 3n‑2

    # ----- bookkeeping ------------------------------------------------
    best = a0.copy()
    best_sc = 2.0 * n * _max_conv(best, L)   # sum==1 ⇒ denominator =1
    cur = best.copy()
    cur_sc = best_sc

    # ----- hyper‑parameters -----------------------------------------
    base_alpha = 0.5          # base step length (before normalisation)
    step_decay = 0.9997       # geometric decay per iteration
    no_improve = 0
    it = 0

    # ----- main optimisation loop ------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution of current iterate
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # active mask (indices attaining the maximum)
        active = (conv_full == max_val).astype(np.float64)
        act_sum = active.sum()
        if act_sum == 0:   # should not happen, but guard against numerical issues
            active = np.ones_like(conv_full) / conv_full.size
            act_sum = 1.0

        # normalised weight vector for the active set
        w = active / act_sum
        w_rev = w[::-1]                     # reversed for convolution → correlation
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)   # correlation a⊛w
        idx = (2 * n - 2) + np.arange(n)               # correspondence derived above
        g = 2.0 * grad_full[idx]                       # sub‑gradient of max(conv)

        # ----- descent step -----------------------------------------
        grad_norm = np.linalg.norm(g) + 1e-12
        direction = -g / grad_norm                      # unit descent direction
        alpha = base_alpha * (step_decay ** it) / math.sqrt(it + 1.0)

        improved = False
        for factor in (1.0, 0.5, 0.25):
            step = alpha * factor
            cand = cur + step * direction
            cand = _project_simplex(cand)

            cand_sc = 2.0 * n * _max_conv(cand, L)
            if cand_sc < cur_sc - 1e-14:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best, best_sc = cand.copy(), cand_sc
                    no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1

            # occasional random mass‑transfer when stuck
            if no_improve > 3000:
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.45
                    j = rng.integers(0, n)
                    while j == i:
                        j = rng.integers(0, n)
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = 2.0 * n * _max_conv(tmp, L)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        if tmp_sc < best_sc:
                            best, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        # ----- periodic symmetrisation -------------------------------
        if it % 1200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = 2.0 * n * _max_conv(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ----- early exit --------------------------------------------
        if best_sc < 1.50274349:
            break

    # ----- final local random polishing (time‑aware) -----------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if best[i] > 0.0:
            delta = best[i] * rng.random() * 0.3
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            tmp = best.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = 2.0 * n * _max_conv(tmp, L)
            if tmp_sc < best_sc:
                best, best_sc = tmp, tmp_sc

    # ----- return ----------------------------------------------------
    return [float(x) for x in best]
