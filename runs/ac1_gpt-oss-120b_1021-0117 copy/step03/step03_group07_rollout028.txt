# --------------------------------------------------------------
# Helper utilities

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via FFT (real input)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    return np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _make_exponential(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    return a / a.sum()


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    return a / a.sum()


def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()


def _make_sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / s
    return a

            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break
        else:
            # very occasional uphill move to diversify
            if rng.random() < 0.01:
                cand_up = cur + alpha * g
                v = cand_up - np.mean(cand_up) + 1.0 / n
                u = np.sort(v)[::-1]
                css = np.cumsum(u) - 1.0
                rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
                if rho.size:
                    rho = rho[-1]
                    theta = css[rho] / (rho + 1)
                else:
                    theta = 0.0
                cand_up = np.maximum(v - theta, 0.0)
                cand_up = _sorted_desc(cand_up)
                cand_up_sc = _score(cand_up)
                if cand_up_sc < cur_sc:
                    cur, cur_sc = cand_up, cand_up_sc
                    if cand_up_sc < best_sc:
                        best_sc, best_vec = cand_up_sc, cand_up.copy()
                        if best_sc < target:
                            break

        # occasional random mass‑transfer exploring move
        if it2 % 1500 == 0 and rng.random() < 0.2:
            nz = np.where(cur > 1e-12)[0]
            if nz.size:
                i = int(rng.choice(nz))
                j = int(rng.integers(0, n - 1))
                if j == i:
                    j = (j + 1) % n
                delta = cur[i] * rng.random() * 0.35
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _sorted_desc(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                        if best_sc < target:
                            break

        if time.monotonic() >= deadline:
            break

    # ----------------------------------------------------------
    # Phase 3 – Focused random mass‑transfer moves (pure hill‑climb)
    # ----------------------------------------------------------
    while time.monotonic() < deadline:
        conv = _conv_fft(cur)
        k = int(np.argmax(conv))
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = cur[idx] * cur[k - idx]
        if prod.size:
            i = int(idx[np.argmax(prod)])
        else:
            i = int(rng.integers(0, n))
        j = int(np.argmin(cur))
        if j == i:
            j = int(rng.integers(0, n - 1))
            if j == i:
                j = (j + 1) % n

        max_delta = cur[i] * 0.45
        if max_delta < 1e-12:
            continue
        delta = max_delta * rng.random()
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _sorted_desc(cand)
        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break

        # rare tiny jitter to keep exploration alive
        if rng.random() < 0.001:
            jitter = rng.normal(scale=5e-5, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = tmp / tmp.sum()
            tmp = _sorted_desc(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    if best_sc < target:
                        break

    # ----------------------------------------------------------
    # Safety fallback (should never happen)
    # ----------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # Return plain Python list of floats (as required by the harness)
    return [float(v) for v in best_vec]
