def _optimize_improved(a0, deadline, rng):
    a = a0.copy()
    n = a.size
    L = 1 << ((2*n-1).bit_length())
    max_val, max_idx = _conv_max_and_index(a, L)
    cur_score = 2.0*n*max_val
    best = a.copy()
    best_score = cur_score
    step = 0.5
    min_step = 1e-7
    decay = 0.9995
    momentum = np.zeros_like(a)
    beta = 0.9

    while time.monotonic() < deadline:
        # compute gradient of current max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:
            j = np.arange(lo, hi+1)
            g[j] = 2.0 * a[max_idx - j]
        # momentum
        momentum = beta * momentum + (1-beta) * g
        # try step sizes in decreasing geometric series
        best_local_score = cur_score
        best_candidate = None
        best_maxval = None
        candidate = None
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            step_sz = step * factor
            cand = a - step_sz * momentum
            cand = _project_simplex(cand)
            maxv, idx = _conv_max_and_index(cand, L)
            cand_score = 2.0 * n * maxv
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_candidate = cand
                best_maxval = maxv
                best_idx = idx
        if best_candidate is not None:
            a = best_candidate
            cur_score = best_local_score
            max_idx = best_idx
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # random move if no improvement
            if rng.random() < 0.01:
                i = rng.integers(0, n)
                if a[i] > 0:
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    delta = a[i] * rng.random() * 0.5
                    cand2 = a.copy()
                    cand2[i] -= delta
                    cand2[j] += delta
                    cand2 = _project_simplex(cand2)
                    max2, idx2 = _conv_max_and_index(cand2, L)
                    score2 = 2.0 * n * max2
                    if score2 < cur_score:
                        a = cand2
                        cur_score = score2
                        max_idx = idx2
                        if score2 < best_score:
                            best_score = score2
                            best = cand2.copy()
            # smoothing
            if rng.random() < 0.005:
                kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
                smooth = np.convolve(a, kernel, mode='same')
                smooth = _project_simplex(smooth)
                max_s, idx_s = _conv_max_and_index(smooth, L)
                score_s = 2.0*n*max_s
                if score_s < cur_score:
                    a = smooth
                    cur_score = score_s
                    max_idx = idx_s
                    if score_s < best_score:
                        best_score = score_s
                        best = smooth.copy()

        step = max(min_step, step * decay)
        # break early if near time end
        if time.monotonic() + 0.5 > deadline:
            break
    # return best and best_score
    return best, best_score

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto the probability simplex, sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    # work on a copy
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # guard against degenerate case
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w

# ------------------------------------------------------------
# Convolution utilities (FFT, max value + index)
# ------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    """Return max convolution value and its index for vec*vec (full)."""
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

# ------------------------------------------------------------
# Analytic sqrt‑series used in earlier work (optional start)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ------------------------------------------------------------
# Core optimizer (gradient descent with line‑search, random moves)
# ------------------------------------------------------------
def _optimize_improved(start_vec: np.ndarray, deadline: float, rng: np.random.Generator):
    n = start_vec.size
    L = 1 << ((2 * n - 1).bit_length())  # power‑of‑two >= 2n‑1
    a = start_vec.copy()
    max_val, max_idx = _conv_max_and_index(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # hyper‑parameters
    step = 0.5
    min_step = 1e-7
    decay = 0.9995
    beta = 0.9
    momentum = np.zeros_like(a)

    # counters to trigger occasional restarts
    no_improve = 0
    RESTART_AFTER = 20000

    while time.monotonic() < deadline:
        # -------- gradient of the current maximum entry --------
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:
            j = np.arange(lo, hi + 1, dtype=np.int64)
            g[j] = 2.0 * a[max_idx - j]

        # -------- momentum update --------
        momentum = beta * momentum + (1.0 - beta) * g

        # -------- line‑search over a few step‑sizes --------
        improved = False
        best_local = cur_score
        best_cand = None
        best_max_idx = max_idx

        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            cand = a - step * factor * momentum
            cand = _project_simplex(cand)
            mval, midx = _conv_max_and_index(cand, L)
            cand_score = 2.0 * n * mval
            if cand_score < best_local - 1e-14:
                best_local = cand_score
                best_cand = cand
                best_max_idx = midx
                improved = True

        if improved:
            a = best_cand
            cur_score = best_local
            max_idx = best_max_idx
            no_improve = 0
            if cur_score < best_score:

        else:

                if score2 < cur_score:
                    a = cand2
                    cur_score = score2
                    max_idx = idx2
                    no_improve = 0

            sm = np.convolve(a, kernel, mode='same')
            sm = _project_simplex(sm)

            if score_s < cur_score:
                a = sm

                max_idx = idx_s

        step = max(min_step, step * decay)

# Entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # --------------------------------------------------------
    # Determine length & basic helpers
    # --------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 256  # fallback length if no parent is supplied

    # --------------------------------------------------------
    # Build a collection of diverse initial vectors
    # --------------------------------------------------------
    inits = []

    # 1) parent if present
    if 'height_sequence_1' in globals():
        inits.append(np.asarray(height_sequence_1, dtype=np.float64))

    # 2) sqrt series (analytic start from earlier work)
    inits.append(_sqrt_series(n))

    # 3) half‑sine shape (smooth, zero at ends)
    i = np.arange(n, dtype=np.float64)
    half_sine = np.sin(np.pi * (i + 0.5) / n)
    inits.append(half_sine)

    # 4) Hann window (cosine taper)
    if n > 1:
        hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    else:
        hann = np.ones(1)
    inits.append(hann)

    # 5) symmetric triangular ramp
    tri = np.concatenate([
        np.linspace(0.0, 1.0, (n + 1) // 2, endpoint=False),
        np.linspace(1.0, 0.0, n // 2, endpoint=False)
    ])
    inits.append(tri)

    # 6) random Dirichlet‑like start
    rand_vec = rng.random(n)
    inits.append(rand_vec)

    # --------------------------------------------------------
    # Normalise each start onto the simplex and evaluate
    # --------------------------------------------------------
    best_overall = None
    best_score_overall = float('inf')

    for vec in inits:
        a0 = np.maximum(vec, 0.0)
        a0 = _project_simplex(a0)

        # quick score (no full evaluate_sequence needed)
        L = 1 << ((2 * n - 1).bit_length())
        max_val, _ = _conv_max_and_index(a0, L)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score_overall:
            best_score_overall = cur_score
            best_overall = a0.copy()

        # run the optimizer if we still have time
        if time.monotonic() + 5.0 < deadline:
            cand, cand_score = _optimize_improved(a0, deadline, rng)
            if cand_score < best_score_overall:
                best_score_overall = cand_score
                best_overall = cand.copy()

        # stop early if time is almost up
        if time.monotonic() + 2.0 > deadline:
            break

    # --------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # --------------------------------------------------------
    if best_overall is None:
        best_overall = np.ones(n, dtype=np.float64) / n

    # --------------------------------------------------------
    # Return plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_overall]
