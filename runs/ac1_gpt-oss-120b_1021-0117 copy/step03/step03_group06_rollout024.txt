Standard algorithm:

def proj_simplex(v):
    # assures sum(v)=1 and v>=0
    if v.sum() == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1] # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def _proj_simplex(v):
    # Using algorithm from Wang & Carreira-Perpinan 2013
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import time, math, numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT convolution, scoring, simplex projection)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full convolution of x with itself using real FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]
    return conv


def _score_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Score = 2*n*max(conv)/sum_a**2 (same as evaluate_sequence)."""
    if sum_a == 0:
        return float("inf")
    return 2.0 * n * conv.max() / (sum_a * sum_a)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = v.astype(np.float64, copy=False)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond].max() - 1  # zero‑based index
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # guard against numerical drift
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function that (hopefully) improves the AC1 bound.
    The routine runs for at most `budget_s` seconds and returns the best
    sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------------
    # Determine problem size and initialise candidates
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: a plain uniform of length 2000
        base = np.full(2000, 1.0, dtype=np.float64)

    n = base.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two length for FFT

    def _normalise(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = np.full_like(v, 1.0 / v.size)
        else:
            v = v / s
        return v

    # a few diverse starting points (always projected onto the simplex)
    starts = []

    # 1) the provided parent (if any)
    starts.append(_normalise(base))

    # 2) pure uniform
    starts.append(_normalise(np.full(n, 1.0)))

    # 3) slightly jittered uniform (to break symmetry immediately)
    jitter = rng.normal(scale=1e-5, size=n)
    starts.append(_normalise(np.full(n, 1.0) + jitter))

    # 4) random Dirichlet (two different concentration levels)
    for conc in (0.5, 2.0):
        x = rng.gamma(conc, 1.0, size=n)
        starts.append(_normalise(x))

    # 5) power‑law “U‑shape” a_i ∝ 1/(min(i,n-1-i)+1)^p
    for p in (0.3, 0.7, 1.2):
        idx = np.arange(n)
        dist = np.minimum(idx, n - 1 - idx) + 1
        a = 1.0 / (dist ** p)
        starts.append(_normalise(a))

    # --------------------------------------------------------------
    # Evaluate initial pool and pick the best
    # --------------------------------------------------------------
    best = None
    best_sc = float("inf")
    for a0 in starts:
        conv0 = _conv_fft(a0, L)
        sc0 = _score_from_conv(conv0, n, a0.sum())
        if sc0 < best_sc:
            best_sc = sc0
            best = a0.copy()
    if best is None:                     # should never happen
        best = _normalise(np.full(n, 1.0))

    cur = best.copy()
    cur_sc = best_sc

    # --------------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------------
    it = 0
    step0 = 0.7            # base additive step magnitude
    move_prob = 0.12       # probability of a random mass‑transfer move
    eps_move = 0.45        # max fraction of mass moved in the random move
    jitter_scale = 5e-4    # Gaussian jitter scale for restarts
    stagnation_limit = 9000
    no_improve = 0
    target_score = 1.50274349

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- gradient based additive mirror step ----
        conv = _conv_fft(cur, L)
        k = int(conv.argmax())                    # index attaining the current max
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]               # sub‑gradient of max‑term

        eta = step0 / math.sqrt(it)               # decaying step size
        cand = cur - eta * g
        cand = np.maximum(cand, 0.0)
        cand = _proj_simplex(cand)

        cand_sc = _score_from_conv(_conv_fft(cand, L), n, cand.sum())

        # ---- acceptance ----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # occasional Metropolis‑type acceptance to escape plateaus
            prob = math.exp(-(cand_sc - cur_sc) / (eta + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        if cand_sc < best_sc:
            best_sc, best = cand_sc, cand.copy()

        # ---- random mass‑transfer move (exploration) ----
        if rng.random() < move_prob:
            i = rng.integers(0, n)
            if cur[i] > 1e-15:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * eps_move
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                tmp = _proj_simplex(tmp)
                tmp_sc = _score_from_conv(_conv_fft(tmp, L), n, tmp.sum())
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best = tmp_sc, tmp.copy()

        # ---- jittered restart from the best point (periodic) ----
        if it % 3000 == 0:
            jitter = rng.normal(scale=jitter_scale, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _proj_simplex(tmp)
            tmp_sc = _score_from_conv(_conv_fft(tmp, L), n, tmp.sum())
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
            if tmp_sc < best_sc:
                best_sc, best = tmp_sc, tmp.copy()

        # ---- full restart after long stagnation ----
        if no_improve > stagnation_limit:
            jitter = rng.normal(scale=jitter_scale * 5, size=n)
            cur = best + jitter
            cur = np.maximum(cur, 0.0)
            cur = _proj_simplex(cur)
            cur_sc = _score_from_conv(_conv_fft(cur, L), n, cur.sum())
            no_improve = 0

        # ---- early exit if target reached ----
        if best_sc < target_score:
            break

    # --------------------------------------------------------------
    # Return best candidate as a plain Python list
    # --------------------------------------------------------------
    best = _proj_simplex(best)          # final safety projection
    return [float(x) for x in best]
