        conv = np.fft.irfft(fx * fx, n=L)
        return conv[:2 * n - 1]

        fw = np.fft.rfft(w, n=L)
        fv = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(fw * fv, n=L)
        return conv[:len(w) + len(v) - 1]   # 3n-2

        return 2.0 * n * M   # because sum=1

        # using algorithm from https://arxiv.org/abs/1101.6081
        u = np.sort(v)[::-1]

            theta = 0.0
        else:
            rho = rho[-1]
            theta = (cssv[rho] - 1) / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

def _fft_convolve(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of x and y using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)
    return conv[:len(x) + len(y) - 1]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u + (1.0 - cssv) / (np.arange(1, len(v) + 1)) > 0)[0]

    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
