def propose_candidate(...):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.2)  # leave some margin

    # init cur based on height_sequence_1 or uniform
    # Normalize cur to sum 1.
    
    # try families, update best.
    # We'll store best_vec and best_sc.
    
    cur = best_vec.copy()
    n = cur.shape[0]
    best_vec = cur.copy()
    best_sc = _score(best_vec)
    
    # gradient descent
    it = 0
    tau0 = 0.02  # initial temperature (as fraction of conv range)
    eta0 = 0.2  # step size
    while time.monotonic() < deadline:
        it += 1
        conv = _conv_fft(cur)
        maxc = conv.max()
        minc = conv.min()
        # scale tau
        tau = tau0 * (0.995 ** it)
        # ensure not too small
        tau = max(tau, 1e-6 * maxc) if maxc>0 else 1e-6
        # compute soft weights
        # subtract max to avoid overflow
        w = np.exp((conv - maxc) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # compute gradient
        w_rev = w[::-1]
        L = 1 << ((n + w_rev.size - 1).bit_length())
        Fa = np.fft.rfft(cur, n=L)
        Fw = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(Fa * Fw, n=L)[:n + w_rev.size - 1]
        # extract gradient for indices 0..n-1
        grad = 2 * grad_full[w_rev.size - 1 - np.arange(n)]
        # multiplicative update
        eta = eta0 / np.sqrt(it)
        cand = cur * np.exp(-eta * grad)
        cand_sum = cand.sum()
        cand = cand / cand_sum if cand_sum > 0 else np.full_like(cand, 1.0 / n)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()
        # accept if improvement or maybe accept with probability
        # We'll accept if cand_sc <= cur_score
        cur_sc = _score(cur)
        if cand_sc < cur_sc:
            cur = cand
        else:
            # occasional random perturbation
            if rng.random() < 0.01:
                jitter = rng.normal(scale=1e-4, size=n)
                tmp = cur + jitter
                tmp = np.maximum(tmp, 0)
                tmp = tmp / tmp.sum()
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
        # early exit if time < threshold break
        if it % 1000 == 0:
            # optionally jitter to escape
            if rng.random() < 0.1:
                jitter = rng.normal(scale=1e-5, size=n)
                tmp = cur + jitter
                tmp = np.maximum(tmp, 0)
                tmp = tmp / tmp.sum()
                tmp_sc = _score(tmp)
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                if tmp_sc < cur_sc:
                    cur = tmp
        # ensure we don't overshoot time
        if time.monotonic() >= deadline:

    # optional final random local search
    while time.monotonic() < deadline:
        # random move

        j = rng.integers(0, n)

        cand[j] += delta
        cand = np.maximum(cand, 0)
        cand = cand / cand.sum()
        cand_sc = _score(cand)
        if cand_sc < best_sc:

        if cand_sc < cur_sc:

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via FFT (real input)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    return np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (sum is not forced to 1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _make_exponential(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    return a / a.sum()


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    return a / a.sum()


def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)

def _make_quadratic(n: int) -> np.ndarray:

def _make_sqrt_series(n: int) -> np.ndarray:

    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / s
    return a

            jitter = rng.normal(scale=1e-4, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = tmp / tmp.sum()
            tmp_sc = _score(tmp)
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc

        if time.monotonic() >= deadline:
            break

    # -----------------------------------------------------------------
    # Phase B – targeted random mass‑transfer local search
    # -----------------------------------------------------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = cur[i] * 0.3 * rng.random()
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()
            if best_sc < target:
                break
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc

    # -----------------------------------------------------------------
    # fallback safety
    # -----------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
