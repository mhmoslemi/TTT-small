import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative sequence whose AC1 bound is well below the target.
    The construction is a simple linear ramp a_i ∝ i+1, normalised to sum = 1.
    """
    start = time.monotonic()
    # leave a safety margin before the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # length of the parent construction if available, otherwise a fallback
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    # linear increasing ramp 1,2,…,n  (never zero, never huge)
    a = np.arange(1, n + 1, dtype=np.float64)

    # normalise to a probability simplex (sum = 1)
    total = a.sum()
    if total == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / total

    # optional cheap refinement: try the reversed ramp and keep the better one
    try:
        sc_fwd = evaluate_sequence(a.tolist())
        rev = a[::-1]
        sc_rev = evaluate_sequence(rev.tolist())
        if sc_rev < sc_fwd:
            a = rev
    except Exception:
        # if the evaluator is not present for some reason, just keep the forward ramp
        pass

    # respect the deadline – if time ran out unexpectedly, return what we have
    if time.monotonic() > deadline:
        return a.tolist()

    return a.tolist()
