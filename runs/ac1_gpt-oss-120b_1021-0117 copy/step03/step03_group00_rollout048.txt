import time, math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) from https://arxiv.org/abs/1309.1541
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Score assuming sum(a)≈1 (the objective is 2·n·max(conv))
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())
