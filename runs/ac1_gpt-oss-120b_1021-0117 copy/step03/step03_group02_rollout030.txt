def propose_candidate(seed=42, budget_s=1100):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 0.5

    # initialization
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = _sqrt_series(256)
    # ensure nonnegative, clip to max 1000
    a = np.clip(a, 0, 1000)
    # project to simplex
    a = _project_simplex(a / max(a.sum(), 1e-12))

    n = a.size
    L = 1 << ((2*n - 1).bit_length())

    max_val, conv = _conv_full_max(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # hyperparameters
    step = 0.4
    min_step = 1e-5
    decay = 0.9995
    alpha = 20.0
    temp = 0.01   # simulated annealing temperature for occasional uphill.

    no_improve = 0
    max_no_improve = 200

    while time.monotonic() < deadline:
        # compute gradient using full softmax
        max_val = conv.max()
        # weight exponent
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0:
            wnorm = np.full_like(w, 1.0 / w.size)
        else:
            wnorm = w / w_sum

        # correlation wnorm with a, valid mode yields length n
        corr = correlate(wnorm, a, mode='valid')
        grad = 2.0 * corr

        # gradient step
        cand = a - step * grad
        cand = _project_simplex(cand)

        # evaluate
        max_val_c, conv_c = _conv_full_max(cand, L)
        cand_score = 2.0 * n * max_val_c

        improved = False
        if cand_score < cur_score:
            a = cand
            conv = conv_c
            cur_score = cand_score
            improved = True
        else:
            # accept uphill with prob exp(-(cand_score-cur)/temp)
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-12)):
                a = cand
                conv = conv_c
                cur_score = cand_score
                improved = True

        # update global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            no_improve = 0
        elif not improved:
            no_improve += 1
        else:
            no_improve = 0

        # occasional random perturbation
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.3
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                max_val2, conv2 = _conv_full_max(cand2, L)
                cand2_score = 2.0 * n * max_val2
                if cand2_score < cur_score:
                    a = cand2
                    conv = conv2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best = cand2.copy()
                        best_score = cand2_score
                        no_improve = 0

        # step decay
        step = max(min_step, step * decay)

        # restart if stuck
        if no_improve >= max_no_improve:
            a = np.random.random(n)
            a = _project_simplex(a)
            max_val, conv = _conv_full_max(a, L)
            cur_score = 2.0 * n * max_val
            no_improve = 0
            step = 0.4

        if time.monotonic() + 0.5 > deadline:
            break

    # return best
    return [float(x) for x in best]

    def project_simplex(v):
        n = v.shape[0]
        if n == 0:
            return v

        rho_arr = u - css / np.arange(1, n+1)
        rho = np.where(rho_arr > 0)[0]
        if rho.size:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        else:
            theta = 0.0
        return np.maximum(v - theta, 0.0)

import time
import math
import numpy as np
from scipy.signal import correlate

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_pos = np.where(rho_arr > 0)[0]
    if rho_pos.size:
        rho = rho_pos[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

def _conv_full_max(a: np.ndarray):
    """Return max element of the linear convolution a*a."""
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())  # power‑of‑two length for FFT
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(conv.max()), conv
