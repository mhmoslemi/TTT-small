import time, math, numpy as np

# --------------------------------------------------------------
# Helper functions (FFT convolution and fast score computation)
# --------------------------------------------------------------
def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of x with itself using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]
    return conv


def _score(conv: np.ndarray, n: int, s: float) -> float:
    """Exact score 2 * n * max(conv) / s**2 (same as evaluate_sequence)."""
    return 2.0 * n * conv.max() / (s * s)


# --------------------------------------------------------------
# Main candidate generator
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound evaluated by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # tiny safety margin

    # -------------------- initial sequence --------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s0 = a0.sum()
        if s0 <= 0.0:
            a0 = None
    else:
        a0 = None

    # uniform fallback of the same length (or a default length if a0 missing)
    if a0 is None:
        n = 2500
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a0.size

    if a0.sum() > 0.0:
        a0 = a0 / a0.sum()
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # possibly improve the start by taking the better of uniform / parent
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    def fast_score(arr):
        conv = _conv_fft(arr, L_fft)
        return _score(conv, n, arr.sum())

    L_fft = 1 << ((2 * n - 1).bit_length())          # power‑of‑two for FFT
    cur = a0
    cur_sc = fast_score(cur)
    uni_sc = fast_score(uniform)
    if uni_sc < cur_sc:

    best = cur.copy()

    # -------------------------- parameters --------------------------
    it = 0

    top_k = 5                    # number of largest convolution points to use

    while time.monotonic() < deadline:

        # indices of the largest entries (max‑k)
