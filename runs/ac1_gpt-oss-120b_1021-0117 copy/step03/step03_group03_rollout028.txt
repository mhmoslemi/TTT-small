import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0:
        return np.full_like(v, 1.0 / n)
    return w / w.sum()


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y using FFT (returns full length)."""
    n = len(x) + len(y) - 1
    fft_len = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:n]
    return conv


def propose_candidate(seed=42, budget_s=1100, **kwargs) -> list:
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Initial sequence (parent if present, else uniform)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = parent.copy()
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # normalise to sum = 1

    # ------------------------------------------------------------------
    # Best-so-far bookkeeping
    # ------------------------------------------------------------------
    conv = _fft_convolve(a, a)
    best = a.copy()
    best_score = 2.0 * n * conv.max()
    target_score = 1.50274349

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    power = 12                 # exponent for weight sharpening
    lr0 = 5.0                  # initial learning rate for exponentiated update
    jitter_every = 250
    jitter_std = 0.2
    max_stale = 2500
    no_improve = 0
    it = 0

    # ------------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # convolution a*a
        conv = _fft_convolve(a, a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ------- smoothed weight vector from convolution peaks -------
        # ratio in [0,1]; raise to power to concentrate on large peaks
        ratio = conv / (max_val + 1e-30)
        w_raw = ratio ** power
        w = w_raw / w_raw.sum()            # sums to 1

        # ------- gradient of the smooth max (a + conv(w, a_rev)) -------
        a_rev = a[::-1]
        conv2 = _fft_convolve(w, a_rev)    # length 3n-2
        start = n - 1
        grad2 = conv2[start:start + n]     # same size as a
        grad = a + grad2

        # ------- exponentiated (mirror) gradient step -------
        it += 1
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * grad)
        total = a.sum()
        if total <= 0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / total

        # ------- occasional random jitter to escape plateaus -------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = a * jitter
            a = a / a.sum()
            no_improve = 0

        # ------- restart if stagnated -------
        if no_improve >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = a * perturb
            a = a / a.sum()
            no_improve = 0

    # ------------------------------------------------------------------
    # Return best candidate as plain Python floats
    # ------------------------------------------------------------------
    return [float(v) for v in best]
