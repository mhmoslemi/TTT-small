import time
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Full convolution via real FFT, returning max value and its indices
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int):
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    max_val = conv.max()
    # treat numerically equal values as active
    eps = 1e-13
    max_idx = np.where(conv >= max_val - eps)[0]
    return float(max_val), max_idx


# ------------------------------------------------------------
# Sub‑gradient of max_{k} (a*a)[k] at the active indices
# ------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1)
        g[i] += 2.0 * a[k - i]
    return g


# ------------------------------------------------------------
# Random mass‑transfer perturbation (keeps L1 norm = 1)
# ------------------------------------------------------------
def _random_transfer(a: np.ndarray, rng: np.random.Generator, frac_max: float = 0.35):
    n = a.shape[0]
    i = rng.integers(0, n)
    if a[i] <= 0.0:
        return a
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    frac = rng.random() * frac_max
    delta = a[i] * frac
    cand = a.copy()
    cand[i] -= delta
    cand[j] += delta
    # small negative values may appear due to rounding – project back
    cand = _project_simplex(cand)
    return cand

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.5)

    # ------------------ initialise -------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = rng.random(256, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s
    a = _project_simplex(a)               # ensures exact sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT length

    # ------------------ initial score --------------------
    max_val, max_idx = _max_convolution(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # ------------------- parameters ----------------------
    step = 0.08                # initial additive step size
    min_step = 1e-8
    max_step = 0.5
    decay = 0.998              # gradual decay when progress stalls
    increase = 1.02            # slight boost after a successful step
    no_improve = 0
    max_no_improve = 2500      # trigger a restart
    prob_random = 0.015        # probability of a random perturbation each loop

    # ----------------------------------------------------
    while time.monotonic() < deadline:
        # ---- gradient of the active convolution term ----
        g = _subgradient_maxconv(a, max_idx)

        # ---- try an additive (projected) update ----
        cand = a - step * g
        cand = _project_simplex(cand)

        max_val_c, max_idx_c = _max_convolution(cand, L)
        cand_score = 2.0 * n * max_val_c

        improved = False
        if cand_score < cur_score - 1e-14:
            a, cur_score, max_idx = cand, cand_score, max_idx_c
            step = min(max_step, step * increase)
            improved = True
        else:
            # ---- try an exponentiated‑gradient update ----
            exp_factor = np.exp(-step * g)
            cand2 = a * exp_factor
            cand2_sum = cand2.sum()
            if cand2_sum == 0.0:
                cand2 = np.full_like(cand2, 1.0 / n)
            else:
                cand2 = cand2 / cand2_sum
            max_val_c2, max_idx_c2 = _max_convolution(cand2, L)
            cand_score2 = 2.0 * n * max_val_c2

            if cand_score2 < cur_score - 1e-14:
                a, cur_score, max_idx = cand2, cand_score2, max_idx_c2
                step = min(max_step, step * increase)
                improved = True
            else:
                # ---- no improvement – shrink step -------------------------------------------------
                step = max(min_step, step * decay)

        # ---- occasional random mass‑transfer move ----
        if rng.random() < prob_random:
            cand3 = _random_transfer(a, rng)
            max_val_c3, max_idx_c3 = _max_convolution(cand3, L)
            cand_score3 = 2.0 * n * max_val_c3
            if cand_score3 < cur_score - 1e-14:
                a, cur_score, max_idx = cand3, cand_score3, max_idx_c3
                improved = True
                step = min(max_step, step * increase)

        # ---- keep global best ------------------------------------------------------------
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            if not improved:
                no_improve += 1

        # ---- restart if stuck -----------------------------------------------------------
        if no_improve >= max_no_improve:
            a = rng.random(n, dtype=np.float64)
            a = _project_simplex(a)
            max_val, max_idx = _max_convolution(a, L)
            cur_score = 2.0 * n * max_val
            step = 0.08
            no_improve = 0

        # ---- early exit safeguard -------------------------------------------------------
        if time.monotonic() + 0.6 >= deadline:
            break

    # return the best candidate found (as a plain Python list)
    return [float(x) for x in best]
