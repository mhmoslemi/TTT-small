# ----- helper functions (top level, no closures) ---------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_full(v: np.ndarray, L: int, n: int) -> np.ndarray:
    """Linear convolution v * v (length 2*n-1) via FFT of length L."""
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[: 2 * n - 1]

            w /= w_sum

        # ----- gradient of the smooth objective --------------------------------
        rev_cur = cur[::-1]
        grad = _gradient_from_weights(w, rev_cur, L, n)

        # ----- step size (diminishing) -----------------------------------------
        eta = eta0 / np.sqrt(it)

        # ----- additive (projected‑gradient) update -----------------------------
        cand_add = cur - eta * grad
        cand_add = _project_simplex(cand_add)
        cand_add_conv = _conv_full(cand_add, L, n)
        cand_add_sc = 2.0 * n * cand_add_conv.max()

        # ----- multiplicative (exponential‑weights) update ----------------------
        exp_factor = np.exp(-eta * grad)
        cand_mul = cur * exp_factor
        s = cand_mul.sum()
        if s == 0.0:
            cand_mul = np.full_like(cur, 1.0 / n)
        else:
            cand_mul = cand_mul / s
        cand_mul_conv = _conv_full(cand_mul, L, n)
        cand_mul_sc = 2.0 * n * cand_mul_conv.max()

        # ----- choose the best candidate ----------------------------------------
        if cand_add_sc < cur_sc and cand_add_sc <= cand_mul_sc:
            cur, cur_sc = cand_add, cand_add_sc
        elif cand_mul_sc < cur_sc:
            cur, cur_sc = cand_mul, cand_mul_sc
        else:
            # occasional Metropolis‑type acceptance to escape plateaus
            prob = np.exp(-(min(cand_add_sc, cand_mul_sc) - cur_sc) / (eta + 1e-12))
            if rng.random() < prob:
                if cand_add_sc < cand_mul_sc:
                    cur, cur_sc = cand_add, cand_add_sc
                else:
                    cur, cur_sc = cand_mul, cand_mul_sc

        # ----- occasional random mass‑transfer when progress stalls --------------
        if it % 4000 == 0 and rng.random() < 0.3:
            jitter = rng.normal(scale=5e-4, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_conv = _conv_full(tmp, L, n)
            tmp_sc = 2.0 * n * tmp_conv.max()
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

    # ----- short local refinement (random pair transfers) ------------------------
    refin_time = min(0.4, deadline - time.monotonic())
    refin_deadline = time.monotonic() + refin_time
    while time.monotonic() < refin_deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * rng.random() * 0.25
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_conv = _conv_full(tmp, L, n)
        tmp_sc = 2.0 * n * tmp_conv.max()
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()

    # return the best vector as plain Python list
    return [float(x) for x in best_vec]
