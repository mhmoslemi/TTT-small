import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Find a non‑negative step‑function (length n) that minimizes
    2 * n * max(conv(a,a))  (the evaluator normalizes sum(a)=1 internally).
    Uses a soft‑max approximation of the max and gradient descent with momentum.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1. Load possible parent construction
    # ------------------------------------------------------------------
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent_raw.ndim == 1
            and parent_raw.size > 0
            and np.isfinite(parent_raw).all()
        ):
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            parent = parent_raw.copy()
            s = parent.sum()
            if s < 0.01:
                parent = None
            else:
                parent = parent / s                # simplex normalization
        else:
            parent = None
    except Exception:
        parent = None

    # ------------------------------------------------------------------
    # 2. Determine length and build initial candidates
    # ------------------------------------------------------------------
    if parent is not None:
        n = parent.shape[0]
    else:
        n = 256

    cand_list = []

    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    cand_list.append(uniform)

    # random Dirichlet‑like
    rand = rng.random(n)
    rand = np.maximum(rand, 0.0)
    rand = rand / rand.sum()
    cand_list.append(rand)

    # triangular (tent) shape
    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1.0, n - idx) - 1.0
    tent = np.maximum(tent, 0.0)
    if tent.sum() > 0:
        tent = tent / tent.sum()
        cand_list.append(tent)

    # parent if available
    if parent is not None:
        cand_list.append(parent)

    # ------------------------------------------------------------------
    # 3. Pick best starting point (fast score = 2*n*max conv)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    # length sufficient for all convolutions (a * a and a * w_rev)
    len_g = 3 * n - 2
    fft_len = 1 << ((len_g - 1).bit_length())  # power of two ≥ len_g
    best = None
    best_score = float("inf")
    for c in cand_list:
        conv = np.convolve(c, c)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best = c.copy()
    a = best.copy()

    # ------------------------------------------------------------------
    # 4. Optimisation parameters
    # ------------------------------------------------------------------
    mu = 0.85                 # momentum coefficient
    eta0 = 0.6                # initial step magnitude
    v = np.zeros_like(a)     # momentum vector
    i_arr = np.arange(n, dtype=np.int64)
    base_idx = conv_len - 1   # = 2*n-2
    rev_idx = base_idx - i_arr
    double_i = 2 * i_arr
    target = 1.50274349
    it = 0

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- full convolution and current score -----
        fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_score = 2.0 * n * conv_full.max()
        if cur_score < best_score:

            best = a.copy()
            if best_score <= target:

        # temperature grows slowly, sharpening the approximation
        lam = 1.0 + (it / 2000.0) * 30.0   # from 1 up to ~31 over 2000 iters

        w = np.exp(x)                 # all non‑negative, max weight =1

        if w_sum == 0:

        else:
            w = w / w_sum

        w_rev = w[::-1]

    # 6. Return the best found sequence as plain Python floats
    # ------------------------------------------------------------------
    return [float(x) for x in best]
