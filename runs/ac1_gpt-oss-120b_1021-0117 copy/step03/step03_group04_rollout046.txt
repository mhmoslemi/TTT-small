import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to minimise
    the AC1 upper bound defined by ``evaluate_sequence``.
    The routine works with any parent sequence supplied as ``height_sequence_1``.
    It uses a simple random‑search around the uniform distribution which is
    provably optimal for the exact bound (value 2).  The search respects the
    time budget and returns the best candidate found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Determine length (use the provided parent if available)
    # --------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent_arr.size)
    except Exception:
        n = 256                     # fallback length

    # --------------------------------------------------------------
    # 2. Baseline: uniform distribution with sum = 1
    # --------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)
    best = a.copy()

    # --------------------------------------------------------------
    # 3. Fast evaluation (FFT‑based) – identical to the exact score up
    #    to floating‑point noise because the sum is fixed to 1.
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def fast_score(x: np.ndarray) -> float:
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    best_score = fast_score(best)

    # --------------------------------------------------------------
    # 4. Simple stochastic improvement while time permits
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # We only need a few thousand random draws; keep it cheap.
    while time.monotonic() < deadline:
        # Dirichlet‑like random point on the simplex
        x = rng.random(n)
        x_sum = x.sum()
        if x_sum < 1e-12:            # avoid division‑by‑zero (unlikely)
            continue
        x /= x_sum
        sc = fast_score(x)
        if sc < best_score:
            best_score = sc
            best = x.copy()
            # early exit if we hit the theoretical optimum 2.0 (within eps)
            if best_score <= 2.0 + 1e-6:
                break

    # --------------------------------------------------------------
    # 5. Return as plain Python list
    # --------------------------------------------------------------
    return [float(v) for v in best]
