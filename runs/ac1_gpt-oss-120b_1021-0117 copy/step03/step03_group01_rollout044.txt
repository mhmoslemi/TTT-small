    a = start_vec.copy()
    n = a.size
    L = 1 << ((2*n - 1).bit_length())

    max_val, max_idx = _conv_max_and_index(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()

    step = 1.0

    decay = 0.9995
    beta = 0.9
    momentum = np.zeros_like(a)

    no_improve_count = 0
    max_no_improve = 2000  # after this, maybe add noise

    while time.monotonic() < deadline:
        # gradient for max point
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (L1=1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # safeguard against numerical issues
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w

# ----------------------------------------------------------------------
# Convolution via FFT – return maximal value and its index
# ----------------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

# ----------------------------------------------------------------------
# Core optimizer – gradient descent on the peak with momentum,
# random perturbations and symmetry enforcement.
# ----------------------------------------------------------------------

    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # hyper‑parameters
    step = 1.0
    min_step = 1e-7
    decay = 0.9995
    beta = 0.9
    momentum = np.zeros_like(a)

    no_improve = 0
    max_no_improve = 5000

    while time.monotonic() < deadline:

        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:

            g[j] = 2.0 * a[max_idx - j]
        # enforce symmetry
        g = (g + g[::-1]) * 0.5

        # ---- momentum update and candidate step ----

        cand = a - step * momentum
        cand = (cand + cand[::-1]) * 0.5   # symmetry

                best_score = cand_score
                best = cand.copy()
        else:
            # occasional uphill acceptance (simulated annealing)
            T = step
            if rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                a = cand
                cur_score = cand_score
                max_idx = cand_idx
            no_improve += 1

        # ---- random mass transfer (pairwise) ----
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.5
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = (cand2 + cand2[::-1]) * 0.5
                cand2 = _project_simplex(cand2)
                max2, idx2 = _conv_max_and_index(cand2, L)
                score2 = 2.0 * n * max2
                if score2 < cur_score:
                    a = cand2
                    cur_score = score2
                    max_idx = idx2
                    no_improve = 0
                    if score2 < best_score:
                        best_score = score2
                        best = cand2.copy()

        # ---- occasional smoothing ----
        if rng.random() < 0.005:
            kernel = np.array([0.2, 0.6, 0.2], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode='same')
            smooth = (smooth + smooth[::-1]) * 0.5
            smooth = _project_simplex(smooth)
            max_s, idx_s = _conv_max_and_index(smooth, L)
            score_s = 2.0 * n * max_s
            if score_s < cur_score:
                a = smooth
                cur_score = score_s
                max_idx = idx_s
                no_improve = 0
                if score_s < best_score:
                    best_score = score_s
                    best = smooth.copy()

        # ---- escape from stagnation ----
        if no_improve > max_no_improve:
            noise = rng.normal(scale=0.02, size=n)
            cand = a + noise
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)
            max_c, idx_c = _conv_max_and_index(cand, L)
            score_c = 2.0 * n * max_c
            if score_c < cur_score:
                a = cand
                cur_score = score_c
                max_idx = idx_c
                no_improve = 0
                if score_c < best_score:
                    best_score = score_c
                    best = cand.copy()
            else:
                no_improve = int(max_no_improve * 0.5)

        # ---- step‑size schedule ----
        step = max(min_step, step * decay)

        # ---- early termination guard ----
        if time.monotonic() + 0.5 >= deadline:
            break

    return best, best_score

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ---- start vector -------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: randomly generated simplex vector of size 2048
        n_fallback = 2048
        a0 = rng.random(n_fallback)
        a0 = _project_simplex(a0)

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    # ---- primary optimisation -----------------------------------------
    best, best_score = _optimize(a0, deadline, rng)

    # ---- secondary start if time permits -------------------------------
    if time.monotonic() + 5.0 < deadline:
        rand_vec = rng.random(a0.shape)
        rand_vec = _project_simplex(rand_vec)
        alt_best, alt_score = _optimize(rand_vec, deadline, rng)
        if alt_score < best_score:
            best, best_score = alt_best, alt_score

    # ---- return as plain Python list ----------------------------------
    return [float(x) for x in best]
