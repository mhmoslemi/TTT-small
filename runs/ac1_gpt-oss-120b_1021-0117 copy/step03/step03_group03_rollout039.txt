def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # init
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
    except Exception:
        parent = None

    if parent is not None:
        n = int(parent.size)
        a = np.clip(parent, 0.0, 1000.0)
    else:
        n = 256
        a = np.ones(n, dtype=np.float64)

    # project onto simplex (sum=1)
    a = _proj_simplex(a)

    # other parameters
    best = a.copy()
    # compute best_score using exact evaluate_sequence for safety
    best_score = float(2 * n * np.max(np.convolve(best, best)))
    # target may be lower bound
    target_score = 1.50274349
    # momentum buffer optionally
    v = np.zeros_like(a)
    beta = 0.9

    # Precompute FFT sizes
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # Additional FFT len for gradient convolution (w convolve reversed a) if we use w
    # We'll compute separate per iteration based on conv_len + n -1
    grad_len = conv_len + n - 1
    fft_len2 = 1 << ((grad_len - 1).bit_length())

    # parameters
    eta0 = 1.0
    alpha = 5.0
    jitter_every = 150
    jitter_std = 0.02
    no_improve = 0
    max_stale = 3000
    it = 0

    while time.monotonic() < deadline:
        # compute convolution via FFT
        Fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]

        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # compute gradient via softmax weighting
        # w vector (conv_len)
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            # fallback to top-k indicator
            top_k = max(5, n // 20)
            idxs = np.argpartition(conv, -top_k)[-top_k:]
            w = np.zeros_like(conv)
            w[idxs] = 1.0 / top_k
            w_sum = 1.0
        else:
            w /= w_sum

        # correlation g = 2 * (w ⋆ a)
        # compute linear correlation via convolution of w with reversed a
        rev_a = a[::-1]
        W = np.fft.rfft(w, n=fft_len2)
        A_rev = np.fft.rfft(rev_a, n=fft_len2)
        corr_full = np.fft.irfft(W * A_rev, n=fft_len2)[:grad_len]
        # extract middle n entries (index offset=n-1)
        g = 2.0 * corr_full[(n-1):(n-1 + n)]

        # momentum & step
        it += 1
        eta = eta0 / math.sqrt(it)
        v = beta * v + (1.0 - beta) * g
        a = a - eta * v
        # project back onto simplex
        a = _proj_simplex(a)

        # jitter occasionally
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v[:] = 0.0
            no_improve = 0

        # restart if stagnation
        if no_improve >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * perturb)
            v[:] = 0.0
            no_improve = 0

    # final fallback if best empty? best initialized nonempty.
    return [float(x) for x in best]

def _proj_simplex(v):
    # Ensure non-negative and sum=1.
    # v is a numpy array of shape (n,)
    # Works for any v.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, len(v)+1)) > 0)[0]
    if rho.size == 0:
        # all zero or negative => uniform
        return np.full_like(v, 1.0/len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import math
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimiser for the AC1 upper bound.
    Returns a non‑empty list of non‑negative coefficients.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # 1. Time budget
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 2. Base sequence (fallback if the parent is unavailable)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = parent.copy()
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3. Normalise to the probability simplex (sum = 1, non‑negative)
    # ------------------------------------------------------------------
    a = _proj_simplex(a)

    # ------------------------------------------------------------------
    # 4. Pre‑compute FFT sizes (for convolution & correlation)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1                      # length of a*a
    fft_len = 1 << ((conv_len - 1).bit_length())
    grad_len = conv_len + n - 1               # (2n-1)+n-1
    fft_len2 = 1 << ((grad_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 5. Optimisation parameters
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(2.0 * n * np.max(np.convolve(best, best)))
    target_score = 1.50274349

    # momentum buffer
    v_buf = np.zeros_like(a)
    beta = 0.9

    # gradient sampling
    top_k = max(5, n // 30)                  # at most a few percent of n
    top_k = min(top_k, conv_len)             # cannot exceed convolution length

    eta0 = 2.0                               # initial step magnitude
    jitter_every = 200
    jitter_std = 0.02
    max_stale = 4000
    no_improve = 0
    it = 0

    # ------------------------------------------------------------------
    # 6. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution a*a (FFT‑based) --------------------------------
        Fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]

        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # ----- update best -------------------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ----- gradient from top‑k convolution peaks -----------------------
        # construct weight vector w (sum = 1) that puts mass on the top-k entries
        idxs = np.argpartition(conv, -top_k)[-top_k:]
        w = np.zeros(conv_len, dtype=np.float64)
        w[idxs] = 1.0 / top_k

        # correlation g = 2 * (w ⋆ a)  (linear correlation via FFT)
        rev_a = a[::-1]
        W = np.fft.rfft(w, n=fft_len2)
        A_rev = np.fft.rfft(rev_a, n=fft_len2)
        corr_full = np.fft.irfft(W * A_rev, n=fft_len2)[:grad_len]
        g = 2.0 * corr_full[(n - 1):(n - 1 + n)]

        # ----- momentum step & projection ---------------------------------
        it += 1
        eta = eta0 / math.sqrt(it)
        v_buf = beta * v_buf + (1.0 - beta) * g
        a = a - eta * v_buf
        a = _proj_simplex(a)

        # ----- occasional random jitter to escape plateaus ---------------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v_buf[:] = 0.0
            no_improve = 0

        # ----- restart if stagnation ---------------------------------------
        if no_improve >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * perturb)
            v_buf[:] = 0.0
            no_improve = 0

    # ------------------------------------------------------------------
    # 7. Return the best sequence (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
