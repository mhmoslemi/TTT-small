    # Implementation:
    if np.sum(v) == 1 and np.all(v >=0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / (np.arange(len(v)) + 1))[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)

    w[w < 0] = 0

    sum_w = w.sum()
    if sum_w > 0:

    return w

def simplex_projection(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # should not happen, return uniform
        return np.full_like(v, 1.0/n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time, math, numpy as np

# ---------- Utility functions ----------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of x with itself using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)
    return conv[: 2 * x.size - 1]


def _score_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Score = 2*n*max(conv)/sum_a**2."""
    if sum_a <= 0.0:
        return np.inf
    return 2.0 * n * conv.max() / (sum_a * sum_a)


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback uniform
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # due to rounding, renormalise
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


def _initial_candidates(n: int, rng: np.random.Generator) -> list:
    """Generate a modest set of diverse start vectors (sum to 1)."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # triangular (mountain)
    if n % 2 == 0:
        half = n // 2
        tri = np.concatenate([np.arange(1, half + 1),
                              np.arange(half, 0, -1)])
    else:
        half = n // 2
        tri = np.concatenate([np.arange(1, half + 2),
                              np.arange(half + 1, 0, -1)])
    tri = tri.astype(float)
    cand.append(tri / tri.sum())

    # half‑sine
    sine = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    cand.append(sine / sine.sum())

    # exponential decays
    for lam in (0.001, 0.002, 0.005, 0.01, 0.02):
        exp = np.exp(-lam * np.arange(n))
        cand.append(exp / exp.sum())

    # Gaussian bump
    sigma = n / 6.0
    x = np.arange(n)
    gauss = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    cand.append(gauss / gauss.sum())

    # block of mass at start
    blk_len = max(1, n // 3)
    block = np.concatenate([np.full(blk_len, 1.0 / blk_len),
                            np.zeros(n - blk_len)])
    cand.append(block / block.sum())

    # parent if present
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.maximum(parent, 0.0)
        s = parent.sum()
        if s > 0.0:
            parent = parent / s
        else:
            parent = np.full_like(parent, 1.0 / parent.size)
        cand.append(parent)

    # a few random simplices
    for _ in range(3):
        r = rng.random(n)
        cand.append(r / r.sum())

    return cand

    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # safety margin so we can return before hard timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine length and auxiliary FFT size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

        n = 2000  # reasonable default if no parent is given
    L = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    candidates = _initial_candidates(n, rng)

    best_sc = np.inf
    # evaluate each cheap candidate (one conv each)

        conv = _conv_fft(cand, L)
        sc = _score_from_conv(conv, n, cand.sum())
        if sc < best_sc:
            best_sc = sc
            best_seq = cand.copy()
    # safety fallback if nothing was valid
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_sc = _score_from_conv(_conv_fft(best_seq, L), n, best_seq.sum())

    cur_seq = best_seq.copy()
    cur_sc = best_sc

    # ------------------------------------------------------------------
    # Parameters for the projected‑gradient / stochastic search
    # ------------------------------------------------------------------
    eta0 = 0.6               # base step size
    eps_move = 0.25          # fraction of mass moved in random shuffle
    jitter_scale = 1e-4
    jitter_interval = 2500

    stagnation_limit = 8000
    stale = 0                # iterations without improvement of cur_seq
    it = 0

    while time.monotonic() < deadline:

        hi = min(k, n - 1)

        if cand_sc < cur_sc:

            stale = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (eta + 1e-12))
            if rng.random() < prob:
                cur_seq, cur_sc = cand_seq, cand_sc
            stale += 1

        # ---- keep global best ----
        if cand_sc < best_sc:
            best_sc, best_seq = cand_sc, cand_seq.copy()
            stale = 0

        # ---- occasional random mass‑shuffle move ----
        if rng.random() < random_move_prob:
            i = rng.integers(0, n)
            if cur_seq[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur_seq[i] * rng.random() * eps_move
                tmp = cur_seq.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _simplex_projection(tmp)
                tmp_sc = _score_from_conv(_conv_fft(tmp, L), n, tmp.sum())
                if tmp_sc < cur_sc:

            tmp_sc = _score_from_conv(_conv_fft(tmp, L), n, tmp.sum())
