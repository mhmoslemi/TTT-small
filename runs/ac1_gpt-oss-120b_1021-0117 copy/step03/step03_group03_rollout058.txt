import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Gradient‑based search for a step‑function minimizing the AC1 upper bound.
    Uses a smooth (power‑weighted) sub‑gradient of the self‑convolution,
    momentum, occasional jitter and restarts.  Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time limit (leave a little margin for the return)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Starting vector (fallback to uniform if parent unavailable)
    # ------------------------------------------------------------------
    try:
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        n = int(a.size)
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # simplex normalisation (sum = 1)

    # ------------------------------------------------------------------
    # Pre‑computations for FFT‑based convolutions
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # Helper to compute the score (2·n·max(conv)) and the convolution
    # ------------------------------------------------------------------
    def _score_and_conv(x):
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max(), conv

    # ------------------------------------------------------------------
    # Initialise best solution
    # ------------------------------------------------------------------
    best = a.copy()
    best_score, _ = _score_and_conv(best)

    # also try pure uniform (may be better for some parents)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_score, _ = _score_and_conv(uniform)
    if u_score < best_score:
        best = uniform.copy()
        best_score = u_score
        a = uniform.copy()

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    eta0 = 1.0          # initial step magnitude
    beta = 0.9          # momentum decay
    power = 8.0         # exponent for power‑weighted smoothing
    jitter_every = 200  # steps between multiplicative jitter
    jitter_std = 0.02   # std‑dev of log‑normal jitter
    max_stale = 2500    # steps without improvement before a restart
    target_score = 1.50274349

    v = np.zeros_like(a)          # momentum buffer
    it = 0
    stale = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cur_score, conv = _score_and_conv(a)

        # ---- keep the best global candidate --------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            stale = 0
            if best_score <= target_score:
                break
        else:
            stale += 1

        # ---- smooth sub‑gradient (power‑weighted) ---------------------------
        max_val = conv.max()
        # avoid divide‑by‑zero (should never happen with non‑empty a)
        if max_val == 0:
            weights = np.ones_like(conv)
        else:
            ratios = conv / max_val                # in [0,1]
            weights = ratios ** power              # emphasise large entries

        # convolution of weights with reversed a gives the gradient
        a_rev = a[::-1]
        w_fft = np.fft.rfft(weights, n=fft_len)
        rev_fft = np.fft.rfft(a_rev, n=fft_len)
        grad_full = np.fft.irfft(w_fft * rev_fft, n=fft_len)[:conv_len]
        g = 2.0 * grad_full[n - 1:n - 1 + n]      # slice length n

        # ---- momentum step with diminishing learning rate ------------------
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g
        a = _proj_simplex(a - eta * v)

    # Return the best sequence (as plain Python floats)
    # ------------------------------------------------------------------
