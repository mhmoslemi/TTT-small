import time, math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # force exact sum 1 to avoid drift
    s = w.sum()
    if s > 0:
        w /= s
    return w

# Analytic sqrt series (fallback if no parent is present)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ------------------------------------------------------------
# Full convolution via real FFT (returns the whole array)
# ------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return conv

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # -------------------- initialise --------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = _sqrt_series(256)

    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a[:] = 1.0 / a.size
    else:
        a = _project_simplex(a / a.sum())

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())      # next power‑of‑two for FFT

    # initial evaluation
    conv = _conv_full(a, L)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score

    # -------------------- optimisation parameters --------------------
    step = 0.5               # base step magnitude (in L2 norm)
    min_step = 1e-6
    step_decay = 0.9995

    temp = 0.02              # annealing temperature
    temp_decay = 0.9995

    no_improve = 0
    max_no_improve = 500

    prob_random = 0.025
    max_transfer = 0.3

    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- current convolution & gradient from all maximal lags ----
        conv = _conv_full(a, L)
        max_val = conv.max()
        # tolerance to catch ties caused by floating‑point noise
        tol = 1e-12 * max_val
        max_idxs = np.where(conv >= max_val - tol)[0]

        g = np.zeros_like(a)
        for k in max_idxs:
            lo = k - (n - 1)
            if lo < 0:
                lo = 0
            hi = k
            if hi > n - 1:
                hi = n - 1
            j = np.arange(lo, hi + 1, dtype=int)
            g[j] += 2.0 * a[k - j]

        # ---- gradient step (norm‑scaled) ----
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            direction = g / norm_g
        else:
            direction = g
        cand = a - step * direction
        cand = _project_simplex(cand)

        # ---- evaluate candidate ----
        conv_c = _conv_full(cand, L)
        cand_score = 2.0 * n * conv_c.max()

        # ---- accept / reject (simulated annealing) ----
        accept = False
        if cand_score < cur_score - 1e-15:
            accept = True
        else:
            if temp > 0:
                prob = math.exp(-(cand_score - cur_score) / temp)
                if rng.random() < prob:
                    accept = True

        if accept:
            a = cand
            cur_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

        # ---- global best tracking ----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        # ---- occasional random mass‑transfer move ----
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                frac = rng.random() * max_transfer
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                conv2 = _conv_full(cand2, L)
                cand2_score = 2.0 * n * conv2.max()
                accept2 = False
                if cand2_score < cur_score - 1e-15:
                    accept2 = True
                else:
                    if temp > 0:
                        prob2 = math.exp(-(cand2_score - cur_score) / temp)
                        if rng.random() < prob2:
                            accept2 = True
                if accept2:
                    a = cand2
                    cur_score = cand2_score
                    no_improve = 0
                    if cur_score < best_score:
                        best_score = cur_score
                        best = a.copy()

        # ---- decay step size and temperature ----
        step = max(min_step, step * step_decay)
        temp = max(1e-9, temp * temp_decay)

        # ---- restart if stuck ----
        if no_improve >= max_no_improve:
            # restart near current best with slight perturbation
            a = best + 0.01 * rng.random(n)
            a = _project_simplex(a)
            cur_score = best_score
            no_improve = 0
            step = 0.5
            temp = 0.02

        # ---- safety break ----
        if time.monotonic() + 0.5 > deadline:
            break

    # Return plain Python list as required
    return [float(x) for x in best]
