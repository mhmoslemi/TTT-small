import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:                     # degenerate case
        w[:] = 1.0 / n
    return w

# ------------------------------------------------------------
# Analytic sqrt‑series (a*a = 1+…+x^{2n‑2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ------------------------------------------------------------
# Convolution (FFT) – return maximal value and its index
# ------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

        rand_vec = rng.random(a0.shape)
        rand_vec = _project_simplex(rand_vec)
        alt_best, alt_score = _optimize(rand_vec, deadline, rng)
        if alt_score < best_score:
            best, best_score = alt_best, alt_score

    # ---- fallback safety (in case optimisation failed) -----------------
    if best_score == float('inf') or not np.isfinite(best_score):
        best = a0.copy()

    # ---- return plain python list ---------------------------------------
    return [float(x) for x in best]
