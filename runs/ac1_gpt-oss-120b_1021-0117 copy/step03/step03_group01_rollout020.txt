import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # numerical safety: if rounding produced zero sum, fall back to uniform
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w


# ----------------------------------------------------------------------
# Compute full linear convolution via FFT (returns float array)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a * a using FFT, length = 2*len(a)-1."""
    fv = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fv * fv, n=L)[: 2 * len(a) - 1]
    return conv_full

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ---- initial vector -------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: generate a simple sqrt‑series of the same length as the
        # budget permits (use 256 as a generic size)
        def _sqrt_series(m):
            s = np.empty(m, dtype=np.float64)
            s[0] = 1.0
            for k in range(1, m):
                # convolution of prefix with itself, ignoring the newest term
                if k == 1:
                    inner = 0.0
                else:
                    inner = float(np.dot(s[1:k], s[k - 1 : 0 : -1]))
                s[k] = (1.0 - inner) / 2.0
            return s
        a0 = _sqrt_series(256)

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    # ---- first optimisation run -----------------------------------------
    best, best_score = _optimize(a0, deadline, rng)

    # ---- optional second start (random) ---------------------------------
    if time.monotonic() + 6.0 < deadline:
        rand_vec = rng.random(a0.shape)
        rand_vec = _project_simplex(rand_vec)
        alt_best, alt_score = _optimize(rand_vec, deadline, rng)
        if alt_score < best_score:
            best, best_score = alt_best, alt_score

    # ---- final polishing via tiny local moves ---------------------------
    if time.monotonic() + 2.0 < deadline:
        best = _local_search(best, deadline, rng)

    # Return a plain Python list as required
    return [float(x) for x in best]
