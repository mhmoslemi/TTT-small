import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast score for a vector with sum == 1:
#   2 * n * max(conv)  where conv = a * a (linear convolution)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # -------------------- initial vector --------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # simple exponentially decaying start, length taken from the test harness later
        n_guess = 2500
        a0 = np.exp(-np.arange(n_guess) / 200.0)
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                     # normalise to sum == 1

    n = a0.size
    # FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # -------------------- bookkeeping --------------------
    best_vec = a0.copy()
    best_sc = _score(best_vec, L)

    cur = best_vec.copy()
    cur_sc = best_sc

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t_iter = 0

    # smoothing / step schedule
    eps0 = 0.5
    eps_decay = 0.9997
    eps_min = 1e-8
    lr0 = 0.4
    no_improve = 0
    TARGET = 1.50274349

    # -------------------- optimisation loop --------------------
    while time.monotonic() < deadline:
        t_iter += 1

        # ----- convolution of current iterate -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- smoothed max weights (softmax) -----
        eps = max(eps_min, eps0 * (eps_decay ** t_iter))
        w = np.exp((conv - max_val) / eps)          # stable: max element -> 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)   # length L
        idx = (2 * n - 2) - np.arange(n)               # map to gradient indices

        m_hat = m / (1.0 - beta1 ** t_iter)

    # ------------------------------------------------------------

    if best_sc == np.inf or best_sc > 10.0:

    return [float(x) for x in best_vec]
