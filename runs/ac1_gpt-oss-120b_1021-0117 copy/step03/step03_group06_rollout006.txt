def score_seq(seq, n, L):
    # seq: numpy array float64
    # note: using rfft; but we need to ensure correct type.
    # compute rfft of length L not full length.
    fx = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[:2*n-1]
    max_conv = conv.max()
    sum_a = seq.sum()
    return 2.0 * n * max_conv / (sum_a * sum_a)

Time complexity O(L log L) ~ O(n log n). Good.

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math, random

    start=time.monotonic()
    deadline=start+budget_s-0.1 (small margin)

    # get n
    if 'height_sequence_1' in globals():
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback uniform of length 2000
        init_seq = np.full(2000, 1.0/2000, dtype=np.float64)

    # ensure nonneg
    init_seq = np.maximum(init_seq, 0.0)
    # Normalize to sum 1
    s = init_seq.sum()
    if s == 0:
        init_seq = np.full_like(init_seq, 1.0 / init_seq.size)
    else:
        init_seq = init_seq / s

    n = init_seq.size
    L = 1 << ((2*n - 1).bit_length())  # power of two length for FFT

    # function to compute score
    def compute_score(arr):
        # arr is numpy 1d array
        # Using FFT
        fx = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fx * fx, n=L)[:2*n-1]
        max_conv = conv.max()
        sum_a = arr.sum()
        if sum_a < 1e-12:
            return float('inf')
        return 2.0 * n * max_conv / (sum_a * sum_a)

    # Initialize population
    pop_size = 12  # moderate
    rng = np.random.default_rng(seed)
    pop = np.empty((pop_size, n), dtype=np.float64)
    scores = np.empty(pop_size, dtype=np.float64)

    # initial individuals
    # first individual is init_seq (height_sequence_1 normalized)
    pop[0] = init_seq.copy()
    scores[0] = compute_score(pop[0])
    best = pop[0].copy()
    best_score = scores[0]

    # fill rest randomly
    for i in range(1, pop_size):
        # random Dirichlet: sample exponential
        rand = rng.random(n)
        rand_sum = rand.sum()
        arr = rand / rand_sum
        pop[i] = arr
        scores[i] = compute_score(arr)
        if scores[i] < best_score:
            best_score = scores[i]
            best = arr.copy()

    # DE parameters
    F = 0.7
    CR = 0.9
    # As we may adapt F per generation?

    # iteration counters
    gen = 0

    # main loop
    while time.monotonic() < deadline:
        gen += 1
        for i in range(pop_size):
            # select three distinct others
            idxs = list(range(pop_size))
            idxs.remove(i)
            a,b,c = rng.choice(idxs, size=3, replace=False)

            mutant = pop[a] + F * (pop[b] - pop[c])

            # enforce nonnegativity
            np.clip(mutant, 0, None, out=mutant)

            # Renorm
            sum_mut = mutant.sum()
            if sum_mut <= 0:
                # random reinitialize
                mutant = rng.random(n)
                sum_mut = mutant.sum()
                mutant = mutant / sum_mut
            else:
                mutant = mutant / sum_mut

            # crossover
            cross = rng.random(n) < CR
            # ensure at least one dimension from mutant
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])

            # renorm again (since trial may have changed sum)
            sum_trial = trial.sum()
            if sum_trial <= 0:
                trial = rng.random(n)
                sum_trial = trial.sum()
                trial = trial / sum_trial
            else:
                trial = trial / sum_trial

            # evaluate trial score
            trial_score = compute_score(trial)

            # replacement condition
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
            # else maybe allow occasional acceptance with probability (like SA)
            else:
                # optionally allow occasional acceptance
                # temperature schedule: decreasing over time
                # Using simple probability; not necessary for improvement.

                pass

        # periodic jitter on best to explore

            # jitter

            cand = best + jitter

import time
import math
import numpy as np

TARGET = 1.50274349  # stop early if we beat this


def _next_pow2(x: int) -> int:
    """Return the next power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _full_conv_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of seq with itself using an FFT of appropriate length."""
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    return conv


def _score_seq(seq: np.ndarray) -> float:
    """Exact score as used by evaluate_sequence."""
    n = seq.shape[0]
    conv = _full_conv_fft(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


def _gradient_step(seq: np.ndarray, eta: float) -> np.ndarray:
    """Mirror‑descent step that reduces the active autocorrelation peak."""
    n = seq.shape[0]
    conv = _full_conv_fft(seq)
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)

    # build sub‑gradient
    g = np.zeros_like(seq)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g[idx] = 2.0 * seq[k - idx]

    mul = np.exp(-eta * g)
    new = seq * mul
    sm = new.sum()
    if sm <= 0.0 or not np.isfinite(sm):
        return seq
    return new / sm


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that attempts to
    minimise the AC‑1 upper bound.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # tiny safety margin

    # ----------------------------------------------------------------------
    # Initialise a base sequence (use the provided parent if any)
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        base = np.full(2000, 1.0 / 2000, dtype=np.float64)

    base = np.maximum(base, 0.0)
    sb = base.sum()
    if sb <= 0.0:
        base = np.full_like(base, 1.0 / base.size)
    else:
        base = base / sb                     # normalise to the probability simplex

    n = base.shape[0]

    # ----------------------------------------------------------------------
    # Build a modest population for differential evolution
    # ----------------------------------------------------------------------
    pop_size = 12
    pop = np.empty((pop_size, n), dtype=np.float64)
    scores = np.empty(pop_size, dtype=np.float64)

    # first individual = the parent (normalised)
    pop[0] = base
    scores[0] = _score_seq(pop[0])

    # a few gradient‑improved copies of the parent
    for i in range(1, 4):
        tmp = base.copy()
        for _ in range(3):
            tmp = _gradient_step(tmp, eta=0.5)
        pop[i] = tmp
        scores[i] = _score_seq(tmp)

    # the remaining individuals are random Dirichlet draws
    for i in range(4, pop_size):
        rand = rng.random(n)
        pop[i] = rand / rand.sum()
        scores[i] = _score_seq(pop[i])

    best_idx = int(scores.argmin())
    best = pop[best_idx].copy()
    best_score = float(scores[best_idx])

    # ----------------------------------------------------------------------
    # Differential evolution parameters
    # ----------------------------------------------------------------------
    F = 0.78                     # differential weight
    CR = 0.9                     # crossover probability
    grad_step_counter = 0

    # ----------------------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- differential evolution sweep -----
        for i in range(pop_size):

            idxs = list(range(pop_size))
            idxs.remove(i)

            mutant = pop[a] + F * (pop[b] - pop[c])

                s_trial = trial.sum()
