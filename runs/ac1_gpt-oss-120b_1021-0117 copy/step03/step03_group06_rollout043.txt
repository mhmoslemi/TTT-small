import time, math, numpy as np

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full convolution of x with itself using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]
    return conv


def _score_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Score = 2*n*max(conv)/sum_a**2 (identical to evaluate_sequence)."""
    return 2.0 * n * conv.max() / (sum_a * sum_a)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # ----------------------------------------------------------------------
    # Initialise: use the provided parent if present, otherwise a simple uniform.
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = np.full(2000, 1.0 / 2000, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                     # put onto the probability simplex

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two length for FFT

    cur = a.copy()
    cur_conv = _conv_fft(cur, L)
    cur_sc = _score_from_conv(cur_conv, n, cur.sum())

    best = cur.copy()
    best_sc = cur_sc

    # ----------------------------------------------------------------------
    # Hyper‑parameters for the optimisation loop.
    # ----------------------------------------------------------------------
    it = 0
    alpha0 = 1.0               # initial step multiplier (larger than before)
    beta_mom = 0.9             # momentum decay
    eps_move = 0.25            # max fraction of mass shuffled in a random move
    jitter_scale = 5e-4        # scale of Gaussian jitter around the best point
    temp0 = 0.2                # base temperature for uphill Metropolis steps
    no_improve = 0
    mom = np.zeros_like(cur)   # momentum accumulator for the sub‑gradient

    while time.monotonic() < deadline:
        it += 1

        # ---------- current convolution & dominant index ----------
        conv = _conv_fft(cur, L)
        k = int(conv.argmax())                     # shift where max occurs
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # ---------- sub‑gradient of max‑convolution (mirrored step) ----------
        g = np.zeros_like(cur)
        # each contributing pair appears twice (except the centre), hence factor 2
        g[idx] = 2.0 * cur[k - idx]

        # momentum smoothing of the sub‑gradient
        mom = beta_mom * mom + (1.0 - beta_mom) * g

        # ---------- multiplicative (exponentiated) update ----------
        eta = alpha0 / math.sqrt(it)               # decreasing step size
        # protect against overflow in the exponent
        mul = np.exp(-eta * mom)
        cand = cur * mul
        cand_sum = cand.sum()
        if cand_sum <= 0.0 or not np.isfinite(cand_sum):
            cand = np.full_like(cur, 1.0 / n)
        else:
            cand = cand / cand_sum

        cand_sc = _score_from_conv(_conv_fft(cand, L), n, cand.sum())

        # ---------- acceptance (mostly greedy, occasional uphill) ----------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # Metropolis criterion with a slowly decaying temperature
            prob = math.exp(-(cand_sc - cur_sc) / (temp0 + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        if cand_sc < best_sc:
            best_sc, best = cand_sc, cand.copy()

        # ---------- directed random mass‑transfer move ----------
        if rng.random() < 0.08:
            # pick a few indices with largest sub‑gradient (sources)
            src = np.argpartition(-g, 3)[:3]
            # pick a few indices with smallest current weight (destinations)
            dst = np.argpartition(cur, 3)[:3]
            tmp = cur.copy()
            for i in src:
                if tmp[i] > 0.0:
                    j = rng.choice(dst)
                    delta = tmp[i] * eps_move * rng.random()
                    tmp[i] -= delta
                    tmp[j] += delta
            tmp = np.maximum(tmp, 0.0)
            tmp_sum = tmp.sum()
            if tmp_sum > 0.0:
                tmp = tmp / tmp_sum
                tmp_sc = _score_from_conv(_conv_fft(tmp, L), n, tmp.sum())
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best = tmp_sc, tmp.copy()

        # ---------- jittered restart from the current best ----------
        if it % 2500 == 0:
            jitter = rng.normal(scale=jitter_scale, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp_sum = tmp.sum()
            if tmp_sum > 0.0:
                tmp = tmp / tmp_sum
                tmp_sc = _score_from_conv(_conv_fft(tmp, L), n, tmp.sum())
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best = tmp_sc, tmp.copy()
                no_improve = 0

        # ---------- long‑stagnation restart ----------
        if no_improve > 12000:
            cur = best + rng.normal(scale=jitter_scale * 5, size=n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur_sc = _score_from_conv(_conv_fft(cur, L), n, cur.sum())
            no_improve = 0

        # ---------- early exit if we have already beaten the target ----------
        if best_sc < 1.50274349:
            break

    # Return a plain Python list (the evaluator expects a list, not an ndarray)
    return [float(x) for x in best]
