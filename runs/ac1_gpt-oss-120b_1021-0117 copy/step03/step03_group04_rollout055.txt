def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # reserve buffer

    # Get sequence length n from height_sequence_1 if present else fallback n=256
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_raw.size
        # Ensure nonnegative and within bounds
        parent_raw = np.clip(parent_raw, 0.0, 1000.0)
        # Normalize to sum=1
        parent_sum = parent_raw.sum()
        if parent_sum < 0.01:
            parent_raw = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            parent_raw = parent_raw / parent_sum
    except Exception:
        n = 256
        parent_raw = None

    # define function to compute max convolution via FFT
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def max_conv(a):
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return conv.max(), np.argmax(conv), conv

    # Simplex projection (existing code)
    def proj_simplex(v):
        # v may be any real array; output nonnegative sum=1
        # Implementation using sorting.
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(1, u.size + 1)
        cond = u - cssv / ind > 0
        if not np.any(cond):
            # return uniform
            return np.full_like(v, 1.0 / v.size)
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    # Initialize a population of candidates
    init_candidates = []

    # Add parent if valid
    if parent_raw is not None:
        init_candidates.append(parent_raw.copy())

    # uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # tent shape
    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1, n - idx) - 1
    tent = np.maximum(tent, 0)
    if tent.sum() > 0:
        tent = tent / tent.sum()
    else:
        tent = init_candidates[-1].copy()
    init_candidates.append(tent)

    # random Dirichlet
    rand = rng.random(n)
    rand = np.maximum(rand, 0)
    rand = rand / rand.sum()
    init_candidates.append(rand)

    # Gaussian-like shape symmetrical
    sigma = n / 6.0  # choose width
    gauss = np.exp(-0.5 * ((idx - n/2) / sigma) ** 2)
    gauss = gauss / gauss.sum()
    init_candidates.append(gauss)

    # Additional random perturbations of uniform maybe 3 more
    for _ in range(3):
        r = rng.random(n)
        r = r / r.sum()
        init_candidates.append(r)

    best_seq = None
    best_score = float('inf')
    # for each candidate run gradient descent for a limited number of iterations or until deadline.

    # We'll define a function to run descent from start:
    def run_descent(start):
        a = start.copy()
        it = 0
        # optionally keep previous a for momentum
        a_prev = a.copy()
        while time.monotonic() < deadline:
            # compute conv and max
            conv_max, max_idx, conv_full = max_conv(a)
            # compute objective
            obj = 2.0 * n * conv_max
            # update best
            nonlocal best_seq, best_score
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                # early exit if good enough
                if best_score <= target:  # define target later
                    return
            it += 1
            # step size schedule
            eta = 0.5 / np.sqrt(it)  # you can adjust
            # gradient of max convex at index max_idx
            # compute j = max_idx - i for each i
            i = np.arange(n)
            j = max_idx - i
            mask = (j >= 0) & (j < n)
            g = np.zeros_like(a)
            g[mask] = 2.0 * a[j[mask]]
            # gradient descent step
            a_new = a - eta * g
            # project onto simplex
            a_new = proj_simplex(a_new)
            # optionally add small jitter every few iterations
            if it % 200 == 0:
                jitter = np.exp(rng.normal(0.0, 0.01, size=n))
                a_new = proj_simplex(a_new * jitter)
            a = a_new
        # loop ends due to time
        return

    # set target
    target = 1.50274349

    for cand in init_candidates:
        if time.monotonic() >= deadline:
            break
        run_descent(cand)

    # If best_seq still None (shouldn't happen) fallback to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Return as list of floats
    return [float(x) for x in best_seq]

def optimize(start, max_iters, deadline):
    a = start.copy()
    v = np.zeros_like(a)
    mu = 0.9
    eta0 = 1.0
    eps_factor = 0.1
    eps_decay = 0.9995
    for it in range(1, max_iters+1):
        # check time
        if time.monotonic() >= deadline:
            break
        # compute conv and max
        fa = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fa * fa, n=L)[:conv_len]
        cur_max = conv_full.max()
        if cur_max <= 0: break
        # update best
        cur_score = 2.0 * n * cur_max / (a.sum()**2)  # sum ~1
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            if cur_score <= target:
                return True  # early stop

        # compute weights (softmax)
        eps = max(cur_max * eps_factor, 1e-12)
        w = np.exp((conv_full - cur_max) / eps)
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        # gradient via convolution of a_rev and w
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, n=L)
        fw = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(fa_rev * fw, n=L)
        g = 2.0 * grad_full[n-1:n-1+n]  # length n

        # momentum
        v = mu * v + (1.0 - mu) * g

        eta = eta0 / np.sqrt(it)

        a = a - eta * v

            a = proj_simplex(a * jitter)

        # maybe decay eps_factor
        eps_factor *= eps_decay
    return False
