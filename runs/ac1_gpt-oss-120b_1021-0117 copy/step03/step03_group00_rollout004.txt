import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Scoring function (exact same formula as evaluate_sequence for sum=1)
# ------------------------------------------------------------
def _score(v: np.ndarray, L: int, n: int) -> float:
    af = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
