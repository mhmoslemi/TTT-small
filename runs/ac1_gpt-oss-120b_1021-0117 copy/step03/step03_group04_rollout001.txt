def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # reserve 5 secs

    # load parent if exists

        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if not (parent_arr.ndim == 1 and parent_arr.size > 0 and np.isfinite(parent_arr).all()):
            raise ValueError
    except Exception:
        parent_arr = None

    # Determine n
    if parent_arr is not None:
        n = int(parent_arr.size)
    else:
        n = 1024  # fallback length
    # create initial candidates
    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    # tent shape
    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1, n - idx) - 1
    tent = np.clip(tent, 0, None)
    if tent.sum() > 0:
        tent = tent / tent.sum()
    else:
        tent = uniform.copy()
    # random Dirichlet
    rand_vec = rng.random(n)
    rand_vec = np.clip(rand_vec, 0, None)
    rand_vec = rand_vec / rand_vec.sum()
    candidates = [uniform, tent, rand_vec]
    if parent_arr is not None:
        a_parent = np.maximum(parent_arr, 0.0)
        s = a_parent.sum()
        if s >= 0.01:
            a_parent = a_parent / s
        else:
            a_parent = uniform.copy()
        candidates.append(a_parent)

    # fast scoring
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    def fast_score(x):
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    best = None
    best_score = float('inf')
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()

    a = best.copy()
    # Parameters
    eta0 = 0.5
    mu = 0.9
    v = np.zeros_like(a)
    it = 0
    target = 1.50274349
    # tolerance for max identification
    tol_rel = 1e-12

    while time.monotonic() < deadline:
        it += 1
        # compute conv and score
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            if best_score <= target:
                break

        # compute subgradient for max
        max_val = conv.max()
        tol = max(tol_rel * max_val, 1e-12)
        # get indices where conv within tol of max
        idx_max = np.where(conv >= max_val - tol)[0]
        # compute gradient
        g = np.zeros_like(a)
        # compute contributions
        for idx_shift in idx_max:
            d = idx_shift - (n - 1)
            if d >= 0:
                length = n - d
                # forward term: a[d:] adds to g[:length]
                g[:length] += a[d:]
                # backward term: a[:length] adds to g[d:]
                g[d:] += a[:length]
            else:
                d_abs = -d
                length = n - d_abs
                # forward term: a[:length] adds to g[d_abs:]
                g[d_abs:] += a[:length]
                # backward term: a[d_abs:] adds to g[:length]
                g[:length] += a[d_abs:]

        # momentum update
        v = mu * v + (1.0 - mu) * g

        # step size schedule
        eta = eta0 / np.sqrt(it)

        # projected gradient descent
        a = a - eta * v

        # ensure nonnegativity and simplex (use projection)
        a = _proj_simplex(a)

    # Return as Python list

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Projected (sub)gradient descent on the max–autocorrelation objective.
    Returns a non‑empty list of non‑negative coefficients.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Time budget
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2. Load parent if present, otherwise fall back to generated candidates
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if not (parent_arr.ndim == 1 and parent_arr.size > 0 and np.isfinite(parent_arr).all()):
            raise ValueError
    except Exception:
        parent_arr = None

    if parent_arr is not None:
        n = int(parent_arr.size)
        a_parent = np.maximum(parent_arr, 0.0)
        s = a_parent.sum()
        a_parent = a_parent / s if s >= 0.01 else np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 256  # reasonable default when no parent is given
        a_parent = None

    # ------------------------------------------------------------------
    # 3. Generate a few cheap starting points
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1, n - idx) - 1.0
    tent = np.clip(tent, 0.0, None)
    if tent.sum() > 0:
        tent = tent / tent.sum()
    else:
        tent = uniform.copy()

    rand_vec = rng.random(n)
    rand_vec = np.clip(rand_vec, 0.0, None)
    rand_vec = rand_vec / rand_vec.sum()

    candidates = [uniform, tent, rand_vec]
    if a_parent is not None:
        candidates.append(a_parent)

    # ------------------------------------------------------------------
    # 4. Helper: fast evaluation (sum is 1, so the bound equals 2*n*max(conv))
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def fast_score(x: np.ndarray) -> float:
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # 5. Pick best initial candidate
    # ------------------------------------------------------------------
    best = None
    best_score = float("inf")
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    a = best.copy()

    # ------------------------------------------------------------------
    # 6. Optimisation parameters
    # ------------------------------------------------------------------
    eta0 = 0.5          # base step size
    mu = 0.9            # momentum decay
    v = np.zeros_like(a)
    it = 0
    target = 1.50274349
    tol_rel = 1e-12
    last_improve = 0

    # ------------------------------------------------------------------
    # 7. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution & score -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

            if best_score <= target:
                break

        # ----- sub‑gradient of max(conv) -----
        max_val = conv.max()
        tol = max(tol_rel * max_val, 1e-12)
        idx_max = np.where(conv >= max_val - tol)[0]

        for idx_shift in idx_max:

            if d >= 0:
                length = n - d
                g[:length] += a[d:]          # forward term
                g[d:]     += a[:length]      # backward term
            else:
                d_abs = -d
                length = n - d_abs
                g[d_abs:] += a[:length]      # forward term (negative shift)
                g[:length] += a[d_abs:]      # backward term

        # ----- momentum update -----
        v = mu * v + (1.0 - mu) * g

        # ----- step size -----
        eta = eta0 / np.sqrt(it)

        # ----- descent step (normalize direction) -----
        direction = v / (np.linalg.norm(v) + 1e-12)
        a = a - eta * direction

        # ----- projection onto simplex -----

    # 8. Return the best sequence found
    # ------------------------------------------------------------------
    if best is None:
        best = a
    return [float(x) for x in best]
