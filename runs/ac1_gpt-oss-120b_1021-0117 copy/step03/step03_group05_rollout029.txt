    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)

        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

    a0 = _project_simplex(a0)

        fv = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
        return conv.max()

    def score(v):
        return 2.0 * n * conv_max(v) / (v.sum() ** 2)

    # Keep best

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w.fill(1.0 / n)
    return w


def _conv_full(v):
    """Full convolution v*v using an FFT of minimal power‑of‑two length."""
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


def _score(v):
    """Exact score used by the evaluator (scale‑invariant)."""
    n = v.shape[0]
    conv = _conv_full(v)
    return float(2.0 * n * conv.max() / (v.sum() ** 2))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)                           # sum = 1
    n = a.shape[0]

    cur = a.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------------
    # parameters
    # ------------------------------------------------------------------
    base_alpha = 0.8
    base_eta = 0.7
    K = max(10, int(0.015 * (2 * n - 1)))   # active set size
    max_no_improve = 4000
    no_improve = 0
    it = 0

    # ------------------------------------------------------------------

    while time.monotonic() < deadline and it < 180:
        it += 1

        rnd = rng.dirichlet(np.full(n, alpha))
        sc = _score(rnd)
        if sc < best_sc:

            cur, cur_sc = best_vec.copy(), best_sc

    # ------------------------------------------------------------------

    while time.monotonic() < deadline:
