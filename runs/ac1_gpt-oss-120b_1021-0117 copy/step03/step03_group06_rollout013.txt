def _full_conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution x*x using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    return np.fft.irfft(fx * fx, n=L)[: 2 * x.shape[0] - 1]

def _grad_from_w(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """Gradient of Σ_i w_i·(a*a)[i] w.r.t. a (mirror‑descent direction)."""
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=L)
    fa = np.fft.rfft(a_rev, n=L)
