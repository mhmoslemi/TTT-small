import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (non‑negative, sum=1). """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, u.size + 1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / u.size
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Returns a list of non‑negative coefficients (float) that minimises
    evaluate_sequence(seq).  The method uses momentum‑gradient descent on the
    largest convolution peaks with projection onto the simplex.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2.  Starting sequence (fallback if parent not present)
    # ------------------------------------------------------------------
    try:
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
    except Exception:
        a = np.ones(256, dtype=np.float64)

    n = a.size

    # normalise to the probability simplex (sum = 1, non‑negative)
    a = a - np.min(a)                     # shift to non‑negative
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / a.sum()
    a = _proj_simplex(a)

    # ------------------------------------------------------------------
    # 3.  Baselines and best‑so‑far tracking
    # ------------------------------------------------------------------
    def _score_from_max(max_val):
        return 2.0 * n * max_val

    # current best (list) and its score
    best = a.copy()
    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    # evaluate initial scores (max conv computed via FFT)
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def _max_conv(x):
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return conv.max()

    best_score = _score_from_max(_max_conv(best))
    uni_score = _score_from_max(_max_conv(uniform))
    if uni_score < best_score:
        best = uniform.copy()
        best_score = uni_score
        a = uniform.copy()

    # ------------------------------------------------------------------
    # 4.  Optimisation parameters
    # ------------------------------------------------------------------
    eta0 = 0.9               # initial step magnitude
    beta = 0.9               # momentum factor
    top_k = 30               # number of convolution peaks used for gradient
    jitter_every = 250       # iterations between random perturbations
    jitter_std = 0.004       # log‑normal jitter strength
    max_stale = 6000         # restart after this many non‑improving steps
    target_score = 1.50274349

    v = np.zeros_like(a)      # momentum buffer
    it = 0
    stale = 0

    # ------------------------------------------------------------------
    # 5.  Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution (FFT) -------------------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        cur_score = _score_from_max(max_val)

        # ----- update best -------------------------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            stale = 0
            if best_score <= target_score:
                break
        else:
            stale += 1

        # ----- gradient from top‑k peaks ----------------------------------------
        if conv.size > top_k:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        else:
            idxs = np.arange(conv.size)

        g = np.zeros_like(a)
        for k in idxs:
            # valid i such that 0 <= i < n and 0 <= k-i < n
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            partner = k - i_range
            g[i_range] += a[partner]
        # average over selected peaks and account for the factor 2 in the gradient
        g = 2.0 * g / max(1, len(idxs))

        # ----- momentum step & projection ---------------------------------------
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g
        a = _proj_simplex(a - eta * v)

        # ----- occasional jitter to escape plateaus -----------------------------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v.fill(0.0)
            stale = 0

        # ----- restart if stagnated ---------------------------------------------
        if stale >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * perturb)
            v.fill(0.0)
            stale = 0

    # ------------------------------------------------------------------
    # 6.  Return best sequence as plain python floats
    # ------------------------------------------------------------------
    return [float(x) for x in best]
