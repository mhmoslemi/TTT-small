    decay = 0.9995
    # mass transfer epsilon (max fraction moved)
    eps_max = 0.2
    # iteration loop
    while time.monotonic() < deadline:

            j += 1

        delta = a[i] * frac

        a_new = a.copy()

        a_new[j] += delta
        # project onto simplex? Since sum unchanged and nonneg constraints satisfied (we ensure not negative)

        if a_new[i] < 0:

        # Compute new score

        new_score = 2.0 * n * max_val

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex. """
    if v.size == 0:
        return v
    # shift to make sum(v) = 1 before projection
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Analytic sqrt series (fallback initialisation)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """A simple decreasing series that roughly satisfies the constraints."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    return a


# ------------------------------------------------------------
# Full convolution via FFT (returns max and full array)
# ------------------------------------------------------------
def _conv_full_max(a: np.ndarray, L: int):
    """Return max(conv(a,a)) and the full convolution (linear, not circular)."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ---- initialise -------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = _sqrt_series(256)

    if a.size == 0:
        a = np.ones(1, dtype=np.float64)

    # normalise to the probability simplex
    s = a.sum()
    a = _project_simplex(a / (s if s != 0 else 1.0))

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # smallest power‑of‑two ≥ 2n‑1

    # initial score
    max_val, conv = _conv_full_max(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # ---- optimisation parameters ------------------------------------
    step = 0.5                 # base gradient step
    min_step = 1e-7
    step_decay = 0.9996

    temp = 1.0                 # simulated‑annealing temperature
    temp_decay = 0.9998
    min_temp = 1e-4

    beta = 0.9                 # momentum coefficient
    momentum = np.zeros_like(a)

    rel_tol = 1e-5             # tolerance for the active set (relative)

    # ---- main loop --------------------------------------------------
    while time.monotonic() < deadline:
        # convolution & active set
        max_val, conv = _conv_full_max(a, L)
        tol = rel_tol * max_val
        active = np.where(conv >= max_val - tol)[0]
        if active.size == 0:               # fallback to the true maximum index
            active = np.array([conv.argmax()], dtype=int)

        # gradient of the (soft) max
        g = np.zeros_like(a)
        for idx in active:
