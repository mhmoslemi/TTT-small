def _gradient(a, tau):
    n = a.size
    conv = _conv_full(a)      # length 2n-1
    max_c = conv.max()
    # softmax
    w = np.exp((conv - max_c) / tau)
    w_sum = w.sum()
    if w_sum == 0:
       w = np.full_like(w, 1.0 / w.size)
    else:
       w /= w_sum
    a_rev = a[::-1]
    # convolution of w and a_rev
    Lg = 1 << ((len(w)+len(a_rev)-1).bit_length())

    conv2 = np.fft.irfft(fw*fa, n=Lg)[:len(w)+len(a_rev)-1]
    # extract g segment
    start = n-1
    g = 2 * conv2[start:start+n]
    return g, max_c

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Small analytic fallback (sqrt series)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Full linear convolution via FFT (returns full array)
# ------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Convolution a*a using a power‑of‑two FFT length L."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return conv

    tau_decay = 0.9995

        if time.monotonic() + 0.3 > deadline:
            break

    # final safeguard: return best found (as plain Python list)
    return [float(x) for x in best]
