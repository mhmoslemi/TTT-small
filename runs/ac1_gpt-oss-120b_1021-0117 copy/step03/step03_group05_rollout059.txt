def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    # get initial vector
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a = rng.random(n)
    # enforce nonnegative & project to simplex (sum=1)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a.sum()
    n = a.size
    # FFT length
    L = 1 << ((2 * n - 1).bit_length())
    # helper functions
    def conv_self(v):
        fv = np.fft.rfft(v, n=L)
        return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    def score(v):
        conv = conv_self(v)
        return float(2.0 * n * conv.max() / (v.sum() ** 2))
    def project_simplex(v):
        # Euclidean projection onto {x>=0, sum x = 1}
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
