def _conv_max_and_grad(a, L, top_k=3):
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(a) - 1]
    max_val = float(conv.max())
    # Get indices of top k values
    if top_k >= len(conv):
        idxs = np.arange(len(conv))
    else:
        # Use argpartition for efficiency
        idxs = np.argpartition(conv, -top_k)[-top_k:]
    # Compute gradient as sum of contributions from each idx.
    n = a.size
    g = np.zeros_like(a)
    for p in idxs:
        lo = max(0, p - (n - 1))
        hi = min(n - 1, p)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            # a[p - j] indices reversed
            g[j] += 2.0 * a[p - j]
    return max_val, g

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # init a0
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # generate random vector length maybe 2500 and pad? We'll pick length 2535
        n = 2535
        a0 = rng.random(n, dtype=np.float64)
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    best_vec = a0.copy()
    # compute initial score
    L = 1 << ((2 * a0.size - 1).bit_length())
    cur_max, grad = _conv_max_and_grad(best_vec, L, top_k=3)
    best_score = 2.0 * a0.size * cur_max

    # define sub functions for conv_max and score
    def score_of(vec):
        max_val = _conv_max_and_grad(vec, L, top_k=1)[0]  # compute conv max, we can ignore gradient
        return 2.0 * len(vec) * max_val

    # optimizer parameters
    step = 0.3
    min_step = 1e-7
    decay = 0.9995
    beta = 0.9
    momentum = np.zeros_like(a0)

    prob_random = 0.03
    prob_smooth = 0.01
    top_k = 3

    while time.monotonic() < deadline:
        # compute conv and gradient at current a0
        max_val, g = _conv_max_and_grad(a0, L, top_k=top_k)
        cur_score = 2.0 * a0.size * max_val

        # momentum
        momentum = beta * momentum + (1.0 - beta) * g

        cand = a0 - step * momentum
        cand = _project_simplex(cand)

        cand_max, _ = _conv_max_and_grad(cand, L, top_k=1)
        cand_score = 2.0 * a0.size * cand_max

        if cand_score < cur_score:
            a0 = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
        else:

import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # make a copy to avoid modifying the input
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # safeguard against pathological all‑zero case
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w


# ------------------------------------------------------------
# Convolution max and (optional) gradient from top‑k peaks
# ------------------------------------------------------------
def _conv_max_and_grad(a: np.ndarray, L: int, top_k: int = 1):
    """Return (max convolution value, gradient from the top_k peaks)."""
    n = a.size
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    max_val = float(conv.max())

    # pick the indices of the largest k entries (unsorted)
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(conv, -top_k)[-top_k:]

    # gradient = sum_{p in idxs} 2 * a shifted by p
    g = np.zeros_like(a)
    for p in idxs:
        lo = max(0, p - (n - 1))
        hi = min(n - 1, p)
        if lo <= hi:
            j = np.arange(lo, hi + 1, dtype=int)
            g[j] += 2.0 * a[p - j]
    return max_val, g
