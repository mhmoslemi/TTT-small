import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast AC1 upper‑bound (uses FFT, sum is 1 after projection)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
# Gradient from the K largest autocorrelation peaks
# ----------------------------------------------------------------------
def _gradient_from_top(a: np.ndarray, conv: np.ndarray, topK: int, mu: float) -> np.ndarray:
    n = a.size
    if topK < conv.size:
        idxs = np.argpartition(-conv, topK)[:topK]
    else:
        idxs = np.arange(conv.size)
    vals = conv[idxs]
    wsum = vals.sum()
    if wsum > 0:
        weights = vals / wsum
    else:
        weights = np.full_like(vals, 1.0 / vals.size)

    g = np.zeros_like(a)
    for idx, w in zip(idxs, weights):
        offset = idx - (n - 1)
        if offset >= 0:
            i = np.arange(offset, n, dtype=np.int64)
            j = idx - i
        else:
            i = np.arange(0, n + offset, dtype=np.int64)
            j = idx - i
        # main contribution (each pair counted twice)
        g[i] += w * 2.0 * a[j]
        # correct the diagonal where i == j (only one contribution)
        if idx % 2 == 0:
            diag = idx // 2
            if 0 <= diag < n:
                g[diag] -= w * a[diag]   # we added 2*a[diag], need only a[diag]
    return g


# ----------------------------------------------------------------------
# Local optimisation by projected momentum steps
# ----------------------------------------------------------------------
def _local_optimize(start, max_iters, fft_len, conv_len,
                    eta0=1.0, mu=0.9, topK=5,
                    jitter_every=200, jitter_sigma=0.02, rng=None):
    n = start.size
    a = start.copy()
    v = np.zeros_like(a)
    best = a.copy()
    best_score = _fast_score(a, fft_len, conv_len)

    for it in range(1, max_iters + 1):
        # FFT convolution and score
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_score = float(2.0 * n * conv.max())

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            # early stop if we are already extremely good
            if best_score <= 1.50274349:
                break

        # gradient from top peaks
        g = _gradient_from_top(a, conv, topK, mu)

        # momentum update
        v = mu * v + (1.0 - mu) * g

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # projected gradient step
        a = _proj_simplex(a - eta * v)

        # occasional jitter to escape plateaus
        if jitter_every and (it % jitter_every == 0):
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = _proj_simplex(a * jitter)

    return best, best_score
