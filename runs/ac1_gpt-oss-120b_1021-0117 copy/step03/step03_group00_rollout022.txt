import time, math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Simple score (2*n*max(conv)) assuming sum(a)=1
# ------------------------------------------------------------
def _score_vec(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---------- initialise ----------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a.fill(1.0 / a.size)
    else:
        a /= s                     # normalize to sum‑1
    n = a.size
    L = 1 << ((3 * n - 2).bit_length())   # FFT length for both convolutions

    best_vec = a.copy()

    # ---------- Adam parameters ----------

    eps0 = 0.5

    min_eps = 1e-6

    # ---------- stagnation handling ----------

    it = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and current score -----

        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            last_improve = it
            if best_sc < 1.50274349:      # target reached
