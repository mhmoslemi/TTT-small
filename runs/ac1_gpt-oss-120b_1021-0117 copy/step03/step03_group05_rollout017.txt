```
def project_simplex(v):
    # v: 1D numpy array
    # returns projection onto {x>=0, sum(x)=1}
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0
    return w
```

import time
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The routine respects the allotted time budget, using a sub‑gradient descent
    on the maximal self‑convolution entry together with occasional stochastic
    moves and (if available) a tiny LP refinement.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the final conversion
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------------
    # initialise candidate (use provided parent if present)
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate a simple random start
        n0 = 256
        a = rng.random(n0, dtype=np.float64)

    # enforce non‑negativity and project to the simplex (sum = 1)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / total

    n = a.size
    # ----------------------------------------------------------------------
    # helper utilities (FFT convolution, score, simplex projection)
    # ----------------------------------------------------------------------
    _L = 1 << ((2 * n - 1).bit_length())  # power‑of‑two length for FFT

    def _conv_fft(v: np.ndarray) -> np.ndarray:
        fv = np.fft.rfft(v, n=_L)
        cv = np.fft.irfft(fv * fv, n=_L)
        return cv[: 2 * n - 1]

    def _score(v: np.ndarray) -> float:
        """Objective 2·n·max(conv) (sum(v) == 1 after projection)."""
        conv = _conv_fft(v)
        return 2.0 * n * conv.max()

    def _project_simplex(v: np.ndarray) -> np.ndarray:
        """Euclidean projection onto {x≥0, sum(x)=1}."""
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        # renormalise to avoid tiny sum‑drift
        s = w.sum()
        if s == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= s
        return w

    # ----------------------------------------------------------------------
    # optional tiny LP refinement (few constraints, many variables)
    # ----------------------------------------------------------------------
    try:
        import cvxpy as cp
        _lp_available = True
    except Exception:
        _lp_available = False

    def _solve_lp(cur_vec: np.ndarray, active_idx: np.ndarray, M: float):
        """Solve  max ∑b  s.t. b≥0 , (cur * b)_k ≤ M for k in active_idx """
        K = active_idx.size
        # build constraint matrix A (K × n)
        A = np.zeros((K, n), dtype=np.float64)
        for r, k in enumerate(active_idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # indices j in [lo,hi] have coefficient f[k‑j]
            j = np.arange(lo, hi + 1, dtype=np.int64)
            A[r, j] = cur_vec[k - j]
        b = cp.Variable(n, nonneg=True)
        constraints = [A @ b <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        try:

        except Exception:
            return None
        if b.value is None:
            return None
        return np.asarray(b.value, dtype=np.float64)
