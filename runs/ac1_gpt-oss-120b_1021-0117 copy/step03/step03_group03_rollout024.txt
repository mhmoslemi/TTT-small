def _proj_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0/len(v))
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    if v.ndim != 1:
        v = v.ravel()
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # fallback to uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 bound.
    Returns the best non‑negative step‑function heights found before the budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # time budget (leave a small safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # initialise a sequence (fallback to uniform if parent missing)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = parent.copy()
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = _proj_simplex(a)                     # enforce Σa = 1, a_i ≥ 0

    # ------------------------------------------------------------------
    # keep the best candidate seen so far
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def _conv_max(x):
        fx = np.fft.rfft(x, n=fft_len)
        c = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
        return c.max(), c

    # evaluate the initial vector and uniform vector
    cur_max, _ = _conv_max(a)
    best = a.copy()
    best_score = float(2.0 * n * cur_max)

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_max, _ = _conv_max(uniform)
    uni_score = float(2.0 * n * uni_max)

    if uni_score < best_score:
        best = uniform.copy()
        best_score = uni_score
        a = uniform.copy()

    # ------------------------------------------------------------------
    # optimisation hyper‑parameters
    # ------------------------------------------------------------------
    lr0 = 1.0                 # initial step size (will decay as 1/√t)
    beta = 0.9                # momentum decay
    momentum = np.zeros_like(a)

    it = 0
    stale = 0
    max_stale = 3000          # restart after this many non‑improving iterations
    jitter_interval = 150
    jitter_std = 0.02
    target_score = 1.50274349
    rel_tol = 1e-12           # tolerance for active‑set detection

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution & current score -----
        cur_max, conv = _conv_max(a)
        cur_score = 2.0 * n * cur_max

        # ----- keep best -----
        if cur_score + 1e-12 < best_score:
            best = a.copy()
            best_score = cur_score
            stale = 0
            if best_score <= target_score:
                break
        else:
            stale += 1

        # ----- active set (indices attaining the maximum, within tolerance) -----
        act = np.where(np.isclose(conv, cur_max, rtol=0, atol=rel_tol * cur_max))[0]
        if act.size == 0:                        # safety fallback
            act = np.array([int(np.argmax(conv))], dtype=np.int64)

        # ----- sub‑gradient of the max‑convolution -----
        g = np.zeros_like(a)
        for k in act:
            # indices i that participate in a[i] * a[k-i]
            i0 = max(0, k - (n - 1))
            i1 = min(n - 1, k)
            idx = np.arange(i0, i1 + 1, dtype=np.int64)
            partner = k - idx
            g[idx] += a[partner]                 # a_i * a_{k-i} → derivative w.r.t a_i
            # diagonal terms (i == partner) contribute twice
            mask_diag = (idx == partner)
            if mask_diag.any():
                g[idx[mask_diag]] += a[idx[mask_diag]]

        g /= float(act.size)                     # average over active constraints

        # ----- momentum update & projected step -----
        it += 1
        eta = lr0 / np.sqrt(it)
        momentum = beta * momentum + (1.0 - beta) * g
        a_candidate = a - eta * momentum
        a = _proj_simplex(a_candidate)           # stay on the simplex

        if it % jitter_interval == 0:

            momentum.fill(0.0)

        if stale >= max_stale:

            perturb = np.exp(rng.normal(0.0, 0.05, size=n))
            a = _proj_simplex(best * perturb)
            momentum.fill(0.0)
            stale = 0

    # ------------------------------------------------------------------
    # return the best sequence as plain python floats
    # ------------------------------------------------------------------
    return [float(v) for v in best]
