import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # guard against pathological all‑zero case
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w


# ------------------------------------------------------------
# Full linear convolution via FFT (returns the whole array)
# ------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Gradient built from the K largest convolution entries
# ------------------------------------------------------------
def _active_gradient(a: np.ndarray, conv: np.ndarray, K: int = 7) -> np.ndarray:
    """Return ∇ max_k conv_k ≈ sum of gradients of the K largest lags."""
    n = a.size
    if conv.size <= K:
        act = np.arange(conv.size)
    else:
        # K largest (unsorted) – cheap
        act = np.argpartition(conv, -K)[-K:]
    g = np.zeros_like(a)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            j = np.arange(lo, hi + 1, dtype=int)
            g[j] += 2.0 * a[k - j]
    return g

    # initial state

    cur_score = 2.0 * n * conv.max()
    best = a.copy()

    # hyper‑parameters
    step = 0.20            # initial learning rate
    min_step = 1e-8
    decay = 0.9995
    beta = 0.9

    while time.monotonic() < deadline:

        grad = _active_gradient(a, conv, K=7)
