import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    # implementation based on sorting, O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Simple stochastic refinement of the provided parent sequence.
    Returns a non‑negative step‑function (list of floats) whose AC1
    upper‑bound score is expected to be competitive.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Time management – keep a small safety margin.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 2. Obtain a starting sequence.
    # ------------------------------------------------------------------
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)                # respect evaluator clamps
        n = a.size
        if n == 0:
            raise ValueError
    except Exception:
        # Fallback: uniform of modest length
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Normalise to a probability simplex (sum = 1) – scaling does not affect
    # the evaluator's ratio but ensures a well‑behaved search space.
    s = a.sum()
    if s < 0.01:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    # ------------------------------------------------------------------
    # 3. Convolution utilities (FFT based, O(n log n)).
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def score(x: np.ndarray) -> float:
        """   2·n·max(a*a)  for a vector with sum(x)=1   """
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # 4. Keep best solution seen so far.
    # ------------------------------------------------------------------
    best = a.copy()
    best_val = score(best)

    # ------------------------------------------------------------------
    # 5. Stochastic refinement loop.
    #    – multiplicative jitter,
    #    – simplex projection,
    #    – accept if improvement.
    # ------------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline and it < 5000:
        # jitter magnitude decays slowly
        sigma = 0.02 / np.sqrt(it + 1)
        jitter = np.exp(rng.normal(0.0, sigma, size=n))
        cand = best * jitter
        cand = _proj_simplex(cand)

        val = score(cand)
        if val < best_val:
            best_val = val
            best = cand
            # occasional aggressive exploration reset
            if it % 500 == 0:
                # re‑jitter a bit more aggressively
                jitter = np.exp(rng.normal(0.0, 0.1, size=n))
                best = _proj_simplex(best * jitter)

        it += 1

    # ------------------------------------------------------------------
    # 6. Return plain Python list of floats.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
