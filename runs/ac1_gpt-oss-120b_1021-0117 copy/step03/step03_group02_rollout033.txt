import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Construct a low‑AC1‑bound sequence by extending a good base sequence
    with a block of tiny equal entries.  The base sequence is taken from the
    pre‑imported `height_sequence_1` if present; otherwise a simple uniform
    fallback is used.
    """
    # ------------------------------------------------------------------
    # time budget
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # base sequence -----------------------------------------------------
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: a short uniform distribution (sum≈1)
        a = np.full(5, 1.0 / 5, dtype=np.float64)

    # enforce evaluator’s clipping rules (non‑negative, ≤1000)
    a = np.clip(a, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # compute current max‑autocorrelation (full convolution)
    # ------------------------------------------------------------------
    conv0 = np.convolve(a, a)               # linear convolution, O(n²) but tiny n
    M0 = float(conv0.max())                 # peak of a*a
    S0 = float(a.sum())                     # total mass of base

    # safety guard – if something went wrong just return the base
    if M0 <= 0.0 or S0 < 0.01:
        return [float(x) for x in a.tolist()]

    # ------------------------------------------------------------------
    # augment with a block of tiny entries
    # ------------------------------------------------------------------
    # Choose K so that the cross‑term (eps * S0) does not exceed M0.
    # Condition: eps = sqrt(M0 / K)  and  eps * S0 <= M0
    #   ⇒  K >= (S0 / sqrt(M0))**2
    K_min = int(np.ceil((S0 / np.sqrt(M0)) ** 2))
    # a modest safety margin; also keep K modest to respect memory/CPU limits
    K = max(K_min, 4000)                    # at least a few thousand entries

    eps = float(np.sqrt(M0 / K))            # height of each added entry
    eps_block = np.full(K, eps, dtype=np.float64)

    # concatenate – order does not matter for the bound
    cand = np.concatenate([a, eps_block])

    # ------------------------------------------------------------------
    # final sanity check (optional, cheap)
    # ------------------------------------------------------------------
    # ensure we stay within evaluator limits
    cand = np.clip(cand, 0.0, 1000.0)
    if time.monotonic() >= deadline:
        # time is up – fall back to whatever we have already
        return [float(x) for x in cand.tolist()]

    # ------------------------------------------------------------------
    # return Python list as required
    # ------------------------------------------------------------------
    return [float(x) for x in cand.tolist()]
