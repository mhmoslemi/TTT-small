    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # simple initial shape: linear ramp then down symmetry
        n_guess = 2000 # but we need same length as? Not required; but we can pick length typical?
        # In case no height_sequence_1, we might define length like 1024? but no guarantee length predetermined.
        n = 1024 possibly?
        # Actually if no parent, we can just define n = 1000 (some length) - but we need to produce any sequence; but score depends on n. Could we produce trivial shape for any n? I'd prefer to rely on parent being present; but specification says may be not. We'll handle fallback by
        n = 1000
        a0 = np.linspace(1,0,n)  # decreasing shape

    # Ensure nonnegative
    a0 = np.maximum(a0, 0.0)
    # Normalize to sum=1
    total = a0.sum()
    if total <= 0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 /= total

    n = a0.size
    # Compute FFT length
    L = 1 << ((3 * n - 2).bit_length())

    # Helper functions
    def _project_simplex(v):
        # Use algorithm by Duchi et al.
        n_ = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1
        rho = np.where(u - css / np.arange(1, n_+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    def _score_fast(a, a_fft=None):
        # assuming sum(a) = 1
        if a_fft is None:
            a_fft_use = np.fft.rfft(a, n=L)
        else:
            a_fft_use = a_fft
        conv = np.fft.irfft(a_fft_use * a_fft_use, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())

    best_vec = a0.copy()
    best_score = _score_fast(best_vec)
    cur = best_vec.copy()
    cur_score = best_score

    # Adam params
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.2  # learning rate
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # softmax eps schedule
    eps0 = 1.0
    eps_decay = 0.995  # faster decay
    eps_min = 1e-5

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # compute convolution and softmax weights
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        eps = max(eps_min, eps0 * (eps_decay ** it))

        # softmax weights
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand_score = _score_fast(cand)

        # Accept if improvement else maybe reject

            if cand_score < best_score:
                best_vec, best_score = cand.copy(), cand_score
                no_improve = 0
        else:

            no_improve += 1

        # periodic symmetrize
        if it % 1000 == 0:

            sym = _project_simplex(sym)
            sym_score = _score_fast(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best_vec, best_score = sym.copy(), sym_score
                    no_improve = 0

        # random perturbation if stagnant
        if no_improve >= 2000:
            # random mass transfer
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score_fast(tmp)
                if tmp_score < cur_score:

                if tmp_score < best_score:
                    best_vec, best_score = tmp.copy(), tmp_score
                    no_improve = 0
                else:
                    no_improve = 0

        # early exit

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex using the algorithm of Duchi et al."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound for a vector that is already on the simplex
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2*n*max(conv) assuming sum(a)=1."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:

    # enforce symmetry right away (optimal solutions are symmetric)
    a0 = (a0 + a0[::-1]) * 0.5
    a0 = _project_simplex(a0)

    n = a0.size
    # FFT length big enough for both 2n‑1 and 3n‑2 linear convolutions
