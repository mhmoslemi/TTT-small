def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.5)

    # initial vector
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 1024
        a = rng.random(n0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a[:] = 1.0/a.size
    else:
        a /= total

    n = a.size
    L = 1 << ((2*n - 1).bit_length())  # FFT size

    def conv_fft(x):
        fx = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(fx*fx, n=L)[:2*n-1]
        return conv

    def score(x):
        conv = conv_fft(x)
        return 2.0*n*conv.max()/(x.sum()**2)

    # initial score
    cur_score = score(a)
    best_a = a.copy()
    best_score = cur_score

    # parameters
    K = 5 # active indices
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # compute conv and active indices
        conv = conv_fft(a)
        # compute top K indices (largest conv)
        if K >= conv.size:
            active = np.arange(conv.size)
        else:
            # argpartition largest
            idx_part = np.argpartition(-conv, K-1)[:K]
            # sort descending for maybe tie
            active = idx_part

        # compute gradient g
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi+1, dtype=np.int64)
            g[idx] += 2.0 * a[k - idx]

        # get indices for mass transfer
        i_hi = int(np.argmax(g))
        i_lo = int(np.argmin(g))

        # ensure i_hi has positive mass
        if a[i_hi] > 1e-12:
            # step factor diminishing
            step = min(a[i_hi], max(1e-6, a[i_hi] * (0.5 / (1 + it*1e-4))))
            # optionally random factor
            delta = step * rng.random() * 0.5  # up to half of step
            # make sure not zero
            if delta > 0:
                cand = a.copy()
                cand[i_hi] -= delta
                cand[i_lo] += delta
                # no need to renorm sum, unchanged
                cand_score = score(cand)
                if cand_score < cur_score or rng.random() < np.exp(-(cand_score-cur_score)/max(1e-3,cur_score*1e-4)):
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_a = cand.copy()
                        best_score = cand_score
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            # fallback random move
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.3
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand_score = score(cand)
                if cand_score < cur_score:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_a = cand.copy()
                        best_score = cand_score
                        no_improve = 0
                    else:
                        no_improve += 1

        # random restart or jitter every something
        if it % 5000 == 0:
            # jitter around best
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best_a + jitter
            tmp = np.maximum(tmp, 0.0)
            # re-normalize sum to 1
            s = tmp.sum()
            if s == 0:
                tmp = np.full(n, 1.0/n)
            else:
                tmp /= s
            tmp_score = score(tmp)
            if tmp_score < cur_score:
                a = tmp
                cur_score = tmp_score
                if tmp_score < best_score:
                    best_a = tmp.copy()
                    best_score = tmp_score
                    no_improve = 0

        # early stop if below target bound
        if best_score < 1.50274349:
            break

    # final small local refinement (a few random transfers) until deadline
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        if a[i] == 0:
            continue
        delta = a[i] * rng.random() * 0.2
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = score(cand)
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_a = cand.copy()
                best_score = cand_score

    return [float(x) for x in best_a]

def conv_fft2(x, y):
    # compute convolution x*y using FFT with size L2
    L2 = 1 << ((len(x) + len(y) - 1).bit_length())
    fx = np.fft.rfft(x, n=L2)
    fy = np.fft.rfft(y, n=L2)

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The routine respects the time budget and keeps the best solution found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # leave margin

    # ------------------------------------------------------------------ #
    # initialise a (use the provided parent if it exists)
    # ------------------------------------------------------------------ #
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a simple random start
        n0 = 2048
        a = rng.random(n0, dtype=np.float64)

    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total <= 0.0:
        a.fill(1.0 / a.size)
    else:
        a /= total                     # enforce sum = 1

    n = a.size

    # ------------------------------------------------------------------ #
    # helper utilities (FFT convolution, score, simplex projection)
    # ------------------------------------------------------------------ #
    # FFT length large enough for any convolution we need (2n-1 + n-1 = 3n-2)
    L = 1 << ((3 * n - 2).bit_length())

    def _conv_fft(v: np.ndarray) -> np.ndarray:
        """Full linear convolution v*v (real). Returns length 2n-1 array."""
        fv = np.fft.rfft(v, n=L)
        c = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        return c

    def _score_from_max(m: float) -> float:
        """Score given max entry of convolution (sum(a)=1)."""
        return 2.0 * n * m

    def _project_simplex(v: np.ndarray) -> np.ndarray:
        """Euclidean projection onto {x>=0, sum x = 1}."""
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rng_idx = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rng_idx.size == 0:
            theta = 0.0
        else:
            rho = rng_idx[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    # bookkeeping
    # ------------------------------------------------------------------ #
    cur = a.copy()
    conv_cur = _conv_fft(cur)
    cur_max = conv_cur.max()
    cur_sc = _score_from_max(cur_max)
    best_vec = cur.copy()
    best_sc = cur_sc

    # algorithmic parameters
    K = 20                        # number of active convolution entries
    add_alpha0 = 0.7              # additive step scaling
    mul_eta0 = 0.7                # multiplicative step scaling

    no_improve = 0

        it += 1

        # --------------------------------------------------------------
        # convolution and active set
        # --------------------------------------------------------------
        conv_cur = _conv_fft(cur)
        cur_max = conv_cur.max()
        cur_sc = _score_from_max(cur_max)

        if K >= conv_cur.size:
            active = np.arange(conv_cur.size)
        else:

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        g = np.zeros_like(cur)
        maxc = cur_max if cur_max != 0 else 1.0
        for k in active:

            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * w * cur[k - idx]
