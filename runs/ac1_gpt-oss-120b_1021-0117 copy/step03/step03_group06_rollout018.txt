import time, math, numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a * a using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv

def _softmax_weights(conv: np.ndarray, T: float) -> np.ndarray:
    """Soft‑max over convolution values with temperature T."""
    # shift by max to avoid overflow
    m = conv.max()
    ex = np.exp((conv - m) / max(T, 1e-12))
    s = ex.sum()
    if s == 0 or not np.isfinite(s):
        return np.full_like(conv, 1.0 / conv.size)
    return ex / s

def _gradient_wrt_a(a: np.ndarray, w: np.ndarray, M: int) -> np.ndarray:
    """
    Compute g_i = 2 * sum_j w_{i+j} * a_j  (valid correlation).
    Uses FFT of length M >= len(w) + len(a) - 1.
    """
    # FFTs (real)
    Fa = np.fft.rfft(a, n=M)
    Fw = np.fft.rfft(w, n=M)
    # correlation = ifft(Fw * conj(Fa))
    corr = np.fft.irfft(Fw * np.conj(Fa), n=M)
    n = a.size
    # valid part: indices n-1 .. 2n-2 (inclusive)
    g = corr[n - 1: 2 * n - 1]
    return 2.0 * g  # factor 2 from derivative of a*a

                    if tmp_sc < cur_sc:
                        a, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best = tmp_sc, tmp.copy()

        # ---- restart after prolonged stagnation ---------------------------
        if no_improve > 8000:
            a = best + rng.normal(scale=jitter_std * 5, size=n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = np.full_like(a, 1.0 / n)
            no_improve = 0

    # return plain Python list
    return [float(v) for v in best]
