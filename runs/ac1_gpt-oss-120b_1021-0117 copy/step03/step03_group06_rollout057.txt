    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # init a
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()

        a0 = np.full_like(a0, 1.0/a0.size)
    a0 = a0 / a0.sum()
    n = a0.size

    L = next_pow2(3*n)  # safe for both conv and grad conv
    # Precompute some constants
    beta_max = 30.0
    total_iters = 200000  # will be limited by deadline
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5

    # initialize logits
    z = np.log(a0 + 1e-12)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_a = a0.copy()
    # compute initial score
    fx = np.fft.rfft(best_a, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[:2*n-1]
    best_score = float(2.0 * n * conv.max())
    # You may also compute using evaluate_sequence for safety but fine

    it = 0
    while time.monotonic() < deadline and it < total_iters:
        it += 1
        # compute a via softmax
        z_shift = z - np.max(z)  # for numeric stability
        a = np.exp(z_shift)
        a_sum = a.sum()
        a = a / a_sum

        # compute convolution via FFT
        fx = np.fft.rfft(a, n=L)

import time, math, numpy as np

def _next_pow2(x: int) -> int:
    """Small helper: smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC‑1 bound.  The routine respects the time budget and
    keeps the best solution found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Initial distribution (use provided parent if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = np.full(2000, 1.0 / 2000, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                     # normalise to a probability vector

    n = a0.size
    # FFT length that safely accommodates all convolutions (full a*a and p*a_rev)
    L = _next_pow2(3 * n)

    # ------------------------------------------------------------------
    # Adam optimiser on logits (soft‑max parametrisation)
    # ------------------------------------------------------------------
    # Initialise logits from the current distribution
    z = np.log(a0 + 1e-12)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.5
    beta_max = 30.0          # final smooth‑max sharpening factor
    max_iters = 50000        # will be cut short by the deadline

    # keep the best solution seen so far
    best_a = a0.copy()
    # score = 2 * n * max(conv) because sum(a)=1
    conv0 = np.fft.irfft(np.fft.rfft(best_a, n=L) * np.fft.rfft(best_a, n=L), n=L)[: 2 * n - 1]
    best_score = float(2.0 * n * conv0.max())

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # ----- soft‑max to obtain a probability vector -----
        z_shift = z - np.max(z)               # numeric stability
        a = np.exp(z_shift)
        a = a / a.sum()

        # ----- full self‑convolution -----
        fx = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fx * fx, n=L)[: 2 * n - 1]

        # ----- smoothed max (log‑sum‑exp) -----
        beta = 1.0 + (beta_max - 1.0) * (it / max_iters)   # linear schedule
        mx = conv.max()
        exp_shift = np.exp(beta * (conv - mx))             # all <= 1
        sum_exp = exp_shift.sum()
        p = exp_shift / sum_exp                             # softmax over indices

        # ----- gradient of the smoothed max w.r.t. a -----
        a_rev = a[::-1]
        fp = np.fft.rfft(p, n=L)
        fr = np.fft.rfft(a_rev, n=L)
        conv_grad = np.fft.irfft(fp * fr, n=L)
        grad_a = 2.0 * conv_grad[n - 1: n - 1 + n]          # size n

        # ----- chain rule to get gradient w.r.t. logits -----
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        lr_t = lr0 * math.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it)
        z -= lr_t * m / (np.sqrt(v) + eps)

        # ----- occasional noise injection to escape plateaus -----
        if it % 5000 == 0:
            z += rng.normal(scale=0.1, size=n)

        # ----- keep track of the best scoring vector -----
        if it % 100 == 0:
            cur_score = float(2.0 * n * conv.max())
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                if best_score < 1.50274349:
                    break

    # ------------------------------------------------------------------
    # Fine‑tuning stage: plain exponentiated‑gradient on the *true* max
    # ------------------------------------------------------------------
    cur = best_a.copy()
    cur_score = float(2.0 * n *
                      np.fft.irfft(np.fft.rfft(cur, n=L) *
                                  np.fft.rfft(cur, n=L), n=L)[: 2 * n - 1].max())

    for sub_it in range(2000):
        if time.monotonic() > deadline:
            break

        # compute convolution and index of the current maximum
        fx = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fx * fx, n=L)[: 2 * n - 1]
        k = int(conv.argmax())

        # sub‑gradient of max‑convolution
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]

        # exponentiated mirror step
        eta = 0.4 / math.sqrt(sub_it + 1.0)
        mul = np.exp(-eta * g)
        cand = cur * mul
        cand = cand / cand.sum()

        cand_score = float(2.0 * n *
                           np.fft.irfft(np.fft.rfft(cand, n=L) *
                                       np.fft.rfft(cand, n=L), n=L)[: 2 * n - 1].max())
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best_score, best_a = cand_score, cand.copy()
                if best_score < 1.50274349:
                    break

        # simple random mass‑transfer move (occasionally)
        if sub_it % 200 == 0:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * 0.2 * rng.random()
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                tmp /= tmp.sum()
                tmp_score = float(2.0 * n *
                                 np.fft.irfft(np.fft.rfft(tmp, n=L) *
                                             np.fft.rfft(tmp, n=L), n=L)[: 2 * n - 1].max())
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    if tmp_score < best_score:
                        best_score, best_a = tmp_score, tmp.copy()
                        if best_score < 1.50274349:
                            break

    # ------------------------------------------------------------------
    # Return the best vector as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best_a]
