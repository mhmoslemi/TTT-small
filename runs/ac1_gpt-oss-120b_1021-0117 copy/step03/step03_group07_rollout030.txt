import time, numpy as np

def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv*fv, n=L)[:2*n-1]

def _score(a):
    n = a.shape[0]
    return float(2.0 * n * _conv_fft(a).max() / (a.sum() ** 2))

def _project_simplex(v):
    # Euclidean projection onto simplex sum=1, v>=0
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _make_exp_decay(n, lam, p):
    x = np.arange(n, dtype=np.float64) / (n-1)
    a = np.exp(-lam * np.power(x, p))
    s = a.sum()
    if s == 0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

def _make_beta_shape(n, alpha, beta):
    i = np.arange(1, n+1, dtype=np.float64)
    a = np.power(i, alpha) * np.power(n - i + 1, beta)
    s = a.sum()
    if s == 0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

            break

    for alpha in np.linspace(0.1, 5.0, 12):
        for beta in np.linspace(0.1, 5.0, 12):
            if time.monotonic() >= deadline:
                break
            cand = _make_beta_shape(n, alpha, beta)
            sc = _score(cand)
            if sc < best_sc:
                best_sc, best_vec = sc, cand.copy()
        if time.monotonic() >= deadline:
            break

    # -----------------------------------------------------------------
    # initialise the main iterative refinement
    # -----------------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    cur_conv = _conv_fft(cur)

    it = 0
    eta0 = 0.6                     # initial learning‑rate factor
    while time.monotonic() < deadline:
        it += 1

        # ---- sub‑gradient from the current active lag -----------------
        k = int(np.argmax(cur_conv))
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]

        # ---- exponential‑gradient update -------------------------------
        eta = eta0 / np.sqrt(it + 1.0)
        cand = cur * np.exp(-eta * g)
        cand_sum = cand.sum()
        cand = cand / cand_sum if cand_sum > 0 else np.full_like(cur, 1.0 / n)

        cand_sc = _score(cand)
        if cand_sc < cur_sc:                     # accept improvement
            cur, cur_sc = cand, cand_sc
            cur_conv = _conv_fft(cur)
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            # occasional uphill move (tiny probability)
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc
                cur_conv = _conv_fft(cur)

        # ---- periodic stochastic jitter & monotone projection ----------
        if it % 2000 == 0:
            # small Gaussian noise
            jitter = rng.normal(scale=1e-4, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp_sum = tmp.sum()
            if tmp_sum > 0:
                tmp /= tmp_sum
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                cur_conv = _conv_fft(cur)
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

            # enforce (soft) monotonicity – sorting descending
            mono = np.sort(cur)[::-1]
            mono_sum = mono.sum()
            if mono_sum > 0:
                mono /= mono_sum
            mono_sc = _score(mono)
            if mono_sc < cur_sc:
                cur, cur_sc = mono, mono_sc
                cur_conv = _conv_fft(cur)
                if mono_sc < best_sc:
                    best_sc, best_vec = mono_sc, mono.copy()

        # ---- targeted random mass‑transfer move -------------------------
        if rng.random() < 0.03:
            prod = cur[idx] * cur[k - idx]
            if prod.size:
                i_local = int(idx[np.argmax(prod)])
            else:
                i_local = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if j == i_local:
                j = (j + 1) % n

            delta = cur[i_local] * rng.random() * 0.4
            cand2 = cur.copy()
            cand2[i_local] -= delta
            cand2[j] += delta
            cand2 = np.maximum(cand2, 0.0)
            s2 = cand2.sum()
            cand2 = cand2 / s2 if s2 > 0 else np.full_like(cand2, 1.0 / n)

            cand2_sc = _score(cand2)
            if cand2_sc < cur_sc:
                cur, cur_sc = cand2, cand2_sc
                cur_conv = _conv_fft(cur)
                if cand2_sc < best_sc:
                    best_sc, best_vec = cand2_sc, cand2.copy()

        # ---- deadline check --------------------------------------------
        if time.monotonic() >= deadline:
            break

    # -----------------------------------------------------------------
    # guarantee a valid fallback
    # -----------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
