def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[:2*n - 1]

def _make_exponential(n, r):
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s

def _make_powerlaw(n, p):
    a = 1.0 / np.power(np.arange(1, n+1, dtype=np.float64), p)
    s = a.sum()
    return a / s

def _make_linear(n):
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()

def _make_quadratic(n):
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    #  Initial vector (use parent if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    else:
        n = 256
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur.size
    best_vec = cur.copy()
    best_score = _score_vec(best_vec)

    # ------------------------------------------------------------------
    #  Scan a few analytical families for a better start
    # ------------------------------------------------------------------
    for r in np.linspace(0.90, 0.9999, 30):
        cand = _make_exponential(n, r)
        sc = _score_vec(cand)
        if sc < best_score:
            best_score, best_vec = sc, cand.copy()
    for p in np.linspace(0.2, 4.0, 30):
        cand = _make_powerlaw(n, p)
        sc = _score_vec(cand)
        if sc < best_score:
            best_score, best_vec = sc, cand.copy()
    for maker in (_make_linear, _make_quadratic, _make_sqrt_series):
        cand = maker(n)
        sc = _score_vec(cand)
        if sc < best_score:
            best_score, best_vec = sc, cand.copy()

    cur = best_vec.copy()
    cur_score = best_score

    target = 1.50274349

    # ------------------------------------------------------------------
    #  Smoothed sub‑gradient descent (mirror descent on the simplex)
    # ------------------------------------------------------------------
    eta0 = 2.0                 # initial step size
    top_k = 15                 # number of largest convolution entries to average
    it = 0

    while time.monotonic() < deadline:
        it += 1
        conv = _conv_fft(cur)
        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)

        if cur_score < best_score:
            best_score, best_vec = cur_score, cur.copy()
            if best_score < target:
                break

        # ----- gradient from several large lags (uniform averaging) -----
        k = min(top_k, conv.size)
        idxs = np.argpartition(-conv, k - 1)[:k]       # k indices with largest conv
        weight = 1.0 / k                               # uniform weight over the chosen lags
        g = np.zeros_like(cur)

        for idx in idxs:
            lo = max(0, idx - (n - 1))
            hi = min(idx, n - 1)
            seg = cur[lo:hi + 1][::-1]                # reversed segment a[idx‑p]
            g[lo:hi + 1] += 2.0 * weight * seg

        # ----- mirror‑descent step on the simplex -----
        eta = eta0 / math.sqrt(it)
        exp_arg = -eta * g
        exp_arg = np.clip(exp_arg, -50, 50)          # avoid extreme under/overflow
        cand = cur * np.exp(exp_arg)
        s = cand.sum()
        if s > 0:
            cand = cand / s
        else:
            cand = np.full_like(cur, 1.0 / n)

        # occasional tiny jitter to escape plateaus
        if rng.random() < 0.006:
            jitter = rng.normal(scale=8e-6, size=n)
            cand = np.maximum(cand + jitter, 0.0)
            cand = cand / cand.sum()

        cur = cand

    # ------------------------------------------------------------------
    #  Safety fallback
    # ------------------------------------------------------------------
    if best_score == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
