def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Get parent if exists
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_raw.ndim != 1 or parent_raw.size == 0 or not np.isfinite(parent_raw).all():
            raise ValueError
        parent_raw = np.clip(parent_raw, 0.0, 1000.0)
        n = int(parent_raw.size)
        a_parent = parent_raw / parent_raw.sum() if parent_raw.sum() >= 0.01 else np.full(n, 1.0/n)
    except Exception:
        # fallback length
        n = 256
        a_parent = None

    # function for fast score
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    def fast_score(x):
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    # candidates
    candidates = []

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    candidates.append(uniform)

    # Tent shape (peak in middle)
    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1.0, n - idx) - 1.0
    if tent.sum() > 0:
        tent = tent / tent.sum()
    else:
        tent = uniform.copy()
    candidates.append(tent)

    # Gaussian centered
    sigma = n / 6.0
    x = np.arange(n) - (n-1)/2
    gauss = np.exp(-0.5 * (x / sigma) ** 2)
    if gauss.sum() > 0:
        gauss = gauss / gauss.sum()
    else:
        gauss = uniform.copy()
    candidates.append(gauss)

    # Exponential decay shapes, try few betas
    for beta in [0.001, 0.003, 0.005, 0.01, 0.02]:
        expdec = np.exp(-beta * np.arange(n))
        if expdec.sum() > 0:
            expdec = expdec / expdec.sum()
            candidates.append(expdec)

    # Random Dirichlet-like
    rand_vec = rng.random(n)
    if rand_vec.sum() > 0:
        rand_vec /= rand_vec.sum()
    candidates.append(rand_vec)

    if a_parent is not None:
        candidates.append(a_parent)

    # select best initial candidate
    best = None
    best_score = float('inf')
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    a = best.copy()

    eta0 = 0.8        # initial step scaling
    mu = 0.9          # momentum factor
    v = np.zeros_like(a)

    K_mid = 20

        if cur_score < best_score:

            best = a.copy()
            if best_score <= target:

        # choose K based on iteration count
        if it < 2000:

        elif it < 7000:
            K = K_mid
        else:

        top_idx = np.argpartition(-conv, K-1)[:K]
        top_vals = conv[top_idx]
        wsum = top_vals.sum()

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic optimizer for the AC1 bound.
    Uses a fast FFT‑based score, a sub‑gradient from the top convolution
    peaks (with correct reversal), momentum, and a simple back‑track line search.
    Returns the best coefficient list found before the deadline.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------

    try:

        n = int(raw.size)
        if raw.sum() >= 0.01:
            a_parent = raw / raw.sum()
        else:
            a_parent = np.full(n, 1.0 / n, dtype=np.float64)
    except Exception:
        n = 256
        a_parent = None

    # ------------------------------------------------------------------
    # 2. Fast AC1 score (FFT based, exact for linear convolution)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def fast_score(x: np.ndarray) -> float:
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # 3. Candidate pool (algorithmic generation, no huge literals)
    # ------------------------------------------------------------------
    candidates = []

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    candidates.append(uniform)

    # Tent (peak in the centre)
    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1.0, n - idx) - 1.0
    if tent.sum() > 0:
        tent = tent / tent.sum()
    else:
        tent = uniform.copy()
    candidates.append(tent)

    # Symmetric Gaussian
    sigma = n / 6.0
    x = np.arange(n) - (n - 1) / 2.0
    gauss = np.exp(-0.5 * (x / sigma) ** 2)
    if gauss.sum() > 0:
        gauss = gauss / gauss.sum()
    else:
        gauss = uniform.copy()
    candidates.append(gauss)

    # Exponential decays (few betas)
    for beta in (0.001, 0.003, 0.006, 0.01, 0.02):
        expd = np.exp(-beta * np.arange(n))
        if expd.sum() > 0:
            expd = expd / expd.sum()
            candidates.append(expd)

    # Random Dirichlet‑like vector

    if a_parent is not None:

    # ------------------------------------------------------------------
    # 4. Pick the best starter (fast evaluation)
    # ------------------------------------------------------------------
    best = None
    best_score = float("inf")
    for cand in candidates:

    # ------------------------------------------------------------------
    return [float(v) for v in best]
