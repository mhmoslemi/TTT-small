        # ----- additive projected‑gradient step -----
        step = step0 / np.sqrt(it)
        cand = a - step * grad
        cand = _project_simplex(cand)
        cand_score = _evaluate(cand, L)

        # ----- occasional multiplicative (exponential‑weights) step -----
        if it % 3 == 0:
            eta = step0 / np.sqrt(it)
            exp_factor = np.exp(-eta * grad)
            cand_mul = a * exp_factor
            total = cand_mul.sum()
            cand_mul = np.full_like(cand_mul, 1.0 / n) if total == 0.0 else cand_mul / total
            cand_mul_score = _evaluate(cand_mul, L)
        else:
            cand_mul = None
            cand_mul_score = float("inf")

        # ----- choose best among current, additive, multiplicative -----
        if cand_score < cur_score and cand_score <= cand_mul_score:
            a = cand
            cur_score = cand_score
        elif cand_mul_score < cur_score:
            a = cand_mul
            cur_score = cand_mul_score

        # ----- global best update -----
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000 and rng.random() < 0.2:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.4
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _evaluate(tmp, L)
                if tmp_score < cur_score:
                    a = tmp
                    cur_score = tmp_score
                    if tmp_score < best_score:
                        best_score = tmp_score
                        best_a = tmp.copy()
                        no_improve = 0

        # ----- occasional jittered restart toward the best solution -----
        if it % 8000 == 0 and rng.random() < 0.3:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best_a + jitter
            tmp = np.maximum(tmp, 0.0)
            a = _project_simplex(tmp)

        # ----- early exit if target reached -----
        if best_score < 1.50274349:
            break

        if time.monotonic() >= deadline:
            break

    # ---------- final local refinement ----------
    refin_end = time.monotonic() + min(0.4, deadline - time.monotonic())
    while time.monotonic() < refin_end:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or best_a[i] == 0.0:
            continue
        delta = best_a[i] * rng.random() * 0.3
        tmp = best_a.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_score = _evaluate(tmp, L)
        if tmp_score < best_score:
            best_score = tmp_score
            best_a = tmp

    return [float(x) for x in best_a]
