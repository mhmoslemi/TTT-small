    # Then inverse transform

def project_simplex(v):
    """Project onto simplex sum=1, v>=0."""
    # Implementation from previous code (efficient O(n log n)).
    # Steps:
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0
    return w

import time
import math
import numpy as np

def _simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    w = np.maximum(v, 0.0)
    s = w.sum()
    if s > 0.0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return w


def _conv_fft(v: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of v with itself using an FFT of length L."""
    fv = np.fft.rfft(v, n=L)
    conv_full = np.fft.irfft(fv * fv, n=L)[: 2 * v.size - 1]
    return conv_full


def _score(v: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 upper bound for a non‑negative vector v."""
    conv = _conv_fft(v, L)
    max_conv = conv.max()
    s = v.sum()
    if s == 0.0:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)
