import time
import numpy as np

def _sidon_positions(n: int, k: int, rng: np.random.Generator) -> np.ndarray | None:
    """
    Try to find a Sidon set of size k inside {0,…,n‑1}.
    Returns the array of positions if successful, otherwise None.
    """
    for _ in range(2000):                     # limited attempts per k
        pos = rng.choice(n, size=k, replace=False)
        pos.sort()
        seen = set()
        ok = True
        for i_idx, i in enumerate(pos):
            for j in pos[i_idx:]:            # j >= i  (include self‑sums)
                s = i + j
                if s in seen:                # collision → not Sidon
                    ok = False
                    break
                seen.add(s)
            if not ok:
                break
        if ok:
            return pos
    return None


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) beats the 1.5 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # small safety margin

    # Length of the candidate – use the provided parent if present.
    n = len(height_sequence_1) if "height_sequence_1" in globals() else 2000

    # Fallback: uniform distribution (score = 2).
    best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    best_sc = 2.0

    # Theoretical upper bound on Sidon size: ⌊√(2n)⌋.
    max_k = int(np.floor(np.sqrt(2 * n)))

    # Search for the largest possible Sidon set.
    for k in range(max_k, 1, -1):
        if time.monotonic() >= deadline:
            break
        positions = _sidon_positions(n, k, rng)
        if positions is None:
            continue

        # Build the candidate: equal mass on each chosen index.
        a = np.zeros(n, dtype=np.float64)
        weight = 1.0 / k
        a[positions] = weight

        # Exact convolution (linear, full) – cheap because a is very sparse.
        conv = np.convolve(a, a)
        max_b = conv.max()
        sum_a = a.sum()
