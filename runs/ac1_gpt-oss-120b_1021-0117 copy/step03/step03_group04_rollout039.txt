import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function minimizing the AC1 bound.
    Returns the best coefficient list found before the deadline.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1. Base length / parent
    # ------------------------------------------------------------------
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_raw.ndim != 1 or parent_raw.size == 0:
            raise ValueError
        parent_raw = np.clip(parent_raw, 0.0, 1000.0)
        n = int(parent_raw.size)
        a_parent = parent_raw / parent_raw.sum()
    except Exception:
        parent_raw = None
        n = 256
        a_parent = None

    # ------------------------------------------------------------------
    # 2. Simple candidate generators
    # ------------------------------------------------------------------
    def uniform():
        return np.full(n, 1.0 / n, dtype=np.float64)

    def geometric(decay):
        idx = np.arange(n, dtype=np.float64)
        raw = np.exp(-decay * idx)
        return raw / raw.sum()

    def linear_inc():
        idx = np.arange(1, n + 1, dtype=np.float64)
        return idx / idx.sum()

    def linear_dec():
        return linear_inc()[::-1]

    candidates = [uniform(), linear_dec(), linear_inc()]
    # a few geometric decays
    for mult in (1.5, 2.0, 3.0):
        decay = mult / n
        candidates.append(geometric(decay))
    if a_parent is not None:
        candidates.append(a_parent)

    # ------------------------------------------------------------------
    # 3. Fast AC1 score (sum is 1 → denominator drops out)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def fast_score(x: np.ndarray) -> float:
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    # pick best init
    best_vec = None
    best_val = np.inf
    for cand in candidates:
        val = fast_score(cand)
        if val < best_val:
            best_val = val
            best_vec = cand.copy()
    a = best_vec.copy()
    best = a.copy()
    best_score = best_val

    # ------------------------------------------------------------------
    # 4. Sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    mu = 0.9
    eta0 = 0.5
    v = np.zeros_like(a)
    it = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            if best_score <= target:
                break

        # indices attaining the maximum (allow tiny tolerance)
        max_ids = np.where(conv >= cur_max - 1e-12)[0]
        # exact sub‑gradient of max(conv) (average over active shifts)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for i in max_ids:
            partner = i - idx
            ok = (partner >= 0) & (partner < n)
            g[idx[ok]] += 2.0 * a[partner[ok]]
        g /= max_ids.size

        # momentum & step
        v = mu * v + (1.0 - mu) * g
        eta = eta0 / np.sqrt(it)
        a = a - eta * v
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:

        else:

            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = uniform()

    # ------------------------------------------------------------------
    # 5. Return best found as Python floats
    # ------------------------------------------------------------------
    return [float(v) for v in best]
