        L = 1 << ((conv_len - 1).bit_length())

    uniform_score = compute_score(uniform)
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
    a = best.copy()
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1  # initial learning rate

    v = np.zeros_like(a)

    no_improve = 0

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:

def _proj_simplex(v):
    n = v.size
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum == 1, v>=0). """
    n = v.size
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    s = a0.sum()
    if s < 1e-12:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a0 = a0 / s                      # normalise to sum = 1

    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2. helper to evaluate the objective (2·n·max(conv))
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    len_am = n + conv_len - 1          # = 3·n-2
    L_fft = 1 << ((len_am - 1).bit_length())

    def _score(seq):
        fa = np.fft.rfft(seq, n=L_fft)
        conv = np.fft.irfft(fa * fa, n=L_fft)[:conv_len]
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # 3. pick the better of parent / uniform as start point
    # ------------------------------------------------------------------
    best = a0.copy()
    best_score = _score(best)
    uni_score = _score(uniform)
    if uni_score < best_score:
        best = uniform.copy()
        best_score = uni_score

    a = best.copy()                    # working variable

    # ------------------------------------------------------------------
    # 4. optimisation parameters (Adam)
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5                           # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    no_improve = 0
    max_stale = 3000
    jitter_every = 200
    jitter_std = 0.02
    target_score = 1.50274349

    idx_rev = (2 * n - 2) - np.arange(n)    # indices for gradient extraction

    # ------------------------------------------------------------------
    # 5. main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution a*a -------------------------------------------------
        Fa = np.fft.rfft(a, n=L_fft)
        conv_full = np.fft.irfft(Fa * Fa, n=L_fft)[:conv_len]
        max_val = conv_full.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ----- mask of maximal convolution entries ----------------------------
        eps_mask = max(1e-14, max_val * 1e-12)
        mask = (conv_full >= max_val - eps_mask).astype(np.float64)

        # ----- gradient:  g_i = 2 * sum_k mask[k] * a_{k-i} --------------------
        Fm = np.fft.rfft(mask, n=L_fft)
        conv_am = np.fft.irfft(Fa * Fm, n=L_fft)[:len_am]
        g = 2.0 * conv_am[idx_rev]

        # ----- Adam update ----------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _proj_simplex(a)

        # ----- occasional jitter to escape plateaus -------------------------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # ----- restart if stagnated -----------------------------------------
        if no_improve >= max_stale:
            a = best.copy()
            pert = np.exp(rng.normal(0.0, 0.05, size=n))
            a = _proj_simplex(a * pert)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

    # ------------------------------------------------------------------
    # 6. return best found sequence as plain Python floats
    # ------------------------------------------------------------------
    return [float(x) for x in best]
```
