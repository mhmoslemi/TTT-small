import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (L1=1)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # safeguard against all‑zero output
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w

# --------------------------------------------------------------
# Helper: full convolution a * a via FFT (real inputs)
# --------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Return the linear convolution of a with itself (length 2*len(a)-1)."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return conv

# --------------------------------------------------------------
# Helper: sub‑gradient of max_k (a*a)[k] – average over near‑max indices
# --------------------------------------------------------------
def _subgrad_max(conv: np.ndarray, a: np.ndarray, tol: float = 1e-12) -> np.ndarray:
    n = a.size
    max_val = conv.max()
    # indices whose value is within a relative tolerance of the max
    eps = max(max_val * tol, tol)
    idxs = np.where(conv >= max_val - eps)[0]
    g = np.zeros_like(a)
    for idx in idxs:
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        j = np.arange(lo, hi + 1, dtype=np.int64)
        g[j] += 2.0 * a[idx - j]
    if idxs.size > 1:
        g /= idxs.size
    return g

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Initialise sequence ------------------------------------------------
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: a short sqrt‑series (any non‑negative vector works)
        a = np.exp(-np.arange(256, dtype=np.float64) / 80.0)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)                 # ensure sum = 1
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())     # power‑of‑two for FFT

    # track the best solution ever seen
    best = a.copy()
    conv = _conv_full(a, L)
    best_score = 2.0 * n * conv.max()

    # optimisation state
    momentum = np.zeros_like(a)
    step = 0.5
    min_step = 1e-7
    decay = 0.99993          # slower decay than the reference implementation
    beta = 0.9               # momentum factor
    prob_transfer = 0.04     # random mass‑transfer probability
    prob_smooth = 0.008      # occasional smoothing
    max_transfer_frac = 0.45 # how much of a source entry may be moved

    # ------------------------------------------------------------------
    # Main loop ---------------------------------------------------------
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- evaluate current state -----
        conv = _conv_full(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ----- compute sub‑gradient at (near) maximal indices -----
        g = _subgrad_max(conv, a)

        # ----- momentum + gradient step -----
        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        conv_cand = _conv_full(cand, L)
        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max

        improved = False
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            cur_max = cand_max
            improved = True
        else:
            # simulated‑annealing style uphill acceptance
            T = step
            if rng.random() < math.exp(-(cand_score - cur_score) / (max(T, 1e-12))):
                a = cand
                cur_score = cand_score
                cur_max = cand_max

        # ----- keep global best -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        # ----- occasional random mass transfer -----
        if rng.random() < prob_transfer:
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * max_transfer_frac
                delta = a[i] * frac
                a2 = a.copy()
                a2[i] -= delta
                a2[j] += delta
                a2 = _project_simplex(a2)
                conv2 = _conv_full(a2, L)
                score2 = 2.0 * n * conv2.max()
                if score2 < cur_score:
                    a = a2
                    cur_score = score2
                    cur_max = conv2.max()
                    improved = True
                    if score2 < best_score:
                        best_score = score2
                        best = a2.copy()

        # ----- occasional smoothing (3‑point kernel) -----
        if rng.random() < prob_smooth:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)
            conv_s = _conv_full(smooth, L)
            score_s = 2.0 * n * conv_s.max()
            if score_s < cur_score:
                a = smooth
                cur_score = score_s
                cur_max = conv_s.max()
                improved = True
                if score_s < best_score:
                    best_score = score_s
                    best = smooth.copy()

        # ----- step‑size decay -----
        step = max(min_step, step * decay)

        # ----- early‑exit guard -----
        if time.monotonic() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # Optional second start (random) if time permits --------------------
    # ------------------------------------------------------------------
    if time.monotonic() + 5.0 < deadline:
        a_rand = rng.random(n)
        a_rand = _project_simplex(a_rand)
        # reuse the same loop body but with a tighter iteration budget
        sub_deadline = deadline
        momentum = np.zeros_like(a_rand)
        step = 0.4
        while time.monotonic() < sub_deadline:
            conv = _conv_full(a_rand, L)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            g = _subgrad_max(conv, a_rand)
            momentum = beta * momentum + (1.0 - beta) * g
            cand = a_rand - step * momentum
            cand = _project_simplex(cand)

            conv_c = _conv_full(cand, L)
            cand_score = 2.0 * n * conv_c.max()

            if cand_score < cur_score:
                a_rand = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best_score = cur_score
                    best = cand.copy()
            else:
                if rng.random() < math.exp(-(cand_score - cur_score) / (max(step, 1e-12))):
                    a_rand = cand
                    cur_score = cand_score

            # occasional random tweak
            if rng.random() < prob_transfer:
                i = rng.integers(0, n)
                if a_rand[i] > 1e-12:
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    delta = a_rand[i] * (rng.random() * max_transfer_frac)
                    a2 = a_rand.copy()
                    a2[i] -= delta
                    a2[j] += delta
                    a2 = _project_simplex(a2)
                    score2 = 2.0 * n * _conv_full(a2, L).max()
                    if score2 < cur_score:
                        a_rand = a2
                        cur_score = score2
                        if score2 < best_score:
                            best_score = score2
                            best = a2.copy()

            step = max(min_step, step * decay)
            if time.monotonic() + 0.5 > sub_deadline:
                break

    # ------------------------------------------------------------------
    # Final safety fallback (uniform) ------------------------------------
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = 2.0 * n * _conv_full(uniform, L).max()
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # Return as plain Python list of floats
    return [float(x) for x in best]
