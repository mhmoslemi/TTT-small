def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initial vector
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        cur = np.full(n, 1.0/n, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s > 0:
        cur = cur / s
    else:
        cur = np.full_like(cur, 1.0/cur.size)

    n = cur.shape[0]

    def obj(a):
        conv = _conv_fft(a)
        return 2.0 * n * conv.max()

    best_vec = cur.copy()
    best_sc = obj(best_vec)

    # explore families
    for r in np.linspace(0.90, 0.9999, 25):
        cand = _make_exponential(n, r)
        sc = obj(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    for p in np.linspace(0.2, 4.0, 25):
        cand = _make_powerlaw(n, p)
        sc = obj(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    cand = _make_linear(n)
    sc = obj(cand)
    if sc < best_sc:
        best_sc, best_vec = sc, cand.copy()
    cand = _make_quadratic(n)
    sc = obj(cand)
    if sc < best_sc:
        best_sc, best_vec = sc, cand.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    # Main iterative improvement
    it = 0
    eta0 = 0.5
    beta = 5000.0
    target = 1.50274349

    # Precompute length for weight convolution
    len_w = 2 * n - 1
    L = 1 << ((3 * n - 2).bit_length())   # >= len_w + n - 1

    while time.monotonic() < deadline:
        it += 1

        # compute convolution a*a
        fa = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(fa * fa, n=L)[:len_w]
        maxc = conv_full.max()

        # compute weighted gradient
        w = np.exp(beta * (conv_full - maxc))
        fw = np.fft.rfft(w, n=L)
        fr = np.fft.rfft(cur[::-1], n=L)
        corr = np.fft.irfft(fw * fr, n=L)[:len_w + n - 1]
        g = 2.0 * corr[n-1 : n-1 + n]

        eta = eta0 / np.sqrt(it)
        cand = cur * np.exp(-eta * g)
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand = cand / cand_sum
        else:
            cand = np.full_like(cand, 1.0/n)

        cand_sc = obj(cand)

        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()
            if best_sc < target:
                break

        # accept if better, else occasional uphill move
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
        else:
            T = eta0 / (np.sqrt(it) * 10 + 1e-12)
            if rng.random() < np.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # occasional jitter

    if best_sc == float("inf") or best_vec.size == 0:

import time
import numpy as np

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    return np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]


def _make_exponential(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    return a / a.sum()


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    return a / a.sum()


def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()


def _objective(a: np.ndarray) -> float:
    """Exact objective (sum of a is assumed = 1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # -----------------------------------------------------------------
    # initial vector (parent if available)
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    else:
        n = 256
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur.shape[0]

    best_vec = cur.copy()
    best_sc = _objective(best_vec)

    # -----------------------------------------------------------------
    # quick scan of simple families for a better start
    # -----------------------------------------------------------------
    for r in np.linspace(0.90, 0.9999, 25):
        cand = _make_exponential(n, r)
        sc = _objective(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    for p in np.linspace(0.2, 4.0, 25):
        cand = _make_powerlaw(n, p)
        sc = _objective(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    cand = _make_linear(n)
    sc = _objective(cand)
    if sc < best_sc:
        best_sc, best_vec = sc, cand.copy()
    cand = _make_quadratic(n)
    sc = _objective(cand)
    if sc < best_sc:
        best_sc, best_vec = sc, cand.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    # -----------------------------------------------------------------
    # main iterative refinement (smooth weighted gradient + exponentiated update)
    # -----------------------------------------------------------------
    it = 0
    eta0 = 0.5          # initial step size factor
    beta = 5000.0       # smoothing parameter for max approximation
    target = 1.50274349
    len_w = 2 * n - 1
    L = 1 << ((3 * n - 2).bit_length())  # >= len_w + n - 1

    while time.monotonic() < deadline:
        it += 1

        # convolution a*a (full)
        fa = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(fa * fa, n=L)[:len_w]
        maxc = conv_full.max()

        # smooth weight vector approximating arg‑max
        w = np.exp(beta * (conv_full - maxc))

        # weighted gradient: g = 2 * (w * a_rev)_{centered}
        fw = np.fft.rfft(w, n=L)
        fr = np.fft.rfft(cur[::-1], n=L)          # a reversed
        corr = np.fft.irfft(fw * fr, n=L)[:len_w + n - 1]
        g = 2.0 * corr[n - 1:n - 1 + n]

        # multiplicative (exponentiated) update, then renormalise
        eta = eta0 / np.sqrt(it)
        cand = cur * np.exp(-eta * g)
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand = cand / cand_sum
        else:
            cand = np.full_like(cand, 1.0 / n)

        cand_sc = _objective(cand)

        # keep global best
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()
            if best_sc < target:
                break

        # accept if improvement, occasionally accept uphill steps
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
        else:
            T = eta0 / (np.sqrt(it) * 10 + 1e-12)
            if rng.random() < np.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # occasional small random jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=1e-4, size=n)
            tmp = np.maximum(cur + jitter, 0.0)
            tmp = tmp / tmp.sum()
            tmp_sc = _objective(tmp)
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
                if best_sc < target:
                    break
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc

        # time guard
        if time.monotonic() >= deadline:
            break

    # -----------------------------------------------------------------
    # safety fallback
    # -----------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return best_vec.tolist()
