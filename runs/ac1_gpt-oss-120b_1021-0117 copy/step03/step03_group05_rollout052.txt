import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list) with a low AC1 bound.
    The search respects the time budget and keeps the best solution found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # ------------------------------------------------------------------
    # initialise the vector (use the provided parent if it exists)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random positive vector of moderate length
        n0 = 256
        cur = rng.random(n0, dtype=np.float64) + 1e-3
    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s <= 0.0:
        cur = np.full_like(cur, 1.0 / cur.size)
    else:
        cur = cur / s                     # enforce sum = 1

    n = cur.size
    # FFT length (next power of two >= 2*n-1)
    _L = 1 << ((2 * n - 1).bit_length())

    def _conv_fft(v: np.ndarray) -> np.ndarray:
        """full convolution v*v using FFT of length _L."""
        fv = np.fft.rfft(v, n=_L)
        conv = np.fft.irfft(fv * fv, n=_L)[: 2 * n - 1]
        return conv

    def _score(v: np.ndarray) -> float:
        """Exact AC1 score (assumes v may be unnormalised)."""
        conv = _conv_fft(v)
        return float(2.0 * n * conv.max() / (v.sum() ** 2))

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # algorithmic parameters
    it = 0
    no_improve = 0
    add_alpha0 = 0.7
    mul_eta0 = 0.7
    target = 1.50274349
    stall_limit = 2000   # trigger random moves after this many non‑improving steps

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        # ----- convolution and active set (argmax) -----
        conv = _conv_fft(cur)
        max_val = conv.max()
        # tolerance to capture ties (absolute)
        tol = max(1e-12, max_val * 1e-12)
        active = np.where(conv >= max_val - tol)[0]

        # ----- exact subgradient for max(conv) -----
        g = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]

        # ----- additive projected‑gradient step -----
        alpha = add_alpha0 / np.sqrt(it)
        cand_add = cur - alpha * g
        cand_add = np.maximum(cand_add, 0.0)
        s_add = cand_add.sum()
        if s_add == 0.0:
            cand_add = cur.copy()
            s_add = 1.0
        cand_add = cand_add / s_add
        cand_add_sc = _score(cand_add)

        # ----- multiplicative (exponential‑weights) step -----
        eta = mul_eta0 / np.sqrt(it)
        exp_factor = np.exp(-eta * g)
        cand_mul = cur * exp_factor
        s_mul = cand_mul.sum()
        if s_mul == 0.0 or np.isnan(s_mul):
            cand_mul = cur.copy()
            s_mul = 1.0
        cand_mul = cand_mul / s_mul
        cand_mul_sc = _score(cand_mul)

        # ----- select best candidate -----
        # compare three candidates: cur (cur_sc), cand_add, cand_mul
        if cand_add_sc < cand_mul_sc and cand_add_sc < cur_sc:
