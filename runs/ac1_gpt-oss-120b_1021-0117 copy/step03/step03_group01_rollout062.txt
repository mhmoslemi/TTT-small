def _optimize(start_vec, deadline, rng):
    a = start_vec.copy()
    n = a.size
    L = 1 << ((2*n - 1).bit_length())
    max_val, max_idx = _conv_max_and_index(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    step = 0.5
    min_step = 1e-8
    decay = 0.99995
    beta = 0.9
    momentum = np.zeros_like(a)

    while time.monotonic() < deadline:
        # Compute gradient at current max index
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:
            j = np.arange(lo, hi + 1, dtype=int)
            g[j] = 2.0 * a[max_idx - j]
        # Momentum update
        momentum = beta * momentum + (1 - beta) * g

        candidates = []

        # additive update with momentum
        cand = a - step * momentum
        cand = _project_simplex(cand)
        candidates.append(cand)

        # multiplicative update (mirror descent) using gradient (not momentum)
        # small epsilon to avoid overflow; use log
        # compute exponent = -step * g  (some entries may be zero)
        exp_term = np.exp(-step * g)
        cand2 = a * exp_term
        cand2_sum = cand2.sum()
        if cand2_sum > 0:
            cand2 = cand2 / cand2_sum
            # ensure within bounds 0..1000; after normalization values <=1 likely
            cand2 = np.clip(cand2, 0, 1000)
            # renormalize again after clipping to preserve sum=1
            cand2 = cand2 / cand2.sum()
            candidates.append(cand2)

        # random mass transfer occasionally
        if rng.random() < 0.03:
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.4
                delta = a[i] * frac
                cand3 = a.copy()
                cand3[i] -= delta
                cand3[j] += delta
                cand3 = _project_simplex(cand3)
                candidates.append(cand3)

        # smoothing occasionally
        if rng.random() < 0.01:
            kernel = np.array([0.25, 0.5, 0.25])
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)
            candidates.append(smooth)

        # Evaluate all candidates
        for cand in candidates:
            max_val_c, idx_c = _conv_max_and_index(cand, L)
            score_c = 2.0 * n * max_val_c
            if score_c < cur_score - 1e-12:
                a = cand
                cur_score = score_c
                max_idx = idx_c
                if score_c < best_score:
                    best_score = score_c
                    best = cand.copy()
                break  # accept first improvement
            else:
                # maybe accept uphill with small probability
                if rng.random() < math.exp(-(score_c - cur_score) / max(step, 1e-12)):
                    a = cand
                    cur_score = score_c
                    max_idx = idx_c
                    break

        # step decay
        step = max(min_step, step * decay)

        # early exit guard
        if time.monotonic() + 0.2 > deadline:
            break

    return best, best_score

import time
import math
import numpy as np
import scipy.linalg
import scipy.optimize

# ----------------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w


# ----------------------------------------------------------------------
# Analytic sqrt‑series (a*a = 1+…+x^{2n‑2}), used only if no parent is given
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ----------------------------------------------------------------------
# Convolution (via FFT) – return max value and its index
# ----------------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Projected (momentum) sub‑gradient descent on the peak of a*a
# ----------------------------------------------------------------------
def _optimize(start_vec: np.ndarray, deadline: float, rng: np.random.Generator):
    a = start_vec.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    max_val, max_idx = _conv_max_and_index(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    step = 0.5
    min_step = 1e-8
    decay = 0.99995
    prob_random = 0.03
    prob_smooth = 0.01
    beta = 0.9
    momentum = np.zeros_like(a)

    while time.monotonic() < deadline:
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:
            j = np.arange(lo, hi + 1, dtype=int)
            g[j] = 2.0 * a[max_idx - j]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # build a few candidates
        candidates = []

        # additive (momentum) candidate
        cand = a - step * momentum
        cand = _project_simplex(cand)
        candidates.append(cand)

        # multiplicative (mirror‑descent) candidate
        exp_term = np.exp(-step * g)
        cand2 = a * exp_term
        s2 = cand2.sum()
        if s2 > 0:
            cand2 = cand2 / s2
            cand2 = np.clip(cand2, 0, 1000)
            cand2 = cand2 / cand2.sum()
            candidates.append(cand2)

        # occasional random mass transfer
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.4
                cand3 = a.copy()
                cand3[i] -= delta
                cand3[j] += delta
                cand3 = _project_simplex(cand3)
                candidates.append(cand3)

        # occasional smoothing
        if rng.random() < prob_smooth:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)
            candidates.append(smooth)

        improved = False
        for cand in candidates:
            max_val_c, idx_c = _conv_max_and_index(cand, L)
            score_c = 2.0 * n * max_val_c
            if score_c < cur_score - 1e-12:
                a = cand
                cur_score = score_c
                max_idx = idx_c
                improved = True
                if score_c < best_score:
                    best = cand.copy()
                    best_score = score_c
                break
            else:
                # uphill acceptance (simulated annealing)
                if rng.random() < math.exp(-(score_c - cur_score) / max(step, 1e-12)):
                    a = cand
                    cur_score = score_c
                    max_idx = idx_c
                    improved = True
                    break

        if not improved:
            # no candidate succeeded – keep current a
            pass

        step = max(min_step, step * decay)

        if time.monotonic() + 0.5 > deadline:
            break

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming step: maximise sum(g) s.t. f*g <= M, g>=0
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float):
    n = f.size
    c = np.r_[f, np.zeros(n - 1, dtype=f.dtype)]
    r = np.r_[f[0], np.zeros(n - 1, dtype=f.dtype)]
    # Toeplitz matrix of shape (2n‑1, n)
    A = scipy.linalg.toeplitz(c, r)
    b = np.full(2 * n - 1, M, dtype=f.dtype)
    # maximise sum(g)  -> minimise -sum(g)
    obj = -np.ones(n, dtype=f.dtype)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(obj, A_ub=A, b_ub=b, bounds=bounds,
                                 method='highs')
    if res.success:
        return res.x
    return None


def _lp_improve(f0: np.ndarray, deadline: float, rng: np.random.Generator):
    """Run a few LP‑based mixing steps starting from f0."""
    n = f0.size
    L = 1 << ((2 * n - 1).bit_length())
    f = f0.copy()
    M, _ = _conv_max_and_index(f, L)
    best = f.copy()
    best_score = 2.0 * n * M
    mix_factor = 0.2
    max_iters = 8
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        g0 = _solve_lp_max_sum(f, M)
        if g0 is None:
            break
        s = g0.sum()
        if s <= 0:
            break
        g = g0 / s                     # normalise to unit mass
        f = (1.0 - mix_factor) * f + mix_factor * g
        f = _project_simplex(f)
        M_new, _ = _conv_max_and_index(f, L)
        score_new = 2.0 * n * M_new
        if score_new < best_score:
            best = f.copy()
            best_score = score_new
        # stop if improvement is negligible
        if M_new >= M * (1 - 1e-5):
            break
        M = M_new
        it += 1

    return best, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------

    if 'height_sequence_1' in globals():

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    # ------------------------------------------------------------------
    # Stage 1 – LP based mixing (if time permits)
    # ------------------------------------------------------------------
    if time.monotonic() + 30.0 < deadline:

    else:
        best = a0
        Ltmp = 1 << ((2 * best.size - 1).bit_length())
        best_score = 2.0 * best.size * _conv_max_and_index(best, Ltmp)[0]

    # ------------------------------------------------------------------
    # Stage 2 – fine‑grained projected gradient descent
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        opt, opt_score = _optimize(best, deadline, rng)
        if opt_score < best_score:
            best = opt
            best_score = opt_score

    # ------------------------------------------------------------------
    # Stage 3 – tiny random hill‑climbing until the clock expires
    # ------------------------------------------------------------------
    n = best.size
    L = 1 << ((2 * n - 1).bit_length())
    while time.monotonic() + 0.5 < deadline:

        if best[i] <= 0.0:

        j = rng.integers(0, n - 1)

            j += 1

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        max_val, _ = _conv_max_and_index(cand, L)
        score = 2.0 * n * max_val
        if score < best_score:

    # ------------------------------------------------------------------
