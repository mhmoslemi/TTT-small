def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(v: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of v with itself via FFT, length L must be >= 2*len(v)-1."""
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[: 2 * v.size - 1]


def _score(v: np.ndarray, L: int) -> float:
    """AC1 upper bound for a non‑negative vector v (sum assumed >0)."""
    s = v.sum()
    if s < 0.01:
        return float("inf")
    conv = _conv_fft(v, L)
    return float(2.0 * v.size * conv.max() / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x >= 0, sum x = 1}."""
    # Reference: https://arxiv.org/abs/1309.1541
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _gradient_from_active(cur: np.ndarray, active: np.ndarray) -> np.ndarray:

        for _ in range(6):
            tmp = cur - step * grad
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L)
            if tmp_sc < cur_sc:
                cand_add = tmp
                cand_add_sc = tmp_sc
                break
            step *= 0.5          # halve step if no improvement

        # ------------------------------------------------------------------
        # multiplicative (exponential‑weights) update
        # ------------------------------------------------------------------
        eta = eta0 / np.sqrt(it)
        exp_factor = np.exp(-eta * grad)          # grad >= 0, so factor <= 1
        cand_mul = cur * exp_factor
        cand_mul = _project_simplex(cand_mul)
        cand_mul_sc = _score(cand_mul, L)

        # ------------------------------------------------------------------
        # pick the best of the three candidates
        # ------------------------------------------------------------------
        # list of (vector, score)
        candidates = [(cur, cur_sc), (cand_mul, cand_mul_sc)]
        if cand_add is not None:
            candidates.append((cand_add, cand_add_sc))

        # select the lowest‑score candidate
        vec, sc = min(candidates, key=lambda x: x[1])

        # accept the selected candidate
        cur, cur_sc = vec, sc

        # ------------------------------------------------------------------
        # bookkeeping of global best
        # ------------------------------------------------------------------
        if cur_sc < best_sc:
            best_vec, best_sc = cur.copy(), cur_sc
            no_improve = 0
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # occasional random mass‑transfer move if progress stalls
        # ------------------------------------------------------------------
        if no_improve > 1500 and rng.random() < 0.3:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0

        # ------------------------------------------------------------------
        # jittered restart toward the best known solution
        # ------------------------------------------------------------------
        if it % 5000 == 0 and rng.random() < 0.5:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        # early exit if target passed (safety, not required)
        if best_sc < 1.50274349:
            break

    # ----------------------------------------------------------------------
    # final light refinement phase (random transfers)
    # ----------------------------------------------------------------------
    refin_deadline = time.monotonic() + min(0.5, deadline - time.monotonic())
    while time.monotonic() < refin_deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * rng.random() * 0.3
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp, L)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc

    # ----------------------------------------------------------------------
    # return best vector as plain Python list
    # ----------------------------------------------------------------------
    return [float(x) for x in best_vec]
