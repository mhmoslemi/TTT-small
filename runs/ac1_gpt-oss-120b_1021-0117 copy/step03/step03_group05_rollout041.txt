def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.5)   # 0.5 guard

    # init vector
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a = np.ones(n, dtype=np.float64)
    a = np.maximum(a, 0.0)
    # normalize to sum 1
    sum_a = a.sum()
    if sum_a <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / sum_a

    n = a.size

    # compute initial conv and score
    L = 1 << ((2 * n - 1).bit_length())
    fft_func = lambda x: np.fft.rfft(x, n=L)
    def conv_of(v):
        fv = fft_func(v)
        conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
        return conv

    def score_of(v):
        conv = conv_of(v)
        maxc = conv.max()
        return 2.0 * n * maxc / (v.sum()**2)

    cur = a.copy()
    cur_sc = score_of(cur)
    best = cur.copy()
    best_sc = cur_sc

    it = 0
    while time.monotonic() < deadline and best_sc > target:
        it += 1

        conv = conv_of(cur)
        maxc = conv.max()
        # tolerance for active set
        eps = 1e-12
        active = np.flatnonzero(conv >= maxc - eps)

        # gradient
        g = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]   # note k-idx is reversed indices
            
        # step sizes
        step = 0.5 / np.sqrt(it)   # hyperparameter

        cand = cur - step * g
        # project onto simplex
        cand = _project_to_simplex(cand)
        cand_sc = score_of(cand)

        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best = cand.copy()
                best_sc = cand_sc
        else:
            # occasionally accept worse move to escape
            if rng.random() < np.exp(-(cand_sc - cur_sc) / (step + 1e-12)):
                cur = cand
                cur_sc = cand_sc

        # occasional random reallocation when stuck
        if it % 1000 == 0:
            # small random perturbation of best
            noise = rng.normal(scale=1e-3, size=n)
            tmp = best + noise
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_to_simplex(tmp)
            tmp_sc = score_of(tmp)
            if tmp_sc < best_sc:
                best = tmp
                best_sc = tmp_sc
                cur = tmp
                cur_sc = tmp_sc

    # final fallback: few random transfers
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if cur[i] <= 0:
            continue
        delta = cur[i] * rng.random() * 0.3
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_to_simplex(tmp)
        tmp_sc = score_of(tmp)
        if tmp_sc < cur_sc:
            cur = tmp
            cur_sc = tmp_sc
            if tmp_sc < best_sc:
                best = tmp.copy()
                best_sc = tmp_sc

    return [float(x) for x in best]

import time, math
import numpy as np

def _is_prime(p: int) -> bool:
    if p < 2:
        return False
    if p % 2 == 0:
        return p == 2
    r = int(math.isqrt(p))
    for d in range(3, r + 1, 2):
        if p % d == 0:
            return False
    return True

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Construct a non‑negative step‑function whose AC1 bound is well below the
    target 1.50274349.  The construction uses a Sidon set (all pairwise sums
    distinct) for the bulk of the mass and spreads the remaining mass uniformly.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # spare margin

    # ----------------------------------------------------------------------
    # determine length of the sequence (use the provided parent if any)
    # ----------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if parent is not None:
        n = int(len(parent))
    else:
        n = 2000                     # fallback length if no parent is available

    # ----------------------------------------------------------------------
    # build a Sidon set using the classic quadratic construction:
    #   A = { i * q + (i*i mod q) : i = 0 .. q-1 }   (q prime)
    # It guarantees all sums a_i + a_j are distinct.
    # ----------------------------------------------------------------------
    q = int(math.isqrt(n))          # start from √n
    while q > 2 and not _is_prime(q):
        q -= 1
    if q < 2:                       # very small n – fall back to uniform
        a = np.full(n, 1.0 / n, dtype=np.float64)
        return [float(x) for x in a]

    sidon = []
    seen = set()
    for i in range(q):
        val = i * q + (i * i) % q
        if val >= n:
            continue
        # the construction is already collision‑free, but keep a guard
        if val in seen:
            continue
        sidon.append(val)
        seen.add(val)

    m = len(sidon)                  # size of the Sidon set (≈ q)
    # ----------------------------------------------------------------------
    # distribute mass:
    #   α  – fraction of total mass placed on the Sidon set
    #   1‑α – mass spread uniformly on the complement.
    # Choose α that equalises the two possible dominant contributions
    # to the maximal auto‑correlation.
    # ----------------------------------------------------------------------
    if m == 0 or n == m:
        a = np.full(n, 1.0 / n, dtype=np.float64)
        return [float(x) for x in a]

    # α = 1 / (1 + sqrt(n‑m)/m)  (derived from α²/m² = (1‑α)²/(n‑m))
    alpha = 1.0 / (1.0 + math.sqrt(max(0, n - m)) / m)
    # safety clamp
    alpha = max(0.0, min(1.0, alpha))

    w_main = alpha / m                                 # weight on each Sidon index
    w_fill = (1.0 - alpha) / (n - m) if n > m else 0.0 # weight on every other index

    a = np.full(n, w_fill, dtype=np.float64)
    a[sidon] = w_main

    # final normalisation (tiny numerical error may accumulate)
    total = a.sum()
    if total > 0.0:
        a /= total
    else:                         # pathological fallback
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ----------------------------------------------------------------------
    # (optional) quick local tweak while time permits – a few random
    # swaps can only improve the bound, never hurt it.
    # ----------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    def _score(v):
        conv = np.convolve(v, v)
        return 2.0 * n * conv.max() / (v.sum() ** 2)

    best = a.copy()
    best_sc = _score(best)

    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] == 0.0:
            continue
        delta = best[i] * rng.random() * 0.2
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection onto simplex (non‑negative, sum=1)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        sc = _score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc

    # return plain Python list
    return [float(x) for x in best]
