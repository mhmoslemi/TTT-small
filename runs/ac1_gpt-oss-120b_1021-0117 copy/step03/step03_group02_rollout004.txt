import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

def _conv_full_max(a: np.ndarray, L: int):
    """Full self‑convolution via FFT; returns (max value, argmax, full array)."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx, conv
