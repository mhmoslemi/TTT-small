def _grad_smoothed(a, alpha):
    conv = _conv_fft(a)   # size 2n-1
    m = conv.max()
    weights = np.exp(alpha * (conv - m))  # <=1
    a_rev = a[::-1]
    L = 1 << ((len(a_rev) + len(weights) - 1).bit_length())
    fr = np.fft.rfft(a_rev, n=L)
    fw = np.fft.rfft(weights, n=L)
    g_full = np.fft.irfft(fr * fw, n=L)[:len(a_rev)+len(weights)-1]
    # g_i = 2 * g_full[i + len(a)-1]
    offset = len(a) - 1
    g = 2.0 * g_full[offset:offset+len(a)]
    return g

def _grad_from_weights(a, w):
    # a is array length n
    # w is length 2n-1
    n = a.size
    a_rev = a[::-1]
    L = 1 << ((n + len(w) - 1).bit_length())
    fr = np.fft.rfft(a_rev, n=L)

    g_full = np.fft.irfft(fr * fw, n=L)[:n + len(w) - 1]
    offset = n - 1
    g = 2.0 * g_full[offset:offset+n]
    return g

import time
import numpy as np

# ----------------------------------------------------------------------
# Core numeric utilities (FFT based, linear memory)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


def _score_norm1(a: np.ndarray) -> float:
    """Score when sum(a)==1, i.e. 2·n·max(conv)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())


def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a*a)_k  w.r.t. a.
    Implements  g_i = 2·∑_j a_j·w_{i+j}  using one FFT.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = 1 << ((n + w.size - 1).bit_length())
    fr = np.fft.rfft(a_rev, n=L)
    fw = np.fft.rfft(w, n=L)
    full = np.fft.irfft(fr * fw, n=L)[: n + w.size - 1]
    offset = n - 1
    return 2.0 * full[offset: offset + n]


# ----------------------------------------------------------------------
# Simple family generators (all normalized to sum==1)
# ----------------------------------------------------------------------
def _make_exponential(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s else np.full(n, 1.0 / n)


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s else np.full(n, 1.0 / n)


def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()


def _make_sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s else np.full(n, 1.0 / n)


def _make_sine(n: int) -> np.ndarray:
    k = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * k / (n + 1))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s else np.full(n, 1.0 / n)


def _make_cosine(n: int) -> np.ndarray:
    k = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 - np.cos(np.pi * k / (n + 1))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ---------- initialise ----------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    else:
        n = 256
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur.size
    best_vec = cur.copy()
    best_sc = _score_norm1(best_vec)

    # ---------- explore simple families ----------
    families = [
        lambda: _make_exponential(n, rng.uniform(0.90, 0.9999)),
        lambda: _make_powerlaw(n, rng.uniform(0.2, 4.0)),
        lambda: _make_linear(n),
        lambda: _make_quadratic(n),
        lambda: _make_sqrt_series(n),
        lambda: _make_sine(n),
        lambda: _make_cosine(n),
    ]

    for gen in families:
        cand = gen()
        sc = _score_norm1(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    target = 1.50274349
    # ---------- Phase A : smoothed exponential‑gradient descent ----------
    eta0 = 0.8      # initial stepsize factor
    alpha0 = 4.0    # initial smooth‑max sharpness
    it = 0
    while time.monotonic() < deadline:
        it += 1
        eta = eta0 / np.sqrt(it)
        alpha = alpha0 * (1.0 + 0.0005 * it)   # slowly sharpen

        # convolution & weights
        conv = _conv_fft(cur)
        maxc = conv.max()
        # exponential weight (max at 1, others ≤1)
        w = np.exp(alpha * (conv - maxc))

        # gradient from weighted sum of convolutions
        g = _grad_from_weights(cur, w)

        # exponentiated‑gradient update
        cand = cur * np.exp(-eta * g)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / cand_sum

        cand_sc = _score_norm1(cand)

        # accept if better, else occasional Metropolis move
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break
        else:
            # small probability to accept worsening move (annealing)
            T = 0.02 / (np.sqrt(it) + 1.0)
            if rng.random() < np.exp(-(cand_sc - cur_sc) / T):
                cur, cur_sc = cand, cand_sc

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            jitter = rng.normal(scale=1e-4, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = tmp / tmp.sum()
            tmp_sc = _score_norm1(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    if best_sc < target:
                        break

    # ---------- Phase B : occasional random mass‑transfer ----------
    it2 = 0
    while time.monotonic() < deadline:
        it2 += 1
        conv = _conv_fft(cur)
        k = int(np.argmax(conv))
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # product of the contributing pairs
        prod = cur[idx] * cur[k - idx]
        i = int(idx[np.argmax(prod)]) if prod.size else int(rng.integers(0, n))
        j = int(np.argmin(cur))
        if j == i:
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1

        delta = cur[i] * rng.random() * 0.4
        if delta < 1e-12:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_sc = _score_norm1(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break

        # very occasional tiny jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=5e-5, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = tmp / tmp.sum()
            tmp_sc = _score_norm1(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    if best_sc < target:
                        break

    # ------------------------------------------------------------------
    # Final safety fallback (should never trigger)
    # ------------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # Return as plain Python list of floats
    return [float(v) for v in best_vec]
