import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # Guard against pathological all‑zero output
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w

# ----------------------------------------------------------------------
# Full convolution via FFT (real → real)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray, L: int) -> np.ndarray:
    """Return the full linear convolution x*x using an FFT of length L."""
    fv = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * x.size - 1]
    return conv

# ----------------------------------------------------------------------
# Gradient of the max‑peak (or near‑peak) entries of a*a
# ----------------------------------------------------------------------
def _peak_gradient(a: np.ndarray, conv: np.ndarray, rtol: float = 1e-10) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv) w.r.t. a.
    All indices where conv is within rtol*max are treated as active.
    """
    n = a.size
    M = conv.max()
    eps = max(1e-12, rtol * M)
    idxs = np.where(conv >= M - eps)[0]
    g = np.zeros_like(a)
    for i in idxs:
        lo = i - (n - 1)
        if lo < 0:
            lo = 0
        hi = i
        if hi > n - 1:
            hi = n - 1
        j = np.arange(lo, hi + 1, dtype=np.int64)
        g[j] += 2.0 * a[i - j]
    return g

# ----------------------------------------------------------------------
# Core optimizer – projected momentum descent using the peak gradient
# ----------------------------------------------------------------------
def _optimize(start_vec: np.ndarray, deadline: float, rng: np.random.Generator):
    a = start_vec.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two for FFT speed

    # initial score (sum is 1 after simplex projection)
    conv = _full_convolution(a, L)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score

    momentum = np.zeros_like(a)

    # hyper‑parameters
    step = 0.2
    min_step = 1e-7
    decay = 0.9995
    beta = 0.9
    prob_random = 0.04
    prob_smooth = 0.01
    rtol = 1e-10

    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # gradient of the (near) maximal autocorrelation entry
        # ------------------------------------------------------------------
        conv = _full_convolution(a, L)
        g = _peak_gradient(a, conv, rtol)

        # momentum update and projected candidate
        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        cand = _project_simplex(cand)

        # evaluate candidate
        conv_c = _full_convolution(cand, L)
        cand_score = 2.0 * n * conv_c.max()

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            # occasional uphill move (simulated annealing style)
            T = step
            if rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                a = cand
                cur_score = cand_score

        # ------------------------------------------------------------------
        # random mass transfer (exploration)
        # ------------------------------------------------------------------
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.4          # up to 40 % shift
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                conv2 = _full_convolution(cand2, L)
                score2 = 2.0 * n * conv2.max()
                if score2 < cur_score:
                    a = cand2
                    cur_score = score2
                    if score2 < best_score:
                        best = cand2.copy()
                        best_score = score2

        # ------------------------------------------------------------------
        # occasional smoothing (3‑point kernel)
        # ------------------------------------------------------------------
        if rng.random() < prob_smooth:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)
            conv_s = _full_convolution(smooth, L)
            score_s = 2.0 * n * conv_s.max()
            if score_s < cur_score:
                a = smooth
                cur_score = score_s
                if score_s < best_score:
                    best = smooth.copy()
                    best_score = score_s

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
        step = max(min_step, step * decay)

        if time.monotonic() + 0.5 > deadline:

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------

    #               simplex‑projected random start.

    else:
        a0 = rng.random(256)
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    # ------------------------------------------------------------------
    # primary optimisation run

        if alt_score < best_score:
            best = alt_best
            best_score = alt_score

    # ------------------------------------------------------------------
    # return a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best]
