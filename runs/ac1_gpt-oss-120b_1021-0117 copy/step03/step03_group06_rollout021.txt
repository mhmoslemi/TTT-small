def _conv_fft(x, L):
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[:2*len(x)-1]

    start = time.time()

    else:

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    # trim zeros at ends (optional)
    nz = np.where(a > 1e-15)[0]
    if nz.size > 0:
        a = a[nz[0]:nz[-1]+1]

    n = a.size
    # size for FFT: at least 3n-2
    L = 1 << ((3*n-2).bit_length())

    cur = a.copy()
    cur_conv = _conv_fft(cur, L)
    cur_sc = 2.0 * n * cur_conv.max() / (cur.sum()**2)  # sum=1
    best = cur.copy()
    best_sc = cur_sc

    # parameters
    eta0 = 1.0   # initial step size
    beta = 0.5   # smoothing parameter
    beta_inc = 1.001   # multiplicative increase per iteration
    beta_max = 20.0
    no_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # compute convolution
        conv = _conv_fft(cur, L)
        conv_max = conv.max()
        cur_sc = 2.0 * n * conv_max / (cur.sum()**2)

        # compute softmax weights over convolution entries
        # shift for numerical stability
        w = np.exp(beta * (conv - conv_max))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = w / w_sum

        # compute gradient via correlation of w with a
        rev_cur = cur[::-1]
        # FFTs
        fw = np.fft.rfft(w, n=L)
        frev = np.fft.rfft(rev_cur, n=L)
        corr_full = np.fft.irfft(fw * frev, n=L)[:3*n-2]
        grad = 2.0 * corr_full[n-1 - np.arange(n)]

        # step size
        eta = eta0 / math.sqrt(it)

        # exponentiated update
        # limit exponent argument
        exp_arg = -eta * grad
        exp_arg = np.clip(exp_arg, -30, 30)
        cand = cur * np.exp(exp_arg)
        cand_sum = cand.sum()

        else:
            cand = cand / cand_sum

        # evaluate candidate

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of x with itself using an FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]
    return conv
