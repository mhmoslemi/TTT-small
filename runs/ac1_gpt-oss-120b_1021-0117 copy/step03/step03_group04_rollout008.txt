def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # get parent if available
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_raw.ndim != 1 or parent_raw.size == 0:
            raise ValueError
        parent_raw = np.clip(parent_raw, 0.0, 1000.0)
        n = parent_raw.size
        a_parent = np.maximum(parent_raw, 0.0)
        s = a_parent.sum()
        if s >= 0.01:
            a_parent = a_parent / s
        else:
            a_parent = np.full(n, 1.0 / n)
    except Exception:
        parent_raw = None
        n = 256
        a_parent = None

    # base candidates
    # uniform
    uniform = np.full(n, 1.0 / n)
    # triangular (tent)
    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1, n - idx) - 1   # shape triangular with zero edges, may be negative on edges
    tent = np.maximum(tent, 0.0)
    if tent.sum() > 0:
        tent = tent / tent.sum()
    else:
        tent = uniform.copy()
    # random dirichlet-like
    rand_vec = rng.random(n)
    rand_vec = np.maximum(rand_vec, 0.0)
    rand_vec = rand_vec / rand_vec.sum()
    candidates = [uniform, tent, rand_vec]
    if a_parent is not None:
        candidates.append(a_parent)

    # choose best initial by fast_score
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    def fast_score(x):
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()
    best = None
    best_score = np.inf
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    a = best.copy()
    # momentum
    v = np.zeros_like(a)
    mu = 0.9
    eta0 = 1.0

    # track improvement
    last_improve_iter = 0
    it = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # compute conv and gradient
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve_iter = it
            if best_score <= target:
                break

        # compute subgradient of max conv
        max_val = conv.max()
        max_idxs = np.flatnonzero(conv == max_val)
        g = np.zeros_like(a)
        offset_center = n - 1
        for idx_conv in max_idxs:
            offset = idx_conv - offset_center
            if offset >= 0:
                # i from 0 .. n-1-offset
                i = np.arange(0, n - offset, dtype=np.int64)
                j = i + offset
            else:
                # offset negative
                i = np.arange(-offset, n, dtype=np.int64)
                j = i + offset
            # accumulate
            # Ensure i and j are arrays of same length
            # Add a[j] to g[i], a[i] to g[j]
            g[i] += a[j]
            g[j] += a[i]

        # momentum update
        v = mu * v + (1.0 - mu) * g

        # step size
        eta = eta0 / np.sqrt(it)

        # update
        a = _proj_simplex(a - eta * v)

        # occasional jitter to escape stagnation
        if it % 300 == 0:
            # if no improvement recently
            if it - last_improve_iter > 3000:
                # jitter: random multiplicative factor around 1
                jitter = np.exp(rng.normal(0.0, 0.02, size=n))
                a = _proj_simplex(a * jitter)
                v[:] = 0.0  # reset momentum

    # return best sequence as list
    return [float(x) for x in best]

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
