import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Full linear convolution via FFT (returns max value and full array)
# ----------------------------------------------------------------------
def _conv_full_max(a: np.ndarray, L: int):
    """Linear convolution a*a, return (max, full-array)."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv

    a = np.empty(n, dtype=np.float64)

            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

                    conv = conv_pert
                    cur_max = max_pert
                    cur_score = score_pert
                    if score_pert < best_score:
                        best = pert.copy()
                        best_score = score_pert
                        no_improve = 0

        # ---- step‑size decay --------------------------------------------------
        eta = max(eta_min, eta * eta_decay)

        # ---- restart if stuck -------------------------------------------------
        if no_improve >= max_no_improve:
            a = rng.random(n)
            a = a / a.sum()
            cur_max, conv = _conv_full_max(a, L)
            cur_score = 2.0 * n * cur_max
            no_improve = 0

        iter_count += 1

        # ---- safeguard against the hard timeout --------------------------------
        if time.monotonic() + 0.5 > deadline:
            break

    # return the best sequence found (as a plain Python list)
    return [float(x) for x in best]
