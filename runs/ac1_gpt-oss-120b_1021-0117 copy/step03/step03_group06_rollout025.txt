import time, math, numpy as np

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of x with itself using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    return np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all entries zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
