def _optimize_start(start_vec, deadline, rng):

    a = start_vec.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    best = a.copy()
    # compute initial conv and score
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2*n-1]
    max_val = conv.max()
    cur_score = 2 * n * max_val

    best_score = cur_score

    step = 0.5
    min_step = 1e-7
    decay = 0.9995
    beta = 0.9
    momentum = np.zeros_like(a)

    while time.monotonic() < deadline:
        # compute gradient for all max peaks
        tol = max_val * 1e-12 + 1e-15
        idxs = np.where(np.abs(conv - max_val) <= tol)[0]
        g = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(n-1, k)
            i = np.arange(lo, hi+1)
            g[i] += 2 * a[k - i]
        momentum = beta * momentum + (1 - beta) * g
        cand = a - step * momentum
        cand = _project_simplex(cand)

        # compute conv for cand via fft
        fv = np.fft.rfft(cand, n=L)
        conv_cand = np.fft.irfft(fv * fv, n=L)[:2*n-1]
        max_val_cand = conv_cand.max()
        cand_score = 2 * n * max_val_cand

        if cand_score < cur_score:
            a = cand
            conv = conv_cand
            max_val = max_val_cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            # accept with probability via temperature
            T = step
            # compute probability
            if rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                a = cand
                conv = conv_cand
                max_val = max_val_cand
                cur_score = cand_score

        # random mass transfer chance
        if rng.random() < 0.02:

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1, dtype=np.float64)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:                     # guard against degenerate case
        w[:] = 1.0 / n
    return w


# ----------------------------------------------------------------------
# Compute full convolution via FFT and return its maximal value
# ----------------------------------------------------------------------
def _conv_max(vec: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * vec.size - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Gradient of the maximal convolution entries (all peaks)
# ----------------------------------------------------------------------
def _gradient(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    n = a.size
    # treat every entry whose value is within 1e-12 relative of the max as a peak
    tol = max_val * 1e-12 + 1e-15
    peak_idxs = np.where(np.abs(conv - max_val) <= tol)[0]
    g = np.zeros_like(a)
    for k in peak_idxs:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    return g

    linear = np.arange(n, 0, -1, dtype=np.float64)
    linear = linear * (2.0 / (n * (n + 1)))
    starts.append(linear)

    # 4. Exponential with α ≈ 2 (theoretically good)
    starts.append(_exp_vec(alpha=2.0))

    # 5. Random simplex point
    rand = _project_simplex(rng.random(n))
    starts.append(rand)

    # 6. A few random exponentials (different α)
    for _ in range(3):
        a = _exp_vec(alpha=rng.uniform(0.5, 4.0))
        starts.append(a)

    best_overall = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Run optimisation from each start while time permits
    # ------------------------------------------------------------------
    for start in starts:
        if time.monotonic() + 5.0 > deadline:
            break
        cand, sc = _optimize(start, deadline, rng)
        if sc < best_score:
            best_score = sc
            best_overall = cand
        # early exit if we already beat the target bound
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # As a final safety net, if nothing succeeded use the uniform vector
    # ------------------------------------------------------------------
    if best_overall is None:
        best_overall = uniform

    # ------------------------------------------------------------------
    # Return plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best_overall]
