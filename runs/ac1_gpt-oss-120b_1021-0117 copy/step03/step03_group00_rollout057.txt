def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    rng = np.random.default_rng(seed)

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s  # simplex

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    # Pre-alloc
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon_adam = 1e-8

    best = a.copy()
    # compute initial conv, score
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    best_sc = 2.0 * n * conv.max()
    cur = a
    cur_sc = best_sc

    eps0 = 1.0
    eps_decay = 0.9995
    eps_min = 1e-8

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # compute conv for current cur
        cur_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(cur_fft * cur_fft, n=L)[:2*n-1]
        max_val = conv.max()

        # compute softmax weights
        eps = max(eps_min, eps0 * (eps_decay ** it))
        # subtract max for stability
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)  # length >= 3n-2
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = 0.5 / math.sqrt(it)  # base learning rate decays
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon_adam)

        cand = cur - step

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Project to the probability simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
