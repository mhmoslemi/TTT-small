        alpha = alpha0 / np.sqrt(it)
        cand_add = proj_simplex(cur_a - alpha * momentum)
        cand_add_sc = score_vec(cand_add)

        # ----- multiplicative candidate (exponential weights) -----
        eta = eta0 / np.sqrt(it)
        cand_mul = proj_simplex(cur_a * np.exp(-eta * grad))
        cand_mul_sc = score_vec(cand_mul)

        # ----- select the best among current and candidates -----
        if cand_add_sc < cur_sc and cand_add_sc <= cand_mul_sc:
            cur_a, cur_sc = cand_add, cand_add_sc
        elif cand_mul_sc < cur_sc:
            cur_a, cur_sc = cand_mul, cand_mul_sc
        else:
            # occasional Metropolis‑style acceptance of a worse move
            worse_sc = min(cand_add_sc, cand_mul_sc)
            prob = np.exp(-(worse_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                if cand_add_sc < cand_mul_sc:
                    cur_a, cur_sc = cand_add, cand_add_sc
                else:
                    cur_a, cur_sc = cand_mul, cand_mul_sc

        # ----- record best ever -----
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cur_a.copy()

        # ----- early exit -----
        if best_sc < target:
            break

        # ----- occasional jitter around the best solution -----
        if it % 4000 == 0 and rng.random() < 0.3:
            jitter = rng.normal(scale=1e-4, size=n)
            tmp = proj_simplex(best_a + jitter)
            tmp_sc = score_vec(tmp)
            if tmp_sc < best_sc:
                best_a, best_sc = tmp.copy(), tmp_sc
                cur_a, cur_sc = tmp.copy(), tmp_sc

    # ------------------------------------------------------------------
    # final local refinement (random mass‑transfer moves)
    # ------------------------------------------------------------------
    refin_end = time.time() + min(0.4, deadline - time.time())
    while time.time() < refin_end:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_a[i] == 0.0:
            continue
        delta = cur_a[i] * rng.random() * 0.3
        tmp = cur_a.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = proj_simplex(tmp)
        tmp_sc = score_vec(tmp)
        if tmp_sc < cur_sc:
            cur_a, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_a, best_sc = tmp.copy(), tmp_sc

    # ------------------------------------------------------------------
    # return the best sequence found
    # ------------------------------------------------------------------
    return [float(x) for x in best_a]
