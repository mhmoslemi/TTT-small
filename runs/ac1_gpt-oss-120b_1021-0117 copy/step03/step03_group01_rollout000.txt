import time
import numpy as np

def _sqrt_series(n: int) -> np.ndarray:
    """Return the length‑n nonnegative coefficients a such that
    (a * a)_k = 1 for all k = 0,…,2n‑2."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        # sum_{i=1}^{k-1} a_i * a_{k-i}
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k-1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that
    minimizes the AC1 upper bound.
    """
    # allow a tiny safety margin before the hard timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # length of the sequence: prefer the parent length if available
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        # a reasonable default when no parent is present
        n = 2000

    # generate the analytically optimal sqrt‑series
    a = _sqrt_series(n)

    # safety clamps (they are essentially no‑ops for the sqrt series)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)

    # In the unlikely event that time runs out before generation,
    # return the best we have (the current a).
    # The computation above is tiny, so we simply return a.
    # Ensure we exit well before the deadline.
    if time.monotonic() > deadline:
        # fallback to a uniform simple vector (still valid)
        a = np.full(n, 1.0 / n)

    return [float(v) for v in a]
