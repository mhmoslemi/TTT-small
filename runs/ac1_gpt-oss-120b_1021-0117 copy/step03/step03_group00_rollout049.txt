def _project_simplex(v):
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1,n+1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0)
    return w

import time, math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------------
    # initialise a on the simplex (sum == 1)
    # --------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: uniform distribution of length 2500
        n_fallback = 2500
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                     # normalize to simplex

    n = a.size
    # FFT length – enough for 2n‑1 and 3n‑2 convolutions
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------------
    # score on the simplex (sum == 1)
    # --------------------------------------------------------------------
    def _score_vec(v: np.ndarray) -> float:
        vf = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(vf * vf, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    cur = a.copy()
    cur_score = _score_vec(cur)
    best = cur.copy()
    best_score = cur_score

    # --------------------------------------------------------------------
    # Adam state
    # --------------------------------------------------------------------
    m = np.zeros_like(cur)
    vexp = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # smoothing for soft‑max approximation
    eps0 = 1.0
    eps_decay = 0.9995
    eps_min = 1e-7

    lr_base = 0.2
    it = 0
    no_improve = 0

    # --------------------------------------------------------------------
    # optimisation loop
    # --------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution and its maximum
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weights for the smoothed max
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        vexp = beta2 * vexp + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = vexp / (1.0 - beta2 ** it)

        cand = cur - step
        cand = _project_simplex(cand)

        cand_score = _score_vec(cand)

        # acceptance test
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
        else:

            if rng.random() < 0.001:
                cur, cur_score = cand, cand_score
            no_improve += 1

        # record best
        if cur_score < best_score:
            best, best_score = cur.copy(), cur_score
            no_improve = 0

        # ----------------------------------------------------------------
        # periodic symmetrisation (helps reduce peak autocorrelation)
        # ----------------------------------------------------------------
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_vec(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score

            break

    # return as plain list
    return [float(x) for x in best]
