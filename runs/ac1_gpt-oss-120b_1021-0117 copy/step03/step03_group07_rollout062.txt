import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _make_convolution_helper(n: int):
    """Pre‑compute the FFT length and return a convolution function."""
    L = 1 << ((2 * n - 1).bit_length())

    def conv(a: np.ndarray) -> np.ndarray:
        fv = np.fft.rfft(a, n=L)
        return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]

    return conv


def _obj(a: np.ndarray, conv_func) -> float:
    """Exact objective used by the evaluator (scale‑invariant)."""
    n = a.shape[0]
    c = conv_func(a)
    return float(2.0 * n * c.max() / (a.sum() ** 2))

        else:
            T = 1e-3 / (np.sqrt(it) + 1.0)
            if rng.random() < np.exp(-(cand_sc - cur_sc) / T):
                cur = cand
                cur_sc = cand_sc

        # occasional random jitter to shake out of plateaus
        if it % 2000 == 0:
            jitter = rng.normal(scale=8e-5, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = tmp / tmp.sum()
            tmp_sc = _obj(tmp, conv)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    if best_sc < target:
                        break

        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase B – Euclidean projection sub‑gradient refinement
    # ------------------------------------------------------------------
    it2 = 0
    while time.monotonic() < deadline:
        it2 += 1
        full_conv = conv(cur)
        max_val = full_conv.max()
        cur_sc = float(2.0 * n * max_val / (cur.sum() ** 2))

        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            if best_sc < target:
                break

        # top‑k lags again
        if full_conv.size <= top_k:
            idxs = np.arange(full_conv.size)
        else:
            idxs = np.argpartition(-full_conv, top_k - 1)[:top_k]

        g = np.zeros_like(cur)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]

        alpha = alpha0 / np.sqrt(it2)
        cand = cur - alpha * g
        cand = _project_simplex(cand)
        cand_sc = _obj(cand, conv)

        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break
        else:
            # rare exploratory move
            if rng.random() < 0.015:
                up = cur + alpha * g
                up = _project_simplex(up)
                up_sc = _obj(up, conv)
                if up_sc < cur_sc:
                    cur, cur_sc = up, up_sc
                    if up_sc < best_sc:
                        best_sc, best_vec = up_sc, up.copy()
                        if best_sc < target:
                            break

        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # safety fall‑back
    # ------------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # convert to plain Python floats for the harness
    return [float(v) for v in best_vec]
