import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto the probability simplex {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Convolution via FFT – returns max and full array
# ------------------------------------------------------------
def _conv_full_max(a: np.ndarray, L: int):
    """Convolution a*a using FFT of length L, returns (max, full conv)."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv


# ------------------------------------------------------------
# Exact sub‑gradient of the pointwise maximum of the autocorrelation
# ------------------------------------------------------------
def _subgradient(a: np.ndarray, conv: np.ndarray, max_val: float, eps: float = 1e-12):
    """Compute a sub‑gradient of max_k (a*a)[k]."""
    active = np.where(conv >= max_val - eps)[0]
    n = a.size
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1)
        j = k - i
        # contribution a_j to ∂/∂a_i and a_i to ∂/∂a_j
        g[i] += a[j]
        g[j] += a[i]
    if active.size:
        g /= float(active.size)          # any convex combination works
    return g


# ------------------------------------------------------------
# Fallback analytic sqrt series (used when no parent is present)
# ------------------------------------------------------------
def _sqrt_series(n: int):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a
