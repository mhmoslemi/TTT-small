        u = np.sort(v)[::-1]

        rho = np.where(u - css / np.arange(1, v.size+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        return w

    best_vec = cur.copy()

import time
import numpy as np

def _conv_fft(v: np.ndarray, L: int) -> np.ndarray:
    """Convolution v*v using FFT of length L (real output)."""
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    # keep only the true convolution length 2*len(v)-1
    return conv[: 2 * len(v) - 1]

def _score(v: np.ndarray, L: int) -> float:
    """AC1 upper bound for vector v (expects non‑negative entries)."""
    n = len(v)
    conv = _conv_fft(v, L)
    return 2.0 * n * conv.max() / (v.sum() ** 2)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects 1‑d array")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

    # ------------------------------------------------------------------

    s = cur.sum()
    if s <= 0.0:

    else:

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())  # FFT length (next power of two)

    # ------------------------------------------------------------------
