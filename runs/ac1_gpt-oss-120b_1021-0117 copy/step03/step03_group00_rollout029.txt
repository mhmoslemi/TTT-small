import time, math, numpy as np

def _project_simplex(v):
    # Euclidean projection onto simplex sum=1, v>=0.
    # Implementation from earlier.
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    n = a.shape[0]
    # compute conv via FFT
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., float(budget_s) - 0.5)

    # Determine problem size
    if "height_sequence_1" in globals():
        a_parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a_parent = np.maximum(a_parent, 0.0)
        s = a_parent.sum()
        if s <= 0:
            a_parent = np.full_like(a_parent, 1.0 / a_parent.size)
        else:
            a_parent = a_parent / s  # normalize to sum 1
        n = a_parent.size
    else:
        n = 2535  # fallback length
        a_parent = None

    # Determine FFT length for gradient full conv
    L = 1 << ((3 * n - 2).bit_length())

    best_vec = None
    best_sc = np.inf

    def _maybe_update(vec):
        nonlocal best_vec, best_sc
        sc = _score(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
        return sc

    # Include parent as one initial candidate
    if a_parent is not None:
        _maybe_update(a_parent)
    # Also include a uniform vector
    uniform = np.full(n, 1.0 / n)
    _maybe_update(uniform)

    # Settings for optimizer
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.5    # initial learning rate
    eps0 = 1.0   # smoothing parameter initial
    eps_decay = 0.9995
    eps_min = 1e-7
    max_iters_per_run = 5000   # safe upper bound per restart

    def _run_from(x0):
        a = x0.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        it = 0
        no_improve = 0
        while time.monotonic() < deadline:

            conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
            max_val = conv.max()
            # smoothing epsilon
            eps = max(eps_min, eps0 * (eps_decay ** it))
            # softmax weights
            exp_arg = (conv - max_val) / eps
            # to avoid overflow, subtract max? Already sub max_val
            w = np.exp(exp_arg)
            w_sum = w.sum()

    if best_vec is None:

        best_sc = _score(best_vec)

    # Return list
    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Project a vector onto the probability simplex (sum = 1, all >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Exact objective (same as evaluate_sequence) for sum == 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(conv) assuming sum(a)=1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # Determine problem size and initial vectors
    # --------------------------------------------------------
    if "height_sequence_1" in globals():

        else:

        n = a_parent.size
    else:
        # fallback: reasonable length if no parent is supplied
        n = 2535
        a_parent = None

    # FFT length sufficient for all convolutions needed (3n-2)
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------

        sc = _score(vec)
        if sc < best_sc:

    # seed the optimisation with a few obvious candidates
    if a_parent is not None:

    uniform = np.full(n, 1.0 / n)
