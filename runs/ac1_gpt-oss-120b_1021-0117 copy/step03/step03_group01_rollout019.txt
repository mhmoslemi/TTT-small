import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum == 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:          # safety, should not happen
        w[:] = 1.0 / n
    return w

# ----------------------------------------------------------------------
# Convolution via FFT together with its maximum and the indices attaining it
# ----------------------------------------------------------------------
def _conv_and_max(vec: np.ndarray, L: int):
    """Return full convolution, its maximal value and all indices attaining it."""
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * vec.size - 1]
    max_val = conv.max()
    idxs = np.where(conv >= max_val - 1e-12)[0]   # ties treated as subgradient set
    return conv, max_val, idxs

# ----------------------------------------------------------------------
# Sub‑gradient of the pointwise maximum of the convolution
# ----------------------------------------------------------------------
def _subgradient(vec: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max_k (a*a)[k] w.r.t. a, using the given arg‑max set."""
    n = vec.size
    g = np.zeros_like(vec)
    for idx in idxs:
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        j = np.arange(lo, hi + 1, dtype=int)
        g[j] += 2.0 * vec[idx - j]
    if idxs.size > 1:
        g /= float(idxs.size)
    return g

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)   # spare margin

    # ---------- problem size ----------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 256                         # fallback size

    # ---------- initialise start vectors ----------
    starts = []

    # 1) parent if available (projected onto simplex)
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)
        starts.append(a0)
        # also try its reversal – often yields a different basin
        rev = a0[::-1].copy()
        rev = _project_simplex(rev)
        starts.append(rev)

    # 2) uniform simplex
    starts.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 3) a few random simplexes (if time permits)
    while time.monotonic() + 5.0 < deadline and len(starts) < 6:
        rand = rng.random(n)
        rand = _project_simplex(rand)
        starts.append(rand)

    best_vec = None
    best_score = np.inf

    # ---------- optimisation over each start ----------
    for vec in starts:
        if time.monotonic() + 0.5 > deadline:
            break
        cand, cand_score = _optimize(vec, deadline, rng)
        if cand_score < best_score:
            best_score = cand_score
            best_vec = cand

    # fallback (should never happen)
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # return plain python list
    return [float(x) for x in best_vec]
