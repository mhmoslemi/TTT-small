import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

            cur_score = cand_score
            max_val = conv_c.max()          # keep for next loop's stale check
        # otherwise keep the old a (momentum stays, we may still jitter later)

        # occasional random jitter to escape flat regions
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = a * jitter
            a = a / a.sum()
            v[:] = 0.0
            stale = 0

        # restart if we have been stagnant for a long period
        if stale >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.03, size=n))
            a = a * perturb
            a = a / a.sum()
            v[:] = 0.0
            stale = 0

    # ----- return the best sequence (plain python floats) -------------------
    return [float(x) for x in best]
```
