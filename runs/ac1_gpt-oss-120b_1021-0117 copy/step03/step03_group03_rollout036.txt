import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v >= 0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Uses a momentum‑accelerated sub‑gradient method with a larger
    top‑k set and more aggressive restarts. Returns the best
    non‑negative sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Time management
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 2. Starting sequence
    # --------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        a = parent.copy()
    except Exception:
        # fallback: uniform of moderate length (same length as parent if known)
        a = np.ones(256, dtype=np.float64)

    n = int(a.size)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # normalise to sum = 1

    # --------------------------------------------------------------
    # 3. Initial best (also try pure uniform)
    # --------------------------------------------------------------
    best = a.copy()
    # direct score (sum = 1)
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    best_score = float(2.0 * n * conv.max())

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    fa_u = np.fft.rfft(uniform, n=fft_len)
    conv_u = np.fft.irfft(fa_u * fa_u, n=fft_len)[:conv_len]
    uniform_score = float(2.0 * n * conv_u.max())
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # --------------------------------------------------------------
    # 4. Optimisation parameters
    # --------------------------------------------------------------
    eta0 = 0.9                # initial step magnitude
    beta = 0.95               # momentum decay
    top_k = 30                # number of top convolution peaks used for gradient
    jitter_every = 150        # steps between random multiplicative jitter
    jitter_std = 0.02         # std‑dev of log‑normal jitter

    v = np.zeros_like(a)      # momentum buffer
    it = 0

    # --------------------------------------------------------------
    while time.monotonic() < deadline:

        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
