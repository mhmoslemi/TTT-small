def _project_simplex(v):
    # Ensure v is np.ndarray 1D

    # sort descending

import time, math, numpy as np, scipy.sparse as sp, scipy.optimize as opt

def _conv_fft(v: np.ndarray, L: int = None) -> np.ndarray:
    """Full linear self‑convolution of v using a power‑of‑two FFT."""
    n = v.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----------------------------------------------------------------------
    # Initialise: use the provided parent if present, otherwise a uniform.
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = np.full(2500, 1.0 / 2500, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (power of two)

    # ----------------------------------------------------------------------
    # Book‑keeping
    # ----------------------------------------------------------------------
    best = cur.copy()
    best_score = 2.0 * n * _conv_fft(best, L).max()
    no_improve = 0
    it = 0

    # algorithmic parameters
    alpha0 = 2.0                # step‑size multiplier for exponentiated update
    eps_active = 0.03           # relative tolerance for “active’’ convolution indices
    jitter_scale = 2e-5
    tau_mix = 0.5               # mixing factor after LP step
    lp_max_cons = 40           # cap on number of LP constraints (active indices)

    # ----------------------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- current score -------------------------------------------------
        conv = _conv_fft(cur, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ---- keep the best -------------------------------------------------
        if cur_score < best_score:
            best_score, best = cur_score, cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ---- early exit ----------------------------------------------------
        if best_score < 1.50274349:
            break

        # ---- active set (indices where convolution is near its maximum) ---
        act_mask = conv >= cur_max * (1.0 - eps_active)
        active_idx = np.nonzero(act_mask)[0]
        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv)], dtype=int)

        # keep at most lp_max_cons of the largest values
        if active_idx.size > lp_max_cons:
            order = np.argsort(-conv[active_idx])[:lp_max_cons]
            active_idx = active_idx[order]

        m = active_idx.size

        # ---- build the small LP (minimise t subject to f * b ≤ t on active) --
        rows, cols, data = [], [], []
        for r, k in enumerate(active_idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j = np.arange(lo, hi + 1, dtype=int)
            coeff = cur[k - j]                     # weight of b_j in (f*b)[k]
            rows.extend([r] * len(j))
            cols.extend(j.tolist())
            data.extend(coeff.tolist())
            # column for the auxiliary variable t (coefficient -1)
            rows.append(r)
            cols.append(n)                         # last column index = n
            data.append(-1.0)

        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(m, n + 1))
        b_ub = np.zeros(m, dtype=np.float64)

        # equality: sum(b) = 1
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0], dtype=np.float64)

        # bounds: keep b non‑negative and not too spiky
        eps = 1e-12
        ub = 0.5                       # generous but prevents a single spike
        bounds = [(eps, ub)] * n + [(0.0, None)]

        # objective: minimise t
        c = np.zeros(n + 1, dtype=np.float64)
        c[n] = 1.0

        # ---- solve LP ------------------------------------------------------
        try:
            res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                              A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds, method='highs')
        except Exception:
            res = None

        if res is not None and res.success:
            b_opt = res.x[:n]

            # ---- mix current and LP solution --------------------------------
            cand = (1.0 - tau_mix) * cur + tau_mix * b_opt
            cand = _project_simplex(cand)
            cand_max = _conv_fft(cand, L).max()
            cand_score = 2.0 * n * cand_max

            if cand_score < cur_score:
                cur = cand
                continue
            else:
                # try the LP point alone
                b_max = _conv_fft(b_opt, L).max()
                b_score = 2.0 * n * b_max
                if b_score < cur_score:
                    cur = b_opt
                    continue

        # ------------------------------------------------------------------
        # Fall back to an exponentiated‑gradient step using the full active set
        # ------------------------------------------------------------------
        g = np.zeros_like(cur)
        for k in active_idx:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] += 2.0 * cur[k - idx]

        eta = alpha0 / math.sqrt(it)
        mul = np.exp(-eta * g)
        cand = cur * mul
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand = cand / cand_sum
        else:
            cand = np.full_like(cur, 1.0 / n)

        cand_score = 2.0 * n * _conv_fft(cand, L).max()
        if cand_score < cur_score:
            cur = cand
        else:
            # occasional uphill acceptance (Metropolis)
            prob = math.exp(-(cand_score - cur_score) / (eta + 1e-12))
            if rng.random() < prob:
                cur = cand

        # ------------------------------------------------------------------
        # Random mass‑transfer move (small perturbation)
        # ------------------------------------------------------------------
        if rng.random() < 0.07:
            i = rng.integers(0, n)
            if cur[i] > 0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.25
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                tmp_sum = tmp.sum()
                if tmp_sum > 0:
                    tmp = tmp / tmp_sum
                    tmp_score = 2.0 * n * _conv_fft(tmp, L).max()
                    if tmp_score < cur_score:
                        cur = tmp

        # ------------------------------------------------------------------
        # Jittered restart from the best point every few thousand iterations
        # ------------------------------------------------------------------
        if it % 3000 == 0:
            jitter = rng.normal(scale=jitter_scale, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp_sum = tmp.sum()
            if tmp_sum > 0:
                tmp = tmp / tmp_sum
                tmp_score = 2.0 * n * _conv_fft(tmp, L).max()
                if tmp_score < cur_score:
                    cur = tmp

        # ------------------------------------------------------------------
        # Heavy restart after long stagnation
        # ------------------------------------------------------------------
        if no_improve > 8000:
            cur = best + rng.normal(scale=jitter_scale * 5, size=n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            no_improve = 0

    # ----------------------------------------------------------------------
    # Return the best vector found (as a plain Python list)
    # ----------------------------------------------------------------------
    return [float(x) for x in best]
```
