import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Returns the best non‑negative step‑function heights found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ---------- time management ----------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ---------- initialise sequence ----------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = parent.copy()
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # normalise to sum = 1

    # ---------- helpers ----------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def _score(seq):
        fa = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * np.max(conv)

    # ---------- best-so-far ----------
    best = a.copy()
    best_score = _score(best)

    # Uniform fallback (might be better than a badly‑scaled parent)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = _score(uniform)
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # ---------- optimisation parameters ----------
    eta0 = 0.7               # initial step magnitude
    beta = 0.9               # momentum decay
    top_k = 20               # number of convolution peaks to average
    jitter_every = 80        # steps between multiplicative jitter
    jitter_std = 0.015       # jitter strength
    max_stale = 2500         # restart after this many non‑improving steps
    target_score = 1.50274349

    v = np.zeros_like(a)     # momentum buffer
    it = 0
    no_improve = 0

    # ---------- main optimisation loop ----------
    while time.monotonic() < deadline:
        # convolution via FFT
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        cur_score = 2.0 * n * np.max(conv)

        # update best
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # gradient built from top‑k peaks
        if conv.size > top_k:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        else:
            idxs = np.arange(conv.size)

        g = np.zeros_like(a)
        for k in idxs:
            start = max(0, k - n + 1)
            stop = min(n - 1, k)
            if start <= stop:
                idx = np.arange(start, stop + 1, dtype=np.int64)

        g = 2.0 * g / max(1, len(idxs))
