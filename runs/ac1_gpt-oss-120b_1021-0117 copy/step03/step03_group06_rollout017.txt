def _full_conv_fft(a, L):
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[:2*a.size-1]

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]          # sub‑gradient (non‑negative)

        # ---- mirror‑descent step with a few step‑size candidates ----
        best_local_sc = cur_sc
        best_local_a = cur
        best_local_eta = eta
        for factor in (1.0, 0.5, 2.0):
            eta_i = eta * factor
            mul = np.exp(-eta_i * g)
            cand = cur * mul
            cand_sum = cand.sum()
            if cand_sum <= 0.0 or not np.isfinite(cand_sum):
                continue
            cand = cand / cand_sum
            cand_conv = _full_conv_fft(cand, L)
            cand_sc = _score_from_conv(cand_conv, cand.sum(), n)
            if cand_sc < best_local_sc:
                best_local_sc = cand_sc
                best_local_a = cand
                best_local_eta = eta_i

        # accept the best mirror move (allow a tiny uphill jump)
        if best_local_sc < cur_sc:
            cur, cur_sc = best_local_a, best_local_sc
            eta = best_local_eta
        else:
            if rng.random() < 0.01:           # occasional uphill acceptance
                cur, cur_sc = best_local_a, best_local_sc
                eta = best_local_eta

        if cur_sc < best_sc:
            best, best_sc = cur.copy(), cur_sc

        # ---- random mass‑transfer move (exploration) ----
        if rng.random() < mass_move_prob:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                delta = cur[i] * rng.random() * move_frac
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                s_tmp = tmp.sum()
                if s_tmp > 0.0:
                    tmp = tmp / s_tmp
                    tmp_conv = _full_conv_fft(tmp, L)
                    tmp_sc = _score_from_conv(tmp_conv, tmp.sum(), n)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc

        # ---- smoothing move (low‑pass filter) ----
        kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
        smooth = np.convolve(cur, kernel, mode='same')
        smooth = np.maximum(smooth, 0.0)
        s_smooth = smooth.sum()
        if s_smooth > 0.0:
            smooth = smooth / s_smooth
            smooth_conv = _full_conv_fft(smooth, L)
            smooth_sc = _score_from_conv(smooth_conv, smooth.sum(), n)
            if smooth_sc < cur_sc:
                cur, cur_sc = smooth, smooth_sc
                if smooth_sc < best_sc:
                    best, best_sc = smooth.copy(), smooth_sc

        # ---- periodic jittered restart from the best point ----
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=jitter_sigma, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            s_tmp = tmp.sum()
            if s_tmp > 0.0:
                tmp = tmp / s_tmp
                tmp_conv = _full_conv_fft(tmp, L)
                tmp_sc = _score_from_conv(tmp_conv, tmp.sum(), n)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc

        # ---- decay step size ----
        eta *= decay

        # ---- early exit if target reached ----
        if best_sc <= target:
            break

    # ----------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ----------------------------------------------------------------------
    best = np.maximum(best, 0.0)
    total = best.sum()
    if total <= 0.0:
        best = np.full_like(best, 1.0 / best.size)
    else:
        best = best / total                     # normalise to sum‑1 (any scale works)
    return [float(x) for x in best]
