    deadline = start + max(0.0, float(budget_s) - 0.5)

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Exact score used by the evaluator (sum is assumed 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(conv) for a normalized to sum==1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # tiny safety margin

    # --------------------------------------------------------
    # Initialise a feasible vector (use parent if present)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: a short exponentially decaying tail
        n_fallback = 2500
        a0 = np.exp(-np.arange(n_fallback) / 80.0)
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                       # normalise to the simplex

    n = a0.size
    # FFT length sufficient for linear convolutions of length up to 3n‑2
    L = 1 << ((3 * n - 2).bit_length())

    cur = a0.copy()
    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------
    eps = 1.0                # current smoothing parameter
    eps_decay = 0.9996
    eps_min = 1e-9
    step0 = 0.6              # base step magnitude
    it = 0
    no_improve = 0

    # --------------------------------------------------------
    # Optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---------- convolution of current iterate ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ---------- smoothed weights (softmax) ----------
        eps = max(eps_min, eps * eps_decay)
        w_exp = np.exp((conv - max_val) / eps)          # stable because max_val is subtracted
        w_sum = w_exp.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_exp / w_sum

        # ---------- gradient of the smoothed max ----------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)                # slice picking the needed part
        g = 2.0 * grad_full[idx]

        # ---------- step and projection ----------
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm
        cand = cur - alpha * g
        cand = _project_simplex(cand)

        # ---------- evaluate ----------
        cand_sc = _score(cand)

        # ---------- acceptance ----------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # occasional Metropolis‑type acceptance to escape shallow traps
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ---------- random mass‑transfer when stuck ----------
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0
                else:
                    no_improve = 0

        # ---------- periodic symmetrisation ----------
        if it % 1500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # optional early exit if we have already passed the target bound
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best]
