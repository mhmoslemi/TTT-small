def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback sqrt series of length 256 is small; we then upscale via interpolation
        n_fallback = 256
        a0 = _sqrt_series(n_fallback)
        # upsample to target length using linear interpolation
        target_n = 2535
        x = np.linspace(0, 1, n_fallback)
        xi = np.linspace(0, 1, target_n)
        a0 = np.interp(xi, x, a0)

    # Ensure nonnegative and simplex
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())

    # internal scoring: sum is 1
    def score_of(vec):
        max_val, _ = _conv_max_and_index(vec, L)
        return 2.0 * n * max_val

    best = a0.copy()
    best_score = score_of(best)

    # main loop
    a = best.copy()
    cur_score = best_score
    max_val, max_idx = _conv_max_and_index(a, L)
    # momentum for gradient descent
    momentum = np.zeros_like(a)
    step = 0.5
    min_step = 1e-7
    decay = 0.9995
    beta = 0.9
    # probabilities
    prob_swap = 0.07
    prob_smooth = 0.02
    prob_random = 0.005

    while time.monotonic() < deadline:
        # recompute max_idx if needed (max_idx may be outdated after a changes)
        # but we have it from last conv compute; after acceptance we update.
        # compute gradient of active max shift
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            g[j] = 2.0 * a[max_idx - j]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        cand = _project_simplex(cand)

        cand_score = score_of(cand)
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            max_val, max_idx = _conv_max_and_index(a, L)
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            # simulated annealing acceptance
            T = step
            if rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                a = cand
                cur_score = cand_score
                max_val, max_idx = _conv_max_and_index(a, L)

        # random swap move
        if rng.random() < prob_swap:
            # choose i with substantial mass
            nonzero = np.where(a > 1e-12)[0]
            if nonzero.size > 0:
                i = rng.choice(nonzero)
                j = rng.integers(0, n)
                if i != j:
                    frac = rng.random() * 0.4
                    delta = a[i] * frac
                    cand2 = a.copy()
                    cand2[i] -= delta
                    cand2[j] += delta
                    cand2 = _project_simplex(cand2)
                    cand2_score = score_of(cand2)
                    if cand2_score < cur_score:
                        a = cand2
                        cur_score = cand2_score
                        max_val, max_idx = _conv_max_and_index(a, L)
                        if cand2_score < best_score:
                            best = cand2.copy()
                            best_score = cand2_score

        # smoothing move
        if rng.random() < prob_smooth:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(a, kernel, mode='same')
            smooth = _project_simplex(smooth)
            smooth_score = score_of(smooth)
            if smooth_score < cur_score:
                a = smooth
                cur_score = smooth_score
                max_val, max_idx = _conv_max_and_index(a, L)
                if smooth_score < best_score:
                    best = smooth.copy()
                    best_score = smooth_score

        # tiny random perturbation
        if rng.random() < prob_random:
            # add small Gaussian noise then project
            noise = rng.normal(scale=0.01, size=n)
            cand_r = a + noise
            cand_r = _project_simplex(cand_r)
            cand_r_score = score_of(cand_r)
            if cand_r_score < cur_score:
                a = cand_r
                cur_score = cand_r_score
                max_val, max_idx = _conv_max_and_index(a, L)
                if cand_r_score < best_score:
                    best = cand_r.copy()
                    best_score = cand_r_score

        # step size adaptation
        step = max(min_step, step * decay)

        # early exit
        if time.monotonic() + 0.5 > deadline:
            break

    # second stage: random restarts if time remains
    while time.monotonic() + 2.0 < deadline:
        # random init
        rand_vec = rng.random(n)
        rand_vec = _project_simplex(rand_vec)
        a = rand_vec.copy()
        cur_score = score_of(a)
        momentum = np.zeros_like(a)
        step = 0.5
        max_val, max_idx = _conv_max_and_index(a, L)
        # short inner loop
        inner_deadline = time.monotonic() + 5.0  # 5 seconds per restart
        while time.monotonic() < inner_deadline and time.monotonic() < deadline:
            lo = max(0, max_idx - (n - 1))
            hi = min(n - 1, max_idx)
            g = np.zeros_like(a)
            if lo <= hi:
                j = np.arange(lo, hi + 1)
                g[j] = 2.0 * a[max_idx - j]
            momentum = beta * momentum + (1.0 - beta) * g
            cand = a - step * momentum
            cand = _project_simplex(cand)
            cand_score = score_of(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                max_val, max_idx = _conv_max_and_index(a, L)
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
            else:
                T = step
                if rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                    a = cand
                    cur_score = cand_score
                    max_val, max_idx = _conv_max_and_index(a, L)
            step = max(min_step, step * decay)
            if time.monotonic() + 0.2 > inner_deadline:
                break
        # after this run, break if time low
        if time.monotonic() + 0.5 > deadline:
            break

    # final fallback and return
    # Ensure best is a list of Python floats
    result = [float(x) for x in best]
    return result

# ----------------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_idx = np.where(rho_arr > 0)[0]
    if rho_idx.size:
        rho = rho_idx[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # guard against all‑zero (should not happen)
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w

def _sqrt_series(n: int) -> np.ndarray:

            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0

    idx = int(conv.argmax())
