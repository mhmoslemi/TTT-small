def _project_simplex(v):
    # Project onto simplex sum(v)=1, v>=0
    v = v - np.max(v)  # maybe shift for numerics? Not needed.
    # Use the standard algorithm:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w
```

    total = a.sum()
    if total == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / total
    a = _project_simplex(a)

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    max_val, _ = _conv_full_max(a, L)
    best = a.copy()
    best_max = max_val

    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    step = 0.2
    step_decay = 0.995
    min_step = 1e-8
    K = 20                     # top lags used for gradient
    alpha = 12.0               # soft‑max sharpness
    prob_random = 0.015        # occasional mass‑transfer
    max_no_improve = 250
    no_improve = 0
    vel = np.zeros_like(a)     # momentum / velocity
    beta = 0.9                  # momentum coefficient

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        max_val, conv = _conv_full_max(a, L)

        # keep global best
        if max_val < best_max - 1e-15:
            best_max = max_val
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # top‑K lags
        if K >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -K)[-K:]

        # soft‑max weights
        w_raw = np.exp(alpha * (conv[idxs] - max_val))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # sub‑gradient (weighted over selected lags)
        g = np.zeros_like(a)
        for idx, wi in zip(idxs, w):
            lo = max(0, idx - (n - 1))
            hi = min(idx, n - 1)
            if lo <= hi:
                j = np.arange(lo, hi + 1)
                g[j] += 2.0 * wi * a[idx - j]

        # heavy‑ball / momentum update
        vel = beta * vel - step * g
        cand = a + vel
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_max, _ = _conv_full_max(cand, L)
        if cand_max < max_val:
            a = cand
        else:
            vel = beta * vel  # damp velocity on reject

        # occasional random mass transfer
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.25
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                max2, _ = _conv_full_max(cand2, L)
                if max2 < max_val:
                    a = cand2
                    if max2 < best_max:
                        best_max = max2
                        best = cand2.copy()

        # step schedule
        step = max(min_step, step * step_decay)

        # restart if stuck
        if no_improve >= max_no_improve:
            a = rng.random(n)
            a = _project_simplex(a)
            vel = np.zeros_like(a)
            step = 0.2
            no_improve = 0

        # safeguard against overshooting the deadline
        if time.monotonic() + 0.5 > deadline:
            break

    return [float(x) for x in best]
