    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    a = np.maximum(a, 0.0)

    else:

    lam_end = 20000.0

    while time.monotonic() < deadline:

            # early exit if surpass target
            if best_score < 1.50274349:

import time, math, numpy as np

def _conv_self(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of x with itself using FFT length L."""
    fx = np.fft.rfft(x, n=L)
    return np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]

def _grad_wrt_a(w: np.ndarray, a_rev: np.ndarray, L: int) -> np.ndarray:
    """Gradient = 2 * (w * a_rev) sliced to length len(a_rev)."""
    fw = np.fft.rfft(w, n=L)
    fa = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(fw * fa, n=L)
    n = a_rev.size
    # slice indices n-1 .. 2n-2 (inclusive) → length n
    return 2.0 * full[n - 1 : n - 1 + n]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    # --- initialise ---------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = np.full(2000, 1.0 / 2000, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                     # put onto the probability simplex

    n = a.size
    L = 1 << ((3 * n - 2).bit_length())   # power‑of‑two length for all convolutions

    cur = a.copy()
    # initial score (sum(cur) == 1 → score = 2*n*max_conv)
    cur_conv = _conv_self(cur, L)
    cur_score = 2.0 * n * cur_conv.max()
    best = cur.copy()
    best_score = cur_score

    # --- hyper‑parameters ---------------------------------------------------
    it = 0
    eta0 = 1.0                     # base step size for exponentiated update
    lam_start = 0.0                # start smoothing weight
    lam_end = 20000.0              # final weight (sharp soft‑max)
    lam_ramp = 5000                # iterations over which λ grows linearly
    jitter_scale = 1e-4
    jitter_interval = 4000
    mass_move_prob = 0.07
    target = 1.50274349

    # --- main optimisation loop --------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # 1) current convolution and max
        conv = _conv_self(cur, L)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score, best = cur_score, cur.copy()
            if best_score < target:
                break

        # 2) smoothing parameter λ (linear schedule)
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp)

        # 3) soft‑max weights over the convolution entries
        #    w_k = exp(λ*(conv_k - max_conv))  (max weight = 1)
        w = np.exp(lam * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # 4) gradient of the smoothed objective
        grad = _grad_wrt_a(w, cur[::-1], L)

        # 5) exponentiated (mirror) update
        eta = eta0 / math.sqrt(it)
        mul = np.exp(-eta * grad)
        cand = cur * mul
        cand_sum = cand.sum()
        if cand_sum <= 0.0 or not np.isfinite(cand_sum):
            cand = np.full_like(cur, 1.0 / n)
        else:
            cand = cand / cand_sum

        # 6) evaluate candidate
        cand_conv = _conv_self(cand, L)
        cand_score = 2.0 * n * cand_conv.max()

        # 7) acceptance (always accept improvements)
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
        else:
            # occasional uphill move (tiny temperature schedule)
            temp = 0.01 / math.sqrt(it)
            if rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):
                cur, cur_score = cand, cand_score

        # 8) random mass‑transfer move (helps escape plateaus)
        if rng.random() < mass_move_prob:
            nz = np.where(cur > 1e-12)[0]
            if nz.size > 1:
                i = rng.choice(nz)
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                delta = cur[i] * rng.random() * 0.2
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                tmp = tmp / tmp.sum()
                tmp_conv = _conv_self(tmp, L)
                tmp_score = 2.0 * n * tmp_conv.max()
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best_score, best = tmp_score, tmp.copy()
                    if best_score < target:
                        break

        # 9) jittered restart from the best point (periodic)
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=jitter_scale, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = tmp / tmp.sum()
            tmp_conv = _conv_self(tmp, L)
            tmp_score = 2.0 * n * tmp_conv.max()
            if tmp_score < cur_score:
                cur, cur_score = tmp, tmp_score
            if tmp_score < best_score:
                best_score, best = tmp_score, tmp.copy()
                if best_score < target:
                    break

        # safety break if time runs out
        if time.monotonic() >= deadline - 0.05:
            break

    # return as plain Python list
    return [float(x) for x in best]
