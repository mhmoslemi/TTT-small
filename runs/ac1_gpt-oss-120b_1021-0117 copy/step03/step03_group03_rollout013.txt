def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # try to get parent
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = len(parent)
        a0 = parent.copy()
    except Exception:
        n = 2048 # maybe choose power of 2 for faster FFT
        a0 = np.ones(n)
    
    # normalise to sum=1:
    a0 = np.maximum(a0, 0.0)
    if a0.sum() < 1e-12:
        a0 = np.full(n, 1.0/n)
    else:
        a0 = a0 / a0.sum()

    # keep best candidate
    best = a0.copy()
    best_score = 2.0 * n * np.max(np.convolve(best, best))
    # optionally also evaluate uniform for baseline.
    uniform = np.full(n, 1.0/n)
    uniform_score = 2.0 * n * np.max(np.convolve(uniform, uniform))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
    
    # Precompute FFT length
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    
    # Adam optimizer variables:
    m = np.zeros_like(a0)
    v2 = np.zeros_like(a0)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0.2 # initial step size
    # We'll adapt step schedule: step * 1/sqrt(step_number)
    
    # parameters controlling gradient selection
    threshold_ratio = 0.99  # consider peaks above threshold*max
    jitter_interval = 200
    jitter_std = 0.05

    while time.monotonic() < deadline:
        # FFT conv
        fa = np.fft.rfft(a0, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        score = 2.0 * n * max_val
        
        # update best
        if score < best_score:
            best = a0.copy()
            best_score = score
            no_improve = 0
            if best_score <= 1.50274349: # target
                break
        else:
            # keep count of no improve iterations
            pass
        
        # Determine active peaks
        # indices where conv >= threshold_ratio * max_val
        thresh = max_val * threshold_ratio
        active_idx = np.where(conv >= thresh)[0]
        # compute gradient for each such index
        g = np.zeros_like(a0)
        for k in active_idx:
            offset = k - (n - 1)
            # overlapping i range:
            if offset >= 0:
                i_start = offset
                i_end = n  # exclusive
            else:
                i_start = 0
                i_end = n + offset
            idx = np.arange(i_start, i_end, dtype=np.int64)
            partner = k - idx
            g[idx] += a0[partner]
        # derivative factor (2)
        g = 2.0 * g / max(1, len(active_idx))

        # Adam update
        step += 1
        lr = step**(-0.5) * 0.5  # initial lr decays
        m = beta1 * m + (1 - beta1) * g
        v2 = beta2 * v2 + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**step)
        v_hat = v2 / (1 - beta2**step)
        a_next = a0 - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project to simplex
        # simple Euclidean projection
        # reuse projection function from earlier code
        a0 = _proj_simplex(a_next)

        # jitter
        if step % jitter_interval == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a0 = _proj_simplex(a0 * jitter)
            # reset moments
            m.fill(0.0)
            v2.fill(0.0)

    # return best as list of floats
    return [float(v) for v in best]

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑d array")
    n = v.shape[0]
    # shift by max for numerical stability
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # all entries are zero – return uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Gradient‑based search for a step‑function that minimises the AC‑1 bound.
    Returns the best non‑negative sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Time handling
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 2.  Starting sequence (use parent if provided)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = parent.copy()
    except Exception:
        # fallback to a modest length uniform vector
        n = 256
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # normalise to sum = 1

    # ------------------------------------------------------------------
    # 3.  Initialise best candidate
    # ------------------------------------------------------------------
    best = a.copy()
    conv_len = 2 * n - 1
    # use dense convolution just once for the initial score (cheap for n≈2k)
    best_score = float(2.0 * n * np.max(np.convolve(best, best)))

    # also keep uniform as a cheap fallback
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(2.0 * n * np.max(np.convolve(uniform, uniform)))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score

    # ------------------------------------------------------------------
    # 4.  FFT preparation (fast convolution each iteration)
    # ------------------------------------------------------------------
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 5.  Optimisation parameters
    # ------------------------------------------------------------------
    eta0 = 0.6                 # initial step magnitude
    beta = 0.95                # momentum decay
    top_k = min(60, n)         # number of top peaks used for gradient
    jitter_every = 150         # steps between multiplicative jitters
    jitter_std = 0.04          # log‑normal jitter std‑dev
    max_stale = 2500           # restart after this many non‑improving steps
    target_score = 1.50274349  # stop as soon as this is reached

    v = np.zeros_like(a)       # momentum buffer
    it = 0
    stale = 0

    # ------------------------------------------------------------------
    # 6.  Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- convolution via FFT ------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ---- update best if we made progress -----------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            stale = 0
            if best_score <= target_score:
                break
        else:
            stale += 1

        # ---- build a sub‑gradient from the largest convolution peaks
        # (use the top‑k peaks to obtain a smoother direction)
        if top_k < conv.size:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        else:
            idxs = np.arange(conv.size)
        g = np.zeros_like(a)

        for k in idxs:
            offset = k - (n - 1)
            if offset >= 0:
                i_start = offset
                i_end = n
            else:
                i_start = 0
                i_end = n + offset
            i = np.arange(i_start, i_end, dtype=np.int64)
            j = k - i
            g[i] += a[j]

        # Scale gradient – the factor 2 comes from the derivative of a[i]*a[j]
        g = 2.0 * g / max(1, len(idxs))

        # ---- momentum update + projection onto the simplex -------------
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g
        a = _proj_simplex(a - eta * v)

        # ---- occasional random jitter to escape plateaus --------------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v.fill(0.0)
            stale = 0

        # ---- long stagnation → restart around the current best ----------
        if stale >= max_stale:
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * perturb)
            v.fill(0.0)
            stale = 0

    # ------------------------------------------------------------------
    # 7.  Return the best sequence (plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
