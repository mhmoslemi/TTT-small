def _project_simplex(v):
    # Projection onto simplex {x >= 0, sum x = 1}
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # ensure sum exactly 1 (floating rounding)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / v.size)
    return w

def _conv_fft(v, L):
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * v.size - 1]
    return conv

import time
import numpy as np
import scipy.sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# utility functions (all top‑level, no closures)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    cond = u - css / np.arange(1, v.size + 1)
    rho = np.where(cond > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0.0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / v.size)
    return w


def _conv_fft(v: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution v*v using a power‑of‑two FFT length L."""
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * v.size - 1]
    return conv


def _score(v: np.ndarray, conv: np.ndarray = None) -> float:
    """Exact score used by the evaluator (scale‑invariant, sum may be 0)."""
    n = v.shape[0]
    if conv is None:
        L = 1 << ((2 * n - 1).bit_length())
        conv = _conv_fft(v, L)
    maxb = conv.max()
    s = v.sum()
    if s == 0.0:
        return np.inf
    return 2.0 * n * maxb / (s * s)


def _gradient_softmax(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Approximate sub‑gradient of max(conv) using a soft‑max weighting.
    Returns a vector of length len(a).
    """
    maxc = conv.max()
    w = np.exp((conv - maxc) / temperature)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum
    # gradient = 2 * (a_rev * w)[n-1 : n-1+len(a)]
    rev = a[::-1]
    total_len = rev.size + w.size - 1          # = 3*len(a)-2
    Lg = 1 << (total_len.bit_length())
    fr = np.fft.rfft(rev, n=Lg)
    fw = np.fft.rfft(w, n=Lg)
    full = np.fft.irfft(fr * fw, n=Lg)[:total_len]
    start = rev.size - 1                      # = len(a)-1
    grad = 2.0 * full[start:start + rev.size]
    return grad


def _build_convolution_matrix(a: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build a sparse (2n‑1) × n matrix M such that M @ g = conv(a, g) (linear convolution).
    Zero entries are omitted for efficiency.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    # each column i contributes a[i] to rows i … i+n‑1
    nnz_est = n * n
    data = np.empty(nnz_est, dtype=np.float64)
    row_ind = np.empty(nnz_est, dtype=np.int32)
    col_ind = np.empty(nnz_est, dtype=np.int32)
    pos = 0
    for i, val in enumerate(a):
        if val == 0.0:
            continue
        r = np.arange(i, i + n, dtype=np.int32)
        data[pos:pos + n] = val
        row_ind[pos:pos + n] = r
        col_ind[pos:pos + n] = i
        pos += n
    data = data[:pos]
    row_ind = row_ind[:pos]
    col_ind = col_ind[:pos]
    return scipy.sparse.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
