def _conv_fft(v: np.ndarray) -> np.ndarray:
    """FFT-based convolution for linear convolution."""
    n_v = len(v)
    n2 = 2 * n_v - 1
    n_fft = 1 << (n2).bit_length()
    fv = np.fft.rfft(v, n_fft)
    conv = np.fft.irfft(fv * fv, n_fft)
    return conv[:n2]

def _score(v: np.ndarray) -> float:
    """Compute AC1 score using full convolution and scaling."""
    if not isinstance(v, np.ndarray):
        v = np.array(v)
    if v.size == 0:
        return float('inf')
    conv = _conv_fft(v)
    n = len(v)
    max_conv = np.max(conv)
    s = np.sum(v)
    if s < 0.01:
        return np.inf
    return 2 * n * max_conv / (s * s)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector onto the simplex (sum=1, non-negative)."""
    u = np.sort(v, kind='mergesort')[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(np.cumsum(u) < (len(u) + 1) * np.arange(1, len(u)+1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = css[-1] / float(len(v))
    return np.maximum(v - theta, 0.0)

def _exponentiate(a, b):
    """Compute exponential for multiplicative updates."""
    return a * b
