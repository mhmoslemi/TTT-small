def _project_simplex(v):
    # Ensure all non-negative after projection onto simplex of sum=1.
    if np.sum(v) == 1 and np.all(v>=0):
        return v.copy()
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    rho = np.where(u - css / ind > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, >=0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    rho = np.where(u - css / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        cand = _project_simplex(cand)

        # evaluate candidate
        cand_sc = _score(cand)

        # keep the best ever seen
        if cand_sc < best_sc:
            best_sc = cand_sc
            best = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        # accept if improvement
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc

        # ----- occasional random mass transfer when stuck ---------------
        if no_improve >= max_no_improve:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best = tmp.copy()
                    best_sc = tmp_sc
            no_improve = 0

        # ----- periodic symmetrisation ---------------------------------
        if it % 300 == 0:
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)
            cur_sc = _score(cur)
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        # early exit if target reached
        if best_sc < 1.50274349:
            break

    return [float(x) for x in best]
