def _next_pow2(x): return 1 << (x - 1).bit_length() if x>0 else 1

def _conv_fft(v, L):
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[:2*v.size-1]

def _score(v, L):
    conv = _conv_fft(v, L)
    max_conv = conv.max()
    s = v.sum()
    if s < 0.01:
        return float('inf')
    n = v.size
    return 2.0 * n * max_conv / (s * s)

def _gradient(v, L):
    n = v.size
    conv = _conv_fft(v, L)
    max_conv = conv.max()
    # active mask: those which are equal to max within tolerance
    eps = 1e-12 * max_conv  # tolerance relative
    mask = (conv >= max_conv - eps).astype(np.float64)  # length 2n-1
    # reversed a
    a_rev = v[::-1]
    # compute convolution of a_rev and mask via FFT size L
    fa_rev = np.fft.rfft(a_rev, n=L)
    fm = np.fft.rfft(mask, n=L)
    grad_full = np.fft.irfft(fa_rev * fm, n=L)
    # extract slice
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]
    return grad

def _project_simplex(v):
    """Project onto simplex sum=1, v>=0."""
    # Using algorithm from Duchi et al.
    # Works for any real vector
    n = v.size
    u = np.sort(v)[::-1]  # descending
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

import time
import numpy as np

# ---------- helper functions (top level) ----------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x (x>0)."""
    return 1 << (x - 1).bit_length()


def _conv_fft(v: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution v*v using FFT length L, returning length 2*len(v)-1."""
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[: 2 * v.size - 1]


def _score(v: np.ndarray, L: int) -> float:
    """AC1 upper bound for vector v (assumed non‑negative)."""
    conv = _conv_fft(v, L)
    max_conv = conv.max()
    s = v.sum()
    if s < 0.01:
        return float("inf")
    n = v.size
    return 2.0 * n * max_conv / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient(v: np.ndarray, L: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (v*v)[k] w.r.t. v.
    Returns a vector of length n.
    """
    n = v.size
    # full auto‑convolution
    conv = _conv_fft(v, L)
    max_val = conv.max()
    # mask of active indices (allow tiny tolerance)
    eps = 1e-12 * max_val
    mask = (conv >= max_val - eps).astype(np.float64)  # length 2*n-1

    # reversed v
    v_rev = v[::-1]

    # convolution of v_rev and mask (correlation)
    fv_rev = np.fft.rfft(v_rev, n=L)
    fm = np.fft.rfft(mask, n=L)
    corr_full = np.fft.irfft(fv_rev * fm, n=L)

    start = n - 1
    grad = 2.0 * corr_full[start : start + n]
    return grad
