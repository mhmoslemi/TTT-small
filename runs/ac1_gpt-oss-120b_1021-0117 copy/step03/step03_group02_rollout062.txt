import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
# Convolution (exact, returns max value)
# ------------------------------------------------------------
def _conv_max(a: np.ndarray) -> float:
    """Full linear convolution a*a and return its maximum."""
    conv = np.convolve(a, a)
    return float(conv.max())

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a list of non‑negative heights that (heuristically) minimise
    the AC1 upper bound as measured by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ----- initialise -------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # simple fallback: start from a smooth decreasing curve
        n = 256
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.001 * x) + 0.01
    # normalise to the probability simplex
    a = _project_simplex(a / (a.sum() if a.sum() != 0 else 1.0))

    n = a.size
    # keep the best candidate seen so far
    best = a.copy()
    best_score = 2.0 * n * _conv_max(a)

    # ----- optimisation parameters ------------------------------------
    step = 0.2                # initial gradient step size
    decay = 0.9995            # exponential decay of step
    min_step = 1e-7
    no_improve = 0
    max_no_improve = 250      # restart after this many stalls
    prob_swap = 0.03          # occasional random swap probability
    K = 8                     # top‑lag count for gradient approximation
    alpha = 10.0              # soft‑max sharpness for lag weighting

    # ----- main optimisation loop -------------------------------------
    while time.monotonic() < deadline:
        # current objective
        cur_max = _conv_max(a)
        cur_score = 2.0 * n * cur_max

        # ---- gradient approximation ---------------------------------
        # full convolution (exact) – cheap enough for n≈2500
        conv = np.convolve(a, a)
        # pick K largest lags (incl. the centre)
        if K >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -K)[-K:]

        # soft‑max weights
        w_raw = np.exp(alpha * (conv[idxs] - cur_max))
        w = w_raw / w_raw.sum() if w_raw.sum() else np.full_like(w_raw, 1.0 / w_raw.size)

        # gradient of max‑lag contribution
        g = np.zeros_like(a)
        for idx, wi in zip(idxs, w):
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            if lo <= hi:
                j = np.arange(lo, hi + 1, dtype=int)
                g[j] += wi * 2.0 * a[idx - j]

        # ----   gradient step + projection ---------------------------
        cand = a - step * g
        cand = _project_simplex(cand)

        cand_score = 2.0 * n * _conv_max(cand)

        # accept if improvement, otherwise occasional uphill step

        if cand_score < cur_score - 1e-14:

            improved = True
        else:

                improved = True

        # ---- keep global best ---------------------------------------
        if cand_score < best_score:
