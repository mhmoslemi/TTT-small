    eps_adam = 1e-8
    lr = 0.2                              # base learning rate

    eps0 = 0.5                            # smoothing for the max
    eps_decay = 0.9995

    best_a = a0.copy()
    # exact score for a simplex vector: 2 n · max(conv)
    best_sc = 2.0 * n * np.max(np.convolve(best_a, best_a))
    no_improve = 0

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----- Soft‑max to get the current probability vector ----------
        max_x = np.max(x)
        y = np.exp(x - max_x)
        a = y / y.sum()                     # always non‑negative, sums to 1

        # ----- Convolution a⋆a (via FFT) -------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- Smoothed max weights ------------------------------------
        eps = eps0 * (eps_decay ** it)
        if eps < 1e-6:
            eps = 1e-6
        w = np.exp((conv - max_val) / eps)  # stable: max subtracted
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- Gradient of the smoothed max w.r.t. a --------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)   # slice producing the needed n values
        g = 2.0 * grad_full[idx]           # ∇_a (smoothed max)

        # ----- Chain rule to get ∇_x (soft‑max parameters) ------------
        dot_ag = np.dot(a, g)
        grad_x = a * (g - dot_ag)          # because ∂a_i/∂x_j = a_i(δ_ij−a_j)

        # ----- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- Score tracking -------------------------------------------
        cur_sc = 2.0 * n * max_val          # sum(a)=1 ⇒ score = 2n·max(conv)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0
            if best_sc < 1.50274349:
                break
        else:
            no_improve += 1

        # ----- Periodic symmetrisation (helps for many instances) -----
        if it % 5000 == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym_fft = np.fft.rfft(a_sym, n=L)
            conv_sym = np.fft.irfft(a_sym_fft * a_sym_fft, n=L)[: 2 * n - 1]
            sc_sym = 2.0 * n * conv_sym.max()
            if sc_sym < best_sc:
                best_sc = sc_sym
                best_a = a_sym.copy()
                # re‑initialise logits to match the symmetric vector
                y = np.exp(np.log(a_sym + 1e-16) - np.max(np.log(a_sym + 1e-16)))
                x = np.log(y / y.sum())
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0

        # ----- Random shake‑up if no improvement for a while ----------
        if no_improve > 10000:
            # Gaussian jitter on logits (small magnitude)
            x += rng.normal(scale=0.02, size=n)
            no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate found as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best_a]
