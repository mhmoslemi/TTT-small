        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size+1) > 0)[0]

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The routine respects the time budget and keeps the best solution found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # spare margin

    # ------------------------------------------------------------------
    # initialise a (use the provided parent if it exists)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a simple random start if no parent is provided
        a = rng.random(2000, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                       # enforce sum = 1

    n = a.shape[0]

    # ------------------------------------------------------------------
    # FFT parameters (powers of two for speed)
    # ------------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())   # length for convolutions
    idx_slice = 2 * n - 2 - np.arange(n)   # indices needed for gradient

    # ------------------------------------------------------------------
    # projection onto the simplex {x >= 0, sum x = 1}
    # ------------------------------------------------------------------
    def _project_simplex(v: np.ndarray) -> np.ndarray:
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    # Adam‑style state
    m = np.zeros_like(a)
    v2 = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    eps = 1e-8
    lr = 0.12                     # base learning rate

    # soft‑max temperature (beta in exp(beta*conv))
    temp = 0.5
    temp_max = 50.0
    temp_mul = 1.0015

    # bookkeeping
    best_vec = a.copy()
    # exact bound (2*n*max(conv) because sum(a)=1)
    fa = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    best_sc = float(2.0 * n * conv_full.max())
    no_improve = 0
    target = 1.50274349

    it = 0
    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # FFT of current vector (re‑used for self‑convolution and gradient)
        fa = np.fft.rfft(a, n=L)

        # self‑convolution (full length 2n‑1)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_c = conv.max()
        cur_sc = float(2.0 * n * max_c)

        # keep the best solution seen so far
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            no_improve = 0
            if best_sc < target:
                break
        else:
            no_improve += 1

        # ------- soft‑max weights for a smooth approximation of max -------
        w = np.exp(temp * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ------- gradient of the soft‑max objective -----------------------
        # g = 2 * (w[::-1] * a) convolution, sliced appropriately
        fw = np.fft.rfft(w[::-1], n=L)
        conv_w_a = np.fft.irfft(fw * fa, n=L)[: 3 * n - 2]
        g = 2.0 * conv_w_a[idx_slice]

        # ------- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v2 = beta2 * v2 + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v2 / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # projected gradient step
        a = a - step
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # increase temperature (sharpen the soft‑max)
        temp = min(temp_max, temp * temp_mul)

        # ------- occasional random mass transfer if stuck ---------------
        if no_improve > 3000 and rng.random() < 0.25:
