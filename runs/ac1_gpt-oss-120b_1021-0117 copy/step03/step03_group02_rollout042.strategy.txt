## Approach and Rationale
We tackle the non-convex optimization problem of minimizing the AC1 upper bound by combining projected gradient descent with an efficient FFT-based convolution for gradient computation. The key bottleneck addressed is the prohibitive computational cost of direct convolutions for large sequence lengths, which would exceed the time budget. By leveraging FFT-based convolution, which reduces complexity from O(n²) to O(n log n), we enable thousands of evaluations within the time allowance. The core idea is:

1. **Projected Gradient Descent**: Start with a normalized version of the given `height_sequence_1` (sum=1, values in [0,1000]). In each iteration, compute gradients using the gradient of the convolution loss function, project the updated sequence back to the feasible set, and evaluate using the authoritative evaluator to ensure correctness.

2. **FFT-based Loss Calculation**: Compute gradients by first approximating the gradient of the max convolution norm. Use FFT for fast convolution evaluation, but compare against the evaluator's full convolution for gradient validation.

3. **Adaptive Step Sizing**: Gradually decrease the step sizes as the algorithm progresses toward convergence to ensure local stability and avoid diverging updates.

4. **Smart Updates**: Only evaluate the candidate if it shows potential improvement to save time on domain matters. 

5. **Fallbacks**: Maintain a best-known candidate and ensure recovery if listings become degenerate or numeral instability occurs.

This approach achieves gains by directly targeting the AC1 bound's gradient with adaptive projection, making steady progress toward solutions with scores below 1.5.

## Mathematical Specification
The central objective is the AC1 upper bound score:

$$score = \frac{2n \cdot \max_{k} |(convolve(a,a))[k]||}{(\sum_i a_i)^2}$$

where convolve(a,a) is the full linear convolution.

The distribution evolves under:
- Constraint: $0 \leq a_i \leq 1000$ for each $i$
- Normalization: $\sum_i a_i = 1$ (achievable by positive scaling without changing the score)

The search minimizes this score using gradient-  projection, with each update based on the gradient of the loss function $\max_{k} (convolve(a,a))[k])$.

def clip_normalize(x):
    """Clip values to [0,1000] and normalize to sum 1.0"""
    x = np.clip(x, 0, 1000)
    s = sum(x)
    if s < 0.01:  # Handle very small sums
        return np.ones_like(x) * 10 if s==0 else np.divide(x, s) * 1000
    return x / s

def standardize(sequence):
    """Convert to normalized form for validation."""
    seq = sequence.copy()
    for i, x in enumerate(seq):
        if isinstance(x, int) or isinstance(x, float):
            seq[i] = float(x)
        else:
            # Fallback to 0 but with careful handling
            pass
    return clip_normalize(seq)

### FFT-based Prediction
```python
def convolve_full(x, padder=None):
    """Compute full convolution using Fast Fourier Transform (FFT)"""
    n_val = len(x)
    conv_length = 2 * n_val - 1
    L = 1 << (conv_length).bit_length()   # Next power of two >= conv_length
    a_arr = np.asarray(clip_normalize(x))
    fa = rfft(a_arr, L)
    conv = irfft(fa * fa, L)[:conv_length]
    return conv

def get_loss_increase(x):
    """Approximate the gradient component of the loss function using a max stability norm."""
    conv = convolve_full(x)  # Local max via FFT for efficiency
    if not np.any(conv) or np.isnan(conv).any():
        return np.ones_like(x)  # Not well defined, Initialization fallback
    idx = np.argmax(conv)
    return convolve_full_np(x, idx)  # Detailed gradient at the max index (implemented below)

def convolve_full_np(x, idx):
    """Compute gradient contribution from the max index via direct convolution"""
    n_val = len(x)
    conv_length = 2 * n_val - 1
    conv = np.zeros(conv_length, dtype=float)
    # Direct convolution specific to index max
    for k in range(conv_length):
        total = 0.0
        for j in range(n_val):
            if k - j < 0 or k - j >= n_val: 
                continue
            total += x[j] * x[k - j]
        conv[k] = total

    # Return the gradient specific to max index
    max_contribution = conv[idx - 0] if idx else conv.max()
    if abs(max_contribution) < 1e-12:
        return np.ones(n_val)
    grad = np.zeros(n_val)
    for i in range(n_val):
        for j in range(n_val):
            doubled_index = i + j
            if doubled_index == idx and doubled_index < conv_length:
                grad[i] += 2 * x[j]   # Derivative contribution
    return grad / (max_contribution if max_contribution!=0 else 10)  # Normalize gradient

def evaluate_sequence_np(sequence):
    """Handler to provide fallback or native evaluation"""
    # Validate with direct mode
    try:
        valid = [float(x) for x in sequence]
    except:
        valid = np.zeros_like(sequence)
    valid = clip_normalize(valid)
    conv_vals = convolve_full_np(valid)
    max_val = max(conv_vals) if len(conv_vals) > 0 else 0.0
    n_val = len(valid)
    sum_val = sum(valid)
    if sum_val < 0.01:
        return np.inf
    return 2.0 * n_val * max_val / (sum_val ** 2)

```

### Core Optimization Loop
```python
def iterate_sequence(initial_sequence, budget_in_seconds=1100):
    x = clip_normalize(initial_sequence)
    best_x = x.copy()
    best_score = evaluate_sequence(x)
    current_score = best_score
    candidate = x.copy()
    step_size = 0.5
    time_limit = time.time() + budget_in_seconds - 5  # Addition buffer

    while time.time() < time_limit:
        try:
            pred = get_loss_increase(x)
            x_out = x - step_size * pred
            candidate = clip_normalize(x_out)
            score_val = evaluate_sequence(candidate)

            if score_val < current_score:
                current_score = score_val
                best_x = candidate.copy()
                best_score = score_val
                step_size *= 0.99  # Reduce steps as optimization advances
            elif time.time() + 0.5 < time_limit - budget_in_seconds/2 and np.random.random() < 0.1:  # Periodic scrambling
                # Random perturbation: promote diversity
                candidate_rand = best_x.copy()
                amplitude = 0.01 * np.ones_like(candidate_rand)
                np.add(candidate_rand, amplitude * np.random.randn(len(candidate_rand)), out=candidate_rand, casting='same')
                candidate_rand = clip_normalize(candidate_rand)
                cand_score = evaluate_sequence(candidate_rand)
                if cand_score < best_score + 0.1:  # Chance improvement
                    best_score = cand_score
                    best_x = candidate_rand.copy()

            time_remaining = time_limit - time.time()
            # Early break if stuck or too slow
            if best_score > 1.4 or time_remaining < 10 and not np.array_equal(candidate, best_x):
                break

        except Exception as e:
            print(f"Evaluation exception: {e}")
            # Fallback if calculation errors relative to the norm
            break

    return [float(x) for x in best_x]
```
