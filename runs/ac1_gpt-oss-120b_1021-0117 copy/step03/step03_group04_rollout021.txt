def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    # Clip negatives, then renormalise; this is a cheap approximation
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def fast_score(x: np.ndarray) -> float:
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    best = None
    best_score = float("inf")
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    a = best.copy()

    # --------------------------------------------------------------
    # 4. Optimisation parameters
    # --------------------------------------------------------------
    it = 0
    eta = 0.5          # initial step magnitude
    decay = 0.9995     # per‑iteration decay
    tol = 0.005        # 0.5 % tolerance for “active” shifts
    mu = 0.9           # momentum factor
    momentum = np.zeros_like(a)
    target = 1.50274349

    # --------------------------------------------------------------
    # 5. Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution and current score
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            if best_score <= target:
                break

        # active shifts (within tolerance of the maximum)
        thresh = cur_max * (1.0 - tol)
        active = np.where(conv >= thresh)[0]

        # gradient from active shifts
        g = np.zeros_like(a)
        if active.size:
            # accumulate contributions from each active shift
            for sh in active:
                offset = sh - np.arange(n)
                ok = (offset >= 0) & (offset < n)
                g[ok] += a[offset[ok]]
            g = 2.0 * g / active.size
        else:
            # fallback: use the single maximal shift
            sh = int(np.argmax(conv))
            offset = sh - np.arange(n)
            ok = (offset >= 0) & (offset < n)
            g[ok] = 2.0 * a[offset[ok]]

        # momentum update
        momentum = mu * momentum + (1.0 - mu) * g

        # step‑size schedule
        eta *= decay

        # gradient step & projection onto simplex
        a = _proj_simplex(a - eta * momentum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _proj_simplex(a * jitter)

    # --------------------------------------------------------------
    # 6. Return the best sequence as plain Python floats
    # --------------------------------------------------------------
    return [float(v) for v in best]
