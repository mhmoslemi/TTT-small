def _grad_smooth(a, tau):
    # a : ndarray length n, sum = 1
    conv = _conv_fft(a)  # length 2n-1
    exp_vec = np.exp(conv / tau)
    w = exp_vec / exp_vec.sum()

import time

# --------------------------------------------------------------
# Helper utilities (simple, matrix‑free)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via real FFT (no zero‑padding beyond power‑of‑2)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    # convolution with itself: multiply spectra
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (sum is forced to 1 in our usage)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())

    a = np.power(r, np.arange(n, dtype=np.float64))
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)

    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)

    s = a.sum()
    if s == 0.0:

    else:

    return a
