def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    from math import exp

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2) # keep some margin

    rng = np.random.default_rng(seed)

    # Determine n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        # ensure nonnegative + normalize
        base = np.maximum(base, 0.0)
        s = base.sum()
        if s > 0:
            base /= s
        else:
            base = np.full_like(base, 1.0 / n)
    else:
        n = 256
        base = np.full(n, 1.0 / n)

    # define helper functions
    def _project_simplex(v):
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    def _conv_fft(a):
        n_loc = a.shape[0]
        L = 1 << ((2*n_loc - 1).bit_length())
        fv = np.fft.rfft(a, n=L)
        return np.fft.irfft(fv * fv, n=L)[: 2*n_loc - 1]

    def _score(a):
        n_loc = a.shape[0]
        conv = _conv_fft(a)
        return float(2.0 * n_loc * conv.max())

    def _make_exponential(r):
        a = r ** np.arange(n, dtype=np.float64)
        a_sum = a.sum()
        return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)

    def _make_powerlaw(p):
        a = 1.0 / np.power(np.arange(1, n+1, dtype=np.float64), p)
        a_sum = a.sum()
        return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)

    def _make_linear():
        a = np.arange(n, 0, -1, dtype=np.float64)
        a_sum = a.sum()
        return a / a_sum

    def _make_quadratic():
        a = np.square(np.arange(n, 0, -1, dtype=np.float64))
        a_sum = a.sum()
        return a / a_sum

    def _make_sqrt_series():
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for k in range(1, n):
            if k == 1:
                s = 0.0
            else:
                s = float(np.dot(a[1:k], a[k-1:0:-1]))
            a[k] = (1.0 - s) / 2.0
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            return np.full(n, 1.0 / n)
        return a / s

    def _make_edge_power(p):
        idx = np.arange(1, n+1, dtype=np.float64)
        prod = idx * (n + 1 - idx)  # (i+1)*(n-i)
        a = prod ** (-p)
        a = np.maximum(a, 0.0)
        s = a.sum()
        return a / s if s > 0 else np.full(n, 1.0 / n)

    def _make_beta(alpha, beta):
        idx = np.arange(1, n+1, dtype=np.float64)
        rev = np.arange(n, 0, -1, dtype=np.float64)
        a = np.power(idx, alpha - 1.0) * np.power(rev, beta - 1.0)
        a = np.maximum(a, 0.0)
        s = a.sum()
        return a / s if s > 0 else np.full(n, 1.0 / n)

    # start with best from families
    candidates = []

    # uniform baseline (if parent not better)
    candidates.append(base.copy())

    # exponential grid
    for r in np.linspace(0.90, 0.99995, 25):
        candidates.append(_make_exponential(r))

    # powerlaw grid
    for p in np.linspace(0.2, 4.0, 25):
        candidates.append(_make_powerlaw(p))

    # linear, quadratic, sqrt_series
    candidates.append(_make_linear())
    candidates.append(_make_quadratic())
    candidates.append(_make_sqrt_series())

    # edge-power grid
    for p in np.linspace(0.1, 5.0, 20):
        candidates.append(_make_edge_power(p))

    # beta grid
    alphas = np.linspace(0.3, 2.5, 12)
    betas = np.linspace(0.3, 2.5, 12)
    for a_alpha in alphas:
        for b_beta in betas:
            candidates.append(_make_beta(a_alpha, b_beta))

    best_vec = None
    best_sc = float('inf')
    for cand in candidates:
        cand = np.maximum(cand, 0.0)
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()
        if time.monotonic() >= deadline:
            break

    # Refinement loop (exponentiated + projected + annealing)
    cur = best_vec.copy()
    cur_sc = best_sc
    cur_conv = _conv_fft(cur)

    target = 1.50274349
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # active lag
        k = int(np.argmax(cur_conv))

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]

        # exponentiated step
        eta = 0.6 / np.sqrt(it)
        cand1 = cur * np.exp(-eta * g)
        cand1_sum = cand1.sum()
        if cand1_sum > 0:
            cand1 = cand1 / cand1_sum
        else:
            cand1 = np.full_like(cur, 1.0 / n)

        sc1 = _score(cand1)

        # Euclidean projection step
        alpha = 0.8 / np.sqrt(it)
        cand2 = cur - alpha * g
        cand2 = _project_simplex(cand2)
        sc2 = _score(cand2)

        # Choose best among cur, cand1, cand2
        if sc1 < cur_sc or sc2 < cur_sc:
            if sc1 <= sc2 and sc1 < cur_sc:
                cur = cand1
                cur_sc = sc1
                cur_conv = _conv_fft(cur)
            elif sc2 < cur_sc:
                cur = cand2
                cur_sc = sc2
                cur_conv = _conv_fft(cur)

            if cur_sc < best_sc:
                best_sc = cur_sc
                best_vec = cur.copy()
                if best_sc < target:
                    break

        # occasional random mass transfer (simulated annealing)
        if rng.random() < 0.02:
            i = int(rng.choice(np.where(cur > 1e-12)[0]))
            j = int(rng.integers(0, n))
            if i == j:
                j = (j + 1) % n
            max_delta = cur[i] * 0.5
            if max_delta > 1e-12:
                delta = max_delta * rng.random()
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)
                cand_sc = _score(cand)
                # accept with annealing probability
                if cand_sc < cur_sc:
                    cur = cand
                    cur_sc = cand_sc
                    cur_conv = _conv_fft(cur)
                else:
                    # temperature decreasing linearly
                    temp = max(0.001, (deadline - time.monotonic()) / (deadline - start))
                    if rng.random() < exp((cur_sc - cand_sc) / temp):
                        cur = cand
                        cur_sc = cand_sc
                        cur_conv = _conv_fft(cur)

                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best_vec = cur.copy()
                    if best_sc < target:
                        break

        # small jitter every few thousand iterations
        if it % 4000 == 0:
            jitter = rng.normal(scale=5e-5, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp_sum = tmp.sum()
            if tmp_sum > 0:
                tmp = tmp / tmp_sum
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur = tmp
                cur_sc = tmp_sc
                cur_conv = _conv_fft(tmp)
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best_vec = cur.copy()
                    if best_sc < target:
                        break

        # update cur_conv for next iteration (if not updated)
        if cur_sc == cur_sc:  # always
            cur_conv = _conv_fft(cur)

    if best_vec is None or best_vec.size == 0:
        return [1.0]  # fallback

    return [float(v) for v in best_vec]

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # next power of two
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective employed by the evaluator (sum not forced to 1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _make_exponential(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_linear(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _make_quadratic(n: int) -> np.ndarray:
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()


def _make_sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_edge_power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n + 1 - idx)          # (i+1)*(n-i)
    a = prod ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    rev = np.arange(n, 0, -1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(rev, beta - 1.0)
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _make_inverse_sum(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / i ** p + 1.0 / (n + 1 - i) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)
