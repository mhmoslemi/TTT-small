    rows = 2*n - 1
    indptr = np.empty(rows+1, dtype=np.int32)
    col_inds = np.empty(n*n, dtype=np.int32)

    rng = np.random.default_rng(seed)

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.size
    # precompute sparse convolution structure
    rows = 2*n - 1
    indptr = np.empty(rows+1, dtype=np.int32)
    col_inds = np.empty(n*n, dtype=np.int32)
    row_idx = np.empty(n*n, dtype=np.int32)
    pos = 0
    for k in range(rows):
        lo = max(0, k - (n-1))
        hi = min(k, n-1)
        cnt = hi - lo + 1
        indptr[k] = pos
        col_inds[pos:pos+cnt] = np.arange(lo, hi+1, dtype=np.int32)
        row_idx[pos:pos+cnt] = k
        pos += cnt
    indptr[rows] = pos
    idx = row_idx - col_inds   # constant int32 array

    def max_conv(v):
        L = 1 << ((2*n - 1).bit_length())
        ft = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(ft * ft, n=L)[:2*n-1]

    best = a.copy()

    # ensure low bound initial

        cur_score = 2.0 * n * M
        if cur_score < best_score:

import time, math, numpy as np, scipy.sparse as sp, scipy.optimize as opt

# ---------- helpers ---------------------------------------------------------

def _max_autocorr(v: np.ndarray) -> float:
    """Maximum value of the full convolution v*v (FFT, linear time)."""
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    return float(conv.max())


def _build_conv_structure(n: int):
    """Return (rows, indptr, col_inds, idx) describing the n×n Toeplitz
    matrix T where (T @ g)[k] = sum_i f[i] * g[k-i] for a given f."""
    rows = 2 * n - 1
    indptr = np.empty(rows + 1, dtype=np.int32)
    col_inds = np.empty(n * n, dtype=np.int32)   # exactly n² non‑zeros
    row_idx = np.empty(n * n, dtype=np.int32)

    pos = 0
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        cnt = hi - lo + 1
        indptr[k] = pos
        col_inds[pos:pos + cnt] = np.arange(lo, hi + 1, dtype=np.int32)
        row_idx[pos:pos + cnt] = k
        pos += cnt
    indptr[rows] = pos
    idx = row_idx - col_inds               # constant index mapping into f
    return rows, indptr, col_inds, idx


# ---------- main entry point ----------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # ---- initial vector ----------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                                            # put on simplex

    n = a.size
    rows, indptr, col_inds, idx = _build_conv_structure(n)

    # ---- bookkeeping -------------------------------------------------------
    best = a.copy()
    best_score = 2.0 * n * _max_autocorr(a)                 # exact score = 2n·max/1²
    target = 1.50274349
    it = 0
    max_lp_iters = 12                                        # enough for convergence

    # ---- main LP‑based improvement loop ------------------------------------
    while time.monotonic() < deadline and it < max_lp_iters:
        it += 1
        M = _max_autocorr(a)
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score, best = cur_score, a.copy()
            if best_score <= target:
                break

        # matrix for the constraints T_f · g ≤ M
        data = a[idx]                                        # f[row‑col] for every non‑zero
        T = sp.csr_matrix((data, col_inds, indptr), shape=(rows, n))

        # linear program: max 1ᵀg  ⇔  min −1ᵀg
        c = -np.ones(n, dtype=np.float64)
        ub = np.full(rows, M, dtype=np.float64)
        bounds = [(0, None)] * n
        remaining = deadline - time.monotonic()
        if remaining <= 0.1:
            break

        # highs respects a time_limit option (newer SciPy); ignore if unavailable
        try:

                                 method='highs',

                                 method='highs')

        if not lp_res.success:

        if sum_g <= 0.0:

        t = 0.4 / math.sqrt(it)

        a = np.maximum(a, 0.0)

    while time.monotonic() < deadline:

            a = cand
            if best_score <= target:
                break
