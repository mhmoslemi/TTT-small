import time, numpy as np

def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv*fv, n=L)[:2*n-1]

def _conv_fft_pair(x, y):
    L = 1 << ((len(x) + len(y) - 1).bit_length())
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    return np.fft.irfft(fx * fy, n=L)[:len(x)+len(y)-1]

def _score(a):
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum()**2))

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    return np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x*y using FFT (real input)."""
    L = 1 << ((len(x) + len(y) - 1).bit_length())
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    return np.fft.irfft(fx * fy, n=L)[: len(x) + len(y) - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (no forced normalisation)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # initialise from parent if available, otherwise a simple fallback
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 256
        cur = np.full(n0, 1.0 / n0, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    n = cur.size

    # --------------------------------------------------------------
    # quick search over a few analytic families (exponential, power‑law,
    # linear) to obtain a decent starting point
    # --------------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    for r in np.linspace(0.90, 0.9999, 40):
        a = np.power(r, np.arange(n, dtype=np.float64))
        a /= a.sum()
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    for p in np.linspace(0.1, 5.0, 40):
        a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        a /= a.sum()
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    sc = _score(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    # --------------------------------------------------------------
    # optimisation phases
    # --------------------------------------------------------------
    total_time = deadline - start
    phase1_end = start + 0.20 * total_time          # active‑lag EG first 20 %
    it = 0
    eta0 = 0.5                                      # step‑size for phase 1
    alpha = 1.0                                     # smoothness for phase 2
    alpha_max = 80.0
    alpha_inc = (alpha_max - alpha) / 80000.0      # slow increase
    lr0 = 0.2                                       # base learning‑rate for phase 2
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # ------------------------------------------------------------------
        # common data: full convolution, maximal index
        # ------------------------------------------------------------------
        conv = _conv_fft(cur)
        k = int(np.argmax(conv))

        # ------------------------------------------------------------------
        # Phase 1 – sub‑gradient on the active lag (first 20 % of time)
        # ------------------------------------------------------------------
        if time.monotonic() < phase1_end:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g = np.zeros_like(cur)
            g[idx] = 2.0 * cur[k - idx]

            eta = eta0 / np.sqrt(it)
            cand = cur * np.exp(-eta * g)
            cand_sum = cand.sum()
            cand = cand / cand_sum if cand_sum > 0 else np.full_like(cur, 1.0 / n)

        # ------------------------------------------------------------------
        # Phase 2 – smoothed‑max gradient descent (remainder of the budget)
        # ------------------------------------------------------------------
        else:
            maxc = conv.max()
            shifted = conv - maxc                # max becomes zero
            w = np.exp(alpha * shifted)          # soft‑max weights
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient = 2 * (w ⋆ cur)  (cross‑correlation)
            g = 2.0 * _conv_fft_pair(w, cur[::-1])[n - 1:n - 1 + n]

            lr = lr0 / np.sqrt(it)
            cand = cur * np.exp(-lr * g)
            cand_sum = cand.sum()
            cand = cand / cand_sum if cand_sum > 0 else np.full_like(cur, 1.0 / n)

            # slowly sharpen the soft‑max
            if alpha < alpha_max:
                alpha = min(alpha_max, alpha + alpha_inc)

        # ------------------------------------------------------------------
        # evaluate and possibly accept the candidate
        # ------------------------------------------------------------------
        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break
        else:
            # occasional uphill move to escape shallow traps
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
                    if best_sc < target:
                        break

        # ------------------------------------------------------------------
        # occasional small random jitter (helps exploration)
        # ------------------------------------------------------------------
        if it % 3000 == 0:
            jitter = rng.normal(scale=5e-5, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp_sum = tmp.sum()
            tmp = tmp / tmp_sum if tmp_sum > 0 else np.full_like(tmp, 1.0 / n)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    if best_sc < target:
                        break

        # final early‑stop guard
        if time.monotonic() >= deadline:
            break

    # ----------------------------------------------------------------------
    # safety net – fallback to uniform if something went wrong
    # ----------------------------------------------------------------------
    if best_sc == float("inf") or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
