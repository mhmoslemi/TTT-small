def _project_simplex(v):
    # v is 1D array.
    # ensure sum(v) = 1 after projection onto nonnegative simplex.
    # Implementation from Duchi.
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Convolution (full) and its maximum, via FFT.
# ------------------------------------------------------------
def _conv_full_max(a: np.ndarray, L: int):
    """Return (max(convolution), full convolution) of a with itself."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv


# ------------------------------------------------------------
# Subgradient of the max‑convolution (with respect to a).
# ------------------------------------------------------------
def _subgradient_max(conv: np.ndarray, a: np.ndarray, tol: float):
    """Subgradient of max(conv) w.r.t. a (a is non‑negative, sum=1)."""
    M = conv.max()
    idxs = np.where(conv >= M - tol)[0]
    n = a.shape[0]
    j = np.arange(n)
    g = np.zeros_like(a)
    for k in idxs:
        # term a[k-j] where indices are valid
        mask1 = (k - j >= 0) & (k - j < n)
        if mask1.any():
            g[j[mask1]] += a[k - j[mask1]]
        # term a[j-k] where indices are valid
        mask2 = (j - k >= 0) & (j - k < n)
        if mask2.any():
            g[j[mask2]] += a[j[mask2] - k]
    return g


# ------------------------------------------------------------
# Optional linear‑programming improvement step (matrix‑free).
# ------------------------------------------------------------
def _lp_mix_step(a: np.ndarray, max_val: float, active_idx: np.ndarray, mix: float = 0.2):
    """Solve a tiny LP to find a vector b with larger ℓ₁‑norm under the same bound,
    then return the mixture (1‑mix)*a + mix*b projected onto the simplex.
    Returns None if cvxpy is unavailable or the solve fails."""
    try:
        import cvxpy as cp
    except Exception:
        return None

    n = a.shape[0]
    b = cp.Variable(n, nonneg=True)
    constraints = []
    # build a linear constraint for each active lag
    for k in active_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        a_slice = a[lo:hi + 1]
        # b indices in the reverse order to line up with a_slice
        b_idx = np.arange(k - lo, k - hi - 1, -1)  # descending
        expr = cp.sum(cp.multiply(a_slice, b[b_idx]))
        constraints.append(expr <= max_val)

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
        if b.value is None:
            raise RuntimeError
    except Exception:
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=2000)
            if b.value is None:
                raise RuntimeError
        except Exception:
            return None

    b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    mixed = (1.0 - mix) * a + mix * b_val
    return _project_simplex(mixed)


# ------------------------------------------------------------
# Simple fallback: analytic sqrt series (used only if no parent).
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Main entry point required by the harness.
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)  # safety margin

    # -----------------------------------------------------------------
    # initialise a (use parent if present, else a compact analytic series)
    # -----------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = _sqrt_series(256)

    if a.size == 0:
        a = np.ones(1, dtype=np.float64)

    # normalise to the simplex (sum = 1)
    a = _project_simplex(a / (a.sum() if a.sum() != 0 else 1.0))

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())  # next power of two for FFT

    # --------------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------------
    max_val, _ = _conv_full_max(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    step = 0.2
    step_decay = 0.9996
    min_step = 1e-7
    beta = 0.8                     # momentum coefficient
    momentum = np.zeros_like(a)

    no_improve = 0
    max_no_improve = 400
    prob_random = 0.015
    lp_interval = 120
    iteration = 0

    while time.monotonic() < deadline:
        # ----------------------------------------------------------
        # evaluate current convolution and gradient
        # ----------------------------------------------------------
        max_val, conv = _conv_full_max(a, L)
        cur_score = 2.0 * n * max_val

        # subgradient of max‑convolution
        tol = max_val * 1e-12 + 1e-15
        g = _subgradient_max(conv, a, tol)

        # ----------------------------------------------------------
        # momentum step + projection
        # ----------------------------------------------------------
        momentum = beta * momentum - step * g
        cand = a + momentum
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_max, _ = _conv_full_max(cand, L)
        cand_score = 2.0 * n * cand_max

        improved = False
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            improved = True
        else:
            # occasional uphill acceptance
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-12)):
                a = cand
                cur_score = cand_score
                improved = True

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            if not improved:
                no_improve += 1
            else:
                no_improve = 0

        # ----------------------------------------------------------
        # occasional random mass‑transfer perturbation
        # ----------------------------------------------------------
        if rng.random() < prob_random:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.random() * 0.25
                delta = a[i] * frac
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)

        # ----------------------------------------------------------
        # optional LP‑based mixing step (sparse, cheap)
        # ----------------------------------------------------------
        if iteration % lp_interval == 0:
            # active set: all lags attaining the maximum (within tolerance)
            rel_tol = max_val * 1e-10 + 1e-15
            active = np.where(conv >= max_val - rel_tol)[0]
            if active.size and active.size < 10:  # keep the LP tiny
                mixed = _lp_mix_step(a, max_val, active, mix=0.2)
                if mixed is not None:
                    mixed_max, _ = _conv_full_max(mixed, L)
                    mixed_score = 2.0 * n * mixed_max
                    if mixed_score < cur_score - 1e-12:
                        a = mixed
                        cur_score = mixed_score
                        improved = True
                        if mixed_score < best_score:
                            best_score = mixed_score
                            best = mixed.copy()
                            no_improve = 0

        # ----------------------------------------------------------
        # step size decay
        # ----------------------------------------------------------
        step = max(min_step, step * step_decay)

        # ----------------------------------------------------------
        # restart when stuck
        # ----------------------------------------------------------
        if no_improve >= max_no_improve:
            a = rng.random(n)
            a = _project_simplex(a)
            momentum = np.zeros_like(a)
            step = 0.2
            no_improve = 0

        iteration += 1

        # quick safety check
        if time.monotonic() + 0.4 > deadline:
            break

    # return best candidate as a plain Python list
    return [float(x) for x in best]
