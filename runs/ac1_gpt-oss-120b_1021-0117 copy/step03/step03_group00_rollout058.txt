def project_simplex(v):
    v = np.asarray(v)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w
```

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _grad_smooth_max(a: np.ndarray, eps: float, L: int):
    """Return gradient of the smooth max of a*a and the full convolution."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)

    idx = (2 * n - 2) - np.arange(n)            # slice giving dL/d a
    g = 2.0 * grad_full[idx]
    return g, conv, maxc

    it = 0

    # --------------------------------------------------------
    # Optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # gradient of smooth max
        g, conv, maxc = _grad_smooth_max(a, eps, L)
        cur_score = 2.0 * n * maxc / (a.sum() ** 2)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = a - step
        cand = _project_simplex(cand)
        # keep symmetry
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        # evaluate candidate
        af_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(af_c * af_c, n=L)[: 2 * n - 1]
        cand_score = 2.0 * n * conv_c.max() / (cand.sum() ** 2)

        # accept if better, otherwise probabilistic Metropolis step
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            no_improve = 0
        else:
            prob = math.exp(-(cand_score - cur_score) / (lr + 1e-12))
            if rng.random() < prob:
                a = cand
                cur_score = cand_score
            no_improve += 1

        # update best ever
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0

        # decay smoothing factor
        eps = max(eps_min, eps * eps_decay)

        # occasional random mass transfer if stuck
        if no_improve > 8000:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                af_tmp = np.fft.rfft(tmp, n=L)
                conv_tmp = np.fft.irfft(af_tmp * af_tmp, n=L)[: 2 * n - 1]
                tmp_score = 2.0 * n * conv_tmp.max() / (tmp.sum() ** 2)
                if tmp_score < best_score:
                    best_score = tmp_score
                    best = tmp.copy()
                    a = tmp
                    cur_score = tmp_score
                    no_improve = 0
                else:
                    no_improve = 0

        # early exit if target reached
        if best_score < 1.50274349:
            break

    # --------------------------------------------------------
    # Return result as plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best]
