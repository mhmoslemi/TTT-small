from scipy import sparse
import cvxpy as cp

def solve_lp_max_sum(f, M):
    n = len(f)
    # Build sparse matrix A
    # Row count = 2*n - 1
    rows = []
    cols = []
    data = []
    for i in range(n):
        # f_i contributes to rows i through i + n - 1
        start = i
        end = i + n  # exclusive
        rows.extend(range(start, end))
        cols.extend([i] * n)
        data.extend([f[i]] * n)
    # Build csr
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # variable
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    return g.value

import time
import math
import numpy as np
import cvxpy as cp

# ------------------------------------------------------------
# Helper: max–convolution value and AC1 score (sum = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int = None) -> float:
    """2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Linear program: maximise sum(g) subject to f∗g ≤ M
# ------------------------------------------------------------
def _lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """Return a feasible g (sum(g)=1 after rescaling) or None."""
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    conv_fg = cp.conv(f, g)               # length 2n‑1, linear in g
    prob = cp.Problem(cp.Maximize(cp.sum(g)),
                      [conv_fg <= M])
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=2000, abstol=1e-8,
                   reltol=1e-6, feastol=1e-8)
    except Exception:
        return None
    if g.value is None:
        return None
    gval = np.asarray(g.value, dtype=np.float64)
    gval[gval < 0] = 0.0
    return gval

# Main entry point

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # simple uniform fallback
        cur = np.full(1024, 1.0, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)

    n = cur.shape[0]
    L_fft = 1 << ((3 * n - 2).bit_length())   # safe length for all convolutions

    best = cur.copy()
    best_sc = _score(best, L_fft)

    # mixing parameter – starts aggressive, then dampens
    mix = 0.5
    it = 0

    # ---- optimisation loop -----------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current max‑convolution
        cur_fft = np.fft.rfft(cur, n=L_fft)
        conv_cur = np.fft.irfft(cur_fft * cur_fft, n=L_fft)[: 2 * n - 1]
        M = conv_cur.max()

        # linear‑program step
        g = _lp_max_sum(cur, M)
        if g is None:
            break                     # give up if LP fails

        # rescale g to match the simplex
        g_sum = g.sum()
        if g_sum <= 0:
            g = np.full_like(g, 1.0 / n)
        else:
            g = g / g_sum

        # convex combination and renormalisation
        cur = (1.0 - mix) * cur + mix * g
        cur = np.maximum(cur, 0.0)
        cur_sum = cur.sum()
