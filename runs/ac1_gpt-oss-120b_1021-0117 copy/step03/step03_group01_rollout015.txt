import time
import math
import numpy as np

# ---------------------------------------------------------
# Simplex projection (L1=1, non‑negative)
# ---------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:               # extremely unlikely, but safe
        w[:] = 1.0 / n
    return w

# ---------------------------------------------------------
# Full convolution via FFT (real, length 2n‑1)
# ---------------------------------------------------------
def _full_conv(v: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[: 2 * v.size - 1]

# ---------------------------------------------------------
# Max value of convolution and all indices attaining it
# ---------------------------------------------------------
def _conv_max_and_positions(v: np.ndarray, L: int):
    c = _full_conv(v, L)
    mx = float(c.max())
    # tolerate tiny numerical jitter
    pos = np.nonzero(c >= mx - 1e-12)[0]
    return mx, pos

# ---------------------------------------------------------
# Gradient of the max‑convolution (sub‑gradient using all
# active positions).  For each active shift k,
# ∂(a*a)_k/∂a_j = a_{k-j} + a_j  (if the index exists).
# ---------------------------------------------------------
def _gradient_from_max(v: np.ndarray, max_pos: np.ndarray) -> np.ndarray:
    n = v.size
    g = np.zeros_like(v)
    for k in max_pos:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        j = np.arange(lo, hi + 1, dtype=np.int64)
        g[j] += v[k - j] + v[j]
    return g

# random mass‑transfer moves and a small smoothing step.

    n = a.size

    cur_score = 2.0 * n * cur_max
    best = a.copy()
