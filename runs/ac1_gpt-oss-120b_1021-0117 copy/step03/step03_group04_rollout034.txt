import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Gradient‑based optimizer for the AC1 upper bound.
    Uses a smooth‑max (soft‑max) surrogate and a projected momentum update.
    Returns the best coefficient list found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2. Determine problem size and base candidates
    # ------------------------------------------------------------------
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_raw.ndim != 1 or parent_raw.size == 0:
            raise ValueError
        parent_raw = np.clip(parent_raw, 0.0, 1000.0)
    except Exception:
        parent_raw = None

    if parent_raw is not None:
        n = int(parent_raw.size)
    else:
        n = 256                     # fallback length (should not happen)

    # Normalise all candidates to have sum = 1 (the evaluator is scale‑invariant)
    def _norm(x):
        s = x.sum()
        return x / s if s > 0 else np.full_like(x, 1.0 / n)

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    # Tent shape (triangular)
    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1.0, n - idx) - 1.0
    tent = np.maximum(tent, 0.0)
    tent = _norm(tent)
    # Random Dirichlet‑like start
    rand = rng.random(n)
    rand = _norm(rand)

    candidates = [uniform, tent, rand]
    if parent_raw is not None:
        a_parent = _norm(np.maximum(parent_raw, 0.0))
        candidates.append(a_parent)

    # ------------------------------------------------------------------
    # 3. Fast evaluation (FFT based)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    # length sufficient for a*a and a*w (3n‑2)
    fft_len = 1 << ((3 * n - 2 - 1).bit_length())

    def fast_score(x: np.ndarray) -> float:
        fa = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    # Choose initial best candidate
    best = None
    best_score = float("inf")
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    a = best.copy()

    # ------------------------------------------------------------------
    # 4. Optimisation parameters
    # ------------------------------------------------------------------
    mu = 0.9                       # momentum decay
    v = np.zeros_like(a)           # momentum buffer
    it = 0
    beta = 1.0                     # soft‑max sharpness
    target = 1.50274349
    eta0 = 2.0                     # initial step magnitude

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of a with itself -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            if best_score <= target:
                break

        # ----- smooth‑max weights (soft‑max) -----
        maxc = conv.max()
        # avoid overflow, shift by max
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # ----- gradient of smooth‑max (matrix‑free) -----
        w_rev = w[::-1]                     # reverse to obtain Toeplitz action
        fw = np.fft.rfft(w_rev, n=fft_len)
        g_full = np.fft.irfft(fa * fw, n=fft_len)  # length fft_len
        # slice that corresponds to indices n‑1 … 2n‑2 (inclusive)
        slice_h = g_full[n - 1 : 2 * n - 1]   # length n

        # ----- momentum update -----
        v = mu * v + (1.0 - mu) * g

        # ----- step size (diminishing) -----
        eta = eta0 / np.sqrt(it)

        a = _proj_simplex(a - eta * v)

            a = _proj_simplex(a * jitter)

    # ------------------------------------------------------------------
    # 6. Return the best sequence (plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
