def _conv_fft(x, L):
    fx = np.fft.rfft(x, n=L)
    return np.fft.irfft(fx*fx, n=L)[:2*x.size-1]

def _score(arr, n, L):
    conv = _conv_fft(arr, L)
    return 2.0 * n * conv.max()

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = np.full(2000, 1.0 / 2000, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s == 0.0:
        cur = np.full_like(cur, 1.0 / cur.size)
    else:
        cur /= s

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    def score_vec(v):
        conv = _conv_fft(v, L)
        return 2.0 * n * conv.max()

    cur_score = score_vec(cur)
    best = cur.copy()
    best_score = cur_score

    it = 0
    # hyperparameters
    alpha0 = 1.0
    jitter_base = 0.005
    eps_move = 0.2
    temp = 0.2 # for occasional uphill acceptance

    while time.monotonic() < deadline:
        it += 1

        # compute convolution and gradient for max index
        conv = _conv_fft(cur, L)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi+1, dtype=np.int64)
        g = np.zeros_like(cur)
        # contributions for subgradient
        # g[idx] = 2 * cur[k - idx] ; note cur[k-idx] is valid
        g[idx] = 2.0 * cur[k - idx]

        eta = alpha0 / math.sqrt(it+1)

        # mirror descent candidate
        mul = np.exp(-eta * g)
        cand = cur * mul
        cand_sum = cand.sum()
        if cand_sum == 0.0 or not np.isfinite(cand_sum):
            cand = np.full_like(cur, 1.0 / n)
        else:
            cand /= cand_sum

        cand_score = score_vec(cand)

        # simple projected gradient candidate
        cand2 = cur - eta * g
        cand2 = np.maximum(cand2, 0.0)
        sum2 = cand2.sum()
        if sum2 == 0.0 or not np.isfinite(sum2):
            cand2 = np.full_like(cur, 1.0 / n)
        else:
            cand2 /= sum2
        cand2_score = score_vec(cand2)

        # choose best among candidates
        if cand2_score < cand_score:
            cand, cand_score = cand2, cand2_score

        # acceptance
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
        else:
            # occasional uphill with Metropolis
            prob = math.exp(-(cand_score - cur_score) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        # update best
        if cur_score < best_score:

                tmp[i] -= delta
                tmp[j] += delta

                tmp_sum = tmp.sum()
                if tmp_sum > 0.0:

                    tmp_score = score_vec(tmp)

import time, math, numpy as np

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of x with itself via real FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    return np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]

def _score_vec(v: np.ndarray, n: int, L: int) -> float:
    """Score = 2 * n * max(convolution) (sum(v) = 1 is assumed)."""
    conv = _conv_fft(v, L)
    return 2.0 * n * conv.max()

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step function with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Initialise – use provided parent if present, otherwise a uniform guess.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = np.full(2000, 1.0 / 2000, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s == 0.0:
        cur = np.full_like(cur, 1.0 / cur.size)
    else:
        cur /= s                       # bring onto the probability simplex

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two length for FFT

    cur_score = _score_vec(cur, n, L)
    best = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Hyper‑parameters (tuned empirically)
    # ------------------------------------------------------------------
    it = 0
    alpha0 = 0.9          # initial step multiplier for exponential update
    eps_move = 0.25       # max fraction of mass shuffled in a random move
    jitter_base = 0.006   # base scale for Gaussian jitter
    temp = 0.15           # temperature for occasional uphill steps
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # ---- sub‑gradient from the current maximal convolution entry ----
        conv = _conv_fft(cur, L)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(cur)

        # ---- step size (decays like 1/√it) ----
        eta = (alpha0 if it < 5000 else 0.25) / math.sqrt(it + 1)
