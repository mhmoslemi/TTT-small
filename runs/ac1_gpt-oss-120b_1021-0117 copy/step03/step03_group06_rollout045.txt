                if s_tmp > 0.0:
                    tmp = tmp / s_tmp
                    fx_t = np.fft.rfft(tmp, n=L)
                    conv_t = np.fft.irfft(fx_t * fx_t, n=L)[: 2 * n - 1]
                    tmp_sc = 2.0 * n * conv_t.max() / (tmp.sum() ** 2)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best = tmp_sc, tmp.copy()

        # ---- jittered restart from the best point -------------------------
        if it % 2500 == 0:
            jitter = rng.normal(scale=jitter_scale, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            s_tmp = tmp.sum()
            if s_tmp > 0.0:
                tmp = tmp / s_tmp
                fx_t = np.fft.rfft(tmp, n=L)
                conv_t = np.fft.irfft(fx_t * fx_t, n=L)[: 2 * n - 1]
                tmp_sc = 2.0 * n * conv_t.max() / (tmp.sum() ** 2)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best = tmp_sc, tmp.copy()
                no_improve = 0

        # ---- long stagnation restart ---------------------------------------
        if no_improve > 10000:
            cur = best + rng.normal(scale=jitter_scale * 5, size=n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            fx = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fx * fx, n=L)[: 2 * n - 1]
            cur_sc = 2.0 * n * conv.max() / (cur.sum() ** 2)
            no_improve = 0

        # ---- early stop ----------------------------------------------------
        if best_sc < target:
            break

    return [float(x) for x in best]
