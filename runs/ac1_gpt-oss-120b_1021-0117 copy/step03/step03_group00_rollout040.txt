import time
import math
import numpy as np
import scipy.sparse as sp

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a vector assumed to sum to 1 (exact evaluator)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max(conv)  (sum(a) is assumed 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Build sparse matrix for (f * g)[s] <= M constraints
# ------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Return sparse matrix A with A[s, j] = f[s-j] (zero when out of range)."""
    n = f.size
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)

    pos = 0
    for j in range(n):
        rows[pos : pos + n] = np.arange(j, j + n, dtype=np.int32)
        cols[pos : pos + n] = j
        data[pos : pos + n] = f
        pos += n
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
