import time, numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.5)

    # init
    if "height_sequence_1" in globals():
        a0 = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 512
        a0 = rng.random(n0)
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s == 0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s

    n = a0.shape[0]
    # compute FFT length power of two >= 2n-1
    L = 1 << ((2 * n - 1).bit_length())

    def conv_fft(v):
        fv = np.fft.rfft(v, n=L)
        c = np.fft.irfft(fv * fv, n=L)[:2*n-1]
        return c

    def score_from_conv(conv):
        M = conv.max()
        # since sum(v)=1 after projection, denominator=1
        return float(2.0 * n * M)

    def full_score(v):
        conv = conv_fft(v)
        return score_from_conv(conv)

    # simplex projection
    def project_simplex(v):
        # onto nonnegative simplex sum=1
        # algorithm from Duchi et al.
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0
        return w

    cur = a0.copy()
    cur_sc = full_score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # ADAM parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    it = 0
    no_improve = 0
    target = 1.50274349
    K = 64  # number of top conv entries to use for gradient
    # main loop
    while time.monotonic() < deadline:
        it += 1
        conv = conv_fft(cur)
        M = conv.max()
        # active set: entries within tolerance of max
        tol = 1e-12 * max(1.0, M)
        active = np.where(conv >= M - tol)[0]
        if active.size == 0 or active.size > K:
            # fallback to top K entries
            active = np.argpartition(-conv, K-1)[:K]

        # compute subgradient g
        g = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n-1))
            hi = min(k, n-1)
            idx = np.arange(lo, hi+1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]

        # ADAM update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        # step size schedule
        lr = 0.05 / np.sqrt(it)  # decaying
        cand_adam = cur - lr * m_hat / (np.sqrt(v_hat) + eps)
        cand_adam = project_simplex(cand_adam)
        cand_adam_sc = full_score(cand_adam)

        # multiplicative update (exponential weights)
        eta = 0.5 / np.sqrt(it)
        exp_factor = np.exp(-eta * g)
        cand_mul = cur * exp_factor
        s_mul = cand_mul.sum()
        if s_mul == 0:
            cand_mul = np.full_like(cur, 1.0 / n)
        else:
            cand_mul = cand_mul / s_mul
        cand_mul_sc = full_score(cand_mul)

        # pick best candidate among cur, adam, mul
        # also keep possibility of staying current
        # Evaluate cur_sc already
        # Determine best candidate
        best_cand = cur
        best_cand_sc = cur_sc
        if cand_adam_sc < best_cand_sc:
            best_cand = cand_adam
            best_cand_sc = cand_adam_sc
        if cand_mul_sc < best_cand_sc:
            best_cand = cand_mul
            best_cand_sc = cand_mul_sc

        # occasional random transfer move, especially when stuck
        if no_improve > 2000 and rng.random() < 0.3:

            i = rng.integers(0, n)

                if j >= i: j += 1
                delta = cur[i] * rng.random() * 0.3
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = project_simplex(tmp)
                tmp_sc = full_score(tmp)
                if tmp_sc < best_cand_sc:
                    best_cand = tmp
                    best_cand_sc = tmp_sc

            tmp = project_simplex(tmp)
            tmp_sc = full_score(tmp)
            if tmp_sc < best_cand_sc:
                best_cand = tmp
                best_cand_sc = tmp_sc

        # accept best candidate
        if best_cand_sc < cur_sc:
            cur = best_cand
            cur_sc = best_cand_sc
            if cur_sc < best_sc:

            else:

        else:

        # early exit

    # final refinement - random local swaps for remaining time
    refinement_deadline = time.monotonic() + min(0.4, deadline - time.monotonic())

            cur_sc = tmp_sc

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (as a Python list) with a low AC1 bound.
    The routine respects the given time budget and keeps the best solution found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------------
    # initialise
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random positive vector
        a0 = rng.random(1024, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 == 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s0                     # project to simplex (sum = 1)

    n = a0.size
    # length of FFT – next power of two ≥ 2n‑1
    L = 1 << ((2 * n - 1).bit_length())

    # ----------------------------------------------------------------------
    # helper utilities
    # ----------------------------------------------------------------------
    def conv_fft(v: np.ndarray) -> np.ndarray:
        """Self‑convolution via FFT, returns length 2n‑1 array."""
        fv = np.fft.rfft(v, n=L)
        c = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        return c

    def score_from_conv(conv: np.ndarray) -> float:
        """Score assuming sum(v)=1 (the denominator is therefore 1)."""
        return float(2.0 * n * conv.max())

    def full_score(v: np.ndarray) -> float:
        """Compute score from a vector."""
        return score_from_conv(conv_fft(v))

    def project_simplex(v: np.ndarray) -> np.ndarray:
        """Euclidean projection onto {x≥0, sum x = 1}."""
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        return w

    # ----------------------------------------------------------------------
    # optimisation state
    # ----------------------------------------------------------------------
    cur = a0.copy()
    cur_sc = full_score(cur)

    best = cur.copy()
    best_sc = cur_sc

    # ADAM moments
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    pow1 = 1.0
    pow2 = 1.0

    it = 0
    no_improve = 0
    target = 1.50274349
    K = 64                                 # max number of active conv entries

    # ----------------------------------------------------------------------
    # main optimisation loop
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        conv = conv_fft(cur)
        M = conv.max()
        # active indices – within a tiny tolerance of the maximum
        tol = 1e-12 * max(1.0, M)
        active = np.where(conv >= M - tol)[0]
        if active.size == 0 or active.size > K:
            # fallback to the K largest entries
            active = np.argpartition(-conv, K - 1)[:K]

        # subgradient of max‑convolution (convex hull of active gradients)
        g = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]

        # ---- ADAM step ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        pow1 *= beta1
        pow2 *= beta2
        m_hat = m / (1.0 - pow1)
        v_hat = v / (1.0 - pow2)

        lr = 0.07 / np.sqrt(it)            # decaying learning rate
        cand_adam = cur - lr * m_hat / (np.sqrt(v_hat) + eps)
        cand_adam = project_simplex(cand_adam)
        cand_adam_sc = full_score(cand_adam)

        # ---- multiplicative (exp‑weights) step ---------------------------
        eta = 0.5 / np.sqrt(it)
        exp_factor = np.exp(-eta * g)
        cand_mul = cur * exp_factor
        s_mul = cand_mul.sum()
        if s_mul == 0.0:
            cand_mul = np.full_like(cur, 1.0 / n)
        else:
            cand_mul = cand_mul / s_mul
        cand_mul_sc = full_score(cand_mul)

        # ---- select the best among current, ADAM, multiplicative ----------
        best_cand = cur
        best_cand_sc = cur_sc
        if cand_adam_sc < best_cand_sc:
            best_cand, best_cand_sc = cand_adam, cand_adam_sc
        if cand_mul_sc < best_cand_sc:
            best_cand, best_cand_sc = cand_mul, cand_mul_sc

        # ---- occasional mass‑transfer targeting large gradients ----------
        if no_improve > 2000 and rng.random() < 0.3:
            i = int(np.argmax(g))
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.35
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = project_simplex(tmp)
                tmp_sc = full_score(tmp)
                if tmp_sc < best_cand_sc:
                    best_cand, best_cand_sc = tmp, tmp_sc

        # ---- jitter around the best known solution occasionally ----------
        if (it % 5000 == 0) and rng.random() < 0.2:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = project_simplex(tmp)
            tmp_sc = full_score(tmp)
            if tmp_sc < best_cand_sc:
                best_cand, best_cand_sc = tmp, tmp_sc

        # ---- accept the chosen candidate ---------------------------------
        if best_cand_sc < cur_sc:
            cur = best_cand
            cur_sc = best_cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc
                no_improve = 0
            else:
                no_improve += 1
        else:
            # small chance to move anyway (escaping plateaus)
            if rng.random() < 0.01:
                cur = best_cand
                cur_sc = best_cand_sc

        # early exit if the target is already reached
        if best_sc < target:
            break

    # ----------------------------------------------------------------------
    # final local refinement within remaining time
    # ----------------------------------------------------------------------
    refinement_deadline = time.monotonic() + min(0.4, deadline - time.monotonic())
    while time.monotonic() < refinement_deadline:
        i = rng.integers(0, n)
        if cur[i] == 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = cur[i] * rng.random() * 0.45
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = project_simplex(tmp)
        tmp_sc = full_score(tmp)
        if tmp_sc < cur_sc:
            cur = tmp
            cur_sc = tmp_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

    # return the best vector as a plain Python list
    return [float(x) for x in best]
