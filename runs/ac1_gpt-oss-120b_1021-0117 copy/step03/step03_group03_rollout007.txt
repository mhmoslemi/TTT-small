import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    # Based on algorithm by Wang & Carreira-Perpinan (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
