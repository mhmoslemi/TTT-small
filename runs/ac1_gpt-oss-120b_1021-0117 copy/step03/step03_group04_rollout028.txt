    # projection function
    def proj_simplex(v):
        # Implementation returns a vector on simplex (sum=1, >=0)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(1, len(v) + 1)
        cond = u - cssv / ind > 0
        if not np.any(cond):
            return np.full_like(v, 1.0 / len(v))
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
        return w

    # fast_score
    def fast_score(a):
        fa = np.fft.rfft(a, n=fft_len)

        return 2.0 * n * conv.max(), conv

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_conv(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Return the full linear convolution a*a using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def _score_and_conv(a: np.ndarray, n: int, fft_len: int, conv_len: int):
    """Compute the AC1 score (2*n*max(conv)) and the convolution array."""
    conv = _full_conv(a, fft_len, conv_len)
    max_b = conv.max()
    return 2.0 * n * max_b, conv, max_b


def _subgradient(a: np.ndarray, conv: np.ndarray, max_b: float, n: int) -> np.ndarray:
    """Sub‑gradient of 2*n*max(conv) w.r.t. a (averaged over all active shifts)."""
    # indices where the convolution attains its maximum (tolerance for floating error)
    tol = max_b * 1e-12 + 1e-14
    max_idx = np.where(conv >= max_b - tol)[0]
    if max_idx.size == 0:
        return np.zeros_like(a)

    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        t = np.arange(lo, hi + 1, dtype=np.int64)
        g[t] += 2.0 * a[k - t]  # a[k - t] is always in‑range here
    g /= max_idx.size
    return g


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic optimizer for the AC1 upper bound.
    Uses a sub‑gradient descent on the simplex combined with occasional random
    perturbations (jitter / simulated‑annealing style).  Returns the best sequence
    encountered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1. Determine problem size
    # ----------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent_arr.size)
        have_parent = True
    except Exception:
        parent_arr = None
        n = 256                 # fallback length if no parent is present
        have_parent = False

    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())  # next power of two

    # ----------------------------------------------------------------------
    # 2. Helper to evaluate fast AC1 score (sum is 1 → score = 2*n*max(conv))
    # ----------------------------------------------------------------------
    def fast_score(seq):
        sc, _, _ = _score_and_conv(seq, n, fft_len, conv_len)
        return sc

    # ----------------------------------------------------------------------
    # 3. Build a few diverse starting points (all normalized to sum=1)
    # ----------------------------------------------------------------------
    candidates = []

    # Uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Tent (triangular, peak in the middle)
    idx = np.arange(n, dtype=np.float64)
    tent = np.minimum(idx + 1, n - idx) - 1.0
    tent = np.clip(tent, 0.0, None)
    if tent.sum() > 0:
        candidates.append(tent / tent.sum())

    # U‑shaped (large at the edges, small in the centre)
    ushape = 1.0 / (np.arange(1, n + 1, dtype=np.float64)) + \
             1.0 / (np.arange(n, 0, -1, dtype=np.float64))
    ushape = np.clip(ushape, 0.0, None)

    # Random Dirichlet‑like point
    rnd = rng.random(n)
