import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_seq(seq: np.ndarray, fft_len: int, n: int) -> float:
    """Fast score = 2*n*max(conv) for a sequence with sum(seq)=1."""
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Returns a list of non‑negative coefficients (length = parent length or 256)
    that attempts to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length and a starting point (parent if available)
    # ------------------------------------------------------------------
    parent_arr = None
    try:
        parent_arr = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
    except Exception:
        pass

    if parent_arr is not None and parent_arr.size:
        n = int(parent_arr.size)
        base = np.clip(parent_arr, 0.0, 1000.0)
    else:
        n = 256
        base = np.ones(n, dtype=np.float64)

    # Normalise any base to sum = 1 (simplex)
    base = np.maximum(base, 0.0)
    s = base.sum()
    if s < 1e-12:
        base = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        base = base / s

    # ------------------------------------------------------------------
    # Pre‑compute FFT size (next power of two)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    cand_list = []

    # parent (if present)
    cand_list.append(base.copy())

    # uniform

    cand_list.append(quad / quad.sum())

    # exponential family
    for r in np.linspace(0.85, 0.99, 8):

        cand_list.append(exp / exp.sum())

    cand_list.append(sin / sin.sum())

    # raised cosine
    cos = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n, dtype=np.float64) / (n - 1)))
    cand_list.append(cos / cos.sum())

        cand_list.append(rand / rand.sum())

            v[:] = 0.0
            no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
