def propose_candidate(seed=42, budget_s=1100):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.5)

    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: maybe exponential decay.
        n = 2000
        a0 = np.exp(-np.arange(n)/50.0)
    # Nonnegatives and normalize sum=1
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s

    n = a0.size
    L = 1 << ((2*n-1).bit_length())
    def conv_fft(v):
        fv = np.fft.rfft(v, n=L)
        c = np.fft.irfft(fv * fv, n=L)[:2*n-1]
        return c

    def score(v):
        c = conv_fft(v)
        return 2.0 * n * c.max() / (v.sum()**2)

    best_vec = a0.copy()
    best_score = score(best_vec)
    cur = a0.copy()
    cur_score = best_score

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # compute conv
        conv = conv_fft(cur)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # compute subgradient
        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        idx = np.arange(lo, hi+1, dtype=int)
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[max_idx - idx]   # same as described

        # step size schedule
        alpha = 0.5 / np.sqrt(it) # maybe adjust

        cand = cur - alpha * g
        # ensure nonnegative and proj to simplex
        # projecting onto simplex sum=1, non-negative. Use efficient O(n log n) algorithm.
        # Build simple projection function.

        def project_simplex(v):
            # Compute projection onto simplex {x>=0, sum=1}
            # v is 1D array
            # using sorting algorithm
            u = np.sort(v)[::-1]
            css = np.cumsum(u) - 1.0
            rho = np.where(u - css / np.arange(1, v.size+1) > 0)[0]
            if rho.size == 0:
                theta = 0.0
            else:
                rho = rho[-1]
                theta = css[rho] / (rho + 1)
            w = np.maximum(v - theta, 0.0)
            return w

        cand = project_simplex(cand)
        cand_sc = score(cand)

        # optionally also try multiplicative
        # but we can also try candidate = cur * np.exp(-beta * g) normalized
        beta = 0.5 / np.sqrt(it)
        exp_factor = np.exp(-beta * g)
        cand_mul = cur * exp_factor

        else:

        cand_mul_sc = score(cand_mul)

        # pick move with best improvement

# ---------- helper functions (top level) ----------
def _conv_fft(v: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of v with itself using a power‑of‑two FFT."""
    fv = np.fft.rfft(v, n=L)
    c = np.fft.irfft(fv * fv, n=L)[: 2 * v.shape[0] - 1]
    return c


def _score(v: np.ndarray, L: int) -> float:
    """Exact evaluation used by the official scorer (sum‑normalised)."""
    conv = _conv_fft(v, L)
    n = v.shape[0]
    return float(2.0 * n * conv.max() / (v.sum() ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_from_indices(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """
    Sum of sub‑gradients of the convolution entries whose indices are in `idxs`.
    For a given k, the gradient w.r.t. a_j is 2*a_{k-j} when the term exists.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in idxs:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # indices of a that participate in the term (k)
        pos = np.arange(lo, hi + 1, dtype=np.int64)
        # contribution 2*a_{k-pos}
        g[pos] += 2.0 * a[k - pos]
    return g
