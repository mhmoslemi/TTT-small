def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a power‑of‑two FFT."""
    n = x.size + y.size - 1
    L = 1 << ((n).bit_length())
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[:n]
    return conv

    """Self‑convolution v*v (full length)."""
    return _conv_full(v, v)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum(x) = 1}."""
