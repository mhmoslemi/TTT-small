import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:               # guard against all‑zero output
        w[:] = 1.0 / n
    return w

# ------------------------------------------------------------
# Full convolution via real FFT (returns the whole array)
# ------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(a, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]

# ------------------------------------------------------------
# Sub‑gradient from the K largest convolution entries
# ------------------------------------------------------------
def _subgradient_topk(a: np.ndarray, conv: np.ndarray, topk: int) -> np.ndarray:
    n = a.size
    if topk <= 0:
        return np.zeros_like(a)
    idx = np.argpartition(conv, -topk)[-topk:]
    g = np.zeros_like(a)
    for k in idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            g[j] += 2.0 * a[k - j]
    # average contribution of the selected entries
    g *= 1.0 / topk
    return g

# ------------------------------------------------------------

    min_step = 1e-7
    decay = 0.99995

    momentum = np.zeros_like(a)

        n = 256
        a0 = np.zeros(n, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    # ---- build a small set of diverse starts ----
    starts = [a0]

    # random simplexes
    for _ in range(2):
        rand = rng.random(n)
        starts.append(_project_simplex(rand))

    # edge‑heavy shape (large at the ends, tiny in the centre)
    idx = np.arange(n)
    shape = 1.0 / ((idx + 1) * (n - idx) + 1e-12)
    starts.append(_project_simplex(shape))

    best_overall = None
    best_score = float('inf')

    # ---- run optimisation from each start while time permits ----
    for start in starts:
        if time.monotonic() + 5.0 > deadline:
            break
        cand, cand_score = _optimize(start, deadline, rng, topk=7)
        if cand_score < best_score:
            best_score = cand_score
            best_overall = cand

    # ---- fallback (should never happen) ----
    if best_overall is None:
        best_overall = a0

    # return plain Python list
    return [float(x) for x in best_overall]
