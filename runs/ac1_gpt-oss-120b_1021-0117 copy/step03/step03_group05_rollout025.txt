import time, numpy as np, math

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initialization
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a0 = np.ones(n, dtype=np.float64) / n

    # ensure nonneg, normalize sum to 1
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s

    n = a0.size
    # precompute FFT length
    L = 1 << ((2 * n - 1).bit_length())

    def conv_fft(v):
        fv = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[:2 * n - 1]
        return conv

    def score(v):
        # avoid compute using external evaluate_sequence for speed
        c = conv_fft(v)
        return 2.0 * n * c.max() / (v.sum() ** 2)

    def project_simplex(v):
        # Euclidean projection onto simplex {x>=0, sum=1}
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0
        # re-normalize to exactly 1 to avoid numerical drift
        sum_w = w.sum()
        if sum_w > 0:
            w = w / sum_w
        else:
            w = np.full_like(v, 1.0 / v.size)
        return w

    cur = a0.copy()
    cur_sc = score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # algorithm parameters
    it = 0
    no_improve = 0
    # take active set size up to
    K = 30
    rel_thresh = 0.005  # relative threshold for active set near max
    add_alpha0 = 0.8
    mul_eta0 = 0.8
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # compute convolution
        conv = conv_fft(cur)
        maxc = conv.max()
        # determine active indices: either top K or those near max
        # start with those near max
        thresh = maxc * (1 - rel_thresh)
        active = np.where(conv >= thresh)[0]
        if active.size > K:
            # pick top K among them
            # use argpartition on conv[active] descending
            idx = np.argpartition(-conv[active], K-1)[:K]
            active = active[idx]
        elif active.size < K:
            # pad with top values across whole conv
            need = K - active.size
            extra = np.argpartition(-conv, K-1)[:K]
            # combine unique indices
            active = np.unique(np.concatenate([active, extra]))[:K]
        # compute subgradient
        g = np.zeros_like(cur)
        # vectorized loop
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]   # derivative of (a*a)_k

        # additive update (projected gradient)
        alpha = add_alpha0 / np.sqrt(it)
        cand_add = cur - alpha * g
        cand_add = project_simplex(cand_add)
        cand_add_sc = score(cand_add)

        # multiplicative update (exponential weights)
        eta = mul_eta0 / np.sqrt(it)
        # avoid overflow/underflow by clipping exponent
        exp_factor = np.exp(-eta * g)
        cand_mul = cur * exp_factor
        sum_mul = cand_mul.sum()
        if sum_mul == 0:
            cand_mul = np.full_like(cur, 1.0 / n)
        else:
            cand_mul = cand_mul / sum_mul
        cand_mul_sc = score(cand_mul)

        # choose best
        # cache best among candidates and current
        best_local_sc = min(cur_sc, cand_add_sc, cand_mul_sc)
        if best_local_sc == cand_add_sc and cand_add_sc < cur_sc:
            cur, cur_sc = cand_add, cand_add_sc
        elif best_local_sc == cand_mul_sc and cand_mul_sc < cur_sc:
            cur, cur_sc = cand_mul, cand_mul_sc

        # optionally accept a worse move with small probability to escape local minima
        else:
            # acceptance probability based on temperature decreasing with iteration
            temp = 1.0 / np.sqrt(it)  # raw temperature
            # difference between best local and current
            diff = min(cand_add_sc, cand_mul_sc) - cur_sc
            if diff < 0:
                # shouldn't happen because min() was not less than cur_sc due to earlier check
                pass
            else:
                prob = math.exp(-diff / (temp + 1e-12))
                if rng.random() < prob:
                    if cand_add_sc < cand_mul_sc:
                        cur, cur_sc = cand_add, cand_add_sc
                    else:
                        cur, cur_sc = cand_mul, cand_mul_sc

        # update best found
        if cur_sc < best_sc:
            best_vec = cur.copy()
            best_sc = cur_sc
            no_improve = 0
        else:
            no_improve += 1

        # random mass transfer when stagnating
        if no_improve > 2000:
            # attempt a random transfer
            i = rng.integers(0, n)
            if cur[i] > 0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = project_simplex(tmp)
                tmp_sc = score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0

        # jittered restart toward best
        if it % 6000 == 0:
            jitter = rng.normal(scale=1e-4, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = project_simplex(tmp)
            tmp_sc = score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        if best_sc < target:
            break

    # final local refinement within remaining time
    remaining = deadline - time.monotonic()
    if remaining > 0.1:
        # do simple random swaps for a while
        ref_end = time.monotonic() + min(0.3, remaining)
        while time.monotonic() < ref_end:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur[i] == 0:
                continue
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = project_simplex(tmp)
            tmp_sc = score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

    # evaluate best_vec with official evaluator (ensuring no rounding errors)
    # optionally ensure that all entries are finite and non-negative
    # convert to list of floats
    return [float(x) for x in best_vec]

            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * cur[k - idx]

        # --------------------------------------------------------------
        # additive (projected‑gradient) candidate
        # --------------------------------------------------------------
        alpha = add_alpha0 / math.sqrt(it)
        cand_add = cur - alpha * g
        cand_add = _project_simplex(cand_add)
        cand_add_sc = _score(cand_add)

        # --------------------------------------------------------------
        # multiplicative (exponential‑weights) candidate
        # --------------------------------------------------------------
        eta = mul_eta0 / math.sqrt(it)
        exp_factor = np.exp(-eta * g)
        cand_mul = cur * exp_factor
        s_mul = cand_mul.sum()
        if s_mul == 0.0:
            cand_mul = np.full_like(cur, 1.0 / n)
        else:
            cand_mul = cand_mul / s_mul
        cand_mul_sc = _score(cand_mul)

        # --------------------------------------------------------------
        # choose the best candidate (including possible Metropolis step)
        # --------------------------------------------------------------
        if cand_add_sc < cur_sc and cand_add_sc <= cand_mul_sc:
            cur, cur_sc = cand_add, cand_add_sc
        elif cand_mul_sc < cur_sc:
            cur, cur_sc = cand_mul, cand_mul_sc
        else:
            # occasional acceptance of a worse move
            temp = 1.0 / math.sqrt(it)
            diff = min(cand_add_sc, cand_mul_sc) - cur_sc
            if diff > 0 and rng.random() < math.exp(-diff / (temp + 1e-12)):
                if cand_add_sc < cand_mul_sc:
                    cur, cur_sc = cand_add, cand_add_sc
                else:
                    cur, cur_sc = cand_mul, cand_mul_sc

        # --------------------------------------------------------------
        # keep the best ever seen
        # --------------------------------------------------------------
        if cur_sc < best_sc:
            best_vec, best_sc = cur.copy(), cur_sc
            no_improve = 0
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # random mass‑transfer when progress stalls
        # --------------------------------------------------------------
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0

        # --------------------------------------------------------------
        # jittered restart towards the best solution
        # --------------------------------------------------------------
        if it % 6000 == 0:
            jitter = rng.normal(scale=1e-4, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        # --------------------------------------------------------------
        # occasional tiny LP step (matrix‑free, sparse)
        # --------------------------------------------------------------
        if it % lp_interval == 0 and (deadline - time.monotonic()) > 5.0:
            K_lp = min(20, active.size)
            if K_lp > 0:
                active_lp = active[:K_lp]
                rows, cols, data_vals = [], [], []
                for row_idx, k in enumerate(active_lp):
                    lo = max(0, k - (n - 1))
                    hi = min(k, n - 1)
                    i_range = np.arange(lo, hi + 1, dtype=np.int64)
                    j_range = k - i_range
                    rows.extend([row_idx] * len(i_range))
                    cols.extend(j_range.tolist())
                    data_vals.extend(cur[i_range].tolist())
                A = sp.csr_matrix((data_vals, (rows, cols)), shape=(len(active_lp), n))
                c = -np.ones(n)                     # maximise sum(b)  ≡  minimise –sum(b)
                bounds = [(0, None)] * n
                res = opt.linprog(c, A_ub=A, b_ub=np.full(A.shape[0], maxc),
                                  bounds=bounds, method='highs')
                if res.success:
                    b = res.x
                    sb = b.sum()
                    if sb > 0.0:
                        g_lp = b / sb
                        g_lp_sc = _score(g_lp)
                        if g_lp_sc < cur_sc:
                            cur, cur_sc = g_lp, g_lp_sc
                            if g_lp_sc < best_sc:
                                best_vec, best_sc = g_lp.copy(), g_lp_sc

        # --------------------------------------------------------------
        # early exit if we already beat the target
        # --------------------------------------------------------------
        if best_sc < target:
            break

    # ----------------------------------------------------------------------
    # final local refinement (few random mass‑transfer moves)
    # ----------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 0.2:
        refine_end = time.monotonic() + min(0.4, remaining)
        while time.monotonic() < refine_end:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur[i] == 0.0:
                continue
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

    # return the best vector found as a plain Python list
    return [float(x) for x in best_vec]
