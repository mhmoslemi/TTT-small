    fx = np.fft.rfft(x, n=L)

import time, math, numpy as np

def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution x*x using a power‑of‑two FFT length L."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]
    return conv

def _score_seq(x: np.ndarray, n: int, L: int) -> float:
    """Return the AC1 score 2·n·max(conv)/sum(x)^2."""
    conv = _conv_fft(x, L)
    return 2.0 * n * conv.max() / (x.sum() * x.sum())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a non‑negative step function with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a short safety margin before the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # --------------------------------------------------------------
    # Initial sequence: use the supplied parent if present,
    # otherwise a uniform distribution.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = np.full(2000, 1.0 / 2000, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s <= 0.0 or not np.isfinite(s):
        cur = np.full_like(cur, 1.0 / cur.size)
    else:
        cur = cur / s                     # probability simplex

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two length for FFT

    # --------------------------------------------------------------
    # Basic statistics for the current best solution
    # --------------------------------------------------------------
    cur_sc = _score_seq(cur, n, L)
    best = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------------
    # Mirrors of the exponentiated gradient (log‑space Adam)
    # --------------------------------------------------------------
    # start from a neutral log‑probability vector
    logits = np.log(np.maximum(cur, 1e-12))
    m = np.zeros_like(cur)          # first moment
    v = np.zeros_like(cur)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    eta0 = 0.6                      # base learning rate

    # --------------------------------------------------------------
    # Misc. parameters
    # --------------------------------------------------------------
    move_prob = 0.07                # how often a random mass‑transfer is tried
    eps_move = 0.25                 # max fraction of the source mass that can be moved
    jitter_base = 1e-4
    stagnation_limit = 12000        # restart after this many non‑improving steps
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # ---------- current convolution & active index ----------
        conv = _conv_fft(cur, L)
        k = int(conv.argmax())                 # index of the current maximum
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        pair = k - idx

        # ---------- sub‑gradient of the max‑convolution ----------
        g = np.zeros_like(cur)
        g[idx] = cur[pair]
        # double contribution when i == k-i (the diagonal)
        g[idx][idx == pair] *= 2.0

        # ---------- Adam‑style update in log‑space ----------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        lr = eta0 / math.sqrt(it)                       # simple schedule
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits in a reasonable numerical window
        np.clip(logits, -20.0, 20.0, out=logits)

        # ---------- candidate from the updated logits ----------
        cand = np.exp(logits - np.max(logits))   # numerical stability
        cand_sum = cand.sum()
        if cand_sum > 0.0:
            cand = cand / cand_sum
        else:
            cand = np.full_like(cur, 1.0 / n)

        cand_sc = _score_seq(cand, n, L)

        # ---------- Metropolis acceptance ----------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # occasional uphill move
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
            else:
                no_improve += 1

        if cand_sc < best_sc:
            best_sc, best = cand_sc, cand.copy()

        # ---------- random mass‑transfer move ----------
        if rng.random() < move_prob:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * eps_move
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                tmp_sum = tmp.sum()
                if tmp_sum > 0.0:
                    tmp = tmp / tmp_sum
                    tmp_sc = _score_seq(tmp, n, L)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best = tmp_sc, tmp.copy()

        # ---------- jittered restart from the best point ----------
        if it % 3000 == 0:
            jitter = rng.normal(scale=jitter_base, size=n)
            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)
            ts = tmp.sum()
            if ts > 0.0:
                tmp = tmp / ts
                tmp_sc = _score_seq(tmp, n, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best = tmp_sc, tmp.copy()
                no_improve = 0

        # ---------- stagnation restart ----------
        if no_improve > stagnation_limit:
            jitter = rng.normal(scale=jitter_base * 5.0, size=n)
            cur = best + jitter
            cur = np.maximum(cur, 0.0)
            cs = cur.sum()
            if cs > 0.0:
                cur = cur / cs
            cur_sc = _score_seq(cur, n, L)
            no_improve = 0

        # ---------- early exit if the target is reached ----------
        if best_sc < 1.50274349:
            break

    # Return a plain Python list (the evaluator expects a list)
    return [float(x) for x in best]
