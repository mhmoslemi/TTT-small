def solve_lp(f, M):
    import scipy.sparse as sp
    from scipy.optimize import linprog
    n = f.shape[0]
    rows = 2*n-1
    # Construct Toeplitz convolution constraint
    data = [np.full(n, f[d]) for d in range(n)]
    offsets = np.arange(n)
    A = sp.diags(data, offsets, shape=(rows, n), format='csr')
    # Upper bound sum(g) <= M / max(f) (if max_f>0)
    max_f = f.max()
    constraints = [A]
    b = np.full(rows, M)
    if max_f > 0:
        sum_ub = M / max_f
        # Append row of ones
        # Convert to csr and vertical stack
        ones_row = sp.csr_matrix(np.ones((1,n)))
        A = sp.vstack([A, ones_row])
        b = np.append(b, sum_ub)
    # objective
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
    if not res.success:
        return None
    g = res.x
    # Normalize to sum 1
    if g.sum() <= 0:
        return None
    g = g / g.sum()
    return g
```

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
