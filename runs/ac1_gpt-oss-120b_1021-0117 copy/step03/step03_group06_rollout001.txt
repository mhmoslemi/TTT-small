    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    # initialization
    if "height_sequence_1" in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2000
        a = np.full(n, 1.0/n, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0/a.size)
        s = a.sum()
    a = a / s
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    def conv_fft(x):
        fx = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(fx * fx, n=L)[: 2 * n - 1]
        return conv
    def score(x):
        s = x.sum()
        if s == 0:
            return float('inf')
        conv = conv_fft(x)
        return 2.0 * n * conv.max() / (s * s)
    cur = a.copy()
    cur_sc = score(cur)
    best = cur.copy()
    best_sc = cur_sc
    it = 0
    no_improve = 0

    alpha0 = 0.45
    eps_move = 0.3
    jitter_scale = 2e-4

    p_transfer = 0.17
    restart_interval = 2500
    stagnation_limit = 6000
