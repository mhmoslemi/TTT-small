import time
import numpy as np

def _score_vec(x: np.ndarray) -> float:
    """Return 2*n*max(conv(x,x)). Fast via FFT (zero‑padded to power of two)."""
    n = int(x.size)
    conv_len = 2 * n - 1
    # next power of two for efficient FFT
    fft_len = 1 << ((conv_len - 1).bit_length())
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return 2.0 * n * conv.max()


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Generate a few simple step‑function candidates and return the one with the

    """

        else:

    candidates = []

    dec = np.arange(n, 0, -1, dtype=np.float64)
