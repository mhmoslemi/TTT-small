import time
import numpy as np

# -------------------------------------------------------
# Simplex projection (Euclidean) – reusable utility
# -------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (sum = 1, v >= 0)."""
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------
# Main search routine
# -------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Returns a non‑negative step‑function (list of floats) whose
    evaluation score is as low as possible within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ---- time budget -------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ---- initial sequence ---------------------------------------------
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = parent.copy()
    except Exception:
        n = 1024
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # normalise to sum = 1

    # ---- best‑so‑far tracking -----------------------------------------
    best = a.copy()
    # compute current bound (2*n*max_conv) – does not need sum because sum=1
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(best, n=fft_len)
    best_max = np.max(np.fft.irfft(fa * fa, n=fft_len)[:conv_len])
    best_score = float(2.0 * n * best_max)

    # ---- optimisation hyper‑parameters --------------------------------
    eta0 = 1.0          # initial step magnitude
    beta = 0.95         # momentum decay
    top_k = 30          # active peaks to use for gradient
    jitter_every = 180  # steps between random jitter
    jitter_std = 0.02   # jitter intensity (log‑normal)
    max_stale = 2500    # steps without improvement before a restart
    target_score = 1.50274349

    v = np.zeros_like(a)   # momentum buffer
    it = 0
    no_improve = 0

    # ---- auxiliary function: compute max convolution value -------------
    def _max_conv(x: np.ndarray) -> float:
        fx = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
        return float(conv.max())

    # ---- main optimisation loop ---------------------------------------
    while time.monotonic() < deadline:
        # ---- convolution (FFT) -----------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        max_val = float(conv.max())
        cur_score = 2.0 * n * max_val

        # ---- record improvement ----------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ---- choose active convolution indices -------------------------
        # pick the top‑k peaks (or all if fewer)
        if conv.size > top_k:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        else:
            idxs = np.arange(conv.size, dtype=np.int64)

        # ---- gradient of max‑conv (sub‑gradient) -----------------------
        g = np.zeros_like(a)
        for k in idxs:
            # i runs over all positions where both i and k-i are valid
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            partner = k - idx
            g[idx] += a[partner]
        # factor 2 because each unordered pair contributes twice
        g = 2.0 * g / max(1, len(idxs))

        # ---- momentum update -------------------------------------------
        it += 1
        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g

        # ---- gradient step + simplex projection -------------------------
        a = _proj_simplex(a - eta * v)

        # ---- occasional random jitter to escape plateaus ---------------
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_std, size=n))
            a = _proj_simplex(a * jitter)
            v.fill(0.0)
            no_improve = 0

        # ---- long stagnation → restart around the best -----------------
        if no_improve >= max_stale:
            # perturb best a a little and continue
            perturb = np.exp(rng.normal(0.0, 0.03, size=n))
            a = _proj_simplex(best * perturb)
            v.fill(0.0)
            no_improve = 0

        # ---- opportunistic random draws (small budget slice) ----------
        if it % 500 == 0:                     # every few hundred iterations
            # draw a random Dirichlet vector (alpha=0.5 tends to be sparse)
            rand_vec = rng.dirichlet([0.5] * n)
            rand_score = 2.0 * n * _max_conv(rand_vec)
            if rand_score < best_score:
                best = rand_vec.copy()
                best_score = rand_score
                a = rand_vec.copy()
                v.fill(0.0)
                no_improve = 0

    # ---- return best candidate as plain Python list --------------------
    return [float(x) for x in best]
