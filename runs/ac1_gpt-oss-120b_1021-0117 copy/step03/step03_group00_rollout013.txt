import time

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (sum = 1, >=0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    n = vec.shape[0]
    vf = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(vf * vf, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            cur = cand
            cur_fft = cand_fft
            cur_conv = cand_conv
            cur_max = cand_conv.max()
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            # occasional probabilistic acceptance (simulated‑annealing flavour)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_fft = cand_fft
                cur_conv = cand_conv
                cur_max = cand_conv.max()
                cur_sc = cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # ------- occasional symmetrisation (helps in many runs) -------
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L_full)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L_full)[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:
                cur = sym
                cur_fft = sym_fft
                cur_conv = sym_conv
                cur_max = sym_conv.max()
                cur_sc = sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()

        # ------- random mass‑transfer when stuck -------
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L_full)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L_full)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_fft = tmp_fft
                    cur_conv = tmp_conv
                    cur_max = tmp_conv.max()
                    cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                no_improve = 0

        # early exit if we already beat the target
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
