import time, math, numpy as np

# -------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# -------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w

# -------------------------------------------------------------
# Convolution max (FFT, returns value and index)
# -------------------------------------------------------------
def _conv_max_and_index(a: np.ndarray, L: int):
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx
