import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex using the algorithm of Wang & Carreira-Perpinan."""
    n = v.size
    u = np.sort(v)[::-1]                       # descending sort
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_candidates = np.where(rho_arr > 0)[0]
    if rho_candidates.size == 0:
        theta = 0.0
    else:
        rho = rho_candidates[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# convolution via FFT (full linear convolution a*a)
# ------------------------------------------------------------
def _conv(vec: np.ndarray) -> np.ndarray:
    n = vec.shape[0]
    L = 1 << ((2 * n - 1).bit_length())       # next power of two
    fv = np.fft.rfft(vec, n=L)
    conv_full = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv_full

# ------------------------------------------------------------
# score exactly as in the evaluator (no clipping needed here)
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    n = vec.shape[0]
    conv = _conv(vec)
    return 2.0 * n * conv.max() / (vec.sum() ** 2)

# ------------------------------------------------------------
# optional analytic start: sqrt-series (used only if no parent)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

    """Return a non‑empty list of non‑negative numbers with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    # leave a small safety margin before the sandbox timeout

    # --------------------------------------------------------
    # initialise a feasible sequence

    if "height_sequence_1" in globals():

    # ensure non‑negativity and normalise to sum = 1 (simplex)

    else:
        base = base / base.sum()
    a = base.copy()
    n = a.size
    i_arr = np.arange(n)

    best_vec = a.copy()
    best_sc = _score(best_vec)

    # --------------------------------------------------------
    # algorithm parameters
    # --------------------------------------------------------
    step = 0.2                     # initial step size for gradient moves
    idle_counter = 0               # counts iterations without improvement
    max_idle = 1500                # after this we jitter the current best
    # probability of performing a gradient step vs a random hill step
    prob_gradient = 0.75

    # --------------------------------------------------------
    # main optimisation loop

    while time.monotonic() < deadline:
        # decide move type
