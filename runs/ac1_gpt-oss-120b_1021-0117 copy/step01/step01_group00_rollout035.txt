import numpy as np, time, math

def _sidon_like_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (1.0 - 1.0/(2.0*i))
    return a

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 2.0)

    # Determine length
    try:
        n = int(len(height_sequence_1))
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        parent_available = True
    except Exception:
        n = 256
        parent = None
        parent_available = False

    # base Sidon-like values (nonnegative)
    base = _sidon_like_sequence(n)

    # initial candidate: random permutation of base
    perm = rng.permutation(n)
    current_seq = base[perm].astype(np.float64)
    # ensure clipping (though all <1)
    current_seq = np.clip(current_seq, 0.0, 1000.0)

    # FFT length
    fft_len = 1 << ((2*n - 1).bit_length())

    def compute_score(seq):
        # seq: np.ndarray float64, length n
        # convolution via real FFT
        f = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(f * f, fft_len)[:2*n-1]
        max_b = conv.max()
        s = seq.sum()
        return 2.0 * n * max_b / (s * s) if s > 0 else np.inf

    best_seq = current_seq.copy()

    # simple linear cooling from T_start to T_end over time
    T_start = 0.01
    T_end = 1e-6

    max_transfer_factor = 0.05  # up to 5% of max value
    # probability of doing a transfer vs swap
    prob_transfer = 0.5

            current_score = cand_score

def _sidon_like_sequence(n):
    """Classic a_i = C(2i,i)/4**i (decreasing, sum≈√n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a
