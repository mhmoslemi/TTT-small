def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)   # leave margin

    # get n and base array
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = parent.shape[0]
        # ensure parent not empty
        if n == 0:
            raise Exception
        init = parent.copy()
    except Exception:
        n = 256
        init = np.ones(n, dtype=np.float64)

    # generate sidon-like seq length n
    def sidon(n):
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1,n):
            a[i] = a[i-1]*(1.0 - 1.0/(2.0*i))
        return a

    # If parent not given or we want maybe better shape, we combine.
    if n > 0:
        # use sidon-like if parent has many repeated values? parent may have shape similar
        seq = sidon(n)
        # perhaps combine: take average of parent and sidon?
        # simply use sidon as base if its evaluation better; else use parent
        seq = np.clip(seq, 0.0, 1000.0)
        # random shuffle for variety
        seq = seq[rng.permutation(n)]

        # compute scores
        best = None; best_score = np.inf
        # Evaluate initial parent and seq
        for candidate in [init, seq]:
            score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = candidate.copy()
        if best is None:
            best = seq.copy()
            best_score = float(evaluate_sequence(best.tolist()))

    # start annealing
    cur = best.copy()
    cur_score = best_score
    # Parameter schedules
    max_iters = 1000000  # large but break if time limit
    # temperature schedule exponential: T0 = 1e-2 * initial? We'll compute.
    # We need acceptance probability = exp((cur_score - new_score)/temp)
    # Since scores around 1.5, differences maybe 1e-3 etc. So scale.
    # We'll define temperature decays from T0 to near 0.
    T0 = 0.1   # arbitrary
    # compute time fraction for schedule
    start_time = time.monotonic()
    while time.monotonic() < deadline:
        # adjust temperature based on elapsed fraction
        elapsed = time.monotonic() - start_time
        # fraction of budget (e.g., 0.9)
        t_frac = elapsed / (deadline - start_time)
        T = T0 * (1 - t_frac)  # linear decay
        # propose move: swap two indices i,j
        i,j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        cand = cur.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_score = float(evaluate_sequence(cand.tolist()))
        # acceptance
        if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / (T+1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
        # occasional random global shuffle to escape
        if rng.random() < 0.001:
            perm = rng.permutation(n)
            cur = cur[perm]
            cur_score = float(evaluate_sequence(cur.tolist()))
        # occasional reset to best
        if rng.random() < 0.0005:
            cur = best.copy()
            cur_score = best_score

    return [float(x) for x in best]

import numpy as np
import time

def _sidon_like_sequence(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4**i for i=0..n-1 (monotone decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def _ac1_fast(seq: np.ndarray) -> float:
    """Fast AC1 bound using FFT (identical to evaluate_sequence formula)."""
    n = seq.shape[0]
    # size for linear convolution (next power of two ≥ 2n‑1)
    m = 1 << ((2 * n - 1).bit_length())
    # convolution via real FFT
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_b = conv.max()
    s = seq.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a sequence of non‑negative heights that (hopefully) improves
    the AC1 upper bound.  The routine uses a simulated‑annealing style
    search on permutations of a Sidon‑like distribution.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # Acquire problem size and a fallback parent if present
    # ------------------------------------------------------------------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = int(parent.shape[0])
    except Exception:
        # generic fallback
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    # ------------------------------------------------------------------
    # Build initial candidates: the given parent and a shuffled Sidon‑like seq
    # ------------------------------------------------------------------
    sidon = _sidon_like_sequence(n)
    sidon = np.clip(sidon, 0.0, 1000.0)
    sidon = sidon[rng.permutation(n)]

    candidates = [parent, sidon]
    best_seq = None
    best_score = np.inf
    best_fast = np.inf

    for cand in candidates:
        fast = _ac1_fast(cand)
        exact = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(exact) and exact < best_score:
            best_seq = cand.copy()
            best_score = exact
            best_fast = fast

    if best_seq is None:
        best_seq = sidon.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        best_fast = _ac1_fast(best_seq)

    # ------------------------------------------------------------------
    # Simulated annealing on permutations of the current multiset
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_fast = best_fast

    start = time.monotonic()
    total_time = deadline - start
    restart_prob = 0.001

    while time.monotonic() < deadline:
        # temperature linearly decays from 0.1 to 0.001
        elapsed = time.monotonic() - start
        prog = elapsed / total_time
        temp = 0.1 * (1.0 - prog) + 0.001 * prog

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # perform swap in place
        cur[i], cur[j] = cur[j], cur[i]
        cand_fast = _ac1_fast(cur)

        accept = False
        if cand_fast < cur_fast:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(cand_fast - cur_fast) / (temp + 1e-12)):

                exact = float(evaluate_sequence(cur.tolist()))

                    best_score = exact
                    best_fast = cand_fast
        else:

        # occasional random restart from the current best
        if rng.random() < restart_prob:
            cur = best_seq.copy()
            cur = cur[rng.permutation(n)]
            cur_fast = _ac1_fast(cur)

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # --------------------------------------------------------------
    return [float(x) for x in best_seq]
