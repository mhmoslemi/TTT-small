def project_simplex(v):
    # v is 1D numpy array
    # ensure non-negative sum = 1
    # using sorting O(n log n)
    # but n ~ 5000, fine.
    # Implementation:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1  # subtract target sum 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

# ---------- helper utilities -------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    # Duchi et al. algorithm, O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero; return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_self(a):
    """Self‑convolution a*a via FFT (real‑valued, length 2*len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    # eliminate tiny negative round‑off
    conv[conv < 0] = 0.0
    return conv

def _score(a):
    """Exact score used by the evaluator (sum may not be exactly 1)."""
    n = a.shape[0]
    s = a.sum()
    if s <= 0:
        return float('inf')
    conv = _conv_self(a)
    return 2.0 * n * conv.max() / (s * s)
