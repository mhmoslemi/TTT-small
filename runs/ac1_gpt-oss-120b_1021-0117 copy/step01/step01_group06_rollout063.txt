import time
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = u - css / np.arange(1, n + 1)
    mask = rho > 0
    if not np.any(mask):
        theta = 0.0
    else:
        rho_idx = np.where(mask)[0][-1]
        theta = css[rho_idx] / (rho_idx + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic start: coefficients of sqrt(1+…+x^{2n‑2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Full convolution via FFT (full length = 2*n‑1)
# ------------------------------------------------------------
def _fft_conv(vec: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * vec.size - 1]
    return conv


# ------------------------------------------------------------
# Score exactly as used by the evaluator
# ------------------------------------------------------------
def _score(vec: np.ndarray, L: int) -> float:
    n = vec.size
    conv = _fft_conv(vec, L)
    return 2.0 * n * conv.max() / (vec.sum() ** 2)

# Main entry point
# ------------------------------------------------------------

    start = time.monotonic()
    # give a small safety margin for the sandbox timeout
    total = max(0.0, float(budget_s) - 5.0)
    deadline = start + total

    # ------------------------------------------------------------------
    # initialise the working vector (sum forced to 1 for numerical stability)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        if s < 1e-12:

        else:
            cur = cur / s
    else:
        n = 256
        cur = np.ones(n, dtype=np.float64) / n

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())      # FFT length (next power of two)

    # ------------------------------------------------------------------
    # evaluate a few simple starts and pick the best as the initial best
    # ------------------------------------------------------------------
    best = cur.copy()
    best_score = _score(best, L)

    # analytic sqrt‑series start
    ss = _sqrt_series(n)
    ss = ss / ss.sum()
    ss_score = _score(ss, L)
    if ss_score < best_score:
        best = ss.copy()
        best_score = ss_score

    # uniform start (occasionally useful)
    uni = np.ones(n, dtype=np.float64) / n
    uni_score = _score(uni, L)
    if uni_score < best_score:

        best_score = uni_score

    cur = best.copy()

    # annealing schedule
