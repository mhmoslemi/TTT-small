def _conv(vec):
    n=vec.size
    L = 1 << ((2*n-1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
    return conv

import time
import numpy as np

# ------------------------------------------------------------
# simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# analytic sqrt‑series (finite truncation)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of a where a*a = 1 + … + x^{2n‑2}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # dot of a[1:k] with a[k-1:0:-1]
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ------------------------------------------------------------
# fast full convolution via FFT (real)
# ------------------------------------------------------------
def _full_conv(vec: np.ndarray) -> np.ndarray:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv

# ------------------------------------------------------------
# score (identical to the provided evaluator)
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (vec.sum() ** 2)

# ------------------------------------------------------------
# gradient of the (averaged) max‑convolution index
# ------------------------------------------------------------
def _gradient_from_max(vec: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Return a sub‑gradient of max(conv) using all maximal indices."""
    n = vec.size
    m = 2 * n - 1
    mask = np.zeros(m, dtype=np.float64)
    if max_idx.size:
        mask[max_idx] = 1.0 / max_idx.size
    # linear cross‑correlation via FFT
    L = 1 << ((3 * n - 2).bit_length())          # >= m + n - 1
    fw = np.fft.rfft(mask, n=L)
    fra = np.fft.rfft(vec[::-1], n=L)           # reversed vector
    corr = np.fft.irfft(fw * fra, n=L)[: 3 * n - 2]
    start = n - 1
    grad = 2.0 * corr[start:start + n]          # 2 * sum_k mask_k * a_{k-i}
    return grad
