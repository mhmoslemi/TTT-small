import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, m: int) -> np.ndarray:
    """Linear convolution a*a via real FFT of length m."""
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * a.size - 1]
    return conv


def _score_fft(a: np.ndarray, n: int, m: int) -> float:
    """Score defined in the authoritative evaluator (fast version)."""
    conv = _conv_self_fft(a, m)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a low‑score non‑negative step function under the budget."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initialise from the provided parent or a fallback series.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = a0.size
    else:
        n = 5582
        a0 = None

    sqrt_seq = _sqrt_series(n)

    # start with the better of the two
    if a0 is not None:
        best = a0.copy()
    else:
        best = sqrt_seq.copy()

    m = _next_pow2(2 * n - 1)                     # FFT length
    best_score = _score_fft(best, n, m)

    alt_score = _score_fft(sqrt_seq, n, m)
    if alt_score < best_score:
        best = sqrt_seq.copy()
        best_score = alt_score

    # ------------------------------------------------------------------
    # Helper: contributions to the peak of the autocorrelation.
    # ------------------------------------------------------------------
    def _contributions(seq: np.ndarray, peak: int) -> np.ndarray:
        idx = np.arange(n)
        rev = peak - idx
        mask = (rev >= 0) & (rev < n)
        contr = np.zeros(n, dtype=np.float64)
        contr[mask] = seq[idx[mask]] * seq[rev[mask]]
        return contr

    # ------------------------------------------------------------------
    # Main stochastic improvement loop.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = best.copy()
        op = rng.random()

        # --------------------------------------------------------------
        # 1) Targeted mass transfer using the peak contributions.
        # --------------------------------------------------------------
        if op < 0.55:
            conv = _conv_self_fft(cand, m)
            peak = int(np.argmax(conv))
            contrib = _contributions(cand, peak)

            if contrib.max() > 0:
                src = int(np.argmax(contrib))

                # choose a destination with small contribution
                low_thresh = np.percentile(contrib, 20)
                low_idxs = np.where(contrib <= low_thresh)[0]
                if low_idxs.size == 0:
                    dst = rng.integers(0, n - 1)
                    while dst == src:
                        dst = rng.integers(0, n - 1)
                else:

                    if dst == src:

                        if alt.size:

                max_transfer = min(cand[src], 1000.0 - cand[dst])

        # --------------------------------------------------------------
