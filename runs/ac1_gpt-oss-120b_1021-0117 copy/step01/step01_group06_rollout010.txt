def _project_simplex(v):
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
    if rho.size > 0:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 score (identical to evaluate_sequence)
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    maxv = conv.max()
    return 2.0 * n * maxv / (vec.sum() ** 2)


# ------------------------------------------------------------
# Index of maximal convolution term (full mode)
# ------------------------------------------------------------
def _max_conv_idx(vec: np.ndarray) -> int:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return int(conv.argmax())


# ------------------------------------------------------------
# Gradient of max‑convolution term (sub‑gradient)
# ------------------------------------------------------------
def _grad_max(v: np.ndarray, k: int) -> np.ndarray:
    n = v.size
    g = np.zeros_like(v)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    if lo <= hi:
        j = np.arange(lo, hi + 1, dtype=int)
        g[j] = 2.0 * v[k - j]
    return g


# ------------------------------------------------------------
# Analytic sqrt‑series (fallback if no parent given)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function whose AC1 bound is small."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # initialise from the provided parent or from the analytic series
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        init = _sqrt_series(n)

    init = np.maximum(init, 0.0)
    s = init.sum()
    if s < 1e-12:                     # degenerate fallback
        init = np.full_like(init, 1.0)
        s = init.sum()
    f = init / s                       # normalise to sum = 1

    n = f.size
    best_f = f.copy()
    best_sc = _score(best_f)

    # ------------------------------------------------------------------
    # iterative improvement: random swaps, gradient descent, occasional smoothing
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # randomly decide which move to try
        r = rng.random()
        if r < 0.55:                     # gradient step
            k = _max_conv_idx(f)
            g = _grad_max(f, k)
            improved = False
            for step in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
                cand = _project_simplex(f - step * g)
                sc = _score(cand)
                if sc < best_sc:
                    best_sc, best_f = sc, cand.copy()
                    f = cand
                    improved = True
                    break
            if not improved and rng.random() < 0.005:

                cand = _project_simplex(f - 0.02 * g)

                    f = cand

                continue

                j += 1

            delta = f[i] * frac

            cand[i] -= delta

            cand = _project_simplex(cand)
