    l = len(x) + len(y) - 1
    size = 1 << (l - 1).bit_length()

import time
import numpy as np

def _fft_convolve(x, y):
    """Linear convolution using real FFT (size is next power of two)."""
    l = len(x) + len(y) - 1
    size = 1 << (l - 1).bit_length()
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    res = np.fft.irfft(X * Y, size)
    return res[:l]

def _norm(v):
    """Clip, keep non‑negative and renormalise to sum‑one."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

def _score_vec(v):
    """Score for a sequence that already sums to 1."""
    n = v.size
    conv = _fft_convolve(v, v)
    maxb = conv.max()
    return 2.0 * n * maxb
