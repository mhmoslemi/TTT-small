# ------------------------------------------------------------
# Helper to build a classic Sidon‑like decreasing sequence
# ------------------------------------------------------------
def _sidon_like(n: int) -> np.ndarray:
    """a_i ≈ 1/√(i+1), then normalized to sum 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s == 0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

def _exp_decay(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam*i), normalized."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s == 0:
        a[:] = 1.0 / n
    else:
        a /= s

def _arcsine_like(n: int, eps: float) -> np.ndarray:
    """a_i ∝ 1/√((i+eps)*(n-1-i+eps)), normalized."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s == 0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ------------------------------------------------------------
# Core optimizer – a greedy mass‑transfer loop guided by the
# currently maximal convolution entry.
# ------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Determine problem size and a fallback (parent) sequence if present
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    # ------------------------------------------------------------------
    # Initialise best solution with a few cheap candidates
    # ------------------------------------------------------------------
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # Sidon‑like
    cand = _sidon_like(n)
    cand_score = float(evaluate_sequence(cand.tolist()))
    if np.isfinite(cand_score) and cand_score < best_score:
        best_seq, best_score = cand.copy(), cand_score

    # a few exponentials
    for lam in (0.0005, 0.001, 0.002):
        cand = _exp_decay(n, lam)
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best_seq, best_score = cand.copy(), cand_score

    # a few arcsine‑like shapes
    for eps in (0.1, 0.5, 1.0):
        cand = _arcsine_like(n, eps)
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best_seq, best_score = cand.copy(), cand_score

    # ------------------------------------------------------------------
    # Main search – greedy mass transfer guided by the current max‑conv
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score

    # FFT length (next power of two) – used for fast convolution
    fft_len = 1 << ((2 * n - 1).bit_length())

    # Helper to obtain full convolution, its maximum and the index of that maximum
    def _conv_max(a):
        A = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())
        return conv, conv[idx], idx

    # Main loop
    while time.monotonic() < deadline:
        conv, cur_max, k0 = _conv_max(cur)

        # --------------------------------------------------------------
        # Derivative of the maximal entry w.r.t. each a_i.
        # For interior positions derivative = 2 * a_{k0-i},
        # for the central term (i = k0-i) it is a_i.
        # --------------------------------------------------------------
        deriv = np.zeros_like(cur)
        lo = max(0, k0 - (n - 1))
        hi = min(n - 1, k0)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        counterpart = k0 - idx
        deriv[idx] = 2.0 * cur[counterpart]
        if k0 % 2 == 0:
            mid = k0 // 2
            if 0 <= mid < n:
                deriv[mid] = cur[mid]  # correct the double count

        # --------------------------------------------------------------
        # Choose a source index (large contribution) and a sink index
        # --------------------------------------------------------------
        weight = deriv * cur
        if weight.sum() > 0:
            src = rng.choice(n, p=weight / weight.sum())
        else:
            src = rng.integers(0, n)

        # sink: pick among the smallest derivative values
        min_der = deriv.min()
        sink_candidates = np.where(deriv <= min_der + 1e-12)[0]
        if sink_candidates.size == 0:
            sink = rng.integers(0, n)
        else:
            sink = rng.choice(sink_candidates)

        if sink == src:
            sink = (src + 1) % n
        if cur[src] <= 0.0:
            continue

        # --------------------------------------------------------------
        # Transfer a modest amount of mass from src to sink
        # --------------------------------------------------------------
        delta = cur[src] * rng.random() * 0.02  # up to 2% of src
        cand = cur.copy()
        cand[src] -= delta
        cand[sink] += delta
        # enforce hard bounds required by the evaluator
        np.clip(cand, 0.0, 1000.0, out=cand)

        # --------------------------------------------------------------
        # Evaluate candidate
        # --------------------------------------------------------------
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best_seq, best_score = cand.copy(), cand_score

        # Accept if it improves the current state; occasional uphill moves
        if np.isfinite(cand_score) and cand_score <= cur_score:
            cur, cur_score = cand, cand_score
        elif np.isfinite(cand_score) and rng.random() < 0.001:
            cur, cur_score = cand, cand_score

        # --------------------------------------------------------------
        # Rare random restart – shuffles the current best shape
        # --------------------------------------------------------------
        if rng.random() < 0.0004:
            perm = rng.permutation(n)
            cur = best_seq[perm].copy()
            cur_score = float(evaluate_sequence(cur.tolist()))
            if cur_score < best_score:
                best_seq, best_score = cur.copy(), cur_score

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
