import time
import numpy as np

def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n-1 using recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Return (maximum entry, its index) of the linear convolution a*a."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx
