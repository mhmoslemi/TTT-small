def _score(vec):
    n = vec.size
    L = 1 << ((2*n-1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv*fv, n=L)[:2*n-1]
    return 2.0*n*conv.max()

import time
import numpy as np

# ------------------------------------------------------------
# Simple utilities
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients a with a*a = 1 + … + x^{2n-2} (formal series)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # sum_{i=1}^{k-1} a_i * a_{k-i}
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a


def _score(vec: np.ndarray) -> float:
    """Exact AC1 bound for a non‑negative vector (sum may be !=1)."""
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (vec.sum() ** 2))

# Main search routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    # keep a small safety margin for the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # initialise – use the supplied parent if present, otherwise sqrt series
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        a = _sqrt_series(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0.0:
        a = np.ones_like(a) / a.size
        total = 1.0
    a = a / total                     # normalise to sum = 1

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    cur = a.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score

    n = cur.size
    it = 0
    # simple schedule for a temperature used in occasional uphill steps
    start_temp = best_score * 0.02      # a few percent of the current bound
    while time.monotonic() < deadline:
        it += 1
        r = rng.random()

        # --------------------------------------------------------------
        # 1) mass transfer (biased towards larger source entries)
        # --------------------------------------------------------------
        if r < 0.45:
            # source chosen with probability proportional to its mass
            src = rng.choice(n, p=cur / cur.sum())
            if cur[src] <= 1e-15:
                continue
            # distinct target
            tgt = rng.integers(0, n - 1)
            if tgt >= src:
                tgt += 1
            # transfer a random fraction of the source mass
            frac = rng.random() * 0.5               # up to 50 %
            delta = cur[src] * frac
            cand = cur.copy()
            cand[src] -= delta
            cand[tgt] += delta

        # --------------------------------------------------------------
        # 2) push mass from the very first coefficient further back
        # --------------------------------------------------------------
        elif r < 0.60:
            if cur[0] <= 1e-15:
                continue
            # pick a target in the latter half of the vector
            tgt = rng.integers(n // 2, n)
            frac = rng.random() * 0.8               # up to 80 %
            delta = cur[0] * frac
            cand = cur.copy()
            cand[0] -= delta
            cand[tgt] += delta

        # --------------------------------------------------------------
        # 3) symmetrisation – average with its reversal
        # --------------------------------------------------------------
        elif r < 0.80:
            cand = (cur + cur[::-1]) * 0.5

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        else:

            rand = rng.gamma(1.0, 1.0, size=n)      # exponential → Dirichlet(1)
            rand /= rand.sum()
            cand = (1.0 - alpha) * cur + alpha * rand

        # small numerical drift correction
        cand_sum = cand.sum()
        if cand_sum <= 0.0:
            continue
        cand = cand / cand_sum

        cand_score = _score(cand)

        # ------------------------------------------------------------------
        # accept improvement unconditionally
        # ------------------------------------------------------------------
        if cand_score < cur_score - 1e-12:
