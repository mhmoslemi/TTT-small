def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    rng = np.random.default_rng(seed)
    start=time.monotonic()
    deadline=start+budget_s-0.1

    # length
    try:
        n=len(height_sequence_1)
    except:
        n=256

    # initialize sequence
    a = np.array(height_sequence_1, dtype=np.float64)
    # optional normalization to sum 1
    total=a.sum()
    if total==0:
        a = np.ones(n)/n
    else:
        a = a/total

    # ensure each entry is non-negative and within bounds
    a = np.clip(a,0,1000)

    # Precompute FFT size
    conv_len=2*n-1
    fft_len=1 << ((conv_len-1).bit_length())

    # compute convolution and score
    def compute_c(x):
        fa=np.fft.rfft(x, n=fft_len)
        conv=np.fft.irfft(fa*fa, n=fft_len)[:conv_len]
        return conv

    conv = compute_c(a)
    max_c = conv.max()
    best_a = a.copy()
    best_score = float(2*n*max_c/(a.sum()**2))

    # loop
    while time.monotonic() < deadline:
        # pick current max index
        k = int(conv.argmax())
        # compute contributions
        # vectorize: indices i that satisfy 0 <= i < n and 0 <= k-i < n
        # create mask for i range
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # create slice indices
        idx = np.arange(i_start, i_end+1)
        j = k - idx
        contrib = a[idx] * a[j]  # elementwise product
        # maybe consider total contribution (including duplicate counting)
        # We want to pick an i with large product
        if contrib.size == 0:
            # fallback random
            i = rng.integers(0,n)
        else:
            # choose i with top contrib
            top_idx = np.argmax(contrib)
            i = idx[top_idx]
        # choose destination j2 not equal i; maybe random
        j2 = rng.integers(0, n)
        if j2 == i:
            continue
        # Transfer small fraction
        if a[i] < 1e-12:
            continue
        frac = rng.uniform(0.01, 0.1)
        delta = a[i] * frac
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j2] += delta
        # ensure nonnegative
        if new_a[i] < 0:
            continue
        # compute new convolution and score
        conv_new = compute_c(new_a)
        max_c_new = conv_new.max()
        new_score = float(2*n*max_c_new/(new_a.sum()**2))
        # if improved
        if new_score < best_score - 1e-12:
            a = new_a
            conv = conv_new
            max_c = max_c_new
            best_a = new_a
            best_score = new_score
        # else optionally accept some upward moves? Could skip.

    # final list
    final = [float(x) for x in best_a]
    # ensure evaluate_sequence call

    except Exception:
        pass
    return final

        n = 256
        a = np.random.random(n).astype(np.float64)

    a = np.clip(a, 0.0, 1000.0)            # stay inside the evaluator's limits

    total = a.sum()
    if total == 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / total                      # scaling is irrelevant for the score

    # ---------------------------------------------------------
    # 2.   Pre‑compute the FFT size for the convolution routine
    # ---------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # -------------------------------------------------
    # 3.   Helper that returns the evaluator's objective
    # -------------------------------------------------
    def fast_score(x: np.ndarray) -> float:
        max_b = _fft_convolve_max(x, fft_len, conv_len)
        s = x.sum()
        return float(2.0 * n * max_b / (s * s))

    cur = a.copy()
    cur_score = fast_score(cur)

    best = cur.copy()
    best_score = cur_score

    # ---------------------------------------------------------
    # 4.   Stochastic descent focusing on the current convolution peak
    # ---------------------------------------------------------
    temperature = 0.1
    cooling = 0.9995
    min_temp = 1e-6
    no_improve = 0
    max_no_improve = 2000

    while time.monotonic() < deadline:
        # full convolution to locate the index with the largest entry
        fa = np.fft.rfft(cur, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        k_peak = int(conv.argmax())

        # indices that actually contribute to this peak
        i_start = max(0, k_peak - (n - 1))
        i_end = min(n - 1, k_peak)
        idx = np.arange(i_start, i_end + 1)
        partner = k_peak - idx
        contrib = cur[idx] * cur[partner]

        # pick a source index weighted by its contribution to the peak
        if contrib.size == 0 or contrib.sum() == 0.0:
            src = rng.integers(0, n)
        else:
            probs = contrib / contrib.sum()
            src = rng.choice(idx, p=probs)

        # pick a destination distinct from the source
        dst = rng.integers(0, n)
        while dst == src:
            dst = rng.integers(0, n)

        if cur[src] <= 0.0:
            continue

        # propose moving a modest fraction of mass
        frac = rng.uniform(0.01, 0.20)
        delta = cur[src] * frac
        delta = min(delta, 1000.0 - cur[dst])   # respect the hard 1000 ceiling
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= delta
        cand[dst] += delta

        cand_score = fast_score(cand)

        # greedy acceptance (with a tiny annealing wiggle)
        if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / max(temperature, 1e-12)):
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            no_improve += 1
            if no_improve >= max_no_improve:   # restart from the best seen so far
                cur = best.copy()
                cur_score = best_score
                no_improve = 0

        temperature = max(min_temp, temperature * cooling)

    # -------------------------------------------------
    # 5.   Return the best sequence found (validated)
    # -------------------------------------------------
    final = [float(v) for v in best]
    try:
        _ = evaluate_sequence(final)   # trigger any hidden checks
    except Exception:
        # If something went wrong, fall back to a simple uniform vector
        final = [float(1.0 / n) for _ in range(n)]
    return final
