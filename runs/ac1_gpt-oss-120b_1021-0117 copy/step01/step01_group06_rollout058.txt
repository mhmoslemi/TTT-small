import time
import numpy as np

def _sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k-1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a

def _conv_and_score(vec):
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    sc = 2.0 * n * max_b / (vec.sum() ** 2)
    return conv, max_b, sc

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.clip(a, 0.0, 1000.0)
    else:
        n = 256
        a = _sqrt_series(n)

    n = a.size
    if a.sum() < 0.01:
        a = np.full(n, 1.0 / n)

    best_vec = a
    best_conv, _, best_sc = _conv_and_score(best_vec)

    # main optimisation loop
    while time.monotonic() < deadline:
        # locate index of current maximum of convolution
        max_idx = int(best_conv.argmax())

        # interval of indices that influence that entry
        start = max(0, max_idx - (n - 1))
        end   = min(n - 1, max_idx)

        idx_range = np.arange(start, end + 1)
        # contributions of each index to the maximal convolution entry
        contrib = best_vec[idx_range] * best_vec[max_idx - idx_range]

        # source index = one with largest contribution
        src = idx_range[np.argmax(contrib)]

        # pick a target index outside the influencing interval if possible,
        # otherwise the index with smallest contribution (different from src)
        if start > 0:
            tgt = rng.integers(0, start)
        elif end < n - 1:
            tgt = rng.integers(end + 1, n)
        else:
            # full interval, choose minimal contribution not equal to src
            sorted_idx = idx_range[np.argsort(contrib)]
            tgt = sorted_idx[0] if sorted_idx[0] != src else sorted_idx[1 % len(sorted_idx)]

        if tgt == src:
            tgt = (tgt + 1) % n

        # transfer a random fraction of the source mass
        frac = rng.random() * 0.5          # up to 50 %
        delta = best_vec[src] * frac
        if delta <= 0.0:
            continue

        cand = best_vec.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_conv, _, cand_sc = _conv_and_score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand
            best_conv = cand_conv

        # occasional random perturbation to escape plateaus

import time
import numpy as np


def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of the series a with a*a = 1+…+x^{2n-2}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


def _conv_and_score(vec: np.ndarray):
    """Full convolution via FFT and the AC1 bound."""
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    sc = 2.0 * n * max_b / (vec.sum() ** 2)
    return conv, max_b, sc


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty sequence of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # -----------------------------------------------------------------
    # initialise – use the provided parent if present, otherwise sqrt series
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.clip(a, 0.0, 1000.0)
    else:
        a = _sqrt_series(256)

    n = a.size
    if a.sum() < 0.01:                     # safety fallback
        a = np.full(n, 1.0 / n)

    best_vec = a
    best_conv, _, best_sc = _conv_and_score(best_vec)

    # -----------------------------------------------------------------
    # optimisation loop – target‑max‑convolution reduction
    # -----------------------------------------------------------------
    while time.monotonic() < deadline:
        # index where the current convolution attains its maximum
        max_idx = int(best_conv.argmax())

        # interval of indices that actually contribute to that entry
        start = max(0, max_idx - (n - 1))
        end = min(n - 1, max_idx)

        idx_range = np.arange(start, end + 1)
        contrib = best_vec[idx_range] * best_vec[max_idx - idx_range]

        # source = index with the largest contribution to the peak
        src = idx_range[np.argmax(contrib)]

        # pick a target outside the influencing interval if possible,
        # otherwise the index with the smallest contribution (different from src)
        if start > 0:
            tgt = rng.integers(0, start)
        elif end < n - 1:
            tgt = rng.integers(end + 1, n)
        else:
            # full interval – move to the lowest‑contributing position
            sorted_idx = idx_range[np.argsort(contrib)]
            tgt = sorted_idx[0] if sorted_idx[0] != src else sorted_idx[1 % len(sorted_idx)]

        if tgt == src:                      # avoid a no‑op
            tgt = (tgt + 1) % n

        # transfer a random fraction of the source mass
        frac = rng.random() * 0.5            # up to 50 % of the source amount
        delta = best_vec[src] * frac
        if delta <= 0.0:
            continue

        cand = best_vec.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_conv, _, cand_sc = _conv_and_score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand
            best_conv = cand_conv

                    best_vec = cand2
                    best_conv = cand2_conv

    # return the best found sequence as a plain Python list
    # -----------------------------------------------------------------
    return [float(x) for x in best_vec]
