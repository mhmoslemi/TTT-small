        cur = init / total                     # normalise to sum = 1

    # --------------------------------------------------------------
    # 2. Pre‑compute FFT layout (fixed for the whole run)
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # --------------------------------------------------------------
    # 3. Initial score (the evaluator uses 2·n·max_b / sum²; sum=1 here)
    # --------------------------------------------------------------
    cur_max = _conv_max_fft(cur, fft_len)
    cur_score = 2.0 * n * cur_max
    best = cur.copy()
    best_score = cur_score

    # --------------------------------------------------------------
    # 4. Simulated‑annealing / targeted hill‑climbing loop
    # --------------------------------------------------------------
    max_temp = 0.20                           # start temperature
    total_time = deadline - start

    while time.monotonic() < deadline:
        # current full convolution (needed for the “worst lag” index)
        conv = _conv_fft(cur, fft_len)
        max_val = float(conv.max())
        L = int(np.argmax(conv))              # lag with maximal value

        # ----------------------------------------------------------
        # 4a. Choose a move – most of the time a targeted move
        # ----------------------------------------------------------
        if rng.random() < 0.70:
            # indices that actually contribute to the maximal lag
            low = max(0, L - (n - 1))
            high = min(n - 1, L)
            idxs = np.arange(low, high + 1, dtype=np.int64)
            partners = L - idxs

            # contribution of each index to this lag
            contrib = cur[idxs] * cur[partners]
            if contrib.sum() > 0.0:
                probs = contrib / contrib.sum()
                src = int(rng.choice(idxs, p=probs))
            else:
                src = int(rng.integers(0, n))

            if cur[src] <= 1e-16:            # source too small – fall back
                src = int(rng.integers(0, n))

            # destination tends to be a currently small entry
            mask = np.arange(n) != src
            dest = int(np.argmin(np.where(mask, cur, np.inf)))
            # occasionally inject randomness
            if rng.random() < 0.20:
                dest = int(rng.integers(0, n - 1))
                if dest >= src:
                    dest += 1
        else:
            # pure random transfer (keeps exploration alive)
            src = int(rng.integers(0, n))
            dest = int(rng.integers(0, n - 1))
            if dest >= src:
                dest += 1

        if src == dest or cur[src] <= 0.0:
            continue

        # ----------------------------------------------------------
        # 4b. Transfer a modest fraction of the source mass
        # ----------------------------------------------------------
        frac = rng.uniform(0.01, 0.25)
        delta = cur[src] * frac
        new = cur.copy()
        new[src] -= delta
        new[dest] += delta

        # ----------------------------------------------------------
        # 4c. Score the new candidate (sum stays ~1, so we reuse the fast formula)
        # ----------------------------------------------------------
        new_max = _conv_max_fft(new, fft_len)
        new_score = 2.0 * n * new_max

        # ----------------------------------------------------------
        # 4d. Acceptance test (simulated annealing)
        # ----------------------------------------------------------
        remaining = max(0.0, deadline - time.monotonic())
        temp = max_temp * (remaining / max(total_time, 1e-12))   # linear cooling
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability of taking a worsening move
            prob = math.exp(-(new_score - cur_score) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

    # --------------------------------------------------------------
    # 5. Return the best sequence found (converted to plain Python list)
    # --------------------------------------------------------------
    final = [float(x) for x in best]
    # final sanity check – the authoritative evaluator clips internally
    try:
        _ = evaluate_sequence(final)
    except Exception:                     # extremely unlikely, but safe fallback
        final = [1.0 / n] * n
    return final
