def compute_bound(seq, n, fft_len, conv_len):
    # seq: np.ndarray float.
    # Use rfft of seq with length fft_len.
    fs = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(fs*fs, fft_len)
    max_val = conv[:conv_len].max()
    sum_val = seq.sum()
    if sum_val < 0.01: return np.inf
    return 2.0 * n * max_val / (sum_val * sum_val)

import numpy as np
import time
import math

def _sidon_like_sequence(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4**i, length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a

def _fast_bound(seq: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Fast AC1 bound using FFT (no safety checks)."""
    s = seq.sum()
    if s < 0.01:
        return math.inf
    F = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(F * F, fft_len)
    m = conv[:conv_len].max()
    return 2.0 * n * m / (s * s)
