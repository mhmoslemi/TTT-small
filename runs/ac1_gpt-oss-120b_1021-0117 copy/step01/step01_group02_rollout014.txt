import time
import numpy as np

def _score(seq: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative sequence."""
    n = seq.shape[0]
    L = 2 * n - 1
    # next power of two for efficient FFT
    N = 1 << (L - 1).bit_length()
    X = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(X * X, n=N)[:L]
    max_b = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a non‑negative step‑function that improves the AC1 bound.

    The routine runs a fast hill‑climbing search that repeatedly moves a
    small portion of mass from the two indices that contribute most to the
    current convolution peak towards the two smallest entries.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------ #
    # initial sequence -------------------------------------------------- #
    try:
        n = int(len(height_sequence_1))
        cur = np.array(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        n = 5000
        cur = np.empty(n, dtype=np.float64)
        cur[0] = 1.0
        for i in range(1, n):
            cur[i] = cur[i - 1] * (1.0 - 1.0 / (2.0 * i))
    # Clip, force non‑negative and normalise (scale does not affect score)
    cur = np.clip(cur, 0.0, 1000.0)
    s = cur.sum()
    if s < 1e-12:
        cur = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        cur = cur / s

    best = cur.copy()
    best_score = _score(best)

    # optional uniform fallback (score = 2)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = _score(uniform)
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score

    # ------------------------------------------------------------------ #
    # parameters for the stochastic search ------------------------------- #
    step_factor = 0.30          # fraction of the smaller source entry moved
    min_factor = 0.005
    decay = 0.9995              # decay applied when a move is not improving
    target_score = 1.5

    # pre‑compute FFT length (constant because n does not change)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()

    # ------------------------------------------------------------------ #
    while time.monotonic() < deadline:
        # compute full self‑convolution (FFT based, O(N log N))
        X = np.fft.rfft(best, n=N)
        conv = np.fft.irfft(X * X, n=N)[:L]

        k_max = int(conv.argmax())

        # indices that participate in the maximal convolution term
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        prod = best[idx] * best[k_max - idx]

        # source pair (the two entries whose product dominates the max)
        i_src = int(idx[np.argmax(prod)])
        j_src = k_max - i_src

        # two smallest entries – candidates for receiving mass
        smallest = np.argpartition(best, 2)[:2]
        dests = []
        for d in smallest:
            if d not in (i_src, j_src):
                dests.append(d)
        # fill missing destinations with random low‑mass indices
        while len(dests) < 2:
            cand = int(rng.integers(0, n))
            if cand not in (i_src, j_src) and cand not in dests:
                dests.append(cand)
        d1, d2 = dests[0], dests[1]

        # amount to move – limited by the smaller source entry
        src_val = best[i_src]
        partner_val = best[j_src] if j_src != i_src else src_val
        base = min(src_val, partner_val)
        delta = base * step_factor
        if delta < 1e-12:
            delta = base * min_factor
        if delta <= 0.0:

        # build candidate by moving delta from the source pair to the dest pair
        cand = best.copy()
        cand[i_src] -= delta

        # renormalise to keep the total mass fixed (scale invariance)
        s = cand.sum()
        if s <= 0.0:
