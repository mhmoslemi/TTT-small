def targeted_move(vec):
    # compute conv and max index
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
    m = int(conv.argmax())
    # compute pair contributions
    idx = np.arange(n)
    idx2 = m - idx
    mask = (idx2 >= 0) & (idx2 < n)
    if not mask.any():
        return vec
    prod = np.zeros_like(vec)
    prod[mask] = vec[mask] * vec[idx2[mask]]
    # Weighted choice.
    prob = prod[mask] / prod[mask].sum()
    # choose source index from masked indices based on prob
    source_candidates = idx[mask]
    i = rng.choice(source_candidates, p=prob)
    j = int(m - i)
    # compute delta
    delta = min(vec[i], vec[j]) * rng.random() * 0.3  # upto 30% of min
    if delta <= 0:
        return vec
    # subtract
    vec[i] -= delta
    vec[j] -= delta
    # add to two random dest indices (maybe same)
    d1 = rng.integers(0, n)
    d2 = rng.integers(0, n)
    vec[d1] += delta/2
    vec[d2] += delta/2
    # ensure nonnegativity (may be zero)
    # normalize to sum=1
    sumv = vec.sum()
    if sumv==0:
        vec[:] = 1.0/n
    else:
        vec /= sumv
    return vec

import time
import numpy as np

# ------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex.
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Convolution max using FFT (vector assumed non‑negative, sum≈1)
# ------------------------------------------------------------
def _conv_max_fft(a: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())
