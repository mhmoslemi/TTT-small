import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _score(seq: np.ndarray, fft_len: int, n: int):
    """Return (AC1 score, full convolution, max value)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    max_val = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float("inf"), conv, max_val
    return 2.0 * n * max_val / (s * s), conv, max_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that seeks to
    minimize the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # --------------------------------------------------------------
    # Determine problem size and possibly a parent construction.
    # --------------------------------------------------------------
    n = 3809  # fallback length
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            arr.ndim == 1
            and arr.size > 0
            and np.isfinite(arr).all()
        ):
            parent = np.clip(arr, 0.0, 1000.0)
            n = int(parent.size)
    except Exception:
        parent = None

    # --------------------------------------------------------------
    # Initial solution – binomial seed or supplied parent.
    # --------------------------------------------------------------
    if parent is not None:
        best = parent.copy()
    else:
        best = _initial_binomial(n)

    fft_len = 1 << ((2 * n - 1).bit_length())
    best_score, best_conv, _ = _score(best, fft_len, n)

    # If the provided parent is better than the binomial start, use it.
    if parent is not None:
        p_score, p_conv, _ = _score(parent, fft_len, n)
        if p_score < best_score:
            best = parent.copy()
            best_score = p_score
            best_conv = p_conv

    # --------------------------------------------------------------
    # Simulated‑annealing style hill‑climber.

    current = best.copy()
    current_score = best_score

    start_time = time.monotonic()

                best_score = cand_score

        # optional early stop if target reached (safety)
        if best_score <= 1.5028:

    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------
    return [float(x) for x in best]
