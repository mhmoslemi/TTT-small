def propose_candidate(seed=42, budget_s=1100):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # length
    n = int(np.asarray(height_sequence_1).size) if 'height_sequence_1' in globals() else 256

    # initialization: sqrt series normalized to sum 1.
    def sqrt_series(n):
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for k in range(1, n):
            if k == 1:
                s = 0.0
            else:
                s = float(np.dot(a[1:k], a[k-1:0:-1]))
            a[k] = (1.0 - s) / 2.0
        return a

    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=np.float64)
        # ensure positivity: as per evaluator, later clipped
        # Normalize sum to 1
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0:
            a0 = sqrt_series(n)
            a0 /= a0.sum()
        else:
            a0 = a0 / s
    else:
        a0 = sqrt_series(n)
        a0 /= a0.sum()

    best = a0.copy()
    # compute initial score
    L = 1 << ((2 * n - 1).bit_length())
    def compute_score(vec):
        fv = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        return 2.0 * n * conv.max() / (vec.sum() ** 2)
    best_score = compute_score(best)

    # main loop
    iter_count = 0
    while time.monotonic() < deadline:
        iter_count += 1
        # compute current conv and max index only if needed
        # Choose type of move: gradient vs random
        if rng.random() < 0.6:
            # gradient move
            fv = np.fft.rfft(best, n=L)
            conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
            max_idx = int(conv.argmax())
            # gradient vector

            low = max(0, k - (n - 1))
            high = min(k, n - 1)

            # add a[k-j] to g[j]
            # create view of a reversed segment

            for j in range(low, high + 1):

            # Now choose source with highest g and target with lowest g

            target = int(np.argmin(g))
            if source == target:

                j = rng.integers(0, n-1)
                if j >= i: j+=1
                source, target = i, j
            # fractional transfer

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative floats (a step‑function)
    that tries to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # a small safety buffer so we can return before the sandbox terminates
    total_time = max(0.0, float(budget_s) - 4.0)
    deadline = start + total_time

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        height_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = height_seq.size
    else:
        n = 256
        height_seq = None

    # ------------------------------------------------------------------
    # helpers
    # ------------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())     # FFT length (next power of two)

    def _conv_max(vec: np.ndarray):
        """Full convolution of vec with itself; return max value, its index and the whole conv."""
        fv = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        idx = int(conv.argmax())
        return float(conv.max()), idx, conv

    # ------------------------------------------------------------------
    # initial vectors (both the analytic sqrt‑series and the provided parent)
    # ------------------------------------------------------------------
    def _sqrt_series(m: int) -> np.ndarray:
        a = np.empty(m, dtype=np.float64)
        a[0] = 1.0
        for k in range(1, m):
            if k == 1:
                s = 0.0
            else:
                s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
            a[k] = (1.0 - s) / 2.0
        return a

    # start from sqrt‑series, normalised to sum = 1
    base = _sqrt_series(n)
    base = np.maximum(base, 0.0)
    base /= base.sum()
    best = base.copy()
    best_score = None
    best_conv = None
    best_max_idx = None

    # optionally compare with the provided parent construction
    if height_seq is not None:
        cand = np.maximum(height_seq, 0.0)
        if cand.sum() == 0.0:
            cand = base.copy()
        else:
            cand = cand / cand.sum()
        mv, mi, conv = _conv_max(cand)
        sc = 2.0 * n * mv / (cand.sum() ** 2)
        best = cand
        best_score = sc
        best_conv = conv
        best_max_idx = mi

    # ensure we have a score for the sqrt‑series baseline as well
    mv, mi, conv = _conv_max(base)
    sc = 2.0 * n * mv / (base.sum() ** 2)
    if best_score is None or sc < best_score:
        best = base.copy()
        best_score = sc
        best_conv = conv
        best_max_idx = mi

    # ------------------------------------------------------------------
    # main stochastic search
    # ------------------------------------------------------------------
    no_improve = 0
    while time.monotonic() < deadline:
        # linearly decreasing maximal transfer fraction (≈0.3 → 0.001)
        elapsed = time.monotonic() - start
        frac_schedule = 0.3 * (1.0 - elapsed / total_time) + 0.001

        # --------------------------------------------------------------
        # decide move type
        # --------------------------------------------------------------
        if rng.random() < 0.55:                     # gradient‑guided move
            k = best_max_idx
            # gradient g_j = a_j + a_{k-j} (if k-j inside range)
            g = best.copy()
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            for j in range(low, high + 1):
                g[j] += best[k - j]
            src = int(np.argmax(g))
            dst = int(np.argmin(g))
            if src == dst:
                src = rng.integers(0, n)
                dst = rng.integers(0, n - 1)
                if dst >= src:
                    dst += 1
        else:                                        # simple random move
            src = rng.integers(0, n)
            if best[src] <= 0.0:
                continue
            dst = rng.integers(0, n - 1)
            if dst >= src:
                dst += 1

        # transfer a random fraction of the source mass
        if best[src] <= 0.0:
            continue
        delta = best[src] * (rng.random() * frac_schedule)
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta

        # --------------------------------------------------------------
        # evaluate candidate (single FFT)
        # --------------------------------------------------------------
        mv, mi, conv = _conv_max(cand)
        sc = 2.0 * n * mv / (cand.sum() ** 2)

        if sc < best_score:
            best = cand
            best_score = sc
            best_conv = conv
            best_max_idx = mi
            no_improve = 0
            # early exit if we already beat the target
            if best_score < 1.50274349:

                pass
        else:

            noise /= noise.sum()
            best = (1.0 - alpha) * best + alpha * noise

            best_score = 2.0 * n * mv / (best.sum() ** 2)
            best_conv = conv
            best_max_idx = mi
            no_improve = 0

    # ------------------------------------------------------------------
