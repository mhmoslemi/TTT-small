import time

def _catalan_weights(n: int) -> np.ndarray:
    """Return the first n coefficients w_i = C(2i,i)/4**i (Catalan‑style)."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _apply_transfer(a, b, i_src, i_dst, eps):
    """Move eps from a[i_src] to a[i_dst] and update convolution b in‑place."""
    # store old values (caller must keep them for possible revert)
    ai = a[i_src]
    aj = a[i_dst]

    # update a
    a[i_src] = ai - eps
    a[i_dst] = aj + eps

    # convolution indices
    idx2i = i_src + i_src
    idx2j = i_dst + i_dst
    idxmid = i_src + i_dst
    eps2 = eps * eps

    # update b according to the exact formula derived from (a+Δ)*(a+Δ)
    b[idx2i] += -2.0 * eps * ai + eps2
    b[idx2j] += 2.0 * eps * aj + eps2
    b[idxmid] += 2.0 * eps * (ai - aj) - 2.0 * eps2


def _revert_transfer(a, b, i_src, i_dst, old_ai, old_aj,
                     old_b2i, old_b2j, old_bmid):
    """Restore a and b to the state saved before a transfer."""
    a[i_src] = old_ai
    a[i_dst] = old_aj
    b[i_src + i_src] = old_b2i
    b[i_dst + i_dst] = old_b2j
    b[i_src + i_dst] = old_bmid


# ------------------------------------------------------------
# main routine
# ------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises the AC1 upper bound.  A simple gradient‑guided hill‑climb
    is employed; it works directly on the convolution values, updating
    them in O(1) per move.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 1.  Obtain a base sequence (parent if available, otherwise Catalan)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = None

    n = 0
    candidates = []

    # Catalan candidate
    if parent is None or parent.size == 0:
        n = 256
    else:
        n = int(parent.size)

    w = _catalan_weights(n)
    a_catalan = w / w.sum()
    candidates.append(a_catalan)

    # parent candidate (if present)
    if parent is not None:
        a_parent = parent / parent.sum()
        candidates.append(a_parent)

    # pick the best of the two as the starting point
    best = None
    best_score = float('inf')
    for cand in candidates:
        # use fast internal score (sum is 1 by construction)
        b_full = np.convolve(cand, cand)
        score = 2.0 * n * b_full.max()
        if score < best_score:
            best_score = score
            best = cand.copy()
    # ------------------------------------------------------------------
    # 2.  Initialise convolution for the current best
    # ------------------------------------------------------------------
    a = best.copy()
    b = np.convolve(a, a)            # full convolution, length 2n‑1
    best = a.copy()
    best_score = 2.0 * n * b.max()

    target_score = 1.50274349

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # a)  Gradient‑guided move: reduce the current maximal lag
        # --------------------------------------------------------------
        if rng.random() < 0.68:
            k_max = int(b.argmax())
            # indices that contribute to this lag
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            partner = k_max - idx
            prod = a[idx] * a[partner]
            if prod.size == 0:
                continue
            src_local = idx[np.argmax(prod)]
            src = int(src_local)
            prt = int(k_max - src)          # partner index

            # destination = smallest entry not src or partner
            mask = np.ones(n, dtype=bool)
            mask[src] = False
            mask[prt] = False
            # use a large sentinel for masked positions
            sentinel = np.full_like(a, np.inf)
            sentinel[mask] = a[mask]
            dst = int(np.argmin(sentinel))

            if src == dst or a[src] <= 0.0:
                continue

            # move a reasonable fraction of the source mass
            eps = rng.random() * a[src] * 0.35  # up to 35 % of src
            if eps <= 0.0:
                continue

            # keep old state for possible revert
            old_ai = a[src]
            old_aj = a[dst]
            old_b2i = b[src + src]
            old_b2j = b[dst + dst]
            old_bmid = b[src + dst]

            _apply_transfer(a, b, src, dst, eps)

            new_score = 2.0 * n * b.max()
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                if best_score <= target_score:
                    break
            else:
                _revert_transfer(a, b, src, dst,
                                 old_ai, old_aj,
                                 old_b2i, old_b2j, old_bmid)

        # --------------------------------------------------------------
        # b)  Random mass‑transfer move (exploratory)
        # --------------------------------------------------------------
        else:
            i_src = rng.integers(n)
            if a[i_src] <= 0.0:
                continue
            i_dst = rng.integers(n - 1)
            if i_dst >= i_src:
                i_dst += 1
            # ensure distinct indices
            eps = rng.random() * a[i_src] * 0.5
            if eps <= 0.0:
                continue

            old_ai = a[i_src]
            old_aj = a[i_dst]
            old_b2i = b[i_src + i_src]
            old_b2j = b[i_dst + i_dst]
            old_bmid = b[i_src + i_dst]

            _apply_transfer(a, b, i_src, i_dst, eps)

            new_score = 2.0 * n * b.max()
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                if best_score <= target_score:
                    break
            else:
                _revert_transfer(a, b, i_src, i_dst,
                                 old_ai, old_aj,
                                 old_b2i, old_b2j, old_bmid)

        # --------------------------------------------------------------
        # c)  Occasional global jitter (helps escape flat plateaux)
        # --------------------------------------------------------------
        if rng.random() < 0.008:
            jitter = np.exp(rng.normal(0.0, 0.012, size=n))
            a_j = a * jitter
            a_j /= a_j.sum()
            b_j = np.convolve(a_j, a_j)
            score_j = 2.0 * n * b_j.max()
            if score_j < best_score:
                best_score = score_j
                best = a_j.copy()
                a = a_j
                b = b_j
                if best_score <= target_score:
                    break

    # ------------------------------------------------------------------
    # 4.  Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
