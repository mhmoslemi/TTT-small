    grad = np.zeros_like(a)

    n = a.shape[0]
    rows = []
    cols = []
    data = []
    # for each k in indices:
    for row, k in enumerate(indices):
        start = max(0, k - (n-1))
        end = min(n-1, k)
        # indices i from start..end (i corresponds to a[i], g[k-i])
        i_vals = np.arange(start, end+1)
        j_vals = k - i_vals
        rows.extend([row]*len(i_vals))
        cols.extend(j_vals.tolist())
        data.extend(a[i_vals].tolist())
    # Build sparse matrix using scipy.sparse.csr_matrix
    A = sp.csr_matrix((data, (rows, cols)), shape=(len(indices), n))
    g = cp.Variable(n)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if prob.status in [cp.OPTIMAL, cp.FEASIBLE]:
        gval = g.value
        if gval is not None:
            # renormalise to sum 1
            s = np.sum(gval)
            if s > 0:
                gval = gval / s
                return gval
    return None

import scipy.sparse as sp
import cvxpy as cp

TARGET = 1.50274349  # early‑stop target


def _objective(seq: np.ndarray) -> float:
    """Exact objective matching evaluate_sequence (scale‑invariant)."""
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    n = seq.size
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _lp_step(a: np.ndarray, M: float, ks: np.ndarray) -> np.ndarray | None:
    """
    Linear‑programming sub‑step.
    Find a non‑negative vector g with larger ℓ₁‑norm than a
    while keeping (a * g)[k] ≤ M for all k in ks.
    Return the normalised vector (sum = 1) or None on failure.
    """
    n = a.shape[0]
    rows, cols, data = [], [], []
    for row, k in enumerate(ks):
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        i = np.arange(start, end + 1, dtype=np.int64)
        j = k - i
        rows.extend([row] * i.size)
        cols.extend(j.tolist())
        data.extend(a[i].tolist())
    if not rows:  # no constraints – shouldn't happen
        return None

    C = sp.csr_matrix((data, (rows, cols)), shape=(ks.size, n), dtype=np.float64)

    g = cp.Variable(n, nonneg=True)
    constraints = [C @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
        if prob.status not in (cp.OPTIMAL, cp.FEASIBLE):
            return None
        gv = g.value
        if gv is None:
            return None
        total = float(gv.sum())
        if total <= 0.0:
            return None
        return gv / total
    except Exception:
        return None

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():

        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = None

    n = 2535 if parent is None else int(parent.size)

    # Catalan start
    w = _catalan_weights(n)
    a0 = w / w.sum()                     # sum = 1
    best = a0.copy()
    best_score = _objective(best)

    # optionally start from the given parent
    if parent is not None and parent.sum() > 0:
        parent_norm = parent / parent.sum()
        sc = _objective(parent_norm)
        if sc < best_score:
            best = parent_norm.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # 2. Parameters for the stochastic‑gradient search
    # ------------------------------------------------------------------
    step_frac = 0.05          # basic step size factor
    min_step = 1e-4
    grow = 1.2
    shrink = 0.5
    lp_counter = 0
    max_lp_calls = 200        # keep LP usage modest

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----------- compute convolution and identify critical indices -----
        conv = np.convolve(best, best)
        max_val = conv.max()
        # gradient based on all near‑max entries (within 99% of the peak)
        active = np.where(conv >= max_val * 0.99)[0]

        # --------------- 3a. Gradient‑style move ------------------------
        # sub‑gradient of the max (sum of reversed slices over active ks)
        grad = np.zeros_like(best)
        for k in active:
            start = max(0, k - (n - 1))
            end = min(n - 1, k)
            i = np.arange(start, end + 1, dtype=np.int64)
            j = k - i
            grad[i] += best[j]
        grad *= 2.0

        if grad.max() > 0:
            step = step_frac * (best.max() / grad.max())
            cand = best - step * grad
            np.clip(cand, 0.0, None, out=cand)
            s = cand.sum()
            if s > 0.0:
                cand /= s
                sc = _objective(cand)
                if sc < best_score:
                    best, best_score = cand, sc
                    step_frac = min(step_frac * grow, 1.0)
                    if best_score <= TARGET:
                        break
                else:
                    step_frac = max(step_frac * shrink, min_step)

        # --------------- 3b. Random pairwise mass transfer ---------------
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] > 0.0:
            eps = rng.random() * best[i] * 0.5
            cand = best.copy()
            cand[i] -= eps
            cand[j] += eps
            # sum unchanged, only numerical drift possible
            sc = _objective(cand)
            if sc < best_score:
                best, best_score = cand, sc
                if best_score <= TARGET:
                    break

        # --------------- 3c. Global jitter (multiplicative) -------------
        if rng.random() < 0.02:
            jitter = np.exp(rng.normal(0.0, 0.018, size=n))
            cand = best * jitter
            cand_sum = cand.sum()
            if cand_sum > 0.0:
                cand /= cand_sum
                sc = _objective(cand)
                if sc < best_score:
                    best, best_score = cand, sc
                    if best_score <= TARGET:
                        break

        # --------------- 3d. Sparse LP refinement (occasionally) -------
        if lp_counter < max_lp_calls and rng.random() < 0.08:
            # use the same active set as constraints
            new_seq = _lp_step(best, max_val, active)
            lp_counter += 1
            if new_seq is not None:
                sc = _objective(new_seq)
                if sc < best_score:
                    best, best_score = new_seq, sc
                    if best_score <= TARGET:
                        break

    # ------------------------------------------------------------------
    # 4. Return the best found sequence as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best]
