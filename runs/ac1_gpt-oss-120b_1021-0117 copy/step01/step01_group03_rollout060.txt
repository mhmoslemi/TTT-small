import numpy as np, time, math

def _conv_self(a):
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv

def _score(a):
    n = a.shape[0]
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

def _sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2.0 * (i-1) + 1.0) / (2.0 * i)
    return a

def _geom_series(n, r):
    # a_i = r**i, normalized not required
    # ensure first element not zero
    exponents = np.arange(n, dtype=np.float64)
    return r ** exponents

def _power_series(n, p):
    # a_i = (i+1)^{-p}
    idx = np.arange(1, n+1, dtype=np.float64)
    return idx ** (-p)

def _initial_candidates(n, parent):
    cand_list = []
    # parent if provided
    if parent is not None:
        cand_list.append(parent.copy())
    # sqrt series
    cand_list.append(_sqrt_series(n))
    # geometric families
    for r in (0.999, 0.9995, 0.9999, 0.995, 0.99):
        cand = _geom_series(n, r)
        cand_list.append(cand)
    # power families
    for p in (0.5, 1.0, 1.5, 2.0, 2.5):
        cand = _power_series(n, p)
        cand_list.append(cand)
    return cand_list

    # allow early cutoff for overhead

    else:

        base = None

    # Prepare a list of diverse starting vectors
    starts = _initial_candidates(n, base)

    # Pick the best among them as current best
    best = starts[0].copy()
    best_score = _score(best)
    for cand in starts[1:]:
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # Main stochastic improvement loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = best.copy()
        op = rng.random()

        if op < 0.45:
            # Targeted transfer: move mass from indices contributing to the

            conv = _conv_self(cand)
            kmax = int(np.argmax(conv))
            low = max(0, kmax - (n - 1))
            high = min(n - 1, kmax)
            # source indices inside the active window
            width = high - low + 1
            if width > 0:
                src_slice = cand[low:high+1]
                # pick a relatively large source

                src = low + src_rel

import numpy as np, time, math

def _conv_self(a):
    """Fast linear convolution a*a via real FFT."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score(a):
    """Exact score used by the evaluator."""
    n = a.shape[0]
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _sqrt_series(n):
    """Classic a_i = C(2i,i)/4^i sequence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


def _geom_series(n, r):
    """Geometric decay a_i = r**i."""
    return r ** np.arange(n, dtype=np.float64)


def _power_series(n, p):
    """Power‑law decay a_i = (i+1)**(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _initial_candidates(n, parent):
    """A handful of diverse starts."""
    cand = []
    if parent is not None:
        cand.append(parent.copy())

        cand.append(_geom_series(n, r))
    for p in (0.5, 1.0, 1.5, 2.0, 2.5):
        cand.append(_power_series(n, p))
    return cand
