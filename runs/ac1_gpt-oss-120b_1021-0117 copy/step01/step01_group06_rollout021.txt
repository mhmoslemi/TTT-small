import time
import numpy as np

# ------------------------------------------------------------
# fallback analytic start (sqrt series) in case no parent given
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    # keep a small safety margin before the hard sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # -----------------------------------------------------------------
    # initialise the sequence (use provided parent if present, else fallback)
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0.0)
    else:
        n = 256
        a = _sqrt_series(n)

    # normalise to sum == 1 (scale‑invariant for the bound)
    s = a.sum()
    if s <= 0.0:
        a = np.ones_like(a) / a.size
    else:
        a = a / s

    n = a.size
    # ------------------------------------------------------------
    # full convolution (FFT) – used for periodic re‑normalisation
    # ------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())
    def _full_conv(v):
        fv = np.fft.rfft(v, n=L)
        return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]

    conv = _full_conv(a)
    best = a.copy()
    best_score = 2.0 * n * conv.max()

    # ------------------------------------------------------------
    # helper: incremental update of the self‑convolution after moving delta
    # ------------------------------------------------------------
    def _updated_conv(old_conv, vec, i, j, delta):

        new_conv = old_conv.copy()

        new_conv[2 * i] += delta * delta
        new_conv[2 * j] += delta * delta
        new_conv[i + j] -= 2.0 * delta * delta

    # ------------------------------------------------------------
    max_frac = 0.5          # up to 50 % of source mass may be moved
