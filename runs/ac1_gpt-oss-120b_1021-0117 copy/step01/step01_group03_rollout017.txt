import numpy as np
import time

# ----------------------------------------------------------------------
# Convolution and scoring utilities (identical to the reference implementation)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT (O(n log n))."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the authoritative evaluator."""
    n = a.size
    conv = _conv_self(a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    # leave a few seconds for the harness to finish cleanly
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initial sequence – use the supplied parent if present, otherwise the
    # classic sqrt series.  Normalise to L1‑norm = 1 (scale invariance).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = a.size
    else:
        # fallback length (the hidden parent length is 5582)
        n = 5582
        a = None

    if a is None:
        a = _sqrt_series(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total                         # enforce Σa = 1

    best = a.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # Helper: gradient of the maximum‑convolution entry.
    # ------------------------------------------------------------------
    def _max_conv_gradient(x: np.ndarray, k_max: int) -> np.ndarray:
        """Return ∇(conv_kmax) where conv = x*x."""
        # indices i for which 0 ≤ i < n and 0 ≤ k_max‑i < n
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(x)
        grad[idx] = 2.0 * x[k_max - idx]
        return grad

    it = 0
    # main loop
    while time.monotonic() < deadline:
        # ----- evaluate current candidate -----
        cur_score = _score(a)
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score

        # ----- locate the worst (largest) convolution entry -----
        conv = _conv_self(a)
        k_max = int(conv.argmax())

        # ----- stochastic control of which mutation to apply -----
        op = rng.random()

        if op < 0.55:
            # ----- deterministic gradient‑type mass transfer -----
            grad = _max_conv_gradient(a, k_max)

            # step size decays slowly with the iteration count
            base_step = 0.30
            eta = base_step / (1.0 + it * 1e-4)

            cand = a - eta * grad
            np.clip(cand, 0.0, None, out=cand)

        np.clip(cand, 0.0, None, out=cand)

        if s < 1e-12:                # pathological – reset to uniform

    # ------------------------------------------------------------------
