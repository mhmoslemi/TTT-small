import time
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Optional analytic start (sqrt series) – kept for reference
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# Core function: propose a candidate sequence
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)     # safety margin

    # -----------------------------------------------------------------
    # Obtain problem size and an initial feasible vector (sum = 1)
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = np.ones(256, dtype=np.float64)          # fallback length
    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 <= 0.0:
        n = a0.size
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / s0                                 # normalise sum to 1
    n = a0.size

    # -----------------------------------------------------------------
    # Pre‑compute FFT length (power of two) for convolution
    # -----------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # -----------------------------------------------------------------
    # Convolution statistics (max value and its index) using FFT
    # -----------------------------------------------------------------
    def _conv_max_and_idx(v: np.ndarray):
        fv = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        max_idx = int(conv.argmax())
        return float(conv[max_idx]), max_idx

    # -----------------------------------------------------------------
    # Initial solution (start from the provided parent)
    # -----------------------------------------------------------------
    cur = a0.copy()
    cur_max, cur_k = _conv_max_and_idx(cur)
    cur_sc = 2.0 * n * cur_max / (cur.sum() ** 2)   # sum(cur)==1 ⇒ 2*n*cur_max
    best = cur.copy()
    best_sc = cur_sc

    # Sometimes the sqrt‑series (unscaled) is a better baseline
    # (its shape may be superior for certain lengths).
    try:
        sqrt_vec = _sqrt_series(n)
        sqrt_vec = np.maximum(sqrt_vec, 0.0)
        sqrt_vec /= sqrt_vec.sum()
        smax, _ = _conv_max_and_idx(sqrt_vec)
        ssc = 2.0 * n * smax / (sqrt_vec.sum() ** 2)
        if ssc < best_sc:
            best = sqrt_vec.copy()
            best_sc = ssc
            cur = sqrt_vec.copy()
            cur_sc = ssc
    except Exception:
        pass  # ignore if sqrt series generation fails for this n

    # -----------------------------------------------------------------
    # Main search loop (projected random moves + gradient steps)
    # -----------------------------------------------------------------
    it = 0
    stagnated = 0
    while time.monotonic() < deadline:
        it += 1

        # ----- random mass transfer (70% of iterations) -----
        if rng.random() < 0.70:
            i = rng.integers(0, n)
            if cur[i] <= 1e-15:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            frac = rng.random() * 0.20          # up to 20 % of source mass
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            # tiny numerical drift → re‑project onto simplex
            cand = _project_simplex(cand)
        # ----- gradient descent step (30% of iterations) -----
        else:
            # need current max‑index for gradient
            cur_max, cur_k = _conv_max_and_idx(cur)

            # gradient of the active convolution entry (ignoring the constant part)
            grad = np.zeros_like(cur)
            lo = max(0, cur_k - (n - 1))
            hi = min(cur_k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] = 2.0 * cur[cur_k - idx]   # factor 2 from derivative

            # step size shrinks slowly with time
            elapsed = time.monotonic() - start_time
            frac_time = max(0.0, 1.0 - elapsed / (budget_s - 5.0))
            step = rng.random() * 0.08 * frac_time

            cand = cur - step * grad
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_max, _ = _conv_max_and_idx(cand)
        cand_sc = 2.0 * n * cand_max / (cand.sum() ** 2)

        # ----- acceptance rule -----
        if cand_sc < cur_sc or rng.random() < 0.001:   # tiny chance to escape
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best = cand.copy()
                best_sc = cand_sc
                stagnated = 0
            else:
                stagnated += 1
        else:
            stagnated += 1

        # ----- occasional restart if stuck -----
        if stagnated > 5000:
            cur = best.copy()
            cur_sc = best_sc
            stagnated = 0

    # -----------------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    # -----------------------------------------------------------------
    return [float(x) for x in best]
