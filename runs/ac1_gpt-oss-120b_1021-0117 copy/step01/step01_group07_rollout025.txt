def simplex_projection(v):
    # v is np array
    # project onto {x >= 0, sum x = 1}
    # Use algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np, time, math

def simplex_projection(v):
    # v is np.ndarray
    # Implementation like Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def evaluate(seq): # just wrapper for evaluate_sequence (imported)
    return float(evaluate_sequence(seq))

def initial_sequences(n):
    # returns list of candidate sequences (np.ndarray)
    seqs = []
    # parent normalized
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size == n and np.isfinite(parent).all():
            p = np.clip(parent, 0.0, 1000.0)
            p = p / p.sum()
            seqs.append(p)
    except Exception:
        pass
    # uniform
    seqs.append(np.full(n, 1.0/n))
    # Catalan (as previously)
    # compute w
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1,n):
        w[i] = w[i-1]*((2*i)*(2*i-1))/(i*i*4.0)
    a = w / w.sum()
    seqs.append(a)
    # power law
    for alpha in np.linspace(0.5,3.0,11):
        a = (np.arange(1,n+1, dtype=np.float64) ** (-alpha))
        a = a / a.sum()

    # beta shape (U)

        i = np.arange(n)+1

        a = np.sin(np.pi * t) ** gamma
        if a.sum()>0:
            a = a / a.sum()
            seqs.append(a)
    return seqs

import time
import numpy as np

# --------------------------------------------------------------
# Helper: project onto the probability simplex {x >= 0, sum x = 1}
# --------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(v.size) + 1) > 0)[0]
    if rho.size == 0:
        # all entries will be zero
        return np.full_like(v, 1.0 / v.size)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Fast bound calculation (identical to evaluate_sequence for valid a)
# --------------------------------------------------------------
def _bound(a: np.ndarray) -> float:
    """Return 2*n*max(conv(a,a))/sum(a)^2 for a non‑negative vector."""
    s = a.sum()
    if s < 0.01:
        return float("inf")
    b = np.convolve(a, a)
    return 2.0 * a.size * b.max() / (s * s)


# --------------------------------------------------------------
# Gradient of the maximal convolution entry (sub‑gradient)
# --------------------------------------------------------------
def _gradient_at_max(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Gradient of the active max‑convolution entry; also returns its index."""
    n = a.size
    b = np.convolve(a, a)
    k = int(b.argmax())
    # indices i with 0<=i<n and 0<=k-i<n
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g = np.zeros_like(a)
    g[i] = 2.0 * a[j]          # ∂/∂a_i of b_k = 2*a_{k-i}
    return g, k
