def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-5.0)

    # Determine n and optional parent
    if 'height_sequence_1' in globals():
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = base.size
    else:
        n = 5582
        base = None

    def _score(a):
        n_local = a.shape[0]
        m_local = 1 << ((2 * n_local - 1).bit_length())
        fa = np.fft.rfft(a, n=m_local)
        conv = np.fft.irfft(fa * fa, n=m_local)[: 2 * n_local - 1]
        max_b = conv.max()
        s = a.sum()
        if s < 0.01:
            return np.inf
        return 2.0 * n_local * max_b / (s * s)

    # Initialize best
    best = None
    best_score = np.inf

    # sqrt series
    def _sqrt_series(length):
        a = np.empty(length, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, length):
            a[i] = a[i-1] * (2.0 * (i-1) + 1.0) / (2.0 * i)
        return a

    # Evaluate candidates
    if base is not None:
        sc = _score(base)
        if np.isfinite(sc):
            best = base.copy()
            best_score = sc

    sqrt_seq = _sqrt_series(n)
    sc_sqrt = _score(sqrt_seq)
    if sc_sqrt < best_score:
        best = sqrt_seq.copy()
        best_score = sc_sqrt

    # Parameter scans
    indices = np.arange(n, dtype=np.float64)
    indices1 = indices + 1.0

    # power law scan
    for p in np.linspace(0.3, 1.5, 120):
        if time.monotonic() > deadline:
            break
        a = 1.0 / (indices1 ** p)
        sc = _score(a)
        if sc < best_score:
            best = a.copy()
            best_score = sc

    # exponential scan
    for lam in np.linspace(5e-5, 0.02, 200):
        if time.monotonic() > deadline:
            break
        a = np.exp(-lam * indices)
        sc = _score(a)
        if sc < best_score:
            best = a.copy()
            best_score = sc

    # plateau + exponential tail scan
    # try a few plateau fractions
    for frac in (0.0, 0.05, 0.1, 0.2):
        L = int(frac * n)
        tail_len = n - L
        if tail_len <= 0:
            continue
        for lam in np.linspace(5e-5, 0.02, 60):
            if time.monotonic() > deadline:
                break
            a = np.empty(n, dtype=np.float64)
            if L > 0:
                a[:L] = 1.0
            a[L:] = np.exp(-lam * np.arange(tail_len, dtype=np.float64))
            sc = _score(a)
            if sc < best_score:
                best = a.copy()
                best_score = sc

    # Local search / simulated annealing
    cur = best.copy()
    cur_score = best_score

    # Parameters for SA
    init_temp = 0.05
    temp = init_temp
    iter_count = 0

    early_cut = n // 8
    late_cut = n - early_cut

    while True:
        if time.monotonic() > deadline:
            break
        iter_count += 1
        # gradually cool
        if iter_count % 1000 == 0:
            # linear cooling toward zero by fraction
            temp = max(1e-6, temp * 0.995)

        cand = cur.copy()
        op = rng.random()

        if op < 0.55:
            # transfer from early to late
            src = rng.integers(0, early_cut)
            if cand[src] <= 1e-12:
                continue
            dst = rng.integers(late_cut, n)
            max_transfer = min(cand[src], 1000.0 - cand[dst])
            if max_transfer <= 0.0:
                continue
            delta = max_transfer * rng.random() * 0.07   # up to 7%
            cand[src] -= delta
            cand[dst] += delta

        elif op < 0.85:
            # block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = rng.integers(0, max(1, n - width + 1))
            stop = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, 0.07)))
            cand[start:stop] *= factor

        else:
            # localized Gaussian noise
            start = rng.integers(0, n)
            width = int(rng.integers(1, max(2, n // 50)))
            stop = min(n, start + width)
            cand[start:stop] += rng.normal(0.0, 0.015, size=stop-start)

        # enforce bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _score(cand)
        if not np.isfinite(cand_score):
            continue
        delta = cand_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # accept with probability exp(-delta / temp)
            if rng.random() < np.exp(-delta / temp):
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

        # occasional restart to best to avoid drift
        if iter_count % 5000 == 0 and rng.random() < 0.2:
            cur = best.copy()
            cur_score = best_score

    # Return result
    return [float(v) for v in best]

import time
import numpy as np


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution of a real vector using a power‑of‑2 FFT."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the authoritative evaluator."""
    n = a.shape[0]
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _sqrt_series(length: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence (length = length)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a
