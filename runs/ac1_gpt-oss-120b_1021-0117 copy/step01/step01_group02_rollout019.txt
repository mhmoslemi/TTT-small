```
def compute_score_fft(a):
    n = a.shape[0]
    m = 2*n - 1
    L = 1 << ((m-1).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[:m]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01: return np.inf
    return 2.0 * n * max_b / (sum_a*sum_a)
```

```python
import time, math, numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)
    # Sequence length and initial vector
    try:
        # height_sequence_1 is provided by the harness
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.shape[0]
    except Exception:
        n = 4000
        # fallback: binomial-like decreasing sequence
        base = np.empty(n, dtype=np.float64)
        base[0] = 1.0
        for i in range(1, n):
            base[i] = base[i-1] * (1.0 - 1.0/(2.0*i))
    # initial candidate: clip and normalise to unit sum
    a = np.clip(base, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s   # sum==1
    # alternative fallback uniform
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    # helper: length for FFT
    m = 2*n - 1
    L = 1 << ((m-1).bit_length())   # next power of two
    # scoring using FFT convolution
    def _score(v):
        fft_v = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(fft_v * fft_v, n=L)[:m]
        max_b = conv.max()
        sum_v = v.sum()
        if sum_v < 0.01:
            return np.inf
        return 2.0 * n * max_b / (sum_v*sum_v)
    # evaluate initial candidates
    best = a.copy()
    best_score = _score(best)
    uniform_score = _score(uniform)
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
    # optimisation parameters
    step_factor = 0.15
    min_step = 1e-9
    no_improve = 0
    shake_prob = 0.008
    max_iter = 10**9  # effectively unlimited, controlled by time
    iter_cnt = 0
    # main loop
    while time.monotonic() < deadline and iter_cnt < max_iter:
        iter_cnt += 1
        # convolution via FFT
        fft_best = np.fft.rfft(best, n=L)
        conv_full = np.fft.irfft(fft_best * fft_best, n=L)[:m]
        cur_max = conv_full.max()
        k = int(conv_full.argmax())
        # partner vector (shifted copy of best)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        partner = np.zeros(n, dtype=np.float64)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            partner[idx] = best[k - idx]
        # contribution product
        product = best * partner
        src = int(product.argmax())
        dest = int(product.argmin())
        if src == dest:
            # Choose a different dest arbitrarily
            dest = int(rng.integers(0, n))
            while dest == src:
                dest = int(rng.integers(0, n))
        # compute delta to move
        if best[src] <= 0.0:
            delta = 0.0
        else:
            delta = step_factor * best[src]
            if delta < min_step:
                delta = min_step
            delta = min(delta, best[src])
        if delta > 0.0:
            cand = best.copy()
            cand[src] -= delta
            cand[dest] += delta
            # clip and renormalise to unit sum
            cand = np.clip(cand, 0.0, 1000.0)
            s_cand = cand.sum()
            if s_cand > 0:
                cand = cand / s_cand * best.sum()
            cand_score = _score(cand)
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                no_improve = 0

    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)
    return [float(v) for v in final]
```

import time
import numpy as np

def _conv_fft(a: np.ndarray, L: int, m: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length m = 2*len(a)-1)."""
    fft_a = np.fft.rfft(a, n=L)
    return np.fft.irfft(fft_a * fft_a, n=L)[:m]

def _score(a: np.ndarray, n: int, L: int, m: int) -> float:
    """AC1 upper bound for a (sum should be >0)."""
    conv = _conv_fft(a, L, m)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step‑function that minimizes the AC1 bound."""
    rng = np.random.default_rng(seed)
    # a short safety margin before the sandbox kills the process
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # --------------------------------------------------------------

            base[i] = base[i - 1] * (1.0 - 1.0 / (2.0 * i))

    a = np.clip(base, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:

    else:
        a = a / s                     # normalise to sum == 1

    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # FFT layout (next power of two covering linear convolution length)
    m = 2 * n - 1
    L = 1 << ((m - 1).bit_length())   # next power of two >= m

        best_score = uniform_score

    # --------------------------------------------------------------

    step_factor = 0.15
    min_step = 1e-9
    no_improve = 0
    shake_prob = 0.008

    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution and the location of its maximum -------------
        conv = _conv_fft(best, L, m)
        k = int(conv.argmax())

        # ----- partner vector: the shifted copy of best that forms conv[k] -----
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        partner = np.zeros(n, dtype=np.float64)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            partner[idx] = best[k - idx]

        # ----- contribution of each index to conv[k] -----------------
        product = best * partner
        src = int(product.argmax())
        dest = int(product.argmin())
        if src == dest:
            dest = int(rng.integers(0, n))
            while dest == src:
                dest = int(rng.integers(0, n))

        # ----- propose moving a fraction of mass from src to dest -----
        delta = 0.0
        if best[src] > 0.0:

            if delta < min_step:

            delta = min(delta, best[src])

        if delta > 0.0:

        final = np.full_like(final, 1.0 / final.size)
