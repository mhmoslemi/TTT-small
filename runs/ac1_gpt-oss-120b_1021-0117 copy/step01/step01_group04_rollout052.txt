import time, math
import numpy as np

# -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """Classic non‑negative sequence a_i = C(2i,i) / 4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 upper bound using FFT; returns 2·n·max(conv)/sum²."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)

            target = M / S

            # valid index range for the pair i + j = k0
            left = max(0, k0 - (n - 1))
            right = min(n - 1, k0)
            idx = np.arange(left, right + 1)
            partners = k0 - idx
            partner_vals = cur[partners]
            diffs = partner_vals - target

            high_mask = (diffs > 0) & (cur[idx] > 0)
            low_mask = diffs < 0
            if high_mask.any() and low_mask.any():
                high_idx = idx[high_mask]
                high_diff = diffs[high_mask]
                i_high = high_idx[np.argmax(high_diff)]

                low_idx = idx[low_mask]
                low_diff = diffs[low_mask]
                i_low = low_idx[np.argmin(low_diff)]

                # move a modest fraction of mass
                frac = 0.05 + 0.1 * rng.random()          # 5‑15 %
                delta = cur[i_high] * frac
                cand = cur.copy()
                cand[i_high] -= delta
                cand[i_low] += delta

        # ---------------------------------------------------------
        # 2. Random block scaling (≈25 % of moves)
        # ---------------------------------------------------------
        elif move < 0.75:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12) + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start:start + width] *= factor

        # ---------------------------------------------------------
        # 3. Simple mass transfer between two positions (≈25 %)
        # ---------------------------------------------------------
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * 0.4                 # up to 40 %
                delta = cur[i] * frac
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta

        if cand is None:
            continue

        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _ac1_score(cand, n, L)

        # keep the best ever seen
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < target_score:
                break

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # temperature schedule
        temperature *= decay

        # occasional restart from the best to avoid drift
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # return as plain Python floats
    return [float(x) for x in best]
