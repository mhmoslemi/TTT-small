        pass

    # Uniform candidate as fallback
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    uscore = compute_score(uniform)
    if best is None or uscore < best_score:
        best = uniform.copy()
        best_score = uscore

    # grid search over power exponents
    # Determine range based on theoretical optimum ~0.5-0.6
    p_vals = np.linspace(0.48, 0.65, 180)  # step ~0.001

            break

        i = np.arange(1, n+1, dtype=np.float64)
        raw = i ** (-p)
        seq = raw / raw.sum()
        # clip (unlikely needed)
        seq = np.clip(seq, 0.0, 1000.0)
        score = compute_score(seq)
        if score < best_score:

        # mixing with uniform

        for alpha in (0.7, 0.85, 1.0):
            if time.monotonic() >= deadline:

            mixed = mixed / mixed.sum()
            mixed = np.clip(mixed, 0.0, 1000.0)
            s = compute_score(mixed)
            if s < best_score:
                best = mixed.copy()
                best_score = s

    # optional simple hill climbing using mass transfer (like previous)
    step_factor = 0.4
    min_step = 1e-8
    while time.monotonic() < deadline:

                s2 = compute_score(shake)
                if s2 < best_score:

    # final safety
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
