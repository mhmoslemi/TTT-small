import time
import numpy as np

def _random_binary_candidate(rng, n, p=0.5):
    """Generate a binary (0/1) sequence of length n with probability p for 1."""
    return (rng.random(n) < p).astype(np.float64)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 bound defined by `evaluate_sequence`.
    """
    # Determine problem size and a simple fallback parent if available.
    try:
        n = int(len(height_sequence_1))
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    rng = np.random.default_rng(seed)

    # Time management.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Keep the best sequence seen.
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # Try a handful of random binary masks; binary patterns give near‑optimal
    # convolution uniformity, hence a bound close to 1.
    attempts = 0
    max_attempts = 30  # enough to stay well under the time budget
    while time.monotonic() < deadline and attempts < max_attempts:
        # Slightly vary the density to explore a bit.
        p = 0.4 + 0.2 * rng.random()   # p ∈ [0.4, 0.6]
        cand = _random_binary_candidate(rng, n, p)
        # Ensure the sequence is not all zeros (sum check will fail otherwise).
        if cand.sum() < 0.01:
            continue
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand.copy()
        attempts += 1

    # As a final safety net, verify the fallback parent again.
    # (In practice the random binary candidate will already be superior.)
    if best_seq.sum() < 0.01:
        best_seq = np.full_like(best_seq, 1e-2)

    # Return a plain Python list of floats, as required.
    return [float(v) for v in best_seq]
