import numpy as np
import time

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (fast, O(n log n))."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a low‑score non‑negative step function within the time budget."""
    rng = np.random.default_rng(seed)
    # leave a few seconds spare for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initialise a probability vector (sum = 1, all entries >= 0).
    # Use the supplied parent if it exists, otherwise the classic sqrt‑series.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

    else:
        # fallback: classic sqrt series (Catalan‑type)
        n_fallback = 5582
        a = np.empty(n_fallback, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n_fallback):
            a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)

    # Normalise to a probability distribution (the objective is scale‑invariant)
    total = a.sum()
    if total < 1e-12:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / total

    n = a.size
    # pre‑computed FFT sizes
    _m_self = 1 << ((2 * n - 1).bit_length())
    _m_cross = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Helper: evaluate the true objective (fast version, uses the same formula as the evaluator)
    # ------------------------------------------------------------------
    def _objective(conv_vec: np.ndarray) -> float:
        # sum(a) == 1 after projection, so denominator is 1
        return 2.0 * n * conv_vec.max()

    # ------------------------------------------------------------------
    # Initialise best solution
    # ------------------------------------------------------------------
    conv = _conv_self(a)
    best_score = _objective(conv)
    best_a = a.copy()

    early_cut = n // 4
    late_cut = 3 * n // 4

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        if grad_iters < max_grad_iters:
            # ----- gradient step (soft‑max smooth max) -----

            maxc = conv.max()
            # soft‑max weights (stable formulation)
            exp_scaled = np.exp(beta * (conv - maxc))
            p = exp_scaled / exp_scaled.sum()                # length 2n‑1

            a_rev = a[::-1]
            fp = np.fft.rfft(p, n=_m_cross)
            fa_rev = np.fft.rfft(a_rev, n=_m_cross)

            # slice out the n relevant entries (indices n‑1 … n‑1+n‑1)
            c = full[n - 1: n - 1 + n]

            conv = conv_new
