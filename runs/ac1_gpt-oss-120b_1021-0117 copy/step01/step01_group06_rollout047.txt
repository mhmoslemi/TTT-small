def _score_fft(a):
    n = a.size
    L = 1 << ((2*n - 1).bit_length())
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[:2*n-1]
    return 2.0 * n * conv.max() / (a.sum() ** 2)

import time, math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of series a where a*a = 1+…+x^{2n-2}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


def _score(vec: np.ndarray) -> float:
    """Exact AC1 score (same as evaluate_sequence)."""
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (vec.sum() ** 2)


def _conv_and_max(vec: np.ndarray):
    """Return (max_convolution_value, argmax_index)."""
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

                if sc < best_sc:
                    best_sc = sc
                    best_vec = cand.copy()
            curr = cand
            no_improve = 0

    # return as plain Python list
    return [float(x) for x in best_vec]
