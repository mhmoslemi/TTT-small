    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    max_b = conv.max()
    s = arr.sum()

        return float('inf'), 0.0, -1
    score = 2.0 * n * max_b / (s*s)
    # We might also want index

    return score, max_b, idx

import numpy as np, time, math

# ---------- utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a[i] = C(2i,i)/4**i  (i=0..n-1) via recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_info(arr: np.ndarray, n: int, L: int):
    """Fast AC1 bound via FFT; also returns max value and its index."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf'), max_b, -1
    score = 2.0 * n * max_b / (s * s)
    idx = int(conv.argmax())
    return score, max_b, idx
