def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 0.2

    try:
        n = len(height_sequence_1)
        init = np.clip(height_sequence_1.astype(np.float64), 0.0, 1000.0)
    except Exception:
        n = 5000
        init = np.ones(n) / n

    # Ensure sum > 0.01, normalize to sum 1 for consistency (scale invariant)
    if init.sum() < 0.01:
        init = np.full(n, 1.0 / n)
    best = init.copy()
    # Optionally normalize sum to 1
    best = best / best.sum()

    # Compute initial score with function
    def score_of(v):
        c = np.convolve(v, v)
        return 2.0 * n * c.max() / (v.sum() ** 2)

    best_score = score_of(best)

    # Global uniform fallback
    uniform = np.full(n, 1.0 / n)
    uniform_score = score_of(uniform)
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score

    # Main loop
    step_factor = 0.3
    min_step = 1e-8
    # Preallocation of arrays for speed
    while time.monotonic() < deadline:
        conv = np.convolve(best, best)
        k = int(conv.argmax())
        # compute partner array
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # idx range
        idx = np.arange(i_start, i_end+1, dtype=np.int64)
        partner = np.empty_like(best)
        partner.fill(0.0)
        partner[idx] = best[k - idx]

        # compute contributions
        contrib = best * partner
        src = int(contrib.argmax())
        # Destination selection: minimal contribution
        # but ensure not same as src
        # we also may restrict to indices where best is low
        dest_candidates = np.where(contrib == contrib.min())[0]
        # if empty, choose random
        if dest_candidates.size == 0:
            dest = rng.integers(0, n)
        else:
            # exclude src if present; else random among them
            dest_candidates = dest_candidates[dest_candidates != src]
            if dest_candidates.size == 0:
                dest = rng.integers(0, n)
            else:
                dest = int(rng.choice(dest_candidates))

        # delta as fraction of source magnitude
        delta = best[src] * step_factor
        if delta < min_step:
            delta = min_step
        if delta > best[src]:
            delta = best[src]
        # perform transfer
        cand = best.copy()
        cand[src] -= delta
        cand[dest] += delta

        # clip and re-normalize due to numeric drift
        cand = np.clip(cand, 0.0, 1000.0)
        # ensure sum >0
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s * best.sum()

        cand_score = score_of(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            # reduce step_factor incrementally for fine tuning
            step_factor = max(0.01, step_factor * 0.995)
        else:
            # maybe random shake
            if rng.random() < 0.01:
                # pick random segment length and scale
                width = int(rng.integers(max(1, n // 300), max(2, n // 50)))
                start = int(rng.integers(0, n - width + 1))
                stop = start + width
                shake = best.copy()
                factor = np.exp(rng.normal(0.0, 0.2))
                shake[start:stop] *= factor
                shake = np.clip(shake, 0.0, 1000.0)
                # re-normalize
                shake = shake / shake.sum() * best.sum()
                shake_score = score_of(shake)
                if shake_score < best_score:
                    best = shake
                    best_score = shake_score

        # occasional check to further lower step_factor over time
        # The loop will stop when time out

    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)
    return [float(v) for v in final]

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic search for a non‑negative step‑function that minimizes the
    AC1 upper bound evaluated by ``evaluate_sequence``.
    The algorithm repeatedly moves a small amount of mass from an index
    contributing most to the current convolution peak to an index that
    contributes least, using FFT‑based convolution for speed.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # ----- obtain initial sequence -----------------------------------------
    try:
        # parent sequence provided by the harness
        init = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = init.shape[0]
        init = np.clip(init, 0.0, 1000.0)
    except Exception:                     # fallback when no parent is present
        n = 5000
        init = np.full(n, 1.0 / n, dtype=np.float64)

    if init.sum() < 0.01:
        init = np.full(n, 1.0 / n, dtype=np.float64)

    # Normalise to unit sum – the score is scale‑invariant
    best = init / init.sum()

    # ----- FFT configuration -----------------------------------------------
    # convolution length = 2*n-1; choose a power‑of‑two FFT size
    conv_len = 2 * n - 1
    n_fft = 1 << ((conv_len - 1).bit_length())   # next power of two

    # ----- score utilities --------------------------------------------------
    def conv_max_and_index(v):
        """Return (max(convolution of v with itself), argmax)."""
        fv = np.fft.rfft(v, n=n_fft)
        c = np.fft.irfft(fv * fv, n=n_fft)[:conv_len]
        idx = int(np.argmax(c))
        return float(c[idx]), idx

    def score_from_max(max_val, total):
        """Exact score from max convolution value and total L1 norm."""
        return 2.0 * n * max_val / (total * total)

    cur_max, cur_k = conv_max_and_index(best)
    best_score = score_from_max(cur_max, best.sum())
    best_k = cur_k

    # a cheap uniform fallback (score = 2)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_max, _ = conv_max_and_index(uniform)
    uni_score = score_from_max(uni_max, 1.0)
    if uni_score < best_score:
        best = uniform
        best_score = uni_score
        best_k = _

    # ----- optimisation parameters -----------------------------------------
    step_factor = 0.45          # fraction of source mass to move
    min_step = 1e-12
    shrink_rate = 0.997
    grow_rate = 1.02
    shake_prob = 0.008
    # number of iterations improves with time; we only stop at deadline.
    while time.monotonic() < deadline:
        # recompute convolution max for current best
        cur_max, k = conv_max_and_index(best)

        # build partner array: partner[i] = best[k-i] if defined else 0
        partner = np.zeros_like(best)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner[idx] = best[k - idx]

        # contributions to the peak
        contrib = best * partner
        src = int(np.argmax(contrib))
        # choose a destination where contribution is minimal (often zero)
        min_contrib = contrib.min()
        dest_cands = np.where(contrib == min_contrib)[0]
        # make sure we actually move somewhere
        if dest_cands.size == 0 or (dest_cands.size == 1 and dest_cands[0] == src):
            dest = rng.integers(0, n)
            while dest == src:
                dest = rng.integers(0, n)
        else:
            # discard src if it appears among minima
            dest_cands = dest_cands[dest_cands != src]
            if dest_cands.size == 0:
                dest = rng.integers(0, n)
                while dest == src:
                    dest = rng.integers(0, n)
            else:
                dest = int(rng.choice(dest_cands))

        # amount to transfer
        delta = best[src] * step_factor
        if delta < min_step:
            delta = min_step
        if delta > best[src]:
            delta = best[src]

        cand = best.copy()
        cand[src] -= delta
        cand[dest] += delta

        # re‑normalise to keep sum exactly 1 (helps numeric stability)
        s = cand.sum()
        if s <= 0:
            continue
        cand /= s

        cand_max, cand_k = conv_max_and_index(cand)
        cand_score = score_from_max(cand_max, cand.sum())

        if cand_score < best_score:
            best = cand
            best_score = cand_score
            best_k = cand_k
            step_factor = max(min_step, step_factor * shrink_rate)
            continue

        # occasional random perturbation to escape plateaus
        if rng.random() < shake_prob:
            # pick a random short segment and scale it
            width = int(rng.integers(max(1, n // 300), max(2, n // 50)))
            start = int(rng.integers(0, n - width + 1))
            stop = start + width
            shake = best.copy()
            factor = float(np.exp(rng.normal(0.0, 0.18)))
            shake[start:stop] *= factor
            shake = np.clip(shake, 0.0, 1000.0)
            shake /= shake.sum()
            sh_max, _ = conv_max_and_index(shake)
            sh_score = score_from_max(sh_max, 1.0)
            if sh_score < best_score:
                best = shake
                best_score = sh_score
                best_k = _
                # after a successful shake we allow larger steps again
                step_factor = min(0.5, step_factor * grow_rate)
                continue

        # if no improvement, slowly increase step size to explore wider moves
        step_factor = min(0.5, step_factor * grow_rate)

    # ----- final sanity checks -----------------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01 or not np.isfinite(final).all():
        # fallback to a simple uniform distribution
        final = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in final]
