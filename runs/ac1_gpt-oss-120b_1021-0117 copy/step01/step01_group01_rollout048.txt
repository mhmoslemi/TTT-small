def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    # if height_sequence_1 defined?
    try:
        seq_parent = np.array(height_sequence_1, dtype=np.float64, copy=False)
        n = int(seq_parent.shape[0])
    except Exception:
        seq_parent = None
        n = 256
    # initial a
    if seq_parent is not None and n > 0:
        a = seq_parent.copy()
        # ensure nonneg and clip
        a = np.clip(a, 0.0, 1000.0)
    else:
        # generate binomial
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(n-1):
            a[i+1] = a[i] * (2*i+1) / (2*i+2)
    # Normalize sum to 1
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / s

    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # fast evaluation
    def conv_max_and_arg(a):
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        idx = int(conv.argmax())
        return float(conv[idx]), idx

    def score(a):
        M, _ = conv_max_and_arg(a)
        return float(2 * n * M / (a.sum() * a.sum()))

    cur = a.copy()
    cur_score = score(cur)
    best = cur.copy()
    best_score = cur_score

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # keep a buffer

    rng = np.random.default_rng(seed)

    iteration = 0
    # parameters
    max_iters = 1000000  # but will break earlier

    while time.monotonic() < deadline and iteration < max_iters:
        M, p = conv_max_and_arg(cur)

        # compute r = a shifted by p (zero elsewhere)
        idx = np.arange(n)
        rp = p - idx
        mask = (rp >= 0) & (rp < n)
        r = np.zeros_like(cur)
        r[mask] = cur[rp[mask]]
        r_mean = r.mean()
        # direction to reduce M (move opposite to gradient)
        d = r_mean - r

        # schedule step size
        # start with eta0; decay slowly over iterations
        eta0 = 0.5
        # exponential decay: eta = eta0 * (0.99 ** (iteration / 200))
        eta = eta0 * (0.99 ** (iteration / 200.0))

        # apply update
        a_new = cur + eta * d

        # enforce nonnegative
        np.maximum(a_new, 0, out=a_new)
        # renormalize sum
        s_new = a_new.sum()
        if s_new > 0:
            a_new /= s_new
        else:
            # degenerate, fallback
            a_new = cur.copy()

        new_score = score(a_new)

        # acceptance: greedy with occasional random acceptance
        if new_score < cur_score:
            cur = a_new
            cur_score = new_score
            if new_score < best_score:
                best = a_new.copy()
                best_score = new_score
        else:
            # occasionally accept worse move to escape local minima
            # probability decays with temperature
            # define temperature proportional to remaining time
            remaining = deadline - time.monotonic()
            total = float(budget_s)
            temp = max(remaining / total, 1e-3)
            prob = math.exp((cur_score - new_score) / (cur_score * 0.05) * temp)
            if rng.random() < prob:
                cur = a_new
                cur_score = new_score

        # occasional random perturbation to escape stagnation
        if iteration % 500 == 0 and iteration > 0:
            # small random shuffle of mass
            perturb = rng.normal(scale=0.02, size=n)
            a_pert = cur + perturb
            np.maximum(a_pert, 0, out=a_pert)
            s_pert = a_pert.sum()
            if s_pert > 0:
                a_pert /= s_pert
                pert_score = score(a_pert)
                if pert_score < cur_score:
                    cur = a_pert
                    cur_score = pert_score
                    if pert_score < best_score:
                        best = a_pert.copy()
                        best_score = pert_score

        iteration += 1

    # final result as list of floats
    final = [float(x) for x in best]
    # ensure scored finite
    try:
        _ = evaluate_sequence(final)
    except Exception:
        # fallback to a simple uniform
        final = [float(x) for x in np.full(n, 1.0 / n)]
    return final
