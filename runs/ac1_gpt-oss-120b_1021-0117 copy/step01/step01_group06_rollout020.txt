def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Determine n
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = a.size
    else:
        n = 256
        a = np.ones(n, dtype=np.float64)

    # Ensure non-negative and avoid zero-sum problem
    a = np.maximum(a, 0.0)
    sum_a = a.sum()
    if sum_a < 1e-12:
        a = np.full_like(a, 1.0)
        sum_a = a.sum()
    a /= sum_a   # normalized to sum 1

    # Helper functions defined outside?

   # compute initial score
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
    max_conv = conv.max()
    best_score = 2.0 * n * max_conv   # same as evaluator
    best_vec = a.copy()
    cur_vec = a.copy()
    cur_score = best_score

    # parameters for gradient descent
    init_step = 0.8   # initial step size (relative)
    min_step = 1e-4
    # "t" for line search attempts
    # use ring buffer? Not needed

    iter_count = 0
    while time.monotonic() < deadline:
        # recompute conv for cur_vec
        fv = np.fft.rfft(cur_vec, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
        max_conv = conv.max()
        max_idx = int(conv.argmax())
        cur_score = 2.0 * n * max_conv

        # Gradient step:
        # chosen max index, compute subgradient
        k = max_idx
        # compute gradient vector
        inds = np.arange(n)
        src = k - inds
        mask = (src >= 0) & (src < n)
        grad = np.zeros_like(cur_vec)
        grad[mask] = 2.0 * cur_vec[src[mask]]
        # take step size decreasing
        step = init_step / np.sqrt(iter_count + 1)
        # line search attempts
        cand = None
        cand_score = None
        for _ in range(10):
            # candidate = projection of cur - step * grad
            cand_tmp = cur_vec - step * grad
            cand_tmp = _project_simplex(cand_tmp)
            # compute score
            fv_c = np.fft.rfft(cand_tmp, n=L)
            conv_c = np.fft.irfft(fv_c * fv_c, n=L)[:2*n-1]
            score_c = 2.0 * n * conv_c.max()
            if score_c < cur_score:
                cand = cand_tmp
                cand_score = score_c
                break
            else:
                step *= 0.5
                if step < min_step:
                    break
        # if we found improvement, accept
        if cand is not None and cand_score < cur_score:
            cur_vec = cand
            cur_score = cand_score

            if cur_score < best_score:
                best_score = cur_score
                best_vec = cur_vec.copy()
        else:
            # maybe perform random transfer step
            # choose random source with positive mass
            # chance to random try
            if rng.random() < 0.3:
                # random transfer
                # compute allowed indices
                positives = np.where(cur_vec > 1e-12)[0]
                if positives.size > 0:
                    i = rng.choice(positives)
                    j = rng.integers(0, n-1)
                    if j >= i:
                        j += 1
                    frac = rng.random() * 0.3
                    delta = cur_vec[i] * frac
                    cand2 = cur_vec.copy()
                    cand2[i] -= delta

def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ------------------------------------------------------------
# First n coefficients of sqrt(1+x+…+x^{2n-2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initialise a feasible vector (non‑negative, sum = 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = _sqrt_series(256)
    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s < 1e-12:
        cur[:] = 1.0
        s = cur.sum()
    cur /= s                     # normalise to the simplex

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (power of two)

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    init_step = 0.6
    min_step = 1e-7
    iter_cnt = 0
    top_frac = 0.001   # fraction of convolution entries taken as “active”

    # ------------------------------------------------------------------
    # Optimisation loop (projected sub‑gradient + random perturbations)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # current convolution and its maximal value
        fv = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        maxc = conv.max()
        cur_sc = 2.0 * n * maxc

        # ------------------------------------------------------------------
        # Build a sub‑gradient from the top‑few convolution entries
        # ------------------------------------------------------------------
        top_k = max(1, int(top_frac * n))
        if top_k == 1:
            act = np.array([int(conv.argmax())], dtype=int)
        else:
            act = np.argpartition(conv, -top_k)[-top_k:]

        grad = np.zeros_like(cur)
        idx = np.arange(n)
        for k in act:
            src = k - idx
            mask = (src >= 0) & (src < n)
            grad[mask] += 2.0 * cur[src[mask]]
        grad /= len(act)                     # average over the active set

        # ------------------------------------------------------------------
        # Line search on the negative sub‑gradient
        # ------------------------------------------------------------------
        step = init_step / np.sqrt(iter_cnt + 1)
        improved = False
        for _ in range(12):
            cand = cur - step * grad
            cand = _project_simplex(cand)
            cand_sc = _score(cand, L)
            if cand_sc < cur_sc:
                cur = cand
                cur_sc = cand_sc
                improved = True
                if cand_sc < best_sc:
                    best_sc = cand_sc
                    best_vec = cand.copy()
                break
            step *= 0.5
            if step < min_step:
                break

        # ------------------------------------------------------------------
        # If sub‑gradient move failed, try an exponential‑gradient step
        # ------------------------------------------------------------------
        if not improved:
            eta = init_step / np.sqrt(iter_cnt + 1) * 0.5
            exp_move = cur * np.exp(-eta * grad)
            if exp_move.sum() > 0:
                exp_move /= exp_move.sum()
                cand = exp_move
                cand_sc = _score(cand, L)
                if cand_sc < cur_sc:
                    cur = cand
                    cur_sc = cand_sc
                    if cand_sc < best_sc:
                        best_sc = cand_sc
                        best_vec = cand.copy()

        # ------------------------------------------------------------------
        # Occasional random mass transfer (helps escape flat regions)
        # ------------------------------------------------------------------
        if rng.random() < 0.25:
            positives = np.where(cur > 1e-12)[0]
            if positives.size:
                i = int(rng.choice(positives))
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.3
                delta = cur[i] * frac
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)
                cand_sc = _score(cand, L)
                if cand_sc < best_sc:
                    best_sc = cand_sc
                    best_vec = cand.copy()
                if cand_sc < cur_sc:
                    cur = cand
                    cur_sc = cand_sc

        iter_cnt += 1
        if time.monotonic() + 0.01 >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
