            smooth = np.clip(smooth, 0.0, 1000.0)
            smooth = smooth * (target_sum / smooth.sum())
            smooth_score = _score(smooth)
            if smooth_score < best_score:
                best, best_score = smooth, smooth_score
                no_improve = 0

        # ---------------------------------------------------------------- #
        # stagnation handling – random restart ---------------------------- #
        if no_improve > max_no_improve:
            cand = rng.random(n, dtype=np.float64)
            cand = cand * (target_sum / cand.sum())
            cand_score = _score(cand)
            if cand_score < best_score:
                best, best_score = cand, cand_score
            step_factor = 0.6
            no_improve = 0

    # ------------------------------------------------------------------ #
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, target_sum / final.size)

    return [float(v) for v in final]
