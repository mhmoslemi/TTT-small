import time
import numpy as np

def _full_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT (full length)."""
    lx = x.size
    ly = y.size
    size = 1 << ((lx + ly - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: lx + ly - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Fast version of the evaluator’s objective (sum is assumed >0)."""
    b = _full_convolve_fft(a, a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = a.size
    return 2.0 * n * max_b / (s * s)


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence, length n."""
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return seq


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step‑function with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine length and initial vectors
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0)
        n = parent.size
    else:
        parent = None
        n = 5582

    # classic sqrt‑series (normalized)
    a0 = _sqrt_series(n)
    a0 = np.clip(a0, 0.0, 1000.0)
    a0 = a0 / max(a0.sum(), 1e-12)

    # use parent if it gives a better quick score
    if parent is not None:
        p = parent / max(parent.sum(), 1e-12)
        if _score(p) < _score(a0):
            a0 = p

    a = a0.copy()
    best = a.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # 2) Gradient‑descent on a smooth approximation of max(conv(a,a))
    # ------------------------------------------------------------------
    beta = 1.0          # soft‑max temperature (low = smooth)
    beta_max = 30.0
    beta_inc = 0.03
    lr = 0.20           # learning rate
    iters_no_improve = 0
    max_no_improve = 3000

    while time.monotonic() < deadline:

        conv_self = _full_convolve_fft(a, a)               # length 2n‑1

        w_raw = np.exp(beta * (conv_self - max_c))          # max exponent = 0

        else:
