import time
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

def _catalan_weights(n: int) -> np.ndarray:
    """First n Catalan‐style coefficients w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    """
    Build the (2n‑1)×n sparse matrix A such that A @ g == np.convolve(f, g).
    Row k, column j entry is f[k‑j] when 0≤k‑j<n, else 0.
    """
    n = f.shape[0]
    # offsets from -(n‑1) … (n‑1)
    offsets = np.arange(-(n - 1), n, dtype=int)
    data = []
    for off in offsets:
        p = abs(off)                     # which coefficient of f is used
        length = n - p
        if length <= 0:
            continue
        # constant diagonal = f[p]
        data.append(np.full(length, f[p], dtype=np.float64))
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max 1ᵀg   s.t.  A@g ≤ M , g ≥ 0,
    where A is the convolution matrix built from f.
    Returns the optimal g (numpy array) or None on failure.
    """
    if M <= 0.0:
        return None
    A = _conv_matrix(f)
    # scale to avoid tiny numbers
    A = A / M
    n = f.shape[0]
    g = cp.Variable(n)
    cons = [A @ g <= 1.0, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), cons)
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, verbose=False)
    except Exception:               # pragma: no cover
        return None
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None
    gval = np.asarray(g.value).ravel()
    gval = np.clip(gval, 0.0, 1000.0)
    return gval

    except Exception:                # pragma: no cover
        parent = np.ones(256, dtype=np.float64)

    n = parent.size

    # normalised versions (sum == 1)
    f_parent = parent.copy()
    if f_parent.sum() < 1e-12:
        f_parent[:] = 1.0 / n
    f_parent /= f_parent.sum()

    w = _catalan_weights(n)
    f_catalan = w / w.sum()

    # pick the better of the two starters
    score_parent = float(evaluate_sequence(f_parent.tolist()))
    score_catalan = float(evaluate_sequence(f_catalan.tolist()))
    best = f_parent if score_parent <= score_catalan else f_catalan
    best_score = min(score_parent, score_catalan)

    # --------------------------------------------------------------
    # 2.  Main iterative improvement
    # --------------------------------------------------------------
    mix_factors = (0.4, 0.25, 0.15, 0.07, 0.03)
    while time.monotonic() < deadline:
        # ---- LP step ------------------------------------------------
        conv_self = np.convolve(best, best)
        M = float(conv_self.max())
        g_opt = _solve_lp(best, M)

        if g_opt is not None:
            for t in mix_factors:
                cand = (1.0 - t) * best + t * g_opt
                s = cand.sum()
                if s <= 0.0:
                    continue
                cand /= s                        # renormalise to sum == 1
                sc = float(evaluate_sequence(cand.tolist()))
                if sc < best_score:
                    best = cand
                    best_score = sc
                    if best_score <= 1.50274349:
                        return [float(v) for v in best]

        # ---- occasional jitter (helps escape plateaux) -----------
        if rng.random() < 0.045:
            jitter = np.exp(rng.normal(0.0, 0.014, size=n))
            cand = best * jitter
            s = cand.sum()
            if s > 0.0:
                cand /= s
                sc = float(evaluate_sequence(cand.tolist()))
                if sc < best_score:
                    best = cand
                    best_score = sc
                    if best_score <= 1.50274349:
                        break

        # ---- cheap random transfer (fallback) --------------------
        if time.monotonic() > deadline - 4.0:
            break                                   # leave time to return
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] > 0.0:
            eps = rng.random() * best[i] * 0.5
            cand = best.copy()
            cand[i] -= eps
            cand[j] += eps
            cand /= cand.sum()
            sc = float(evaluate_sequence(cand.tolist()))
            if sc < best_score:
                best = cand
                best_score = sc
                if best_score <= 1.50274349:
                    break

    # --------------------------------------------------------------
    # 3.  Return the best sequence found
    # --------------------------------------------------------------
    return [float(v) for v in best]
