import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # degenerate case – return uniform
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic sub‑gradient optimisation for the AC1 upper bound.
    Starts from the supplied parent (``height_sequence_1``) or a fallback,
    then repeatedly:
      * computes the convolution,
      * finds the shift attaining the maximum,
      * takes a sub‑gradient step for that shift,
      * projects back onto the simplex,
    with a diminishing step size and occasional jitter.
    Returns the best sequence encountered.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # 1.  Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2.  Base sequence (fallback if the parent is unavailable)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = None

    if parent is not None:
        n = int(parent.size)
        a = parent.copy()
    else:
        n = 256
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                       # normalize to sum = 1

    # ------------------------------------------------------------------
    # 3.  Initialise best candidate
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # also keep uniform as a cheap alternative
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    target_score = 1.50274349
    conv_len = 2 * n - 1
    # next power‑of‑two for FFT (fast convolution)
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # 4.  Main optimisation loop (sub‑gradient descent)
    # ------------------------------------------------------------------
    it = 0

    while time.monotonic() < deadline:

        # --- convolution via FFT (real‑valued) ---------------------------------

    # ------------------------------------------------------------------
