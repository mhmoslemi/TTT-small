def _conv_max_fft(a, fft_len=None):
    n = a.shape[0]
    conv_len = 2*n - 1
    if fft_len is None:
        fft_len = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(conv.max())

import time
import numpy as np

# ---------- helper functions -------------------------------------------------
def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4**i sequence, decreasing and positive."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def _conv_max_fft(a: np.ndarray, fft_len: int = None) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    if fft_len is None:
        fft_len = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(conv.max())


def _score_from_array(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact score (same as evaluate_sequence) using FFT for speed."""
    max_b = _conv_max_fft(a, fft_len)
    s = a.sum()
    return float(2 * n * max_b / (s * s))


def _apply_move(a: np.ndarray, B: np.ndarray, i: int, j: int, delta: float) -> np.ndarray:
    """
    Return a new convolution array after moving `delta` from a[i] to a[j].
    Does **not** modify `a` or `B` in‑place.
    """
    n = a.shape[0]
    B_new = B.copy()

    # linear part 2*a*Δ
    B_new[i : i + n] -= 2.0 * delta * a
    B_new[j : j + n] += 2.0 * delta * a

    # quadratic Δ*Δ part
    B_new[2 * i] += delta * delta
    B_new[2 * j] += delta * delta
    B_new[i + j] -= 2.0 * delta * delta

    return B_new
