            idx_i = np.arange(lo, hi+1)

            step = step0 / np.sqrt(it)
            cand = cur - step * grad
            cand = project_simplex(cand)
        else:

            # choose source index with probability proportional to mass
            if cur.max() == 0:

            delta = cur[src] * frac
            cand = cur.copy()
            cand[src] -= delta
            cand[tgt] += delta
            # due to floating error, ensure nonnegative:
            cand = np.maximum(cand, 0.0)
            cand = project_simplex(cand)

        # evaluate candidate
        sc = score(cand)

        # acceptance rule: Metropolis with decreasing temperature
        # compute temperature relative to current best score
        # We'll use simple simulated annealing: accept if improves else with prob exp(-(sc - cur_sc)/T)
        # T schedule decreasing from initial to near zero

        if it % 5000 == 0:
            cur = best.copy()
            cur_sc = best_sc

    # Return best solution as Python list
    return [float(x) for x in best]

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
