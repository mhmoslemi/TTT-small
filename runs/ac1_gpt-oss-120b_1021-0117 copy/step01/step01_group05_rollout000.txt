import time
import numpy as np

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fast_score(seq: np.ndarray):
    """Return (score, full convolution, max(conv))."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())           # power of two >= 2n-1
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = float(conv.max())
    s = float(seq.sum())
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val

            total_removed = rem_i + rem_j

            # distribute to a low‑value position (avoid i and j)
            low_idxs = np.argpartition(cand, low_k)[:low_k]
            # ensure we have at least one admissible target
            tgt_candidates = low_idxs[(low_idxs != i) & (low_idxs != j)]
            if tgt_candidates.size == 0:
                tgt = rng.integers(0, n)
                while tgt == i or tgt == j:
                    tgt = rng.integers(0, n)
            else:
                tgt = int(rng.choice(tgt_candidates))
            cand[tgt] += total_removed

        elif op == 1:
            # ----- block scaling (log‑normal perturbation) -----
            width = rng.integers(1, max(2, n // 30))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            cand[start:start + width] *= factor

        elif op == 2:
            # ----- single index log‑normal perturbation -----
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.25))
            cand[idx] *= factor

        else:
            # ----- smooth two adjacent entries -----
            i = rng.integers(0, n - 1)
            avg = (cand[i] + cand[i + 1]) * 0.5
            cand[i] = avg
            cand[i + 1] = avg

        # enforce bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        # --------------------------------------------------------------
        # evaluate the candidate (FFT‑based, O(n log n))
        # --------------------------------------------------------------
        cand_score, cand_conv, _ = _fast_score(cand)

        # --------------------------------------------------------------
        # acceptance criterion (simulated annealing)
        # --------------------------------------------------------------
        elapsed = time.monotonic() - start_time
        time_frac = elapsed / (deadline - start_time)
        temp = max(1e-5, init_temp * (1.0 - time_frac))

        delta = cand_score - cur_score
        accept = (delta < 0.0) or (rng.random() < np.exp(-delta / temp))
        if accept:
            cur = cand
            cur_score = cand_score
            cur_conv = cand_conv

        # --------------------------------------------------------------
        # record global best
        # --------------------------------------------------------------
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            best_conv = cand_conv
            last_best_step = steps

        # --------------------------------------------------------------
        # occasional re‑centering on the best solution to avoid drift
        # --------------------------------------------------------------
        if steps - last_best_step > restart_interval:
            cur = best.copy()
            cur_score = best_score
            cur_conv = best_conv
            last_best_step = steps

    # ----------------------------------------------------------------------
    # Return the best solution found as a plain Python list
    # ----------------------------------------------------------------------
    return [float(x) for x in best]
