    # ----- helper: fast score (FFT based) ------------------------------------
    def fast_score(arr):
        """Return (score, convolution array, max value)."""
        total = arr.sum()
        if total <= 0.0:
            return float('inf'), None, None
        # zero‑pad to length 2*n (next power of two not required for rfft)
        fa = np.fft.rfft(arr, n=2 * n)
        conv = np.fft.irfft(fa * fa, n=2 * n)[: 2 * n - 1]
        m = conv.max()
        score = 2.0 * n * m / (total * total)
        return score, conv, m

    # ----- start from a collection of simple candidates -----------------------
    best = None
    best_score = float('inf')

    # 1. parent (if any)
    if parent is not None:
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() >= 0.01:
            sc, _, _ = fast_score(parent)
            if sc < best_score:
                best = parent.copy()
                best_score = sc

    # 2. uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    sc, _, _ = fast_score(uniform)
    if sc < best_score:
        best = uniform.copy()
        best_score = sc

    # 3. binomial–type baseline (same as previous program)
    a0 = np.empty(n, dtype=np.float64)
    a0[0] = 1.0
    for i in range(1, n):
        a0[i] = a0[i - 1] * (1.0 - 1.0 / (2.0 * i))
    a0 = a0 / a0.sum()
    sc, _, _ = fast_score(a0)
    if sc < best_score:
        best = a0.copy()
        best_score = sc

    # ----- parametric families -------------------------------------------------
    # a) powers of the binomial shape
    exps = np.linspace(0.5, 2.0, 16)
    for p in exps:
        cand = np.power(a0, p)
        cand = cand / cand.sum()
        sc, _, _ = fast_score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # b) power‑law decay a_i ∝ (i+1)^-α
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.0, 2.0, 21)
    for a in alphas:
        cand = 1.0 / np.power(idx + 1.0, a)
        cand = cand / cand.sum()
        sc, _, _ = fast_score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # c) raised‑cosine window  a_i = sin(π(i+0.5)/n)^p
    cos_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    cos_vals = np.clip(cos_vals, 0.0, None)
    for p in (0.5, 1.0, 2.0):
        cand = np.power(cos_vals, p)
        cand = cand / cand.sum()
        sc, _, _ = fast_score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # ----- iterative refinement -----------------------------------------------
    step_factor = 0.5          # initial transfer fraction
    min_step = 1e-8
    no_improve = 0

    while time.monotonic() < deadline:
        # convolution of the current best
        fa = np.fft.rfft(best, n=2 * n)
        conv = np.fft.irfft(fa * fa, n=2 * n)[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (best.sum() ** 2)

        # index where the maximum occurs
        k = int(conv.argmax())

        # partner[i] = best[k-i] if in range, else 0
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        partner = np.zeros_like(best)
        idx_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner[idx_range] = best[k - idx_range]

        # contribution to the peak
        contrib = best * partner
        src = int(contrib.argmax())
        # choose a destination where partner is minimal (often zero)
        min_partner = partner.min()
        dest_cands = np.where(partner == min_partner)[0]
        dest = int(rng.choice(dest_cands)) if dest_cands.size else int(rng.integers(0, n))

        if src == dest:
            # occasional random perturbation when stuck
            if rng.random() < 0.02:
                w = int(rng.integers(max(1, n // 200), max(2, n // 20)))
                s = int(rng.integers(0, max(1, n - w + 1)))
                e = min(n, s + w)
                cand = best.copy()
                cand[s:e] *= float(np.exp(rng.normal(0.0, 0.15)))
                cand = np.clip(cand, 0.0, 1000.0)
                cand = cand / cand.sum() * best.sum()
                sc, _, _ = fast_score(cand)
                if sc < best_score:
                    best = cand
                    best_score = sc
            continue

        # if moving mass would not improve the peak, skip
        if partner[src] <= partner[dest]:
            # very small chance to try a random swap anyway
            if rng.random() < 0.005:
                i, j = rng.choice(n, size=2, replace=False)
                delta = min(best[i], best[j]) * 0.5
                cand = best.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = np.clip(cand, 0.0, 1000.0)
                cand = cand / cand.sum() * best.sum()
                sc, _, _ = fast_score(cand)
                if sc < best_score:
                    best = cand
                    best_score = sc
            continue

        # amount to transfer from src to dest
        delta = best[src] * step_factor
        if delta < min_step:
            delta = min_step
        if delta > best[src]:
            delta = best[src]

        cand = best.copy()
        cand[src] -= delta
        cand[dest] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        # renormalise to keep the total mass unchanged (numerical safety)
        tot = cand.sum()
        if tot == 0.0:
            continue
        cand = cand / tot * best.sum()

        # evaluate the candidate
        sc, _, _ = fast_score(cand)

        if sc < best_score:
            best = cand
            best_score = sc
            no_improve = 0
            # slowly shrink the step for finer adjustments
            step_factor = max(0.02, step_factor * 0.99)
        else:
            no_improve += 1
            if no_improve % 250 == 0:
                # occasionally enlarge the step to escape plateaus
                step_factor = min(0.7, step_factor * 1.07)

        # occasional global shake
        if rng.random() < 0.001:
            w = int(rng.integers(max(1, n // 300), max(2, n // 30)))
            s = int(rng.integers(0, max(1, n - w + 1)))
            e = min(n, s + w)
            shake = best.copy()
            shake[s:e] *= float(np.exp(rng.normal(0.0, 0.2)))
            shake = np.clip(shake, 0.0, 1000.0)
            shake = shake / shake.sum() * best.sum()
            sc, _, _ = fast_score(shake)
            if sc < best_score:
                best = shake
                best_score = sc

    # ----- final safety check -------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.full_like(best, 1.0 / best.size)

    return [float(v) for v in best]
