def project_simplex(v):
    # onto simplex sum=1, v>=0
    # safe: if v is already sum 1 and nonnegative, return v.
    # else use algorithm:
    # shift = v - tau where tau is threshold.
    # Sorting descending.
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted) - 1
    ind = np.arange(1, len(v)+1)
    cond = v_sorted - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

This returns vector with sum=1. Works for our case.

```
def _project_onto_simplex(v):
    # Based on algorithm from Duchi et al. 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (top level, no closures)
# ------------------------------------------------------------

def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def _conv_max_and_idx(a: np.ndarray, fft_len: int):
    """Return (max value, index) of the linear convolution a*a."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _score_from_max_b(max_b: float, sum_a: float, n: int) -> float:
    """Exact AC1 upper bound for a given max_b and sum_a."""
    return float(2 * n * max_b / (sum_a * sum_a))


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return zero vector
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _optimize_sequence(a_init: np.ndarray,
                       n: int,
                       fft_len: int,
                       deadline: float,
                       rng: np.random.Generator):
    """Projected sub‑gradient descent on the max‑autocorrelation objective."""
    a = a_init.astype(np.float64, copy=True)
    a = _project_onto_simplex(a)            # guarantee feasibility
    max_b, _ = _conv_max_and_idx(a, fft_len)
    cur_score = _score_from_max_b(max_b, a.sum(), n)
    best = a.copy()
    best_score = cur_score
    step = 0.12                               # initial step size

    while time.monotonic() < deadline:
        max_b, k_star = _conv_max_and_idx(a, fft_len)
        cur_score = _score_from_max_b(max_b, a.sum(), n)

        # gradient of the active convolution entry (sub‑gradient)
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            g[idx] = 2.0 * a[k_star - idx]

        # line‑search on the step size
        improved = False
        step_try = step
        while step_try > 1e-12:
            cand = a - step_try * g
            cand = _project_onto_simplex(cand)
            max_b_new, _ = _conv_max_and_idx(cand, fft_len)
            new_score = _score_from_max_b(max_b_new, cand.sum(), n)
            if new_score < cur_score - 1e-15:                  # strict improvement
                a = cand
                cur_score = new_score
                improved = True
                if new_score < best_score:
                    best = cand.copy()
                    best_score = new_score
                step = step_try * 1.1                         # slightly enlarge
                break
            step_try *= 0.5

        if not improved:
            # small random mass transfer to escape possible plateaus
            i = rng.integers(0, n)
            if a[i] > 1e-14:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.uniform(0.01, 0.05)
                delta = a[i] * frac
                a2 = a.copy()
                a2[i] -= delta
                a2[j] += delta
                # sum unchanged, all entries stay non‑negative
                max_b2, _ = _conv_max_and_idx(a2, fft_len)
                score2 = _score_from_max_b(max_b2, a2.sum(), n)
                if score2 < cur_score - 1e-15:
                    a = a2
                    cur_score = score2
                    if score2 < best_score:
                        best = a2.copy()
                        best_score = score2
                    step = max(step * 0.9, 1e-6)
            else:
                # nothing to move; shrink step a bit
                step = max(step * 0.7, 1e-8)

    return best, best_score
