    n = seq.size
    # compute convolution via FFT
    m = 1
    while m < 2*n-1:
        m <<= 1
    fft_seq = np.fft.rfft(seq, n=m)
    prod = fft_seq * fft_seq
    conv = np.fft.irfft(prod, n=m)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)

def _sidon_seq(n: int) -> np.ndarray:
    """Sidon‑like decreasing sequence a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a

def _generate_shapes(n: int):
    """Produce a few simple decreasing prototypes."""
    i = np.arange(1, n + 1, dtype=np.float64)
    shapes = [_sidon_seq(n),
              1.0 / np.sqrt(i),
              1.0 / i,
              np.power(i, -0.75)]
    return shapes

def _permute_step(seq: np.ndarray, step: int) -> np.ndarray:
    """Place elements of seq using a coprime step (mod n)."""
    n = seq.shape[0]
    out = np.empty_like(seq)
    pos = 0
    for k in range(n):
        out[pos] = seq[k]
        pos = (pos + step) % n
    return out

def _interleave(seq: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …"""
    n = seq.shape[0]
    out = np.empty_like(seq)
    lo, hi, pos = 0, n - 1, 0
    while lo <= hi:
        out[pos] = seq[hi]
        pos += 1
        hi -= 1
        if lo <= hi:
            out[pos] = seq[lo]
            pos += 1
            lo += 1
    return out
