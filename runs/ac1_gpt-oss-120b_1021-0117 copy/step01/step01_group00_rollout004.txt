def _sidon_like_sequence(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4**i sequence (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _bit_reverse_permutation(n: int) -> np.ndarray:
    """Permutation that sorts indices by their bit‑reversed value."""
    bits = (n - 1).bit_length()
    rev = np.empty(n, dtype=np.int64)
    for i in range(n):
        v = 0
        x = i
        for _ in range(bits):
            v = (v << 1) | (x & 1)
            x >>= 1
        rev[i] = v
    return np.argsort(rev)


def _interleaved_permutation(n: int) -> np.ndarray:
    """Place items 0,1,2,… alternately in the first half and second half."""
    perm = np.empty(n, dtype=np.int64)
    left, right = 0, n // 2
    idx = 0
    while idx < n:
        if left < n:
            perm[left] = idx
            left += 1
            idx += 1
        if idx < n and right < n:
            perm[right] = idx
            right += 1
            idx += 1
    return perm


def _score(seq: np.ndarray, fft_len: int) -> float:
    """Return AC1 bound for seq (uses numpy FFT)."""
    S = np.sum(seq)
    if S < 0.01:
        return float("inf")
    ft = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(ft * ft, n=fft_len)
    max_b = np.max(conv[: 2 * len(seq) - 1])
    return (2.0 * len(seq) * max_b) / (S * S)

        perm = rng.permutation(n)
        cand = values[perm]
        sc = _score(cand, fft_len)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()

    # ----------------------------------------------------------
    # Local hill‑climbing: swap two positions if it improves.
    # ----------------------------------------------------------
    current_seq = best_seq.copy()
    current_score = best_score
    # Limit the number of swap attempts to keep within budget.
    max_swaps = 10000
    swaps_done = 0
    while time.monotonic() < deadline and swaps_done < max_swaps:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cand = current_seq.copy()
        cand[i], cand[j] = cand[j], cand[i]
        sc = _score(cand, fft_len)
        swaps_done += 1
        if sc < current_score:
            current_seq, current_score = cand, sc
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
        # occasional restart from the global best
        if rng.random() < 0.001:
            current_seq = best_seq.copy()
            current_score = best_score

    # ----------------------------------------------------------
    # Return the best sequence found as a plain Python list.
    # ----------------------------------------------------------
    return [float(x) for x in best_seq]
