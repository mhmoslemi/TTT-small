import time
import numpy as np

def _catalan_weights(m: int) -> np.ndarray:
    """First m Catalan‑style weights w_i = C(2i,i)/4**i."""
    w = np.empty(m, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, m):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _conv_max(a: np.ndarray, size: int) -> tuple:
    """Full self‑convolution of a using FFT, returning (max, full array)."""
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)[: 2 * a.size - 1]
    return float(c.max()), c


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic projected‑gradient search for a step‑function that minimises
    the AC1 upper bound defined by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 1.  Load a parent construction if available, otherwise fall back.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = int(parent.size)

    # ------------------------------------------------------------------
    # 2.  Two quick initial candidates: Catalan weights and the parent.
    # ------------------------------------------------------------------
    cand_catalan = _catalan_weights(n)
    cand_catalan = cand_catalan / cand_catalan.sum()          # normalised
    cand_parent = parent / parent.sum()

    score_catalan = float(evaluate_sequence(cand_catalan.tolist()))
    score_parent = float(evaluate_sequence(cand_parent.tolist()))

    if score_catalan <= score_parent:
        best = cand_catalan.copy()
        best_score = score_catalan
    else:
        best = cand_parent.copy()
        best_score = score_parent

    # working copy of the current iterate
    a = best.copy()
    cur_score = best_score

    # pre‑compute FFT size (next power of two >= 2n‑1)
    fft_size = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop
    # ------------------------------------------------------------------
    step_frac = 0.03          # typical transferred fraction of the source mass
    min_step = 1e-4
    jitter_std = 0.012
    no_improve = 0            # counter for stagnation‑triggered restarts

    while time.monotonic() < deadline:
        # ----- convolution & most‑violated index ---------------------------------
        M, conv = _conv_max(a, fft_size)
        k_max = int(conv.argmax())

        # ----- gradient of the active convolution term (a_i + a_{k_max-i}) -----
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        grad = a.copy()

        # ----- choose source (large gradient) and target (small gradient) ----------
        i_src = int(grad.argmax())
        i_tgt = int(grad.argmin())

        # If the source slot is essentially empty, fall back to a random move
        if a[i_src] < 1e-14 or i_src == i_tgt:

            if rng.random() < 0.02:

                cand /= cand.sum()
                s = float(evaluate_sequence(cand.tolist()))
                if s < best_score:
                    best, best_score = cand, s
                    a, cur_score = cand, s
                    step_frac = min(step_frac * 1.2, 0.2)

                no_improve = 0

    # ------------------------------------------------------------------
    # 4.  Return the best sequence found (plain Python list)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
