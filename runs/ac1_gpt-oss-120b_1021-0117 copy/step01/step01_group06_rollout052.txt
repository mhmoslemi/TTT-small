def _sqrt_series(n):

    a[0] = 1.0

            s = 0.0

            s = np.dot(a[1:k], a[k-1:0:-1])  # a[1:k] dot reversed a[1:k]

import time, numpy as np

def _sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = np.dot(a[1:k], a[k-1:0:-1])
        a[k] = (1.0 - s) * 0.5
    return a

def _max_convolution(x):
    n = x.size
    L = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * n - 1]
    return conv.max()

def _score(x):
    s = x.sum()
    if s == 0.0:
        return np.inf
    maxc = _max_convolution(x)
    n = x.size
    return float(2.0 * n * maxc / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine problem size
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).size)
    else:
        n = 256

    # initial vector – analytic sqrt series
    a = _sqrt_series(n)
    a = np.maximum(a, 0.0)
    # scale to sum 1 – scale does not affect the bound but keeps numbers stable
    a = a / a.sum()
    best = a.copy()
    best_score = _score(best)

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # annealing parameters
    T0 = 0.05  # initial temperature (in score units)
    # main loop
    iter_count = 0
    while time.monotonic() < deadline:
        iter_count += 1
        # propose a simple mass transfer
        i = rng.integers(0, n)
        if best[i] == 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # transfer a small fraction of i's mass
        frac = rng.random() * 0.25
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep numeric safety
        if cand[i] < 0:
            continue
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        else:
            # annealing acceptance
            t = (deadline - time.monotonic()) / (deadline - start)
            T = T0 * t
            if T > 0 and np.log(rng.random()) < -(cand_score - best_score) / T:
                best = cand
                best_score = cand_score

        # occasional larger perturbation (3‑way re‑distribution)
        if rng.random() < 0.01:
            idx = rng.choice(n, size=3, replace=False)
            total = best[idx].sum()
            # generate three random fractions that sum to 1
            w = rng.random(3)
            w /= w.sum()
            cand2 = best.copy()
            cand2[idx] = total * w
            cand2_score = _score(cand2)
            if cand2_score < best_score:
                best = cand2
                best_score = cand2_score

    # Return as plain Python list
    return [float(v) for v in best]

import time
import numpy as np

# ------------------------------------------------------------
# analytic √‑series (a*a = 1+…+x^{2n‑2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # sum_{i=1}^{k-1} a_i * a_{k-i}
            s = np.dot(a[1:k], a[k - 1 : 0 : -1])
        a[k] = (1.0 - s) * 0.5
    return a

# ------------------------------------------------------------
# max‑value of the full convolution a*a
# ------------------------------------------------------------
def _max_convolution(v: np.ndarray) -> float:
    n = v.size
    L = 1 << ((2 * n - 1).bit_length())     # next power of two
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(conv.max())

# ------------------------------------------------------------
# AC1 bound (exactly the same formula as evaluate_sequence)
# ------------------------------------------------------------
def _score(v: np.ndarray) -> float:
    s = v.sum()
    if s == 0.0:
        return float("inf")
    maxc = _max_convolution(v)
    return float(2.0 * v.size * maxc / (s * s))

# ------------------------------------------------------------
# main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # problem size – use the supplied parent if present
    if "height_sequence_1" in globals():
        n = int(np.asarray(height_sequence_1).size)
    else:
        n = 256

    # start from the analytic √‑series solution
    a = _sqrt_series(n)
    a = np.maximum(a, 0.0)          # safety
    a = a / a.sum()                 # normalise to sum = 1 (scale‑invariant)
    best = a.copy()
    best_score = _score(best)

    # a mutable copy used for simulated‑annealing moves
    cur = best.copy()
    cur_score = best_score

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # simple linear cooling schedule
    T0 = 0.05   # initial temperature (in score units)

    while time.monotonic() < deadline:
        # ---- propose a two‑index mass transfer -----------------
        i = rng.integers(0, n)
        if cur[i] == 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # transfer a modest fraction of the source mass
        delta = cur[i] * rng.random() * 0.20
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0.0:
            continue
        cand_score = _score(cand)

        # ---- acceptance logic ---------------------------------
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            cur = cand
            cur_score = cand_score
        else:
            # temperature proportional to remaining time
            t_frac = (deadline - time.monotonic()) / (deadline - start)
            T = T0 * t_frac
            if T > 0.0 and np.log(rng.random()) < -(cand_score - cur_score) / T:
                cur = cand
                cur_score = cand_score

        # ---- occasional three‑way redistribution ---------------
