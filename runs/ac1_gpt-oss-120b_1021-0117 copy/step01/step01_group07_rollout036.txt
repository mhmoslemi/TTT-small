    """Return first n Catalan‑style coefficients w_i = C(2i,i)/4**i."""

    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w

            best_score = sc
            best = cand.copy()
    if best is None:               # safety net – should never happen
        best = uniform.copy()
        best_score = _score(best)

    # ----------------------------------------------------------------------
    # 5. Main optimisation loop (projected gradient on the peak location)
    # ----------------------------------------------------------------------
    a = best.copy()
    step = 0.3                     # initial step magnitude
    stagnation = 0                 # counts consecutive non‑improving rounds

    while time.monotonic() < deadline:
        # compute current convolution and locate a maximal index
        conv = np.convolve(a, a)
        M = conv.max()
        max_idxs = np.where(conv >= M - 1e-12)[0]
        j = int(rng.choice(max_idxs))

        # gradient of the peak (sub‑gradient of the max‑norm)
        lo = max(0, j - (n - 1))
        hi = min(n - 1, j)
        grad = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] = 2.0 * a[j - idx]

        improved = False
        # try a few step factors around the current one
        for factor in (step, step * 0.5, step * 1.5):
            cand = a - factor * grad
            cand = np.where(cand < 0.0, 0.0, cand)
            cand = _proj_simplex(cand)
            sc = _score(cand)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                best, best_score = cand.copy(), sc
                a = cand.copy()
                improved = True
                step = min(step * 1.2, 2.0)      # accelerate after success
                stagnation = 0
                break

        if improved:
            if best_score <= 1.50274349:
                break
            continue

        # ---------------------------------------------------------------
        # No improvement – shrink step and possibly jitter
        # ---------------------------------------------------------------
        step = max(step * 0.7, 1e-5)
        stagnation += 1

        if stagnation >= 30 or rng.random() < 0.02:
            # random multiplicative jitter
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            cand = _proj_simplex(a * jitter)
            sc = _score(cand)
            if math.isfinite(sc) and sc < best_score:
                best, best_score = cand.copy(), sc
                a = cand.copy()
                step = 0.3
                stagnation = 0
                continue

            # small random mass transfer
            i, j2 = rng.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                eps = rng.random() * a[i] * 0.4
                cand = a.copy()
                cand[i] -= eps
                cand[j2] += eps
                cand = _proj_simplex(cand)
                sc = _score(cand)
                if math.isfinite(sc) and sc < best_score:
                    best, best_score = cand.copy(), sc
                    a = cand.copy()
                    step = 0.3
                    stagnation = 0

    # ----------------------------------------------------------------------
    # 6. Return the best found coefficient list
    # ----------------------------------------------------------------------
    return [float(v) for v in best]
