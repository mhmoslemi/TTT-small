    n = len(seq)

    L = int(2**np.ceil(np.log2(2*n-1)))  # power of two
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)
    return conv[:2*n-1].max()

import numpy as np
import time

def arcsine_like_sequence(n: int) -> np.ndarray:
    """Return a discretised arcsine density on [0, n‑1].

    a_i = 1 / sqrt((i+0.5)*(n-0.5-i))
    The sequence is non‑negative, sum≈π and its self‑convolution is almost flat,
    giving an AC1 bound close to 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return a


def _sidon_like_sequence(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4**i sequence (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def _fast_score(seq: np.ndarray, n: int, L: int) -> float:
    """Compute the AC1 bound using FFT (much faster than the full evaluator)."""
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)
    max_b = conv[: 2 * n - 1].max()
    s = seq.sum()
    return 2.0 * n * max_b / (s * s)
