def _sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = np.dot(a[1:k], a[k-1:0:-1])
        a[k] = (1.0 - s) / 2.0
    return a

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_idx = np.where(rho_arr > 0)[0]
    if rho_idx.size == 0:
        theta = 0.0
    else:
        rho = rho_idx[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Analytic sqrt series (optional fallback)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of a where a*a = 1+…+x^{2n-2}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # sum_{i=1}^{k-1} a_i * a_{k-i}
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ------------------------------------------------------------
# Fast AC1 bound (same formula as evaluate_sequence)
# ------------------------------------------------------------
def _fast_score(vec: np.ndarray, L: int) -> float:
    n = vec.size
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    M = conv.max()
    s = vec.sum()
    return float(2.0 * n * M / (s * s))

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # -----------------------------------------------------------------
    # initialise a
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        a = _sqrt_series(n)
    n = a.size
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / s                         # normalise to sum = 1

    # -----------------------------------------------------------------
    # pre‑compute FFT length (power of two >= 2n‑1)
    # -----------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # initial score and convolution
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    cur_score = _fast_score(a, L)

    best = a.copy()
    best_score = cur_score

    # algorithmic parameters
    step0 = 0.12               # initial step magnitude
    iter_cnt = 0
    # -----------------------------------------------------------------
    # main optimisation loop
    # -----------------------------------------------------------------
    while time.monotonic() < deadline:
        iter_cnt += 1

        # index where convolution attains its maximum
        i0 = int(conv.argmax())

        # sub‑gradient of the maximal autocorrelation term
        g = np.zeros_like(a)
        start = max(0, i0 - (n - 1))
        end = min(n - 1, i0)
        idx = np.arange(start, end + 1)
        j = i0 - idx
        g[idx] = 2.0 * a[j]
