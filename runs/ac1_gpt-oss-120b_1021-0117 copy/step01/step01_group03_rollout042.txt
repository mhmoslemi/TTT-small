def compute_score(arr):
    a = np.asarray(arr, dtype=np.float64)
    n = a.size
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa*fa, n=m)[: 2*n-1]
    maxb = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * maxb / (s*s)

    def conv_max(arr):

        fa = np.fft.rfft(arr, n=m)
        conv = np.fft.irfft(fa*fa, n=m)[:2*len(arr)-1]
        return conv.max()
    def score(arr):
        nlen = len(arr)
        mx = conv_max(arr)
        s = arr.sum()
        if s < 0.01:
            return np.inf
        return 2.0 * nlen * mx / (s*s)

    def sqrt_series(n):

        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i-1] * (2.0*i - 1.0) / (2.0*i)
        return a

        best = parent.copy()
        best_score = score(best)
    else:
        best = sqrt_series(n)
        best_score = score(best)

    # Also evaluate sqrt series if parent present and better
    sqrt_seq = sqrt_series(n)
    sqrt_score = score(sqrt_seq)

    # Use small number of random draws within time.
    while time.monotonic() < deadline:

def fast_score(arr):
    n = arr.size
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(arr, n=m)

    s = arr.sum()
    if s < .01:
        return np.inf
    return 2.0 * n * maxb / (s * s)

def reversed_part(a, k):
    # a is 1D array length n
    # output b where b[j] = a[k-j] if 0 <= k-j < n else 0
    n = a.size
    # compute lower and upper bounds for j:
    low = max(0, k - (n - 1))
    high = min(k, n - 1)
    b = np.zeros_like(a)
    # For j in [low, high], set b[j] = a[k-j]
    # vectorized:
    idx = np.arange(low, high+1)
    b[idx] = a[k - idx]
    return b

import time
import numpy as np

def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (O(n log n))."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score(arr: np.ndarray) -> float:
    """Exact objective used by the authoritative evaluator."""
    n = arr.size
    mx = _conv_full(arr).max()
    s = arr.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * mx / (s * s))


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a

        best, best_score = sqrt_seq.copy(), sc

    # analytic candidate: p=0.5, beta=1/n (exponential taper)
    i = np.arange(n, dtype=np.float64)
    analytic = (i + 1.0) ** (-0.5) * np.exp(-i / n)
    sc = _score(analytic)
    if sc < best_score:
        best, best_score = analytic, sc

    # ------------------------------------------------------------------
    # Helper to compute the reversed part for a given convolution index.
    # ------------------------------------------------------------------
    def _reversed_part(arr: np.ndarray, k: int) -> np.ndarray:
        """Return vector v where v[j] = arr[k-j] if valid, else 0."""
        nloc = arr.size
        low = max(0, k - (nloc - 1))
        high = min(k, nloc - 1)
        out = np.zeros_like(arr)
        if low <= high:
            idx = np.arange(low, high + 1)
            out[idx] = arr[k - idx]
        return out

    # ------------------------------------------------------------------
    # Main search loop.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 1) Random parametric shapes (power law + optional exponential)
        # --------------------------------------------------------------
        if rng.random() < 0.63:
            p = rng.uniform(0.30, 0.80)                     # power‑law exponent
            # log‑uniform beta in [1e‑6, 0.01]
            logb = rng.uniform(np.log(1e-6), np.log(0.01))
            beta = np.exp(logb)
            alpha = rng.uniform(0.0, 200.0)                 # shift for the denominator
            cand = (i + 1.0 + alpha) ** (-p) * np.exp(-beta * i)
            sc = _score(cand)
            if sc < best_score:
                best, best_score = cand, sc

        # --------------------------------------------------------------
        # 2) Gradient‑like descent using the currently worst convolution.
        # --------------------------------------------------------------
        else:
            arr = best.copy()
            conv = _conv_full(arr)
            k_max = int(conv.argmax())
            max_val = conv[k_max]
            total = arr.sum()
            if total <= 0.0:
                continue
            lam = max_val / total                     # λ = M / Σa
            rev = _reversed_part(arr, k_max)          # a_{k*‑j}
            direction = rev - lam
            # step size (small enough to stay stable)
            eta = 0.07
            cand = arr - eta * direction
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = _score(cand)
            if sc < best_score:
                best, best_score = cand, sc

        # --------------------------------------------------------------
        # 3) Occasional tiny mass‑transfer move (helps escape plateaus).
        # --------------------------------------------------------------
        if rng.random() < 0.07:
            cand = best.copy()
            early = n // 4
            late = 3 * n // 4
            src = rng.integers(0, early)
            dst = rng.integers(late, n)
            max_transfer = min(cand[src], 1000.0 - cand[dst])
            if max_transfer > 0.0:
                delta = max_transfer * rng.random() * 0.05
                cand[src] -= delta
                cand[dst] += delta
                np.clip(cand, 0.0, 1000.0, out=cand)
                sc = _score(cand)
                if sc < best_score:
                    best, best_score = cand, sc

    # Return the best discovered candidate as a plain Python list.
    return [float(v) for v in best]
