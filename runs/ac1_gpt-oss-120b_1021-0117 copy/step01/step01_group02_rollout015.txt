def max_convolution(v, N, n):
    fv = np.fft.rfft(v, n=N)
    conv = np.fft.irfft(fv * fv, n=N)[:2*n - 1]
    return conv.max(), conv

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # obtain base sequence
    try:
        base = np.array(height_sequence_1, copy=False, dtype=np.float64)
    except Exception:
        # fallback length

    n = int(base.shape[0])

    # make sure non‑negative and normalise sum to 1
    seq = np.clip(base, 0.0, 1000.0)
    if seq.sum() < 1e-12:
        seq = np.full_like(seq, 1.0 / n)
    seq = seq / seq.sum()        # sum == 1

    # uniform fallback
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # choose initial current and best
    cur = seq.copy()
    best = seq.copy()

    # FFT size (next power of two >= 2n)
    N = 1 << ((2 * n - 1).bit_length())

    # evaluate scores (sum == 1)
    cur_max, cur_conv = max_convolution(cur, N, n)
    cur_score = 2.0 * n * cur_max
    best_score = cur_score
    best_max = cur_max

    # also consider uniform as a possible better start
    u_max, _ = max_convolution(uniform, N, n)
    u_score = 2.0 * n * u_max
    if u_score < best_score:
        cur = uniform.copy()
        best = uniform.copy()
        cur_max, cur_conv = u_max, None

        best_score = u_score
        best_max = u_max

    step_factor = 0.5
    min_step = 1e-9

    decay = 0.9995
    iter_count = 0

    while time.monotonic() < deadline:
        iter_count += 1

        cur_max, cur_conv = max_convolution(cur, N, n)

    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # ---- FFT size (next power of two ≥ 2n) ------------------------------
    N = 1 << ((2 * n - 1).bit_length())

    # ---- initialise current / best ----------------------------------------
    cur_max, cur_conv = max_convolution(cur, N, n)
    cur_score = 2.0 * n * cur_max

    best = cur.copy()
    best_score = cur_score
    best_max = cur_max

    # maybe uniform is already better (unlikely)
    u_max, _ = max_convolution(uniform, N, n)
    u_score = 2.0 * n * u_max
    if u_score < best_score:
        cur = uniform.copy()
        best = uniform.copy()
        cur_max, cur_conv = u_max, None
        cur_score = u_score
        best_score = u_score
        best_max = u_max

    # ---- parameters -------------------------------------------------------
    step_factor = 0.8          # initial fraction of source mass to move
    min_step = 1e-12
    temperature = 0.12
    decay = 0.9995

    while time.monotonic() < deadline:
        # recompute full convolution for the current state
        cur_max, cur_conv = max_convolution(cur, N, n)
        k = int(cur_conv.argmax())

        # partner[i] = cur[k-i] for indices that overlap, else 0
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = np.zeros(n, dtype=np.float64)
        partner[idx] = cur[k - idx]

        # contributions to the peak = cur[i] * partner[i]
        contrib = cur[idx] * partner[idx]
        src_rel = int(contrib.argmax())
        src = i_start + src_rel

        # destination where partner is minimal (usually zero)
        min_val = partner.min()
        dest_cand = np.where(partner == min_val)[0]
        if dest_cand.size:
            dest = int(rng.choice(dest_cand))
        else:
            dest = int(rng.integers(0, n))

        if src == dest:
            # random local perturbation as a cheap escape
            width = max(1, n // 150)
            start = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.18, size=width))
            cand = cur.copy()
            cand[start:start + width] *= factor
            cand = cand / cand.sum()
            cand_max, _ = max_convolution(cand, N, n)
            cand_score = 2.0 * n * cand_max
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
            continue

        # amount to transfer from src to dest
        delta = cur[src] * step_factor
        if delta < min_step:
            delta = min_step
        if delta > cur[src]:
            delta = cur[src]

        cand = cur.copy()
        cand[src] -= delta
        cand[dest] += delta

        # sum stays (≈) 1, no clipping needed (values stay ≤ 1 ≪ 1000)
        cand_max, _ = max_convolution(cand, N, n)
        cand_score = 2.0 * n * cand_max

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = cand
            cur_score = cand_score

        # record best ever seen
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            best_max = cand_max

        # schedule
        temperature *= decay
        if cand_score < best_score + 1e-12:
            step_factor = max(0.01, step_factor * 0.997)

        # occasional random shake to leave plateaus
        if rng.random() < 0.007:
            width = max(1, n // 80)
            start = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.25, size=width))
            shake = cur.copy()
            shake[start:start + width] *= factor
            shake = shake / shake.sum()
            shake_max, _ = max_convolution(shake, N, n)
            shake_score = 2.0 * n * shake_max
            if shake_score < cur_score:
                cur = shake
                cur_score = shake_score
                if shake_score < best_score:
                    best = shake.copy()
                    best_score = shake_score
                    best_max = shake_max

    # ---- final sanity ----------------------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final.tolist()]
