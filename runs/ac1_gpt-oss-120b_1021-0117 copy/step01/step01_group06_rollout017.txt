import time, numpy as np

def _project_simplex(v):
    # same as before
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max(a, L):
    # compute full convolution via FFT and return max
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*a.size-1]
    return conv.max(), conv

def _score(a, L=None):
    n = a.size
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2 * n - 1]
    return 2.0 * n * np.max(conv) / (np.sum(a) ** 2)

        a = np.maximum(a, 0.0)

        a = np.full(n, 1.0 / n, dtype=np.float64)

    # normalize to simplex (sum=1)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / a.size
    else:
        a /= total

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # initial scores
    cur_score = _score(a, L)
    best = a.copy()
    best_score = cur_score

    # parameters
    step0 = 0.2
    decay = 0.9995
    iteration = 0
    # store current convolution for reuse

        # compute convolution and max index

        max_idx = int(np.argmax(conv_full))

        cand = a - step * g
        cand = _project_simplex(cand)
        cand_score = _score(cand, L)

        if cand_score < cur_score:
            a = cand

        else:

import time
import numpy as np

# ------------------------------------------------------------
# simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# fast convolution max via FFT (full linear convolution)
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int):
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return conv.max(), conv


# ------------------------------------------------------------
# score identical to the evaluator (scale invariant)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.size
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return 2.0 * n * np.max(conv) / (np.sum(a) ** 2)


# ------------------------------------------------------------
# analytic sqrt series (a*a = 1+...+x^{2n-2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # dot of a[1:k] with reversed a[1:k]
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a
