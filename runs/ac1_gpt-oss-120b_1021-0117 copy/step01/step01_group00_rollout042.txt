def proj_simplex(v):
    # ensures sum=1, all >=0
    # v is 1D numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def ac1_score(a):
    n = len(a)
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]

def ac1_fft(a):
    n = a.shape[0]

    fa = np.fft.rfft(a, size)

import numpy as np
import time

def _sidon_like_sequence(n):
    """Classic Sidon-like decreasing sequence a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a

def _next_pow2(x):
    """Small helper: next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _ac1_fast(a, n=None, fft_len=None):
    """Fast AC1 computation using FFT (identical to the official evaluator)."""
    if n is None:
        n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    maxb = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * maxb / (s * s)
