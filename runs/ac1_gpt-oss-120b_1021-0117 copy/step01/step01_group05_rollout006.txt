                best_ever = cand.copy()
                best_ever_score = cand_score
                last_improve = it

        # --------------------------------------------------------------
        # Stagnation handling – restart from the globally best ever if needed
        # --------------------------------------------------------------
        if it - last_improve > no_improve_limit:
            # small jitter around the best ever to break possible plateaus
            best = best_ever.copy()
            jitter = np.exp(rng.normal(0.0, 0.025, size=n))
            best *= jitter
            np.clip(best, 0.0, 1000.0, out=best)
            best_score, best_conv, _ = _fast_score(best, m)
            last_improve = it

        # occasional time check
        if it % 5000 == 0 and time.monotonic() >= deadline:
            break

    # Return the best sequence ever found as a plain Python list
    return [float(x) for x in best_ever]
