import time
import numpy as np

def _score(arr: np.ndarray) -> float:
    """Exact AC1 upper bound for a given 1‑D array."""
    n = arr.size
    conv = np.convolve(arr, arr)
    return 2.0 * n * conv.max() / (arr.sum() ** 2)


def _contrib(f: np.ndarray, peak_idxs: np.ndarray) -> np.ndarray:
    """Sum of contributions of each coordinate to the chosen peak indices."""
    n = f.size
    contrib = np.zeros_like(f)
    max_val = np.max(np.convolve(f, f))
    for k in peak_idxs:
        # weight peaks by their relative height (helps focus on the true maximum)
        w = (np.convolve(f, f)[k] / max_val) if max_val > 0 else 1.0
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        contrib[idx] += f[k - idx] * w
    return contrib

    binom[0] = 1.0
