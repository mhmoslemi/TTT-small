import numpy as np

def _conv_self(arr: np.ndarray) -> np.ndarray:
    """Linear convolution of arr with itself using a real FFT."""
    n = arr.shape[0]
    m = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(arr, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score(arr: np.ndarray) -> float:
    """Exact objective used by the authoritative evaluator."""
    b = _conv_self(arr)
    max_b = b.max()
    s = arr.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * arr.shape[0] * max_b / (s * s)
