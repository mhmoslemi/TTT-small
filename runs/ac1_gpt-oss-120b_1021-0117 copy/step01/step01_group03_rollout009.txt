def _conv_self(a):
    n = a.size
    m = 1 << ((2*n -1).bit_length())
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[:2*n-1]
    return conv
```

import numpy as np
import time

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Fast linear convolution of a with itself using real FFT."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (scale‑invariant)."""
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a
