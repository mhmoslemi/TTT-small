    best_seq = cur.copy()
    best_score = cur_score

    compute T0 = best_score * 0.1 (or 0.05); T = T0

    while time.time < deadline:
        candidate = cur.copy()
        pick op = random 0-5

        apply modifications accordingly.

        ensure 0 <= candidate <=1000.

        cand_score, cand_conv, _ = fast_score(candidate)

        # acceptance
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score)/T):
            cur = candidate; cur_score = cand_score; cur_conv = cand_conv

        if cand_score < best_score:
            best_seq = candidate.copy(); best_score = cand_score; best_conv = cand_conv

        # update temperature: linear cooling
        progress = (time.monotonic() - start_time) / (deadline - start_time)

            start = rng.integers(0, n - width + 1)
            factor = math.exp(rng.normal(0.0, 0.09))
            candidate[start:start+width] *= factor

        elif op == 1:
            # point scaling
            idx = rng.integers(0, n)
            factor = math.exp(rng.normal(0.0, 0.25))
            candidate[idx] *= factor

        elif op == 2:

            # determine max conv index
            k_max = int(np.argmax(cur_conv))
            # compute contributions
            idx_arr = np.arange(n)
            partner = k_max - idx_arr

            contrib.fill(0.0)
            contrib[mask] = cur[mask] * cur[partner[mask]]
            i = int(np.argmax(contrib))
            j = k_max - i
            # remove a fraction from i and j (if distinct)
            alpha = rng.uniform(0.04, 0.12)
            # ensure nonnegative after removal
            delta_i = cur[i] * alpha
            candidate[i] -= delta_i
            total_removed = delta_i
            if i != j:

                candidate[j] -= delta_j
                total_removed += delta_j
            # add to random low-value position
            # choose target not equal to i or j
            tgt = rng.integers(0, n)
            # ensure not i or j; loop limited steps
            while tgt == i or tgt == j:
                tgt = rng.integers(0, n)
            candidate[tgt] += total_removed

        elif op == 3:
            # smooth window of size 2..5
            w = rng.integers(2, 6)
            start = rng.integers(0, n - w + 1)
            avg = np.mean(candidate[start:start+w])

        elif op == 4:

            # pick high idx among top 10% values
            # compute 90th percentile threshold

            high_idxs = np.where(cur >= thresh)[0]
            if high_idxs.size == 0:

            low_thresh = np.percentile(cur, 10)
            low_idxs = np.where(cur <= low_thresh)[0]

                low_idx = int(rng.choice(low_idxs))
            # transfer
            alpha = rng.uniform(0.05, 0.2)
            transfer = cur[high_idx] * alpha
            candidate[high_idx] -= transfer
            candidate[low_idx] += transfer

        else:  # op ==5
            # global smoothing
            # conv with kernel [0.25, 0.5, 0.25]
            # simple moving average
            smooth = np.empty_like(candidate)
            smooth[0] = 0.5 * candidate[0] + 0.5 * candidate[1]
            smooth[-1] = 0.5 * candidate[-2] + 0.5 * candidate[-1]
            # interior
            smooth[1:-1] = 0.25 * candidate[:-2] + 0.5 * candidate[1:-1] + 0.25 * candidate[2:]
            # preserve sum: scale
            sum_before = candidate.sum()
            sum_smooth = smooth.sum()
            if sum_smooth > 0:
                smooth *= (sum_before / sum_smooth)
            candidate = smooth

        # clip to bounds
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        cand_score, cand_conv, _ = _fast_score(candidate)

        # acceptance: Metropolis; if better accept
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur = candidate
            cur_score = cand_score
            cur_conv = cand_conv

        if cand_score < best_score:
            best = candidate.copy()
            best_score = cand_score
            best_conv = cand_conv

    # final return
    return [float(x) for x in best]

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the classic binomial (central‐binomial) decreasing shape
# ----------------------------------------------------------------------
def _initial_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ----------------------------------------------------------------------
# Helper: compute the AC1 upper bound using an FFT‑based self‑convolution
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray):
    """Return (score, full convolution, max value)."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val
