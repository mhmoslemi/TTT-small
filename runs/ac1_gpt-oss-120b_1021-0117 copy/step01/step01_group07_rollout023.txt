import time, numpy as np

def _catalan_weights(n):
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i-1] * ((2*i)*(2*i-1)) / (i*i*4.0)
    return w

def _conv_max(a):
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A*A, L)
    c = C[:2*n-1]
    idx = int(np.argmax(c))
    return float(c[idx]), idx

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Determine length n and fallback parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = None

    if parent is not None:
        n = int(parent.size)
        parent_norm = parent / max(parent.sum(), 1e-12)
    else:
        n = 2000
        parent_norm = None

    # Helper to compute score from array (sum assumed 1)
    def score_of(a):
        max_b, _ = _conv_max(a)
        return 2.0 * n * max_b

    # Initial candidates
    best = np.ones(n, dtype=np.float64) / n   # uniform
    best_score = score_of(best)

    # Catalan
    w = _catalan_weights(n)
    a_cat = w / w.sum()
    sc = score_of(a_cat)
    if sc < best_score:
        best = a_cat.copy()
        best_score = sc

    # Parent (if any)
    if parent_norm is not None:
        a_par = parent_norm.copy()
        sc = score_of(a_par)
        if sc < best_score:
            best = a_par.copy()
            best_score = sc

    # Parameter sweep: power law
    alphas = np.linspace(0.30, 1.20, 19)   # 0.3..1.2 step .05
    i_vals = np.arange(1, n+1, dtype=np.float64)   # 1..n
    for alpha in alphas:
        w = i_vals ** (-alpha)
        a = w / w.sum()
        sc = score_of(a)
        if sc < best_score:
            best = a.copy()
            best_score = sc

    # Parameter sweep: exponential decay
    rs = np.linspace(0.96, 0.9999, 20)
    idx = np.arange(n, dtype=np.float64)
    for r in rs:
        w = r ** idx
        a = w / w.sum()
        sc = score_of(a)
        if sc < best_score:
            best = a.copy()
            best_score = sc

    # Local refinement
    a = best.copy()
    while time.monotonic() < deadline:
        max_b, t = _conv_max(a)
        cur_score = 2.0 * n * max_b
        # record best again (might have improved outside)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            if best_score <= 1.50274349:
                break

        # gradient indices range
        lo = max(0, t - (n - 1))
        hi = min(n - 1, t)
        idx_range = np.arange(lo, hi+1)
        # gradient values proportional to a[t-idx]
        grad = 2.0 * a[t - idx_range]
        # choose source index weighted by gradient (high impact)

            # fallback random move
            i_src = rng.integers(0, n)

import time
import numpy as np

def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‐style coefficients w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _conv_max(a: np.ndarray):
    """Return (max convolution value, index) for a*a using FFT (linear convolution)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A * A, L)                   # cyclic conv of length L
    conv = C[: 2 * n - 1]                        # linear part
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx
