    start = time.monotonic()
    deadline = start + budget_s - 2.0

    # get n, parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.size)
    except Exception:
        n = 256
        parent = None

    L = 1 << ((2*n - 2).bit_length())  # next power of two >= 2n - 1

    def _ac1_score(arr):
        # arr sum maybe not =1
        fa = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fa * fa, n=L)
        max_b = conv[:2*n-1].max()
        s = arr.sum()
        if s < 1e-12:
            return float('inf')
        return 2.0 * n * max_b / (s * s)

    # initial candidates
    # binomial
    base = np.empty(n, dtype=np.float64)
    base[0] = 1.0
    for i in range(1, n):
        base[i] = base[i-1] * (2*i - 1) / (2*i)
    base = base / base.sum()  # normalize
    best = base.copy()
    best_score = _ac1_score(best)

    if parent is not None:
        # ensure parent normalized? not needed; evaluate as is; but must keep sum: evaluator will compute ratio anyway.
        parent_norm = parent.copy()
        # ensure sum >0
        if parent_norm.sum() > 0:
            parent_norm = parent_norm / parent_norm.sum()
        parent_score = _ac1_score(parent_norm)
        if parent_score < best_score:
            best = parent_norm.copy()
            best_score = parent_score

    cur = best.copy()
    cur_score = best_score

    rng = np.random.default_rng(seed)
    temperature = 1.0
    decay = 0.9997

    while time.monotonic() < deadline:
        # compute conv of current for target move selection
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        max_b = conv.max()
        k_max = int(conv.argmax())
        # choose move
        r = rng.random()
        cand = cur.copy()
        if r < 0.3:
            # block flatten
            width = int(rng.integers(max(2, n // 200), max(4, n // 10)))
            start_idx = int(rng.integers(0, max(1, n - width + 1)))
            block = cand[start_idx:start_idx+width]
            mean = block.mean()
            block[:] = mean
        elif r < 0.55:

            width = int(rng.integers(max(2, n // 200), max(4, n // 12)))
            start_idx = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.1))
            cand[start_idx:start_idx+width] *= factor

            j = int(rng.integers(0, n))

                cand[j] += delta

                    cand[q] += eps

            cur_score = best_score

    return [float(x) for x in best]

import time, math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """
    a[i] = C(2i,i)/4**i for i = 0..n-1.
    Recurrence: a[0]=1, a[i]=a[i-1]*(2i-1)/(2i)
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a
