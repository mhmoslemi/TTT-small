import numpy as np
import time
import math

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Build a step‑function whose non‑zero heights form a Sidon set.
    With at most one unordered pair per convolution index the
    maximum convolution value is 2·C² and the AC1 bound becomes
        4·n / s²   (C = 1000, s = number of non‑zero entries).
    Choosing s large enough (≈124 for n≈5582) yields a bound
    below the target 1.50274349.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # length of the hidden parent (if present) or the usual fallback
    if "height_sequence_1" in globals():
        n = int(np.asarray(height_sequence_1, dtype=np.float64).size)
    else:
        n = 5582

    target_bound = 1.50274349
    # minimal size that guarantees a bound < target (4·n / s² < target)
    min_s = int(math.ceil(math.sqrt(4.0 * n / target_bound)))
    # a small safety margin
    goal_s = max(min_s, 130)

    # ------------------------------------------------------------------
    # fallback candidate – uniform heights (bound = 2) – keep it just in case
    # ------------------------------------------------------------------
    uniform = np.full(n, 1000.0, dtype=np.float64)
    best_seq = uniform.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float("inf")

    # ------------------------------------------------------------------
    # Random greedy construction of a Sidon set.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # random ordering of the admissible indices (skip 0 – it creates many collisions)
        order = rng.permutation(np.arange(1, n, dtype=np.int64))
        used_sums = set()
        chosen = []

        for idx in order:
            diag = idx + idx
            if diag in used_sums:
                continue
            # check pairwise sums against the existing set
            conflict = False
            for j in chosen:
                if (idx + j) in used_sums:
                    conflict = True
                    break
            if conflict:
                continue

            # accept this index
            chosen.append(int(idx))
            used_sums.add(diag)
            for j in chosen[:-1]:          # add sums with all previously chosen entries
                used_sums.add(idx + j)

            if len(chosen) >= goal_s:      # enough elements – stop scanning
                break

        # if we have at least the minimal required size, evaluate the candidate
        if len(chosen) >= min_s:
            cand = np.zeros(n, dtype=np.float64)
            cand[chosen] = 1000.0
            score = evaluate_sequence(cand.tolist())
            if math.isfinite(score) and score < best_score:
                best_score = score
                best_seq = cand.copy()
                if best_score < target_bound:
                    break     # we already beat the target, stop early

        # give a chance to try another random ordering if time permits
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq]
