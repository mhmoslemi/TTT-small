def _catalan_weights(k: int) -> np.ndarray:
    """Catalan style coefficients w_i = C(2i,i)/4**i for i=0..k-1."""
    w = np.empty(k, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, k):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w
