import numpy as np, time, math

def _fast_ac1_score(seq):
    # seq: numpy array of nonnegative floats
    s = seq.sum()
    if s < 1e-12:
        return float('inf')
    # use FFT convolution
    n = seq.shape[0]
    # compute next power of 2 size >= 2*n-1
    size = 1 << ((2*n-1).bit_length())
    # FFT
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)
    max_b = conv[:2*n-1].max()
    return 2.0 * n * max_b / (s*s)

    try:

    # C proportional to log(n)
    logn = math.log(n) if n > 1 else 1.0

import numpy as np, time, math

def _fast_ac1_score(seq: np.ndarray) -> float:
    """Fast evaluation of the AC1 bound for a non‑negative sequence."""
    s = seq.sum()
    if s < 1e-12:
        return float('inf')
    n = seq.shape[0]
    # size = next power of two ≥ 2n‑1 (FFT length)
    size = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)
    max_b = conv[: 2 * n - 1].max()
    return 2.0 * n * max_b / (s * s)
