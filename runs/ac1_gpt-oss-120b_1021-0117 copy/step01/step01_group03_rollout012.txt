def shift_move(a):
    conv = _conv_self(a)
    max_idx = int(np.argmax(conv))
    n = a.size
    g = np.zeros_like(a)
    # fill overlapping region
    start_i = max(0, max_idx - (n-1))
    end_i = min(n-1, max_idx)  # inclusive?
    # indices i where 0 <= max_idx - i < n -> i in [max_idx - (n-1), max_idx]
    # We'll compute via vectorized indexing:
    i = np.arange(start_i, end_i+1)
    g[i] = a[max_idx - i]
    contributions = a * g
    src = int(np.argmax(contributions))
    # for dst, choose index with minimal g (maybe zero)
    dst_candidates = np.where(g == 0)[0]  # zero contributions free
    if dst_candidates.size > 0:
        dst = int(rng.choice(dst_candidates))
    else:
        dst = int(np.argmin(g))
    # Transfer mass
    if a[src] <= 1e-12:
        return a
    max_transfer = min(a[src], 1000 - a[dst])
    if max_transfer <= 1e-12:
        return a
    delta = max_transfer * rng.random() * 0.1  # up to 10% of transferable
    a2 = a.copy()
    a2[src] -= delta
    a2[dst] += delta
    return a2
```

def block_scale_move(a):
    n = a.size
    width = int(rng.integers(max(1, n//500), max(2, n//50)))  # small blocks
    start = rng.integers(0, n-width+1)
    factor = 1.0 + rng.normal(0.0, 0.05)  # small scaling
    a2 = a.copy()
    a2[start:start+width] *= factor
    # clip to [0,1000]
    np.clip(a2, 0.0, 1000.0, out=a2)
    return a2
```

import numpy as np
import time

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via real FFT (next power‑of‑2 size)."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (fast proxy)."""
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * a.size * max_b / (s * s)


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def _guided_shift(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a small amount from a high‑contribution index to a low‑gradient one."""
    n = a.size
    conv = _conv_self(a)
    k = int(np.argmax(conv))                     # lag where convolution peaks
    g = np.zeros_like(a)                         # contribution of each entry to that peak
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start > i_end:
        return a.copy()
    idx = np.arange(i_start, i_end + 1, dtype=np.int64)
    g[idx] = a[k - idx]

    contrib = a * g
    src = int(np.argmax(contrib))
    zeros = np.where(g == 0)[0]
    if zeros.size:
        dst = int(rng.choice(zeros))
    else:
        dst = int(np.argmin(g))

    if src == dst or a[src] <= 0.0:
        return a.copy()
    transferable = min(a[src], 1000.0 - a[dst])
    if transferable <= 0.0:
        return a.copy()
    delta = transferable * rng.random() * 0.15   # up to 15 % of what can be moved
    out = a.copy()
    out[src] -= delta
    out[dst] += delta
    return out


def _block_scale(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random block by a modest factor."""
    n = a.size
    low = max(1, n // 500)
    high = max(2, n // 50)
    width = int(rng.integers(low, high))
    start = rng.integers(0, n - width + 1)
    factor = 1.0 + rng.normal(0.0, 0.05)         # ≈ ±5 %
    out = a.copy()
    out[start:start + width] *= factor
    np.clip(out, 0.0, 1000.0, out=out)
    return out


def _add_noise(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Add tiny Gaussian perturbations."""
    out = a + rng.normal(0.0, 0.01, size=a.shape)
    np.clip(out, 0.0, 1000.0, out=out)
    return out


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a low‑score non‑negative step function within the time budget."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initial sequence: use supplied parent if present, else sqrt series.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = cur.size
    else:
        n = 5582                     # typical hidden length
        cur = _sqrt_series(n)

    # ------------------------------------------------------------------
    # Establish a solid starting point (parent vs sqrt series).
    # ------------------------------------------------------------------
    best = cur.copy()
    best_score = _score(best)

    sqrt_seq = _sqrt_series(n)
    sq_score = _score(sqrt_seq)
    if np.isfinite(sq_score) and sq_score < best_score:
        best = sqrt_seq.copy()
        best_score = sq_score

    # current state for exploration (may diverge from global best)
    cur_state = best.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # Main stochastic search loop.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        choice = rng.random()
        if choice < 0.60:
            cand = _guided_shift(cur_state, rng)
        elif choice < 0.85:
            cand = _block_scale(cur_state, rng)
        else:
            cand = _add_noise(cur_state, rng)

        cand_score = _score(cand)

        # simple simulated‑annealing acceptance for the current state
        if cand_score < cur_score or rng.random() < 0.001:
            cur_state = cand
            cur_score = cand_score

        # keep the best ever found
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score

    # ------------------------------------------------------------------
    # Return the best candidate as a plain Python list of floats.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
