# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray):
    """
    Return (max_value, index_of_max) of the full convolution a*a.
    a is assumed to be non‑negative and sum(a) == 1.
    """
    conv = np.convolve(a, a)
    idx = int(conv.argmax())
    return float(conv.max()), idx


def _random_dirichlet(rng, n, alpha=0.5):
    """Generate a random probability vector of length n."""
    x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0:
        x[:] = 1.0 / n
        s = 1.0
    return x / s
