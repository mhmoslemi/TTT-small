                    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
                    cur_score = 2.0 * n * float(conv.max())

        # slowly decay learning rate
        if no_imp % 500 == 0:
            lr = max(lr * 0.9, 1e-4)

    # ------------------------------------------------------------------
    # final answer: convert best vector to plain Python list
    final = [float(x) for x in best]
    try:
        _ = evaluate_sequence(final)          # sanity check – may raise
        return final
    except Exception:
        # safety fallback – a plain binomial sequence
        fallback = _generate_binomial_seq(n)
        fallback = fallback / fallback.sum()
        return [float(x) for x in fallback]
