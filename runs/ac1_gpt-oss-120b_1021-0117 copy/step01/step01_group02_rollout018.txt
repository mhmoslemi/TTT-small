import time, numpy as np

def _fft_score(a, N=None):
    n = a.shape[0]
    if N is None:
        N = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fa*fa, N)[:2*n-1]
    maxc = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * maxc / (s*s)

import time
import numpy as np

def _fft_score(a: np.ndarray, N: int) -> float:
    """Fast evaluation of the AC1 upper bound using FFT convolution."""
    n = a.shape[0]
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    maxc = float(conv.max())
    s = float(a.sum())
    return np.inf if s < 1e-12 else 2.0 * n * maxc / (s * s)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The algorithm mixes a guided mass‑transfer step with occasional random
    multiplicative shakes, both evaluated by an FFT‑based score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ---- initial sequence -------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.shape[0]
    except Exception:
        n = 5000
        base = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.clip(base, 0.0, 1000.0)
    total = a.sum()
    if total < 0.01:            # safety fallback
        a = np.full_like(a, 1.0 / n)
        total = a.sum()

    # ---- constants --------------------------------------------------------
    Nfft = 1 << ((2 * n - 1).bit_length())  # power of two >= 2n‑1
    target_sum = total

    # ---- start with a reasonable candidate ---------------------------------
    best = a.copy()
    best_score = _fft_score(best, Nfft)

    # uniform fallback (score = 2)
    uniform = np.full(n, target_sum / n, dtype=np.float64)
    uniform_score = _fft_score(uniform, Nfft)
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # ---- algorithm parameters ---------------------------------------------
    step_factor = 0.25          # fraction of source mass to move
    min_step = 1e-8
    temp = 0.02                 # annealing temperature
    temp_decay = 0.9995
    shake_sigma = 0.25          # std‑dev for multiplicative shake

    # ---- main optimisation loop -------------------------------------------
    while time.monotonic() < deadline:
        # 60 % of the time we perform a guided mass transfer
        if rng.random() < 0.60:
            # compute current convolution via FFT (only once per loop)
            fa = np.fft.rfft(best, Nfft)
            conv = np.fft.irfft(fa * fa, Nfft)[: 2 * n - 1]
            k = int(conv.argmax())

            # partner[i] = best[k-i] when defined, else 0
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            partner = np.zeros_like(best)
            partner[idx] = best[k - idx]

            src = int(partner.argmax())
            # choose a destination among indices with the smallest partner value
            min_val = float(partner.min())

            if src != dest:

                delta = max(min_step, min(delta, best[src]))
                cand = best.copy()

        # ---- random multiplicative shake -----------------------------------
