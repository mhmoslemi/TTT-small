def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-5.0)

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 256
        a = np.ones(n, dtype=np.float64) / n
    # project to simplex (ensuring sum=1)
    a = a / a.sum()
    best = a.copy()
    best_sc = _score(best)

    n = a.size
    # precompute length for FFT conv of a*a
    L_conv = 1 << ((2 * n - 1).bit_length())
    # length for gradient conv
    L_grad = 1 << ((3 * n - 2).bit_length())

    alpha = 5.0
    alpha_inc = 0.05  # growth per iteration

    # loops
    it = 0
    while time.monotonic() < deadline and it < 200000:
        it += 1

        # compute convolution a*a
        fa = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(fa * fa, n=L_conv)[:2*n-1]
        conv_max = conv.max()
        # approximate softmax
        # if conv_max is large, shift
        exp_terms = np.exp(alpha * (conv - conv_max))
        w = exp_terms / exp_terms.sum()

        # compute gradient
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, n=L_grad)
        fw = np.fft.rfft(w, n=L_grad)
        grad_full = np.fft.irfft(fa_rev * fw, n=L_grad)
        grad = 2.0 * grad_full[n-1:n-1+n]

        # try step sizes
        improved = False
        for step_factor in (0.2, 0.1, 0.05, 0.02, 0.01):
            cand = _project_simplex(a - step_factor * grad)
            sc = _score(cand)
            if sc < best_sc:
                best_sc = sc
                best = cand.copy()
                a = cand
                improved = True
                break
        if improved:
            # maybe increase alpha a little
            alpha = min(alpha + alpha_inc, 100.0)
            continue

        # if not improved, occasional random mass transfer
        if rng.random() < 0.1:
            i = rng.integers(0,n)
            if a[i] > 0:
                j = rng.integers(0,n-1)
                if j >= i: j += 1
                frac = rng.random() * 0.3
                delta = a[i] * frac
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)
                sc = _score(cand)
                if sc < best_sc:
                    best_sc = sc
                    best = cand.copy()
                    a = cand
        # else small random perturbation
        else:
            # small Gaussian noise
            noise = rng.normal(scale=0.001, size=n)
            cand = _project_simplex(a + noise)
            sc = _score(cand)
            if sc < best_sc:
                best_sc = sc
                best = cand.copy()
                a = cand

        # optional: periodically reset alpha to lower to avoid stuck
        if it % 500 == 0:
            alpha = max(alpha * 0.9, 5.0)

    return [float(x) for x in best]

import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast score (identical to evaluate_sequence but without safety checks)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (a.sum() ** 2)

# Optional analytic starter (sqrt series) if no parent is given
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

    n = a.size

    # continuation parameters for the smooth max approximation
    alpha = 5.0            # start smooth

    max_iter = 200_000

        it += 1

        # ------------------- convolution a*a -------------------------
        fa = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(fa * fa, n=L_conv)[: 2 * n - 1]

        # ------------------- soft‑max weights -----------------------
        conv_max = conv.max()
        # stable exponentiation
        w_raw = np.exp(alpha * (conv - conv_max))
        w = w_raw / w_raw.sum()

        # ------------------- gradient of smooth max ----------------
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, n=L_grad)
