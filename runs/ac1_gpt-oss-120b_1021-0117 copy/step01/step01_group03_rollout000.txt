def propose_candidate(...):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 6000
    if 'height_sequence_1' in globals():
        a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    else:
        a = _sqrt_series(n)
    best = a.copy()
    best_score = _score(best)
    best_vec = best.copy()
    # main loop
    while time.monotonic() < deadline:
        # compute conv
        conv = _conv_self(best_vec)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]
        # compute contributions array: for each i, product with counterpart if valid
        # counterpart = max_idx - i
        # Better compute w = best_vec * (best_vec[partner]) where partner in range
        # Use view: create partner array using slicing? Simpler for now:
        w = np.zeros_like(best_vec)
        low = max_idx - (n-1)
        high = max_idx
        # i in [low, high] yields partner j = max_idx - i in [0,n-1]
        i_start = max(0, low)
        i_end = min(n-1, high)
        i_range = np.arange(i_start, i_end+1)
        partner = max_idx - i_range
        w[i_range] = best_vec[i_range] * best_vec[partner]
        # choose src from indices with w > median and value > small eps
        # choose dst from indices with w < median.
        if np.any(w > 0):
            median_w = np.median(w[w > 0])  # median of non-zero contributions
        else:
            median_w = 0
        # sources with high w
        src_candidates = np.where((w >= median_w) & (best_vec > 1e-12))[0]
        # targets with low w (including zeros)
        dst_candidates = np.where(w <= median_w * 0.1)[0]  # low contributions
        if src_candidates.size == 0 or dst_candidates.size == 0:
            # fallback random
            src = rng.integers(0,n)
            dst = rng.integers(0,n)
        else:
            src = rng.choice(src_candidates)
            dst = rng.choice(dst_candidates)
        max_transfer = min(best_vec[src], 1000.0 - best_vec[dst])
        if max_transfer <= 0:
            continue
        # transfer fraction: pick random fraction up to 10% of transfer
        delta = max_transfer * rng.random() * 0.1
        if delta <= 1e-12:
            continue
        # propose new vector copy
        cand = best_vec.copy()
        cand[src] -= delta
        cand[dst] += delta
        # enforce bound may be already within
        # compute score
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best_vec = cand
            best_score = sc
            # possibly record best overall
        # additional: occasional block scaling
        if rng.random() < 0.2:
            # block scaling
            width = int(rng.integers(max(1, n//200), max(2, n//20))) # larger width

            np.clip(cand,0,1000,out=cand)
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_vec = cand
                best_score = sc
        # maybe apply smoothing
        if rng.random() < 0.1:
            # low-pass filter: moving average
            wlen = int(rng.integers(2, n//100))
            if wlen > 1:
                cumsum = np.cumsum(np.concatenate([[0.0], best_vec]))

                smooth = (cumsum[wlen:] - cumsum[:-wlen]) / wlen

                sc = _score(cand)
                if np.isfinite(sc) and sc < best_score:

import numpy as np
import time

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT (fast, O(n log n))."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact AC1 upper‑bound used by the evaluator."""
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i, length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initialise from the provided parent if present, otherwise from the
    # classic sqrt‑series.  Clip to the allowed range.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
        )
        n = init.size
    else:
        n = 5582                      # typical hidden length
        init = _sqrt_series(n)

    best = init.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # Helper: contribution of each index to the current maximal convolution.
    # ------------------------------------------------------------------
    def _contrib(arr: np.ndarray, idx: int) -> np.ndarray:
        """Return a*arr_rev where arr_rev[j] = arr[idx-j] where defined."""
        contrib = np.zeros_like(arr)
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        j = idx - i
        contrib[i] = arr[i] * arr[j]
        return contrib

    # ------------------------------------------------------------------
    # Main search loop.  Two complementary move families:
    #   1. Gradient‑inspired mass transfer based on contributions.
    #   2. Random block scaling / smoothing perturbations.
    # Accept only strictly improving candidates for monotone descent.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --- compute current convolution and locate the maximal index ---
        conv = _conv_self(best)
        k_max = int(conv.argmax())
        # --- contribution vector for the maximal entry --------------------
        contrib = _contrib(best, k_max)

        # ----- 1. Gradient‑inspired transfer ---------------------------------
        # sources: large contribution, non‑zero mass
        src_cand = np.where((contrib > np.median(contrib)) & (best > 1e-12))[0]
        # targets: small contribution, room for mass
        tgt_cand = np.where((contrib < np.median(contrib) * 0.1) &
                            (best < 1000.0))[0]

        if src_cand.size and tgt_cand.size:
            src = int(rng.choice(src_cand))
            dst = int(rng.choice(tgt_cand))

                cand = best.copy()
                cand[src] -= delta

                sc = _score(cand)
