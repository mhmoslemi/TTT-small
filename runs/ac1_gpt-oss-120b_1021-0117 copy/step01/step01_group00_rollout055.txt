def hill_climb(seq, best_score, deadline):
    current = seq.copy()
    cur_score = best_score
    n = seq.size
    while time.monotonic() < deadline:
        # propose move
        move_type = rng.random()
        if move_type < 0.7:  # swap
            i, j = rng.integers(0, n, size=2)

                continue

            cand[i], cand[j] = cand[j], cand[i]

```
import numpy as np, time, math

def compute_score(seq):
    seq = np.asarray(seq, dtype=np.float64)
    # clip
    np.clip(seq, 0.0, 1000.0, out=seq)
    S = np.sum(seq)
    if S < 1e-12:
        return math.inf
    n = seq.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    # use real FFT
    fft_seq = np.fft.rfft(seq, n=L)
    prod = fft_seq * fft_seq
    conv = np.fft.irfft(prod, n=L)
    conv = conv[:2*n-1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (S * S)
```

def generate_sidon_like(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1,n):
        a[i] = a[i-1] * (1.0 - 1.0/(2.0*i))
    return a

import numpy as np
import time
import math

# ------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility to get the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def compute_score(seq: np.ndarray) -> float:
    """
    Fast proxy for evaluate_sequence:
    returns 2*n*max(conv(seq,seq))/(sum(seq)**2)
    uses an FFT‑based convolution, clips to [0,1000].
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    np.clip(a, 0.0, 1000.0, out=a)
    S = float(a.sum())
    if S <= 0.0:
        return math.inf
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    # real FFT of the zero‑padded sequence
    f = np.fft.rfft(a, n=L)
    prod = f * f
    conv = np.fft.irfft(prod, n=L)
    conv = conv[: 2 * n - 1]          # keep the exact length
    max_b = float(conv.max())
    return 2.0 * n * max_b / (S * S)


def _sidon_like_sequence(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4**i (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _exponential_sequence(n: int, c: float) -> np.ndarray:
    """a_i = exp(-c*i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-c * idx)


def _hybrid_sequence(n: int, p: float, c: float) -> np.ndarray:
    """a_i = (i+1)^{-p} * exp(-c*i)."""
    idx = np.arange(1, n + 1, dtype=np.float64) - 1.0
    return (idx + 1.0) ** (-p) * np.exp(-c * idx)


def _spread_sequence(base: np.ndarray, step: int) -> np.ndarray:
    """
    Permutes `base` by a modular step (coprime to len(base) yields a full cycle).
    This tends to spread large values evenly across the index line.
    """
    n = base.shape[0]
    idx = (np.arange(n, dtype=np.int64) * step) % n
    out = np.empty_like(base)
    out[idx] = base
    return out
