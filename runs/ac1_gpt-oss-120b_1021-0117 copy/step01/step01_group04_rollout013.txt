def max_contrib(a, K):
    n = a.shape[0]
    max_vals = np.zeros_like(a)
    idx = np.arange(n)
    for k in K:
        src_idx = k - idx
        # valid where 0 <= src_idx < n
        mask = (src_idx >= 0) & (src_idx < n)
        # For valid entries, get a[src_idx]
        contrib = np.zeros_like(a)
        contrib[mask] = a[src_idx[mask]]
        max_vals = np.maximum(max_vals, contrib)
    return max_vals

import numpy as np
import time
import math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _conv_self(arr: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution via FFT, returned length 2·len(arr)-1."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    n = arr.shape[0]
    return conv[: 2 * n - 1]

def _max_contrib(arr: np.ndarray, idx_max: np.ndarray) -> np.ndarray:
    """For each position i, max_{k∈idx_max} a_{k-i} (zero outside)."""
    n = arr.shape[0]
    idx = np.arange(n)
    out = np.zeros_like(arr)
    for k in idx_max:
        src = k - idx
        mask = (src >= 0) & (src < n)
        if not mask.any():
            continue
        cand = arr[src[mask]]
        out[mask] = np.maximum(out[mask], cand)
    return out
