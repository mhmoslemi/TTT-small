def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)  # keep margin

    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256

    # Prepare initial sequence
    # Try binomial if available else generate.
    if 'height_sequence_1' in globals():
        # Use provided but re-normalize to sum 1
        a = np.array(height_sequence_1, dtype=np.float64)
    else:
        # generate binomial
        def gen_binom(n):
            a = np.empty(n, dtype=np.float64)
            a[0]=1.0
            for i in range(n-1):
                a[i+1]=a[i]*(2*i+1)/(2*i+2)
            return a
        a = gen_binom(n)

    # Clip and scale
    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total == 0:
        a = np.ones(n)/n
        total = 1.0
    a = a / total   # sum to 1

    # FFT config
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # Evaluate initial
    cur = a.copy()
    cur_score = _score_fft(cur, n, fft_len)
    best = cur.copy()
    best_score = cur_score

    # Simulated Annealing parameters
    T0 = 0.01 * cur_score   # initial temperature 1% of score?
    Tmin = 1e-6
    decay = 0.9995   # per iteration approx
    T = T0

    iteration = 0
    # loop
    while time.monotonic() < deadline:
        iteration += 1
        # Choose source and target indices
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # Ensure source > small threshold
        if cur[i] <= 1e-14:
            continue
        # Choose fraction to transfer; random between small and large
        frac = rng.random() * 0.5 + 0.01   # between 0.01 to 0.51
        delta = cur[i] * frac
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        new_score = _score_fft(new, n, fft_len)

        # Acceptance
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
        else:
            # probabilistic acceptance
            # compute acceptance probability
            if T > 0:
                prob = math.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    cur = new
                    cur_score = new_score
        # temperature cooling
        T = max(Tmin, T * decay)

    # Return best as list
    final = [float(x) for x in best]
    # ensure evaluate
    try:
        _ = evaluate_sequence(final)
    except Exception:
        # fallback
        final = [float(x) for x in a]
    return final

import time
import math
import numpy as np

def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4**i via recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def _conv_stats_fft(a: np.ndarray, fft_len: int):
    """Return (max convolution value, index where it occurs) using a single FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _score_fft(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2*n*max(conv)/sum(a)^2 (sum(a) will be 1 in our usage)."""
    max_b, _ = _conv_stats_fft(a, fft_len)
    return 2.0 * n * max_b / (a.sum() ** 2)

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------

    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total == 0.0:

    else:
        a = a / total                     # now sum == 1

    # --------------------------------------------------------------

    conv_len = 2 * n - 1

    cur = a.copy()

    best = cur.copy()

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    # temperature for occasional uphill steps
    T = cur_score * 0.02
    Tmin = 1e-6
    decay = 0.9995

    while time.monotonic() < deadline:

        max_raw, max_idx = _conv_stats_fft(cur, fft_len)

        # b) gradient w.r.t. the active lag (product partners)

        hi = min(n - 1, max_idx)

        # c) choose a source where moving mass reduces the active term most

        src_score = cur * g
        src_idx = int(np.argmax(src_score))
