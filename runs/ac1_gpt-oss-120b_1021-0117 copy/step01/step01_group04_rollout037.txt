def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent_arr.shape[0])
    except Exception:
        n = 4806
        parent_arr = None

    L = 1 << ((2 * n - 2).bit_length())  # next power of two

    def _conv(arr):
        fa = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fa * fa, n=L)
        return conv[: 2 * n - 1]

    def _max_conv(arr):
        conv = _conv(arr)
        return float(conv.max())

    def _score_arr(arr):
        # arr assumed sum 1
        return 2.0 * n * _max_conv(arr)

    # baseline: binomial
    base = np.empty(n, dtype=np.float64)
    base[0] = 1.0
    for i in range(1, n):
        base[i] = base[i-1] * (2*i - 1) / (2*i)
    # normalize
    base = base / base.sum()
    best = base.copy()
    best_score = _score_arr(best)

    if parent_arr is not None:
        # normalize parent to sum 1
        parent_norm = parent_arr.copy()
        total = parent_norm.sum()
        if total > 0:
            parent_norm = parent_norm / total
            parent_score = _score_arr(parent_norm)
            if parent_score < best_score:
                best = parent_norm.copy()
                best_score = parent_score

    cur = best.copy()
    cur_score = best_score
    eta = 0.5   # initial step size for exponentiated gradient
    decay = 0.9999
    stagnation = 0
    # main loop
    while time.monotonic() < deadline:
        # compute convolution and max index for current a
        conv_cur = _conv(cur)
        max_val_cur = float(conv_cur.max())
        # update best
        if max_val_cur * 2 * n < best_score - 1e-12:
            best = cur.copy()
            best_score = max_val_cur * 2 * n
        # find argmax index
        k = int(conv_cur.argmax())
        # compute gradient (subgradient) for index k
        g = np.zeros_like(cur)
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        idx = np.arange(i0, i1 + 1, dtype=np.int64)
        rev = k - idx
        g[idx] = 2.0 * cur[rev]
        # candidate via exponentiated gradient
        cand = cur * np.exp(-eta * g)
        # renormalize to sum 1 (simple projection onto simplex by scaling)
        s = cand.sum()
        if s <= 0:
            # fallback to cur if underflow
            cand = cur.copy()
            s = cand.sum()
        cand = cand / s
        cand_score = 2.0 * n * _max_conv(cand)
        # accept improvement
        if cand_score < cur_score - 1e-12:
            cur = cand
            cur_score = cand_score
            stagnation = 0
        else:
            stagnation += 1
            # occasionally add noise to escape stagnation
            if stagnation >= 200:
                # random multiplicative perturbation
                perturb = np.exp(eta * (rng.random(n) - 0.5))
                cand2 = cur * perturb
                cand2 = cand2 / cand2.sum()
                cand2_score = 2.0 * n * _max_conv(cand2)
                if cand2_score < cur_score:
                    cur = cand2
                    cur_score = cand2_score
                    stagnation = 0
                else:
                    # shrink step size
                    eta *= 0.5
                    stagnation = 0
        # global best tracking
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            # early exit if target achieved
            if best_score < 1.50274349:
                break
        # decay step size slowly
        eta *= decay
        # ensure eta not too small
        if eta < 1e-6:
            eta = 1e-6

    # Scale best to sum sqrt(2n) for nicer values
    target_sum = math.sqrt(2.0 * n)
    scale = target_sum / best.sum()
    best_scaled = best * scale
    # clip to evaluator bounds (not needed but safe)
    np.clip(best_scaled, 0.0, 1000.0, out=best_scaled)
    return [float(x) for x in best_scaled]

import numpy as np
import time
import math

# ---------- helpers ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n-1 (classic binomial step‑function)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _conv_fft(arr: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution arr*arr via FFT, padded to length L."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * arr.shape[0] - 1]

def _max_conv(arr: np.ndarray, L: int) -> float:
    """Maximum entry of the self‑convolution of arr."""
    return float(_conv_fft(arr, L).max())

def _score(arr: np.ndarray, L: int) -> float:
    """Exact AC1 upper bound for a sequence with sum == 1."""
    n = arr.shape[0]
    return 2.0 * n * _max_conv(arr, L)

# ---------- main search -----------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Stochastic mirror‑descent on the simplex aiming to minimise
    the AC1 bound.  Returns a list of coefficients (non‑negative,
    ≤1000) for the given length (taken from height_sequence_1 if
    present, otherwise 4806).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ----- obtain length & optional parent ---------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        n = 4806
        parent = None

    L = _next_pow2(2 * n - 1)          # FFT size (next power of two)

    # ----- initialise with binomial and possibly the parent ----------------
    cur = _gen_binomial(n)
    cur = cur / cur.sum()             # normalise to sum == 1
    best = cur.copy()
    best_score = _score(best, L)

    if parent is not None:
        p = parent.astype(np.float64)
        if p.sum() > 0:
            p = p / p.sum()
            p_score = _score(p, L)
            if p_score < best_score:
                best = p.copy()
                best_score = p_score
                cur = p.copy()

    # ----- parameters for the mirror‑descent -----------------------------
    step = 2.0                         # initial multiplicative step size
    decay = 0.9999                     # slow decay of step
    stagnation = 0                     # counts iterations without improvement

    # ----- main optimisation loop ----------------------------------------
    while time.monotonic() < deadline:
        # current convolution & index attaining the maximum
        conv_cur = _conv_fft(cur, L)
        max_val = float(conv_cur.max())
        k = int(conv_cur.argmax())      # one active index

        # keep best ever seen
        if max_val * 2 * n < best_score - 1e-12:
            best = cur.copy()
            best_score = max_val * 2 * n
            if best_score < 1.50274349:      # early exit condition
                break

        # sub‑gradient of the active linear piece (2*a_{k‑i})
        g = np.zeros_like(cur)
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        if i0 <= i1:
            idx = np.arange(i0, i1 + 1, dtype=np.int64)
            rev = k - idx
            g[idx] = 2.0 * cur[rev]

        # mirror‑descent update (exponentiated gradient)
        gmax = g.max()
        if gmax > 0.0:
            factor = np.exp(-step * (g / gmax))
            cand = cur * factor
        else:
            cand = cur.copy()
        cand = cand / cand.sum()                 # projection onto simplex

        cand_score = _score(cand, L)

        # accept if we improve the current point
        if cand_score < best_score - 1e-12:
            cur = cand
            stagnation = 0
        else:
            stagnation += 1
            # occasional random perturbation to escape flat regions
            if stagnation >= 300:
                perturb = np.exp(step * 0.3 * (rng.random(n) - 0.5))
                cand2 = cur * perturb
                cand2 = cand2 / cand2.sum()
                cand2_score = _score(cand2, L)
                if cand2_score < best_score:
                    cur = cand2
                    best = cand2.copy()
                    best_score = cand2_score
                    stagnation = 0
                else:
                    # shrink step size when progress stalls
                    step *= 0.5
                    stagnation = 0

        # global best tracking (might improve without accepting)
        if cand_score < best_score - 1e-12:
            best = cand.copy()
            best_score = cand_score

        # step‑size schedule
        step *= decay
        if step < 1e-5:
            step = 1e-5

    # ----- final scaling (optional) ---------------------------------------
    # The AC1 bound is scaling‑invariant; we match the conventional
    # normalisation Σ a_i = sqrt(2 n) for nicer numbers.
    target_sum = math.sqrt(2.0 * n)
    scale = target_sum / best.sum()
    result = best * scale
    np.clip(result, 0.0, 1000.0, out=result)

    return [float(v) for v in result]
