import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) aiming at a low
    AC1 upper bound.  The implementation uses the provided parent
    `height_sequence_1` (if available) as a starting point and then performs
    a very small amount of random mass‑transfer search while respecting the
    evaluator’s constraints.
    """
    # ----- basic setup ----------------------------------------------------
    # deadline a little before the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # obtain length and a base vector (parent if present, else uniform)
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:                     # no parent supplied
        n = 5000
        base = np.full(n, 1.0 / n, dtype=np.float64)

    # enforce the evaluator’s clipping limits
    seq = np.clip(base, 0.0, 1000.0)

    # guarantee a non‑trivial sum (the parent already satisfies this)
    if seq.sum() < 0.01:
        seq = np.full(n, 1.0 / n, dtype=np.float64)

    # current best candidate
    best = seq.copy()
    best_score = evaluate_sequence([float(v) for v in best])

    # ----- tiny random search (few iterations, cheap) --------------------
    rng = np.random.default_rng(seed)
    max_iters = 25                     # enough to try a few moves
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # pick a source index with non‑zero mass
        src = int(rng.integers(0, n))
        if best[src] <= 0.0:
            continue

        # pick a destination (could coincide, we skip that case)
        dest = int(rng.integers(0, n))
        if dest == src:
            continue

        # transfer a small fraction of the source mass
        delta = best[src] * 0.01 * rng.random()
        if delta <= 0.0:
            continue

        cand = best.copy()
        cand[src] -= delta
        cand[dest] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        # renormalise to keep the total sum exactly the same (avoids drift)
        s = cand.sum()
        if s <= 0.0:
            continue
        cand = cand * (best.sum() / s)

        # evaluate the candidate
        cand_score = evaluate_sequence([float(v) for v in cand])

        # keep it if it improves the bound
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            # early stop if we already beat the advertised target
            if best_score < 1.50274349:
                break

    # ----- final sanity clip (the evaluator repeats this anyway) ----------
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
