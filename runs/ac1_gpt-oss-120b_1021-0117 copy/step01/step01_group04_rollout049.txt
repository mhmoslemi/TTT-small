def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import required modules.
    compute deadline = time.monotonic() + max(0., budget_s - 2.)
    rng = numpy.random.default_rng(seed)

    # Determine n, base sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    except Exception:
        # fallback generate binomial of length maybe 2048
        n = 2048
        parent = None

    # Precompute FFT length L as next power of two >= 2*n -1
    L = _next_pow2(2*n -1)

    # initial candidate
    if parent is not None:
        cur = parent.copy()
        # ensure within [0,1000] (maybe already)
        np.clip(cur, 0.0, 1000.0, out=cur)
    else:
        cur = _gen_binomial(n)
    # compute scores
    cur_score = _ac1_score(cur, n, L)
    best = cur.copy()
    best_score = cur_score

    # set parameters
    max_temp = 5.0
    temperature = max_temp
    decay = 0.9997
    base_sigma = 0.4  # base std of log-factor

    # loop
    while time.monotonic() < deadline:
        # decide move type probabilities
        move_type = rng.random()
        cand = None
        if move_type < 0.5: # block scaling
            width = int(rng.integers(1, max(2, n//6)))  # block width up to n/6 (maybe)
            start = int(rng.integers(0, n - width + 1))
            sigma = base_sigma * (temperature / max_temp)  # shrink sigma
            factor = math.exp(rng.normal(0.0, sigma))
            cand = cur.copy()
            cand[start:start+width] *= factor
        elif move_type < 0.8: # mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * 0.5 # up to 50%
                delta = cur[i] * frac
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta

            width = int(rng.integers(2, max(3, n//12)))
            start = int(rng.integers(0, n - width + 1))
            # compute avg of block and optionally factor
            block = cur[start:start+width]
            avg = block.mean()

            sigma = base_sigma * 0.5 * (temperature / max_temp)
            factor = 1.0 + rng.normal(0.0, sigma) * 0.1  # small perturbation

import numpy as np, time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound using FFT (linear convolution)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)

    except Exception:

    cur_score = _ac1_score(cur, n, L)
    best = cur.copy()
