import numpy as np
import time

def _initial_binomial(n: int) -> np.ndarray:
    """Monotone decreasing binomial seed a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()                     # normalise to sum‑1 (scale‑invariant)
    return a


def _power_law_seq(n: int, alpha: float) -> np.ndarray:
    """a_i = (i+1)^(-alpha) , normalised to sum‑1."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = idx ** (-alpha)
    a /= a.sum()
    return a


def _fast_score(seq: np.ndarray):
    """Return (score, full convolution, max conv) using FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    mx = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, mx
    return float(2.0 * n * mx / (s * s)), conv, mx


def _targeted_move(seq: np.ndarray, best_conv: np.ndarray, rng: np.random.Generator):
    """Move a small fraction of the mass from the pair that most contributes to the maximal convolution."""
    n = seq.shape[0]
    k = int(np.argmax(best_conv))
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    i_arr = np.arange(lo, hi + 1, dtype=np.int64)
    j_arr = k - i_arr
    prod = seq[i_arr] * seq[j_arr]
    i_max = i_arr[np.argmax(prod)]
    j_max = k - i_max

    alpha = rng.uniform(0.02, 0.12)          # fraction of mass to lift
    if i_max == j_max:
        # diagonal term a_i^2
        removed = seq[i_max] * alpha
        seq[i_max] -= removed
        dst = np.argmin(seq)
        if dst == i_max:
            dst = rng.integers(0, n)
        seq[dst] += removed
    else:
        rem_i = seq[i_max] * alpha
        rem_j = seq[j_max] * alpha
        seq[i_max] -= rem_i
        seq[j_max] -= rem_j
        total = rem_i + rem_j
        dst = np.argmin(seq)
        if dst == i_max or dst == j_max:
            dst = rng.integers(0, n)
        seq[dst] += total
    return seq
