    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv

def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (fast via FFT)."""
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

def _sqrt_series(n: int) -> np.ndarray:
    """Classic Catalan‑type sequence a_i = C(2i,i)/4^i (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a low‑score non‑negative step function under the given budget."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 4.0)   # a few seconds margin

    # ------------------------------------------------------------------
    # Initialise – use supplied parent if present, otherwise the sqrt series.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = a.shape[0]
    else:
        n = 5582
        a = _sqrt_series(n)

    # also have a pure sqrt series as a fallback candidate
    s_series = _sqrt_series(n)

    # pick the better of the two as the start
    best = a.copy()
    best_score = _score(best)
    s_score = _score(s_series)
    if s_score < best_score:
        best = s_series.copy()
        best_score = s_score

    # current working solution
    cur = best.copy()
    cur_score = best_score

    total_time = deadline - start

    # ------------------------------------------------------------------
    # Local search: targeted moves, swaps, block scalings, and noise.
    # ------------------------------------------------------------------
    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        remaining = deadline - now
        frac = remaining / total_time  # 1 → 0 as time runs out

        op = rng.random()
        # ------------------------------------------------------------------
        # 1. Targeted move – reduce the biggest contribution to the current peak.
        # ------------------------------------------------------------------
        if op < 0.55:
            conv = _conv_self(cur)
            k = int(conv.argmax())
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_idx = k - i_idx
            contrib = cur[i_idx] * cur[j_idx]
            maxpos = int(contrib.argmax())
            i_src = int(i_idx[maxpos])
            j_src = int(j_idx[maxpos])
            src = i_src if cur[i_src] >= cur[j_src] else j_src
            dst = rng.integers(0, n)
            while dst == src:
                dst = rng.integers(0, n)
            max_transfer = min(cur[src], 1000.0 - cur[dst])
            if max_transfer <= 0:
                continue
            delta = max_transfer * frac * rng.random() * 0.08   # up‑to 8 % early on
            cand = cur.copy()
            cand[src] -= delta
            cand[dst] += delta

        # ------------------------------------------------------------------
        # 2. Random swap of two positions (re‑ordering can lower autocorrelation).
        # ------------------------------------------------------------------
        elif op < 0.70:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # ------------------------------------------------------------------
        # 3. Block scaling – multiplicative perturbation of a contiguous segment.
        # ------------------------------------------------------------------
        elif op < 0.85:
            w_low = max(1, n // 200)
            w_high = max(2, n // 12)
            width = int(rng.integers(w_low, w_high))
            start_idx = rng.integers(0, max(1, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand = cur.copy()
            cand[start_idx:start_idx + width] *= factor

        # ------------------------------------------------------------------
        # 4. Tiny additive Gaussian noise.
        # ------------------------------------------------------------------
        else:
            noise = rng.normal(0.0, 0.02, size=n)
            cand = cur + noise

        # enforce the hard bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate the candidate
        cand_score = _score(cand)
        if not np.isfinite(cand_score):
            continue

        # keep the global best ever seen
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        # accept for the running state (always accept improvements,
        # occasionally accept small uphill moves)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
        else:
            temp = 0.001 * frac   # tiny temperature that vanishes near the deadline
            if temp > 0.0 and rng.random() < math.exp((cur_score - cand_score) / temp):
                cur = cand
                cur_score = cand_score

    # ------------------------------------------------------------------
    # Return the best solution discovered as a plain Python list.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
