    # Helper conv & score functions fast
    def conv_self(a):
        n_local = a.shape[0]
        m = 1 << ((2 * n_local - 1).bit_length())
        fa = np.fft.rfft(a, n=m)
        prod = fa * fa
        conv = np.fft.irfft(prod, n=m)[: 2 * n_local - 1]
        return conv

    def score(a):
        n_local = a.shape[0]
        b = conv_self(a)
        maxb = b.max()
        s = a.sum()
        if s < 0.01:
            return np.inf
        return 2.0 * n_local * maxb / (s * s)

    best = None
    best_score = np.inf

    # Try initial families
    # 1. Parent if exists
    if init_parent is not None:
        sc = score(init_parent)
        best = init_parent.copy()
        best_score = sc

    # 2. Power law candidates
    for p in np.linspace(0.3, 1.2, 10):
        idx = np.arange(n, dtype=np.float64)
        a = 1.0 / np.power(idx + 1.0, p)
        # Ensure sum > 0.01
        # evaluate score
        sc = score(a)
        if sc < best_score:
            best = a.copy()
            best_score = sc

    # 3. Symmetric sqrt shape
    idx = np.arange(n, dtype=np.float64)

    sc = score(a)
    if sc < best_score:
        best = a.copy()
        best_score = sc

    # 4. Gaussian shapes with various sigma

        a = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
        sc = score(a)
        if sc < best_score:
            best = a.copy()

    # Ensure best exists
    if best is None:
        best = np.ones(n, dtype=np.float64)
        best_score = score(best)

    # Local search loop
    # Determine region parameters
    while time.monotonic() < deadline:
        # compute current conv and max index
        b = conv_self(best)
        max_idx = int(np.argmax(b))  # int for index
        # Determine contributing indices range: i from max(0, max_idx-(n-1)) to min(n-1, max_idx)
        i_min = max(0, max_idx - (n - 1))
        i_max = min(n - 1, max_idx)

        i = rng.integers(i_min, i_max + 1)

        # Choose source = whichever larger value
        src = i if best[i] >= best[j] else j
        # Choose destination far from src: sample uniformly from [0,n) until far enough
        # Ensure dest is at distance > n//4 maybe

        attempts = 0

            dest = rng.integers(0, n)
            attempts += 1

        # Transfer delta from src to dest

                cand = best.copy()
                cand[src] -= delta
                cand[dest] += delta
                # Clip (should already be ensured)
                np.clip(cand, 0.0, 1000.0, out=cand)
                sc = score(cand)
                if sc < best_score:
                    best = cand
                    best_score = sc
                    continue  # move to next iteration

        # Fallback random block scaling operation
        width = rng.integers(max(1, n // 200), max(2, n // 12))
        start = rng.integers(0, max(1, n - width + 1))
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand = best.copy()
        cand[start:start+width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = score(cand)
        if sc < best_score:

            best_score = sc
            continue

        # occasional additive noise
        if rng.random() < 0.1:
            noise = rng.normal(0.0, 0.01, size=n)
            cand = best + noise
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = score(cand)
            if sc < best_score:
                best = cand
                best_score = sc

    # Return best as list
    return [float(v) for v in best]

import numpy as np
import time

def _conv_self(a):
    n = a.size
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    prod = fa*fa
    conv = np.fft.irfft(prod, n=m)[:2*n-1]
    return conv

def _score(a):
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s*s)

def _gradient(a, k):

    i_max = min(n - 1, k)

        i = np.arange(i_min, i_max + 1, dtype=np.int64)
        partner = k - i
        # add partner values
        # Using advanced indexing adds vector to selected positions

import numpy as np
import time

# ---------- convolution and score ---------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution via real FFT (O(n log n))."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the official evaluator."""
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _gradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution value at index k:
    g_i = a_i + a_{k-i}  (missing partners are treated as 0).
    """
    n = a.shape[0]
    g = a.copy()
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    if i_min <= i_max:
        i = np.arange(i_min, i_max + 1, dtype=np.int64)
        partner = k - i
        g[i] += a[partner]
    return g


# ---------- main search -------------------------------------------
