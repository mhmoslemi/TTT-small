def _score(vec):
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
    return 2.0 * n * conv.max() / (vec.sum() ** 2)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    # Determine length n
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: sqrt series of length 256? Let's use 256 as guess maybe.
        n = 256
        base = _sqrt_series(n).astype(np.float64)
    # Normalize to sum=1 (nonnegative)
    base = np.maximum(base, 0.0)
    if base.sum() == 0:
        base = np.ones_like(base) / base.size
    else:
        base = base / base.sum()
    # Ensure simplex projection:
    # (the vector already sums to 1 and nonneg)
    cur = base.copy()
    best = cur.copy()
    best_sc = _score(best)
    n = cur.size
    # Precompute some constants
    max_transfer = 0.25
    it = 0
    # main loop
    while time.monotonic() < deadline:
        it += 1
        # random choice
        if rng.random() < 0.9:  # random mass transfer
            # pick source weighted by current amplitude
            # Ensure source with some mass
            src = rng.choice(n, p=cur)  # probability proportional to a_i
            if cur[src] <= 0:
                continue
            tgt = rng.integers(0, n-1)
            if tgt >= src:
                tgt += 1
            frac = rng.random() * max_transfer
            delta = cur[src] * frac
            cand = cur.copy()
            cand[src] -= delta
            cand[tgt] += delta
            # projection not needed since sum conserved and nonneg (if delta <= cur[src]), but rounding may make tiny negative; enforce nonneg and reproject

            cand = _project_simplex(cand)
            sc = _score(cand)
            if sc < best_sc:
                best_sc = sc
                best = cand
                cur = cand  # move current to new best
            # optionally also accept if sc < cur_sc? but cur_sc = best_sc if cur==best else store cur_sc separately
        else:
            # gradient step
            conv = _conv_full(cur)
            M = conv.max()
            # find indices where conv close to M
            # tolerance relative to M
            tol = 1e-12
            idxs = np.where(conv >= M - tol)[0]
            if idxs.size == 0:
                continue
            grad = np.zeros_like(cur)
            for k in idxs:
                lo = max(0, k - (n - 1))
                hi = min(k, n - 1)
                # For i = lo..hi, contribution = 2 * cur[k - i]
                # vectorized:
                i_range = np.arange(lo, hi+1)
                contrib = 2.0 * cur[k - i_range]
                grad[lo:hi+1] += contrib

            # Choose eta = lr * (cur.mean() / (grad.mean()+1e-12))

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 score (identical to evaluate_sequence)
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (vec.sum() ** 2)


# ------------------------------------------------------------
# Full convolution via FFT (used for gradient)
# ------------------------------------------------------------
def _conv_full(vec: np.ndarray) -> np.ndarray:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Analytic sqrt‑series (fallback initializer)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # initialise a feasible vector (sum = 1, non‑negative)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            a[:] = 1.0
        a /= a.sum()
    else:
        n = 256
        a = _sqrt_series(n).astype(np.float64)
        a = np.maximum(a, 0.0)
        a /= a.sum()

    cur = a.copy()
    best = cur.copy()
    best_sc = _score(best)
    n = cur.size

    # probabilities for the three move types
    prob_early = 0.55   # move mass from early to late indices
    prob_grad  = 0.35   # gradient‑like step based on active convolution peaks
    # remaining 0.10 will be a generic random transfer

    max_transfer = 0.30          # up to 30 % of the source mass in a transfer
    it = 0
    last_improve_iter = 0

    while time.monotonic() < deadline:
        it += 1
        r = rng.random()

        # --------------------------------------------------------------
        # 1) Early→Late mass transfer (biased exploration)
        # --------------------------------------------------------------
        if r < prob_early:
            src = rng.integers(0, max(1, n // 3))
            if cur[src] <= 0.0:
                continue
            tgt = rng.integers(2 * n // 3, n)
            frac = rng.random() * max_transfer
            delta = cur[src] * frac
            cand = cur.copy()
            cand[src] -= delta
            cand[tgt] += delta
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _score(cand)
            if sc < best_sc:
                best_sc, best, cur = sc, cand, cand
                last_improve_iter = it

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        elif r < prob_early + prob_grad:
            conv = _conv_full(cur)
            M = conv.max()
            # indices where conv attains (almost) the maximum

            cur = pert
            last_improve_iter = it

    # ------------------------------------------------------------------
    # Return the best vector discovered (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
