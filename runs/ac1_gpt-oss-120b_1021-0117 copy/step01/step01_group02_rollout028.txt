    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-0.5)
    # n
    try:

        a = np.array(height_sequence_1, dtype=np.float64).copy()
    except Exception:

        # start with power law shape
        i = np.arange(n)

    # Clip and normalize sum to 1

        s = 1.0
    a = a / s

    # Score function using FFT
    def score(vec):

        conv = np.fft.irfft(fft*fft, n=m)[:2*n-1]
        maxc = conv.max()
        return 2.0 * n * maxc / (vec.sum()**2)

    best = a.copy()

        best_score = uniform_score

    # Parameters
    step_factor = 0.4

def project_onto_simplex(v):
    # Return projection onto {x: x >=0, sum x = 1}
    # Using algorithm by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

import time, numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    try:
        n = len(height_sequence_1)
        a = np.array(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        n = 5000
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i-1] * (1 - 1.0/(2.0*i))
    # clip and normalize sum to 1
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0/n)
    else:
        a = a / s

    # fallback uniform
    uniform = np.full(n, 1.0/n)
    # define scoring using FFT
    m = 1 << ((2*n-1).bit_length())
    def score(vec):
        fft = np.fft.rfft(vec, n=m)
        conv = np.fft.irfft(fft*fft, n=m)[:2*n-1]
        maxc = conv.max()
        return 2.0 * n * maxc  # sum is 1 => denominator 1

    best = a.copy()
    best_score = score(best)
    uniform_score = score(uniform)
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # parameters
    eta = 0.5
    decay = 0.9995
    min_eta = 1e-6
    no_improve = 0
    max_no_improve = 1000

    while time.monotonic() < deadline:
        # compute conv and max index
        fft_a = np.fft.rfft(best, n=m)
        conv = np.fft.irfft(fft_a*fft_a, n=m)[:2*n-1]
        maxc = conv.max()
        k = int(conv.argmax())
        # compute partner gradient
        p = np.zeros_like(best)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = slice(i_start, i_end + 1)
        # indices for partner: a[k - i]
        p[idx] = best[k - np.arange(i_start, i_end + 1)]
        # additive projected gradient
        cand = best - eta * p
        # projection onto simplex (sum = 1, nonneg)
        # Duchi algorithm
        u = np.sort(cand)[::-1]
        cssv = np.cumsum(u) - 1
        rho_arr = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
        if rho_arr.size == 0:
            # all zero or negative, fall back to uniform
            cand = uniform.copy()
        else:
            rho = rho_arr[-1]
            theta = cssv[rho] / (rho + 1)
            cand = np.maximum(cand - theta, 0.0)
        cand_score = score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            no_improve = 0
        else:
            # try random local perturbation
            if rng.random() < 0.01:
                width = int(rng.integers(1, max(2, n // 100)))
                idxs = rng.choice(n, size=width, replace=False)
                factor = np.exp(rng.normal(0.0, 0.2, size=width))
                pert = best.copy()
                pert[idxs] *= factor
                pert = np.clip(pert, 0.0, 1000.0)
                pert = pert / pert.sum()
                pert_score = score(pert)
                if pert_score < best_score:
                    best = pert
                    best_score = pert_score
                    no_improve = 0
        # decay eta
        eta = max(min_eta, eta * decay)
        # optionally restart if no improvement long time
        no_improve += 1
        if no_improve > max_no_improve:
            # random restart with Dirichlet
            r = rng.random(n)
            cand = r / r.sum()
            cand_score = score(cand)
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                no_improve = 0
            else:
                no_improve = 0  # reset
    final = best
    final = np.clip(final, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    # optionally verify final score with evaluator
    # but we just return list
    return [float(v) for v in final]

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / v.size)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
