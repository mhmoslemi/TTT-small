import time
import math
import numpy as np

def _generate_binomial_seq(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, decreasing positive sequence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a

def _normalize(a: np.ndarray) -> np.ndarray:
    """Scale to unit sum, guard against zero total."""
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / a.size
    else:
        a /= s
    return a

def _score_fft(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Fast evaluation of the evaluator’s objective for a unit‑sum vector:
    2·n·max(conv(a,a)).
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
