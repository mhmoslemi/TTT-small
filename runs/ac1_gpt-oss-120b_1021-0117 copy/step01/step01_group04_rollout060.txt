def gradient_move(a, step):

    # compute gradient vector (2 * a_rev) on overlap region
    grad = np.zeros_like(a)

    # renormalize
    total = cand.sum()
    if total == 0:
        # revert to original
        return a.copy()
    cand = cand / total
    return cand

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x (x>0)."""
    return 1 << (x - 1).bit_length()

    """C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)

    s = v.sum()
    if s <= 0:
        return v
    return v / s

        return float('inf')

    tri = _normalize(tri)
