import numpy as np, time

def _sidon_like_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (1.0 - 1.0/(2.0*i))
    return a

def _build_sidon_positions(n, m):
    chosen = []
    sums = set()
    for pos in range(n):
        ok = True
        for q in chosen:
            if pos + q in sums:
                ok = False
                break
        if ok and (pos + pos) not in sums:
            # add position
            for q in chosen:
                sums.add(pos + q)
                sums.add(q + pos) # same
            sums.add(pos + pos)
            chosen.append(pos)
            if len(chosen) >= m:
                break
    return chosen

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = int(parent.shape[0])
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        n = 256
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
    # generate base decreasing sequence
    base = _sidon_like_sequence(n)
    base = np.clip(base, 0.0, 1000.0)
    # evaluate base
    base_score = float(evaluate_sequence(base.tolist()))
    if np.isfinite(base_score) and base_score < best_score:
        best_seq = base.copy()
        best_score = base_score
    # try Sidon position arrangement
    m = int(np.sqrt(n))  # target size
    pos = _build_sidon_positions(n, m)
    if len(pos) >= 2:
        # assign largest values to those positions
        sorted_vals = np.sort(base)[::-1]  # descending
        arr = np.empty_like(base)
        arr.fill(0.0)
        # place top len(pos) values
        top_vals = sorted_vals[:len(pos)]
        for idx, p in enumerate(pos):
            arr[p] = top_vals[idx]
        # fill remaining positions with remaining values, in original order maybe
        rest_vals = sorted_vals[len(pos):]
        rem_idx = np.where(arr == 0.0)[0]
        arr[rem_idx] = rest_vals[:len(rem_idx)]
        cand_score = float(evaluate_sequence(arr.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best_seq = arr.copy()
            best_score = cand_score
            current = arr.copy()
        else:
            current = best_seq.copy()
    else:
        current = best_seq.copy()
    # hill climbing loop
    while time.monotonic() < deadline:
        # propose random swap
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cand = current.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score):
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                current = cand
            elif cand_score < float(evaluate_sequence(current.tolist())):
                current = cand
        # occasional random restart with new random permutation of base
        if rng.random() < 0.0005:
            perm = rng.permutation(n)
            current = base[perm]
    # final fallback ensure sum >=0.01
    if best_seq.sum() < 0.01:
        best_seq = np.full_like(best_seq, 0.01 / n)
    return [float(x) for x in best_seq]

import numpy as np, time

def _sidon_like_sequence(n: int) -> np.ndarray:
    """Sidon‑like a_i = ∏_{k=1}^i (1-1/(2k))."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def _build_sidon_positions(n: int, m: int) -> list:
    """Greedy construction of ≤ m positions whose pairwise sums are all distinct."""
    chosen = []
    sums = set()
    for pos in range(n):
        ok = True
        # check unordered sums with already chosen positions
        for q in chosen:
            if (pos + q) in sums:
                ok = False
                break
        if ok and (pos + pos) not in sums:
            for q in chosen:
                sums.add(pos + q)
            sums.add(pos + pos)
            chosen.append(pos)
            if len(chosen) >= m:
                break
    return chosen


def _score(arr: np.ndarray) -> float:
    """Wrapper for the provided evaluator (expects a Python list)."""
