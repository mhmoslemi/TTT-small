import time
import numpy as np

# ------------------------------------------------------------
# fast convolution‑based score and index of the maximal entry
# ------------------------------------------------------------
def _score_and_index(vec: np.ndarray):
    """Return (score, index_of_max_convolution) for a non‑negative vector."""
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())          # power‑of‑2 length for FFT
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])
    sc = 2.0 * n * max_val / (vec.sum() ** 2)
    return sc, max_idx


# ------------------------------------------------------------
# main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------
    # initialise a feasible sequence (sum normalised to 1)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        total = a.sum()
        if total <= 0.0:
            a = np.full_like(a, 1.0 / a.size, dtype=np.float64)
        else:
            a = a / total
    else:
        # fallback: uniform distribution (any length >0 works)
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size

    # best solution seen so far
    best_vec = a.copy()
    best_sc, best_max_idx = _score_and_index(best_vec)

    # current solution (used for the Metropolis walk)
    cur_vec = a.copy()
    cur_sc, cur_max_idx = best_sc, best_max_idx

    # --------------------------------------------------------
    # simulated‑annealing parameters
    # --------------------------------------------------------
    T_start = 0.02            # initial temperature
    T_end = 1e-4              # final temperature
    max_frac = 0.5            # maximal proportion of source mass to move
    half = n // 2

    # --------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        # linear cooling schedule
        elapsed = time.monotonic() - start
        total = deadline - start
        t_frac = min(1.0, max(0.0, elapsed / total))
        T = T_start * (1.0 - t_frac) + T_end * t_frac

        # ----------------------------------------------------
        # pick a move: either a random two‑index transfer or a
        # targeted transfer that tries to reduce the current
        # maximal convolution entry.
        # ----------------------------------------------------
        if rng.random() < 0.6:          # biased move
            if rng.random() < 0.5:      # targeted move based on cur_max_idx
                k = cur_max_idx
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                # contribution of each i to the peak value
                idxs = np.arange(lo, hi + 1, dtype=np.int64)
                contrib = cur_vec[idxs] * cur_vec[k - idxs]
                total_c = contrib.sum()
                if total_c > 0.0:
                    # pick a source proportional to its contribution
                    prob = contrib / total_c
                    i = rng.choice(idxs, p=prob)
                else:
                    i = rng.integers(0, n)
                # choose a far‑away target
                if i < half:
                    j = rng.integers(half, n)
                else:
                    j = rng.integers(0, half)
            else:                      # simple early→late transfer
                i = rng.integers(0, half)
                j = rng.integers(half, n)
        else:
