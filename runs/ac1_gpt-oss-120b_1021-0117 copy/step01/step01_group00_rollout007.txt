Define compute_score(seq):
    seq = np.asarray(seq, dtype=np.float64)
    s = seq.sum()
    if s < 0.01: return np.inf
    # compute linear convolution using FFT
    n = seq.size
    size = 2 * n - 1
    # compute next power of 2 for efficient FFT
    fsize = 1 << (size - 1).bit_length()
    fft_a = np.fft.rfft(seq, n=fsize)
    conv = np.fft.irfft(fft_a * fft_a, n=fsize)[:size]
    max_b = conv.max()
    return 2 * n * max_b / (s * s)

def _sidon_like_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (1.0 - 1.0/(2.0*i))
    return a

def fast_score(arr):
    n = arr.shape[0]

import time
import math
import numpy as np

def _sidon_like_sequence(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4**i (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def _fast_score(seq: np.ndarray) -> float:
    """Efficient evaluation of the AC1 bound (identical to evaluate_sequence)."""
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    size = 2 * n - 1
    # next power of two for an efficient FFT
    fsize = 1 << ((size - 1).bit_length())
    fft = np.fft.rfft(seq, n=fsize)
    conv = np.fft.irfft(fft * fft, n=fsize)[:size]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Return a list of non‑negative heights (step‑function values) that
    aims to minimise the AC1 bound defined by evaluate_sequence().
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # safety margin

    # ------------------------------------------------------------------
    # Determine problem size and an initial sequence.
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is supplied by the harness when available.
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.shape[0]
        if n == 0 or not np.isfinite(parent).all():
            raise Exception
        parent = np.clip(parent, 0.0, 1000.0)
        have_parent = True
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)
        have_parent = False

    # Two natural candidates: the supplied parent (if any) and the Sidon‑like
    # construction.  We keep the one that gives the better fast score.
    cand_parent = parent.copy()
    cand_sidon = _sidon_like_sequence(n)
    cand_sidon = np.clip(cand_sidon, 0.0, 1000.0)

    best_seq = cand_parent if _fast_score(cand_parent) < _fast_score(cand_sidon) else cand_sidon
    best_score = _fast_score(best_seq)

    # ------------------------------------------------------------------
    # Helper to possibly replace the current best.
    # ------------------------------------------------------------------
    def maybe_update(seq: np.ndarray, sc: float):
        nonlocal best_seq, best_score
        if sc < best_score:
            best_seq = seq.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # Systematic permutations of the current best sequence.
    # ------------------------------------------------------------------
    # 1) reversed
    maybe_update(best_seq[::-1], _fast_score(best_seq[::-1]))

    # 2) alternating ends (0, n‑1, 1, n‑2, …)
    alt_idx = np.empty(n, dtype=np.int64)
    lo, hi, pos = 0, n - 1, 0
    while lo <= hi:
        alt_idx[pos] = lo
        pos += 1
        lo += 1
        if lo <= hi:
            alt_idx[pos] = hi
            pos += 1
            hi -= 1
    maybe_update(best_seq[alt_idx], _fast_score(best_seq[alt_idx]))

    # 3) multiplicative permutations with coprime steps
    mul_steps = []
    attempts = 0
    while len(mul_steps) < 12 and attempts < 200:
        p = int(rng.integers(2, n))
        if math.gcd(p, n) == 1:
            mul_steps.append(p)
        attempts += 1
    for p in mul_steps:
        perm = (np.arange(n, dtype=np.int64) * p) % n
        maybe_update(best_seq[perm], _fast_score(best_seq[perm]))

    # 4) a few random shuffles
    for _ in range(5):
        perm = rng.permutation(n)
        maybe_update(best_seq[perm], _fast_score(best_seq[perm]))

    # ------------------------------------------------------------------
    # Stochastic local search (swap moves) until time runs out.
    # ------------------------------------------------------------------
    current_seq = best_seq.copy()
    current_score = best_score

    while time.monotonic() < deadline:

        i, j = rng.integers(0, n, size=2)
        if i == j:

        cand = current_seq.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_score = _fast_score(cand)

        if cand_score < current_score:
            current_seq = cand
            current_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score

        # occasional restart from global best to keep exploration alive
        if rng.random() < 0.001:
            current_seq = best_seq.copy()
            current_score = best_score

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq.tolist()]
