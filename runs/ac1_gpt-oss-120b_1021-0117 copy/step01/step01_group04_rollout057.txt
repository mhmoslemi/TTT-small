import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _eval(arr: np.ndarray, n: int, L: int):
    """Return (score, max_val, max_idx, conv) using FFT convolution."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    conv = conv[: 2 * n - 1]
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf'), max_val, max_idx, conv
    score = 2.0 * n * max_val / (s * s)
    return score, max_val, max_idx, conv

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # ----- problem size ----------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256

    L = _next_pow2(2 * n - 1)

    # ----- initial candidate ------------------------------------------------
    if parent is not None:
        cur = parent.copy()
    else:
        cur = _gen_binomial(n)
    np.clip(cur, 0.0, 1000.0, out=cur)

    cur_score, _, cur_max_idx, _ = _eval(cur, n, L)
    best = cur.copy()
    best_score = cur_score
    best_max_idx = cur_max_idx

    # ----- deterministic exploration of simple families --------------------
    # power‑law a_i = (i+1)^(-p)
    for p in np.linspace(0.1, 1.0, 19):
        cand = 1.0 / np.power(np.arange(n, dtype=np.float64) + 1.0, p)
        np.clip(cand, 0.0, 1000.0, out=cand)
        s, _, _, _ = _eval(cand, n, L)
        if s < best_score:
            best = cand.copy()
            best_score = s
            best_max_idx = None
    # exponential a_i = exp(-lam * i)
    for lam in np.linspace(0.0005, 0.005, 10):
        cand = np.exp(-lam * np.arange(n, dtype=np.float64))
        np.clip(cand, 0.0, 1000.0, out=cand)
        s, _, _, _ = _eval(cand, n, L)
        if s < best_score:
            best = cand.copy()
            best_score = s
            best_max_idx = None
    # ensure we start the stochastic phase from the best we have so far
    cur = best.copy()
    cur_score, _, cur_max_idx, _ = _eval(cur, n, L)

    # ----- stochastic local search -----------------------------------------
    temperature = 1.0
    decay = 0.9993
    prob_pair   = 0.40   # targeted reduction of the dominant contribution
    prob_block  = 0.20   # block scaling with sum‑preserving renorm
    prob_rand   = 0.20   # simple mass transfer
    prob_smooth = 0.20   # smoothing kernel

    while time.monotonic() < deadline:
        move = rng.random()
        # --- 1) targeted pair reduction ------------------------------------

            cand[j] += delta_i + delta_p

            best_score = cand_score
