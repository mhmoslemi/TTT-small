def evaluate_sequence(sequence):
    # check types. Clip to [0,1000].
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

def _score_and_conv(arr, fft_len):
    # return (score, conv_array)
    fa = np.fft.rfft(arr, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*len(arr)-1]
    maxb = conv.max()
    s = arr.sum()
    score = float(2 * len(arr) * maxb / (s * s))

import time, math

def _fft_len_for_n(n: int) -> int:
    """Length of power‑of‑2 FFT covering linear convolution of length n."""
    conv_len = 2 * n - 1

def _score(arr: np.ndarray, fft_len: int) -> float:
    """Fast proxy for evaluate_sequence – works for non‑negative arrays."""
    fa = np.fft.rfft(arr, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * len(arr) - 1]
    maxb = conv.max()
    s = arr.sum()
    return float(2 * len(arr) * maxb / (s * s))

    a = np.empty(n, dtype=np.float64)
