def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    # Get n
    try:
        parent_np = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_np.ndim != 1 or parent_np.size == 0:
            raise Exception
        n = parent_np.shape[0]
    except Exception:
        n = 256
        parent_np = None
    # conv size
    L = 1 << ((2*n-2).bit_length())
    # define ac1 score
    def ac1(arr):
        fa = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        maxb = float(conv.max())
        s = float(arr.sum())
        if s < 1e-12:  # avoid division by zero
            return float('inf')
        return 2.0 * n * maxb / (s * s)
    # initial best: evaluate parent (if any) and a standard base (binomial perhaps)
    best = None
    best_score = float('inf')
    if parent_np is not None:
        # Clip values as tolerant
        arr = np.clip(parent_np, 0.0, 1000.0)
        sc = ac1(arr)
        best = arr.copy()
        best_score = sc
    # also try binomial base
    base = np.empty(n, dtype=np.float64)
    base[0] = 1.0
    for i in range(1, n):
        base[i] = base[i-1] * (2*i - 1) / (2*i)
    base = np.clip(base, 0.0, 1000.0)
    bscore = ac1(base)
    if bscore < best_score:
        best = base.copy()
        best_score = bscore
    # try Gaussian shapes with various sigma fractions
    mids = (n - 1) / 2.0
    # sigma fractions from 0.1 to 0.5 maybe 0.05 increments
    for frac in np.linspace(0.1, 0.6, 11):
        sigma = frac * n
        # avoid sigma too small causing zeros and low sum? still evaluate
        idx = np.arange(n, dtype=np.float64)
        arr = np.exp(-0.5 * ((idx - mids) / sigma) ** 2)
        arr = np.clip(arr, 0.0, 1000.0)
        sc = ac1(arr)
        if sc < best_score:
            best = arr.copy()
            best_score = sc
        if time.monotonic() > deadline:
            break
    # also optionally try an alternative shape: a triangle that peaks at center
    # build triangle shape: a_i = 1 - abs(i - mid)/ (mid+1)
    # maybe scaling to ensure non-negative.
    # But likely not needed.

    # if still best is None (should not), set to base.
    if best is None:
        best = base.copy()
        best_score = ac1(best)
    # start local search from best:
    cur = best.copy()
    cur_score = best_score
    temperature = 1.0
    decay = 0.9995
    # search loop with time check
    while time.monotonic() < deadline:
        # Choose move: produce candidate
        if rng.random() < 0.7:
            # block scaling
            # pick random width
            max_width = max(1, n // 10)
            min_width = max(1, n // 200)
            width = int(rng.integers(min_width, max_width+1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.09))
            cand = cur.copy()
            cand[start:start+width] *= factor
        else:
            # transfer mass
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac_move = rng.random() * 0.3
                delta = cur[i] * frac_move
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                continue
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = ac1(cand)
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            # early break if below target
            if best_score < 1.50274349:
                break
        # acceptance
        if cand_score < cur_score or rng.random() < math.exp((cur_score - cand_score) / (temperature + 1e-12)):
            cur = cand
            cur_score = cand_score
        temperature *= decay
        # occasional random restart from best
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0
        # maybe early exit if time near.
        # Continue

    # return best list of floats
    return [float(x) for x in best]
