import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ------------------------------------------------------------
# auxiliary: FFT based full convolution (real input)
# ------------------------------------------------------------
def _conv_full(v: np.ndarray) -> np.ndarray:
    """Full linear convolution v*v using real FFT."""
    n = v.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# score exactly as used by the evaluator
# ------------------------------------------------------------
def _score(v: np.ndarray) -> float:
    n = v.size
    conv = _conv_full(v)
    return 2.0 * n * conv.max() / (v.sum() ** 2)


# ------------------------------------------------------------
# sqrt‑series (analytic start if no parent is provided)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty sequence of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = _sqrt_series(256)

    a = np.maximum(a, 0.0)                     # safety
    if a.sum() < 0.01:                         # avoid degenerate start
        a += 0.01 / a.size

    best_vec = a
    best_sc = _score(best_vec)

    n = best_vec.size
    target_sum = np.sqrt(2.0 * n)               # theoretical normalisation

    # ----- pre‑compute index arrays for the Toeplitz constraint matrix --
    i_arr = np.repeat(np.arange(n, dtype=np.int32), n)
    j_arr = np.tile(np.arange(n, dtype=np.int32), n)
    row_arr = i_arr + j_arr                     # shape (n*n,)

    # ----- main optimisation loop (LP + mixing) -----------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    lp_iter = 0
    max_lp_iter = 12                 # enough for noticeable progress but cheap

    while time.monotonic() < deadline and lp_iter < max_lp_iter:
        lp_iter += 1
        a = best_vec
        n = a.size

        # current convolution and its maximum
        conv = _conv_full(a)
        M = conv.max()

        # build sparse convolution matrix A where A @ b = a * b (full)
        data = a[i_arr]                               # repeats of a_i
        A = sp.csr_matrix((data, (row_arr, j_arr)), shape=(2 * n - 1, n))

        # linear programme: maximise sum(b)  subject to  A @ b <= M , b >= 0
        c = -np.ones(n, dtype=np.float64)             # min –sum(b) == max sum(b)
        ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        try:
            res = opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                              method='highs')
        except Exception:                             # any failure → stop LP
            break
        if res.status != 0:                           # not optimal
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # rescale to the standard normalisation
        g = b * (target_sum / sum_b)

        # try a few mixing fractions; keep the first improvement
        improved = False
        for t in (0.5, 0.25, 0.1, 0.05):

                improved = True

            if best_vec[i] == 0.0:

            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = best_vec[i] * rng.random() * 0.1
            cand = best_vec.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_sc:
                best_sc, best_vec = sc, cand

        # leave a small margin for the final hill‑climb

    # ----- final local search (simple hill‑climb) --------------------

        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand

    # ----- return ----------------------------------------------------
    return [float(x) for x in best_vec]
