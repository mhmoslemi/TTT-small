# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (same formula as the evaluator)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ------------------------------------------------------------
# Gradient of the maximal convolution entry
# ------------------------------------------------------------
def _grad_at_shift(a: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of conv[a] at the given shift (full convolution index)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if shift < n:
        # indices 0 .. shift inclusive
        g[: shift + 1] = 2.0 * a[shift::-1]
    else:
        # right half of the convolution
        length = 2 * n - 1 - shift          # number of overlapping terms
        start = shift - (n - 1)            # first index that participates
        # take the last `length` entries of `a` in reverse order
        g[start:] = 2.0 * a[-1 : -length - 1 : -1]
    return g


# ------------------------------------------------------------
# Optional analytic start (sqrt series) if no parent is supplied
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of the series a with a*a = 1+…+x^{2n-2}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialise a feasible point (use the provided parent if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
    else:
        n = 256
        a = _sqrt_series(n)
    a = _project_simplex(a)               # ensures sum = 1 and non‑negative
    n = a.shape[0]

    # ------------------------------------------------------------------
    # pre‑compute FFT length (next power of two)
    # ------------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    best_vec = a.copy()
    best_sc = _score(best_vec)

    # step size for gradient moves; will adapt during the run
    step = 0.5

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- full convolution and current max index -----
        fv = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        max_idx = int(conv_full.argmax())
        # ----- gradient of the active shift -----
        grad = _grad_at_shift(a, max_idx)

        improved = False
        cur_step = step
        # ----- try decreasing step sizes along -grad direction -----
        for _ in range(6):
            cand = a - cur_step * grad
            cand = _project_simplex(cand)
            sc = _score(cand)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_vec = cand.copy()
                a = cand
                improved = True
                step = min(step * 1.1, 1.0)       # slowly enlarge step after success
                break
            cur_step *= 0.5

        if not improved:
            # ----- random mass‑transfer move (exploration) -----
            i = rng.integers(0, n)
            if a[i] > 1e-14:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.2            # up to 20 % of source mass
                delta = a[i] * frac
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)
                sc = _score(cand)
                if sc < best_sc:
                    best_sc = sc
                    best_vec = cand.copy()
                    a = cand
                    step = max(step * 0.7, 0.01)      # shrink step after a disruptive move

        # occasionally inject a tiny Gaussian jitter to escape possible plateaus
        if rng.random() < 0.001:
            noise = rng.normal(scale=0.001, size=n)
            cand = best_vec + noise
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            sc = _score(cand)
            if sc < best_sc:
                best_sc = sc
                best_vec = cand.copy()
                a = cand

    # ------------------------------------------------------------------
    # return the best vector found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
