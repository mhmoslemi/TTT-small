import numpy as np, time

# ------------------------------------------------------------
# 1.  Initial binomial seed (C(2i,i)/4^i) – monotone decreasing.
# ------------------------------------------------------------
def _initial_binomial(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

# ------------------------------------------------------------
# 2.  Fast AC‑1 evaluation via FFT (size is next power of two).
# ------------------------------------------------------------
def _fast_score(seq: np.ndarray):
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # power‑of‑two ≥ 2n‑1
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val

    # Detect parent sequence (height_sequence_1) if provided.
    # --------------------------------------------------------
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
        else:
            parent = None
            n = 3809
    except Exception:
        parent = None
        n = 3809

    # --------------------------------------------------------
    # Initial candidate – binomial or the supplied parent if better.
    # --------------------------------------------------------
    cand = _initial_binomial(n)
    best_score, best_conv, _ = _fast_score(cand)

    if parent is not None:
        p_score, p_conv, _ = _fast_score(parent)
        if p_score < best_score:
            cand = parent.copy()
            best_score, best_conv = p_score, p_conv

    # current state (used for proposing moves)
    cur = cand.copy()
    cur_score, cur_conv, _ = _fast_score(cur)

    # constants reused each iteration
    idx = np.arange(n, dtype=np.int64)

    # temperature schedule (simulated‑annealing style)
    T0 = 0.02
    T = T0
    decay = 0.9995

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # ------------------------------------------------------------------
        #  Choose a move type.
        # ------------------------------------------------------------------
        r = rng.random()
        nxt = cur.copy()

        if r < 0.50:
            # --------------------------------------------------------------
            #  Gradient‑guided mass relocation (core improvement move).
            # --------------------------------------------------------------
            k_max = int(np.argmax(cur_conv))               # index of max conv
            partner = k_max - idx                           # partner index for each i
            mask = (partner >= 0) & (partner < n)

            # product a_i * a_partner_i for all valid i
            prod = np.zeros_like(nxt)
            prod[mask] = nxt[idx[mask]] * nxt[partner[mask]]

            # select a modest number of largest contributing pairs
            donor_cnt = max(1, n // 200)                    # ≈1 % of entries
            top = np.argpartition(-prod, donor_cnt - 1)[:donor_cnt]

            # set of indices that will give up a fraction of their mass
            donors = set()
            for i in top:
                j = partner[i]
                if 0 <= j < n:
                    donors.add(i)
                    donors.add(j)

            if donors:
                alpha = rng.uniform(0.04, 0.12)            # fraction to move
                moved = 0.0
                for d in donors:
                    amt = nxt[d] * alpha
                    nxt[d] -= amt
                    moved += amt

                # pick low‑value locations to receive the moved mass
                not_donor = np.ones(n, dtype=bool)
                not_donor[list(donors)] = False
                avail = np.nonzero(not_donor)[0]
                if avail.size:
                    rec_cnt = min(len(donors), avail.size)
                    rec = avail[np.argpartition(nxt[avail], rec_cnt - 1)[:rec_cnt]]
                    inc = moved / rec_cnt
                    nxt[rec] += inc

        elif r < 0.80:
            # --------------------------------------------------------------
            #  Small block scaling – gentle random perturbation.
            # --------------------------------------------------------------
            max_w = max(2, n // 100)
            w = rng.integers(1, max_w + 1)
            s = rng.integers(0, n - w + 1)
            factor = np.exp(rng.normal(0.0, 0.06))
            nxt[s:s + w] *= factor
        else:
            # --------------------------------------------------------------
            #  Single entry scaling – larger exploration steps.
            # --------------------------------------------------------------
            i = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.12))
            nxt[i] *= factor

        # enforce bounds
        np.clip(nxt, 0.0, 1000.0, out=nxt)

        # ------------------------------------------------------------------
        #  Evaluate the proposal.
        # ------------------------------------------------------------------
        cand_score, cand_conv, _ = _fast_score(nxt)

        # ------------------------------------------------------------------
        #  Acceptance test (greedy with occasional uphill moves).
        # ------------------------------------------------------------------
        delta = cand_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # simulated‑annealing style acceptance
            prob = np.exp(-delta / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = nxt
            cur_score, cur_conv = cand_score, cand_conv
            # decay temperature only on successful move to keep some vigor
            T *= decay
        else:
            # even if rejected, still cool a tiny bit
            T *= decay

        # ------------------------------------------------------------------
        #  Keep track of the best solution ever seen.
        # ------------------------------------------------------------------
        if cand_score < best_score:
            best = nxt
            best_score = cand_score
            best_conv = cand_conv

        # ------------------------------------------------------------------
        #  Light “smooth‑pair” tweak every few thousand steps – it may
        #  reduce isolated peaks without hurting the overall shape.
        # ------------------------------------------------------------------
        if it % 2500 == 0:
            smo = best.copy()
            i = rng.integers(0, n - 1)
            avg = (smo[i] + smo[i + 1]) * 0.5
            smo[i] = smo[i + 1] = avg
            np.clip(smo, 0.0, 1000.0, out=smo)
            s_score, s_conv, _ = _fast_score(smo)
            if s_score < best_score:
                best, best_score, best_conv = smo, s_score, s_conv

        # ------------------------------------------------------------------
        #  Early exit if we already beat the target.
        # ------------------------------------------------------------------
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------
    # Return the best sequence as a plain Python list.
    # ------------------------------------------------------------
    return [float(v) for v in best]
