def random_transfer(v):
    n = v.shape[0]; i = rng.integers(n); if v[i] < 1e-12: return v; else j = rng.integers(n-1); if j >= i: j+=1
    fraction = rng.random() * 0.2 # up to 20% of source mass
    delta = v[i] * fraction
    cum = v.copy(); cum[i] -= delta; cum[j] += delta
    cum = _project_simplex(cum)  # ensures sum=1
    return cum

import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1, dtype=np.float64)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast score: 2 * n * max(conv) / (sum^2)
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (vec.sum() ** 2))


# ------------------------------------------------------------
# Subgradient of max(conv) for given maximal indices
# ------------------------------------------------------------
def _subgradient(v: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    grad = np.zeros_like(v)
    for k in idxs:
        left = max(0, k - (n - 1))
        right = min(k, n - 1)
        # range of i where both v[i] and v[k-i] are defined
        i = np.arange(left, right + 1, dtype=np.int64)
        rev = k - i
        grad[left : right + 1] += 2.0 * v[rev]
    if len(idxs) > 1:
        grad /= len(idxs)
    return grad

# Analytic sqrt‑series (fallback if no parent vector)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

                frac = rng.random() * 0.25
                delta = best_vec[i] * frac
                cand = best_vec.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)
                cand = (cand + cand[::-1]) * 0.5
                sc = _score(cand)
                if sc < best_sc:
                    best_sc = sc
                    best_vec = cand

        # minimal time‑check to avoid overrun
        if time.monotonic() + 0.05 >= deadline:
            break

    # --------------------------------------------------------
    # return result as plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
