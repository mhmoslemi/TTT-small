def _conv_self_fft(a):
    n = a.shape[0]
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=m)
    prod = fa*fa
    conv = np.fft.irfft(prod, n=m)[:2*n-1]
    return conv

def _score_norm1(a):  # sum=1 approx
    n = a.shape[0]
    conv = _conv_self_fft(a)
    M = conv.max()
    return 2.0 * n * M

def _gradient_at_shift(a, shift):
    n = a.shape[0]
    g = a.copy()
    # compute overlapping range
    start_j = max(0, shift - (n - 1))
    end_j = min(n - 1, shift)
    if start_j <= end_j:
        j = np.arange(start_j, end_j+1)
        i = shift - j
        g[i] += a[j]
    return g

import numpy as np
import time


def _conv_full(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_norm1(a: np.ndarray, fft_len: int) -> float:
    """Objective when the coefficients sum to 1 (the scaling used by the optimiser)."""
    n = a.size
    conv = _conv_full(a, fft_len)
    return 2.0 * n * conv.max()


def _gradient_at_shift(a: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient of the convolution entry at index ``shift`` w.r.t. ``a``.
    For each i, ∂(a*a)[shift]/∂a_i = a_i + a_{shift-i}  (second term omitted if out of range).
    """
    n = a.size
    g = a.copy()
    start = max(0, shift - (n - 1))
    end = min(n - 1, shift)
    if start <= end:
        j = np.arange(start, end + 1, dtype=int)
        i = shift - j
        g[i] += a[j]
    return g
