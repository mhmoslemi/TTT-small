def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step‐function that minimises the AC1 bound.
    The routine starts from the supplied parent (height_sequence_1) or a
    simple fallback, then performs a few sparse‑LP refinements followed by
    cheap gradient / random moves.  It respects the time budget and returns
    the best sequence found.
    """
    rng = np.random.default_rng(seed)
    # reserve a few seconds for the harness shutdown
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 1.  Base sequence (fallback to Catalan‑type if needed)
    # ------------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        if a0.ndim != 1 or a0.size == 0:
            raise ValueError
    except Exception:
        # fallback: uniform of modest length (will be resized later)
        a0 = np.ones(256, dtype=np.float64)

    n = int(a0.shape[0])
    a = np.clip(a0, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:                # avoid division‑by‑zero pathologies
        a = np.full(n, 1.0 / n)
    else:
        a = a / s                # normalise to sum = 1 (scaling does not affect score)

    # ------------------------------------------------------------------
    # 2.  Helper utilities
    # ------------------------------------------------------------------
    def objective(seq):
        """Exact objective value (2·n·max(conv(seq,seq))) – faster than evaluate_sequence."""
        conv = np.convolve(seq, seq)
        return 2.0 * n * float(conv.max())

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))   # keep a fully verified score

    # pre‑compute sparse convolution matrix pattern (rows/cols are independent of a)
    idx = np.arange(n, dtype=np.int32)
    rows = (idx[:, None] + idx[None, :]).ravel()          # shape (n*n,)
    cols = np.tile(idx, n)                               # shape (n*n,)
    shape = (2 * n - 1, n)

    # ------------------------------------------------------------------
    # 3.  LP refinement loop (few iterations, each cheap for n≈2500)
    # ------------------------------------------------------------------
    mix_factor = 0.5                     # convex‑combination weight
    target_score = 1.50274349

    for _ in range(8):                      # at most eight LP steps
        if time.monotonic() >= deadline:
            break
        M = float(np.max(np.convolve(a, a)))        # current max autocorrelation

        # Build the data for the sparse matrix C where (C @ b)[k] = (a * b)[k]
        data = np.repeat(a, n)                       # n*n entries, fits in memory
        C = sp.csr_matrix((data, (rows, cols)), shape=shape)

        # LP: maximise Σb  ⇔  minimise −Σb  subject to C·b ≤ M, b ≥ 0
        c = -np.ones(n, dtype=np.float64)            # objective for minimisation

        res = opt.linprog(
