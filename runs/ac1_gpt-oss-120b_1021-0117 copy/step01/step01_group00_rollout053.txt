            cand[i] -= delta
            cand[j] += delta
            # values stay far below the 1000 bound, clipping not required

        cand_score = _eval(cand)

        # ----- update global best ----------------------------------------
        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score

        # ----- accept/reject for the current walk -----------------------
        remaining = deadline - now
        temperature = temp0 * max(remaining / total_time, 0.0)

        if cand_score < cur_score:
            cur_seq = cand
            cur_score = cand_score
        else:
            # Metropolis criterion (down‑hill passes automatically)
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                cur_seq = cand
                cur_score = cand_score

        # ----- occasional perturbation of the best to escape basins ------
        if rng.random() < 0.001:
            perturb = (rng.random(n) - 0.5) * 0.02   # ±1 % multiplicative noise
            cand = best_seq * (1.0 + perturb)
            cand = np.clip(cand, 0.0, 1000.0)
            total = cand.sum()
            if total > 0.0:
                cand = cand * (cur_seq.sum() / total)
            cand_score = _eval(cand)
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
            if cand_score < cur_score:
                cur_seq = cand
                cur_score = cand_score

    # ----------------------------------------------------------------------
    # Return a plain‑Python list as required by the harness
    # ----------------------------------------------------------------------
    return [float(x) for x in best_seq]
