import time
import numpy as np

def _catalan_weights(n: int) -> np.ndarray:
    """Return the first n Catalan‑type weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _grad_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the maximum convolution value.
    a – current vector (sum = 1, non‑negative)
    conv – full convolution a*a (np.convolve)
    Returns a gradient with the same shape as a.
    """
    n = a.size
    max_val = conv.max()
    # Use a tiny absolute tolerance; the maximum is typically unique.
    tol = max_val * 1e-12 if max_val != 0 else 1e-12
    idxs = np.where(conv >= max_val - tol)[0]
    shifts = idxs - (n - 1)               # map index to shift (k)
    g = np.zeros_like(a)

    for k in shifts:
        if k >= 0:
            length = n - k
            if length <= 0:
                continue
            left = a[:length]            # a_i
            right = a[k:]                # a_{i+k}
            g[:length] += right          # derivative w.r.t. a_i
            g[k:] += left                # derivative w.r.t. a_{i+k}
        else:
            kk = -k
            length = n - kk
            if length <= 0:
                continue
            left = a[kk:]                # a_i where i >= kk
            right = a[:length]           # a_{i-kk}
            g[kk:] += right
            g[:length] += left
    return g
