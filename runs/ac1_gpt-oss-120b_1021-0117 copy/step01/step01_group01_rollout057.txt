def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # keep buffer

    try:
        n = len(height_sequence_1)
        a = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64) / n

    # Ensure nonnegativity & clip to 0-1000
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        a = np.ones_like(a) / a.size

    # Normalise sum to 1 (scaling doesn't matter)
    a = a / a.sum()
    # We can keep sum=1 for easier updates; final scaling irrelevant for objective.

    # Precompute FFT length
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # Helper: compute objective quickly via fft
    def score(arr):
        # arr is 1d np array sum=1 maybe
        fa = np.fft.rfft(arr, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        maxb = conv.max()
        return float(2 * n * maxb / (arr.sum() ** 2))

    cur = a.copy()
    cur_score = score(cur)
    best = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng(seed)

    # We'll do a loop that attempts pairwise transfers directed by peak contributions

    while time.monotonic() < deadline:
        # Compute convolution and find index of max
        fa = np.fft.rfft(cur, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # compute contributions for each index i
        # j = max_idx - i
        # contributions where 0 <= j < n
        i_arr = np.arange(n)
        j_arr = max_idx - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[mask] = cur[i_arr[mask]] * cur[j_arr[mask]]

        # choose a source index i with high contribution
        # Use weighted random based on contribution (or top-k)
        # We also want to avoid zero entries.
        # We'll sample from distribution proportional to contrib.
        total_contrib = contrib.sum()
        if total_contrib <= 0:
            # fallback random
            i = rng.integers(0, n)
        else:
            # sample i weighted by contrib
            probs = contrib / total_contrib
            # Using rng.choice with p
            i = rng.choice(n, p=probs)

        # Choose destination j random (different)
        # But we may also pick one with low contribution.
        # Let's sample uniformly for simplicity.
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1

        # Ensure source amount positive
        if cur[i] <= 1e-12:
            continue

        # Transfer a fraction of cur[i] to j, e.g., up to 30% but limited to keep positivity.
        frac = rng.uniform(0.02, 0.25)  # adjustable
        delta = cur[i] * frac

        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # Evaluate candidate
        new_score = score(new)

        # Accept if improved
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        # Early break if we achieved target threshold
        if best_score < 1.503:  # good
            break

    # Ensure final best is list float within bounds
    final = [float(x) for x in best]
    # optionally clip (though evaluator will clip)
    return final
```

def _normalize(arr):
    """Clip to [0,1000] and scale to unit sum (sum>0)."""
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s < 1e-12:
        arr = np.ones_like(arr) / arr.size
    else:
        arr = arr / s
    return arr


def _conv_max(arr, fft_len):
    """Maximum entry of a*a using a single FFT."""
    fa = np.fft.rfft(arr, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * arr.size - 1]
    return float(conv.max()), conv


def _score(arr, n, fft_len):
    """Exact objective (scale invariant, sum is 1)."""
    max_b, _ = _conv_max(arr, fft_len)
    # sum(arr) is 1 after normalization, kept for safety
    s = arr.sum()
    return float(2 * n * max_b / (s * s))


def _generate_binomial_seq(n):
    """C(2i,i)/4^i via recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0

        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a

def _hill_climb(start, n, fft_len, deadline, rng):

    cur_score = _score(cur, n, fft_len)
    best = cur.copy()

    while time.monotonic() < deadline:

        max_b, conv = _conv_max(cur, fft_len)
        max_idx = int(conv.argmax())

                if best_score < 1.50274349:
                    break
    return best, best_score


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a list of non‑negative heights with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # -----------------------------------------------------------------
    # Determine length and a reasonable starting vector (parent if present)
    # -----------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        parent = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64) / n

    # -----------------------------------------------------------------
    # FFT layout (fixed for the whole run)
    # -----------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # -----------------------------------------------------------------
    # Build a collection of candidate shapes
    # -----------------------------------------------------------------
    candidates = []

    # parent (if any)
    candidates.append(('parent', _normalize(parent)))

    # binomial
    candidates.append(('binomial', _normalize(_generate_binomial_seq(n))))

    # uniform
    candidates.append(('uniform', np.full(n, 1.0 / n, dtype=np.float64)))

    # triangular
    candidates.append(('triangular', _normalize(_triangular(n))))

    # Hann window
    candidates.append(('hann', _normalize(_hann_window(n))))

    # Gaussian (several sigmas)
    for sigma_factor in (0.12, 0.18, 0.25):
        sigma = n * sigma_factor
        candidates.append((f'gauss_{sigma_factor:.2f}', _normalize(_gaussian(n, sigma))))

    # Exponential (several taus)
    for tau_factor in (0.15, 0.25, 0.35):
        tau = n * tau_factor
        candidates.append((f'exp_{tau_factor:.2f}', _normalize(_exponential(n, tau))))

    # Power‑law exponents
    for p in (0.20, 0.35, 0.50, 0.65, 0.80):
        candidates.append((f'pow_{p:.2f}', _normalize(_power_law(n, p))))

    # DPSS windows (if SciPy is available)
    for nw in (1.5, 2.0, 2.5, 3.0, 3.5):
        w = _dpss_window(n, nw)
        if w is not None:
            candidates.append((f'dpss_{nw:.1f}', _normalize(w)))

    # -----------------------------------------------------------------
    # Evaluate each raw candidate, remember the best
    # -----------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    for name, seq in candidates:
        s = _score(seq, n, fft_len)
        if s < best_score:
            best_score = s
            best_seq = seq.copy()
        # early exit if we already beat the target
        if best_score < 1.50274349:
            break
        # respect overall deadline
        if time.monotonic() >= deadline:
            break

    # -----------------------------------------------------------------
    # Perform a focused hill‑climb on the current best candidate
    # -----------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 0.5 and best_seq is not None:
        # allocate 90 % of the remaining time to hill climbing
        hc_deadline = time.monotonic() + 0.9 * remaining
        improved_seq, improved_score = _hill_climb(best_seq, n, fft_len, hc_deadline, rng)
        if improved_score < best_score:
            best_seq = improved_seq
            best_score = improved_score

    # -----------------------------------------------------------------
    # Prepare final list (the evaluator will clip again, but we stay safe)
    # -----------------------------------------------------------------
    final = [float(x) for x in best_seq]
    return final
