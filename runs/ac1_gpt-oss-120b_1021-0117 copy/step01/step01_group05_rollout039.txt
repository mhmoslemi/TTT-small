    m = 1 << ((2*n - 1).bit_length())

    s = seq.sum()

    single_log_std = 0.30  # log‑normal single‑index scaling

    # ------------------------------------------------------------------
    # Main optimisation loop – runs until the time budget expires.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- Gradient‑driven move ------------------------------------
        kmax = int(np.argmax(best_conv))
        a_shift = _shifted(best, kmax)
        sum_a = best.sum()
        max_val = best_conv.max()
        # Direction derived from sub‑gradient of 2 n·max_conv / sum²
        direction = a_shift - (max_val / sum_a) * best

        improved = False
        for alpha in grad_alphas:
            cand = best - alpha * direction
            np.clip(cand, 0.0, 1000.0, out=cand)
            s = cand.sum()
            if s <= 0.0:
                continue
            # Rescale to keep the ℓ¹‑norm unchanged (scale‑invariant objective)
            cand *= (sum_a / s)
            cand_score, cand_conv, _ = _fast_score(cand, m)
            if cand_score < best_score:
                best, best_score, best_conv = cand, cand_score, cand_conv
                improved = True
                break
        if improved:
            continue

        # ----- Stochastic perturbations (fallback) --------------------
        cand = best.copy()
        op = rng.integers(0, 4)

        if op == 0:                      # block scaling
            width = rng.integers(1, max(2, n // 30))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, block_log_std))
            cand[start:start + width] *= factor

        elif op == 1:                    # single‑index perturbation
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, single_log_std))
            cand[idx] *= factor

        elif op == 2:                    # move mass away from the maximal pair
            lo = max(0, kmax - (n - 1))
            hi = min(kmax, n - 1)
            i = rng.integers(lo, hi + 1)
            j = kmax - i
            if i != j:                     # avoid the diagonal case
                frac = rng.uniform(0.04, 0.12)
                rem_i = cand[i] * frac
                rem_j = cand[j] * frac
                cand[i] -= rem_i
                cand[j] -= rem_j
                tot = rem_i + rem_j
                tgt = rng.integers(0, n)
                while tgt == i or tgt == j:
                    tgt = rng.integers(0, n)
                cand[tgt] += tot

        else:                              # smooth two neighbours
            i = rng.integers(0, n - 1)
            avg = (cand[i] + cand[i + 1]) * 0.5
            cand[i] = avg
            cand[i + 1] = avg

        # enforce bounds and (re)normalise ℓ¹‑norm
        np.clip(cand, 0.0, 1000.0, out=cand)
        s = cand.sum()
        if s > 0.0:
            cand *= (sum_a / s)

        cand_score, cand_conv, _ = _fast_score(cand, m)
        if cand_score < best_score:
            best, best_score, best_conv = cand, cand_score, cand_conv

    # ------------------------------------------------------------------
    # Return a plain Python list – the evaluator will recompute the exact score.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
