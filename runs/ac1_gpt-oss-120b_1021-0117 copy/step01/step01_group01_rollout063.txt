def propose_candidate(seed=42, budget_s=1100,...):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., float(budget_s) - 1.)

    # Determine n
    try:
        n = int(len(height_sequence_1))
        # Use height_sequence_1 as fallback maybe not needed.
    except:
        n=256

    conv_len = 2*n-1
    fft_len = 1 << ((conv_len-1).bit_length()) # nearest power of two >= conv_len

    def conv_max(a):
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return float(conv.max())

    def score(a):
        s = a.sum()
        if s == 0:
            return math.inf
        maxb = conv_max(a)
        return float(2 * n * maxb / (s * s))

    # 1. Generate candidates: binomial, power-laws, gaussian, etc.
    candidates = []

    # binomial:
    a_bin = np.empty(n, dtype=np.float64)
    a_bin[0] = 1.0
    for i in range(n-1):
        a_bin[i+1] = a_bin[i] * (2*i + 1) / (2*i + 2)
    a_bin = a_bin / a_bin.sum()
    candidates.append(a_bin)

    # power law alpha values
    alphas = np.linspace(0.1, 0.9, 9) # maybe 0.1 to 0.9 step 0.1
    i_arr = np.arange(n, dtype=np.float64) + 1.0
    for alpha in alphas:
        a = 1.0 / (i_arr ** alpha)
        a = a / a.sum()
        candidates.append(a)

    # Gaussian centered
    # sigma choose a fraction of n; we try few sigma values:
    center = (n - 1) / 2.0
    for sigma_frac in [0.2, 0.25, 0.33, 0.4]:
        sigma = sigma_frac * n
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        a = a / a.sum()
        candidates.append(a)

    # Raised cosine (Hann) window:
    x = np.arange(n)
    a = 0.5 * (1 - np.cos(2 * np.pi * (x+1) / (n+1))) # Hann window positive
    a = a / a.sum()
    candidates.append(a)

    # Evaluate all and keep best
    best = None
    best_score = math.inf
    for cand in candidates:
        s = score(cand)
        if s < best_score:
            best_score = s
            best = cand.copy()
    # Keep best

    # 2. Gradient descent refinement:
    cur = best.copy()
    cur_score = best_score

    # parameters
    eta = 0.02  # step size
    # track iteration
    iteration = 0

    while time.monotonic() < deadline:
        # compute convolution, max position
        fa = np.fft.rfft(cur, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_b = float(conv.max())
        k_max = int(np.argmax(conv))

        # compute gradient g
        g = np.zeros(n, dtype=np.float64)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        # i_range inclusive
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k_max - i_range
        g[i_range] = 2.0 * cur[j_range]

        # descent direction
        d = -g
        # adjust to zero sum:
        d += (g.sum() / n) * np.ones(n)  # because sum(-g) = -sum(g); we want sum(d)=0 => add sum(g)/n

        # optionally scale step adaptively
        # try a few step scales
        accepted = False
        for scale in (1.0, 0.5, 0.25, 0.125):
            step = eta * scale
            new = cur + step * d
            # enforce nonnegativity
            new = np.maximum(new, 0.0)
            # renorm sum to 1
            s_new = new.sum()
            if s_new <= 0:
                continue
            new = new / s_new
            new_score = score(new)
            if new_score < cur_score:
                cur = new
                cur_score = new_score
                if new_score < best_score:
                    best = new.copy()
                    best_score = new_score
                accepted = True
                break
        if not accepted:
            # optionally reduce eta
            eta *= 0.5
            if eta < 1e-6:
                break
        else:
            # maybe increase eta slightly
            eta = min(eta * 1.1, 0.1)

        iteration += 1
    # final best candidate as list
    final = [float(x) for x in best]
    # make sure evaluator accepts
    try:
        _ = evaluate_sequence(final)
    except Exception:
        # fallback to binomial normalized
        final = [float(x) for x in a_bin]
    return final

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., float(budget_s) - 1.0)

        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return float(conv.max())

    def score_from_max(max_b):
        # sum a is 1 after normalization
        return float(2 * n * max_b)

    best = None
    best_score = math.inf

    # Helper to update candidate
    def consider(cand):
        nonlocal best, best_score
        maxb = conv_max(cand)
        sc = score_from_max(maxb)
        if sc < best_score:
            best_score = sc
            best = cand.copy()

    # 1. Binomial sequence
    a_bin = np.empty(n, dtype=np.float64)
    a_bin[0] = 1.0
    for i in range(n - 1):
        a_bin[i + 1] = a_bin[i] * (2 * i + 1) / (2 * i + 2)
    a_bin = a_bin / a_bin.sum()
    consider(a_bin)

    # 2. Power‑law exponents search
    i_arr = np.arange(1, n + 1, dtype=np.float64)

    # coarse grid
    alphas = np.linspace(0.4, 0.8, 81)   # step 0.005
    for a in alphas:
        if time.monotonic() >= deadline:
            break

    # refine around best alpha if we have stored it? We didn't store alpha. We can just try more fine grid

            break
        seq = np.power(i_arr, -a)

            new[j] += delta
            maxb = _conv_max(new)
            new_score = _score_from_max(maxb)
            if new_score < cur_score:
                cur = new
                cur_score = new_score
                if new_score < best_score:
                    best_seq = new.copy()
                    best_score = new_score

    # ------------------------------------------------------------------
    # 5. Final fallback (should never be needed)
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64) / n

    final = [float(v) for v in best_seq]
    try:
        _ = evaluate_sequence(final)   # sanity check
    except Exception:
        final = [float(v) for v in a_bin]   # fallback to the binomial construction
    return final
```
