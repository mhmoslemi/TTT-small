def project_simplex(v):
    # ensure sum to 1, nonnegative
    # algorithm: sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    lambda_ = cssv[rho] / (rho + 1)
    w = np.maximum(v - lambda_, 0)
    return w

def _project_simplex(v):
    # returns projection onto simplex {x >=0, sum x = 1}
    # based on Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all entries zero; return uniform?
        return np.full_like(v, 1.0/n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    # test catalan

        best_score = cand_score

    # parameters

    min_step = 1e-6
    max_step = 0.5

    # Option to compute factor for gradient step scaling: approx normalized?

    while time.monotonic() < deadline:

        conv = np.fft.irfft(fa * fa, n=m)[:conv_len]
        max_idx = int(np.argmax(conv))
        # Compute gradient for max idx

            cand = _project_simplex(cand)

            conv_j = np.fft.irfft(fa_j * fa_j, n=m)[:conv_len]

import time
import numpy as np

def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style central binomial coefficients w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _project_nonneg_normalise(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum‑one (simple simplex projection)."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s <= 0.0:
        # fallback to a uniform vector if everything vanished
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def _conv_max_and_index(a: np.ndarray):
    """Return (maximum value, index) of the linear convolution a*a using FFT."""
    n = a.size
    conv_len = 2 * n - 1
    m = 1 << (conv_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:conv_len]
    idx = int(conv.argmax())
    return float(conv[idx]), idx
