def max_convolution_and_grad(a):
    conv = np.convolve(a, a)
    max_val = conv.max()
    max_idx = int(np.argmax(conv))
    # gradient: for each i compute 2*a[max_idx - i] if in range

    grad[lo:hi+1] = 2.0 * a[max_idx - np.arange(lo, hi+1)]

def simplex_projection(v):
    # v is numpy array, may contain negative values
    # returns projection onto {x >= 0, sum x = 1}
    # standard algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero, return uniform simplex point
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _full_conv_fft(a: np.ndarray) -> np.ndarray:

    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[:m]

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = None

    if parent is not None:
        n = parent.size
        a = np.clip(parent, 0.0, None)
        if a.sum() <= 0.0:
            a = np.ones(n, dtype=np.float64)
    else:
        # modest fallback length – the original code used 256
        n = 256
        a = np.ones(n, dtype=np.float64)

    # normalise to the simplex (sum = 1)
    a = a / a.sum()
    best = a.copy()
    best_score = 2.0 * n * np.max(_full_conv_fft(best))   # sum = 1 ⇒ denominator = 1

    # also keep a uniform baseline (sometimes useful)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = 2.0 * n * np.max(_full_conv_fft(uniform))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score

    # ------------------------------------------------------------------
    # 2.  Parameters for the simple sub‑gradient descent / stochastic mix
    # ------------------------------------------------------------------
    step_frac = 0.12                 # relative step size
    min_step = 1e-6
    max_step = 5.0
    no_improve = 0
    max_no_improve = 200            # trigger extra jitter after stagnation

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- a) sub‑gradient step based on the worst convolution entry -----
        conv = _full_conv_fft(best)
        max_idx = int(np.argmax(conv))
        # gradient of the active entry (2*a[mirror] where it exists)
        grad = np.zeros_like(best)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1, dtype=int)
        if idx.size:
            grad[idx] = 2.0 * best[max_idx - idx]

        max_grad = grad.max()
        # protect against division by zero (should not happen)
        if max_grad <= 0.0:
            max_grad = 1.0

        step = step_frac * (best.max() / max_grad)
        cand = best - step * grad
        cand = _simplex_projection(cand)

        # evaluate
        cand_score = 2.0 * n * np.max(_full_conv_fft(cand))
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            step_frac = min(step_frac * 1.2, max_step)
            no_improve = 0
            if best_score <= 1.50274349:
                break
        else:
            step_frac = max(step_frac * 0.7, min_step)
            no_improve += 1

        # ----- b) occasional random mass transfer (exploration) -----
        if rng.random() < 0.25:
            i, j = rng.choice(n, size=2, replace=False)
            if best[i] > 0.0:
                eps = rng.random() * best[i] * 0.4
                cand2 = best.copy()
                cand2[i] -= eps
                cand2[j] += eps
                cand2 = _simplex_projection(cand2)
                cand2_score = 2.0 * n * np.max(_full_conv_fft(cand2))
                if cand2_score < best_score:
                    best = cand2
                    best_score = cand2_score
                    step_frac = min(step_frac * 1.1, max_step)
                    no_improve = 0
                    if best_score <= 1.50274349:
                        break

        # ----- c) global jitter when stagnating -----
        if no_improve >= max_no_improve or rng.random() < 0.015:
            jitter = np.exp(rng.normal(0.0, 0.018, size=n))
            cand3 = best * jitter
            cand3 = _simplex_projection(cand3)
            cand3_score = 2.0 * n * np.max(_full_conv_fft(cand3))
            if cand3_score < best_score:
                best = cand3
                best_score = cand3_score
                step_frac = max(step_frac * 0.9, min_step)
                no_improve = 0
                if best_score <= 1.50274349:
                    break
            else:
                no_improve = 0   # reset stagnation counter

    # ------------------------------------------------------------------
    # 4.  Return the best sequence found as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best]
