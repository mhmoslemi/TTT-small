    a = a / total

    # Initial candidates
    best_a = a.copy()
    best_conv = np.convolve(best_a, best_a)
    best_score = 2.0 * n * best_conv.max()

    # Uniform fallback

    uniform_score = 2.0 * n * uniform_conv.max()
    if uniform_score < best_score:
        best_a = uniform
        best_conv = uniform_conv
        best_score = uniform_score

    # Parameters
    step_factor = 0.5
    min_step = 1e-8
    iteration = 0
    recompute_interval = 3000

    while time.monotonic() < deadline:
        # Periodically recompute convolution from scratch to avoid drift
        if iteration % recompute_interval == 0:
            best_conv = np.convolve(best_a, best_a)
            best_score = 2.0 * n * best_conv.max()

        conv = best_conv
        a = best_a

        # Locate maximum of convolution
        k = int(conv.argmax())

        # Build partner contributions for index k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # partner[i] = a[k-i] where defined
        partner = np.zeros(n, dtype=np.float64)

        partner[idx] = a[k - idx]

        src = int(partner.argmax())

        min_val = partner.min()
        dest_candidates = np.where(partner == min_val)[0]

        if src == dest:

                best_score = cand_score

import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step‑function that lowers the AC1 bound.

    The algorithm starts from the provided `height_sequence_1` (if any) or a
    uniform fallback, then repeatedly moves a fraction of mass from an index
    that contributes most to the current convolution maximum to an index that
    contributes the least.  Convolutions are updated incrementally (O(n)) so
    many iterations fit into the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # --- initialise sequence -------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        a = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 5000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total < 0.01:
        a = np.full(n, 1.0 / n, dtype=np.float64)
        total = a.sum()
    a = a / total                     # normalise (scale does not affect score)

    # --- baseline candidates --------------------------------------------------
    best_a = a.copy()
    best_conv = np.convolve(best_a, best_a)
    best_score = 2.0 * n * best_conv.max()   # sum == 1 → denominator = 1

    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_conv = np.convolve(uniform, uniform)
    uniform_score = 2.0 * n * uniform_conv.max()
    if uniform_score < best_score:
        best_a = uniform
        best_conv = uniform_conv
        best_score = uniform_score

    # binomial (C(2i,i)/4^i) series
    binom = np.empty(n, dtype=np.float64)
    binom[0] = 1.0
    for i in range(1, n):
        binom[i] = binom[i - 1] * (1.0 - 1.0 / (2.0 * i))
    binom = np.clip(binom, 0.0, 1000.0)
    binom = binom / binom.sum()
    binom_conv = np.convolve(binom, binom)
    binom_score = 2.0 * n * binom_conv.max()
    if binom_score < best_score:
        best_a = binom
        best_conv = binom_conv
        best_score = binom_score

    # --- parameters -----------------------------------------------------------
    step_factor = 0.6      # start with a relatively aggressive move size
    min_step = 1e-9
    iteration = 0
    recompute_interval = 4000   # re‑compute full convolution occasionally

    # ------------------------------------------------------------------------
    while time.monotonic() < deadline:
        # occasional full recompute to keep numerical drift in check
        if iteration % recompute_interval == 0:
            best_conv = np.convolve(best_a, best_a)
            best_score = 2.0 * n * best_conv.max()

        conv = best_conv
        a = best_a

        # locate the current convolution maximum
        k = int(conv.argmax())

        # partner[i] = a[k-i] if defined, else 0
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        partner = np.zeros(n, dtype=np.float64)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner[idx] = a[k - idx]

        src = int(partner.argmax())

        # destination: any index with minimal partner (often zero)
        min_val = partner.min()
        dest_candidates = np.where(partner == min_val)[0]
        if dest_candidates.size == 0:
            dest = int(rng.integers(0, n))
        else:
            dest = int(rng.choice(dest_candidates))

        # If source and destination coincide, perform a random “shake”
        if src == dest:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            cand = best_a.copy()
            cand[start:stop] *= np.exp(rng.normal(0.0, 0.12))
            cand = np.clip(cand, 0.0, 1000.0)
            cand /= cand.sum()
            cand_conv = np.convolve(cand, cand)
            cand_score = 2.0 * n * cand_conv.max()
            if cand_score < best_score:
                best_a = cand
                best_conv = cand_conv
                best_score = cand_score
            iteration += 1
            continue

        # try moving either the whole mass or a fraction of it
        fractions = (1.0, step_factor, step_factor * 0.5)
        improved = False
        for frac in fractions:
            delta = best_a[src] * frac
            if delta < min_step:
                continue
            if delta > best_a[src]:
                delta = best_a[src]

            # candidate sequence
            cand = best_a.copy()
            cand[src] -= delta
            cand[dest] += delta

            # incremental update of the convolution
            new_conv = best_conv.copy()
            # subtract src contribution
            new_conv[src:src + n] -= 2.0 * delta * best_a
            # add dest contribution
            new_conv[dest:dest + n] += 2.0 * delta * best_a
            # e*e term
            d2 = delta * delta
            new_conv[2 * src] += d2
            new_conv[2 * dest] += d2
            new_conv[src + dest] -= 2.0 * d2

            cand_score = 2.0 * n * new_conv.max()
            if cand_score < best_score:
                best_a = cand
                best_conv = new_conv
                best_score = cand_score
                improved = True
                step_factor = max(0.02, step_factor * 0.998)  # fine‑tune
                break   # accept the first improving move

        if not improved:
            # occasional random perturbation to escape plateaus
            if rng.random() < 0.001:
                width = int(rng.integers(max(1, n // 300), max(2, n // 50)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                stop = start + width
                cand = best_a.copy()
                cand[start:stop] *= np.exp(rng.normal(0.0, 0.15))
                cand = np.clip(cand, 0.0, 1000.0)
                cand /= cand.sum()
                cand_conv = np.convolve(cand, cand)
                cand_score = 2.0 * n * cand_conv.max()
                if cand_score < best_score:
                    best_a = cand
                    best_conv = cand_conv
                    best_score = cand_score

        iteration += 1

    # --- final clean‑up -------------------------------------------------------
    final = np.clip(best_a, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
