    # maybe also ensure each entry <=1000, they likely already less.

    # Predefine FFT length
    conv_len = 2*n-1
    fft_len = 1 << ((conv_len - 1).bit_length)

    # function to compute conv full via FFT
    def conv_full(x):
        fx = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
        return conv

    # fast scoring function:
    def score_of(x):
        conv = conv_full(x)
        M = conv.max()
        s = x.sum()
        return 2 * n * M / (s * s)

    # projecting onto simplex with tolerance to keep sum=1
    def project_simplex(v):
        # algorithm from https://doi.org/10.1007/11866555_16
        # Works for nonnegative sum=1
        # v arbitrary real
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.where(u + (1 - cssv) / (np.arange(len(v)) + 1) > 0)[0][-1]
        theta = (cssv[rho] - 1) / (rho + 1)
        w = np.maximum(v - theta, 0)
        return w

    # iteration loop
    it = 0
    while time.monotonic() < deadline:
        # compute convolution and find max index
        conv = conv_full(a)
        k_max = int(np.argmax(conv))
        M = conv[k_max]
        # compute gradient g = shift contributions
        g = np.zeros_like(a)
        # for each i where i-k_max in range:
        #   g_i += a[i - k_max]
        #   g_i += a[i + k_max]   (if i+k_max < n)
        if k_max <= n-1:
            # contributions from left shift
            g[k_max:] += a[:n - k_max]   # g_i for i >= k_max
        if k_max > 0:
            # contributions from right shift
            g[:n - k_max] += a[k_max:]   # i from 0 to n-k_max-1
        # gradient of M w.r.t a is g (since derivative of sum)

    # safety evaluate

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # get n
    try:
        n = int(len(height_sequence_1))
        a = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64) / n

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / total

    # FFT parameters
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length() + 0)  # next power of 2

    def max_conv(x):
        fx = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
        return float(conv.max())

    def fast_score(x):
        return 2.0 * n * max_conv(x)  # since sum(x)=1

    cur_score = fast_score(a)
    best = a.copy()
    best_score = cur_score

    # simulated annealing parameters
    max_iter = 200000  # may be limited by time
    init_temp = 0.015   # scale same units as score (score~1.57)
    min_temp = 1e-5
    init_eps = 0.2
    min_eps = 0.001

    it = 0
    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        if it >= max_iter:
            break
        # temperature and epsilon schedule
        elapsed = now - start
        progress = elapsed / (deadline - start)
        temp = init_temp * (1 - progress) + min_temp * progress
        eps = init_eps * (1 - progress) + min_eps * progress

        # choose source i with largish value
        i = rng.integers(0, n)
        if a[i] < 1e-14:
            continue
        # pick target j different
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # transfer
        delta = a[i] * eps
        a[i] -= delta
        a[j] += delta

        new_score = fast_score(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # metropolis acceptance
            if temp > 0:
                prob = math.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta

        it += 1

    # fallback to initial if no improvement
    final = [float(x) for x in best]
    # ensure valid for evaluator
    try:
        _ = evaluate_sequence(final)
    except Exception:
        final = [float(x) for x in a]  # should be safe
    return final

def propose_candidate(...):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.5

    # init a from height_sequence_1 if present else binomial
    try:
        a = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 256
        a = _generate_binomial_seq(n).astype(np.float64)

    # make nonnegative and normalize
    a = np.maximum(a, 0)
    total = a.sum()
    if total == 0:
        a = np.ones_like(a) / a.size
    else:
        a = a / total

    n = a.size
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    def fast_score(x):
        fx = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
        return 2.0 * n * float(conv.max())

    cur_score = fast_score(a)
    best = a.copy()
    best_score = cur_score

    # parameters
    max_iter = 1000000 # large
    init_temp = 0.01
    min_temp = 1e-6
    init_eps = 0.2
    min_eps = 1e-4

    for it in range(max_iter):
        now = time.monotonic()
        if now >= deadline:
            break

        # schedule
        progress = (now - start) / (deadline - start)
        temp = init_temp * (1 - progress) + min_temp * progress
        eps = init_eps * (1 - progress) + min_eps * progress

        # choose source from first half weighted by a
        src = rng.integers(0, n//2)
        if a[src] < 1e-12:
            continue
        # target from second half (maybe uniform)
        tgt = rng.integers(n//2, n)
        # ensure distinct (src != tgt automatically)
        delta = a[src] * eps
        a[src] -= delta
        a[tgt] += delta

        new_score = fast_score(a)

        if new_score < cur_score:
            accept = True
        else:
            # metropolis
            if temp > 0:
                accept = rng.random() < math.exp(-(new_score - cur_score) / temp)
            else:
                accept = False

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            # revert
            a[src] += delta
            a[tgt] -= delta

    final = [float(x) for x in best]
    _ = evaluate_sequence(final)  # trigger any hidden checks
    return final

import time
import math
import numpy as np

def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 using the recurrence a_{i+1}=a_i*(2i+1)/(2i+2)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 upper bound
    (as computed by evaluate_sequence) is as low as possible within the time
    budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)   # leave a margin

    # ------------------------------------------------------------------
    # 1. Initialise the sequence (use the supplied parent if present)
    # ------------------------------------------------------------------
    try:
        a = np.array(height_sequence_1, dtype=np.float64)   # parent provided
    except Exception:
        # fallback: binomial coefficients of moderate length
        n_fallback = 256
        a = _generate_binomial_seq(n_fallback)

    a = np.clip(a, 0.0, 1000.0)           # safety limits required by the evaluator
    s = a.sum()
    if s == 0.0:
        a = np.ones_like(a) / a.size
    else:
        a = a / s                         # normalise sum to exactly 1.0

    n = a.size
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two

    def _fast_score(x: np.ndarray) -> float:
        """Rapid proxy for evaluate_sequence (sum(x) == 1)."""
        fx = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
        return 2.0 * n * float(conv.max())

    cur_score = _fast_score(a)
    best = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # 2. Stochastic simulated‑annealing search
    # ------------------------------------------------------------------
    # parameters (tuned for the typical size n≈6700)
    max_iter = 800_000
    init_temp = 0.015          # Metropolis temperature at start
    min_temp = 1e-6            # final temperature
    init_eps = 0.20            # maximum fraction of source mass transferred early
    min_eps = 1e-4             # final fraction

    # restrict source to the first half (large entries) and target to the second half
    half = n // 2

    it = 0
    while it < max_iter:
        now = time.monotonic()
        if now >= deadline:
            break

        # linear schedules for temperature and step size
        prog = (now - start) / (deadline - start)
        temp = init_temp * (1.0 - prog) + min_temp * prog
        eps = init_eps * (1.0 - prog) + min_eps * prog

        # pick a source index that currently holds a noticeable amount
        src = rng.integers(0, half)
        if a[src] < 1e-14:
            it += 1
            continue

        # pick a target index far from the source (second half of the vector)
        tgt = rng.integers(half, n)
        if tgt == src:
            it += 1
            continue

        delta = a[src] * eps                # amount to relocate
        a[src] -= delta
        a[tgt] += delta

        new_score = _fast_score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if temp > 0.0:
                # probability of accepting a worsening move
                prob = math.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            # revert the trial move
            a[src] += delta
            a[tgt] -= delta

        it += 1

    # ------------------------------------------------------------------
    # 3. Return the best sequence found (authoritative check)
    # ------------------------------------------------------------------
    final = [float(v) for v in best]
    try:
        # trigger any hidden checks in the supplied evaluator
        _ = evaluate_sequence(final)
    except Exception:
        # safety fallback – return the current best even if the check fails
        final = [float(v) for v in a]
    return final
