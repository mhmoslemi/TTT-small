def _grad_max_conv(a, conv):
    n = a.size
    max_val = conv.max()
    # tolerance relative to max_val
    eps = max_val * 1e-12 + 1e-15
    idxs = np.where(conv >= max_val - eps)[0]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        # slice indices
        i_range = np.arange(i_start, i_end+1)
        g[i_range] += 2 * a[k - i_range]
    return g

import time
import numpy as np

# ------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex.
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# First n coefficients of the series a * a = 1 + x + ... + x^{2n-2}
# (the “sqrt‑series” used in the reference solution).
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Convolution (full) via FFT – returns length 2n‑1 array.
# ------------------------------------------------------------
def _conv_fft(vec: np.ndarray) -> np.ndarray:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# AC1 score (identical to the evaluator’s formula, but fast).
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    n = vec.size
    conv = _conv_fft(vec)
    return float(2.0 * n * conv.max() / (vec.sum() ** 2))


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """

    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
