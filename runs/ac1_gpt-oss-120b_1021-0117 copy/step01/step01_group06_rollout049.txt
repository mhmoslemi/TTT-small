def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        # Normalize to sum=1 (scale invariance)
        total = a.sum()
        if total == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / total
    else:
        # fallback: sqrt series
        n = 256
        a = _sqrt_series(n)
        a = a / a.sum()
    n = a.size
    best_vec = a.copy()
    best_sc = _score(best_vec)
    # parameters
    step0 = 0.5
    iter_cnt = 0
    no_imp = 0
    # main loop
    while time.monotonic() < deadline:
        # compute convolution, gradient
        L = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_val = conv.max()
        # find active indices (with tolerance)
        tol = 1e-12 * max_val
        active = np.where(conv >= max_val - tol)[0]
        # compute subgradient
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # slice: a[k - hi : k - lo + 1] reversed
            seg = a[k - hi : k - lo + 1][::-1]  # length hi-lo+1
            grad[lo:hi+1] += 2.0 * seg
        # step size
        eps = step0 / np.sqrt(iter_cnt + 1.0)
        cand = a - eps * grad
        # projection onto simplex
        cand = _project_simplex(cand)
        # evaluate
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()
            no_imp = 0
        else:
            no_imp += 1
        # accept if improvement or with some probability (maybe always keep best)
        if sc < best_sc * (1 + 1e-12):  # small tolerance
            a = cand
        else:
            # optionally random perturbation to avoid stagnation
            if no_imp > 2000:
                # random restart
                a = rng.dirichlet(np.ones(n))
                no_imp = 0
            else:
                # keep old a
                a = a
        iter_cnt += 1

    # return best vector as list
    return [float(x) for x in best_vec]

    """Project v onto the simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic sqrt‑series (a*a = 1+…+x^{2n-2}) – fallback initializer
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Fast AC1 bound (exactly the evaluator’s formula)
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (vec.sum() ** 2)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty sequence of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # -------------------- initialise --------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total <= 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / total
    else:
        nfallback = 256
        a = _sqrt_series(nfallback)
        a = a / a.sum()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # -------------------- best‑so‑far --------------------
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    best_val = conv.max()
    best_sc = 2.0 * n * best_val / (a.sum() ** 2)
    best_vec = a.copy()

    # -------------------- subgradient descent ------------
    step0 = 0.5                      # initial step magnitude
    it = 0
    stagnation = 0

    while time.monotonic() < deadline:
        # current convolution & active set
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_val = conv.max()
        tol = 1e-12 * max_val
        active = np.where(conv >= max_val - tol)[0]

        # subgradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        for k in active:
            lo = k - (n - 1)
            if lo < 0:
                lo = 0
            hi = k
            if hi > n - 1:
                hi = n - 1
            # add 2 * a[k-i] for i in [lo,hi]
            seg = a[k - hi : k - lo + 1][::-1]      # reversed slice
            grad[lo : hi + 1] += 2.0 * seg

        # projected sub‑gradient step
        eps = step0 / np.sqrt(it + 1.0)
        cand = a - eps * grad
        cand = _project_simplex(cand)

        # evaluate candidate
        fc = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(fc * fc, n=L)[: 2 * n - 1]
        max_c = conv_c.max()
        sc_c = 2.0 * n * max_c / (cand.sum() ** 2)

        # improvement?
        if sc_c < best_sc:
            best_sc = sc_c
            best_vec = cand.copy()
            a = cand
            stagnation = 0
        else:
            stagnation += 1
            # occasional random restart to avoid pathological stalls
            if stagnation >= 5000:
                a = rng.dirichlet(np.ones(n))
                stagnation = 0

        it += 1

    # --------------------------------------------------------
    # return the best candidate found (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
