def simplex_proj(v):
    # Project onto simplex sum=1, v>=0 automatically after projection.
    # Input v (array). Output x.
    # From Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

# ---------- helper functions (top level) ---------------------------------

def full_convolution(a: np.ndarray, n: int, n_fft: int) -> np.ndarray:
    """Return the full linear convolution a*a (length 2n-1) using FFT."""
    af = np.fft.rfft(a, n_fft)
    conv = np.fft.irfft(af * af, n_fft)
    return conv[: 2 * n - 1]


def max_conv_and_index(a: np.ndarray, n: int, n_fft: int):
    """Compute max value of a*a and its index (full convolution)."""
    conv = full_convolution(a, n, n_fft)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project vector v onto the probability simplex {x >= 0, sum x = 1}.
    Duchi et al. (2008) algorithm.
    """
    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, len(v) + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
