import time
import numpy as np

# ---------- helper utilities -------------------------------------------------
def _catalan_weights(n: int) -> np.ndarray:
    """Central‑binomial (Catalan‑style) weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x≥0, sum=1}. """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                         # all non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: re‑normalise exactly
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


def _active_gradient(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max‑convolution at shift k (single active index)."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad


def _gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """One projected‑gradient descent step on the AC1 objective."""
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))
    grad = _active_gradient(a, k)
    cand = a - step * grad
    cand = np.clip(cand, 0.0, None)
    cand = _project_to_simplex(cand)
    return cand


def _random_transfer(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Mass‑conserving random transfer between two distinct indices."""
    n = a.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    if a[i] <= 0.0:
        return a
    delta = rng.random() * 0.5 * a[i]          # up to half of a[i]
    cand = a.copy()
    cand[i] -= delta
    cand[j] += delta
    # projection keeps the sum exactly 1
    cand = _project_to_simplex(cand)
    return cand
