    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2*a.size - 1]

def evaluate_fast(a, fft_len, n):
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]   # 2*n-1 length
    maxb = np.max(conv)
    s = np.sum(a)
    return maxb, s, maxb * 2*n / (s*s) (score)

def _full_convolution_max(a: np.ndarray, fft_len: int) -> tuple[float, np.ndarray]:
    """Return (max value, full convolution) of a with itself using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max()), conv

def _score_from_max(n: int, max_conv: float, total: float) -> float:
    """Exact score used by the evaluator."""
    return 2.0 * n * max_conv / (total * total)
