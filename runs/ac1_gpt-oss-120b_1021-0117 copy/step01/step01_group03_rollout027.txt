import numpy as np
import time

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (next power‑of‑2 length)."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # smallest power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the authoritative evaluator."""
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i   (decreasing, non‑negative)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Find a non‑negative step function with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # Initial sequence – parent if supplied, otherwise the classic series.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = parent.size
    else:
        n = 5582
        parent = None

    base = _sqrt_series(n)
    best = base.copy()
    best_score = _score(best)

    if parent is not None:
        pscore = _score(parent)
        if np.isfinite(pscore) and pscore < best_score:
            best = parent.copy()
            best_score = pscore

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    early_cut = n // 8                # region “early” (source of mass)
    late_start = n - early_cut        # region “late”  (sink of mass)

    # temperature for simulated‑annealing‑style acceptance
    total_time = deadline - time.monotonic()
    def temperature():
        # linear cooling from 1 to ~0
        remaining = deadline - time.monotonic()
        return max(0.0, remaining / total_time)

    # ------------------------------------------------------------------
    # Main loop – mixed moves with a bias toward draining the early part.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = best.copy()
        op = rng.random()

        # ---- targeted mass transfer (dominant move) --------------------
        if op < 0.65:
            # perform 1–3 small transfers in one iteration
            for _ in range(rng.integers(1, 4)):
                src = rng.integers(0, early_cut)
                if cand[src] <= 1e-12:
                    continue
                dst = rng.integers(late_start, n)
                cap = 1000.0 - cand[dst]
                if cap <= 0.0:
                    continue
                max_transfer = min(cand[src], cap)
                # early iterations allow larger steps; shrink with temperature
                delta = max_transfer * rng.random() * (0.05 + 0.30 * temperature())
                cand[src] -= delta
                cand[dst] += delta

        # ---- blockwise scaling (exploratory) -------------------------
        elif op < 0.85:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = rng.integers(0, max(1, n - width + 1))
            stop = start + width
            # shrink early blocks, enlarge late ones – guided by temperature
            if start < n // 2:
                mean = -0.06 * temperature()
            else:
                mean = 0.06 * temperature()
            factor = float(np.exp(rng.normal(mean, 0.08)))
            cand[start:stop] *= factor

            best, best_score = cand, sc
        else:
            # uphill moves accepted with probability proportional to temperature
            if rng.random() < 0.001 * temperature():
                best, best_score = cand, sc

    # ------------------------------------------------------------------
    # Return the best candidate found.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
