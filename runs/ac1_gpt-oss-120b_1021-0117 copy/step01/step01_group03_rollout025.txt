def _conv_full(x, y):
    n = x.size
    m = y.size
    size = n + m - 1
    N = 1 << ((size - 1).bit_length())
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:size]
    return conv

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time

    # find n
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = parent.size
    else:
        # fallback length (maybe typical length)
        n = 5582
        parent = None

    # base sequences
    # sqrt series (maybe not needed)
    def _sqrt_series(n):
        a = np.empty(n, dtype=np.float64)
        a[0]=1.0
        for i in range(1,n):
            a[i] = a[i-1] * (2.0*i -1.0)/(2.0*i)  # as earlier but correct formula
        return a

    def _conv_self(a):
        n = a.size
        m = 1 << ((2*n-1).bit_length())
        fa = np.fft.rfft(a, n=m)
        prod = fa * fa
        conv = np.fft.irfft(prod, n=m)[:2*n-1]
        return conv

    def _score(a):
        n = a.size
        b = _conv_self(a)
        max_b = b.max()
        s = a.sum()
        if s < 0.01:
            return np.inf
        return 2.0 * n * max_b / (s * s)

    # compute sqrt series candidate
    sqrt_seq = _sqrt_series(n)

    # exponential candidate
    c_const = 3.0  # constant to adjust
    r = 1.0 - c_const / n
    if r <= 0.0:
        r = 0.0
    expo_seq = np.power(r, np.arange(n, dtype=np.float64))

    # collect candidates
    candidates = []
    candidates.append(sqrt_seq)
    if parent is not None:
        candidates.append(parent)
    candidates.append(expo_seq)

    best = None
    best_score = np.inf
    for cand in candidates:
        # range check
        cand = np.clip(cand, 0.0, 1000.0)  # ensure bounds
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = cand

    # (optional) could also perform a few simple gradient steps on best to maybe improve further; but not needed.

    return [float(v) for v in best]

import numpy as np

def _conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using real FFT (O(n log n))."""
    n = a.size + b.size - 1
    m = 1 << ((n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    fb = np.fft.rfft(b, n=m)
    conv = np.fft.irfft(fa * fb, n=m)[:n]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the authoritative evaluator."""
    n = a.size
    b = _conv_full(a, a)          # full self‑convolution
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a
