    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    try:
        n = len(height_sequence_1)
        a = height_sequence_1.astype(np.float64).copy()
    except NameError:
        n = 5000
        a = np.ones(n, dtype=np.float64) / n
    # Normalize to sum 1
    total = a.sum()
    if total <= 0: 
        a = np.full(n, 1.0/n)
        total = 1.0
    a = a / total
    # Clip to 0..1000 (ensured by non-neg)
    a = np.clip(a, 0.0, 1000.0)

    conv = np.convolve(a, a)
    cur_max = conv.max()
    best_a = a.copy()
    best_conv = conv.copy()
    best_score = 2.0 * n * cur_max / (a.sum()**2)

    step_factor = 0.3
    min_step = 1e-8
    iter_count = 0
    while time.monotonic() < deadline:
        # find src/dest
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end+1)
        partner = a[k - idx]  # partner values
        prod = a[idx] * partner
        src = idx[np.argmax(prod)]
        # dest candidates with minimal partner
        # Build full partner array for dest selection
        partner_full = np.zeros(n, dtype=np.float64)
        partner_full[idx] = partner
        min_val = partner_full.min()
        dest_candidates = np.where(partner_full == min_val)[0]
        if dest_candidates.size == 0:
            dest = int(rng.integers(0, n))
        else:
            dest = int(rng.choice(dest_candidates))
        if src == dest:
            # random perturb and continue
            width = int(rng.integers(max(1, n//200), max(2, n//12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand = a.copy()
            factor = np.exp(rng.normal(0.0, 0.12, size=stop-start))
            cand[start:stop] *= factor
            # renorm
            cand_sum = cand.sum()
            if cand_sum > 0:
                cand = cand / cand_sum * a.sum()
                cand_conv = np.convolve(cand, cand)
                cand_score = 2.0 * n * cand_conv.max() / (cand.sum()**2)
                if cand_score < best_score:
                    a = cand
                    conv = cand_conv
                    best_a = cand.copy()
                    best_conv = cand_conv.copy()
                    best_score = cand_score
            continue

        delta = a[src] * step_factor
        if delta < min_step:
            delta = min_step
        if delta > a[src]:
            delta = a[src]
        # compute candidate conv incrementally
        cand_conv = conv.copy()
        # add contributions for dest and src
        cand_conv[dest:dest+n] += 2*delta*a
        cand_conv[src:src+n] -= 2*delta*a

        cand_conv[idx2src] += delta*delta
        cand_conv[idx2dest] += delta*delta
        cand_conv[idxsum] -= 2*delta*delta

        cand_max = cand_conv.max()
        cand_score = 2.0 * n * cand_max / (a.sum()**2)   # sum unchanged

        if cand_score < best_score:
            # accept
            a[src] -= delta
            a[dest] += delta
            conv = cand_conv
            best_a = a.copy()
            best_conv = conv.copy()
            best_score = cand_score
            # Possibly decay step_factor slowly

        else:

        n = 5000                       # reasonable default
        parent = None

    # start with a simple uniform candidate (score = 2)
    best_a = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = 2.0
    # ----------------------------------------------------
    # helper to evaluate a candidate (sum is assumed = 1)
    # ----------------------------------------------------
    def _score(arr):
        conv = np.convolve(arr, arr)
        return 2.0 * n * conv.max()          # sum(arr) == 1

    # ----------------------------------------------------
    # grid search over power‑law exponents p where a_i ∝ (i+1)^(-p)
    # ----------------------------------------------------
    # coarse grid
    coarse = np.linspace(0.10, 0.95, 86)      # step ≈0.01
    for p in coarse:
        if time.monotonic() >= deadline:
            break
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = np.power(idx, -p)
        a /= a.sum()                         # normalise to L1 = 1
        s = _score(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()

    # ----------------------------------------------------
    # refine around the best exponent found
    # ----------------------------------------------------
    # extract the exponent that produced best_a (approximate)
    # we recompute it from the last evaluation (simple heuristic)
    # by a quick log‑regression on the tail (not critical)
    # Instead, just explore a neighbourhood of p values.
    # Determine a neighbourhood centre from the coarse grid:
    best_p_est = 0.5                     # fallback centre
    # pick the p from coarse that gave the best score:
    # (re‑evaluate on the grid to know which p it was)
    scores = []
    for p in coarse:
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = np.power(idx, -p)
        a /= a.sum()
        scores.append(_score(a))
    best_idx = int(np.argmin(scores))
    best_p_est = coarse[best_idx]

    # fine grid around the estimate
    fine_low = max(0.10, best_p_est - 0.05)
    fine_high = min(0.95, best_p_est + 0.05)
    fine = np.linspace(fine_low, fine_high, 101)
    for p in fine:
        if time.monotonic() >= deadline:
            break
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = np.power(idx, -p)
        a /= a.sum()
        s = _score(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()

    # ----------------------------------------------------
    # optional tiny random perturbations to escape plateaus
    # ----------------------------------------------------
    while time.monotonic() < deadline:
        # pick a random block, scale it a little, re‑normalise
        width = rng.integers(max(1, n // 400), max(2, n // 80))
        start = rng.integers(0, n - width + 1)
        stop = start + width
        cand = best_a.copy()
        scale = np.exp(rng.normal(0.0, 0.07, size=width))
        cand[start:stop] *= scale
        cand /= cand.sum()
        s = _score(cand)
        if s < best_score:
            best_score = s
            best_a = cand
        # occasional larger shake
        if rng.random() < 0.001:
            width = rng.integers(max(1, n // 200), max(2, n // 30))
            start = rng.integers(0, n - width + 1)
            stop = start + width
            cand = best_a.copy()
            scale = np.exp(rng.normal(0.0, 0.3, size=width))
            cand[start:stop] *= scale
            cand /= cand.sum()
            s = _score(cand)
            if s < best_score:
                best_score = s
                best_a = cand

    # ----------------------------------------------------
    # final sanitisation
    # ----------------------------------------------------
    final = np.clip(best_a, 0.0, 1000.0)
    if final.sum() < 0.01:
        # fallback uniform
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
