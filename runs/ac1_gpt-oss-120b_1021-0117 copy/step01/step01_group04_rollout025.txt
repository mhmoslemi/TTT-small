import time, math
import numpy as np

# ---------- utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound using FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _gradient_move(a: np.ndarray, n: int, L: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a candidate by moving mass away from positions that contribute most
    to the current maximal autocorrelation value.
    """
    # find the index of the maximal convolution value
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    idx_max = int(conv.argmax())
    d = idx_max - (n - 1)               # offset where max occurs

    # overlapping region for this offset
    if d >= 0:
        length = n - d
        a_i = a[:length]
        a_j = a[d:]
    else:
        d_abs = -d
        length = n - d_abs
        a_i = a[d_abs:]
        a_j = a[:length]

    term = a_i * a_j                     # contribution of each pair
    total = term.sum()
    if total <= 0.0:
        # fallback: simple random transfer
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        while i == j:
            j = int(rng.integers(0, n))
        frac = rng.random() * 0.3
        delta = a[i] * frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)
        return cand

    # pick a source index with probability proportional to its contribution
    probs = term / total
    cum = np.cumsum(probs)
    src_off = int(np.searchsorted(cum, rng.random(), side='right'))

    # map to original index
    if d >= 0:
        i = src_off
    else:
        i = src_off + d_abs

    # pick a destination that is not the source nor its paired partner
    j = int(rng.integers(0, n))
    attempts = 0
    while (j == i) or (d >= 0 and j == i + d) or (d < 0 and j == i - d_abs):
        j = int(rng.integers(0, n))
        attempts += 1
        if attempts > 5:
            break

    # move a modest fraction of the source mass
    frac = rng.random() * 0.2  # up to 20 %
    delta = a[i] * frac
    cand = a.copy()
    cand[i] -= delta
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ---------- main search -------------------------------------------------------

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic search for a non‑negative step function that minimizes the AC1
    upper bound. Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)   # safety margin

    # ---- initialise ---------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        base = parent.copy()
    except Exception:
        n = 256
        base = _gen_binomial(n)

    # also keep a binomial fallback (often decent)
    try:
        binom = _gen_binomial(n)
        binom_score = _ac1_score(binom, n, _next_pow2(2 * n - 1))
        base_score = _ac1_score(base, n, _next_pow2(2 * n - 1))
        if binom_score < base_score:
            best = binom.copy()
            best_score = binom_score
        else:
            best = base.copy()
            best_score = base_score
    except Exception:
        best = base.copy()
        best_score = _ac1_score(best, n, _next_pow2(2 * n - 1))

    cur = best.copy()
    cur_score = best_score

    # search parameters
    temperature = 1.0
    decay = 0.9996
    L = _next_pow2(2 * n - 1)
    target = 1.50274349

    # ------------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- choose a move -------------------------------------------------
        move_type = rng.random()
        if move_type < 0.5:
            cand = _gradient_move(cur, n, L, rng)
        elif move_type < 0.85:          # block scaling
            w_low = max(1, n // 250)
            w_high = max(2, n // 12)
            width = int(rng.integers(w_low, w_high + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start:start + width] *= factor
        else:                           # simple mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * 0.4
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        # ----- respect bound constraints ------------------------------------
        np.clip(cand, 0.0, 1000.0, out=cand)

        # keep overall sum (scaling does not affect bound but keeps numbers sane)
        s_cur = cur.sum()
        s_cand = cand.sum()
        if s_cand > 1e-12:
            cand *= (s_cur / s_cand)

        # ----- evaluate ------------------------------------------------------
        cand_score = _ac1_score(cand, n, L)

        # ----- update best ---------------------------------------------------
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < target:
                break

        # ----- accept/reject for the current walk ---------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= decay

        # ----- occasional restart from best ----------------------------------
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # return as plain Python list
    return [float(x) for x in best]
