import time
import numpy as np

# ------------------------------------------------------------
# Project a vector onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_idx = np.where(rho_arr > 0)[0]
    if rho_idx.size == 0:
        theta = 0.0
    else:
        rho = rho_idx[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w.fill(1.0 / n)
    return w


# ------------------------------------------------------------
# Generate the "sqrt series" a such that a * a = 1 + x + … + x^{2n-2}
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # dot product of a[1:k] and reversed a[1:k]
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Fast score computation, identical to the official evaluator
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    max_val = conv.max()
    # sum(vec) is expected to be 1, but keep the formula general
    s = vec.sum()
    if s == 0.0:
        return float("inf")
    return 2.0 * n * max_val / (s * s)
