import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT convolution, score)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (O(n log n))."""
    n = a.size
    # length of full convolution = 2*n-1, next power of two for efficient FFT
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the authoritative evaluator."""
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with low AC1 score."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initial sequence: use supplied parent if present, otherwise the classic
    # sqrt‑series (central binomial coefficients divided by 4^i).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        best = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).copy()
        n = best.size
    else:
        # fallback length – typical hidden length
        n = 5582
        best = np.empty(n, dtype=np.float64)
        best[0] = 1.0
        for i in range(1, n):
            best[i] = best[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)

    best_score = _score(best)
    best_conv = _conv_self(best)  # keep the convolution of the best candidate

    # ------------------------------------------------------------------
    # Search loop – we only accept strictly improving moves (lower score).
    # Three mutation types are mixed:
    #   1) Targeted mass transfer from an index that participates in the
    #      current maximal convolution term to an index that does not.
    #   2) Multiplicative scaling of a random sub‑block.
    #   3) Small additive Gaussian noise.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = best.copy()
        op = rng.random()

        # --------------------------------------------------------------
        # 1) Targeted transfer – tries to shrink the current maximal
        #    convolution entry without raising any other entry above it.
        # --------------------------------------------------------------
        if op < 0.60:
            # locate a k where the convolution attains its maximum
            k_max = int(np.argmax(best_conv))
            # pick a source index that actually contributes to this k
            src = rng.integers(0, min(k_max + 1, n))
            # pick a destination that does NOT influence this k
            if k_max + 1 >= n:
                continue
            dst = rng.integers(k_max + 1, n)

            if cand[src] <= 1e-14:
                continue
            max_transfer = min(cand[src], 1000.0 - cand[dst])
            if max_transfer <= 0.0:
                continue
            # move a modest fraction of the transferable mass
            delta = max_transfer * rng.random() * 0.07  # up to 7 %
            cand[src] -= delta
            cand[dst] += delta

        # --------------------------------------------------------------
        # 2) Block scaling – multiplicative perturbation of a contiguous block.
        # --------------------------------------------------------------
        elif op < 0.80:
            width = int(rng.integers(max(1, n // 250), max(2, n // 15)))
            start = rng.integers(0, max(1, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.09)))  # log‑normal factor
            cand[start:start + width] *= factor

        # --------------------------------------------------------------
        # 3) Tiny additive noise.
        # --------------------------------------------------------------
        else:
            noise = rng.normal(0.0, 0.008, size=n)
            cand += noise

        # enforce the problem bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate – cheap FFT based score
        conv = _conv_self(cand)
        max_val = conv.max()
        # early discard if the maximal entry cannot improve the best score
        # (score = 2*n*max / sum^2). Since sum does not increase, we need
        # max strictly smaller than the best maximal value.
        if max_val >= best_conv.max():
            continue
        sc = float(2.0 * n * max_val / (cand.sum() ** 2))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
            best_conv = conv

    # ------------------------------------------------------------------
    # Return the best discovery as a plain Python list.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
