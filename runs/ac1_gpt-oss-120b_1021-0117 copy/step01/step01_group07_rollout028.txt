import time
import numpy as np

def _catalan_weights(n: int) -> np.ndarray:
    """First n Catalan‑style weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _score(a: np.ndarray) -> float:
    """Exact objective 2*n*max(conv)/sum(a)^2 for a >= 0."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * a.shape[0] * b.max() / (s * s))


def _grad_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the max convolution entry (index k) w.r.t. a."""
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    if lo <= hi:
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        rev = k - idx
        g[idx] = 2.0 * a[rev]
    return g
