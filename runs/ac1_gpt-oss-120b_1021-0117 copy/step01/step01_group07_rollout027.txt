    # progress iteration loop
    while time.monotonic() < deadline:

        conv = np.convolve(best, best)
        i_max = int(conv.argmax())
        # gradient vector built as shifted best
        # compute reversed slice
        # Determine range of k where i_max - k in [0,n-1], which is k ∈ [i_max-(n-1), i_max]
        lo = max(0, i_max - (n - 1))
        hi = min(n - 1, i_max)   # inclusive
        grad = np.zeros_like(best)
        # for k from lo to hi, set grad[k] = 2*best[i_max - k]
        # we can efficiently compute slice
        grad[lo:hi+1] = 2.0 * best[i_max - np.arange(lo, hi+1)]
        # propose candidate
        cand = best - step * grad
        # clip
        np.clip(cand, 0.0, None, out=cand)
        s = cand.sum()
        if s > 0:
            cand /= s
            score = float(evaluate_sequence(cand.tolist()))
            if score < best_score:
                best = cand
                best_score = score
                # increase step
                step = min(step * 1.1, max_step)
                if best_score <= 1.50274349:

        step = max(step * 0.5, min_step)

            eps = rng.random() * best[i] * 0.3  # smaller to be safe
            cand = best.copy()
            cand[i] -= eps
            cand[j] += eps
            cand /= cand.sum()
            score = float(evaluate_sequence(cand.tolist()))

import time
import numpy as np

def _catalan_weights(n: int) -> np.ndarray:
    """Return the first n Catalan‑style coefficients w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _sine_weights(n: int) -> np.ndarray:
    """Half‑sine positive window."""
    idx = np.arange(n, dtype=np.float64)
    return np.sin(np.pi * (idx + 0.5) / n)


def _exp_weights(n: int, lam: float = 0.005) -> np.ndarray:
    """Simple exponential decay."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- gradient step based on the current maximal convolution entry
        conv = np.convolve(best, best)
        i_max = int(conv.argmax())

        grad = np.zeros_like(best)
        lo = max(0, i_max - (n - 1))
        hi = min(n - 1, i_max)
        if lo <= hi:
            ks = np.arange(lo, hi + 1, dtype=np.int64)
            grad[ks] = 2.0 * best[i_max - ks]

        cand = best - step * grad
        np.clip(cand, 0.0, None, out=cand)
        s = cand.sum()
        if s > 0.0:
            cand /= s
            sc = _score(cand)
            if sc < best_score:
                best, best_score = cand, sc
                step = min(step * 1.2, max_step)
                if best_score <= target:
                    break
                continue  # accept and skip the rest of this iteration
        # if not accepted, shrink the step size
        step = max(step * 0.5, min_step)

        # ----- random mass transfer (conserve total mass)
        i, j = rng.integers(n, size=2)
        if best[i] > 0.0:
            eps = rng.random() * best[i] * 0.35  # move up to 35% of source
            cand = best.copy()
            cand[i] -= eps
            cand[j] += eps
            cand /= cand.sum()
            sc = _score(cand)
            if sc < best_score:
                best, best_score = cand, sc
                if best_score <= target:
                    break

        # ----- occasional jitter to escape plateaux
        if rng.random() < jitter_prob:
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            cand = best * jitter
            cand /= cand.sum()
            sc = _score(cand)
            if sc < best_score:
                best, best_score = cand, sc
                if best_score <= target:
                    break

    # ------------------------------------------------------------------
    # 6.   Return best sequence as plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best]
```
