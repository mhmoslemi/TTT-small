def conv_self(a):
    n = a.shape[0]

    fa = np.fft.rfft(a, n=m)

    s = a.sum()
    if s < 0.01:

    return 2.0*n*maxb/(s*s)

            k = int(conv.argmax())

            j = k - all_idx
            mask = (j >= 0) & (j < n)

            partner = np.zeros(n, dtype=np.float64)
            partner[mask] = cand[j[mask]]

            contrib = np.zeros(n, dtype=np.float64)
            contrib[mask] = cand[mask] * partner[mask]

            src_cands = np.where(contrib > 0)[0]
            if src_cands.size == 0:
                continue
            # pick source biased by its contribution
            w = contrib[src_cands]
            w /= w.sum()
            src = rng.choice(src_cands, p=w)

            # destination where this peak is untouched (partner == 0)
            zero_mask = (partner == 0.0) & (cand < 1000.0)
            if zero_mask.any():
                dst = rng.choice(np.flatnonzero(zero_mask))
            else:
                low_thr = np.percentile(partner[mask], 5) if mask.any() else 0.0
                low_mask = (partner <= low_thr) & (cand < 1000.0)
                if low_mask.any():
                    dst = rng.choice(np.flatnonzero(low_mask))
                else:
                    free = np.where(cand < 1000.0)[0]
                    if free.size == 0:
                        continue
                    dst = rng.choice(free)

            if src == dst:
                continue
            max_tr = min(cand[src], 1000.0 - cand[dst])
            if max_tr <= 0.0:
                continue
            delta = max_tr * rng.random() * 0.4   # up to 40 % of transferable mass
            cand[src] -= delta
            cand[dst] += delta

        elif op < 0.85:        # blockwise multiplicative scaling
            min_w = max(1, n // 250)
            max_w = max(2, n // 8)
            width = int(rng.integers(min_w, max_w + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.11)))
            cand[start:start + width] *= factor

        else:                  # small additive Gaussian noise
            cand += rng.normal(0.0, 0.018, size=n)

        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_cheap = _cheap_score(cand)
        if cand_cheap < cur_cheap - 1e-12:
            # move accepted for the hill‑climb
            cur_seq = cand
            cur_cheap = cand_cheap

            # verify with the authoritative evaluator
            try:
                cand_off = evaluate_sequence(cand.tolist())
            except Exception:
                cand_off = np.inf
            if np.isfinite(cand_off) and cand_off < best_off_score:
                best_off_seq = cand.copy()
                best_off_score = cand_off

    # ------------------------------------------------------------------
    # Return the best verified sequence (or the current one if none verified)
    # ------------------------------------------------------------------
    final = best_off_seq if np.isfinite(best_off_score) else cur_seq
    # final sanity checks
    np.clip(final, 0.0, 1000.0, out=final)
    if final.sum() < 0.01:
        final += 0.01 / n
    return [float(v) for v in final]
