def subgrad_step(a, step_scale):
    n = a.shape[0]
    size = 2*n
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)[:2*n-1]
    max_val = conv.max()
    # idx where conv nearly max
    idxs = np.where(conv >= max_val - 1e-14)[0]
    # shift values:
    ks = idxs - (n-1)
    grad = np.zeros_like(a)
    for k in ks:
        if k >= 0:
            grad[:n-k] += a[k:]
            grad[k:] += a[:n-k]
        else:
            k = -k
            grad[k:] += a[:n-k]
            grad[:n-k] += a[k:]
    # normalize gradient magnitude
    if grad.max() <= 0:
        return a
    scale = step_scale * a.max() / (grad.max() + 1e-12)
    cand = a - scale * grad
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s <= 0:
        return a
    cand /= s
    return cand
```

import time
import numpy as np

def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style central binomial weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w

def _score(arr: np.ndarray) -> float:
    """Fast AC1 upper‑bound for a non‑negative step function."""
    n = arr.shape[0]
    size = 2 * n                     # zero‑pad length for a full convolution
    fa = np.fft.rfft(arr, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    max_b = conv.max()
    s = arr.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

def _subgradient_move(a: np.ndarray, step_frac: float) -> np.ndarray:
    """Move opposite to a subgradient of the active convolution constraints."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    max_val = conv.max()
    # indices attaining (or almost attaining) the maximum
    idx = np.where(conv >= max_val - 1e-14)[0]
    ks = idx - (n - 1)               # signed shifts
    grad = np.zeros_like(a)
    for k in ks:
        if k >= 0:
            grad[: n - k] += a[k:]
            grad[k:] += a[: n - k]
        else:
            kk = -k
            grad[kk:] += a[: n - kk]
            grad[: n - kk] += a[kk:]
    if grad.max() == 0:
        return a
    scale = step_frac * a.max() / (grad.max() + 1e-12)
    cand = a - scale * grad
    np.clip(cand, 0.0, None, out=cand)
    cand_sum = cand.sum()
    if cand_sum > 0.0:
        cand /= cand_sum
    return cand
