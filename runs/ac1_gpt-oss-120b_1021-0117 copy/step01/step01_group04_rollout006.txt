def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(m: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..m-1."""
    a = np.empty(m, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, m):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_and_max(arr: np.ndarray, n: int, L: int):
    """Return (bound, index_of_max_conv, full_conv) using FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_val = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        bound = float('inf')
    else:
        bound = 2.0 * n * max_val / (s * s)
    idx = int(conv.argmax())
    return bound, idx, conv


def _gradient_at_offset(arr: np.ndarray, k: int) -> np.ndarray:
    """Gradient of conv[k] w.r.t. arr (2 * reversed slice)."""
    n = arr.shape[0]
    grad = np.empty_like(arr)
    if k < n:
        # positions 0..k get 2*arr[k::-1]
        rev = arr[: k + 1][::-1]
        grad[: k + 1] = 2.0 * rev
        if k + 1 < n:
            grad[k + 1 :] = 0.0
    else:
        start = k - n + 1
        grad[:start] = 0.0
        rev = arr[start:][::-1]            # length = n-start
        grad[start:] = 2.0 * rev
    return grad

    cur_score, cur_k, _ = _ac1_score_and_max(cur, n, L)
    best = cur.copy()
    best_score = cur_score

    # try the pure binomial as a secondary fallback (sometimes better)
    if parent_arr is not None:
        binom = _gen_binomial(n)
        bin_score, bin_k, _ = _ac1_score_and_max(binom, n, L)
        if bin_score < best_score:
            best = binom.copy()
            best_score = bin_score
            cur = best.copy()
            cur_score = best_score
            cur_k = bin_k

    # ----- annealing parameters -----------------------------------------------
    temperature = 1.0
    decay = 0.9996

    # pre‑computed helpers for move generation
    min_block = max(1, n // 250)
    max_block = max(2, n // 8)

    # -------------------------------------------------------------------------
    while time.monotonic() < deadline:
        # decide move type
        if rng.random() < 0.65:          # gradient‑guided mass transfer
            # compute gradient for the current worst offset
            grad = _gradient_at_offset(cur, cur_k)

            top_cnt = max(1, n // 20)    # ~5 % of indices
            # donors: large gradient & positive mass
            donors = np.argpartition(grad, -top_cnt)[-top_cnt:]
            donors = donors[cur[donors] > 1e-12]
            if donors.size == 0:
                i = int(rng.integers(n))
            else:
                i = int(rng.choice(donors))

            # recipients: small gradient
            recips = np.argpartition(grad, top_cnt)[:top_cnt]
            recips = recips[recips != i]
            if recips.size == 0:
                j = (i + 1) % n
            else:
                j = int(rng.choice(recips))

            # transfer a modest fraction of donor's mass
            max_move = cur[i] * 0.35
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move
            delta = min(delta, 1000.0 - cur[j])   # respect upper bound
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        else:                              # simple block scaling move
            width = int(rng.integers(min_block, max_block + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.11))
            cand = cur.copy()
            block = cand[start:start + width]
            block *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate candidate
        cand_score, cand_k, _ = _ac1_score_and_max(cand, n, L)

        # acceptance rule (simulated annealing)
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = cand
            cur_score = cand_score
            cur_k = cand_k

        # record best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < 1.50274349:   # target already reached
                break

        # temperature schedule
        temperature *= decay

        # occasional restart from best to avoid drift
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            cur_k = None
            temperature = 1.0

        # recompute cur_k lazily if needed
        if cur_k is None:
            cur_score, cur_k, _ = _ac1_score_and_max(cur, n, L)

    # -------------------------------------------------------------------------
    return [float(v) for v in best]
