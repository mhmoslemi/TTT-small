def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # Determine n and vector
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        n = int(parent.shape[0])
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    # Convolution length power of two
    m = 1 << ((2 * n - 1).bit_length())

    def _max_conv(a):
        A = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(A * A, n=m)
        return conv[:2*n-1].max()

    def _metric(a):
        maxc = _max_conv(a)
        s = a.sum()
        if s < 1e-12:
            return math.inf
        return 2.0 * n * maxc / (s * s)

    best_seq = None
    best_score = math.inf

    # Initial candidates
    candidates = []

    # parent fallback as candidate (as list)
    candidates.append(parent)

    # sidon original
    sid = _sidon_like_sequence(n)
    candidates.append(sid)

    # interleaved sidon
    candidates.append(_interleave(sid))

    # permuted sidon (random)
    candidates.append(_permute(sid, rng))

    # exponential families
    for beta in np.linspace(0.0, 0.0015, 15):
        expo = np.exp(-beta * np.arange(n, dtype=np.float64))
        candidates.append(expo)

    # power law
    power = 1.0/np.sqrt(np.arange(1, n+1, dtype=np.float64))
    candidates.append(power)

    # evaluate initial candidates
    for cand in candidates:
        score = _metric(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()

    # hill climbing / simulated annealing
    current = best_seq.copy()
    current_score = best_score

    # Temperature schedule parameters
    max_temp = 1.0
    min_temp = 1e-6

    # Helper to generate neighbor move
    while time.monotonic() < deadline:
        # decide move type
        move = rng.integers(0, 5)
        cand = current.copy()
        if move == 0:
            # swap two values
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif move == 1:

            cand[start:start+length] = cand[start:start+length][::-1]
        else:

def _sidon_like_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (1.0 - 1.0/(2.0*i))
    return a

def _interleave(values):
    n = values.shape[0]
    out = np.empty_like(values)
    left = 0
    right = n-1
    toggle = True
    idx = 0
    while left <= right:
        if toggle:
            out[left] = values[idx]
            left += 1
        else:
            out[right] = values[idx]
            right -= 1
        idx += 1
        toggle = not toggle
    return out

def _permute(values, rng):
    perm = rng.permutation(values.shape[0])

import numpy as np
import time
import math

def _sidon_like_sequence(n):
    """Classic Sidon‑like decreasing sequence a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a

def _interleave(values):
    """Place the largest values at the two ends, then the next largest,
    and so on (high‑low‑high‑low ordering)."""
    n = values.shape[0]
    out = np.empty_like(values)
    l, r = 0, n - 1
    toggle = True
    idx = 0
    while l <= r:
        if toggle:
            out[l] = values[idx]
            l += 1
        else:
            out[r] = values[idx]
            r -= 1
        idx += 1
        toggle = not toggle
    return out

def _permute(values, rng):
    """Random permutation using the supplied rng."""
    perm = rng.permutation(values.shape[0])
    return values[perm]

def _max_conv_fft(a, m):
    """Maximum of the linear convolution a*a using FFT (rfft + irfft)."""
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    n = a.shape[0]
    return conv[:2 * n - 1].max()

def _metric(a, n, m):
    """Exact AC1 metric used by the evaluator (scale‑invariant)."""
    maxc = _max_conv_fft(a, m)
    s = a.sum()
    if s <= 0.0:
        return math.inf
    return 2.0 * n * maxc / (s * s)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine length and a fallback parent sequence (if provided)
    # ------------------------------------------------------------------

                         0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        n = int(parent.shape[0])
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    # FFT length: next power of two ≥ 2·n
    m = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Assemble a few deterministic starters
    # ------------------------------------------------------------------
    starters = []
    starters.append(parent)

    sid = _sidon_like_sequence(n)
    starters.append(sid)
    starters.append(_interleave(sid))
    starters.append(_permute(sid, rng))

    # exponential families (beta from 0 up to a modest value)
    for beta in np.linspace(0.0, 0.0015, 15):
        starters.append(np.exp(-beta * np.arange(n, dtype=np.float64)))

    # power‑law tail
    starters.append(1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64)))

    # ------------------------------------------------------------------
    # Evaluate the deterministic starters
    # ------------------------------------------------------------------
    best_seq = None
    best_score = math.inf
    for cand in starters:
        sc = _metric(cand, n, m)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # Simulated annealing / hill‑climbing phase
    # ------------------------------------------------------------------
    current = best_seq.copy()
    current_score = best_score

    max_temp = 1.0          # start temperature
    min_temp = 1e-6         # final temperature

    # pool for occasional restarts (pre‑computed, cheap)
