import numpy as np
import time


def _initial_binomial(n: int) -> np.ndarray:
    """Generate decreasing coefficients a_i = C(2i,i)/4^i (i=0..n-1)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of x with itself using a power‑of‑two FFT."""
    n = x.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(fx * fx, n=m)[: 2 * n - 1]
    return conv


def _score_from_conv(conv: np.ndarray, sum_a: float, n: int) -> float:
    """Exact AC1 score from a convolution array."""
    return float(2.0 * n * np.max(conv) / (sum_a * sum_a))


def propose_candidate(seed=42, budget_s=1100, **kwargs):

    # leave a little head‑room for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
    except Exception:
        parent = None

    # ------------------------------------------------------------------
    # Initialise a feasible non‑negative sequence.

    if parent is not None:
        a = parent.copy()
    else:
        a = _initial_binomial(n)
