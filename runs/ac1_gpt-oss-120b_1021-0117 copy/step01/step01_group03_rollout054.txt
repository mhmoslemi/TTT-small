import numpy as np
import time

def _conv_self_fft(a):
    n = a.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa*fa, n=m)[:2*n-1]
    return conv

def _score_fast(a):
    n = a.shape[0]
    conv = _conv_self_fft(a)
    maxb = conv.max()
    return float(2.0 * n * maxb)  # sum is 1 after normalization

def gradient_step(a, lr):
    conv = _conv_self_fft(a)
    idx = int(conv.argmax())
    # compute gradient g
    n = a.shape[0]
    indices = np.arange(n)
    comp = idx - indices
    mask = (comp >= 0) & (comp < n)
    g = np.zeros_like(a)
    g[mask] = 2 * a[comp[mask]]
    # apply update
    new = a * np.exp(-lr * g)   # improbable large exponent if lr big
    # ensure positivity, renorm
    total = new.sum()
    if total == 0:
        return a
    new /= total
    return new

import numpy as np, time

def _conv_self_fft(a):
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv

def _objective(a):
    n = a.shape[0]
    conv = _conv_self_fft(a)
    maxb = conv.max()
    s = a.sum()
    return float(2.0 * n * maxb / (s * s))

def _sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2.0 * i - 1.0) / (2.0 * i)
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
    else:
        n = 5582
        a = _sqrt_series(n)

    n = a.shape[0]
    if a.sum() == 0.0:
        a[:] = 1.0 / n
    else:
        a = a / a.sum()

    # compute initial objective
    conv = _conv_self_fft(a)
    cur_score = float(2.0 * n * conv.max() / (a.sum() ** 2))
    best = a.copy()
    best_score = cur_score

    # parameters
    lr0 = 0.8          # initial learning rate for exponential update
    temp0 = 0.05       # initial temperature for occasional worse acceptance
    total_time = deadline - start_time

    # main loop
    while time.monotonic() < deadline:
        # elapsed fraction
        elapsed = time.monotonic() - start_time
        frac = elapsed / total_time
        # current learning rate and temperature (linear decay)
        lr = lr0 * (1.0 - frac)
        temp = temp0 * (1.0 - frac)

        # gradient step targeted at current max index
        max_idx = int(conv.argmax())
        idx = np.arange(n)
        partner = max_idx - idx
        mask = (partner >= 0) & (partner < n)
