def ac1_score(arr, n, L):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    # conv length L; linear conv up to 2*n-1
    conv = conv[:2*n-1]
    max_b = conv.max()
    s = arr.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (s * s)

import time, math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 bound using FFT (linear convolution)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    conv = conv[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4**i, i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a
