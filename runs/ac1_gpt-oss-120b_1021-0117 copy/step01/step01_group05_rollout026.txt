```
def construct_sparse_sequence(n, m_target, c_max=2, seed=...) -> np.ndarray:
    rng = np.random.default_rng(seed)
    # We'll attempt to build set of indices.
    indices = []
    sum_counts = {}
    attempts = 0
    max_attempts = 1000000
    while len(indices) < m_target and attempts < max_attempts:
        i = rng.integers(0, n)
        # compute required sum counts if we add i
        ok = True
        # check self sum i+i:
        k_self = i + i
        cnt = sum_counts.get(k_self, 0) + 1
        if cnt > c_max:
            ok = False
        else:
            # check sums with existing indices
            for j in indices:
                k = i + j
                cnt = sum_counts.get(k, 0) + 1
                if cnt > c_max:
                    ok = False
                    break
        if ok:
            # add i
            indices.append(i)
            # update sum_counts
            sum_counts[i + i] = sum_counts.get(i + i, 0) + 1
            for j in indices[:-1]: # all previous
                k = i + j
                sum_counts[k] = sum_counts.get(k, 0) + 1
        attempts += 1
    # If not enough indices, we could relax c_max or increase attempts.
```
