import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Heuristic search for a non‑negative step function reducing the
    AC‑1 bound.  Uses a combination of multiplicative‑gradient updates,
    random segment perturbations and greedy mass transfers.  The
    candidate with the lowest score seen before the deadline is returned."""
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:                      # no parent provided
        n = 5000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / n
    else:
        a /= s                               # normalise to sum == 1

    # ----- helpers ----------------------------------------------------
    def conv_max(x: np.ndarray) -> float:
        """Maximum of the self‑convolution using an FFT (no dense matrices)."""
        m = 1 << ((2 * n - 2).bit_length())          # next power of two ≥ 2n‑1
        fftx = np.fft.rfft(x, n=m)
        c = np.fft.irfft(fftx * fftx, n=m)[: 2 * n - 1]
        c = np.maximum(c, 0.0)                      # numerical safety
        return float(c.max())

    def score_from_max(conv_max_val: float) -> float:
        """Exact score (sum is kept equal to 1)."""
        return 2.0 * n * conv_max_val

    # initial score
    cur_max = conv_max(a)
    best = a.copy()
    best_score = score_from_max(cur_max)

    # ----- parameters -------------------------------------------------
    eta = 0.5                     # step size for multiplicative update
    min_eta = 1e-5
    step_factor = 0.3            # fraction of source mass moved in greedy step
    min_step = 1e-12
    no_improve = 0               # counter for stagnation

    # ----- main optimisation loop ------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    while time.monotonic() < deadline:
        # ---------- current convolution & worst lag -------------------
        m = 1 << ((2 * n - 2).bit_length())
        fft_a = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(fft_a * fft_a, n=m)[: 2 * n - 1]
        conv = np.maximum(conv, 0.0)
        k = int(conv.argmax())
        cur_max = float(conv.max())

        # ---------- partner vector (= derivative of the max lag) ------
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = np.zeros_like(a)
        partner[idx] = a[k - idx]                # a_{k-i} where defined

        # ---------- 1) multiplicative‑gradient update --------------
        mul = np.exp(-eta * partner)
        a_mul = a * mul
        a_mul_sum = a_mul.sum()
        if a_mul_sum > 0:
            a_mul /= a_mul_sum
        new_max = conv_max(a_mul)
        new_score = score_from_max(new_max)

        if new_score < best_score:
            best = a_mul.copy()
            best_score = new_score
            a = a_mul
            eta = max(min_eta, eta * 0.99)      # slowly decay step size
            no_improve = 0
            continue

        # ---------- 2) occasional random segment perturbation -------
        if rng.random() < 0.02:
            width = int(rng.integers(max(1, n // 300), max(2, n // 30)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            cand = a.copy()
            scale = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start:stop] *= scale
            cand = np.clip(cand, 0.0, 1000.0)
            cand_sum = cand.sum()
            if cand_sum > 0:
                cand /= cand_sum
                cand_score = score_from_max(conv_max(cand))
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
                    a = cand
                    eta = max(min_eta, eta * 0.95)
                    no_improve = 0
                    continue

        # ---------- 3) greedy mass transfer -------------------------
        # contribution of each index to the current maximum
        contrib = a * partner
        src = int(contrib.argmax())
        # destination = index with smallest partner (often zero)
        min_val = partner.min()
        dest_candidates = np.where(partner == min_val)[0]
        if dest_candidates.size == 0:
            dest = int(rng.integers(0, n))
        else:
            dest = int(rng.choice(dest_candidates))
        if src != dest:
            delta = step_factor * a[src]
            delta = max(delta, min_step)
            delta = min(delta, a[src])
            cand = a.copy()
            cand[src] -= delta
            cand[dest] += delta
            cand = np.clip(cand, 0.0, 1000.0)
            cand_sum = cand.sum()
            if cand_sum > 0:
                cand /= cand_sum
                cand_score = score_from_max(conv_max(cand))
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
                    a = cand
                    step_factor = max(0.01, step_factor * 0.97)
                    no_improve = 0
                    continue

        # ---------- fallback: decay parameters when stuck -----------
        no_improve += 1
        if no_improve > 2000:                 # prolonged stagnation
            eta = min(1.0, eta * 1.5)          # try a larger step
            step_factor = min(0.5, step_factor * 1.2)
            no_improve = 0

    # ----- final sanity check -----------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
