def project_simplex(v):
    # from Wikipedia: projection onto simplex {x: sum=1, x>=0}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _catalan_weights(n):
    w = np.empty(n, dtype=np.float64)
    w[0]=1.0
    for i in range(1,n):
        w[i] = w[i-1] * ((2*i)*(2*i-1)) / (i*i*4.0)
    return w
```

We will also define:

```
def project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def _catalan_weights(n: int) -> np.ndarray:
    """First n Catalan‑style weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the AC1 upper bound defined by ``evaluate_sequence``.
    A cheap LP refinement (maximising total mass under the current convolution
    bound) is performed once, then a projected‑gradient descent on a smooth
    approximation of the max‑convolution is applied until time runs out.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1.  Load parent (if present) and set length n
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = int(parent.size)

    # ------------------------------------------------------------------
    # 2.  Helper to evaluate the bound (no need to call the external evaluator)
    # ------------------------------------------------------------------
    def _bound(a: np.ndarray) -> float:
        s = a.sum()
        if s < 1e-12:
            return np.inf
        conv = np.convolve(a, a)
        return float(2.0 * n * np.max(conv) / (s * s))

    # ------------------------------------------------------------------
    # 3.  Assemble a few deterministic starting points and keep the best
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    def _try_candidate(cand):
        nonlocal best_a, best_score
        sc = _bound(cand)
        if np.isfinite(sc) and sc < best_score:
            best_a = cand.copy()
            best_score = sc

    # parent (normalised)
    if parent.sum() > 0.0:
        _try_candidate(parent / parent.sum())

    # Catalan weights
    w = _catalan_weights(n)
    _try_candidate(w / w.sum())

    # Power‑law sweep 0.30 … 0.70
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in np.linspace(0.30, 0.70, 41):
        a_pow = idx ** (-p)
        _try_candidate(a_pow / a_pow.sum())

    # fallback safety
    if best_a is None:
        best_a = np.ones(n, dtype=np.float64) / n
        best_score = _bound(best_a)

    # ------------------------------------------------------------------
    # 4.  One LP refinement using the current best sequence
    # ------------------------------------------------------------------
    def _lp_refine(f):
        """Return a normalised vector with larger total mass under the same
        convolution bound, or None if the LP fails."""
        M = np.max(np.convolve(f, f))
        # Build sparse convolution matrix C (size (2n‑1) × n)
        N2 = n * n
        rows = np.empty(N2, dtype=np.int32)
        cols = np.empty(N2, dtype=np.int32)
        data = np.empty(N2, dtype=np.float64)
        pos = 0
        for j in range(n):
            rows[pos:pos + n] = np.arange(j, j + n, dtype=np.int32)
            cols[pos:pos + n] = j
            data[pos:pos + n] = f
            pos += n
        C = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b_vec = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise sum(b)  ⇔ minimise -sum(b)
        try:

                c,
                A_ub=C,
                b_ub=b_vec,
                bounds=[(0, None)] * n,
                method="highs",
                options={"presolve": True, "time_limit": 30},
            )
            if res.success:
                g = np.maximum(res.x, 0.0)
                if g.sum() > 0.0:
                    g = g / g.sum()
                    return g
        except Exception:
            pass
        return None

            for cand in (g, (best_a + g) * 0.5):
                cand = _project_simplex(cand)

                    best_a = cand.copy()
                    best_score = sc

        a = best_a.copy()
    else:

    # ------------------------------------------------------------------

        cand = np.maximum(cand, 0.0)

                break
        else:

        # occasional random jitter to escape plateaus
        if rng.random() < 0.02:
            jitter = np.exp(rng.normal(0.0, 0.018, size=n))
            a = _project_simplex(a * jitter)

        # gradually increase the soft‑max temperature
        if rng.random() < 0.1:
            alpha = min(alpha * 1.07, 200.0)

        # occasional small random mass transfer
        if rng.random() < 0.01:

    # ------------------------------------------------------------------

        best_a = np.ones(n, dtype=np.float64) / n
    return [float(v) for v in best_a]
