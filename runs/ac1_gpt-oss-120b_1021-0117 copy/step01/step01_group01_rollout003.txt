            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, min_temp))
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            cur_score = new_score
            if new_score < best_score:
                best = a.copy()
                best_score = new_score

        # ----- temperature schedule -----
        temp = max(temp * decay, min_temp)

        # ----- early exit if we reach the target bound -----
        if best_score <= 1.50274349:
            break

    # --------------------------------------------------------------
    # 5. Return the best sequence found (list of plain floats)
    # --------------------------------------------------------------
    final = [float(x) for x in best]
    # final sanity check – the harness will also do this
    try:
        _ = evaluate_sequence(final)
    except Exception:
        final = [float(x) for x in np.ones(n) / n]
    return final
