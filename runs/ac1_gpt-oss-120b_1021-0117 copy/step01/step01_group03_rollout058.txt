import numpy as np
import time

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (O(n log n))."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the authoritative evaluator."""
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def _max_contrib_pair(a: np.ndarray, k: int):
    """
    For convolution index k, return the pair (i,j) with the largest
    contribution a[i]*a[j] (i+j==k) together with the contribution value.
    """
    n = a.shape[0]
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    i = np.arange(start, end + 1, dtype=np.int64)
    j = k - i
    prod = a[i] * a[j]
    idx = int(np.argmax(prod))
    return int(i[idx]), int(j[idx]), float(prod[idx])


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a low‑score non‑negative step function under the given budget."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ----------------------------------------------------------------------
    # Initialise – use the supplied parent if it exists, otherwise the sqrt series.
    # Keep the better of the two as a starting point.
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = parent.size
        best = parent.copy()
        best_score = _score(best)
    else:
        n = 5582
        best = _sqrt_series(n)
        best_score = _score(best)

    # also evaluate the raw sqrt series (useful when a parent exists)
    sqrt_seq = _sqrt_series(n)
    sqrt_score = _score(sqrt_seq)
    if sqrt_score < best_score:
        best = sqrt_seq.copy()
        best_score = sqrt_score

    # ----------------------------------------------------------------------
    # Main optimisation loop.
    #   – identify the current worst convolution index,
    #   – pull a small fraction of mass from the most responsible entry,
    #   – relocate it to a random position (preserving the total sum),
    #   – accept the change only if the score improves.
    # Occasionally perform a cheap smoothing step to escape plateaus.
    # ----------------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        # current convolution and its worst index
        conv = _conv_self(best)
        max_idx = int(np.argmax(conv))

        # pair (i,j) that contributes most to this index
        i_max, j_max, _ = _max_contrib_pair(best, max_idx)

        # pick the larger side of the pair as source
        src = i_max if best[i_max] >= best[j_max] else j_max
        if best[src] <= 1e-12:
            # nothing to move – try a tiny random perturbation instead
            noise = rng.normal(0.0, 0.02, size=n)
            cand = best + noise
            np.clip(cand, 0.0, 1000.0, out=cand)
        else:
            # destination must be different from source
            dst = rng.integers(0, n)
            if dst == src:
                continue

            # how much to transfer (3–10 % of the source entry)
            frac = rng.random() * 0.07 + 0.03
            delta = best[src] * frac
            # respect upper bound at destination
            delta = min(delta, 1000.0 - best[dst])
            # respect non‑negativity at source
            delta = min(delta, best[src])
            if delta <= 0.0:
                continue

            cand = best.copy()
            cand[src] -= delta
            cand[dst] += delta
            np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate candidate
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

        # every few iterations try a cheap averaging (smoothing) move
        if iteration % 250 == 0:
            width = int(rng.integers(max(2, n // 200), max(5, n // 30)))
            start = rng.integers(0, max(1, n - width + 1))
            stop = start + width
            avg_val = best[start:stop].mean()
            cand2 = best.copy()
            cand2[start:stop] = avg_val
            sc2 = _score(cand2)
            if sc2 < best_score:
                best = cand2
                best_score = sc2

        iteration += 1

        # early stop if we already beat the target bound
        if best_score <= 1.5028:
            break

    # ----------------------------------------------------------------------
    # Return the best discovered candidate as a plain Python list.
    # ----------------------------------------------------------------------
    return [float(v) for v in best]
