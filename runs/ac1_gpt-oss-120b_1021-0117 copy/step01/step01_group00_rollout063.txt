def fast_ac1(seq):
    n = len(seq)
    a = np.asarray(seq, dtype=np.float64)
    # zero-pad to length 2*n-1
    # use next power of two
    size = 1 << (2*n-1).bit_length()
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    maxb = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2 * n * maxb / (sum_a**2)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1) # leave margin.
    # get n and base sequence
    try:
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
        n = int(base.size)
    except Exception:
        n = 256
        base = np.full(n, 1.0, dtype=np.float64)
    # define fast_score func
    size_fft = 1 << ((2*n-1).bit_length())
    def fast_score(seq):
        a = np.asarray(seq, dtype=np.float64)
        fa = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(fa*fa, n=size_fft)[:2*n-1]
        maxb = np.max(conv)
        sum_a = np.sum(a)
        if sum_a < 1e-12:
            return np.inf
        return 2.0 * n * maxb / (sum_a * sum_a)
    # initial candidates
    best_seq = base.copy()
    best_fast = fast_score(best_seq)
    # try sidon-like
    def sidon(n):
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i-1] * (1.0 - 1.0/(2.0*i))
        return a
    sid = sidon(n)
    sid = np.clip(sid, 0.0, 1000.0)
    sf = fast_score(sid)
    if sf < best_fast:
        best_seq = sid.copy()
        best_fast = sf
    # local search loop
    current = best_seq.copy()
    cur_fast = best_fast
    # parameters
    MUTATE_COUNT = 3
    mut_scale = 0.3
    # start iteration
    while time.monotonic() < deadline:
        # create mutated copy
        cand = current.copy()
        # decide how many indices to mutate (1 to MUTATE_COUNT)
        k = rng.integers(1, MUTATE_COUNT+1)
        idx = rng.choice(n, size=k, replace=False)
        for i in idx:
            factor = 1.0 + rng.normal(scale=mut_scale)
            new_val = cand[i] * factor
            if new_val < 0.0:
                new_val = 0.0
            cand[i] = new_val
        # optionally random small Gaussian noise to entire array (prob small)
        if rng.random() < 0.01:
            cand = cand * (1.0 + rng.normal(scale=0.05, size=n))
            cand = np.maximum(cand, 0.0)
        # ensure max bound 1000 (clip)
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_fast = fast_score(cand)
        if cand_fast < cur_fast:
            current = cand
            cur_fast = cand_fast
            if cand_fast < best_fast:
                best_seq = cand.copy()
                best_fast = cand_fast
        # occasional reset to best_seq to avoid drift
        if rng.random() < 0.005:
            current = best_seq.copy()
            cur_fast = best_fast
    # final authoritative evaluation of best_seq
    final_score = float(evaluate_sequence(best_seq.tolist()))
    # maybe try a few small perturbations evaluated with evaluate_sequence if time left
    # ...
    return [float(x) for x in best_seq]

import numpy as np
import time

def _analytic_score(n, u, a):
    """Analytic bound for the piecewise constant sequence:
    first u entries have height a, remaining n‑u entries have height 1.
    """
    L = n - u                     # length of the low part
    # three possible contributions to the convolution maximum
    max_b = max(L, 2.0 * u * a, u * a * a)
    sum_a = u * a + L
    if sum_a <= 0.0:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _make_sequence(n, u, a):
    """Construct the sequence described above."""
    seq = np.empty(n, dtype=np.float64)
    seq[:u] = a
    seq[u:] = 1.0
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq
