import numpy as np
import time
import math

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (O(n log n))."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score_arr(a: np.ndarray) -> float:
    """Exact objective (same as the authoritative evaluator)."""
    n = a.shape[0]
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _gradient_step(a: np.ndarray, eta: float) -> np.ndarray:
    """One projected‑gradient move that reduces the current maximal
    convolution coefficient."""
    n = a.shape[0]
    b = _conv_self(a)                     # full self‑convolution
    k = int(np.argmax(b))                 # index of the current worst lag

    # gradient of max(b) w.r.t. a is 2*a reflected around k.
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx = np.arange(i_start, i_end + 1)
    grad = np.zeros_like(a)
    grad[idx] = a[k - idx]                # a reflected, factor 2 omitted (absorbed in eta)

    # multiplicative update and re‑normalization (scale invariance)
    factor = np.exp(-eta * grad)
    a_new = a * factor
    total = a_new.sum()
    if total == 0.0:
        return a.copy()
    a_new /= total
    return a_new
