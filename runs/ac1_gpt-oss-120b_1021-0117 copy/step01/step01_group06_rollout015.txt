def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    if 'height_sequence_1' in globals():
        init_arr = np.asarray(height_sequence_1, dtype=np.float64).copy()
        init_arr = np.maximum(init_arr, 0.0)
        # clip large values
        init_arr = np.minimum(init_arr, 1000.0)
        n = init_arr.size
        init_arr = init_arr / init_arr.sum()
    else:
        n = 256
        init_arr = rng.random(n)
        init_arr = init_arr / init_arr.sum()

    # also compute sqrt series as alt initial
    sqrt_init = _sqrt_series(n)
    sqrt_init = sqrt_init / sqrt_init.sum()

    # evaluate both
    best_vec = init_arr
    best_sc = _score(best_vec)
    sqrt_sc = _score(sqrt_init)
    if sqrt_sc < best_sc:
        best_vec = sqrt_init
        best_sc = sqrt_sc

    # Precompute L for FFT
    L = 1 << ((2 * n - 1).bit_length())

    # main loop
    no_improve_iters = 0
    while time.monotonic() < deadline:
        # random choice for move type
        if rng.random() < 0.6:
            # gradient step at max conv
            # compute conv of best_vec
            fv = np.fft.rfft(best_vec, n=L)
            conv = np.fft.irfft(fv * fv, n=L)[:2 * n - 1]
            idx_max = int(conv.argmax())
            # compute gradient
            idx = idx_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad = np.zeros_like(best_vec)
            grad[mask] = 2.0 * best_vec[idx[mask]]
            step = rng.random() * 0.2  # up to 0.2
            cand = best_vec - step * grad
            # project to simplex
            cand = _project_simplex(cand)
        else:
            # random mass transfer
            i = rng.integers(0, n)
            if best_vec[i] <= 0:
                continue
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            frac = rng.random() * 0.25
            delta = best_vec[i] * frac
            cand = best_vec.copy()
            cand[i] -= delta
            cand[j] += delta
            # simple projection not needed because sum unchanged and nonneg; but due to round errors ensure positivity
            # still we may project to simplex to be safe
            cand = _project_simplex(cand)

        # evaluate
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand
            no_improve_iters = 0
        else:
            no_improve_iters += 1

        # occasional restart if stuck
        if no_improve_iters > 2000:
            # random Dirichlet vector
            rand_vec = rng.random(n)
            rand_vec = _project_simplex(rand_vec)
            sc_rand = _score(rand_vec)
            if sc_rand < best_sc:
                best_sc = sc_rand
                best_vec = rand_vec
            no_improve_iters = 0

        # Additionally, sometimes reduce first coefficient aggressively
        if rng.random() < 0.02:
            # attempt to reduce first coefficient (index 0)
            # gradient for max conv might not target 0; we apply a simple step
            grad0 = np.zeros_like(best_vec)
            grad0[0] = 2.0 * best_vec[0]  # as earlier heuristics
            step0 = rng.random() * 0.5
            cand2 = _project_simplex(best_vec - step0 * grad0)
            sc2 = _score(cand2)
            if sc2 < best_sc:
                best_sc = sc2
                best_vec = cand2

    return [float(x) for x in best_vec]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]

    else:

    return w

    cand_list.append(sqrt_vec / sqrt_vec.sum())

    cand_list.append(_hann_window(n))
    cand_list.append(_triangular_window(n))
    cand_list.append(_gaussian_window(n, 0.25))
    cand_list.append(_gaussian_window(n, 0.40))

    # pick the best of the initial candidates
    best_vec = cand_list[0]
    best_sc = _score(best_vec)
    for v in cand_list[1:]:
        sc = _score(v)
        if sc < best_sc:
            best_sc = sc
            best_vec = v

    # --------------------------------------------------------
    # Pre‑compute FFT length and helpers
    # --------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    def _conv_info(v):
        fv = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        idx = int(conv.argmax())
        return conv, idx

    # --------------------------------------------------------
    # Stochastic local optimisation
    # --------------------------------------------------------
    no_improve = 0
    target_sc = 1.50274349  # early‑stop if we beat the target

    while time.monotonic() < deadline:
        move_type = rng.random()

        if move_type < 0.55:
            # ----------------------------------------------------
            # Targeted mass transfer based on the current peak
            # ----------------------------------------------------
            conv, idx_max = _conv_info(best_vec)

            idx = idx_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            contrib = np.zeros(n, dtype=np.float64)
            contrib[mask] = best_vec[mask] * best_vec[idx[mask]]

            if np.any(contrib):
                src = int(np.argmax(contrib))
            else:
                src = rng.integers(0, n)

            # pick a distinct destination
            dst = rng.integers(0, n - 1)
            if dst >= src:
                dst += 1

            if best_vec[src] <= 0.0:
                continue

            frac = rng.random() * 0.25
            delta = best_vec[src] * frac
            cand = best_vec.copy()
            cand[src] -= delta
            cand[dst] += delta
            cand = _project_simplex(cand)

        elif move_type < 0.85:
            # ----------------------------------------------------
            # Gradient‑type step at the maximal autocorrelation index
            # ----------------------------------------------------
            conv, idx_max = _conv_info(best_vec)

            idx = idx_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad = np.zeros_like(best_vec)
            grad[mask] = 2.0 * best_vec[idx[mask]]
            step = rng.random() * 0.20
            cand = best_vec - step * grad
            cand = _project_simplex(cand)

        else:
            # ----------------------------------------------------
            # Classic random mass shuffle (hill climbing)
            # ----------------------------------------------------
            src = rng.integers(0, n)
            if best_vec[src] <= 0.0:
                continue
            dst = rng.integers(0, n - 1)
            if dst >= src:
                dst += 1
            frac = rng.random() * 0.25
            delta = best_vec[src] * frac
            cand = best_vec.copy()
            cand[src] -= delta
            cand[dst] += delta
            cand = _project_simplex(cand)

        # evaluate the candidate
        sc = _score(cand, L=L)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand
            no_improve = 0
            if best_sc <= target_sc:
                break
        else:
            no_improve += 1

        # occasional aggressive reduction of the first entry (historical trick)
        if rng.random() < 0.02:
            grad0 = np.zeros_like(best_vec)
            grad0[0] = 2.0 * best_vec[0]
            step0 = rng.random() * 0.5
            cand0 = _project_simplex(best_vec - step0 * grad0)
            sc0 = _score(cand0, L=L)
            if sc0 < best_sc:
                best_sc = sc0
                best_vec = cand0
                no_improve = 0

        # restart if stuck for many iterations
        if no_improve > 3000:
            rnd = rng.random(n)
            rnd = _project_simplex(rnd)
            sc_r = _score(rnd, L=L)
            if sc_r < best_sc:
                best_sc = sc_r
                best_vec = rnd
                no_improve = 0
            else:
                no_improve = 0  # give another chance

    # --------------------------------------------------------
    # Return the discovered best vector as a plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
