    n = len(seq)
    # gradient from conv peak
    # for i from max(0, k_max-(n-1)) to min(k_max, n-1):
    lo = max(0, k_max - (n-1))
    hi = min(k_max, n-1)
    # gradient vector:
    grad = np.zeros_like(seq)
    # i_range = np.arange(lo, hi+1)
    idx = np.arange(lo, hi+1)
    grad[idx] = seq[k_max - idx]
    # choose i_high weighted by grad
    if grad.sum() == 0:
        return seq
    # high probabilities
    p_high = grad / grad.sum()
    i_high = rng.choice(n, p=p_high)
    # low probabilities (prefer low grad)
    inv = grad.max() - grad + 1e-12  # ensures all >0
    p_low = inv / inv.sum()
    i_low = rng.choice(n, p=p_low)
    # ensure not equal
    if i_low == i_high:
        i_low = (i_low + 1) % n
    # transfer fraction
    delta = seq[i_high] * 0.2  # 20% of source
    # if delta > 0: else skip
    new_seq = seq.copy()
    new_seq[i_high] -= delta
    new_seq[i_low] += delta
    np.clip(new_seq, 0, 1000, out=new_seq)
    return new_seq

def _fast_score(seq):
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]
    max_val = conv.max()
    score = 2.0 * n * max_val / (seq.sum() ** 2)
    max_idx = int(conv.argmax())
    return float(score), conv, max_idx
```

    # constants for move probabilities
    prob_grad = 0.6
    prob_block = 0.2
    prob_single = 0.1
    prob_smooth = 0.1
    # Main loop
    iter_count = 0
    while time.monotonic() < deadline:
        candidate = best.copy()
        r = rng.random()
        if r < prob_grad:
            # gradient transfer move
            n = best.shape[0]
            S = best.sum()
            k = best_max_idx
            B = best_conv[k]
            # shift vector
            shift = np.zeros_like(best)
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            shift[idx] = best[k - idx]
            # gradient of objective (ignoring constant factor)
            ref = B / S
            grad = shift - ref
            # choose source with positive gradient
            pos = np.where(grad > 0)[0]
            neg = np.where(grad < 0)[0]
            if pos.size == 0 or neg.size == 0:
                # fallback to random move
                i_src = rng.integers(0, n)
                i_tgt = rng.integers(0, n)
            else:
                i_src = pos[grad[pos].argmax()]
                i_tgt = neg[grad[neg].argmin()]  # most negative
            # transfer fraction
            if candidate[i_src] > 0:
                delta = candidate[i_src] * rng.uniform(0.05, 0.2)  # 5-20%
                candidate[i_src] -= delta
                candidate[i_tgt] += delta
        elif r < prob_grad + prob_block:
            # block scaling
            width = rng.integers(1, max(2, n // 30))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.08))
            candidate[start:start+width] *= factor
        elif r < prob_grad + prob_block + prob_single:
            # single perturb
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.25))
            candidate[idx] *= factor
        else:
            # smooth neighbours
            i = rng.integers(0, n-1)
            avg = (candidate[i] + candidate[i+1]) * 0.5
            candidate[i] = avg
            candidate[i+1] = avg
        # enforce bounds
        np.clip(candidate, 0.0, 1000.0, out=candidate)
        # compute score
        cand_score, cand_conv, cand_max_idx = _fast_score(candidate)
        if cand_score < best_score:
            best = candidate
            best_score = cand_score
            best_conv = cand_conv
            best_max_idx = cand_max_idx
        iter_count += 1
    # Return best as list of floats
    return [float(x) for x in best]

import numpy as np, time, math

def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fast_score(seq: np.ndarray):
    """Return (AC1 score, convolution array, argmax index) using FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # power‑of‑2 length
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = conv.max()
    score = 2.0 * n * max_val / (seq.sum() ** 2)
    return float(score), conv, int(conv.argmax())


def _interleave_desc(arr: np.ndarray) -> np.ndarray:
    """Place the sorted descending values alternately at the left and right ends."""
    desc = np.sort(arr)[::-1]                     # descending
    n = desc.size
    out = np.empty_like(desc)
    lo, hi = 0, n - 1
    left = True
    for v in desc:
        if left:
            out[lo] = v
            lo += 1
        else:
            out[hi] = v
            hi -= 1
        left = not left
    return out
