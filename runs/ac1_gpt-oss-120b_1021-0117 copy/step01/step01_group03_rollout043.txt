    # Helper to evaluate sequence via evaluator

        return evaluate_sequence([float(v) for v in seq])

    # candidate list: power law exponents
    # include p from 0.55 to 0.95 inclusive
    for p in np.linspace(0.55, 0.95, 21):

def _score_seq(a):
    n = a.size
    # compute conv via FFT
    m = 1<<((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=m)
    prod = fa*fa
    conv = np.fft.irfft(prod, n=m)[:2*n-1]
    M = conv.max()
    s = a.sum()
    if s < 0.01: return np.inf
    return float(2.0*n*M/(s*s))
```

import numpy as np
import time

# ------------------------------------------------------------
# Fast convolution and score (identical to the evaluator but
# without the safety checks – scaling of the vector does not
# affect the score, so this routine is sufficient for the
# internal search.
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    n = a.size
    b = _conv_self(a)
    M = b.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * M / (s * s))
