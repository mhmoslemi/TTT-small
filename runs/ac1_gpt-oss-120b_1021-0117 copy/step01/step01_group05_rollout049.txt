def fast_score(seq):
    n = len(seq)
    m = 1 << ((2*n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]
    max_val = conv.max()
    sum_val = seq.sum()
    if sum_val <= 0: return np.inf, conv, max_val
    score = 2.0 * n * max_val / (sum_val*sum_val)
    return float(score), conv, max_val

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a decreasing positive seed a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fast_score(seq: np.ndarray):
    """Return (score, full convolution, max convolution) using FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val
