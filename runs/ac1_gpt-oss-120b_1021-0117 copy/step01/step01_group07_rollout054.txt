    # Returns w = argmin ||w-v||^2 st w>=0, sum w = s

import time, numpy as np

def project_simplex(v, s):
    # Duchi algorithm
    # handle case all negative => return zeros
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho_arr.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def compute_obj(a, S):
    # objective = max convolution; with sum S fixed, returns M
    conv = np.convolve(a, a)
    return np.max(conv)

def subgrad_max(a, conv, M):
    # returns subgradient for max term(s)
    # conv length = 2n-1
    # mask for indices where conv close to M
    tol = 1e-12
    mask = (conv >= M - tol)
    # Convert mask to float array
    K = mask.astype(np.float64)
    # Compute grad using convolution
    rev_a = a[::-1]
    # convolution full
    grad_full = np.convolve(K, rev_a)  # length 3n-2
    n = a.shape[0]
    grad = 2 * grad_full[n-1 : n-1 + n]   # slice length n
    return grad
```

    # ensure sum(v) = s and v >=0

import time
import numpy as np

def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x)=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style decreasing weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w

    # ------------------------------------------------------------------
