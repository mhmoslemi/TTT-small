import time
import numpy as np


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT (O(n log n))."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a non‑negative step function with a low AC1 bound.

    Returns a list of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)
    # leave a few seconds margin for finalisation
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initial sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = a.size
    else:
        n = 5582
        a = _sqrt_series(n)

    # compare with pure sqrt‑series if a came from the parent
    sqrt_seq = _sqrt_series(n)
    conv_sqrt = _conv_self(sqrt_seq)
    score_sqrt = 2.0 * n * conv_sqrt.max() / (sqrt_seq.sum() ** 2)
    conv_a = _conv_self(a)
    score_a = 2.0 * n * conv_a.max() / (a.sum() ** 2)
    if score_sqrt < score_a:
        a = sqrt_seq.copy()
        conv_a = conv_sqrt.copy()
        score_a = score_sqrt

    best_a = a.copy()
    best_score = score_a

    # target sum (we keep it essentially constant)
    S_target = a.sum()

    # ------------------------------------------------------------------
    # Helper for full score (using current convolution)
    # ------------------------------------------------------------------
    def _current_score(conv_arr, total):
        """objective = 2*n*max(conv)/(total**2)"""
        return 2.0 * n * conv_arr.max() / (total * total)

    # ------------------------------------------------------------------
    # Main stochastic optimisation loop
    # ------------------------------------------------------------------
    early_cut = n // 4
    late_cut = 3 * n // 4
    reset_interval = 4000          # recompute convolution from scratch occasionally
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        op = rng.random()

        # --------------------------------------------------------------
        # 1) Targeted mass transfer – tries to lower the current peak.
        # --------------------------------------------------------------
        if op < 0.55:
            # locate the index where the convolution is maximal
            kmax = int(np.argmax(conv_a))

            # choose a source index that belongs to the early part and whose
            # partner (kmax - src) lies inside the array.
            src_min = max(0, kmax - (n - 1))
            src_max = min(early_cut - 1, kmax)
            if src_max >= src_min:
                src = rng.integers(src_min, src_max + 1)
                partner = kmax - src
                if partner == src or partner < 0 or partner >= n:
                    # degenerate case – skip this iteration
                    continue

                # two destination indices far in the tail
                dst1 = rng.integers(late_cut, n)
                dst2 = rng.integers(late_cut, n)

                # transferable amount (keep sum constant)
                max_transfer = min(a[src], a[partner])
                if max_transfer <= 0.0:
                    continue
                eps = max_transfer * rng.random() * 0.15   # up to 15 % of the smaller entry

                # deltas (sum to zero)
                d_src = -eps
                d_partner = -eps
                d_dst1 = eps
                d_dst2 = eps

                # ----------------------------------------------------------
                # Update convolution incrementally (using the *old* a)
                # ----------------------------------------------------------
                changes = [(src, d_src), (partner, d_partner),
                           (dst1, d_dst1), (dst2, d_dst2)]

                # a*Δ part
                for idx, d in changes:
                    if d != 0.0:
                        conv_a[idx: idx + n] += 2.0 * d * a

                # Δ*Δ part (pairwise)
                L = len(changes)
                for i in range(L):
                    ii, di = changes[i]
                    for j in range(i, L):
                        ij, dj = changes[j]
                        if i == j:
                            conv_a[2 * ii] += di * dj
                        else:
                            conv_a[ii + ij] += 2.0 * di * dj

                # apply the deltas to the sequence
                for idx, d in changes:
                    a[idx] += d
                    # clipping keeps the variables in the admissible range
                    if a[idx] < 0.0:
                        a[idx] = 0.0
                    elif a[idx] > 1000.0:
                        a[idx] = 1000.0

        # --------------------------------------------------------------
        # 2) Block scaling – a moderate scaling of a contiguous block,
        #    followed by renormalisation to keep the total sum fixed.
        # --------------------------------------------------------------
        elif op < 0.85:
            width = int(rng.integers(max(2, n // 200), max(3, n // 12)))
            start = rng.integers(0, max(1, n - width + 1))
            stop = min(n, start + width)
            factor = np.exp(rng.normal(0.0, 0.12))

            inds = np.arange(start, stop, dtype=np.int64)
            old_vals = a[inds].copy()
            new_vals = old_vals * factor
            deltas = new_vals - old_vals
            if np.allclose(deltas, 0.0):

            for offset, idx in enumerate(inds):
                d = deltas[offset]
                if d == 0.0:

                conv_a[idx: idx + n] += 2.0 * d * a

            L = len(inds)
            for i in range(L):
                ii = inds[i]
                di = deltas[i]
                for j in range(i, L):
                    ij = inds[j]
                    dj = deltas[j]
                    if i == j:
                        conv_a[2 * ii] += di * dj
                    else:
                        conv_a[ii + ij] += 2.0 * di * dj

            # apply the scaling
            a[inds] = new_vals
            np.clip(a, 0.0, 1000.0, out=a)

            # renormalise to keep the total sum exactly S_target
            total = a.sum()
            if total > 0.0:
                a *= (S_target / total)
                np.clip(a, 0.0, 1000.0, out=a)
            # after a global scaling the simple incremental update is no longer exact
            conv_a = _conv_self(a)

        # --------------------------------------------------------------
        # 3) Small additive noise – perturbs the whole vector a little.
        # --------------------------------------------------------------
        else:
            noise = rng.normal(0.0, 0.015, size=n)
            a += noise
            np.clip(a, 0.0, 1000.0, out=a)
            # re‑normalise the total sum
            total = a.sum()
            if total > 0.0:
                a *= (S_target / total)
                np.clip(a, 0.0, 1000.0, out=a)
            conv_a = _conv_self(a)

        # ------------------------------------------------------------------
        # Periodically recompute the convolution from scratch to avoid drift.
        # ------------------------------------------------------------------
        if iteration % reset_interval == 0:
            conv_a = _conv_self(a)

        # ------------------------------------------------------------------
        # Evaluate current candidate (fast, uses the maintained convolution).
        # ------------------------------------------------------------------
        total = a.sum()
        if total < 0.01:
            # sanity – should never happen
            continue
        cur_score = _current_score(conv_a, total)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    # ------------------------------------------------------------------
    # Return best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(v) for v in best_a]
