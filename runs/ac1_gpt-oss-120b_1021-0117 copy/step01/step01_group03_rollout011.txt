def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    else:
        n = 5582
        parent = None

    # precompute index array
    idx = np.arange(n, dtype=np.float64)

    # function to compute score with FFT (fast)
    def fast_score(arr):
        s = arr.sum()
        if s < 0.01:
            return float('inf')
        n_local = arr.size
        m = 1 << ((2*n_local - 1).bit_length())
        fa = np.fft.rfft(arr, n=m)
        conv = np.fft.irfft(fa*fa, n=m)[:2*n_local-1]
        max_b = conv.max()
        return 2.0*n_local*max_b / (s*s)

    best_seq = None
    best_score = float('inf')
    
    # consider parent
    if parent is not None:
        cand_score = fast_score(parent)
        if np.isfinite(cand_score) and cand_score < best_score:
            best_seq = parent.copy()
            best_score = cand_score

    # baseline geometric with r0
    r0 = 1.0 - 2.0 / n
    seq0 = np.power(r0, idx)
    cand_score = fast_score(seq0)
    if np.isfinite(cand_score) and cand_score < best_score:
        best_seq = seq0
        best_score = cand_score

    # search over t = 1 - r (small)
    # choose logspace range
    lo = max(1e-7, 0.1 * 2.0 / n)   # 10% of 2/n
    hi = max(lo*10, 5.0 * 2.0 / n)  # up to 5 times 2/n
    # create 80 samples log spaced
    num_samples = 80
    t_vals = np.logspace(np.log10(lo), np.log10(hi), num=num_samples)
    for t in t_vals:
        if time.monotonic() >= deadline:
            break
        r = 1.0 - t
        if r <= 0.0:
            continue
        seq = np.power(r, idx)
        sc = fast_score(seq)
        if sc < best_score:
            best_seq = seq
            best_score = sc

    # small random perturbations / local search
    # Use simple move: swap some mass
    early_cut = n // 4
    late_cut = 3 * n // 4
    while time.monotonic() < deadline:
        cand = best_seq.copy()

        if op < 0.6:
            src = rng.integers(0, early_cut)

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution via real FFT (linear convolution a*a)
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Fast evaluation of the AC1 upper bound (identical to the official one
# except using the FFT based convolution for speed)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = a.size
    max_b = _conv_self_fft(a).max()
    return 2.0 * n * max_b / (s * s)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a low‑score non‑negative step function.
    Returns a list of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Determine problem size and optional parent candidate.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    else:
        n = 5582               # fallback length used in the reference solution
        parent = None

    idx = np.arange(n, dtype=np.float64)   # reused index vector

    # ------------------------------------------------------------------
    # Initialise best solution with parent (if any) and a geometric shape.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')

    if parent is not None:
        sc = _fast_score(parent)
        if np.isfinite(sc) and sc < best_score:
            best_seq = parent.copy()
            best_score = sc

    # Geometric sequence a_i = r^i (r < 1).  Scaling is irrelevant for the score.
    r0 = 1.0 - 2.0 / n                     # analytic guess from the continuous model
    seq0 = np.power(r0, idx)
    sc = _fast_score(seq0)
    if np.isfinite(sc) and sc < best_score:
        best_seq = seq0
        best_score = sc

    # ------------------------------------------------------------------
    # One‑dimensional search over the decay factor r (equivalently t = 1‑r).
    # The optimum lies near t ≈ 2/n, we explore a log‑spaced neighbourhood.
    # ------------------------------------------------------------------
    t_target = 2.0 / n
    lo = max(1e-7, 0.1 * t_target)
    hi = max(lo * 10, 5.0 * t_target)
    t_vals = np.logspace(np.log10(lo), np.log10(hi), num=80)

    for t in t_vals:
        if time.monotonic() >= deadline:
            break
        r = 1.0 - t
        if r <= 0.0:
            continue
        cand = np.power(r, idx)
        sc = _fast_score(cand)
        if sc < best_score:
            best_seq = cand
            best_score = sc

    # ------------------------------------------------------------------
    # Light‑weight stochastic local search to polish the best candidate.
    # ------------------------------------------------------------------
    early_cut = n // 4
    late_cut = 3 * n // 4
    max_iters = 25000

    # ------------------------------------------------------------------
