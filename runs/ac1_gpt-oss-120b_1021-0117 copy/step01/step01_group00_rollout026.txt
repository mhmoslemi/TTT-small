import numpy as np
import time

# ------------------------------------------------------------
# Helper to apply an additive delta to a single index and update
# the convolution array in O(n) time.
def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float):
    """In‑place update of array a and its full self‑convolution conv.
    a[idx] changes by `delta`.  conv is length 2*len(a)-1.
    """
    if delta == 0.0:
        return
    n = a.shape[0]
    # add 2*delta * a (old values) to the slice where index sum = idx + j
    conv[idx : idx + n] += 2.0 * delta * a
    # correct the self‑product term (appears only once in the slice)
    conv[2 * idx] += delta * delta
    # finally update the coefficient itself
    a[idx] += delta


# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a candidate sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    # reserve a little time for finalisation
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

        parent = np.clip(parent, 0.0, 1000.0)

    # ------------------------------------------------------------------
