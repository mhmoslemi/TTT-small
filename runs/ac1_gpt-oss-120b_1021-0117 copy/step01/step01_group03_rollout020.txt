import numpy as np
import time
import math

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using real FFT (O(n log n))."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the authoritative evaluator."""
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence of length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


def _grad_max_convolution(a: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the maximal convolution entry (index k) w.r.t. a."""
    n = a.size
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    i = np.arange(i_start, i_end + 1, dtype=int)
    j = k - i
    # where i == j the term appears once, otherwise twice
    mask = i == j
    g[i[mask]] = a[i[mask]]
    g[i[~mask]] = 2.0 * a[j[~mask]]
    return g
