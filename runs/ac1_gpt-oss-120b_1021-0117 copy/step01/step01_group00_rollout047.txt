def conv_max(seq):
    n = seq.shape[0]
    L = 1 << ((2*n-2).bit_length())  # next pow of 2 >= 2n-1
    fft = np.fft.rfft(seq, n=L)
    prod = fft * fft
    conv = np.fft.irfft(prod, n=L)
    # keep only first 2*n-1 entries
    return np.max(conv[:2*n-1])
```

import numpy as np
import time
import math

def _sidon_like(n: int) -> np.ndarray:
    """Sidon‑like sequence a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def _power_seq(n: int, alpha: float) -> np.ndarray:
    """a_i = (i+1)^(-alpha)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-alpha)


def _alternating(vals: np.ndarray) -> np.ndarray:
    """Place largest values alternately at the leftmost and rightmost positions."""
    n = vals.shape[0]
    out = np.empty_like(vals)
    left, right = 0, n - 1
    k = 0
    while left <= right:
        out[left] = vals[k]
        k += 1
        left += 1
        if left <= right:
            out[right] = vals[k]
            k += 1
            right -= 1
    return out


def _conv_max(seq: np.ndarray) -> float:
    """Maximum of the full auto‑convolution using an FFT (real‑to‑real)."""
    n = seq.shape[0]
    L = 1 << ((2 * n - 2).bit_length())      # next power of two ≥ 2n‑1
    fft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft * fft, n=L)
    return float(np.max(conv[: 2 * n - 1]))   # discard padded tail


def _score(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum²."""
    s = np.sum(seq)
    if s < 1e-12:
        return math.inf
    return 2.0 * seq.shape[0] * _conv_max(seq) / (s * s)

    # ------------------------------------------------------------------
    value_sets = [_sidon_like(n)]

                else:
                    # revert the swap
                    cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
                    stagnation += 1

                temp *= decay
                if temp < 1e-12:
                    temp = 1e-12

                # occasional random restart if stuck
                if stagnation > 3000:
                    cur_seq = vals.copy()
                    rng.shuffle(cur_seq)
                    cur_score = _score(cur_seq)
                    stagnation = 0

                # early quit a little before the absolute deadline
                if time.monotonic() > deadline - 0.05:
                    break

            if time.monotonic() > deadline - 0.05:
                break
        if time.monotonic() > deadline - 0.05:
            break

    # ------------------------------------------------------------------
    # final safety net – ensure a valid return
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(256, 1.0, dtype=np.float64)

    return [float(x) for x in best_seq.tolist()]
