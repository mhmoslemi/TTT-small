def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise Exception
        n = parent_arr.shape[0]
        # clip to bounds
        parent_arr = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        n = 256
        parent_arr = np.full(n, 1.0, dtype=np.float64)

    # Normalize to sum=1
    sum_parent = parent_arr.sum()
    if sum_parent == 0:
        init_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        init_seq = parent_arr / sum_parent

    # Also generate Sidon-like sequence as alternative
    def sidon_seq(m):
        a = np.empty(m, dtype=np.float64)
        a[0] = 1.0
        for i in range(1,m):
            a[i] = a[i-1] * (1.0 - 1.0/(2.0*i))
        return a

    sid_seq = sidon_seq(n)
    # Rescale to sum=1
    sid_seq = sid_seq / sid_seq.sum()

    # Evaluate both via our fast metric
    L = 1 << ((2*n-1).bit_length())
    def max_corr(seq):
        fft = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(fft*fft, n=L)[:2*n-1]
        return np.max(conv)

    # Choose initial sequence based on lower score
    score_parent = 2*n*max_corr(init_seq)
    score_sid = 2*n*max_corr(sid_seq)
    if score_sid < score_parent:
        current_seq = sid_seq.copy()
        best_score = score_sid
    else:
        current_seq = init_seq.copy()
        best_score = score_parent

    best_seq = current_seq.copy()

    # Simulated annealing parameters
    temperature = 1.0
    cooling_rate = 0.9995  # slower cooling for many iterations
    min_temp = 1e-4

    # Determine a base step size proportion of typical value
    # Typical value magnitude ~ 1/n? Actually sum=1, average =1/n ~ 0.00014, but values vary.
    # We'll set base_step as 0.02 (2% of total sum) which ensures delta not too large relative to values.
    base_step = 0.02

    # Main loop
    while time.monotonic() < deadline and temperature > min_temp:
        # Propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        step = temperature * base_step
        # Determine feasible delta
        # delta can be added to i, sub from j
        delta = rng.uniform(-step, step)
        # clip to feasible intervals
        # lower bound: -current_seq[i] (cannot go below 0)
        # upper bound: current_seq[j] (cannot exceed due to conservation)
        lo = -current_seq[i]
        hi = current_seq[j]
        if delta < lo:
            delta = lo
        elif delta > hi:
            delta = hi
        if delta == 0.0:
            continue

        cand_seq = current_seq.copy()
        cand_seq[i] += delta
        cand_seq[j] -= delta

        # compute score
        cand_max = max_corr(cand_seq)
        cand_score = 2*n * cand_max

        # accept?
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
        # Metropolis acceptance
        if cand_score <= 2*n * max_corr(current_seq) or rng.random() < math.exp(-(cand_score - 2*n * max_corr(current_seq))/temperature):

        temperature *= cooling_rate

    # Ensure final sequence respects bounds
    final_seq = np.clip(best_seq, 0.0, 1000.0)

    # Ensure sum >= 0.01; if sum too small, scale up proportionally (maintaining shape)
    s = final_seq.sum()
    if s < 0.01:
        # Scale to 0.01
        final_seq = final_seq * (0.01 / s)

    # Convert to list of floats
    return [float(x) for x in final_seq.tolist()]

import numpy as np
import time
import math

def sidon_like_sequence(n):
    """Classic a_i = C(2i,i)/4**i (unnormalized)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a

def max_corr(seq, L):
    """Maximum value of the linear convolution seq*seq using FFT."""
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[:2 * seq.shape[0] - 1]
    return float(conv.max())

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 1. Load or build a parent sequence
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise Exception
        n = int(parent_arr.shape[0])
        parent_arr = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        n = 256
        parent_arr = np.full(n, 1.0, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2. Normalise candidates to have sum == 1 (scale–invariant objective)
    # ------------------------------------------------------------------
    def normalise(v):
        s = v.sum()
        return v / s if s > 0 else np.full_like(v, 1.0 / v.size)

    parent_norm = normalise(parent_arr)
    sidon = sidon_like_sequence(n)
    sidon_norm = normalise(sidon)

    # ------------------------------------------------------------------
    # 3. Choose the better of the two as starting point
    # ------------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())        # FFT length (power of two)
    score_parent = 2.0 * n * max_corr(parent_norm, L)
    score_sidon = 2.0 * n * max_corr(sidon_norm, L)

    if score_sidon < score_parent:
        current_seq = sidon_norm.copy()
        best_score = score_sidon
    else:
        current_seq = parent_norm.copy()
        best_score = score_parent

    best_seq = current_seq.copy()
    curr_score = best_score

    # ------------------------------------------------------------------
    # 4. Simulated annealing – transfer a small amount between two
    #    coordinates while preserving the total sum.
    # ------------------------------------------------------------------
    temperature = 1.0
    cooling_rate = 0.9995
    min_temp = 1e-4
    base_step = 0.02          # relative to total sum (which equals 1)

    while time.monotonic() < deadline and temperature > min_temp:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        step = temperature * base_step
        low = max(-current_seq[i], -step)
        high = min(current_seq[j], step)
        if low >= high:
            continue
        delta = rng.uniform(low, high)

        cand_seq = current_seq.copy()
        cand_seq[i] += delta
        cand_seq[j] -= delta

        cand_score = 2.0 * n * max_corr(cand_seq, L)

        # keep the best ever seen
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()

        # Metropolis acceptance
        if cand_score < curr_score or rng.random() < math.exp(-(cand_score - curr_score) / temperature):
            current_seq = cand_seq
            curr_score = cand_score

        temperature *= cooling_rate

    # ------------------------------------------------------------------
    # 5. Light-weight local refinement using swaps (sum‑preserving)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cand_seq = best_seq.copy()
        cand_seq[i], cand_seq[j] = cand_seq[j], cand_seq[i]
        cand_score = 2.0 * n * max_corr(cand_seq, L)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq

    # ------------------------------------------------------------------
    # 6. Final safety checks and return
    # ------------------------------------------------------------------
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = final_seq.sum()
    if s < 0.01:                     # enforce evaluator's lower bound
        final_seq = final_seq * (0.01 / s)

    return [float(x) for x in final_seq.tolist()]
