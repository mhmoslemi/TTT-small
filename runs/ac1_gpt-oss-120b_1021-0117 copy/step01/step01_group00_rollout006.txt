def fast_score(seq):
    n = len(seq)
    # zero pad to length m = 2*n - 1
    m = 2*n - 1
    # compute next power of 2 >= m for efficient FFT
    mfft = 1 << (m - 1).bit_length()
    # using rfft
    f = np.fft.rfft(seq, n=mfft)
    conv = np.fft.irfft(f * f, n=mfft)[:m]   # take first m entries
    maxb = np.max(conv)
    sum_a = np.sum(seq)
    return 2 * n * maxb / (sum_a**2)

def _sidon_like_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (1.0 - 1.0/(2.0*i))

def _sidon_like_sequence(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4**i, length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _score_and_conv(seq: np.ndarray):
    """
    Return (score, full convolution) where
    score = 2 * n * max(conv) / (sum(seq) ** 2) .
    Uses FFT (matrix‑free, O(n log n)).
    """
    n = seq.shape[0]
    m = 2 * n - 1                     # length of linear convolution
    mfft = _next_pow2(m)               # zero‑pad size (power of two)
    f = np.fft.rfft(seq, n=mfft)       # FFT of zero‑padded sequence
    conv = np.fft.irfft(f * f, n=mfft)[:m]   # linear convolution
    maxb = np.max(conv)
    s = np.sum(seq)
    if s == 0:
        return np.inf, conv
    score = 2.0 * n * maxb / (s * s)
    return float(score), conv


def _fast_score(seq: np.ndarray) -> float:
    """Score only (no convolution returned)."""
    sc, _ = _score_and_conv(seq)
    return sc


def _exact_score(seq: np.ndarray) -> float:
    """Use the authoritative evaluator (pre‑imported)."""
    try:
        return float(evaluate_sequence(seq.tolist()))
    except Exception:
        return np.inf

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # Determine length and an initial parent sequence (if supplied)
    # --------------------------------------------------------------
    try:
        parent = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64),
            0.0,
            1000.0,
        )
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        n = int(parent.shape[0])
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    # --------------------------------------------------------------
    # Initialise best candidate (parent + a few deterministic families)
    # --------------------------------------------------------------
    best_seq = parent.copy()
    best_score, _ = _score_and_conv(best_seq)
    # check exact version – it also forces the clipping that the evaluator does
    exact = _exact_score(best_seq)
    if np.isfinite(exact):
        best_score = exact

    # ----- Sidon‑like -------------------------------------------------
    sidon = _sidon_like_sequence(n)
    sidon = np.clip(sidon, 0.0, 1000.0)
    sidon /= sidon.sum()
    sc = _fast_score(sidon)
    if sc < best_score:
        best_seq = sidon.copy()
        best_score = sc

    # ----- Exponential families ----------------------------------------
    for r in np.linspace(0.90, 0.9999, 30):
        seq = np.power(r, np.arange(n, dtype=np.float64))
        seq /= seq.sum()
        sc = _fast_score(seq)
        if sc < best_score:
            best_seq = seq.copy()
            best_score = sc

    # ----- Power‑law families -------------------------------------------
    for p in np.linspace(0.1, 2.5, 30):
        idx = np.arange(1, n + 1, dtype=np.float64)
        seq = idx ** (-p)
        seq /= seq.sum()
        sc = _fast_score(seq)
        if sc < best_score:
            best_seq = seq.copy()
            best_score = sc

    # ----- Gaussian families --------------------------------------------
    centre = (n - 1) / 2.0
    for sf in np.linspace(0.05, 0.4, 12):
        sigma = sf * n
        seq = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        seq /= seq.sum()
        sc = _fast_score(seq)
        if sc < best_score:
            best_seq = seq.copy()
            best_score = sc

    # ----- Linear mix of Sidon and an exponential ----------------------
    exp_seq = np.power(0.95, np.arange(n, dtype=np.float64))
    exp_seq /= exp_seq.sum()
    for lam in np.linspace(0.0, 1.0, 11):
        mix = lam * sidon + (1.0 - lam) * exp_seq
        mix /= mix.sum()
        sc = _fast_score(mix)
        if sc < best_score:
            best_seq = mix.copy()
            best_score = sc

    # --------------------------------------------------------------
    # Local optimisation – hill climbing with targeted moves
    # --------------------------------------------------------------
    current = best_seq.copy()
    curr_score, conv = _score_and_conv(current)

    # Settings for the stochastic search
    prob_targeted = 0.35          # fraction of iterations that are “guided”
    move_fraction = 0.05          # up‑to‑5 % of the source entry may be moved
    restart_prob = 0.001          # occasional random perturbation from best

    while time.monotonic() < deadline:
        # ---- guided move -------------------------------------------------
        if rng.random() < prob_targeted:
            # index of the maximal convolution entry
            d = int(np.argmax(conv))
            # valid overlapping index range for this lag
            i0 = max(0, d - (n - 1))
            i1 = min(n - 1, d)
            idx = np.arange(i0, i1 + 1, dtype=np.int64)
            jdx = d - idx
            # products a_i * a_j that realise the max value
            prods = current[idx] * current[jdx]
            max_loc = int(np.argmax(prods))
            i_src = idx[max_loc]
            j_src = jdx[max_loc]
            # choose the larger of the two as the donor
            src = i_src if current[i_src] >= current[j_src] else j_src
            # pick a random receiver distinct from src
            dst = rng.integers(0, n)
            while dst == src:
                dst = rng.integers(0, n)
            delta = current[src] * rng.random() * move_fraction
            if delta <= 0.0:
                continue
            cand = current.copy()
            cand[src] -= delta
            cand[dst] += delta
        else:
            # ---- simple random transfer -----------------------------------
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            if current[i] <= 0.0:
                continue
            delta = current[i] * rng.random() * move_fraction
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta

        # occasional global perturbation (restart)
        if rng.random() < restart_prob:
            cand = best_seq + rng.normal(scale=0.001, size=n)
            cand = np.clip(cand, 0.0, None)
            total = np.sum(cand)
            if total > 0.0:
                cand /= total

        # evaluate candidate
        cand_score, cand_conv = _score_and_conv(cand)

        # accept only improvements (monotone descent)
        if cand_score < curr_score:
            current, curr_score, conv = cand, cand_score, cand_conv
            if cand_score < best_score:
                best_seq, best_score = cand.copy(), cand_score
                # early exit if we beat the target bound
                if best_score < 1.5028:
                    break

    # ------------------------------------------------------------------
    # Final validation with the authoritative evaluator
    # ------------------------------------------------------------------
    final_val = _exact_score(best_seq)
    if not np.isfinite(final_val):
        # Fallback to parent (guaranteed finite)
        best_seq = parent.copy()

    # Return Python list of floats
    return [float(x) for x in best_seq]
