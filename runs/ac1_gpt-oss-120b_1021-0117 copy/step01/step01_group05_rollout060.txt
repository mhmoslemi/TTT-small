def _score(seq):
    n = seq.shape[0]
    m = 1 << ((2*n-1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f*f, n=m)[:2*n-1]
    max_val = conv.max()
    sum_val = seq.sum()
    if sum_val <= 0:
        return np.inf, conv, max_val
    score = 2.0 * n * max_val / (sum_val * sum_val)
    return float(score), conv, max_val
```

import numpy as np, time, math

def _score_seq(seq):
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]
    max_val = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float('inf'), conv, max_val
    return 2.0 * n * max_val / (s * s), conv, max_val

def _binomial_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2.0 * i - 1.0) / (2.0 * i)
    return a

def _power_law_seq(n, p):
    i = np.arange(1, n+1, dtype=np.float64)
    return i ** (-p)

def _flattened_binomial_seq(n, gamma):
    b = _binomial_seq(n)
    return np.power(b, gamma)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Determine length and optional parent
    n = 3809
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
    except Exception:
        parent = None

    # Helper to evaluate sequence and return score
    def fast_score(seq):
        return _score_seq(seq)

    # Build a set of starting sequences
    seeds = []
    seeds.append(_binomial_seq(n))
    seeds.append(_flattened_binomial_seq(n, 0.8))
    seeds.append(_flattened_binomial_seq(n, 0.6))
    for p in (0.45, 0.5, 0.55):
        seeds.append(_power_law_seq(n, p))
    # uniform
    seeds.append(np.full(n, 1.0, dtype=np.float64))

    if parent is not None:
        seeds.append(parent.copy())

    best_seq = None
    best_score = float('inf')
    best_conv = None

    # Evaluate all seed candidates
    for s in seeds:
        sc, conv, _ = fast_score(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()
            best_conv = conv

    # Current state for simulated annealing
    cur_seq = best_seq.copy()
    cur_score = best_score
    cur_conv = best_conv

    # Temperature schedule
    init_temp = max(0.5, best_score)  # start with half the best score or at least 0.5
    final_temp = 1e-5
    start_time = time.monotonic()

    def mutate(seq, conv_arr):
        """Return a mutated copy of seq."""
        nloc = seq.shape[0]
        cand = seq.copy()
        op = rng.integers(0, 10)
        if op == 0:
            # weighted transfer from a high value to a low value
            total = cand.sum()
            if total > 0.0:
                probs = cand / total
                src = rng.choice(nloc, p=probs)
                inv = 1.0 / (cand + 1e-12)
                tgt_probs = inv / inv.sum()
                tgt = rng.choice(nloc, p=tgt_probs)
                if src != tgt:
                    alpha = rng.uniform(0.02, 0.2)
                    delta = cand[src] * alpha
                    cand[src] -= delta
                    cand[tgt] += delta
        elif op == 1:
            # block scaling
            width = rng.integers(1, max(2, nloc // 30))
            start = rng.integers(0, nloc - width + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            cand[start:start+width] *= factor
        elif op == 2:
            # single element scaling
            idx = rng.integers(0, nloc)
            factor = np.exp(rng.normal(0.0, 0.2))
            cand[idx] *= factor
        elif op == 3:
            # smooth neighbor pair
            i = rng.integers(0, nloc-1)
            avg = (cand[i] + cand[i+1]) * 0.5
            cand[i] = avg
            cand[i+1] = avg
        elif op == 4:
            # block averaging (preserves sum)
            width = rng.integers(2, max(4, nloc // 20))
            start = rng.integers(0, nloc - width + 1)
            avg = cand[start:start+width].mean()

        elif op == 5:

                lo = max(0, k_max - (nloc - 1))
                hi = min(k_max, nloc - 1)
                i = rng.integers(lo, hi + 1)
                j = k_max - i
                if i != j:
                    alpha = rng.uniform(0.04, 0.12)
                    delta_i = cand[i] * alpha
                    delta_j = cand[j] * alpha
                    cand[i] -= delta_i
                    cand[j] -= delta_j
                    total = delta_i + delta_j
                    tgt = rng.integers(0, nloc)

        elif op == 8:

        np.clip(cand, 0.0, 1000.0, out=cand)
        return cand

    # main SA loop
    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        # progress
        prog = (now - start_time) / (deadline - start_time)
        temp = init_temp * ((final_temp / init_temp) ** prog)

        candidate = mutate(cur_seq, cur_conv)
        cand_score, cand_conv, _ = fast_score(candidate)

        # Acceptance relative to current
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            cur_seq = candidate
            cur_score = cand_score
            cur_conv = cand_conv

        # Update global best
        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate
            best_conv = cand_conv

    # Ensure we have a valid answer
    if best_seq is None or not np.isfinite(best_score):
        best_seq = _binomial_seq(n)

    return [float(x) for x in best_seq]

import numpy as np
import time
import math

def _score_seq(seq: np.ndarray):
    """Return (score, convolution array, max value) for a non‑negative sequence."""
    n = seq.shape[0]
    # Length for a power‑of‑two FFT that covers the full convolution.
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float("inf"), conv, max_val
    return 2.0 * n * max_val / (s * s), conv, max_val


def _binomial_seq(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i  (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def _power_law_seq(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p)


def _flattened_binomial_seq(n: int, gamma: float) -> np.ndarray:
    """a_i = (binomial_i)^{gamma}, 0<gamma<1 flattens the shape."""
    b = _binomial_seq(n)
    return np.power(b, gamma)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic search (simulated‑annealing style) for a step‑function that
    minimises the AC1 bound.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Problem size and optional parent supplied by the harness
    # ------------------------------------------------------------------
    n = 3809
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
    except Exception:
        parent = None

    # ------------------------------------------------------------------
    # Assemble a few diverse seed sequences
    # ------------------------------------------------------------------
    seeds = [
        _binomial_seq(n),
        _flattened_binomial_seq(n, 0.8),
        _flattened_binomial_seq(n, 0.6),
        _power_law_seq(n, 0.45),
        _power_law_seq(n, 0.5),
        _power_law_seq(n, 0.55),
        np.full(n, 1.0, dtype=np.float64),  # uniform
    ]
    if parent is not None:
        seeds.append(parent.copy())

    best_seq = None
    best_score = float("inf")
    best_conv = None

    for s in seeds:
        sc, conv, _ = _score_seq(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()
            best_conv = conv

    # ------------------------------------------------------------------
    # Simulated‑annealing core: current state and global best
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score
    cur_conv = best_conv

    init_temp = max(0.5, best_score)   # start fairly warm
    final_temp = 1e-5
    start_time = time.monotonic()

    # --------------------------------------------------------------
    # Mutation operator (uses current convolution for a focused move)
    # --------------------------------------------------------------
    def mutate(seq: np.ndarray, conv_arr: np.ndarray) -> np.ndarray:
        nloc = seq.shape[0]
        cand = seq.copy()
        op = rng.integers(0, 10)

        if op == 0:
            # transfer a fraction from a high‑value index to a low‑value one
            total = cand.sum()
            if total > 0.0:
                probs = cand / total
                src = rng.choice(nloc, p=probs)
                inv = 1.0 / (cand + 1e-12)
                tgt_probs = inv / inv.sum()
                tgt = rng.choice(nloc, p=tgt_probs)
                if src != tgt:
                    alpha = rng.uniform(0.02, 0.20)
                    delta = cand[src] * alpha
                    cand[src] -= delta
                    cand[tgt] += delta

        elif op == 1:
            # scale a random block
            width = rng.integers(1, max(2, nloc // 30))
            start = rng.integers(0, nloc - width + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            cand[start:start + width] *= factor

        elif op == 2:
            # scale a single element
            idx = rng.integers(0, nloc)
            factor = np.exp(rng.normal(0.0, 0.20))
            cand[idx] *= factor

        elif op == 3:
            # smooth two neighbours jointly
            i = rng.integers(0, nloc - 1)
            avg = (cand[i] + cand[i + 1]) * 0.5
            cand[i] = avg
            cand[i + 1] = avg

        elif op == 4:
            # replace a block by its average (preserves sum)
            width = rng.integers(2, max(4, nloc // 20))
            start = rng.integers(0, nloc - width + 1)
            avg = cand[start:start + width].mean()
            cand[start:start + width] = avg

        elif op == 5:
            # move a fraction away from the currently maximal convolution index
            if conv_arr is not None:
                k_max = int(np.argmax(conv_arr))
                lo = max(0, k_max - (nloc - 1))
                hi = min(k_max, nloc - 1)
                i = rng.integers(lo, hi + 1)
                j = k_max - i
                if i != j:
                    alpha = rng.uniform(0.04, 0.12)
                    di = cand[i] * alpha
                    dj = cand[j] * alpha
                    cand[i] -= di
                    cand[j] -= dj
                    total = di + dj
                    tgt = rng.integers(0, nloc)
                    while tgt == i or tgt == j:
                        tgt = rng.integers(0, nloc)
                    cand[tgt] += total

        elif op == 6:
            # swap two entries
            i, j = rng.choice(nloc, size=2, replace=False)
            cand[i], cand[j] = cand[j], cand[i]

        elif op == 7:
            # redistribute mass among a few randomly chosen indices
            k = rng.integers(2, min(20, nloc // 10) + 1)
            idxs = rng.choice(nloc, size=k, replace=False)
            tot = cand[idxs].sum()
            if tot > 0.0:
                proportions = rng.random(k)
                proportions /= proportions.sum()
                cand[idxs] = tot * proportions

        elif op == 8:
            # global multiplicative noise
            factor = np.exp(rng.normal(0.0, 0.12, size=nloc))
            cand *= factor

        else:
            # zero a short block
            width = rng.integers(1, max(2, nloc // 50))
            start = rng.integers(0, nloc - width + 1)
            cand[start:start + width] = 0.0

        np.clip(cand, 0.0, 1000.0, out=cand)
        return cand

    # --------------------------------------------------------------
    # Main simulated‑annealing loop
    # --------------------------------------------------------------
    while True:
        now = time.monotonic()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        temp = init_temp * ((final_temp / init_temp) ** prog)

        cand = mutate(cur_seq, cur_conv)
        cand_score, cand_conv, _ = _score_seq(cand)

        # accept according to SA criterion
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            cur_seq = cand
            cur_score = cand_score
            cur_conv = cand_conv

        # update global best
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            best_conv = cand_conv

    # ------------------------------------------------------------------
    # Fallback in case something went wrong (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None or not np.isfinite(best_score):
        best_seq = _binomial_seq(n)

    # Return a plain Python list – the evaluator will re‑evaluate it exactly.
    return [float(x) for x in best_seq]
