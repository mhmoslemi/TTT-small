        parent = None
        have_parent = False

    target_sum = math.sqrt(2.0 * n)          # the usual normalisation
    L = _next_pow2(2 * n - 1)                # FFT size (power of two)

    # ---- initial vector ------------------------------------------------
    cur = _gen_binomial(n)
    cur *= target_sum / cur.sum()            # normalise to target_sum
    if have_parent:
        p = parent.copy()
        p_sum = p.sum()
        if p_sum > 0:
            p *= target_sum / p_sum
        # keep the better of the two as the start
        # (both have the same normalisation, so compare max‑convolution)
        fa = np.fft.rfft(p, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        p_score = 2.0 * n * conv.max() / (p.sum() ** 2)
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max() / (cur.sum() ** 2)
        if p_score < cur_score:
            cur = p

    best = cur.copy()
    # compute the initial bound (max‑convolution, because sum == sqrt(2n))
    fa = np.fft.rfft(best, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    best_score = 2.0 * n * conv.max() / (best.sum() ** 2)

    # ---- optimisation parameters ---------------------------------------
    step0 = 0.5                    # initial step magnitude
    beta = 0.9                     # momentum factor
    momentum = np.zeros_like(cur)
    it = 0
    # --------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution and bound
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        maxM = conv.max()
        cur_score = 2.0 * n * maxM / (cur.sum() ** 2)

        # record best ever seen
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score
            if best_score < 1.50274349:          # target achieved
                break

        # indices where the max is attained (allow tiny numerical slack)
        tol = maxM * 1e-12 + 1e-14
        max_idx = np.where(conv >= maxM - tol)[0]

        # sub‑gradient of the max‑convolution (averaged if several maxima)
        g = np.zeros_like(cur)
        idxs = np.arange(n)
        for k in max_idx:
            off = k - idxs
            mask = (off >= 0) & (off < n)
            g[mask] += 2.0 * cur[off[mask]]
        if max_idx.size:
            g /= max_idx.size

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # diminishing step size
        step = step0 / math.sqrt(it)

        # propose a new point and keep feasibility
        cand = cur - step * momentum
        np.clip(cand, 0.0, None, out=cand)

        # occasional tiny random shake to avoid stagnation
        if rng.random() < 0.01:
            noise = rng.normal(scale=0.01 * target_sum, size=n)
            cand += noise
            np.clip(cand, 0.0, None, out=cand)

        s = cand.sum()
        if s <= 0.0:                     # safeguard against collapse
            cand = _gen_binomial(n)
            cand *= target_sum / cand.sum()
        else:
            cand *= target_sum / s        # re‑normalise to the fixed L1 norm

        np.clip(cand, 0.0, 1000.0, out=cand)   # respect evaluator bounds

        cur = cand

    # final safety: recompute exact score via the official evaluator (optional)
    # (the list we return will be clipped again by the evaluator anyway)
    return [float(v) for v in best]
