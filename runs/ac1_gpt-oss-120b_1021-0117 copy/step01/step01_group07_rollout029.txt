import time
import numpy as np

def _max_conv_fft(x: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Fast max of the full linear convolution using FFT."""
    X = np.fft.rfft(x, fft_len)
    conv = np.fft.irfft(X * X, fft_len)[:conv_len]
    # Numerical noise may produce tiny negative values – clip for safety
    return float(conv.max())

def _simplex_project(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w >=0, sum=1}."""
    # Based on the algorithm by Wang & Carreira‑Perpinán (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    rho = ind[u - cssv / ind > 0].max()
    theta = cssv[rho - 1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    probability simplex.  It runs until the supplied time budget expires.
    """

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)  # safety margin

    # ------------------------------------------------------------------

    conv_len = 2 * n - 1

    cur = _simplex_project(cur)                  # enforce sum = 1

    # also try a pure uniform vector – often close to the optimum
