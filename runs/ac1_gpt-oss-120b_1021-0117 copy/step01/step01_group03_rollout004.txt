import numpy as np, time, math

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT (fast, O(n log n))."""
    n = a.size
    # next power of two >= 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (scale‑invariant)."""
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    Uses a targeted “mass‑transfer” move that explicitly attacks the lag
    contributing the current maximal autocorrelation, combined with occasional
    block‑scaling perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initialise: use provided parent if available, otherwise a sqrt‑series.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

        a[0] = 1.0
