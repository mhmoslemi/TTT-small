def propose_candidate(seed=..., budget_s=1100, **kwargs):
    import time, numpy as np
    # set deadline
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(seed)

    # get parent or default initial
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        parent = np.clip(parent, 0, 1000)
    except Exception:
        parent = np.ones(512, dtype=np.float64) # if none

    n = parent.size

    # initial candidate: normalized Catalan or normalized parent whichever better
    a = np.copy(parent)
    # Ensure sum not zero
    if a.sum() == 0:
        a = np.ones(n)
    # Normalize sum to 1 for stable scaling
    a = a / a.sum()

    # Compute baseline score
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Also try Catalan
    # compute Catalan weights length n
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i-1] * ((2*i)*(2*i-1)) / (i*i*4.0)
    w = w / w.sum()
    score_w = float(evaluate_sequence(w.tolist()))
    if score_w < best_score:
        best = w.copy()
        best_score = score_w

    # set a = best start
    a = best.copy()

    # gradient descent loop
    grad_frac = 0.05
    min_grad_frac = 1e-5

    # we also have parameter for random moves probability
    prob_random = 0.2

    while time.monotonic() < deadline:
        # compute convolution and max
        conv = np.convolve(a, a)
        max_val = conv.max()
        # early exit if meets target
        if best_score <= 1.50274349:
            break

        # Choose move type
        if rng.random() < prob_random:
            # random mass transfer
            i, j = rng.integers(n, size=2, endpoint=False)
            if a[i] > 0:
                eps = rng.random() * a[i] * 0.5
                cand = a.copy()
                cand[i] -= eps
                cand[j] += eps
            else:
                continue
        else:
            # gradient step based on active indices
            active = np.flatnonzero(conv == max_val)
            # compute gradient
            g = np.zeros_like(a)
            # for each active k
            for k in active:
                shift = k - np.arange(n)
                mask = (shift >= 0) & (shift < n)
                # add 2*a[shift[mask]]
                g[mask] += 2.0 * a[shift[mask]]
            max_g = g.max()
            if max_g <= 0:
                # fallback to random move
                continue
            step = grad_frac * (a.max() / max_g)
            # update candidate
            cand = a - step * g
            # ensure nonnegative
            cand = np.maximum(cand, 0.0)
            # renormalize (sum may have decreased due to clipping)
            s = cand.sum()
            if s <= 0:
                continue
            cand /= s

        # evaluate cand
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
            a = cand
            # increase step size up to some limit
            grad_frac = min(grad_frac * 1.2, 0.5)
            # possibly reduce random prob
            prob_random = max(prob_random * 0.99, 0.05)
            if best_score <= 1.50274349:
                break
        else:
            # not improved
            # reduce step
            grad_frac = max(grad_frac * 0.7, min_grad_frac)
            # maybe keep a unchanged

        # occasionally jitter (small multiplicative noise)
        if rng.random() < 0.02:
            jitter = np.exp(rng.normal(0.0, 0.01, size=n))
            cand_j = a * jitter
            cand_j = cand_j / cand_j.sum()
            cand_j_score = float(evaluate_sequence(cand_j.tolist()))
            if cand_j_score < best_score:
                best = cand_j
                best_score = cand_j_score
                a = cand_j

    # final fallback: return best as list

import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that minimises the
    AC1 upper bound.  The routine builds several analytically‑motivated
    candidates (uniform, Catalan, power‑law) and then runs a cheap
    gradient‑descent style refinement while respecting the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1.  Parent sequence (if provided) as a fallback initialisation
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(512, dtype=np.float64)   # safe default

    n = int(parent.size)

    # ------------------------------------------------------------------
    # 2.  Helper: score of a sequence (identical to evaluate_sequence)
    # ------------------------------------------------------------------
    def _score(seq: np.ndarray) -> float:
        s = seq.sum()
        if s <= 0.0:
            return np.inf
        conv = np.convolve(seq, seq)
        return 2.0 * seq.size * conv.max() / (s * s)

    # ------------------------------------------------------------------
    # 3.  Assemble a handful of analytic candidates
    # ------------------------------------------------------------------
    best = None
    best_score = np.inf

    # (a)   Normalised parent
    if parent.sum() > 0.0:
        cand = parent / parent.sum()
        sc = _score(cand)
        best, best_score = cand, sc

    # (b)   Catalan (central‑binomial) weights
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    w /= w.sum()
    sc = _score(w)
    if sc < best_score:
        best, best_score = w.copy(), sc

    # (c)   Power‑law families a_i ∝ (i+1)^{-p}
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1‑based indices
    p_list = [0.5, 0.55, 0.56, 0.57, 0.58, 0.60]
    for p in p_list:
        raw = idx ** (-p)
        cand = raw / raw.sum()
        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand.copy(), sc

    # start the descent from the best found so far
    a = best.copy()
    grad_frac = 0.05
    min_grad_frac = 1e-5

    # ------------------------------------------------------------------
    # 4.  Gradient‑descent style refinement (bounded by deadline)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # compute convolution and the current objective
        conv = np.convolve(a, a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val / (a.sum() ** 2)

        if cur_score < best_score:
            best, best_score = a.copy(), cur_score
            # early exit if we already beat the target
            if best_score <= 1.50274349:
                break

        # active indices where the maximum is attained
        act = np.flatnonzero(conv == max_val)
        # assemble the (sub‑)gradient of the active max‑convolution
        g = np.zeros_like(a)
        ar = a
        rng_idx = np.arange(n)
        for k in act:
            shift = k - rng_idx
            mask = (shift >= 0) & (shift < n)
            g[mask] += 2.0 * ar[shift[mask]]

        max_g = g.max()
        if max_g <= 0.0:
            # fallback: random mass transfer
            i, j = rng.integers(n, size=2, replace=False)
            if a[i] == 0.0:
                continue
            eps = rng.random() * a[i] * 0.5
            cand = a.copy()
            cand[i] -= eps
            cand[j] += eps
        else:
            # step proportional to the largest entry of a
            step = grad_frac * (a.max() / max_g)
            cand = a - step * g
            cand = np.maximum(cand, 0.0)

        # re‑normalise (ensures invariance of the objective)
        total = cand.sum()
        if total == 0.0:
            continue
        cand /= total

        # evaluate candidate (fast, no extra clipping)
        cand_sc = _score(cand)

        if cand_sc < best_score:
            a = cand
            best, best_score = cand.copy(), cand_sc
            grad_frac = min(grad_frac * 1.2, 1.0)   # accelerate after success
            continue
        else:
            # no improvement – shrink step size and occasionally jitter
            grad_frac = max(grad_frac * 0.7, min_grad_frac)
            if rng.random() < 0.03:
                jitter = np.exp(rng.normal(0.0, 0.015, size=n))
                cand_j = a * jitter
                cand_j /= cand_j.sum()
                cand_j_sc = _score(cand_j)
                if cand_j_sc < best_score:
                    a = cand_j
                    best, best_score = cand_j.copy(), cand_j_sc
                    grad_frac = min(grad_frac * 1.1, 0.5)

    # ------------------------------------------------------------------
    # 5.  Return the best sequence discovered (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
