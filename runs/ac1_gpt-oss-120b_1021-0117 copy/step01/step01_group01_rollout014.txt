def project_simplex(v):
    # projects v onto probability simplex (nonnegative, sum = 1)
    # from https://github.com/ashudeep/optimize
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all nonpositive
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1  # zero-index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Returns the convolution array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _score_fft(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Fast proxy for the authoritative evaluator (identical for non‑negative a).
    """
    conv = _conv_full_fft(a, fft_len)
    max_b = float(conv.max())
    s = float(a.sum())
    # sum is forced to be positive during the search
    return 2.0 * n * max_b / (s * s)


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x >= 0, sum x = 1}.
    Implements the algorithm of Duchi et al. (O(n log n)).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑d array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive; return the unit vector at the max component
        w = np.zeros_like(v)
        w[np.argmax(v)] = 1.0
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # force exact sum to 1 (numerical safety)
    w_sum = w.sum()
    if w_sum == 0:
        # fallback to uniform
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w
