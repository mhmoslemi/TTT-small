def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

    """Linear convolution a*a using a single FFT (real inputs)."""
    fa = np.fft.rfft(a, fft_n)
    conv = np.fft.irfft(fa * fa, fft_n)[:conv_len]
    return conv

    return 2.0 * n * max_val
