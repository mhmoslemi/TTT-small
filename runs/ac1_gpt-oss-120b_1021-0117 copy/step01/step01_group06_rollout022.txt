import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_candidates = np.where(rho_arr > 0)[0]
    if rho_candidates.size == 0:
        theta = 0.0
    else:
        rho = rho_candidates[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic “sqrt‑series” (a*a = 1+…+x^{2n-2}) – fallback initializer
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Fast AC1 bound (identical to evaluate_sequence)
# ------------------------------------------------------------
def _score(v: np.ndarray) -> float:
    n = v.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (v.sum() ** 2)
