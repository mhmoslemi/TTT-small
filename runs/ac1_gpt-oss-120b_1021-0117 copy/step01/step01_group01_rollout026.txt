def compute_score(a):
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    max_b = conv.max()
    s = a.sum()
    # avoid division by zero
    if s == 0:

import time
import numpy as np

# ---------- helper utilities ----------
def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def _score_fft(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact score used by the evaluator (fast via FFT)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    max_b = conv.max()
    s = a.sum()
    if s == 0.0:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))


def _make_shape(n: int, p: float, lam: float, shift: float) -> np.ndarray:
    """Return a shape a_i = (i+1+shift)^-p * exp(-lam*i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.power(i + 1.0 + shift, -p) * np.exp(-lam * i)
    # scaling does not affect the score, but keep numbers moderate
    a = np.clip(a, 0.0, 1000.0)
    return a
