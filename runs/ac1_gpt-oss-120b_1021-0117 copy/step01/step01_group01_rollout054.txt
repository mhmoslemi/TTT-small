import time
import numpy as np

def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 via recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full) using a single FFT pair."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _score_fft(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact score used by the official evaluator (sum=1 assumed)."""
    max_b = _conv_full_fft(a, fft_len).max()
    return float(2 * n * max_b)   # sum(a)=1 -> denominator = 1


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # Length of the problem (provided by the harness when available)
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    # ---- initial sequence ---------------------------------------------------
    a = _generate_binomial_seq(n)
    total = a.sum()
    if total <= 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / total                     # normalise sum to 1 (scale‑invariant)

    # Pre‑compute FFT size (next power of two)
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    cur = a.copy()
    cur_score = _score_fft(cur, n, fft_len)
    best = cur.copy()
    best_score = cur_score

    # ---- main optimisation loop --------------------------------------------
    # Greedy moves that reduce the dominant convolution entry.
    # At each step we locate the shift giving the current maximum,

    # that contributes the least to that shift, and keep the move if it

    no_improve = 0
    while time.monotonic() < deadline:

        conv_full = _conv_full_fft(cur, fft_len)
