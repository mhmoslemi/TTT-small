        # ensure non-negative and <=1000
        np.clip(candidate, 0.0, 1000.0, out=candidate)

            best = candidate
            best_score = cand_score
        # occasionally perform random perturbations to escape local minima
        if rng.random() < 0.05:

import time
import math
import numpy as np
from scipy import signal


def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fast_score(seq: np.ndarray):
    """Return (score, full convolution array, max value) using an FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # power‑of‑two length
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]   # full convolution a*a
    max_val = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_val
    return 2.0 * n * max_val / (s * s), conv, max_val


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step‑function that lowers the AC1 upper bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine problem size and possibly a provided parent sequence.
    # ------------------------------------------------------------------
    n = 3809
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)  # may be undefined
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = int(parent.size)
    except Exception:
        parent = None

    # ---------------------------------------------------------
    # Start from the best of the binomial seed and the parent.
    # ---------------------------------------------------------
    best = _initial_binomial(n)
    best_score, best_conv, _ = _fast_score(best)

    if parent is not None:
        p_score, p_conv, _ = _fast_score(parent)
        if p_score < best_score:
            best = parent.copy()
            best_score = p_score
            best_conv = p_conv

    # ------------------------------------------------------------------
    # Normalise the sum (the objective is homogeneous, this fixes the scale).
    # ------------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)          # any constant works – convenient choice
    cur_sum = best.sum()
    if cur_sum > 0:
        best *= target_sum / cur_sum
    best_score, best_conv, _ = _fast_score(best)   # recompute after scaling

    # ------------------------------------------------------------------
    # Parameters for the deterministic gradient step.
    # ------------------------------------------------------------------
    step = 0.005       # absolute magnitude before scaling by a normalised direction
    step_decay = 0.9995
    min_step = 1e-4

    n = best.size
    idx_for_grad = n - 1 - np.arange(n)     # indices to extract gradM from correlation

    # ------------------------------------------------------------------

    while time.monotonic() < deadline:
        # ---- Gradient of the maximal convolution value (M) ----------
        M = best_conv.max()
        atol = max(1e-12 * M, 1e-15)         # tolerant equality test
        mask = np.abs(best_conv - M) <= atol

        # correlation gives Σ_k mask[k] * best[k - i] = Σ_k mask[k] * a_{k-i}
        corr = signal.correlate(mask.astype(np.float64), best, mode='full', method='fft')
        gradM = 2.0 * corr[idx_for_grad]    # ∂M/∂a_i

        step = max(step * step_decay, min_step)

    # ------------------------------------------------------------------
    # Return the best sequence found (list of pure Python floats).
    # ------------------------------------------------------------------
    return [float(x) for x in best]
