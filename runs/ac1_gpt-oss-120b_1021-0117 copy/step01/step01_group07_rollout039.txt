        score = _score_from_max(max_val, n)
        if score < best_score:
            best = a0.copy()

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def _catalan_weights(n: int) -> np.ndarray:
    """First n Catalan-type weights w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _fft_size_for_len(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n."""
    m = 1
    while m < 2 * n:
        m <<= 1
    return m


def _max_conv_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum of the linear convolution a*a using FFT (size fft_len)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # convolution length = 2·n‑1, the rest are zero‑padding artefacts
    return conv[: 2 * a.shape[0] - 1].max()


def _score_from_max(max_val: float, n: int) -> float:
    """Score = 2·n·max_conv (the sum is normalised to 1)."""
    return 2.0 * n * max_val


def _improve_by_targeted_moves(
    a: np.ndarray,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    deadline: float,
    target: float,
) -> tuple[np.ndarray, float]:
    """
    Reduce max‑convolution by repeatedly moving a small fraction of mass
    from indices that contribute most to the current peak.
    Returns the best vector seen and its score.
    """
    best = a.copy()
    best_score = _score_from_max(_max_conv_fft(best, fft_len), n)

    # step‑size controls the transferred fraction of the source coordinate
    step_frac = 0.12
    min_step = 1e-4

    while time.monotonic() < deadline:
        # current convolution and the index where it attains the maximum
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        k_max = int(conv.argmax())

        # gradient of the active convolution term:
        # g[i] = 2·a[k_max‑i]  if 0≤k_max‑i<n else 0
        idx = k_max - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[idx[valid]]

        # source index: biggest contribution to the peak
        i_src = int(np.argmax(g))
        if g[i_src] == 0.0:
            break  # nothing to improve

        # size of the transfer (kept proportional to the source mass)
        eps = step_frac * a[i_src] * 0.5
        if eps <= 0.0:
            break

        # destination index – choose a random position different from source
        j_dst = rng.integers(0, n)
        while j_dst == i_src:
            j_dst = rng.integers(0, n)

        cand = a.copy()
        cand[i_src] -= eps
        cand[j_dst] += eps

        cand_score = _score_from_max(_max_conv_fft(cand, fft_len), n)

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            a = cand
            step_frac = min(step_frac * 1.2, 0.5)
            if best_score <= target:
                break
        else:
            step_frac = max(step_frac * 0.7, min_step)

        # occasional global jitter to escape plateaus
        if rng.random() < 0.018:
            jitter = np.exp(rng.normal(0.0, 0.019, size=n))
            cand2 = a * jitter
            cand2 /= cand2.sum()
            cand2_score = _score_from_max(_max_conv_fft(cand2, fft_len), n)
            if cand2_score < best_score:
                best = cand2.copy()
                best_score = cand2_score
                a = cand2
                if best_score <= target:
                    break

    return best, best_score
