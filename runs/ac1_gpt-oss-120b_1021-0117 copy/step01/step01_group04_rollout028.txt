move_block_scale(arr, rng, n):
    width = rng.integers(max(1, n // 300), max(2, n // 30))  # adapt block size
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.06))  # small scaling
    cand = arr.copy()
    cand[start:start+width] *= factor
    # clip
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

move_mass_transfer:
    i = rng.integers(0, n); j = rng.integers(0, n)
    while i == j: j = rng.integers(0, n)   # ensure distinct
    frac = rng.random() * 0.5  # up to 50%
    delta = arr[i] * frac
    cand = arr.copy()
    cand[i] -= delta
    cand[j] += delta
    np.clip(cand, 0, 1000, out=cand)
    return cand

import numpy as np, time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n-1 (recurrence)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound via FFT (same formula as official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _move_block_scale(arr: np.ndarray, rng, n: int) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor."""
    # block size between n/300 and n/30 (at least 1)
    lo = max(1, n // 300)
    hi = max(2, n // 30)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = math.exp(rng.normal(0.0, 0.07))  # modest perturbation
    cand = arr.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _move_mass_transfer(arr: np.ndarray, rng, n: int) -> np.ndarray:
    """Transfer a random fraction of mass from i to j."""
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    while j == i:
        j = int(rng.integers(0, n))
    frac = rng.random() * 0.45   # up to 45 % of source
    delta = arr[i] * frac
    cand = arr.copy()
    cand[i] -= delta
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _move_reduce_peak(arr: np.ndarray, conv: np.ndarray,
                      p: int, rng, n: int) -> np.ndarray:
    """
    Reduce the contribution to the current maximal convolution index p.
    Choose a source inside the support of p and a target outside,
    moving a modest amount of mass.
    """
    start = max(0, p - (n - 1))
    end = min(n - 1, p)
    if start > end:
        return _move_mass_transfer(arr, rng, n)

    # source inside support
    src = int(rng.integers(start, end + 1))

    # pick a target outside support
    if start > 0 and end < n - 1:
        if rng.random() < 0.5:
            tgt = int(rng.integers(0, start))
        else:
            tgt = int(rng.integers(end + 1, n))
    elif start > 0:
        tgt = int(rng.integers(0, start))
    elif end < n - 1:
        tgt = int(rng.integers(end + 1, n))
    else:
        # support covers whole array – fallback
        return _move_mass_transfer(arr, rng, n)

    frac = rng.random() * 0.5
    delta = arr[src] * frac
    cand = arr.copy()
    cand[src] -= delta
    cand[tgt] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
