import numpy as np

# ----------------------------------------------------------------------
# Helper: decreasing binomial coefficients (C(2i,i)/4^i)
# ----------------------------------------------------------------------
def _initial_binomial(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ----------------------------------------------------------------------
# Helper: FFT‑based convolution and AC1 score
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray):
    """
    Returns (score, full convolution array, max value).
    Score = 2 * n * max(conv) / (sum(seq)**2)
    """
    n = seq.shape[0]
    # next power of two covering full length 2n-1
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val
