    else:
        a = a / total
    
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    
    # function to compute conv array
    def full_conv(vec):
        fa = np.fft.rfft(vec, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return conv
    
    cur = a.copy()
    cur_conv = full_conv(cur)
    cur_max = float(cur_conv.max())
    cur_score = 2.0 * n * cur_max
    best = cur.copy()
    best_score = cur_score
    
    init_lr = 0.5
    min_lr = 1e-4
    # main loop
    it = 0
    while time.monotonic() < deadline:
        it += 1

        max_idx = int(np.argmax(cur_conv))
        # gradient
        idx = np.arange(n)
        mask = (max_idx - idx >= 0) & (max_idx - idx < n)
        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[max_idx - idx[mask]]
        # step size schedule
        lr = max(init_lr / np.sqrt(it), min_lr)
        # propose update
        new = cur - lr * g
        # clip negatives
        np.maximum(new, 0.0, out=new)
        s = new.sum()
        if s == 0.0:
            # fallback to uniform
            new = np.ones(n, dtype=np.float64) / n
        else:
            new /= s
        # compute new conv and score
        new_conv = full_conv(new)
        new_max = float(new_conv.max())
        new_score = 2.0 * n * new_max
        # accept if better
        if new_score < cur_score:
            cur = new
            cur_conv = new_conv
            cur_max = new_max
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
        else:

                    cand_conv = full_conv(cand)
                    cand_score = 2.0 * n * float(cand_conv.max())
                    if cand_score < cur_score:
                        cur = cand
                        cur_conv = cand_conv
                        cur_score = cand_score
                        if cand_score < best_score:
                            best = cand.copy()
                            best_score = cand_score
        # periodic check for time
        if it % 1000 == 0 and time.monotonic() > deadline:
            break
    # return best as list
    final = [float(x) for x in best]
    # optional final validation - call evaluate_sequence (ensures ok)
    try:
        _ = evaluate_sequence(final)
    except Exception:
        # fallback to initial
        final = [float(x) for x in a]
    return final

def _generate_binomial_seq(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 using recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the linear convolution a*a (full) via a single FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return float(conv[: 2 * a.size - 1].max())


def _full_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (real) via FFT; length = 2*len(a)-1."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at the index max_idx (d(conv[max])/da)."""
    n = a.shape[0]
    idx = np.arange(n)
    mask = (max_idx - idx >= 0) & (max_idx - idx < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[max_idx - idx[mask]]
    return g
