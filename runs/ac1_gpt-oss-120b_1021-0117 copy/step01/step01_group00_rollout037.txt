def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # margin for returning
    
    # Determine n and base parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        n = int(parent.shape[0])
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)
    
    # Helper functions
    def conv_max(seq):
        # compute max of linear convolution using FFT
        m = 1 << ((2*n - 1).bit_length())
        # Use real FFT
        fft = np.fft.rfft(seq, n=m)
        prod = fft * fft
        conv = np.fft.irfft(prod, n=m)
        # linear convolution first 2n-1 entries
        return float(conv[:2*n-1].max())
    
    def score(seq):
        s = float(seq.sum())
        if s < 0.01:
            return float('inf')
        maxb = conv_max(seq)
        return 2.0 * n * maxb / (s * s)
    
    # Ensure sequence within bounds
    def clip(seq):
        np.clip(seq, 0.0, 1000.0, out=seq)
        # Rescale if sum too small
        s = seq.sum()
        if s < 0.01:
            seq *= (0.01 / (s + 1e-12))
        return seq
    
    # Record best
    best_seq = np.copy(parent)
    best_score = score(best_seq)
    
    # Baseline: parent maybe already good, ensure clip
    best_seq = clip(best_seq)
    best_score = score(best_seq)
    
    # Generate candidate families
    # 1. Sidon-like
    def sidon_seq():
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i-1] * (1.0 - 1.0/(2.0*i))
        return a
    
    candidates = []
    candidates.append(sidon_seq())
    candidates.append(np.full(n, 1.0))
    # power law exponents
    for p in np.linspace(0.1, 2.0, 10):
        seq = (np.arange(1, n+1, dtype=np.float64) ** (-p))
        candidates.append(seq)
    # geometric decays
    for r in [0.999, 0.995, 0.99, 0.985, 0.98]:
        seq = r ** np.arange(n, dtype=np.float64)
        candidates.append(seq)
    # symmetric product shape
    indices = np.arange(n, dtype=np.float64) + 1
    rev = np.arange(n, 0, -1, dtype=np.float64) + 1
    for p in [0.5, 1.0]:
        seq = (indices ** (-p)) * (rev ** (-p))
        candidates.append(seq)
    
    # Evaluate each base candidate
    for seq in candidates:
        seq = clip(seq)
        sc = score(seq)
        if sc < best_score:
            best_seq = seq.copy()
            best_score = sc
    
    # Now hill climbing
    current = best_seq.copy()
    current_score = best_score
    
    # Main loop
    while time.monotonic() < deadline:
        # Choose operation
        op = rng.integers(0, 5)
        cand = current.copy()
        if op == 0:
            # swap two positions
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif op == 1:
            # perturb a single entry multiplicatively
            i = rng.integers(0, n)
            delta = rng.normal(loc=0.0, scale=0.05)  # 5% change
            cand[i] = max(0.0, cand[i] * (1.0 + delta))
        elif op == 2:
            # transfer mass between two entries
            i, j = rng.integers(0, n, size=2)
            if i != j:
                amount = cand[i] * rng.random() * 0.5
                cand[i] = cand[i] - amount
                cand[j] = cand[j] + amount
        elif op == 3:
            # small smoothing on a random window
            start_idx = rng.integers(0, n-2)
            win = rng.integers(5, 50)  # window size
            end_idx = min(n, start_idx + win)
            seg = cand[start_idx:end_idx]
            # moving average of length 3
            if seg.size >= 3:
                smoothed = np.convolve(seg, np.ones(3)/3, mode='same')
                cand[start_idx:end_idx] = smoothed
        else:
            # random perturbation of whole vector
            noise = rng.normal(loc=0.0, scale=0.02, size=n)
            cand = cand * (1.0 + noise)
            # ensure nonnegative
            cand = np.maximum(cand, 0.0)
        
        cand = clip(cand)
        cand_score = score(cand)
        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score
        if cand_score < current_score:
            current = cand.copy()
            current_score = cand_score
        
        # occasional restart from best with small random tweak
        if rng.random() < 0.001:
            current = best_seq.copy()
            # apply a random perturbation
            pert = rng.normal(scale=0.02, size=n)
            current = np.maximum(current * (1+pert), 0.0)
            current = clip(current)
            current_score = score(current)
    
    return [float(x) for x in best_seq]

import time
import numpy as np

# ---------------------------------------------------------
# Helper functions (top‑level, no closures)

def _max_convolution(seq: np.ndarray):
    """Return max value of the linear convolution a*a using FFT."""
    n = seq.shape[0]
    # length of linear convolution is 2n-1, pad to next power of two
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)
    # keep only the linear part
    conv = conv[: 2 * n - 1]
    return float(conv.max())


def _max_convolution_with_index(seq: np.ndarray):
    """Return (max value, index) of the linear convolution a*a using FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)
    conv = conv[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _score(seq: np.ndarray):
    """Exact AC1 bound used by the official evaluator (scale‑invariant)."""
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    n = seq.shape[0]
    maxb = _max_convolution(seq)
    return 2.0 * n * maxb / (s * s)


def _clip(seq: np.ndarray):
    """Force values into the allowed interval and guarantee a non‑zero sum."""
    np.clip(seq, 0.0, 1000.0, out=seq)
    s = seq.sum()
    if s < 0.01:               # avoid the infinite evaluator case
        seq *= (0.01 / (s + 1e-12))
        np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def _sidon_like_sequence(n: int):
    """Classic a_i = C(2i,i)/4**i sequence (non‑negative, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def _power_law_sequence(n: int, p: float):
    """a_i = (i+1)^{-p}  (positive, decreasing)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)
