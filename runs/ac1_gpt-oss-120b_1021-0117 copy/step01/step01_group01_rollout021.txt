import time
import math
import numpy as np

def _generate_binomial_seq(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4**i via recurrence a_{i+1}=a_i*(2i+1)/(2i+2)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a

def _conv_max_fft(a: np.ndarray, fft_len: int):
    """Maximum entry of a*a (full linear convolution) using one FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    L = 2 * a.size - 1
    conv = conv[:L]
    idx = int(conv.argmax())
    return float(conv[idx]), idx
