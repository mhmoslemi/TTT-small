    best = a0.copy()

        max_val = conv.max()

        grad = partner   # proportional to derivative

        candidates = np.where(grad <= low_thresh)[0]

import time
import numpy as np

def _full_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a using FFT (real inputs)."""
    n = a.shape[0]
    size = 2 * n - 1
    # next power of two for efficient FFT
    m = 1 << ((size - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:size]
    return conv

def _score(v: np.ndarray, n: int) -> float:
    """Exact score used by the evaluator (n = len(v))."""
    conv = _full_convolve(v)
    max_b = conv.max()
    s = v.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (s * s)
