def _ac1_score_and_conv(arr,n,L):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
    max_b = conv.max()
    s = arr.sum()
    if s < 1e-12:
        return float('inf'), conv
    return 2.0 * n * max_b / (s*s), conv

import numpy as np, time, math

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def _score_and_conv(arr: np.ndarray, n: int, L: int):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf'), conv
    return 2.0 * n * max_b / (s * s), conv

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # length and fallback handling
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise ValueError
        n = int(parent_arr.size)
    except Exception:
        n = 2048
        parent_arr = None

    L = _next_pow2(2 * n - 1)

    # ----- generate a few plausible starts ---------------------------------
    starts = []

    # binomial
    starts.append(_gen_binomial(n))

    # arcsine‑like 1/√((i+1)(n‑i))
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 1) * (n - i))
    starts.append(arcs)

    # sin‑shaped window
    starts.append(np.sin(np.pi * (i + 0.5) / n))

    # parent if available
    if parent_arr is not None:
        starts.append(parent_arr)

    # random positive vector
    starts.append(rng.random(n))

    best = None
    best_score = float('inf')
    for cand in starts:
        cand = np.clip(cand, 0.0, 1000.0)
        cand = (cand + cand[::-1]) * 0.5          # enforce symmetry
        sc = _ac1_score(cand, n, L)
        if sc < best_score:
            best_score = sc
            best = cand.copy()

    # ---- local search ----------------------------------------------------
    cur = best.copy()
    cur_score, conv_cur = _score_and_conv(cur, n, L)
    best = cur.copy()
    best_score = cur_score

    temperature = 1.0
    decay = 0.99985

    while time.monotonic() < deadline:
        # choose move type
        typ = rng.random()
        if typ < 0.7:                               # targeted mass shift
            kmax = int(conv_cur.argmax())
            i_start = max(0, kmax - (n - 1))
            i_end = min(n - 1, kmax)
            i_range = np.arange(i_start, i_end + 1)
            j_range = kmax - i_range
            contrib = cur[i_range] * cur[j_range]
            if contrib.size == 0:
                continue
            src_idx = i_range[int(contrib.argmax())]

            # pick destination outside the support if possible
            if i_start > 0 or i_end < n - 1:
                if i_start > 0 and i_end < n - 1:
                    if rng.random() < 0.5:
                        dst = rng.integers(0, i_start)
                    else:
                        dst = rng.integers(i_end + 1, n)
                elif i_start > 0:
                    dst = rng.integers(0, i_start)
                else:
                    dst = rng.integers(i_end + 1, n)
            else:
                dst = rng.integers(0, n)
                while dst == src_idx:
                    dst = rng.integers(0, n)

            frac = rng.random() * 0.2 + 0.05
            delta = cur[src_idx] * frac
            cand = cur.copy()
            cand[src_idx] -= delta
            cand[dst] += delta
        elif typ < 0.9:                             # block scaling
            min_w = max(1, n // 200)
            max_w = max(2, n // 12)
            width = rng.integers(min_w, max_w + 1)
            start = rng.integers(0, n - width + 1)
            factor = math.exp(rng.normal(0.0, 0.07))
            cand = cur.copy()
            cand[start:start + width] *= factor
        else:                                        # random transfer
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                continue
            frac = rng.random() * 0.3
            delta = cur[i_src] * frac
            cand = cur.copy()
            cand[i_src] -= delta
            cand[i_dst] += delta

        # enforce bounds and symmetry
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand = (cand + cand[::-1]) * 0.5

        cand_score = _ac1_score(cand, n, L)

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < 1.50274349:
                break

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score
            _, conv_cur = _score_and_conv(cur, n, L)

        temperature *= decay

        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            _, conv_cur = _score_and_conv(cur, n, L)
            temperature = 1.0

    return [float(v) for v in best]
