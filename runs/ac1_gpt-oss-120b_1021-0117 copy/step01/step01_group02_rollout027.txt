def hill_climb(seq, steps, time_deadline):
    best = seq.copy()
    best_score = compute_score(best)
    step_factor = 0.2
    min_step = 1e-6
    while steps>0 and time.monotonic()<time_deadline:
        conv = np.convolve(best, best)
        max_idx = int(conv.argmax())
        # compute partner similarly
        i_start = max(0, max_idx - (n - 1))
        i_end = min(max_idx, n - 1)
        # partner values
        idx_range = np.arange(i_start, i_end+1)
        partner = np.zeros_like(best)
        partner[idx_range] = best[max_idx - idx_range]
        src = int(partner.argmax())
        # dest where partner minimal
        dest_candidates = np.where(partner == partner.min())[0]
        dest = int(np.random.choice(dest_candidates)) if dest_candidates.size>0 else np.random.randint(0,n)
        if src == dest:
            continue
        delta = best[src] * step_factor
        if delta < min_step:
            delta = min_step
        cand = best.copy()
        cand[src] -= delta
        cand[dest] += delta
        # renorm to sum=1
        s = cand.sum()
        cand /= s
        # compute score
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            # optionally adjust step size
            step_factor = max(0.01, step_factor * 0.995)
        steps -= 1
    return best, best_score
```

import time

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound 2·n·max(conv(a,a))/sum(a)^2.
    The method searches a simple one‑parameter power‑law family
    a_i ∝ (i+offset)^(-p) and then performs a short hill‑climb.
    """

    # ---- length ---------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
    except Exception:                     # no parent available
        n = 5000

    # ---- helper to evaluate a candidate quickly -------------------------
    def _score_arr(v: np.ndarray) -> float:
        """2·n·max(conv(v,v))/sum(v)^2 – same formula as evaluate_sequence."""
        c = np.convolve(v, v)
        return 2.0 * n * c.max() / (v.sum() ** 2)

    best_seq = None
    best_score = float("inf")
    best_p = None
    best_off = None

    # ---- optional parent as a baseline ----------------------------------
    if "height_sequence_1" in globals():
