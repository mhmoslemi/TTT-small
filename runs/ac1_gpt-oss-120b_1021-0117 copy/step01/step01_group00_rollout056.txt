def compute_bound(a):
    # compute max convolution via FFT
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length()) # next power of 2
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A*A, n=size)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np
import time
from math import log2, ceil

# ------------------------------------------------------------
# Helper: fast convolution (full) using FFT, returns the full array
def _full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Convolution a*a (full) via FFT with a pre‑computed size."""
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.size - 1]
    # Numerical noise may produce tiny negatives – clamp to zero.
    np.maximum(conv, 0.0, out=conv)
    return conv


# ------------------------------------------------------------
# Helper: evaluate the AC1 bound without calling the supplied evaluator.
def _ac1_bound(a: np.ndarray, size: int) -> float:
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = _full_conv_fft(a, size)
    return 2.0 * a.size * conv.max() / (s * s)

    memory.
    """

        seq = np.empty(m, dtype=np.float64)

        for i in range(1, m):
            seq[i] = seq[i - 1] * (1.0 - 1.0 / (2.0 * i))
        return seq

    base = _sidon_like_sequence(n)

    # power‑scaled Sidon
    for p in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0):

    # simple power‑law tails
