```python
import numpy as np, time, math

def _initial_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def _pow_decay(n, p):
    # a_i ∝ (i+1)^(-p)
    i = np.arange(n) + 1
    a = i.astype(np.float64) ** (-p)
    # normalize to sum=1 (scaling not needed but keep)
    return a / np.sum(a)

def _exp_decay(n, lam):
    i = np.arange(n)
    a = np.exp(-lam * i)
    return a / np.sum(a)

def _fast_score_seq(seq):
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]
    maxc = np.max(conv)
    s = np.sum(seq)
    if s <= 0:
        return float('inf'), conv
    score = 2.0 * n * maxc / (s * s)
    return float(score), conv
```

import numpy as np, time, math

def _initial_binomial(n: int) -> np.ndarray:
    """Monotone decreasing a_i = C(2i,i)/4^i, i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _pow_decay(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    return a / a.sum()

def _exp_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam*i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()

def _fast_score(seq: np.ndarray):
    """Return (score, full convolution) using FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    maxc = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv
    return 2.0 * n * maxc / (s * s), conv
