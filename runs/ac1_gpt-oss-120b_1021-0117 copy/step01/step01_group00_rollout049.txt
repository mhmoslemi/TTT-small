    rng = np.random.default_rng(seed)
    deadline = time.time() + max(0.0, float(budget_s) - 1.0)  # leave margin

    # Determine base sequence length n

        n = parent.shape[0]
        # ensure 1-d
        if parent.ndim != 1:
            raise Exception
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    # function to compute AC1 score
    fft_len = 1 << ((2*n - 1).bit_length())
    def compute_score(arr):
        # arr will be a np.ndarray
        a = np.clip(arr, 0.0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        # FFT conv
        f = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(f * f, n=fft_len)[:2*n - 1]
        max_b = conv.max()
        return (2.0 * n * max_b) / (s * s)

    # baseline candidates
    best_seq = np.clip(parent, 0.0, 1000.0)
    best_score = compute_score(best_seq)
    # compute sidon-like sequence
    def sidon_seq(m):
        a = np.empty(m, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, m):
            a[i] = a[i-1] * (1.0 - 1.0/(2.0*i))
        return a
    sid = sidon_seq(n)
    sid = np.clip(sid, 0.0, 1000.0)
    sid_score = compute_score(sid)
    if sid_score < best_score:
        best_seq = sid.copy()
        best_score = sid_score

    # start local search from best_seq
    current = best_seq.copy()
    current_score = best_score

    # parameters for SA
    start_temp = 1.0
    end_temp = 1e-4
    max_iters = 2000000  # large but will be limited by time
    i = 0
    while time.time() < deadline:

        progress = i / max_iters
        temp = start_temp * (1.0 - progress) + end_temp * progress

        # propose move
        r = rng.random()
        if r < 0.4:

            cand = current.copy()
            cand[idx] = max(0.0, min(1000.0, cand[idx] + delta))

        cand_score = compute_score(cand)

        # ------------------------------------------------------------------
        # Acceptance rule (Metropolis)
        # ------------------------------------------------------------------
        if cand_score < current_score or rng.random() < math.exp(
                -(cand_score - current_score) / max(temp, 1e-12)):
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score

        # ------------------------------------------------------------------
        # Occasionally inject fresh diversity
        # ------------------------------------------------------------------
        if it % 50000 == 0 and rng.random() < 0.1:
            # restart from the Sidon shape with a random permutation
            cur = sid.copy()
            perm = rng.permutation(n)
            cur = cur[perm]
            cur_score = compute_score(cur)
            current = cur
            current_score = cur_score
            if cur_score < best_score:
                best_seq = cur.copy()
                best_score = cur_score

        it += 1

    # ------------------------------------------------------------------ #
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------ #
    return [float(x) for x in best_seq]
```
