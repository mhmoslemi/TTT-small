import numpy as np, time, math, random
from numpy.fft import rfft, irfft

def _initial_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i-1) / (2*i)
    return a

def _score_and_conv(seq, m=None):
    n = seq.shape[0]
    if m is None:
        m = 1 << ((2*n-1).bit_length())
    f = rfft(seq, n=m)
    conv = irfft(f * f, n=m)[:2*n-1]
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    sum_seq = float(seq.sum())
    if sum_seq <= 0:
        return float('inf'), conv, max_val, max_idx
    score = 2.0 * n * max_val / (sum_seq*sum_seq)
    return score, conv, max_val, max_idx

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0) # leave margin

    # Determine n and possibly parent
    n = 3809
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
    except Exception:
        parent = None

    if parent is not None:
        best = parent.copy()
    else:
        best = _initial_binomial(n)

    best_score, best_conv, best_max, best_argmax = _score_and_conv(best)

    # Additional candidate: binomial if parent worse
    if parent is None:
        # using binomial already
        pass
    else:
        # try binomial and pick best
        bin_seq = _initial_binomial(n)
        bin_score, bin_conv, _, _ = _score_and_conv(bin_seq)
        if bin_score < best_score:
            best = bin_seq
            best_score = bin_score
            best_conv = bin_conv
            best_argmax = int(np.argmax(best_conv))

    # Parameters for simulated annealing/hill climbing
    # define schedule for temperature
    # start with sigma for scaling moves maybe 0.2 gradually decreasing
    init_sigma = 0.3
    min_sigma = 0.02
    # We'll reduce sigma linearly based on progress towards deadline
    # Also have probability of accepting worse moves decreasing.

    # main loop
    # track iteration
    iter_count = 0
    while time.monotonic() < deadline:
        # update sigma based on remaining fraction
        frac = (deadline - time.monotonic()) / (budget_s - 2.0)
        sigma = min_sigma + (init_sigma - min_sigma) * frac

        # choose operation randomly weighted
        op = rng.integers(0, 6)  # 0-5 includes new moves
        cand = best.copy()
        if op == 0:  # block scaling
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            factor = math.exp(rng.normal(0.0, sigma*0.5))
            cand[start:start+width] *= factor
        elif op == 1:  # single index scaling
            idx = rng.integers(0, n)
            factor = math.exp(rng.normal(0.0, sigma))
            cand[idx] *= factor
        elif op == 2:  # mass shift from high product pair
            # Find current argmax pair
            k = best_argmax
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # Choose index i in overlap region proportional to a_i * a_{k-i}
            # compute weights
            idxs = np.arange(lo, hi+1)
            js = k - idxs
            # products
            prod = cand[idxs] * cand[js]
            total_prod = prod.sum()
            if total_prod > 0:
                # sample a pair with probability proportional to product
                # accumulate cdf
                probs = prod / total_prod
                # choose one pair
                pick = rng.random()
                cum = 0.0
                for ii, p in zip(idxs, probs):
                    cum += p
                    if pick <= cum:
                        i = int(ii)
                        j = int(k - i)
                        break
                else:
                    i = int(idxs[-1]); j = int(k - i)
                # move fraction from both i and j
                alpha = rng.uniform(0.04, 0.12)  # fraction of each to move
                take_i = cand[i] * alpha
                take_j = cand[j] * alpha
                cand[i] -= take_i
                cand[j] -= take_j
                # deposit total to random low index (prefer low values)
                tgt = rng.integers(0, n)
                attempts = 0
                while (tgt == i or tgt == j) and attempts < 5:
                    tgt = rng.integers(0, n)
                    attempts += 1
                # optionally bias to low-value index:
                if rng.random() < 0.7:
                    # choose from among indices with small value
                    # get quantile threshold
                    thresh = np.percentile(cand, 30)
                    low_idxs = np.where(cand <= thresh)[0]
                    if low_idxs.size > 0:
                        tgt = int(rng.choice(low_idxs))
                cand[tgt] += (take_i + take_j)
        elif op == 3:  # smooth neighboring pair
            i = rng.integers(0, n-1)
            avg = (cand[i] + cand[i+1]) * 0.5
            cand[i] = avg
            cand[i+1] = avg
        elif op == 4:  # random mass relocation
            src = rng.integers(0, n)
            if cand[src] > 0:
                frac = rng.uniform(0.05, 0.3)
                amount = cand[src] * frac
                cand[src] -= amount
                dst = rng.integers(0, n)
                cand[dst] += amount
        else:  # op == 5: perturb two random indices
            i, j = rng.integers(0, n, size=2)
            factor_i = math.exp(rng.normal(0.0, sigma*0.6))
            factor_j = math.exp(rng.normal(0.0, sigma*0.6))
            cand[i] *= factor_i
            cand[j] *= factor_j

        np.clip(cand, 0.0, 1000.0, out=cand)

        # compute score
        cand_score, cand_conv, cand_max, cand_argmax = _score_and_conv(cand)

        # acceptance: if better -> accept; else maybe accept with probability exp(-Δ/ T)
        # define temperature based on remaining fraction
        T = sigma  # simple mapping
        if cand_score < best_score or rng.random() < math.exp(-(cand_score - best_score) / (T + 1e-12)):
            best = cand
            best_score = cand_score
            best_conv = cand_conv
            best_argmax = cand_argmax

        iter_count += 1

    # Final: return best as python list
    return [float(x) for x in best]

import numpy as np
import time
import math
from numpy.fft import rfft, irfft

def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _score_and_conv(seq: np.ndarray, n: int, m: int):
    """Fast AC1 score + full convolution via FFT."""
    f = rfft(seq, n=m)
    conv = irfft(f * f, n=m)[: 2 * n - 1]
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    s = float(seq.sum())
    if s <= 0.0:
        return float("inf"), conv, max_val, max_idx
    score = 2.0 * n * max_val / (s * s)
    return score, conv, max_val, max_idx
