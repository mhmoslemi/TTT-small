import time, math
import numpy as np

def _conv_fft(x):
    """Linear self‑convolution via FFT (real)."""
    n = x.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two >= 2n‑1
    X = np.fft.rfft(x, n=L)
    c = np.fft.irfft(X * X, n=L)
    return c[: 2 * n - 1]

def _proj_simplex(v, s=1.0):
    """Projection of v onto the simplex {w >= 0, sum(w)=s}."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = cssv[-1] / v.size
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Sub‑gradient descent on max(self‑convolution) with simplex projection."""
    rng = np.random.default_rng(seed)
    # ---- initialise ---------------------------------------------------------
    try:
        n = len(height_sequence_1)               # provided parent, if any
        a = np.array(height_sequence_1, dtype=np.float64)
    except Exception:                           # no parent – fall back to binomial shape
        n = 5000
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))

    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    a = a / total if total > 1e-12 else np.full(n, 1.0 / n)   # normalise to sum = 1

    # ---- baseline candidates -------------------------------------------------
    conv = _conv_fft(a)
    best = a.copy()
    best_score = 2.0 * n * conv.max()         # sum = 1, so score = 2·n·max(conv)

    # uniform fallback (score = 2) – keep if better
    uniform = np.full(n, 1.0 / n)
    u_score = 2.0 * n * _conv_fft(uniform).max()
    if u_score < best_score:
        best = uniform.copy()
        best_score = u_score

    # ---- optimisation loop --------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    it = 0
    alpha0 = 0.4                               # initial step‑size factor

    while time.monotonic() < deadline:

        score = 2.0 * n * M                     # sum = 1 → exact objective value
        if score < best_score:

        # indices attaining (or almost attaining) the maximum
        eps = 1e-12

        # sub‑gradient of max‑convolution (sum over all active k)
        g = np.zeros_like(a)
        for k in max_idx:
