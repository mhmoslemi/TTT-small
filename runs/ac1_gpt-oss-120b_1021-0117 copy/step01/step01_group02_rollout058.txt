def propose_candidate(seed=42, budget_s=1100):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-0.5)

    # get length
    try:
        n = len(height_sequence_1)
        init = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 5000
        init = np.full(n, 1.0/n, dtype=np.float64)

    # Normalize sum to 1
    if init.sum() == 0:
        init = np.full(n, 1.0/n)
    init = init / init.sum()

    a = init.copy()
    # ensure nonnegative, clip, but clipping may affect sum; after clipping we renormalize to sum=1.
    a = np.clip(a, 0.0, 1000.0)
    a = a / a.sum()

    # compute initial score using FFT
    N = 1 << (2*n - 1).bit_length()
    # maybe compute conv via function compute_conv
    def max_conv(arr):
        Fa = np.fft.fft(arr, n=N)
        conv = np.fft.ifft(Fa * Fa).real[:2*n-1]
        return conv.max()
    cur_max = max_conv(a)
    cur_score = 2.0 * n * cur_max

    best_seq = a.copy()
    best_score = cur_score

    # parameters
    step_factor = 0.8
    min_step = 1e-8
    no_improve_iter = 0

    # main loop
    while time.monotonic() < deadline:
        # compute full conv and index of max
        Fa = np.fft.fft(a, n=N)
        conv = np.fft.ifft(Fa * Fa).real[:2*n-1]
        k = int(conv.argmax())
        cur_max = conv[k]
        # indices overlapping
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner_idx = k - idx
        left = a[idx]
        right = a[partner_idx]
        contributions = left * right
        # src: max contribution
        src_local = int(idx[np.argmax(contributions)])
        # dest: minimal contribution (including zero for out-of-range)
        # compute contributions for all indices (0..n-1)
        # for each j, partner index = k - j; if out of bounds, contribution=0
        # vectorize
        j = np.arange(n)
        partner = k - j
        valid = (partner >= 0) & (partner < n)
        partner_vals = np.zeros(n, dtype=np.float64)
        partner_vals[valid] = a[partner[valid]]
        # contribution = a * partner_vals
        contrib_all = a * partner_vals
        # dest = argmin contrib_all (prefer zero)
        dest = int(np.argmin(contrib_all))
        # ensure dest != src
        if dest == src_local:
            # random dest
            dest = int(rng.integers(0, n))
            if dest == src_local:
                dest = (dest + 1) % n

        # decide delta
        delta = a[src_local] * step_factor
        if delta < min_step:
            delta = min_step
        if delta > a[src_local]:
            delta = a[src_local]
        # if delta == 0 (src zero) then random perturbation
        if delta <= 0:
            # random perturb
            width = int(rng.integers(max(1, n//200), max(2, n//12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand = a.copy()
            # multiplicative random factor
            factor = np.exp(rng.normal(0.0, 0.15, size=stop-start))
            cand[start:stop] *= factor
            # ensure nonnegative clip and renormalise
            cand = np.clip(cand, 0.0, 1000.0)
            cand_sum = cand.sum()
            if cand_sum == 0:
                continue
            cand /= cand_sum
            new_max = max_conv(cand)
            new_score = 2.0 * n * new_max
            if new_score < best_score:
                best_seq, best_score = cand, new_score
                a = cand
                cur_score = new_score
            continue

        cand = a.copy()
        cand[src_local] -= delta
        cand[dest] += delta
        # clip just in case negative due to numerical errors
        cand = np.clip(cand, 0.0, 1000.0)
        # renormalize to sum=1 (should be unchanged, but ensure)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum
        new_max = max_conv(cand)
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best_seq, best_score = cand.copy(), new_score
            a = cand
            cur_score = new_score
            step_factor = min(0.95, step_factor * 1.001)  # slowly increase small step if progress
            no_improve_iter = 0
        else:
            # optionally accept some negative moves with small prob to escape
            no_improve_iter += 1
            if rng.random() < 0.001:
                a = cand  # accept
            # occasionally shrink step factor
            if no_improve_iter % 1000 == 0:
                step_factor = max(0.02, step_factor * 0.9)

        # Additional random shake out-of-loop occasionally
        if rng.random() < 0.003:
            # random segment multiplicative noise
            width = int(rng.integers(max(1, n//300), max(2, n//50)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand2 = a.copy()
            cand2[start:stop] *= np.exp(rng.normal(0.0, 0.2, size=stop-start))
            cand2 = np.clip(cand2, 0.0, 1000.0)
            cand2 /= cand2.sum()
            new_max2 = max_conv(cand2)
            new_score2 = 2.0 * n * new_max2
            if new_score2 < best_score:
                best_seq, best_score = cand2.copy(), new_score2
                a = cand2
                cur_score = new_score2

    # final sanity: ensure sum > 0.01 and nonnegative
    final = np.clip(best_seq, 0.0, 1000.0)
    s = final.sum()
    if s < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
