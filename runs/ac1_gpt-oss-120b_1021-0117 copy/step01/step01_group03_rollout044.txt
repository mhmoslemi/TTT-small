import time
import numpy as np

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT (size ~next power of two)."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _sqrt_series(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4^i sequence (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a low‑score non‑negative step function within the time budget."""
    rng = np.random.default_rng(seed)
    # keep a margin to exit cleanly
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initial sequence: parent if present, otherwise sqrt series.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = a.size
    else:
        n = 5582
        a = _sqrt_series(n)
        a = np.clip(a, 0.0, 1000.0)

    # evaluate initial candidate (homogeneous objective – scaling invariant)
    conv = _conv_self(a)
    sum_a = a.sum()
    cur_score = 2.0 * n * conv.max() / (sum_a * sum_a) if sum_a >= 0.01 else float("inf")
    best = a.copy()
    best_score = cur_score

    # helpers for early‑late transfers (fallback when gradient step stalls)
    early_cut = n // 4
    late_cut = 3 * n // 4
    target = 1.50274349  # goal – stop early if reached

    # ------------------------------------------------------------------
    # Main iterative improvement loop.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # recompute convolution and score at the beginning of each iteration
        conv = _conv_self(a)
        max_conv = conv.max()
        sum_a = a.sum()
        if sum_a < 0.01:                 # safety guard
            a[0] += 0.02
            sum_a = a.sum()
        cur_score = 2.0 * n * max_conv / (sum_a * sum_a)

        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target:
                break  # good enough – exit early

        # ------------------------------------------------------------------
        # Choose a stochastic transformation.
        # ------------------------------------------------------------------
        op = rng.random()
        if op < 0.55:
            # ----- gradient‑guided mass transfer (reduce the current peak) -----
            k = int(np.argmax(conv))           # index of the current maximum
            idx = np.arange(n)
            rev = k - idx
            mask = (rev >= 0) & (rev < n)
            w = np.zeros(n, dtype=np.float64)
            w[mask] = a[rev[mask]]             # partner values for this k

            # consider only positions that actually contribute to the peak
            contrib_mask = w > 0
            if np.any(contrib_mask):
                contrib = a * w
                src = int(np.argmax(contrib))  # source with largest contribution
                # pick a destination with small w and room to grow
                space = 1000.0 - a
                cand = np.where((space > 1e-12) & (~contrib_mask))[0]
                if cand.size == 0:
                    cand = np.where(space > 1e-12)[0]
                dst = int(cand[np.argmin(w[cand])])
                if src != dst and a[src] > 1e-12:
                    max_transfer = min(a[src], space[dst])
                    if max_transfer > 0.0:

            else:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
