import time, math, numpy as np

def _sidon_like_sequence(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4**i, generated by recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a

def _max_conv_fft(a: np.ndarray, L: int, n: int) -> float:
    """Maximum value of full convolution a*a using a length‑L FFT."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return float(conv[: 2 * n - 1].max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ---- obtain problem size and a default parent -----------------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = int(parent.size)
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    # ---- initial candidates ----------------------------------------
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    sid = _sidon_like_sequence(n)
    sid = np.clip(sid, 0.0, 1000.0)
    sid_score = float(evaluate_sequence(sid.tolist()))
    if sid_score < best_score:
        best_seq, best_score = sid.copy(), sid_score

    # Uniform start (useful for early exploration)
    uniform = np.full(n, 1.0, dtype=np.float64)
    uni_score = float(evaluate_sequence(uniform.tolist()))
    if uni_score < best_score:
        best_seq, best_score = uniform.copy(), uni_score

    # ---- search parameters -----------------------------------------
    cur = best_seq.copy()
    total = float(cur.sum())               # invariant sum ( > 0 )
    # FFT length (next power of two ≥ 2n)
    L = 1 << (int(math.log2(2 * n - 1)) + 1)

    # helper: quick bound from FFT (no scaling needed)
    def approx_score(arr: np.ndarray) -> float:
        mx = _max_conv_fft(arr, L, n)
        return 2.0 * n * mx / (total * total)

    # start with current bound (may be slightly different from best_score)
    cur_score = approx_score(cur)

    # ---- main stochastic improvement loop --------------------------
    while time.monotonic() < deadline:
        # choose a source (large) and a target (small) index
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # direction: move mass from larger to smaller
        if cur[i] < cur[j]:
            i, j = j, i
        if cur[i] <= 0.0:
            continue
        max_move = min(cur[i], 1000.0 - cur[j])
        if max_move <= 0.0:
            continue
        # transfer a modest random fraction
        delta = max_move * rng.random() * 0.08  # ≤8 % of allowed move
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # quick approximate test
        cand_approx = approx_score(cand)
        if cand_approx + 1e-12 >= best_score:
            # not promising – skip expensive exact evaluation
            continue

        # exact evaluation (uses np.convolve, slower but reliable)
        cand_score = float(evaluate_sequence(cand.tolist()))
        if not math.isfinite(cand_score):
            continue

        if cand_score < best_score:
            best_seq, best_score, cur, cur_score = (
                cand.copy(),
                cand_score,
                cand.copy(),
                cand_approx,
            )
        else:
            # keep the candidate as current only if it improves the approximate score

        # occasional random perturbation to avoid dead‑ends

            idx = rng.choice(n, size=min(8, n), replace=False)
            cur[idx] = rng.random(size=idx.shape) * cur[idx]
            cur_score = approx_score(cur)

    # ensure the returned sequence respects the evaluator’s clipping
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
