import time

# ----- helper functions -------------------------------------------------
def _generate_binomial_seq(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def _conv_max_fft(a: np.ndarray, fft_len: int):
    """Return (max value, full convolution) of a*a via one FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max()), conv


def _gradient_at_max(a: np.ndarray, conv: np.ndarray, max_val: float):
    """Sub‑gradient of the max‑convolution w.r.t. a (sum of 2*a_{p-i})."""
    n = a.size
    # indices where convolution attains the maximum (allow tiny tolerance)
    tol = max_val * 1e-12 + 1e-15
    max_pos = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for p in max_pos:
        i_start = max(0, p - (n - 1))
        i_end = min(p, n - 1)
        # i runs i_start .. i_end  (inclusive)
        ip = np.arange(i_start, i_end + 1)
        grad[i_start:i_end + 1] += 2.0 * a[p - ip]
    return grad
