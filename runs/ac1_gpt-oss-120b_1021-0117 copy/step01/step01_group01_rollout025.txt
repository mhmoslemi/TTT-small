        elapsed = time.monotonic() - start
        remaining = deadline - time.monotonic()
        # temperature schedule: start at 0.1 (or higher) and linearly to 0 ending
        # We'll base on proportion of elapsed / (deadline - start)
        t_frac = elapsed / (deadline - start + 1e-12)

        accept = False
        if new_score < cur_score:
            accept = True
        else:

                prob = math.exp(-(new_score - cur_score) / temperature)
                accept = rng.random() < prob

        if accept:
            a = new_a
            cur_score = new_score
            if new_score < best_score:
                best = new_a.copy()
                best_score = new_score

        iter_count += 1

        # occasional re-normalize to avoid drift
        if iter_count % 1000 == 0:
            # re-normalize sum to 1 (should be already near)
            s = a.sum()
            if abs(s - 1.0) > 1e-12:

    # after loop
    final = [float(x) for x in best]
    try:
        _ = evaluate_sequence(final)
    except Exception:

    return final

import time
import math
import numpy as np

def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale array to have sum 1 (leaving zeros untouched)."""
    s = float(v.sum())
    if s <= 0.0:
        v = np.ones_like(v, dtype=np.float64)
        s = float(v.sum())
    return v / s

def _score_fft(v: np.ndarray, n: int, fft_len: int) -> float:
    """Exact score using FFT – identical to evaluate_sequence for non‑negative v."""
    fa = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(v.sum())
    return 2.0 * n * max_b / (s * s)
