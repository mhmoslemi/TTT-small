import numpy as np, time, math

def _next_pow2(x: int) -> int:
    """smallest power of two >= x"""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4**i for i=0..n-1"""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """exact AC1 bound via FFT (2*n*max(conv)/sum^2)"""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _conv_full(arr: np.ndarray, n: int, L: int) -> np.ndarray:
    """full linear convolution a*a (length 2*n-1) using FFT"""
    fa = np.fft.rfft(arr, n=L)
    return np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]

    base = _gen_binomial(n)                # classic binomial construction
    best = base.copy()
    best_score = _ac1_score(best, n, L)

    if parent_seq is not None:
        ps = _ac1_score(parent_seq, n, L)
        if ps < best_score:
            best = parent_seq.copy()
            best_score = ps

    cur = best.copy()
    cur_score = best_score

    # ----- search parameters --------------------------------------------
    temperature = 1.0
    decay = 0.9997
    target = 1.50274349          # stop early if we beat this

    # ----- main loop ----------------------------------------------------
    while time.monotonic() < deadline:
        # pick a mutation type
        move = rng.integers(0, 7)
        cand = cur.copy()

        if move == 0:                                 # block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand[start:start + width] *= factor

        elif move == 1:                               # mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * 0.4
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta

        elif move == 2:                               # smoothing
            k = int(rng.integers(3, 11))              # odd window size 3..10
            kernel = np.ones(k, dtype=np.float64) / k
            smooth = np.convolve(cand, kernel, mode='same')
            cand[:] = smooth

        elif move == 3:                               # swap two entries
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            cand[i], cand[j] = cand[j], cand[i]

        elif move == 4:                               # power‑law family
            p = rng.random() * 0.6 + 0.3               # p ∈ [0.3,0.9]
            beta = rng.random() * n * 0.1              # β up to 0.1·n
            idx = np.arange(n, dtype=np.float64)
            cand = (idx + beta) ** (-p)

        elif move == 5:                               # exponential decay family
            lam = rng.random() * 0.005 + 0.0005        # λ ∈ [5e‑4,5.5e‑3]
            cand = np.exp(-lam * np.arange(n, dtype=np.float64))

        else:                                          # targeted max‑conv reduction
            # compute current convolution once
            conv = _conv_full(cur, n, L)
            kmax = int(conv.argmax())
            # indices that contribute to this lag
            i_min = max(0, kmax - (n - 1))
            i_max = min(n - 1, kmax)
            if i_max >= i_min:
                isel = np.arange(i_min, i_max + 1, dtype=np.int64)
                jsel = kmax - isel
                contrib = cur[isel] * cur[jsel]
                total_contrib = contrib.sum()
                if total_contrib > 0:
                    # pick a pair weighted by contribution
                    probs = contrib / total_contrib
                    idx = rng.choice(isel.shape[0], p=probs)
                    i = int(isel[idx])
                    j = int(jsel[idx])
                    # move a fraction from i (and optionally j) to a random location
                    take_i = cur[i] * rng.random() * 0.3
                    cand[i] -= take_i
                    k = rng.integers(0, n)
                    cand[k] += take_i
                    if j != i:
                        take_j = cur[j] * rng.random() * 0.3
                        cand[j] -= take_j
                        cand[k] += take_j

        # enforce the evaluator’s hard bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate candidate
        cand_score = _ac1_score(cand, n, L)

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < target:
                break

        # simple simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # temperature schedule
        temperature *= decay

        # occasional full restart from current best
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # return python list of floats (non‑empty, finite)
    return [float(x) for x in best]
