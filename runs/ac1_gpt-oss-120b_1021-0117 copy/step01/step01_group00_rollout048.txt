def _sidon_like_sequence(n):
    """Generate a_i = C(2i,i)/4**i for i=0..n-1 (decreasing, sum ~sqrt(2n))."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a

    return [float(x) for x in final]
