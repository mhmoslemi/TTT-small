import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 upper bound using FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step function minimising the AC1 bound."""
    rng = np.random.default_rng(seed)
    # a few seconds safety margin for the outer harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # obtain problem size and an initial candidate
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256

    L = _next_pow2(2 * n - 1)

    # classic binomial start (C(2i,i)/4^i)
    base = np.empty(n, dtype=np.float64)
    base[0] = 1.0
    for i in range(1, n):
        base[i] = base[i - 1] * (2 * i - 1) / (2 * i)

    best = base.copy()
    best_score = _ac1_score(best, n, L)

    if parent is not None:
        parent_score = _ac1_score(parent, n, L)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    cur = best.copy()
    cur_score = best_score

    # simulated‑annealing parameters
    temperature = 1.0
    decay = 0.9998

    # ------------------------------------------------------------------
    # main stochastic descent
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        move = rng.random()

        # 0 – targeted reduction of the peak convolution term
        if move < 0.4:
            # compute full convolution of the current vector
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            k = int(conv.argmax())

            # indices i,j with i+j = k (overlap region)
            start = max(0, k - (n - 1))
            end = min(n - 1, k)
            idx = np.arange(start, end + 1, dtype=np.int64)
            idxj = k - idx

            # product a_i * a_j, pick the biggest contributor
            prod = cur[idx] * cur[idxj]
            if prod.size == 0:
                continue
            local_max = int(prod.argmax())
            i = int(idx[local_max])
            j = int(idxj[local_max])

            # shrink the two entries slightly and move the lost mass far right
            factor = 1.0 - rng.uniform(0.02, 0.15)   # 2‑15 % reduction
            cand = cur.copy()
            loss = 0.0
            loss_i = cand[i] * (1.0 - factor)
            cand[i] = cand[i] * factor
            loss += loss_i
            if j != i:
                loss_j = cand[j] * (1.0 - factor)
                cand[j] = cand[j] * factor
                loss += loss_j

            # dump the removed mass onto a far‑right index (to avoid creating
            # new large overlaps).  If that index saturates, spread the excess.
            dest = int(rng.integers(n // 2, n))
            cand[dest] += loss
            overflow = max(0.0, cand[dest] - 1000.0)
