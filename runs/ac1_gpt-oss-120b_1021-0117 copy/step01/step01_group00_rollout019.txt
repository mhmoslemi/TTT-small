import time
import numpy as np

def _power_seq(length: int, alpha: float) -> np.ndarray:
    """Return a decreasing non‑negative step sequence a_i = (i+1)^{-alpha}."""
    idx = np.arange(1, length + 1, dtype=np.float64)
    seq = np.power(idx, -alpha)
    # respect evaluator clipping limits
    return np.clip(seq, 0.0, 1000.0)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step‑function minimizing the AC1 bound."""
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine problem size and a fallback parent sequence
    try:
        n = int(len(height_sequence_1))
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0)
        if parent.ndim != 1 or not np.isfinite(parent).all() or parent.size == 0:
            raise Exception
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    # best solution so far
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    # ------------------------------------------------------------------
    # Explore a family a_i = (i+1)^{-alpha} for several alphas
    alphas = np.linspace(0.55, 0.98, 22)   # 22 candidates, cheap to evaluate
    for alpha in alphas:
        if time.monotonic() >= deadline:
            break
        cand = _power_seq(n, alpha)
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best_seq, best_score = cand, cand_score

    # ------------------------------------------------------------------
    # Light local search: random swaps on the current best shape
    rng = np.random.default_rng(seed)
    current = best_seq.copy()
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cand = current.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best_seq, best_score = cand, cand_score
            current = cand
        # occasional restart to avoid stagnation
        if rng.random() < 0.001:
            current = best_seq.copy()
            perm = rng.permutation(n)
            current = current[perm]

    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
