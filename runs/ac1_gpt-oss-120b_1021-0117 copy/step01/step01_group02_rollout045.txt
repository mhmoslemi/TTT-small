import time
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to minimise
    the AC1 upper bound.  The construction is a simple exponentially decaying
    sequence a_i = r**i with r = 1 - log(n)/n, which analytically gives a very
    low value of 2·n·max(a*a)/(sum a)^2.
    """
    # ---- time budget -------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ---- length ------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 5000                     # fallback length if parent not provided

    # ---- exponential decay parameter ----------------------------------------
    # epsilon = log(n)/n gives a near‑optimal trade‑off between sum and max‑auto‑correlation
    eps = max(1e-12, math.log(max(n, 2)) / n)
    r = 1.0 - eps
    if r <= 0.0:                     # safety guard (should never happen for n>=2)
        r = 0.5

    # ---- generate the sequence ---------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    # a_i = r**i = exp(i * log(r))
    seq = np.exp(idx * math.log(r))

    # ---- enforce evaluator limits -------------------------------------------
    seq = np.clip(seq, 0.0, 1000.0)

    # ---- sanity checks ------------------------------------------------------
    if seq.sum() < 0.01:
        # extremely unlikely, but fall back to a uniform distribution
        seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ---- respect the deadline -----------------------------------------------
    if time.monotonic() >= deadline:
        # if we run out of time (unlikely here), return a trivial uniform sequence
        seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ---- final conversion ----------------------------------------------------
    return [float(v) for v in seq]
