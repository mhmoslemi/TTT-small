def gradient_contrib(seq, idx):
    n = seq.shape[0]
    out = np.zeros_like(seq)
    i_start = max(0, idx - (n - 1))

def _initial_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def _fast_score(seq):
    n = seq.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]
    max_val = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float('inf'), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val

import numpy as np
import time
import math

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fast_score(seq: np.ndarray):
    """Return (score, full convolution, max conv) using FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # power‑of‑two length
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val


def _gradient_update(seq: np.ndarray, idx: int, M: float, eta: float) -> np.ndarray:
    """One projected gradient step (scale‑invariant) on seq."""
    n = seq.shape[0]
    s = seq.sum()
    # gradient of the convolution maximum (sub‑gradient for the max index)
    gradM = np.zeros_like(seq)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1)
    j_range = idx - i_range
    gradM[i_range] = 2.0 * seq[j_range]
    # gradient of the objective up to a positive factor
    g = s * gradM - 2.0 * M
    new_seq = seq - eta * g
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    # renormalise to keep the ℓ₁‑norm (scaling does not affect the objective)
    sum_new = new_seq.sum()
    if sum_new > 0.0:
        scale = s / sum_new
        new_seq *= scale
        np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq
