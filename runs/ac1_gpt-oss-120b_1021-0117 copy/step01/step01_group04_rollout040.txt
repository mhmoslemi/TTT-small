import numpy as np
import time
import math

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Evaluate the AC1 upper bound for a candidate."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

        n = 256
        parent = None

    L = _next_pow2(2 * n - 1)

    base = _gen_binomial(n)
    best = base.copy()

    if have_parent:

        if parent_score < best_score:

    cur = best.copy()
