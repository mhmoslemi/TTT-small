import time
import numpy as np

def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style weights w_i = C(2i,i)/4**i for i=0..n-1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _max_conv_and_arg(a: np.ndarray, m: int):
    """Return (max value, index of max) of a*a using FFT of length m."""
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def _score_from_max(n: int, max_b: float, sum_a: float):
    """Exact objective from max convolution and sum."""
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)
