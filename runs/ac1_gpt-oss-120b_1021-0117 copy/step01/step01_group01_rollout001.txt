def _max_conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum entry of the linear convolution a*a using a single FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(conv.max())

    deadline = start + max(0.0, float(budget_s) - 0.5)
