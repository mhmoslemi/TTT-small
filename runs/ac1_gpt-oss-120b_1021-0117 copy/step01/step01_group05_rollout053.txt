def _score_and_conv(seq):
    n = seq.shape[0]
    m = 1 << ((2*n-1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f*f, n=m)[:2*n-1]
    s = np.sum(seq)
    if s <=0: return (float('inf'), conv)
    max_val = np.max(conv)
    score = 2.0 * n * max_val / (s*s)
    return (float(score), conv)

def _targeted_move(seq, conv, rng):
    n = seq.shape[0]
    k_max = int(np.argmax(conv))
    lo = max(0, k_max - (n-1))
    hi = min(k_max, n-1)
    # compute contributions
    idx_range = np.arange(lo, hi+1)
    # partner indices
    partners = k_max - idx_range
    products = seq[idx_range] * seq[partners]
    # maybe exclude diagonal (i=j) to move from both
    # find i where product is max and i != partners
    # but if i==partner (k_max even, i=j), we skip? we could treat same.
    # Let's just find max product among i != partner
    mask = idx_range != partners
    if not np.any(mask):
        # fallback to random
        i = rng.integers(lo, hi+1)
        j = k_max - i
    else:
        args = np.argmax(products[mask])
        i_candidates = idx_range[mask][args]
        i = i_candidates
        j = k_max - i

    # remove small fraction
    alpha = rng.uniform(0.03, 0.12)
    # ensure we don't go negative
    delta_i = seq[i] * alpha
    delta_j = seq[j] * alpha
    seq[i] -= delta_i
    seq[j] -= delta_j

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)  # give 1 sec margin

    # Determine n, parent as above.
    n = ...

    # Define fast score
    # Define _initial_binomial

    # Candidate seeds
    seeds = []

    # Binomial
    bin_seq = _initial_binomial(n)
    bin_score, bin_conv = _fast_score(bin_seq)
    seeds.append((bin_seq, bin_score, bin_conv))

    # Parent if exists
    if parent is not None:
        p_seq = np.clip(parent, 0, 1000)
        p_score, p_conv = _fast_score(p_seq)
        seeds.append((p_seq, p_score, p_conv))

    # Exponential seeds: sample c values.
    for c in [0.001, 0.002, 0.004, 0.008, 0.016, 0.032, 0.064, 0.128]:
        exp_seq = np.exp(-c * np.arange(n, dtype=np.float64))
        # Clip maybe not needed; values <=1.
        exp_score, exp_conv = _fast_score(exp_seq)
        seeds.append((exp_seq, exp_score, exp_conv))

    # Choose best
    best_seq, best_score, best_conv = min(seeds, key=lambda x: x[1])

    # Set current = best copy
    current_seq = best_seq.copy()
    current_score = best_score
    current_conv = best_conv

    # Main loop
    while time.monotonic() < deadline:
        # Compute acceptance temperature
        t_frac = (deadline - time.monotonic()) / (deadline - start_time)
        # but we want temperature high early, low later; maybe use (1 - t_frac)
        temp = 0.05 * (1 - t_frac) + 0.001 * t_frac  # high early 0.05, low 0.001

        candidate = current_seq.copy()
        op = rng.integers(0, 5) # 5 ops

        if op == 0:  # block scaling
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.04))
            candidate[start : start + width] *= factor
        elif op == 1:  # single scaling
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.2))
            candidate[idx] *= factor
        elif op == 2:  # targeted mass redistribution
            # Use current_conv
            k_max = int(np.argmax(current_conv))
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx_arr = np.arange(lo, hi + 1)
            partner_arr = k_max - idx_arr
            contrib = candidate[idx_arr] * candidate[partner_arr]
            # choose max contribution
            max_idx = idx_arr[np.argmax(contrib)]
            min_idx = np.argmin(candidate)  # index with minimal value
            # ensure distinct
            if max_idx == min_idx:
                # just pick random different low
                min_idx = rng.integers(0, n)
                while min_idx == max_idx:
                    min_idx = rng.integers(0, n)
            # fraction to move
            alpha = rng.uniform(0.02, 0.12)
            # remove from max_idx and its partner
            partner = k_max - max_idx
            if max_idx != partner:
                removed_i = candidate[max_idx] * alpha
                removed_j = candidate[partner] * alpha
                candidate[max_idx] -= removed_i
                candidate[partner] -= removed_j
                total_removed = removed_i + removed_j
                candidate[min_idx] += total_removed
            else:
                # diagonal case
                removed = candidate[max_idx] * alpha
                candidate[max_idx] -= removed
                candidate[min_idx] += removed
        elif op == 3:  # smooth two neighbours
            i = rng.integers(0, n - 1)
            avg = (candidate[i] + candidate[i + 1]) * 0.5
            candidate[i] = avg
            candidate[i + 1] = avg
        else:  # small random noise additive
            # add Gaussian noise scaled relative to current values
            idxs = rng.choice(n, size= max(1, n // 200), replace=False)
            noise = rng.normal(0.0, 0.02, size=idxs.shape)  # relative factor
            candidate[idxs] *= np.exp(noise)

        # clip to allowed range
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        cand_score, cand_conv = _fast_score(candidate)

        # acceptance
        if cand_score < current_score or rng.random() < np.exp(-(cand_score - current_score) / max(temp, 1e-12)):
            current_seq = candidate
            current_score = cand_score
            current_conv = cand_conv

        # update best
        if cand_score < best_score:
            best_seq = candidate
            best_score = cand_score
            best_conv = cand_conv

    # Return best as list
    return [float(x) for x in best_seq]

import numpy as np
import time

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _initial_exponential(n: int, c: float) -> np.ndarray:
    """a_i = exp(-c*i)  (decreasing exponential)."""
    return np.exp(-c * np.arange(n, dtype=np.float64))


def _fast_score(seq: np.ndarray):
    """Return (score, full convolution array) for a non‑negative seq."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())      # next power of two
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv
    max_val = np.max(conv)
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv
