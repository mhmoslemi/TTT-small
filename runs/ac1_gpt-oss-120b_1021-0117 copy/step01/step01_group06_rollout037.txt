def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0,float(budget_s)-0.2)

    # Determine n, get initial vectors.
    if "height_sequence_1" in globals():
        a_parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = a_parent.size
    else:
        n = 256
        a_parent = np.ones(n, dtype=np.float64)

    # sqrt_series
    def sqrt_series(n):
        a = np.empty(n, dtype=np.float64)
        a[0]=1.0
        for k in range(1,n):
            if k==1:
                s=0.0
            else:
                s = float(np.dot(a[1:k], a[k-1:0:-1]))
            a[k]=(1.0 - s)/2.0
        return a

    a_sqrt = sqrt_series(n)

    # Normalize to simplex sum=1
    a_parent = np.maximum(a_parent,0.0)
    if a_parent.sum()==0:
        a_parent = np.ones(n)
    a_parent = a_parent / a_parent.sum()
    a_sqrt = a_sqrt / a_sqrt.sum()

    # start from best of them
    L = 1 << ((2*n-1).bit_length())
    def conv_full(x):
        fv = np.fft.rfft(x, n=L)
        return np.fft.irfft(fv*fv, n=L)[:2*n-1]

    def ac1_score(x):
        conv = conv_full(x)
        maxb = conv.max()
        # sum x should be 1 after projection, but compute anyway
        s = x.sum()
        if s==0: return np.inf
        return 2.0 * n * maxb / (s*s)

    # compute scores
    score_parent = ac1_score(a_parent)
    score_sqrt = ac1_score(a_sqrt)
    if score_parent <= score_sqrt:
        cur = a_parent.copy()
        cur_score = score_parent
    else:
        cur = a_sqrt.copy()
        cur_score = score_sqrt

    best_vec = cur.copy()
    best_score = cur_score

    momentum = np.zeros_like(cur)

    step = 0.2
    beta = 0.9
    iter_count=0
    while time.monotonic() < deadline:
        iter_count+=1
        # Gradient descent step
        conv = conv_full(cur)
        maxb = conv.max()
        kmax = int(conv.argmax())

        # subgradient
        idx = kmax - np.arange(n)
        mask = (idx >=0) & (idx < n)
        g = np.zeros_like(cur)
        g[mask] = 2.0*cur[idx[mask]]

        # momentum update
        momentum = beta*momentum - step * g
        cand = cur + momentum
        cand = _project_simplex(cand)

        cand_score = ac1_score(cand)

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            # possibly enlarge step
            step = min(step*1.05, 5.0)
            # update best
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
        else:
            # reduce step and reset momentum
            step *= 0.6
            momentum = np.zeros_like(cur)
            # occasional random hill climbing
            if rng.random() < 0.02:
                i = rng.integers(0, n)

                    j = rng.integers(0, n-1)

                        cur = trial

def _project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]

    """First n coefficients of a where a*a = 1+...+x^{2n-2}."""
    a = np.empty(n, dtype=np.float64)

        a[k] = (1.0 - s) / 2.0

    base = base / base.sum()

    sqrt_vec = _sqrt_series(n)
    sqrt_vec = np.maximum(sqrt_vec, 0.0)
    sqrt_vec = sqrt_vec / sqrt_vec.sum()

    # ------------------------------------------------------------------
    # Convolution helpers (FFT‑based, reuse a fixed length)
    # ------------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    def _conv_full(x: np.ndarray) -> np.ndarray:
        fv = np.fft.rfft(x, n=L)
        return np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]

    def _ac1(x: np.ndarray) -> float:
        conv = _conv_full(x)
        maxb = conv.max()
        s = x.sum()
        return float(2.0 * n * maxb / (s * s)) if s > 0 else np.inf

    # ------------------------------------------------------------------
    # Choose the better of the two starters
    # ------------------------------------------------------------------
    cur = base if _ac1(base) <= _ac1(sqrt_vec) else sqrt_vec
    cur = cur.copy()
    cur_score = _ac1(cur)
    best_vec = cur.copy()
    best_score = cur_score

    step = 1.0                      # initial line‑search step size

    # ------------------------------------------------------------------
    # Main optimisation loop (projected sub‑gradient + occasional hill‑climb)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        conv = _conv_full(cur)
        maxb = conv.max()
        kmax = int(conv.argmax())

        # sub‑gradient of the active convolution entry:
        idx = kmax - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[idx[mask]]

        # back‑tracking line search on the simplex
        improved = False
        t = step
        for _ in range(5):
            cand = _project_simplex(cur - t * g)
            cand_score = _ac1(cand)
            if cand_score < cur_score:
                cur, cur_score = cand, cand_score
                if cand_score < best_score:
                    best_vec, best_score = cand.copy(), cand_score
                step = t * 1.1               # modestly enlarge after success
                improved = True
                break
            t *= 0.5
        if not improved:
            step *= 0.7                       # shrink if no progress

        # occasional random mass‑transfer move (helps escape plateaus)
        if rng.random() < 0.01:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.3
                delta = cur[i] * frac
                trial = cur.copy()
                trial[i] -= delta
                trial[j] += delta
                trial = _project_simplex(trial)
                trial_score = _ac1(trial)
                if trial_score < cur_score:
                    cur, cur_score = trial, trial_score
                    if trial_score < best_score:
                        best_vec, best_score = trial.copy(), trial_score

        if step < 1e-8:          # numerical stall
            break

    # --------------------------------------------------------------
    # Return a clean simplex vector (values already lie in [0,1000])
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total > 0:
        best_vec = best_vec / total
    return [float(v) for v in best_vec]
