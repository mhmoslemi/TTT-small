def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 3.0)
    # Determine n
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.shape[0]
        if n == 0:
            raise Exception
    except Exception:
        n = 256
        parent = None

    L = 1 << (2*n - 2).bit_length()  # next pow2

    # define score function
    def ac1_score(arr):
        fa = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
        maxb = float(conv.max())
        s = float(arr.sum())
        if s < 1e-12:
            return float('inf')
        return 2.0 * n * maxb / (s*s), conv, maxb

    # initial candidates
    best = None
    best_score = float('inf')

    # candidate from binomial
    base = np.empty(n, dtype=np.float64)
    base[0] = 1.0
    for i in range(1,n):
        base[i] = base[i-1] * (2*i - 1) / (2*i)
    s, _, _ = ac1_score(base)
    best = base.copy(); best_score = s

    # candidate from parent
    if parent is not None:
        s = ac1_score(parent)[0]
        if s < best_score:
            best = parent.copy(); best_score = s

    # candidate from power law initial guess p=0.5 (c=0)
    i_arr = np.arange(1,n+1,dtype=np.float64)

    s = ac1_score(a)[0]
    if s < best_score:
        best = a.copy(); best_score = s

    # current = best
    cur = best.copy()
    cur_score, cur_conv, cur_maxb = ac1_score(cur)

    # parameters for param search
    p_best = 0.5

    best_params = (p_best, c_best)

    temperature = 1.0
    decay = 0.9995
    # iteration loops
    iter_count = 0

        iter_count += 1
        # 10% chance do param search
        if rng.random() < 0.12:
            # propose new (p,c)
            if rng.random() < 0.5:

                    cand[j] += delta

            else:

                m = int(np.argmax(cur_conv))

                j = m - i
                if 0 <= j < n:
                    # remove fraction from i and j
                    frac = rng.random() * 0.25
                    delta_i = cur[i] * frac
                    delta_j = cur[j] * frac
                    total = delta_i + delta_j
                    k = int(rng.integers(0, n))
                    while k == i or k == j:

                    cand = cur.copy()
                    cand[i] -= delta_i
                    cand[j] -= delta_j
                    cand[k] += total
                else:
                    # fallback to simple transfer
                    i = int(rng.integers(0, n))
                    j = int(rng.integers(0, n))
                    if i == j:

                    frac = rng.random() * 0.35

            np.clip(cand, 0.0, 1000.0, out=cand)

import time, math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score(arr: np.ndarray, n: int, L: int):
    """
    Returns (bound, full convolution, max(conv)).
    Bound = 2*n*max(conv) / (sum(arr)^2).
    """
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf"), conv, max_b
    return float(2.0 * n * max_b / (s * s)), conv, max_b
