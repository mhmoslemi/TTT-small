def _step_gradient(cur, g, step_factor):
    # compute η max as described to keep nonnegative after subtraction
    positive = g > 0
    if np.any(positive):
        # compute max step to keep cur - η*g >= 0
        # cur_i - η*g_i >= 0 => η <= cur_i / g_i for each i
        # take min ratio over positive g
        eta_max = np.min(cur[positive] / g[positive])
        # restrict to e.g. 0.1 of that
        eta = step_factor * eta_max
    else:
        eta = 0.0
    cand = cur - eta * g
    # ensure non-negative and renorm
    cand = np.maximum(cand, 0.0)
    total = cand.sum()
    if total == 0:
        cand = np.full_like(cand, 1.0 / len(cand))
    else:
        cand /= total
    return cand

import time
import numpy as np

# ------------------------------------------------------------
# utilities
# ------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of a where a*a = 1+…+x^{2n-2}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # sum_{i=1}^{k-1} a_i * a_{k-i}
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


def _conv_full(x: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution x*x, full length 2*len(x)-1, using FFT."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]
    return conv


def _score_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact bound used by the evaluator."""
    return float(2.0 * n * conv.max() / (total * total))


def _gradient_at_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of the maximal convolution coefficient."""
    n = a.size
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    g[mask] = a[j[mask]]
    if k_max % 2 == 0:
        i_mid = k_max // 2
        if 0 <= i_mid < n:
            g[i_mid] = 2.0 * a[i_mid]
    return g


def _make_candidate_by_gradient(a: np.ndarray, g: np.ndarray, factor: float) -> np.ndarray:
    """Move a a little opposite to the gradient and re‑normalise."""
    pos = g > 1e-12
    if not np.any(pos):
        return a.copy()
    # maximal step that keeps all entries non‑negative after subtraction
    eta_max = np.min(a[pos] / g[pos])
    eta = factor * eta_max
    cand = a - eta * g
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s == 0.0:
        cand.fill(1.0 / cand.size)
    else:
        cand /= s
    return cand


def _make_candidate_random(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random fraction of mass from one index to another."""
    n = a.size
    i = rng.integers(0, n)
    if a[i] == 0.0:
        return a.copy()
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    frac = rng.random() * 0.25          # up to 25 % of source mass
    delta = a[i] * frac
    cand = a.copy()
    cand[i] -= delta
    cand[j] += delta
    # simple renormalisation (keeps sum = 1)
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    return cand

    rng = np.random.default_rng(seed)
    # give a safety margin for the sandbox
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # -----------------------------------------------------------------
    # initialise: provided parent or analytic sqrt‑series
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():

    a = np.maximum(a, 0.0)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length – power of two

    # -----------------------------------------------------------------
    # evaluate the start point
    # -----------------------------------------------------------------
    conv = _conv_full(a, L)
    cur_score = _score_from_conv(conv, a.sum(), n)
    best_vec = a.copy()
    best_score = cur_score

    # -----------------------------------------------------------------
    # optimisation loop
    # -----------------------------------------------------------------
    step_factor = 0.5      # for gradient moves
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # recompute convolution and the index of the current maximum
        conv = _conv_full(a, L)
        max_idx = int(np.argmax(conv))

        # ---------- gradient‑driven move ----------
        g = _gradient_at_max(a, max_idx)
        cand = _make_candidate_by_gradient(a, g, step_factor)
        conv_cand = _conv_full(cand, L)
        sc = _score_from_conv(conv_cand, cand.sum(), n)
        if sc < best_score:

            a, cur_score = cand, sc
        else:

        cand = _make_candidate_random(a, rng)

        if sc < best_score:

            a, cur_score = cand, sc

            # simple Metropolis‑type acceptance for exploration

        # safety stop if we have already found a very good bound

    # return the best candidate found
    # -----------------------------------------------------------------
    return [float(v) for v in best_vec]
