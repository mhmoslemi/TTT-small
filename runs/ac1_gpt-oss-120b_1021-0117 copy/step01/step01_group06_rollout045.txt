def sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        # compute s = sum_{i=1}^{k-1} a[i] * a[k-i]; for i from 1 to k-1
        # Equivalent to dot product of a[1:k] and a[k-1:0:-1]
        s = 0.0
        if k > 1:
            s = np.dot(a[1:k], a[k-1:0:-1])
        a[k] = (1.0 - s) / 2.0
    return a

import numpy as np

def _sqrt_series(n: int) -> np.ndarray:
    """Return the first n coefficients a satisfying (a*a)_k = 1 for 0≤k<n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k > 1:
            # sum_{i=1}^{k-1} a_i * a_{k-i}
            s = np.dot(a[1:k], a[k-1:0:-1])
        else:
            s = 0.0
        a[k] = (1.0 - s) * 0.5
    # numerical safety
    a[a < 0.0] = 0.0
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty sequence of non‑negative floats with a low AC1 bound."""
    # length of the problem (use provided parent if present)
    if "height_sequence_1" in globals():
        n = int(np.asarray(height_sequence_1).size)
    else:
        n = 256

    # analytic sqrt‑series solution (no further normalisation)
    vec = _sqrt_series(n)

    # ensure evaluator constraints (already satisfied, but clamp just in case)
    vec = np.clip(vec, 0.0, 1000.0)

    # convert to plain Python list
    return [float(x) for x in vec]
