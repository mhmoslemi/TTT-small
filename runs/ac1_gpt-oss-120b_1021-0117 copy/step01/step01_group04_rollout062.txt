    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    try:

        parent = None

    L = 1 << ((2*n - 2).bit_length())   # next power of 2 > 2*n-2

    def score(arr):
        fa = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fa * fa, n=L)
        max_b = conv[:2*n-1].max()
        s = arr.sum()
        if s < 1e-12:
            return float('inf')
        return 2.0 * n * max_b / (s * s)

    # generate candidates
    def gen_binomial():
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i-1] * (2*i - 1) / (2*i)
        return a

    def gen_power(p):
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.power(i + 1.0, p)
        return a

    def gen_geo(r):
        i = np.arange(n, dtype=np.float64)
        a = np.power(r, i)
        return a

    # initial candidates list
    cand_list = []
    base = gen_binomial()
    cand_list.append(base)
    # add geometric candidates with r from 0.5 to 0.9999 in logscale maybe
    for r in [0.9, 0.95, 0.97, 0.98, 0.99]:
        cand_list.append(gen_geo(r))
    # power law candidates
    for p in [0.3, 0.4, 0.5, 0.6, 0.7]:
        cand_list.append(gen_power(p))

    # evaluate them and keep best
    best = None
    best_score = float('inf')
    for c in cand_list:
        sc = score(c)
        if sc < best_score:
            best_score = sc
            best = c.copy()
    # include parent if present
    if parent is not None:
        s = score(parent)
        if s < best_score:
            best = parent.copy()
            best_score = s

    cur = best.copy()
    cur_score = best_score

    temperature = 1.0
    decay = 0.9985   # slower decay
    # main loop
    while time.monotonic() < deadline:
        # Choose move type: 0-block scaling, 1-mass transfer, 2-random perturb
        move_type = rng.random()
        cand = cur.copy()
        if move_type < 0.5:
            # block scaling
            width = int(rng.integers(max(1, n//200), max(2, n//10)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.15))
            cand[start:start+width] *= factor
        elif move_type < 0.8:

            frac = rng.random() * 0.5

            cand[i] -= delta
            cand[j] += delta
        else:

            factor = math.exp(rng.normal(0.0, 0.06))
            cand *= factor

        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = score(cand)
        # update best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            # if we beat target, break
            if best_score < 1.50274349:
                break

        # Simulated annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:

        if accept:

        # occasional restart from best
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    return [float(x) for x in best]

import math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _gen_geo(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i = r**i (a_0 = 1)."""
    return np.power(r, np.arange(n, dtype=np.float64))


def _gen_pow(n: int, p: float) -> np.ndarray:
    """Power‐law a_i = 1/(i+1)**p."""
    return 1.0 / np.power(np.arange(n, dtype=np.float64) + 1.0, p)


def _ac1_info(arr: np.ndarray, n: int, L: int):
    """
    Returns (score, max_conv, max_index, sum) using FFT.
    """
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]

    max_idx = int(conv.argmax())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf'), max_conv, max_idx, s
    score = 2.0 * n * max_conv / (s * s)
    return score, max_conv, max_idx, s
