def compute_conv_and_score(a, n, L):
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    conv = conv[:2*n-1]
    M = conv.max()
    s = a.sum()
    if s < 1e-12:
        return None, None, None, None # not used
    score = 2.0 * n * M / (s * s)
    return conv, M, s, score

def compute_gradient(a, conv, M, s, n):
    # conv is full conv array
    max_indices = np.flatnonzero(np.abs(conv - M) <= 1e-12)  # tolerance
    # compute g_temp = 2 * sum_{k in max_indices} a[k-i]
    # We'll compute using vectorized approach with loops
    i_vals = np.arange(a.size)
    g2 = np.zeros_like(a)
    for k in max_indices:
        # valid indices for i: 0 <= k-i < n
        # compute mask
        valid = (k - i_vals >= 0) & (k - i_vals < a.size)
        # indices in a to fetch
        idx = k - i_vals[valid]
        g2[valid] += a[idx]  # add a_{k-i}
    # multiply by 2 to get derivative of M
    g2 *= 2.0
    factor_M = (2.0 * n) / (s * s)
    factor_S = (4.0 * n * M) / (s * s * s)
    grad = factor_M * g2 - factor_S
    return grad

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a[i] = C(2i,i)/4**i for i=0..n-1 via recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a
