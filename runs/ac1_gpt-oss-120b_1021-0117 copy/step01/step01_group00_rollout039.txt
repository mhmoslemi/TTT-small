import time
import numpy as np

def _sidon_like_sequence(n):
    """Classic Sidon‑like coefficients a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def _conv_max(a, L):
    """Maximum of full convolution a*a using FFT of length L."""
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * len(a) - 1]
    return conv.max()

        max_idx = int(conv.argmax())

        # ------------------------------------------------------------------
        # If the peak is the centre (most common for Sidon‑like) we try to
        # shift mass from indices with a large partner product to those with
        # a tiny partner product.
        # ------------------------------------------------------------------
        if max_idx == n - 1:
            prod = cur * cur[rev_idx]                # a_i * a_{n-1-i}
            i_max = int(prod.argmax())
            i_min = int(prod.argmin())
            if i_max != i_min and cur[i_max] > 0:
                eps = min(cur[i_max] * 0.4, 0.01)     # modest amount
                cand = cur.copy()
                cand[i_max] -= eps
                cand[i_min] += eps
                cand = np.clip(cand, 0.0, None)       # safety
                cand /= cand.sum()                    # renormalise
                new_score = score_of(cand)

                if new_score < cur_score:
                    cur = cand
                    cur_score = new_score
                    if new_score < best_score:
                        best_vec = cand.copy()
                        best_score = new_score
                        if best_score < 1.50274349:
                            break
            # occasional random shake
            if rng.random() < 0.001:
                cur = best_vec.copy()
                cur_score = best_score
            continue

        # ------------------------------------------------------------------
        # Otherwise the peak is at some shift < centre.  Reduce it by moving
        # mass out of the contributing interval.
        # ------------------------------------------------------------------
        if max_idx < n:
            low, high = 0, max_idx
        else:
            low, high = max_idx - (n - 1), n - 1

        # pick a donor inside the contributing interval
        i = rng.integers(low, high + 1)
        if cur[i] <= 1e-12:
            continue
        eps = cur[i] * rng.random() * 0.5
        # pick a receiver outside that interval
        for _ in range(5):
            k = rng.integers(0, n)
            if not (low <= k <= high):
                break
        else:
            # deterministic fallback
            k = (high + 1 + rng.integers(0, n - (high - low + 1))) % n

        cand = cur.copy()
        cand[i] -= eps
        cand[k] += eps
        cand = np.clip(cand, 0.0, None)
        cand /= cand.sum()
        new_score = score_of(cand)

        if new_score < cur_score:
            cur = cand
            cur_score = new_score
            if new_score < best_score:
                best_vec = cand.copy()
                best_score = new_score
                if best_score < 1.50274349:
                    break

        # occasional restart from the overall best
        if rng.random() < 0.001:
            cur = best_vec.copy()
            cur_score = best_score

    # ------------------------------------------------------------------
    # final scaling (any positive scale works; keep values modest)
    # ------------------------------------------------------------------
    out = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in out]
