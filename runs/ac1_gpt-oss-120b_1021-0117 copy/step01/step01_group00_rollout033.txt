import numpy as np
import time

def _sidon_like_sequence(n: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4**i sequence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a

def _interleaved_sequence(seq: np.ndarray) -> np.ndarray:
    """Place the largest values on even positions, the smallest on odd."""
    v = np.sort(seq)[::-1]          # descending
    n = v.size
    half = (n + 1) // 2
    large = v[:half]
    small = v[half:]
    out = np.empty_like(v)
    out[0::2] = large
    out[1::2] = small
    return out
