def simplex_projection(v):
    # Ensure non-negative sum 1 projection onto simplex.
    # v can be any real array.
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero? return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    try:
        n = len(height_sequence_1)
        a0 = np.maximum(height_sequence_1.astype(np.float64), 0.0)
        a0_sum = a0.sum()
        if a0_sum < 1e-12:
            # fallback to binomial series
            raise ValueError
    except Exception:
        # fallback generate binomial series normalized to sum=1
        n = 5000
        a0 = np.empty(n, dtype=np.float64)
        a0[0] = 1.0
        for i in range(1, n):
            a0[i] = a0[i-1]*(1.0 - 1.0/(2.0*i))
        # ensure nonneg
        a0 = np.clip(a0, 0.0, None)
        a0 = a0 / a0.sum()

    # normalize to sum=1 (simplex)
    a = a0 / a0.sum()

    # compute initial score
    def compute_score(arr):
        conv = np.convolve(arr, arr)
        maxc = conv.max()
        s = arr.sum()
        if s == 0:
            return np.inf
        return 2.0 * len(arr) * maxc / (s**2)

    best = a.copy()
    best_score = compute_score(best)

    # optionally compute uniform baseline
    uniform = np.full_like(best, 1.0 / n)
    uniform_score = compute_score(uniform)
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score

    # projection onto simplex
    def proj_simplex(v):
        # returns vector on simplex (nonnegative, sum=1)
        if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
            return v
        # sort descending
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(len(v)) + 1
        cond = u - cssv / ind > 0
        if not np.any(cond):
            # fallback uniform
            return np.full_like(v, 1.0 / len(v))
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0)
        return w

    # iterate
    eta = 0.5  # initial step size
    decay = 0.9995
    min_eta = 0.001

    while time.monotonic() < deadline:

def local_refine(a, current_score, time_limit):
    n = len(a)
    best_local = a.copy()
    best_local_score = current_score
    # parameters
    step_factor=0.3
    min_step=1e-7
    while time.monotonic() < time_limit:
        # compute conv, find max index
        conv = np.convolve(best_local, best_local)
        maxc = conv.max()
        k = int(conv.argmax())
        # partner find src: index i where a_i * a_{k-i} contributes most? simplest: source index which appears in sum and has large value.
        # Use partner as previous approach
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1)
        partner = np.zeros_like(best_local)
        partner[idx] = best_local[k - idx]
        src = int(partner.argmax())
        # destination choose index with minimal partner or random
        min_val = partner.min()
        dest_candidates = np.where(partner == min_val)[0]
        if dest_candidates.size > 0:
            dest = int(rng.choice(dest_candidates))
        else:
            dest = int(rng.integers(0, n))
        if src == dest:
            # random perturb
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand = best_local.copy()
            cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.2)))
            cand = np.clip(cand, 0.0, 1000.0)
            s = cand.sum()
            if s > 0:
                cand = cand / s
            cand_score = compute_score(cand)
            if cand_score < best_local_score:
                best_local, best_local_score = cand, cand_score
            continue

        delta = best_local[src] * step_factor
        if delta < min_step:
            delta = min_step
        if delta > best_local[src]:
            delta = best_local[src]
        cand = best_local.copy()
        cand[src] -= delta
        cand[dest] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        if cand_score < best_local_score:
            best_local, best_local_score = cand, cand_score
            # optionally reduce step
            step_factor = max(0.02, step_factor * 0.998)
    return best_local, best_local_score

import time, numpy as np

def simplex_projection(v):
    # Project onto simplex sum=1, v >= 0
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def compute_score(arr, n):
    conv = np.convolve(arr, arr)
    maxc = conv.max()
    s = arr.sum()
    if s == 0:
        return np.inf
    return 2.0 * n * maxc / (s * s)

def compute_gradient(a, n, max_idxs):
    g = np.zeros_like(a)
    for k in max_idxs:
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1)
        g[idx] += a[k - idx]
        if k < n:
            idx2 = np.arange(k, n)
            g[idx2] += a[idx2 - k]
    return g

def local_refine(a, n, current_score, rng, deadline):
    # simple mass‑transfer refinement similar to previous program
    best = a.copy()
    best_score = current_score
    step_factor = 0.3
    min_step = 1e-7
    while time.monotonic() < deadline:
        conv = np.convolve(best, best)
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = np.zeros_like(best)
        partner[idx] = best[k - idx]
        src = int(partner.argmax())
        min_val = partner.min()
        dest_candidates = np.where(partner == min_val)[0]
        if dest_candidates.size:
            dest = int(rng.choice(dest_candidates))
        else:
            dest = int(rng.integers(0, n))
        if src == dest:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))

        final = np.full_like(final, 1.0 / final.size)
