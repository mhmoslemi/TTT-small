def _conv_self(a):
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2 * n - 1]
    return conv

def _score_seq(a):
    conv = _conv_self(a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * a.shape[0] * max_b / (s * s)

import time
import numpy as np

# --------------------------------------------------------------
# Fast linear convolution using a real FFT (O(n log n)).
# --------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------
# Exact objective used by the evaluator.
# --------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


# --------------------------------------------------------------
# Classic sqrt‑series a_i = C(2i,i)/4^i  (length n).
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


# --------------------------------------------------------------
# Main entry point required by the harness.
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    # leave a small safety margin (the harness may kill the process)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initial sequence – use the supplied parent if it exists,
    # otherwise fall back to the classic sqrt‑series.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = cur.size
    else:
        n = 5582
        cur = None

    sqrt_seq = _sqrt_series(n)

    # pick the better of the two as the starting point
    best = sqrt_seq.copy()
    best_score = _score(best)
    if cur is not None:
        parent_score = _score(cur)
        if np.isfinite(parent_score) and parent_score < best_score:
            best = cur.copy()
            best_score = parent_score

    # current working candidate (not necessarily the best so far)
    cur = best.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # Helper: pick a source index weighted by its contribution to the
    # current maximum of the convolution.
    # ------------------------------------------------------------------
    def _select_src(a, max_idx):
        if max_idx < n:                             # left‑hand overlap
            start, stop = 0, max_idx + 1
        else:                                        # right‑hand overlap
            start = max_idx - (n - 1)
            stop = n
        idx = np.arange(start, stop, dtype=np.int64)
        contrib = a[idx] * a[max_idx - idx]           # a_i * a_{max_idx-i}

        if total == 0.0:
            return rng.integers(start, stop)
        # weighted choice

        src_local = int(np.searchsorted(np.cumsum(contrib), rnd, side='right'))
        return int(idx[src_local])

    # ------------------------------------------------------------------
    # Main optimisation loop.

    eps_transfer = 0.03           # up‑to 3 % of transferable mass each move
    block_scale_prob = 0.02
