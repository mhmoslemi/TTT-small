import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT‑based linear convolution and score)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT of length fft_len."""
    f = np.fft.rfft(a, n=fft_len)
    c = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    return c


def _score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact AC1 upper‑bound score for a non‑negative step function."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max() / (s * s))


def _second_max(arr: np.ndarray) -> float:
    """Return the second largest value (may be equal to the maximum)."""
    if arr.size <= 1:
        return arr.max()
    # -2 picks the second‑largest element irrespective of ties
    return np.partition(arr, -2)[-2]
