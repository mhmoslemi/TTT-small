def compute_score(v):
    n = len(v)
    L = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(A * A, n=L)[:2*n-1]

    sum_v = np.sum(v)
    return 2.0 * n * max_conv / (sum_v * sum_v)

Simplex projection algorithm:

def proj_simplex(v, s=1.0):
    """Projection of vector v onto simplex {x >=0, sum x = s}."""
    # v can be any shape
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    # find rho
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # get n
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 5000
    # next power of 2 for convolution
    L = 1 << ((2 * n - 1).bit_length())

    # initial candidate
    def init_seq():
        # if have parent, jitter it
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            base = np.clip(base, 0.0, 1000.0)
        except Exception:
            base = np.full(n, 1.0 / n, dtype=np.float64)
        # add small random noise
        noise = rng.normal(0.0, 0.02, size=n)
        cand = base * (1.0 + noise)
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s < 1e-12:
            cand = np.full_like(cand, 1.0 / n)
            s = cand.sum()
        cand = cand / s   # normalize to sum 1
        return cand

    def compute_score_and_conv(v):
        A = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(A * A, n=L)[:2 * n - 1]
        max_conv = conv.max()
        score = 2.0 * n * max_conv / (v.sum() ** 2)
        return score, conv

    # generate several starting points
    best_seq = init_seq()
    best_score, _ = compute_score_and_conv(best_seq)

    # also consider uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_score, _ = compute_score_and_conv(uniform)
    if uni_score < best_score:
        best_seq = uniform
        best_score = uni_score

    # hill climbing parameters
    step_factor = 0.4
    min_step = 1e-8

    cur_seq = best_seq.copy()
    cur_score = best_score

    # main loop
    while time.monotonic() < deadline:
        # compute conv and max index
        A = np.fft.rfft(cur_seq, n=L)
        conv = np.fft.irfft(A * A, n=L)[:2 * n - 1]
        k = int(conv.argmax())
        # partners: a_at = cur_seq[k - idx] for idx in range(i_start,i_end+1)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)

        src = int(idx_range[np.argmax(partner)])
        # dest: index where partner is minimal (could be many)
        min_val = partner.min()
        dest_candidates = idx_range[partner == min_val]
        if dest_candidates.size == 0:
            dest = rng.integers(0, n)
        else:
            dest = int(rng.choice(dest_candidates))

        if src == dest:
            # random perturbation: scale a random block
            width = int(rng.integers(max(1, n // 200), max(2, n // 50)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand = cur_seq.copy()
            factor = np.exp(rng.normal(0.0, 0.1, size=stop-start))
            cand[start:stop] *= factor
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s > 0:
                cand = cand / s
            cand_score, _ = compute_score_and_conv(cand)
            if cand_score < cur_score:
                cur_seq = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best_seq = cur_seq.copy()
                    best_score = cur_score
            continue

        delta = cur_seq[src] * step_factor
        if delta < min_step:
            delta = min_step

        if cand_score < cur_score:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
            # gradually increase step_factor up to 0.5
            step_factor = min(0.5, step_factor * 1.02)
        else:

            step_factor = max(0.01, step_factor * 0.95)

        # occasional random shake to escape plateau
        if rng.random() < 0.003:
            width = int(rng.integers(max(1, n // 300), max(2, n // 50)))
            start = int(rng.integers(0, max(1, n - width + 1)))

            factor = np.exp(rng.normal(0.0, 0.2, size=stop-start))
            cand2[start:stop] *= factor
            cand2 = np.clip(cand2, 0.0, None)

            cand2_score, _ = compute_score_and_conv(cand2)
            if cand2_score < cur_score:

                cur_score = cand2_score
                if cand2_score < best_score:
                    best_seq = cand2.copy()
                    best_score = cand2_score
                step_factor = max(step_factor, 0.05)  # reset step

    final = np.clip(best_seq, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)
    return [float(v) for v in final]

def proj_simplex(v, s=1.0):
    # ensure v is numpy array
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _proj_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                      # descending
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_and_conv(v: np.ndarray, L: int, n: int):
    """Return (score, full linear convolution) for a non‑negative vector."""
    A = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    max_conv = conv.max()
    score = 2.0 * n * max_conv / (v.sum() ** 2)
    return float(score), conv


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step function with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # -------------------------------------------------------------
    # base sequence (parent if available, otherwise uniform)
    try:
        n = int(len(height_sequence_1))
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        n = 5000
        base = np.full(n, 1.0 / n, dtype=np.float64)

    base = np.clip(base, 0.0, 1000.0)
    if base.sum() < 1e-12:
        base = np.full(n, 1.0 / n, dtype=np.float64)
    base = base / base.sum()                     # normalize to sum == 1

    # -------------------------------------------------------------
    # auxiliary functions (no closures)
    L = _next_pow2(2 * n - 1)

    # -------------------------------------------------------------
    # generate a few random starts and keep the best one
    best_seq = base.copy()
    best_score, _ = _score_and_conv(best_seq, L, n)

    def _rand_start():
        # Dirichlet‑like (gamma) sampling; parameters >1 keep entries relatively even
        x = rng.gamma(2.0, 1.0, size=n)
        s = x.sum()
        return x / s if s > 0 else np.full(n, 1.0 / n)

    for _ in range(3):
        if time.monotonic() >= deadline:
            break
        cand = _rand_start()
        cand_score, _ = _score_and_conv(cand, L, n)
        if cand_score < best_score:
            best_seq, best_score = cand, cand_score

    cur_seq = best_seq.copy()
    cur_score = best_score

    # -------------------------------------------------------------
    # Phase 1: gradient‑like descent on the active convolution index
    step = 0.5
    gd_iters = 0
    max_gd_iters = 20000
    while time.monotonic() < deadline - 5 and gd_iters < max_gd_iters:
        cur_score, conv = _score_and_conv(cur_seq, L, n)
        k = int(conv.argmax())

        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_start > i_end:
            break
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = cur_seq[k - idx]

        # gradient of conv[k] w.r.t. a (each contributing index appears twice)
        g = np.zeros_like(cur_seq)
        g[idx] = 2.0 * partner

        cand = cur_seq - step * g
        cand = _proj_simplex(cand, s=cur_seq.sum())
        cand_score, _ = _score_and_conv(cand, L, n)

        if cand_score < cur_score:

            if cand_score < best_score:

        if src == dest:
