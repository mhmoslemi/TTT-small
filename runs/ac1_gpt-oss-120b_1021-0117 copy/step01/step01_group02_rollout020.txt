import time
import numpy as np

# ---------- helper functions (top‑level, no closures) -----------------

def _next_pow_two(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _full_conv(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (real input)."""
    n = a.shape[0]
    size = _next_pow_two(2 * n - 1)
    ft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(ft * ft, n=size)[: 2 * n - 1]
    return conv.real


def _score(a: np.ndarray) -> float:
    """Objective 2*n*max(conv)/(sum(a)^2).  Assumes a non‑negative."""
    n = a.shape[0]
    conv = _full_conv(a)
    maxc = conv.max()
    s = a.sum()
    if s == 0.0:
        return np.inf
    return 2.0 * n * maxc / (s * s)


def _partner_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a given convolution index k (0 <= k < 2n-1),
    return the vector p where p[i] = a[k-i] if 0<=k-i<n else 0.
    """
    n = a.shape[0]
    p = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=np.int64)
    p[idx] = a[k - idx]
    return p

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.shape[0]
        a0 = base.copy()
        s0 = a0.sum()
        if s0 <= 0.0:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a0 = a0 / s0                     # normalise to sum == 1
    except Exception:
        # fallback if no parent is provided
        n = 5000
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    a0 = np.clip(a0, 0.0, 1000.0)

    # ----------- keep best so far -----------------------------------------
    best = a0
    best_score = _score(best)

    # uniform fallback (easy baseline)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = _score(uniform)
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # ----------- parameters ------------------------------------------------
    step_factor = 0.5          # fraction of source height to move
    min_step = 1e-8
    target_score = 1.50274349  # desired upper bound
    no_improve = 0            # counter for stagnation

    # ----------- main loop ------------------------------------------------
    while time.monotonic() < deadline:
        # compute full convolution once per iteration
        conv = _full_conv(best)
        k = int(conv.argmax())               # index of current peak

        # partner values for that peak
        partner = _partner_vector(best, k)

        # contribution of each position to the peak (product a_i * partner_i)
        contrib = best * partner
        src = int(contrib.argmax())           # most damaging index

        # destination: index with smallest contribution (often zero)
        min_contrib = contrib.min()
        cand_dest = np.where(contrib == min_contrib)[0]
        if cand_dest.size == 0:
            dest = rng.integers(0, n)
        else:
            dest = int(rng.choice(cand_dest))
            if dest == src:
                # ensure a move; pick a random other index
                dest = (src + rng.integers(1, n)) % n

        # amount to transfer
        delta = best[src] * step_factor
        if delta < min_step:
            delta = min_step
        if delta > best[src]:
            delta = best[src]

        # generate candidate by moving delta from src to dest
        cand = best.copy()
        cand[src] -= delta
        cand[dest] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        # renormalise to keep total mass unchanged (protect against drift)
        total = cand.sum()
        if total > 0.0:
            cand = cand / total * best.sum()
        else:
            continue

        cand_score = _score(cand)

        # accept improvements (or very tiny uphill moves)
        if cand_score < best_score:
            best, best_score = cand, cand_score
            step_factor = max(0.01, step_factor * 0.995)
            no_improve = 0
        else:
            no_improve += 1
            # occasional random shake – helps escape plateaus
            if rng.random() < 0.008:
                width = int(rng.integers(max(1, n // 250), max(2, n // 30)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                stop = start + width
                block = cand[start:stop] * np.exp(rng.normal(0.0, 0.14))
                cand2 = cand.copy()
                cand2[start:stop] = block
                cand2 = np.clip(cand2, 0.0, 1000.0)
                total2 = cand2.sum()
                if total2 > 0.0:
                    cand2 = cand2 / total2 * best.sum()
                    cand2_score = _score(cand2)
                    if cand2_score < best_score:
                        best, best_score = cand2, cand2_score
                        step_factor = max(0.01, step_factor * 0.995)
                        no_improve = 0

        # if we haven't improved for a while, shrink step size to fine‑tune
        if no_improve > 2500:
            step_factor = max(0.005, step_factor * 0.9)
            no_improve = 0

        # early exit if target reached
        if best_score <= target_score:
            break

    # ----------- final sanity check ---------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)

    return [float(v) for v in final]
