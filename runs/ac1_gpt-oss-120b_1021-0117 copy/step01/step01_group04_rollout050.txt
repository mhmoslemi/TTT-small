import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4**i   (i = 0..n-1)"""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 upper bound using FFT (no scaling of the whole vector)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

    # Problem size and optional parent construction
    # ------------------------------------------------------------------

        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)

        parent = None

                    grad[diag] += cur[diag]

        # ----- move type -------------------------------------------------
        if rng.random() < 0.68:          # mass‑transfer guided by gradient
            src = int(np.argmax(grad))
            dst = int(np.argmin(grad))
            if src != dst and cur[src] > 0:
                delta = cur[src] * rng.random() * 0.32   # up to 32 % moved
                delta = min(delta, 1000.0 - cur[dst])
                if delta > 0:
                    cand = cur.copy()
                    cand[src] -= delta
                    cand[dst] += delta
                else:
                    continue
            else:
                continue
        else:                             # small random block scaling
            w = int(rng.integers(max(1, n // 250), max(2, n // 15)))
            start = int(rng.integers(0, max(1, n - w + 1)))
            factor = math.exp(rng.normal(0.0, 0.07))
            cand = cur.copy()
            cand[start:start + w] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)

        # ------------------------------------------------------------------
        cand_score = _ac1_score(cand, n, L)

        # Record global optimum
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            # Early stop if we beat the target bound
            if best_score < 1.50274349:
                break

        # Simulated‑annealing style acceptance for the current walk
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= decay

        # Occasionally restart from the best found so far
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
