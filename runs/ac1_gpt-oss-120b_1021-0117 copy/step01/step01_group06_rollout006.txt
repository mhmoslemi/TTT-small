# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_mask = rho_arr > 0
    if not np.any(rho_mask):
        theta = 0.0
    else:
        rho = np.where(rho_mask)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 score (identical to evaluate_sequence)
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (vec.sum() ** 2))


# ------------------------------------------------------------
# Conventional sqrt series (fallback initialisation)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """First n terms of the series a with a*a = 1 + ... + x^{2n-2}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    # safety margin for the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # -----------------------------------------------------------------
    # Initial vector
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur_vec = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur_vec = _sqrt_series(256)
    # make sure we are on the probability simplex
    cur_vec = np.maximum(cur_vec, 0.0)
    if cur_vec.sum() == 0.0:
        cur_vec[:] = 1.0 / cur_vec.size
    else:
        cur_vec = cur_vec / cur_vec.sum()

    n = cur_vec.size
    cur_score = _score(cur_vec)
    best_vec = cur_vec.copy()
    best_score = cur_score

    # pre‑compute FFT size
    L = 1 << ((2 * n - 1).bit_length())
    step0 = 0.5  # initial step magnitude for the sub‑gradient step
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        # -------------------------------------------------------------
        # 70 % gradient (sub‑gradient) move, 30 % random mass transfer
        # -------------------------------------------------------------
        if rng.random() < 0.7:
            # ----- gradient step -----
            fv = np.fft.rfft(cur_vec, n=L)
            conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
            k_max = int(conv.argmax())

            # gradient of the active convolution entry
            idx = np.arange(n)
            j = k_max - idx
            mask = (j >= 0) & (j < n)
            g = np.zeros_like(cur_vec)
            g[mask] = 2.0 * cur_vec[j[mask]]

            eta = step0 / np.sqrt(iteration)
            cand = _project_simplex(cur_vec - eta * g)
            cand_score = _score(cand)

            if cand_score < cur_score:
                cur_vec, cur_score = cand, cand_score
                if cand_score < best_score:
                    best_score, best_vec = cand_score, cand.copy()
        else:
            # ----- random mass‑transfer move -----
            i = rng.integers(0, n)
            if cur_vec[i] == 0.0:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            frac = rng.random() * 0.5          # up to 50 % of source mass
            delta = cur_vec[i] * frac
            cand = cur_vec.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)
            cand_score = _score(cand)

            if cand_score < cur_score:
                cur_vec, cur_score = cand, cand_score
                if cand_score < best_score:
                    best_score, best_vec = cand_score, cand.copy()

        # -------------------------------------------------------------
        # occasional jitter to avoid stagnation
        # -------------------------------------------------------------
        if iteration % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            cand = _project_simplex(cur_vec + jitter)
            cand_score = _score(cand)
            if cand_score < cur_score:
                cur_vec, cur_score = cand, cand_score
                if cand_score < best_score:
                    best_score, best_vec = cand_score, cand.copy()

        # brief safety check that we still have time
        if time.monotonic() >= deadline:
            break

    # return the best found vector as plain Python floats
    return [float(x) for x in best_vec]
