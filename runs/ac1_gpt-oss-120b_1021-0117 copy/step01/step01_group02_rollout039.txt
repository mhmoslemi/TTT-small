def _max_convolution(arr, n):
    """Return (max(conv), full conv) of arr with itself (linear convolution)."""
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)
    return conv.max(), conv
