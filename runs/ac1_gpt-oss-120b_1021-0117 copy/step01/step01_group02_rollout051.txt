def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    try:
        n = int(len(height_sequence_1))
        cur = height_sequence_1.astype(np.float64).copy()
    except Exception:
        n = 5000
        # initialize, maybe use binomial series similar as earlier
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i-1] * (1.0 - 1.0/(2.0*i))
        cur = a
    # ensure sum > 0.01
    if cur.sum() < 0.01:
        cur = np.full_like(cur, 1.0 / cur.size)

    cur = np.clip(cur, 0.0, 1000.0)
    best = cur.copy()
    best_score = evaluate_sequence(best.tolist())
    # maybe also consider uniform candidate for fallback
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_score = evaluate_sequence(uniform.tolist())
    if uni_score < best_score:
        best = uniform.copy()
        best_score = uni_score

    min_step = 1e-8
    # main loop
    while time.monotonic() < deadline:
        n = cur.size
        conv = np.convolve(cur, cur)
        cur_max = conv.max()
        k = int(conv.argmax())
        # second max distinct
        # we need to ignore the max entries; find next max
        # Could use mask
        mask = conv < cur_max - 1e-15
        if mask.any():
            second_max = conv[mask].max()
        else:
            second_max = cur_max
        # Compute partner contributions at index k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = np.zeros_like(cur)
        # partner[i] = cur[k-i] for i in given range
        partner[idx] = cur[k - idx]
        src = int(partner.argmax())
        # candidate dest indices
        min_partner_val = partner.min()
        dest_candidates = np.where(partner == min_partner_val)[0]
        # ensure at least one candidate
        if dest_candidates.size == 0:
            dest_candidates = np.arange(n)
        # also add a few random dest to explore
        # Randomly pick up to 3 new dest indices
        if rng.random() < 0.3:
            # add random indices
            rand_ids = rng.choice(n, size=min(3, n), replace=False)
            dest_candidates = np.unique(np.concatenate((dest_candidates, rand_ids)))
        improved = False
        # iterate over dest candidates limited
        for dest in dest_candidates[:5]:
            if dest == src:
                continue
            max_delta = cur[src]
            # avoid zero partner src (should not happen)
            psrc = partner[src]
            # compute delta aiming to reduce cur_max to second_max
            if psrc > 0:
                delta_target = (cur_max - second_max) / (2.0 * psrc + 1e-15)
                delta = min(max_delta, max(delta_target * 0.5, max_delta * 0.2)) # combine
                # ensure not too small
                if delta < min_step:
                    delta = min_step
            else:
                delta = max_delta * 0.2  # fallback
            # ensure delta <= max_delta
            if delta > max_delta:
                delta = max_delta
            cand = cur.copy()
            cand[src] -= delta
            cand[dest] += delta
            # clip
            np.clip(cand, 0.0, 1000.0, out=cand)
            # sum preserve
            # compute score
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                cur = cand.copy()
                improved = True
                break # break dest loop
        if not improved:
            # try a random local perturbation (shake)
            if rng.random() < 0.05:
                width = rng.integers(max(1, n // 200), max(2, n // 12))
                start = rng.integers(0, max(1, n - width + 1))
                stop = min(n, start + width)
                cand = cur.copy()
                # multiplicative perturbation
                factor = np.exp(rng.normal(0.0, 0.15, size=stop-start))
                cand[start:stop] *= factor
                np.clip(cand, 0.0, 1000.0, out=cand)
                # sum constant? not automatically, but we can renorm to same sum as cur
                s = cand.sum()
                if s > 0.0:
                    cand = cand / s * cur.sum()
                cand_score = evaluate_sequence(cand.tolist())
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
                    cur = cand.copy()
            # else occasional full random shuffle
            elif rng.random() < 0.02:
                perm = rng.permutation(n)
                cand = cur[perm]
                cand_score = evaluate_sequence(cand.tolist())
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
                    cur = cand.copy()
        # optionally reduce step/adjust attempts after some time?
        # Not needed
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)
    return [float(v) for v in final]

import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic search for a non‑negative step‑function that minimizes
    2·n·max(conv)/(sum²) where conv = a*a (linear convolution).
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # Use the provided parent if it exists, otherwise fall back on a simple
    # decreasing series.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = base.shape[0]
    except Exception:
        n = 5000
        base = np.empty(n, dtype=np.float64)
        base[0] = 1.0
        for i in range(1, n):
            base[i] = base[i - 1] * (1.0 - 1.0 / (2.0 * i))

    base = np.clip(base, 0.0, 1000.0)
    if base.sum() < 0.01:
        base = np.full_like(base, 1.0 / base.size)

    # ------------------------------------------------------------------
    # Small pool of initial candidates (original, reversed, sorted, uniform)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    init_candidates = [
        base,
        base[::-1],
        np.sort(base)[::-1],
        np.sort(base),
        uniform,
    ]

    best = base.copy()
    best_score = evaluate_sequence(best.tolist())
    cur = best.copy()

    for cand in init_candidates:
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best = cand.copy()
            cur = cand.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    min_step = 1e-8
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        n = cur.size
        conv = np.convolve(cur, cur)                     # linear convolution
        cur_max = conv.max()
        k = int(conv.argmax())

        # second (distinct) maximum – used for a target delta
        mask = conv < cur_max - 1e-12
        if mask.any():
            second_max = conv[mask].max()
        else:
            second_max = cur_max

        # partner[i] = cur[k-i] where defined, else 0
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = np.zeros_like(cur)
        partner[idx] = cur[k - idx]

        src = int(partner.argmax())

        # choose a few destination indices (minimal partner + some random)
        min_val = partner.min()
        dest_candidates = np.where(partner == min_val)[0]
        if dest_candidates.size == 0:
            dest_candidates = np.arange(n)
        if rng.random() < 0.25:
            extra = rng.choice(n, size=min(3, n), replace=False)
            dest_candidates = np.unique(np.concatenate((dest_candidates, extra)))

        improved = False
        for dest in dest_candidates[:5]:
            if dest == src:
                continue
            max_delta = cur[src]
            if max_delta < min_step:
                continue

            psrc = partner[src]                     # contribution of src to conv[k]
            # candidate delta values
            deltas = [max_delta * 0.5, max_delta * 0.25, max_delta * 0.1]
            if psrc > 0.0:
                delta_target = (cur_max - second_max) / (2.0 * psrc + 1e-15)
                if delta_target > 0.0:
                    deltas.append(min(delta_target, max_delta))
            # evaluate each delta, keep the best improvement
            for delta in deltas:
                if delta < min_step:
                    continue
                cand = cur.copy()
                cand[src] -= delta
                cand[dest] += delta
                np.clip(cand, 0.0, 1000.0, out=cand)

                sc = evaluate_sequence(cand.tolist())
                if sc < best_score:
                    best = cand.copy()
                    cur = cand.copy()
                    best_score = sc
                    improved = True
                    break
            if improved:
                break

        # ------------------------------------------------------------------
        # If no improvement, try a small random perturbation or a permutation.
        # ------------------------------------------------------------------
        if not improved:
            # random multiplicative perturbation of a short window
            if rng.random() < 0.02:
                width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
                start = int(rng.integers(0, max(1, n - width + 1)))

            pass

    if final.sum() < 0.01:
        final = np.full_like(final, 1.0 / final.size)
    return [float(v) for v in final]
