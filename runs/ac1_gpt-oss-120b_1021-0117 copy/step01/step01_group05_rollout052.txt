def _fast_score(seq, m):
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*len(seq)-1]
    max_val = np.max(conv)
    s = np.sum(seq)
    if s <= 0:
        return float('inf'), conv, max_val
    score = 2.0 * len(seq) * max_val / (s * s)
    return score, conv, max_val

def move_mass_from_max_pair(seq, conv, beta):
    n = len(seq)
    k = int(np.argmax(conv))
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    i_vals = np.arange(lo, hi+1)
    j_vals = k - i_vals
    # compute contributions
    contrib = seq[i_vals] * seq[j_vals]
    max_idx = np.argmax(contrib)
    i = i_vals[max_idx]
    j = j_vals[max_idx]
    if i == j:
        removed = seq[i] * beta
        seq[i] -= removed
        total_removed = removed
    else:
        removed_i = seq[i] * beta
        removed_j = seq[j] * beta
        seq[i] -= removed_i
        seq[j] -= removed_j
        total_removed = removed_i + removed_j
    # choose target index with low value (use random)
    tgt = rng.integers(0, n)
    seq[tgt] += total_removed
    # clip
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Determine n and start sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 0 and np.isfinite(parent_arr).all():
            start_seq = np.clip(parent_arr, 0.0, 1000.0)
        else:
            raise Exception
    except Exception:
        # produce binomial
        n_est = 3809
        start_seq = np.empty(n_est, dtype=np.float64)
        start_seq[0] = 1.0
        for i in range(1, n_est):
            start_seq[i] = start_seq[i-1] * (2*i-1)/(2*i)
    n = start_seq.size
    m = 1 << ((2*n - 1).bit_length())

    def fast_score(seq):
        f = np.fft.rfft(seq, n=m)
        conv = np.fft.irfft(f * f, n=m)[:2*n-1]
        max_val = conv.max()
        total = seq.sum()
        if total <= 0.0:
            return float('inf'), conv, max_val
        score = 2.0 * n * max_val / (total * total)
        return score, conv, max_val

    best = start_seq.copy()
    best_score, best_conv, _ = fast_score(best)

    # optional: also try binomial if parent not given and parent worse
    # but we already used start_seq = parent if present else binomial.

    # Additional random initialization: try exponential shape to see if lower
    # We'll evaluate a few param values and replace best if lower.
    # Choose some r values.
    def gen_exponential(r):
        return r**np.arange(n, dtype=np.float64)

    # Evaluate grid of r
    for r in np.linspace(0.998, 0.9999, 50):
        cand_seq = gen_exponential(r)
        # ensure within bounds
        cand_seq = np.clip(cand_seq, 0.0, 1000.0)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes the AC1 bound.
    Uses a combination of targeted mass relocation, smoothing and block
    scaling moves.  The algorithm respects the time budget and always returns
    a feasible candidate.
    """
    rng = np.random.default_rng(seed)
    # give a short safety margin before the hard timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        start = np.clip(parent, 0.0, 1000.0)
    except Exception:
        # binomial seed a_i = C(2i,i)/4^i
        n = 3809
        start = np.empty(n, dtype=np.float64)
        start[0] = 1.0
        for i in range(1, n):
            start[i] = start[i - 1] * (2 * i - 1) / (2 * i)

    n = start.size
    # length for power‑of‑two FFT that covers full 2n‑1 convolution
    m = 1 << ((2 * n - 1).bit_length())

    # ----------------------------------------------------------------------
    # Helper: compute AC1 score together with the convolution array
    # ----------------------------------------------------------------------
    def fast_score(seq):
        f = np.fft.rfft(seq, n=m)
        conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
        max_val = conv.max()
        total = seq.sum()
        if total <= 0.0:
            return float("inf"), conv, max_val
        score = 2.0 * n * max_val / (total * total)
        return score, conv, max_val

    best = start.copy()
    best_score, best_conv, _ = fast_score(best)

    # ----------------------------------------------------------------------
    # Try a few exponential families (r**i) – cheap global exploration
    # ----------------------------------------------------------------------
    exponents = np.linspace(0.998, 0.9999, 35)
    arr = np.arange(n, dtype=np.float64)
    for r in exponents:
        cand = np.power(r, arr)
        cand = np.clip(cand, 0.0, 1000.0)
        sc, conv, _ = fast_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
            best_conv = conv

    # ----------------------------------------------------------------------

    while time.monotonic() < deadline:
