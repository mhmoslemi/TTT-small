import numpy as np
import time

# ----------------------------------------------------------------------
# Convolution + score (exactly as in the reference evaluator)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (O(n log n))."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the authoritative evaluator."""
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Classic Catalan (sqrt) series – a known good baseline.
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) beats the
    target AC1 bound.  The routine respects the time budget, keeps the
    best discovered sequence and returns it as a plain list."""
    rng = np.random.default_rng(seed)
    # keep a tiny safety margin before the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initial candidate: use the provided parent if it exists, otherwise
    # fall back to the classic sqrt series.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = init.size
    else:
        init = None
        n = 5582

    sqrt_seq = _sqrt_series(n)

    # pick the better of the two starters
    best = sqrt_seq.copy()
    best_score = _score(best)
    if init is not None:
        sc = _score(init)
        if np.isfinite(sc) and sc < best_score:
            best = init.copy()
            best_score = sc

    cur = best.copy()
    cur_score = best_score

    # target for early exit
    TARGET = 1.50274349

    # ------------------------------------------------------------------
    # Helper: gradient of the current maximal convolution entry.
    # For the index k_max where the autocorrelation attains its maximum,
    # the partial derivative w.r.t a[i] is a[k_max - i] (if that index
    # lies inside the array).  We build that vector explicitly.
    # ------------------------------------------------------------------
    def _grad_for_max(a: np.ndarray, k_max: int) -> np.ndarray:
        n = a.size
        grad = np.zeros_like(a)
        start = max(0, k_max - (n - 1))
        end = min(k_max, n - 1)
        # i runs from start … end inclusive
        i = np.arange(start, end + 1, dtype=np.int64)
        grad[i] = a[k_max - i]
        return grad

    # ------------------------------------------------------------------
    # Main optimisation loop.  Three move types are mixed:
    #   1) targeted mass transfer guided by the gradient of the current max;
    #   2) modest block‑wise scaling (exploratory);
    #   3) tiny additive noise (to escape plateaus).
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        op = rng.random()
        # ------------------------------------------------------------------
        # 1) Gradient‑guided transfer
        # ------------------------------------------------------------------
        if op < 0.55:
            # current convolution & its maximum index
            b = _conv_self(cur)
            k_max = int(np.argmax(b))
            grad = _grad_for_max(cur, k_max)

            # source = index that most increases the current max,
            # destination = index that hardly influences it.
            src = int(np.argmax(grad))
            zero_inds = np.where(grad == 0)[0]
            if zero_inds.size:

                    if best_score <= TARGET:
                        break

    # ----------------------------------------------------------------------
    # Return the best solution discovered as a plain Python list.
    # ----------------------------------------------------------------------
