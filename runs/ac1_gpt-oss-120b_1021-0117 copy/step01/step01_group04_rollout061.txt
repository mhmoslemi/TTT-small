def _ac1_score(arr, n, L):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    max_b = conv[:2*n-1].max()
    s = arr.sum()
    if s < 1e-12: return float('inf')
    return 2.0 * n * max_b / (s * s)

import time, math
import numpy as np

# ----- helpers -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """
    Fast evaluation of the AC1 upper bound via FFT.
    Returns 2*n*max(conv)/sum(arr)^2  (np.inf if sum is too small).
    """
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _gen_binomial(n: int) -> np.ndarray:
    """Classic binomial construction a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ----- main search ---------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a few seconds free for cleanup/return
    deadline = start + max(0.0, float(budget_s) - 3.0)

    # ----- problem size ----------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256  # fallback length
    L = _next_pow2(2 * n - 1)

    # ----- baseline candidates ---------------------------------------------
    best = None
    best_score = float('inf')

    # parent (if supplied)
    if parent is not None:
        sc = _ac1_score(parent, n, L)
        if sc < best_score:
            best = parent.copy()
            best_score = sc

    # binomial start (known decent solution)
    base = _gen_binomial(n)
    sc = _ac1_score(base, n, L)
    if sc < best_score:
        best = base.copy()
        best_score = sc

    # ----- parametric search (exponential + power law) --------------------
    # shape a_i = (i+1)^(-p) * exp(-eps*i)
    i_arr = np.arange(n, dtype=np.float64)

    # parameter grids
    p_vals = np.linspace(0.3, 1.2, 12)          # 12 points between 0.3 and 1.2
    eps_vals = np.concatenate([
        np.array([0.0]),                       # pure power law
        np.linspace(5e-5, 0.0015, 25)          # exponential decay
    ])

    for p in p_vals:
        power = (i_arr + 1.0) ** (-p)
        for eps in eps_vals:
            if time.monotonic() >= deadline:
                break
            if eps == 0.0:
                cand = power
            else:
                cand = power * np.exp(-eps * i_arr)
            # evaluate
            sc = _ac1_score(cand, n, L)
            if sc < best_score:
                best = cand.copy()
                best_score = sc
                if best_score < 1.50274349:
                    # good enough – we can stop the grid search early
                    break
        if time.monotonic() >= deadline:
            break

    # ensure we have a candidate
    if best is None:
        best = np.ones(n, dtype=np.float64)
        best_score = _ac1_score(best, n, L)

    # ----- local stochastic refinement ------------------------------------
    cur = best.copy()
    cur_score = best_score
    temperature = 1.0
    decay = 0.9995

    while time.monotonic() < deadline:
        # early exit if target reached
        if best_score < 1.50274349:
            break

        # ----- propose a move --------------------------------------------
        if rng.random() < 0.6:                 # block scaling
            width = int(rng.integers(1, max(2, n // 30)))
            start_idx = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.07))
            cand = cur.copy()
            cand[start_idx:start_idx + width] *= factor
        else:                                   # transfer a fraction of mass
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * 0.25          # up to 25 % moved
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        np.clip(cand, 0.0, 1000.0, out=cand)

        # ----- evaluate ----------------------------------------------------
        cand_score = _ac1_score(cand, n, L)

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:

            if rng.random() < prob:

        if accept:

            cur_score = cand_score

        temperature *= decay

        # occasional restart from the best solution
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # ----- return -----------------------------------------------------------
    return [float(x) for x in best]
