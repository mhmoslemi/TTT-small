# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# First n coefficients of the series a where a*a = 1+…+x^{2n-2}
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# fast AC1 upper‑bound for a vector that already sums to 1
# ------------------------------------------------------------
def _score_and_conv(x: np.ndarray, L: int):
    fv = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * x.size - 1]
    mx = conv.max()
    # sum(x) is (numerically) 1, but we keep the exact formula
    s = x.sum()
    return 2.0 * x.size * mx / (s * s), conv


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty sequence of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the sandbox
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------------
    # initialise: use the provided parent if it exists, otherwise sqrt series
    # --------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = a / a.sum()                     # simplex normalisation
    else:
        n = 256
        a = _sqrt_series(n).astype(np.float64)
        a = a / a.sum()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())      # size for FFT (next power of two)

    # --------------------------------------------------------------------
    # keep the best vector seen so far
    # --------------------------------------------------------------------
    best_vec = a.copy()
    best_sc, _ = _score_and_conv(best_vec, L)

    # --------------------------------------------------------------------
    # parameters for the hybrid sub‑gradient / random walk search
    # --------------------------------------------------------------------
    step0 = 0.35            # initial step magnitude for the sub‑gradient move
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # ------------------------------------------------------------
        # evaluate current vector (also yields the index of the worst
        # convolution entry)
        # ------------------------------------------------------------
        cur_sc, conv = _score_and_conv(a, L)
        max_idx = int(conv.argmax())

        # ------------------------------------------------------------
        # decide which move to perform
        # ------------------------------------------------------------
        if rng.random() < 0.5:
            # ----- sub‑gradient step ---------------------------------
            # gradient of the maximal convolution entry:
            grad = np.zeros_like(a)
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            # g[i] = 2 * a[max_idx - i]  (zero outside the valid range)
            grad[idx] = 2.0 * a[max_idx - idx]
            # diminishing step size
            step = step0 / np.sqrt(it)
            cand = a - step * grad
        else:
            # ----- random mass‑transfer move -------------------------
            # pick a source index weighted by current mass (more likely)
            src = rng.choice(n, p=a)
            # pick a distinct target index uniformly
            tgt = rng.integers(0, n - 1)
            if tgt >= src:
                tgt += 1
            # transfer a random fraction of the source mass
            frac = rng.random() * 0.25
            delta = a[src] * frac
            cand = a.copy()
            cand[src] -= delta
            cand[tgt] += delta

        # enforce simplex constraints (non‑negativity and unit sum)
        cand = _project_simplex(cand)

        # ------------------------------------------------------------
        # evaluate candidate
        # ------------------------------------------------------------
        cand_sc, _ = _score_and_conv(cand, L)

        # ------------------------------------------------------------
        # accept if it improves the global best; otherwise keep current
        # ------------------------------------------------------------
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            # occasionally allow a non‑improving move to escape plateaus
            if rng.random() < 0.02:
                a = cand

        # ------------------------------------------------------------
        # occasional re‑initialisation when stuck for a while
        # ------------------------------------------------------------
        if no_improve > 3000:
            # perturb the best solution lightly and project back
            pert = rng.normal(scale=0.01, size=n)
            cand2 = best_vec + pert
            cand2 = np.maximum(cand2, 0.0)
            cand2 = _project_simplex(cand2)
            cand2_sc, _ = _score_and_conv(cand2, L)
            if cand2_sc < best_sc:
                best_sc = cand2_sc
                best_vec = cand2.copy()
            a = best_vec.copy()
            no_improve = 0

    # --------------------------------------------------------------------
    # Return the best vector found as a plain Python list
    # --------------------------------------------------------------------
    return [float(v) for v in best_vec]
