def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Determine n and parent
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_raw.ndim != 1 or parent_raw.size == 0 or not np.isfinite(parent_raw).all():
            raise Exception
        n = parent_raw.shape[0]
        parent = np.clip(parent_raw, 0.0, 1000.0)
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    # helper functions
    def max_conv(seq):
        m = 1 << ((2*n-1).bit_length())
        fft_seq = np.fft.rfft(seq, m)
        conv = np.fft.irfft(fft_seq * fft_seq, m)
        return conv[:2*n-1].max()

    def score(seq):
        s = seq.sum()
        if s < 0.01:
            return np.inf
        mb = max_conv(seq)
        return 2.0 * n * mb / (s * s)

    # initial candidates
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))  # use evaluator for correctness

    # Sidon-like sequence
    def sidon_seq(n):
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i-1] * (1.0 - 1.0 / (2.0 * i))
        return a

    # choose whichever better
    seq0 = sidon_seq(n)
    seq0 = np.clip(seq0, 0.0, 1000.0)
    score0 = float(evaluate_sequence(seq0.tolist()))
    if score0 < best_score:
        best_seq, best_score = seq0.copy(), score0

    # current for hill climbing
    cur = best_seq.copy()
    cur_score = best_score

    # parameters
    max_iter = 200000  # just upper bound
    iter_count = 0
    # Simulated annealing temperature schedule
    T0 = 1.0
    T_min = 1e-4
    # We'll define iterations = approx 0.9 * budget seconds / time per iteration maybe 0.001s

    while time.monotonic() < deadline:
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a fraction of source to target
        delta = cur[i] * rng.random() * 0.5  # up to 50% of source
        if delta <= 0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure bounds
        if cand[i] < 0.0 or cand[j] > 1000.0:
            continue
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
        # else possibly accept with probability (optional)
        # occasional random restart to escape
        if rng.random() < 0.0005:
            # restart with small random perturb of best_seq or sidon
            cur = best_seq.copy()
            perm = rng.permutation(n)
            cur = cur[perm]

        # optional break if iterations too many
        iter_count += 1
        if iter_count > max_iter:
            break

    # final fallback: ensure non-empty list valid
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(n, 1.0, dtype=np.float64)

    # Fast score

        conv = np.fft.irfft(fft_seq * fft_seq, m)

    # Sidon-like construction

        a = np.empty(m, dtype=np.float64)
        a[0] = 1.0

    # (optional) but skip due to time.

    # Parameters

import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step sequence with a low AC1 bound.
    Uses a fast FFT‑based score and a simple simulated‑annealing style
    hill‑climbing that moves mass between coefficients while keeping the
    total sum fixed."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # obtain problem size and a feasible starting vector (the provided
    # parent if present, otherwise a uniform vector)
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent_arr.ndim != 1
            or parent_arr.size == 0
            or not np.isfinite(parent_arr).all()
        ):
            raise Exception
        n = int(parent_arr.size)
        parent = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    # ------------------------------------------------------------------
    # fast evaluation of the AC1 bound (identical to the official one
    # except it uses an FFT based convolution)
    _fft_len = 1 << ((2 * n - 1).bit_length())

    def fast_score(seq: np.ndarray) -> float:
        s = seq.sum()
        if s < 0.01:
            return np.inf
        f = np.fft.rfft(seq, _fft_len)
        conv = np.fft.irfft(f * f, _fft_len)
        maxb = conv[: 2 * n - 1].max()
        return 2.0 * n * maxb / (s * s)

    # ------------------------------------------------------------------
    # evaluate the supplied parent
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    cur = best_seq.copy()
    cur_score = fast_score(cur)

    # ------------------------------------------------------------------
    # classic Sidon‑like construction (a_i = C(2i,i)/4**i)
    def _sidon_like_sequence(m: int) -> np.ndarray:
        a = np.empty(m, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, m):
            a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
        return a

    sid = _sidon_like_sequence(n)
    sid = np.clip(sid, 0.0, 1000.0)
    sid_score = float(evaluate_sequence(sid.tolist()))
    if sid_score < best_score:
        best_seq = sid.copy()
        best_score = sid_score
        cur = sid.copy()
        cur_score = fast_score(cur)

    # ------------------------------------------------------------------
    # hill‑climbing / simulated annealing loop
    max_iter = 200_000               # safety cap
    it = 0
    move_factor_initial = 0.5        # up‑to this fraction of a source entry may move
    while time.monotonic() < deadline and it < max_iter:
        # gradually shrink the step size
        progress = it / max_iter
        move_factor = move_factor_initial * (1.0 - progress) + 0.05 * progress

        # pick a source index – 10 % of the time the current maximal entry
        if rng.random() < 0.1:

        max_delta = cur[src] * move_factor
        if max_delta <= 0.0:

        cand[src] -= delta
        cand[dst] += delta
        if cand[src] < 0.0 or cand[dst] > 1000.0:
            it += 1
            continue

        cand_score = fast_score(cand)
