import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 bound defined by `evaluate_sequence`.
    The routine works with a provided parent `height_sequence_1` (if any);
    otherwise it falls back to a uniform sequence.
    A tiny matrix‑free smoothing loop is used as a cheap local optimiser.
    """
    # ------------------------------------------------------------------
    # Basic setup – obtain the length and a starting candidate.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    try:
        # height_sequence_1 is supplied by the harness (may be missing)
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0)
        n = int(parent.size)
    except Exception:
        # no parent – work with a modest default length
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    # a simple uniform candidate (always feasible)
    uniform = np.full(n, 1.0, dtype=np.float64)

    # ------------------------------------------------------------------
    # Helper to compute the bound using the analytical form
    #   B = 2 * n * sum(a_i^2) / (sum(a_i)^2)
    # This avoids the O(n log n) convolution for intermediate checks.
    # ------------------------------------------------------------------
    def bound_fast(vec):
        s = np.sum(vec)
        if s < 0.01:
            return np.inf
        ss = np.dot(vec, vec)
        return 2.0 * n * ss / (s * s)

    # ------------------------------------------------------------------
    # Initialise the best sequence with the better of {parent, uniform}
    # ------------------------------------------------------------------
    best_seq = uniform.copy()
    best_score = bound_fast(best_seq)

    parent_score = bound_fast(parent)
    if np.isfinite(parent_score) and parent_score < best_score:
        best_seq = parent.copy()
        best_score = parent_score

    # ------------------------------------------------------------------
    # Simple matrix‑free smoothing search.
    # The smoothing operator preserves the total sum (circular averaging)
    # and drives the distribution towards uniformity, which is optimal for
    # the bound.  The loop stops when the time budget is exhausted.
    # ------------------------------------------------------------------
    current = best_seq.copy()
    rng = np.random.default_rng(seed)

    while time.monotonic() < deadline:
        # circular three‑point averaging
        new_seq = (current + np.roll(current, 1) + np.roll(current, -1)) / 3.0

        # fast evaluation
        cand_score = bound_fast(new_seq)

        if cand_score < best_score:
            best_seq = new_seq.copy()
            best_score = cand_score
            current = new_seq
            continue  # keep smoothing from the improved point

        # occasional tiny random perturbation to escape flat plateaux
        if rng.random() < 0.005:
            perturb = rng.normal(scale=0.02, size=n)
            perturbed = np.clip(new_seq + perturb, 0.0, 1000.0)
            pert_score = bound_fast(perturbed)
            if pert_score < best_score:
                best_seq = perturbed.copy()
                best_score = pert_score
                current = perturbed
                continue

        # otherwise keep the original current (no improvement)
        # a small chance to restart from the original parent to explore
        if rng.random() < 0.001:
            current = parent.copy()

    # ------------------------------------------------------------------
    # Return a plain Python list of floats – the evaluator expects this.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
