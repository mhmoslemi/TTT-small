def propose_candidate(...):

    w = _catalan_weights(n)

    # also consider parent normalized: maybe best
    parent_norm = parent / np.sum(parent)
    if evaluate_sequence(parent_norm.tolist()) < evaluate_sequence(a.tolist()):
        a = parent_norm

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Additional candidate: exponential distribution for baseline
    def exp_init(alpha):
        x = np.arange(n, dtype=np.float64)
        a_exp = np.exp(-alpha*x)
        a_exp /= a_exp.sum()
        return a_exp

    # try some alphas quickly (maybe 5) to find best
    for alpha in np.linspace(0.0005, 0.005, 10):
        cand = exp_init(alpha)
        s = float(evaluate_sequence(cand.tolist()))
        if s < best_score:
            best = cand.copy()
            best_score = s

    # main loop
    step_frac = 0.1  # initial fraction for mass transfer
    min_step = 1e-5
    temperature = 0.02  # acceptance tolerance for worse moves
    cooling = 0.9995

    while time.monotonic() < deadline:
        # compute b and max shift
        b = np.convolve(best, best)
        k_max = int(np.argmax(b))
        max_val = b[k_max]
        # compute indices contributing to this max
        lo = max(0, k_max - (n-1))
        hi = min(k_max, n-1)
        idxs = np.arange(lo, hi+1)
        partners = k_max - idxs
        contrib = best[idxs] * best[partners]
        total_contrib = contrib.sum()
        # probability distribution weighted by contribution
        if total_contrib == 0:
            i = rng.integers(lo, hi+1)  # fallback
        else:
            probs = contrib / total_contrib
            i = rng.choice(idxs, p=probs)

        # choose a target j random not equal i
        j = rng.integers(0, n-1)
        while j == i:
            j = rng.integers(0, n-1)

        # amount to transfer
        eps = step_frac * best[i]
        if eps <= 0:
            # pick small random
            eps = step_frac * 1e-6

        cand = best.copy()
        cand[i] -= eps
        cand[j] += eps

        # ensure nonnegative (clipping small negative due to numeric)
        if cand[i] < 0:
            # adjust eps to not make negative
            cand[i] = 0.0
        # renorm (still sum ~1 but due to rounding may differ)
        s = cand.sum()
        if s > 0:
            cand /= s
        else:
            continue

        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score):
            # accept if better or maybe accept with probability
            if score < best_score or rng.random() < np.exp(-(score - best_score) / temperature):
                best = cand
                best_score = score
                # reduce temperature maybe
                temperature *= cooling
                # optionally increase step_frac if improvement
                step_frac = min(step_frac * 1.1, 0.2)
                if best_score <= 1.50274349:

                step_frac = max(step_frac * 0.9, min_step)

        best = np.full(n, 1.0/n)
        best_score = float(evaluate_sequence(best.tolist()))

    # return list
    return [float(v) for v in best]

import time
import numpy as np

# --------------------------------------------------------------
# Helper to generate the Catalan weights used in the previous version
# --------------------------------------------------------------
def _catalan_weights(n: int) -> np.ndarray:
    """Return the first n coefficients w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w
