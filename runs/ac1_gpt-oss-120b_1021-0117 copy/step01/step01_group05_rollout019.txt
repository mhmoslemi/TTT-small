def move_mass_redistribute(seq, conv, rng):
    n = len(seq)
    k_max = int(np.argmax(conv))
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    # compute contributions
    idx_range = np.arange(lo, hi+1)
    partners = k_max - idx_range
    contrib = seq[idx_range] * seq[partners]
    # pick i with max contribution
    i = int(idx_range[np.argmax(contrib)])
    j = k_max - i
    # if i==j handle
    # choose alpha fraction to move (0.05 to 0.15)
    alpha = rng.uniform(0.05, 0.15)
    # compute removed mass
    if i == j:
        removed = seq[i] * alpha
        new_seq = seq.copy()
        new_seq[i] = max(0.0, seq[i] - removed)
        total_removed = removed
    else:
        removed_i = seq[i] * alpha
        removed_j = seq[j] * alpha
        new_seq = seq.copy()
        new_seq[i] = max(0.0, seq[i] - removed_i)
        new_seq[j] = max(0.0, seq[j] - removed_j)
        total_removed = removed_i + removed_j
    # find low-value target
    # we can pick random index from the lower 20% values
    # compute sorted indices of seq (or use approximate).
    # For speed, choose random candidate indices until we find one with value less than median.
    tgt = rng.integers(0, n)
    # maybe try few attempts
    attempts = 0
    while attempts < 5 and new_seq[tgt] > np.median(seq):
        tgt = rng.integers(0, n)
        attempts += 1
    new_seq[tgt] += total_removed
    return new_seq
```

def _fast_score(seq):
    n = seq.shape[0]
    m = 1 << ((2*n-1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]
    max_val = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float('inf'), conv, max_val
    score = 2.0 * n * max_val / (s*s)
    return float(score), conv, max_val

def move_mass_redistribute(seq, conv, rng):
    n = seq.shape[0]
    k_max = int(np.argmax(conv))
    lo = max(0, k_max - (n-1))
    hi = min(k_max, n-1)
    i_range = np.arange(lo, hi+1)
    j_range = k_max - i_range
    # contributions
    contrib = seq[i_range] * seq[j_range]
    # pick index with highest contribution
    idx = np.argmax(contrib)
    i = int(i_range[idx])
    j = k_max - i
    # fraction to move
    alpha = rng.uniform(0.08, 0.18)
    if i == j:
        removed = seq[i] * alpha
        seq_new = seq.copy()
        seq_new[i] = seq[i] - removed
        total_removed = removed
    else:
        removed_i = seq[i] * alpha
        removed_j = seq[j] * alpha
        seq_new = seq.copy()
        seq_new[i] = seq[i] - removed_i
        seq_new[j] = seq[j] - removed_j
        total_removed = removed_i + removed_j
    # choose target t as a low-value index (random until low)
    # low threshold: median
    median_val = np.median(seq)
    t = rng.integers(0, n)
    attempts = 0
    while attempts < 5 and seq_new[t] > median_val:
        t = rng.integers(0, n)
        attempts += 1
    seq_new[t] += total_removed
    return seq_new
```

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fast_score(seq: np.ndarray):
    """Return (score, full convolution array, max conv) using FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val
