import time
import numpy as np

try:
    import cvxpy as cp

    cp = None


def _sidon_like_sequence(n):
    """Classic decreasing Sidon‑like sequence a_i = ∏_{j=1}^i (1-1/(2j))."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a


def _conv_max(seq):
    """Return (max(conv), full convolution) using an FFT (linear convolution)."""
    n = seq.size
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    return conv.max(), conv
