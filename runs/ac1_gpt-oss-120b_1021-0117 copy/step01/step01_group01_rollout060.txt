import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that yields a
    low value of evaluate_sequence.  Uses a gradient‑guided stochastic hill‑climb
    with a simple simulated‑annealing acceptance rule.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)   # keep a safety margin

    # --------------------------------------------------------------------- #
    # 1. Obtain problem size and an initial feasible sequence
    # --------------------------------------------------------------------- #
    try:
        n = int(len(height_sequence_1))
        a = np.array(height_sequence_1, dtype=np.float64)
    except Exception:                     # fallback if the parent is missing
        n = 256
        a = np.ones(n, dtype=np.float64)

    # make sure everything is within the evaluator's hard bounds
    a = np.clip(a, 0.0, 1000.0)

    # normalise the sum (any positive scale leaves the score unchanged)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s

    # --------------------------------------------------------------------- #
    # 2. Pre‑computed FFT layout
    # --------------------------------------------------------------------- #
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two

    # --------------------------------------------------------------------- #
    # 3. Helper to evaluate the objective (exactly the same as evaluate_sequence)
    # --------------------------------------------------------------------- #
    def _score(vec: np.ndarray) -> float:
        fa = np.fft.rfft(vec, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_b = float(conv.max())
        s = float(vec.sum())
        if s < 1e-12:
            return np.inf
        return 2.0 * n * max_b / (s * s)

    cur = a.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score

    # simulated‑annealing temperature schedule
    T = cur_score * 0.05 + 1e-12
    decay = 0.9997

    # --------------------------------------------------------------------- #
    # 4. Main optimisation loop
    # --------------------------------------------------------------------- #
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # full convolution and location of its maximum
        fa = np.fft.rfft(cur, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = float(conv.max())
        k = int(conv.argmax())                     # index where max occurs

        # contributions of each coefficient to this maximal entry
        idx = np.arange(n)
        valid = (k - idx >= 0) & (k - idx < n)
        rev = np.zeros_like(cur)
        rev[valid] = cur[k - idx[valid]]           # a_{k-i} where defined
        contrib = cur * rev                         # a_i * a_{k-i}

        # pick a source index with a large contribution and a sink with a tiny one
        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))
        if i_src == i_dst:
            i_dst = int(rng.integers(0, n))
            while i_dst == i_src:
                i_dst = int(rng.integers(0, n))

        if cur[i_src] <= 0.0:
            continue

        # move a modest fraction of mass from source to destination
        frac = rng.uniform(0.01, 0.20)
        delta = cur[i_src] * frac
        new = cur.copy()
        new[i_src] -= delta
        new[i_dst] += delta

        # Ensure feasibility (non‑negative, upper bound already far away)
        if new[i_src] < 0.0:
            continue

        new_score = _score(new)

        # Accept if better, otherwise with a probability that falls with temperature
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        # cool down
        T *= decay

        # occasional random jitter to avoid stagnation
        if it % 4000 == 0:

            if i != j and cur[i] > 0.0:
                eps = cur[i] * rng.uniform(0.001, 0.01)
                cur[i] -= eps
                cur[j] += eps
                cur_score = _score(cur)

    # --------------------------------------------------------------------- #
    # 5. Return the best found sequence (as plain Python floats)
    # --------------------------------------------------------------------- #
    final = [float(max(0.0, min(1000.0, x))) for x in best.tolist()]
    # final check – if something went wrong fall back to the normalised start
    try:
        _ = evaluate_sequence(final)
        return final
    except Exception:

        return fallback
