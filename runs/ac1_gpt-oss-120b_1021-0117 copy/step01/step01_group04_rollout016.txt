import time, math
import numpy as np

# ---------- helper utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_and_conv(arr: np.ndarray, n: int, L: int):
    """Return (score, full linear convolution) of arr with itself."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf'), conv
    score = 2.0 * n * max_b / (s * s)
    return score, conv


def _move_impact(arr: np.ndarray, conv: np.ndarray, rng: np.random.Generator):
    """Transfer a fraction from an index heavily involved in the maximal
    convolution term to a (mostly) unrelated index."""
    n = arr.size
    k0 = int(np.argmax(conv))
    idx = np.arange(n)
    partner = k0 - idx
    valid = (partner >= 0) & (partner < n)
    impact = np.zeros_like(arr)
    if valid.any():
        # product a_i * a_{k0-i}
        impact[valid] = arr[valid] * arr[partner[valid]]
    source = int(np.argmax(impact))
