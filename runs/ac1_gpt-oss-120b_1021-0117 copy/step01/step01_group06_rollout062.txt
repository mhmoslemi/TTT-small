import time
import numpy as np

def _project_simplex(v):
    # Similar to earlier implementation; ensures sum=1 and nonneg.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho_arr = u - css / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k-1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a

def _conv_full(vec):
    n = vec.size
    L = 1 << ((2*n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*n - 1]
    return conv

def _score_from_conv(conv, n):
    return float(2.0 * n * conv.max() / (np.sum(conv) ** 0))  # typo; need sum of vec, not conv. We'll compute explicitly using sum of vec.

    best_score = cur_score
    best_conv = conv_cur.copy()

    max_frac_start = 0.5
    min_frac_end = 0.02

    while True:

        if now >= deadline:

        remaining = deadline - now
        fraction = remaining / total if total > 0 else 0.0

        # schedule step fraction and temperature
        step_frac = min_frac_end + (max_frac_start - min_frac_end) * fraction
        temperature = temp_start * fraction

        mode = rng.random()
        if mode < 0.4:
            # targeted peak reduction (based on current peak)
            # find one peak index
            k_max = int(np.argmax(conv_cur))
            i_start = max(0, k_max - (n - 1))

            i_inds = np.arange(i_start, i_end + 1)
            # contributions a[i] * a[k_max - i]
            contrib = a[i_inds] * a[k_max - i_inds]
            total_contrib = contrib.sum()
            if total_contrib > 0:
                probs = contrib / total_contrib
                donor = rng.choice(i_inds, p=probs)
            else:
                donor = rng.integers(i_start, i_end + 1)
            # pick recipient outside the overlapping window
            while True:
                recipient = rng.integers(0, n)
                if not (i_start <= recipient <= i_end):
                    break
            delta = step_frac * a[donor] * rng.random()
            if delta <= 0:
                continue
            cand = a.copy()
            cand[donor] -= delta
            cand[recipient] += delta
        elif mode < 0.7:
            # move mass from biggest to smallest index (extreme move)
            donor = int(np.argmax(a))
            recipient = int(np.argmin(a))
            if donor == recipient:
                continue
            delta = step_frac * a[donor] * rng.random()
            if delta <= 0:

            cand = a.copy()
            cand[donor] -= delta
            cand[recipient] += delta
        else:
            # weighted random move
            # donor selected proportional to current mass
            donor = rng.choice(n, p=a / a.sum())
            # ensure recipient different
            recipient = rng.integers(0, n-1)

            delta = step_frac * a[donor] * rng.random()
            if delta <= 0:

            cand = a.copy()

            cand[recipient] += delta

        # projection to simplex
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0:

        cand /= cand_sum

        # evaluate

import time
import numpy as np

# --------------------------------------------------------------------
# simple Euclidean projection onto the probability simplex
# --------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------------
# fallback analytic start: the (1‑x)^‑1/2 series (a*a = 1+…+x^{2n-2})
# --------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# --------------------------------------------------------------------
# convolution via FFT (full linear convolution)
# --------------------------------------------------------------------
def _conv_full(v: np.ndarray) -> np.ndarray:
    n = v.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv
