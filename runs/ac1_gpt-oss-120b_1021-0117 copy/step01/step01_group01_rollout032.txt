import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _generate_binomial_seq(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i via recurrence; all positive, decreasing."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x >= 0, sum x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    mask = u - css / ind > 0
    if not np.any(mask):
        # all entries are zero or negative: return uniform point
        w = np.full_like(v, 1.0 / len(v))
        return w
    rho = np.where(mask)[0][-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise exactly
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


def _conv_max_and_grad(a: np.ndarray, fft_len: int, conv_len: int):
    """
    Compute the linear convolution a*a via FFT, return
    - max value,
    - argmax index,
    - gradient of the max entry w.r.t. a (2 * a shifted).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:conv_len]
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])

    n = a.shape[0]
    idx = max_idx - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[idx[mask]]
    return max_val, max_idx, g

    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:conv_len]
    return float(conv.max())

                best_max = p_max
                cur = pert_seq.copy()
                cur_max = p_max
                cur_grad = _conv_max_and_grad(cur, fft_len, conv_len)[2]

    # --------------------------------------------------------------
    # Light hill‑climbing refinement (mass transfer moves)
    cur = best.copy()
    cur_max = best_max
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if cur[i] <= 1e-12:
            continue
        frac = rng.uniform(0.01, 0.2)
        delta = cur[i] * frac
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_max = _conv_max(new, fft_len, conv_len)
        if new_max < cur_max:
            cur = new
            cur_max = new_max
            if new_max < best_max:
                best = new.copy()
                best_max = new_max

    # --------------------------------------------------------------
    # Return the best candidate (as a plain Python list)
    final = [float(x) for x in best]
    try:
        _ = evaluate_sequence(final)  # sanity check
    except Exception:
        # safety fallback: uniform simplex point
        final = [float(x) for x in _project_simplex(np.ones(n))]
    return final
