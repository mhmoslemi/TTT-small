def _project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic sqrt–series (a * a = 1 + … + x^{2n-2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Fast max‑convolution using FFT (vector assumed real)
# ------------------------------------------------------------
def _conv_max(vec: np.ndarray) -> float:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Score for a vector that already satisfies sum(vec)=1
# ------------------------------------------------------------
def _score_normed(vec: np.ndarray) -> float:
    n = vec.size
    return 2.0 * n * _conv_max(vec)

    f = base / s                     # sum(f) == 1

    max_lp_iters = 6

        lp_iter += 1

        # current bound on the convolution
        M = _conv_max(f)

        # Build sparse Toeplitz matrix representing convolution with f:
        # (A_ub)_{k,j} = f_{k-j}  (0 <= k-j < n), else 0.
        # This is a collection of constant diagonals.
        data = [np.full(n, fv) for fv in f]          # n diagonals, each length n
        A_ub = sp.diags(data, offsets=range(n),
                        shape=(2 * n - 1, n), format='csc')
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # maximise sum(b)  <=> minimise -sum(b)
        c = -np.ones(n, dtype=np.float64)
