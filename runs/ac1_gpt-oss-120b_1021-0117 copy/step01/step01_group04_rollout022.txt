def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-2.0)
    try: parent = np.asarray(height_sequence_1, dtype=float)
    except: parent = None
    n = parent.size if parent is not None else 256
  L = _next_pow2(2*n-1)
  # initial candidates
  base = _gen_binomial(n)
  candidates = [base]
  if parent is not None:
      candidates.append(parent)
  # evaluate each, pick best
  best = candidates[0].copy()
  best_score = ac1_fft(best,n,L)
  for cand in candidates[1:]:
      cand_score = ac1_fft(cand, n, L)
      if cand_score < best_score:
          best = cand.copy(); best_score = cand_score
  cur = best.copy()
  cur_score = best_score

  temperature = 1.0
  decay = 0.9995

  # start search
  while time.monotonic() < deadline:
      # compute gradient for cur
      # compute conv with FFT
      fa = np.fft.rfft(cur, n=L)
      conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
      max_val = float(conv.max())
      max_idx = int(conv.argmax())
      S = float(cur.sum())
      # compute gradient vector
      idxs = max_idx - np.arange(n)
      # mask
      mask = (idxs >= 0) & (idxs < n)
      grad = np.zeros_like(cur)
      grad[mask] = 2.0 * cur[idxs[mask]]
      # gradient of objective
      grad_f = (2.0 * n / (S * S)) * grad - (4.0 * n * max_val / (S ** 3)) * np.ones_like(cur)
      # propose move
      if rng.random() < 0.7:
          # gradient guided mass transfer
          i_up = int(np.argmax(grad_f))  # highest positive gradient (increase harmful)
          i_down = int(np.argmin(grad_f)) # most negative gradient (increase helpful)
          if i_up != i_down and grad_f[i_up] > 0:
              # move a fraction from up to down
              delta = cur[i_up] * 0.1  # 10%
              if delta > 0:
                  cand = cur.copy()
                  cand[i_up] -= delta
                  cand[i_down] += delta
              else:
                  continue
          else:
              # fallback random move
              i = int(rng.integers(0, n))
              j = int(rng.integers(0, n))
              if i == j:
                  continue
              frac = rng.random() * 0.4
              delta = cur[i] * frac
              cand = cur.copy()
              cand[i] -= delta
              cand[j] += delta
      else:
          # random block scaling
          width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
          start = int(rng.integers(0, max(1, n - width + 1)))
          factor = math.exp(rng.normal(0.0, 0.12))
          cand = cur.copy()
          cand[start:start+width] *= factor

      # clip
      np.clip(cand, 0.0, 1000.0, out=cand)

      cand_score = ac1_fft(cand, n, L)

      # keep best
      if cand_score < best_score:
          best = cand.copy()
          best_score = cand_score
          # early stop

          cur_score = best_score

import numpy as np
import time
import math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4**i for i=0..n-1 via recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 bound using FFT (linear convolution)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step‑function with a small AC1 bound.

    Returns a list of Python floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ---- problem size ----------------------------------------------------

    L = _next_pow2(2 * n - 1)

    # ---- initial candidates ------------------------------------------------
    base = _gen_binomial(n)
    candidates = [base]
    if parent is not None:
        candidates.append(parent)

    best = candidates[0].copy()
    best_score = _ac1_score(best, n, L)
    for cand in candidates[1:]:
        sc = _ac1_score(cand, n, L)
        if sc < best_score:

            best_score = sc

    cur = best.copy()
    cur_score = best_score

    # ---- main search loop -------------------------------------------------
    while time.monotonic() < deadline:
        # compute convolution for current sequence (used for gradient)

        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]

        # gradient of max(conv) w.r.t. each entry
