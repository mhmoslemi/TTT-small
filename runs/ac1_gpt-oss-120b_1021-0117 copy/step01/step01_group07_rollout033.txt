def targeted_move(a, step_factor, rng):
    n = a.size
    # compute convolution
    c = np.convolve(a, a)
    max_idx = int(np.argmax(c))
    # find contributions
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    i_range = np.arange(i_start, i_end + 1)
    j = max_idx - i_range
    # contributions = a[i_range] * a[j]
    contrib = a[i_range] * a[j]
    # choose i with largest contribution
    if contrib.size == 0:
        return None
    top_idx = i_range[int(np.argmax(contrib))]
    i = int(top_idx)
    j = max_idx - i
    # if either index out-of-range or a[i] or a[j] zero, fallback random
    if i < 0 or i >= n or j < 0 or j >= n:
        return None
    # compute epsilon
    if i == j:
        eps = step_factor * a[i]
        if eps <= 0:
            return None
        # move eps from i to random other k
        k = rng.integers(n)
        while k == i:
            k = rng.integers(n)
        cand = a.copy()
        cand[i] -= eps
        cand[k] += eps
        return cand
    else:
        eps = step_factor * min(a[i], a[j])
        if eps <= 0:
            return None
        # pick two random target positions distinct from i,j
        k = rng.integers(n)
        while k == i or k == j:
            k = rng.integers(n)
        l = rng.integers(n)
        while l == i or l == j or l == k:
            l = rng.integers(n)
        cand = a.copy()
        cand[i] -= eps
        cand[j] -= eps
        cand[k] += eps
        cand[l] += eps
        return cand

def random_pair_move(a, rng):

    i_start = max(0, m - (n-1))
    i_end = min(n-1, m)
    # i_range
    i_range = np.arange(i_start, i_end+1)
    g = np.zeros_like(a)
    j = m - i_range
    g[i_range] = 2 * a[j]
    max_g = g.max()
    if max_g <= 0:
        return None
    # Determine step magnitude as fraction of max_g relative to a.max()
    max_a = a.max()
    step = step_frac * (max_a / max_g)  # ensures step <= step_frac * a.max() / max_g
    cand = a - step * g
    cand = np.clip(cand, 0, None)
    return cand

import time
import numpy as np

def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style weights w_i = C(2i,i)/4**i for i=0..n‑1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _gradient_move(a: np.ndarray, step_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Take a small step opposite the gradient of the currently maximal convolution entry."""
    n = a.size
    c = np.convolve(a, a)
    m = int(np.argmax(c))
    # gradient of c[m] w.r.t. a_i is 2*a_{m-i} when the index exists
    grad = np.zeros_like(a)
    i_start = max(0, m - (n - 1))
    i_end = min(n - 1, m)
    i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = m - i_range
    grad[i_range] = 2.0 * a[j]
    max_g = grad.max()
    if max_g <= 0.0:
        return None
    # scale step so it is a fraction of the largest component of a
    max_a = a.max()
    step = step_frac * (max_a / max_g)
    cand = a - step * grad
    cand = np.clip(cand, 0.0, None)
    return cand


def _targeted_move(a: np.ndarray, step_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Shift a little mass away from the pair that contributes most to the peak convolution."""
    n = a.size
    c = np.convolve(a, a)
    m = int(np.argmax(c))
    # indices that actually take part in c[m]
    i_start = max(0, m - (n - 1))
    i_end = min(n - 1, m)
    i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = m - i_range
    contrib = a[i_range] * a[j]                       # contribution of each ordered pair
    if contrib.size == 0:
        return None
    # pick the ordered index with the largest product
    idx = int(i_range[np.argmax(contrib)])
    i = idx
    j = m - i
    if not (0 <= i < n and 0 <= j < n):
        return None
    if i == j:
        eps = step_frac * a[i]
        if eps <= 0.0:
            return None
        k = rng.integers(n)
        while k == i:
            k = rng.integers(n)
        cand = a.copy()
        cand[i] -= eps
        cand[k] += eps
        return cand
    # i != j
    eps = step_frac * min(a[i], a[j])
    if eps <= 0.0:
        return None
    # two distinct target positions
    k = rng.integers(n)
    while k == i or k == j:
        k = rng.integers(n)
    l = rng.integers(n)
    while l == i or l == j or l == k:
        l = rng.integers(n)
    cand = a.copy()
    cand[i] -= eps
    cand[j] -= eps
    cand[k] += eps
    cand[l] += eps
    return cand


def _random_pair_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random fraction of mass from one index to another."""
    n = a.size
    pos = np.where(a > 1e-16)[0]
    if pos.size < 1:
        return None
    i = rng.choice(pos)
    j = rng.integers(n)
    while j == i:
        j = rng.integers(n)
    eps = rng.random() * a[i] * 0.5
    cand = a.copy()
    cand[i] -= eps
    cand[j] += eps
    return cand


def _jitter_move(a: np.ndarray, std: float, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative jitter (log‑normal) followed by renormalisation."""
    jitter = np.exp(rng.normal(0.0, std, size=a.shape))
    cand = a * jitter
    return cand

    # ------------------------------------------------------------------

    try:
