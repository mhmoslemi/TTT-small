import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic sqrt‑series (a*a = 1+…+x^{2n-2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Fast score (same as evaluate_sequence)
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (vec.sum() ** 2)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty sequence of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------
    # Determine length and a reasonable start vector
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        init1 = np.asarray(height_sequence_1, dtype=np.float64)
        n = init1.size
        init1 = np.maximum(init1, 0.0)
        init1 /= init1.sum()
    else:
        n = 256
        init1 = None

    sqrt_init = _sqrt_series(n)
    sqrt_init = sqrt_init / sqrt_init.sum()

    if init1 is None or _score(sqrt_init) < _score(init1):
        cur = sqrt_init.copy()
    else:
        cur = init1.copy()

    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # Pre‑compute FFT length for the whole run
    # --------------------------------------------------------
    L_fft = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Parameters for the projected sub‑gradient descent
    # --------------------------------------------------------
    base_step = 0.8          # initial step scale
    max_iter = 400000        # safety cap
    iter_cnt = 0
    no_improve = 0           # counter for stagnation

    while time.monotonic() < deadline and iter_cnt < max_iter:
        iter_cnt += 1

        # ----- convolution and active index(es) -----
        fv = np.fft.rfft(cur, n=L_fft)
        conv = np.fft.irfft(fv * fv, n=L_fft)[: 2 * n - 1]
        max_val = conv.max()
        # indices within a tiny tolerance are treated as active

        # ----- sub‑gradient (sum over active indices) -----
        g = np.zeros_like(cur)
        for idx in active:
