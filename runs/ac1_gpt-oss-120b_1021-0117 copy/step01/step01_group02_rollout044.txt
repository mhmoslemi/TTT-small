def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # get length and starting vector
    try:
        n = int(len(height_sequence_1))
        init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        n = 5000
        # exponential decay sequence as simple guess
        x = np.exp(-np.arange(n) * 0.001)
        init = x / x.sum() * np.sqrt(2 * n)  # scaling irrelevant
        init = np.clip(init, 0.0, 1000.0)

    if init.sum() < 0.01:
        init = np.full(n, 1.0 / n, dtype=np.float64)

    # Helper to compute score and convolution via FFT
    n_fft = 1 << ((2 * n - 1).bit_length())
    def score_and_conv(seq):
        # FFT of seq
        f = np.fft.rfft(seq, n=n_fft)
        conv = np.fft.irfft(f * f, n=n_fft)[:2 * n - 1]
        cur_max = conv.max()
        sc = 2.0 * n * cur_max / (seq.sum() ** 2)
        return float(sc), conv

    best = init.copy()
    best_score, best_conv = score_and_conv(best)

    # fallback uniform (maybe better)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_score, _ = score_and_conv(uniform)
    if uni_score < best_score:
        best = uniform
        best_score = uni_score
        best_conv = None

    # early exit if already good enough
    target = 1.50274349
    if best_score <= target:
        return [float(v) for v in np.clip(best, 0.0, 1000.0)]

    # parameters
    step_factor = 0.3
    min_step = 1e-7
    random_shake_prob = 0.006
    shake_sigma = 0.15

    # Main loop
    while time.monotonic() < deadline:
        # Compute conv if not cached
        if best_conv is None:
            _, best_conv = score_and_conv(best)
        conv = best_conv
        cur_max = conv.max()
        k = int(conv.argmax())

        # partner vector
        partner = np.zeros_like(best)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner[idx] = best[k - idx]

        # product contributions
        prod = best * partner
        src = int(prod.argmax())
        # destination: minimal partner, break ties by minimal best value
        min_partner = partner.min()
        dest_candidates = np.where(partner == min_partner)[0]
        if dest_candidates.size == 0:
            dest = int(rng.integers(0, n))
        else:
            # choose among candidates the one with smallest best value
            cand_best_vals = best[dest_candidates]
            dest = int(dest_candidates[int(cand_best_vals.argmin())])

        if src == dest:
            # fallback random perturbation
            if rng.random() < random_shake_prob:
                width = int(rng.integers(max(1, n // 300), max(2, n // 50)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                stop = start + width
                cand = best.copy()
                factor = np.exp(rng.normal(0.0, shake_sigma, size=width))
                cand[start:stop] *= factor
                cand = np.clip(cand, 0.0, 1000.0)
                cand *= best.sum() / cand.sum()
                cand_score, _ = score_and_conv(cand)
                if cand_score < best_score:
                    best = cand
                    best_score = cand_score
                    best_conv = None
            continue

        delta = best[src] * step_factor
        if delta < min_step:
            delta = min_step
        if delta > best[src]:
            delta = best[src]

        cand = best.copy()
        cand[src] -= delta
        cand[dest] += delta

        cand = np.clip(cand, 0.0, 1000.0)
        # renormalize to keep sum constant (avoid drift)
        s = cand.sum()
        if s <= 0:
            continue
        cand *= best.sum() / s

            best = cand
            best_score = cand_score
            best_conv = None
            # increase step_factor a bit for momentum

        else:

def propose_candidate(seed=42, budget_s=1100, **kwargs):

    Return a non‑negative step‑function (list of floats) whose AC1 upper bound
    (2·n·max(conv)/sum²) is as low as possible.

    except Exception:                     # no parent supplied

        best = x / x.sum() * np.sqrt(2 * n)   # scaling irrelevant for the bound
        best = np.clip(best, 0.0, 1000.0)

    # ------------------------------------------------------------------ #

    def score_and_conv(seq):
        f = np.fft.rfft(seq, n=n_fft)
