def _norm_one(v: np.ndarray) -> np.ndarray:
    """Scale to have sum = 1 (preserve non‑negativity)."""
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _self_conv_max(a: np.ndarray, L: int):
    """Return (max convolution, FFT of a, full convolution)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _softmax_weights(conv: np.ndarray, cur_max: float, eps: float) -> np.ndarray:

    return w / s if s != 0.0 else np.full_like(w, 1.0 / w.size)

    tol = 1e-15

    # ---- main loop -------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        cur_max, cur_fft, conv = _self_conv_max(cur, L)

        # soft‑max weighting
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = _softmax_weights(conv, cur_max, eps)

        # gradient (matrix‑free)
        grad = _grad(cur_fft, w, n, L)

        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # candidate generation (exponentiated update)
        step = step0 / math.sqrt(it + 1.0)
        cand = None
        cand_max = None
        for _ in range(12):
            exp = -step * momentum
            exp = np.clip(exp, -50.0, 50.0)
            cand_tmp = cur * np.exp(exp)
            cand_tmp = _norm_one(cand_tmp)
            cand_tmp_max, _, _ = _self_conv_max(cand_tmp, L)
            if cand_tmp_max <= cur_max + tol:
                cand, cand_max = cand_tmp, cand_tmp_max
                break
            step *= 0.5

        if cand is None:
            cand, cand_max = cand_tmp, cand_tmp_max

        # acceptance
        if cand_max < cur_max - tol:
            cur = cand
            cur_max = cand_max
            no_improve = 0
            if cand_max < best_max:
                best_max = cand_max
                best_vec = cand.copy()
                best_score = 2.0 * n * best_max
        else:
            no_improve += 1
            prob = math.exp(-(cand_max - cur_max) / (step + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_max = cand_max

        # periodic symmetrisation
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _norm_one(sym)
            sym_max, _, _ = _self_conv_max(sym, L)
            if sym_max + tol < cur_max:
                cur = sym
                cur_max = sym_max
                if sym_max < best_max:
                    best_max = sym_max
                    best_vec = sym.copy()
                    best_score = 2.0 * n * best_max
                no_improve = 0

        # occasional random mass transfer to escape plateaus
        if no_improve > 500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _norm_one(tmp)
                tmp_max, _, _ = _self_conv_max(tmp, L)
                if tmp_max < cur_max:
                    cur = tmp
                    cur_max = tmp_max
                    no_improve = 0
                    if tmp_max < best_max:
                        best_max = tmp_max
                        best_vec = tmp.copy()
                        best_score = 2.0 * n * best_max
                else:
                    no_improve = 0

        # early stop if we beat target
        if best_score < target_score:
            break

    # -----------------------------------------------------------------
    return [float(x) for x in best_vec]
