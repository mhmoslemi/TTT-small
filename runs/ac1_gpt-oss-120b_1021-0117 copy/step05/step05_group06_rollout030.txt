import time
import math
import numpy as np

# --------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Maximum of the full auto‑convolution a * a (linear convolution)
# --------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max(conv_full) for a*a using zero‑padded FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.clip(a0, 0.0, None)
        a0 = _project_simplex(a0)
        n = a0.size
    else:
        n = 2048
        a0 = rng.random(n, dtype=np.float64)
        a0 = _project_simplex(a0)

    # FFT length large enough for all convolutions involved
    L = 1 << ((3 * n - 2).bit_length())

    # ----- bookkeeping ------------------------------------------------
    best = a0.copy()
    best_sc = _max_conv(best, L)
    best_bound = 2.0 * n * best_sc

    # ----- optimisation parameters ------------------------------------
    eps0 = 0.8               # initial smoothing temperature
    eps_decay = 0.9995
    lr0 = 0.5                # base learning‑rate
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(a0)   # Adam first moment
    v = np.zeros_like(a0)   # Adam second moment
    beta1_pow, beta2_pow = 1.0, 1.0
    cur = a0.copy()
    cur_sc = best_sc
    cur_bound = best_bound
    target_bound = 1.50274349
    it = 0

    # ----- main gradient‑descent loop --------------------------------
    while time.monotonic() < deadline - 2.0:
        it += 1
        # convolution of current iterate
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # smooth‑max weights (softmax) – switch to hard sub‑gradient later
        eps = max(1e-9, eps0 * (eps_decay ** it))
        if eps > 1e-6:
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0 or not np.isfinite(w_sum):
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum
        else:
            mask = conv >= max_val - 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = mask.astype(float) / cnt

        # gradient of the smoothed max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        direction = m_hat / (np.sqrt(v_hat) + 1e-8)

        # back‑tracking line search (max 8 attempts)
        alpha = lr0 / math.sqrt(it)
        improved = False
        for _ in range(8):
            cand = cur - alpha * direction
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc <= cur_sc:            # accept non‑worsening step
                cur = cand
                cur_sc = cand_sc
                cur_bound = 2.0 * n * cand_sc
                improved = True
                if cur_bound < best_bound:
                    best = cur.copy()
                    best_sc = cand_sc
                    best_bound = cur_bound
                break
            alpha *= 0.5

        # occasional symmetrisation (helps with the Toeplitz structure)
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _max_conv(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                cur_bound = 2.0 * n * sym_sc
                if cur_bound < best_bound:
                    best = cur.copy()
                    best_sc = sym_sc
                    best_bound = cur_bound

        # early exit
        if best_bound < target_bound:
            break

    # ----- deterministic anti‑identity permutation --------------------
    vals = np.sort(best)[::-1]
    perm = np.empty_like(vals)
    left, right = 0, n - 1
    k = 0
    while left <= right:
        perm[left] = vals[k]; k += 1
        if left < right:
            perm[right] = vals[k]; k += 1
        left += 1
        right -= 1
    perm_sc = _max_conv(perm, L)
    perm_bound = 2.0 * n * perm_sc
    if perm_bound < best_bound:
        best = perm.copy()
        best_sc = perm_sc
        best_bound = perm_bound

    # ----- hill‑climbing swaps ----------------------------------------
    cur = best.copy()
    cur_bound = best_bound
    while time.monotonic() < deadline - 0.2:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_sc = _max_conv(cur, L)
        new_bound = 2.0 * n * new_sc
        if new_bound < cur_bound:
            cur_bound = new_bound
            best = cur.copy()
            best_sc = new_sc
            best_bound = new_bound
            if best_bound < target_bound:
                break
        else:
            cur[i], cur[j] = cur[j], cur[i]   # revert

    # ----- final safety check -----------------------------------------
    if best.size == 0:
        best = np.array([1.0])

    return [float(x) for x in best]
