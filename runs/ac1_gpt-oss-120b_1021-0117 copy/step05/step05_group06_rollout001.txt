import time, math, numpy as np

def _project_simplex(v):
    # v: 1d np array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _max_conv(a, L):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*a.size-1]
    return float(conv.max())

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _max_conv(a: np.ndarray, L: int) -> float:
    """Maximum value of the (full) auto‑convolution a*a using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialisation – use the provided parent if present, otherwise random.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()

        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)            # enforce sum = 1, non‑negative
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT length

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_max = _max_conv(best, L)
    best_bound = 2.0 * n * best_max
    cur = best.copy()
    cur_max = best_max

    # ------------------------------------------------------------------
    # Gradient‑descent phase (matrix‑free)
    # ------------------------------------------------------------------
    beta = 0.9
    momentum = np.zeros_like(cur)
    step0 = 1.0
    eps = 1.0
    eps_decay = 0.9995
    min_eps = 1e-9
    it = 0
    target = 1.50274349

    while time.monotonic() < deadline - 30.0:

        # current convolution
        cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # smoothing parameter for the soft‑max approximation of max()
        eps = max(min_eps, eps * eps_decay)

        # soft‑max weights
        if eps > 1e-7:
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:

        # gradient of the smoothed max via FFT (same as in the reference)

                best, best_max = cur.copy(), new_max
                best_bound = 2.0 * n * best_max
                if best_bound < target:
                    break
        else:

            cur[i], cur[j] = cur[j], cur[i]

        if rng.random() < 0.001:
