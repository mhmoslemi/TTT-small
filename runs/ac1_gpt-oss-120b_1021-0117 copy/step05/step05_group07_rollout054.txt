import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the return path
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # Initialise – parent if present, otherwise random uniform start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()

        cur = rng.random(2000, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)                 # force sum = 1
    n = cur.size

    # ------------------------------------------------------------
    # FFT planning – length at least next power of two ≥ 2n‑1
    # ------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)            # mapping for gradient indices

    # ------------------------------------------------------------
    # Book‑keeping

    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    best_sc = 2.0 * n * conv_full.max()
    best_vec = cur.copy()
    cur_sc = best_sc

    # momentum and schedule parameters
    beta = 0.94
    momentum = np.zeros_like(cur)
    step0 = 0.7
    no_improve = 0
    target = 1.50274349

    it = 0
    while time.monotonic() < deadline:

        # ----- momentum update (Nesterov‑like) -----
