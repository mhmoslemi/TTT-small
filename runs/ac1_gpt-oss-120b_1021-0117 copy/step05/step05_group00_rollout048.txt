import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.size
    # sort in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast AC1 bound for sequences with sum≈1 (score = 2*n*max(conv))
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise a feasible point (normalize to sum=1)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s
    else:
        # fallback: exponential decay
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()

    # optional symmetrisation (helps a lot)
    a0 = (a0 + a0[::-1]) * 0.5
    a0 = _project_simplex(a0)

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())  # FFT length for full convolution

    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------
    it = 0
    no_improve = 0
    # smoothing (epsilon) parameters
    eps = None
    eps_factor = 0.2
    eps_decay = 0.998

    # Adam optimisation parameters
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1_pow = 1.0
    beta2_pow = 1.0
    base_lr = 0.6
    eps_adam = 1e-8

    sym_interval = 300          # frequency of forced symmetrisation
    improve_target = 1.50274349
    max_line_trials = 4         # backtracking attempts

    while time.monotonic() < deadline:
        it += 1

        # ----- autocorrelation (full) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- smoothing weight -----
        if eps is None:
            eps = max(max_val * eps_factor, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed max (via FFT) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)      # pick the needed n entries
        g = 2.0 * grad_full[idx]

        # ----- Adam moment updates -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        direction = m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- step size (decaying) -----
        norm_dir = np.linalg.norm(direction) + 1e-12
        alpha = base_lr / math.sqrt(it) / norm_dir

        # ----- backtracking line‑search -----
        improved = False
        cand = None
        cand_sc = None
        cur_alpha = alpha
        for _ in range(max_line_trials):
            cand_try = cur - cur_alpha * direction
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            cur_alpha *= 0.5

        # ----- occasional uphill acceptance (simulated annealing) -----
        if not improved:
            cand_try = cur - cur_alpha * direction
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            prob = math.exp(-(cand_sc - cur_sc) / (cur_alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic enforced symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 1800:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0

        # ----- restart from best with small noise -----
        if no_improve > 3500:
            cur = best_vec + 0.03 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

    # Return the best vector as plain Python floats
    return [float(x) for x in best_vec]
