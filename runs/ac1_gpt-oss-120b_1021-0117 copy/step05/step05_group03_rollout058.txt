def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.3) # margin
    
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 1024
        a = rng.random(n)
    a = np.maximum(a,0.0)
    a = _project_simplex(a)
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())  # next power of 2
    
    def compute_score(vec):
        af = np.fft.rfft(vec, n=L)
        conv_full = np.fft.irfft(af * af, n=L)[:2*n-1]
        return 2.0 * n * conv_full.max()
    best = a.copy()
    best_sc = compute_score(best)
    cur = a.copy()
    cur_sc = best_sc
    
    step0 = 1.0
    beta = 0.85
    momentum = np.zeros_like(cur)
    alpha_temp = 5.0   # softmax temperature parameter (i.e., scaling)
    # We'll call it beta_s = 1/temperature? Actually larger alpha makes weight more peaked.
    # Use alpha_factor to increase gradually towards maybe 200.
    # We'll start at 2 and increase linearly.
    
    iter = 0
    no_improve = 0
    
    target = 1.50274349
    sym_period = 300
    while time.monotonic() < deadline:
        iter += 1
        # compute conv and gradient via softmax
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_conv = conv_full.max()
        # softmax weights
        # shift for stability
        shift = conv_full - max_conv
        w = np.exp(alpha_temp * shift)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # compute gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]
        # momentum
        momentum = beta * momentum + (1-beta) * grad
        # step size
        base_alpha = step0 / math.sqrt(iter)
        # line search
        cand_best = None
        cand_best_sc = None
        for factor in [1.0, 0.7, 0.5, 0.3, 0.1]:
            step = base_alpha * factor
            tmp = cur - step * momentum
            tmp = _project_simplex(tmp)
            tmp_sc = compute_score(tmp)
            if cand_best_sc is None or tmp_sc < cand_best_sc:
                cand_best = tmp
                cand_best_sc = tmp_sc
            if tmp_sc < cur_sc:
                # accept early
                break
        # accept
        if cand_best_sc < cur_sc:
            cur, cur_sc = cand_best, cand_best_sc
            no_improve = 0
            if cand_best_sc < best_sc:
                best, best_sc = cand_best.copy(), cand_best_sc
        else:
            # optionally random up-hill
            prob = math.exp(-(cand_best_sc - cur_sc) / (base_alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand_best, cand_best_sc
            no_improve += 1
        # symmetrize
        if iter % sym_period == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = compute_score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                no_improve = 0
        # random mass transfer if stuck
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = compute_score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0
        # increase softmax temperature gradually
        if iter % 500 == 0:
            alpha_temp = min(alpha_temp * 1.3, 200.0)   # cap
        # early exit
        if best_sc < target:
            break
    # return best as Python list
    return [float(x) for x in best]

import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Full‑convolution score (exactly the evaluator's formula)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2).  a may be any non‑negative vector."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv_full.max() / (s * s))

# Main optimisation routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients that (hopefully)
    yields a low AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for final conversion
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the supplied parent if present, otherwise random.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(1024, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1
    n = cur.size

    # ------------------------------------------------------------
    # FFT length – power of two >= 2*n‑1 (full convolution size)
    # ------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    best = cur.copy()
    best_sc = _score(best, L)
    cur_sc = best_sc

    # hyper‑parameters (tuned heuristically)
    step0 = 1.0          # base step magnitude
    beta = 0.85          # momentum factor
    momentum = np.zeros_like(cur)
    alpha_smooth = 2.0   # start with a gentle soft‑max
    target = 1.50274349
    sym_interval = 350   # how often to enforce symmetry
    no_improve = 0
    it = 0

    # ------------------------------------------------------------
    # Optimisation loop – matrix‑free, O(n log n) per iteration
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of the current iterate -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_c = conv_full.max()

        # ----- smooth (soft‑max) weight vector -----
        # shift for numerical stability (max_c >= all entries)
        shift = conv_full - max_c
        w = np.exp(alpha_smooth * shift)   # all entries <= 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- step size (inverse‑sqrt decay) -----
        base_alpha = step0 / math.sqrt(it)

                cand_sc = sc

            no_improve = 0

        else:

                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- slowly tighten the soft‑max (approach true maximum) -----
        if it % 250 == 0:
            alpha_smooth = min(alpha_smooth * 1.25, 200.0)

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------
    # Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------
    return [float(x) for x in best]
