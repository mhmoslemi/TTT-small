def propose_candidate(seed=42, budget_s=1100, **kwargs):

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2048
        cur = rng.random(n, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s == 0.0:
        cur = np.full_like(cur, 1.0 / cur.size)
    else:
        cur /= s

    n = cur.size
    L = 1 << ((2*n - 1).bit_length())

    # Precompute constants
    target_score = 1.50274349   # for early exit if feasible, though may not be reached

    # Use Adam-like updates
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.1

    best_vec = cur.copy()

    a_fft = np.fft.rfft(cur, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    cur_max = conv.max()
    best_max = cur_max
    best_score = 2.0 * n * best_max   # sum normalized to 1

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # recompute convolution FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        cur_max = conv.max()

        # compute softmax weights over conv
        # temperature schedule
        temp = max(1e-12, 0.5 * (0.9995 ** it))
        # shift by max to avoid overflow
        w = np.exp((conv - cur_max) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient: 2 * convolution of cur with reversed weights
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)

        # exponentiated gradient step
        cand = cur * np.exp(-step * m_hat)
        # ensure nonneg and renormalize
        cand = np.maximum(cand, 0.0)
        sum_cand = cand.sum()
        if sum_cand == 0.0:
            cand = cur.copy()
        else:
            cand /= sum_cand

        # compute candidate max
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[:2*n-1]
        cand_max = cand_conv.max()

            cur_max = cand_max
            # update best

                    best_vec = sym.copy()

                tmp = np.maximum(tmp, 0.0)

                if tmp_sum > 0:
