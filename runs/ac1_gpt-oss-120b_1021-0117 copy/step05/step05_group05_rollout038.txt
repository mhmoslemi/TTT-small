def _grad(a):
    n = len(a)
    L = 1 << ((3 * n - 2).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    mask = conv >= max_val - 1e-12

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """2*n*max(conv(a,a))   (sum of a is assumed to be 1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _grad_and_max(a):
    """Sub‑gradient of max(conv) w.r.t. a and the current max value."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())

    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    mx = conv.max()

    # mask of the (approximately) active lag(s)
    mask = conv >= mx - 1e-12
    w = mask.astype(np.float64)
    cnt = w.sum()
    if cnt > 0:
        w /= cnt

    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)

    # indices that map the linear convolution back to a‑gradient
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g, mx
