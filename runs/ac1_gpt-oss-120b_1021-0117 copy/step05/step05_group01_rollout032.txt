import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – sum of vector set to 1, non‑negative
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = z}."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Compute full linear convolution via FFT (real‑valued)
# ------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (full length) using real FFT of size L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    # result length = 2*len(a)-1; extra tail is zero‑padded
    return conv[: 2 * a.size - 1]


# ------------------------------------------------------------
# Exact sub‑gradient of max‑convolution (averaged over all maximisers)
# ------------------------------------------------------------
def _grad_max_conv(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """Return the sub‑gradient of max_k (a*a)[k] (averaged over all attaining indices)."""
    n = a.size
    max_val = conv.max()
    # indices where the maximum is attained (tolerance for floating errors)
    idx = np.where(conv >= max_val - max(eps, eps * max_val))[0]
    cnt = idx.size
    g = np.zeros_like(a)
    for k in idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    if cnt > 1:
        g /= float(cnt)
    return g

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The routine runs for at most ``budget_s`` seconds and returns the
    best vector seen.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # --------------------------------------------------------
    # Initial vector – use parent if present, otherwise uniform
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # uniform start – works well for any n
        a = np.full(2560, 1.0, dtype=np.float64)  # size will be trimmed later
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1

    n = a.size
    # FFT length for convolution a*a (full length 2n‑1)
    L_conv = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    best_vec = a.copy()
    conv_full = _conv_full(a, L_conv)
    best_sc = 2.0 * n * conv_full.max()
    cur_sc = best_sc

    # optimisation hyper‑parameters
    step0 = 0.8               # base step magnitude
    beta = 0.90               # momentum factor
    momentum = np.zeros_like(a)
    sym_interval = 250        # symmetrisation period
    no_improve = 0
    it = 0
    target = 1.50274349       # stop early if reached

    # --------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution & score -----
        conv_full = _conv_full(a, L_conv)
        cur_max = conv_full.max()
        cur_sc = 2.0 * n * cur_max

        # ----- keep best -----
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit -----
        if best_sc < target:
            break

        # ----- gradient of the max‑convolution -----
        g = _grad_max_conv(a, conv_full)

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- back‑tracking line search -----
        base_alpha = step0 / math.sqrt(it)
        alpha = base_alpha
        cand = None
        cand_sc = None
        for _ in range(7):                     # up to 7 attempts
            a_tmp = a - alpha * momentum
            a_tmp = _project_simplex(a_tmp)     # stay on simplex
            conv_tmp = _conv_full(a_tmp, L_conv)
            cand_sc = 2.0 * n * conv_tmp.max()
            if cand_sc <= cur_sc:
                cand, cand_sc = a_tmp, cand_sc
                break
            alpha *= 0.5

        # if no descent found, keep the last trial (may be uphill)
        if cand is None:
            cand, cand_sc = a_tmp, cand_sc

        # ----- accept / reject (with occasional uphill move) -----
        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:

        # ----- periodic symmetrisation (helps the Toeplitz structure) -----
        if it % sym_interval == 0:

            sym_sc = 2.0 * n * conv_sym.max()
            if sym_sc < cur_sc:
                a, cur_sc = a_sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

    # --------------------------------------------------------
    # Return the best candidate seen (plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
