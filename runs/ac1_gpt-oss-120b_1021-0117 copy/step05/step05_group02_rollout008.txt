import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 upper‑bound (scale‑invariant, assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of the smoothed max‑convolution
# ------------------------------------------------------------
def _gradient(a: np.ndarray, eps: float, L: int):
    """Return (grad, max_conv) where grad is ∇ of the soft‑max objective."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)

    # convolution a*a
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    mx = conv.max()

    # soft‑max weights
    w = np.exp((conv - mx) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2 * (a * reverse(w)) convolution
    w_rev = w[::-1]
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, mx

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The routine respects the time budget and keeps the best candidate found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # leave a safety margin

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = np.full(1024, 1.0 / 1024, dtype=np.float64)

    a = np.clip(a, 0.0, 1000.0)
    a = _project_simplex(a)           # enforce sum = 1
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (power of two)

    best = a.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    # ---- Adam moments ------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999

    # ---- hyper‑parameters -------------------------------------------
    eps = 1.0          # soft‑max smoothing
    eps_decay = 0.9995
    eps_min = 1e-8
    base_lr = 0.5
    no_improve = 0
    it = 0
    target = 1.50274349

    # ---- optimisation loop ------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # gradient of smoothed objective
        g, mx = _gradient(a, eps, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / np.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # candidate 1 : Adam step on simplex
        cand1 = _project_simplex(a - step)
        cand1_sc = _score(cand1)

        # candidate 2 : mirror‑descent (exponentiated gradient) step
        eta = 0.3 / np.sqrt(it)
        exp_arg = -eta * g
        np.clip(exp_arg, -50, 50, out=exp_arg)   # avoid overflow
        cand2 = a * np.exp(exp_arg)
        s = cand2.sum()
        if s > 0.0:
            cand2 /= s
        else:
            cand2 = a.copy()
        cand2_sc = _score(cand2)

        # pick the better of the two proposals
        if cand1_sc <= cand2_sc:
            cand, cand_sc = cand1, cand1_sc
        else:
            cand, cand_sc = cand2, cand2_sc

        # accept if improvement, otherwise Metropolis‑like acceptance
        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            # temperature schedule (very gentle)
            temp = 0.02 / np.sqrt(it) + 1e-12
            prob = np.exp(-(cand_sc - cur_sc) / temp)
            if rng.random() < prob:
                a, cur_sc = cand, cand_sc
            no_improve += 1

        # ---- periodic symmetrisation ----
        if it % 500 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                a, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

        # ---- random mass‑transfer when stuck ----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    a, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ---- decay smoothing parameter ----
        eps = max(eps_min, eps * eps_decay)

        # early exit if target reached
        if best_sc < target:
            break

    # return the best candidate found as a plain Python list
    return [float(x) for x in best]
