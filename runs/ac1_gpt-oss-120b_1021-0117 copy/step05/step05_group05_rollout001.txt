def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start=time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.5)
    # get init

    n = cur.shape[0]
    L = 1 << ((3 * n - 2).bit_length())

    best_sc = _score(best_vec)
    cur_sc = best_sc
    momentum = np.zeros_like(cur)
    beta = 0.95
    step0 = 1.0
    no_improve = 0
    it = 0
    target=1.50274349
    # main loop
    while time.monotonic() < deadline:
        it +=1
        # compute convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)

import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _lp_optimize(a, M):
    try:
        import cvxpy as cp
    except Exception:
        return None
    n = a.shape[0]
    g = cp.Variable(n, nonneg=True)
    a_param = cp.Parameter(n, value=a, constant=True)
    conv_expr = cp.conv(a_param, g)  # length 2n-1
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv_expr <= M])
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        return None
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None
    gv = g.value
    if gv is None:
        return None
    return np.asarray(gv, dtype=np.float64)
