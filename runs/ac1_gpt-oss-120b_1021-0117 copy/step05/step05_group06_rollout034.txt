# ------------------------------------------------------------
# Simple projection onto the simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _proj_simplex_one(x: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and renormalise to sum = 1."""
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s > 0.0:
        x /= s
    else:
        x.fill(1.0 / x.size)
    return x


# ------------------------------------------------------------
# Maximum value of the (linear) convolution a*a (full length)
# ------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

            if eps < eps_min:
                eps = eps_min

            # Soft‑max weights (or exact sub‑gradient when eps tiny)
            if eps > 1e-7:
                w = np.exp((conv - max_val) / eps)
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(conv, 1.0 / conv.size)
                else:
                    w /= w_sum
            else:
                mask = conv >= max_val - 1e-12
                cnt = int(mask.sum())
                if cnt == 0:
                    w = np.full_like(conv, 1.0 / conv.size)
                else:
                    w = np.where(mask, 1.0 / cnt, 0.0)

            # Gradient of the smoothed objective
            w_rev = w[::-1]
            g = _gradient(a_fft, w_rev, L, n)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            a = a - step
            a = _proj_simplex_one(a)

            # Periodic evaluation
            if it % 50 == 0:
                sc = _max_conv(a, L)
                bound = 2.0 * n * sc
                if sc < best_sc:
                    best_sc = sc
                    best_seq = a.copy()
                    best_bound = bound
                if best_bound < target_bound:
                    break

        # final evaluation for this restart
        sc = _max_conv(a, L)
        bound = 2.0 * n * sc
        if sc < best_sc:
            best_sc = sc
            best_seq = a.copy()
            best_bound = bound
        if best_bound < target_bound:
            break

    # --------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # --------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_sc = _max_conv(best_seq, L)
        best_bound = 2.0 * n * best_sc

    # --------------------------------------------------------
    # Deterministic anti‑identity permutation (large ↔ small)
    # --------------------------------------------------------
    vals = np.sort(best_seq)[::-1]
    perm = np.empty_like(best_seq)
    l, r = 0, n - 1
    k = 0
    while l <= r:
        perm[l] = vals[k]
        k += 1
        if l < r:
            perm[r] = vals[k]
            k += 1
        l += 1
        r -= 1
    sc_perm = _max_conv(perm, L)
    bound_perm = 2.0 * n * sc_perm
    if bound_perm < best_bound:
        best_seq = perm.copy()
        best_sc = sc_perm
        best_bound = bound_perm

    # --------------------------------------------------------
    # Random cyclic shifts (few attempts)
    # --------------------------------------------------------
    shift_trials = min(20, n // 10)
    for _ in range(shift_trials):
        if time.monotonic() > deadline - 0.2:
            break
        shift = rng.integers(0, n)
        shifted = np.roll(best_seq, shift)
        sc_s = _max_conv(shifted, L)
        bound_s = 2.0 * n * sc_s
        if bound_s < best_bound:
            best_seq = shifted.copy()
            best_sc = sc_s
            best_bound = bound_s

    # --------------------------------------------------------
    # Simple hill‑climbing swaps (remaining time)
    # --------------------------------------------------------
    cur = best_seq.copy()
    cur_sc = best_sc
    cur_bound = best_bound
    while time.monotonic() < deadline - 0.1:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        sc_new = _max_conv(cur, L)
        bound_new = 2.0 * n * sc_new
        if bound_new < cur_bound:
            cur_bound = bound_new
            cur_sc = sc_new
            best_seq = cur.copy()
            best_sc = sc_new
            best_bound = bound_new
            if best_bound < target_bound:
                break
        else:
            cur[i], cur[j] = cur[j], cur[i]

    # --------------------------------------------------------
    # Return the best candidate found
    # --------------------------------------------------------
    if best_seq.size == 0:
        best_seq = np.array([1.0])
    return [float(x) for x in best_seq]
