def propose_candidate(seed=42,budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.15   # ensure return.

    # get base vector.
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=float)
    else:
        base = np.full(1024, 1/1024) # maybe fallback
    # clamp to [0,1000], ensure sum>0.
    base = np.clip(base,0,1000)
    s = base.sum()
    if s < 0.01:
        base = np.ones_like(base) / base.size
    else:
        base = base / s   # normalized to sum=1.

    n = base.size
    L = 1 << ((2*n - 1).bit_length())
    # ensures convolution length = power of two.

    # helper functions:
    def project_simplex(v):
        # returns v >=0 sum=1
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1
        rho = np.where(u - css / np.arange(1,len(v)+1) > 0)[0]
        if len(rho)==0:
            theta = 0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho+1)
        w = np.maximum(v - theta, 0)
        return w

    # Score compute using FFT (maybe faster) but we can reuse evaluate_sequence for final but for internal we compute same.
    def fast_score(a):
        # compute conv and max quickly.
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        # conv length L; we need first 2n-1 values.
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        return 2.0 * n * max_val   # not dividing by sum^2 because sum=1

    # initialize
    cur = base.copy()
    cur_score = fast_score(cur)

    best_vec = cur.copy()
    best_score = cur_score

    # Adam variables
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # parameters
    step_size = 0.5  # base learning rate
    eps_smoothing = 0.1 # initial smoothing for softmax
    eps_decay = 0.9995
    eps_min = 1e-8

    # counters
    iter_count = 0
    no_improve = 0
    restart_counter = 0

    target_score = 1.50274349 # we want less.

    while time.monotonic() < deadline:
        iter_count += 1

        # compute convolution and gradient
        # compute FFT of cur once
        cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # softmax weights
        eps_cur = max(eps_min, eps_smoothing * (eps_decay ** iter_count))
        # compute weights w_i = exp((conv_i - max_val)/eps_cur)
        # to avoid overflow, subtract max.
        w = np.exp((conv - max_val) / eps_cur)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient: derivative of sum_i w_i * conv_i w.r.t. a
        # conv_i = sum_j a_j * a_{i-j}
        # gradient = 2 * (a * reverse(w))[k]; compute using FFT:
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
        # mapping indices: for each index j in cur, gradient is: 2 * sum_{i} a_i * w_{i+j}
        # Actually conv = a * a, w is weight per index of conv. gradient d/d a_j = 2 * sum_{k} a_{k} * w_{j+k} (convolution of a with w). Implementation used earlier:
        # using idx = (2n-2) - np.arange(n)
        idx = (2*n - 2) - np.arange(n)  # mapping for reverse indexing as earlier
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iter_count)
        v_hat = v / (1 - beta2 ** iter_count)
        step = step_size / (np.sqrt(v_hat) + eps)
        cand = cur - step * m_hat
        # project to simplex (ensures nonneg and sum=1)
        cand = project_simplex(cand)
        cand_score = fast_score(cand)

        # if not improved, apply optional random perturb
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_vec, best_score = cand.copy(), cand_score
        else:
            # accept with prob exp(-(cand_score-cur_score)/temp)
            temp = max(0.01, step_size / np.sqrt(iter_count))
            if rng.random() < math.exp(-(cand_score - cur_score)/temp):
                cur, cur_score = cand, cand_score
            no_improve += 1

        # occasional random mass transfer
        if no_improve > 3000:
            # random redistribution: pick i with mass >0, move part to random j
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.4  # move up to 40% mass
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = project_simplex(tmp)
                tmp_score = fast_score(tmp)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    no_improve = 0
                if tmp_score < best_score:
                    best_vec, best_score = tmp.copy(), tmp_score

        # occasional restart if stuck too long
        if no_improve > 6000 and restart_counter < 3:
            # restart with random simplex point
            rand_vec = rng.random(n)
            rand_vec = project_simplex(rand_vec)
            cur = rand_vec
            cur_score = fast_score(cur)
            m.fill(0); v.fill(0)
            no_improve = 0
            restart_counter += 1

        # symmetrical averaging sometimes beneficial
        if iter_count % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = project_simplex(sym)
            sym_score = fast_score(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best_vec, best_score = sym.copy(), sym_score
                no_improve = 0

        if best_score < target_score:
            break

    # Return best solution as list
    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – sum = 1, non‑negative
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast objective (sum = 1 ⇒ denominator omitted)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a safety margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------

    # --------------------------------------------------------

    # FFT length sufficient for a * w (size n + (2n‑1) – 1 = 3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # --------------------------------------------------------

    step0 = 1.0               # base step for momentum variant

    eps_min = 1e-9
    beta = 0.9                # momentum coefficient
    momentum = np.zeros_like(cur)
    it = 0

    target = 1.50274349

    # --------------------------------------------------------

    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution -----

        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weights approximating the max -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
