def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale non‑negative vector to have sum = 1 (fallback uniform if sum = 0)."""
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _conv_max(a: np.ndarray, L: int):
    """Return (max convolution value, FFT of a, full convolution array)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _gradient_from_max(a: np.ndarray, kmax: int) -> np.ndarray:
    """Exact gradient of the maximal convolution entry w.r.t. a."""
    n = a.size
    g = np.zeros_like(a)
    if kmax < n:

        # fill from offset..n‑1 with reversed tail a[n‑1:offset-1:-1]
        g[offset:] = 2.0 * a[n - 1 : offset - 1 : -1]
    return g
