def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a0 = rng.random(n, dtype=np.float64) # random

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)  # sum=1

    n = a0.size
    L = 1 << ((2*n - 1).bit_length())

    cur = a0.copy()
    cur_sc = _max_conv(cur, L)
    best = cur.copy()
    best_sc = cur_sc
    best_bound = 2.0 * n * best_sc

    # hyperparameters
    step0 = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    it = 0

    # main loop
    while time.monotonic() < deadline - 1.0:
        it += 1
        # compute conv and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2*n - 1]
        max_val = conv.max()
        # subgradient weight mask
        mask = conv == max_val
        cnt = int(mask.sum())
        if cnt == 0:
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = np.where(mask, 1.0 / cnt, 0.0)

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)

        # learning rate decay
        alpha = step0 / math.sqrt(it)

        # line-search: try decreasing step
        taken = False
        for _ in range(6):
            step = alpha * (m_hat / (np.sqrt(v_hat) + eps_adam))
            cand = cur - step
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc <= cur_sc:
                cur = cand
                cur_sc = cand_sc
                taken = True
                break
            else:
                alpha *= 0.5
        if not taken:
            # fallback to simple gradient step with small alpha
            cand = cur - 0.1 * g
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc < cur_sc:
                cur = cand; cur_sc = cand_sc

        # update best
        if cur_sc < best_sc:
            best = cur.copy()
            best_sc = cur_sc
            best_bound = 2.0 * n * best_sc
            # early exit if good enough
            if best_bound < 1.50274349:
                break

        # periodic symmetrization
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _max_conv(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if cur_sc < best_sc:
                    best = cur.copy()
                    best_sc = cur_sc
                    best_bound = 2.0 * n * best_sc

    # final hillclimb swaps
    cur_swap = best.copy()
    cur_sc_swap = best_sc
    while time.monotonic() < deadline - 0.2:
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        cur_swap[i], cur_swap[j] = cur_swap[j], cur_swap[i]
        new_sc = _max_conv(cur_swap, L)
        if new_sc < cur_sc_swap:
            cur_sc_swap = new_sc
            best = cur_swap.copy()
            best_sc = new_sc
            best_bound = 2.0 * n * new_sc
            if best_bound < 1.50274349:
                break
        else:
            # revert
            cur_swap[i], cur_swap[j] = cur_swap[j], cur_swap[i]

    # final output
    if best.size == 0:
        best = np.array([1.0])
    # Ensure finite list
    return [float(x) for x in best]

import time
import math
import numpy as np

# --------------------------------------------------------------
# Simplex projection (L2 onto {x>=0, sum=1})
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (sum=1, non‑negative)."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
# Maximum of full autocorrelation a*a using zero‑padded FFT
# --------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max(conv_full) for a*a (full length 2n‑1) via FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())


# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep small safety margin

    # ------------------------------------------------------------------
    # Initial vector – use the provided parent if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: uniform random start of moderate length
        a0 = rng.random(2000, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)          # enforce sum = 1, non‑negative
    n = a0.size

    # ------------------------------------------------------------------
    # FFT length (next power of two ≥ 2*n)
    # ------------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    cur = a0.copy()
    cur_sc = _max_conv(cur, L)
    best = cur.copy()
    best_sc = cur_sc
    best_bound = 2.0 * n * best_sc

    # ------------------------------------------------------------------
    # Parameters for Adam‑style sub‑gradient descent
    # ------------------------------------------------------------------
    step0 = 1.0
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Main (matrix‑free) optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline - 2.0:
        it += 1

        # ---- current autocorrelation and its maximum ------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ---- sub‑gradient weight (exact max, not softmax) ----------
        mask = np.isclose(conv, max_val, atol=1e-12)
        cnt = int(mask.sum())
        if cnt == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = np.where(mask, 1.0 / cnt, 0.0)

        # ---- gradient of the max‑convolution ------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ---- Adam moments -------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ---- adaptive step (line‑search with back‑tracking) ---------
        alpha = step0 / math.sqrt(it)
        improved = False
        for _ in range(6):
            step = alpha * (m_hat / (np.sqrt(v_hat) + eps_adam))
            cand = cur - step
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc <= cur_sc:
                cur = cand
                cur_sc = cand_sc
                improved = True
                break
            alpha *= 0.5

        # fallback: tiny gradient step if back‑tracking never succeeded
        if not improved:
            cand = cur - 0.1 * g
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc < cur_sc:
                cur = cand
                cur_sc = cand_sc

        # ---- keep global best ----------------------------------------
        if cur_sc < best_sc:
            best = cur.copy()
            best_sc = cur_sc
            best_bound = 2.0 * n * best_sc
            if best_bound < target:

        # ---- occasional symmetrisation (helps Toeplitz structure) ---
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _max_conv(sym, L)
            if sym_sc < cur_sc:

                cur_sc = sym_sc
                if cur_sc < best_sc:

                    best_sc = cur_sc

    # Post‑processing: simple hill‑climbing swaps to polish the result
    # ------------------------------------------------------------------
    cur_swap = best.copy()
    cur_sc_swap = best_sc
    while time.monotonic() < deadline - 0.2:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur_swap[i], cur_swap[j] = cur_swap[j], cur_swap[i]
        new_sc = _max_conv(cur_swap, L)
        if new_sc < cur_sc_swap:
            cur_sc_swap = new_sc
            best = cur_swap.copy()
            best_sc = new_sc
            best_bound = 2.0 * n * new_sc
            if best_bound < target:
                break
        else:
            # revert the swap
            cur_swap[i], cur_swap[j] = cur_swap[j], cur_swap[i]

    # ------------------------------------------------------------------
    # Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best]
