import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a vector with sum = 1 (i.e. 2*n*max(conv))
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.size
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    Runs for up to `budget_s` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a uniform start
    # --------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = np.full(1024, 1.0 / 1024, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)  # enforce sum = 1
    n = cur.size

    # --------------------------------------------------------------------
    # FFT planning – length ≥ 3n‑2, power of two (for gradient step)
    # --------------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # --------------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------------
    step0 = 0.9               # momentum step base
    step0_mirror = 0.5        # mirror‑descent step base
    step0_adam = 0.8          # Adam step base
    eps0 = 0.5                # initial soft‑max smoothing
    eps_decay = 0.98          # faster decay than before
    eps_min = 1e-8
    beta = 0.9                # momentum coefficient
    momentum = np.zeros_like(cur)

    beta1, beta2 = 0.9, 0.999   # Adam coefficients
    m_adam = np.zeros_like(cur)
    v_adam = np.zeros_like(cur)

    no_improve = 0
    it = 0
    target = 1.50274349

    # --------------------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # --------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # --------- convolution of current vector ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_sc = 2.0 * n * max_val

        # --------- soft‑max weights for smoothed max ----------
        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps < 1e-6:                     # switch to exact sub‑gradient
            w = (conv_full == max_val).astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0.0:
                w[:] = 1.0 / w.size
            else:
                w /= w_sum
        else:
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum

        # --------- gradient of the smoothed objective ----------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # --------- momentum update ----------
        momentum = beta * momentum + (1.0 - beta) * g

        # --------- Adam moments ----------
        m_adam = beta1 * m_adam + (1.0 - beta1) * g
        v_adam = beta2 * v_adam + (1.0 - beta2) * (g ** 2)

        # --------- candidate 1 : momentum (gradient) step ----------
        alpha = step0 / math.sqrt(it)
        cand1 = cur - alpha * momentum
        cand1 = _project_simplex(cand1)
        cand1_sc = _score(cand1, L)

        # --------- candidate 2 : mirror descent ----------
        eta = step0_mirror / math.sqrt(it)
        exp_arg = -eta * g
        np.clip(exp_arg, -50, 50, out=exp_arg)          # avoid overflow
        cand2 = cur * np.exp(exp_arg)
        s = cand2.sum()
        if s > 0:
            cand2 /= s
        else:
            cand2 = cur.copy()
        cand2_sc = _score(cand2, L)

        # --------- candidate 3 : Adam ----------
        # bias‑correction
        m_hat = m_adam / (1.0 - beta1 ** it)
        v_hat = v_adam / (1.0 - beta2 ** it)
        step_adam = step0_adam / math.sqrt(it)
        adam_step = step_adam * m_hat / (np.sqrt(v_hat) + 1e-8)

        cand3 = _project_simplex(cand3)
        cand3_sc = _score(cand3, L)

        cand_sc = min(cand1_sc, cand2_sc, cand3_sc)

        elif cand_sc == cand2_sc:
            cand, cand_sc = cand2, cand2_sc
        else:
            cand, cand_sc = cand3, cand3_sc

        # --------- accept if improvement, otherwise Metropolis ----------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            # small uphill probability
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # --------- periodic symmetrisation ----------
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # --------- random mass‑transfer when stuck ----------
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # --------- early exit if target reached ----------
        if best_sc < target:
            break

        # --------- time check (avoid overshoot) ----------
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
