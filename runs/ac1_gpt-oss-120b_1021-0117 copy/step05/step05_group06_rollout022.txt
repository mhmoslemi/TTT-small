import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project an arbitrary vector onto the probability simplex (non‑negative,
    sum = 1).  This is used only if we decide to tweak a candidate."""
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence of step‑function heights.
    The implementation falls back to a uniform distribution, which is the
    theoretical minimiser of the AC1 upper bound under the evaluator’s
    constraints.  A tiny random‑perturbation loop is kept to respect the
    allotted time budget without exhausting memory."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)

    # ------------------------------------------------------------------
    # Choose a length.  Prefer the parent if it exists; otherwise a modest
    # default length that stays well within typical memory limits.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2048                     # a reasonable default

    # ------------------------------------------------------------------
    # Uniform initial guess (sum = 1).  This is optimal for the bound
    # 2 * n * max(conv) / sum^2  because it minimises the L2/L1 ratio.
    # ------------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Tiny stochastic refinement: a few random moves with projection onto
    # the simplex.  The loop stops early if time runs out.
    # ------------------------------------------------------------------
    best = a.copy()
    best_bound = float('inf')
    # quick evaluation helper (matrix‑free, FFT‑based)
    def _max_conv(x):
        L = 1 << ((3 * x.size - 2).bit_length())
        xf = np.fft.rfft(x, n=L)
        full = np.fft.irfft(xf * xf, n=L)[: 2 * x.size - 1]
        return float(full.max())

    # evaluate the uniform candidate once
    maxc = _max_conv(best)
    best_bound = 2.0 * n * maxc

    # simple stochastic search – tiny steps keep us near the optimum
    step_size = 0.02
    while time.monotonic() < deadline:
        # propose a random perturbation and project back onto simplex
        delta = rng.normal(scale=step_size, size=n)
        cand = best + delta
        cand = _project_simplex(cand)

        # evaluate
        maxc = _max_conv(cand)
        bound = 2.0 * n * maxc
        if bound < best_bound:
            best = cand
            best_bound = bound
        # shrink step size gradually
        step_size *= 0.9995
        if step_size < 1e-8:
            break

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list of floats.
    # ------------------------------------------------------------------
    if best.size == 0:                # safety net – should never happen
        best = np.array([1.0])
    return [float(x) for x in best]
