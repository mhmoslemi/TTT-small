def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale to have sum 1, keeping non‑negativity."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s

    """
    Return max of a*a (full linear convolution) together with FFT data
    and the index where the maximum occurs.
    """
    a_fft = np.fft.rfft(a, n=L)

    max_val = conv.max()
    k = int(conv.argmax())
    return max_val, a_fft, conv, k

    mask = (src >= 0) & (src < n)
    g = np.zeros_like(a)
    g[mask] = a[src[mask]]
    return g

        cand, sc = _candidate_from_array(a)
        if sc < best_score:
            best_score, best_vec = sc, cand

    # Triangular shape
    mid = (n - 1) // 2
    tri = np.maximum(0.0, (mid + 1) - np.abs(np.arange(n) - mid))
    cand, sc = _candidate_from_array(tri)
    if sc < best_score:
        best_score, best_vec = sc, cand

    # Sine shape
    sine = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    cand, sc = _candidate_from_array(sine)
    if sc < best_score:
        best_score, best_vec = sc, cand

    # --------------------------------------------------------
    # Return the best sequence found (plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
