import time, math, numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Score used by the evaluator – assumes sum(a)=1 (the evaluator
# rescales internally, but keeping sum=1 makes the objective homogeneous)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a, a))."""
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())          # length for FFT (>=3n‑2)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    Runs for up to `budget_s` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # tiny safety margin

    # ----- initialise ----------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 1024
        cur = np.full(n0, 1.0 / n0, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)               # enforce sum = 1
    n = cur.size

    # ----- FFT planning (power of two >= 3n‑2) ---------------------------
    L = 1 << ((3 * n - 2).bit_length())

    # ----- book‑keeping -------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # ----- optimisation hyper‑parameters --------------------------------
    # Adam moments
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.30                                 # base learning rate

    # smoothed‑max (soft‑max) parameters
    eps0, eps_decay, eps_min = 0.5, 0.9995, 1e-9

    # mirror‑descent step size
    step0_mirror = 0.8

    # plain momentum (kept for occasional back‑track)
    beta = 0.95

    target = 1.50274349
    no_improve = 0
    it = 0

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current vector ----------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weighting (smooth approximation of max) -------
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective ----------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- occasionally blend exact sub‑gradient --------------------
        if it > 2000 and (it % 10 == 0):
            tol = 1e-12
            max_idx = np.where(np.abs(conv - max_val) <= tol)[0]
            if 0 < max_idx.size <= 8:          # cheap exact gradient
                g_exact = np.zeros_like(cur)
                for k in max_idx:
                    i_start = max(0, k - (n - 1))
                    i_end = min(k, n - 1)
                    if i_start <= i_end:
                        i_range = np.arange(i_start, i_end + 1)
                        g_exact[i_range] += cur[k - i_range]
                    i_start2 = max(k, 0)
                    i_end2 = n - 1
                    if i_start2 <= i_end2:
                        i_range2 = np.arange(i_start2, i_end2 + 1)
                        g_exact[i_range2] += cur[i_range2 - k]
                g = 0.5 * g + 0.5 * g_exact

        # ----- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps_adam)
        cand = cur - step * m_hat
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # ----- mirror‑descent candidate --------------------------------
        eta = step0_mirror / math.sqrt(it)
        exp_arg = -eta * g
        np.clip(exp_arg, -50, 50, out=exp_arg)
        cand2 = cur * np.exp(exp_arg)
        s = cand2.sum()
        cand2 = cand2 / s if s > 0 else cur.copy()
        cand2_sc = _score(cand2)

        # ----- select the better of the two ----------------------------
        if cand_sc <= cand2_sc:
            cand, cand_sc = cand, cand_sc
        else:
            cand, cand_sc = cand2, cand2_sc

        # ----- accept if improvement, otherwise occasional uphill ------
        if cand_sc < cur_sc - 1e-12:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (step0_mirror + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps with Toeplitz structure) --
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random mass‑transfer when stuck -------------------------
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target reached ----------------------------
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
