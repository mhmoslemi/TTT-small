def _grad_max(a, conv):
    n = a.size
    max_val = conv.max()
    # tolerance
    tol = max_val * 1e-12
    idxs = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # for i in [i_start, i_end]
        # g[i] += 2 * a[k-i]
        # vectorized: indices = np.arange(i_start, i_end+1)
        # g[indices] += 2 * a[k - indices]
        inds = np.arange(i_start, i_end+1, dtype=int)
        g[inds] += 2 * a[k - inds]
    return g

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Exact sub‑gradient of max_k (a*a)[k]
# ------------------------------------------------------------
def _grad_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the maximum convolution entry."""
    n = a.size
    max_val = conv.max()
    # tolerance to catch ties caused by floating point noise
    tol = max_val * 1e-12
    idxs = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        inds = np.arange(i_start, i_end + 1, dtype=int)
        g[inds] += 2.0 * a[k - inds]
    return g


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # ------------------------------------------------------------------
    # Initialise from the parent if available; otherwise use a half‑sine.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2048
        i = np.arange(n0, dtype=np.float64)
        a = np.sin(np.pi * (i + 0.5) / (2 * n0))
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())  # next power of two >= 2n‑1

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    best_sc = 2.0 * n * conv_full.max()
    best_vec = a.copy()

    # optimisation parameters
    step0 = 0.6          # base step size
    beta = 0.9           # momentum factor
    momentum = np.zeros_like(a)
    no_improve = 0
