import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Compute the maximum coefficient of the self‑convolution of a
# ------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int, n: int) -> float:
    """Return max( a * a ) using an FFT of length L (n = len(a))."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return float(conv_full[: 2 * n - 1].max())

        # ----- candidate generation -----
        cand = cur - step * momentum
        cand = _project_simplex(cand)

        # ----- evaluate candidate (hard max) -----
        cand_score = 2.0 * n * _max_conv(cand, L, n)

        # ----- accept if better, else metropolis move -----
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
        else:
            prob = math.exp(-(cand_score - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score
            no_improve += 1

        # ----- periodic symmetrisation (helps with the Toeplitz structure) -----
        if it % sym_period == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = 2.0 * n * _max_conv(sym, L, n)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = 2.0 * n * _max_conv(tmp, L, n)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score
                no_improve = 0

        # ----- early exit if target reached -----
        if best_score < target:
            break

    # ------------------------------------------------------------
    # Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------
    return [float(x) for x in best]
