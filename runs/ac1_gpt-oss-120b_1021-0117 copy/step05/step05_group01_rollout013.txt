import time

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x) = z}."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u > css / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Gradient from the top‑k largest convolution entries
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    n = a.size
    # indices of the `top_k` largest values (unsorted)
    idxs = np.argpartition(-conv, top_k - 1)[:top_k]

    for k in idxs:

        i_end = min(n - 1, k)
        if i_start <= i_end:
            i = np.arange(i_start, i_end + 1, dtype=int)
            g[i] += 2.0 * a[k - i]
    return g / max(1, top_k)
