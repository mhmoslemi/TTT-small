import time
import numpy as np

def _fast_score(arr: np.ndarray) -> float:
    """Fast evaluation of 2*n*max(conv)/sum(arr)^2 using FFT."""
    n = arr.shape[0]
    # length for zero‑padded linear convolution (next power of two)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    s = arr.sum()
    # sum is guaranteed >0 by the caller
    return 2.0 * n * max_b / (s * s)
