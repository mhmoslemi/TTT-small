import time, math, numpy as np

def _project_simplex(v):
    # Simple Euclidean projection onto probability simplex
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    # compute 2*n*max(convolution) with sum normalized to anything (here we use sum=1)
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initialise
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # default length 2000 uniform
        n0 = 2000
        cur = np.full(n0, 1.0 / n0, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)
    n = cur.size

    # FFT length for gradient (convolution of a with weight ~2n length)
    L = 1 << ((3 * n - 2).bit_length())

    # hyper‑parameters
    eps0 = 0.5
    eps_decay = 0.9994
    eps_min = 1e-12
    step0 = 0.6               # additive momentum base step
    step0_mirror = 0.4        # mirror‑descent base step
    beta = 0.9                # momentum coefficient
    momentum = np.zeros_like(cur)

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    it = 0
    no_improve = 0
    target = 1.50274349

    # main optimisation loop
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and soft‑max weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- candidate from momentum step -----
        alpha = step0 / math.sqrt(it)
        cand1 = cur - alpha * momentum
        cand1 = _project_simplex(cand1)
        cand1_sc = _score(cand1)

        # back‑track if necessary (up to 6 halvings)
        if cand1_sc > cur_sc:
            for _ in range(6):
                alpha *= 0.5
                cand_tmp = cur - alpha * momentum
                cand_tmp = _project_simplex(cand_tmp)
                tmp_sc = _score(cand_tmp)
                if tmp_sc <= cur_sc:
                    cand1, cand1_sc = cand_tmp, tmp_sc
                    break
            else:
                cand1, cand1_sc = cand_tmp, tmp_sc

        # ----- mirror‑descent step -----
        eta = step0_mirror / math.sqrt(it)
        exp_arg = -eta * g
        np.clip(exp_arg, -50, 50, out=exp_arg)
        cand2 = cur * np.exp(exp_arg)
        s = cand2.sum()
        if s > 0:
            cand2 /= s
        else:
            cand2 = cur.copy()
        cand2_sc = _score(cand2)

        # ----- choose better candidate -----
        if cand1_sc <= cand2_sc:
            cand, cand_sc = cand1, cand1_sc
        else:
            cand, cand_sc = cand2, cand2_sc

        # ----- accept if improvement, otherwise occasional uphill -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            # Metropolis‑like uphill acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps Toeplitz structure) -----
        if it % 300 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- targeted random move when stuck -----
        if no_improve > 1500:
            # move mass from index with large gradient to index with small gradient
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0
            else:
                # fallback random mass transfer
                i = rng.integers(0, n)
                if cur[i] > 0:
                    delta = cur[i] * rng.random() * 0.5
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        # ----- early exit if target met -----
        if best_sc < target:
            break

    # final random refinement phase (if time left)
    while time.monotonic() < deadline - 0.2:
        # small perturbation
        i = rng.integers(0, n)
        if cur[i] == 0:
            continue
        delta = cur[i] * rng.random() * 0.3
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        # low‑probability uphill move
        elif rng.random() < 0.001:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc

    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Scoring function used by the evaluator (sum is assumed = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a, a))."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑empty non‑negative sequence whose AC1 bound is as low as
    possible within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # --------------------------------------------------------
    # Initial vector – use the supplied parent if present
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: a uniform distribution of length 2000
        n0 = 2000
        cur = np.full(n0, 1.0 / n0, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)
    n = cur.size

    # --------------------------------------------------------
    # FFT length sufficient for a * w (a length‑n convolved with length‑(2n‑1) weights)
    # --------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------
    # Hyper‑parameters (tuned to improve over the reference method)
    # --------------------------------------------------------
    eps0 = 0.5                 # initial soft‑max smoothing
    eps_decay = 0.9994         # faster decay → sharper approximation earlier
    eps_min = 1e-12
    step0 = 0.6                # base step for the additive (momentum) update
    step0_mirror = 0.4         # base step for the multiplicative (mirror) update
    beta = 0.9                 # momentum coefficient
    momentum = np.zeros_like(cur)

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    it = 0
    no_improve = 0
    target = 1.50274349

    # --------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution (needed for both score and gradient) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weights approximating the max -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective (via convolution) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]                     # ∇_a (soft‑max max)

        # ----- momentum update -----

                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0
            else:

                    tmp[i] -= delta

    while time.monotonic() < deadline - 0.2:
