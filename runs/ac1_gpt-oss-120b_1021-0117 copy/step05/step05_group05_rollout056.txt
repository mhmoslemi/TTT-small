    rng = np.random.default_rng(seed)
    start = time.monotonic()

    else:

    cur_sc = best_sc

    v = np.zeros_like(cur)

    beta2 = 0.999

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Score = 2*n*max(conv(a,a)) (sum is 1 after projection)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _gradient(a, L):
    """Sub‑gradient of max‑convolution at a (using a convex combination of active indices)."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)

    # full autocorrelation
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()

    # active set (indices where the max is attained)
    mask = conv_full >= max_val - 1e-12
    w = mask.astype(np.float64)
    cnt = w.sum()
    if cnt > 1:
        w /= cnt

    # reverse for convolution with a
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)

    # gradient = 2 * (a * w_rev) evaluated at appropriate shift
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val
