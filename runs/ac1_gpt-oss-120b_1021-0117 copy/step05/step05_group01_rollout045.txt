# ------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x)=s}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - s
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Compute max value of the (full) autocorrelation using FFT
# ------------------------------------------------------------
def _conv_max_fft(a: np.ndarray, L: int) -> float:
    """Return max_k (a*a)[k] using an FFT of length L (>=2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return float(conv[: 2 * a.size - 1].max())

# ------------------------------------------------------------
def _grad_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:

    no_improve = 0
    it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution and score
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        cur_max = conv.max()
        cur_sc = 2.0 * n * cur_max

        # keep best
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # early exit if target reached
        if best_sc < target:
            break

        # sub‑gradient of the max‑convolution
        g = _grad_max_conv(a, conv)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # back‑tracking line search
        alpha = step0 / np.sqrt(it)
        for _ in range(8):
            a_tmp = a - alpha * momentum
            a_tmp = _project_simplex(a_tmp, s=1.0)
            # symmetrise candidate (keeps the search in the symmetric subspace)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = _project_simplex(a_tmp, s=1.0)

            cand_sc = 2.0 * n * _conv_max_fft(a_tmp, L)
            if cand_sc <= cur_sc:
                a, cur_sc = a_tmp, cand_sc
                break
            alpha *= 0.5
        else:
            # no descent found – accept with Metropolis probability
            prob = np.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                a, cur_sc = a_tmp, cand_sc

        # periodic random mass‑transfer when stuck
        if no_improve > max(5000, 2 * n):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.2
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, s=1.0)
                a = (a + a[::-1]) * 0.5
                a = _project_simplex(a, s=1.0)
            no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
