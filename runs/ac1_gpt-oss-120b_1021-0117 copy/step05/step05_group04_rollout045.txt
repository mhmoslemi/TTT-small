import time

import numpy as np

# ------------------------------------------------------------
# Helper functions (matrix‑free, linear memory)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale non‑negative vector to have sum = 1 (fallback to uniform)."""
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _conv_max_fft(a: np.ndarray, L: int):
    """Return (max convolution, FFT of a, full convolution array)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _gradient_sub(a_fft: np.ndarray, conv: np.ndarray, max_val: float,
                  L: int, n: int):
    """
    Subgradient of max (a*a) using indicator at maximal convolution entries.
    Returns a length‑n gradient vector.
    """
    # Indicator (or near‑indicator) weight vector w on convolution indices
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)
    total = w.sum()
    if total == 0.0:
        w = np.full_like(conv, 1.0 / conv.size, dtype=np.float64)
    else:
        w /= total
    # Reverse for convolution with a
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)  # selects n entries
    return 2.0 * grad_full[idx]
