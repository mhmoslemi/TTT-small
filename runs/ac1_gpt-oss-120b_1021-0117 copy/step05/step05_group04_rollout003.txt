import time, math, numpy as np

def _normalize(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

def _self_conv_max(a, L):
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*a.size-1]
    return conv.max(), a_fft, conv

def _softmax_weights(conv, max_val, eps):
    w = np.exp((conv - max_val) / eps)
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / s

def _gradient(a_fft, w, n, L):
    # w length 2*n-1
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # init vector
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _normalize(cur)
    n = cur.size

    # FFT length: next pow of two >= 2*n-1
    L = 1 << ((2 * n - 1).bit_length())
    # Alternatively a bit larger to avoid aliasing? Use as is.

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # Smoothing schedule
    eps0 = 0.5
    eps_decay = 0.9995
    eps_min = 1e-10

    # step schedule
    step0 = 0.8

    # tracking
    best_vec = cur.copy()
    best_max, _, _ = _self_conv_max(cur, L)
    best_score = 2.0 * n * best_max
    no_improve = 0

    it = 0
    # main loop
    while time.monotonic() < deadline:
        it += 1

        cur_max, a_fft, conv = _self_conv_max(cur, L)

        # choose weighting
        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps > 1e-6:
            w = _softmax_weights(conv, cur_max, eps)
        else:
            # exact max weight: uniform over indices achieving max within tolerance
            mask = (conv >= cur_max - 1e-12)
            count = mask.sum()
            if count == 0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = mask.astype(float) / count

        grad = _gradient(a_fft, w, n, L)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        direction = m_hat / (np.sqrt(v_hat) + eps_adam)

        step = step0 / math.sqrt(it + 1.0)

def _normalize(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex {v ≥ 0, sum(v)=1}."""

    s = v.sum()

    """Return (max of a*a, rfft(a), full convolution)."""
    a_fft = np.fft.rfft(a, n=L)

    return conv.max(), a_fft, conv


def _softmax_weights(conv: np.ndarray, max_val: float, eps: float) -> np.ndarray:
    """Soft‑max over convolution values with temperature eps."""
