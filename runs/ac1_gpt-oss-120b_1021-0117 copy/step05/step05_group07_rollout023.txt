import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

        cur = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))  # Hann
        cur = cur.astype(np.float64)

    # enforce positivity and simplex constraint (sum = 1)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    # FFT planning – length at least 3n‑2 (next power of two)
    # ------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())

    max_val = conv_full.max()
    cur_sc = 2.0 * n * max_val
    best_sc = cur_sc

    beta1, beta2 = 0.9, 0.999

    v = np.zeros_like(cur)

    step0 = 0.8               # base step magnitude
    eps0 = 1.0                # initial soft‑max temperature
    eps_decay = 0.995         # slower cooling
    eps_min = 1e-8
    target = 1.50274349
    it = 0
    no_improve = 0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:

        # ----- current convolution and its maximum -----

        max_val = conv_full.max()
        cur_sc = 2.0 * n * max_val

        # ----- keep best ever -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- soft‑max weights for the gradient -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps > 1e-12:

            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum
        else:

            mask = conv_full == max_val

            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))

            # retain the no_improve counter (already incremented above)

        # ----- periodic symmetrisation (helps with Toeplitz structure) -----
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
