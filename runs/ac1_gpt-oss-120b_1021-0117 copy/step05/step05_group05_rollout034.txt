import time, math, numpy as np

def _project_simplex(v):
    """Projection of a vector onto the probability simplex (sum = 1, non‑negative)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score_vec(a):
    """Return 2*n*max(conv(a,a)) – the same quantity used by the evaluator when sum=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initial vector ---------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: a modest random start
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)                # sum = 1, non‑negative
    n = a.size
    L = 1 << ((3 * n - 2).bit_length())    # enough for all convolutions used

    # optimisation hyper‑parameters -----------------------------------------
    step0 = 0.8
    beta = 0.9                               # momentum decay
    momentum = np.zeros_like(a)
    max_alpha = 50.0                         # final soft‑max sharpness
    alpha = 1.0                              # initial soft‑max sharpness

    best_a = a.copy()
    best_sc = _score_vec(best_a)
    cur_sc = best_sc
    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution of current iterate --------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- smooth approximation of the max (soft‑max) --------------------
        alpha = min(max_alpha, 1.0 + it * 0.005)            # raise slowly
        exp_term = np.exp(alpha * (conv - max_val))       # max term -> 1.0
        w = exp_term / exp_term.sum()                     # weights sum to 1

        # ----- gradient of the smooth max ------------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)                  # select needed slice
        grad = 2.0 * grad_full[idx]

        # ----- momentum -------------------------------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- back‑tracking line search --------------------------------------
        step = step0 / math.sqrt(it)
        cand = _project_simplex(a - step * momentum)
        cand_sc = _score_vec(cand)
        tmp_step = step
        for _ in range(8):
            if cand_sc <= cur_sc:
                break
            tmp_step *= 0.5
            cand = _project_simplex(a - tmp_step * momentum)
            cand_sc = _score_vec(cand)

        # ----- accept / reject -------------------------------------------------
        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # occasional simulated‑annealing move
            temp = max(1e-4, step)
            if rng.random() < math.exp(-(cand_sc - cur_sc) / temp):
                a, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- keep best -------------------------------------------------------
        if cur_sc < best_sc:
            best_a, best_sc = a.copy(), cur_sc

        # ----- periodic symmetrisation (helps for many instances) ------------
        if it % 600 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < cur_sc:
                a, cur_sc = sym, sym_sc
                no_improve = 0
            if sym_sc < best_sc:
                best_a, best_sc = sym.copy(), sym_sc

        # ----- random local perturbation after stagnation ----------------------
        if no_improve > 1500:
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                delta = a[i] * rng.random() * 0.3
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_vec(tmp)
                if tmp_sc < cur_sc:
                    a, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_a, best_sc = tmp.copy(), tmp_sc

        # ----- early exit if target reached ------------------------------------
        if best_sc <= target:
            break

    return [float(x) for x in best_a]
