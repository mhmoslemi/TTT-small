    from numpy.fft import rfft, irfft

    rng = np.random.default_rng(seed)

    # Project onto simplex (sum=1)
    def proj_simplex(v):
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)
    a = proj_simplex(np.maximum(a, 0.0))

    def score(vec):
        af = rfft(vec, n=L)
        conv = irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())
    
    best_vec = a.copy()
    best_sc = score(best_vec)

    # Adam-like parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta2 = 0.999

    it = 0

    target = 1.50274349

        af = rfft(a, n=L)
        conv_full = irfft(af * af, n=L)[:2*n-1]
        max_val = conv_full.max()

        tol = 1e-12  # absolute tolerance
        mask = conv_full >= max_val - tol
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 1:
            w /= cnt
        # reversed w to compute subgradient
        w_rev = w[::-1]
        wf = rfft(w_rev, n=L)
        grad_full = irfft(af * wf, n=L)

        m_hat = m / (1 - beta1**it)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # projection onto simplex
    def proj_simplex(v):
        v = np.asarray(v, dtype=np.float64)
        n = v.shape[0]
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    # initialize
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a0 = rng.random(1024, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    a0 = (a0 + a0[::-1]) * 0.5
    a = proj_simplex(a0)

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    from numpy.fft import rfft, irfft

    def score_fft(vec):
        af = rfft(vec, n=L)
        conv = irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())

    best_vec = a.copy()
    best_sc = score_fft(best_vec)

    mom = np.zeros_like(a)
    beta = 0.9
    alpha0 = 1.0
    it = 0
    no_improve = 0
    target = 1.50274349
    # tolerance for active set
    act_tol = 1e-12

    while time.monotonic() < deadline:
        it += 1
        af = rfft(a, n=L)
        conv_full = irfft(af * af, n=L)[:2*n-1]
        max_val = conv_full.max()
        # active indices
        active = np.where(conv_full >= max_val - act_tol)[0]
        # pick random active index
        if active.size > 0:
            idx_active = rng.choice(active)
        else:
            idx_active = np.argmax(conv_full)
        w = np.zeros(2*n-1, dtype=np.float64)
        w[idx_active] = 1.0
        w_rev = w[::-1]
        wf = rfft(w_rev, n=L)
        grad_full = irfft(af * wf, n=L)
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum
        mom = beta * mom + (1.0 - beta) * g

        # step schedule
        step = alpha0 / math.sqrt(it)

        # backtracking search
        cand = None
        cand_sc = None
        for _ in range(12):
            cand_tmp = proj_simplex(a - step * mom)
            cand_tmp_sc = score_fft(cand_tmp)
            if cand_tmp_sc <= best_sc:  # accept any improvement over best
                cand, cand_sc = cand_tmp, cand_tmp_sc
                break
            if cand_tmp_sc < score_fft(a):
                cand, cand_sc = cand_tmp, cand_tmp_sc
                break
            step *= 0.5
        if cand is None:
            cand, cand_sc = proj_simplex(a - step * mom), score_fft(proj_simplex(a - step * mom))

        # Decide acceptance
        cur_sc = score_fft(a)
        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            # occasional uphill acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                a, cur_sc = cand, cand_sc
            no_improve += 1

        # periodic symmetry
        if it % 200 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = proj_simplex(sym)
            sym_sc = score_fft(sym)
            if sym_sc < best_sc:
                best_sc, best_vec = sym_sc, sym.copy()
            if sym_sc < cur_sc:
                a, cur_sc = sym, sym_sc
                no_improve = 0

        # stagnation handling

            # random mass transfer
            i = rng.integers(0, n)
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n)

                a = proj_simplex(a)
                cur_sc = score_fft(a)
                if cur_sc < best_sc:
                    best_sc, best_vec = cur_sc, a.copy()
                no_improve = 0

        # early stop if reached target
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
