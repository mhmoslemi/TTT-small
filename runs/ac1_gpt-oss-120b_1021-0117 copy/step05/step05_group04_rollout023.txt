def _max_conv(v, L):
    fft = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fft*fft, n=L)
    # conv length L; relevant length 2*len(v)-1
    return float(conv[:2*len(v)-1].max())

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(1024, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    # normalize sum to 1
    total = cur.sum()
    if total == 0.0:
        cur = np.full_like(cur, 1.0/cur.size)
    else:
        cur /= total
    
    n = cur.size
    L = 1 << ((2*n-2).bit_length())  # next power of 2 >= 2n-1
    
    cur_max = _max_conv(cur, L)
    best_vec = cur.copy()
    best_max = cur_max
    best_score = 2.0 * n * best_max
    
    target_score = 1.50274349
    
    # SA parameters
    temp = cur_max * 0.5  # initial temperature half of current max
    temp_min = 1e-9
    decay = 0.9997
    alpha = 0.1  # fraction of mass moved per step
    
    iteration = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        iteration += 1
        
        # choose source index with probability favouring large entries
        if rng.random() < 0.7:
            i = int(np.argmax(cur))
        else:
            i = rng.integers(0, n)
        if cur[i] <= 1e-12:
            continue
        # choose destination different from i
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        
        # delta mass transfer
        delta = cur[i] * rng.random() * alpha
        if delta <= 0:
            continue
        
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure nonnegativity (should hold)
        # recompute max conv
        cand_max = _max_conv(cand, L)
        diff = cand_max - cur_max
        
        # acceptance
        accept = False
        if diff < 0:
            accept = True
        else:
            # metropolis
            prob = math.exp(-diff / (temp + 1e-12))
            if rng.random() < prob:
                accept = True

            cur = cand
            cur_max = cand_max
            no_improve = 0

                best_max = cur_max
                best_vec = cur.copy()
                best_score = 2.0 * n * best_max
                # early stop if target reached

import time
import math
import numpy as np

def _max_conv(v: np.ndarray, L: int) -> float:
    """Return max of the linear convolution v*v using FFT of length L."""
    fft = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fft * fft, n=L)
    # relevant part length = 2*len(v)-1
    return float(conv[: 2 * v.size - 1].max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Simple simulated‑annealing search that improves on the supplied parent.
    Returns a non‑negative list of coefficients.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin before the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise candidate – use the provided parent if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(1024, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s == 0.0:
        cur = np.full_like(cur, 1.0 / cur.size)
    else:
        cur /= s                       # normalise to sum = 1

    n = cur.size
    # FFT length – next power of two ≥ 2n‑1
    L = 1 << ((2 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Evaluation of the current state
    # ------------------------------------------------------------------
    cur_max = _max_conv(cur, L)
    best_vec = cur.copy()
    best_max = cur_max
    best_score = 2.0 * n * best_max
    target_score = 1.50274349

    # ------------------------------------------------------------------
    # Simulated‑annealing parameters
    # ------------------------------------------------------------------
    temp = cur_max * 0.5                # initial temperature
    temp_min = 1e-9
    decay = 0.9997                      # geometric decay per iteration
    alpha = 0.10                        # fraction of source mass moved
    it = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- propose a local mass‑transfer move -----------------------
        # bias source index towards the current largest element
        if rng.random() < 0.7:
            i = int(np.argmax(cur))
        else:
            i = rng.integers(0, n)
        if cur[i] <= 1e-12:
            continue

        # destination index, distinct from i
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # amount transferred (always ≤ cur[i])
        delta = cur[i] * rng.random() * alpha
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_max = _max_conv(cand, L)
        diff = cand_max - cur_max

        # ---- Metropolis acceptance ------------------------------------
        accept = False
        if diff < 0.0:
            accept = True
        else:
            prob = math.exp(-diff / (temp + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_max = cand_max
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_vec = cur.copy()
                best_score = 2.0 * n * best_max
                if best_score < target_score:
                    break
        else:
            no_improve += 1

        # ---- temperature schedule --------------------------------------
        temp = max(temp * decay, temp_min)

        # ---- occasional larger perturbation ---------------------------
        if it % 5000 == 0:
            i2 = rng.integers(0, n)
            j2 = rng.integers(0, n - 1)
            if j2 >= i2:
                j2 += 1
            big_delta = cur[i2] * rng.random() * 0.5
            if big_delta > 0.0:
                cand2 = cur.copy()
                cand2[i2] -= big_delta
                cand2[j2] += big_delta
                cand2_max = _max_conv(cand2, L)
                if cand2_max < best_max:
                    cur = cand2
                    cur_max = cand2_max
                    best_max = cand2_max
                    best_vec = cand2.copy()
                    best_score = 2.0 * n * best_max

        # ---- escape from long plateaus --------------------------------
        if no_improve > 20000:
            # Randomly remix the current best
            cur = best_vec.copy()
            for _ in range(5):
                src = rng.integers(0, n)
                if cur[src] <= 0.0:
                    continue
                dst = rng.integers(0, n - 1)
                if dst >= src:
                    dst += 1
                mass = cur[src] * rng.random() * 0.3
                cur[src] -= mass
                cur[dst] += mass
            cur_max = _max_conv(cur, L)
            no_improve = 0

        # ---- periodic renormalisation (numerical drift) ---------------
        if it % 10000 == 0:
            cur /= cur.sum()
            cur_max = _max_conv(cur, L)

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
