def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
        # obtain a fallback candidate (projected onto simplex)
        init_seq = np.asarray(height_sequence_1, dtype=np.float64).copy()
        init_seq = np.maximum(init_seq, 0.0)
        init_seq = _project_simplex(init_seq)
    else:
        n = 2000
        init_seq = rng.random(n, dtype=np.float64)
        init_seq = _project_simplex(init_seq)

    # Prepare FFT parameters
    L = 1 << ((3 * n - 2).bit_length())
    idx = (2 * n - 2) - np.arange(n)

    # Helper to compute bound
    def compute_bound(seq):
        a_fft = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return 2.0 * n * conv.max()

    # Initialize best
    cur = init_seq
    cur_sc = compute_bound(cur)
    best_sc = cur_sc
    best_vec = cur.copy()
    
    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05

    # Softmax temperature schedule
    eps0 = 1.0
    eps_decay = 0.9995
    eps_min = 1e-6

    # Counters
    it = 0
    no_improve = 0

    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # Compute convolution and softmax weights
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        # temperature
        eps = max(eps_min, eps0 * (eps_decay ** it))
        # compute softmax weights
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _compute_bound(seq: np.ndarray) -> float:
    """Return 2*n*max(convolve(seq,seq)) for a non‑negative sequence with sum=1."""
    n = seq.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    a_fft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine length and initialise a feasible start vector.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        cur = _project_simplex(cur)          # sum = 1, non‑negative
    else:
        n = 2000
        cur = rng.random(n, dtype=np.float64)
        cur = _project_simplex(cur)

    # ------------------------------------------------------------------
    # FFT planning (shared for all iterations)
    # ------------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())
    idx = (2 * n - 2) - np.arange(n)          # index map for gradient extraction

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    cur_sc = _compute_bound(cur)
    best_sc = cur_sc
    best_vec = cur.copy()

    # Adam optimiser state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.2

    # Soft‑max temperature schedule
    eps0, eps_decay, eps_min = 1.0, 0.9995, 1e-6

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # Current convolution and smooth max weights
        # --------------------------------------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # Gradient of the (soft) max
        # --------------------------------------------------------------
        w_rev = w[::-1]

        grad = 2.0 * grad_full[idx]                     # size n, real‑valued

    # Return the best candidate (as a plain Python list)

    return [float(x) for x in best_vec]
