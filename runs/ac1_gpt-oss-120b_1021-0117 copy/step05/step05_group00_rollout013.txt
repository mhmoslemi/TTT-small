def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1) # margin

    # load initial length n
    if 'height_sequence_1' in globals():
        a_init = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a_init = np.exp(-np.arange(n)/100.0)

    n = a_init.size
    # normalize to sum 1 using clip+norm
    a_init = np.maximum(a_init, 0.0)
    total = a_init.sum()
    if total <= 0:
        a_init = np.full(n, 1.0/n)
    else:
        a_init = a_init / total

        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())

import time, math, numpy as np

# ------------------------------------------------------------
# Simple projection onto the non‑negative simplex (clip & renorm)
# ------------------------------------------------------------
def _clip_and_normalize(v: np.ndarray) -> np.ndarray:
    """Set negatives to 0 and renormalise to sum 1 (fallback uniform)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ------------------------------------------------------------
# Score for a simplex‑normalised vector (exact AC1 bound)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(convolution) assuming sum(a)=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
