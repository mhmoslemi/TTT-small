import numpy as np, time, math

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1,n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    return np.maximum(v - theta, 0.0)

def _score_from_conv(conv):
    # assuming sum=1
    n = (len(conv) + 1) // 2  # not needed
    return float(2.0 * n * conv.max())

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a, b, L):
    """Linear convolution via FFT, zero‑padded to length L."""
    af = np.fft.rfft(a, n=L)
    bf = np.fft.rfft(b, n=L)
    return np.fft.irfft(af * bf, n=L)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) that tries to
    minimise 2·n·max(conv(a,a))/sum(a)^2.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------ #
    # initialise a
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 1024

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)                     # sum(a) == 1
    n = a.shape[0]
    # size for zero‑padded FFT (must cover 3·n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.2

    conv = _conv_fft(a, a, L)[: 2 * n - 1]

    best_a = a.copy()

    it = 0

        it += 1
