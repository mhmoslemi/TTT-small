def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import time, numpy as np, math

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0,float(budget_s)-0.5)

    # Acquire initial vector
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 1024
        a = rng.random(n0, dtype=np.float64)
    a = np.maximum(a,0)
    # Project onto simplex sum=1
    a = _project_simplex(a)

    n = a.shape[0]
    # FFT length power of two >= 3n
    L = 1 << ((3*n - 2).bit_length())

    # Precompute indices for gradient extraction
    idx = (2*n - 2) - np.arange(n)

    # initial compute
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    best_score = 2.0 * n * conv.max()
    best_vec = a.copy()
    cur_score = best_score

    # hyperparameters
    step0 = 1.0               # base step magnitude
    beta_smoothing = 0.2      # initial smoothing parameter for softmax (beta)
    beta_factor = 1.01        # increase factor per iteration (or per 500 iter)
    max_beta = 10.0           # limit smoothing
    momentum_factor = 0.9
    momentum = np.zeros_like(a)

    it = 0
    sym_period = 200
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # Compute FFT and conv
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_conv = conv.max()
        # compute score
        cur_score = 2.0 * n * max_conv

        # Compute smoothed weights
        beta = min(max_beta, beta_smoothing * (beta_factor ** (it // 200)))  # increase every 200 iter maybe
        # softmax: w = exp(beta*conv)/sum
        # to avoid overflow, subtract max conv
        scaled = beta * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)  # shape length 2n-1
        w = exp_scaled / exp_scaled.sum()

        # Compute gradient using w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[idx]   # gradient of weighted sum

        # Update momentum
        momentum = momentum_factor * momentum + (1 - momentum_factor) * g

        # Step size schedule: maybe inversely sqrt
        alpha = step0 / math.sqrt(it)

        # Proposed new a
        a_new = a - alpha * momentum

        # Project onto simplex to enforce nonnegative sum=1
        a_new = _project_simplex(a_new)

        # Optionally symmetrize
        if it % sym_period == 0:
            a_sym = (a_new + a_new[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            # compute its score using fast method
            a_sym_fft = np.fft.rfft(a_sym, n=L)
            conv_sym_full = np.fft.irfft(a_sym_fft * a_sym_fft, n=L)
            conv_sym = conv_sym_full[:2*n-1]
            score_sym = 2.0 * n * conv_sym.max()
            if score_sym < cur_score:
                a_new = a_sym
                cur_score = score_sym

        # Evaluate if improvement
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Random perturbation if stuck
        if no_improve > 2000:
            # randomly transfer mass
            i = rng.integers(0, n)
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.2
                j = rng.integers(0, n)
                if j != i:
                    a_tmp = a.copy()
                    a_tmp[i] -= delta
                    a_tmp[j] += delta
                    a_tmp = _project_simplex(a_tmp)
                    # compute score quickly
                    a_tmp_fft = np.fft.rfft(a_tmp, n=L)
                    conv_tmp_full = np.fft.irfft(a_tmp_fft * a_tmp_fft, n=L)
                    conv_tmp = conv_tmp_full[:2*n-1]
                    score_tmp = 2.0 * n * conv_tmp.max()
                    if score_tmp < cur_score:
                        a_new = a_tmp
                        cur_score = score_tmp
            no_improve = 0

        a = a_new

        # early exit
        if best_score < 1.50274349:
            break

    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Guard against the pathological all‑zero case
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    return w
