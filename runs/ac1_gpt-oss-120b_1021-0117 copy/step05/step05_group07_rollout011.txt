import time, math, numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex (sum = 1, x_i >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Compute AC1 upper bound for a sequence that already sums to 1
# ------------------------------------------------------------
def _ac1_score(a: np.ndarray, L: int, n: int) -> float:
    """Return 2*n*max(conv) for a (assumed sum(a)=1)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max()

        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _ac1_score(cand, L, n)
        if cand_sc < best_sc:
            best_sc, best_a = cand_sc, cand.copy()

    # ------------------------------------------------------------
    # Return the best candidate (as a plain Python list)
    # ------------------------------------------------------------
    return [float(x) for x in best_a]
