import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast computation of 2 * n * max(conv(a, a)) assuming sum(a)=1
# ------------------------------------------------------------
def _score_fft(a: np.ndarray, L: int, n: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑empty non‑negative sequence whose AC1 bound is as low
    as possible within the time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # --------------------------------------------------------
    # Initialise
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = np.full(1024, 1.0 / 1024, dtype=np.float64)

    cur = np.clip(cur, 0.0, 1000.0)
    cur = _project_simplex(cur)               # enforce sum = 1
    n = cur.size

    # --------------------------------------------------------
    # FFT planning
    # --------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())       # power of two ≥ 2n‑1

    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score_fft(best_vec, L, n)
    cur_sc = best_sc

    # --------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------
    step0 = 0.9                # base step for momentum descent
    step0_mirror = 0.6         # base step for mirror descent
    eps0 = 0.08                # initial soft‑max temperature
    eps_decay = 0.9996
    eps_min = 1e-7
    beta = 0.85                # momentum decay
    momentum = np.zeros_like(cur)
    no_improve = 0
    it = 0
    target = 1.50274349

    # --------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution & soft‑max weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----

        # ----- candidate 1 : momentum step with back‑tracking -----
        alpha = step0 / math.sqrt(it)
        cand1 = cur - alpha * momentum
        cand1 = _project_simplex(cand1)
        cand1_sc = _score_fft(cand1, L, n)
        if cand1_sc > cur_sc:

                cand_tmp = cur - alpha * momentum
                cand_tmp = _project_simplex(cand_tmp)
                tmp_sc = _score_fft(cand_tmp, L, n)
                if tmp_sc <= cur_sc:

        eta = step0_mirror / math.sqrt(it)

        cand2_sc = _score_fft(cand2, L, n)

        # ----- periodic symmetrisation (helps with convolution symmetry) -----

        # ----- random mass‑transfer when stuck -----

                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_fft(tmp, L, n)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- safety check to leave a buffer for returning -----
        if time.monotonic() > deadline - 0.05:
            break

    # --------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # --------------------------------------------------
    return [float(x) for x in best_vec]
