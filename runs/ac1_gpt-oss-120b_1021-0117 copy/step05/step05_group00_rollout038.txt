import time, math, numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 score for a vector with sum == 1 (returns 2*n*max(conv))
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # -------------------- initialise --------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / a.size)
            s = 1.0
        a = a / s
    else:
        n0 = 2000
        t = np.arange(n0, dtype=np.float64)
        a = np.exp(-t / 100.0)
        a = a / a.sum()

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # length for FFTs

    cur = a.copy()
    cur_sc = _score(cur, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # -------------------- optimisation parameters --------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0
    no_improve = 0
    last_best_it = 0
    lr = 0.5                     # base learning rate for Adam
    sym_interval = 300
    improve_target = 1.50274349
    # epsilon (smoothing) schedule parameters
    eps_start_factor = 0.5
    eps_decay = 0.0005

    while time.monotonic() < deadline:
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        # ----- convolution & maximum -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- smoothing weight (soft‑max) -----
        eps = max(max_val * max(eps_start_factor * math.exp(-eps_decay * it), 1e-12), 1e-12)
        w = np.exp(np.clip((conv_full - max_val) / eps, -50, 0))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # extract needed n entries
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_sc = _score(cand, L)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                last_best_it = it
                no_improve = 0
        else:
            # occasional uphill move (simulated annealing flavour)
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                    last_best_it = it
                    no_improve = 0

        # ----- stagnation handling -----
        if it - last_best_it > 2000:
            # random perturbation centred on the current best
            cur = best_vec + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta2_pow = 1.0
            last_best_it = it

        if it - last_best_it > 6000:
            # full restart from a fresh Dirichlet sample
            cur = rng.dirichlet(np.full(n, 0.5))
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta2_pow = 1.0
            last_best_it = it

        # ----- early exit if target achieved -----
        if best_sc < improve_target:
            break

    return [float(x) for x in best_vec]
