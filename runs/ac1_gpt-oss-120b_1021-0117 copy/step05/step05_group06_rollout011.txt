def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.2   # leave margin
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2500
        a = rng.random(n, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.size
    L = 1 << ((2*n-1).bit_length())
    # initial best
    best = a.copy()
    best_sc = _max_conv(best, L)
    best_bound = 2.0 * n * best_sc
    cur = a.copy()
    cur_sc = best_sc
    # gradient descent parameters
    eps0 = 0.1
    eps_decay = 0.9995
    alpha0 = 1.0
    beta = 0.9
    momentum = np.zeros_like(cur)
    it = 0
    while time.monotonic() < deadline - 30.0:
        it += 1
        # compute convolution and max
        cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)[:2*n-1]
        max_val = conv_full.max()
        # compute softmax weighting
        eps = max(1e-12, eps0 * (eps_decay ** it))
        if eps > 1e-5:
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
        else:
            mask = conv_full >= max_val - 1e-12
            cnt = mask.sum()
            if cnt > 0:
                w = np.where(mask, 1.0/cnt, 0.0)
            else:
                w = np.full_like(conv_full, 1.0/conv_full.size)
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
        # indices to map to gradient for a: for each i, sum_{j} a_j w_{i+j}
        # retrieving gradient vector: for i in [0,n-1], gradient = 2 * conv of a and w_rev positioned appropriate length
        # The previous code derived using idx = (2*n-2) - np.arange(n). We'll adopt same.
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # momentum
        momentum = beta * momentum + (1 - beta) * g
        # step size schedule
        alpha = alpha0 / math.sqrt(it)
        # line search
        improved = False
        for _ in range(6):
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc < cur_sc - 1e-12:
                cur = cand
                cur_sc = cand_sc
                improved = True
                if cand_sc < best_sc:
                    best = cand.copy()
                    best_sc = cand_sc
                    best_bound = 2.0 * n * best_sc
                break
            alpha *= 0.5
        # if not improved, maybe reset momentum
        if not improved:
            momentum *= 0.0
        if best_bound < 1.50274349:
            break

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _max_conv(a: np.ndarray, L: int) -> float:
    """Maximum of the linear convolution a*a using zero‑padded FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)

    else:
        a = rng.random(2500, dtype=np.float64)

    n = a.size

    best_sc = _max_conv(best, L)          # max convolution (sum = 1)

    # ---- gradient descent -------------------------------------------

    eps_decay = 0.9995

                cur_bound = 2.0 * n * cur_sc
            else:
                cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]  # revert

    return [float(x) for x in best]
