def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.2 (some margin)
    # load initial candidate:
    if height_sequence_1 defined:
        base = np.array(height_sequence_1, dtype=float)
    else:
        n = 2000; base = np.exp(-np.arange(n)/100)
    base = np.clip(base, 0, np.inf)
    # We'll have function to normalize to sum=1
    def normalize(v):
        v = np.maximum(v, 0)
        s = v.sum()
        if s <= 0:
            return np.full_like(v, 1.0/v.size)
        return v / s

    cur = normalize(base)

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    # Score function using FFT
    def score(v):
        # compute full conv via FFT
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())

    cur_sc = score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # parameters
    max_no_improve = 2000
    no_improve = 0

    # Adam hyper-params
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    alpha_adam = 0.05  # learning rate; will tune

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t_iter = 0

    # smoothing parameter epsilon for softmax
    eps_smooth = None
    eps_decay = 0.99  # more aggressive

    # symmetrization: every sym_freq iterations
    sym_freq = 200

    # target bound
    target = 1.50274349

    while time.monotonic() < deadline:
        t_iter += 1
        # compute conv and gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()
        # define epsilon
        if eps_smooth is None:
            eps_smooth = max(max_val * 0.2, 1e-9)
        else:
            eps_smooth = max(eps_smooth * eps_decay, 1e-9)
        # compute softmax weights
        w = np.exp((conv_full - max_val) / eps_smooth)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient via convolution with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # extract gradient for each index j:
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t_iter)
        v_hat = v / (1 - beta2**t_iter)
        step = alpha_adam * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = normalize(cand)

        cand_sc = score(cand)
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            no_improve += 1

        # symmetrization
        if t_iter % sym_freq == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = normalize(sym)
            sym_sc = score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # random mass transfer if stuck
        if no_improve > 1000:
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = normalize(tmp)
                tmp_sc = score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    # accept with probability
                    prob = np.exp(-(tmp_sc - cur_sc) / (max(1e-3, step.mean())+1e-12))
                    if rng.random() < prob:
                        cur = tmp
                        cur_sc = tmp_sc
                        no_improve = 0

        # restart if too long no improvement
        if no_improve > 5000:
            # reinitialize from best with small noise
            cur = best_vec + 0.01 * rng.random(n)
            cur = normalize(cur)
            m[:] = 0.0
            v[:] = 0.0
            eps_smooth = None
            no_improve = 0

        # early stop
        if best_sc < target:
            break

    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    # Shift by the smallest value to avoid negative issues
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Score (2*n*max(convolution)), assumes sum == 1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Core optimisation loop for a single initial vector
# ------------------------------------------------------------
def _optimize_one(init: np.ndarray, L: int, deadline: float,
                   target: float, rng: np.random.Generator) -> tuple:
    """Run Adam‑based optimisation from `init`. Returns (best_vec, best_sc)."""
    n = init.size
    cur = init.copy()
    cur_sc = _score(cur, L)

    best_vec = cur.copy()
    best_sc = cur_sc

    # Adam state
    m = np.zeros_like(cur)
    v2 = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    alpha_adam = 0.05

    eps_smooth = None
    eps_decay = 0.99

    no_improve = 0
    it = 0
    sym_freq = 250

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & smoothing weight -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps_smooth is None:
            eps_smooth = max(max_val * 0.2, 1e-9)
        else:
            eps_smooth = max(eps_smooth * eps_decay, 1e-9)

        w = np.exp((conv_full - max_val) / eps_smooth)
        w_sum = w.sum()
        if w_sum == 0.0:

        else:
