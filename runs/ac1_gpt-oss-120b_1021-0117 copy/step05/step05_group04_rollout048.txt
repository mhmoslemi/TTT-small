def _proj_simplex(v):
    # v is ndarray, returns projection onto simplex {x: sum x = 1, x >= 0}
    # Implementation from https:
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho = max{i | u_i - cssv_i / i > 0}
    cond = u - cssv / ind > 0
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

        tol = max(eps_tol * cur_max, 1e-15)
        w_active = (conv >= cur_max - tol).astype(np.float64)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive; return a unit vector
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, L: int):
    """Return (max(conv), rfft(a), full linear convolution) using FFT."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _gradient_from_weights(a_fft: np.ndarray, w: np.ndarray,
                          n: int, L: int, idx_slice: np.ndarray) -> np.ndarray:
    """
    Compute gradient of the weighted max‑convolution.
    g_i = 2 * (a * w_rev)_ {(2n‑2)‑i}
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    return 2.0 * grad_full[idx_slice]


def _active_weights(conv: np.ndarray, cur_max: float) -> np.ndarray:
    """
    Return a weight vector w of length len(conv) that distributes unit
    weight equally over all indices where conv is within a tiny tolerance
    of the maximum.
    """
    # Absolute tolerance avoids issues when max is very small
    tol = max(1e-15, cur_max * 1e-12)
    mask = conv >= cur_max - tol
    count = np.count_nonzero(mask)
    if count == 0:
        # fallback – put all weight on the argmax
        w = np.zeros_like(conv)
        w[conv.argmax()] = 1.0
        return w
    w = np.empty_like(conv, dtype=np.float64)
    w.fill(0.0)
    w[mask] = 1.0 / count
    return w

    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin
