def _project_simplex(v):
    # Project onto simplex sum=1, v>=0
    # Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero?
        theta = 0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _project_simplex(v):
    # ensure non-negative sum=1
    # If v sums to exactly 1 and all non-negative, return v.
    # Otherwise do projection.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)

import time, math, numpy as np

# ------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ------------------------------------------------------------
def _conv_max_and_fft(a: np.ndarray, L: int):
    """Return max of a*a (full convolution), its FFT and the full conv array."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return float(conv.max()), a_fft, conv


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int):
    """
    Gradient of the smooth‑max objective.
    g_i = 2 * (a * w_rev)_{(2n‑2)-i}.
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]


def _project_simplex(v: np.ndarray):
    """Projection of v onto the probability simplex {x≥0, sum x = 1}."""
    # Duchi et al. 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise (sum may be off by ~1e-12)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients whose AC1 bound
    is (hopefully) as low as possible.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # margin for safety

    # ------------------------------------------------------------------
    # Initial vector – use provided parent if available, else a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
    else:
        # modest default length when no parent is supplied
        n_default = 1500
        a = rng.random(n_default, dtype=np.float64)
    # project onto simplex (sum = 1, non‑negative)
    a = _project_simplex(a)

    n = a.size
    # FFT length sufficient for a (size n) convolved with a vector of length 2n‑1
    L = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2

    # ------------------------------------------------------------
    # Optimisation hyper‑parameters (matrix‑free)
    # ------------------------------------------------------------
    tau = 0.5                 # starting temperature for soft‑max
    tau_decay = 0.9995
    tau_min = 1e-12

    lr = 0.05                 # base learning rate for Adam
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    step_clip = 0.2          # clip absolute step size to keep updates stable

    m = np.zeros_like(a)     # first moment estimate
    v = np.zeros_like(a)     # second moment estimate
    beta1_pow = 1.0
    beta2_pow = 1.0

    sym_interval = 500       # how often to enforce symmetry
    no_improve = 0           # iterations since best improvement

    # ------------------------------------------------------------
    # Record the best solution seen so far
    # ------------------------------------------------------------
    cur_max, a_fft, conv = _conv_max_and_fft(a, L)
    best_max = cur_max
    best_vec = a.copy()

    it = 0
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------
        # 1) Soft‑max weights over the current convolution
        # --------------------------------------------------------
        eps = max(tau_min, tau * (tau_decay ** it))
        # shift by current max for numerical stability
        exp_arg = (conv - cur_max) / eps
        exp_arg = np.clip(exp_arg, -50.0, 0.0)   # values are ≤0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------
        # 2) Gradient of the smoothed objective (matrix‑free via FFT)
        # --------------------------------------------------------
        grad = _gradient(a_fft, w, n, L)

        # --------------------------------------------------------
        # 3) Adam update on the simplex‑projected variables
        # --------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        step = np.clip(step, -step_clip, step_clip)

        a = a - step
        a = _project_simplex(a)

        # --------------------------------------------------------
        # 4) Evaluate the new point (will be used in the next loop)
        # --------------------------------------------------------
        cur_max, a_fft, conv = _conv_max_and_fft(a, L)

        # record best solution
        if cur_max < best_max - 1e-15:
            best_max = cur_max
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # --------------------------------------------------------
        # 5) Periodic symmetrisation (helps due to Toeplitz symmetry)
        # --------------------------------------------------------
        if it % sym_interval == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_max, _, _ = _conv_max_and_fft(sym, L)
            if sym_max < cur_max:
                a = sym
                cur_max = sym_max
                a_fft, _, _ = _conv_max_and_fft(a, L)  # recompute for next iter
                if sym_max < best_max:
                    best_max = sym_max
                    best_vec = sym.copy()
                    no_improve = 0

        # --------------------------------------------------------
        # 6) Random perturbation if stuck for a long while
        # --------------------------------------------------------
        if no_improve > 5000:
            noise = rng.normal(scale=0.01, size=n)
            a = a + noise
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)
            no_improve = 0
            # fresh FFT for the perturbed vector
            cur_max, a_fft, conv = _conv_max_and_fft(a, L)
            if cur_max < best_max:
                best_max = cur_max
                best_vec = a.copy()

        # early exit if we have already beaten the target
        if 2.0 * n * best_max < 1.50274349:
            break

    # ------------------------------------------------------------
    # Final safety evaluation (in case the loop ended before
    # the latest candidate was scored)
    # ------------------------------------------------------------
    final_max, _, _ = _conv_max_and_fft(a, L)
    if final_max < best_max:
        best_vec = a.copy()
        best_max = final_max

    # ------------------------------------------------------------
    # Return the best vector found as a plain Python list
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
