def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2) # safety

    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
    else:
        n = 1024
        a0 = rng.random(n, dtype=np.float64)

    # Normalize sum to 1
    a0_sum = a0.sum()
    if a0_sum <= 0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / a0_sum

    n = a0.size
    L = 1 << ((2*n - 1).bit_length())  # next power of two >= 2n-1

    # Initialize logits u = log(a) + small noise
    eps_log = 1e-30
    u = np.log(a0 + eps_log)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    # Temperature schedule for smoothed max
    eps_temp0 = 0.5  # initial smoothing
    eps_decay = 0.9995
    eps_temp_min = 1e-6

    # Tracking best
    best_a = a0.copy()
    best_score = float('inf')
    best_conv_max = None

    # Compute initial conv
    conv = _conv_via_fft(best_a, L)
    best_conv_max = conv.max()
    best_score = 2.0 * n * best_conv_max

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # compute current a each iteration from logits
        # softmax to get probabilities
        # could use stable softmax: subtract max
        u_max = np.max(u)
        exp_u = np.exp(u - u_max)
        a = exp_u / np.sum(exp_u)

        # optional symmetrize each iteration
        a = (a + a[::-1]) * 0.5
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # compute convolution
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        cur_max = conv.max()

        # compute score
        score = 2.0 * n * cur_max

        # keep best
        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_conv_max = cur_max

        # stop if target achieved
        if best_score < 1.50274349:
            break

        # temperature for softmax
        eps_temp = max(eps_temp_min, eps_temp0 * (eps_decay ** iteration))

        # compute softmax weights over conv
        # w = softmax((conv - cur_max)/eps_temp)
        # To avoid overflow, compute exp((conv - cur_max)/eps)
        w = np.exp((conv - cur_max) / eps_temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # compute gradient wrt a
        # using conv of a with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad_a = 2.0 * grad_full[idx]

        # gradient wrt u
        g_dot_a = np.dot(grad_a, a)
        grad_u = a * (grad_a - g_dot_a)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u * grad_u)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        lr = 0.05  # maybe a base learning rate
        u -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional random perturbation when stuck
        if iteration % 500 == 0:
            # small random normal noise added to u
            noise = rng.normal(scale=0.1, size=n)
            u += noise

        # keep loop in time

    # after loop, output best_a
    return [float(x) for x in best_a]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ------------------------------------------------------------
def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector (sum=1)."""
    m = np.max(logits)
    e = np.exp(logits - m)
    s = e.sum()
    if s == 0.0:
        # fallback uniform
        return np.full_like(e, 1.0 / e.size, dtype=np.float64)
    return e / s


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT of length L."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv


def _softmax_weights(conv: np.ndarray, cur_max: float, eps: float) -> np.ndarray:
    """Weights approximating the max via a soft‑max with temperature eps."""
    # exponent ≤ 0, safe from overflow
    w = np.exp((conv - cur_max) / eps)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size, dtype=np.float64)
    return w / s


def _gradient_conv(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution w.r.t. a.
    Returns vector of length n.
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)          # slice yielding length‑n vector
    return 2.0 * grad_full[idx]
