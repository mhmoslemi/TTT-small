        it += 1

        # compute conv and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)

def _project_simplex(v):
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[:2*n-1]
    return float(2.0 * n * conv.max())

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initialise

    else:
        # fallback random length ~2000
        a0 = rng.random(2000, dtype=np.float64)
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)   # sum = 1
    cur = a0.copy()
    n = cur.size
    L = 1 << ((3 * n - 2).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)

    # hyper‑parameters
    step0 = 0.7
    mom_coeff = 0.9
    smooth_beta = 1.0
    beta_max = 200.0
    beta_factor = 1.001
    target = 1.50274349
    no_improve = 0
    it = 0
    sym_period = 300
    stagnation_limit = 3000

    while time.monotonic() < deadline:
        it += 1

        # convolution of cur with itself
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)

        w_sum = w_raw.sum()

        else:

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
