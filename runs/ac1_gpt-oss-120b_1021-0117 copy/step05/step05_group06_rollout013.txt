import time, math, numpy as np

def _project_simplex(v):
    # as before
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _next_pow2(x):
    return 1 << (x-1).bit_length()

def _max_conv(a, L):
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    # length is 2*n -1, we only need first 2*n-1
    return float(conv_full[:2*a.size-1].max())

    eps_min = 1e-8
    it = 0
    patience = 0
    target = 1.50274349

    # --------------------------------------------------------------
    # Main gradient‑descent loop (soft‑max smoothing)
    # --------------------------------------------------------------
    while time.monotonic() < deadline - 2.0:
        it += 1

        # convolution and its maximum
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weights (annealed)
        eps = max(eps_min, eps_start * (eps_decay ** it))
        if eps > 1e-6:
            w = np.exp((conv - max_val) / eps)
            s = w.sum()
            w = np.full_like(w, 1.0 / w.size) if s == 0.0 else w / s
        else:
            mask = conv >= max_val - 1e-12
            cnt = int(mask.sum())
            w = np.full_like(conv, 1.0 / conv.size) if cnt == 0 else np.where(mask, 1.0 / cnt, 0.0)

        # gradient of smoothed objective:
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # back‑tracking line search
        alpha = step0 / math.sqrt(it)
        improved = False
        for _ in range(5):
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc <= cur_sc:
                cur = cand
                cur_sc = cand_sc
                improved = True
                if cand_sc < best_sc:
                    best = cand.copy()
                    best_sc = cand_sc
                    best_bound = 2.0 * n * best_sc
                    patience = 0
                break
            alpha *= 0.5
        if not improved:
            patience += 1

        # random perturbation if stuck
        if patience >= 200:
            cur += rng.normal(scale=0.02, size=n)
            cur = _project_simplex(np.clip(cur, 0.0, 1.0))
            cur_sc = _max_conv(cur, L)
            patience = 0

        # occasional symmetrisation
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _max_conv(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best = sym.copy()
                    best_sc = sym_sc
                    best_bound = 2.0 * n * best_sc

        # early exit if target met
        if best_bound < target:
            break

    # --------------------------------------------------------------
    # Deterministic anti‑identity permutation (large opposite small)
    # --------------------------------------------------------------
    vals = np.sort(best)[::-1]
    perm = np.empty_like(vals)
    left, right = 0, n - 1
    k = 0
    while left <= right:
        perm[left] = vals[k]; k += 1
        if left < right:
            perm[right] = vals[k]; k += 1
        left += 1
        right -= 1
    perm_sc = _max_conv(perm, L)
    perm_bound = 2.0 * n * perm_sc
    if perm_bound < best_bound:
        best = perm.copy()
        best_sc = perm_sc
        best_bound = perm_bound

    # --------------------------------------------------------------
    # Simple hill‑climbing swaps (limited by remaining time)
    # --------------------------------------------------------------
    cur = best.copy()
    cur_bound = best_bound
    while time.monotonic() < deadline - 0.2:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_sc = _max_conv(cur, L)
        new_bound = 2.0 * n * new_sc
        if new_bound < cur_bound:
            cur_bound = new_bound
            best = cur.copy()
            best_sc = new_sc
            best_bound = new_bound
            if best_bound < target:
                break
        else:
            cur[i], cur[j] = cur[j], cur[i]

    # --------------------------------------------------------------
    # Optional mild smoothing as a final polish
    # --------------------------------------------------------------
    k = 5
    kernel = np.ones(k) / k
    smooth = np.convolve(best, kernel, mode='same')
    smooth = _project_simplex(np.clip(smooth, 0.0, 1.0))
    smooth_sc = _max_conv(smooth, L)
    if smooth_sc < best_sc:
        best = smooth
        best_sc = smooth_sc
        best_bound = 2.0 * n * best_sc

    # --------------------------------------------------------------
    # Return best candidate as plain Python list
    # --------------------------------------------------------------
    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best]
