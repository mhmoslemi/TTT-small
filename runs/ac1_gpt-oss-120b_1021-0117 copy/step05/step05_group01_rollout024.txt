# ------------------------------------------------------------
# Simplex projection (Euclidean) : {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex of size z (default 1)."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Exact sub‑gradient of max_{k} (a*a)[k]  (convolution w/o reversal)
# ------------------------------------------------------------
def _subgradient(a: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the maximal convolution entry at index `shift`."""
    n = a.shape[0]
    g = np.zeros_like(a)
    start = max(0, shift - (n - 1))
    end = min(n - 1, shift)
    # indices of a that contribute to this shift
    idx = shift - np.arange(start, end + 1)
    g[start:end + 1] = 2.0 * a[idx]
    return g
