import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ------------------------------------------------------------
def _self_conv_max(a: np.ndarray, L: int):
    """Return max of a*a (linear convolution) together with FFT data."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int):
    """
    Subgradient of the max‑convolution objective.
    w should be a mask (or weights) of length 2*n‑1.
    Returns an n‑vector gradient.
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]


# ------------------------------------------------------------

    if cur_sum == 0.0:

    else:
        cur = cur / cur_sum

    n = cur.size
