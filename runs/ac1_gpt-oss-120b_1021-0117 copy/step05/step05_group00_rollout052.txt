def _optimize(a0, deadline):
    cur = a0.copy()
    best = cur.copy()
    best_sc = _score(best)
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0
    epsilon_smooth = None
    while time.monotonic() < deadline:
        t_iter += 1
        # compute conv
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()
        # smoothing epsilon schedule
        if epsilon_smooth is None:
            epsilon_smooth = max_val * 0.2
        else:
            epsilon_smooth = max(epsilon_smooth * 0.995, 1e-9)
        # compute weights w
        w = np.exp((conv_full - max_val) / epsilon_smooth)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]  # gradient of smoothed objective
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t_iter)
        v_hat = v / (1 - beta2**t_iter)
        lr = 0.5 / np.sqrt(t_iter)  # decaying lr; scaling issue
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cur = cur - step
        # enforce non-negativity via projection onto simplex
        cur = _project_simplex(cur)
        # enforce symmetry
        cur = (cur + cur[::-1]) * 0.5
        cur = _project_simplex(cur)
        # compute score
        sc = _score(cur)
        if sc < best_sc:
            best_sc = sc
            best = cur.copy()
        # early exit if sc < target
        if best_sc < improve_target:
            break
        # periodical random perturbation if stagnation
        if t_iter % 500 == 0:
            # random mass transfer
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.2

        if time.monotonic() > deadline - 0.1:

    return best, best_sc

def _optimize(a0, deadline, L, target, rng):
    cur = a0.copy()
    n = cur.size
    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    eps_smooth = None
    best = cur.copy()
    best_sc = _score(cur)
    it = 0
    no_improve = 0
    # time per iteration check
    while time.monotonic() < deadline:
        it += 1
        # FFT of cur
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        if eps_smooth is None:
            eps_smooth = max_val * 0.2
        else:
            eps_smooth = max(eps_smooth * 0.995, 1e-12)
        # softmax weights
        exps = np.exp((conv_full - max_val) / eps_smooth)
        w_sum = exps.sum()
        if w_sum == 0.0:
            w = np.full_like(exps, 1.0 / exps.size)
        else:
            w = exps / w_sum
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        lr = 0.5 / (np.sqrt(it) + 1.0)
        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        cur = cur - step
        # projection onto simplex
        cur = _project_simplex(cur)
        # enforce symmetry
        cur = (cur + cur[::-1]) * 0.5
        cur = _project_simplex(cur)
        # score
        sc = _score(cur)
        if sc < best_sc:
            best_sc = sc
            best = cur.copy()
            no_improve = 0
        else:
            no_improve += 1
        # occasional random mass transfer if stuck
        if no_improve >= 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n)

                    j = (j + 1) % n
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)

                if sc_tmp < best_sc:

                    best = tmp.copy()
                if sc_tmp < sc:
                    cur = tmp
                no_improve = 0
        # early stop if target reached
        if best_sc < target:
            break
        # safety break if time near deadline
        if time.monotonic() > deadline - 0.05:

import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# AC1 bound for vectors with sum≈1 (exact evaluator uses the same)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution) for a vector with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Core optimizer (Adam + symmetry + simplex projection)
# ------------------------------------------------------------
def _optimize(a0: np.ndarray, deadline: float, L: int, target: float,
              rng: np.random.Generator) -> tuple[np.ndarray, float]:
    cur = a0.copy()
    n = cur.size
    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8
    beta1_pow, beta2_pow = 1.0, 1.0
    eps_smooth = None
    best = cur.copy()
    best_sc = _score(cur)
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # ---- convolution and smoothing weight ----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        if eps_smooth is None:
            eps_smooth = max_val * 0.2
        else:
            eps_smooth = max(eps_smooth * 0.995, 1e-12)

        exps = np.exp((conv_full - max_val) / eps_smooth)
        w_sum = exps.sum()
        if w_sum == 0.0:
            w = np.full_like(exps, 1.0 / exps.size)
        else:
            w = exps / w_sum

        # ---- gradient of the smoothed max ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ---- Adam update ----
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        lr = 0.5 / (math.sqrt(it) + 1.0)
        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        cur = cur - step
        cur = _project_simplex(cur)

        # ---- enforce symmetry ----
        cur = (cur + cur[::-1]) * 0.5
        cur = _project_simplex(cur)

        # ---- score & bookkeeping ----
        sc = _score(cur)
        if sc < best_sc:
            best_sc = sc
            best = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ---- random mass transfer when stuck ----
        if no_improve >= 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n)
                if j == i:
                    j = (j + 1) % n
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                sc_tmp = _score(tmp)
                if sc_tmp < best_sc:
                    best_sc = sc_tmp
                    best = tmp.copy()
                if sc_tmp < sc:
                    cur = tmp
                no_improve = 0

        # ---- early exit on target ----
        if best_sc < target:
            break

        # ---- keep some time for the harness ----
        if time.monotonic() > deadline - 0.05:
            break

    return best, best_sc
