import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # Keep a small safety margin for final conversion/return
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random positive vector of moderate length
        n = 2048
        a0 = rng.random(n, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    cur = _project_simplex(a0)            # enforce sum = 1
    n = cur.size

    # ------------------------------------------------------------
    # FFT length – must accommodate a (size n) * w (size 2n‑1)
    # ------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2

    # ------------------------------------------------------------
    # Optimisation state
    # ------------------------------------------------------------
    best_vec = cur.copy()
    # initial hard score (exact objective, because sum=1)
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    best_sc = float(2.0 * n * conv.max())

    # Adam parameters
    lr = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # Soft‑max (smooth‑max) temperature schedule
    beta_temp = 5.0          # start smoothness
    beta_factor = 1.005
    beta_max = 500.0

    # Misc. control
    sym_period = 500
    no_improve = 0
    target = 1.50274349
    it = 0

    # ------------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- compute convolution of cur with itself -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]

        # ----- exact hard score (used for best‑so‑far) -----
        cur_sc = float(2.0 * n * conv.max())
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- update smooth‑max temperature -----
        if beta_temp < beta_max:
            beta_temp *= beta_factor

        # ----- soft‑max weights (smooth approximation of the max) -----
        maxc = conv.max()
        # subtract maxc for numerical stability
        w = np.exp(beta_temp * (conv - maxc))
        w /= w.sum()                       # now w sums to 1

        # ----- gradient of the smooth‑max loss -----
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- take a step and project back onto the simplex -----
        cur = _project_simplex(cur - step)

        # ----- periodic symmetrisation (helps with the Toeplitz structure) -----
        if it % sym_period == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            # evaluate symmetrised candidate
            a_fft_sym = np.fft.rfft(sym, n=L)
            conv_sym = np.fft.irfft(a_fft_sym * a_fft_sym, n=L)[: 2 * n - 1]
            sym_sc = float(2.0 * n * conv_sym.max())
            if sym_sc < cur_sc:
                cur = sym
                # reset Adam moments to avoid stale momentum
                m.fill(0.0)
                v.fill(0.0)
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = cur.copy()
                    no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 5000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                cur[i] -= delta
                cur[j] += delta
                cur = _project_simplex(cur)
                # reset optimiser state after perturbation
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
