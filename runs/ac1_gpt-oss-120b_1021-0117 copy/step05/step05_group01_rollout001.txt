def _project_simplex(v, Z=1.0):
    # if sum(v) == Z and all >=0: return v
    # else compute
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _grad_topk(a, conv, top_k):
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k-1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            i = np.arange(i_start, i_end+1)
            g[i] += 2.0 * a[k - i]
    g /= max(1, top_k)
    return g

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    # fast path
    if np.all(v >= 0) and abs(v.sum() - Z) < 1e-12:
        return v
    u = np.sort(v)[::-1]                     # descending sort
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Maximum of the linear convolution a*a (full length) via FFT
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max_k (a*a)[k] using an FFT of length L (>=2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ------------------------------------------------------------
# Approximate sub‑gradient from the top‑k largest convolution entries
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        if i0 <= i1:
            i = np.arange(i0, i1 + 1, dtype=int)
            g[i] += 2.0 * a[k - i]
    g /= max(1, top_k)
    return g

    else:

    # FFT length (next power of two >= 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Optimisation state
    # ------------------------------------------------------------------
    best_vec = a.copy()
    best_sc = _conv_max(a, L)                # = score because sum fixed
    m = np.zeros_like(a)                     # first moment (Adam)
    v = np.zeros_like(a)                     # second moment (Adam)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    top_k = 10                               # start with 10 active constraints

    target_score = 1.50274349
