import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Score for a normalized (sum=1) non‑negative vector
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        # ----- acceptance (allow occasional uphill moves) -----
        if cand_sc <= cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- keep the best ever seen -----
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # --------------------------------------------------------------
    # Return the best vector found as plain Python list
    # --------------------------------------------------------------
    return [float(x) for x in best_vec]
