# ------------------------------------------------------------
# Projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum Z."""
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast max‑convolution via FFT (full linear convolution)
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max_k (a*a)[k] using an FFT of length L (>=2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ------------------------------------------------------------
# Sub‑gradient using the top‑k largest convolution entries
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            i = np.arange(i_start, i_end + 1, dtype=int)
            g[i] += 2.0 * a[k - i]
    g /= max(1, top_k)
    return g

# ------------------------------------------------------------
# Build the (full) convolution matrix for a given f (sparse)
# ------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    """Toeplitz‑like matrix T such that (T @ g) = f * g (full linear conv)."""
    n = f.shape[0]
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)                     # 0 … n‑1
    T = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return T

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_vec = a.copy()
    cur_sc = 2.0 * n * _conv_max(a, L)
    best_sc = cur_sc

    # optimisation parameters
    step0 = 0.7                         # base step magnitude
    beta = 0.90                         # momentum factor
    momentum = np.zeros_like(a)
    top_k = 8                           # number of largest correlations for gradient
    sym_interval = 250                  # how often to symmetrise
    no_improve = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        cur_max = conv_full.max()
        cur_sc = 2.0 * n * cur_max

        # ----- keep the best -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit -----
        if best_sc < target:
            break

        # ----- gradient step (projected SGD) -----
        g = _grad_topk(a, conv_full, top_k)
        momentum = beta * momentum + (1.0 - beta) * g
        alpha = step0 / math.sqrt(it)    # simple decaying step
        a_tmp = a - alpha * momentum
        a_tmp = _project_simplex(a_tmp)   # keep on simplex

        # ----- evaluate candidate -----
        cand_sc = 2.0 * n * _conv_max(a_tmp, L)

        # ----- accept if improvement -----
        if cand_sc < cur_sc:
            a = a_tmp
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, a_tmp.copy()
        else:
            # occasional uphill move (Metropolis‑style)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                a = a_tmp
                cur_sc = cand_sc

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            sym_sc = 2.0 * n * _conv_max(a_sym, L)
            if sym_sc < cur_sc:
                a, cur_sc = a_sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    no_improve = 0

        # ----- occasional random mass‑transfer when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

        # ----- occasional LP refinement (cheap, few iterations) -----
        # Trigger only when we have at least some time left.
        if it % 1500 == 0 and (deadline - time.monotonic()) > 5.0:
            M = _conv_max(a, L)
            # Build sparse convolution matrix for the current a
            A = _build_conv_matrix(a)
            g_var = cp.Variable(n, nonneg=True)
            prob_lp = cp.Problem(cp.Maximize(cp.sum(g_var)),
                                 [A @ g_var <= M + 1e-12])
            try:
                prob_lp.solve(solver="ECOS", warm_start=True, max_iters=2000)
            except Exception:  # solver failure – skip this refinement
                continue
            if g_var.value is not None:
                g_opt = np.maximum(g_var.value, 0.0)
                # Mix a and g_opt
                t = 0.2
                a_mix = (1 - t) * a + t * g_opt
                a_mix = _project_simplex(a_mix)
                mix_sc = 2.0 * n * _conv_max(a_mix, L)
                if mix_sc < cur_sc:
                    a, cur_sc = a_mix, mix_sc
                    if mix_sc < best_sc:
                        best_sc, best_vec = mix_sc, a_mix.copy()
                        no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
