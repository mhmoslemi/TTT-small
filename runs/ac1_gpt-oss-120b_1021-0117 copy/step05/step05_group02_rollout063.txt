import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Scoring – same formula used by the evaluator
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a, a))."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------

        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- candidate 1 : momentum step with back‑tracking -----
        alpha = step0 / math.sqrt(it)
        cand1 = cur - alpha * momentum
        cand1 = _project_simplex(cand1)
        cand1_sc = _score(cand1)

        if cand1_sc > cur_sc:            # back‑track if not improving
            tmp_alpha = alpha
            for _ in range(8):
                tmp_alpha *= 0.5
                cand_tmp = cur - tmp_alpha * momentum
                cand_tmp = _project_simplex(cand_tmp)
                tmp_sc = _score(cand_tmp)
                if tmp_sc <= cur_sc:
                    cand1, cand1_sc = cand_tmp, tmp_sc
                    break
            else:
                cand1, cand1_sc = cand_tmp, tmp_sc

        # ----- candidate 2 : mirror‑descent step -----
        eta = step0_mirror / math.sqrt(it)
        exp_arg = -eta * g
        np.clip(exp_arg, -50, 50, out=exp_arg)      # avoid overflow
        cand2 = cur * np.exp(exp_arg)
        s = cand2.sum()
        if s > 0:
            cand2 /= s
        else:
            cand2 = cur.copy()
        cand2_sc = _score(cand2)

        # ----- candidate 3 : random multiplicative perturbation -----
        sigma = 0.1 / math.sqrt(it + 1)
        log_cur = np.log(cur + 1e-15)
        cand3_log = log_cur + rng.normal(scale=sigma, size=n)
        cand3 = np.exp(cand3_log)
        cand3 = _project_simplex(cand3)
        cand3_sc = _score(cand3)

        # ----- pick the best among candidates (including current) -----
        candidates = [(cur, cur_sc), (cand1, cand1_sc),
                      (cand2, cand2_sc), (cand3, cand3_sc)]
        cand, cand_sc = min(candidates, key=lambda p: p[1])

        # ----- accept / reject (occasionally uphill moves) -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            # small Metropolis‑like acceptance probability
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps with Toeplitz structure) -----
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 1500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------
    # Final fine‑tuning: small random swaps while time permits
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or best_vec[i] == 0.0:
            continue
        delta = best_vec[i] * 0.01 * rng.random()
        tmp = best_vec.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < best_sc:
            best_vec, best_sc = tmp, tmp_sc

    # --------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # --------------------------------------------------
    return [float(x) for x in best_vec]
