import time

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non-negative, sum=1)."""
    # based on algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _max_conv(a: np.ndarray, L: int) -> float:
    """Maximum value of the linear convolution a*a (full length) using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    # keep the first 2*len(a)-1 entries (linear convolution)
    conv = conv_full[: 2 * a.size - 1]
    return float(conv.max())

def _max_conv(a: np.ndarray, L: int) -> float:
    """Maximum value of the linear convolution a * a (full length) using FFT."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * a.size - 1]
    return float(conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence whose AC1 upper bound
    (2·n·max(conv)/sum²) is as low as possible.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)          # enforce sum = 1, non‑negative
    n = a.size

    # FFT length – power of two not smaller than 2 n‑1
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    cur = a.copy()
    cur_sc = _max_conv(cur, L)
    best = cur.copy()
    best_sc = cur_sc
    best_bound = 2.0 * n * best_sc

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    beta = 0.9               # momentum factor
    momentum = np.zeros_like(cur)
    step0 = 1.0              # initial step‑size coefficient
    eps = 1.0                # temperature for soft‑max weights
    eps_min = 1e-12
    idx = (2 * n - 2) - np.arange(n)   # indices to pick gradient entries
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Gradient‑descent phase (matrix‑free)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline - 2.0:
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature schedule (geometric decay)
        eps = max(eps_min, eps * 0.99)

        # construct sub‑gradient weights
        if eps > 1e-8:
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            w = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)
        else:
            mask = conv >= max_val - 1e-14
            cnt = int(mask.sum())
            if cnt == 0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = np.where(mask, 1.0 / cnt, 0.0)

        # gradient of the (smoothed) objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # back‑tracking line search
        alpha = step0 / math.sqrt(it)
        improved = False
        for _ in range(8):
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc <= cur_sc:
                cur = cand
                cur_sc = cand_sc
                improved = True
                if cand_sc < best_sc:
                    best = cand.copy()
                    best_sc = cand_sc
                    best_bound = 2.0 * n * best_sc
                break
            alpha *= 0.5

        if not improved:
            pass  # keep current point

        # occasional symmetrisation (helps with the Toeplitz structure)
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _max_conv(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best = sym.copy()
                    best_sc = sym_sc
                    best_bound = 2.0 * n * best_sc

        if best_bound < target:
            break

    # ------------------------------------------------------------------
    # Deterministic anti‑identity permutation (large values opposite small)
    # ------------------------------------------------------------------
    vals = np.sort(best)[::-1]          # descending order
    perm = np.empty_like(vals)
    left, right = 0, n - 1
    k = 0
    while left <= right:
        perm[left] = vals[k]; k += 1
        if left < right:
            perm[right] = vals[k]; k += 1
        left += 1
        right -= 1
    perm = _project_simplex(perm)       # safety re‑projection
    perm_sc = _max_conv(perm, L)
    perm_bound = 2.0 * n * perm_sc
    if perm_bound < best_bound:
        best = perm.copy()
        best_sc = perm_sc
        best_bound = perm_bound

    # ------------------------------------------------------------------
    # Simple hill‑climbing swaps to polish the permutation
    # ------------------------------------------------------------------
    cur = best.copy()
    cur_bound = best_bound
    while time.monotonic() < deadline - 0.2:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_sc = _max_conv(cur, L)
        new_bound = 2.0 * n * new_sc
        if new_bound < cur_bound:
            cur_bound = new_bound
            best = cur.copy()
            best_sc = new_sc
            best_bound = new_bound
            if best_bound < target:
                break
        else:
            cur[i], cur[j] = cur[j], cur[i]   # revert

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best]
