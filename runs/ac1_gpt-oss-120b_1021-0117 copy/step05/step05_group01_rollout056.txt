import time
import math
import numpy as np
import scipy.signal as signal

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Approximate sub‑gradient of the weighted max‑convolution
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    if top_k > 0:
        g /= float(top_k)
    return g

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (approximately) minimises the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)                 # enforce sum = 1

    n = a.size
    # FFT length for a*a (full convolution)
    L = 1 << ((2 * n - 1).bit_length())

    # ----- optimisation parameters ----------------------------------------
    beta1, beta2, eps = 0.9, 0.999, 1e-8               # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.3                                      # base learning rate
    start_alpha = 5.0                                  # soft‑max sharpness
    blend = 0.5                                        # blend soft‑max / top‑k
    top_k = 5
    target = 1.50274349

    # ----- bookkeeping -----------------------------------------------------
    best_vec = a.copy()
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    best_sc = float(2 * n * conv.max())
    no_improve = 0
    it = 0

    # ------------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution and score
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_conv = conv.max()
        cur_sc = float(2 * n * max_conv)

        # keep the best solution seen so far
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_sc < target:
            break

        # ---------- gradient (soft‑max part) ----------
        alpha = start_alpha * (1.0 + 0.0004 * it)         # slowly increase sharpness
        exp_weights = np.exp(alpha * (conv - max_conv))   # stable soft‑max
        w = exp_weights / exp_weights.sum()

        # correlation of a (first) with w (second) gives ∑_l w_{l+i} a_l
        corr = signal.correlate(a, w, mode='full')
        offset = len(w) - 1                               # lag 0 position
        g_soft = 2.0 * corr[offset: offset + n]

        # ---------- gradient (top‑k part) ----------
        g_top = _grad_topk(a, conv, top_k)

        # blend the two gradient estimates
        g = blend * g_soft + (1.0 - blend) * g_top

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # enforce symmetry (helps Toeplitz structure)
        a_new = (a_new + a_new[::-1]) * 0.5
        a_new = _project_simplex(a_new)

        a = a_new

        # occasional stochastic mass‑transfer when stagnating
        if no_improve > 4000:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.35
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)
            no_improve = 0

    # ------------------------------------------------------------
    # Final local refinement (tiny random moves) if time permits
    # ------------------------------------------------------------
    while time.monotonic() < deadline - 0.1:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if best_vec[i] <= 0.0:
            continue
        delta = best_vec[i] * rng.random() * 0.07
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        # quick feasibility check
        if np.min(cand) < 0:
            continue
        # evaluate
        cand_fft = np.fft.rfft(cand, n=L)
        conv_cand = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_sc = float(2 * n * conv_cand.max())
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()

    # ------------------------------------------------------------
    # Return the best candidate found
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
