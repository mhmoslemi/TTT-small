import time
import numpy as np

# Try to import cvxpy; if unavailable we will fall back to a simple heuristic.
try:
    import cvxpy as cp
    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def _evaluate(seq: np.ndarray, n: int, total: float) -> float:
    """Fast computation of the AC1 bound (identical to evaluate_sequence)."""
    max_conv = np.convolve(seq, seq).max()
    return 2.0 * n * max_conv / (total * total)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a short safety margin for final conversion & return
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64) * 0.1  # small random start

    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:                      # enforce the evaluator's lower bound
        a += (0.01 - a.sum()) / a.size

    n = a.size
    target_sum = a.sum()

    # ------------------------------------------------------------------
    # Keep the best solution seen so far
    # ------------------------------------------------------------------
    best_vec = a.copy()
    best_score = evaluate_sequence(best_vec.tolist())

    # ------------------------------------------------------------------
    # Set up the LP (if possible) that maximises sum(g) under
    #   conv(f, g) <= M   with f fixed and M = max(conv(f,f))
    # ------------------------------------------------------------------
    have_cvx = _CVX_AVAILABLE
    if have_cvx:
        try:
            g = cp.Variable(n)
            f_param = cp.Parameter(n, nonneg=True)
            M_param = cp.Parameter(nonneg=True)
            conv_expr = cp.conv(f_param, g)                 # length 2n‑1
            prob = cp.Problem(cp.Maximize(cp.sum(g)),
                              [g >= 0, conv_expr <= M_param])
        except Exception:                                 # pragma: no cover
            have_cvx = False

    # schedule of mixing coefficients (larger at the start)
    mix_sched = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]
    it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop – matrix‑free, O(n log n) per LP solve,
    # with a few cheap convolution checks for candidate mixes.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # current bound
        cur_score = _evaluate(a, n, target_sum)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            if best_score < 1.50274349:   # target reached
                break

        if not have_cvx:
            # Simple random mass‑transfer fallback when cvxpy is unavailable.
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                a[i] -= delta
                a[j] += delta
                a = a * (target_sum / a.sum())
            it += 1
            continue

        # ----- solve the LP for the current f -----
        M_current = np.convolve(a, a).max()
        f_param.value = a
        M_param.value = M_current
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                       verbose=False)
        except Exception:                                 # pragma: no cover
            break
        if prob.status not in ("optimal", "optimal_inaccurate"):
            break
        g_opt = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        if g_opt.sum() == 0.0:
            break
        # Scale g_opt so its sum equals the target sum (preserves the denominator)
        g_opt = g_opt * (target_sum / g_opt.sum())

        # ----- try several convex combinations and keep the best -----
        best_local = a
        best_local_M = M_current
        for t in mix_sched:
            cand = (1.0 - t) * a + t * g_opt
            cand = np.clip(cand, 0.0, 1000.0)
            cand = cand * (target_sum / cand.sum())
            cand_M = np.convolve(cand, cand).max()
            if cand_M < best_local_M:
                best_local_M = cand_M
                best_local = cand
        a = best_local
        it += 1

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
