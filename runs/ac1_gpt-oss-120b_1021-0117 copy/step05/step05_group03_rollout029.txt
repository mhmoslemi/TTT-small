import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection of a vector onto the probability simplex.
# Guarantees non‑negative entries that sum to 1.
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex {x >= 0, sum x = 1}."""
    if v.size == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Core entry point required by the harness.
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients that (approximately)
    minimise the AC1 upper bound evaluated by `evaluate_sequence`.
    The routine prefers a provided parent construction if it yields a lower
    bound than a simple uniform distribution.
    """
    start = time.monotonic()
    # keep a tiny margin for final conversion
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------- determine length ------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = parent.size
    else:
        # fallback length when no parent is supplied
        n = 1024
        parent = None

    # ------------------- uniform candidate -----------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------- evaluate candidates ----------------------
    # evaluation is cheap (O(n log n) via FFT inside `evaluate_sequence`);
    # we call it only for the few candidates we consider.
    best_vec = uniform
    best_score = evaluate_sequence(best_vec.tolist())

    if parent is not None:
        # enforce non‑negativity (the evaluator also clips, but we keep it clean)
        parent = np.maximum(parent, 0.0)
        # optional normalisation – scaling does not affect the bound,
        # but projecting to the simplex keeps numbers well‑behaved.
        s = parent.sum()
        if s > 0:
            parent = _project_simplex(parent)
        parent_score = evaluate_sequence(parent.tolist())
        if parent_score < best_score:
            best_vec = parent
            best_score = parent_score

    # ------------------- very short local refinement -------------
    # A couple of cheap stochastic moves are attempted only if time permits.
    rng = np.random.default_rng(seed)
    cur = best_vec.copy()
    cur_score = best_score
    it = 0
    while time.monotonic() < deadline and it < 300:

        # generate a small random perturbation and re‑project
        noise = rng.normal(scale=0.01, size=n)
        cand = cur + noise
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_vec = cand.copy()
                best_score = cand_score

    # ensure we return a plain Python list of floats
    return [float(x) for x in best_vec]
