def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale vector to have sum = 1 (non‑negative)."""
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _self_conv_max(a: np.ndarray, L: int):
    """
    Return (max of a*a (linear convolution), FFT of a, full convolution).
    Convolution length = 2*len(a)-1, taken from the zero‑padded FFT.
    """
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _softmax_weights(conv: np.ndarray, max_val: float, eps: float) -> np.ndarray:
    """Soft‑max on convolution values (temperature = eps)."""
    w = np.exp((conv - max_val) / eps)  # ≤1, never overflows
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size, dtype=np.float64)
    return w / s


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution objective.
    g_i = 2 * (a * w_rev)_{(2n‑2)-i}.

    idx = (2 * n - 2) - np.arange(n)          # slice of length‑n vector
