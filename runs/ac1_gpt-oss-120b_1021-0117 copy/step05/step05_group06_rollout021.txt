def subgradient(a, L):
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    # mask of maxima
    mask = conv >= max_val - 1e-12
    cnt = mask.sum()
    if cnt == 0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = np.where(mask, 1.0 / cnt, 0.0)
    # reverse w
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g, max_val
```

def expgrad_update(a, grad, eta):
    # a positive
    factor = np.exp(-eta * grad)  # shape (n,)
    a_new = a * factor
    # renorm to sum=1
    a_new_sum = a_new.sum()
    if a_new_sum == 0:

        a_new = np.full_like(a, 1.0 / a.size)
    else:
        a_new = a_new / a_new_sum
    return a_new

def _project_simplex(v: np.ndarray) -> np.ndarray:
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:

    T0 = cur_bound * 0.05                  # initial temperature (≈5 % of bound)
    T_min = 1e-12
    transfer_scale = 0.2                   # fraction of source height to move
    iter_cnt = 0

    while True:
        now = time.monotonic()
        if now >= deadline - 0.3:
            break

        # decide move type: swap (≈70 %) or transfer (≈30 %)
        if rng.random() < 0.7:
            # ---- swap two positions ------------------------------------------------
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            cur[i], cur[j] = cur[j], cur[i]
        else:
            # ---- transfer a fraction from i to j ------------------------------------
            i = rng.integers(0, n)
            if cur[i] == 0.0:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = rng.random() * cur[i] * transfer_scale
            cur[i] -= delta
            cur[j] += delta

        # evaluate
        new_max = _max_conv(cur, L)
        new_bound = 2.0 * n * new_max / (cur.sum() ** 2)

        # temperature schedule (linear cooling)
        prog = (now - start) / (deadline - start)
        T = max(T_min, T0 * (1.0 - prog))

        # acceptance test
        if new_bound < cur_bound or rng.random() < math.exp(-(new_bound - cur_bound) / T):
            cur_bound = new_bound
            if new_bound < best_bound:
                best_bound = new_bound
                best = cur.copy()
                if best_bound < target:
                    break
        else:
            # revert move
            if rng.random() < 0.7:          # same branch as used for the move
                cur[i], cur[j] = cur[j], cur[i]
            else:
                cur[i] += delta
                cur[j] -= delta

        iter_cnt += 1

    # --------------------------------------------------------------
    # Final greedy polishing (only accept improvements)
    # --------------------------------------------------------------
    cur = best.copy()
    cur_bound = best_bound
    while time.monotonic() < deadline - 0.15:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cur[i], cur[j] = cur[j], cur[i]
        new_max = _max_conv(cur, L)
        new_bound = 2.0 * n * new_max / (cur.sum() ** 2)
        if new_bound < cur_bound:
            cur_bound = new_bound
            if new_bound < best_bound:
                best_bound = new_bound
                best = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]

    # --------------------------------------------------------------
    # Optional brief projected‑gradient refinement (few iterations)
    # --------------------------------------------------------------
    while time.monotonic() < deadline - 0.05:
        g, max_val = _subgrad(best, L)
        step = 0.05
        cand = best - step * g
        cand = _project_simplex(cand)
        cand_max = _max_conv(cand, L)
        cand_bound = 2.0 * n * cand_max / (cand.sum() ** 2)
        if cand_bound < best_bound:
            best = cand
            best_bound = cand_bound
        else:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best]
