import time, math, numpy as np

def _project_simplex(v):
    # v: np.ndarray
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a, L):
    # a assumed normalized to sum=1
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())

def _grad(a, L):
    n = a.size
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2*n - 1]
    max_val = conv.max()
    mask = conv >= max_val - 1e-12
    w = mask.astype(np.float64)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val

    return np.maximum(v - theta, 0.0)


def _score(a, L):
    """Exact AC1 upper bound for a (assumes sum(a)=1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())


def _grad(a, L):
    """Sub‑gradient of the max‑convolution objective."""
    n = a.size
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    m = conv.max()
    mask = conv >= m - 1e-12          # ties at the maximum
    w = mask.astype(np.float64)
    s = w.sum()
    if s > 0:
        w /= s
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, m


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(1024, dtype=np.float64)   # fallback length

    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())          # power‑of‑two for FFT

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # optimisation parameters
    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.8
    it = 0
    stagnation = 0
    target = 1.50274349    # benchmark – stop early if reached

    # ------------------------------------------------------------------
    # main loop
    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # sub‑gradient at current point
        grad, mval = _grad(cur, L)
        cur_sc = 2.0 * n * mval          # exact score for current vector

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        # propose a new point
        cand = cur - step * momentum
        cand = _project_simplex(cand)
        cand_sc = _score(cand, L)

        # accept if improved; otherwise occasional downhill move
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            stagnation = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            stagnation += 1

        # every few hundred iterations enforce symmetry (helps autocorrelation)
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                stagnation = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # random redistribution when stuck for a while
        if stagnation > 3000 and time.monotonic() + 0.2 < deadline:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                stagnation = 0

        # early exit if we already beat the benchmark
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
