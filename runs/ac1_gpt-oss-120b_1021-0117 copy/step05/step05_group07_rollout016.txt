# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- soft‑max weights -----
        eps = max(eps_min, eps_start * (eps_decay ** it))
        if eps > 1e-12:
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum
        else:
            w = np.zeros_like(conv_full)
            mask = conv_full == max_val
            cnt = mask.sum()
            if cnt > 0:
                w[mask] = 1.0 / cnt

        # ----- gradient of the (soft) max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]               # size n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = (lr0 / (math.sqrt(it) + 1.0)) * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- candidate generation -----
        cand = cur - step
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_max = cand_conv.max()
        cand_sc = 2.0 * n * cand_max

        # ----- accept if improvement -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1

        # ----- occasional symmetrisation (helps with Toeplitz structure) -----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > max_no_improve:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)

                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0
            else:
                # full restart if the current vector is degenerate
                cur = rng.random(n, dtype=np.float64)
                cur = _project_simplex(cur)
                m.fill(0.0)
                v.fill(0.0)
                beta1_pow, beta2_pow = 1.0, 1.0
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
