import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative coefficient list with a low AC1 bound.
    The algorithm:
      * start from either the provided parent `height_sequence_1` or a uniform vector;
      * evaluate both and keep the better one;
      * perform a quick random‑simplex search (Dirichlet samples) while time permits.
    The final best candidate is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the final conversion/return
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------------
    # Base length and a deterministic uniform start vector
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        # enforce the evaluator's clamping rules
        base = np.clip(base, 0.0, 1000.0)
        total = base.sum()
        if total > 0.0:
            base = base / total                 # normalise to sum = 1
        n = base.size
    else:
        # fall back to a modest size when no parent is present
        n = 1024
        base = None

    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # ----------------------------------------------------------------------
    # Keep track of the best vector seen so far
    # ----------------------------------------------------------------------
    best_vec = uniform
    best_sc = float('inf')

    # evaluate uniform
    try:
        best_sc = evaluate_sequence(best_vec.tolist())
    except Exception:      # safety – should never happen
        best_sc = float('inf')

    # possibly improve with the supplied parent
    if base is not None:
        try:
            sc_parent = evaluate_sequence(base.tolist())
            if sc_parent < best_sc:
                best_sc = sc_parent
                best_vec = base.copy()
        except Exception:
            pass

    # ----------------------------------------------------------------------
    # Light‑weight random search on the probability simplex
    # ----------------------------------------------------------------------
    # Dirichlet(1,…,1) is equivalent to normalising iid exponentials;
    # a single call to rng.random then normalisation is cheaper.
    while time.monotonic() < deadline:
        # generate a random point on the simplex
        cand = rng.random(n, dtype=np.float64)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum
        # evaluate
        try:
            sc = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()
            # early stop if we already beat the historical target
            if best_sc < 1.50274349:
                break

    # ----------------------------------------------------------------------
    # Return the best candidate found (as plain Python floats)
    # ----------------------------------------------------------------------
    return [float(v) for v in best_vec]
