import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast max of a*a (linear convolution via FFT)
# ------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Gradient of the smooth max (log‑sum‑exp) of the autocorrelation
# ------------------------------------------------------------
def _smooth_grad(a: np.ndarray, L: int, eps: float, idx: np.ndarray):
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    max_val = conv_full.max()
    # softmax weights over the convolution entries
    w = np.exp((conv_full - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient = 2 * (a * w_rev) convolution
    w_rev = w[::-1]
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)[: 2 * a.size - 1]
    g = 2.0 * grad_full[idx]
    return g, max_val, conv_full

        # ----------------------------------------------------
        # Restart from best with small Gaussian noise if hopeless
        # ----------------------------------------------------
        if no_improve > noise_thresh:
            cur = best_vec + 0.001 * rng.standard_normal(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.2
            no_improve = 0

        # ----------------------------------------------------
        # Early exit if target reached
        # ----------------------------------------------------
        if best_sc < improve_target:
            break

    # Return as plain Python list
    return [float(x) for x in best_vec]
```
