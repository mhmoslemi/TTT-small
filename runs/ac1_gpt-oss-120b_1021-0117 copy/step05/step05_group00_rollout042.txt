def _project_simplex(v):
    # project onto simplex sum=1, v>=0
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1

        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s                                   # now sum = 1
    else:
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())                # length for FFTs

    cur = a0.copy()
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    cur_sc = float(2.0 * n * conv_full.max())
    best_vec = cur.copy()
    best_sc = cur_sc

    # ----- optimisation parameters -----
    m = np.zeros_like(cur)            # Adam first moment
    v = np.zeros_like(cur)            # Adam second moment
    beta1, beta2 = 0.9, 0.999
    epsadam = 1e-8
    it = 0
    no_improve = 0
    eps = None
    eps_decay = 0.996
    sym_interval = 400
    improve_target = 1.50274349
    max_trials = 5

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & smoothing weight -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps is None:
            eps = max(max_val * 0.2, 1e-9)
        else:
            eps = max(eps * eps_decay, 1e-9)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max (correlation) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = m_hat / (np.sqrt(v_hat) + epsadam)

        # normalise step magnitude
        step_norm = np.linalg.norm(step) + 1e-12
        alpha0 = 0.5 / step_norm

        # ----- backtracking line‑search -----
        improved = False
        cand = None
        cand_sc = None
        alpha = alpha0
        for _ in range(max_trials):
            cand_try = cur - alpha * step
            cand_try = _project_simplex(cand_try)
            a_fft_c = np.fft.rfft(cand_try, n=L)
            conv_c = np.fft.irfft(a_fft_c * a_fft_c, n=L)[: 2 * n - 1]
            cand_sc = float(2.0 * n * conv_c.max())
            if cand_sc < cur_sc:
                cand = cand_try
                cur_sc = cand_sc
                improved = True
                break
            alpha *= 0.5

        # ----- occasional uphill move -----
        if not improved:
            cand_try = cur - alpha * step
            cand_try = _project_simplex(cand_try)
            a_fft_c = np.fft.rfft(cand_try, n=L)
            conv_c = np.fft.irfft(a_fft_c * a_fft_c, n=L)[: 2 * n - 1]
            cand_sc = float(2.0 * n * conv_c.max())
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                cur_sc = cand_sc
                improved = True

        if improved:
            cur = cand
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_vec = cur.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            a_fft_sym = np.fft.rfft(sym, n=L)
            conv_sym = np.fft.irfft(a_fft_sym * a_fft_sym, n=L)[: 2 * n - 1]
            sym_sc = float(2.0 * n * conv_sym.max())
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                a_fft_tmp = np.fft.rfft(tmp, n=L)
                conv_tmp = np.fft.irfft(a_fft_tmp * a_fft_tmp, n=L)[: 2 * n - 1]
                tmp_sc = float(2.0 * n * conv_tmp.max())
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- restart from best with small noise if hopeless -----
        if no_improve > 5000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

    # return the best found vector as plain Python floats
    return [float(x) for x in best_vec]
