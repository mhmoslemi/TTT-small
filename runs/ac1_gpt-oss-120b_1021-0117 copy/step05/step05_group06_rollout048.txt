    step0 = 0.5
    beta = 0.9
    momentum = np.zeros_like(a)
    eps0 = 0.5
    eps_decay = 0.9995
    eps_min = 1e-9

    best_sc = _max_conv(a, L)
    cur = a.copy()
    cur_sc = best_sc

    it = 0
    target_bound = 1.50274349

    while time.monotonic() < deadline - 0.3:
        it += 1
        # compute convolution and gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)

        eps = max(eps_min, eps0 * (eps_decay ** it))

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (L1 onto {x>=0, sum=1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Maximum of linear (full) convolution a * a using FFT of length L
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max(conv_full) for a*a using zero‑padded FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())


# ----------------------------------------------------------------------
# Anti‑identity permutation (largest values to opposite ends)
# ----------------------------------------------------------------------
def _anti_identity(seq: np.ndarray) -> np.ndarray:
    vals = np.sort(seq)[::-1]
    n = seq.size
    out = np.empty_like(seq)
    left, right = 0, n - 1
    k = 0
    while left <= right:
        out[left] = vals[k]; k += 1
        if left < right:
            out[right] = vals[k]; k += 1
        left += 1
        right -= 1
    return out

                    best_sc = sym_sc

        # early exit if already good enough
        if 2.0 * n * best_sc < target:
            break

    return best, best_sc


# ----------------------------------------------------------------------
# Simple hill‑climbing swaps (random pair swaps)
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray, sc: float, bound: float,
                L: int, deadline: float, rng: np.random.Generator):
    n = seq.size
    cur = seq.copy()
    cur_sc = sc
    cur_bound = bound
    attempts = 0
    max_attempts = 20000
    while time.monotonic() < deadline - 0.2 and attempts < max_attempts:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_sc = _max_conv(cur, L)
        new_bound = 2.0 * n * new_sc
        if new_bound < cur_bound:
            cur_bound = new_bound
            cur_sc = new_sc
            seq = cur.copy()
            sc = new_sc
            bound = new_bound
            if bound < 1.50274349:
                break
        else:
            # revert
            cur[i], cur[j] = cur[j], cur[i]
        attempts += 1
    return seq, sc, bound


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initialise sequence length -----
    if "height_sequence_1" in globals():
        n = int(np.asarray(height_sequence_1).size)
    else:
        n = 2048

    # ----- FFT length (next power of two >= 2n‑1) -----
    L = 1 << ((2 * n - 1).bit_length())

    # ----- generate a few diverse starting points -----
    starts = []

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)
        starts.append(a)

    # uniform simplex point
    starts.append(np.full(n, 1.0 / n))

    # a few random Dirichlet points
    for _ in range(2):
        a = rng.random(n, dtype=np.float64)
        a = _project_simplex(a)
        starts.append(a)

    best_seq = None
    best_bound = np.inf

    # ----- optimisation for each start (early exit if good enough) -----
    for a0 in starts:
        if time.monotonic() >= deadline - 0.5:
            break

        # gradient descent
        seq, sc = _gd_opt(a0, L, deadline, rng)
        bound = 2.0 * n * sc

        # anti‑identity permutation (often beneficial)
        perm = _anti_identity(seq)
        perm_sc = _max_conv(perm, L)
        perm_bound = 2.0 * n * perm_sc
        if perm_bound < bound:
            seq, sc, bound = perm, perm_sc, perm_bound

        # hill‑climbing swaps
        seq, sc, bound = _hill_climb(seq, sc, bound, L, deadline, rng)

        if bound < best_bound:
            best_bound = bound
            best_seq = seq.copy()

        if best_bound < 1.50274349:
            break

    # ----- fallback (should never happen) -----
    if best_seq is None or best_seq.size == 0:
        best_seq = np.array([1.0])

    # ----- convert to plain Python list of floats -----
    return [float(x) for x in best_seq]
