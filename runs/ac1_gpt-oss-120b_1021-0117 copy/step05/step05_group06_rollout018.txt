def hill_climb_perm(vals, init_seq, L, deadline, target):
    best_seq = init_seq.copy()
    best_bound = compute_bound(best_seq, L)
    cur_seq = best_seq.copy()
    cur_bound = best_bound
    rng = np.random.default_rng()
    temp = 1.0
    it = 0
    while time.monotonic() < deadline:
        i, j = rng.integers(0, len(vals), size=2)
        if i==j: continue
        # swap in cur_seq
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_bound = compute_bound(cur_seq, L)
        accept = False
        if new_bound < cur_bound:
            accept = True
        else:
            # simulated annealing acceptance
            # compute probability
            if temp > 1e-8:
                prob = math.exp(-(new_bound - cur_bound) / temp)
                accept = rng.random() < prob
        if accept:
            cur_bound = new_bound
            if new_bound < best_bound:
                best_bound = new_bound
                best_seq = cur_seq.copy()
                if best_bound < target:
                    break
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        # update temperature
        it += 1
        if it % 5000 == 0:
            temp *= 0.9  # cool down
    return best_seq, best_bound
```

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper functions (top‑level, matrix‑free)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum = 1, v >= 0)."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _max_conv(a: np.ndarray, L: int) -> float:
    """Maximum value of the (linear) autocorrelation a*a using zero‑padded FFT."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return float(conv_full[: 2 * a.size - 1].max())


def _bound_from_seq(seq: np.ndarray, L: int, n: int) -> float:
    """Exact bound evaluated by the authoritative evaluator (sum == 1)."""
    return 2.0 * n * _max_conv(seq, L)


# --------------------------------------------------------------
# Main search routine
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the final conversion
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ----------------------------------------------------------
    # Initial sequence (use parent if present)
    # ----------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)                     # enforce sum = 1
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())        # power‑of‑two FFT length

    # ----------------------------------------------------------
    # Book‑keeping
    # ----------------------------------------------------------
    best_seq = a.copy()
    best_sc = _max_conv(best_seq, L)
    best_bound = 2.0 * n * best_sc
    cur = a.copy()
    cur_sc = best_sc

    # ----------------------------------------------------------
    # Gradient–descent phase (matrix‑free sub‑gradient)
    # ----------------------------------------------------------
    step0 = 0.8
    beta = 0.9
    momentum = np.zeros_like(a)
    eps0 = 0.5
    eps_decay = 0.99
    eps_min = 1e-7
    target_bound = 1.50274349
    it = 0

    # keep about 15 % of the time budget for this phase
    gd_deadline = deadline - (budget_s * 0.15)

    while time.monotonic() < gd_deadline:
        it += 1
        # current autocorrelation and its maximum
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weighting (approaches the sub‑gradient as eps → 0)
        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps > 1e-6:
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            mask = conv >= max_val - 1e-12
            cnt = int(mask.sum())
            w = np.where(mask, 1.0 / cnt, 0.0) if cnt else np.full_like(conv, 1.0 / conv.size)

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # back‑tracking line search
        alpha = step0 / math.sqrt(it + 1.0)
        improved = False
        for _ in range(6):
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc <= cur_sc:
                cur = cand
                cur_sc = cand_sc
                improved = True
                if cand_sc < best_sc:
                    best_seq = cand.copy()
                    best_sc = cand_sc
                    best_bound = 2.0 * n * best_sc
                break
            alpha *= 0.5

        if not improved:
            # no progress – keep the current point unchanged
            pass

        # occasional symmetrisation (helps the Toeplitz structure)
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _max_conv(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_seq = sym.copy()
                    best_sc = sym_sc
                    best_bound = 2.0 * n * best_sc

        if best_bound < target_bound:
            break

    # ----------------------------------------------------------
    # Deterministic alternating arrangement (large‑small alternation)
    # ----------------------------------------------------------
    vals = np.sort(best_seq)[::-1]                 # descending
    perm = np.empty_like(vals)
    left, right = 0, n - 1
    k = 0
    while left <= right:
        perm[left] = vals[k]; k += 1
        if left < right:
            perm[right] = vals[k]; k += 1
        left += 1
        right -= 1
    perm_sc = _max_conv(perm, L)
    perm_bound = 2.0 * n * perm_sc
    if perm_bound < best_bound:
        best_seq = perm.copy()
        best_sc = perm_sc
        best_bound = perm_bound

    # ----------------------------------------------------------
    # Stochastic hill‑climbing / simulated annealing on permutations
    # ----------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_bound = best_bound
    # start temperature at 1 % of the current bound
    temp = max(1e-8, best_bound * 0.01)
    swap_prob = 0.7            # probability of performing a simple swap
    it = 0
    # allocate the remaining time budget
    while time.monotonic() < deadline:
        it += 1
        # ---- propose a move -------------------------------------------------
        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
            revert = lambda: cur_seq.__setitem__(slice(i, i + 1), cur_seq[j])  # noqa: E731
        else:
            i = rng.integers(0, n - 1)
            j = rng.integers(i + 1, n)
            cur_seq[i:j + 1] = cur_seq[i:j + 1][::-1]
            revert = lambda: cur_seq.__setitem__(slice(i, j + 1), cur_seq[i:j + 1][::-1])  # noqa: E731

        new_bound = _bound_from_seq(cur_seq, L, n)
        delta = new_bound - cur_bound
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis acceptance
            if temp > 0 and rng.random() < math.exp(-delta / temp):
                accept = True

        if accept:
            cur_bound = new_bound
            if new_bound < best_bound:
                best_bound = new_bound
                best_seq = cur_seq.copy()
                if best_bound < target_bound:
                    break
        else:
            # revert the move
            revert()

        # ---- cooling schedule ------------------------------------------------
        if it % 1000 == 0:
            temp = max(1e-8, temp * 0.96)

    # ----------------------------------------------------------
    # Final optional symmetrisation (may improve the bound a tiny bit)
    # ----------------------------------------------------------
    sym = (best_seq + best_seq[::-1]) * 0.5
    sym = _project_simplex(sym)
    sym_sc = _max_conv(sym, L)
    sym_bound = 2.0 * n * sym_sc
    if sym_bound < best_bound:
        best_seq = sym.copy()
        best_bound = sym_bound

    # ----------------------------------------------------------
    # Return best candidate as a plain Python list
    # ----------------------------------------------------------
    return [float(x) for x in best_seq]
