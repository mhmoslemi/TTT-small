def compute_conv_max_and_grad(a, L=None):
    n = a.shape[0]
    if L is None:
        L = 1 << ((2*n-2).bit_length() + 1) # Actually need length >= 2n-1 for convolution.
    # Use FFT for conv
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    max_val = conv.max()
    k = conv.argmax()
    # gradient: for i such that 0 <= k-i < n, grad[i] = 2*a[k-i]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    # Vectorized: indices = np.arange(i_start, i_end+1)
    # grad[indices] = 2 * a[k - indices]
    idx = np.arange(i_start, i_end+1, dtype=int)
    grad[idx] = 2.0 * a[k - idx]
    return max_val, grad, conv, a_fft

def _project_simplex(v):
    # ensure non-negative and sum=1
    # standard projection
    v = np.asarray(v, dtype=np.float64)
    if v.sum() == 1 and np.all(v >= 0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale to have sum = 1 (non‑negative)."""
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex:
    {x ≥ 0, sum(x)=1}.
    """
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety
    w /= w.sum() if w.sum() > 0 else 1.0
    return w


def _conv_max_and_grad(a: np.ndarray, L: int):
    """
    Compute the linear convolution a*a, its maximum value,
    and the gradient of the (scalar) max‑entry w.r.t. a.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    k = int(conv.argmax())
    # gradient of conv[k] w.r.t. a_i is 2*a_{k-i} if the index exists
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    grad[idx] = 2.0 * a[k - idx]
    return max_val, grad, conv, a_fft

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative list of coefficients whose
    AC1 bound (2 n·max(a*a)/sum(a)^2) is as low as possible.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the sandbox
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initial vector – use the parent if it exists, else random.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():

        cur = np.maximum(cur, 0.0)
    else:
        n = 2048
        cur = rng.random(n, dtype=np.float64)

    cur = _normalize(cur)
    n = cur.shape[0]

    # --------------------------------------------------------
    # FFT length – next power of two ≥ 2n‑1
    # --------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Optimization parameters
    # --------------------------------------------------------
    step0 = 0.8            # base step size
    step_decay = 0.9995    # exponential decay
    momentum_beta = 0.75
    momentum = np.zeros_like(cur)
    best_vec = cur.copy()
    best_max, _, _, _ = _conv_max_and_grad(best_vec, L)
    best_score = 2.0 * n * best_max
    target_score = 1.50274349
    no_improve = 0
    it = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- current convolution and gradient ----
        cur_max, grad, conv, a_fft = _conv_max_and_grad(cur, L)

        # ---- momentum update ----
        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad

        # ---- step size (decaying) ----
        step = step0 * (step_decay ** it)
        if step < 1e-6:
            step = 1e-6

        # ---- candidate 1: exponentiated (multiplicative) update ----

        cand1 = _normalize(cand1)
