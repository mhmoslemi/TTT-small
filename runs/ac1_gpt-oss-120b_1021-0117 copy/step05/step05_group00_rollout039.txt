import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a) == 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of a smoothed max (soft‑max temperature = eps)
# ------------------------------------------------------------
def _grad_soft_max(a: np.ndarray, eps: float, L: int) -> np.ndarray:
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    c_max = conv.max()
    w = np.exp((conv - c_max) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g

# Main entry point
# ------------------------------------------------------------

    start = time.monotonic()
    # leave a tiny safety margin for the final return

    else:                               # fallback: smooth exponential

        t = np.arange(n_fallback, dtype=np.float64)

        a0 = a0 / a0.sum()

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())   # enough for a * w_rev convolution

    cur = a0.copy()
    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # ---------- Adam optimiser state ----------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 0.3

    # temperature for the soft‑max, will be decayed
    eps_temp = None
    temp_decay = 0.9995

    no_improve = 0
    sym_interval = 250          # how often to symmetrise

            eps_temp = max(conv_tmp.max() * 0.2, 1e-12)
