import time

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (non‑negative,
# sum = 1).  Returns a new array with the same shape as the input.
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weights (smooth approximation of the max) -----
        w = np.exp(temp * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = mom * momentum + (1.0 - mom) * grad

        # ----- step size (inverse‑sqrt decay) -----
        base_alpha = step0 / math.sqrt(it)

        # ----- line‑search over a few scaling factors -----
        cand = None
        cand_sc = None
        for factor in (1.0, 0.7, 0.5, 0.3, 0.1, 0.05):
            alpha = base_alpha * factor
            tmp = cur - alpha * momentum
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L)

            if cand_sc is None or tmp_sc < cand_sc:
                cand, cand_sc = tmp, tmp_sc

            if tmp_sc < cur_sc:
                break

        # ----- accept / reject (with occasional uphill move) -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (base_alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps with the Toeplitz structure) -----
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random perturbation if we have been stuck too long -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- increase the temperature for a tighter max approximation -----
        if temp < temp_max:
            temp = min(temp_max, temp + temp_inc)

        # ----- early exit if the target bound has been reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found as a plain Python list.
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
