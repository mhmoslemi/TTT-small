def _project_simplex(v):
    v = np.asarray(v)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

import time, math, numpy as np

# ------------------------------------------------------------------
# Euclidean projection onto the probability simplex (non‑negative, sum=1)
# ------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------------
# Helper: score for a vector with sum≈1 (2·n·max(conv))
# ------------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Initialise – parent if present, otherwise random
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.clip(a, 0.0, 1000.0)
        # optional mild smoothing of the parent (helps suppress spikes)
        if a.size >= 3:
            kernel = np.ones(3) / 3.0
            a = np.convolve(a, kernel, mode="same")
    else:
        n = 1024
        a = rng.random(n, dtype=np.float64)

    # enforce feasibility (non‑negative, sum = 1)
    a = _project_simplex(a)
    # enforce symmetry – the optimal Toeplitz pattern is typically symmetric
    a = (a + a[::-1]) * 0.5
    a = _project_simplex(a)

    n = a.shape[0]
    # FFT length sufficient for convolutions up to length 3·n‑2
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------

    target = 1.50274349

    idx = (2 * n - 2) - np.arange(n)   # indices for gradient extraction

                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate discovered
    # ------------------------------------------------------------------
