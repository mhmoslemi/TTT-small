import time
import math
import numpy as np

# ------------------------------------------------------------
# Helpers
# ------------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v>=0)."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray, L: int) -> float:
    """Fast evaluation of 2 * n * max(conv(a,a)) (assumes sum(a)=1)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    start = time.monotonic()
    # keep a small safety margin for post‑processing
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = rng.random(a.shape, dtype=np.float64)
        else:
            a /= s
    else:
        # fallback: random start of length ~1024
        n = 1024
        a = rng.random(n, dtype=np.float64)
    a = _project_simplex(a)          # enforce sum = 1
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # FFT length, power of two

    # ---- Adam parameters ------------------------------------------------
    lr = 0.2
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # ---- bookkeeping ----------------------------------------------------
    best_a = a.copy()
    best_sc = _score(best_a, L)
    cur_sc = best_sc
    no_improve = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------
    # Optimisation loop – matrix‑free, O(n log n) per iteration
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & max (current objective) -----------------
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_conv = conv.max()
        cur_sc = 2.0 * n * max_conv

        # ----- soft‑max weights on convolution (smooth max) ----------
        tau = max(1e-8, 0.2 * max_conv)        # temperature
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smooth max --------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update -------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        step = lr * m / (np.sqrt(v) + eps)
        a_new = a - step
        a_new = _project_simplex(a_new)

        # ----- evaluate candidate --------------------------------------
        sc_new = _score(a_new, L)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # ----- periodic symmetrisation --------------------------------
        if it % 500 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sc_sym = _score(sym, L)
            if sc_sym < best_sc:
                best_sc = sc_sym
                best_a = sym.copy()
                no_improve = 0
            a = sym

        # ----- occasional random perturbation when stuck -------------
        if no_improve > 3000:
            perturb = rng.normal(scale=0.02, size=n)
            a = a + perturb
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)
            no_improve = 0

        # ----- early exit --------------------------------------------
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best solution discovered (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_a]
