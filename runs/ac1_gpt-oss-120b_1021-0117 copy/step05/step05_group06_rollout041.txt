import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Project onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}. """
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Maximum value of the full convolution a * a (length 2n‑1) using FFT.
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max(conv_full) for a*a using zero‑padded FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())

# ----------------------------------------------------------------------
# Anti‑identity permutation: largest, smallest, 2nd‑largest, 2nd‑smallest, …
# ----------------------------------------------------------------------
def _anti_identity_perm(vals: np.ndarray) -> np.ndarray:
    """Given a 1‑D array, return the anti‑identity permuted copy."""
    n = vals.size
    out = np.empty_like(vals)
    left, right = 0, n - 1
    idx = 0
    while left <= right:
        out[left] = vals[idx]
        idx += 1
        if left < right:
            out[right] = vals[idx]
            idx += 1
        left += 1
        right -= 1
    return out
