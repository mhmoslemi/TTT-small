def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    from scipy.sparse import diags
    from scipy.optimize import linprog
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.5)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 2048
        a = rng.random(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    sum_a = a.sum()
    if sum_a == 0:
        a = np.full_like(a, 1.0 / a.size)
        sum_a = 1.0
    else:
        a = a / sum_a

    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    # functions defined locally for speed

    def conv_full(vec):
        fft = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(fft * fft, n=L)
        return conv[:2*n - 1]

    cur = a.copy()
    best = cur.copy()
    conv_cur = conv_full(cur)
    best_score = 2.0 * n * conv_cur.max()
    cur_score = best_score

    # Main Loop
    iter_count = 0
    t = 0.2  # mixing factor
    # Reduce mixing factor over time if needed

    while time.monotonic() < deadline:
        iter_count += 1
        # Compute current max convolution
        conv_cur = conv_full(cur)
        M = conv_cur.max()
        # Solve LP for g
        # Build Toeplitz matrix
        offsets = np.arange(n)
        diag_data = [np.full(n, cur[d]) for d in range(n)]
        A = diags(diag_data, offsets, shape=(2*n - 1, n), format='csr')
        b_ub = np.full(2*n - 1, M)
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not res.success or res.x is None:
            # fallback: break loop
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        g = g0 / sum_g0  # normalize to sum 1

        # mix
        new_cur = (1 - t) * cur + t * g
        # enforce nonnegative (should hold)
        new_cur = np.maximum(new_cur, 0.0)
        # renormalize (sum may be 1 due to mixing)
        sum_new = new_cur.sum()
        if sum_new == 0:
            continue
        new_cur = new_cur / sum_new

        # compute score
        conv_new = conv_full(new_cur)
        new_score = 2.0 * n * conv_new.max()

        if new_score < best_score:
            best = new_cur.copy()
            best_score = new_score

        # accept if improved else maybe occasional acceptance
        if new_score <= cur_score:
            cur = new_cur
            cur_score = new_score
        else:
            # metropolis to allow occasional uphill if improvement
            # compute delta
            delta = new_score - cur_score
            # step size based gamma maybe
            prob = math.exp(-delta / (M + 1e-12))  # not great
            if rng.random() < prob:
                cur = new_cur
                cur_score = new_score

        # adapt t gradually
        if iter_count % 10 == 0:
            # optionally reduce t
            t = max(0.01, t * 0.98)

        # occasional symmetrization
        if iter_count % 20 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = sym / sym.sum()
            conv_sym = conv_full(sym)
            sym_score = 2.0 * n * conv_sym.max()
            if sym_score < best_score:
                best = sym.copy()
                best_score = sym_score
                cur = sym
                cur_score = sym_score

        # early exit if below target
        if best_score < 1.50274349:
            break

    # Return best as list
    return [float(x) for x in best]

import time
import math
import numpy as np
from scipy.sparse import diags
from scipy.optimize import linprog

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and scale to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback uniform
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _conv_full(v: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution of v using FFT length L (returns length 2·n‑1)."""
    fft = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fft * fft, n=L)
    return conv[: 2 * v.size - 1]


def _solve_lp(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max 1·b  s.t.  a * b ≤ M   and b ≥ 0.
    Returns the optimal b (not normalized) or None on failure.
    """
    n = a.size
    # Toeplitz matrix with constant diagonals = a[d] on offset d (row‑col = d)
    offsets = np.arange(n, dtype=np.int32)           # 0 … n‑1
    diag_data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    A = diags(diag_data, offsets, shape=(2 * n - 1, n), format="csr")
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)               # minimise –sum(b)  ⇒ maximise sum(b)
    bounds = [(0.0, None)] * n
    # use the high‑performance HiGHS interior point solver
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if res.success and res.x is not None:
        return res.x
    return None

    """Enforce reflection symmetry and renormalise."""
    sym = (v + v[::-1]) * 0.5
