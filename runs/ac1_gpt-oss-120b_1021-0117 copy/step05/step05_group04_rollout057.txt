import time, math, numpy as np

def _normalize(v):
    s = v.sum()
    if s == 0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

def _conv_max(a, L):
    # compute autocorrelation (self-convolution) max using FFT
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)
    # conv length = L; relevant part 0..(2*n-2)
    m = np.max(conv[:2*a.size - 1])
    return m

def _make_shape(n, shape, param):
    # returns array of length n
    if shape == 'cosine':
        x = np.arange(n)
        a = np.sin(math.pi * (x + 0.5) / n)
        if param != 1.0:
            a = a ** param
        return a
    elif shape == 'quadratic':
        x = np.arange(n)
        t = (2.0*x/(n-1) - 1.0)
        a = 1.0 - t*t
        if param != 1.0:
            a = a ** param
        return a
    elif shape == 'exp':
        center = (n-1)/2.0
        dist = np.abs(np.arange(n) - center)
        a = np.exp(-param * dist)
        return a
    elif shape == 'tent':
        x = np.arange(n)
        a = (x+1)*(n - x)
        if param != 1.0:
            a = a ** param
        return a
    else:
        return np.ones(n)

    deadline = start + max(0.0, float(budget_s) - 0.5)  # margin

        cur0 = rng.random(n0, dtype=np.float64)

    # ensure non-negative and normalized sum=1 for consistency
    cur0 = _normalize(cur0)
    n = cur0.size
    L = 1 << ((2 * n - 1).bit_length())  # power of two for FFT

    # ----------------------------------------------------------------------
    # 1. Evaluate initial vector and keep as baseline
    # ----------------------------------------------------------------------
    best_vec = cur0.copy()
    best_max = _conv_max(best_vec, L)
    best_score = 2.0 * n * best_max / (best_vec.sum() ** 2)

    # ----------------------------------------------------------------------
    shape_params = [

        ('exp', 0.005), ('exp', 0.01), ('exp', 0.02), ('exp', 0.04),

        a = _normalize(a)

        score = 2.0 * n * m / (a.sum()**2)
        if score < best_score:

            best_max = m
            best_vec = a.copy()

    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------
    cur = best_vec.copy()
    cur_max = best_max
    no_improve = 0

import numpy as np

# ------------------------------------------------------------
# Basic utilities (matrix‑free, linear memory)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale to have sum == 1 (leaving zeros untouched)."""
    s = v.sum()
    if s == 0.0:
        # fallback uniform – should never happen for a sensible start
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _self_conv_max(a: np.ndarray, L: int):
    """
    Return (max of a*a, FFT of a, full convolution array).
    Convolution is performed by FFT of length L ≥ 2·len(a)‑1.
    """
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)
    # relevant part: first 2·len(a)‑1 entries
    max_val = conv[: 2 * a.size - 1].max()
    return max_val, a_fft, conv


def _softmax_weights(conv: np.ndarray, max_val: float, eps: float) -> np.ndarray:
    """Soft‑max over the convolution values, temperature = eps."""
    w = np.exp((conv - max_val) / eps)          # all ≤ 1, no overflow
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size, dtype=np.float64)
    return w / s


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of the smoothed‑max objective.
    Implements  g_i = 2·(a * w_rev)[(2n‑2)‑i]  (matrix‑free via FFT).
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)          # slice yielding length‑n vector
    return 2.0 * grad_full[idx]

def _make_shape(n: int, kind: str, param: float) -> np.ndarray:

    x = np.arange(n, dtype=np.float64)
    if kind == "cosine":
        a = np.sin(math.pi * (x + 0.5) / n)
        if param != 1.0:
            a = a ** param
    elif kind == "quadratic":
        t = (2.0 * x / (n - 1) - 1.0)
        a = 1.0 - t * t
        if param != 1.0:
            a = a ** param
    elif kind == "exp":

        a = np.exp(-param * np.abs(x - centre))
    elif kind == "tent":
        a = (x + 1.0) * (n - x)
        if param != 1.0:
            a = a ** param
    else:
        a = np.ones(n, dtype=np.float64)
    return a
