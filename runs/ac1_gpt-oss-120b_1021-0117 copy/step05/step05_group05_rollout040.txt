import time, math, numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _softmax(logits: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector (sum==1, non‑negative)."""
    m = np.max(logits)
    e = np.exp(logits - m)
    s = e.sum()
    # avoid division by zero – logits are never all -inf here
    return e / (s if s != 0 else 1.0)


def _score(a: np.ndarray) -> float:
    """Score = 2 * n * max( a * a )  (convolution, no denominator)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _grad_a(a: np.ndarray, conv_max: float, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max‑convolution w.r.t. a.
    w – weights on the active set (sums to 1).
    Returns vector of length n.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)          # map to correct positions
    return 2.0 * grad_full[idx]               # factor 2 from derivative of a·a

    best_vec = a0.copy()

    # ------------------------------------------------------------

    it = 0
