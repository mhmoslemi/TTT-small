import time
import math
import numpy as np

# ------------------------------------------------------------
# Convolution utilities (matrix‑free, FFT based)
# ------------------------------------------------------------
def _self_conv_max(a: np.ndarray, L: int):
    """Return max of a*a (linear convolution) and the FFT of a."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int):
    """
    Gradient of the weighted convolution objective.
    Returns 2 * (a * w_rev) evaluated at the required slice.
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that minimizes
    the AC1 upper bound.  The routine works directly on the parent
    sequence `height_sequence_1` when it is present; otherwise it starts
    from a random vector.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin before the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initial vector – use the parent if it exists
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
    else:
        # modest random start if no parent is supplied
        cur = rng.random(2048, dtype=np.float64)

    # Normalise to a probability simplex (sum = 1)
    cur_sum = cur.sum()
    if cur_sum == 0.0:
        cur = np.full_like(cur, 1.0 / cur.size, dtype=np.float64)
    else:
        cur = cur / cur_sum

    n = cur.size
    # Length for FFT (must accommodate convolution of length n and 2n‑1)
    L = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2

    # --------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------
    beta = 0.9                     # momentum factor
    momentum = np.zeros_like(cur)  # initialise momentum
    eta = 0.5                      # initial learning rate (exponentiated)
    eta_min = 1e-12
    eta_max = 10.0
    best_vec = cur.copy()
    best_max, _, _ = _self_conv_max(cur, L)
    best_score = 2.0 * n * best_max
    target_score = 1.50274349
    no_improve = 0
    it = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and max -----
        cur_max, a_fft, conv = _self_conv_max(cur, L)

        # ----- detect support of the maximum -----
        # points within a tiny tolerance of the maximum receive weight
        tol = max(cur_max * 1e-9, 1e-12)
        sup_idx = np.where(conv >= cur_max - tol)[0]
        if sup_idx.size == 0:
            # fallback to the absolute maximum index
            sup_idx = np.array([conv.argmax()], dtype=np.int64)
        w = np.zeros_like(conv)
        w[sup_idx] = 1.0 / sup_idx.size

        # ----- gradient (matrix‑free via FFT) -----
        grad = _gradient(a_fft, w, n, L)

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- line‑search for a descent step -----
        success = False
        trial_eta = eta
        cand = None
        cand_max = None
        for _ in range(8):
            exponent = -trial_eta * momentum
            exponent = np.clip(exponent, -50.0, 50.0)   # avoid overflow
            cand_tmp = cur * np.exp(exponent)
            cand_tmp = cand_tmp / np.sum(cand_tmp)     # project onto simplex

            cand_tmp_max, _, _ = _self_conv_max(cand_tmp, L)
            if cand_tmp_max <= cur_max + 1e-15:
                cand, cand_max = cand_tmp, cand_tmp_max
                success = True
                break
            trial_eta *= 0.5

        if success:
            cur = cand
            cur_max = cand_max
            eta = min(eta_max, trial_eta * 2.0)   # modestly increase step size
            no_improve = 0
        else:
            eta = max(eta_min, eta * 0.5)         # shrink step size
            no_improve += 1

        # ----- keep the best solution seen so far -----
        if cur_max < best_max:
            best_max = cur_max
            best_vec = cur.copy()
            best_score = 2.0 * n * best_max
            if best_score < target_score:
                break

        # ----- periodic symmetrisation (helps due to convolution symmetry) -----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = sym / np.sum(sym)
            sym_max, _, _ = _self_conv_max(sym, L)
            if sym_max < cur_max:
                cur = sym
                cur_max = sym_max
                if sym_max < best_max:
                    best_max = sym_max
                    best_vec = sym.copy()
                    best_score = 2.0 * n * best_max
                no_improve = 0

                j += 1

                    cur_max = tmp_max
                    no_improve = 0

                        best_score = 2.0 * n * best_max

                    no_improve = 0

        if time.monotonic() >= deadline:

    # Optional local refinement with cheap random perturbations
    # --------------------------------------------------------
    while time.monotonic() < deadline - 0.1:

        cand[i] -= delta
        cand[j] += delta
        cand = cand / np.sum(cand)
        cand_max, _, _ = _self_conv_max(cand, L)
        cand_score = 2.0 * n * cand_max
        if cand_score < best_score:
            best_vec = cand
            best_score = cand_score
            if best_score < target_score:
                break

    # --------------------------------------------------------
    # Return the best sequence found (plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
