def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    # init a

        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        length = 2048

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.size
    L = 1 << ((3*n-2).bit_length())
    # parameters
    eps0 = 1.0
    eps_decay = 0.9998
    eps_min = 1e-10
    beta1 = 0.9
    beta2 = 0.999
    adam_lr = 0.5
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_score = _score(best)
    it = 0
    no_improve = 0
    # loops
    while time.monotonic() < deadline:
        it += 1
        # compute FFT of a
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2*n-1]
        max_val = conv.max()
        # smoothing weight
        eps = max(eps_min, eps0 * (eps_decay ** it))
        # compute softmax weights
        w = np.exp((conv - max_val) / eps)  # shift by max
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = adam_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        cand = a - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        exp_arg = -eta * g
        np.clip(exp_arg, -50, 50, out=exp_arg)
        cand2 = a * np.exp(exp_arg)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – enforces non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score used by the evaluator – assumes sum(a)=1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a, a))."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The function runs for up to `budget_s` seconds and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise – use the parent if present, otherwise a uniform start
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = np.full(1024, 1.0 / 1024, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)            # enforce sum = 1
    n = cur.size

    # --------------------------------------------------------
    # FFT planning – length ≥ 3n‑2, power of two
    # --------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------
    eps0 = 1.0
    eps_decay = 0.9998
    eps_min = 1e-12

    beta1, beta2 = 0.9, 0.999      # Adam parameters
    adam_lr = 0.4                  # base learning rate
    m = np.zeros_like(cur)        # first moment
    v = np.zeros_like(cur)        # second moment

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    it = 0
    no_improve = 0
    target = 1.50274349

    # --------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of cur with itself -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weights approximating the max -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update (projected onto simplex) -----
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = adam_lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # ----- Mirror‑descent (exponential) candidate -----
        eta = 0.35 / math.sqrt(it)
        exp_arg = -eta * g
        np.clip(exp_arg, -50, 50, out=exp_arg)
        cand2 = cur * np.exp(exp_arg)
        s = cand2.sum()
        if s > 0:
            cand2 /= s
        else:
            cand2 = cur.copy()
        cand2_sc = _score(cand2)

        # ----- choose better of the two -----
        if cand_sc <= cand2_sc:
            new_vec, new_sc = cand, cand_sc
        else:
            new_vec, new_sc = cand2, cand2_sc

        # ----- accept if improvement, otherwise keep current -----
        if new_sc < cur_sc:
            cur, cur_sc = new_vec, new_sc
            no_improve = 0
            if new_sc < best_sc:
                best_vec, best_sc = new_vec.copy(), new_sc
        else:
            no_improve += 1

        # ----- occasional symmetrisation (helps with Toeplitz structure) -----
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # --------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # --------------------------------------------------
    return [float(x) for x in best_vec]
