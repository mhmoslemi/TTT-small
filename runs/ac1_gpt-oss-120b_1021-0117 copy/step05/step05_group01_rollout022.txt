import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Compute full linear convolution via FFT and return max value
# ------------------------------------------------------------
def _conv_max_fft(a: np.ndarray, L: int) -> (np.ndarray, float):
    """Return (full convolution array, its maximum) using FFT length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * a.size - 1]
    return conv, float(conv.max())
