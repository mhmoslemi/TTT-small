import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence whose AC1 bound
    is (hopefully) below the target 1.50274349.
    """

    start = time.monotonic()
    # keep a small safety margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = cur.size
    else:
        n = 2000
        cur = rng.random(n, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1

    # --------------------------------------------------------
    # Pre‑computed FFT size and index map
    # --------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())           # power of two ≥ 3n‑2
    idx = (2 * n - 2) - np.arange(n)               # slice to extract gradient

    # --------------------------------------------------------
    # Helper to evaluate the objective (FFT based, O(n log n))
    # --------------------------------------------------------
    def _score_from_fft(a_fft: np.ndarray) -> float:
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return 2.0 * n * conv.max()

    # initial state
    a_fft = np.fft.rfft(cur, n=L)
    cur_score = _score_from_fft(a_fft)

    eps_min = 1e-12

    it = 0

        # ---- current convolution & soft‑max weights ----
        a_fft = np.fft.rfft(cur, n=L)

        max_val = conv_full.max()

        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps > 1e-12:
