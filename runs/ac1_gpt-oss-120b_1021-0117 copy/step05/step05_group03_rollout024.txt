import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project non‑negative vector v onto the simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound for a vector normalised to sum = 1.
# Returns 2 * n * max(conv) where conv = a * a (linear convolution).
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int, n: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence whose AC1 bound is as low as possible.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin before the hard timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – parent if present, else a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)                # enforce sum = 1
    cur = a0.copy()
    n = cur.size

    # ------------------------------------------------------------
    # FFT length – enough for n‑by‑(2n‑1) convolution
    # ------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec, L, n)
    cur_sc = best_sc

    # algorithmic parameters
    step0 = 0.7            # base step magnitude
    beta = 0.9             # momentum factor
    momentum = np.zeros_like(cur)
    it = 0
    no_improve = 0
    target = 1.50274349    # known target bound
    mu_factor = 0.96       # smoothing decay
    mu_min_frac = 1e-6     # relative minimum smoothing
    # initialise smoothing parameter mu from the current max convolution
    af = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    mu = max(conv.max() * 0.15, 1e-12)

    # ------------------------------------------------------------
    # Optimisation loop – matrix‑free, O(n log n) per iteration
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and its maximum -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_c = conv.max()

        # ----- smoothed weight vector (soft max) -----
        # w_i ∝ exp((c_i - max_c) / mu)
        exp_arg = (conv - max_c) / mu
        w = np.exp(exp_arg)                 # values in (0,1]
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- step size (inverse‑sqrt decay) -----
        base_alpha = step0 / math.sqrt(it)

        # ----- try a few scaled steps and keep the best -----
        cand = None
        cand_sc = None
        for factor in (1.0, 0.7, 0.5, 0.3, 0.1):
            alpha = base_alpha * factor
            tmp = cur - alpha * momentum
            tmp = _project_simplex(tmp)
            sc = _score(tmp, L, n)

            no_improve = 0

        else:

            prob = math.exp(-(cand_sc - cur_sc) / (base_alpha + 1e-12))

                j = rng.integers(0, n - 1)

                    j += 1

                tmp[j] += delta

                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- decay smoothing parameter -----
        mu = max(mu * mu_factor, max_c * mu_min_frac)

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Final optional symmetrisation (may improve the bound a bit)
    # ------------------------------------------------------------------
    sym = (best_vec + best_vec[::-1]) * 0.5
    sym = _project_simplex(sym)
    sym_sc = _score(sym, L, n)
    if sym_sc < best_sc:
        best_vec = sym
        best_sc = sym_sc

    # ------------------------------------------------------------------
    # Return the best candidate found (plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
