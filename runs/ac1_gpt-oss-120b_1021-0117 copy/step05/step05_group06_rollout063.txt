import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Maximum value of a*a (full convolution) using an FFT of length L
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())

            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best = sym.copy()
                    best_sc = sym_sc
                    best_bound = 2.0 * n * best_sc

        if best_bound < target_bound:
            break

    # ------------------------------------------------------------------
    # Permutation & simple hill‑climbing phase -------------------------
    # ------------------------------------------------------------------
    # Helper to evaluate a candidate and keep the best one
    def _maybe_update(candidate):
        nonlocal best, best_sc, best_bound
        sc = _max_conv(candidate, L)
        if sc < best_sc:
            best = candidate.copy()
            best_sc = sc
            best_bound = 2.0 * n * sc

    # anti‑identity permutation (largest –‑ opposite smallest)
    vals_desc = np.sort(best)[::-1]
    perm = np.empty_like(vals_desc)
    left, right = 0, n - 1
    k = 0
    while left <= right:
        perm[left] = vals_desc[k]; k += 1
        if left < right:
            perm[right] = vals_desc[k]; k += 1
        left += 1
        right -= 1
    _maybe_update(perm)

    # simple reverse
    _maybe_update(best[::-1])

    # step‑size based deterministic permutations
    steps = [2, 3, 5, 7, 11, 13, 17, 31, 37, 97, 151, 997]
    for step in steps:
        order = (np.arange(n) * step) % n
        cand = np.empty_like(best)
        cand[order] = vals_desc
        _maybe_update(cand)

    # a handful of random shuffles
    for _ in range(30):
        idx = rng.permutation(n)
        cand = best[idx]
        _maybe_update(cand)

    # iterative random swaps (hill climbing)
    cur = best.copy()
    cur_sc = best_sc
    while time.monotonic() < deadline - 0.2:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_sc = _max_conv(cur, L)
        if new_sc < cur_sc:
            cur_sc = new_sc
            if new_sc < best_sc:
                best = cur.copy()
                best_sc = new_sc
                best_bound = 2.0 * n * new_sc
                if best_bound < target_bound:
                    break
        else:
            cur[i], cur[j] = cur[j], cur[i]

    # ------------------------------------------------------------------
    # Return the best candidate found -----------------------------------
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best]
