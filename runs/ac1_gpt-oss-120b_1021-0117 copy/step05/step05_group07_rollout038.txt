def propose_candidate(...):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0,budget_s - 0.2)
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a,0)
    a = _project_simplex(a)   # sum=1.

    n = a.size
    L = 1 << ((2*n - 1).bit_length())
    idx = (2*n - 2) - np.arange(n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps_adam=1e-8
    step_size=0.1
    eps_temp=1.0
    eps_decay=0.995

    best_a = a.copy()
    best_score = compute_score(a)  # 2*n*max(conv)
    it=0

    while time.monotonic()<deadline:
        it+=1
        # compute conv and max
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # compute softmax weights
        eps_temp = max(1e-12, eps_temp * eps_decay)
        # compute w
        # To avoid overflow when eps is tiny: subtract max first.
        w = np.exp((conv - max_val) / eps_temp)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam updates
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        delta = step_size * m_hat / (np.sqrt(v_hat) + eps_adam)

        # candidate
        cand = a - delta
        cand = _project_simplex(cand)

        # evaluate cand
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[:2*n-1]
        cand_score = 2.0 * n * cand_conv.max()

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
        # acceptance: if cand better accept else stay
        if cand_score <= 2.0*n*max_val:  # improvement else maybe accept with prob
            a = cand
        else:
            # occasional random perturbation
            if rng.random() < 0.01:
                # random jitter small
                jitter = (rng.random(n)-0.5) * 0.01
                a = _project_simplex(np.clip(a + jitter,0, None))
        # occasionally symmetrize
        if it % 100 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[:2*n-1]
            sym_score = 2.0 * n * sym_conv.max()
            if sym_score < best_score:
                best_score = sym_score
                best_a = sym.copy()
            if sym_score < 2.0*n*max_val:
                a = sym

        # Check early exit
        if best_score < target:
            break

    return best_a.tolist()

import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Core optimisation – Adam‑style with soft‑max smoothing
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a tiny safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # Initial vectors (parent, uniform, tilted, linear, random)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        init_parent = np.maximum(init_parent, 0.0)
        init_parent = _project_simplex(init_parent)
        n = init_parent.size
    else:
        n = 2000
        init_parent = np.full(n, 1.0 / n, dtype=np.float64)

    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # slight linear tilt (more mass left, less right)
    tilt = np.linspace(0.5, -0.5, n)
    tilted = uniform + 0.05 * tilt
    tilted = np.maximum(tilted, 0.0)
    tilted = _project_simplex(tilted)

    # linear decreasing (no‑zero support)
    lin_desc = np.arange(n, 0, -1, dtype=np.float64)
    lin_desc = _project_simplex(lin_desc)

    # random Dirichlet‑like start
    random_vec = rng.random(n, dtype=np.float64)
    random_vec = _project_simplex(random_vec)

    candidates = [init_parent, uniform, tilted, lin_desc, random_vec]

    # ------------------------------------------------------------
    # Helper to compute the AC1 score (2*n*max(conv))
    # ------------------------------------------------------------
    def _score(a: np.ndarray, L: int, idx: np.ndarray) -> float:
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------
    # Pick the best among the cheap initial candidates
    # ------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())          # enough for a * w_rev (len 2n‑1) conv
    idx = (2 * n - 2) - np.arange(n)              # gradient slice indices
    best_vec = None
    best_sc = float("inf")
    for cand in candidates:
        sc = _score(cand, L, idx)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()

    # ------------------------------------------------------------
    # Optimisation state
    # ------------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    m = np.zeros_like(cur)        # Adam first moment
    v = np.zeros_like(cur)        # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_step = 0.5
    eps_temp = 1.0                # temperature for soft‑max
    eps_decay = 0.995
    eps_min = 1e-12
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution & score -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_sc = 2.0 * n * max_val

        # ----- update best -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- soft‑max weights for smoothed max -----
        eps_temp = max(eps_min, eps_temp * eps_decay)
        w = np.exp((conv_full - max_val) / eps_temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of soft‑max max (matrix‑free) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update (with bias correction) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step_factor = base_step / math.sqrt(it)
        delta = step_factor * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = _project_simplex(cur - delta)

        # ----- evaluate candidate -----
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_sc = 2.0 * n * cand_conv.max()

        # ----- accept if better, otherwise occasional Metropolis move -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # small probability to accept uphill moves (helps escape plateaus)
            prob = math.exp(-(cand_sc - cur_sc) / (step_factor + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ----- periodic random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta_mass = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta_mass
                tmp[j] += delta_mass
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
