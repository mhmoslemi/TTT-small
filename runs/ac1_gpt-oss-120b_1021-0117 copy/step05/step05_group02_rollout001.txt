def _project_simplex(v):
    # same as before
    n=v.size
    u=np.sort(v)[::-1]
    css=np.cumsum(u)-1
    rho=np.where(u - css/np.arange(1,n+1) > 0)[0]
    if rho.size==0:
        theta=0.0
    else:
        rho=rho[-1]
        theta=css[rho]/(rho+1)
    return np.maximum(v-theta,0.0)

def _objective(a, L=None):
    n = a.shape[0]
    if L is None:
        L = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af*af, n=L)[:2*n-1]
    maxb = conv.max()
    return 2.0 * n * maxb

def _subgradient(a, conv):

        length = n - offset
        g[offset:] = 2.0 * a[:length][::-1]

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, L: int) -> float:
    """2 * n * max(convolution(a, a)), assumes sum(a)=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _subgradient(a: np.ndarray, conv: np.ndarray):
    """Subgradient of max(conv) w.r.t a (a is non‑negative, sum=1)."""
    k = int(np.argmax(conv))               # index of maximal autocorrelation
    n = a.shape[0]
    g = np.zeros_like(a)
    if k < n:
        g[: k + 1] = 2.0 * a[k :: -1]       # a[k], a[k-1], ..., a[0]
    else:
        offset = k - n + 1
        length = n - offset
        g[offset:] = 2.0 * a[:length][::-1]
    return g, k


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initialise ---------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 1024
        cur = np.full(n0, 1.0 / n0, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)            # enforce sum = 1
    n = cur.size

    # FFT length (power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _objective(best_vec, L)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)

    # ---- hyper‑parameters ----------------------------------------------------
    alpha0 = 1.0          # base step for momentum
    eta0 = 0.8            # base step for mirror‑descent
    beta = 0.9            # momentum decay
    target = 1.50274349
    it = 0
    no_improve = 0

    # ---- main optimisation loop --------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current autocorrelation
        af = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]

        # subgradient at current maximum
        g, _ = _subgradient(cur, conv_full)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- candidate 1 : momentum step with back‑tracking -----------------
        alpha = alpha0 / math.sqrt(it)
        cand1 = cur - alpha * momentum
        cand1 = _project_simplex(cand1)
        cand1_sc = _objective(cand1, L)
        if cand1_sc > cur_sc:
            for _ in range(8):
                alpha *= 0.5
                cand_tmp = cur - alpha * momentum
                cand_tmp = _project_simplex(cand_tmp)
                tmp_sc = _objective(cand_tmp, L)
                if tmp_sc <= cur_sc:
                    cand1, cand1_sc = cand_tmp, tmp_sc
                    break
            else:
                cand1, cand1_sc = cand_tmp, tmp_sc

        # ----- candidate 2 : exponentiated (mirror) step --------------------
        eta = eta0 / math.sqrt(it)
        exp_arg = -eta * g
        np.clip(exp_arg, -50, 50, out=exp_arg)
        cand2 = cur * np.exp(exp_arg)
        s = cand2.sum()
        if s > 0:
            cand2 /= s
        else:
            cand2 = cur.copy()
        cand2_sc = _objective(cand2, L)

        # ----- pick better candidate -----------------------------------------
        if cand1_sc <= cand2_sc:
            cand, cand_sc = cand1, cand1_sc
        else:
            cand, cand_sc = cand2, cand2_sc

        # ----- acceptance ----------------------------------------------------
        if cand_sc < cur_sc - 1e-12:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- occasional symmetrisation ------------------------------------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _objective(sym, L)
            if sym_sc < cur_sc - 1e-12:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random mass transfer when stuck -------------------------------
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _objective(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early stop ----------------------------------------------------
        if best_sc < target:
            break

    # return best candidate as plain Python list
    return [float(x) for x in best_vec]
