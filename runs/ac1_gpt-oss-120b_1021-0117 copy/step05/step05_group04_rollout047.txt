# Helper utilities (top‑level, no closures)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Return a non‑negative vector with sum = 1."""
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _conv_max(a: np.ndarray, L: int):
    """Return (max of a*a, FFT of a, full convolution array)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _grad_sub(a_fft: np.ndarray, w: np.ndarray, n: int, L: int):
    """Exact sub‑gradient of the max‑convolution using the weight mask w."""
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

            cand_tmp = _normalize(cand_tmp)

            cand_tmp_max, _, _ = _conv_max(cand_tmp, L)
            if cand_tmp_max <= cur_max + 1e-15:
                cand, cand_max = cand_tmp, cand_tmp_max
                break
            step_cur *= 0.5

        if cand is None:
            # fallback to last trial (may be uphill)
            cand, cand_max = cand_tmp, cand_tmp_max

        # ----------------------------------------------------
        # acceptance test
        # ----------------------------------------------------
        if cand_max < cur_max - 1e-15:
            cur = cand
            cur_max = cand_max
            no_improve = 0
            if cand_max < best_max:
                best_max = cand_max
                best_vec = cand.copy()
                best_score = 2.0 * n * best_max
        else:
            no_improve += 1
            # occasional uphill acceptance to escape plateaus
            prob = math.exp(-(cand_max - cur_max) / (step + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_max = cand_max

        # ----------------------------------------------------
        # periodic symmetrisation (helps due to convolution symmetry)
        # ----------------------------------------------------
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _normalize(sym)
            sym_max, _, _ = _conv_max(sym, L)
            if sym_max < cur_max - 1e-15:
                cur = sym
                cur_max = sym_max
                if sym_max < best_max:
                    best_max = sym_max
                    best_vec = sym.copy()
                    best_score = 2.0 * n * best_max
                no_improve = 0

        # ----------------------------------------------------
        # random mass‑transfer when stagnating
        # ----------------------------------------------------
        if no_improve > 5000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _normalize(tmp)
                tmp_max, _, _ = _conv_max(tmp, L)
                if tmp_max < cur_max:
                    cur = tmp
                    cur_max = tmp_max
                    if tmp_max < best_max:
                        best_max = tmp_max
                        best_vec = tmp.copy()
                        best_score = 2.0 * n * best_max
                    no_improve = 0
                else:
                    no_improve = 0

        # ----------------------------------------------------
        # early exit if target already reached
        # ----------------------------------------------------
        if best_score < target_score:
            break

    # --------------------------------------------------------
    # Return best sequence found (plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
