def _normalize(v):
    s = v.sum()
    if s == 0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _self_conv(a, L=None):
    n = a.size
    if L is None:
        L = _next_pow2(2*n-1)
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    return conv, a_fft

def _gradient_from_weights(a_fft, w, n, L):
    # w: binary mask of length 2n-1
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale vector to have sum = 1 (non‑negative)."""
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _self_conv(a: np.ndarray, L: int):
    """
    Return full linear convolution a*a (length 2·n‑1) and the FFT of a.
    """
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * a.size - 1]
    return conv, a_fft


def _gradient_from_weights(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of the (soft) max‑convolution objective.
    w is a weight vector of length 2·n‑1 (e.g. softmax or binary mask).
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients that (hopefully)
    yields an AC1 bound lower than the target 1.50274349.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # Initialise from parent if present, otherwise random.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
    else:
        cur = rng.random(2000, dtype=np.float64)

    cur = _normalize(cur)
    n = cur.size
    L = _next_pow2(3 * n - 2)          # enough for full a*g convolution (len ≈ 3n‑2)

    # --------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------
    beta = 0.9                         # momentum factor
    momentum = np.zeros_like(cur)
    step0 = 0.5                        # base step size
    step_min = 1e-5
    no_improve = 0
    best_vec = cur.copy()
    conv, a_fft = _self_conv(cur, L)
    cur_max = conv.max()
    best_max = cur_max
    best_score = 2.0 * n * best_max   # sum(cur)=1 → denominator = 1
    target_score = 1.50274349
    it = 0
    base_tol = 1e-12

    # --------------------------------------------------------
    # Main optimisation loop (matrix‑free gradient descent)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # --- current convolution + binary mask of active constraints ---
        conv, a_fft = _self_conv(cur, L)
        cur_max = conv.max()
        tol = max(base_tol, cur_max * 1e-10)
        mask = conv >= cur_max - tol
        w = mask.astype(np.float64)                     # binary weights

        # --- sub‑gradient of the max‑convolution ---
        grad = _gradient_from_weights(a_fft, w, n, L)

        # --- momentum update ---
        momentum = beta * momentum + (1.0 - beta) * grad

        # --- candidate generation (exponentiated update) ---
        step = max(step_min, step0 / math.sqrt(it))
        cand = None
        cand_max = None
        for _ in range(8):                     # simple back‑tracking line search
            exp_arg = -step * momentum
            exp_arg = np.clip(exp_arg, -50.0, 50.0)  # avoid overflow
            cand_tmp = cur * np.exp(exp_arg)
            cand_tmp = _normalize(cand_tmp)

            cand_conv, _ = _self_conv(cand_tmp, L)
            cand_tmp_max = cand_conv.max()
            if cand_tmp_max <= cur_max + 1e-15:
                cand, cand_max = cand_tmp, cand_tmp_max
                break
            step *= 0.5

        if cand is None:                       # fallback to last trial
            cand = cand_tmp
            cand_max = cand_tmp_max

        # --- acceptance test ---
        if cand_max < cur_max - 1e-15:
            cur = cand
            cur_max = cand_max
            no_improve = 0
            if cand_max < best_max:
                best_max = cand_max
                best_vec = cand.copy()
                best_score = 2.0 * n * best_max
        else:
            no_improve += 1
            prob = math.exp(-(cand_max - cur_max) / (step + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_max = cand_max

        # --- periodic symmetrisation (helps due to Toeplitz symmetry) ---
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _normalize(sym)
            sym_conv, _ = _self_conv(sym, L)
            sym_max = sym_conv.max()
            if sym_max < cur_max:
                cur = sym
                cur_max = sym_max
                if sym_max < best_max:
                    best_max = sym_max
                    best_vec = sym.copy()
                    best_score = 2.0 * n * best_max
                no_improve = 0

        # --- random mass‑transfer when stagnating ---
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _normalize(tmp)
                tmp_conv, _ = _self_conv(tmp, L)
                tmp_max = tmp_conv.max()
                if tmp_max < cur_max:
                    cur = tmp
                    cur_max = tmp_max
                    if tmp_max < best_max:
                        best_max = tmp_max
                        best_vec = tmp.copy()
                        best_score = 2.0 * n * best_max
                    no_improve = 0
                else:
                    no_improve = 0

        # --- early exit if target already attained ---
        if best_score < target_score:
            break

    # --------------------------------------------------------
    # Post‑processing: try modest up‑scaling (clipping at 1000)
    # --------------------------------------------------------
    max_entry = best_vec.max()
    scaling_limit = 1000.0 / max_entry if max_entry > 0 else 1.0
    max_factor = max(1.0, min(scaling_limit, 10.0))   # explore up to 10× or limit
    factors = np.geomspace(1.0, max_factor, num=30)

    best_scaled = best_vec
    best_scaled_score = best_score

    for f in factors:
        cand = np.clip(best_vec * f, 0.0, 1000.0)
        s = cand.sum()
        if s < 0.01:                       # safeguard (should never happen)
            continue
        cand_conv, _ = _self_conv(cand, L)
        cand_max = cand_conv.max()
        cand_score = 2.0 * n * cand_max / (s * s)
        if cand_score < best_scaled_score:
            best_scaled_score = cand_score
            best_scaled = cand.copy()

    # --------------------------------------------------------
    # Return the best sequence discovered (plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_scaled]
