import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Project a vector onto the probability simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Core routine: propose a candidate sequence.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑empty list of non‑negative coefficients whose AC1 bound
    (as evaluated by `evaluate_sequence`) is as low as possible within the
    allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # keep margin

    # --------------------------------------------------------------
    # Initial sequence (use parent if present)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)
    n = cur.size

    # --------------------------------------------------------------
    # FFT length (must accommodate convolution of length 2n‑1 with gradient)
    # --------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------

    best_vec = cur.copy()
    # initial score (no need for extra FFT: compute once)

    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    cur_max = conv_full[: 2 * n - 1].max()
    best_sc = cur_sc = float(2 * n * cur_max)

    # Adam moments
    m = np.zeros_like(cur)

    while time.monotonic() < deadline:
        it += 1

        a_fft = np.fft.rfft(cur, n=L)

        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
