import time
import math
import numpy as np

def _conv_fft(v: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution v*v using real FFT, length 2*len(v)-1."""
    vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(vf * vf, n=L)[: 2 * v.size - 1]
    return conv

def _active_gradient(a: np.ndarray, act: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max{a*a} w.r.t. a for the active indices."""
    n = a.size
    i = np.arange(n)
    grad = np.zeros_like(a)
    for k in act:
        j = k - i
        mask = (j >= 0) & (j < n)
        # contribution a_{k-i}
        grad[i[mask]] += a[j[mask]]
        # diagonal term counted twice
        if (k & 1) == 0:
            cen = k // 2
            if 0 <= cen < n:
                grad[cen] += a[cen]
    return grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative coefficient list with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # ---------- initialise ----------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
    else:
        a = rng.random(1024, dtype=np.float64)

    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.size, dtype=np.float64)
    else:
        a = a / s                       # normalise to sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # power of two ≥ 2n‑1

    # ---------- bookkeeping ----------
    conv = _conv_fft(a, L)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    best_score = 2.0 * n * best_max      # sum = 1 → score = 2 n max

    # parameters
    step0 = 0.4
    momentum_factor = 0.9
    momentum = np.zeros_like(a)
    no_improve = 0
    it = 0
    target_score = 1.50274349

    # ---------- main loop ----------
    while time.monotonic() < deadline:
        it += 1

        # recompute convolution and active set
        conv = _conv_fft(a, L)
        cur_max = conv.max()
        # active indices: within tiny tolerance of the maximum
        tol = max(1e-12, cur_max * 1e-9)
        act = np.where(conv >= cur_max - tol)[0]

        grad = _active_gradient(a, act)

        momentum = momentum_factor * momentum + (1.0 - momentum_factor) * grad

        step = step0 / math.sqrt(it)

        exponent = np.clip(exponent, -50.0, 50.0)   # avoid overflow
