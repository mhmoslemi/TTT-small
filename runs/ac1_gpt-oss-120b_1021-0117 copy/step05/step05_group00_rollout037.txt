import time

import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        it += 1

        # ----- convolution -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_sc = 2.0 * n * max_val

        # ----- smoothing temperature -----
        if eps_smoothing is None:
            eps_smoothing = max(max_val * 0.2, 1e-12)
        else:
            eps_smoothing = max(eps_smoothing * eps_decay, 1e-12)

        # ----- soft‑max weights -----
        w = np.exp((conv_full - max_val) / eps_smoothing)
        ws = w.sum()
        if ws == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= ws

        # ----- gradient of smoothed max -----
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
        g = 2.0 * grad_full[idx]          # gradient vector

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / (math.sqrt(it) + 1.0)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        cand = cur - step
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_sc = _score(cand)

        # ----- accept if improvement, else occasional uphill -----
        improved = False
        if cand_sc < cur_sc:
            improved = True
        else:
            # simulated‑annealing style acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()
                cur = sym
                cur_sc = sym_sc
                no_improve = 0
            elif sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                no_improve = 0
            # reset Adam moments after a symmetry step
            m.fill(0.0)
            v.fill(0.0)

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                trial = cur.copy()
                trial[i] -= delta
                trial[j] += delta
                trial = _project_simplex(trial)
                trial_sc = _score(trial)
                if trial_sc < cur_sc:
                    cur = trial
                    cur_sc = trial_sc
                    if trial_sc < best_sc:
                        best_sc = trial_sc
                        best_vec = trial.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- restart from best with small noise if hopeless -----
        if no_improve > 5000:
            cur = best_vec + 0.005 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps_smoothing = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # Return as plain Python list
    return [float(x) for x in best_vec]
