import time, math, numpy as np

# ------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, ≥0)
# ------------------------------------------------------------
def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
# Score of a candidate (exactly the evaluator’s formula)
# ------------------------------------------------------------
def _score(vec, L):
    a = np.asarray(vec, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_val / (sum_a * sum_a))
