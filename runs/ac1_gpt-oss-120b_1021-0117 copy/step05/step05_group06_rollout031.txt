import time
import math
import numpy as np

# --------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, x >= 0)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# Max convolution (full) of a vector with itself, using FFT
# --------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())

                w = np.full_like(conv_full, 1.0 / conv_full.size)
            else:
                w = np.where(mask, 1.0 / cnt, 0.0)

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
        # mapping back to the original indices
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # back‑tracking line‑search
        alpha = step0 / math.sqrt(it)
        improved = False
        for _ in range(8):
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)
            cand_bound = _bound(cand)
            if cand_bound <= cur_bound:
                cur = cand
                cur_bound = cand_bound
                improved = True
                if cand_bound < best_bound:
                    best = cand.copy()
                    best_bound = cand_bound
                break
            alpha *= 0.5

        if not improved:
            # occasional reset of momentum to escape plateaus
            momentum *= 0.0

        # occasional symmetrisation (helps due to the Toeplitz structure)
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_bound = _bound(sym)
            if sym_bound < cur_bound:
                cur = sym
                cur_bound = sym_bound
                if sym_bound < best_bound:
                    best = sym.copy()
                    best_bound = sym_bound

        if best_bound < target:
            break

    # ----------------------------------------------------------
    # Local refinement: small mass transfers between coordinates
    # ----------------------------------------------------------
    cur = best.copy()
    cur_bound = best_bound
    while time.monotonic() < deadline - 0.3:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a modest fraction of mass from i to j
        delta = (rng.random() - 0.5) * 0.02          # up to ±1 % of total mass
        if delta > cur[i]:
            delta = cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = _project_simplex(new)
        new_bound = _bound(new)
        if new_bound < cur_bound:
            cur = new
            cur_bound = new_bound
            if new_bound < best_bound:
                best = new.copy()
                best_bound = new_bound
                if best_bound < target:
                    break

    # ----------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ----------------------------------------------------------
    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best.tolist()]
