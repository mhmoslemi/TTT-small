def _optimize(initial, time_limit, start_time):

  cur = initial.copy()
  cur_sc = _score(cur)
  best = cur.copy()
  best_sc = cur_sc

  n = len(cur)

  L = 1 << ((3*n - 2).bit_length())

  momentum = np.zeros_like(cur)
  beta = 0.9
  base_alpha = 0.7  # bigger step maybe
  eps = None
  eps_decay = 0.99
  max_line_search = 5
  no_improve_counter = 0

  while time.monotonic() < start_time + time_limit:
      # compute convolution via FFT
      a_fft = np.fft.rfft(cur, n=L)
      conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n - 1]
      max_val = conv_full.max()
      # smoothing epsilon
      if eps is None:
         eps = max(max_val * 0.3, 1e-12)
      else:
         eps = max(eps * eps_decay, 1e-12)
      # weights
      w = np.exp((conv_full - max_val) / eps)
      w_sum = w.sum()
      if w_sum == 0.0:
         w = np.full_like(w, 1.0 / w.size)
      else:
         w /= w_sum
      # compute gradient via FFT
      w_rev = w[::-1]
      w_fft = np.fft.rfft(w_rev, n=L)
      grad_full = np.fft.irfft(a_fft * w_fft, n=L)
      # mapping to gradient size n: indices:
      # conv_full index k corresponds to sum_{i} a_i a_{k-i}, with i range.
      # gradient expression for smoothed max given weight w is sum_k w_k * gradient of conv_k.
      # The gradient computing via fft with reversed w gives this.
      # In original code they extracted first n components using idx = (2 * n - 2) - np.arange(n)
      idx = (2 * n - 2) - np.arange(n)
      g = 2.0 * grad_full[idx]

      # momentum update
      momentum = beta * momentum + (1.0 - beta) * g

      # step size: normalizing
      norm_m = np.linalg.norm(momentum) + 1e-12
      alpha0 = base_alpha / (np.sqrt(1 + no_improve_counter)) / norm_m

      # line search
      improved = False
      cand = None
      cand_sc = None
      alpha = alpha0
      for _ in range(max_line_search):

          cand_try = _project_simplex(cand_try)

      # occasional uphill acceptance

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound for a vector whose sum is (approximately) 1.
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) of a with itself."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Core optimizer working on a single seed vector.
# ------------------------------------------------------------
def _optimize(initial: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Projected gradient descent with momentum and occasional random moves."""
    n = initial.size
    L = 1 << ((3 * n - 2).bit_length())          # length for gradient convolution
    cur = initial.copy()
    cur_sc = _score(cur)

    best = cur.copy()
    best_sc = cur_sc

    momentum = np.zeros_like(cur)
    beta = 0.9
    base_alpha = 0.5
    eps = None
    eps_decay = 0.995
    max_trials = 5
    no_improve = 0
    target = 1.50274349

    deadline = time.monotonic() + time_limit
    while time.monotonic() < deadline:
        # ----- convolution & smoothed max weight -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps is None:
            eps = max(max_val * 0.2, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max (via FFT) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)               # map to length‑n gradient
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- step size (decaying) -----
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(1 + no_improve) / norm_m

        # ----- back‑tracking line search -----
        improved = False
        cand = None
        cand_sc = None
        alpha = alpha0
        for _ in range(max_trials):
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5

        # ----- occasional uphill move (simulated annealing) -----
        if not improved:
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            temp = 1e-3
            prob = math.exp(-(cand_sc - cur_sc) / temp) if cand_sc > cur_sc else 1.0
            if rng.random() < prob:
                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc - 1e-12:
                best_sc = cand_sc
                best = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- periodic symmetrisation (helps a lot) -----
        if no_improve % 500 == 0 and no_improve > 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best = sym.copy()
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    return best, best_sc
