import time, math, numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a list of non‑negative coefficients (step‑function heights) that
    aims to minimise the AC1 upper bound evaluated by `evaluate_sequence`.
    """
    start = time.monotonic()
    # keep a small safety margin for the harness timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Base vector (use the supplied construction if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(1024, dtype=np.float64)

    n = a0.shape[0]

    # ------------------------------------------------------------------
    # 2.  Utility helpers
    # ------------------------------------------------------------------
    def _project_simplex(v):
        """Projection onto the probability simplex (non‑negative, sum‑to‑1)."""
        v = np.asarray(v, dtype=np.float64)
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, len(v) + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    # Length for FFT (next power of two for a linear convolution)
    L = 1 << ((2 * n - 1).bit_length())

    def _score_fft(vec):
        """Score 2·n·max(conv) assuming sum(vec)=1 (convolution via FFT)."""
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max()), conv

    # Initialise best solution with the (projected) parent if it exists
    cur = _project_simplex(np.maximum(a0, 0.0))
    best_vec = cur.copy()
    best_sc, _ = _score_fft(best_vec)

    # ------------------------------------------------------------------
    # 3.  Phase 1 – exponential‑decay sweep (cheap global search)
    # ------------------------------------------------------------------
    low, high = 1e-6, 5e-3          # search interval for the decay rate
    best_alpha = None
    for _ in range(6):              # successive refinements
        alphas = np.logspace(np.log10(low), np.log10(high), num=30)
        for alpha in alphas:
            if time.monotonic() >= deadline:
                break
            a = np.exp(-alpha * np.arange(n, dtype=np.float64))
            a /= a.sum()                     # enforce sum = 1
            sc, _ = _score_fft(a)
            if sc < best_sc:
                best_sc, best_vec, best_alpha = sc, a.copy(), alpha
        if best_alpha is None:
            break
        # tighten the interval around the current best
        low = max(best_alpha / 2.0, 1e-12)
        high = best_alpha * 2.0

    # ------------------------------------------------------------------
    # 4.  Phase 2 – sub‑gradient descent on the non‑smooth objective
    # ------------------------------------------------------------------
    a = best_vec.copy()
    a_sc = best_sc
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.8
    it = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # convolution and maximal value
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_val = conv.max()
        a_sc = 2.0 * n * max_val

        # build a (sub‑)gradient of the maximum entry
        # mask of ties (or nearly ties) to obtain a valid subgradient
        mask = conv >= max_val - 1e-12
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 1:
            w /= cnt
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum‑aided step
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        cand = a - step * momentum
        cand = _project_simplex(cand)
        cand_sc, _ = _score_fft(cand)

        # simple acceptance rule (always take improvement)
        if cand_sc < a_sc:
            a, a_sc = cand, cand_sc
        else:
            # occasional uphill moves (simulated‑annealing style)
            prob = math.exp(-(cand_sc - a_sc) / (step + 1e-12))
            if rng.random() < prob:
                a, a_sc = cand, cand_sc

        # update global best
        if a_sc < best_sc:
            best_sc, best_vec = a_sc, a.copy()
            if best_sc < target:
                break

        # occasional random jitter to escape flat regions
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.05, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = _project_simplex(a_j)
            a_j_sc, _ = _score_fft(a_j)
            if a_j_sc < best_sc:
                best_sc, best_vec = a_j_sc, a_j.copy()
                a, a_sc = a_j, a_j_sc

    # ------------------------------------------------------------------
    # 5.  Return the best found vector as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
