import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a safety margin (0.5 s) before the sandbox kills the process
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a modest default length – the exact length is not critical
        cur = rng.random(2000, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1
    n = cur.size

    # ------------------------------------------------------------
    # FFT preparation – length at least 3n‑2 (power of two)
    # ------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())   # smallest power of two ≥ 3n‑2
    idx = (2 * n - 2) - np.arange(n)      # indices to extract the gradient

    # ------------------------------------------------------------
    # Adam state
    # ------------------------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr_base = 0.6          # base learning rate (will be scaled by Adam)

    # ------------------------------------------------------------
    # Temperature schedule for the soft‑max smoothing of max(conv)
    # ------------------------------------------------------------
    temp0 = 1.0
    temp_decay = 0.995

    # ------------------------------------------------------------
    # Book‑keeping of best solution
    # ------------------------------------------------------------
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()
    cur_sc = 2.0 * n * max_val
    best_sc = cur_sc
    best_vec = cur.copy()

    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------
    # Optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- softmax weights (smooth approximation of max) -----
        temp = max(temp0 * (temp_decay ** it), 1e-12)
        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the (soft) max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
