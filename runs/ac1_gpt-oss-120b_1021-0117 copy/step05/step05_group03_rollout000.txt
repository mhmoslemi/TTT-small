import time, math, numpy as np

def _project_simplex(v):
    # docstring or comment...
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a, L):
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[:2*n-1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate a random vector of length about 2000
        n = 2000
        a = rng.random(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.size
    L = 1 << ((3*n - 2).bit_length())

    best = a.copy()
    best_sc = _score(best, L)

    # algorithmic parameters
    step0 = 0.8
    beta = 1.0          # smoothing factor
    beta_inc = 1.001
    beta_max = 500.0
    momentum = np.zeros_like(a)
    no_improve = 0
    target = 1.50274349
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(a, n=L)

        # convolution a*a
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        # smooth weight (softmax) w
        # shift by max for numerical stability
        w_raw = np.exp(beta * (conv - max_val))
        w = w_raw / w_raw.sum()

        # gradient of smooth max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = 0.9 * momentum + 0.1 * grad

        # step size
        alpha = step0 / math.sqrt(it)

        # candidate update
        cand = a - alpha * momentum
        cand = _project_simplex(cand)

        cand_sc = _score(cand, L)

        # acceptance
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            no_improve = 0
        else:
            no_improve += 1

        a = cand

        # increase smoothing
        if beta < beta_max:
            beta = min(beta_max, beta * beta_inc)

        # periodic symmetrisation
        if it % 600 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
                no_improve = 0
            a = sym

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray, L: int) -> float:
    """Score = 2 * n * max(convolution(a, a)) assuming sum(a)=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())
