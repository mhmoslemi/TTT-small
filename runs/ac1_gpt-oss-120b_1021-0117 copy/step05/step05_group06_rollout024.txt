import time, math, numpy as np

# ----------------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Maximum value of full convolution a * a using zero‑padded FFT
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())


# ----------------------------------------------------------------------
# Anti‑identity permutation: largest, smallest, 2nd‑largest, 2nd‑smallest …
# ----------------------------------------------------------------------
def _anti_permute(v_desc: np.ndarray) -> np.ndarray:
    n = v_desc.size
    out = np.empty(n, dtype=v_desc.dtype)
    left, right = 0, n - 1
    k = 0
    while left <= right:
        out[left] = v_desc[k]; k += 1
        if left < right:
            out[right] = v_desc[k]; k += 1
        left += 1
        right -= 1
    return out

# ----------------------------------------------------------------------
