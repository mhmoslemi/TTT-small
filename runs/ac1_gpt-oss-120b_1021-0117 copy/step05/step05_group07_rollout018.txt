import time, math, numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    # make a copy to avoid modifying the input
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative heights that (hopefully)
    improves the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin for the final return
    deadline = start_time + max(0.0, float(budget_s) - 0.3)

    # --------------------------------------------------------
    # Initialise – use the parent if present, otherwise a quick start
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # build a monotone decreasing start (helps anti‑symmetry)
        n = 2500
        cur = np.sort(rng.exponential(scale=1.0, size=n))[::-1]
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # sum = 1
    n = cur.size

    # --------------------------------------------------------
    # FFT set‑up (single length for the whole run)
    # --------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length() + 1)   # next power of two ≥ 2n
    idx = (2 * n - 2) - np.arange(n)          # indexing for gradient

    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()
    cur_sc = 2.0 * n * max_val
    best_sc = cur_sc
    best_vec = cur.copy()

    # --------------------------------------------------------
    # Optimisation hyper‑parameters
    # --------------------------------------------------------
    step0 = 0.25          # base step size
    beta = 0.95           # momentum decay
    momentum = np.zeros_like(cur)

    eps0 = 0.07           # initial soft‑max temperature
    eps_decay = 0.996
    eps_min = 1e-12

    it = 0
    no_improve = 0
    target = 1.50274349   # stop if we beat this

    # --------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- soft‑max weights around the current maximum -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps > 5e-12:
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum
        else:
            # hard max – equal weight on all ties
            w = np.zeros_like(conv_full)
            mask = conv_full == max_val
            cnt = mask.sum()
            if cnt:
                w[mask] = 1.0 / cnt

        # ----- gradient of the (soft) max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]               # shape (n,)

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- step size (inverse‑sqrt decay) with back‑tracking line search -----
        base_alpha = step0 / math.sqrt(it)
        alpha = base_alpha
        cand = None
        cand_sc = None

        for _ in range(10):
            cand_tmp = cur - alpha * momentum
            cand_tmp = _project_simplex(cand_tmp)

            cand_fft = np.fft.rfft(cand_tmp, n=L)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
            cand_max = cand_conv.max()
            cand_sc = 2.0 * n * cand_max

            if cand_sc <= cur_sc:
                cand, cand_sc = cand_tmp, cand_sc
                break
            alpha *= 0.5

        # if no improvement in the line‑search, keep the last tried point
        if cand is None:
            cand, cand_sc = cand_tmp, cand_sc

        # ----- accept / reject (occasionally allow uphill moves) -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            # Metropolis‑type uphill acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- occasional random rotation (helps escape edge effects) -----
        if it % 800 == 0:

            rot_sc = 2.0 * n * rot_conv.max()
            if rot_sc < cur_sc:
                cur, cur_sc = rot, rot_sc
                if rot_sc < best_sc:
                    best_sc, best_vec = rot_sc, rot.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 4000:
            # move a fraction of mass from a high‑gradient index to a low‑gradient one
            i_max = int(np.argmax(grad))
            i_min = int(np.argmin(grad))
            if cur[i_max] > 0.0 and i_max != i_min:
                delta = cur[i_max] * rng.random() * 0.45
                tmp = cur.copy()
                tmp[i_max] -= delta
                tmp[i_min] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0

        # ----- early exit if target already beaten -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate encountered (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
