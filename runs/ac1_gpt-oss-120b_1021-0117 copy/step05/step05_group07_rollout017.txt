def _project_simplex(v):
    # v is np.ndarray
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    # compute rho via condition
    cond = u - css / np.arange(1, n+1)
    rho = np.where(cond > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        cur = rng.random(n, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)   # sum=1, nonnegative
    n = cur.shape[0]

    # Precompute convenient indices
    L = 1 << ((3 * n - 2).bit_length())
    idx = (2 * n - 2) - np.arange(n)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_opt = 1e-8
    lr0 = 0.5

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # Softmax temperature schedule
    eps0 = 1.0
    eps_decay = 0.995
    eps_min = 1e-12

    # bookkeeping
    # compute initial score
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()
    cur_sc = 2.0 * n * max_val
    best_sc = cur_sc
    best_vec = cur.copy()

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # recompute conv and gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_sc = 2.0 * n * max_val

        # Softmax weights
        eps = max(eps_min, eps0 * (eps_decay ** it))
        # compute w safely
        diff = conv_full - max_val
        # to avoid overflow, we compute exp( diff / eps )
        # diff <= 0, so exponent <=0
        w = np.exp(diff / eps)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isinf(w_sum) or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / math.sqrt(it)  # decay
        step = lr / (np.sqrt(v_hat) + eps_opt)
        cand = cur - step * m_hat
        # project
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_max = cand_conv.max()
        cand_sc = 2.0 * n * cand_max

        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            # occasional uphill move
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_sc = cand_sc
            no_improve += 1

        # periodic symmetrization
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # random mass transfer if stuck
        if no_improve > 5000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                no_improve = 0

        if best_sc < target:
            break

    # Final return as list
    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project v onto the probability simplex {x >= 0, sum(x) = 1}.
    Returns a new array with the same shape as v.
    """
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    cond = u - css / np.arange(1, n + 1)
    rho = np.where(cond > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients that (hopefully)
    yields a low AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the parent sequence if available, otherwise random.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)              # enforce sum = 1, non‑negative
    n = cur.size

    # ------------------------------------------------------------------
    # FFT set‑up (length is next power of two ≥ 3n‑2)
    # ------------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())
    idx = (2 * n - 2) - np.arange(n)        # indices needed for gradient

    # ------------------------------------------------------------------
    # Adam optimiser state
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_opt = 1e-8
    lr0 = 0.5                               # base learning rate

    m = np.zeros_like(cur)                  # first moment
    v = np.zeros_like(cur)                  # second moment

    # soft‑max temperature schedule for the max‑convolution
    eps0 = 1.0
    eps_decay = 0.995
    eps_min = 1e-12

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()
    cur_sc = 2.0 * n * max_val               # current bound (sum = 1)
    best_sc = cur_sc
    best_vec = cur.copy()

    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_sc = 2.0 * n * max_val

        # ----- soft‑max weights over convolution positions -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        diff = conv_full - max_val                # ≤ 0
        w = np.exp(diff / eps)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the (soft) max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]               # size n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)                  # simple decay
        step = lr / (np.sqrt(v_hat) + eps_opt)

        cand = cur - step * m_hat
        cand = _project_simplex(cand)             # simplex projection

        # ----- evaluate candidate -----
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_max = cand_conv.max()
        cand_sc = 2.0 * n * cand_max

        # ----- accept/reject -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- occasional symmetrisation (helps with Toeplitz structure) -----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 5000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
