import time, math, numpy as np

# ---------- helper utilities ----------
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_and_fft(a, L):
    """ Return linear convolution (length 2n-1), its maximum and the FFT of a. """
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    n = a.shape[0]
    conv = conv_full[: 2 * n - 1]
    return conv, conv.max(), a_fft

def _gradient(a_fft, w, L, n):
    """ Gradient of weighted max (sub‑gradient of max‑convolution) """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)          # slice that matches derivative indices
    return 2.0 * grad_full[idx]

    start = time.monotonic()

    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initial vector ----------------------------------------------------

    else:
        # fallback: a random positive vector of moderate length
        a = rng.random(2048, dtype=np.float64)

    a = _project_simplex(np.maximum(a, 0.0))
    n = a.shape[0]
    # FFT length comfortably larger than linear convolution length
    L = 1 << ((3 * n - 2).bit_length())

    # ----- optimisation state ------------------------------------------------
    beta_mom = 0.9               # momentum coefficient
    step0 = 0.6                  # initial step scale
    momentum = np.zeros_like(a)
    best_vec = a.copy()
    conv, cur_max, a_fft = _conv_and_fft(a, L)
    cur_score = float(2.0 * n * cur_max)          # sum(a)=1 => denominator =1
    best_score = cur_score

    no_improve = 0
    iteration = 0
    target = 1.50274349

    # ----- main loop ---------------------------------------------------------
    while True:
