import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Score for a vector already normalised to sum 1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initialise from the provided parent if present, otherwise random
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.size
    # one FFT length large enough for both a*a and a*w convolutions
    L = 1 << ((3 * n - 2).bit_length())

    best = cur.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)
    beta = 0.9          # momentum factor
    step0 = 1.0         # initial step scale
    no_improve = 0
    it = 0
    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution of current vector with itself -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- smooth approximation of the max (softmax) -----
        # temperature decays with sqrt(iteration)
        temp = max_val * 0.02 / math.sqrt(it)
        if temp < 1e-12:
            temp = 1e-12
        # weights concentrate on near‑max entries
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.ones_like(conv) / conv.size
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        step = step0 / math.sqrt(it)
        cand = None
