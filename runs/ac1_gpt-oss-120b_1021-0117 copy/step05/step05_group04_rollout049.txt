# ------------------------------------------------------------
# Basic linear‑algebra utilities (matrix‑free)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1."""
    s = v.sum()
    if s == 0.0:
        # fallback uniform vector (should never happen)
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _conv_fft(x: np.ndarray, L: int):
    """
    Compute FFT of x (zero‑padded to length L), the linear convolution
    x*x and its maximum value.
    Returns (fft_x, conv, max(conv)).
    """
    fft_x = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fft_x * fft_x, n=L)[: 2 * x.size - 1]
    return fft_x, conv, conv.max()


def _softmax_weights(conv: np.ndarray, max_val: float, eps: float) -> np.ndarray:
    """Soft‑max over convolution values with temperature = eps."""
    w = np.exp((conv - max_val) / eps)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size, dtype=np.float64)
    return w / s


def _gradient(fft_x: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of the smoothed max‐convolution objective.
    Implements g = 2 * (x * w_rev)_{(2n‑2)‑i}.
    """
    w_rev = w[::-1]
    fft_w = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(fft_x * fft_w, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients that (hopefully)
    yields a low AC1 bound.  No assumptions are made about the existence of
    a parent construction; if `height_sequence_1` is present it is used as
    a starting point.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # Initial vector
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
    else:
        # fallback: random positive vector of moderate size
        cur = rng.random(2048, dtype=np.float64)
    cur = _normalize(cur)
    n = cur.size

    # FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------
    eps0 = 0.5                # initial soft‑max temperature
    eps_decay = 0.9985        # exponential decay factor per iteration
    eps_min = 1e-9
    step0 = 1.0               # base step size for exponentiated update
    beta = 0.96               # momentum coefficient
    momentum = np.zeros_like(cur)

    # best solution so far
    fft_cur, conv_cur, cur_max = _conv_fft(cur, L)
    best_vec = cur.copy()
    best_max = cur_max
    best_score = 2.0 * n * best_max   # sum(cur) == 1, so denominator = 1
    target_score = 1.50274349

    it = 0
    no_improve = 0

    # --------------------------------------------------------
    # Main optimisation loop (matrix‑free, FFT‑based)
    # --------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- current convolution & max -----
        fft_cur, conv_cur, cur_max = _conv_fft(cur, L)

        # ----- smoothing weight (soft‑max or exact) -----
        eps = eps0 * (eps_decay ** it)
        if eps < eps_min:
            eps = eps_min

        if eps > 1e-4:
            w = _softmax_weights(conv_cur, cur_max, eps)
        else:
            # exact sub‑gradient: uniform over all maximal positions
            mask = conv_cur == cur_max
            cnt = mask.sum()
            if cnt == 0:
                w = np.full_like(conv_cur, 1.0 / conv_cur.size, dtype=np.float64)
            else:
                w = mask.astype(np.float64) / cnt

        # ----- gradient via FFT (matrix‑free) -----
        grad = _gradient(fft_cur, w, n, L)

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- exponentiated gradient step -----
        step = step0 / math.sqrt(it)          # diminishing step size
        exponent = -step * momentum
        exponent = np.clip(exponent, -50.0, 50.0)   # avoid overflow
        cand = cur * np.exp(exponent)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            cand = cur.copy()
        else:
            cand = cand / cand_sum

        # ----- evaluate candidate -----
        _, _, cand_max = _conv_fft(cand, L)

        # ----- acceptance -----
        if cand_max < cur_max - 1e-12:
            cur = cand
            cur_max = cand_max
            no_improve = 0
            if cand_max < best_max:
                best_max = cand_max
                best_vec = cand.copy()
                best_score = 2.0 * n * best_max
                if best_score < target_score:
                    break
        else:
            no_improve += 1
            # occasional uphill move (Metropolis rule)
            if rng.random() < math.exp(-(cand_max - cur_max) / (step + 1e-12)):
                cur = cand
                cur_max = cand_max

        # ----- periodic symmetrisation (helps due to Toeplitz symmetry) -----
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _normalize(sym)
            _, _, sym_max = _conv_fft(sym, L)
            if sym_max < cur_max:
                cur = sym
                cur_max = sym_max
                if sym_max < best_max:
                    best_max = sym_max
                    best_vec = sym.copy()
                    best_score = 2.0 * n * best_max
                    if best_score < target_score:
                        break
                no_improve = 0

        # ----- random mass transfer when progress stalls -----
        if no_improve > 5000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _normalize(tmp)
                _, _, tmp_max = _conv_fft(tmp, L)
                if tmp_max < cur_max:
                    cur = tmp
                    cur_max = tmp_max
                    no_improve = 0
                    if tmp_max < best_max:
                        best_max = tmp_max
                        best_vec = tmp.copy()
                        best_score = 2.0 * n * best_max
                        if best_score < target_score:
                            break
                else:
                    no_improve = 0

        # ----- hard restart after a long stagnation -----
        if no_improve > 20000:
            cur = rng.random(n, dtype=np.float64)
            cur = _normalize(cur)
            momentum = np.zeros_like(cur)
            no_improve = 0

    # --------------------------------------------------------
    # Return the best coefficient list found
    # --------------------------------------------------------
    return [float(v) for v in best_vec]
