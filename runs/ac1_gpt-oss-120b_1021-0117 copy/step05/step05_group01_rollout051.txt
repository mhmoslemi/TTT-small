    deadline = start + max(0.0, float(budget_s)-0.5)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2500
        a = rng.random(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()
    n = a.size

    # FFT length:
    L = 1 << ((3*n-2).bit_length())
    # allocate placeholders for best
    best_a = a.copy()
    # compute initial score:
    def score_vec(v):
        # v sum normalized to 1
        # compute conv max
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return 2.0 * n * conv.max()

    best_score = score_vec(a)

    # parameter schedule
    step0 = 1.0
    beta = 0.9
    momentum = np.zeros_like(a)

    # tolerance for max equality:
    atol = 1e-12

    # main loop
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # compute conv and max
        af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(af * af, n=L)[:2*n-1]
        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # optional early exit if below target
            if best_score < 1.50274349:
                break

        # compute mask of max entries
        mask = (np.abs(conv_full - cur_max) <= atol).astype(np.float64)  # length 2n-1

        # compute subgradient via correlation
        # FFT of mask:
        mask_fft = np.fft.rfft(mask, n=L)
        # correlation (linear) a ⋆ mask = ifft(af * conj(mask_fft))
        corr = np.fft.irfft(af * np.conj(mask_fft), n=L)
        # extract gradient entries
        offset = len(mask) - 1  # 2n-2
        g = 2.0 * corr[offset: offset + n]   # length n

        # momentum update
        momentum = beta * momentum + (1 - beta) * g

        # step size schedule, decays sqrt
        alpha = step0 / math.sqrt(it)

    """Euclidean projection of v onto simplex {x >=0, sum x = z}."""
    if z < 0:

# Euclidean projection onto the simplex {x >= 0, sum(x) = z}

def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum z."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    n = v.size

        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
