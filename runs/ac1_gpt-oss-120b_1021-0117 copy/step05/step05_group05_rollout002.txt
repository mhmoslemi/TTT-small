import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(a):
    """Score used during the search (sum of a is 1, so denominator = 1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ----- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = rng.random(1024, dtype=np.float64)
    a = _project_simplex(np.maximum(a, 0.0))

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t_iter = 0

    # temperature schedule (softmax smoothing)
    T = 1.0
    T_min = 1e-4
    T_decay = 0.93  # multiplicative decay every decay_step
    decay_step = 250

    best_a = a.copy()
    best_sc = _score(best_a)

    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        t_iter += 1

        # ----- convolution -----
        af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(af * af, n=L)
        conv = conv_full[: 2 * n - 1]

        # ----- temperature update -----
        if t_iter % decay_step == 0:
            T = max(T_min, T * T_decay)

        # ----- softmax weights -----
        maxc = conv.max()
        # stable softmax
        exps = np.exp((conv - maxc) / T)
        sum_exps = exps.sum()
        if sum_exps == 0.0:  # numerically unlikely but safe
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = exps / sum_exps

        # ----- gradient of smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]  # subgradient of smoothed max

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = 0.2 * m_hat / (np.sqrt(v_hat) + eps)  # scalar 0.2 as base LR
        a_new = a - step
        a_new = _project_simplex(a_new)

        # ----- optional symmetry enforcement -----
        if t_iter % 400 == 0:
            sym = (a_new + a_new[::-1]) * 0.5
            a_new = _project_simplex(sym)

        # ----- evaluate -----
        cur_sc = _score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # ----- stagnation handling -----
        if no_improve > 5000:
            # small random perturbation then re‑project
            perturb = rng.normal(scale=0.02, size=n)
            a = _project_simplex(a + perturb)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # ----- early exit -----
        if best_sc < target:
            break

    # Return the best found sequence as a plain Python list
    return [float(x) for x in best_a]
