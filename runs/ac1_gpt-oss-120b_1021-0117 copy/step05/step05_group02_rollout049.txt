        v = np.zeros_like(cur)

        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]

        // Compute score
        cur_score = 2.0 * n * max_val

        // Compute softmax weights w
        eps_cur = max(min_eps, eps_temp)
        w = np.exp((conv - max_val) / eps_cur)

        // Compute gradient of smoothed max wrt a:

        w_rev = w[::-1]

        # compute gradient entries for a_i: indices (2n-2)-i
        idx = (2*n - 2) - np.arange(n)

import time, math, numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence that (hopefully) improves the AC1 bound.
    Uses an Adam‑style projected gradient descent with soft‑max smoothing of the
    convolution maximum.  The best vector seen before the time limit is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initialise -------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)
    else:
        n0 = 1024
        cur = np.full(n0, 1.0 / n0, dtype=np.float64)

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # FFT length ≥ 2n‑1, power of two

    # ----- helper to compute max convolution --------------------------------
    def _conv_max(a_fft):
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return conv.max()

    # ----- initial score ----------------------------------------------------
    a_fft = np.fft.rfft(cur, n=L)
    cur_max = _conv_max(a_fft)
    cur_score = 2.0 * n * cur_max

    best_vec = cur.copy()
    best_sc = cur_score

    # ----- Adam hyper‑parameters --------------------------------------------
    base_lr = 0.5
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # ----- soft‑max temperature ---------------------------------------------
    eps_temp = 0.5
    eps_decay = 0.9998
    min_eps = 1e-6

    # ----- other controls ----------------------------------------------------
    no_improve = 0
    target = 1.50274349
    it = 1

    while time.monotonic() < deadline:
        # ----- soft‑max weighting of the convolution -------------------------
        eps_cur = max(min_eps, eps_temp)
        # recompute full convolution for current a_fft
