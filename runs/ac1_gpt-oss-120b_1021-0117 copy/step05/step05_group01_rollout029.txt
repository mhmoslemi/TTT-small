import time, math, numpy as np

def _simplex_proj(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex {x>=0, sum(x)=1} by simple renormalisation."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 1e-12:
        v /= s
    else:
        v[:] = 1.0 / v.size
    return v

def _full_conv_fft(a: np.ndarray, L: int):
    """Return full linear convolution a*a (length 2n-1) and its FFT."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv, af

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
