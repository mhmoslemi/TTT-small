   import time, math, numpy as np
   rng = np.random.default_rng(seed)
   start = time.monotonic()
   deadline = start + max(0.0, float(budget_s) - 0.1)  # small safety

        a0 = a0 / a0.sum()
   n = a0.size
   L = 1 << ((2*n - 1).bit_length())
   cur = a0.copy()
   cur_sc = _score(cur)
   best = cur.copy()
   best_sc = cur_sc

   target = 1.50274349

   sym_interval = 300

        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]

        idx = conv_full.argmax()

        # subgradient

        idx_arr = np.arange(n)
        src = idx - idx_arr
        valid = (src >= 0) & (src < n)
        g = np.zeros_like(cur)

        # learning rate
        eta = 0.5 / math.sqrt(it)

        # candidate 1: projected GD
        cand1 = cur - eta * g

        sc1 = _score(cand1)

        else:

        sc2 = _score(cand2)

        # Choose best among current and candidates
        improved = False

                cur_sc = sc1

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast AC1 bound for a sequence with sum(a)=1  (returns 2*n*max(conv))
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
