import time, math, numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex: sum(x)=1, x>=0
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Core optimisation routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1
    n = cur.size

    # ------------------------------------------------------------
    # FFT planning – length big enough for full convolution (len = 3n-2)
    # ------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())   # next power of two
    # index slice to extract the needed n entries after circular convolution
    idx = (2 * n - 2) - np.arange(n)

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    a_fft = np.fft.rfft(cur, n=L)
    full_conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = full_conv.max()
    best_sc = 2.0 * n * max_val          # 2*n*max_b (sum = 1)
    best_vec = cur.copy()
    cur_sc = best_sc

    # optimisation parameters
    step0 = 0.4          # base step size
    beta = 0.94          # momentum factor
    momentum = np.zeros_like(cur)
    eps0 = 0.8           # initial temperature for the soft‑max
    eps_decay = 0.985    # slower cooling – more accurate gradients
    eps_min = 1e-12
    it = 0
    no_improve = 0
    target = 1.50274349   # target bound to chase (if reachable)

    # ------------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- smoothed weights for the (soft) max -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps > 1e-12:
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum
        else:
            # hard max: uniform weight on all maximal positions
            w = np.zeros_like(conv_full)
            mask = conv_full == max_val
            cnt = mask.sum()
            if cnt:
                w[mask] = 1.0 / cnt

        # ----- gradient of the (soft) max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]          # size n

        # ----- Nesterov‑style momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad
        lookahead = cur - step0 / math.sqrt(it + 1) * momentum
        lookahead = _project_simplex(lookahead)

        # ----- line‑search (back‑tracking) -----
        base_alpha = step0 / math.sqrt(it + 1)
        alpha = base_alpha
        cand = None
        cand_sc = None
        for _ in range(10):
            cand_tmp = cur - alpha * momentum
            cand_tmp = _project_simplex(cand_tmp)

            cand_fft = np.fft.rfft(cand_tmp, n=L)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
            cand_max = cand_conv.max()
            cand_sc = 2.0 * n * cand_max

            if cand_sc <= cur_sc:
                cand, cand_sc = cand_tmp, cand_sc
                break
            alpha *= 0.5
        # if no improvement, keep the last tried candidate
        if cand is None:
            cand, cand_sc = cand_tmp, cand_sc

        # ----- accept / reject (with occasional uphill moves) -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- symmetrisation every 250 iterations (helps Toeplitz) -----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stagnating -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # As a safe fallback, also consider the perfectly uniform distribution
    # (the theoretical optimum for the given objective).
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_fft = np.fft.rfft(uniform, n=L)
    uniform_conv = np.fft.irfft(uniform_fft * uniform_fft, n=L)[: 2 * n - 1]
    uniform_sc = 2.0 * n * uniform_conv.max()
    if uniform_sc < best_sc:
        best_vec = uniform
        best_sc = uniform_sc

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
