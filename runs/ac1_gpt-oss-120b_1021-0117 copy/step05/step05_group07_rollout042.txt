import time, math, numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex, returning a new array."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Score evaluation (identical to the provided evaluator, but lightweight)
# ----------------------------------------------------------------------
def _score(a: np.ndarray, n: int, L: int) -> float:
    """Return 2*n*max(conv)/(sum(a)**2) for a given vector a."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin so the harness can still collect the answer
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # sum = 1
    n = cur.size

    # ------------------------------------------------------------------
    # FFT planning – zero‑pad to a power of two >= 2n‑1
    # ------------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())
    # indices for extracting the linear‑convolution part of the gradient
    idx = (2 * n - 2) - np.arange(n)

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(cur, n, L)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)
    beta = 0.95          # momentum coefficient
    base_step = 0.5      # initial step magnitude
    it = 0
    no_improve = 0
    target = 1.50274349

    # occasional uniform template (helps escaping plateaus)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- gradient of the hard max -----
        # consider every index whose value is (almost) the maximum
        tol = max_val * 1e-12 + 1e-15
        mask = np.abs(conv - max_val) <= tol
        w = mask.astype(np.float64)               # binary weight vector
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]               # size n

        # ----- momentum update (Nesterov‑style look‑ahead) -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- step size (inverse‑sqrt decay) with simple back‑track -----
        step = base_step / math.sqrt(it)
        cand = None
        cand_sc = None
        for _ in range(5):
            cand_tmp = cur - step * momentum
            cand_tmp = _project_simplex(cand_tmp)
            cand_sc_tmp = _score(cand_tmp, n, L)
            if cand_sc_tmp <= cur_sc:
                cand, cand_sc = cand_tmp, cand_sc_tmp
                break
            step *= 0.5
        if cand is None:
            # fall back to the last tried candidate (may be worse)
            cand, cand_sc = cand_tmp, cand_sc_tmp

        # ----- accept / reject -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            # occasional uphill move (Metropolis style)
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps with the Toeplitz structure) -----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- random blend with uniform when stagnating -----
        if no_improve > 1200:
            mix = min(0.8, 0.1 + 0.0005 * (no_improve - 1200))
            cand_tmp = (1.0 - mix) * cur + mix * uniform
            cand_tmp = _project_simplex(cand_tmp)
            cand_sc_tmp = _score(cand_tmp, n, L)
            if cand_sc_tmp < cur_sc:
                cur, cur_sc = cand_tmp, cand_sc_tmp
                no_improve = 0
                if cand_sc_tmp < best_sc:
                    best_sc, best_vec = cand_sc_tmp, cand_tmp.copy()
            else:
                no_improve += 1

        # ----- random mass‑transfer when really stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, n, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
