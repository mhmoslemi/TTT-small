import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = z}."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Initial vector – use the parent if present, else a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1 (simplex)

    n = a.size
    # FFT length sufficient for convolutions up to length 3n‑2
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------
    # Optimisation bookkeeping
    # ------------------------------------------------------------
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # hyper‑parameters
    step0 = 0.5
    alpha0 = 1.0
    alpha_rate = 0.005          # increase of soft‑max temperature per iteration
    alpha_max = 30.0
    target = 1.50274349
    sym_interval = 250
    no_improve = 0

    # initial score
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    best_sc = 2.0 * n * conv_full.max()
    best_vec = a.copy()
    cur_sc = best_sc
    it = 0

    # ------------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        cur_max = conv_full.max()
        cur_sc = 2.0 * n * cur_max

        # ----- keep the best seen so far -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- smooth approximation of the max (soft‑max) -----
        alpha = alpha0 + alpha_rate * it
        if alpha > alpha_max:
            alpha = alpha_max
        shift = cur_max
        p = np.exp(alpha * (conv_full - shift))   # values ≤ 1
        p_sum = p.sum()
        if p_sum == 0.0:
            p = np.full_like(p, 1.0 / p.size)
        else:
            p = p / p_sum

        # ----- gradient of the smooth max (matrix‑free) -----
        a_rev = a[::-1]
        a_rev_fft = np.fft.rfft(a_rev, n=L)
        p_fft = np.fft.rfft(p, n=L)
        grad_full = np.fft.irfft(p_fft * a_rev_fft, n=L)[: 3 * n - 2]
        g = 2.0 * grad_full[n - 1 : n - 1 + n]      # size n

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step_vec = step0 / (np.sqrt(v_hat) + eps)

        # ----- back‑tracking line search (few attempts) -----
        cand = None
        cand_sc = None

            cand_sc = 2.0 * n * conv_tmp.max()

        if it % sym_interval == 0:

            a_sym_fft = np.fft.rfft(a_sym, n=L)

                    no_improve = 0

            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.4
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
            no_improve = 0

        # safety check for the deadline
        if time.monotonic() >= deadline - 0.05:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
