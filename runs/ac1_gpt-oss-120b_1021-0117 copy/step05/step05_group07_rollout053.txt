import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _softmax_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """Return a probability vector concentrating on the maximal entries of conv."""
    max_val = conv.max()
    if eps > 1e-12:
        w = np.exp((conv - max_val) / eps)
        s = w.sum()
        if s == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= s
    else:
        w = np.zeros_like(conv)
        mask = conv == max_val
        cnt = mask.sum()
        if cnt > 0:
            w[mask] = 1.0 / cnt
    return w
