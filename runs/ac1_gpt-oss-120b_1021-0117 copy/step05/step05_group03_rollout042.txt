    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):

    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5  # step magnitude
    # temperature schedule
    temp0 = 0.001
    temp_decay = 0.9995

    no_improve = 0

        conv = conv_full[: 2 * n - 1]

        w_sum = w.sum()
        if w_sum == 0.0:

        # gradient via weight

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection – Euclidean projection onto {x >= 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Score for a vector with sum = 1  (returns 2 * n * max(convolution))
# ----------------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())
