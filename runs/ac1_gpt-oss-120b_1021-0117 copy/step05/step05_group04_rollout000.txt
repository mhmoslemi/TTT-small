import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Make non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _conv_max_fft(a: np.ndarray, L: int):
    """Return (max conv, FFT of a, full conv) for a*a using FFT length L."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _softmax_weights(conv: np.ndarray, max_val: float, eps: float) -> np.ndarray:
    """Soft‑max over convolution values (temperature = eps)."""
    w = np.exp((conv - max_val) / eps)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size, dtype=np.float64)
    return w / s


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of the smoothed max‑conv objective.
    g_i = 2 * (a * w_rev)_{(2n‑2)‑i}.
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # Initialise from parent if available, otherwise random
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2048, dtype=np.float64)

    cur = _normalize(cur)
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())          # power‑of‑two FFT length

    # --------------------------------------------------------
    # Initial evaluation
    # --------------------------------------------------------
    cur_max, cur_fft, conv = _conv_max_fft(cur, L)
    best_max = cur_max
    best_vec = cur.copy()
    best_score = 2.0 * n * best_max               # sum is 1, so denominator = 1
    target_score = 1.50274349

    # --------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------
    step0 = 0.70
    eps0 = 0.80
    eps_decay = 0.9995
    eps_min = 1e-14
    momentum = np.zeros_like(cur)
    beta = 0.90
    no_improve = 0
    it = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution metrics -----
        cur_max, cur_fft, conv = _conv_max_fft(cur, L)
        cur_score = 2.0 * n * cur_max

        # ----- keep best -----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_max = cur_max
            best_vec = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- soft‑max temperature -----
        eps = max(eps_min, eps0 * (eps_decay ** it))

        # ----- weights and gradient -----
        w = _softmax_weights(conv, cur_max, eps)
        grad = _gradient(cur_fft, w, n, L)

        # ----- momentum -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- candidate generation (exponentiated update) -----
        step = step0 / math.sqrt(it)
        exponent = -step * momentum
        exponent = np.clip(exponent, -50.0, 50.0)   # avoid overflow
        cand = cur * np.exp(exponent)
        cand = _normalize(cand)
        cand_max, _, _ = _conv_max_fft(cand, L)

        # ----- acceptance test -----
        if cand_max < cur_max - 1e-15:
            cur = cand
            cur_max = cand_max
            # immediate best update
            cand_score = 2.0 * n * cand_max
            if cand_score < best_score:
                best_score = cand_score
                best_max = cand_max
                best_vec = cand.copy()
            no_improve = 0
        else:
            # occasional uphill move (Metropolis style)
            prob = math.exp(-(cand_max - cur_max) / (step + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_max = cand_max

        # ----- periodic symmetrisation -----
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _normalize(sym)
            sym_max, _, _ = _conv_max_fft(sym, L)
            if sym_max < cur_max:
                cur = sym
                cur_max = sym_max
                sym_score = 2.0 * n * sym_max
                if sym_score < best_score:
                    best_score = sym_score
                    best_max = sym_max
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 1500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.25
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _normalize(tmp)
                tmp_max, _, _ = _conv_max_fft(tmp, L)
                if tmp_max < cur_max:
                    cur = tmp
                    cur_max = tmp_max
                    tmp_score = 2.0 * n * tmp_max
                    if tmp_score < best_score:
                        best_score = tmp_score
                        best_max = tmp_max
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    # small chance to accept non‑improving move
                    if rng.random() < 0.05:
                        cur = tmp
                        cur_max = tmp_max
                        no_improve = 0

        # ----- occasional mixing with uniform distribution -----
        if it % 500 == 0 and rng.random() < 0.3:
            uniform = np.full_like(cur, 1.0 / n)
            mix = 0.1
            cand_mix = (1.0 - mix) * cur + mix * uniform
            cand_mix = _normalize(cand_mix)
            mix_max, _, _ = _conv_max_fft(cand_mix, L)
            if mix_max < cur_max:
                cur = cand_mix
                cur_max = mix_max
                mix_score = 2.0 * n * mix_max
                if mix_score < best_score:
                    best_score = mix_score
                    best_max = mix_max
                    best_vec = cand_mix.copy()
                no_improve = 0

        # ----- early termination if target reached -----
        if best_score < target_score:
            break

        # ----- time guard -----
        if time.monotonic() >= deadline:
            break

    # --------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
