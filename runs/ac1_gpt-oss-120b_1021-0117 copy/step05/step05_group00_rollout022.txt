def _make_exponential(n: int) -> np.ndarray:
    """Return a normalized exponential decreasing sequence of length n."""
    # Choose λ so that (1-λ) ~ 1/(2n); this yields a very low AC1 bound.
    lam = 1.0 - 1.0 / (2.0 * n)
    # geometric progression λ^i
    powers = np.power(lam, np.arange(n, dtype=np.float64))
    # normalize to sum 1
    seq = powers / powers.sum()
    return seq

            best_seq = parent
            best_score = parent_score

    # Very cheap local refinement: a few random redistributions
    rng = np.random.default_rng(seed)
    # The refinement loop respects the overall time budget.
    # It performs simple mass‑transfer moves that tend to lower the peak
    # of the autocorrelation while preserving the sum.
    for _ in range(2000):
        if time.time() >= deadline:
            break
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a small random fraction from the larger entry to the smaller
        if best_seq[i] > best_seq[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        delta = best_seq[src] * rng.random() * 0.1  # up to 10% of source mass
        if delta <= 0:
            continue
        new_seq = best_seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta
        # project onto the simplex (ensures non‑negativity and sum = 1)
        new_seq = np.maximum(new_seq, 0.0)
        new_seq_sum = new_seq.sum()
        if new_seq_sum == 0:
            continue
        new_seq /= new_seq_sum
        sc = evaluate_sequence(new_seq.tolist())
        if sc < best_score:
            best_seq = new_seq
            best_score = sc

    # Return as plain Python floats
    return [float(x) for x in best_seq]
