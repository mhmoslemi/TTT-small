def _project_simplex(v, z=1.0):
    n = v.size
    if z < 0:
        raise ValueError
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
