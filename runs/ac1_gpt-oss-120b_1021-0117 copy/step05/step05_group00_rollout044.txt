def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3) # small safety

    # Initial vector:
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0:
            a0 = np.ones_like(a0) / a0.size
            s = 1.0
        a0 = a0 / s
    else:
        n = 2000
        t = np.arange(n, dtype=np.float64)
        a0 = np.exp(-t / 50.0)
        a0 = a0 / a0.sum()
    n = a0.shape[0]
    cur = a0.copy()
    best_vec = cur.copy()
    # compute current score and track best
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    max_val = conv_full.max()
    cur_sc = 2.0 * n * max_val
    best_sc = cur_sc

import time
import math
import numpy as np

# ------------------------------------------------------------
# projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the simplex. """
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
