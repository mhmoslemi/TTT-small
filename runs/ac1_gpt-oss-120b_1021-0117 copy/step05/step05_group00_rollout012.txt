import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Compute 2*n*max(convolution) for a (assuming sum(a)=1)."""
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length() + 1)  # a bit larger power of two for safety
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())
