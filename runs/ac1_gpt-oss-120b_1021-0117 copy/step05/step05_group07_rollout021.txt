import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---------- initial vector ----------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)                 # sum = 1
    cur = (cur + cur[::-1]) * 0.5                # enforce symmetry
    cur = cur / cur.sum()                        # renormalise exactly

    n = cur.size
    # FFT length (power of two) for linear convolution of length 2n‑1
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)            # slice for gradient extraction

    # ---------- bookkeeping ----------
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    best_sc = 2.0 * n * conv_full.max()
    best_vec = cur.copy()
    cur_sc = best_sc

    # ---------- optimizer state ----------
    momentum = np.zeros_like(cur)
    # hyper‑parameters
    step0 = 0.6
    beta = 0.9
    eps0 = 0.5
    eps_decay = 0.9995
    eps_min = 1e-8
    target = 1.50274349
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # current convolution and its maximum
        a_fft = np.fft.rfft(cur, n=L)

        max_val = conv_full.max()
        cur_sc = 2.0 * n * max_val

        if cur_sc < best_sc:
