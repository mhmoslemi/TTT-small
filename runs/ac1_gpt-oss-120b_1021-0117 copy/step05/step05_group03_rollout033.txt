import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast score for a vector already on the simplex (sum = 1)
# Returns 2 * n * max(conv(a, a))
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep margin for return

    # --------------------------------------------------------
    # Initialise – use the parent if present, otherwise uniform
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback length – reasonably large to give a good bound
        cur = np.full(2000, 1.0 / 2000, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1
    n = cur.size

    # --------------------------------------------------------
    # FFT length – power of two >= 2*n‑1 (full convolution)
    # --------------------------------------------------------
    L = 1 << ((2 * n - 2).bit_length())

    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # algorithmic parameters
    step0 = 1.2               # initial step magnitude
    beta = 0.88               # momentum factor
    momentum = np.zeros_like(cur)
    no_improve = 0
    it = 0
    target = 1.50274349
    sym_period = 300          # periodic symmetrisation
    stuck_thr = 2500          # when to apply a random mass‑transfer
    tau0 = 0.02               # initial smoothing temperature for sub‑gradient

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current vector -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- smoothed sub‑gradient of the max -----
        tau = max(1e-9, tau0 / math.sqrt(it))
        diff = conv - max_val                     # <= 0
        # softmax weighting; values far from max get exponentially small weight
        w = np.exp(diff / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # gradient of (soft‑max) convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # map to gradient w.r.t a
        g = 2.0 * grad_full[idx]

        momentum = beta * momentum + (1.0 - beta) * g

                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc <= target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
