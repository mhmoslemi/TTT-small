import time
import numpy as np

# ------------------------------------------------------------
# Simple utilities
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale non‑negative vector to have L1‑norm = 1."""
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform positive vector
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _conv_max(v: np.ndarray) -> tuple:
    """Return (max(conv), full convolution) for v * v."""
    conv = np.convolve(v, v)
    return conv.max(), conv


def _gradient_max(v: np.ndarray, peak_idx: int) -> np.ndarray:
    """
    Gradient of the peak convolution value with respect to v
    (sub‑gradient of the max, taken at the peak index).
    For non‑negative v the derivative is 2*v[peak_idx - i] where the
    partner index lies inside the array.
    """
    n = v.size
    i = np.arange(n)
    partner = peak_idx - i
    mask = (partner >= 0) & (partner < n)
    g = np.zeros_like(v)
    g[mask] = 2.0 * v[partner[mask]]
    return g
