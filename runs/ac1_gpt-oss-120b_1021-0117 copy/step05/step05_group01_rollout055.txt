import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x)=z}
# ------------------------------------------------------------
def _project_simplex(v, z=1.0):
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Full convolution a*a via FFT (length L >= 2n-1)
# ------------------------------------------------------------
def _conv_full_fft(a, L):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv


# ------------------------------------------------------------
# Gradient of smoothed max using softmax weights
# ------------------------------------------------------------
def _grad_softmax_max(a, conv, beta, Lg):
    n = a.size
    # softmax over convolution entries (numerically stable)
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()               # weights, sum to 1
    # gradient = 2 * (w_rev * a) evaluated at indices (2n-2-i)
    w_rev = w[::-1]                                 # length 2n-1
    wr_fft = np.fft.rfft(w_rev, n=Lg)
    a_fft = np.fft.rfft(a, n=Lg)
    corr = np.fft.irfft(wr_fft * a_fft, n=Lg)       # cross‑correlation
    idx_base = 2 * n - 2
    grad = 2.0 * corr[idx_base - np.arange(n)]
    return grad
