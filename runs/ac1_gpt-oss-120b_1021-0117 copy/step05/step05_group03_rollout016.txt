import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum=1, >=0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast internal score for a vector with sum=1:
#   2 * n * max(conv(a, a))
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    deadline = start + max(0.0, float(budget_s) - 0.5)

    # Initialise – use the parent if present, otherwise a random start
    # ------------------------------------------------------------------

        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s if s > 0 else cur
    else:
        cur = rng.random(1024, dtype=np.float64)
        cur = _project_simplex(cur)

    cur = _project_simplex(cur)            # enforce sum = 1
    n = cur.shape[0]

    # ------------------------------------------------------------
    # FFT length – power of two >= 2n‑1
    # ------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # optimisation parameters
    beta = 0.85               # momentum factor
    step0 = 1.0               # base step size
    momentum = np.zeros_like(cur)
    no_improve = 0
    it = 0
    target_sc = 1.50274349    # target value (sum already = 1)
    idx = (2 * n - 2) - np.arange(n, dtype=int)   # index map for gradient

    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        a_fft = np.fft.rfft(cur, n=L)

        conv = conv_full[: 2 * n - 1]

        cand = _project_simplex(cand)

        if cand_sc < best_sc:

    # ------------------------------------------------------------
