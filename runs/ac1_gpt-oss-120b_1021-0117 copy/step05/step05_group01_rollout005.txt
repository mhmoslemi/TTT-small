def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # init a
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2048
        a = rng.random(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()

    n = a.size
    # FFT length power of two >= 2n-1
    L = 1 << ((2 * n - 1).bit_length())

    # initial best
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    best_sc = 2.0 * n * max_val
    best_vec = a.copy()
    # store gradient momentum buffer
    momentum = np.zeros_like(a)

    # parameters
    base_step = 0.6
    beta = 0.85
    sym_interval = 250
    no_improve = 0
    it = 0
    target = 1.50274349

    # Helper to evaluate score quickly (using conv max)
    def score_of(vec):
        af = np.fft.rfft(vec, n=L)
        cf = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return 2.0 * n * cf.max()

    # Random search phase
    # allocate some of the budget: maybe 5% of time or 2000 iterations whichever earlier.
    # We'll use while loop checking time and number of samples.
    rand_iters = 0

            mask[rng.integers(0, n)] = True
        a_rand = rng.random(n) * mask
        a_rand = a_rand / a_rand.sum()
        sc = score_of(a_rand)
        if sc < best_sc:
            best_sc, best_vec = sc, a_rand.copy()
            a = a_rand  # restart from new good point
            momentum.fill(0.0)
            no_improve = 0
        # early exit if good enough
        if best_sc < target:
            return [float(x) for x in best_vec]

    # Main optimization loop
    while time.monotonic() < deadline:
        it += 1

        # compute conv and max
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        # update best
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_sc < target:
            break

        # gradient from exact max positions
        idxs = np.where(np.isclose(conv, max_val, atol=1e-12))[0]
        grad = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # i range
            i = np.arange(i_start, i_end + 1)
            grad[i] += 2.0 * a[k - i]

        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size
        alpha = base_step / math.sqrt(it)

        # backtracking line search
        cand = None
        cand_sc = None
        for _ in range(8):
            a_tmp = a - alpha * momentum
            a_tmp = _project_simplex(a_tmp)   # enforce simplex sum=1, nonneg
            sc_tmp = score_of(a_tmp)
            if sc_tmp <= cur_sc:
                cand, cand_sc = a_tmp, sc_tmp
                break
            alpha *= 0.5
        if cand is None:
            cand, cand_sc = a_tmp, sc_tmp

        # accept
        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            # occasional uphill with probability
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                a, cur_sc = cand, cand_sc

        # symmetrization
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            sc_sym = score_of(a_sym)
            if sc_sym < cur_sc:
                a, cur_sc = a_sym, sc_sym
                if sc_sym < best_sc:
                    best_sc, best_vec = sc_sym, a_sym.copy()
                no_improve = 0

        # random mass transfer if stuck
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.3
                a[i] -= delta
                a[j] += delta
                # ensure sum=1 (should already)
                a = a / a.sum()
                momentum.fill(0.0)
                no_improve = 0

    # finalize: ensure values within bounds (0..1000) and sum >=0.01
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    if best_vec.sum() < 0.01:
        # fallback: uniform
        best_vec = np.full_like(best_vec, 1.0 / best_vec.size)

    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with sum z."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Score = 2 * n * max(convolution) (sum of a is 1)
# ------------------------------------------------------------
def _score(vec: np.ndarray, L: int, n: int) -> float:
    af = np.fft.rfft(vec, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return 2.0 * n * conv_full.max()

# ------------------------------------------------------------
# Sub‑gradient of max‑convolution (exact: only indices attaining the max)
# ------------------------------------------------------------
def _grad_max_subgrad(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    n = a.size
    idxs = np.where(np.isclose(conv, max_val, atol=1e-12))[0]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # vector of indices i_start .. i_end
        i = np.arange(i_start, i_end + 1, dtype=int)
        # contribution 2 * a[k - i] to each g[i]
        g[i] += 2.0 * a[k - i]
    return g
