import time, math, numpy as np

def _project_simplex(v):
    """Projection onto simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2000:
            for _ in range(3):
                i = rng.integers(0, n)
                if cur[i] <= 1e-12:
                    continue
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                if j == i:
                    continue
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0
                    break

        # ----- occasional LP‑based refinement (matrix‑free) -----
        if _cvxpy_available and it % 500 == 0 and (deadline - time.monotonic()) > 5.0:
            # active set: largest convolution entries
            K = max(10, int(0.01 * n))
            top_idx = np.argpartition(-conv_full, K)[:K]

            g = cp.Variable(n, nonneg=True)
            constraints = []
            for k in top_idx:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                if i_start > i_end:
                    continue
                f_slice = cur[i_start:i_end + 1]
                g_slice = g[i_start:i_end + 1]
                constraints.append(cp.sum(cp.multiply(f_slice, g_slice)) <= max_val)

            # ensure zero‑lag constraint explicitly
            zero_shift = n - 1
            if zero_shift not in top_idx:
                constraints.append(cp.sum(cp.multiply(cur, g)) <= max_val)

            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                           abstol=1e-6, reltol=1e-6)
            except Exception:
                prob = None
            if prob is not None and g.value is not None:
                g_opt = np.asarray(g.value).flatten()
                mix = 0.12
                cand_lp = (1.0 - mix) * cur + mix * g_opt
                cand_lp = _project_simplex(cand_lp)
                cand_lp_fft = np.fft.rfft(cand_lp, n=L)
                cand_lp_conv = np.fft.irfft(cand_lp_fft * cand_lp_fft, n=L)[: 2 * n - 1]
                cand_lp_sc = 2.0 * n * cand_lp_conv.max()
                if cand_lp_sc < cur_sc:
                    cur, cur_sc = cand_lp, cand_lp_sc
                    if cand_lp_sc < best_sc:
                        best_sc, best_vec = cand_lp_sc, cand_lp.copy()
                    no_improve = 0

        # ----- deadline check -----
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as a plain Python list).
    # ------------------------------------------------------------------
    return [float(v) for v in best_vec]
