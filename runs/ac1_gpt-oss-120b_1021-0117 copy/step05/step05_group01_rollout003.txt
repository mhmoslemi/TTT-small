    start = time.monotonic()

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of vector v onto the simplex."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Convolution via FFT (returns the full length 2n‑1 vector)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Compute linear convolution a*a using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    n = a.size
    return conv_full[: 2 * n - 1]

# ------------------------------------------------------------
# Exact sub‑gradient of max‑convolution (sum over all argmax indices)
# ------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray, L: int, max_entries: int = 10):
    """Return (M, g) where M = max(a*a) and g is a sub‑gradient."""
    conv = _conv_fft(a, L)
    M = conv.max()
    # indices where conv attains the maximum (with tolerance)
    mask = np.isclose(conv, M, atol=1e-12)
    idxs = np.flatnonzero(mask)
    if idxs.size > max_entries:
        half = max_entries // 2
        idxs = np.concatenate((idxs[:half], idxs[-half:]))
    n = a.size
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # accumulate 2*a[k-i] for i in [i_start, i_end]
        for i in range(i_start, i_end + 1):
            g[i] += 2.0 * a[k - i]
    return M, g

    """
    Return a non‑empty list of non‑negative coefficients with a low AC1 bound.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------

    # ------------------------------------------------------------
    momentum = np.zeros_like(a)
