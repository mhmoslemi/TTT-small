import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    # shift by median to improve numerical stability
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast AC1 bound for a vector already normalised to sum=1
# ------------------------------------------------------------
def _score_vec(a: np.ndarray, L: int) -> float:
    """2*n*max(convolution) for a vector with sum≈1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()

    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length())     # power‑of‑two length for FFT

    best_vec = a0.copy()
    best_sc = _score_vec(best_vec, L)

    # --------------------------------------------------------
    # 1) Structured parametric search (exponential, power‑law, two‑level)
    # --------------------------------------------------------
    # ---- exponential family ------------------------------------------------
    # delta = 1 - r ; we sweep log‑space deltas from 1e‑6 to 1e‑1
    for delta in np.logspace(-6, -1, num=300):
        if time.monotonic() >= deadline:
            break
        r = 1.0 - delta
        if r <= 0.0:
            continue
        log_r = math.log(r)
        i = np.arange(n, dtype=np.float64)
        a = np.exp(i * log_r)               # raw exponential
        a /= a.sum()                         # project onto simplex (just normalise)
        sc = _score_vec(a, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = a.copy()

    # ---- fine tuning around the best delta ---------------------------------
    # locate the delta that produced best_vec (if it was exponential)
    # we keep a simple local grid around the last used delta
    # (the variable `delta` still holds the last value; use its neighbourhood)
    if 'delta' in locals():
        delta_center = delta
        for delta in np.logspace(math.log10(max(delta_center * 0.3, 1e-9)),
                                 math.log10(min(delta_center * 3.0, 0.5)),
                                 num=100):
            if time.monotonic() >= deadline:
                break
            r = 1.0 - delta
            if r <= 0.0:
                continue
            log_r = math.log(r)
            a = np.exp(i * log_r)
            a /= a.sum()
            sc = _score_vec(a, L)
            if sc < best_sc:
                best_sc = sc
                best_vec = a.copy()

    # ---- power‑law family ----------------------------------------------------
    for p in np.linspace(0.05, 5.0, 120):
        if time.monotonic() >= deadline:
            break
        i = np.arange(1, n + 1, dtype=np.float64)
        a = i ** (-p)
        a /= a.sum()
        sc = _score_vec(a, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = a.copy()

    # ---- two‑level piecewise constant (high block + low tail) -----------------
    ks = [int(f * n) for f in (0.02, 0.05, 0.1, 0.2, 0.3, 0.5)]
    for k in ks:
        if not (0 < k < n):
            continue
        for logt in np.linspace(math.log(1.1), math.log(30.0), 40):
            if time.monotonic() >= deadline:
                break
            t = math.exp(logt)                     # ratio high/low > 1
            l = 1.0 / (k * t + n - k)              # low value satisfying sum=1
            h = t * l                               # high value
            a = np.concatenate([np.full(k, h, dtype=np.float64),
                                np.full(n - k, l, dtype=np.float64)])
            sc = _score_vec(a, L)
            if sc < best_sc:
                best_sc = sc
                best_vec = a.copy()

    # --------------------------------------------------------
    # 2) Local refinement via smoothed‑gradient descent
    # --------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    momentum = np.zeros_like(cur)
    beta = 0.9
    eps = None
    it = 0
    max_iters = 5000

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        # FFT of current iterate
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # smoothing temperature
        if eps is None:
            eps = max(max_val * 0.2, 1e-12)
        else:
            eps = max(eps * 0.995, 1e-12)

        # soft‑max weights over the convolution entries
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)

        # gradient of the smoothed max (via FFT)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # indices that map to original vector
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # step size (decaying)
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = 0.5 / math.sqrt(it) / norm_m

        # candidate step & projection
        cand = cur - alpha * momentum
        cand = _project_simplex(cand)

        cand_sc = _score_vec(cand, L)
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        # occasional symmetrisation (helps occasional stagnation)
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
        # early exit if we have already crossed the target threshold
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
