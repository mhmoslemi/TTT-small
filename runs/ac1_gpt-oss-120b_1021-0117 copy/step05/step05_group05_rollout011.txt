import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score_from_conv(conv, n):
    """ score = 2 * n * max(conv)   (sum of a is 1 after projection) """
    return float(2.0 * n * conv.max())

def _conv_fft(a, L):
    """ linear convolution a * a using FFT of length L """
    af = np.fft.rfft(a, n=L)
    full = np.fft.irfft(af * af, n=L)
    return full[: 2 * a.size - 1]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a0 = rng.random(1024, dtype=np.float64)

    a = _project_simplex(np.maximum(a0, 0.0))
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two >= 2n-1

    # ----- bookkeeping ----------------------------------------------------
    momentum = np.zeros_like(a)
    beta = 0.9               # momentum factor
    step0 = 1.0              # initial step magnitude
    it = 0
    no_improve = 0
    best = a.copy()
    # initial scores
    conv = _conv_fft(a, L)
    cur_score = _score_from_conv(conv, n)
    best_score = cur_score
    target = 1.50274349       # stop when we beat this

    # ----- main optimisation loop -----------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution and maximum
        conv = _conv_fft(a, L)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # active‑set mask (sub‑gradient of the max)
        mask = conv >= max_val - 1e-12
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 1:
            w /= cnt

        # gradient of max (sub‑gradient of the objective)
        a_fft = np.fft.rfft(a, n=L)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
        grad = 2.0 * grad_full[(2 * n - 2) - np.arange(n)]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule
        step = step0 / math.sqrt(it)

        # ----- line‑search / backtracking ---------------------------------
        cand = None
        cand_score = None
        s = step
        for _ in range(8):
            tmp = _project_simplex(a - s * momentum)
            tmp_conv = _conv_fft(tmp, L)
            tmp_score = _score_from_conv(tmp_conv, n)
            if tmp_score <= cur_score:
                cand, cand_score = tmp, tmp_score
                break
            s *= 0.5
        if cand is None:
            cand = tmp
            cand_score = tmp_score

        # acceptance (allow occasional uphill moves)
        if cand_score < cur_score:
            a, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            # simulated‑annealing style acceptance
            temperature = 0.02 * (1.0 - (time.monotonic() - start) / (deadline - start))
            if rng.random() < math.exp(-(cand_score - cur_score) / max(temperature, 1e-12)):
                a, cur_score = cand, cand_score
            no_improve += 1

        # ----- periodic symmetrisation ------------------------------------
        if it % 200 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_conv = _conv_fft(sym, L)
            sym_score = _score_from_conv(sym_conv, n)
            if sym_score < cur_score:
                a, cur_score = sym, sym_score
                no_improve = 0
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score

        # ----- random mass‑transfer when stuck -----------------------------

            if a[i] > 1e-12:

                no_improve = 0

        # ----- early exit if target reached --------------------------------
        if best_score < target:
            break

    # return the best vector found (as plain Python list)
    return [float(x) for x in best]
