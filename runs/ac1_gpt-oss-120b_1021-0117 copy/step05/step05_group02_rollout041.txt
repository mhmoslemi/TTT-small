def solve_lp(f, M):
    n = f.shape[0]
    rows = 2*n - 1
    # Build sparse matrix
    diags = []
    offsets = []
    for d in range(n):
        length = rows - d
        diags.append(np.full(length, f[d]))
        offsets.append(d)
    A = sp.diags(diags, offsets, shape=(rows, n), format='csr')
    # Setup LP
    c = -np.ones(n)   # maximize sum(g)
    b_ub = np.full(rows, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        g = res.x
        sum_g = g.sum()
        if sum_g > 1e-12:
            g_norm = g / sum_g
            return g_norm
    return None

import time, math, numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Score used by the evaluator – assumes sum(a)=1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a, a))."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Linear‑program sub‑routine: find a complementary sequence
# ------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float):
    """Return a non‑negative vector g (sum(g)=1) with f*g ≤ M, or None."""
    n = f.size
    rows = 2 * n - 1
    # build Toeplitz‑like constraint matrix A (sparse)
    diags = []
    offsets = []
    for d in range(n):
        length = rows - d
        diags.append(np.full(length, f[d], dtype=np.float64))
        offsets.append(d)
    A = sp.diags(diags, offsets, shape=(rows, n), format='csr')
    c = -np.ones(n)                     # maximise sum(g)  → minimise -sum(g)
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        return None
    if not res.success:
        return None
    g = res.x
    s = g.sum()
    if s <= 1e-12:
        return None
    return g / s

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The function runs for up to `budget_s` seconds and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size

    # FFT length sufficient for gradient computation (a * w_rev)
    L = 1 << ((3 * n - 2).bit_length())

    # ---- hyper‑parameters -------------------------------------------

    step0_mirror = 0.8        # mirror‑descent base step
    eps0 = 1.5                # initial soft‑max temperature

    momentum = np.zeros_like(cur)
    no_improve = 0
    it = 0
    target = 1.50274349
    lp_interval = 600         # how often to invoke the LP routine

    while time.monotonic() < deadline:
        it += 1

        # ---- convolution of cur with itself -------------------------

        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ---- smooth gradient (soft‑max) -----------------------------
