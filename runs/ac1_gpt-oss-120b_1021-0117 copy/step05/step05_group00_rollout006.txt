def evaluate_and_gradient(a):

    a_fft = np.fft.rfft(a, n=L)

    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]

    max_val = conv_full.max()

    s = a.sum()
    if s < 0.01:
        score = np.inf
    else:
        score = 2 * n * max_val / (s * s)

    # smoothing weight
    eps = max(max_val * 0.12, 1e-12) # any factor
    w = np.exp((conv_full - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    # indices for gradient
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]

    return score, grad, conv_full, max_val

import time

# Helper functions (top‑level, no closures)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max_fft(a: np.ndarray, L: int):
    """Return full convolution of a with itself, its maximum, and the FFT of a."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    max_val = conv_full.max()
    return conv_full, max_val, a_fft


def _score_fft(a: np.ndarray, L: int) -> float:
    """Exact AC1 bound for a (assumes sum(a) > 0)."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_val / (s * s))
