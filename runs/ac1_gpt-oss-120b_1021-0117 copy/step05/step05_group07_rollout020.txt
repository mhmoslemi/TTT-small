import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (L2) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Core optimiser – matrix‑free gradient descent with momentum
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: a random vector of length ~2500
        cur = rng.random(2500, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1
    n = cur.size

    # ----- FFT planning – need length >= 3n‑2 -----------------------
    L = 1 << ((3 * n - 2).bit_length())   # next power of two
    # slice that extracts the gradient components from the full convolution
    idx = (2 * n - 2) - np.arange(n)

    # ----- bookkeeping ------------------------------------------------
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()
    cur_sc = 2.0 * n * max_val                # current score
    best_sc = cur_sc
    best_vec = cur.copy()

    # optimisation hyper‑parameters
    step0 = 0.3          # base step magnitude
    beta = 0.95          # momentum factor (higher ≈ smoother)
    momentum = np.zeros_like(cur)
    eps0 = 1.0           # initial temperature for soft‑max
    eps_decay = 0.995    # slower cooling than the reference
    eps_min = 1e-12
    target = 1.50274349
    it = 0
    no_improve = 0

    # ------------------------------------------------------------
    # main optimisation loop – matrix‑free, O(n log n) per iteration
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- temperature (soft‑max smoothing) --------------------
        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps > 1e-12:
            # soft‑max weights; clip exponent to avoid overflow
            diff = (conv_full - max_val) / eps
            diff = np.clip(diff, -50.0, 0.0)          # max occurs at 0
            w = np.exp(diff)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum
        else:
            # hard max – uniform weight on all maximal positions
            w = np.zeros_like(conv_full)
            mask = conv_full == max_val
            cnt = mask.sum()
            if cnt > 0:
                w[mask] = 1.0 / cnt

        # ----- gradient of the (soft) max -------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]          # size n

        # ----- momentum update ------------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- step‑size (inverse‑sqrt decay) with back‑tracking --
        base_alpha = step0 / math.sqrt(it + 1.0)
        alpha = base_alpha
        cand = None
        cand_sc = None

        # try up to 10 back‑tracking steps
        for _ in range(10):
            cand_tmp = cur - alpha * momentum
            cand_tmp = _project_simplex(cand_tmp)

            cand_fft = np.fft.rfft(cand_tmp, n=L)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
            cand_max = cand_conv.max()
            cand_sc = 2.0 * n * cand_max

            if cand_sc <= cur_sc:
                cand, cand_sc = cand_tmp, cand_sc
                break
            alpha *= 0.5

        # if no improvement, keep the last trial
        if cand is None:
            cand, cand_sc = cand_tmp, cand_sc

        # ----- accept / reject ------------------------------------
        if cand_sc < cur_sc:

            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:

            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:

            no_improve += 1

        if it % 150 == 0:

            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck --------------------
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0

        # ----- early exit if target reached -----------------------
        if best_sc < target:
            break

        # ----- keep an eye on the deadline ------------------------
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
