import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast score (matrix‑free) – same formula as the official evaluator
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max(convolution(a, a)) / (sum(a)**2)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ------------------------------------------------------------
# Subgradient of max‑convolution (exact, O(n))
# ------------------------------------------------------------
def _max_conv_subgrad(a: np.ndarray, k: int) -> np.ndarray:
    """
    Subgradient of max(conv(a,a)) at the peak index k.
    Returns a vector g where g[t] = 2*a[k-t] if 0<=k-t<n else 0.
    """
    n = a.size
    g = np.zeros_like(a)
    t_start = max(0, k - (n - 1))
    t_end = min(n - 1, k)
    if t_start <= t_end:
        t = np.arange(t_start, t_end + 1, dtype=int)
        idx = k - t
        g[t] = 2.0 * a[idx]
    return g


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The function runs up to `budget_s` seconds and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise – use the parent if present, otherwise a uniform start
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # default length 2048, uniform distribution
        n_default = 2048
        cur = np.full(n_default, 1.0 / n_default, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)               # enforce sum = 1
    n = cur.size

    # --------------------------------------------------------
    # FFT length (power of two ≥ 2n‑1)
    # --------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # --------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------
    step0 = 0.55               # base step for gradient descent
    step0_mirror = 0.38        # base step for mirror (exponential) update
    beta = 0.90                # momentum factor
    momentum = np.zeros_like(cur)
    T0 = 0.012                 # initial temperature for uphill moves
    target = 1.50274349
    it = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- convolution of current vector ----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        k = int(conv.argmax())

        # ---- exact subgradient at the peak ----
        g = _max_conv_subgrad(cur, k)

        # ---- momentum update ----
        momentum = beta * momentum + (1.0 - beta) * g

        # ---- step sizes (decay with sqrt(iter)) ----
        alpha = step0 / math.sqrt(it)
        eta = step0_mirror / math.sqrt(it)

        # ---- candidate 1 : projected gradient descent step ----
        cand1 = cur - alpha * momentum
        cand1 = _project_simplex(cand1)
        cand1_sc = _score(cand1)

        # ---- candidate 2 : mirror (exponential) update ----
        exp_arg = -eta * g
        np.clip(exp_arg, -50, 50, out=exp_arg)      # avoid overflow
        cand2 = cur * np.exp(exp_arg)
        s = cand2.sum()
        if s > 0:
            cand2 /= s
        else:
            cand2 = cur.copy()
        cand2_sc = _score(cand2)

        # ---- pick the better of the two deterministic candidates ----
        if cand1_sc <= cand2_sc:
            cand, cand_sc = cand1, cand1_sc
        else:
            cand, cand_sc = cand2, cand2_sc

        # ---- temperature for stochastic acceptance ----
        T = T0 * (0.9995 ** it)

        # ---- accept or reject (allow occasional uphill moves) ----
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc

        # ---- occasional random mass‑transfer mutation ----
        if it % 30 == 0:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                # choose a different index j
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.28
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                if tmp_sc < cur_sc or rng.random() < math.exp(-(tmp_sc - cur_sc) / (T + 1e-12)):
                    cur, cur_sc = tmp, tmp_sc

        # ---- periodic symmetrisation (helps with Toeplitz structure) ----
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_vec, best_sc = sym.copy(), sym_sc
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc

        # ---- early exit if target reached ----
        if best_sc < target:
            break

    # --------------------------------------------------
    # Return the best candidate (as plain Python list)
    # --------------------------------------------------
    return [float(x) for x in best_vec]
