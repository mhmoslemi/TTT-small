    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    best_score = 2.0 * n * conv.max()
    best_vec = a.copy()
    
    # parameters
    step0 = 0.5
    beta = 0.9

    tau = 0.5  # temperature

    tau_min = 1e-4
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # compute convolution and gradient
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # compute weights
        # shift for numeric stability
        shifted = conv - max_val
        w = np.exp(shifted / max(tau, 1e-12))
        w_sum = w.sum()
        if w_sum == 0:

            w[conv.argmax()] = 1.0
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # momentum update (Adam like)
        momentum = beta * momentum + (1 - beta) * g
        # step size
        step = step0 / (it**0.5)
        cand = a - step * momentum
        cand = _project_simplex(cand)
        # symmetrize optionally

        cand_score = 2.0 * n * cand_conv.max()
        if cand_score < best_score:
            best_score = cand_score
            best_vec = cand.copy()
            a = cand
            no_improve = 0

    return [float(x) for x in best_vec]

    v = np.asarray(v, dtype=np.float64)

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # step size (decaying)
        step = step0 / math.sqrt(it)

        # ---- backtracking line search ------------------------------------
        cand = None
        cand_score = None
        alpha = step
        for _ in range(6):
            tmp = a - alpha * momentum
            tmp = _project_simplex(tmp)
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp = _project_simplex(tmp)
            tmp_fft = np.fft.rfft(tmp, n=L)
            tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[:2 * n - 1]
            sc = 2.0 * n * tmp_conv.max()
            if sc < cur_score:
                cand, cand_score = tmp, sc
                break
            alpha *= 0.5

        # if no descent found, maybe accept a tiny worsening move
        if cand is None:
            if rng.random() < 0.001:
                cand, cand_score = tmp, sc
            else:
                cand, cand_score = a, cur_score

        # ---- accept / record ---------------------------------------------
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
        else:
            no_improve += 1

        # occasional explicit symmetrisation (already done each step, but
        # reinforces numerical symmetry)
        if it % 250 == 0:
            a = (a + a[::-1]) * 0.5
            a = _project_simplex(a)
            a_fft = np.fft.rfft(a, n=L)
            conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_vec = a.copy()

        # random reshuffle when stuck
        if no_improve > 3000:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[:2 * n - 1]
                sc = 2.0 * n * tmp_conv.max()
                if sc < best_score:
                    best_score = sc
                    best_vec = tmp.copy()
                if sc < cur_score:
                    a = tmp
                    cur_score = sc
                no_improve = 0

        # linear‑programming refinement after prolonged stagnation
        if no_improve > 6000 and time.monotonic() + 30.0 < deadline:
            g_opt = _lp_optimize(a, float(max_val))
            if g_opt is not None and g_opt.size == n:
                s = g_opt.sum()
                if s > 0.0:
                    g_norm = g_opt / s
                    g_proj = _project_simplex(g_norm)
                    g_proj = (g_proj + g_proj[::-1]) * 0.5
                    g_proj = _project_simplex(g_proj)
                    g_fft = np.fft.rfft(g_proj, n=L)
                    conv_g = np.fft.irfft(g_fft * g_fft, n=L)[:2 * n - 1]
                    sc = 2.0 * n * conv_g.max()
                    if sc < best_score:
                        best_score = sc
                        best_vec = g_proj.copy()
                        a = g_proj
                        cur_score = sc
                        no_improve = 0

        # temperature decay
        tau = max(tau * tau_decay, tau_min)

        if best_score < target:
            break

    # final LP polish if time permits
    if time.monotonic() + 10.0 < deadline:
        best_fft = np.fft.rfft(best_vec, n=L)
        conv_best = np.fft.irfft(best_fft * best_fft, n=L)[:2 * n - 1]
        M_best = conv_best.max()
        g_opt = _lp_optimize(best_vec, float(M_best))
        if g_opt is not None and g_opt.size == n:
            s = g_opt.sum()
            if s > 0.0:
                g_norm = g_opt / s
                g_proj = _project_simplex(g_norm)
                g_proj = (g_proj + g_proj[::-1]) * 0.5
                g_proj = _project_simplex(g_proj)
                g_fft = np.fft.rfft(g_proj, n=L)
                conv_g = np.fft.irfft(g_fft * g_fft, n=L)[:2 * n - 1]
                sc = 2.0 * n * conv_g.max()
                if sc < best_score:
                    best_score = sc
                    best_vec = g_proj.copy()

    return [float(x) for x in best_vec]
