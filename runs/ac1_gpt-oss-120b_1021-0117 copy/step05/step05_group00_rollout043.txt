def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # get n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    # define projection
    def _project_simplex(v):
        n_ = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1
        rho = np.where(u - css / np.arange(1, n_+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    def _score(a):
        s = a.sum()
        if s < 1e-12:
            return np.inf
        L = 1 << ((2*n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        maxb = conv.max()
        return float(2.0 * n * maxb / (s*s))

    best_vec = None
    best_sc = np.inf

    # generate seed vectors
    seeds = []
    # height_sequence_1 if available
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        a0 = a0 / a0.sum()
        seeds.append(a0)

    # uniform
    uniform = np.full(n, 1.0 / n)
    seeds.append(uniform)

    # exponential (geometric) with several alphas
    for alpha in (0.0005, 0.001, 0.002, 0.005):
        a = np.exp(-alpha * np.arange(n, dtype=np.float64))
        a = a / a.sum()
        seeds.append(a)

    # power-law
    for p in (0.3, 0.5, 0.8):
        a = 1.0 / (np.arange(1, n+1, dtype=np.float64) ** p)
        a = a / a.sum()
        seeds.append(a)

    # random Dirichlet-like
    for _ in range(5):
        a = rng.random(n)
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            continue
        a = a / a.sum()
        seeds.append(a)

    # optimization parameters common
    improve_target = 1.50274349
    max_trials = 5
    sym_interval = 400
    eps_decay = 0.995
    base_alpha = 0.5
    beta = 0.9

    # precompute L for gradient calculations (larger to avoid wrap-around)
    L = 1 << ((3 * n - 2).bit_length())

    # process each seed
    for seed_vec in seeds:
        cur = seed_vec.copy()
        cur_sc = _score(cur)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
        # early exit
        if best_sc < improve_target:
            break

        momentum = np.zeros_like(cur)
        eps = None
        no_improve = 0
        it = 0

        while it < 2000 and time.monotonic() < deadline:
            it += 1

            # convolution & weight
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
            max_val = conv_full.max()

            if eps is None:
                eps = max(max_val * 0.2, 1e-12)
            else:
                eps = max(eps * eps_decay, 1e-12)

            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:

        """Euclidean projection onto {x>=0, sum(x)=1}."""
        n_ = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n_ + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    def _score(a: np.ndarray) -> float:
        """Exact AC1 bound for a (no assumptions on sum)."""
        s = a.sum()
        if s < 1e-12:
            return np.inf
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max() / (s * s))
