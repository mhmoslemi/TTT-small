def _softmax_weights(conv, max_val, eps):
    exp_term = (conv - max_val) / eps
    # clip to avoid overflow
    exp_term = np.clip(exp_term, -50.0, 0.0)
    w = np.exp(exp_term)
    w_sum = w.sum()
    if w_sum == 0.0:
        # fallback uniform
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = w / w_sum
    return w

import time, math, numpy as np

# ------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale non‑negative vector to have sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _max_and_fft(a: np.ndarray, L: int):
    """Return max of a*a, its FFT (rfft) and the full convolution."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _softmax_weights(conv: np.ndarray, max_val: float, eps: float) -> np.ndarray:
    """Soft‑max weights for a smooth approximation of max."""
    exp_term = (conv - max_val) / eps
    exp_term = np.clip(exp_term, -50.0, 0.0)          # avoid overflow/underflow
    w = np.exp(exp_term)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size, dtype=np.float64)
    return w / s


def _gradient_fft(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate.
    Returns 2·(a ⊛ w_rev) evaluated at the needed indices.
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]
