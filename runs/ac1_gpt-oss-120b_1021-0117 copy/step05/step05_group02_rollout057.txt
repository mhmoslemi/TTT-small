import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score used by the evaluator (sum is assumed to be 1)
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    """Return 2 * n * max(convolution(vec, vec))."""
    n = vec.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The function runs for up to `budget_s` seconds and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------ initial vector -------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fall back to a uniform vector of moderate size
        cur = np.full(2048, 1.0 / 2048, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)        # enforce sum = 1
    n = cur.shape[0]

    # FFT length (power of two, ≥ 3n‑2 to handle a*w convolutions)
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------ optimisation state --------------------
    m = np.zeros_like(cur)            # 1st moment (Adam)
    v = np.zeros_like(cur)            # 2nd moment (Adam)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.6                  # base step size for Adam
    eta0 = 0.5                        # base step for mirror descent
    topk_start = max(1, int(0.01 * (2 * n - 1)))  # start with ~1% of shifts
    topk_decay = 0.9995

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # ----------- compute convolution and max ---------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----------- build weight vector (top‑k soft max) -------
        topk = max(1, int(topk_start * (topk_decay ** it)))
        if topk >= conv.size:

        else:

        # ------------------- candidate 1 (Adam step) ----------
        cand1 = cur - step
        cand1 = _project_simplex(cand1)
        cand1_sc = _score(cand1)

        if cand1_sc > cur_sc:

                bt_step *= 0.5
                tmp = cur - bt_step
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc <= cur_sc:
                    cand1, cand1_sc = tmp, tmp_sc
                    break
            else:
                cand1, cand1_sc = tmp, tmp_sc

            no_improve += 1

        # ------------------- periodic symmetrisation ------------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ------------------- random mass‑transfer when stuck ----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0

        # ------------------- early stopping --------------------
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
