def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    # helper functions defined as nested? but must be top-level per spec. So we define at top of file.

    # init
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 0.5)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # try default length: 2048
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)  # sum=1
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())

    # define evaluate bound
    best = a.copy()
    best_sc = _max_conv(best, L)
    best_bound = 2.0 * n * best_sc

    cur = a.copy()
    cur_sc = best_sc

    # hyperparameters
    eps0 = 0.5   # start smoothing parameter
    eps_decay = 0.9995
    eps_min = 1e-12
    step0 = 0.5
    beta = 0.9
    mom = np.zeros_like(a)
    it = 0
    target = 1.50274349

    # gradient descent loop
    while time.time() < deadline - 5.0:
        it += 1
        # compute convolution of cur
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()

        eps = max(eps_min, eps0 * (eps_decay ** it))
        # compute softmax weights
        # to avoid overflow, subtract max_val
        exp_arg = (conv - max_val) / eps
        # clamp large negative values to avoid underflow? np.exp handles fine.
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isinf(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient via convolution with reversed weights
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # map to original shape
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum
        mom = beta * mom + (1.0 - beta) * grad

        # line search
        alpha = step0 / math.sqrt(it)
        improved = False
        for _ in range(10):
            cand = cur - alpha * mom
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc <= cur_sc:
                cur = cand
                cur_sc = cand_sc
                improved = True
                if cand_sc < best_sc:
                    best = cand.copy()
                    best_sc = cand_sc
                    best_bound = 2.0 * n * best_sc
                break
            alpha *= 0.5
        if not improved:
            # maybe try a random perturbation
            pass
        # periodic symmetrization
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _max_conv(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best = sym.copy()
                    best_sc = sym_sc
                    best_bound = 2.0 * n * best_sc

        # early exit
        if best_bound < target:
            break

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Project onto the probability simplex (sum == 1, x >= 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Maximum entry of the (full) autocorrelation a*a
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max value of the full linear convolution a*a using FFT length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * a.size - 1]
    return float(conv.max())

# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # Initialise – use parent if provided, else random
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2048, dtype=np.float64)
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)           # enforce sum == 1
    n = a0.size

    # --------------------------------------------------------------
    # FFT length (next power of two >= 2n-1)
    # --------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    best = a0.copy()
    best_sc = _max_conv(best, L)
    best_bound = 2.0 * n * best_sc
    cur = a0.copy()
    cur_sc = best_sc

    # --------------------------------------------------------------
    # Hyper‑parameters for the smoothed gradient descent
    # --------------------------------------------------------------
    eps0 = 0.5                # initial temperature for soft‑max
    eps_decay = 0.9999        # geometric cooling
    eps_min = 1e-12
    step0 = 0.8               # base step size
    beta = 0.9                # momentum factor
    momentum = np.zeros_like(cur)
    it = 0
    target_bound = 1.50274349

    # --------------------------------------------------------------
    # Phase 1 – smoothed gradient descent (matrix‑free)
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        it += 1
        # compute full autocorrelation
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weights (smoothed arg‑max)
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)            # map to original coordinates
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # back‑tracking line search on the simplex
        alpha = step0 / math.sqrt(it)
        improved = False
        for _ in range(10):
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc <= cur_sc:
                cur = cand
                cur_sc = cand_sc
                improved = True
                if cand_sc < best_sc:
                    best = cand.copy()
                    best_sc = cand_sc
                    best_bound = 2.0 * n * best_sc
                break
            alpha *= 0.5

        # if no improvement, try a small random jitter (still projected)
        if not improved and time.time() < deadline - 5.0:
            jitter = rng.normal(scale=0.01, size=n)
            cand = cur + jitter
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc < cur_sc:
                cur = cand
                cur_sc = cand_sc
                if cand_sc < best_sc:
                    best = cand.copy()
                    best_sc = cand_sc
                    best_bound = 2.0 * n * best_sc

        # occasional symmetrisation (helps reduce peaks)
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _max_conv(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best = sym.copy()
                    best_sc = sym_sc
                    best_bound = 2.0 * n * best_sc

        # early exit if target reached
        if best_bound < target_bound:
            break

    # --------------------------------------------------------------
    # Phase 2 – deterministic “anti‑identity” re‑ordering
    # --------------------------------------------------------------
    vals = np.sort(best)[::-1]                 # descending
    perm = np.empty_like(vals)
    left, right = 0, n - 1
    k = 0
    while left <= right:
        perm[left] = vals[k]; k += 1
        if left < right:
            perm[right] = vals[k]; k += 1
        left += 1
        right -= 1
    perm_sc = _max_conv(perm, L)
    perm_bound = 2.0 * n * perm_sc
    if perm_bound < best_bound:
        best = perm.copy()
        best_sc = perm_sc
        best_bound = perm_bound

    # --------------------------------------------------------------
    # Phase 3 – simple hill‑climbing swaps on the current best
    # --------------------------------------------------------------
    cur = best.copy()
    cur_bound = best_bound
    while time.time() < deadline - 0.3:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_sc = _max_conv(cur, L)
        new_bound = 2.0 * n * new_sc
        if new_bound < cur_bound:
            cur_bound = new_bound
            if new_bound < best_bound:
                best = cur.copy()
                best_sc = new_sc
                best_bound = new_bound
                if best_bound < target_bound:
                    break
        else:
            # revert the swap
            cur[i], cur[j] = cur[j], cur[i]

    # ------------------------------------------------------------------
    # Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best]
