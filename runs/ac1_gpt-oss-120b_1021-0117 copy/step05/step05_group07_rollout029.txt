import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the sandbox
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1
    n = cur.size

    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------

    beta1 = 0.9

    it = 0

    # temperature for the soft‑max weighting of the convolution max
    T0 = 0.2                # initial temperature
    T_decay = 0.9995        # very slow decay – keeps gradient smooth

    # bookkeeping for the best solution seen so far
    best_vec = cur.copy()
    best_sc = float('inf')
    target = 1.50274349

    # --------------------------------------------------------

    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and its current maximum -----
        a_fft = np.fft.rfft(cur, n=L)

        max_b = conv_full.max()
        cur_sc = 2.0 * n * max_b
