import time

# ------------------------------------------------------------
# Projection onto the probability simplex (sum=1, a_i>=0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 score (assumes sum(a)=1)
# ------------------------------------------------------------
def _score_fft(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(convolution) for a vector with sum≈1."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Max convolution value and subgradient (subgradient of max)
# ------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray, L: int):
    """Compute max convolution value and a subgradient w.r.t. a."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)

    # full convolution (linear, not circular)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()

    # weights for subgradient: uniform over argmax positions
    tol = max_val * 1e-12 + 1e-15
    mask = conv_full >= max_val - tol
    cnt = np.count_nonzero(mask)
    if cnt == 0:  # should not happen, but safe fallback
        w = np.full_like(conv_full, 1.0 / conv_full.size)
    else:
        w = mask.astype(np.float64) / cnt

    # subgradient = 2 * (a * w_rev) restricted to n components
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)   # map to required n entries
    grad = 2.0 * grad_full[idx]
    return max_val, grad
