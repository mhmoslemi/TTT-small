import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0.0)
    return w

    return np.maximum(v - theta, 0.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(1024, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)          # now sum(a) == 1
    n = a.shape[0]

    # ----- FFT parameters -------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1

    # ----- helper to compute the exact score -------------------------------
    def _score_vec(v):
        s = v.sum()
        if s < 0.01:
            return np.inf
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return 2.0 * n * conv.max() / (s * s)

    cur = a
    cur_sc = _score_vec(cur)
    best = cur.copy()
    best_sc = cur_sc

    # ----- optimisation parameters -----------------------------------------
    eta0 = 0.8                # base step size for exponentiated update
    no_improve = 0
    it = 0
    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution and active‑set mask ----------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        mask = conv >= max_val - 1e-12
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        # ----- gradient of the active constraints ------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- multiplicative (exponentiated) update -----------------------
        eta = eta0 / math.sqrt(it)
        upd = np.exp(-eta * grad)
        cand = cur * upd
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / s

        cand_sc = _score_vec(cand)

        # ----- accept / reject (Metropolis) --------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (eta + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- keep global best -------------------------------------------
        if cur_sc < best_sc:
            best, best_sc = cur.copy(), cur_sc

        # ----- occasional random perturbation to escape stagnation ----------
        if no_improve > 2000:
            perturb = rng.random(n)
            cand = cur + 0.05 * (perturb - perturb.mean())
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            cand_sc = _score_vec(cand)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
                if cand_sc < best_sc:
                    best, best_sc = cand.copy(), cand_sc

        # ----- periodic symmetrisation (helps many instances) -------------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- early exit if target reached --------------------------------
        if best_sc < target:
            break

    # ----- return the best candidate found ---------------------------------
    return [float(x) for x in best]
