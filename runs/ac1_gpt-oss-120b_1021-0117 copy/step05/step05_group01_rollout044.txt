import time, math
import numpy as np

def _conv_full(a, L):
    af = np.fft.rfft(a, n=L)
    return np.fft.irfft(af * af, n=L)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initial vector
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # arbitrary length if no parent provided
        n0 = 2048
        a = rng.random(n0, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # evaluate current best
    conv_full = _conv_full(a, L)[: 2 * n - 1]
    best_score = 2.0 * n * conv_full.max()
    best_a = a.copy()
    no_improve = 0
    it = 0

    base_eta = 0.9
    sym_period = 200
    stuck_thresh = 2000

    while time.monotonic() < deadline:
        it += 1
        # convolution
        conv_full = _conv_full(a, L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_score < 1.50274349:
            break

        # subgradient of max convolution
        eps = max_val * 1e-12 + 1e-12
        idxs = np.where(conv_full >= max_val - eps)[0]
        g = np.zeros_like(a)
        for k in idxs:
            i0 = max(0, k - (n - 1))
            i1 = min(n - 1, k)
            # slice i0..i1
            i = np.arange(i0, i1 + 1, dtype=int)
            g[i] += 2.0 * a[k - i]

        # multiplicative update
        eta = base_eta / math.sqrt(it)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # symmetrize periodically
        if it % sym_period == 0:
            a = (a + a[::-1]) * 0.5
            a = a / a.sum()

        # random mass transfer when stuck
        if no_improve > stuck_thresh:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

    # hill‑climbing refinement with remaining time
    while time.monotonic() < deadline:
        # small random transfer from current best copy
        a_hc = best_a.copy()
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if a_hc[i] <= 0:
            continue
        delta = a_hc[i] * rng.random() * 0.3
        a_hc[i] -= delta
        a_hc[j] += delta
        conv_full = _conv_full(a_hc, L)[: 2 * n - 1]
        sc = 2.0 * n * conv_full.max()
        if sc < best_score:
            best_score = sc
            best_a = a_hc.copy()

    return [float(x) for x in best_a]

def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full convolution a*a using an FFT of length L."""
    af = np.fft.rfft(a, n=L)
    return np.fft.irfft(af * af, n=L)
