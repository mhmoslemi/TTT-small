def _normalize(v: np.ndarray) -> np.ndarray:
    """Make non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v = v / s
    else:
        v = np.full_like(v, 1.0 / v.size)
    return v

    """Return max of the full convolution a*a using FFT length L."""
    af = np.fft.rfft(a, n=L)

    return float(conv.max())

    return 2.0 * n * max_val

    """Convenient wrapper that computes the evaluator score for sum=1."""
    return _score_from_max(_conv_max(a, L), a.size)

                best_vec, best_sc = cand.copy(), cand_sc
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps due to convolution symmetry) -----
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _normalize(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 800:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _normalize(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # --------------------------------------------------------
    # Return best solution discovered (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
