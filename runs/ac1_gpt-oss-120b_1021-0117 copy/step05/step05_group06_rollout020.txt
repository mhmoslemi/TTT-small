# Helper: project onto the probability simplex {x>=0, sum=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return float(conv_full[: 2 * a.size - 1].max())

            else:
                w = np.where(mask, 1.0 / cnt, 0.0)

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # back‑tracking line search
        alpha = step0 / math.sqrt(it)
        improved = False
        for _ in range(6):
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc <= cur_sc:
                cur, cur_sc = cand, cand_sc
                improved = True
                if cand_sc < best_sc:
                    best, best_sc = cand.copy(), cand_sc
                    best_bound = 2.0 * n * best_sc / (best.sum() ** 2)
                break
            alpha *= 0.5
        if not improved:
            pass

        # occasional symmetrisation (helps with Toeplitz structure)
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _max_conv(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    best_bound = 2.0 * n * best_sc / (best.sum() ** 2)

        if best_bound < target:
            break

    # -----------------------------------------------------------------
    # Generate a handful of permutation‑based candidates
    # -----------------------------------------------------------------
    sorted_vals = np.sort(best)[::-1]          # descending
    cand_list = []

    # 1) alternating‑extremes (anti‑identity)
    perm = np.empty_like(sorted_vals)
    left, right, p = 0, n - 1, 0
    while left <= right:
        perm[left] = sorted_vals[p]; p += 1
        if left < right:
            perm[right] = sorted_vals[p]; p += 1
        left += 1
        right -= 1
    cand_list.append(perm)

    # 2) reversed version
    cand_list.append(perm[::-1].copy())

    # 3) a few rotations of the anti‑identity pattern
    for shift in (1, 2, 3):
        cand_list.append(np.roll(perm, shift))

    # 4) modular‑step traversals (step coprime with n)
    for step in (3, 5, 7, 11, 13):
        if math.gcd(step, n) == 1:
            idx = (np.arange(n) * step) % n
            cand_list.append(sorted_vals[idx])

    # 5) a few random permutations of the sorted values
    for _ in range(10):
        cand_list.append(rng.permutation(sorted_vals))

    # evaluate all generated candidates
    for c in cand_list:
        sc = _max_conv(c, L)
        bound = 2.0 * n * sc / (c.sum() ** 2)
        if bound < best_bound:
            best, best_sc = c.copy(), sc
            best_bound = bound

    # -----------------------------------------------------------------
    # Intensive hill‑climbing / simulated annealing swaps
    # -----------------------------------------------------------------
    cur = best.copy()
    cur_bound = best_bound
    temperature = 1e-3
    temp_decay = 0.9999

    while time.monotonic() < deadline - 0.2:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]               # propose swap
        new_sc = _max_conv(cur, L)
        new_bound = 2.0 * n * new_sc / (cur.sum() ** 2)

        accept = False
        if new_bound < cur_bound:
            accept = True
        else:
            prob = math.exp(-(new_bound - cur_bound) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_bound = new_bound
            if new_bound < best_bound:
                best, best_sc = cur.copy(), new_sc
                best_bound = new_bound
                if best_bound < target:
                    break
        else:
            cur[i], cur[j] = cur[j], cur[i]           # revert

        temperature = max(temperature * temp_decay, 1e-12)

    # -----------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # -----------------------------------------------------------------
    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best]
