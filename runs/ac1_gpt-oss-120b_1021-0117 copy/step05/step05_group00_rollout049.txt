def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0, float(budget_s) - 0.5)  # margin

    # initialization
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total <= 0:
            a = np.full_like(a, 1.0 / a.size)
            total = 1.0
        a = a / total
    else:
        n = 2000
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-t / 100.0)
        a = a / a.sum()
    n = a.size
    L = 1 << ((3*n - 2).bit_length())

    # optimizer states
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    alpha = 0.05  # base learning rate

    best = a.copy()
    best_score = compute_score(best)

    no_improve = 0
    it = 0

    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # compute convolution, max, weights
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n - 1]
        max_conv = conv.max()
        # temperature schedule: combine max_conv magnitude and iteration
        tau = max( max_conv * 0.2 * (0.995**it), 1e-12)  # may adapt
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = alpha / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step * m_hat

        # projection onto simplex
        a_new = project_simplex(a_new)

        # periodic symmetrisation
        if it % 50 == 0:
            a_new = (a_new + a_new[::-1]) * 0.5
            a_new = project_simplex(a_new)

        cur_score = compute_score(a_new)

        # accept (always accept new candidate; but we keep best)
        if cur_score < best_score:
            best_score = cur_score
            best = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # iterate
        a = a_new

        # random perturbation if stuck
        if no_improve > 3000:
            # small random noise around best
            a = best + 0.01 * rng.random(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # early exit
        if best_score < target:
            break

    # final evaluation to ensure best meets constraints
    # convert to list of floats
    return [float(x) for x in best]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (scale‑invariant; sum≈1)
# ------------------------------------------------------------
def _score_vec(a: np.ndarray, Lscore: int) -> float:
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    n = a.shape[0]
    af = np.fft.rfft(a, n=Lscore)
    conv = np.fft.irfft(af * af, n=Lscore)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total <= 0.0:
            a = np.full_like(a, 1.0 / a.size)
            total = 1.0
        a = a / total                     # normalise to sum = 1
    else:                                 # fallback when parent absent
        n0 = 2000
        t = np.arange(n0, dtype=np.float64)
        a = np.exp(-t / 100.0)
        a = a / a.sum()
    n = a.size

    # ----- FFT lengths ------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())      # for gradient convolution
    Lscore = 1 << ((2 * n - 1).bit_length()) # for scoring

    # ----- Adam optimiser state ---------------------------------------
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.08

    # ----- best‑so‑far tracking ---------------------------------------
    best = a.copy()
    best_score = _score_vec(best, Lscore)
    cur_score = best_score

    no_improve = 0
    it = 0
    target = 1.50274349

    # ----- optimisation loop -----------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution of current a with itself
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_conv = conv.max()

        # smooth approximation of the max (soft‑max)
        tau = max(max_conv * 0.2 * (0.995 ** it), 1e-12)
        w = np.exp((conv - max_conv) / tau)
        ws = w.sum()
        if ws == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= ws

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update (minimise)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            best = a_new.copy()
            no_improve = 0

        # early termination when target reached
        if best_score < target:
            break

    # final sanity: ensure non‑trivial sum (evaluator requires ≥0.01)
    if best.sum() < 0.01:
        best = best / best.sum() * 0.01

    return [float(x) for x in best]
