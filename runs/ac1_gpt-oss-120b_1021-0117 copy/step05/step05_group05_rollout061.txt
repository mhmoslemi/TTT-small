def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def _gradient(a, L, max_val):
    # compute subgradient w.r.t a of maximum convex combination (uniform on max indices)
    # a_fft cached
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*a.size-1]
    eps = 1e-12
    mask = conv >= max_val - eps
    w = mask.astype(np.float64)
    cnt = w.sum()
    if cnt > 1:
        w /= cnt
    # reverse for convolution
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*a.size-2) - np.arange(a.size)   # mapping to gradient indices
    g = 2.0 * grad_full[idx]
    return g, max_val, a_fft, conv

import time, math, numpy as np

try:
    import cvxpy as cp
    _has_cvxpy = True
except Exception:  # pragma: no cover
    _has_cvxpy = False


def _project_simplex(v):
    """ Euclidean projection onto the probability simplex (sum=1, non‑negative). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(a):
    """ 2·n·max(a∗a) for a non‑negative vector with sum = 1 (scale‑invariant). """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _lp_optimize(a, M):
    """
    Solve  max 1ᵀg   s.t.  conv(a,g) ≤ M ,  g ≥ 0 .
    Returns the optimal g as a dense ndarray or None on failure.
    """
    n = a.shape[0]
    g = cp.Variable(n, nonneg=True)
    a_par = cp.Parameter(n, value=a, constant=True)
    conv_expr = cp.conv(a_par, g)                 # length 2n‑1
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv_expr <= M])
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                    max_iters=4000, verbose=False)
    except Exception:  # pragma: no cover
        return None
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None
    gv = g.value
    if gv is None:
        return None
    return np.asarray(gv, dtype=np.float64)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a list of non‑negative step heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
        cur = np.maximum(cur, 0.0)
    else:
        cur = rng.random(1024, dtype=np.float64)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((3 * n - 2).bit_length())          # safe length for a⋆w

    best = cur.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)
    beta = 0.95
    step0 = 1.0
    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # --- convolution of cur with itself -----------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # --- sub‑gradient of the max‑convolution -----------------------
        mask = conv >= max_val - 1e-12
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 1:
            w /= cnt
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # --- momentum and line‑search step ------------------------------
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        alpha = step
        cand = None
        cand_sc = None
        for _ in range(10):
            cand_tmp = cur - alpha * momentum
            cand_tmp = _project_simplex(cand_tmp)
            cand_tmp_sc = _score(cand_tmp)
            if cand_tmp_sc < cur_sc:
                cand, cand_sc = cand_tmp, cand_tmp_sc
                break
            alpha *= 0.5
        if cand is None:
            cand, cand_sc = cand_tmp, cand_tmp_sc

        # --- acceptance -------------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # --- frequent symmetrisation (helps to reduce peaks) ----------
        if it % 250 == 0:

            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:

                no_improve = 0

        if no_improve >= 2000:

            if cur[i] > 0.0:

                j = rng.integers(0, n - 1)

            if g_opt is not None:

                    cur_sc = _score(cur)
                    if cur_sc < best_sc:
                        best, best_sc = cur.copy(), cur_sc
                    no_improve = 0

        # --- early exit if target reached --------------------------------
        if best_sc < target:
            break

    return [float(x) for x in best]
