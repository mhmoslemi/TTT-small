    import time, numpy as np, math

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2048, dtype=np.float64)
    cur = np.maximum(cur, 0.0)

    # normalize to sum 1
    cur /= max(cur.sum(), 1e-12)

    n = len(cur)
    # FFT length: next power of 2 >= 2*n-1
    L = 1 << ((2*n - 2).bit_length())

    def max_conv(v):
        a_fft = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2*n -1]
        return conv.max()

    cur_max = max_conv(cur)
    best_vec = cur.copy()
    best_max = cur_max
    best_score = 2 * n * cur_max # sum=1

    # parameters
    it = 0
    temperature = 0.5
    temp_decay = 0.9995
    step_base = 0.2
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # random local move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # proportion to move
        delta = cur[i] * rng.random() * 0.25
        if delta == 0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # renormalize (should already sum to 1)
        # small numerical errors corrected
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum

        cand_max = max_conv(cand)
        cand_score = 2 * n * cand_max

        # acceptance: metropolis
        if cand_score < best_score or rng.random() < math.exp((best_score - cand_score) / temperature):
            cur = cand
            cur_max = cand_max
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
                best_max = cand_max
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # decrease temperature
        temperature = max(1e-8, temperature * temp_decay)

        # occasional symmetrization
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym /= sym.sum()
            sym_max = max_conv(sym)
            sym_score = 2 * n * sym_max
            if sym_score < best_score:
                cur = sym
                cur_max = sym_max
                best_score = sym_score
                best_vec = sym.copy()
                best_max = sym_max

    return [float(x) for x in best_vec]

def _self_conv_max(a: np.ndarray, L: int):
    """Return max of a*a (linear convolution) together with its FFT."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _softmax_weights(conv: np.ndarray, max_val: float, eps: float) -> np.ndarray:
    """Soft‑max over convolution values, temperature = eps."""
    w = np.exp((conv - max_val) / eps)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size, dtype=np.float64)
    return w / s


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:

    w_rev = w[::-1]

    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]
