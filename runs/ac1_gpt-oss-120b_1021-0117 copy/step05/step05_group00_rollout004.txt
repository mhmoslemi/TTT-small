import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)≈1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a non‑negative vector normalized to sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for cleanup/return
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # --------------------------------------------------------
    # initialise a feasible point on the simplex
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
    else:  # fallback if parent not present
        n_fallback = 2500
        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 100.0)

    s0 = a0.sum()
    if s0 < 0.01:
        a0 = np.full_like(a0, 1.0 / a0.size)
        s0 = a0.sum()
    a0 = a0 / s0                      # normalise to sum = 1

    cur = a0.copy()
    n = cur.size
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------
    it = 0
    no_improve = 0
    eps = None
    eps_decay = 0.992
    base_alpha = 0.8
    beta = 0.95                     # momentum
    momentum = np.zeros_like(cur)
    sym_interval = 250
    improve_target = 1.50274349
    max_trials = 5                  # back‑tracking steps

    # Pre‑compute FFT length (power of two)
    L = 1 << ((2 * n - 1).bit_length())

    while time.monotonic() < deadline:
        it += 1

        # ----------- convolution and smoothing weight ------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps is None:
            eps = max(max_val * 0.25, 1e-9)
        else:
            eps = max(eps * eps_decay, max_val * 1e-5)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ------------------- gradient ----------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # pick n components
        g = 2.0 * grad_full[idx]

        # ------------------- momentum ----------------------------
        momentum = beta * momentum + (1.0 - beta) * g

        # ------------------- step size ---------------------------
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / (math.sqrt(it) * norm_m)

        # ------------------- backtracking line‑search ------------
        improved = False
        cand = None
        cand_sc = None
        cur_alpha = alpha
        for _ in range(max_trials):
            cand_try = cur - cur_alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            cur_alpha *= 0.5

        # ------------------- occasional uphill move -------------
        if not improved:
            cand_try = cur - cur_alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            # simple Metropolis criterion
            prob = math.exp(-(cand_sc - cur_sc) / (cur_alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        # ------------------- accept if improved -----------------
        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ------------------- periodic symmetrisation ------------
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ------------------- random mass‑transfer when stuck ---
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ------------------- restart from best with noise -------
        if no_improve > 5000:
            cur = best_vec + 0.005 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        # ------------------- early exit if target reached -------
        if best_sc < improve_target:
            break

        # safety check to avoid overrunning the deadline
        if time.monotonic() >= deadline:
            break

    # Return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
