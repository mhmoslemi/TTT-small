import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1 (avoid division by zero)."""
    s = v.sum()
    if s == 0.0:
        # uniform fallback – should never happen for a sensible start
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _conv_max_fft(a: np.ndarray, L: int):
    """
    Compute the linear convolution a*a via FFT of length L
    and return (max(conv), fft_of_a, conv_array).
    """
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _softmax_weights(conv: np.ndarray, max_val: float, eps: float) -> np.ndarray:
    """Return a soft‑max distribution over the convolution entries."""
    w = np.exp((conv - max_val) / eps)          # (conv‑max)/eps ≤ 0
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size, dtype=np.float64)
    return w / s


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution objective.
    Implements 2·(a * w_rev) evaluated at the required indices.
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    # indices producing the length‑n gradient vector
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients whose
    AC1 bound (2 n·max(a*a)/sum(a)^2) is as low as possible.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a safety margin before the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initial vector – use the parent if it exists, else random
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
    else:
        cur = rng.random(1024, dtype=np.float64)

    cur = _normalize(cur)
    n = cur.size

    # FFT length – next power of two ≥ 2n‑1 (covers linear convolution)
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------
    eps0 = 0.6           # initial soft‑max temperature
    eps_decay = 0.9995   # exponential decay per iteration
    eps_min = 1e-13
    step0 = 0.45         # base step magnitude for exponentiated update
    beta = 0.92          # momentum factor
    momentum = np.zeros_like(cur)
    no_improve = 0
    best_vec = cur.copy()
    cur_max, a_fft, _ = _conv_max_fft(cur, L)
    best_max = cur_max
    best_score = 2.0 * n * best_max
    target_score = 1.50274349          # stop early if we beat the target
    it = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- current convolution and max ----
        cur_max, a_fft, conv = _conv_max_fft(cur, L)

        # ---- soft‑max weights (smooth approximation of max) ----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = _softmax_weights(conv, cur_max, eps)

        # ---- gradient (FFT‑based, matrix‑free) ----
        grad = _gradient(a_fft, w, n, L)

        # ---- momentum update (exponential moving average) ----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ---- candidate generation (exponentiated gradient step) ----
        step = step0 / math.sqrt(it)
        exponent = -step * momentum
        exponent = np.clip(exponent, -50.0, 50.0)      # protect against overflow
        cand = cur * np.exp(exponent)
        cand = _normalize(cand)

        cand_max, cand_fft, _ = _conv_max_fft(cand, L)

        # ---- acceptance test ----
        if cand_max < cur_max - 1e-15:

            a_fft = cand_fft

            no_improve = 0

                best_vec = cand.copy()
                best_score = 2.0 * n * best_max
        else:

            no_improve += 1
            prob = math.exp(-(cand_max - cur_max) / (step + 1e-12))
            if rng.random() < prob:
                cur = cand
                a_fft = cand_fft
                cur_max = cand_max

        # ---- random mass‑transfer when stalled for a while ----
        if no_improve > 2500:

        if best_score < target_score:

    # Return the best sequence found (plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
