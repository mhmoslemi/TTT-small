import time
import numpy as np

def _score(v: np.ndarray) -> float:
    """Exact AC1 bound for a vector (full convolution, max)."""
    n = v.shape[0]
    # FFT length: next power of two ≥ 2n‑1
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound.  The routine respects the time budget,
    falls back to a uniform vector when no parent is present, and performs a
    tiny stochastic local‑search while time permits.
    """
    start = time.monotonic()
    # keep a safety margin so we always return a value before the sandbox kills us
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # Initialise – use the provided parent if it exists, otherwise uniform.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 1024
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    # enforce the evaluator's hard bounds
    cur = np.clip(cur, 0.0, 1000.0)

    # make the sum positive (the evaluator requires sum ≥ 0.01)
    s = cur.sum()
    if s < 0.01:
        cur += (0.01 - s) / cur.size
        s = cur.sum()

    # normalise to the probability simplex – scaling does not affect the score
    cur = cur / s

    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # ------------------------------------------------------------------
    # Simple stochastic mass‑transfer search (matrix‑free, O(n log n) per step)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    it = 0
    # temperature schedule (simulated‑annealing style)
    T = 0.02
    decay = 0.9995

    while time.monotonic() < deadline and it < 8000:
        it += 1

        # propose: move a tiny amount of mass from i to j
        i = rng.integers(0, cur.size)
        j = rng.integers(0, cur.size - 1)
        if j >= i:
            j += 1
        delta = cur[i] * rng.random() * 0.04  # up to 4 % of the donor mass
        if delta == 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep non‑negativity and renormalise to the simplex
        cand = np.clip(cand, 0.0, None)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        cand_sc = _score(cand)

        # accept if better, otherwise with Metropolis probability
        if cand_sc < best_sc:
            best_vec, best_sc = cand.copy(), cand_sc
            cur = cand
        else:
            prob = np.exp(-(cand_sc - best_sc) / max(T, 1e-12))
            if rng.random() < prob:
                cur = cand

        # cool down
        T *= decay

    # ------------------------------------------------------------------
    # Return the best vector discovered (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
