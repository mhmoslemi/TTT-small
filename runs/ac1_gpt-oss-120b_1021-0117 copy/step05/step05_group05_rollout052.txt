import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Return 2·n·max(conv(a,a))  (the numerator of the evaluator)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)          # indices for gradient extraction

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # learning‑rate schedule
    base_lr = 0.8

    # beta (soft‑max temperature) schedule
    beta_start, beta_end = 0.5, 20.0
    beta_switch_frac = 0.7   # after this fraction start using hard mask

    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        it += 1

        # ----- FFT of current vector -----
        a_fft = np.fft.rfft(cur, n=L)

        # ----- autocorrelation -----
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # ----- weight vector w (smooth max) -----
        elapsed = now - start
        total = deadline - start
        frac = elapsed / total if total > 0 else 0.0
        if frac < beta_switch_frac:
            # smooth soft‑max weighting
            beta = beta_start + (beta_end - beta_start) * (frac / beta_switch_frac)
            exps = np.exp(beta * (conv - max_conv))
            w = exps / exps.sum()
        else:
            # hard max (mask) for fine‑tuning
            mask = conv >= max_conv - 1e-12
            w = mask.astype(np.float64)
            cnt = w.sum()
            if cnt > 0:
                w /= cnt

        # ----- gradient of the smooth max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = grad_full[: 2 * n - 1]
        g = 2.0 * grad[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / math.sqrt(it)
        step = lr * (m_hat / (np.sqrt(v_hat) + eps))

        cand = cur - step
        # enforce symmetry (helps due to symmetry of the objective)
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # ----- accept / reject -----
        if cand_sc <= cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # occasional stochastic acceptance
            prob = math.exp(-(cand_sc - cur_sc) * 0.2)
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- keep the best -----
        if cand_sc < best_sc:
            best_vec, best_sc = cand.copy(), cand_sc

        # ----- periodic random perturbation when stuck -----
        if no_improve > 3000:
            noise = rng.normal(scale=0.02, size=n)
            cur = _project_simplex(np.clip(cur + noise, 0, None))
            cur_sc = _score(cur)
            # reset momentum
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
