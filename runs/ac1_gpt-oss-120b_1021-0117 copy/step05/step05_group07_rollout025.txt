import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a uniform start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # sum = 1
    n = cur.size

    # ------------------------------------------------------------
    # FFT planning – length at least 3n‑2 (next power of two)
    # ------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())
    idx = (2 * n - 2) - np.arange(n)    # slice needed for gradient

    # ------------------------------------------------------------
    # Book‑keeping and hyper‑parameters
    # ------------------------------------------------------------
    # ADAM parameters
    alpha = 0.05          # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # temperature for soft‑max (log‑sum‑exp) smoothing
    T = 5.0               # initial temperature
    T_decay = 0.9995
    T_min = 0.001

    # optimisation loop control
    it = 0
    no_improve = 0
    target = 1.50274349

    # moments for ADAM
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    pow_beta1 = 1.0
    pow_beta2 = 1.0

    # initial score
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()
    cur_score = 2.0 * n * max_val
    best_vec = cur.copy()
    best_sc = cur_score

    # ------------------------------------------------------------
    # Optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- recompute convolution and its maximum for current vector -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_score = 2.0 * n * max_val

        # ----- soft‑max weights (log‑sum‑exp) -----
        T = max(T_min, T * T_decay)          # geometric cooling
        w = np.exp((conv_full - max_val) / T)
        w_sum = w.sum()
        if w_sum == 0.0:

        else:
            w /= w_sum

        # ----- gradient of the smoothed max (matrix‑free) -----

                    no_improve = 0

                    no_improve = 0

        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
