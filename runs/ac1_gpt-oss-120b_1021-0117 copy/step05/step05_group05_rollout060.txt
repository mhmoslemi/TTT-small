import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(a, fft_len=None):
    """Internal objective: 2 * n * max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _grad(a, beta=15.0, fft_len=None, idx_slice=None):
    """Gradient of the smoothed max using a softmax weight on the convolution."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)
    conv = conv_full[: 2 * n - 1]

    # soft‑max weighting of the convolution entries
    w = np.exp(beta * (conv - conv.max()))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum

    # convolution of a with reversed weights gives the sub‑gradient
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=fft_len)
    grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)
    if idx_slice is None:
        idx_slice = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx_slice]

    mu = 0.9                # Nesterov momentum factor
    step0 = 1.0             # initial step length
    beta_soft = 15.0        # soft‑max temperature
    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- gradient based on soft‑max of the full convolution ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        w = np.exp(beta_soft * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[idx]

        # ----- Nesterov‑accelerated projected step -------------------------
        y = cur + mu * (cur - prev)
        step = step0 / math.sqrt(it)
        cand = _project_simplex(y - step * g)
        cand_sc = _score(cand, fft_len=L)

        if cand_sc < cur_sc:
            prev, cur, cur_sc = cur, cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            # occasional probabilistic acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                prev, cur, cur_sc = cur, cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation ------------------------------------
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, fft_len=L)
            if sym_sc < cur_sc:
                prev, cur, cur_sc = cur, sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random mass‑transfer when stuck -----------------------------
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, fft_len=L)
                if tmp_sc < cur_sc:
                    prev, cur, cur_sc = cur, tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        if best_sc < target:
            break

    return [float(x) for x in best]
