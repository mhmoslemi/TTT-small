        n = v.shape[0]
        if n == 0:
            return v

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast evaluation of 2 * n * max(conv(a, a))  (a is assumed to sum to 1)
# ------------------------------------------------------------
def _score_vec(a: np.ndarray, L: int) -> float:
    """Return 2 * n * max(convolution(a, a)) using FFT of length L."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    """

    start = time.monotonic()
    # keep a small safety margin for final conversion
    deadline = start + max(0.0, float(budget_s) - 0.5)

    target = 1.50274349          # aim to beat this value

    # ------------------------------------------------------------------

    if "height_sequence_1" in globals():

        # otherwise normalise it to the simplex

    L = 1 << ((2 * n - 1).bit_length())

    # temperature for the soft‑max smoothing of the max‑operator
    tau = 0.01

    best = a.copy()
