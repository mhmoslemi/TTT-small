def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # init cur
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = np.full(1024, 1.0/1024, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)
    n = cur.size

    # FFT length L for convolution and for gradient (need size >= 3n-2)
    L = 1 << ((3*n - 2).bit_length())
    # Also L_score perhaps same L, conv via a_fft * a_fft.

    best_vec = cur.copy()
    best_sc = _score_fast(best_vec, L)   # define
    cur_sc = best_sc

    # parameters
    step0 = 0.8
    beta = 0.9
    momentum = np.zeros_like(cur)
    tol_mask = 1e-12
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # compute convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2*n-1]
        max_val = conv.max()

        # mask for max indexes
        # use tolerance relative to max
        tol = max(tol_mask, 1e-12 * max_val)
        mask = (conv >= max_val - tol).astype(np.float64)

        # compute gradient via cross-correlation
        a_rev = cur[::-1]
        mask_fft = np.fft.rfft(mask, n=L)
        a_rev_fft = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(a_rev_fft * mask_fft, n=L)
        # Extract index range n-1 .. n-1 + n-1? Actually length of grad_full = L
        # Want segment start = n-1, length n
        start_idx = n-1
        grad_seg = grad_full[start_idx : start_idx + n]
        grad = 2.0 * grad_seg

        # momentum
        momentum = beta * momentum + (1-beta) * grad

        # step size
        alpha = step0 / math.sqrt(it)

        cand = cur - alpha * momentum
        cand = _project_simplex(cand)
        cand_sc = _score_fast(cand, L)

        # Accept if improvement
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            # maybe small probability acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # random mass transfer if stuck
        if no_improve > 2000:
            # pick two distinct indices
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            # move random fraction from i to j
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_fast(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # optional symmetric averaging every 500 iterations
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fast(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        if best_sc < target:
            break

    return [float(x) for x in best_vec]

def _score_fast(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2*n - 1]
    max_val = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_val / (sum_a * sum_a))

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_fast(a: np.ndarray, L: int) -> float:
    """Fast evaluation of the AC1 bound for a (assumes non‑negative a)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_val / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initial sequence -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

    cur = np.maximum(cur, 0.0)

    n = cur.size

    # ---- FFT length (>=3n-2) ---------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())

    # ---- bookkeeping ------------------------------------------------------

    best_sc = _score_fast(best_vec, L)
    cur_sc = best_sc

    # ---- hyper‑parameters -------------------------------------------------
    step0 = 0.8
    beta = 0.9
    momentum = np.zeros_like(cur)
    tol_mask = 1e-12
    it = 0
    no_improve = 0
    target = 1.50274349

    # ---- main optimisation loop -------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution and maximum
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # mask of maximal entries (subgradient support)
        tol = max(tol_mask, 1e-12 * max_val)
        mask = (conv >= max_val - tol).astype(np.float64)

        # gradient = 2 * (mask * reverse(cur)) correlation
        a_rev = cur[::-1]
        mask_fft = np.fft.rfft(mask, n=L)

        start_idx = n - 1

        # momentum update
