    while time.monotonic() < deadline:

        # compute conv and score
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        cur_max = conv.max()
        cur_sc = 2.0 * n * cur_max

        # update best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if best_sc < target:
            break

        # gradient of max conv
        g = _grad_max_subgrad(a, conv)

        # momentum
        momentum = beta * momentum + (1.0 - beta) * g

        # step size
        alpha = step0 / math.sqrt(it)

        # candidate
        a_tmp = a - alpha * momentum
        a_tmp = _project_simplex(a_tmp)

        # evaluate candidate
        a_tmp_fft = np.fft.rfft(a_tmp, n=L)
        conv_tmp = np.fft.irfft(a_tmp_fft * a_tmp_fft, n=L)[:2*n-1]
        cand_sc = 2.0 * n * conv_tmp.max()

        # accept if improvement or occasional uphill
        if cand_sc <= cur_sc:
            a = a_tmp
            cur_sc = cand_sc
        else:
            # Metropolis acceptance with temperature decreasing
            T = max(1e-4, step0 / math.sqrt(it))
            prob = math.exp(-(cand_sc - cur_sc) / T)
            if rng.random() < prob:
                a = a_tmp
                cur_sc = cand_sc

        # periodic symmetrisation

            a_sym_fft = np.fft.rfft(a_sym, n=L)
            conv_sym = np.fft.irfft(a_sym_fft * a_sym_fft, n=L)[:2*n-1]

            if sym_sc < cur_sc:

                cur_sc = sym_sc
                if sym_sc < best_sc:

                    best_vec = a_sym.copy()
                    stagnation = 0

            i = rng.integers(0, n)

            if i != j:

                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
            stagnation = 0

    return [float(x) for x in best_vec]

import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean), sum(z)=1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project onto {x >= 0, sum(x)=z}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Max of full convolution a*a (FFT based)
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ------------------------------------------------------------
# Exact sub‑gradient of max_k (a*a)[k]
# ------------------------------------------------------------
def _grad_max_subgrad(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.size
    max_val = conv.max()
    idxs = np.where(np.abs(conv - max_val) <= 1e-12)[0]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        lt = i < j
        if lt.any():
            ii = i[lt]
            jj = j[lt]
            g[ii] += a[jj]
            g[jj] += a[ii]
        eq = i == j
        if eq.any():
            ii = i[eq]
            g[ii] += 2.0 * a[ii]
    return g

# ------------------------------------------------------------
# Optional sparsification (zero‑out tiny entries)
# ------------------------------------------------------------
def _sparsify(a: np.ndarray, thr: float = 1e-6) -> np.ndarray:
    a[a < thr] = 0.0
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / a.size
    else:
        a /= s
    return a
