import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex using the algorithm of Duchi et al."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    def _score(vec: np.ndarray) -> float:
        a_fft = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return 2.0 * n * conv.max()      # sum(vec) == 1 ⇒ denominator = 1

    cur_score = _score(cur)
    best_vec = cur.copy()
    best_score = cur_score

    # --------------------------------------------------------
    # Optimisation hyper‑parameters
    # --------------------------------------------------------
    step_base = 0.55          # base step magnitude
    beta = 0.95               # momentum coefficient
    momentum = np.zeros_like(cur)

    eps0 = 1.0                # initial smoothing temperature
    eps_decay = 0.97
    eps_min = 1e-9

    it = 0
    no_improve = 0
    target = 1.5025           # stop early if we beat the target

    # --------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iter)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- smooth approximation of the max (softmax) -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps > 1e-12:
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            w = np.zeros_like(conv)
            mask = conv == max_val
            cnt = mask.sum()
            if cnt > 0:
                w[mask] = 1.0 / cnt

        # ----- gradient of the (soft) max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]            # shape (n,)

        # ----- momentum update (Nesterov‑style could be added) -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- line search with backtracking -----
        alpha = step_base / math.sqrt(it)
        cand = None
        cand_sc = None
        # try a few reductions of the step size
        for _ in range(7):
            cand_tmp = cur - alpha * momentum
            # enforce symmetry (helps with the Toeplitz structure)
            cand_tmp = (cand_tmp + cand_tmp[::-1]) * 0.5
            cand_tmp = _project_simplex(cand_tmp)
            cand_tmp_sc = _score(cand_tmp)

            if cand_tmp_sc < cur_score:
                cand, cand_sc = cand_tmp, cand_tmp_sc
                break
            alpha *= 0.5

        # if none of the trials improved, fall back to the last attempt
        if cand is None:
            cand, cand_sc = cand_tmp, cand_tmp_sc

        # ----- acceptance -----
        if cand_sc < cur_score:
            cur, cur_score = cand, cand_sc
            no_improve = 0
            if cand_sc < best_score:
                best_score, best_vec = cand_sc, cand.copy()
        else:
            # occasional uphill move (Metropolis criterion)
            prob = math.exp(-(cand_sc - cur_score) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_sc
            no_improve += 1

        # ----- periodic full symmetrisation -----
        if it % 100 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_score:
                cur, cur_score = sym, sym_sc
                if sym_sc < best_score:
                    best_score, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_score:
                    cur, cur_score = tmp, tmp_sc
                if tmp_sc < best_score:
                    best_score, best_vec = tmp_sc, tmp.copy()
                no_improve = 0

        # ----- early exit if target reached -----
        if best_score < target:
            break

    # --------------------------------------------------------
    # Return the best candidate as a plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
