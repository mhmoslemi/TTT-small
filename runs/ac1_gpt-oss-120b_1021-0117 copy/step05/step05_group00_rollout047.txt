import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a) = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function vector with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.4)

    # --------------------------------------------------------
    # initialise a feasible point on the simplex
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s0 = a0.sum()
        if s0 <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s0                                 # sum = 1
        # enforce (optional) symmetry – helpful for the autocorrelation
        a0 = (a0 + a0[::-1]) * 0.5
        a0 = _project_simplex(a0)
    else:                                           # fallback construction
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()

    n = a0.size
    # pre‑computed FFT length (next power of two >= 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    cur = a0.copy()
    cur_score = _score(cur)

    best_vec = cur.copy()
    best_sc = cur_score

    # --------------------------------------------------------
    # Adam optimiser state
    # --------------------------------------------------------
    m = np.zeros_like(cur)           # first moment
    v = np.zeros_like(cur)           # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 0.5

    # --------------------------------------------------------
    # control variables
    # --------------------------------------------------------
    it = 0
    no_improve = 0
    sym_interval = 200
    improve_target = 1.50274349
    gamma_hard = 0.2                # mix soft and hard gradient

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current point -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_score = 2.0 * n * max_val

        # ----- update best if needed -----
        if cur_score < best_sc:
            best_sc = cur_score
            best_vec = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- softmax weighting for smoothed max -----
        eps = max(max_val * 0.2, 1e-12)
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed max (soft) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # slice length n
        g_soft = 2.0 * grad_full[idx]

        # ----- hard gradient at exact max index -----
        max_idx = int(np.argmax(conv_full))
        indices = max_idx - np.arange(n)
        mask = (indices >= 0) & (indices < n)
        g_hard = np.zeros_like(cur)
        g_hard[mask] = 2.0 * cur[indices[mask]]

        # ----- mixed gradient -----
        g = (1.0 - gamma_hard) * g_soft + gamma_hard * g_hard

        # ----- Adam update (learning rate decays with sqrt(iter)) -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # ----- accept if improves or with a tiny uphill probability -----
        if cand_sc < cur_score:
            cur = cand
            cur_score = cand_sc
        else:
            # occasional uphill move (simulated‑annealing style)
            prob = math.exp(-(cand_sc - cur_score) / (cur_score * 1e-3 + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_score = cand_sc

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_score:
                cur = sym
                cur_score = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_score:
                    cur = tmp
                    cur_score = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- restart from best with small noise if hopeless -----
        if no_improve > 5000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = _project_simplex(cur)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

    # Return as plain Python list
    return [float(x) for x in best_vec]
