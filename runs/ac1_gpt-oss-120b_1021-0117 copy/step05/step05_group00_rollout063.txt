import time, math, numpy as np

def _sidon_greedy(n, rng):
    perm = rng.permutation(n)
    indices = []
    used = set()
    for pos in perm:
        # Check self sum
        s_self = pos + pos
        if s_self in used:
            continue
        # Check sums with existing
        ok = True
        for i in indices:
            if (i + pos) in used:
                ok = False
                break
        if not ok:
            continue
        # add
        for i in indices:
            used.add(i + pos)
        used.add(s_self)
        indices.append(pos)
    return indices

def _evaluate_sequence(seq):
    try:
        return evaluate_sequence(seq)
    except Exception:
        return float('inf')

def propose_candidate(seed=42,budget_s=1100,...):
    import time, numpy as np, math
    start = time.monotonic()
    deadline = start + max(0.0,float(budget_s)-0.1)

    # initial vector
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.clip(a,0,1000)
        s = a.sum()
        if s < 0.01:
            a = np.full_like(a,1.0/a.size)
            s = a.sum()
        a = a / s
    else:
        n = 2000
        a = np.exp(-np.arange(n)/100)
        a = a / a.sum()

    n = a.size
    L = 1 << ((3*n-2).bit_length())

    def score_vec(v):
        # fast through FFT like earlier
        L_local = L
        af = np.fft.rfft(v, n=L_local)
        conv = np.fft.irfft(af*af, n=L_local)[: 2*n-1]
        return float(2.0 * n * conv.max())

    cur = a.copy()
    cur_sc = score_vec(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # Optimization parameters
    momentum = np.zeros_like(cur)
    beta = 0.95
    base_alpha = 1.0
    eps = None
    eps_decay = 0.997
    sym_interval = 250
    max_trials = 5
    it = 0
    no_improve = 0
    improve_target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # compute convolution and smoothed weights
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft*a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()
        if eps is None:
            eps = max(max_val*0.15, 1e-9)
        else:
            eps = max(eps*eps_decay, 1e-9)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # Need proper indexing for gradient where convolution of length 2n-1: gradient for indices 0..n-1 is 2 * grad_full[(2n-2)-np.arange(n)]
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum
        momentum = beta * momentum + (1 - beta) * g

        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(it) / norm_m

        # line search
        improved = False
        alpha = alpha0
        for _ in range(max_trials):
            cand = cur - alpha * momentum
            # project onto simplex (sum=1)
            # We'll use same _project_simplex as before
            cand = _project_simplex(cand)
            cand_sc = score_vec(cand)
            if cand_sc < cur_sc:
                improved = True
                break
            alpha *= 0.5
        if not improved:
            # occasional uphill
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)
            cand_sc = score_vec(cand)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # symmetrize
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = score_vec(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # random mass-transfer if stagnated

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (sum of coefficients ≈ 1)
# ------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
