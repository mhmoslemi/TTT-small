def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(1024, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.05  # base learning rate
    it = 0
    no_improve = 0
    target = 1.50274349
    sym_period = 400
    alpha = 5.0
    alpha_max = 200.0
    alpha_inc = 1.001

    while time.monotonic() < deadline:
        it += 1
        # compute convolution of cur with itself
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # softmax weights
        # handle large coordinates
        exp_arg = alpha * (conv - max_val)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient via convolution with reversed weights
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        tmp = cur - step
        tmp = _project_simplex(tmp)

        tmp_sc = _score(tmp, L)

        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            no_improve = 0
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc
        else:
            # occasional uphill acceptance
            prob = math.exp(-(tmp_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = tmp, tmp_sc
            no_improve += 1

        # symmetrization
        if it % sym_period == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
            if sym_sc < best_sc:
                best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # random mass transfer when stuck
        if no_improve > 4000:
            # pick two positions and move mass
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp2 = cur.copy()
                tmp2[i] -= delta
                tmp2[j] += delta
                tmp2 = _project_simplex(tmp2)
                tmp2_sc = _score(tmp2, L)
                if tmp2_sc < cur_sc:
                    cur, cur_sc = tmp2, tmp2_sc
                if tmp2_sc < best_sc:
                    best_vec, best_sc = tmp2.copy(), tmp2_sc
                no_improve = 0

        # increase softmax temperature gradually
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_inc)

        # early termination
        if best_sc < target:
            break

    # return best as list
    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a probability vector (sum = 1):
#   2 * n * max_{k} (a * a)[k]
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Compute 2*n*max(convolution(a, a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------

    # ------------------------------------------------------------

    # Adam optimiser parameters
