def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, cvxpy as cp, math

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # init sequence
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        rng = np.random.default_rng(seed)
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        a[:] = 1.0
        total = a.sum()
    a = a / total   # sum=1

    n = a.size
    # prepare FFT length
    L = 1 << ((2 * n - 1).bit_length())

    # compute initial conv and best score
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    M = conv_full.max()
    best_score = 2.0 * n * M
    best_seq = a.copy()

    # parameters
    t = 0.5   # mixing factor
    # we also use small random perturbations or symmetrizations occasionally
    iteration = 0
    # counters for stagnation
    no_improve = 0

    while time.monotonic() < deadline:
        iteration += 1

        # Solve LP to get g
        f_const = a   # constant array
        g = cp.Variable(n, nonneg=True)
        # convolution expression
        conv_fg = cp.conv(f_const, g)  # length 2n-1
        constraints = [conv_fg <= M + 1e-12]   # small epsilon
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception:
            # fallback if solver fails
            break
        if g.value is None:
            break
        g_val = np.array(g.value, dtype=np.float64)
        # Normalize g to sum 1
        g_sum = g_val.sum()
        if g_sum <= 1e-12:
            # if g ended up zero, break
            break
        g_val = g_val / g_sum

        # mixing
        a_new = (1 - t) * a + t * g_val

        # compute new conv
        a_new_fft = np.fft.rfft(a_new, n=L)
        conv_new = np.fft.irfft(a_new_fft * a_new_fft, n=L)[:2*n-1]
        M_new = conv_new.max()
        score_new = 2.0 * n * M_new

        # accept if improved
        if score_new < best_score * (1 - 1e-8):  # some tolerance
            best_score = score_new
            best_seq = a_new.copy()
            a = a_new
            M = M_new
            no_improve = 0
        else:
            no_improve += 1
            # maybe reduce mixing factor if no improvement
            if no_improve > 10:
                t = max(0.1, t * 0.7)
                no_improve = 0
        # optional symmetrize
        if iteration % 10 == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = np.maximum(a_sym, 0)
            a_sym = a_sym / a_sym.sum()
            # eval sym
            a_sym_fft = np.fft.rfft(a_sym, n=L)
            conv_sym = np.fft.irfft(a_sym_fft * a_sym_fft, n=L)[:2*n-1]
            M_sym = conv_sym.max()
            score_sym = 2.0 * n * M_sym
            if score_sym < best_score:
                best_score = score_sym
                best_seq = a_sym.copy()
                a = a_sym
                M = M_sym
            else:
                # maybe keep sym as a candidate for next iteration
                a = a_sym if score_sym < 2.0 * n * M else a

        # early exit
        if best_score < 1.50274349:
            break

    # Return best result as list
    return [float(x) for x in best_seq]

import time, math, numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = z}."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ------------------------------------------------------------
def _conv_max(arr: np.ndarray, L: int):
    """Return (max value, full convolution) of arr with itself."""
    af = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * arr.size - 1]
    return float(conv.max()), conv

def _score(arr: np.ndarray, L: int):
    """Score = 2 * n * max(conv(arr,arr))."""
    n = arr.size
    m, _ = _conv_max(arr, L)
    return 2.0 * n * m

# Main entry point
# ------------------------------------------------------------

    """Return a non‑empty non‑negative sequence with a low AC1 bound."""

    # Initialise a (use parent if present)
    # --------------------------------------------------------

    a = np.maximum(a, 0.0)

    if s < 1e-12:
        a[:] = 1.0
        s = a.sum()

        max_lp_iters = 30

        eps = 1e-12
        lp_iter = 0
        while time.monotonic() < deadline and lp_iter < max_lp_iters:
            lp_iter += 1

            # current convolution peak
            M, _ = _conv_max(a, L)

            # ----------------------------------------------------
            # Linear program: maximise sum(g) s.t. a * g <= M
            # ----------------------------------------------------
            g = cp.Variable(n, nonneg=True)
            conv_ag = cp.conv(a, g)          # length 2n‑1

                               [conv_ag <= M + eps])

            except Exception:               # pragma: no cover

            if gs <= 1e-12:

        cur_max = float(conv_full.max())

        if cur_sc < best_score:
