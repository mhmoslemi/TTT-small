import time
import math
import numpy as np

# --------------------------------------------------------------
# Simplex projection (L2 onto {x >= 0, sum = 1})
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # ensure exact sum = 1 (possible tiny numerical drift)
    s = w.sum()
    if s > 0:
        w /= s
    return w

    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())

        # soft‑max weights (smooth approximation of arg‑max)
        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps > 1e-6:
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            mask = conv >= max_val - 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = np.where(mask, 1.0 / cnt, 0.0)

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)   # map full‑conv indices back to a
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # back‑tracking line search
        alpha = step0 / math.sqrt(it)
        improved = False
        for _ in range(8):
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc <= cur_sc:
                cur = cand
                cur_sc = cand_sc
                improved = True
                if cand_sc < best_sc:
                    best = cand.copy()
                    best_sc = cand_sc
                    best_bound = 2.0 * n * best_sc
                break
            alpha *= 0.5

        # occasional random jitter if stuck
        if not improved and rng.random() < 0.005:
            noise = rng.normal(scale=0.01, size=n)
            cand = cur + noise
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc < cur_sc:
                cur = cand
                cur_sc = cand_sc
                if cand_sc < best_sc:
                    best = cand.copy()
                    best_sc = cand_sc
                    best_bound = 2.0 * n * best_sc

        # occasional symmetrisation (helps with convolution structure)
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _max_conv(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best = sym.copy()
                    best_sc = sym_sc
                    best_bound = 2.0 * n * best_sc

        # early exit
        if best_bound < target_bound:
            break

    # ----------------------------------------------------------
    # Permutation search (random shuffles + simulated annealing)
    # ----------------------------------------------------------
    # multiset of values to permute
    vals = np.sort(best)[::-1]               # descending order

    # evaluate a few random permutations openly
    cur_perm = vals.copy()
    rng.shuffle(cur_perm)
    cur_sc = _max_conv(cur_perm, L)
    cur_bound = 2.0 * n * cur_sc
    if cur_sc < best_sc:
        best = cur_perm.copy()
        best_sc = cur_sc
        best_bound = cur_bound

    # number of random re‑shuffles (time‑bounded)
    shuffle_trials = 200
    for _ in range(shuffle_trials):
        if time.monotonic() >= deadline - 5.0:
            break
        perm = vals.copy()
        rng.shuffle(perm)
        sc = _max_conv(perm, L)
        if sc < best_sc:
            best = perm.copy()
            best_sc = sc
            best_bound = 2.0 * n * sc
            cur_perm = perm
            cur_sc = sc
            cur_bound = 2.0 * n * sc

    # simulated annealing on permutations
    cur_seq = cur_perm.copy()
    cur_sc = _max_conv(cur_seq, L)
    cur_bound = 2.0 * n * cur_sc
    # temperature scheduler
    T = max(0.1, (best_bound - target_bound) * 0.5)
    decay = 0.9997
    step = 0

    while time.monotonic() < deadline - 0.5:
        step += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_sc = _max_conv(cur_seq, L)
        new_bound = 2.0 * n * new_sc
        delta = new_bound - cur_bound
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < math.exp(-delta / T)
        if accept:
            cur_sc = new_sc
            cur_bound = new_bound
            if new_bound < best_bound:
                best = cur_seq.copy()
                best_sc = new_sc
                best_bound = new_bound
                if best_bound < target_bound:
                    break
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        # cool
        T *= decay
        if T < 1e-12:
            T = 0.0

    # ----------------------------------------------------------
    # Return best candidate (as plain Python list)
    # ----------------------------------------------------------
    if best.size == 0:
        return [1.0]
    return [float(x) for x in best]
