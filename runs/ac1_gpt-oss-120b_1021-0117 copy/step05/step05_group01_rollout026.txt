import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=z}."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Choose an initial vector (parent if present, otherwise a linear decay)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a_parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a_parent = np.maximum(a_parent, 0.0)
        if a_parent.sum() < 1e-12:
            a_parent[:] = 1.0
        a_parent = a_parent / a_parent.sum()
        n = a_parent.size
        # linear decreasing start of the same length
        a_lin = np.arange(n, 0, -1, dtype=np.float64)
        a_lin = a_lin / a_lin.sum()
        # pick the better of the two
        def _score(a):
            L = 1 << ((2 * a.size - 1).bit_length())
            fft_a = np.fft.rfft(a, n=L)
            conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * a.size - 1]
            return 2.0 * a.size * conv.max()
        if _score(a_lin) < _score(a_parent):
            a = a_lin.copy()
        else:
            a = a_parent.copy()
    else:
        # no parent – use a simple decreasing start (length 2048)
        n = 2048
        a = np.arange(n, 0, -1, dtype=np.float64)
        a = a / a.sum()

    # ------------------------------------------------------------
    # Pre‑computed FFT length (next power‑of‑two >= 2n‑1)
    # ------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    cur_max = conv_full[: 2 * n - 1].max()
    best_sc = 2.0 * n * cur_max
    best_vec = a.copy()

    # optimisation parameters
    momentum = np.zeros_like(a)
    beta = 0.9                     # momentum decay
    step0 = 1.0                    # base step magnitude
    top_k = 7                     # number of largest correlations used for gradient
    sym_interval = 150            # how often to symmetrise
    no_improve = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        fft_a = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fft_a * fft_a, n=L)
        conv = conv_full[: 2 * n - 1]
        cur_max = conv.max()
        cur_sc = 2.0 * n * cur_max

        # ----- keep the best seen so far -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- gradient from the `top_k` largest convolution entries -----
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # vectorised contribution for this lag
            i = np.arange(i_start, i_end + 1, dtype=int)
            g[i] += 2.0 * a[k - i]
        g /= max(1, top_k)

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- step size (decays like 1/sqrt(it)) -----
        alpha = step0 / math.sqrt(it)

        # ----- candidate generation and projection -----
        a_cand = a - alpha * momentum
        a_cand = _project_simplex(a_cand)   # keep on simplex (sum=1, non‑negative)

        # ----- evaluate candidate -----
        fft_c = np.fft.rfft(a_cand, n=L)
        conv_c = np.fft.irfft(fft_c * fft_c, n=L)[: 2 * n - 1]
        cand_sc = 2.0 * n * conv_c.max()

        # ----- accept / reject (with occasional uphill move) -----
        if cand_sc < cur_sc:
            a = a_cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                a = a_cand
                cur_sc = cand_sc

        # ----- periodic symmetrisation (helps with the Toeplitz structure) -----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            fft_s = np.fft.rfft(a_sym, n=L)
            conv_s = np.fft.irfft(fft_s * fft_s, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * conv_s.max()
            if sym_sc < cur_sc:
                a = a_sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 5000:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

        # ----- adapt top_k if we have been stuck for a while -----
        if no_improve > 2000:
            top_k = min(conv.size, top_k + 2)
        else:
            top_k = 7

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
