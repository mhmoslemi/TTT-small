import time, math, numpy as np
from scipy.special import i0

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Window generators (non‑negative, easy to normalise)
# ------------------------------------------------------------
def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Return a Kaiser window of length n with shape parameter beta."""
    if n == 1:
        return np.ones(1, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    arg = beta * np.sqrt(1.0 - ((i - alpha) / alpha) ** 2)
    w = i0(arg) / i0(beta)
    return w


def _hann_window(n: int) -> np.ndarray:
    """Return a Hann (raised‑cosine) window."""
    return 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))


def _hamming_window(n: int) -> np.ndarray:
    """Return a Hamming window."""
    return 0.54 - 0.46 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))


def _blackman_window(n: int) -> np.ndarray:
    """Return a Blackman window."""
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    i = np.arange(n, dtype=np.float64)
    return (a0
            - a1 * np.cos(2.0 * np.pi * i / (n - 1))
            + a2 * np.cos(4.0 * np.pi * i / (n - 1)))

# Core entry point

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the sandbox timeout

    # 1. Initialise – use the parent if present, otherwise random
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # sum = 1
    # enforce (approximate) symmetry – the optimum tends to be symmetric
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)

    n = cur.size
    # --------------------------------------------------------
    # 2. Pre‑compute FFT length and index map
    # --------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())   # next power of two >= 3n-2
    idx = (2 * n - 2) - np.arange(n)      # slice needed for gradient

    # --------------------------------------------------------
    # 3. Quick window‑based search (cheap, dense‑free)
    # --------------------------------------------------------
    def _bound_from_vec(v):
        """Return 2*n*max(conv(v,v)) assuming sum(v)=1."""
        a_fft = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return 2.0 * n * conv.max()

    best_vec = cur.copy()
    best_sc = _bound_from_vec(best_vec)

    # a small collection of families & parameters
    window_families = [
        (lambda n: np.ones(n, dtype=np.float64), []),                       # rect

        alpha = base_alpha

    # ------------------------------------------------------------

    # ------------------------------------------------------------
