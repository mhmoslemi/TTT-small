def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

                if tmp_sc < cur_score:
                    cur, cur_score = tmp, tmp_sc
                if tmp_sc < best_score:
                    best_seq, best_score = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- cheap random perturbation (exploration) -----
        if rng.random() < 0.01:
            pert = rng.normal(scale=0.001, size=n)
            cand_tmp = cur + pert
            cand_tmp = _project_simplex(cand_tmp)
            cand_sc = _score(cand_tmp, fft_len)
            if cand_sc < cur_score:
                cur, cur_score = cand_tmp, cand_sc
                if cand_sc < best_score:
                    best_seq, best_score = cand_tmp.copy(), cand_sc

        # ----- occasional linear‑programming boost -----
        if it % 1500 == 0:
            M = max_val
            # Toeplitz constraint matrix  (2n‑1) × n
            diags_data = [np.full(n - i, cur[i]) for i in range(n)]
            offsets = np.arange(n)
            A = diags(diags_data, offsets, shape=(2 * n - 1, n), format='csc')
            b_ub = np.full(2 * n - 1, M)
            c_obj = -np.ones(n)                    # maximise sum(b)
            try:
                res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                              bounds=(0, None), method='highs')
                if res.success:
                    b = np.maximum(res.x, 0.0)
                    if b.sum() > 0:
                        g = b / b.sum()           # normalise to simplex
                        t = 0.2
                        cand_lp = (1 - t) * cur + t * g
                        cand_lp = _project_simplex(cand_lp)
                        cand_lp_sc = _score(cand_lp, fft_len)
                        if cand_lp_sc < cur_score:
                            cur, cur_score = cand_lp, cand_lp_sc
                            if cand_lp_sc < best_score:
                                best_seq, best_score = cand_lp.copy(), cand_lp_sc
            except Exception:
                pass

        # ----- early exit if target reached -----
        if best_score < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
