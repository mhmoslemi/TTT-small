def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    # initialization

        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)

        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 /= s
    else:
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 /= a0.sum()
    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    cur = a0.copy()
    cur_sc = _score(cur, n, L)  # need to pass n, L
    best_vec = cur.copy()
    best_sc = cur_sc
    momentum = np.zeros_like(cur)
    beta = 0.95
    base_alpha = 1.0
    max_trials = 12
    sym_interval = 250
    improve_target = 1.50274349
    no_improve = 0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # compute FFT and convolution
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        # compute second max
        # set argmax to -inf
        # Using copy to not modify conv: 
        idx_max = conv.argmax()
        sec_max = conv.copy()
        sec_max[idx_max] = -np.inf
        second_val = sec_max.max()
        diff = max_val - second_val
        # epsilon for weight mask
        eps = max(diff, max_val * 1e-3, 1e-12)  # ensure non-zero
        # active mask
        mask = (conv >= max_val - eps).astype(np.float64)
        w_sum = mask.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = mask / w_sum
        # gradient via convolution with reversed weights
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        # step size
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / math.sqrt(it) / norm_m
        improved = False
        cand = None
        cand_sc = None
        a_try = cur
        for _ in range(max_trials):
            cand_try = a_try - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try, n, L)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            # occasional uphill acceptance (simulated annealing)
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try, n, L)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True
        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        # symmetrisation
        if it % sym_interval == 0:

            sym = _project_simplex(sym)
            sym_sc = _score(sym, n, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0
        # random mass transfer when stuck
        if no_improve > 1500:
            # pick two positions, move some mass
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n)
                if j == i:
                    j = (j + 1) % n
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, n, L)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    # not improving; maybe keep trying random
                    no_improve = max(no_improve - 100, 0)
        if no_improve > 5000:

            cur = best_vec + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur /= cur.sum()
            momentum.fill(0.0)
            no_improve = 0
        if best_sc < improve_target:
            break
    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Exact AC1 bound for a sequence with sum ≈ 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
