import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _lp_optimize(a, M):
    """Solve the linear program that maximises sum(g) subject to
    a * g (convolution) ≤ M and g ≥ 0.
    Returns the optimal g (numpy array) or None on failure."""
    try:
        import cvxpy as cp
    except Exception:
        return None
    n = a.shape[0]
    g = cp.Variable(n, nonneg=True)
    a_param = cp.Parameter(n, value=a, constant=True)
    conv_expr = cp.conv(a_param, g)          # length 2n‑1
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv_expr <= M])
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=3000,
                   verbose=False)
    except Exception:
        return None
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None
    gv = g.value
    if gv is None:
        return None
    return np.asarray(gv, dtype=np.float64)
