def _project_simplex(v: np.ndarray) -> np.ndarray:
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)

def _max_conv(a: np.ndarray, L: int) -> float:

    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())

            sym_sc = _max_conv(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    best_bound = 2.0 * n * best_sc
        if best_bound < 1.50274349:
            break

    # ---- deterministic anti‑identity permutation ----
    vals = np.sort(best)[::-1]
    perm = np.empty_like(vals)
    l, r, k = 0, n - 1, 0
    while l <= r:
        perm[l] = vals[k]; k += 1
        if l < r:
            perm[r] = vals[k]; k += 1
        l += 1; r -= 1
    perm_sc = _max_conv(perm, L)
    perm_bound = 2.0 * n * perm_sc
    if perm_bound < best_bound:
        best, best_sc = perm.copy(), perm_sc
        best_bound = perm_bound

    # ---- intensive hill‑climbing swaps ----
    cur, cur_bound = best.copy(), best_bound
    while time.monotonic() < deadline - 0.5:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_sc = _max_conv(cur, L)
        new_bound = 2.0 * n * new_sc
        if new_bound < cur_bound:
            cur_bound = new_bound
            best, best_sc = cur.copy(), new_sc
            best_bound = new_bound
            if best_bound < 1.50274349:
                break
        else:
            cur[i], cur[j] = cur[j], cur[i]

    # ---- final gradient refinement (optional) ----
    cur = best.copy()
    cur_sc = best_sc
    momentum = np.zeros_like(cur)
    it = 0
    while time.monotonic() < deadline - 0.2:
        it += 1
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        mask = np.abs(conv - max_val) <= 1e-12
        cnt = mask.sum()
        w = mask.astype(np.float64) / max(cnt, 1)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        momentum = beta * momentum + (1.0 - beta) * g
        alpha = step0 / math.sqrt(it)
        cand = _project_simplex(cur - alpha * momentum)
        cand_sc = _max_conv(cand, L)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                best_bound = 2.0 * n * best_sc
        if best_bound < 1.50274349:
            break

    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best]
