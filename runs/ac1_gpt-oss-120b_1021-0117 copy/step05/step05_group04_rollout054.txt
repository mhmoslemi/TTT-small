def _full_conv_fft(a: np.ndarray, L: int):
    """Return rFFT of a and its full linear convolution a*a (length 2·len(a)-1)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return a_fft, conv

def _subgradient_from_fft(a_fft: np.ndarray, conv: np.ndarray,
                          idx_max: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Exact subgradient of max(conv) w.r.t. a.
    idx_max – indices where conv attains its maximum.
    """
    w = np.zeros_like(conv)
    w[idx_max] = 1.0
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
    # grad_i = 2 * sum_{k in S} a_{k-i}
    slice_idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[slice_idx]
