def _conv_max(v):
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(vf * vf, n=L)[: 2 * n - 1]
    return conv.max()

def _score(v):

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(v: np.ndarray) -> float:
    """Maximum value of the full autocorrelation of v (sum(v) assumed = 1)."""
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(vf * vf, n=L)[: 2 * n - 1]
    return float(conv.max())


def _score(v: np.ndarray) -> float:
    """Exact AC1 bound (no scaling) for a vector with sum≈1."""
    s = v.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * v.shape[0] * _conv_max(v) / (s * s))
