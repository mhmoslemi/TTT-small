    # momentum optional
    momentum = np.zeros_like(a)
    beta = 0.9
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # compute conv via FFT
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        cur_max = conv.max()
        cur_score = 2 * n * cur_max
        # check best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            # maybe early exit if below target
            if best_score < 1.50274349:
                break
        # compute gradient for max index
        k = int(np.argmax(conv))
        # gradient

        i_end = min(n-1, k)
        # i_start ... i_end inclusive
        idx = np.arange(i_start, i_end+1)
        g[idx] = 2 * a[k - idx]
        # accumulate momentum
        momentum = beta * momentum + (1 - beta) * g
        # step size schedule
        alpha = 0.5 / np.sqrt(it)

            a_candidate = a - alpha * momentum

            conv_c = np.fft.irfft(afc * afc, n=L)[:2*n-1]
            cand_score = 2 * n * conv_c.max()
            if cand_score <= cur_score:
                a = a_candidate
                break

        else:

            pass

import time

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Convolution via FFT, returning full array and its maximum
# ------------------------------------------------------------
def _conv_full_and_max(a: np.ndarray, L: int):
    """Full a*a convolution (real) using an FFT of length L, plus max."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv, float(conv.max())

        # accept if better, otherwise Metropolis‑style uphill move
        if cand_sc < cur_sc:
            a, conv, cur_max = cand, conv_c, max_c
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (lr0 + 1e-12))
            if rng.random() < prob:
                a, conv, cur_max = cand, conv_c, max_c

        # ------------------------------------------------------------------
        # Periodic symmetrisation – often reduces the central peak
        # ------------------------------------------------------------------
        if it % 250 == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            conv_sym, max_sym = _conv_full_and_max(a_sym, L)
            sc_sym = 2.0 * n * max_sym
            if sc_sym < cur_sc:
                a, conv, cur_max = a_sym, conv_sym, max_sym
                if sc_sym < best_sc:
                    best_sc, best_vec = sc_sym, a_sym.copy()
                    no_improve = 0

        # ------------------------------------------------------------------
        # Random mass‑transfer when progress stalls
        # ------------------------------------------------------------------
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.45
                a[i] -= delta
                a[j] += delta
                a /= a.sum()
                no_improve = 0

        # ------------------------------------------------------------------
        # Occasional restart from the best point with a tiny perturbation
        # ------------------------------------------------------------------
        if no_improve > 8000:
            a = best_vec.copy()
            a += rng.normal(scale=0.005, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    # ------------------------------------------------------------------
    # Final short fine‑tuning (very cheap)
    # ------------------------------------------------------------------
    a = best_vec.copy()
    for _ in range(150):
        conv, cur_max = _conv_full_and_max(a, L)
        cur_sc = 2.0 * n * cur_max
        k = int(np.argmax(conv))
        g = _grad_max_entry(a, k)
        step = 0.02 * g
        cand = a - step
        cand = _project_simplex(cand)
        conv_c, max_c = _conv_full_and_max(cand, L)
        cand_sc = 2.0 * n * max_c
        if cand_sc < cur_sc:
            a = cand
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            break

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
