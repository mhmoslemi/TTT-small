import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _score(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initial vector ----------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2000, dtype=np.float64)
    a0 = np.maximum(a0, 0.0)
    cur = _project_simplex(a0)            # sum = 1, all non‑negative
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_score = _score(best)
    cur_score = best_score

    # Adam parameters ---------------------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # soft‑max (smooth max) parameters ---------------------------------------
    beta = 0.2          # initial inverse temperature
    beta_max = 20.0
    beta_inc = 0.001

    lr0 = 0.5            # base learning rate
    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ---- convolution and smooth max ------------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # soft‑max weights over the convolution entries
        exp_w = np.exp(beta * (conv_full - max_val))      # max → 1, others ≤ 1
        w = exp_w / exp_w.sum()

        # ---- gradient of the smooth max ------------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ---- Adam update ---------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step

        # enforce symmetry (helps flatten the convolution)
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_score = _score(cand)

            cur_score = cand_score
            if cand_score < best_score:

                best_score = cand_score

                best_score = tmp_score
