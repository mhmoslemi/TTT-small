def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        init = np.array(height_sequence_1, dtype=np.float64)
        init = np.clip(init, 0.0, 1000.0)
        s = init.sum()
        if s < 1e-12:
            init = np.full(n, 1.0/n, dtype=np.float64)
        else:
            init = init / s   # scale to sum 1
    else:
        n = 2000
        init = np.full(n, 1.0/n, dtype=np.float64)

    # Ensure sum exactly 1
    init = init / init.sum()

    # Evaluate
    try:
        best_score = evaluate_sequence(init.tolist())
    except Exception:
        best_score = float('inf')
    best_vec = init.copy()
    cur = init.copy()
    cur_score = best_score

    rng = np.random.default_rng(seed)

    # Parameters
    max_iters = 10**7  # big bound, but will stop by time.
    step_factor = 0.5   # max portion of source mass to move
    large_move_prob = 0.005
    temperature0 = 0.1
    iter_count = 0

    while time.monotonic() < deadline and iter_count < max_iters:
        iter_count += 1
        # adapt step_factor gradually? Keep constant.
        if rng.random() < large_move_prob:
            # large rearrangement: pick three indices
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # move entire mass of i to j (zero i)
            delta = cur[i]
            if delta <= 0:
                continue
            cand = cur.copy()
            cand[i] = 0.0
            cand[j] += delta
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # Move a random fraction up to step_factor of source mass
            max_move = cur[i] * step_factor
            if max_move <= 0:
                continue
            delta = rng.random() * max_move
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        # Renormalize to sum 1 (to correct floating errors)
        cand_sum = cand.sum()
        if cand_sum <= 0:
            continue
        cand = cand / cand_sum

        # Evaluate
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue

        # Accept if better, else maybe accept worse with prob
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_vec = cand.copy()
                best_score = cand_score
        else:
            # simulated annealing acceptance
            # temperature decreasing linearly
            t = temperature0 * max(0.0, (deadline - time.monotonic()) / (deadline - start))
            if t > 0 and rng.random() < np.exp(-(cand_score - cur_score) / t):
                cur = cand
                cur_score = cand_score

    return [float(x) for x in best_vec]

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound as measured by the pre‑imported
    ``evaluate_sequence``.
    The routine works with any parent sequence ``height_sequence_1`` that may
    already be present; otherwise it falls back to a random simplex start.
    """
    start = time.monotonic()
    # keep a safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ----------------------------------------------------------------------
    # Problem size and initial vector
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        cur = np.asarray(height_sequence_1, dtype=np.float64)
        cur = np.clip(cur, 0.0, 1000.0)
        s = cur.sum()
        if s <= 0.0:
            cur = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            cur = cur / s                     # normalise to sum == 1
    else:
        n = 2000
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    # ----------------------------------------------------------------------
    # Helper to evaluate a candidate safely
    def _score(vec):
        try:
            return float(evaluate_sequence(vec.tolist()))
        except Exception:
            return float("inf")

    # initialise best
    best_vec = cur.copy()
    best_score = _score(best_vec)
    cur_score = best_score

    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Parameters for the stochastic search
    prob_grad = 0.25          # how often to use a gradient step
    prob_large = 0.005        # occasional large rearrangement
    step_factor = 0.5         # max proportion of source mass to move
    alpha_grad = 0.12         # step size for gradient descent
    no_improve_limit = 5000   # restart after this many failed steps
    iters_without_improve = 0

    # ----------------------------------------------------------------------
    # Main optimisation loop
    while time.monotonic() < deadline:
        # compute convolution and current peak information (shared for both move types)
        conv = np.convolve(cur, cur)
        max_val = conv.max()
        kmax = int(conv.argmax())

        # decide which move to make
        if rng.random() < prob_large:
            # large move: transfer the whole mass of one index to another
            i, j = rng.integers(0, n, size=2)
            if i == j or cur[i] <= 0.0:
                continue
            cand = cur.copy()
            delta = cand[i]
            cand[i] = 0.0
            cand[j] += delta
        elif rng.random() < prob_grad:
            # gradient descent step aimed at the current maximal convolution entry
            grad = np.zeros_like(cur)
            lo = max(0, kmax - (n - 1))
            hi = min(n - 1, kmax)
            # indices lo..hi receive gradient 2 * cur[kmax - i]
            idx = np.arange(lo, hi + 1)
            grad[idx] = 2.0 * cur[kmax - idx]
            # take a step opposite to the gradient
            cand = cur - alpha_grad * grad
            cand = np.clip(cand, 0.0, None)
        else:
            # simple local transfer of a fraction of mass from i to j
            i, j = rng.integers(0, n, size=2)
            if i == j or cur[i] <= 0.0:
                continue
            max_move = cur[i] * step_factor
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        # renormalise to keep sum exactly 1 (helps the evaluator)
        s = cand.sum()
        if s <= 0.0:
            continue
        cand /= s

        cand_score = _score(cand)
        # accept if we improve the current point
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            iters_without_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
        else:
            iters_without_improve += 1
            # occasional uphill move (simulated annealing style)
            # temperature linearly decays to 0
            remaining = max(0.0, deadline - time.monotonic())
            total = max(1e-12, deadline - start)
            temp = remaining / total * 0.1  # max temperature 0.1
            if temp > 0 and rng.random() < np.exp(-(cand_score - cur_score) / temp):
                cur = cand
                cur_score = cand_score
                iters_without_improve = 0

        # restart if stuck too long
        if iters_without_improve >= no_improve_limit:
            # fresh random point on the simplex
            rnd = rng.exponential(scale=1.0, size=n)
            rnd /= rnd.sum()
            cur = rnd
            cur_score = _score(cur)
            iters_without_improve = 0
            # keep the global best
            if cur_score < best_score:
                best_score = cur_score
                best_vec = cur.copy()

    # ----------------------------------------------------------------------
    # Fallback safety: if somehow we never obtained a finite score, use uniform
    if not np.isfinite(best_score):
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # Ensure final answer respects evaluator clipping rules
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    return [float(x) for x in best_vec]
