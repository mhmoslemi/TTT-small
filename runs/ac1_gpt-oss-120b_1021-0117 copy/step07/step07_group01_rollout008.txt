def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # init
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2500, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    # project onto simplex sum=1
    a = _project_simplex(a)
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    # helpers
    def score(v):
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())
    def grad(v):
        af = np.fft.rfft(v, n=L)
        conv_full = np.fft.irfft(af * af, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
        else:
            w = mask.astype(np.float64) / cnt
        w_rev = w[::-1]
        wf = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * wf, n=L)
        idx = (2*n-2) - np.arange(n)
        return 2.0 * grad_full[idx]
    best = a.copy()
    best_sc = score(best)

    # Adam state
    m = np.zeros_like(a)
    v_hat = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.5
    it = 0
    no_improve = 0
    target = 1.50274349

    # loop
    while time.monotonic() < deadline:
        it += 1
        # compute gradient
        g = grad(a)
        # Adam moments
        m = beta1 * m + (1 - beta1) * g
        v_hat = beta2 * v_hat + (1 - beta2) * (g * g)

import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(v):
    """Score 2*n*max(conv(v,v)) assuming sum(v)=1."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _grad(v):
    """Sub‑gradient of max(conv(v,v)) w.r.t. v (sum(v)=1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(v, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = conv >= max_val - eps
    cnt = mask.sum()
    if cnt == 0:
        w = np.zeros_like(conv)
    else:
        w = mask.astype(np.float64) / cnt          # uniform on active set
    w_rev = w[::-1]

    grad_full = np.fft.irfft(af * wf, n=L)
    idx = (2 * n - 2) - np.arange(n)
