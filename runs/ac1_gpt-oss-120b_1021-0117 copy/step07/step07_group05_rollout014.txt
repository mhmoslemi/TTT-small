import time

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x>=0, sum(x)=1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1, dtype=np.float64)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = css[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Helper: full convolution via FFT (no dense Toeplitz)
def _conv_full(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x with itself."""
    n = x.size
    size = 1
    while size < 2 * n:
        size <<= 1
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)
    return conv[: 2 * n - 1]

# Helper: objective value (the AC1 upper bound)
def _objective(x: np.ndarray) -> float:
    """Compute 2*n*max(conv)/sum(x)^2 for a non‑negative vector."""
    s = np.sum(x)
    if s <= 0.0:
        return math.inf
    conv = _conv_full(x)
    m = np.max(conv)
    n = x.size
    return 2.0 * n * m / (s * s)

# indices attaining the maximum).
def _gradient(x: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """Gradient of max(conv(x,x)) w.r.t. x (sum over all maximal positions)."""
    n = x.size
    grad = np.zeros_like(x)

        i_range = np.arange(lo, hi + 1, dtype=np.intp)
        j_range = k - i_range
        grad[i_range] += 2.0 * x[j_range]
