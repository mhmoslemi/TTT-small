import time, math, numpy as np

def _project_simplex(v):
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    """Fast score for a vector a with sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (alpha / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + epsilon)

        cand = cur - step
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_sc = _score_simplex(cand, n)

        # ----- accept or reject -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            # occasional uphill move (simulated annealing)
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # ----- random redistribution when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_simplex(tmp, n)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

        # ----- symmetrisation every 1000 iterations -----
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_simplex(sym, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # Return best vector as plain Python floats
    return [float(x) for x in best_vec]
