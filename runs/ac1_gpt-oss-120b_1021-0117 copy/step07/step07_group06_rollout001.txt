import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a):
    """Score when sum(a)=1: 2 * n * max autocorrelation."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    start = time.monotonic()

    # ----- initialization -----
    if "height_sequence_1" in globals():
        cur = _project_simplex(np.maximum(height_sequence_1, 0.0))
    else:
        cur = _project_simplex(rng.random(2000) + 1e-6)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # Adam parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # schedule parameters
    lr0 = 0.6                # initial learning rate
    alpha_max = 80.0        # max softmax temperature
    switch_iter = 800       # when to use hard max mask
    target = 1.50274349

    it = 0

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        a_fft = np.fft.rfft(cur, n=L)

        conv = conv_full[: 2 * n - 1]

        # ----- weighting for gradient -----
        if it < switch_iter:

            w = np.exp(alpha * (conv - max_val))

            else:

            cand, cand_sc = cand_sym, cand_sym_sc

        # ----- acceptance -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1

        # ----- periodic forced symmetrisation -----
        if it % 300 == 0:
            sym = _project_simplex((cur + cur[::-1]) * 0.5)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # ----- random perturb when stuck -----
        if no_improve > 2500:
            tmp = cur + rng.normal(scale=0.02, size=n)
            tmp = np.maximum(tmp, 0.0)
            cur = _project_simplex(tmp)
            no_improve = 0
            m.fill(0.0)
            v.fill(0.0)

        # ----- early exit -----
        if best_sc < target:
            break

    # ----- return best found -----
    return [float(x) for x in best_vec]
