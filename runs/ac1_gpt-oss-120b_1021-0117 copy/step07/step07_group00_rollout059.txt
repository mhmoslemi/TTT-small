import time, math, numpy as np

def _project_simplex(v):
    # Euclidean projection onto simplex sum=1, v>=0
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of vector v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # hyper‑parameters
    step0 = 1.1          # initial step size
    beta = 0.92          # momentum decay
    momentum = np.zeros_like(cur)
    temp = None
    decay = 0.998        # temperature decay (slow)
    min_temp_factor = 5e-5
    it = 0
    no_improve = 0
    target = 1.50274349

    # --------------------------------------------------------------
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # --------- convolution and temperature ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.08
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # --------- soft‑max weights ----------
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:                     # hard max fallback
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # --------- gradient ----------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # --------- momentum & step ----------
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        # --------- additive candidate ----------
        cand_add = _project_simplex(cur - step * momentum)
        cand_add_sc = _score(cand_add, L)

        improved = False
        # accept additive if it improves
        if cand_add_sc < cur_sc:
            cur, cur_sc = cand_add, cand_add_sc
            improved = True
        else:
            # --------- multiplicative (exponential) candidate ----------
            expo = -step * grad
            expo = np.clip(expo, -20, 20)          # avoid overflow
            cand_mul = cur * np.exp(expo)
            cand_mul = _project_simplex(cand_mul)
            cand_mul_sc = _score(cand_mul, L)

            # pick the better of the two proposals
            if cand_mul_sc < cand_add_sc:
                cand, cand_sc = cand_mul, cand_mul_sc
            else:
                cand, cand_sc = cand_add, cand_add_sc

            # simulated‑annealing acceptance
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                improved = True
            else:
                prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
                if rng.random() < prob:
                    cur, cur_sc = cand, cand_sc
                    improved = True

        # --------- bookkeeping ----------
        if improved:
            no_improve = 0
            if cur_sc < best_sc:
                best_vec, best_sc = cur.copy(), cur_sc
        else:
            no_improve += 1

        # --------- random redistribution when stuck ----------
        if no_improve > 1500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.35
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # --------- periodic symmetrisation ----------
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # --------- occasional smoothing ----------
        if it % 600 == 0:
            kernel = np.array([0.25, 0.5, 0.25])
            smooth = np.convolve(cur, kernel, mode="same")
            cur = _project_simplex(smooth)
            cur_sc = _score(cur, L)
            if cur_sc < best_sc:
                best_vec, best_sc = cur.copy(), cur_sc

        # early exit if target reached
        if best_sc < target:
            break

    # --------------------------------------------------------------
    # final lightweight local search (pairwise mass transfer)
    while time.time() < deadline - 0.1:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps = cur[i] * rng.random() * 0.18
        if eps <= 0:
            continue
        tmp = cur.copy()
        tmp[i] -= eps
        tmp[j] += eps
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp, L)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc

        # very occasional symmetrisation to keep structure
        if rng.random() < 0.02:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

    # --------------------------------------------------------------
    # return the best candidate found
    return [float(x) for x in best_vec]
