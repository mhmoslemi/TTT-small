import time, numpy as np

def _project_simplex(v):
    v = np.asarray(v)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

            max_sym = conv_sym_full[: 2 * n - 1].max()
            sym_score = 2.0 * n * max_sym
            if sym_score < best_score:
                best = sym.copy()
                best_score = sym_score
                a = sym
                cur_score = sym_score
                no_improve = 0

        # ----- random redistribution when stuck -------------------------
        if no_improve > 5000:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                a_tmp = a.copy()
                a_tmp[i] -= delta
                a_tmp[j] += delta
                a_tmp = _normalize(a_tmp)

                a_tmp_fft = np.fft.rfft(a_tmp, n=L)
                conv_tmp_full = np.fft.irfft(a_tmp_fft * a_tmp_fft, n=L)
                max_tmp = conv_tmp_full[: 2 * n - 1].max()
                tmp_score = 2.0 * n * max_tmp

                if tmp_score < best_score:
                    best = a_tmp.copy()
                    best_score = tmp_score
                a = a_tmp
                cur_score = tmp_score
                no_improve = 0

        # ----- early termination -----------------------------------------
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # safety fallback (should never be needed)
    if best_score == float("inf") or np.isnan(best_score):
        best = np.full(n, 1.0 / n)
    return [float(x) for x in best]
