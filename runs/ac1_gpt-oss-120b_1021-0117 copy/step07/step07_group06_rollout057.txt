    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2) # safety margin

    # Determine length n
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = a0.size
        init_vec = a0
    else:
        # choose some default length, say 2000
        n = 2000
        init_vec = rng.random(n, dtype=np.float64) + 1e-3

    best_score = _score(best_overall)

    # Also consider uniform start

    uniform_sc = _score(uniform)
    if uniform_sc < best_score:
        best_overall = uniform.copy()
        best_score = uniform_sc

    # Define function to perform a single optimization run
    def optimize(start_vec, max_iters, time_limit):
        cur = start_vec.copy()
        cur_sc = _score(cur)
        best_vec = cur.copy()
        best_sc = cur_sc

        m = np.zeros_like(cur)
        v = np.zeros_like(cur)
        beta1 = 0.9
        beta2 = 0.999
        epsilon = 1e-8
        step0 = 0.6
        alpha0 = 1.0
        alpha_max = 200.0

        L = 1 << ((2 * n - 1).bit_length())
        it = 0
        last_improve_it = 0

        while it < max_iters and time.monotonic() < time_limit:
            it += 1
            # FFT of current vector
            a_fft = np.fft.rfft(cur, n=L)
            # full convolution via FFT
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()

            # softmax weighting
            alpha = min(alpha_max, alpha0 * math.log1p(it))
            # shift for stability
            w = np.exp(alpha * (conv - max_val))
            w_sum = w.sum()
            if not np.isfinite(w_sum) or w_sum == 0.0:
                # fallback: equal weight on max entries
                mask = conv == max_val
                w = mask.astype(np.float64)
                w_sum = w.sum()

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a):
    """Score for a vector already normalized to sum=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
