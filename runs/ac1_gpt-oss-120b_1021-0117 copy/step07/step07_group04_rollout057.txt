import time, math, numpy as np

def _project_simplex(v):
    """Project onto the probability simplex (non‑negative, sum=1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Fix possible numerical drift
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast evaluation of the AC1 upper bound (assumes a >= 0)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # --------------------------------------------------------------
    # Length of the problem
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2000
    L = 1 << ((2 * n - 1).bit_length())

    xs = np.arange(n, dtype=np.float64)

    # --------------------------------------------------------------
    # Assemble deterministic initial candidates
    candidates = []

    # Uniform
    candidates.append(np.full(n, 1.0 / n))

    # Triangular
    tri = np.minimum(xs + 1, n - xs)
    candidates.append(tri / tri.sum())

    # Sine bump
    sine = np.sin(np.pi * (xs + 1) / (n + 1))
    candidates.append(sine / sine.sum())

    # Raised cosine
    bump = (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2
    candidates.append(bump / bump.sum())

    # Exponential decays
    for alpha in (0.001, 0.003, 0.006):
        exp = np.exp(-alpha * xs)
        candidates.append(exp / exp.sum())

    # Power laws
    for p in (0.5, 1.0, 1.5):
        pw = (xs + 1) ** (-p)
        candidates.append(pw / pw.sum())

    # Gaussian bump
    sigma = n / 10.0
    center = (n - 1) / 2.0
    gauss = np.exp(-((xs - center) / sigma) ** 2)
    candidates.append(gauss / gauss.sum())

    # Bimodal (two Gaussians)
    sigma2 = n / 20.0
    mu1, mu2 = n / 4.0, 3 * n / 4.0
    bimodal = np.exp(-((xs - mu1) / sigma2) ** 2) + np.exp(-((xs - mu2) / sigma2) ** 2)
    candidates.append(bimodal / bimodal.sum())

    # Random Dirichlet
    rand = rng.random(n)
    candidates.append(rand / rand.sum())

    # Parent if supplied
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.sum() > 0:
            parent = parent / parent.sum()
            candidates.append(parent)

    # --------------------------------------------------------------
    # Pick best deterministic start
    best_vec = None
    best_sc = float('inf')
    for vec in candidates:
        sc = _score_fft(vec, n, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    # --------------------------------------------------------------
    # Adam optimiser state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.05
    beta_soft = 20.0   # softness of max‑weighting
    target = 1.50274349

    it = 0
    no_improve = 0

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution and max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # ----- softmax weights over the convolution -----
        w_raw = np.exp(beta_soft * (conv - max_conv))
        w = w_raw / w_raw.sum()

        # ----- gradient (sub‑gradient of max) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam moment updates -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- adaptive step -----
        step_factor = lr0 / math.sqrt(it)
        step_vec = step_factor * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step_vec
        cand = _project_simplex(cand)
        cand_sc = _score_fft(cand, n, L)

        # ----- acceptance -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional simulated‑annealing move
            temp = 0.1 / math.sqrt(it)
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (temp + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # ----- global best update -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()

        # ----- periodic symmetrisation -----
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
            if sym_sc < best_sc:
                best_sc, best_vec = sym_sc, sym.copy()

        # ----- random perturbation when stuck -----
        if no_improve >= 1000:
            pert = cur + rng.normal(scale=0.01, size=n)
            pert = np.maximum(pert, 0.0)
            pert = _project_simplex(pert)
            pert_sc = _score_fft(pert, n, L)
            if pert_sc < cur_sc:
                cur, cur_sc = pert, pert_sc
                no_improve = 0
                if pert_sc < best_sc:
                    best_sc, best_vec = pert_sc, pert.copy()

        # ----- full restart after long stagnation -----
        if no_improve >= 5000:
            cur = rng.random(n)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            cur_sc = _score_fft(cur, n, L)

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
