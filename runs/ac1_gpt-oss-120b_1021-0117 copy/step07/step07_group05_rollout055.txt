def project_simplex(v):
    # v is np array
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)
    rng = np.random.default_rng(seed)

    # Determine n
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    # Helper: projection onto simplex
    def project_simplex(v):
        # using Duchi algorithm
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1
        rho = np.where(u - css / np.arange(1, v.size+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    # Initial candidates
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    best_vec = uniform
    try:
        best_score = evaluate_sequence(best_vec.tolist())

        best_score = float('inf')

        parent = np.clip(parent, 0.0, 1000.0)
        s = parent.sum()
        if s > 0:
            parent = parent / s  # normalize sum to 1
            try:
                parent_score = evaluate_sequence(parent.tolist())
                if parent_score < best_score:
                    best_score = parent_score
                    best_vec = parent
            except Exception:
                pass

    cur_score = best_score

    max_iters = 50000
    it = 0
    while time.monotonic() < deadline and it < max_iters:

        max_frac = 0.2

                if parent_score < best_score:
                    best_score = parent_score
                    best_vec = parent
            except Exception:
                pass

    cur = best_vec.copy()
    cur_score = best_score

    # --------------------------------------------------------------
    # Search parameters
    max_iters = 20000            # generous but safe
    base_frac = 0.20             # maximum fraction of source mass to move
    min_frac = 0.01              # lower bound for the fraction

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # ---- simple random transfer between two indices -----------------
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # linear cooling of the step size
        frac = base_frac * (1.0 - it / max_iters)
        if frac < min_frac:
            frac = min_frac
        delta = cur[i] * rng.random() * frac
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0.0:
            cand[i] = 0.0
        cand = _project_simplex(cand)

        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand

        # ---- every 200 steps: targeted move to reduce the current max ----
        if it % 200 == 0:
            conv = np.convolve(cur, cur)                # full convolution
            idx_max = int(np.argmax(conv))

            low = max(0, idx_max - (n - 1))
            high = min(n - 1, idx_max)
            contrib_idx = np.arange(low, high + 1)

            probs = cur[contrib_idx]
            if probs.sum() > 0.0:
                source = rng.choice(contrib_idx, p=probs / probs.sum())
            else:
                source = rng.integers(0, n)

            target = rng.integers(0, n)
            while target == source:
                target = rng.integers(0, n)

            delta2 = cur[source] * rng.random() * 0.5   # more aggressive
            cand2 = cur.copy()
            cand2[source] -= delta2
            cand2[target] += delta2
            if cand2[source] < 0.0:
                cand2[source] = 0.0
            cand2 = _project_simplex(cand2)

            try:
                cand2_score = evaluate_sequence(cand2.tolist())
            except Exception:
                continue

            if cand2_score < cur_score:
                cur = cand2
                cur_score = cand2_score
                if cand2_score < best_score:
                    best_score = cand2_score
                    best_vec = cand2

        # ---- occasional tiny perturbation to escape plateaus ------------
        if it % 1000 == 0:
            noise = rng.random(n) * 1e-5
            cand3 = cur + noise
            cand3 = _project_simplex(cand3)
            try:
                cand3_score = evaluate_sequence(cand3.tolist())
                if cand3_score < cur_score:
                    cur = cand3
                    cur_score = cand3_score
                    if cand3_score < best_score:
                        best_score = cand3_score
                        best_vec = cand3
            except Exception:
                pass

    # --------------------------------------------------------------
    # Return the best vector found (as a plain Python list of floats)
    return [float(x) for x in best_vec]
