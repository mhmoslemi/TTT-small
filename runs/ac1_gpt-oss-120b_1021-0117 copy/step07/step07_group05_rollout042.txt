    Returns np.inf if the input is invalid.

    if not isinstance(sequence, list):

    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):

    sequence = [float(x) for x in sequence]

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as measured by ``evaluate_sequence``.
    The routine works with any parent sequence ``height_sequence_1`` that may
    already be present; otherwise it starts from a uniform distribution and
    performs a fast stochastic projected‑gradient‑like search.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ----------------------------------------------------------------------
    # problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000                     # reasonable default

    # ----------------------------------------------------------------------
    # Euclidean projection onto the probability simplex (sum == 1, non‑negative)
    def _project_simplex(v):
        v = np.asarray(v, dtype=np.float64)
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Initial candidate: uniform distribution (sum == 1)
    a = np.full(n, 1.0 / n, dtype=np.float64)
    a = _project_simplex(a)

    # Evaluate the initial candidate
    try:
        cur = a.copy()
        cur_score = evaluate_sequence(cur.tolist())
    except Exception:                     # defensive – should not happen
        cur = a.copy()
        cur_score = float("inf")

    best_vec = cur.copy()
    best_score = cur_score

    # ----------------------------------------------------------------------
    # Optional parent sequence – keep it if it happens to be better
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        try:
            parent_score = evaluate_sequence(parent.tolist())
            if parent_score < best_score:
                best_score = parent_score
                best_vec = parent.copy()
                cur = parent.copy()
                cur_score = parent_score
        except Exception:
            pass

    # ----------------------------------------------------------------------
    # Search parameters
    step_frac = 0.3                 # maximal fraction of source mass to move
    it = 0
    decay_interval = 5000
    decay_factor = 0.9

    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        # full convolution and the index of the maximal entry
        conv = np.convolve(cur, cur)
        k_max = int(np.argmax(conv))

        # contributions of each coordinate to that maximal entry
        contrib = np.zeros(n, dtype=np.float64)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        idx2 = k_max - idx
        contrib[idx] = cur[idx] * cur[idx2]

        # source index – bias toward large contributions
        src_weights = contrib + 1e-16
        src = rng.choice(n, p=src_weights / src_weights.sum())

        # destination index – bias toward small contributions
        dst_weights = (contrib.max() - contrib) + 1e-16
        if dst_weights.sum() == 0:
            dst = rng.integers(0, n)
        else:
            dst = rng.choice(n, p=dst_weights / dst_weights.sum())

        if src == dst:
            continue

        # move a random fraction of the source mass to the destination
        delta = cur[src] * step_frac * rng.random()
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = _project_simplex(cand)

        # evaluate the candidate
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue

        # accept the move (simple hill‑climb with a tiny annealing tail)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
        else:
            # temperature proportional to remaining wall‑clock time
            temp = max(1e-6, (deadline - time.monotonic()) / float(budget_s))

                cur_score = cand_score

        # keep the best ever seen
        if cand_score < best_score:
            best_score = cand_score
            best_vec = cand.copy()
            if best_score < 1.50274349:      # target reached – stop early
                break

        # adapt step size slowly
        it += 1
        if it % decay_interval == 0:
            step_frac = max(step_frac * decay_factor, 0.005)

    # Return the best vector as a plain Python list
    return [float(x) for x in best_vec]
