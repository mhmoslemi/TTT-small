import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast evaluation of the AC1 bound for a non‑negative vector a (sum assumed 1)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # --------------------------------------------------------------
    # Length and FFT size
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))

        n = 2000

    candidates.append(sine / sine.sum())
    bump = (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2        # raised cosine
