import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    step0_adam = 0.6
    step0_exp = 0.5
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ---- convolution ----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ---- smooth weights for sub‑gradient ----
        alpha = min(30.0, it * 0.01)               # continuation parameter
        w_unnorm = np.exp(alpha * (conv - max_val))
        w_sum = w_unnorm.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = w_unnorm / w_sum

        # ---- gradient of smooth max ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---- Adam additive update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step_adam = step0_adam / (math.sqrt(it) + 1.0)
        update = step_adam * m_hat / (np.sqrt(v_hat) + eps)
        cand_add = _project_simplex(a - update)

        # ---- exponentiated (multiplicative) update ----
        step_exp = step0_exp / (math.sqrt(it) + 1.0)
        exp_factor = np.exp(-step_exp * grad)
        cand_mul = a * exp_factor
        s = cand_mul.sum()
        if s > 0:
            cand_mul = cand_mul / s
        else:
            cand_mul = a.copy()

        # ---- evaluate candidates ----
        cand_add_sc = _score(cand_add, L)
        cand_mul_sc = _score(cand_mul, L)

        if cand_add_sc < cand_mul_sc:
            cand, cand_sc = cand_add, cand_add_sc
        else:
            cand, cand_sc = cand_mul, cand_mul_sc

        # ---- acceptance ----
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            a = cand
            no_improve = 0
        elif cand_sc < _score(a, L):
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - best_sc) / (step_adam + 1e-12))
            if rng.random() < prob:
                a = cand

        # ---- periodic symmetrisation ----
        if it % 300 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
                a = sym
                no_improve = 0

        # ---- occasional smoothing kernel ----
        if it % 500 == 0:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            a_smooth = np.convolve(a, kernel, mode='same')
            a_smooth = _project_simplex(a_smooth)
            smooth_sc = _score(a_smooth, L)
            if smooth_sc < best_sc:
                best = a_smooth.copy()
                best_sc = smooth_sc
                a = a_smooth
                no_improve = 0

        # ---- random mass transfer when stagnating ----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.25
                j = rng.integers(0, n)
                if j == i:
                    j = (j + 1) % n
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                a = tmp
                no_improve = 0

        # ---- full restart after long stagnation ----
        if no_improve > 5000:
            rand_vec = rng.random(n, dtype=np.float64)
            rand_vec = _project_simplex(rand_vec)
            a = 0.5 * best + 0.5 * rand_vec
            a = _project_simplex(a)
            no_improve = 0

        # ---- early exit ----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # post‑processing: fine‑grained random swaps while time permits
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] == 0.0:
            continue
        delta = a[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        tmp = a.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = np.maximum(tmp, 0.0)
        sm = tmp.sum()
        if sm == 0.0:
            continue
        tmp = tmp / sm
        sc = _score(tmp, L)
        if sc < best_sc:
            best = tmp.copy()
            best_sc = sc
            a = tmp
            if best_sc < target:
                break

    return [float(x) for x in best]
