def _project_simplex(v):
    """Project onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        update = step * m_hat / (np.sqrt(v_hat) + eps)

        # additive candidate
        cand_add = _project_simplex(a - update)

        # multiplicative candidate
        exp_factor = np.exp(-step * grad)
        cand_mul = a * exp_factor
        s = cand_mul.sum()
        cand_mul = cand_mul / s if s > 0 else a.copy()

        # random local move (small perturbation)
        cand_rand = a.copy()
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if cand_rand[i] > 0:
            delta = cand_rand[i] * 0.05 * rng.random()
            cand_rand[i] -= delta
            cand_rand[j] += delta
            cand_rand = _project_simplex(cand_rand)

        # evaluate all candidates
        sc_add = _score_simplex(cand_add)
        sc_mul = _score_simplex(cand_mul)
        sc_rand = _score_simplex(cand_rand)

        # pick the best among them
        cand, cand_sc = cand_add, sc_add
        if sc_mul < cand_sc:
            cand, cand_sc = cand_mul, sc_mul
        if sc_rand < cand_sc:
            cand, cand_sc = cand_rand, sc_rand

        # acceptance logic
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            a = cand
            no_improve = 0
        elif cand_sc < cur_sc:
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - best_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = cand

        # occasional symmetrisation
        if it % 200 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_simplex(sym)
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
                a = sym
                no_improve = 0

        # random redistribution when stuck
        if no_improve > 2000:
            i = rng.integers(0, n)
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                a = _project_simplex(tmp)
            no_improve = 0

        if no_improve > 5000:
            rand_vec = rng.random(n, dtype=np.float64)
            rand_vec = _project_simplex(rand_vec)
            a = 0.5 * best + 0.5 * rand_vec
            a = _project_simplex(a)
            no_improve = 0

        if best_sc < target:
            break

    # final safety clipping
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
