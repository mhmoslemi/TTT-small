def _score_fft(a, n, L):
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    return float(2.0 * n * conv.max() / (s * s))
```

import time

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w


# ----------------------------------------------------------------------
def _score_fft(a: np.ndarray, n: int, L: int) -> float:
    """Fast score (no clipping, assumes a>=0)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

             start_score: float,
             n: int,
             L: int,
             rng: np.random.Generator,
             deadline: float,

        it += 1

    for p in (0.5, 1.0, 1.5):

        candidates.append(pw / pw.sum())                     # power laws
    rand = rng.random(n)

    # ----- global best so far -----
    best_vec = scored[0][1].copy()
    best_sc = scored[0][0]

    target = 1.50274349
