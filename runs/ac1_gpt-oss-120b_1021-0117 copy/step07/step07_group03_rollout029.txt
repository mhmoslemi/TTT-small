def _conv_softmax_grad(a, L, beta=20.0):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]

    idx = (2*n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]

import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex (non‑negative, sum=1). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
def _conv_max(a, L):
    """Return max of full convolution a*a (linear, not circular)."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    return conv.max()

def _conv_max_softmax_grad(a, L, beta):
    """
    Return (max value, gradient) where gradient is the derivative of the
    smooth approximation of max given by softmax with temperature beta.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    max_val = conv.max()
    # softmax weights (stable)
    shifted = conv - max_val          # ≤0
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / exp_vals.sum()

    # gradient = 2 * (a * w_rev) linear convolution, extracted for first n entries
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimizes
    the autocorrelation upper bound used by evaluate_sequence().
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2000, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    # enforce (optional) symmetry – often beneficial
    a0 = (a0 + a0[::-1]) * 0.5
    cur = _project_simplex(a0)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1

    # ---- Adam optimizer state -------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.12                     # base learning rate
    beta1_pow, beta2_pow = 1.0, 1.0

    # ---- bookkeeping ----------------------------------------------------
    max_val, grad = _conv_max_softmax_grad(cur, L, beta=5.0)
    best_sc = 2.0 * n * max_val
    best_vec = cur.copy()
    cur_sc = best_sc

    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        # temperature schedule for smooth max (slowly tighten)
        beta_temp = min(30.0, 5.0 + it * 0.01)

        # gradient at current point
        max_val, grad = _conv_max_softmax_grad(cur, L, beta=beta_temp)
        cur_sc = 2.0 * n * max_val

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        delta = alpha * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - delta)

        # occasional symmetrisation (helps reduce peaks)
        if it % 250 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        cand_max = _conv_max(cand, L)
        cand_sc = 2.0 * n * cand_max

        # greedy acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # update best solution
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        # random perturbation when stuck
        if no_improve > 3000:
            sigma = 0.05 / math.sqrt(it)
            pert = cur + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(pert)
            pert_max = _conv_max(pert, L)
            pert_sc = 2.0 * n * pert_max
            if pert_sc < cur_sc:
                cur, cur_sc = pert, pert_sc
                if pert_sc < best_sc:
                    best_sc = pert_sc
                    best_vec = pert.copy()
                no_improve = 0
            else:
                # reset optimizer moments to escape flat region
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0

        # early termination if target reached
        if best_sc < target:
            break

    # return the best vector as plain Python floats
    return [float(x) for x in best_vec]
