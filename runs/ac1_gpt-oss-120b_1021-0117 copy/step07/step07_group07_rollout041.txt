    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)  # margin

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)  # sum = 1
    else:
        n = rng.integers(2000, 3000)
        a = rng.random(n)
        a = _project_simplex(a)

    n = a.shape[0]
    conv = np.convolve(a, a)  # length 2n-1
    max_conv = conv.max()

    best_a = a.copy()
    best_max = max_conv

    # Parameters
    step_factor = 0.5
    min_step_factor = 0.001
    decay = 0.9999  # per iteration, decays slowly

    no_improve = 0
    # maybe also track iteration count for periodic symmetrization
    it = 0

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # Choose i weighted by a
        # Ensure a has mass >0; if all equal maybe any.
        if a.max() <= 0:
            # all zeros? avoid; break
            break
        i = rng.choice(n, p=a)  # probabilities sum to 1
        # Ensure i's mass is > 0
        if a[i] <= 1e-12:
            continue
        # Choose j arbitrary not equal to i
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # delta random
        delta = a[i] * step_factor * rng.random()
        if delta <= 0:
            continue
        # compute diff update
        diff = np.empty_like(conv)
        diff.fill(0.0)
        # diff[j:j+n] += a
        diff[j:j+n] = a
        # diff[i:i+n] -= a

        new_conv = conv + 2.0 * delta * diff
        # delta^2 contributions
        # positions:
        pos2j = 2*j
        pos2i = 2*i
        posij = i + j

        d2 = delta * delta

        new_conv[pos2i] += d2

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, non‑negative)."""
    v = np.asarray(v, dtype=np.float64)

        theta = 0.0

        theta = css[rho] / (rho + 1)

        if j == i_star:
            continue

        # propose moving a small amount of mass from i_star to j
        delta = a[i_star] * step_factor * rng.random()
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear + quadratic parts)
        new_conv = conv.copy()
        new_conv[j:j + n] += 2.0 * delta * a
        new_conv[i_star:i_star + n] -= 2.0 * delta * a
        d2 = delta * delta
        new_conv[2 * j] += d2
        new_conv[2 * i_star] += d2
        new_conv[i_star + j] -= 2.0 * d2

        new_max = new_conv.max()

        # ------------------------------------------------------------------
        # accept the move only if it reduces the peak
        if new_max < max_conv - 1e-15:
            a[i_star] -= delta
            a[j] += delta
            conv = new_conv
            max_conv = new_max
            no_imp = 0
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
        else:
            no_imp += 1
            step_factor = max(min_step, step_factor * decay)

        # ------------------------------------------------------------------
        # periodic symmetrisation (helps to lower the peak)
        if it % sym_interval == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            conv_sym = np.convolve(sym, sym)
            max_sym = conv_sym.max()
            if max_sym < max_conv:
                a = sym
                conv = conv_sym
                max_conv = max_sym
                if max_sym < best_max:
                    best_max = max_sym
                    best_a = a.copy()
                # after a successful symmetrisation allow a slightly larger step
                step_factor = min(0.5, step_factor / decay)

        # ------------------------------------------------------------------
        # if stuck for a long time, perform a random redistribution to shake things
        if no_imp > max_no_imp:
            i2 = rng.choice(n, p=a)               # bias toward larger entries
            j2 = rng.integers(0, n - 1)
            if j2 != i2:
                delta2 = a[i2] * 0.4 * rng.random()
                a[i2] -= delta2
                a[j2] += delta2
                conv = np.convolve(a, a)
                max_conv = conv.max()
                no_imp = 0
                step_factor = 0.5

        # ------------------------------------------------------------------
        # early exit when the target bound is reached
        if 2.0 * n * max_conv < 1.50274349:
            best_a = a.copy()
            break

    # ----------------------------------------------------------------------
    # return the best vector found (as plain Python floats)
    return [float(x) for x in best_a]
