def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w
```

    v = np.zeros_like(cur)

    beta2 = 0.999
    eps = 1e-8
    lr = 0.3   # base learning rate

    it = 0
    no_improve = 0
    target = 1.50274349

    temp = None
    decay = 0.998

        it += 1

        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.07
        else:

        # compute weights: blend softmax and hard mask
        # softmax weight
        w_soft = np.exp((conv - max_val) / (temp + 1e-12))
        w_soft_sum = w_soft.sum()

        else:

            mask = (conv == max_val).astype(np.float64)

            w = 0.8 * w_soft + 0.2 * mask

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast AC1 upper‑bound score, works for any non‑negative vector."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s == 0:
        return float('inf')
    return float(2.0 * n * max_val / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initial vector -------------------------------------------------
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # ---- Adam parameters ------------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.30                     # base learning rate

    # ---- temperature for soft‑max weighting ----------------------------
    temp = None

    while True:

        it += 1

        else:
