import time
import numpy as np

def _normalize(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex by simple renormalisation."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return v
    return v / s

def _conv_full(v: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution using FFT (length 2*len(v)-1)."""
    vf = np.fft.rfft(v, n=L)
    c = np.fft.irfft(vf * vf, n=L)
    return c[: 2 * v.size - 1]

def _score_vec(v: np.ndarray, L: int) -> float:
    """Exact AC1 upper bound for a non‑negative vector."""
    n = v.size
    conv = _conv_full(v, L)
    maxc = conv.max()
    s = v.sum()
    return 2.0 * n * maxc / (s * s)

def _gradient(cur: np.ndarray, w_rev: np.ndarray, L: int, idx: np.ndarray) -> np.ndarray:
    """Gradient of the weighted convolution objective."""
    cur_fft = np.fft.rfft(cur, n=L)
    w_rev_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L)
    return 2.0 * grad_full[idx]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # small safety margin

    # ------------------------------------------------------------------
    # initialise candidate
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        cur = _normalize(a0)
    else:
        n0 = rng.integers(2000, 3000)
        cur = _normalize(rng.random(n0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())          # power‑of‑two FFT length
    idx = (2 * n - 2) - np.arange(n)             # slice to retrieve gradient

    # best ever solution
    best = cur.copy()
    best_sc = _score_vec(best, L)
    cur_sc = best_sc

    # Adam state
    m = np.zeros_like(cur)
    v_acc = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # temperature (soft‑max) schedule – start smooth, cool down
    temp = 200.0
    temp_min = 0.5
    temp_decay = 0.9995

    step0 = 0.8
    it = 0
    no_imp = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ---- convolution and soft‑max weighting -----------------------
        conv = _conv_full(cur, L)
        maxc = conv.max()
        w = np.exp((conv - maxc) / temp)          # soft‑max weights
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ---- gradient -----------------------------------------------
        w_rev = w[::-1]
        grad = _gradient(cur, w_rev, L, idx)

        # ---- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v_acc = beta2 * v_acc + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_acc / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)

        cand = cur - step * m_hat
        cand = _normalize(cand)
        cand_sc = _score_vec(cand, L)

                    no_imp = 0
                    if tmp_sc < best_sc:

        # ---- temperature cooling --------------------------------------
