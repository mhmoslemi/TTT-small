import scipy.sparse as sp

def toeplitz_conv_mat(a):
    n = a.shape[0]
    diags = []
    offsets = []
    for d in range(-(n-1), n):
        length = n - abs(d)
        if d >= 0:
            val = a[d]
        else:
            val = a[-d]
        diags.append(np.full(length, val))
        offsets.append(d)
    return sp.diags(diags, offsets, shape=(2*n-1, n), format='csr')

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        v = v.ravel()
    n = v.size
    u = np.sort(v)[::-1]

        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):

    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
