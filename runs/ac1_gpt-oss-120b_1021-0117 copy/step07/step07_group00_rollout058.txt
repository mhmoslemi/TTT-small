                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

        # ---- early exit ---------------------------------------------------------
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # final polishing: hard‑max gradient steps (few iterations)
    for _ in range(200):
        if time.time() >= deadline:
            break
        a_fft = np.fft.rfft(best, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        mask = (conv_full == max_val).astype(np.float64)
        cnt = mask.sum()
        if cnt == 0:
            continue
        w = mask / cnt
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # simple normalized step
        step_vec = 0.15 * grad / (np.linalg.norm(grad) + 1e-12)
        cand = _project_simplex(best - step_vec)
        cand_sc = _fast_score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand.copy(), cand_sc

    # ------------------------------------------------------------------
    # Return best vector as plain Python floats
    return [float(x) for x in best]
