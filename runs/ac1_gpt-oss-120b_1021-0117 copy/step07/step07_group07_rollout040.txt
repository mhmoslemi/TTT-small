import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum = 1, v >= 0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max(a, fft_a=None, L=None):
    """Return max value of the full self‑convolution of a (real, non‑negative)."""
    if fft_a is None:
        fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    conv = conv_full[: 2 * a.size - 1]
    return conv.max()

def _score_from_max(n, max_val, sum_a):
    """Exact score used by the evaluator."""
    return float(2.0 * n * max_val / (sum_a * sum_a))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.4)

    # -------------------- initialization --------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        a = _project_simplex(a0)
    else:
        n0 = rng.integers(2000, 3000)
        a = _project_simplex(rng.random(n0))

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())  # power‑of‑two length for FFT

    # initial score
    a_fft = np.fft.rfft(a, n=L)
    cur_max = _conv_max(a, fft_a=a_fft, L=L)
    cur_score = _score_from_max(n, cur_max, a.sum())
    best_vec = a.copy()
    best_score = cur_score

    # algorithmic parameters
    step = 0.6                     # initial step length
    step_decay = 0.9992            # exponential decay
    min_step = 1e-7
    temperature = 0.001            # for occasional uphill moves
    temp_decay = 0.9999
    stagnation = 0                 # counter for lack of improvement
    ITER_SYM = 800                 # symmetrisation period
    ITER_RESTART = 4000            # when to inject a random perturbation
    target = 1.50274349            # theoretical target

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution & max -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- sub‑gradient of the max -----
        # identify all positions attaining the maximum (within tolerance)
        eps = 1e-12
        mask = np.abs(conv - max_val) <= eps
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt == 0:
            w[:] = 1.0 / w.size
        else:
            w /= cnt

        # gradient = 2 * (a * w_rev) convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- gradient step + projection -----
        cand = a - step * grad
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_fft = np.fft.rfft(cand, n=L)
        cand_max = _conv_max(cand, fft_a=cand_fft, L=L)
        cand_score = _score_from_max(n, cand_max, cand.sum())

        # ----- acceptance -----
        improved = cand_score < cur_score
        if improved:
            a, cur_score, cur_max = cand, cand_score, cand_max
            stagnation = 0
            if cand_score < best_score:
                best_vec, best_score = cand.copy(), cand_score
        else:
            stagnation += 1
            # occasional Metropolis acceptance
            if rng.random() < np.exp(-(cand_score - cur_score) / temperature):
                a, cur_score, cur_max = cand, cand_score, cand_max

        # ----- schedule updates -----
        step = max(min_step, step * step_decay)
        temperature = max(1e-7, temperature * temp_decay)

        # ----- periodic symmetrisation (helps due to convolution symmetry) -----
        if it % ITER_SYM == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_max = _conv_max(sym, fft_a=sym_fft, L=L)
            sym_score = _score_from_max(n, sym_max, sym.sum())
            if sym_score < cur_score:
                a, cur_score, cur_max = sym, sym_score, sym_max
                stagnation = 0
                if sym_score < best_score:
                    best_vec, best_score = sym.copy(), sym_score

        # ----- random perturbation when stuck -----
        if stagnation > ITER_RESTART:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                pert = a.copy()
                pert[i] -= delta
                pert[j] += delta
                pert = _project_simplex(pert)
                pert_fft = np.fft.rfft(pert, n=L)
                pert_max = _conv_max(pert, fft_a=pert_fft, L=L)
                pert_score = _score_from_max(n, pert_max, pert.sum())
                if pert_score < cur_score:
                    a, cur_score, cur_max = pert, pert_score, pert_max
                    stagnation = 0
                    if pert_score < best_score:
                        best_vec, best_score = pert.copy(), pert_score

        # ----- early exit if target reached -----
        if best_score < target:
            break

    # Return the best sequence found (as plain Python floats)
    return [float(x) for x in best_vec]
