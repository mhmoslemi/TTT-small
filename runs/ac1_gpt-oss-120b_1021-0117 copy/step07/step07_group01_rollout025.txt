def generate_sidon_set(n, target_size, rng, max_trials=100000):
    sums = set()
    support = []
    attempts = 0
    while len(support) < target_size and attempts < max_trials:
        cand = rng.integers(0, n)
        if cand in support:
            attempts += 1
            continue
        ok = True
        for x in support:
            s = x + cand
            if s in sums:
                ok = False
                break
        if not ok:
            attempts += 1
            continue
        # also check self-sum
        if cand + cand in sums:
            attempts += 1
            continue
        # add
        for x in support:
            sums.add(x + cand)
        sums.add(cand + cand)
        support.append(int(cand))
        attempts = 0 # reset attempts after success
    return support
```

def _extend_sidon_set(support, n, rng, target):
    sums_used = np.zeros(2*n, dtype=bool)
    # mark existing sums
    for i, idx in enumerate(support):
        sums_used[idx+idx] = True
        for j in range(i):
            sums_used[idx + support[j]] = True

        # add
        for idx in support:
            sums_used[cand + idx] = True
        sums_used[cand+cand] = True
        support.append(int(cand))

import time
import math
import numpy as np

# -------------------- helpers ---------------------------------

def _greedy_sidon_set(n: int, rng: np.random.Generator, target: int) -> list:
    """
    Build a Sidon set (all unordered pair sums distinct) by greedy scan of a
    random permutation. Stops when `target` elements are collected or the
    permutation is exhausted.
    Returns the list of selected indices (may be shorter than target).
    """
    sums_used = np.zeros(2 * n, dtype=bool)          # sums 0 .. 2n-2
    support = []
    order = rng.permutation(n)
    for idx in order:
        # self‑sum collision?
        s_self = idx + idx
        if sums_used[s_self]:
            continue
        if support:
            # check sums with all already chosen indices
            existing = idx + np.array(support, dtype=int)
            if sums_used[existing].any():
                continue
        # accept this index
        support.append(int(idx))
        sums_used[s_self] = True
        if len(support) > 1:
            sums_used[idx + np.array(support[:-1], dtype=int)] = True
        if len(support) >= target:
            break
    return support


def _extend_sidon_set(support: list, n: int, rng: np.random.Generator,
                      target: int, max_trials: int = 2000) -> list:
    """
    Try to enlarge an existing Sidon set by random insertions.
    """
    if len(support) >= target:
        return support
    sums_used = np.zeros(2 * n, dtype=bool)
    # mark the sums already occupied by the current support
    for i, a in enumerate(support):
        sums_used[a + a] = True
        for b in support[:i]:
            sums_used[a + b] = True
    trials = 0
    while len(support) < target and trials < max_trials:
        cand = rng.integers(0, n)
        if cand in support:
            trials += 1
            continue
        s_self = cand + cand
        if sums_used[s_self]:
            trials += 1
            continue
        if support:
            existing = cand + np.array(support, dtype=int)
            if sums_used[existing].any():
                trials += 1
                continue
        # add the candidate
        for a in support:
            sums_used[cand + a] = True
        sums_used[s_self] = True
        support.append(int(cand))
        trials = 0          # reset counter after a success
    return support


def _find_sidon_set(n: int, target: int, rng: np.random.Generator,
                    max_outer: int = 30, max_extend: int = 2000) -> list:
    """
    Attempt several random constructions (greedy + random extension) to obtain
    a Sidon set of size at least `target`. Returns the best set found (maybe
    smaller than target).
    """
    best = []
    for _ in range(max_outer):
        cand = _greedy_sidon_set(n, rng, target)
        if len(cand) < target:
            cand = _extend_sidon_set(cand, n, rng, target, max_trials=max_extend)
        if len(cand) > len(best):
            best = cand
            if len(best) >= target:
                break
    return best

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Produce a non‑negative step‑function (list of floats) that tries to minimise
    the AC1 upper bound.  The strategy builds a large Sidon set of indices,
    assigns equal mass to them (so the L1 norm is 1) and returns the resulting
    sequence.  If time runs out before a sufficiently large Sidon set is found,
    the best candidate obtained so far is returned (fallbacking to the
    supplied `height_sequence_1` if present).
    """
    # timing
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # spare a bit for cleanup

    rng = np.random.default_rng(seed)

    # length of the required sequence

    else:

    best_score = float('inf')

    # theoretical upper bound on Sidon size: s(s+1)/2 <= 2n-1  ->  s <= 2*sqrt(n)
    max_possible = int(2 * math.isqrt(n))          # safe integer bound
    max_possible = min(max_possible, n)

    # we need at least the size that beats the target:
    # score = 4*n / s^2  < 1.50274349  ==>  s > sqrt(4*n / 1.5027)
    needed = int(math.sqrt(4 * n / 1.50274349)) + 1
    target = max(needed, 20)                       # never go below 20

    # try decreasing target sizes until we run out of time
    while target >= 20 and time.monotonic() < deadline:
        # attempt to build a Sidon set of this size

                                  max_outer=30, max_extend=2000)
        if len(support) < target:
            # could not reach this size – lower the target and continue
            target -= 1
            continue

        a[support] = 1.0 / len(support)

        # good enough – return immediately
