import time
import numpy as np

def _simplex_normalize(v):
    """Project onto the probability simplex by clipping and renormalising."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    # fallback – uniform distribution
    n = v.shape[0]
    return np.full_like(v, 1.0 / n)


def _score_simplex(a, L):
    """Score assuming sum(a)=1: 2·n·max(convolution)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # ------------------------------------------------------------------

    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = rng.integers(2000, 3000)
        init = rng.random(n0)
    cur = _simplex_normalize(np.maximum(init, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_sc = _score_simplex(best, L)
    cur_sc = best_sc

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 0.8

    # soft‑max temperature for max‑approximation
    temp = 0.5
    temp_factor = 1.004
    temp_max = 250.0

    target = 1.50274349
    it = 0
    no_improve = 0

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution & weighted gradient -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
