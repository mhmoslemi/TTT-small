def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that (hopefully) beats
    the previous AC1 upper bound.  The routine works with any parent sequence
    ``height_sequence_1`` that may already be present; otherwise it starts from a
    random simplex point and improves it by a projected (sub)gradient descent
    that directly reduces the maximal autocorrelation value.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # --------------------------------------------------------------
    # problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000               # a reasonable fallback

    # --------------------------------------------------------------
    # helper: project onto the probability simplex (non‑negative, sum == 1)
    def _project_simplex(v):
        v = np.asarray(v, dtype=np.float64)
        n_local = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n_local + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    # --------------------------------------------------------------
    # convolution helper (FFT, returns max value and its index)
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    def _max_conv(a_vec):
        fa = np.fft.rfft(a_vec, n=m)
        conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
        max_val = conv.max()
        max_idx = int(conv.argmax())
        return max_val, max_idx

    # --------------------------------------------------------------
    # initialise a (on the simplex)
    if "height_sequence_1" in globals():
