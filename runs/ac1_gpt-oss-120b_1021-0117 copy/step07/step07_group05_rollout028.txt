        else:
            a = parent / total   # normalize to sum 1
        n = a.size

        a = np.full(n, 1.0 / n, dtype=np.float64)

    #projection function
    def _project_simplex(v):
        v = np.asarray(v, dtype=np.float64)
        if v.ndim != 1:
            v = v.ravel()
        # ensure nonneg
        # Use algorithm: sort descending
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        # Sometimes due to numerical error sum may deviate, renormalize
        s = w.sum()
        if s > 0:
            w = w / s
        else:
            w = np.full_like(v, 1.0 / v.size)
        return w

    # evaluate current vector
    try:
        best_score = evaluate_sequence(a.tolist())
    except Exception:
        best_score = float('inf')

            off_j = j_range[~diag]

            cand = _project_simplex(cand)

            # compute product for all indices to find low
            # For efficiency just pick random low-mass index
            target = rng.integers(0, n)
            # ensure target not equal to i0 or j0
            if target == i0 or target == j0:
                target = (target + 1) % n
            # move fraction
            if i0 == j0:
                delta = a[i0] * 0.1
                a_mod = a.copy()
                a_mod[i0] -= delta
                a_mod[target] += delta
            else:
                delta = min(a[i0], a[j0]) * 0.1
                a_mod = a.copy()
                a_mod[i0] -= delta
                a_mod[j0] -= delta
                # split delta equally between two random targets
                t1 = target
                t2 = rng.integers(0, n)
                if t2 == i0 or t2 == j0:
                    t2 = (t2 + 1) % n
                a_mod[t1] += delta/2
                a_mod[t2] += delta/2
            cand = _project_simplex(a_mod)
        # evaluate candidate
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        if cand_score < best_score:

            best_vec = cand.copy()

            break
    # final fallback: ensure best_vec not empty
    if best_vec.size == 0:
        best_vec = a
    return [float(x) for x in best_vec]

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        v = v.ravel()
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:                               # fallback – uniform vector
        w = np.full_like(v, 1.0 / v.size)
    return w
