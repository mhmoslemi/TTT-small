def softmax_weights(conv, alpha):
    maxc = conv.max()
    scaled = alpha * (conv - maxc)  # negative numbers
    exp_scaled = np.exp(scaled)
    sum_exp = exp_scaled.sum()
    if sum_exp == 0:
        w = np.zeros_like(conv)
    else:
        w = exp_scaled / sum_exp
    return w

def _softmax_weights(conv, alpha):
    maxc = conv.max()
    scaled = alpha * (conv - maxc)
    # to avoid overflow: np.exp may underflow for large negative; that's fine.
    exp = np.exp(scaled)
    s = exp.sum()
    if s == 0:
        # fallback uniform
        return np.full_like(conv, 1.0/conv.size)
    return exp / s

        v = np.zeros_like(a)

            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()
            cur_score = 2.0 * n * max_val

            # Update global best
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Early exit if target reached
            if best_score < 1.50274349:
                break

            # adaptive alpha
            # linear schedule based on iteration count upto 2000
            if it < 2000:
                alpha = alpha_start + (alpha_end - alpha_start) * (it / 2000.0)
            else:
                alpha = alpha_end

            # softmax weights
            # to improve numerical stability subtract max
            maxc = max_val
            scaled = alpha * (conv - maxc)
            # exponentiate
            exp_scaled = np.exp(scaled)
            sum_exp = exp_scaled.sum()
            if sum_exp == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = exp_scaled / sum_exp

            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # Adam moments
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # learning rate schedule
            lr = base_lr / (math.sqrt(it) + 1.0)
            update = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = _project_simplex(a - update)

            # occasional symmetrisation to enforce evenness
            if it % 800 == 0:
                sym = (cand + cand[::-1]) * 0.5
                cand = _project_simplex(sym)

            # random perturb if stagnated
            if no_improve > 3000:

                rnd = _project_simplex(rng.random(n))
                cand = _project_simplex(0.5 * best + 0.5 * rnd)
                # reset moments
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0

            a = cand
        # break if target reached
        if best_score < 1.50274349:
            break

    if best is None:
        best = np.full(n, 1.0/n)

    # Convert to list of Python floats
    return [float(x) for x in best]

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of heights) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ----------------------------------------------------------------------
    # Initial vectors -------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: a modest random vector
        base = rng.random(2000, dtype=np.float64)

    base = np.maximum(base, 0.0)
    n = base.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two for FFTs

    inits = [
        _project_simplex(base),                 # given parent (if any)
        np.full(n, 1.0 / n),                    # uniform
        _project_simplex(np.exp(-0.001 * np.arange(n))),  # exponential decay
        _project_simplex(rng.random(n)),        # random simplex
    ]

    # ----------------------------------------------------------------------
    # Optimisation parameters -----------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    base_lr = 0.5               # initial learning rate
    target_score = 1.50274349   # stop when reached
    best = None
    best_score = float("inf")
    no_improve_global = 0

    # ----------------------------------------------------------------------
    # Main optimisation loop ------------------------------------------------
    for a0 in inits:
        if time.monotonic() >= deadline:
            break
        a = a0.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        it = 0
        no_improve = 0

        while True:
            if time.monotonic() >= deadline:
                break
            it += 1

            # ----- convolution and current score --------------------------------
            a_fft = np.fft.rfft(a, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()
            cur_score = 2.0 * n * max_val

            # ----- keep the best ever -------------------------------------------
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                no_improve = 0
                no_improve_global = 0
            else:
                no_improve += 1
                no_improve_global += 1

            if best_score < target_score:
                break

            # ----- soft‑max weights (smoothed max) -------------------------------
            # alpha grows from a small to a larger value as we iterate
            alpha = 20.0 + (200.0 - 20.0) * min(1.0, it / 2000.0)
            # stable log‑sum‑exp trick
            maxc = max_val
            scaled = alpha * (conv - maxc)
            exp_scaled = np.exp(scaled)
            sum_exp = exp_scaled.sum()
            if sum_exp == 0.0:                     # numerical safeguard
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = exp_scaled / sum_exp

            # ----- gradient of the soft‑max (via FFT) ---------------------------
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # ----- Adam update --------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = base_lr / (math.sqrt(it) + 1.0)
            update = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = _project_simplex(a - update)

            # ----- occasional symmetrisation ------------------------------------
            if it % 800 == 0:
                sym = (cand + cand[::-1]) * 0.5
                cand = _project_simplex(sym)

            # ----- random perturbation when stuck -------------------------------
            if no_improve > 3000:
                rnd = _project_simplex(rng.random(n))
                cand = _project_simplex(0.5 * best + 0.5 * rnd)
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0

            a = cand

        if best_score < target_score:
            break

    # ----------------------------------------------------------------------
    # Fallback (should never be needed) --------------------------------------
    if best is None:
        best = np.full(n, 1.0 / n)

    # Return as plain Python list of floats
    return [float(x) for x in best]
