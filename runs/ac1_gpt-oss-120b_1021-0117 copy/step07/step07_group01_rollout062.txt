import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score = 2 * n * max(conv(a,a))  (a is assumed to sum to 1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # Initial vector
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2000, dtype=np.float64)
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())

    # Global best
    best = a0.copy()
    best_sc = _score(best)

    # Adam state (will be re‑initialised on each restart)
    def _new_adam_state():
        return np.zeros_like(best), np.zeros_like(best)

    # ------------------------------------------------------------------
    # Main search loop – multiple short restarts
    RESTART_STEPS = 3000          # iterations per restart block
    step0 = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    max_no_improve = 3000

    it_total = 0
    while True:
        # time check
        if time.monotonic() >= deadline:
            break

        # initialise a fresh start (or a perturbed version of the best)
        if it_total % RESTART_STEPS == 0:
            # perturb the current global best to keep diversity
            rand_vec = rng.random(n, dtype=np.float64)
            rand_vec = _project_simplex(rand_vec)
            a = 0.5 * best + 0.5 * rand_vec
            a = _project_simplex(a)
            m, v = _new_adam_state()
            it = 0
            no_improve = 0
        # ------------------------------------------------------------------
        # one optimisation step
        it += 1
        it_total += 1

        # ----- convolution and max -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- sub‑gradient (average over active constraints) -----
        eps_mask = max_val * 1e-12 + 1e-15

        w = mask.astype(np.float64)

        w_rev = w[::-1]

            break

    # ------------------------------------------------------------------
    return [float(x) for x in best]
