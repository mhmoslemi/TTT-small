import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    """Score assuming sum(a)==1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # reasonable fallback length
        init = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(np.maximum(init, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # optimizer state (Adam)
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.5

    # soft‑max temperature for the max‑convolution
    temp_factor = 0.02          # relative to current max value
    temp_decay = 0.9997
    min_temp_factor = 1e-6

    best_vec = cur.copy()
    best_sc = _score_vec(best_vec)
    cur_sc = best_sc

    it = 0

    target = 1.50274349

        # soft‑max weighting approximating the max

        if w_sum > 0:

        idx = (2 * n - 2) - np.arange(n)

    return [float(x) for x in best_vec]
