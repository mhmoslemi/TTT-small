def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score for a simplex‑normalised vector (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # initialise a simplex‑normalised vector
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.6

    # smoothing for the max (soft‑max temperature)
    beta = 10.0          # start fairly smooth
    beta_max = 200.0

    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ------------------------------------------------------------------
        # convolution of current a
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        # soft‑max weights over the convolution
        w = np.exp(beta * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            # fallback to uniform over max entries
            mask = conv >= max_val - (max_val * 1e-12 + 1e-15)
            w = mask.astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum

        # gradient of the (soft) max‑convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        update = step * m_hat / (np.sqrt(v_hat) + eps)

        beta = min(beta_max, beta * 1.001)

        # early exit if target reached
        if best_sc < target:
            break

    # ----------------------------------------------------------------------
    return [float(x) for x in best]
