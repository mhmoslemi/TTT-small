def subgradient_max(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)

    right = np.zeros_like(a)

    right[idx_max:] = a[:n - idx_max]
    grad = left + right
    return grad, conv.max()

import time
import numpy as np

# ---------- helpers ---------------------------------------------------------

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_grad(a, L):
    """
    Return (max_convolution, gradient_of_max) for a*a.
    a : 1‑d ndarray, non‑negative, sum≈1
    L : length of zero‑padded FFT (power of two ≥ 2*len(a)-1)
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    k = int(np.argmax(conv))          # index of the maximal entry
    max_val = float(conv[k])

    # gradient: g_i = a_{k-i} (if in range) + a_{i-k} (if in range)
    grad = np.zeros_like(a)

    # first term: a_{k-i}
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = a[k - idx]

    # second term: a_{i-k}
    j_start = max(k, 0)
    j_end = min(k + n - 1, n - 1)
    if j_start <= j_end:
        idx = np.arange(j_start, j_end + 1)
        grad[idx] += a[idx - k]

    return max_val, grad
