import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise to eliminate tiny round‑off loss
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a, fft_len):
    """Fast evaluation of 2·n·max(a∗a) for a already on the simplex."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # initial vector
    if "height_sequence_1" in globals():

    n = cur.size
    fft_len = 1 << ((2 * n - 1).bit_length())        # power‑of‑two length for FFT

    # ------------------------------------------------------------------
    # optimiser state
    momentum = np.zeros_like(cur)

    it = 0

        # ----- linear convolution (auto‑correlation) -----
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weighting for a smooth gradient of the max -----
        alpha = min(alpha_max, alpha0 * math.log1p(it))
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:                     # overflow guard
