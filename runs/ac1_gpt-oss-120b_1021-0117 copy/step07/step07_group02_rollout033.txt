import time, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a list of non‑negative step‑function heights.
    The implementation keeps a uniform distribution (theoretical optimum)
    and optionally performs a brief random search, keeping the best vector.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin before the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # length of the problem – use the supplied parent if available
    if 'height_sequence_1' in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2500  # reasonable fallback length

    # The uniform distribution (sum = 1) is the provable minimiser of the target.
    best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    # initial score (for bookkeeping; evaluate_sequence is pre‑imported)
    best_score = evaluate_sequence(best_vec.tolist())

    # ---- light‑weight random exploration (optional) --------------------
    # Only run while time permits; we keep the best vector seen.
    while time.monotonic() < deadline:
        # draw a random point in the simplex via exponential‑(Gamma) sampling
        rnd = rng.random(n, dtype=np.float64)
        cand = _project_simplex(rnd)
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
        # check time every few hundred iterations automatically via the loop condition

    # Return a plain Python list of floats (required by the evaluator)
    return [float(x) for x in best_vec]
