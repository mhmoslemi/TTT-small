def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3) # margin
    # Determine n
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)
        n = seq0.size
    else:
        n = 2000
    L = 1 << ((2 * n - 1).bit_length())
    # projection onto simplex
    def project_simplex(v):
        # Implementation: same as before
        v = np.asarray(v, dtype=np.float64)
        n_v = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n_v+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        # due to float errors correct sum
        excess = w.sum() - 1.0
        if abs(excess) > 1e-12:
            w -= excess / n_v
            w = np.maximum(w, 0.0)
        return w
    # Fast score using FFT (expects sum=1)
    def score_fft(a):
        # a is numpy array length n, sum = 1
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        max_val = conv.max()
        return 2.0 * n * max_val   # since sum=1
    # Build candidate set
    xs = np.arange(n, dtype=np.float64)
    candidates = []
    # uniform
    candidates.append(np.full(n, 1.0 / n))
    # triangular
    tri = np.minimum(xs + 1, n - xs)
    candidates.append(tri / tri.sum())
    # quadratic (parabolic) shape
    quad = (xs + 1) * (n - xs)
    candidates.append(quad / quad.sum())
    # sine bump
    sine = np.sin(np.pi * (xs + 1) / (n + 1))
    candidates.append(sine / sine.sum())
    # raised cosine bump (as before)
    bump = (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2
    candidates.append(bump / bump.sum())
    # exponential decays
    for alpha in (0.0005, 0.001, 0.002, 0.004, 0.008):
        exp = np.exp(-alpha * xs)
        candidates.append(exp / exp.sum())
    # power law
    for p in (0.2, 0.5, 1.0, 1.5, 2.0):
        pw = (xs + 1) ** (-p)
        candidates.append(pw / pw.sum())
    # sine^p
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        sinp = np.sin(np.pi * (xs + 1) / (n + 1)) ** p
        # ensure not all zero
        if sinp.max() > 0:
            candidates.append(sinp / sinp.sum())
    # random Dirichlet
    rand = rng.random(n)
    candidates.append(rand / rand.sum())
    # parent if present
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.sum() > 0:
            parent = parent / parent.sum()
            candidates.append(parent)
    # evaluate best among these
    best_vec = None
    best_score = float('inf')
    for vec in candidates:
        sc = score_fft(vec)
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_score = best_score
    best_global = best_vec.copy()
    best_global_score = best_score
    # gradient descent parameters
    momentum = np.zeros_like(cur)
    beta = 0.95
    step0 = 1.0
    it = 0
    no_improve = 0
    target = 1.50274349
    # Extra: use coordinate random moves schedule
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1
        # compute convolution and gradient via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        # mask for active entries
        eps = max_val * 1e-12 + 1e-15
        mask = (conv >= max_val - eps).astype(np.float64)
        cnt = mask.sum()
        if cnt > 0:
            mask /= cnt
        # gradient
        w_rev = mask[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # map back to indices 0..n-1
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        # step size schedule
        step = step0 / math.sqrt(it)
        # line search
        best_local_vec = cur
        best_local_score = cur_score
        for factor in (0.4, 0.7, 1.0, 1.3, 1.6):
            cand = project_simplex(cur - step * factor * momentum)
            sc = score_fft(cand)
            if sc < best_local_score:

                best_local_vec = cand
        # Acceptance

            cur, cur_score = best_local_vec, best_local_score
            no_improve = 0

        # Update global best
        if cur_score < best_global_score:
            best_global_score, best_global = cur_score, cur.copy()

        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = project_simplex(sym)
            sc_sym = score_fft(sym)
            if sc_sym < cur_score:
                cur, cur_score = sym, sc_sym
                no_improve = 0
                if sc_sym < best_global_score:
                    best_global_score, best_global = sc_sym, sym.copy()
        # Random coordinate moves when stuck
        if no_improve > 3000:
            # attempt several random swaps
            for _ in range(5):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = project_simplex(tmp)
                sc_tmp = score_fft(tmp)
                if sc_tmp < cur_score:
                    cur, cur_score = tmp, sc_tmp
                    no_improve = 0
                    if sc_tmp < best_global_score:
                        best_global_score, best_global = sc_tmp, tmp.copy()
                    break
            # if still not improve after attempts, reset momentum
            if no_improve > 3000:
                momentum = np.zeros_like(cur)
        # Early exit
        if best_global_score < target:
            break
    # Return list
    return [float(x) for x in best_global]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Score when sum(a)=1, using FFT for convolution."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max()

def _max_and_grad(a, n, L):
    """Return max conv value and sub‑gradient of max(conv(a,a))."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad
