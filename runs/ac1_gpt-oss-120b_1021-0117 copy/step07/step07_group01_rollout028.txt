def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)

    # init a
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2500
        a = rng.random(n)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    step0 = 1.0
    it = 0
    target = 1.50274349

    # Temperature schedule for softmax
    temp_factor = 0.005  # initial factor relative to max

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # compute FFT of a
        a_fft = np.fft.rfft(a, n=L)
        # compute convolution
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # compute temperature (tau)
        tau = max_conv * temp_factor / np.sqrt(it + 1)
        if tau < 1e-12:
            tau = 1e-12

        # softmax weights
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.zeros_like(w)
        else:
            w = w / w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (math.sqrt(it) + 1.0)

        update = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(a - update)

        cand_sc = _score(cand)

        # acceptance
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            a = cand
        elif cand_sc < _score(a):
            a = cand

        # occasional symmetrisation
        if it % 300 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
                a = sym

        # random perturbation when stuck: we can track no_improve counter
        # implement simple moving condition
        # Keep track of last improvement steps
        # For simplicity, we can add random perturbations every 2000 iterations
        if it % 2000 == 0:
            # perturb a little
            perturb = rng.normal(scale=0.01, size=n)
            pert_a = np.maximum(a + perturb, 0.0)
            pert_a = _project_simplex(pert_a)
            a = pert_a

        # early exit if target reached
        if best_sc < target:
            break

    return [float(x) for x in best]

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score for a vector already normalized to sum=1 (2*n*max(convolution))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random start, length about 2500
        a = rng.random(2500, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    # Adam optimizer state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 1.0
    it = 0
    no_improve = 0
    target = 1.50274349

    # parameters for soft‑max smoothing of the max‑operator
    base_temp = 0.5   # fraction of current max used as temperature

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution of current a -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()
        cur_sc = float(2.0 * n * max_conv)

        # ----- soft‑max weighting -----
        tau = max_conv * base_temp / math.sqrt(it + 1.0)
        if tau < 1e-12:
            tau = 1e-12
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(w)
        else:
            w /= w_sum

        # ----- gradient of smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (math.sqrt(it) + 1.0)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(a - update)
        cand_sc = _score(cand)

        # ----- acceptance -----
        if cand_sc < best_sc:
            best, best_sc = cand.copy(), cand_sc
            a, cur_sc = cand, cand_sc
            no_improve = 0
        elif cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional uphill move
            if rng.random() < math.exp(-(cand_sc - best_sc) / (step + 1e-12)):
                a = cand

        # ----- periodic symmetrisation -----
        if it % 300 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best, best_sc = sym.copy(), sym_sc
                a, cur_sc = sym, sym_sc
                no_improve = 0

        # ----- random perturbation when stuck -----
        if no_improve > 4000:
            rand_vec = _project_simplex(rng.random(n))
            a = 0.5 * best + 0.5 * rand_vec
            a = _project_simplex(a)
            cur_sc = _score(a)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # final short local search around the best candidate
    cur = best.copy()
    cur_sc = best_sc
    for _ in range(500):
        if time.monotonic() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a small random amount from i to j
        delta = cur[i] * rng.random() * 0.1
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        sc = _score(tmp)
        if sc < cur_sc:
            cur, cur_sc = tmp, sc
            if sc < best_sc:
                best, best_sc = tmp.copy(), sc

    # ------------------------------------------------------------------
    return [float(x) for x in best]
