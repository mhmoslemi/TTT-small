import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # correct possible rounding error
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast computation of the evaluator score (assumes a>=0)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------
    # Determine problem size
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2000
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT length

    # ------------------------------------------------------------
    # Build a small suite of deterministic initial candidates
    xs = np.arange(n, dtype=np.float64)
    candidates = []

    candidates.append(np.full(n, 1.0 / n))                         # uniform
    tri = np.minimum(xs + 1, n - xs)                             # triangular
    candidates.append(tri / tri.sum())
    sine = np.sin(np.pi * (xs + 1) / (n + 1))                     # sine bump
    candidates.append(sine / sine.sum())
    bump = (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2.0          # raised cosine
    candidates.append(bump / bump.sum())

    for alpha in (0.001, 0.003, 0.006):                           # exponentials
        exp = np.exp(-alpha * xs)
        candidates.append(exp / exp.sum())
    for p in (0.5, 1.0, 1.5):                                    # power laws
        pw = (xs + 1) ** (-p)
        candidates.append(pw / pw.sum())

    rand = rng.random(n)                                          # random simplex
    candidates.append(rand / rand.sum())

    if "height_sequence_1" in globals():                         # parent if present
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.sum() > 0:
            parent = parent / parent.sum()
            candidates.append(parent)

    # ------------------------------------------------------------
    # Pick the best deterministic start
    best_vec = None
    best_sc = float('inf')
    for vec in candidates:
        sc = _score_fft(vec, n, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # ------------------------------------------------------------

    it = 0
    no_improve = 0
    best_global = cur.copy()
    best_global_sc = cur_sc

    step0 = 0.5          # base step size

    temp_initial = 0.5   # relative to current max

            break

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]

        T = max_val * max(temp_min_factor,
                          temp_initial * math.exp(-it / 250.0))

        # ----- soft‑max weights -----
        w = np.exp((conv - max_val) / T)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient (smooth) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- step size (inverse sqrt decay) -----
        step = step0 / math.sqrt(it)

        # ----- line search over a few scales -----
        local_best = cur
        local_best_sc = cur_sc
        for factor in (0.5, 1.0, 1.5):

                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)

                    no_improve = 0
                    if tmp_sc < best_global_sc:
                        best_global_sc = tmp_sc
                        best_global = tmp.copy()

        if best_global_sc < 1.50274349:
            break

    # ------------------------------------------------------------
