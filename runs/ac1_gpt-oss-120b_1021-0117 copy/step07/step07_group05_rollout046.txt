Implementation:

```
def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.nonzero(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a, n):
    """Fast replica of evaluate_sequence (a already non‑negative)."""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    rng = np.random.default_rng(seed)

    # ---- initial candidates -------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # start from parent if available
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.sum() > 0:
            parent = parent / parent.sum()
        else:
            parent = uniform.copy()
    else:
        parent = uniform.copy()

    # evaluate both
    best_vec = uniform
    best_score = _score(best_vec, n)
    try:
        pscore = _score(parent, n)
        if pscore < best_score:
            best_score = pscore
            best_vec = parent.copy()
    except Exception:
        pass

    cur = best_vec.copy()
    cur_score = best_score

    # ---- search parameters -------------------------------------------------
    max_iters = 10 ** 7          # huge upper bound, real stop is time
    stale_limit = 3000           # iterations without improvement before restart
    last_improve = 0
    T0 = 0.5
    Tmin = 1e-6

    # main loop -------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        # temperature (linear decay)
        prog = it / 200000.0
        T = max(Tmin, T0 * (1.0 - prog))

        # propose a simple mass transfer
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # fraction of source mass to move (adaptive)
        max_frac = max(0.01, 0.5 * (1.0 - prog))
        delta = cur[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)

        cand_score = _score(cand, n)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
                last_improve = it

        # occasional restart if stuck
        if it - last_improve > stale_limit:
            # perturb best solution modestly
            perturb = rng.normal(scale=0.02, size=n)
            cand = best_vec + perturb
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            try:
                cand_score = _score(cand, n)
                cur = cand
                cur_score = cand_score
                last_improve = it
            except Exception:
                pass

    # final sanity clipping (evaluate_sequence would also clip)
    final = np.clip(best_vec, 0.0, 1000.0)
    # ensure non‑zero sum
    if final.sum() < 0.01:
        final = uniform.copy()

    return [float(x) for x in final]
