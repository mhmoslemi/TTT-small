    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)  # keep margin

    v = np.zeros_like(a)

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:

        # Use tolerance relative to max_val
        tol = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - tol
        w = mask.astype(np.float64)
        cnt = w.sum()

        w_rev = w[::-1]

        idx = (2 * n - 2) - np.arange(n)

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score for a vector already normalized to sum=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
