import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # correct possible rounding error
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a: np.ndarray, n: int, L: int) -> float:
    """Fast version of the evaluator (assumes a >= 0, sum(a) = 1)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

        grad = 2.0 * grad_full[idx]

        # ---- momentum update ----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ---- step size (inverse‑sqrt decay) ----
        step = step0 / math.sqrt(it)

        # ---- try a few scale factors and keep the best ----
        local_best_vec = cur
        local_best_sc = cur_sc
        for factor in (1.0, 0.5, 0.2):
            cand = _project_simplex(cur - step * factor * momentum)
            cand_sc = _score_fft(cand, n, L)
            if cand_sc < local_best_sc:
                local_best_sc = cand_sc
                local_best_vec = cand

        # ---- acceptance ----
        if local_best_sc < cur_sc:
            cur, cur_sc = local_best_vec, local_best_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional Metropolis‑like acceptance
            prob = math.exp(-(local_best_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = local_best_vec, local_best_sc

        # ---- update global best ----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            no_improve = 0

        # ---- periodic symmetrisation (helps escape shallow traps) ----
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, n, L)
            if sym_sc < best_sc:
                best_sc, best_vec = sym_sc, sym.copy()
                cur, cur_sc = sym, sym_sc
                no_improve = 0

        # ---- random jitter when stuck for a while ----
        if no_improve > 4000:
            jitter = rng.normal(scale=0.02, size=n)
            cand = _project_simplex(cur + jitter)
            cand_sc = _score_fft(cand, n, L)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
            else:
                momentum = np.zeros_like(cur)   # reset momentum

        # ---- early exit if target reached ----
        if best_sc < target:
            break

    # --------------------------------------------------------------
    return [float(x) for x in best_vec]
