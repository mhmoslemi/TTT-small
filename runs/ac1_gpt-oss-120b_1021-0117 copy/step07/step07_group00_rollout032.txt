import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w
```

        v = beta2 * v + (1 - beta2) * (grad ** 2)

        v_hat = v / (1 - beta2 ** it)
        step_size = 0.5 / (np.sqrt(it) + 1)  # base step decaying
        step = step_size * m_hat / (np.sqrt(v_hat) + eps)

        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand = cand / cand_sum  # renormalize to sum 1 (simple projection)
        else:
            cand = _project_simplex(cand)

        if cand_score < best_score:
            best_vec = cand.copy()
            best_score = cand_score
        
        # accept if improvement; else maybe accept with prob.
        if cand_score < score_vec(cur):
            cur = cand
        else:
            # simulated annealing based on step magnitude
            delta = cand_score - score_vec(cur)
            prob = math.exp(-delta / (step_size + 1e-12))
            if rng.random() < prob:
                cur = cand

        if it % 200 == 0:

            i = int(np.argmax(grad))

            if cur[i] > 1e-12:

        # early exit if score already good
        if best_score < 1.45:  # safety for target
            break
    
    # Return as list of floats
    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # leave a few hundred ms margin for return handling
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------ #
    # initialization (use the provided parent if present)
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())          # next power‑of‑two FFT size

    # scoring helper (vector assumed to sum ~1)
    def _score(v):
        a_fft = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    best_vec = cur.copy()
    best_score = _score(best_vec)
    cur_score = best_score

    # ------------------------------------------------------------------ #
    # optimizer state (Adam‑style)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # soft‑max temperature schedule
    temp = None
    temp_factor = 0.07      # initial fraction of max(conv)
    temp_decay = 0.997

    it = 0
