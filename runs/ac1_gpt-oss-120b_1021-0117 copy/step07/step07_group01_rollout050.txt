import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a: np.ndarray) -> float:
    """Score for a vector already normalized to sum=1: 2*n*max(conv)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a list of non‑negative heights that (hopefully) improve the AC1 bound.
    The routine respects the time budget, uses only linear memory and FFT‑based
    convolutions, and falls back to the best solution found so far.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2500, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)           # enforce sum=1, non‑negative

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.8

    it = 0
    no_improve = 0
    target = 1.50274349

    # ----- main optimisation loop ----------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # FFT of current iterate
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        conv_max = conv.max()

        # Soft‑max weighting of the convolution entries
        alpha = 5.0 + min(45.0, it * 0.01)          # gradually tighten
        # stable soft‑max
        w_exp = np.exp(alpha * (conv - conv_max))
        w_sum = w_exp.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_exp / w_sum

        # Gradient of the smoothed max‑convolution
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (math.sqrt(it) + 1.0)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- two candidate proposals -----
        cand_add = _project_simplex(a - update)

        cand_mul = a * np.exp(-step * grad)
        s = cand_mul.sum()
        if s > 0:
            cand_mul = cand_mul / s
        else:
            cand_mul = a.copy()

        sc_add = _score(cand_add)
        sc_mul = _score(cand_mul)
        if sc_add < sc_mul:
            cand, cand_sc = cand_add, sc_add
        else:
            cand, cand_sc = cand_mul, sc_mul

        # ----- acceptance logic -----
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            a = cand
            no_improve = 0
        else:
            cur_sc = _score(a)
            if cand_sc < cur_sc:
                a = cand
                no_improve = 0
            else:
                no_improve += 1
                prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
                if rng.random() < prob:
                    a = cand

        # ----- occasional symmetrisation -----
        if it % 300 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sc_sym = _score(sym)
            if sc_sym < best_sc:
                best = sym.copy()
                best_sc = sc_sym
                a = sym
                no_improve = 0

        # ----- random perturbation when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                a = _project_simplex(tmp)
            no_improve = 0

        # ----- occasional full restart -----
        if no_improve > 5000:
            rand_vec = rng.random(n, dtype=np.float64)
            rand_vec = _project_simplex(rand_vec)
            a = 0.5 * best + 0.5 * rand_vec
            a = _project_simplex(a)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ----- final simple local search (tiny moves) -------------------------
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.02
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best = cand
            best_sc = cand_sc

    return [float(x) for x in best]
