def _grad_sub(a, L, temp_factor):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_val = conv.max()
    temp = max(max_val * temp_factor, 1e-12)
    w = np.exp((conv - max_val) / temp)
    w_sum = w.sum()
    if w_sum>0:
        w /= w_sum
    w_rev = w[::-1]
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    idx = (2*n-2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val
```

import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _objective(a, L):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*a.shape[0]-1]
    return float(2.0 * a.shape[0] * conv.max())

def _gradient(a, L, temp_factor):
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_val = conv.max()
    temp = max(max_val * temp_factor, 1e-12)
    w = np.exp((conv - max_val) / temp)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    w_rev = w[::-1]
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    idx = (2*n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)

    # initialization
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random length 2500
        a0 = rng.random(2500, dtype=np.float64)
    a = _project_simplex(np.maximum(a0, 0.0))
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    # also evaluate a few simple candidates and keep the best
    best_a = a.copy()
    best_sc = _objective(best_a, L)

    # Uniform
    uniform = np.full(n, 1.0 / n)
    sc = _objective(uniform, L)
    if sc < best_sc:
        best_sc, best_a = sc, uniform.copy()

    # Cosine bell (raised cosine)
    x = np.arange(n)
    cosbell = (1 - np.cos(2 * np.pi * x / (n - 1))) / 2
    cosbell = _project_simplex(cosbell)
    sc = _objective(cosbell, L)
    if sc < best_sc:
        best_sc, best_a = sc, cosbell.copy()

    # Sine based window
    sinewin = np.sin(np.pi * (x + 0.5) / n)
    sinewin = _project_simplex(sinewin)
    sc = _objective(sinewin, L)
    if sc < best_sc:
        best_sc, best_a = sc, sinewin.copy()

    # some random starts
    for _ in range(3):
        rnd = _project_simplex(rng.random(n))
        sc = _objective(rnd, L)
        if sc < best_sc:
            best_sc, best_a = sc, rnd.copy()

    # set current to best
    cur = best_a.copy()
    cur_sc = best_sc

    # optimizer state
    momentum = np.zeros_like(cur)
    beta = 0.95
    step0 = 2.0
    temp_factor = 1e-2
    temp_decay = 0.9995
    min_temp_factor = 1e-6
    no_improve = 0
    target = 1.50274349
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # compute gradient
        grad, _ = _gradient(cur, L, temp_factor)

        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size
        step = step0 / math.sqrt(it)

        # candidate update
        cand = cur - step * momentum
        cand = _project_simplex(cand)

        # enforce symmetry
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _objective(cand, L)

        if cand_sc < best_sc:
            best_sc = cand_sc
            best_a = cand.copy()

        # accept if improvement, else occasional Metropolis
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # periodic symmetrisation
        if it % 200 == 0:
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)
            cur_sc = _objective(cur, L)

        # random local move when stuck
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_sc = _objective(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_a = tmp_sc, tmp.copy()
            else:
                # full random restart
                rnd = _project_simplex(rng.random(n))
                rnd = (rnd + rnd[::-1]) * 0.5
                rnd = _project_simplex(rnd)
                rnd_sc = _objective(rnd, L)
                if rnd_sc < cur_sc:
                    cur, cur_sc = rnd, rnd_sc
                    momentum.fill(0.0)
                    no_improve = 0

        # decay temperature
        temp_factor = max(temp_factor * temp_decay, min_temp_factor)

        # early stop
        if best_sc < target:
            break

        # occasional time check
        if it % 1000 == 0 and time.monotonic() >= deadline:
            break

    # final short hill-climbing using remaining time
    remaining = deadline - time.monotonic()
    if remaining > 0.1:
        hill_deadline = time.monotonic() + remaining
        while time.monotonic() < hill_deadline:
            i = rng.integers(0, n)
            if best_a[i] <= 1e-12:
                continue
            delta = best_a[i] * rng.random() * 0.02
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)
            cand_sc = _objective(cand, L)
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_a = cand

    # return as Python list
    return [float(x) for x in best_a]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _objective(a, L):
    """Objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
