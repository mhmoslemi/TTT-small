import time, numpy as np, math

def _project_simplex(v):
    # Projects onto simplex sum(v)=1, v>=0
    # Works with float array
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _evaluate(a):
    # fast via FFT
    n = a.shape[0]
    # use next power of two >= 2n-1
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2 * n * max_b / (sum_a * sum_a)

import time
import numpy as np

def _project_simplex(v):
    """Project onto the probability simplex (sum = 1, non‑negative)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max(a, fft_len):
    """Maximum value of the full convolution a*a using FFT."""
    # a is 1‑D float array, sum(a) > 0
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*a.shape[0] - 1]
    return conv.max()

def _objective(a, fft_len):
    """Compute the AC1 upper bound for a (sum is not assumed to be 1)."""

        fa = np.fft.rfft(a, fft_len)

                    best_a = tmp.copy()

def _project_simplex(v):
    """Project a vector onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
