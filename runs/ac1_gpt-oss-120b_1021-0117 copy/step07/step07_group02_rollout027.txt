def optimize(a0, deadline):
    a = project_simplex(a0)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1  # base step size
    best_vec = a.copy()
    best_score = score(a)
    t = 0
    while time.monotonic() < deadline:
        t += 1
        max_val, max_idx = compute max conv and idx
        # compute gradient
        g = grad_for_max(a, max_idx)
        # Adam update before projection
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_simplex(a_new)
        cur_score = score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a_new.copy()
        # check for convergence
        if cur_score < target: break
        a = a_new
    return best_vec, best_score

import time, numpy as np

def _project_simplex(v):
    # Euclidean projection onto probability simplex (sum=1, v>=0)
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max_and_idx(a, L):
    # returns max value, index, and fft of a for reuse
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * a.shape[0] - 1]
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    return max_val, max_idx, a_fft

def _subgradient_max(a, max_idx):
    n = a.shape[0]
    grad = np.zeros_like(a)
    j_start = max(0, max_idx - (n - 1))
    j_end = min(n - 1, max_idx)
    if j_start <= j_end:
        js = np.arange(j_start, j_end + 1)
        grad[js] = 2.0 * a[max_idx - js]
    return grad

    else:
        # fallback: Dirichlet with moderate concentration
        n_fallback = 2500
        init = rng.dirichlet(np.full(n_fallback, 0.5))
    n = init.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    # project onto simplex to ensure feasibility
    cur = _project_simplex(np.maximum(init, 0.0))
    # score function (sum=1)
    def _score(a):
        max_val, _, _ = _conv_max_and_idx(a, L)
        return 2.0 * n * max_val

    cur_score = _score(cur)
    best_vec = cur.copy()
    best_score = cur_score

    # Adam state

    beta2 = 0.999
    eps = 1e-8

    v = np.zeros_like(cur)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = _project_simplex(cur - step)
        cand_score = _score(cand)

        # accept improvements; occasional Metropolis step to escape plateaux
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
        else:
            no_improve += 1
            if rng.random() < 0.001:          # tiny probabilistic move
                cur, cur_score = cand, cand_score

        # record global best
        if cur_score < best_score:
            best_score, best_vec = cur_score, cur.copy()
            if best_score < target:
                break

        # periodic symmetrisation (helps many optimal patterns)
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                m.fill(0.0)
                v.fill(0.0)
            if sym_score < best_score:
                best_score, best_vec = sym_score, sym.copy()
                if best_score < target:
                    break

        # small random mass redistribution when stuck
        if no_improve % 2000 == 0 and no_improve > 0:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score(tmp)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    no_improve = 0

        # restart if no improvement for a long stretch
        if no_improve >= restart_limit:
            alpha = 0.3                                 # sparse Dirichlet
            cur = _project_simplex(rng.dirichlet(np.full(n, alpha)))
            cur_score = _score(cur)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # slowly decay the learning rate
        lr = max(lr * 0.9999, 1e-4)

        # safety check
        if time.monotonic() >= deadline:
            break

    return [float(x) for x in best_vec]
