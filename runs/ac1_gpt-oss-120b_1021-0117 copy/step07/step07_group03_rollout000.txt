import time
import math
import numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_max_and_grad(a, L):
    """
    Compute (max convolution value, subgradient) of a*a (full convolution).
    a is non‑negative, length n. L is a power‑of‑two >= 2n‑1.
    Returns max_val and a gradient vector of length n.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()

        a0 = rng.random(2000, dtype=np.float64) + 1e-3

    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two >= 2n‑1

    # ----- keep best -------------------------------------------------------
    max_val, grad = _conv_max_and_grad(cur, L)
    cur_sc = 2.0 * n * max_val
    best_sc = cur_sc
    best_vec = cur.copy()

    # ----- Adam optimizer state --------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha0 = 0.5

    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # current gradient
        max_val, grad = _conv_max_and_grad(cur, L)
        cur_sc = 2.0 * n * max_val

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
