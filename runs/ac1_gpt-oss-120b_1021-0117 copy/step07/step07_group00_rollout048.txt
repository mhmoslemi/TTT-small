import time, math, numpy as np

def _project_simplex(v):
    # Euclidean projection onto simplex sum=1
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2*n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # leave small margin for final return
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # initialization
    if 'height_sequence_1' in globals():
        cur = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        # default length 2000 if no parent exists
        n = 2000
        cur = rng.random(n, dtype=np.float64)
    # ensure nonnegative and project to simplex (sum=1)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # optimizer state
    momentum = np.zeros_like(cur)
    beta = 0.85          # momentum
    step0 = 2.0          # initial step scaling factor
    it = 0
    no_improve = 0
    target = 1.50274349

    # temperature for softmax
    temp = None
    decay = 0.998
    min_temp_factor = 1e-4

    # warmup iteration threshold to switch to hard max
    warmup_iters = 2000

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2*n - 1]
        max_val = conv.max()

        # temperature scheduling (only used in warmup)
        if it <= warmup_iters:
            if temp is None:
                temp = max_val * 0.09
            else:
                temp = max(temp * decay, max_val * min_temp_factor)
            # compute softmax weights
            w = np.exp((conv - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w = w / w_sum
        else:
            # hard max weight (average over all maxima)
            mask = (conv == max_val)
            cnt = mask.sum()
            if cnt == 0:
                w = np.ones_like(conv) / conv.size
            else:
                w = mask.astype(np.float64) / cnt

        # gradient via convolution with reversed weights
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size decay
        step = step0 / math.sqrt(it)

        # candidate update
        cand = _project_simplex(cur - step * momentum)

        # evaluate candidate
        cand_sc = _score(cand)

            no_improve = 0
            if cand_sc < best_sc:

                best_sc = cand_sc

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast AC1 upper bound for a vector already normalized to sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
