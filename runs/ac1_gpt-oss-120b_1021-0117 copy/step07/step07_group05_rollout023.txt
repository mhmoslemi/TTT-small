import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum==1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _score_fft(seq):
    """Fast (O(n log n)) evaluation of the AC1 bound used by `evaluate_sequence`."""
    seq = np.asarray(seq, dtype=np.float64)
    n = seq.size
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    # next power‑of‑two for efficient FFT
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()
    # convolution via FFT (real‑valued)
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[:m]
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a * sum_a)
