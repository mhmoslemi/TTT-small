Implementation details: Projection onto simplex algorithm O(n log n). Use standard implementation:

def project_simplex(v):
    # v is 1D numpy array
    # projects onto {x >= 0, sum x = 1}
    # Implementation from Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    # Input 1D array, returns projection onto {x >=0, sum=1}
    v = np.asarray(v, dtype=np.float64)
    # Sort descending
    u = np.sort(v)[::-1]

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, sum_i x_i = 1}
    """
    v = v.astype(np.float64, copy=False)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, fft_len: int) -> float:
    """Return max of the full linear convolution a*a using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _gradient(a: np.ndarray, fft_len: int):
    """
    Compute a sub‑gradient of max_b = max(conv(a,a)) and the value itself.
    The sub‑gradient is 2 * a reversed and shifted according to the
    index where the maximum occurs.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    k = int(conv.argmax())
    n = a.size
    g = np.zeros_like(a)
    if k < n:
        # overlap length = k+1
        g[: k + 1] = 2.0 * a[k :: -1]
    else:
        start = k - (n - 1)          # first index that participates
        g[start:] = 2.0 * a[start:][::-1]
    return g, float(conv.max()), k


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that attempts to
    minimise the AC1 upper bound defined by ``evaluate_sequence``.
    The routine works with any pre‑existing ``height_sequence_1``; otherwise
    it starts from the uniform distribution.
    """
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.05)

    # ------------------------------------------------------------------
    # problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000                     # fallback size

    # ------------------------------------------------------------------
    # initial vector (projected onto simplex, sum == 1)
    if "height_sequence_1" in globals():
        cur = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        cur = _project_simplex(cur)
    else:
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = cur.copy()
    # ------------------------------------------------------------------
    # helper: evaluate current sequence (no clipping – evaluator already clips)
    def _score(vec):
        s = vec.sum()
        if s < 0.01:
            return np.inf
        m = _conv_max(vec, fft_len)
        return 2.0 * n * m / (s * s)

    # pre‑compute FFT length (next power of two >= 2n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    best_score = _score(best_vec)

    # ------------------------------------------------------------------
    # deterministic baseline: uniform (already present when no parent)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = _score(uniform)
    if uniform_score < best_score:
        best_score = uniform_score
        best_vec = uniform.copy()

    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)

    it = 0
    # main optimisation loop – projected sub‑gradient descent + occasional noise
    while time.monotonic() < deadline and it < 200_000:
        it += 1

        # sub‑gradient step
        grad, maxb, _ = _gradient(cur, fft_len)
        # diminishing step size
        eta = 0.6 / np.sqrt(it)
        cand = cur - eta * grad
        cand = _project_simplex(cand)

        cand_score = _score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_vec = cand.copy()

        # accept improvement for the current trajectory
        if cand_score < _score(cur):
            cur = cand

        # every few iterations inject a small random perturbation to escape plateaus
        if it % 150 == 0:
            perturb = cur + rng.normal(scale=0.02, size=n)
            perturb = _project_simplex(perturb)
            pert_score = _score(perturb)
            if pert_score < best_score:
                best_score = pert_score
                best_vec = perturb.copy()
            if pert_score < _score(cur):
                cur = perturb

    # ------------------------------------------------------------------
    # final safety – evaluate a few random candidates in case the loop
    # never improved (very unlikely)
    for _ in range(5):
        rnd = rng.random(n)
        rnd = _project_simplex(rnd)
        sc = _score(rnd)
        if sc < best_score:
            best_score = sc
            best_vec = rnd.copy()

    # ------------------------------------------------------------------
    # Return the best seen vector as plain Python floats
    return [float(x) for x in best_vec]
