def _max_conv_and_grad(a, L, use_smooth=False, alpha=10.0):
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    if use_smooth:
        # compute softmax weights
        # shift to avoid overflow: subtract max_val from conv before exp
        exp_term = np.exp(alpha * (conv - max_val))
        weight_sum = exp_term.sum()
        if weight_sum == 0:
            w = np.zeros_like(conv)
        else:
            w = exp_term / weight_sum
    else:
        eps = max_val * 1e-12 + 1e-15
        mask = (conv >= max_val - eps).astype(np.float64)
        cnt = mask.sum()
        if cnt > 0:
            w = mask / cnt
        else:
            w = mask
    # reverse weights for convolution
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

import time, math, numpy as np

# --------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _max_conv_and_grad(a, L, use_smooth=False, alpha=10.0):
    """
    Compute max of full convolution a*a and a subgradient.
    If use_smooth=True, use a softmax weighting with parameter alpha.
    Returns (max_value, gradient_vector).
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)

    # full linear convolution via FFT
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    max_val = conv.max()

    if use_smooth:
        # softmax over convolution entries, shifted for numerical stability
        exps = np.exp(alpha * (conv - max_val))
        w_sum = exps.sum()
        w = exps / w_sum if w_sum != 0 else np.zeros_like(conv)
    else:
        eps = max_val * 1e-12 + 1e-15
        mask = (conv >= max_val - eps).astype(np.float64)
        cnt = mask.sum()
        w = mask / cnt if cnt > 0 else mask

    # gradient = 2 * (a * w_rev) evaluated at appropriate indices
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]

    return float(max_val), grad
