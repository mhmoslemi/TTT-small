def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.nonzero(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _score_exact(a):
    n = a.shape[0]
    conv = np.convolve(a, a)

    sum_a = a.sum()

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _score_fft(a):
    """Score 2*n*max(conv) using FFT (exact for our sizes)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    cur_sc = _score_fft(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # optimizer state
    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 1.0
    it = 0
    no_improve = 0
    target = 1.50274349

    # temperature for soft‑max weighting of the max‑convolution
    temp_factor = 1e-2
    temp_decay = 0.9995
    min_temp_factor = 1e-6

    # ----- main loop ------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # forward FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)

        # full convolution a*a
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weighting over lag indices
        temp = max_val * max(temp_factor, min_temp_factor)
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(0.0)

        # gradient of weighted max‑convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule
        step = step0 / math.sqrt(it)

        # candidate step
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score_fft(cand)

        # acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # periodic symmetrisation
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # random local redistribution when stuck
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_fft(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
            else:
                # full random restart
                cur = _project_simplex(rng.random(n))
                cur_sc = _score_fft(cur)
                momentum.fill(0.0)
                no_improve = 0

        # decay temperature factor
        temp_factor = max(temp_factor * temp_decay, min_temp_factor)

        # early exit if target reached
        if best_sc < target:
            break

        # time guard
        if it % 5000 == 0 and time.time() >= deadline:
            break

    # ----- final local refinement (optional) -----------------------------
    while time.time() < deadline - 0.2:
        i = rng.integers(0, n)
        if best_vec[i] <= 1e-12:
            continue
        delta = best_vec[i] * rng.random() * 0.05
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score_fft(cand)
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
