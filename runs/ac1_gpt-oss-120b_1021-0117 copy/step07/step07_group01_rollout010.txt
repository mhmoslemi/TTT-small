import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score = 2 * n * max(conv(a,a)).  a is assumed to sum to 1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small margin before the sandbox kills us
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------

        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2000, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    a = _project_simplex(a)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best)

    # hyper‑parameters
    step0 = 0.9          # initial step size
    tau0 = 0.005         # smoothing for the soft‑max weight

    it = 0

    while True:
        if time.monotonic() >= deadline:

        it += 1

        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()
        cur_sc = 2.0 * n * max_conv

        # ----- smooth weight over convolution entries -----
        tau = tau0 * (0.997 ** it)
        if tau < 1e-6:
            tau = 1e-6
        exp_arg = (conv - max_conv) / tau   # ≤ 0

        if w_sum == 0.0 or np.isnan(w_sum):

            if cnt > 0:
