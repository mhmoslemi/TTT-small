def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # Get n and init vector
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        # Use as fallback later
    else:
        n = 2000
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = None
    best_score = float('inf')

    # function to evaluate
    def eval_vec(v):
        # v assumed non-negative, sum general
        # compute objective using conv_max_and_grad
        max_val, _ = _conv_max_and_grad(v, L)
        sum_v = v.sum()
        if sum_v < 0.01:
            return float('inf')
        return 2.0 * n * max_val / (sum_v**2)

    # candidate generation: baseline from height_sequence_1
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
        # Symmetrize?
        a0 = (a0 + a0[::-1]) * 0.5
        score = eval_vec(a0)
        if score < best_score:
            best_score = score
            best_vec = a0.copy()

    # window families
    # Define helpers for generating shapes
    # compute index array
    idx = np.arange(n)
    center = (n - 1) / 2.0
    # Hann
    hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))
    # Hamming
    hamming = 0.54 - 0.46 * np.cos(2 * np.pi * idx / (n - 1))
    # Blackman
    blackman = 0.42 - 0.5 * np.cos(2 * np.pi * idx / (n - 1)) + 0.08 * np.cos(4 * np.pi * idx / (n - 1))

    for base in (hann, hamming, blackman):
        v = base.copy()
        if v.sum() == 0:
            continue
        v = v / v.sum()

# -------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_max_and_grad(a, L):
    """
    Compute (max convolution, gradient) of a*a (full convolution).
    a is non‑negative; L is a power‑of‑two >= 2*len(a)-1.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()

    w_rev = mask[::-1]

    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad
