import time, math, numpy as np

def _project_simplex(v):
    """Project onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _subgrad_maxconv(a, n, L):
    """Return sub‑gradient of max(conv) and the max value itself."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = conv >= max_val - eps
    w = mask.astype(np.float64)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 0.2)

    # --------------------------------------------------------------
    # Length and FFT size
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    else:
        n = 2000
        parent = None
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Initial candidates
    xs = np.arange(n, dtype=np.float64)
    cand_list = [
        np.full(n, 1.0 / n),                                   # uniform
        np.minimum(xs + 1, n - xs) / np.minimum(xs + 1, n - xs).sum(),  # triangle
        np.sin(np.pi * (xs + 1) / (n + 1)) / np.sin(np.pi * (xs + 1) / (n + 1)).sum(),  # sine bump
        (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2 / ((1 - np.cos(2 * np.pi * xs / (n - 1))) / 2).sum(),  # raised cosine
    ]
    for a in (0.001, 0.003, 0.006):
        exp = np.exp(-a * xs)
        cand_list.append(exp / exp.sum())
    for p in (0.5, 1.0, 1.5):
        pw = (xs + 1) ** (-p)
        cand_list.append(pw / pw.sum())
    rand = rng.random(n)
    cand_list.append(rand / rand.sum())
    if parent is not None:
        p = np.maximum(parent, 0.0)
        if p.sum() > 0:
            cand_list.append(p / p.sum())

    # --------------------------------------------------------------
    # Optimisation parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.08          # base learning rate for Adam
    target_score = 1.50274349

    best_vec = None

    # --------------------------------------------------------------

    for start in cand_list:

                conv_full = np.fft.irfft(a_fft * a_fft, n=L)

    return [float(x) for x in best_vec]
