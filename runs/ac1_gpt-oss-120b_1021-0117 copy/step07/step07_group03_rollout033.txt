    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5) # safe margin

    # initial vector
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
        cur = np.maximum(cur, 0.0)
    else:
        n = 2000
        cur = rng.random(n, dtype=np.float64)

    # project onto simplex (sum=1)
    cur = _project_simplex(cur)
    n = cur.size
    L = 1 << ((2*n - 1).bit_length())

    # optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_size = 0.1
    t = 0
    best_vec = cur.copy()
    best_sc = float('inf')
    cur_sc = float('inf')
    no_improve = 0

    target = 1.50274349

    while True:
        if time.time() >= deadline:
            break

        max_val, grad = _conv_max_and_grad(cur, L)
        cur_sc = 2.0 * n * max_val

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_sc < target:
            break

        t += 1
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step_size * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)

        # occasional symmetrization: every 500 iterations
        if t % 500 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        # evaluate candidate
        cand_max, _ = _conv_max_and_grad(cand, L)
        cand_sc = 2.0 * n * cand_max

        # accept if improvement else maybe reject
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            # occasional random perturb when not improving
            if no_improve > 2000:
                sigma = 0.1 / np.sqrt(t)
                pert = cur + sigma * rng.standard_normal(size=n)
                pert = _project_simplex(pert)
                pert_max, _ = _conv_max_and_grad(pert, L)
                pert_sc = 2.0 * n * pert_max
                if pert_sc < cur_sc:
                    cur = pert
                    cur_sc = pert_sc
                    no_improve = 0
                else:
                    # reset moments to allow new direction
                    m.fill(0.0)
                    v.fill(0.0)
                    no_improve = 0

    # final return: ensure vector robust to clipping
    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_max_and_grad(a, L):
    """
    Compute the maximum of the full convolution a*a and a sub‑gradient.
    a is a non‑negative real vector.
    Returns (max_value, gradient) where gradient has the same shape as a.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad
