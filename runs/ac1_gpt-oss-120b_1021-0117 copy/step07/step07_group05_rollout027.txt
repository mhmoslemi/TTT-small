def project_simplex(v):
    # v is 1D array, not necessarily sum=1.
    # algorithm from Duchi 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / (np.arange(1, n+1)) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def max_convolution(a):
    n = a.shape[0]
    m = 2*n - 1
    # compute FFT with zero padding
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    return np.max(conv)

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum==1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that strives to
    minimise the AC1 upper bound measured by ``evaluate_sequence``.
    The routine works with any pre‑existing ``height_sequence_1``; otherwise it
    falls back to a uniform distribution.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ----------------------------------------------------------------------
    # Problem size and initial vector
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = np.sum(parent)
        if s <= 0:
            # degenerate case – fall back to uniform
            n = len(parent)
            cur = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            cur = parent / s                     # scale to sum==1
    else:
        n = 2000                                 # reasonable default
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur.shape[0]
    m = 2 * n - 1

    # ----------------------------------------------------------------------
    # Pre‑compute convolution of the initial point
    conv = np.convolve(cur, cur)                # length 2n‑1
    cur_score = 2.0 * n * np.max(conv)          # sum(cur)==1 ⇒ simple formula
    best_vec = cur.copy()
    best_score = cur_score

    # ----------------------------------------------------------------------
    # Random engine and parameters
    rng = np.random.default_rng(seed)
    move_factor = 0.5            # maximal fraction of a[i] that may be moved
    temperature = max(1e-12, cur_score * 0.02)
    cooling = 0.9999
    min_temp = 1e-12
    min_step = 1e-12
    it = 0

    # ----------------------------------------------------------------------
    # Main optimisation loop – incremental convolution updates
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # delta ≤ cur[i]; move a random fraction
        delta = cur[i] * rng.random() * move_factor
        if delta < min_step:
            continue

        # ------------------------------------------------------------------
        # Incremental update of the autocorrelation
        conv_new = conv.copy()
        # linear part 2*delta*(a shifted at j) – 2*delta*(a shifted at i)
        conv_new[j:j + n] += 2.0 * delta * cur
        conv_new[i:i + n] -= 2.0 * delta * cur
        # quadratic part
        conv_new[2 * j] += delta * delta
        conv_new[2 * i] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        # ------------------------------------------------------------------
        # Evaluate the candidate (sum is preserved ⇒ score = 2 n max_conv)
        cand_score = 2.0 * n * np.max(conv_new)

        # ------------------------------------------------------------------
        # Simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, min_temp))
            if rng.random() < prob:
                accept = True

        if accept:
            # apply the move
            cur[i] -= delta
            cur[j] += delta
            conv = conv_new
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cur.copy()

        # ------------------------------------------------------------------
        # Temperature schedule
        temperature = max(min_temp, temperature * cooling)

        # occasional renormalisation to avoid drift (cheap, every 20000 iters)
        if it % 20000 == 0:
            total = np.sum(cur)
            if total != 0.0:
                cur /= total
                # recompute convolution exactly to keep consistency
                conv = np.convolve(cur, cur)
                cur_score = 2.0 * n * np.max(conv)
                if cur_score < best_score:
                    best_score = cur_score
                    best_vec = cur.copy()

    # ----------------------------------------------------------------------
    # Final safety – enforce constraints before returning
    best_vec = np.maximum(best_vec, 0.0)
    total = np.sum(best_vec)
    if total == 0.0:
        # extremely unlikely, fall back to uniform
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_vec = best_vec / total   # keep sum == 1 (scaling does not affect score)

    return [float(x) for x in best_vec]
