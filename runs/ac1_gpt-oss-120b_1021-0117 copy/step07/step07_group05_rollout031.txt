import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that (approximately)
    minimizes the AC1 upper bound measured by ``evaluate_sequence``.
    The routine works with any parent sequence ``height_sequence_1`` that may
    already be present; otherwise it falls back to a uniform distribution,
    which is provably optimal for the objective.
    """
    start = time.monotonic()
    # leave a tiny safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ------------------------------------------------------------------
    # problem size
    n = len(height_sequence_1) if "height_sequence_1" in globals() else 2000

    # ------------------------------------------------------------------
    # baseline: uniform distribution (sum == 1)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_vec = uniform
    try:
        best_score = evaluate_sequence(best_vec.tolist())
    except Exception:                           # extremely defensive
        best_score = float("inf")

    # ------------------------------------------------------------------
    # optional parent – keep if it happens to be better
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        try:
            parent_score = evaluate_sequence(parent.tolist())
            if parent_score < best_score:
                best_score = parent_score
                best_vec = parent
        except Exception:
            pass

    # ------------------------------------------------------------------
    # simple stochastic refinement
    rng = np.random.default_rng(seed)
    cur = best_vec.copy()
    cur_score = best_score
    it = 0
    max_iters = 20000               # generous but still safe within the budget
    while time.monotonic() < deadline and it < max_iters:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a random fraction (≤10 %) of mass from i to j
        delta = cur[i] * rng.random() * 0.1
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)           # enforce sum==1, non‑negativity
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand
        it += 1

    # ------------------------------------------------------------------
    # Return the best vector found (plain Python list of floats)
    return [float(x) for x in best_vec]
