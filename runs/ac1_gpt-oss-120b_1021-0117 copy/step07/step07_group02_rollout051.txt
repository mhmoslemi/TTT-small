def _project_simplex(v):

   v = np.asarray(v, dtype=np.float64)

   n = v.size

   u = np.sort(v)[::-1]

   css = np.cumsum(u) - 1

   ind = np.arange(1, n+1)

   cond = u - css/ind > 0

   if not np.any(cond):

       theta = 0

   else:

       rho = np.where(cond)[0][-1]

       theta = css[rho] / (rho+1)

   w = np.maximum(v - theta, 0.0)

   return w

import time, math, numpy as np

def _project_simplex(v):
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a, L=None):
    """Score for a vector a with sum(a)=1 (returns 2·n·max(conv))."""
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    L = 1 << ((2 * n - 1).bit_length())
