```
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
```

def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_grad(a, L):
    """
    Return (max convolution value, gradient of that max) for a*a.
    Gradient is a sub‑gradient of the max‑function.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad


def _conv_max(a, L):
    """Maximum entry of the full convolution a*a (FFT‑based)."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    return conv.max()

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---------- initial vector ----------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # uniform fallback
        a0 = np.full(2000, 1.0 / 2000, dtype=np.float64)

    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1

    # ---------- bookkeeping ----------
    max_val = _conv_max(cur, L)
    best_sc = 2.0 * n * max_val
    best_vec = cur.copy()
    cur_sc = best_sc

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 0.25

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    it = 0
    no_improve = 0
    target = 1.50274349

    # ---------------------------------------------------------
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # gradient of current point
        max_val, grad = _conv_max_and_grad(cur, L)
        cur_sc = 2.0 * n * max_val

        # Adam update (biased moments)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size decays like 1/√it
        step = step0 / (math.sqrt(it) + 1.0)

        # propose a new point and project onto the simplex
        cand = _project_simplex(cur - step * m_hat / (np.sqrt(v_hat) + eps))

        # evaluate candidate (only max needed)
        cand_max = _conv_max(cand, L)
        cand_sc = 2.0 * n * cand_max

        # greedy acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # keep the best ever seen
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            no_improve = 0

        # ----- occasional edge damping (helps to suppress large end‑values) -----
        if it % 500 == 0:
            edge = max(1, int(0.01 * n))
            tmp = cur.copy()
            tmp[:edge] *= 0.5
            tmp[-edge:] *= 0.5
            tmp = _project_simplex(tmp)
            tmp_max = _conv_max(tmp, L)
            tmp_sc = 2.0 * n * tmp_max
            if tmp_sc < best_sc:
                best_sc = tmp_sc
                best_vec = tmp.copy()
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0

        # ----- random perturbation when stuck -----
        if no_improve > 3000:
            sigma = 0.07 / math.sqrt(it)
            pert = cur + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(pert)
            pert_max = _conv_max(pert, L)
            pert_sc = 2.0 * n * pert_max
            if pert_sc < best_sc:
                best_sc = pert_sc
                best_vec = pert.copy()
                cur, cur_sc = pert, pert_sc
            else:
                # reset momentum to escape flat regions
                m.fill(0.0)
                v.fill(0.0)
            no_improve = 0

        # early stop if target already reached
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
