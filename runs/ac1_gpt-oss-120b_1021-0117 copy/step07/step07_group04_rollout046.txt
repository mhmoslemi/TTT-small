def _subgrad_max_conv(a, n, L):
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = conv >= max_val - eps
    w = mask.astype(np.float64)
    cnt = w.sum()
    if cnt > 0:
        w /= cnt
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val
```

def _objective_grad(a, n, L):
    s = a.sum()
    if s < 1e-12:
        return np.zeros_like(a), np.inf
    grad_f, max_val = _subgrad_max_conv(a, n, L)
    factor1 = 2.0 * n / (s ** 2)
    factor2 = 4.0 * n * max_val / (s ** 3)
    grad = factor1 * grad_f - factor2 * np.ones_like(a)
    return grad, max_val

def _score_fft(a, n, L):
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    max_val = conv.max()
    return float(2.0 * n * max_val / (s * s))
```

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast evaluation of the AC1 bound for a non‑negative vector a (sum is irrelevant)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    return float(2.0 * n * max_val / (s * s))

def _objective_grad(a, n, L):
    """Gradient (sub‑gradient) of the AC1 objective w.r.t. a."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = conv >= max_val - eps
    w = mask.astype(np.float64)
    cnt = w.sum()
    if cnt > 0:
        w /= cnt
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad_f = 2.0 * grad_full[idx]

    grad = factor1 * grad_f - factor2 * np.ones_like(a)
