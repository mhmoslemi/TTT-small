import time, math, numpy as np

def _project_simplex(v):
    # projection onto simplex sum=1, v >=0
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to minimise
    the AC1 upper bound as measured by ``evaluate_sequence``.
    The routine works with a possible parent sequence ``height_sequence_1``;
    otherwise it falls back to a uniform start.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Initialise a feasible vector (sum == 1, non‑negative)
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
    else:
        n = 2000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = _project_simplex(a)

    # --------------------------------------------------------------
    # Baseline evaluation
    try:
        cur_score = evaluate_sequence(a.tolist())
    except Exception:
        cur_score = float("inf")
    best_vec = a.copy()
    best_score = cur_score

    # --------------------------------------------------------------
    # Parameters for the search
    max_iters = 100_000          # hard upper bound; loop also checks time
    step0 = 0.5                  # initial step size for gradient steps
    max_move_frac = 0.3          # max fraction of source mass moved in random steps
    grad_prob = 0.7              # probability to perform a gradient step
    eps = 1e-12                  # tolerance for accepting improvements

    # --------------------------------------------------------------
    # Main optimisation loop
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # ---- decide move type -------------------------------------------------
        if rng.random() < grad_prob:
            # ----- gradient step -------------------------------------------------
            # convolution and location of current maximum
            conv = np.convolve(a, a)
            k_max = int(np.argmax(conv))

            # gradient of max_b = (a*a)[k_max] w.r.t. a
            g = np.zeros_like(a)
            if k_max < n:
                # overlap region i = 0 .. k_max
                g[: k_max + 1] = 2.0 * a[k_max :: -1]
            else:
                offset = k_max - (n - 1)
                g[offset:] = 2.0 * a[::-1]

            step = step0 / math.sqrt(it + 1.0)
            cand = a - step * g
        else:
            # ----- random mass transfer -----------------------------------------
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = a[i] * rng.random() * max_move_frac
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta

        # ---- projection onto simplex -------------------------------------------
        cand = _project_simplex(cand)

        # ---- evaluate ----------------------------------------------------------
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue

        # ---- accept if it improves the current vector -------------------------
        if cand_score < cur_score - eps:
            a = cand
            cur_score = cand_score

        # ---- keep the best ever ------------------------------------------------
        if cand_score < best_score - eps:
            best_score = cand_score
            best_vec = cand.copy()

    # --------------------------------------------------------------
    # Final safety clipping (evaluator also clips, but we do it here)
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Return as plain Python list
    return [float(x) for x in best_vec]
