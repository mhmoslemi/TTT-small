def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)

### projection onto simplex

We'll define:

def simplex_projection(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time, numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    start = time.monotonic()
    deadline = start + max(0., float(budget_s) - 0.1)  # margin
    rng = np.random.default_rng(seed)

    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        total = np.sum(parent)
        if total < 1e-12:
            a = np.full(parent.shape, 1.0 / parent.size)
        else:
            a = parent / total
        n = parent.size
    else:
        n = 2000
        a = np.full(n, 1.0 / n)

    # compute initial score
    c = np.convolve(a, a)
    best_score = 2 * n * np.max(c)
    best_vec = a.copy()

    # parameters
    max_iters = 20000
    iter_count = 0
    no_improve = 0
    # set step fraction
    step_frac = 0.2

    while time.monotonic() < deadline and iter_count < max_iters:
        iter_count += 1

        # compute convolution and max
        c = np.convolve(a, a)
        max_b = np.max(c)
        # quick compute score
        cur_score = 2 * n * max_b

        # In case we already have the best_score and no further improvement after many steps, break
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Choose to either do gradient move or random move
        if rng.random() < 0.7:
            # gradient guided move

            # identify indices near max
            # tolerance as relative small fraction of max_b
            tol = max_b * 1e-12  # high precision
            k_max = np.where(c >= max_b - tol)[0]

            # Compute subgradient
            grad = np.zeros_like(a)
            count = np.zeros_like(a, dtype=np.int32)
            for k in k_max:
                i_start = k - (n - 1)
                if i_start < 0:
                    i_start = 0
                i_end = k
                if i_end >= n:
                    i_end = n - 1
                # slice indices
                # compute a_sub = a[k - np.arange(i_start, i_end+1)]
                idx = k - np.arange(i_start, i_end + 1)
                # compute a_sub
                a_sub = a[idx]
                # accumulate
                grad[i_start:i_end + 1] += a_sub
                count[i_start:i_end + 1] += 1

            # Subgradient including a_i * count
            subgrad = grad + a * count

            # choose source with highest subgrad, target with lowest
            src = np.argmax(subgrad)
            tgt = np.argmin(subgrad)
            if src == tgt or a[src] == 0:
                # fallback random
                src, tgt = rng.choice(n, size=2, replace=False)
        else:
            # purely random pair
            src, tgt = rng.choice(n, size=2, replace=False)

        # Determine delta
        # Use a fraction of source mass
        delta = a[src] * rng.random() * step_frac
        if delta <= 0:
            continue
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        # projection
        cand = simplex_projection(cand)

        # evaluate candidate score quickly
        c_cand = np.convolve(cand, cand)
        cand_score = 2 * n * np.max(c_cand)

        # acceptance condition: accept if better (or maybe also accept with probability)
        if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / max(1e-8, cur_score) * 0.1):
            a = cand
            cur_score = cand_score
            # update best
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
                no_improve = 0
        # else reject

        # occasional random restart (if stuck)
        if no_improve >= 500 and time.monotonic() < deadline:
            # random perturbation to escape
            a = best_vec.copy()
            # add random noise
            noise = rng.random(n) * step_frac * 0.05  # small noise
            a = a + noise
            a = simplex_projection(a)
            no_improve = 0

    # final fallback: evaluate best vector using evaluate_sequence to confirm not infinite
    try:
        final_score = evaluate_sequence(best_vec.tolist())
        if not np.isfinite(final_score):
            # fallback uniform
            best_vec = np.full(n, 1.0 / n)
    except Exception:
        best_vec = np.full(n, 1.0 / n)

    return [float(x) for x in best_vec]

import time, numpy as np

def _simplex_projection(v):
    """Euclidean projection of v onto the probability simplex (sum=1, non‑negative)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    """Return the AC1 upper bound for a vector with sum=1."""
    # a should be non‑negative, sum≈1
    # direct convolution, take maximum, multiply by 2n
    return 2.0 * n * np.max(np.convolve(a, a))
