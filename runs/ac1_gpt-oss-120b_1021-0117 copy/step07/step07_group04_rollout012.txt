def subgradient_maxconv(a, n, L):
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = conv >= max_val - eps
    cnt = mask.sum()
    if cnt == 0:
        w = mask.astype(np.float64)
    else:
        w = mask.astype(np.float64) / cnt
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    # we need gradient for each a_i: gradient_i = 2 * sum_{k} a_{k} * w_{i+k}? The earlier idx mapping.
    # earlier code: idx = (2*n - 2) - np.arange(n); grad = 2.0 * grad_full[idx]
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val

def optimize_from(init_vec, n, L, deadline):
   cur = init_vec.copy()
   cur = project_onto_simplex(cur)    # ensure sum=1
   cur_sc = score_fft(cur, n, L)
   best_local = cur_sc
   best_local_vec = cur.copy()
   # Adam hyperparameters
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8
   step_alpha = 0.5
   m = np.zeros(n)
   v = np.zeros(n)
   t = 0
   no_improve = 0
   while time.monotonic() < deadline:
       t += 1
       # compute grad
       grad, _ = subgradient_maxconv(cur, n, L)
       # Adam update
       m = beta1 * m + (1 - beta1) * grad
       v = beta2 * v + (1 - beta2) * (grad * grad)
       m_hat = m / (1 - beta1**t)
       v_hat = v / (1 - beta2**t)
       update = step_alpha * m_hat / (np.sqrt(v_hat) + eps)
       cand = project_onto_simplex(cur - update)
       cand_sc = score_fft(cand, n, L)
       # Accept if better OR occasionally accept worse?
       if cand_sc < cur_sc:
           cur = cand
           cur_sc = cand_sc
           no_improve = 0

               cur_sc = cand_sc

    # define deterministic inits
    xs = np.arange(n, dtype=np.float64)
    inits = []
    inits.append(np.full(n, 1.0/n))
    tri = np.minimum(xs+1, n - xs)
    inits.append(tri/tri.sum())
    sine = np.sin(np.pi*(xs+1)/(n+1))
    inits.append(sine/sine.sum())
    bump = (1 - np.cos(2*np.pi*xs/(n-1)))/2
    inits.append(bump/bump.sum())
    for alpha in (0.001, 0.003, 0.006):
        exp = np.exp(-alpha*xs)
        inits.append(exp/exp.sum())
    for p in (0.5, 1.0, 1.5):
        pw = (xs+1)**(-p)
        inits.append(pw/pw.sum())
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.sum() > 0:
            parent = parent/parent.sum()
            inits.append(parent)

    best_score = float('inf')
    best_vec = None

    # iterate over deterministic inits
    for init in inits:
        # compute remaining time
        if time.monotonic() >= deadline:
            break
        # run optimization with subdeadline
        sub_dead = min(deadline, time.monotonic() + 5.0) # allocate up to 5 seconds per init
        sc, vec = optimize_from(init, n, L, sub_dead)
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()

    # random restarts until time runs out
    while time.monotonic() < deadline:

        rand_vec = rng.random(n)

    # fallback: if best_vec still None (unlikely), set uniform

    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # fix possible rounding error
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast score (no clipping, a assumed non‑negative)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

def _grad_and_score(a, n, L):
    """Return sub‑gradient of max‑convolution, the max value and the score."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = conv >= max_val - eps
    cnt = mask.sum()
    if cnt > 0:
        w = mask.astype(np.float64) / cnt
    else:
        w = mask.astype(np.float64)
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    s = a.sum()
    score = float(2.0 * n * max_val / (s * s))
    return grad, max_val, score

    cur = _project_simplex(init.copy())
    cur_grad, cur_max, cur_sc = _grad_and_score(cur, n, L)
    best_sc = cur_sc
    best_vec = cur.copy()

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.5
    t = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        t += 1
        grad, _, _ = _grad_and_score(cur, n, L)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)
