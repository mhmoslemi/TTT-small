def lp_step(f, M, maxiter=100):
    n = len(f)
    A = build_conv_matrix(f)  # shape (2n-1, n)
    # upper bound vector:
    ub = np.full(2*n-1, M, dtype=np.float64)
    # objective: maximize sum(b) => minimize -1^T b
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=ub, bounds=(0, None), method='highs')
    if not res.success:
        return None
    b = res.x
    return b

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex {x>=0, sum(x)=1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Compute the AC1 upper bound for a given (non‑negative) sequence.
def _ac1_score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = np.convolve(a, a, mode='full')
    max_conv = conv.max()
    # denominator is (sum a)^2 ; sum is usually 1 after normalisation
    s = a.sum()
    if s == 0:
        return float('inf')
    return 2.0 * n * max_conv / (s * s)

# ----------------------------------------------------------------------
# Build the sparse convolution matrix A such that (A @ x)[k] = (f * x)[k].
def _conv_matrix(f: np.ndarray) -> csr_matrix:
    """Return sparse matrix of shape (2*n-1, n) with rows = f * x."""
    n = f.shape[0]
    # generate all pairs (i,j) -> row = i+j, col = j, data = f[i]
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    ii, jj = np.meshgrid(i, j, indexing='ij')
    rows = (ii + jj).ravel()                     # shape (n*n,)
    cols = jj.ravel()                            # shape (n*n,)
    data = np.repeat(f, n)                       # f[i] repeated n times
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimises the
    AC1 upper bound as measured by ``evaluate_sequence``.
    The routine starts from the supplied parent (if any) or from a uniform
    distribution, then performs a few sparse‑LP iterations (the classic
    alternating‑projection scheme) followed by a cheap stochastic local search.
    """
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ------------------------------------------------------------------
    # initialise the sequence
    if "height_sequence_1" in globals():
        f = np.asarray(height_sequence_1, dtype=np.float64)
        f = np.clip(f, 0.0, 1000.0)
    else:
        # a reasonable default length when no parent is present
        n_default = 2000
        f = np.full(n_default, 1.0 / n_default, dtype=np.float64)

    # normalise to sum == 1  (objective is scale‑invariant)
    s = f.sum()
    if s < 1e-12:
        f = np.full_like(f, 1.0 / f.size, dtype=np.float64)
    else:
        f = f / s

    n = f.size

    # ------------------------------------------------------------------
    # keep the best solution found so far
    best_seq = f.copy()
    best_score = _ac1_score(best_seq)

    # also compare with the plain uniform vector (might be better)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = _ac1_score(uniform)
    if uniform_score < best_score:
        best_seq = uniform.copy()
        best_score = uniform_score

    # ------------------------------------------------------------------
    # parameters for the alternating LP scheme
    mix_factor = 0.2               # how much of the new direction to mix in
    max_lp_iters = 12              # a dozen iterations are usually enough
    lp_iter = 0

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # main optimisation loop (LP steps first, then cheap local search)
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 1) LP step – try to reduce the maximal autocorrelation
        if lp_iter < max_lp_iters:
            lp_iter += 1
            # current bound M
            conv = np.convolve(f, f, mode='full')
            M = conv.max()

            # build the sparse convolution matrix for the current f
            A = _conv_matrix(f)                # shape (2n-1, n)

            # linprog formulation: maximise sum(b)  <=> minimise -1^T b
            c = -np.ones(n, dtype=np.float64)
            ub = np.full(2 * n - 1, M, dtype=np.float64)

            # solve the LP (highs handles sparse matrices well)
            res = linprog(
                c,
                A_ub=A,
                b_ub=ub,
                bounds=[(0, None)] * n,
                method='highs',
                options={'presolve': True, 'time_limit': max(0.0, deadline - time.monotonic())}
            )
            if not res.success:
                # give up the LP part if the solver fails
                max_lp_iters = lp_iter  # stop trying further LP steps
            else:
                b = res.x
                b_sum = b.sum()
                if b_sum > 0:
                    g = b / b_sum                     # normalise to sum == 1
                    # convex combination
                    f_new = (1.0 - mix_factor) * f + mix_factor * g
                    # clip and renormalise (tiny numerical errors may appear)
                    f_new = np.clip(f_new, 0.0, 1000.0)
                    f_new_sum = f_new.sum()
                    if f_new_sum > 0:
                        f_new = f_new / f_new_sum
                    # evaluate
                    new_score = _ac1_score(f_new)
                    if new_score < best_score:
                        best_score = new_score
                        best_seq = f_new.copy()
                    # accept if it improves the current f, otherwise keep f
                    if new_score < _ac1_score(f):
                        f = f_new
                else:
                    max_lp_iters = lp_iter  # nothing useful can be extracted
            continue  # restart loop to respect time budget

        # --------------------------------------------------------------
        # 2) Simple stochastic local search (after LP phase)
        #    Move a random fraction of mass from i to j and re‑project.
        i, j = rng.integers(0, n, size=2)
        if i != j:
            cur = best_seq.copy()
            delta = cur[i] * rng.random() * 0.3          # up to 30 % of the source mass
            if delta > 0:
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)            # restore exact sum == 1
                cand_score = _ac1_score(cand)
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand
        # give the loop a chance to exit if time is exhausted
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # final safety: clip to the evaluator's admissible range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return as plain Python list of floats
    return [float(x) for x in best_seq]
