import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Fix possible tiny numerical drift
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast evaluation of the AC1 upper bound (assumes a>=0 and sum(a)=1)."""
    s = a.sum()
    if s < 1e-12:

    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

                    delta = tmp[i] * rng.random() * 0.3
                    tmp[i] -= delta
                    tmp[j] += delta
            tmp = _project_simplex(tmp)
            cur, cur_sc = tmp, _score_fft(tmp, n, L)
            no_improve = 0

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain floats)
    return [float(x) for x in best_vec]
