    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    """Score assuming sum(a)==1 (denominator becomes 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score_vec(best_vec)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)
    beta = 0.9          # momentum factor
    step0 = 1.0         # initial step magnitude
    it = 0
    target = 1.50274349

    # ----- optimisation loop -----------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        k_max = int(conv.argmax())
        cur_sc = 2.0 * n * max_val

        # gradient of the active (maximum) convolution entry
        grad = np.zeros_like(cur)
        j_start = max(0, k_max - (n - 1))

            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                delta = cur[i] * rng.random() * 0.3
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_vec(tmp)
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc

        # ----- symmetrisation (helps for many optimal shapes) -------
        if it % 4000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # early exit if target reached
        if best_sc < target:
            break

        # periodic time check
        if it % 1000 == 0 and time.monotonic() >= deadline:
            break

    # ----- return best candidate found --------------------------------
    return [float(x) for x in best_vec]
