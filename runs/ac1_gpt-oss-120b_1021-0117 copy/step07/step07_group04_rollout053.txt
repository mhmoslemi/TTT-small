import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # adjust rounding errors:
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w
```

Now _score_fft:

```python
def _score_fft(a, n, L):
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2*n-1]
    max_c = conv.max()
    return float(2.0 * n * max_c / (s * s))
```

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)

    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w
