import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam step --------------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = lr0 / (1 + it * 1e-4)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)

        # ----- evaluate candidate -----------------------------------------
        cand_sc = 2.0 * n * _max_conv(cand)

        # ----- acceptance --------------------------------------------------
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (lr + 1e-12)):
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # ----- keep global best -------------------------------------------
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()
            no_improve = 0

        # ----- periodic symmetrisation ------------------------------------
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = 2.0 * n * _max_conv(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ----- random perturb when stuck ----------------------------------
        if no_improve > 2000:
            noise = rng.normal(scale=0.02, size=n)
            tmp = best_vec + noise
            tmp = _project_simplex(np.clip(tmp, 0.0, None))
            tmp_sc = 2.0 * n * _max_conv(tmp)
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
                cur, cur_sc = tmp, tmp_sc
            no_improve = 0

        # ----- early exit -------------------------------------------------
        if best_sc < target:
            break

    # ----- final quick local tweaks ---------------------------------------
    while time.monotonic() < deadline - 0.1:
        i, j = rng.choice(n, size=2, replace=False)
        delta = best_vec[i] * rng.random() * 0.08
        tmp = best_vec.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = 2.0 * n * _max_conv(tmp)
        if tmp_sc < best_sc:
            best_sc, best_vec = tmp_sc, tmp.copy()

    return [float(x) for x in best_vec]
