def _conv_full_and_smooth_grad(a, L, beta):
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    # compute smooth weights
    if beta > 0:
        # shift for stability
        shifted = conv - max_val
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum) or np.isinf(w_sum):
            # fallback to mask
            mask = (conv >= max_val - 1e-12 * max_val).astype(np.float64)
            cnt = mask.sum()
            if cnt > 0:
                mask /= cnt
            w = mask
        else:
            w = w / w_sum
    else:
        # beta=0 => uniform weights
        w = np.full_like(conv, 1.0/(conv.size))
    # gradient:
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    # extract entries
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad, w

import time

import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _conv_max(a, L):
    """Maximum value of full convolution a*a using FFT of length L."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    return conv.max()


def _conv_max_and_smooth_grad(a, L, beta):
    """
    Return (max_convolution, gradient) where the gradient is computed from a
    smoothed version of the max (softmax with temperature 1/beta).
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)

    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()

    # --- smooth weights -------------------------------------------------
    # relative deviation, avoids underflow when max is tiny
    rel = (conv - max_val) / (max_val + 1e-15)   # lies in [-1,0]
    if beta > 0:
        w = np.exp(beta * rel)                 # more concentrated as beta grows
        w_sum = w.sum()
        if not np.isfinite(w_sum) or w_sum == 0.0:
            # fall back to strict mask (hard max)
            eps = max_val * 1e-12 + 1e-15
            mask = (conv >= max_val - eps).astype(np.float64)
            cnt = mask.sum()
            w = mask / cnt if cnt > 0 else mask
        else:
            w = w / w_sum
    else:
        w = np.full_like(conv, 1.0 / conv.size)

    # --- gradient (2 * a * w_rev) --------------------------------------
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)   # selects the needed slice
    grad = 2.0 * grad_full[idx]
    return max_val, grad
