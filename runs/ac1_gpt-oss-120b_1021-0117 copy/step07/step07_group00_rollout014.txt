    # given logits, return softmax distribution
    e = np.exp(z - np.max(z))
    return e / e.sum()

    n = a.shape[0]

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]

import time, math, numpy as np

def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / e.sum()


def _score(a: np.ndarray) -> float:
    """Fast surrogate score (exact when sum(a)=1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
