import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    # Find rho
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())
```

    v = np.zeros_like(cur)

    beta2 = 0.999
    eps = 1e-8

    temp_factor = 2e-3
    temp_decay = 0.9997
    min_temp_factor = 1e-6

        a_fft = np.fft.rfft(cur, n=L)
        # convolution full
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        temp = max_val * max(temp_factor, min_temp_factor)
        # to avoid overflow, subtract max_val before exp
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum > 0:

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    """Score for a vector with sum(a)==1 (returns 2*n*max(conv))."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score_vec(best_vec)
    cur_sc = best_sc

    # ----- Adam optimisation parameters ----------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha0 = 0.5                     # base learning rate

    # ----- soft‑max temperature -----------------------------------------
    temp_factor = 2e-3               # relative to current max value
    temp_decay = 0.9997
    min_temp_factor = 1e-6

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # forward FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)

        # full convolution a*a
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weighting over lag indices
        temp = max_val * max(temp_factor, min_temp_factor)
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:
            w.fill(0.0)

        # gradient of weighted max‑convolution via FFT
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (alpha0 / math.sqrt(it)) * (m_hat / (np.sqrt(v_hat) + eps))

        # optional clipping to keep step moderate
        step_norm = np.linalg.norm(step)
        if step_norm > 1.0:
            step = step * (1.0 / step_norm)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score_vec(cand)

        # acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional probabilistic acceptance to escape plateaus
            prob = math.exp(-(cand_sc - cur_sc) / (max_val + 1e-12))
            if rng.random() < prob * 0.01:
                cur, cur_sc = cand, cand_sc
                no_improve = 0

        # record global best
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()

        # periodic symmetrisation (helps for many near‑optimal patterns)
        if it % 600 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < best_sc:
                best_sc, best_vec = sym_sc, sym.copy()
                cur, cur_sc = sym, sym_sc
                no_improve = 0

        # random local redistribution when stuck
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_vec(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
            else:
                # full random restart
                cur = _project_simplex(rng.random(n))
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0
                cur_sc = _score_vec(cur)

        # decay temperature factor
        temp_factor = max(temp_factor * temp_decay, min_temp_factor)

        # early exit if target achieved
        if best_sc < target:
            break

    # Return best-found vector as plain Python floats
    return [float(x) for x in best_vec]
