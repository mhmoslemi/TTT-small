import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    cond = u - css / np.arange(1, n + 1)
    rho = np.where(cond > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_seq(a: np.ndarray) -> float:
    """Exact AC1 upper bound for a non‑negative sequence a."""
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return float("inf")
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _grad_max_corr(a: np.ndarray, max_idx: int) -> np.ndarray:
    """Sub‑gradient of max_k (a * a)[k] at the index where the maximum occurs."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, max_idx - (n - 1))
    hi = min(n - 1, max_idx)
    if lo <= hi:
        js = np.arange(lo, hi + 1, dtype=np.int64)
        grad[js] = 2.0 * a[max_idx - js]
    return grad

        else:
