import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    """Score when sum(a)≈1: 2·n·max(a*a)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    sum_a = a.sum()
    if sum_a == 0:
        return np.inf
    return float(2.0 * n * max_val / (sum_a ** 2))

def _grad_subconv(a):
    """Subgradient of max_{k} (a*a)(k) w.r.t a (assuming a≥0)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    # indices where the maximum is attained (within tolerance)
    atol = max(1e-15, 1e-12 * max_val)
    mask = conv >= max_val - atol
    cnt = mask.sum()
    if cnt == 0:
        # fallback to single max index
        mask = np.zeros_like(conv, dtype=bool)
        mask[np.argmax(conv)] = True
        cnt = 1
    w = np.empty_like(conv)
    w.fill(0.0)
    w[mask] = 1.0 / cnt
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val
