def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., float(budget_s) - 0.5)   # buffer

    # Determine n and L

        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.shape[0]
    else:
        n = 2000
        parent_arr = None

    L = 1 << ((2*n - 1).bit_length())

    # Build initial pool
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0/n))

    xs = np.arange(n, dtype=np.float64)

    # triangular base shape
    tri = np.minimum(xs+1, n - xs)
    # param p
    for p in (-4, -2, -1.5, -1, -0.5, 0, 0.5, 1, 2, 4):
        a = np.power(tri, p)
        a_sum = a.sum()

            candidates.append(a / a_sum)

    # cosine bump
    bump = (1 - np.cos(2 * np.pi * xs / (n-1))) / 2
    for p in (-4, -2, -1, -0.5, 0, 0.5, 1, 2, 4):

        s = a.sum()

    # sine bump
    sine = np.sin(np.pi * (xs+1) / (n+1))

        s = a.sum()

    # exponential symmetric shape
    for alpha in (0.0005, 0.001, 0.002, 0.005, 0.01, 0.02):

        exp_right = np.exp(-alpha * (n-1 - xs))
        a = exp_left + exp_right
        s = a.sum()

    # beta distribution shapes
    xs_norm = xs / (n-1)
    betas = [(0.5,0.5), (1,1), (2,2), (0.5,2), (2,0.5), (1,2), (2,1), (0.2,0.2), (5,5)]

        s = a.sum()

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # correct possible tiny drift
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast computation of the evaluator score (assumes a >= 0, sum(a) > 0)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))
