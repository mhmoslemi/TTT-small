def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.2

    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64) + 1e-8
    # Ensure nonnegative, project onto simplex (sum=1)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)
    n = cur.size

    L = 1 << ((2 * n - 1).bit_length())
    best = cur.copy()
    best_score = _score(best)

    momentum = np.zeros_like(cur)
    beta = 0.85
    step0 = 1.0

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # compute convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # indices where max occurs
        max_idx = np.flatnonzero(conv == max_val)

        # gradient compute: average over maxima
        grad = np.zeros_like(cur)
        for k in max_idx:
            i = np.arange(n)
            mask = (k - i >= 0) & (k - i < n)
            grad[mask] += 2.0 * cur[k - i[mask]]
        if max_idx.size > 0:
            grad /= max_idx.size

        # momentum
        momentum = beta * momentum + (1 - beta) * grad

        step = step0 / math.sqrt(iteration)
        cand = cur - step * momentum
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _score(cand)

        # Accept if better
        if cand_score < best_score:
            best, best_score = cand.copy(), cand_score
        # Use simple acceptance rule: accept if better, else occasional random chance
        if cand_score < _score(cur):
            cur = cand
        else:
            # occasional random perturbation
            if rng.random() < 0.01:
                # small random perturbation
                pert = rng.normal(scale=0.01, size=n)
                cur = cur + pert
                cur = np.maximum(cur, 0.0)
                cur = _project_simplex(cur)

        # Symmetrization periodically
        if iteration % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score(sym)
            if sym_score < best_score:
                best, best_score = sym.copy(), sym_score
            if sym_score < _score(cur):
                cur = sym

        # Early stop if best_score already below target (but we can use same target)
        if best_score < 1.50274349:
            break

    return [float(x) for x in best]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a, L):
    """Fast surrogate of the evaluator (sum of a is assumed 1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # ------------------------------------------------------------------
    # initialise
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64) + 1e-8
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())  # power‑of‑2 length for FFT

    # best so far
    best_vec = cur.copy()
    best_sc = _score(best_vec, L)

    # optimiser state
    momentum = np.zeros_like(cur)
    beta = 0.88
    step0 = 0.8
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ---------- convolution ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        max_idx = np.flatnonzero(conv == max_val)

        # ---------- sub‑gradient of the true max ----------
        grad = np.zeros_like(cur)
        for k in max_idx:
            i = np.arange(n)
            mask = (k - i >= 0) & (k - i < n)
            grad[mask] += 2.0 * cur[k - i[mask]]
        if max_idx.size:
            grad /= max_idx.size

        # ---------- momentum ----------
        momentum = beta * momentum + (1.0 - beta) * grad

        # ---------- step ----------
        step = step0 / math.sqrt(it)
        cand = cur - step * momentum
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0:
            cand /= s
        else:
            # fallback to uniform if everything vanished
            cand = np.full_like(cand, 1.0 / n)

        cand_sc = _score(cand, L)

        # ---------- acceptance ----------
        cur_sc = _score(cur, L)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional uphill move
            temp = step * 0.4 + 1e-12
            if rng.random() < math.exp(-(cand_sc - cur_sc) / temp):
                cur, cur_sc = cand, cand_sc
                no_improve = 0

        # ---------- record best ----------
        if cur_sc < best_sc:
            best_vec, best_sc = cur.copy(), cur_sc
        if cand_sc < best_sc:
            best_vec, best_sc = cand.copy(), cand_sc

        # ---------- periodic symmetrisation ----------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < best_sc:
                best_vec, best_sc = sym.copy(), sym_sc
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0

        # ---------- random local redistribution when stuck ----------
        if no_improve > 3500:
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc

        # ---------- early exit ----------
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
