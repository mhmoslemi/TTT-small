import time, math, numpy as np

def _project_simplex(v):
    # same code as earlier (optimized)
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    n = a.shape[0]
    # compute convolution via FFT
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    s = np.sum(a)
    if s == 0:
        return np.inf
    return float(2.0 * n * conv.max() / (s * s))

# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v):
    """Project onto {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]                     # descending
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    s = np.sum(a)
    if s == 0.0:

                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_vec(tmp)
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best = tmp.copy()
                    a = tmp
                    momentum.fill(0.0)
                    no_improve = 0
                else:
                    # fallback: tiny Gaussian noise
                    noise = rng.normal(scale=0.005, size=n)
                    tmp = _project_simplex(a + noise)
                    tmp_sc = _score_vec(tmp)
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best = tmp.copy()
                        a = tmp
                        momentum.fill(0.0)
                        no_improve = 0
            else:
                # if the picked entry is zero, restart from best with small noise
                noise = rng.normal(scale=0.01, size=n)
                a = _project_simplex(best + noise)
                momentum.fill(0.0)
                no_improve = 0

        # ---------- early exit ----------
        if best_sc < target:
            break

        # periodic time‑check (cheap)
        if it % 2000 == 0 and time.monotonic() >= deadline:
            break

    # --------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    # --------------------------------------------------------
    return [float(x) for x in best]
