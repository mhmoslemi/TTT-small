import time, math, numpy as np
import cvxpy as cp

# --------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _max_conv(a, L):
    """Maximum value of the full convolution a*a using FFT."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * a.shape[0] - 1]
    return conv.max()

def _conv_max_and_grad(a, L):
    """Maximum convolution value and a sub‑gradient for the max‑operator."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

                    abstol=1e-8, reltol=1e-8, feastol=1e-8)
    except Exception:                               # fallback to SCS
        try:
            prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        except Exception:
            return None
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None
    val = b.value
    if val is None:
        return None
    return np.maximum(val, 0.0)

# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2000, dtype=np.float64)
    a0 = np.maximum(a0, 0.0)
    cur = _project_simplex(a0)          # simplex (sum = 1)
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())  # power‑of‑two FFT size

    max_val = _max_conv(cur, L)
    cur_sc = 2.0 * n * max_val
    best_sc = cur_sc
    best_vec = cur.copy()

    target = 1.50274349

    # ----- LP refinement (few aggressive steps) -------------------------
    for _ in range(10):
        if time.time() >= deadline:
            break
        M = _max_conv(cur, L)
        b_raw = _solve_lp(cur, M)
        if b_raw is None:
            break
        sb = b_raw.sum()
        if sb <= 0.0:
            break
        b_norm = b_raw / sb
        # small symmetrisation and simplex projection
        b_norm = (b_norm + b_norm[::-1]) * 0.5
        b_norm = _project_simplex(b_norm)

        new_max = _max_conv(b_norm, L)
        new_sc = 2.0 * n * new_max

        if new_sc < best_sc:
            best_sc = new_sc
            best_vec = b_norm.copy()
        if new_sc < cur_sc:
            cur = b_norm
            cur_sc = new_sc
        else:
            break          # no further gain from LP
        if best_sc < target:
            break

    # ----- gradient‑based fine tuning ------------------------------------
    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.8
    it = 0
    no_improve = 0

    while True:
        if time.time() >= deadline:
            break
        it += 1
        max_val, grad = _conv_max_and_grad(cur, L)
        cur_sc = 2.0 * n * max_val

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        cand = _project_simplex(cur - step * momentum)

        # more frequent symmetrisation helps flatten the autocorrelation
        if it % 100 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        cand_max, _ = _conv_max_and_grad(cand, L)
        cand_sc = 2.0 * n * cand_max

        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_sc = cand_sc
                no_improve = 0
            else:
                no_improve += 1

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        if no_improve > 4000:
            # random jitter to escape flat regions
            sigma = 0.05 / math.sqrt(it)
            pert = cur + sigma * rng.standard_normal(n)
            pert = _project_simplex(np.maximum(pert, 0.0))
            pert_max, _ = _conv_max_and_grad(pert, L)
            pert_sc = 2.0 * n * pert_max
            if pert_sc < cur_sc:
                cur = pert
                cur_sc = pert_sc
                if pert_sc < best_sc:
                    best_sc = pert_sc
                    best_vec = pert.copy()
                no_improve = 0
            else:
                momentum.fill(0.0)
                no_improve = 0

        if best_sc < target:
            break

    # ----- return best found vector ---------------------------------------
    return [float(x) for x in best_vec]
