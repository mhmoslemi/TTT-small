def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1) # leave slack
    # init
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
    else:
        n = 2500
        a = rng.random(n, dtype=np.float64)
    # project onto simplex (sum=1)
    a = _project_simplex(a)
    n = a.size
    L = 1 << ((2*n - 1).bit_length())
    # scoring function
    def score_vec(v):
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return 2.0 * n * conv.max()
    cur_score = score_vec(a)
    best = a.copy()
    best_score = cur_score
    # parameters
    eps = 0.2  # initial max fraction move
    temp0 = 0.01
    it = 0
    # loop
    while time.monotonic() < deadline:
        it += 1
        # adapt epsilon slowly decreasing
        eps = max(1e-4, eps * 0.9999)
        # temperature decays
        temp = max(1e-6, temp0 * (0.9999 ** it))
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # delta is small fraction of a[i]
        max_delta = a[i]
        if max_delta == 0:
            continue
        # delta up to eps * max_delta
        delta = eps * rng.random() * max_delta
        # ensure we don't go negative
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # normalize to sum 1 (should already)
        # but due to rounding may deviate slightly:
        # we can renorm by dividing by sum
        s = a_new.sum()
        if s <= 0:
            continue
        a_new = a_new / s
        new_score = score_vec(a_new)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score)/temp):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best = a_new.copy()
                best_score = new_score
                # early exit if good
                if best_score < 1.50274349:
                    return [float(x) for x in best]
        # periodic symmetrization
        if it % 5000 == 0:
            a = (a + a[::-1]) * 0.5
            a = a / a.sum()
            cur_score = score_vec(a)
    # return best
    return [float(x) for x in best]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a = rng.random(n, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    # optional symmetrization
    a = (a + a[::-1]) * 0.5
    a = a / a.sum()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    def _score_vec(vec):
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        max_val = conv.max()
        sum_val = vec.sum()
        if sum_val == 0:
            return float('inf')
        return float(2.0 * n * max_val / (sum_val * sum_val))
    cur_score = _score_vec(a)
    best = a.copy()
    best_score = cur_score
    # simulated annealing parameters
    temp0 = 0.05
    temp_decay = 0.99995
    eps_move_initial = 0.2
    eps_decay = 0.99999
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # adapt parameters
        temp = max(1e-8, temp0 * (temp_decay ** it))
        eps_move = max(1e-5, eps_move_initial * (eps_decay ** it))
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # delta <= eps_move * a[i]
        delta = eps_move * rng.random() * a[i]
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # re-normalize due to rounding
        a_new = a_new / a_new.sum()
        new_score = _score_vec(a_new)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / temp):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best = a_new.copy()
                best_score = new_score
                if best_score < 1.50274349:
                    break
        # occasional symmetrization
        if it % 5000 == 0:

            if cur_score < best_score:
                best = a.copy()
                best_score = cur_score
                if best_score < 1.50274349:
                    break
    return [float(x) for x in best]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(vec, L):
    """Exact AC1 score for a vector (no extra scaling)."""
    n = vec.shape[0]
    af = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = vec.sum()
    if s == 0:
        return float("inf")
    return float(2.0 * n * max_val / (s * s))
