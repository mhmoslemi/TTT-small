import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast score for a vector already on the simplex (sum≈1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ---- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # ---- Adam optimizer state -------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.8                     # base learning rate

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    temp = None
    decay = 0.997
    min_temp_factor = 1e-4

    it = 0
    no_improve = 0
    target = 1.50274349

    # ---- main optimisation loop -----------------------------------------
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # full autocorrelation
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature for soft‑max
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # soft‑max weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of weighted max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 / math.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + eps)
        cand = _project_simplex(cur - update)

        cand_sc = _score(cand)

        # acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # random redistribution when stuck
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # occasional symmetrisation
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        if best_sc < target:
            break

    # ---- output ---------------------------------------------------------
    return [float(x) for x in best_vec]
