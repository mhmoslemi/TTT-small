def subgradient_maxconv(a):
   n = len(a)
   L = 1 << ((2*n-1).bit_length())
   a_fft = np.fft.rfft(a, n=L)
   conv_full = np.fft.irfft(a_fft * a_fft, n=L)
   conv = conv_full[:2*n-1]
   M = conv.max()
   # mask for entries near M using tolerance to include near-equal maxima
   mask = (conv >= M - 1e-12)  # tolerance maybe due to floating errors
   # treat as probabilities uniform over mask entries
   w = mask.astype(np.float64)
   cnt = w.sum()
   if cnt == 0:
       w = np.ones_like(conv) / conv.size
   else:
       w /= cnt
   # compute gradient
   w_rev = w[::-1]
   w_fft = np.fft.rfft(w_rev, n=L)
   grad_full = np.fft.irfft(a_fft * w_fft, n=L)
   idx = (2*n - 2) - np.arange(n)
   grad = 2.0 * grad_full[idx]
   return grad, M

import time, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size > 0:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w
