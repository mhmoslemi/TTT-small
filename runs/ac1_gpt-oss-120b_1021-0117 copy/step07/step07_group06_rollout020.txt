def _max_conv_and_grad(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[:2*n -1]
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    # gradient
    idx = max_idx - np.arange(n)
    mask = (idx >=0) & (idx < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[idx[mask]]
    return max_val, grad
```

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _max_conv_and_grad(a):
    """Return max autocorrelation value and a subgradient (using the max index)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    idx = max_idx - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[idx[mask]]
    return max_val, grad
