import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_score(a, L=None):
    """Fast evaluation of the AC1 bound when sum(a)=1."""
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _mutate_local(a, rng, delta_factor):
    """Move a random amount from one index to another."""
    n = a.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    if a[i] <= 0.0:
        return a
    delta = a[i] * rng.random() * delta_factor
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    # stay on the simplex (sum unchanged, non‑negative)
    if b[i] < 0.0:
        b = _project_simplex(b)
    return b

def _mutate_swap(a, rng):
    """Swap two entries."""
    n = a.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    b = a.copy()
    b[i], b[j] = b[j], b[i]
    return b

def _mutate_random(n, rng):
    """Draw a fresh random point on the simplex."""
    return _project_simplex(rng.random(n))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # --------------------------------------------------------------
    # Determine problem size and initial points
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    # Helper to normalize a vector onto the simplex
    def _norm(v):
        v = np.maximum(v, 0.0)
        return _project_simplex(v)

    # Candidate pool
    candidates = []

    if "height_sequence_1" in globals():
        a0 = _norm(np.asarray(height_sequence_1, dtype=np.float64).copy())
        candidates.append(a0)

    # Semi‑circular start (often good for autocorrelation)
    x = np.linspace(-1.0, 1.0, n)
    a_semi = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    candidates.append(_norm(a_semi))

    # Pure random start
    candidates.append(_norm(rng.random(n)))

    # Choose the best of the small pool as the starting point
    best = candidates[0]
    best_score = _fft_score(best)
    for cand in candidates[1:]:
        sc = _fft_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    current = best.copy()
    current_score = best_score

    # Pre‑compute FFT length
    L_fft = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Simulated annealing / local search parameters
    total_time = deadline - start
    T0 = 0.05                      # initial temperature
    delta_factor = 0.2            # magnitude of local moves
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # linear temperature schedule
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            break
        T = T0 * (remaining / total_time)

        # pick a mutation
        r = rng.random()
        if r < 0.5:
            cand = _mutate_local(current, rng, delta_factor)
        elif r < 0.8:
            cand = _mutate_swap(current, rng)
        else:
            cand = _mutate_random(n, rng)

        cand = _project_simplex(cand)          # keep on simplex
        cand_score = _fft_score(cand, L_fft)

        # Metropolis acceptance
        if cand_score < current_score or rng.random() < math.exp(-(cand_score - current_score) / (T + 1e-12)):
            current = cand
            current_score = cand_score

        # Update global best
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

        # Shrink move size gradually
        if it % 1000 == 0:
            delta_factor = max(0.01, delta_factor * 0.99)

        # Escape stagnation by a small perturbation around the best point
        if no_improve > 5000:
            noise = rng.normal(scale=delta_factor, size=n)
            pert = _project_simplex(best + noise)
            pert_score = _fft_score(pert, L_fft)
            if pert_score < best_score:
                best = pert
                best_score = pert_score
            current = pert
            current_score = pert_score
            no_improve = 0

        # Early exit if we have reached the target bound
        if best_score < target:
            break

    # --------------------------------------------------------------
    return [float(x) for x in best]
