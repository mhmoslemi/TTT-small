import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # init vector
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    # initial projection onto simplex and enforce symmetry
    cur = _project_simplex(np.maximum(cur, 0.0))
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2   # base learning rate
    beta1_pow = 1.0
    beta2_pow = 1.0

    temp = None
    decay = 0.998
    min_temp_factor = 5e-4

    no_improve = 0
    it = 0
    target = 1.50274349

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature schedule
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # softmax weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        # enforce nonnegativity
        cand = np.maximum(cand, 0.0)
        # enforce symmetry
        cand = (cand + cand[::-1]) * 0.5
        # project onto simplex
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # accept if better
        if cand_sc < best_sc:
            best_vec = cand.copy()
            best_sc = cand_sc

            no_improve = 0

        if no_improve > 2500:

            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = np.maximum(tmp, 0.0)
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < best_sc:
                best_vec = tmp.copy()
                best_sc = tmp_sc
            if tmp_sc < cand_sc:
                cur = tmp
                no_improve = 0

        # occasional full symmetrization (even though we keep symmetry); just re-project
        if it % 500 == 0:
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)

        if best_sc < target:
            break

    return [float(x) for x in best_vec]

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]

        theta = 0.0

    w[w < 0] = 0.0

    eps = 1e-8
    lr0 = 0.2
    beta1_pow, beta2_pow = 1.0, 1.0

    # ----- soft‑max temperature -----------------------------------------
    temp = None
    decay = 0.998
    min_temp_factor = 5e-4

    no_improve = 0
    it = 0
    target = 1.50274349

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # convolution of current iterate
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature for soft‑max weighting
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # soft‑max weights (symmetric)
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of weighted max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = np.maximum(cand, 0.0)
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # keep the best ever
        if cand_sc < best_sc:
            best_vec = cand.copy()
            best_sc = cand_sc

        # accept if improves current iterate
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional uphill move
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc

        # random redistribution when stuck
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = np.maximum(tmp, 0.0)
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0

        # occasional re‑symmetrisation (cheap)
        if it % 500 == 0:
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)

        if best_sc < target:
            break

    return [float(x) for x in best_vec]
