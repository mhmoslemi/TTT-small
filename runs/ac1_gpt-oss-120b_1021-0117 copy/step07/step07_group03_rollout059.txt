def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # initialization
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a0 = rng.random(2000, dtype=np.float64)
    a0 = np.maximum(a0, 0.0)
    cur = _project_simplex(a0)
    n = cur.size
    L = 1 << ((2*n - 1).bit_length())

    # compute initial
    max_val, grad = _conv_max_and_grad(cur, L)
    cur_sc = 2.0 * n * max_val
    best_sc = cur_sc
    best_vec = cur.copy()

    # log probabilities
    eps = 1e-12
    w = np.log(cur + eps)

    # parameters
    init_step = 0.5
    it = 0
    no_improve = 0
    target = 1.50274349

    # main loop
    while time.time() < deadline:
        it += 1
        # compute gradient (recalc conv)
        max_val, grad = _conv_max_and_grad(cur, L)
        cur_sc = 2.0*n*max_val

        # step size
        step = init_step / math.sqrt(it)

        # update log probabilities
        w -= step * grad

        # compute probabilities via softmax
        max_w = np.max(w)
        p = np.exp(w - max_w)
        p_sum = p.sum()
        if p_sum == 0 or not np.isfinite(p_sum):
            # fallback: reinitialize w from cur
            w = np.log(cur + eps)
            continue
        p /= p_sum

        # symmetrize
        a = (p + p[::-1]) * 0.5

        # optional smoothing
        if it % 250 == 0:
            # small smoothing kernel
            kernel = np.array([0.2, 0.6, 0.2], dtype=np.float64)
            a = np.convolve(a, kernel, mode='same')
        # ensure nonnegative
        a = np.maximum(a, 0.0)
        # normalize to simplex
        a_sum = a.sum()
        if a_sum == 0 or not np.isfinite(a_sum):
            a = cur  # fallback to previous
        else:
            a = a / a_sum

        # evaluate new candidate
        max_val_cand, _ = _conv_max_and_grad(a, L)
        cand_sc = 2.0 * n * max_val_cand

        # accept if better, else keep current (no Metropolis)
        if cand_sc < cur_sc:
            cur = a
            cur_sc = cand_sc
            no_improve = 0
        else:
            # maybe accept with probability

            if rng.random() < prob:

                cur_sc = cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # update best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

            pert = cur + sigma * rng.standard_normal(size=n)

                w = np.log(cur + eps)
            no_improve = 0

        # early exit
        if best_sc < target:
            break

    # return best vector as list of floats
    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_grad(a, L):
    """
    Returns (max_convolution, gradient) of the max of a*a (full convolution).
    a is assumed non‑negative; no sum‑to‑one requirement.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad


def _softmax(w):
    """Stable softmax returning a probability vector (sum=1)."""
    w = np.asarray(w, dtype=np.float64)
    m = w.max()
    e = np.exp(w - m)
    s = e.sum()
    if s == 0 or not np.isfinite(s):
        # fallback to uniform
        return np.full_like(w, 1.0 / w.size)
    return e / s

    w = np.log(cur + eps)                     # log‑softmax parameters
    m = np.zeros_like(w)                     # first moment
    v = np.zeros_like(w)                     # second moment
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.25                                 # base learning rate
    it = 0
    no_improve = 0
    target = 1.50274349

    # smoothing kernel for occasional low‑pass filtering
    kernel = np.array([0.2, 0.6, 0.2], dtype=np.float64)

    while time.time() < deadline:
        it += 1

        # ----- gradient of current point ---------------------------------
        max_val, grad = _conv_max_and_grad(cur, L)
        cur_sc = 2.0 * n * max_val

        # chain rule: gradient w.r.t. logits w
        avg = np.dot(cur, grad)
        grad_w = cur * (grad - avg)

        # ----- Adam update -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 / math.sqrt(it)
        w -= step * (m_hat / (np.sqrt(v_hat) + 1e-8))

        # ----- map back to simplex ----------------------------------------
        a = _softmax(w)

        # enforce symmetry (helps lower peaks)
        a = (a + a[::-1]) * 0.5

        # occasional mild smoothing
        if it % 200 == 0:
            a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0 or not np.isfinite(s):
            a = cur.copy()
        else:
            a /= s

        # ----- evaluate candidate -----------------------------------------
        max_val_cand, _ = _conv_max_and_grad(a, L)
        cand_sc = 2.0 * n * max_val_cand

        # accept the new point (always, but keep best separately)
        cur = a
        cur_sc = cand_sc

        # ----- update global best -----------------------------------------
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- stagnation handling ---------------------------------------
        if no_improve > 3000:
            sigma = 0.1 / math.sqrt(it)
            w += sigma * rng.standard_normal(size=n)   # perturb logits
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # ----- early termination -----------------------------------------
        if best_sc < target:
            break

    # return the best sequence found
    return [float(x) for x in best_vec]
