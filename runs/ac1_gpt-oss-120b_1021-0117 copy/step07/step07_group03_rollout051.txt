import time, math, numpy as np

def _project_simplex(v):
    """Project onto simplex {x>=0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _conv_max_and_grad(a, L):
    """
    Compute max of full convolution a*a and gradient of that max.
    a must be non‑negative, but function works for any real array.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    # mask of positions attaining (or close to) the max
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    # gradient via convolution with reversed mask
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

    cur_sc = 2.0 * n * max_val
    best_sc = cur_sc
    best_vec = cur.copy()
    prev = cur.copy()

    # optimisation parameters
    step0 = 0.6          # initial step size factor
    mu = 0.9             # Nesterov momentum coefficient
    no_improve = 0
    target = 1.50274349

    it = 0

        it += 1

        # Nesterov look‑ahead point

    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # ------------------------------------------------------------------
    # initialise a feasible non‑negative vector (sum = 1)
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2000, dtype=np.float64)

    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1

    # initial evaluation
    max_val, _ = _conv_max_and_grad(cur, L)
    cur_sc = 2.0 * n * max_val
    best_sc = cur_sc
    best_vec = cur.copy()
    prev = cur.copy()

    # optimisation hyper‑parameters
    step0 = 0.6          # initial step factor
    mu = 0.9             # Nesterov momentum coefficient
    no_improve = 0
    target = 1.50274349

    it = 0
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # --- Nesterov accelerated gradient step --------------------------------
        y = cur + mu * (cur - prev)          # look‑ahead point
        max_val, grad = _conv_max_and_grad(y, L)
        step = step0 / math.sqrt(it)         # simple decay

        cand = y - step * grad                # gradient descent
        cand = _project_simplex(cand)

        # occasional symmetrisation (helps flatten peaks)
        if it % 150 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        # evaluate candidate
        max_cand, _ = _conv_max_and_grad(cand, L)
        cand_sc = 2.0 * n * max_cand

        # accept if improvement
        if cand_sc < cur_sc:
            prev = cur
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # keep global best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        # random perturbation when stuck
        if no_improve > 4000:
            sigma = 0.1 / math.sqrt(it)
            pert = cur + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(pert)
            max_pert, _ = _conv_max_and_grad(pert, L)
            pert_sc = 2.0 * n * max_pert
            if pert_sc < cur_sc:
                prev = cur
                cur = pert
                cur_sc = pert_sc
                if pert_sc < best_sc:
                    best_sc = pert_sc
                    best_vec = pert.copy()
            no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # return the best vector as plain Python floats
    return [float(x) for x in best_vec]
