    # Standard algorithm
    v = np.asarray(v)
    n = v.shape[0]

def fast_objective(a):
    n = a.shape[0]
    conv = np.fft.ifft(np.fft.fft(a, 2*n-1) * np.fft.fft(a, 2*n-1)).real

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1

        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the probability simplex (sum == 1, >=0)
def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast evaluation of the AC1 upper bound (identical to evaluate_sequence
# but using FFT which is O(n log n) instead of the O(n²) np.convolve)
def fast_objective(seq):
    n = seq.shape[0]
    if n == 0:
        return float("inf")
    # full convolution via FFT
    L = 2 * n - 1
    f = np.fft.fft(seq, L)
    conv = np.fft.ifft(f * f).real
    M = np.max(conv)
    S = np.sum(seq)
    if S == 0.0:
        return float("inf")
    return 2.0 * n * M / (S * S)

        arr = sin_vals ** q
        arr_sum = np.sum(arr)
        if arr_sum == 0.0:
            continue
        cand = arr / arr_sum
        sc = fast_objective(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # 4. Symmetric triangular shape raised to a power
    tri = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1)).astype(np.float64)
    for p in np.linspace(-5.0, 5.0, 21):
        if time.monotonic() >= deadline:
            break
        arr = tri ** p
        arr_sum = np.sum(arr)
        if arr_sum == 0.0:
            continue
        cand = arr / arr_sum
        sc = fast_objective(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # ------------------- projected gradient descent ----------------
    max_gd_iters = 200
    eta = 0.5                      # initial step size
    for _ in range(max_gd_iters):
        if time.monotonic() >= deadline:
            break
        # convolution and active set
        L = 2 * n - 1
        f_fft = np.fft.fft(best_seq, L)
        conv = np.fft.ifft(f_fft * f_fft).real
        M = np.max(conv)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]
        if active.size > 30:
            # keep a small representative subset
            idx = np.linspace(0, active.size - 1, 30, dtype=int)
            active = active[idx]

        # gradient of the maximal convolution entry
        grad = np.zeros_like(best_seq)
        for k in active:
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            i_range = np.arange(i_min, i_max + 1)
            j_range = k - i_range
            grad[i_range] += 2.0 * best_seq[j_range]

        # simple back‑tracking line search
        step = eta
        improved = False
        while step > 1e-6:
            cand = best_seq - step * grad
            cand = project_simplex(cand)
            sc = fast_objective(cand)
            if sc < best_score:
                best_seq = cand
                best_score = sc
                improved = True
                break
            step *= 0.5
        if not improved:
            eta *= 0.5
            if eta < 1e-4:
                break

    # ------------------- random hill‑climbing --------------------
    while time.monotonic() < deadline:
        # generate a Dirichlet‑like noise vector and mix
        noise = rng.exponential(scale=1.0, size=n)
        alpha = 0.2
        cand = (1.0 - alpha) * best_seq + alpha * (noise / np.sum(noise))
        cand = project_simplex(cand)
        sc = fast_objective(cand)
        if sc < best_score:
            best_seq = cand
            best_score = sc

    # final safety checks
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if np.sum(best_seq) < 1e-12:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_seq]
