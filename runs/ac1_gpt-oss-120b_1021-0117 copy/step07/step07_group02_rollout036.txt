def _score(a):
    n = len(a)
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n - 1]
    return 2.0 * n * conv.max()
```

def _grad_from_conv(a, m):
    n = a.shape[0]
    # compute gradient g of conv[m] w.r.t a
    # g[i] = 2 * a[m - i] if 0 <= m - i < n else 0
    g = np.zeros_like(a)
    # valid range i in [max(0, m-(n-1)), min(n-1, m)]
    i_start = max(0, m - (n-1))
    i_end = min(n-1, m)
    # number of elements = i_end - i_start + 1
    # for i in range(i_start, i_end+1):
    #   g[i] = 2 * a[m - i]
    # we can vectorize: slice a_rev = a[m - i_start : m - i_end - 1 : -1]
    rev = a[m - i_start : m - i_end - 1 : -1]  # should be length (i_end - i_start + 1)
    g[i_start:i_end+1] = 2 * rev
    return g
```

        w = np.exp((conv - max_val) / temp)

        else:
            w.fill(0.0)
        w_rev = w[::-1]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    """Score assuming sum(a)==1 (denominator becomes 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score_vec(best_vec)
    cur_sc = best_sc

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # soft‑max temperature parameters
    temp_factor = 0.1          # relative to current max value
    temp_decay = 0.995
    min_temp_factor = 1e-4

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weighting for a smooth gradient of the max‑convolution
        temp = max_val * max(temp_factor, min_temp_factor)
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(0.0)

        # gradient of the weighted max‑convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
