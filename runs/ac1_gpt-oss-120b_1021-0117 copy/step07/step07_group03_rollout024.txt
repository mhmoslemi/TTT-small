def _simplex(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0:
        # fallback uniform
        v = np.full_like(v, 1.0 / v.size)
    else:
        v /= s
    return v

import time, math, numpy as np

# ----------------------------------------------------------------------
def _simplex(v):
    """Project onto the probability simplex by clipping negatives and renormalising."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def _conv_max_and_grad(a, L):
    """
    Compute (max convolution value, subgradient) of a*a.
    a is a non‑negative vector (not necessarily normalised).
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return float(max_val), grad
