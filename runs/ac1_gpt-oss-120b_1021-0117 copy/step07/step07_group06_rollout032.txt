import time, math, numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _max_convolution(v):
    """Maximum entry of the self‑convolution of v (full, length 2n‑1)."""
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
    return conv.max()

def _score(v):
    """Score used by the evaluator when sum(v)=1."""
    n = v.shape[0]
    return float(2.0 * n * _max_convolution(v))

def _lp_step(f, M):
    """Solve  max sum(g)  s.t. f * g <= M, g>=0.  Returns a feasible g (or None)."""
    n = f.size
    # Toeplitz matrix where each row is a shifted copy of f (lower‑triangular)
    col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
    A = toeplitz(col, row)                     # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise sum(g)  => minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # HiGHS is fast and accepts dense matrices
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a list of non‑negative coefficients that (empirically) improve
    upon the previous best AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        f = np.asarray(height_sequence_1, dtype=np.float64).copy()
        f = np.maximum(f, 0.0)
    else:
        # fallback: uniform vector of length ~2000
        n0 = 2000
        f = np.full(n0, 1.0 / n0, dtype=np.float64)

    f = _project_simplex(f)            # enforce sum = 1, non‑negative
    n = f.size

    best_vec = f.copy()
    best_sc = _score(best_vec)

    # ---- iterative improvement -----------------------------------------
    it = 0
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # current max convolution bound
        M = _max_convolution(f)

        # linear‑programming step
        g = _lp_step(f, M)
        if g is None:
            # LP failed – fall back to a small random perturbation
            delta = rng.random(n) * 1e-3
            g = f + delta
            g = np.maximum(g, 0.0)

        # normalise g to the simplex (keeps the scale invariant)
        g = _project_simplex(g)

        # mixing coefficient – decreasing schedule
        alpha = 0.5 / math.sqrt(it)          # between ~0.5 and ~0.01
        f_candidate = (1.0 - alpha) * f + alpha * g
        f_candidate = _project_simplex(f_candidate)

        cand_sc = _score(f_candidate)

        # accept if improvement; otherwise keep current f
        if cand_sc < best_sc:
            best_vec, best_sc = f_candidate.copy(), cand_sc
        if cand_sc < _score(f):
            f = f_candidate

        # occasional symmetrisation (helps many instances)
        if it % 8 == 0:
            sym = (f + f[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_vec, best_sc = sym.copy(), sym_sc
            if sym_sc < _score(f):
                f = sym

        # early exit if target reached
        if best_sc < 1.50274349:
            break

    # ----------------------------------------------------------------------
    # Return best vector found as plain Python floats
    return [float(x) for x in best_vec]
