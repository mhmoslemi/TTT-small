def _score_simple(a):

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_simple(a):
    """Score assuming sum(a)==1 (denominator becomes 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ----- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random vector of length ~2500
        a = rng.random(2500, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # next power of two for FFT

    best = a.copy()
    best_score = _score_simple(best)

    # optimizer state
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 2.0          # initial step magnitude
    it = 0
    no_improve = 0
    target = 1.50274349

    # ----- main gradient‑descent loop ------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution a*a
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        max_idx = int(conv_full.argmax())
        cur_score = 2.0 * n * max_val

        # subgradient of the max‑convolution
        idxs = max_idx - np.arange(n)
        mask = (idxs >= 0) & (idxs < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idxs[mask]]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # step size (diminishing)
        step = step0 / math.sqrt(it)

        # back‑tracking line search (max 8 tries)
        for _ in range(8):
            cand = a - step * momentum
            cand = _project_simplex(cand)
            # force symmetry (helps the problem)
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)
            cand_score = _score_simple(cand)
            if cand_score < cur_score:
                break
            step *= 0.5

        # accept if improvement
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            no_improve += 1

        # random redistribution when stuck
        if no_improve > 2500:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                a = _project_simplex(tmp)
            else:
                a = _project_simplex(rng.random(n))
            momentum.fill(0.0)
            no_improve = 0

        # early exit if target reached
        if best_score < target:
            break

        # occasional time check
        if it % 5000 == 0 and time.monotonic() >= deadline:
            break

    # ----- fine‑grained random hill climbing ------------------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if best[i] <= 0.0:
            continue
        delta = best[i] * rng.random() * 0.2
        j = rng.integers(0, n - 1)
        if j >= i:

        trial = best.copy()

        trial = _project_simplex(trial)

        trial = _project_simplex(trial)
        trial_score = _score_simple(trial)
        if trial_score < best_score:
            best = trial
            best_score = trial_score
            if best_score < target:
                break

    # return as plain Python floats
    return [float(x) for x in best]
