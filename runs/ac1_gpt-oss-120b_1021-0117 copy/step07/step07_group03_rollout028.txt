import time, math, numpy as np

# --------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise for safety (numerical errors)
    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w

# --------------------------------------------------------------
def _conv_max_and_grad(a, L):
    """
    Returns (max_convolution, gradient) of the max value of a*a (full convolution).
    a is assumed non‑negative; the function works for any a.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt

    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)

    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

def _make_window(n, kind):
    """Generate a few classical non‑negative windows and project them onto the simplex."""
    t = np.arange(n, dtype=np.float64)
    if kind == "hann":
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * t / (n - 1))
    elif kind == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * t / (n - 1))
    elif kind == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * t / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * t / (n - 1))
        )
    elif kind == "cosine":
        w = np.sin(np.pi * (t + 0.5) / n)
    else:  # fallback uniform
        w = np.ones(n, dtype=np.float64)
    w = np.maximum(w, 0.0)
    return _project_simplex(w)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin for returning the answer
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ----------------------------------------------------------
    # initialise from the parent if present, otherwise random
    if "height_sequence_1" in globals():

            best_vec = w.copy()
    # also keep the parent‑derived vector as a candidate
    max_val, _ = _conv_max_and_grad(a, L)
    sc = 2.0 * n * max_val
    if sc < best_sc:

        best_vec = a.copy()

    # ----------------------------------------------------------
    # optimisation state
    cur = best_vec.copy()            # start from the best of the cheap candidates
    cur_sc = best_sc
    momentum = np.zeros_like(cur)
    beta = 0.95          # momentum decay
    step0 = 0.7          # initial step magnitude
    it = 0

    target = 1.50274349

    # ----------------------------------------------------------
    while True:

        it += 1

        max_val, grad = _conv_max_and_grad(cur, L)
        cur_sc = 2.0 * n * max_val

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size – slowly decaying
        step = step0 / max(1.0, math.sqrt(it))

        # candidate move

                if pert_sc < best_sc:
                    best_sc = pert_sc
                    best_vec = pert.copy()
                no_improve = 0
            else:

                momentum.fill(0.0)
                no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

    # ----------------------------------------------------------
    # return the best vector discovered (as plain Python floats)
    return [float(x) for x in best_vec]
