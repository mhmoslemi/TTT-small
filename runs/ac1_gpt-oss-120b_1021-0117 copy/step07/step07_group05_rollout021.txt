        # v is 1d numpy array
        # Implementation from Duchi et al.
        u = np.sort(v)[::-1]

        else:

            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound as measured by ``evaluate_sequence``.
    The method combines a few cheap parametric initialisations with a
    projected sub‑gradient descent on the convex objective
        f(a) = 2·n·max(conv(a,a)) / (sum(a)²) .
    All work is O(n) or O(n log n) per iteration and respects the time
    budget.
    """
    start = time.monotonic()
    # give a tiny safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # --------------------------------------------------------------
    # problem size and a simple initial vector
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        init_parent = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 2000
        init_parent = None

    # --------------------------------------------------------------
    # Euclidean projection onto the probability simplex {x≥0, sum(x)=1}
    def _project_simplex(v: np.ndarray) -> np.ndarray:
        """Duchi et al. projection, linear‐time after sorting."""
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        ind = np.arange(1, v.size + 1)
        rho = np.where(u - css / ind > 0)[0]
        if rho.size == 0:         # all entries zero – return uniform
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    # --------------------------------------------------------------
    # objective computed directly (identical to evaluate_sequence for
    # non‑negative entries ≤1000)
    def _objective(a: np.ndarray) -> float:
        s = a.sum()
        if s < 1e-12:
            return np.inf
        c = np.convolve(a, a)
        return 2.0 * n * c.max() / (s * s)

    # --------------------------------------------------------------
    # keep the best candidate seen so far
    best_vec = None
    best_score = np.inf

    # uniform starter (sum = 1)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    score = _objective(uniform)
    if score < best_score:
        best_score = score
        best_vec = uniform.copy()

    # parent if supplied (clip then normalise to sum=1)
    if init_parent is not None:
        parent = np.clip(init_parent, 0.0, 1000.0)
        s = parent.sum()
        if s > 0:
            parent = parent / s
        score = _objective(parent)
        if score < best_score:
            best_score = score
            best_vec = parent.copy()

    # --------------------------------------------------------------
    # a few exponential families – a_i ∝ exp(-λ·i)
    # λ is scaled by 1/n so that the shape changes smoothly with n.
    for t in np.linspace(0.0, 10.0, 21):          # 21 candidates
        lam = t / n
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a /= a.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_vec = a.copy()
        if time.monotonic() >= deadline:
            break

    # --------------------------------------------------------------
    # Projected sub‑gradient descent starting from the best vector so far.
    rng = np.random.default_rng(seed)
    a = best_vec.copy()
    cur_score = best_score
    step0 = 1.0                         # initial step magnitude
    max_iters = 4000
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        # convolution and current objective
        c = np.convolve(a, a)
        max_b = c.max()
        cur_score = 2.0 * n * max_b / (a.sum() * a.sum())

        # sub‑gradient at a maximising index
        k = int(np.argmax(c))
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.intp)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        # diminishing step size
        step = step0 / np.sqrt(it)

        # simple back‑tracking line search (max 5 reductions)
        for _ in range(5):
            a_new = _project_simplex(a - step * g)
            new_score = _objective(a_new)
            if new_score <= cur_score:
                a = a_new
                cur_score = new_score
                break
            step *= 0.5
        else:
            # no improvement – inject a small random perturbation
            perturb = rng.normal(scale=0.01, size=n)
            a = _project_simplex(a + perturb)

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # occasional random restart if stuck
        if it % 800 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            a = _project_simplex(best_vec + jitter)

    # --------------------------------------------------------------
    # Final safety – clip to the evaluator's allowed range
    final = np.clip(best_vec, 0.0, 1000.0)

    # Convert to plain Python list of floats
    return [float(x) for x in final]
