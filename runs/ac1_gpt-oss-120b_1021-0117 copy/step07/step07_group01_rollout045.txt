import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    # sort in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a, L):
    """Score for a vector already normalized to sum=1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.4)

    # -------------------- initialization --------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)               # enforce sum==1, non‑negative
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two for FFT efficiency

    # Adam‑style moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # bookkeeping
    best = a.copy()
    best_sc = _score(best, L)
    it = 0
    no_improve = 0
    step0 = 0.6                           # initial step magnitude
    target = 1.50274349

    # -------------------- main loop --------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ---------- convolution and max ----------
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ---------- sub‑gradient ----------
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # gradient of weighted max‑convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # adaptive step that decays like 1/sqrt(it)
        step = step0 / (math.sqrt(it) + 1.0)

        # additive candidate
        cand_add = _project_simplex(a - step * m_hat / (np.sqrt(v_hat) + eps))
        # multiplicative (exponential) candidate
        cand_mul = a * np.exp(-step * grad)
        s = cand_mul.sum()
        cand_mul = cand_mul / s if s > 0 else a.copy()

        # evaluate
        sc_add = _score(cand_add, L)
        sc_mul = _score(cand_mul, L)
        if sc_add < sc_mul:
            cand, cand_sc = cand_add, sc_add
        else:
            cand, cand_sc = cand_mul, sc_mul

        # ---------- acceptance ----------
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            a = cand
            no_improve = 0
        elif cand_sc < _score(a, L):
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            # occasional probabilistic jump
            prob = math.exp(-(cand_sc - best_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = cand

        # ---------- periodic symmetrisation ----------
        if it % 500 == 0:

            sym = _project_simplex(sym)
            sc_sym = _score(sym, L)
            if sc_sym < best_sc:
                best = sym.copy()
                best_sc = sc_sym
                a = sym
                no_improve = 0

        # ---------- random mass redistribution when stuck ----------
        if no_improve > 3000:

            if a[i] > 0:

                j = rng.integers(0, n - 1)
                if j >= i:

                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                a = _project_simplex(tmp)
            no_improve = 0

        # ---------- restart with perturbation ----------
        if no_improve > 6000:

            rand_vec = _project_simplex(rand_vec)

            a = _project_simplex(a)
            no_improve = 0

        # ---------- early exit if target reached ----------

    # ----------------------------------------------------
