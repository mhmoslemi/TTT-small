import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 0.8                     # base learning rate for Adam
    target_score = 1.50274349       # early‑stop threshold
    max_restarts = 3                # how many independent starts
    best_vec = None
    best_sc = float("inf")

    # ------------------------------------------------------------------
    for restart in range(max_restarts):
        if time.monotonic() >= deadline:
            break

        # initial vector
        if restart == 0 and "height_sequence_1" in globals():
            cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
        else:
            cur = _project_simplex(rng.random(n))

        m = np.zeros_like(cur)
        v = np.zeros_like(cur)

        cur_sc = _score(cur)
        if cur_sc < best_sc:
            best_vec, best_sc = cur.copy(), cur_sc

        # temperature for the soft‑max approximation of the max
        temp = 0.1
        temp_factor = 1.015
        temp_max = 5000.0

        it = 0
        no_improve = 0

        while True:
            if time.monotonic() >= deadline:
                break
            it += 1

            # ----- convolution of the current vector with itself -----
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
            max_val = conv_full.max()

            # ----- soft‑max weights approximating the max -----
            diff = conv_full - max_val                     # ≤ 0
            w_exp = np.exp(temp * diff)                    # max term → 1
            w_sum = w_exp.sum()
            if w_sum == 0.0 or np.isnan(w_sum):
                w = np.full_like(conv_full, 1.0 / conv_full.size)
            else:
                w = w_exp / w_sum

            # ----- gradient of the weighted sum -----
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = step0 / (np.sqrt(v_hat) + eps)
            cand = cur - step * m_hat
            cand = _project_simplex(cand)

            cand_sc = _score(cand)

            # ----- acceptance -----
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
                if cand_sc < best_sc:
                    best_vec, best_sc = cand.copy(), cand_sc
            else:
                no_improve += 1
                if rng.random() < 0.001:          # tiny chance to escape
                    cur, cur_sc = cand, cand_sc

            # ----- increase temperature (sharper max) -----
            temp = min(temp_max, temp * temp_factor)

            # ----- occasional symmetrisation (helps structure) -----
            if it % 400 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    no_improve = 0
                    if sym_sc < best_sc:
                        best_vec, best_sc = sym.copy(), sym_sc

            # ----- random redistribution when stuck -----
            if no_improve > 2500:
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.3
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        no_improve = 0
                        if tmp_sc < best_sc:
                            best_vec, best_sc = tmp.copy(), tmp_sc

            # ----- early exit if the target is reached -----
            if best_sc <= target_score:
                break

        # end of this restart
    # ------------------------------------------------------------------

    # ----- fine‑grained local search (tiny mass swaps) -----
    if best_vec is None:
        best_vec = np.ones(n) / n
        best_sc = _score(best_vec)

    cur = best_vec.copy()
    it_local = 0
    while time.monotonic() + 0.05 < deadline:
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = cur[i] * rng.random() * 0.02   # move up to 2 % of the source weight
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        if tmp[i] < 0.0:
            continue
        tmp_sc = _score(tmp)
        if tmp_sc < best_sc:
            cur = tmp
            best_sc = tmp_sc
            best_vec = tmp.copy()
        it_local += 1
        if it_local % 500 == 0 and time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # return the best vector found as a plain Python list
    return [float(x) for x in best_vec]
```
