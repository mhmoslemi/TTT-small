    Return (max convolution value, subgradient) of a*a.
    a must be a non‑negative vector (not necessarily normalized).
    """

    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]                     # linear convolution
    max_val = conv.max()
    # mask of entries attaining (or almost attaining) the maximum
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt                                   # average over all active peaks
    # gradient is 2 * (a * (mask reversed)) restricted to original length
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    # indices that map back to the original vector
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad


def _next_power_of_two(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a list of non‑negative coefficients that (hopefully) improve
    the AC1 upper bound.  The routine respects the time budget, keeps the
    best vector found so far, and falls back to it before the timeout.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # small safety margin

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # random positive start if the parent vector is absent
        a0 = rng.random(2000, dtype=np.float64) + 1e-3

    # project onto the probability simplex (sum = 1)
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total == 0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / total

    a = a0.copy()
    n = a.size
    L = _next_power_of_two(2 * n - 1)

    # best solution tracking
    max_val, grad = _conv_max_and_grad(a, L)
    cur_score = 2.0 * n * max_val          # evaluator value for sum=1
    best_score = cur_score
    best_vec = a.copy()

    # algorithm parameters
    step0 = 0.5
    beta = 0.9          # averaging factor for gradient (momentum‑like)
    grad_avg = np.zeros_like(a)
    it = 0
    no_improve = 0
    target = 1.50274349

    # -------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # current gradient
        max_val, grad = _conv_max_and_grad(a, L)
        cur_score = 2.0 * n * max_val

        # update moving average of the subgradient
        grad_avg = beta * grad_avg + (1.0 - beta) * grad

        # step size schedule
        step = step0 / math.sqrt(it)

        # ----- exponentiated (mirror) update --------------------------------
        # move opposite to the averaged gradient using multiplicative updates
        mul = np.exp(-step * grad_avg)
        # avoid exact zeros which would break normalization
        mul = np.where(mul <= 0, 1e-300, mul)
        a_new = a * mul
        s = a_new.sum()
        if s == 0:
            a_new = a.copy()
        else:
            a_new = a_new / s

        # evaluate candidate
        max_val_new, _ = _conv_max_and_grad(a_new, L)
        new_score = 2.0 * n * max_val_new

        # acceptance rule (deterministic improvement + occasional uphill move)
        if new_score < cur_score:
            a, cur_score = a_new, new_score
            no_improve = 0
        else:
            prob = math.exp(-(new_score - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                a, cur_score = a_new, new_score
                no_improve = 0
            else:
                no_improve += 1

        # track the best vector seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            no_improve = 0

        # ---- periodic symmetrisation (helps reduce peaks) ----
        if it % 500 == 0:
            a = (a + a[::-1]) * 0.5
            a = a / a.sum()
            # recompute score after symmetrisation
            max_val, _ = _conv_max_and_grad(a, L)
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_vec = a.copy()
                no_improve = 0

        # ---- random perturbation when stuck --------------------
        if no_improve > 5000:
            sigma = 0.02 / math.sqrt(it)
            pert = a + sigma * rng.standard_normal(size=n)
            pert = np.maximum(pert, 0.0)
            pert = pert / pert.sum()
            max_val_pert, _ = _conv_max_and_grad(pert, L)
            pert_score = 2.0 * n * max_val_pert
            if pert_score < cur_score:
                a, cur_score = pert, pert_score
                no_improve = 0
                if pert_score < best_score:
                    best_score = pert_score
                    best_vec = pert.copy()
            else:
                # fresh random restart while keeping the best so far
                a = rng.random(n, dtype=np.float64)
                a = a / a.sum()
                no_improve = 0

        # early exit if target reached
        if best_score < target:
            break

    # return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
