def _project_simplex(v):
    """Project onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # convolution and max for current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_sc = 2.0 * n * max_val

        # ------------------- temperature for soft‑max ------------------
        if temp is None:
            temp = max_val * 0.05
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # ------------------- soft‑max weights ------------------------
        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:                     # fallback to hard max
            w = (conv_full == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ------------------- gradient of weighted max ----------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ------------------- Adam update -----------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 / (math.sqrt(it) + 1.0)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - update)

        # ----------- evaluate candidate (full convolution) ----------
        cand_fft = np.fft.rfft(cand, n=L)
        conv_cand = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_sc = 2.0 * n * conv_cand.max()

        # ------------------- keep best -------------------------------
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

        # ------------------- accept / anneal -------------------------
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_sc = cand_sc

        # --------------- occasional random redistribution ------------
        if no_improve > 2500:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_fft = np.fft.rfft(tmp, n=L)
            conv_tmp = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
            tmp_sc = 2.0 * n * conv_tmp.max()
            if tmp_sc < cur_sc:
                cur = tmp
                cur_sc = tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()

        # ------------------- occasional symmetrisation ---------------
        if it % 600 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            conv_sym = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * conv_sym.max()
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()

        # ------------------- early exit ------------------------------
        if best_sc < target:
            break

    # final normalisation (sum should already be ~1)
    total = best_vec.sum()
    if total == 0:
        total = 1.0
    best_vec = np.maximum(best_vec / total, 0.0)

    return [float(x) for x in best_vec]
