import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a):
    """Score = 2 * n * max(convolution(a, a)).  Assumes sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) that
    minimizes the AC1 upper bound under the given time budget."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fall back to a modest random start if no parent is supplied
        a = rng.random(2000, dtype=np.float64) + 1e-3
    a = _project_simplex(np.maximum(a, 0.0))

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = a.copy()
    best_sc = _score(best_vec)

    # Adam optimizer state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    eps = 1e-8
    lr = 0.05          # base learning rate

    # bookkeeping
    t = 0
    target = 1.50274349
    sym_period = 300      # how often to enforce symmetry
    jitter_period = 600   # how often to add a small random perturbation
    jitter_scale = 0.02   # magnitude of jitter (relative to simplex)

    # current score (avoid recomputation)
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    a_sc = float(2.0 * n * conv.max())

    while True:
        if time.monotonic() >= deadline:
            break
        t += 1

        # --------------------------------------------------------------
        # recompute FFT of current vector (needed each iteration)
        a_fft = np.fft.rfft(a, n=L)

        # full autocorrelation
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        a_sc = float(2.0 * n * max_val)

        # --------------------------------------------------------------
        # subgradient of the max‑convolution:
        # uniform average over all lags attaining (or almost attaining) the max
        tol = 1e-12 * max_val + 1e-12
        mask = (conv >= max_val - tol).astype(np.float64)
        cnt = mask.sum()
        if cnt == 0.0:               # extremely unlikely, fallback to single max
            mask = np.zeros_like(conv)
            mask[conv.argmax()] = 1.0
            cnt = 1.0
        w = mask / cnt                # convex combination of active subgradients

        # gradient via convolution with the reversed mask (matrix‑free)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)   # mapping to size‑n gradient
        grad = 2.0 * grad_full[idx]

        # --------------------------------------------------------------
        # Adam update (projected onto simplex after the step)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # accept improvement for the current iterate
        if cand_sc < a_sc:
            a = cand
            a_sc = cand_sc

        # update global best
        if cand_sc < best_sc:
            best_vec = cand.copy()
            best_sc = cand_sc

        # --------------------------------------------------------------
        # periodic symmetrisation (helps many instances)
        if t % sym_period == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < a_sc:
                a = sym
                a_sc = sym_sc
            if sym_sc < best_sc:
                best_vec = sym.copy()
                best_sc = sym_sc

        # --------------------------------------------------------------
        # occasional jitter to escape plateaus
        if t % jitter_period == 0:
            noise = rng.normal(scale=jitter_scale, size=n)
            tmp = a + noise
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < a_sc:
                a = tmp
                a_sc = tmp_sc
            if tmp_sc < best_sc:
                best_vec = tmp.copy()
                best_sc = tmp_sc

        # --------------------------------------------------------------
        # early exit if target is reached
        if best_sc < target:
            break

        # safeguard: ensure we stop before the hard timeout
        if time.monotonic() >= deadline:
            break

    # Return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
