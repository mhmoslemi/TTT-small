def propose_candidate(seed=42,budget_s=1100,**kwargs):
    start = time.monotonic()
    deadline = start+budget_s-0.3
    rng = np.random.default_rng(seed)
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        cur = _project_simplex(np.maximum(a0,0.0))
    else:
        n = 2500
        cur = _project_simplex(rng.random(n))
    n = cur.size
    L = 1 << ((2*n-1).bit_length())
    best_vec = cur.copy()
    # compute initial score
    conv_full = np.fft.irfft(np.fft.rfft(cur, n=L) * np.fft.rfft(cur, n=L), n=L)
    conv = conv_full[:2*n-1]
    best_sc = float(2.0 * n * conv.max())
    best_vec = cur.copy()

    # parameter grid search for better init
    # compute for each param candidate
    def evaluate(a):
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())

    # generate candidate set
    p_vals = np.linspace(0.1, 5.0, 20)
    for p in p_vals:
        a = np.arange(n, dtype=np.float64) + 1.0
        a = a ** (-p)
        a = _project_simplex(a)
        sc = evaluate(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
        # exponential
        a2 = np.exp(-p * np.arange(n, dtype=np.float64))
        a2 = _project_simplex(a2)
        sc2 = evaluate(a2)
        if sc2 < best_sc:
            best_sc, best_vec = sc2, a2.copy()
    # also try random
    for _ in range(5):
        a = rng.random(n)
        a = _project_simplex(a)
        sc = evaluate(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    cur = best_vec.copy()
    cur_sc = best_sc
    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.1
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # compute conv and gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # mask for max positions
        eps = 1e-12 * max_val
        mask = (conv >= max_val - eps)
        # compute gradient via FFT correlation
        mask_f = mask.astype(np.float64)
        fft_mask = np.fft.rfft(mask_f, n=L)
        rev_cur = cur[::-1]
        fft_rev = np.fft.rfft(rev_cur, n=L)
        corr_full = np.fft.irfft(fft_mask * fft_rev, n=L)
        grad = 2.0 * corr_full[n-1:2*n-1]
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        # step size schedule
        step = step0 / (np.sqrt(it) + 1.0)
        cand = _project_simplex(cur - step * momentum)
        # evaluate candidate
        cand_sc = evaluate(cand)
        # accept if improvement
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            # occasionally accept probabilistically
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc
        # symmetrize periodically
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = evaluate(sym)
            if sym_sc < cur_sc:

import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _max_conv(a, L):
    """Return max_{k} (a*a)(k) using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv

    def _score(a):

        return 2.0 * n * maxv

    best_vec = cur.copy()
