import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        v = v.ravel()
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: ensure sum is exactly 1
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(w, 1.0 / n)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as measured by ``evaluate_sequence``.
    """
    start = time.monotonic()
    # keep a tiny safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ----------------------------------------------------------------------
    # Problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000  # fallback length

    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Initial candidates
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_vec = uniform.copy()
    try:
        best_score = evaluate_sequence(best_vec.tolist())
    except Exception:
        best_score = float("inf")

    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() < 1e-12:
            parent = uniform.copy()
        else:
            # project onto simplex (preserves sum=1)
            parent = _project_simplex(parent)
        try:
            parent_score = evaluate_sequence(parent.tolist())
            if parent_score < best_score:
                best_score = parent_score
                best_vec = parent.copy()
        except Exception:
            pass

    # ----------------------------------------------------------------------
    # Current working point
    cur = best_vec.copy()
    cur_score = best_score

    # Parameters for the smoothed objective
    beta = 0.5           # start smoothing parameter
    beta_max = 120.0
    eta0 = 0.25          # base step size
    iteration = 0

    # Main optimisation loop
    while time.monotonic() < deadline:
        iteration += 1

        # ----- convolution of current point
        conv_cur = np.convolve(cur, cur, mode='full')
        c_max = conv_cur.max()

        # ----- soft‑max weights (stable)
        exp_arg = beta * (conv_cur - c_max)      # non‑positive
        w = np.exp(exp_arg)
        w_sum = w.sum()

        else:
            w /= w_sum

        # ----- gradient of the smoothed objective (up to a constant factor)
        rev = cur[::-1]
        conv2 = np.convolve(w, rev, mode='full')
        grad = 2.0 * conv2[n - 1:n - 1 + n]      # length n

        # ----- step size (diminishing)
        eta = eta0 / np.sqrt(iteration)

        cand = cur - eta * grad
        cand = _project_simplex(cand)

        except Exception:

            best_vec = cand.copy()
            no_improve = 0

                if explore_score < cur_score:

                    cur_score = explore_score
                    no_improve = 0

        # increase smoothing gradually
        beta = min(beta * 1.025, beta_max)

        # safety time check
        if time.monotonic() >= deadline:
            break

    # Return the best sequence found (as plain Python list)
    return [float(x) for x in best_vec]
