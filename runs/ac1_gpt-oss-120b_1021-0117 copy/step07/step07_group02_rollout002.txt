def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate random vector of length 2500
        a0 = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.shape[0]
    best_vec = cur.copy()
    best_sc = _score_vec(best_vec)
    cur_sc = best_sc
    # optimization parameters
    step0 = 0.1   # base step
    beta = 0.9
    momentum = np.zeros_like(cur)
    it = 0
    no_improve = 0
    target = 1.50274349
    while time.monotonic() < deadline:
        it += 1
        # compute gradient of max conv
        L = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        M = conv_full.max()
        k = int(conv_full.argmax())
        # gradient of M: dM/da = 2 * a[abs(k - i)]
        i = np.arange(n)
        gradM = 2.0 * cur[np.abs(k - i)]
        # gradient of objective (since sum=1) = 2*n * gradM
        grad = 2.0 * n * gradM
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score_vec(cand)
        # accept if better
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            # occasional random perturb
            if no_improve % 500 == 0:
                # small random redistribution
                idx1, idx2 = rng.choice(n, size=2, replace=False)
                delta = cur[idx1] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[idx1] -= delta
                tmp[idx2] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_vec(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
        # occasional symmetrization
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc

                    best_sc, best_vec = sym_sc, sym.copy()
        # early exit
        if best_sc < target:
            break
        # time check
        if it % 1000 == 0 and time.monotonic() >= deadline:
            break
    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    """Score assuming sum(a)==1 (denominator becomes 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    it = 0
