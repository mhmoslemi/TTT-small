import time, math, numpy as np
import cvxpy as cp

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    """Fast evaluation for a vector already normalized to sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _conv_fft(a, L):
    """Return full convolution (length 2n-1) via FFT."""
    af = np.fft.rfft(a, n=L)
    full = np.fft.irfft(af * af, n=L)
    return full

def _grad_max(a, conv, max_val, L):
    """Subgradient of max(conv) w.r.t. a using binary mask."""
    # binary mask for all maximal entries (tolerance for float)
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)
    # reverse mask for convolution trick
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

def _solve_lp(cur, bound, t_mix=0.2):
    """
    Solve the LP:
        max   sum(g)
        s.t.  conv(cur, g) <= bound, g >= 0
    Returns a new vector after mixing with current one,
    already normalized to sum=1. Returns None on failure.
    """
    n = cur.shape[0]
    g = cp.Variable(n, nonneg=True)
    # conv(cur, g) yields length 2n-1
    constraints = [cp.conv(cp.Constant(cur), g) <= bound]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=3000,
                   abstol=1e-8, reltol=1e-6, feastol=1e-8)
    except Exception:
        return None
    if g.value is None:
        return None
    g0 = np.maximum(g.value, 0.0)
    s = g0.sum()
    if s <= 0:
        return None
    g_norm = g0 / s               # keep sum = 1
    new_vec = (1.0 - t_mix) * cur + t_mix * g_norm
    # convex combination of two simplex vectors stays on simplex
    return new_vec

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ---- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)
    beta = 0.92          # momentum decay
    step0 = 1.0          # initial step magnitude
    it = 0
    no_improve = 0
    target = 1.50274349

    # ---- main loop ------------------------------------------------------
    while time.time() < deadline:
        it += 1
        # convolution and max value
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # gradient of the (binary) max
        grad = _grad_max(cur, conv, max_val, L)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule
        base_step = step0 / math.sqrt(it)

        # backtracking line search (up to 5 reductions)
        step = base_step
        cand = None
        cand_sc = None
        for _ in range(5):
            cand_try = _project_simplex(cur - step * momentum)
            cand_try_sc = _score(cand_try, L)
            if cand_try_sc < cur_sc:
                cand, cand_sc = cand_try, cand_try_sc
                break
            step *= 0.5
        # if no improvement, keep current
        if cand is not None:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # ----- stochastic rescue when stuck --------------------------------
        if no_improve > 1200:
            # attempt a random perturbation
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0
            else:
                # try LP refinement
                lp_vec = _solve_lp(cur, max_val, t_mix=0.15)
                if lp_vec is not None:
                    lp_sc = _score(lp_vec, L)
                    if lp_sc < cur_sc:
                        cur, cur_sc = lp_vec, lp_sc
                        if lp_sc < best_sc:
                            best_vec, best_sc = lp_vec.copy(), lp_sc
                        no_improve = 0

        # ----- occasional symmetrisation ---------------------------------
        if it % 1500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # early exit if target reached
        if best_sc < target:
            break

    # Return best found vector as plain Python floats
    return [float(x) for x in best_vec]
