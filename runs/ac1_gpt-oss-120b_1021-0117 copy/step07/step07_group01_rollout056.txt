def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., float(budget_s) - 0.2)

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

        a = rng.random(2000, dtype=np.float64) # fallback.

    a = np.maximum(a, 0)
    a = _project_simplex(a)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)

    best = a.copy()
    best_sc = _score(best)

    step0 = 0.5
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # FFT of a
        a_fft = np.fft.rfft(a, n=L)
        # compute full convolution
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]

        max_val = conv.max()

        # smoothing weight
        # alpha schedule
        alpha = 1.0 / max(0.01, 0.5 / math.sqrt(it))  # this gives larger alpha over time
        # Actually we want temperature decreasing: temp = 0.5 / sqrt(it+1)
        temp = max(0.02, 0.5 / math.sqrt(it))
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(conv) / conv.size
        else:
            w /= w_sum

        # gradient via convolution with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # step size schedule
        step = step0 / (1.0 + it*0.001)   # decays slowly

        # update
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        sc = _score(a_new)

        if sc < best_sc:
            best = a_new.copy()
            best_sc = sc
            a = a_new
            no_improve = 0
        else:
            # accept if improvement over current a
            cur_sc = _score(a)
            if sc < cur_sc:
                a = a_new
                no_improve = 0
            else:
                no_improve += 1
                # maybe occasional accept with probability:
                prob = math.exp(-(sc - cur_sc) / (step + 1e-12))
                if rng.random() < prob:

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score_vec(a, L):
    """Exact score used by the evaluator (2·n·max(conv))."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    best_sc = float(2.0 * n * conv_full[: 2 * n - 1].max())
