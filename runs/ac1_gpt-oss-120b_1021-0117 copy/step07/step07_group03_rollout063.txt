    max_val, _ = _conv_max_and_grad(a, L)
    best_score = 2 * n * max_val
    # parameters

    iter = 0

    while time.time() < deadline:
        iter += 1
        eta = eta0 / (1.0 + decay*iter)
        max_val, grad = _conv_max_and_grad(a, L)
        cur_score = 2 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # exponentiated update
        exp_term = -eta * grad
        # clipping
        exp_term = np.clip(exp_term, -20, 20)
        a_new = a * np.exp(exp_term)
        sum_a = a_new.sum()
        if sum_a == 0:
            a_new = a.copy()
        else:
            a_new = a_new / sum_a

        # occasional smoothing
        if iter % 50 == 0:
            kernel = np.array([0.25,0.5,0.25])
            a_new = np.convolve(a_new, kernel, mode='same')
            a_new = a_new / a_new.sum()
        # occasional symmetrization
        if iter % 200 == 0:
            a_new = (a_new + a_new[::-1]) * 0.5
            a_new = a_new / a_new.sum()

        # evaluate
        max_val_new, _ = _conv_max_and_grad(a_new, L)
        new_score = 2 * n * max_val_new
        # accept if better else accept with probability
        if new_score < cur_score:
            a = a_new
            no_improve = 0
        else:
            # simulated annealing probability decays
            prob = math.exp(-(new_score - cur_score) / (eta + 1e-12))
            if rng.random() < prob:
                a = a_new
                no_improve = 0
            else:
                # possibly fallback to previous a (keep)
                pass

        # stagnation handling
        if no_improve > 5000:
            # random perturbation
            sigma = 0.1 / math.sqrt(iter)
            pert = a + sigma * rng.standard_normal(size=n)
            pert = np.maximum(pert, 0.0)

            max_val_pert, _ = _conv_max_and_grad(pert, L)

            if pert_score < cur_score:
                a = pert
                if pert_score < best_score:
                    best_score = pert_score
                    best_vec = pert.copy()
                no_improve = 0
            else:
                # reset momentum? Not using momentum.
                no_improve = 0

        # early stop if near target
        if best_score < 1.503:
            break

    return [float(x) for x in best_vec]
