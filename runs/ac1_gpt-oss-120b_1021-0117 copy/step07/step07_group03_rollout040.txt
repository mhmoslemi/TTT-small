import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_max_and_grad(a, L, temperature=0.0):
    """
    Returns (max_convolution, gradient) for the max value of a*a.
    If temperature>0 a softmax mask is used, otherwise a hard mask of the
    maximal entries is applied.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()

    if temperature <= 0.0:
        eps = max_val * 1e-12 + 1e-15
        mask = (conv >= max_val - eps).astype(np.float64)
        cnt = mask.sum()
        if cnt > 0:
            mask /= cnt
    else:
        # soft‑max over the convolution values
        shift = conv - max_val          # max becomes 0, avoids overflow
        w = np.exp(shift / temperature)
        mask = w / w.sum()

    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad
