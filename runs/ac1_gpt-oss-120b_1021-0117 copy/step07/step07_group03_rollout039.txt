import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto {x >= 0, sum(x) = 1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max_and_argmax(a, L):
    """Return (max convolution value, index where it is attained) for a*a."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * a.size - 1]
    max_val = conv.max()
    arg = int(conv.argmax())
    return max_val, arg

def _grad_max_at_arg(a, arg):

    n = a.shape[0]

    i_end = min(arg, n - 1)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1)
        rev_idx = arg - idx
        grad[idx] = 2.0 * a[rev_idx]
    return grad

# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # leave a short margin for cleanup
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # random Dirichlet‑like start, length ≈ 2000
        n0 = 2000
        a0 = rng.random(n0, dtype=np.float64)
    a0 = np.maximum(a0, 0.0)
    a0 = (a0 + a0[::-1]) * 0.5          # enforce symmetry – helpful
    cur = _project_simplex(a0)
    n = cur.size
    L = 1 << ((2 * n - 2).bit_length())   # power‑of‑two ≥ 2n‑1

    # ----- bookkeeping ----------------------------------------------------
    max_val, arg = _conv_max_and_argmax(cur, L)
    cur_sc = 2.0 * n * max_val
    best_sc = cur_sc
    best_vec = cur.copy()

    # parameters
    step0 = 0.8
    momentum = np.zeros_like(cur)
    beta = 0.92
    it = 0
    no_improve = 0
    target = 1.50274349
    sym_interval = 250          # symmetrise every so many iterations
    perturb_interval = 3000    # when stuck, random perturbation

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # ----- gradient of current max ------------------------------------
        max_val, arg = _conv_max_and_argmax(cur, L)
        cur_sc = 2.0 * n * max_val
        grad = _grad_max_at_arg(cur, arg)

        # ----- momentum update -------------------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- step size -------------------------------------------------
        step = step0 / math.sqrt(it)

        # ----- try a few line‑search reductions -------------------------
        accepted = False
        cand = None
        cand_sc = None
        cur_step = step
        for _ in range(4):
            cand_try = _project_simplex(cur - cur_step * momentum)
            # enforce symmetry (keeps search space small)
            cand_try = (cand_try + cand_try[::-1]) * 0.5
            cand_try = _project_simplex(cand_try)

            max_val_c, _ = _conv_max_and_argmax(cand_try, L)
            cand_sc_try = 2.0 * n * max_val_c
            if cand_sc_try < cur_sc:
                cand, cand_sc = cand_try, cand_sc_try
                accepted = True
                break
            cur_step *= 0.5

        # ----- fallback to simple gradient step if line search failed -----
        if not accepted:
            cand_try = _project_simplex(cur - step * grad)
            cand_try = (cand_try + cand_try[::-1]) * 0.5
            cand_try = _project_simplex(cand_try)
            max_val_c, _ = _conv_max_and_argmax(cand_try, L)
            cand_sc_try = 2.0 * n * max_val_c
            if cand_sc_try < cur_sc:
                cand, cand_sc = cand_try, cand_sc_try
                accepted = True

        # ----- acceptance -------------------------------------------------
        if accepted:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # occasional probabilistic acceptance (simulated‑annealing flavour)
            prob = math.exp(-(cand_sc_try - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand_try, cand_sc_try
                no_improve = 0
            else:
                no_improve += 1

        # ----- update global best -----------------------------------------
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        # ----- periodic symmetrisation ------------------------------------
        if it % sym_interval == 0:
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)

        # ----- random perturbation when stuck ------------------------------
        if no_improve > perturb_interval:
            sigma = 0.05 / math.sqrt(it + 1.0)
            pert = cur + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(pert)
            pert = (pert + pert[::-1]) * 0.5
            pert = _project_simplex(pert)
            max_val_p, _ = _conv_max_and_argmax(pert, L)
            pert_sc = 2.0 * n * max_val_p
            if pert_sc < cur_sc:
                cur, cur_sc = pert, pert_sc
                if pert_sc < best_sc:
                    best_sc = pert_sc
                    best_vec = pert.copy()
            else:
                momentum.fill(0.0)  # reset momentum
            no_improve = 0

        # ----- early exit if we beat the target ----------------------------
        if best_sc < target:
            break

    # Return best vector as plain Python floats
    return [float(x) for x in best_vec]
