import time
import numpy as np


def _fast_score(a):
    """Fast computation of 2·n·max(a*a) where * is discrete convolution."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    # length for zero‑padded FFT (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
