import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # Sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

    start = time.time()
    deadline = start + max(0., float(budget_s) - 0.2)

    if 'height_sequence_1' in globals():

    else:

    n = cur.size
    L = 1 << ((2*n - 1).bit_length())

    best_sc = _score(best)
    cur_sc = best_sc

    m = np.zeros_like(cur)  # first moment
    v = np.zeros_like(cur)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # mask for max positions (tolerance)
        mask = (conv >= max_val - 1e-12).astype(np.float64)
        mask_sum = mask.sum()
        if mask_sum == 0:
            mask = np.ones_like(conv) / conv.size
            mask_sum = 1.0
        w = mask / mask_sum

        # gradient via FFT
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Normalize gradient direction
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0:
            # random restart
            cur = _project_simplex(np.maximum(cur + rng.normal(scale=0.01, size=n), 0.0))
            continue

        g = grad / grad_norm

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        step = lr0 / np.sqrt(it)  # decay
        update = - step * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur + update  # note update already negative direction
        cand = _project_simplex(np.maximum(cand, 0.0))
        # optionally enforce symmetry every some iterations
        if it % 500 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            # occasional acceptance of worse move (simulated annealing)
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc

        # random perturb when stuck:
        if no_improve > 4000:

            j = rng.integers(0, n-1)
            if j >= i:

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast evaluation of 2·n·max(conv(a,a)) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        g = grad / g_norm

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr0 / math.sqrt(it)            # simple decay
        update = -step * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur + update
        cand = _project_simplex(np.maximum(cand, 0.0))

        # occasional symmetrisation (helps when the optimum is symmetric)
        if it % 400 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # ---- accept or reject -------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            # tiny probability to accept worsening move (simulated annealing)
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc

        # ---- stagnation handling -----------------------------------------
        if no_improve > 5000:
            # redistribute a small chunk of mass between two random indices
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.25
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc

        # ---- early exit --------------------------------------------------
        if best_sc < target:
            break

    # return simplest Python list
    return [float(x) for x in best]
```
