import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    """Score assuming sum(a)==1 -> 2*n*max(conv)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _full_conv(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]

def _gradient_max_conv(a, k):
    """Sub‑gradient of max‑convolution at index k (for sum(a)==1)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g[mask] += a[idx[mask]]
    if k < n:
        g[k:] += a[: n - k]
    # derivative appears twice (i and its partner)
    return 2.0 * g

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ---- initial vector ----------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size

    best_vec = cur.copy()
    best_sc = _score_vec(best_vec)
    cur_sc = best_sc

    # ---- optimizer state ---------------------------------------------------
    momentum = np.zeros_like(cur)
    beta = 0.85          # lower than before for faster adaptation
    step0 = 1.0
    it = 0
    no_improve = 0
    target = 1.50274349

    # parameters for occasional random moves
    redistrib_thresh = 3000

    while time.monotonic() < deadline:
        it += 1

        # full convolution and max
        conv = _full_conv(cur)
        max_val = conv.max()
        max_idx = int(conv.argmax())

        # gradient of the active lag (exact sub‑gradient)
        grad = _gradient_max_conv(cur, max_idx)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule
        step = step0 / math.sqrt(it)

        # two candidate updates: projected SGD and exponentiated update
        cand_sgd = _project_simplex(cur - step * momentum)

        # exponentiated gradient update (stay on simplex automatically)
        # clip exponent to avoid overflow
        exp_arg = -step * momentum
        exp_arg = np.clip(exp_arg, -50, 50)
        cand_exp = cur * np.exp(exp_arg)
        cand_exp_sum = cand_exp.sum()
        if cand_exp_sum == 0:
            cand_exp = np.full_like(cur, 1.0 / n)
        else:
            cand_exp = cand_exp / cand_exp_sum

        # evaluate candidates (both have sum≈1)
        cand_sgd_sc = _score_vec(cand_sgd)
        cand_exp_sc = _score_vec(cand_exp)

        # pick better of the two
        if cand_sgd_sc < cand_exp_sc:
            cand, cand_sc = cand_sgd, cand_sgd_sc
        else:
            cand, cand_sc = cand_exp, cand_exp_sc

        # accept if improvement, otherwise keep current (possible stochastic tweak later)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1

        # ----- periodic symmetrisation (helps with many optimal patterns) -----
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ----- random local redistribution when stuck ------------------------
        if no_improve > redistrib_thresh:
            nonzero = np.where(cur > 0)[0]
            if nonzero.size > 0:
                i = rng.choice(nonzero)
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_vec(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
            else:
                # complete restart (very unlikely)
                cur = _project_simplex(rng.random(n))
                cur_sc = _score_vec(cur)
                momentum.fill(0.0)

        # ----- early exit ----------------------------------------------------
        if best_sc < target:
            break

        # occasional time check
        if it % 1000 == 0 and time.monotonic() >= deadline:
            break

    # return best vector as plain Python floats
    return [float(x) for x in best_vec]
