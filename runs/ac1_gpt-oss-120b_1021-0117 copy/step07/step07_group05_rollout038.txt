def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # Determine n
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    else:
        n = 2000
        a = np.full(n, 1.0 / n, dtype=np.float64)  # uniform

    # Simplex projection
    def _project_simplex(v):
        v = np.asarray(v, dtype=np.float64)
        # sort descending
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    # Normalize to sum 1
    a = _project_simplex(a)

    try:
        cur_score = evaluate_sequence(a.tolist())
    except Exception:
        cur_score = float('inf')
    best_vec = a.copy()
    best_score = cur_score

    # Gradient descent parameters
    eta = 0.5   # initial step size
    min_eta = 1e-6
    max_iter = 20000   # just large; will be limited by deadline

    it = 0
    while time.monotonic() < deadline and it < max_iter:
        it += 1
        # compute convolution and gradient
        c = np.convolve(a, a)
        m = np.max(c)
        mask = (c >= m - 1e-12)
        K = mask.sum()
        if K == 0:
            break
        # correlation
        conv = np.convolve(mask.astype(np.float64), a[::-1])
        g = conv[n-1:n-1+n]
        avg_a = g / K
        grad = 4.0 * n * (avg_a - m)  # because sum a = 1

        # direction
        d = -grad
        # try step
        cand = _project_simplex(a + eta * d)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
            # increase step size moderately
            eta = min(eta * 1.2, 5.0)
        else:
            # no improvement: shrink step
            eta = max(eta * 0.5, min_eta)

        # occasional random perturbation to escape plateau
        if it % 1000 == 0:
            rng = np.random.default_rng(seed + it)
            # small random perturb p
            delta = rng.normal(scale=0.01, size=n)
            cand2 = _project_simplex(a + delta)
            cand2_score = evaluate_sequence(cand2.tolist())
            if cand2_score < best_score:
                best_score = cand2_score
                best_vec = cand2.copy()
                a = cand2
                cur_score = cand2_score

    # return best vector found
    return [float(x) for x in best_vec]

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
