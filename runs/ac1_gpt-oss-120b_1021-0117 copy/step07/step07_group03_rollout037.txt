import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_max_and_grad(a, L):
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * a.shape[0] - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * a.shape[0] - 2) - np.arange(a.shape[0])
    grad = 2.0 * grad_full[idx]
    return max_val, grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # initial vector
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_guess = 2000
        a0 = rng.random(n_guess, dtype=np.float64)

    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # best-so-far tracking
    max_val, _ = _conv_max_and_grad(cur, L)
    best_score = 2.0 * n * max_val
    best_vec = cur.copy()
    cur_score = best_score

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    lr = 0.5
    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # gradient at current point
        max_val, grad = _conv_max_and_grad(cur, L)
        cur_score = 2.0 * n * max_val

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        cand = _project_simplex(cur - step)

        # occasional symmetrisation
        if it % 250 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        cand_max, _ = _conv_max_and_grad(cand, L)
        cand_score = 2.0 * n * cand_max

        # acceptance rule (simple greedy with occasional random walk)
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
        else:

                no_improve = 0

                no_improve += 1

            no_improve = 0

            if pert_score < cur_score:
                cur, cur_score = pert, pert_score
                if pert_score < best_score:
                    best_score = pert_score
                    best_vec = pert.copy()
                no_improve = 0
            else:
                # reset momentum
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0

        if best_score < target:
            break

    # final safe conversion to plain Python list
    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_max_and_grad(a, L):
    """
    Returns (max_convolution, subgradient) of the full convolution a*a.
    Uses FFT for O(L log L) work; L must be a power of two >= 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * a.shape[0] - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)

    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * a.shape[0] - 2) - np.arange(a.shape[0])
    grad = 2.0 * grad_full[idx]
