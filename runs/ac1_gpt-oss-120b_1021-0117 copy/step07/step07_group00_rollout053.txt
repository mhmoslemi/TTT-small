import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast score for a vector already normalized to sum=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.4)

    # ----- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    # ----- hyper‑parameters ------------------------------------------------
    momentum = np.zeros_like(cur)
    beta = 0.92                # momentum decay
    step0 = 1.1                # initial step size factor
    temp_factor = 0.08         # relative to current max conv
    decay = 0.997              # temperature decay
    target = 1.50274349
    it = 0
    no_improve = 0

    # ----- main optimisation loop ----------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution of current vector with itself
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature scheduling
        if it == 1:
            temp = max_val * temp_factor
        else:
            temp = max(temp * decay, max_val * 1e-6)

        # soft‑max weights (fallback to hard max when temperature tiny)
        if temp < max_val * 1e-12:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(conv)
                w_sum = w.size
        else:
            # protect against overflow
            arg = (conv - max_val) / temp
            arg = np.clip(arg, -700, 700)
            w = np.exp(arg)
            w_sum = w.sum()
        w /= w_sum

        # gradient of the weighted max (via correlation)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size (decreasing)
        step = step0 / math.sqrt(it)

        # evaluate a few candidate step factors
        cand_factors = (0.5, 1.0, 1.5, 2.0)
        cand_sc_best = float('inf')
        cand_best = None
        for f in cand_factors:
            cand = _project_simplex(cur - step * f * momentum)
            cand_sc = _score(cand)
            if cand_sc < cand_sc_best:
                cand_sc_best = cand_sc
                cand_best = cand

        # acceptance
        if cand_sc_best < cur_sc:
            cur, cur_sc = cand_best, cand_sc_best
            no_improve = 0
            if cand_sc_best < best_sc:
                best, best_sc = cand_best.copy(), cand_sc_best
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc_best - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand_best, cand_sc_best
                if cand_sc_best < best_sc:
                    best, best_sc = cand_best.copy(), cand_sc_best
                no_improve = 0

        # random redistribution when stuck
        if no_improve > 1800:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.45
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc

        # periodic symmetrisation (helps because convolution is symmetric)
        if it % 600 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

        # early exit if target reached
        if best_sc < target:
            break

    # ----- final fine‑tuning with exact max subgradient --------------------
    # Only if time permits and we are still above target
    while time.time() < deadline - 0.1 and best_sc > target:
        a_fft = np.fft.rfft(best, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        mask = (conv == max_val).astype(np.float64)
        w_sum = mask.sum()
        if w_sum == 0:
            break
        w = mask / w_sum
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # small aggressive step
        step = 0.3 / math.sqrt(it + 1)
        cand = _project_simplex(best - step * grad)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
        else:
            break
        it += 1

    # Ensure a valid finite list is returned
    return [float(x) for x in best]
