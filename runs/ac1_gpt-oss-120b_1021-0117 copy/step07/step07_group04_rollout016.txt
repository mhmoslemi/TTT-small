import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # correct possible numerical drift
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast evaluator (assumes a>=0)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # Length and FFT size
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2000
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Deterministic initial candidates
    xs = np.arange(n, dtype=np.float64)
    candidates = []

    # 1) Uniform
    candidates.append(np.full(n, 1.0 / n))

    # 2) Triangular (peak centre)
    tri = np.minimum(xs + 1, n - xs)
    candidates.append(tri / tri.sum())

    # 3) Sine bump
    sine = np.sin(np.pi * (xs + 1) / (n + 1))
    candidates.append(sine / sine.sum())

    # 4) Raised cosine
    bump = (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2
    candidates.append(bump / bump.sum())

    # 5) Exponential decays
    for alpha in (0.001, 0.003, 0.006):
        exp = np.exp(-alpha * xs)
        candidates.append(exp / exp.sum())

    # 6) Power laws
    for p in (0.5, 1.0, 1.5):
        pw = (xs + 1) ** (-p)
        candidates.append(pw / pw.sum())

    # 7) Random Dirichlet‑like
    rand = rng.random(n)
    candidates.append(rand / rand.sum())

    # 8) Parent if available
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.sum() > 0:
            parent = parent / parent.sum()
            candidates.append(parent)

    # ------------------------------------------------------------------
    # Choose best deterministic start
    best_vec = None
    best_sc = float('inf')
    for vec in candidates:
        sc = _score_fft(vec, n, L)
        if sc < best_sc:
            best_sc, best_vec = sc, vec.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # ------------------------------------------------------------------
    # Adam optimiser state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    base_lr = 0.5
    eps = 1e-8
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ---- convolution & max mask (single active index) ----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        tol = max_val * 1e-12 + 1e-15
        max_idxs = np.where(conv >= max_val - tol)[0]
        # pick a random active index to get a concrete sub‑gradient
        idx_active = rng.choice(max_idxs)
        w = np.zeros_like(conv)
        w[idx_active] = 1.0

        # ---- gradient (sub‑gradient of max‑convolution) ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx_map = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx_map]

        # ---- Adam update ----
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        direction = m_hat / (np.sqrt(v_hat) + eps)

        # ---- try a few step scalings ----
        local_best = cur
        local_best_sc = cur_sc
        for factor in (0.5, 1.0, 1.5):
            step = base_lr * factor * direction / math.sqrt(it)
            cand = _project_simplex(cur - step)
            cand_sc = _score_fft(cand, n, L)
            if cand_sc < local_best_sc:
                local_best, local_best_sc = cand, cand_sc

        # ---- accept if improved, otherwise occasional annealing ----
        if local_best_sc < cur_sc:
            cur, cur_sc = local_best, local_best_sc
            no_improve = 0
        else:
            no_improve += 1
            # tiny chance to accept worsening move
            prob = math.exp(-(local_best_sc - cur_sc) / (base_lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = local_best, local_best_sc

        # ---- enforce symmetry (helps reduce peak) ----
        cur = (cur + cur[::-1]) * 0.5
        cur = _project_simplex(cur)
        # recompute score after symmetrisation
        cur_sc = _score_fft(cur, n, L)

        # ---- update global best ----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()

        # ---- random perturbation when stuck ----
        if no_improve > 3000:
            # inject small Dirichlet noise
            noise = rng.random(n)
            cand = cur * 0.7 + noise * 0.3
            cand = _project_simplex(cand)
            cand_sc = _score_fft(cand, n, L)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
            else:
                no_improve = 0   # reset to avoid endless stalling

        # ---- early exit if target reached ----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return as plain Python list
    return [float(x) for x in best_vec]
