def max_convolution(a):
    n = len(a)
    size = 1 << (2*n - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    # conv is length size, we need first 2n-1
    return np.max(conv[:2*n-1])
```

def simplex_projection(v):
    # implementation from Duchi et al.
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0

import time
import numpy as np

def _project_simplex(v):
    """Projection of v onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        # all entries zero or negative – return uniform
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(cond)[0][-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical clean‑up
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / v.size)
    return w

    init = np.clip(init, 0.0, 1000.0)

    cur = _project_simplex(init)

    # ----------------------------------------------------------------------

    try:

        best_score = float("inf")
    best = cur.copy()
