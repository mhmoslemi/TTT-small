import time, math, numpy as np

def _conv_max_and_grad(a, L):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

def _exp_update(a, grad, eta):
    # multiplicative update staying on simplex
    delta = -eta * grad
    # clip to avoid overflow
    delta = np.clip(delta, -50.0, 50.0)
    a = a * np.exp(delta)
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        # fallback uniform
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s
    return a

def _conv_max_and_grad(a, L):
    """Return (max of a*a, subgradient at a) using FFT."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

def _exp_update(a, grad, eta):
    """Multiplicative (exponential) update staying on the simplex."""
    delta = -eta * grad
    delta = np.clip(delta, -50.0, 50.0)          # avoid overflow
    a = a * np.exp(delta)
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        a = np.full_like(a, 1.0 / a.size)       # fallback uniform
    else:
        a = a / s
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # ----- initialise ----------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
    else:
        a0 = rng.random(2500, dtype=np.float64)
    a0 += 1e-12                         # avoid exact zeros
    cur = a0 / a0.sum()                 # simplex normalisation
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2n‑1

    max_val, _ = _conv_max_and_grad(cur, L)
    cur_score = 2.0 * n * max_val
    best_score = cur_score
    best_vec = cur.copy()

    # ----- optimisation parameters ---------------------------------------
    eta0 = 0.5          # base learning rate for exponential update
    it = 0
    no_improve = 0
    target = 1.50274349

    # ----- main optimisation loop ----------------------------------------
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # gradient at current point
        max_val, grad = _conv_max_and_grad(cur, L)
        cur_score = 2.0 * n * max_val

        # exponential (mirror) step
        eta = eta0 / math.sqrt(it)
        cand = _exp_update(cur, grad, eta)

        # evaluate candidate
        cand_max, _ = _conv_max_and_grad(cand, L)
        cand_score = 2.0 * n * cand_max

        # acceptance
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
        else:
            if rng.random() < 0.001:   # tiny uphill chance
                cur = cand
                cur_score = cand_score
            no_improve += 1

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_vec = cur.copy()
            no_improve = 0

        # periodic symmetrisation (helps flatten peaks)
        if it % 400 == 0:
            cur = (cur + cur[::-1]) * 0.5
            cur = cur / cur.sum()

        # stagnation handling
        if no_improve > 4000:
            sigma = 0.05 / math.sqrt(it)
            pert = cur + sigma * rng.standard_normal(size=n)
            pert = np.maximum(pert, 0.0)
            pert = pert / pert.sum()
            pert_max, _ = _conv_max_and_grad(pert, L)
            pert_score = 2.0 * n * pert_max
            if pert_score < cur_score:
                cur = pert
                cur_score = pert_score
            no_improve = 0

        # early exit if target reached
        if best_score < target:
            break

    # ----- final local refinement ----------------------------------------
    ref = best_vec.copy()
    for j in range(2000):
        if time.time() >= deadline:
            break
        max_val, grad = _conv_max_and_grad(ref, L)
        step = 0.1 / math.sqrt(j + 1)
        cand = ref - step * grad
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        cand_max, _ = _conv_max_and_grad(cand, L)
        cand_score = 2.0 * n * cand_max
        if cand_score < best_score:
            best_score = cand_score
            best_vec = cand.copy()
            ref = cand.copy()
        # occasional symmetrisation inside refinement
        if j % 500 == 0:
            ref = (ref + ref[::-1]) * 0.5
            ref = ref / ref.sum()

    return [float(x) for x in best_vec]
