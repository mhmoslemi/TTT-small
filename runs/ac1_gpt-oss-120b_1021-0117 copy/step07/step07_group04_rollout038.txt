import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # correct numerical
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, L):

    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*len(a)-1]
    return float(2.0 * len(a) * conv.max() / (s * s))

        # ---- step size (inverse‑sqrt decay) ----
        step = step0 / math.sqrt(it)

        # ---- simple line search over a few scale factors ----
        local_best_vec = cur
        local_best_sc = cur_sc
        for factor in (0.7, 1.0, 1.3):
            cand = _project_simplex(cur - step * factor * momentum)
            cand_sc = _score_fft(cand, L)
            if cand_sc < local_best_sc:
                local_best_sc = cand_sc
                local_best_vec = cand

        # ---- acceptance ----
        if local_best_sc < cur_sc:
            cur, cur_sc = local_best_vec, local_best_sc
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(local_best_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = local_best_vec, local_best_sc

        # ---- global best update ----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()

        # ---- periodic symmetrisation (helps enforce shape) ----
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, L)
            if sym_sc < best_sc:
                best_sc, best_vec = sym_sc, sym.copy()
                cur, cur_sc = sym, sym_sc
                no_improve = 0

        # ---- random redistribution when progress stalls ----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_fft(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0

        # ---- early exit if the target has been reached ----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
