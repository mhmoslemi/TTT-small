def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w


def _score(a):
    """Surrogate score for a vector with sum=1: 2*n*max(conv) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # optimizer state
    momentum = np.zeros_like(cur)
    beta = 0.95            # momentum coefficient
    step0 = 0.9            # initial step magnitude
    it = 0
    no_improve = 0

    # gradient top‑k fraction
    top_frac = 0.01        # use top 1 % of convolution entries
    target = 1.50274349    # early‑stop target

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- FFT of current vector -----
        a_fft = np.fft.rfft(cur, n=L)

        # ----- convolution (auto‑correlation) -----
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- weight vector: top‑k mask -----
        k = max(1, int(top_frac * n))
        idx_top = np.argpartition(-conv, k - 1)[:k]
        w = np.zeros_like(conv, dtype=np.float64)
        w[idx_top] = 1.0 / k

        # ----- gradient of weighted sum of convolution entries -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # map to size‑n gradient
        grad = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- step size (decays like 1/sqrt(it)) -----
        step = step0 / math.sqrt(it)

        # ----- candidate and evaluation -----
        cand = cur - step * momentum
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # ----- acceptance -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            temp = step * 0.5 + 1e-12
            prob = math.exp(-(cand_sc - cur_sc) / temp)
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ----- periodic ascending sort (helps reverse‑dot product) -----
        if it % 400 == 0:
            asc = np.sort(cur)               # ascending order
            asc_sc = _score(asc)
            if asc_sc < cur_sc:
                cur, cur_sc = asc, asc_sc
                no_improve = 0
                if asc_sc < best_sc:
                    best_vec, best_sc = asc.copy(), asc_sc

        # ----- occasional mixing with uniform distribution -----
        if it % 1500 == 0:
            uniform = np.full_like(cur, 1.0 / n)
            mixed = 0.7 * cur + 0.3 * uniform
            mixed = _project_simplex(mixed)
            mixed_sc = _score(mixed)
            if mixed_sc < cur_sc:
                cur, cur_sc = mixed, mixed_sc
                if mixed_sc < best_sc:
                    best_vec, best_sc = mixed.copy(), mixed_sc

        # ----- random local redistribution when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- safeguard time check -----
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
