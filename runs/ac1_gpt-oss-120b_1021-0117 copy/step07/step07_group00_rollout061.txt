def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.1)   # buffer

    # initialize
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    # Ensure nonnegative, then project onto simplex sum=1
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.8   # base learning rate
    # Temperature schedule
    temp = None
    decay_temp = 0.996
    min_temp_factor = 5e-4

    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay_temp, max_val * min_temp_factor)

        # softmax weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        # ensure nonnegative before projecting
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # accept if improvement or occasional acceptance
        if cand_sc < best_sc:
            best_vec, best_sc = cand.copy(), cand_sc
        if cand_sc < _score(cur) or rng.random() < math.exp(-(cand_sc - _score(cur)) / (lr + 1e-12)):
            cur = cand
            no_improve = 0
        else:
            no_improve += 1

        # random perturbation after many non-improves
        if no_improve > 2000:
            # select two indices
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            # transfer fraction from i to j
            delta = cur[i] * rng.random() * 0.5
            cur[i] -= delta
            cur[j] += delta
            cur = _project_simplex(cur)
            cur_sc = _score(cur)
            if cur_sc < best_sc:
                best_vec, best_sc = cur.copy(), cur_sc
            no_improve = 0

        # occasional symmetrization
        if it % 600 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_vec, best_sc = sym.copy(), sym_sc
            if sym_sc < _score(cur):
                cur = sym

        # early exit
        if best_sc < target:
            break

    return [float(x) for x in best_vec]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    # for increased flexibility, skip simplex projection; keep sum maybe not 1
    # but ensure positivity. We'll keep original sum to preserve shape.
    # But for gradient steps we need to keep scale stable; we'll not project onto simplex but
    # regularly renormalize to keep sum close to 1.
    cur = cur / cur.sum()  # normalize to sum=1 for start
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # evaluate
    best_vec = cur.copy()
    best_sc = _score(best_vec)   # raw_score (2*n*maxconv)
    cur_sc = best_sc

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta0 = 0.5  # base learning rate

    temp = None
    decay_temp = 0.998
    min_temp_factor = 5e-4

    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.time() >= deadline:
            break
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature adaptation
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay_temp, max_val * min_temp_factor)

        # softmax weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)

        # candidate
        cand = cur - step
        cand = np.maximum(cand, 0.0)
        # renormalize to sum=1 to keep scales comparable
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand = cand / cand_sum
        else:
            cand = np.full_like(cand, 1.0 / n)

        cand_sc = _score(cand)

        # acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            # simulated annealing acceptance based on raw_score diff
            prob = math.exp(-(cand_sc - cur_sc) / (eta0 + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # occasional random perturb when stuck
        if no_improve > 1800:
            # redistribute mass between random indices proportionally
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            cur[i] -= delta
            cur[j] += delta
            cur = np.maximum(cur, 0.0)
            cur_sum = cur.sum()
            cur = cur / cur_sum
            cur_sc = _score(cur)
            if cur_sc < best_sc:
                best_vec, best_sc = cur.copy(), cur_sc
            no_improve = 0

        # symmetrization
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = np.maximum(sym, 0.0)
            sym = sym / sym.sum()
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_vec, best_sc = sym.copy(), sym_sc
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc

        if best_sc < target:
            break

    # return best vector as list
    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of a vector onto the probability simplex (sum=1, v>=0). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """ Fast score for a vector (assumes sum≈1). Returns 2*n*max(convolution). """
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function heights list attempting to minimise the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = cur / cur.sum()                     # start with sum = 1

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # ----- Adam optimiser state -------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    eta0 = 0.5                                 # base learning rate

    # ----- temperature (soft‑max) -----------------------------------------
    temp = None
    decay_temp = 0.998
    min_temp_factor = 5e-4

    it = 0
    no_improve = 0
    target = 1.50274349   # early‑exit target (may never be met)

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # ----- convolution & max -----------------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- temperature adaptation --------------------------------------
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay_temp, max_val * min_temp_factor)

        # ----- soft‑max weights -------------------------------------------
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum

        # ----- gradient of weighted max ------------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
