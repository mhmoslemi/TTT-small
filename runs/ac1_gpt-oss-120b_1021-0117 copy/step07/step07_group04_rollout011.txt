    tri = np.minimum(xs + 1, n - xs)
    candidates.append(tri / tri.sum())

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # correct possible tiny numerical drift
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast evaluation of the objective (no clipping, assumes a>=0)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

def _conv_and_grad(cur, n, L):
    """
    Return (max_conv, gradient) where gradient is a sub‑gradient of max_conv
    w.r.t. cur (sum fixed). Uses FFT for O(n log n) work.
    """
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps_mask = max_val * 1e-12 + 1e-15
    mask = conv >= max_val - eps_mask

    idx = (2 * n - 2) - np.arange(n)

            cur, cur_sc = local_best_vec, local_best_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional simulated‑annealing step
            prob = math.exp(-(local_best_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = local_best_vec, local_best_sc
                no_improve = 0

        # ----- update global best -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()

        # ----- periodic symmetrisation -----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ----- random redistribution when stuck -----
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_fft(tmp, n, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # --------------------------------------------------------------
    # Final lightweight local search (hill‑climbing) with remaining time
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if best_vec[i] == 0.0:
            continue
        delta = best_vec[i] * rng.random() * 0.2
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score_fft(cand, n, L)
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()
        # avoid tight looping
        if time.monotonic() + 0.001 >= deadline:
            break

    # --------------------------------------------------------------
    return [float(x) for x in best_vec]
