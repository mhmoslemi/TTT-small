import time

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

        cur = _project_simplex(cur)

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # next power of two >= 2n-1

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # best-so-far
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    best_sc = float(2 * n * conv.max())
    best_vec = cur.copy()
    cur_sc = best_sc

    # control variables
    it = 0
    no_improve = 0
    target = 1.50274349
    sym_interval = 200           # how often to symmetrize
    perturb_after = 8000         # attempts before a random perturb

    while time.monotonic() < deadline:
        it += 1

        # ---------- convolution ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        M = conv.max()

        # ---------- active‑set subgradient ----------
        eps_active = max(1e-12, 1e-9 * M)
        idx_active = np.where(conv >= M - eps_active)[0]
        if idx_active.size == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = np.zeros_like(conv)
            w[idx_active] = 1.0 / idx_active.size
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = 0.8 / (np.sqrt(v_hat) + eps)
        cand = cur - step * m_hat
        cand = _project_simplex(cand)

        # ---------- evaluate candidate ----------
        cand_fft = np.fft.rfft(cand, n=L)
        conv_cand_full = np.fft.irfft(cand_fft * cand_fft, n=L)
        cand_sc = float(2 * n * conv_cand_full[: 2 * n - 1].max())

        # ---------- acceptance ----------
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            no_improve += 1
            if rng.random() < 0.001:          # occasional stochastic accept
                cur = cand
                cur_sc = cand_sc

        # ---------- periodic symmetrisation ----------
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            conv_sym_full = np.fft.irfft(sym_fft * sym_fft, n=L)
            sym_sc = float(2 * n * conv_sym_full[: 2 * n - 1].max())
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()

        # ---------- random redistribution when stuck ----------
        if no_improve > perturb_after:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_sc = float(2 * n * np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1].max())
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                else:
                    no_improve = 0
            else:
                no_improve = 0

        # ---------- early termination ----------
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # return the best sequence found (as plain Python floats)
    return [float(x) for x in best_vec]
