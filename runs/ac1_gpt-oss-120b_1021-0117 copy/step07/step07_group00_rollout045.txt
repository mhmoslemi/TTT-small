import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        v = v.ravel()
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, L=None):
    """Score 2*n*max(convolve(a,a)) assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        # exponentiated‑gradient style update
        grad_clip = np.clip(grad, -12.0, 12.0)
        factor = np.exp(-lr * grad_clip)
        cand2 = _project_simplex(cur * factor)

        # ------- evaluate -------
        sc1 = _fast_score(cand1, L)
        sc2 = _fast_score(cand2, L)
        if sc1 <= sc2:
            cand, cand_sc = cand1, sc1
        else:
            cand, cand_sc = cand2, sc2

        # ------- acceptance -------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ------- random redistribution when stuck -------
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _fast_score(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # ------- periodic symmetrisation -------
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # ------- early exit ----------
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # safety fallback
    if best_sc == float('inf') or np.isnan(best_sc):
        best_vec = np.full(n, 1.0 / n)

    return [float(x) for x in best_vec]
