import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, conv_max=None):
    """Score assuming sum(a)=1: 2*n*max(convolve(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if conv_max is None:
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        conv_max = conv.max()
    return float(2.0 * n * conv_max)

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---- Adam update -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - pow1)
        v_hat = v / (1.0 - pow2)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(np.maximum(cand, 0.0))

        cand_sc = _fast_score(cand, conv_max=None)   # recompute fast score

        # ---- acceptance --------------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
        # always keep the global best
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()
            last_improve = it

        # ---- random redistribution if stuck -------------------------------
        if it - last_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _fast_score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                last_improve = it

        # ---- occasional symmetrisation ------------------------------------
        if it % 1500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                last_improve = it

        # ---- early exit --------------------------------------------------
        if best_sc < target:
            break

        # ---- update bias-correction powers --------------------------------
        pow1 *= beta1
        pow2 *= beta2

    # return as plain Python floats
    return [float(x) for x in best_vec]
