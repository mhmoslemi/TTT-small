import time, math, numpy as np

def _project_simplex(v):
    """Project a vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # correct tiny numerical drift
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast computation of the evaluator score (assumes a>=0)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

def _soft_weights(conv, k):
    """Softmax-like weights for smoothing the max operation."""
    maxc = conv.max()
    # subtract max to avoid under/overflow; all entries <= 0
    exp_vals = np.exp(k * (conv - maxc))
    sum_exp = exp_vals.sum()
    if sum_exp == 0:
        # fallback to uniform (should not happen)
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / sum_exp

        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # --------------------------------------------------------------
    # Optimisation state
    momentum = np.zeros_like(cur)
    beta = 0.88          # momentum decay
    step0 = 0.6          # base step size
    it = 0
    no_improve = 0
    target = 1.50274349
    transition = 300    # iterations before switching to hard mask
    k_soft = 30.0       # initial softmax temperature factor

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- FFT of current vector -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]

        # ----- Choose weighting for gradient -----
        if it <= transition:
            # gradually increase sharpness
            k = min(200.0, k_soft * (1.0 + it / 500.0))
            w = _soft_weights(conv, k)
        else:
            # hard mask on the active set (as in previous program)
            max_val = conv.max()
            eps = max_val * 1e-12 + 1e-15
            mask = conv >= max_val - eps
            w = mask.astype(np.float64)
            cnt = w.sum()
            if cnt > 0:
                w /= cnt

        # ----- Gradient from weighted convolution -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- Step size schedule -----
        step = step0 / math.sqrt(it)

        # ----- Line search over a few scaling factors -----
        local_best_vec = cur
        local_best_sc = cur_sc
        for factor in (0.5, 1.0, 1.5):
            cand = _project_simplex(cur - step * factor * momentum)
            cand_sc = _score_fft(cand, n, L)
            if cand_sc < local_best_sc:
                local_best_sc = cand_sc
                local_best_vec = cand

        # ----- Acceptance -----
        if local_best_sc < cur_sc:
            cur, cur_sc = local_best_vec, local_best_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional simulated‑annealing style acceptance
            prob = math.exp(-(local_best_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = local_best_vec, local_best_sc

        # ----- Global best update -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()

        # ----- Periodic symmetrisation -----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ----- Random redistribution when stuck -----
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_fft(tmp, n, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

        # ----- Early exit if target attained -----
        if best_sc < target:
            break

    # --------------------------------------------------------------
    # Return the best-found sequence as a plain Python list
    return [float(x) for x in best_vec]
