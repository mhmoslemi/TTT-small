import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

# ----------------------------------------------------------------------
def _score_fft(a, n, L):
    """Fast evaluation of the AC1 upper bound (assumes a>=0 and sum>0)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

# ----------------------------------------------------------------------
def _gradient(a, a_fft, conv, max_val, L, n):
    """Sub‑gradient of the max‑convolution w.r.t. a."""
    eps = max_val * 1e-12 + 1e-15
    mask = conv >= max_val - eps
    w = mask.astype(np.float64)
    cnt = w.sum()
    if cnt > 0:
        w /= cnt
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

    # --------------------------------------------------------------
    # optimisation parameters
    momentum = np.zeros_like(cur)
    beta = 0.9          # momentum
    step0 = 0.8         # base step size
    it = 0
    no_improve = 0
    target = 1.50274349

    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and gradient -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        grad = _gradient(cur, a_fft, conv, max_val, L, n)

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- step size (inverse sqrt decay) -----
        step = step0 / math.sqrt(it)

        # ----- line search over a few scale factors -----
        local_best_vec = cur
        local_best_sc = cur_sc
        for factor in (0.5, 1.0, 1.5):
            cand = _project_simplex(cur - step * factor * momentum)
            cand_sc = _score_fft(cand, n, L)
            if cand_sc < local_best_sc:
                local_best_sc = cand_sc
                local_best_vec = cand

        # ----- acceptance -----
        if local_best_sc < cur_sc:
            cur, cur_sc = local_best_vec, local_best_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional simulated‑annealing move
            prob = math.exp(-(local_best_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = local_best_vec, local_best_sc

        # ----- update global best -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()

        # ----- periodic symmetrisation -----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ----- random restart when stuck -----
        if no_improve > 4000:
            cur = rng.random(n)
            cur = _project_simplex(cur)
            momentum.fill(0.0)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # --------------------------------------------------------------
    # final local perturbation phase (light hill‑climbing)
    hill_steps = 2000
    for _ in range(hill_steps):
        if time.monotonic() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = cur[i] * rng.random() * 0.2
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score_fft(tmp, n, L)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
        else:
            # tiny Gaussian jitter occasionally accepted
            if rng.random() < 0.01:
                noise = rng.normal(scale=0.005, size=n)
                tmp = cur + noise
                tmp = _project_simplex(tmp)
                tmp_sc = _score_fft(tmp, n, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

    # --------------------------------------------------------------
    return [float(x) for x in best_vec]
