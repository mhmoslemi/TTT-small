def propose_candidate(seed=42,budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 0.3)  # keep margin

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2000, dtype=np.float64)

    a = _project_simplex(a)   # ensures sum=1

    n = a.size
    L = 1 << ((2*n - 1).bit_length())   # next power of two for FFT

    best_a = a.copy()
    best_score = _score(best_a)  # compute fast

    # AdaGrad variables:
    acc_sq = np.zeros_like(a)
    eta = 0.8
    decay = 0.9995
    min_eta = 1e-4
    # other parameters:
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        # compute conv and gradient
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()
        # indicator mask: equality within tolerance 1e-12 relative
        eps = max_val * 1e-12 + 1e-12
        mask = (conv_full >= max_val - eps).astype(np.float64)

        # gradient via convolution with mask reversed
        w_rev = mask[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n-2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # AdaGrad update
        acc_sq += grad**2
        step_vec = eta / np.sqrt(acc_sq + 1e-12)
        cand = _project_simplex(a - step_vec * grad)

        cand_score = _score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            # keep track of improvement
            a = cand
            no_improve = 0
        else:
            # accept with some probability if not too worse
            # use simulated annealing based on temperature derived from max_val
            # define temperature as max_val * 0.02
            temp = max_val * 0.02
            if temp > 0:
                prob = math.exp(-(cand_score - best_score) / temp)
            else:
                prob = 0.0
            if rng.random() < prob:
                a = cand
                no_improve = 0
            else:
                no_improve += 1

        # occasional random redistribution if stuck
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            delta = a[i] * rng.random() * 0.5
            a_tmp = a.copy()
            a_tmp[i] -= delta
            a_tmp[j] += delta
            a_tmp = _project_simplex(a_tmp)
            a_tmp_score = _score(a_tmp)
            if a_tmp_score < best_score:
                best_score = a_tmp_score
                best_a = a_tmp.copy()
                a = a_tmp
                no_improve = 0
            else:
                no_improve = 0  # reset anyway

        # occasionally symmetrize
        if no_improve % 700 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score(sym)
            if sym_score < best_score:
                best_score = sym_score
                best_a = sym.copy()
                a = sym
                no_improve = 0

        # decay step size
        eta = max(min_eta, eta * decay)

        if best_score < target:
            break

    # after loop, return best_a as list floats
    return [float(x) for x in best_a]

import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast score for a vector already normalized to sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
