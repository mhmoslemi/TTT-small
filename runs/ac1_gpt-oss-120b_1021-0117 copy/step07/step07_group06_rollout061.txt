import time, math, numpy as np

def _project_simplex(v):
    """Project onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Construct a non‑negative step function whose autocorrelation
    (convolution with itself) has a small maximal entry.
    The construction builds a Sidon (Mian‑Chowla) set S ⊂ {0,…,n‑1}
    and puts equal mass on the indices in S.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)   # safety margin

    # length of the required vector
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2000                     # fallback length if no parent is present

    # ------------------------------------------------------------------
    # Build a maximal Sidon set (greedy / Mian‑Chowla construction)
    sums_seen = set()
    S = []
    for x in range(n):
        # check whether adding x would create a duplicate sum
        if (x + x) in sums_seen:
            continue
        conflict = False
        for y in S:
            if (x + y) in sums_seen:
                conflict = True
                break
        if conflict:
            continue
        # x is admissible – add it and record its sums
        for y in S:
            sums_seen.add(x + y)
        sums_seen.add(x + x)
        S.append(x)

        # early stop if we already hit the theoretical sqrt(2n) barrier
        if len(S) >= int(math.sqrt(2 * n)) + 2:
            # we have enough points; continue a few more just in case
            # but we do not need to exhaust the whole range
            pass

        if time.monotonic() >= deadline:
            break

    K = len(S)

    # ------------------------------------------------------------------
    # Build the candidate vector
    if K < 2:                     # very unlikely, fallback to uniform
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.zeros(n, dtype=np.float64)
        a[S] = 1.0 / K            # equal mass on the Sidon set

    # The vector already satisfies sum(a)=1 and a_i≥0.
    # Return it as a plain Python list of floats.
    return [float(v) for v in a]
