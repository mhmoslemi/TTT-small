def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    # enforce symmetry from the start
    a = (a + a[::-1]) * 0.5
    a = _project_simplex(a)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best)

    # Adam‑style momentum
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.5
    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution & active set -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # ----- gradient of the max‑convolution -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (math.sqrt(it) + 1.0)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- two candidate proposals -----
        cand_add = _project_simplex(a - update)
        exp_factor = np.exp(-step * grad)
        cand_mul = a * exp_factor
        s = cand_mul.sum()
        if s > 0:
            cand_mul = cand_mul / s
        else:
            cand_mul = a.copy()
        cand_mul = _project_simplex(cand_mul)

        # enforce symmetry on both candidates
        cand_add = (cand_add + cand_add[::-1]) * 0.5
        cand_add = _project_simplex(cand_add)
        cand_mul = (cand_mul + cand_mul[::-1]) * 0.5
        cand_mul = _project_simplex(cand_mul)

        cand_add_sc = _score(cand_add)
        cand_mul_sc = _score(cand_mul)

        if cand_add_sc < cand_mul_sc:
            cand, cand_sc = cand_add, cand_add_sc
        else:
            cand, cand_sc = cand_mul, cand_mul_sc

        cur_sc = _score(a)

        # ----- acceptance logic -----
        if cand_sc < best_sc:
            best, best_sc = cand.copy(), cand_sc
            a = cand
            no_improve = 0
        elif cand_sc < cur_sc:
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = cand

        # ----- periodic symmetrisation -----
        if it % 250 == 0:
            a = (a + a[::-1]) * 0.5
            a = _project_simplex(a)

        # ----- random redistribution when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)
                a = (a + a[::-1]) * 0.5
                a = _project_simplex(a)
            no_improve = 0

        # ----- full restart if really stuck -----
        if no_improve > 8000:
            rand_vec = rng.random(n, dtype=np.float64)
            rand_vec = _project_simplex(rand_vec)
            a = 0.5 * best + 0.5 * rand_vec
            a = _project_simplex(a)
            a = (a + a[::-1]) * 0.5
            a = _project_simplex(a)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    return [float(x) for x in best]
