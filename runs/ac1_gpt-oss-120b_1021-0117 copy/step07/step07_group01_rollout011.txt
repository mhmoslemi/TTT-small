import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a):
    """Score for a vector already normalized to sum = 1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ----- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()

        a = np.ones_like(a)
    a = a / a.sum()                     # simplex projection
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()

    # ----- hyper‑parameters -----------------------------------------------
    step0 = 0.9                         # initial step size
    decay = 0.0004                      # linear decay of step
    alpha0 = 1e-5                       # initial spectral‑smoothing weight
    alpha_max = 0.4                     # cap for smoothing
    target = 1.50274349                 # target bound

    it = 0
    no_improve = 0

    # ----- main loop -------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # current FFT and convolution
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- sub‑gradient (hard max mask) -----
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:

        # ----- gradient of max‑convolution -----

            break

    # return the best candidate found
