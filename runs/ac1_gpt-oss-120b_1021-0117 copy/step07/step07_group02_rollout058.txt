# Helper: Euclidean projection onto the probability simplex (sum = 1, x >= 0)
def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0.0)
    if v.sum() == 0:
        # uniform distribution if all zero (should not happen)
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]

        # ---- convolution a*a (full) ----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ---- soft‑max weights over all lags ----
        temp = max_val * max(temp_factor, min_temp)
        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(0.0)

        # ---- gradient of the weighted max‑convolution ----
        w_fft = np.fft.rfft(w, n=L)
        a_rev_fft = np.fft.rfft(cur[::-1], n=L)
        grad_full = np.fft.irfft(w_fft * a_rev_fft, n=L)
        # slice the part that corresponds to the original indices
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.1 / math.sqrt(it)          # base step size schedule
        delta = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - delta
        cand = _project_simplex(cand)

        # ---- evaluate candidate ----
        cand_sc = _score_vec(cand, L)

        # ---- accept if better; otherwise occasional Metropolis move ----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ---- periodic symmetrisation (optimal patterns are often symmetric) ----
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            # symmetry preserves the sum and non‑negativity
            sym_sc = _score_vec(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ---- random restart if stuck ----
        if no_improve > 5000:
            cur = _project_simplex(rng.random(n))
            cur_sc = _score_vec(cur, L)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # ---- temperature decay ----
        temp_factor = max(temp_factor * temp_decay, min_temp)

        # ---- early exit ----
        if best_sc < target:
            break

        # ---- occasional time check ----
        if it % 1000 == 0 and time.monotonic() >= deadline:
            break

    # Return the best‑found vector as plain Python floats
    return [float(x) for x in best_vec]
