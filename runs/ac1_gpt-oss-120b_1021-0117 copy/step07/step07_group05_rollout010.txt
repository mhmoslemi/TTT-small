def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)
    # Determine n and initial vector
    if 'height_sequence_1' in globals():
        init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    else:
        init = np.full(default_n, 1.0/default_n, dtype=np.float64)
    # Normalize to sum 1
    sum_init = init.sum()
    if sum_init == 0:
        init = np.full_like(init, 1.0/init.size)
    else:
        init = init / sum_init
    # Ensure sum=1
    # Evaluate
    best = init.copy()
    try:
        best_score = evaluate_sequence(best.tolist())
    except Exception:
        best_score = float('inf')
    # Evaluate uniform as baseline
    n = init.size
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    uni_score = evaluate_sequence(uniform.tolist())
    if uni_score < best_score:
        best = uniform.copy()
        best_score = uni_score

    # Simple projection onto simplex
    def project_simplex(v):
        # algorithm from Duchi et al.
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1
        rho = np.where(u - css / np.arange(1, v.size+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    rng = np.random.default_rng(seed)
    cur = best.copy()
    cur_score = best_score

    # Parameters
    max_iter = 1000000  # high limit but break with time
    # cooling schedule parameters
    start_frac = 0.5
    end_frac = 0.001
    # total iterations for schedule
    schedule_len = 200000
    it = 0
    while time.monotonic() < deadline and it < max_iter:
        it += 1
        # dynamic fraction
        if it < schedule_len:
            frac = start_frac - (start_frac - end_frac) * it / schedule_len
        else:
            frac = end_frac
        # Choose move: pick two distinct indices and transfer delta mass
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # delta is fraction of cur[i]
        delta = cur[i] * rng.random() * frac
        if delta <= 0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # Clip and project onto simplex (should already sum to 1)
        cand = project_simplex(cand)
        # Evaluate
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand
                best_score = cand_score
        # Occasionally (every 1000 it) perform subgradient step
        if (it % 1000) == 0:
            # subgradient for current best cur
            a = cur

    # Return best as list

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum == 1, v>=0)."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (no clipping needed if a is legal)."""
    s = a.sum()
    if s < 0.01:
        return float("inf")
    c = np.convolve(a, a)          # full mode, length 2*n-1
    return 2.0 * a.size * c.max() / (s * s)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound.  The routine works with any parent
    sequence ``height_sequence_1`` that may already be present in the
    global namespace; otherwise it falls back to a uniform distribution.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ---------------------------- initialise ----------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 0.0, 1000.0)
        total = a0.sum()
        if total == 0.0:
            a0 = None
        else:
            a0 = a0 / total                     # normalise to sum == 1
    else:
        a0 = None

    if a0 is None:
        n = 2000
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a0.size

    best = a0.copy()
    best_score = _objective(best)

    # also keep the uniform baseline (it is never worse than a bad start)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_score = _objective(uniform)
    if uni_score < best_score:
        best = uniform.copy()
        best_score = uni_score

    # ---------------------------- helpers ----------------------------
    rng = np.random.default_rng(seed)
    cur = best.copy()
    cur_score = best_score

    # schedule for the size of the random mass transfer
    start_frac = 0.3          # up to 30 % of the source mass at the beginning
    end_frac   = 0.001        # shrink to a tiny fraction later
    schedule_len = 40000      # number of iterations over which the schedule decays

    it = 0
    # main loop ---------------------------------------------------------
    while time.monotonic() < deadline and it < 200000:
        it += 1

        # ----- adaptive fraction for the move -----
        if it <= schedule_len:
            frac = start_frac - (start_frac - end_frac) * (it / schedule_len)
        else:
            frac = end_frac

        # ----- random mass transfer -----
        i, j = rng.integers(0, n, size=2)
        if i != j:
            delta = cur[i] * rng.random() * frac      # up to `frac` of source mass
            if delta > 0.0:
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)          # keep sum == 1, non‑negative
                cand_score = _objective(cand)

                if cand_score < cur_score:
                    cur = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best = cand
                        best_score = cand_score

        # ----- occasional sub‑gradient descent step -----
        if (it % 1000) == 0:
            a = cur
            conv = np.convolve(a, a)                    # full convolution
            k0 = int(conv.argmax())                     # index of the maximal entry

            idx = np.arange(n)

            grad = np.zeros_like(a)
            grad[valid] = 2.0 * a[k0 - idx[valid]]

            cand = _project_simplex(cand)

                cur_score = cand_score
                if cand_score < best_score:

        # ----- optional random restart if stagnating (very cheap) -----

            rand = rng.random(n)

        _ = evaluate_sequence(final)

    except Exception:
        return [float(x) for x in uniform]
