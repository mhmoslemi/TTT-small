def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # problem size & initial vector
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    else:
        n = 2000
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
    a0 = _project_simplex(a0)

    cur = a0.copy()
    cur_score = _score_vec(cur, n)
    best = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng(seed)
    step_frac = 0.2          # fraction of source mass to move
    it = 0

    while time.monotonic() < deadline and it < 300000:
        it += 1
        conv = np.convolve(cur, cur)
        k = int(conv.argmax())
        # gradient-like contribution vector g_i = a_{k-i} where defined
        g = np.zeros_like(cur)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] = cur[k - idx]

        # choose source index (large contribution) and target (small contribution)
        prod = g * cur
        if np.any(prod > 0):
            i = int(np.argmax(prod))
        else:
            i = int(rng.integers(0, n))
        j = int(np.argmin(g))
        if i == j:
            j = (i + 1) % n

        delta = cur[i] * step_frac
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)

        cand_score = _score_vec(cand, n)
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        # occasional random shake to escape plateaus
        if it % 1000 == 0:
            i2, j2 = rng.integers(0, n, size=2)
            if i2 != j2:
                d2 = cur[i2] * 0.05
                cur[i2] -= d2
                cur[j2] += d2
                cur = _project_simplex(cur)
                cur_score = _score_vec(cur, n)
                if cur_score < best_score:
                    best, best_score = cur.copy(), cur_score
        # slowly shrink step size
        step_frac = max(0.01, step_frac * 0.9999)

    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
