def propose_candidate(seed=42, budget_s=1100):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.1

    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2000
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(n, dtype=np.float64)
    # Ensure nonnegative
    a = np.maximum(a, 0.0)
    # Project onto simplex to have sum=1
    a = _project_simplex(a)

    # Precompute L for FFT
    L = 1 << ((2*n - 1).bit_length())

    def score(vec):
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())
    
    best = a.copy()
    best_score = score(best)
    current = a.copy()
    current_score = best_score

    # Temperature schedule
    init_temp = 0.1 * best_score  # maybe
    final_temp = 1e-4
    # maybe linear cooling based on elapsed time fraction
    while time.monotonic() < deadline:
        # compute temperature based on progress:
        elapsed = time.monotonic() - start
        frac = elapsed / (deadline - start)
        temp = init_temp * (1 - frac) + final_temp * frac

        # propose move
        # choose two distinct indices i != j
        i, j = rng.choice(n, size=2, replace=False)
        # amount to move
        delta = current[i] * rng.random() * 0.2  # 20% of i's mass
        if delta <= 0:
            continue
        new_vec = current.copy()
        new_vec[i] -= delta
        new_vec[j] += delta
        # Ensure nonnegativity (already ensured)
        # No need for simplex projection cause sum unchanged
        # Nonetheless, enforce tiny negative due to numeric errors
        new_vec = np.maximum(new_vec, 0.0)
        # compute score
        new_score = score(new_vec)
        if new_score < best_score:
            best = new_vec.copy()
            best_score = new_score
        # acceptance
        if new_score < current_score or rng.random() < math.exp(-(new_score - current_score)/temp):
            current = new_vec
            current_score = new_score

        # occasional symmetrization
        if rng.random() < 0.001:
            sym = (current + current[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = score(sym)
            if sym_score < best_score:
                best = sym.copy()
                best_score = sym_score
                current = sym
                current_score = sym_score

    return [float(x) for x in best]
```

import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    # find rho
    rho_arr = u - css / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a, L=None):
    n = a.shape[0]
    if L is None:
        L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)

    # initialization
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score_vec(best, L)

    # Adam optimizer state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.6
    it = 0
    beta1_pow = 1.0
    beta2_pow = 1.0
    last_improve = 0

    # main loop
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        # FFT of current a
        a_fft = np.fft.rfft(a, n=L)

        # linear convolution a*a
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # sub‑gradient mask for max entries
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # gradient of max‑convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = step0 / (math.sqrt(it) + 1.0)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - update
        cand = _project_simplex(cand)

                cand[j] += delta

                cand_sc = _score_vec(cand, L)

        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            a = cand
            last_improve = it
        elif cand_sc < _score_vec(a, L):
            a = cand
        else:

            frac = elapsed / (deadline - start)
            temp = (0.5 * best_sc) * (1.0 - frac) + 1e-4 * frac
            if rng.random() < math.exp(-(cand_sc - best_sc) / (temp + 1e-12)):

            beta2_pow = 1.0
            last_improve = it

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]

        theta = 0.0
