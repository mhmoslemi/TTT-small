def compute_score(seq):
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

            best_vec = base_seq.copy()
