import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score_vec(v):
    """Score 2*n*max(conv(v,v)) when sum(v)=1."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.35)  # safety margin

    # ---------- initialization ----------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a = _project_simplex(np.maximum(a0, 0.0))
    else:
        n = rng.integers(2000, 3000)
        a = rng.random(n)
        a = _project_simplex(a)

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    cur = a.copy()
    cur_score = _score_vec(cur)

    best = cur.copy()
    best_score = cur_score

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 0.5

    # algorithmic parameters
    top_k = 7               # number of largest convolution entries to weight
    no_improve = 0
    it = 0
    target = 1.50274349

    # ---------- main loop ----------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution and top‑k indices -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        # top‑k (largest) indices
        if top_k < conv.size:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
        else:
            idxs = np.arange(conv.size)
        w = np.zeros_like(conv)
        w[idxs] = 1.0 / idxs.size

        # ----- gradient of weighted sum -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx_slice = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx_slice]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)

        cand = cur - step * m_hat
        cand = _project_simplex(cand)

        # enforce symmetry each iteration
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_score = _score_vec(cand)

        # ----- acceptance -----
        if cand_score < cur_score:

            no_improve = 0
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score

                if cur_score < best_score:
                    best, best_score = cur.copy(), cur_score

        # early exit if target achieved
        if best_score < target:

    # ---------- return ----------
    return [float(x) for x in best]
