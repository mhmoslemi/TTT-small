        n = base.size

        n = 2000
        base = None

def _run_descent(a0,...):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    lr = 0.5  # initial learning rate
    cur_score = objective(a)
    no_improve_counter = 0
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        max_val, grad = _conv_max_and_grad(a, L)
        cur_score = 2.0 * n * max_val
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_vec = a.copy()
            no_improve_counter = 0
        else:
            no_improve_counter += 1

        # gradient of objective
        g = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # candidate
        cand = a - step
        # optional symmetrization
        if it % 200 == 0:
            cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_max, _ = _conv_max_and_grad(cand, L)
        cand_score = 2.0 * n * cand_max

        # acceptance: if better accept, else maybe accept with small prob
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
        else:
            # Simulated annealing: temperature decreasing
            temp = max(0.01, 0.5 / np.sqrt(it))
            if rng.random() < np.exp(-(cand_score - cur_score) / temp):
                a = cand
                cur_score = cand_score
                # reset momentum maybe?
                m.fill(0.0)
                v.fill(0.0)

        # If stagnated, random perturb
        if no_improve_counter >= 3000:
            sigma = 0.05 / np.sqrt(it)
            a = a + sigma * rng.standard_normal(size=a.shape)
            a = _project_simplex(a)

        # early break if global target reached (we need to pass global target to break)
        if cur_score < target:
            best_local_score = cur_score
            best_local_vec = a.copy()
            break

    return best_local_vec, best_local_score

import time
import numpy as np

# -------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# -------------------------------------------------------------
def _conv_max_and_grad(a, L):
    """
    Returns (max_convolution, gradient) for a*a (full convolution).
    a is assumed non‑negative. Gradient is with respect to a.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

# -------------------------------------------------------------
def _run_adam_descent(a0, L, n, rng, deadline, target, max_iters=20000):
    """
    Adam‑style descent on the scale‑invariant objective.
    Returns the best vector found in this run and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5                     # base learning rate
    best_local_score = np.inf
    best_local_vec = a.copy()
    no_imp = 0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current objective
        max_val, grad = _conv_max_and_grad(a, L)
        cur_score = 2.0 * n * max_val

        # store best locally
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_vec = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        # gradient of the objective
        g = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # candidate move
        cand = a - step

        # occasional symmetrisation
        if it % 200 == 0:
            cand = (cand + cand[::-1]) * 0.5

        # onto simplex
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_max, _ = _conv_max_and_grad(cand, L)
        cand_score = 2.0 * n * cand_max

        # acceptance
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
        else:
            # simulated‑annealing style acceptance
            temp = max(0.01, 0.5 / np.sqrt(it))
            if rng.random() < np.exp(-(cand_score - cur_score) / temp):
                a = cand
                cur_score = cand_score
                m.fill(0.0)
                v.fill(0.0)

        # perturb if stuck
        if no_imp >= 3000:
            sigma = 0.05 / np.sqrt(it)
            a = a + sigma * rng.standard_normal(size=n)
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            no_imp = 0

        # early exit if global target already reached
        if cur_score < target:
            best_local_score = cur_score
            best_local_vec = a.copy()
            break

    return best_local_vec, best_local_score

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # length and base vector
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    else:
        n = 2000
        base = None

    L = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1
    target = 1.50274349

    best_score = np.inf
    best_vec = None
    used_parent = False

    while time.time() < deadline:
        # ----- initial vector ------------------------------------
        if not used_parent and base is not None:
            a0 = base.copy()
            used_parent = True
        else:
            a0 = rng.exponential(1.0, size=n)   # Dirichlet‑like start
        a0 = _project_simplex(np.maximum(a0, 0.0))

        # ----- run descent ---------------------------------------
        vec, sc = _run_adam_descent(a0, L, n, rng, deadline, target)

        # ----- update global best --------------------------------
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()
            if best_score < target:
                break

    # fallback if nothing succeeded
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    # return as plain Python list
    return [float(x) for x in best_vec]
