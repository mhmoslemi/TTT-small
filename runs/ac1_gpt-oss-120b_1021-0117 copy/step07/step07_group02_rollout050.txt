def _grad_maxconv(v, eps):
    n = v.shape[0]
    L = 1 << ((3*n-2).bit_length())
    a_fft = np.fft.rfft(v, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    # Tolerance epsilon: treat values within eps*max_val as active
    if max_val > 0:
        thresh = max_val - eps * max_val
    else:
        # all zeros
        thresh = max_val
    I = (conv >= thresh).astype(np.float64)  # shape 2n-1
    # reversed indicator for convolution (correlation)
    I_rev = I[::-1]
    I_rev_fft = np.fft.rfft(I_rev, n=L)

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    """Score assuming sum(a) may be ≠1 (the evaluator uses the same formula)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    sum_a = a.sum()
    if sum_a == 0:
        return float('inf')
    return float(2.0 * n * max_val / (sum_a * sum_a))

def _grad_maxconv(a, eps):
    """Sub‑gradient of the max‑convolution (scaled by 2). Returns (grad, max_val)."""
