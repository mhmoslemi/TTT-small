def conv_max(v):
    n = v.shape[0]
    L = 1 << ((2*n - 2).bit_length()) # Actually we need length >= 2n-1 for full conv; using L = next power of 2 >= 2*n -1.
    fft = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fft * fft, n=L)
    return np.max(conv[:2*n-1])

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    
    if 'height_sequence_1' in globals():

    cur_sc = best_sc

        if time.monotonic() > deadline - 5:

            break

        # Build A_ub sparse matrix
        offsets = np.arange(-(n-1), n, dtype=int)

            else:

        # Solve LP

        try:
            res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=M*np.ones(2*n-1), bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(1, int(deadline - time.monotonic()))})
        except Exception:

            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc

                    best_vec, best_sc = cand.copy(), cand_sc
            # continue loop

        g = res.x
        # Mix cur and g
        # Choose mixing weight: e.g., 0.5 or sqrt to reduce M gradually
        alpha = 0.5

        new = _project_simplex(np.maximum(new, 0.0))
        new_sc = _score(new)

        if new_sc < cur_sc:
            cur, cur_sc = new, new_sc
            if new_sc < best_sc:
                best_vec, best_sc = new.copy(), new_sc

        # early exit
        if best_sc < target_sc:

        # occasional random perturb (if no improvement in a few rounds)
        lp_iter += 1
        if lp_iter % 5 == 0:

import time, math, numpy as np

# --------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def _project_simplex(v):
    """Project a vector onto the simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

# Fast evaluation of the AC1 upper bound for a vector whose sum=1
def _score(a):
    """Return 2*n*max(conv) for a (sum‑normalised) vector a."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    Return a list of non‑negative coefficients (length = parent length if
    `height_sequence_1` exists, otherwise ~2500) whose AC1 upper bound is as
    small as possible within the time budget.

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    if "height_sequence_1" in globals():

    # ---- optimisation state --------------------------------------------
    momentum = np.zeros_like(cur)
    beta = 0.9                 # momentum coefficient
    step0 = 0.6                # initial step magnitude
    it = 0
    target_sc = 1.50274349    # early‑stop threshold

    # ---- main loop ------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution and current score
        L = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        # gradient from the (hard) max entry
        k = int(np.argmax(conv))
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(cur)
        grad[mask] = 2.0 * cur[idx[mask]]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size (decays like 1/sqrt(it))
        step = step0 / math.sqrt(it)

        # ----- try a move -------------------------------------------------
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score(cand)

        # backtrack if not improving
        if cand_sc >= cur_sc:

            for _ in range(3):            # a few reductions

            cur_sc = cand_sc
