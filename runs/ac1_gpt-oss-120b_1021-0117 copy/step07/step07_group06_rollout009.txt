        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score_vec(a):
    """Score = 2*n*max(conv(a,a)) / (sum(a)^2).  a is a NumPy array."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s <= 0:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for returning the result
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        cur = rng.random(2000, dtype=np.float64) + 1e-3
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_sc = _score_vec(best)
    cur_sc = best_sc

    # Adam optimizer state
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    lr = 0.2
    beta1_pow = beta1
    beta2_pow = beta2

    # soft‑max schedule for approximating the max
    alpha0, alpha_max = 1.0, 80.0

    it = 0
    no_improve = 0

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ---------- autocorrelation ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ---------- soft‑max weighting ----------
        alpha = min(alpha_max, alpha0 * math.log1p(it))
        # numerically stable exponentiation
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if not np.isfinite(w_sum) or w_sum == 0.0:
            # fallback to an indicator of the maximal entries
            w = (conv >= max_val - 1e-12).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ---------- gradient (matrix‑free via FFT) ----------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        delta = -lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = _project_simplex(cur + delta)

        # ---------- evaluate candidate ----------
        cand_sc = _score_vec(cand)

        # keep global best
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc

        # simple greedy acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional probabilistic acceptance to escape plateaus
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc
                no_improve = 0

        # ---------- periodic symmetrisation ----------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < best_sc:
                best, best_sc = sym.copy(), sym_sc
                cur, cur_sc = sym, sym_sc
                no_improve = 0
            elif sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc

        # ---------- random local redistribution when stuck ----------
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta_mass = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta_mass
                tmp[j] += delta_mass
                tmp = _project_simplex(tmp)
                tmp_sc = _score_vec(tmp)
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                elif tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0

        # ---------- early exit ----------
        if best_sc < 1.50274349:
            break

        # update bias‑correction powers for next iteration
        beta1_pow *= beta1
        beta2_pow *= beta2

    # ------------------------------------------------------------------
    # Return the best sequence found as plain Python floats
    return [float(x) for x in best]
