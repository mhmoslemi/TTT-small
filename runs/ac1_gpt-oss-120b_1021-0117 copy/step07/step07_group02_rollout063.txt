def optimize(initial_vec, time_budget, deadline):
    cur = initial_vec.copy()
    n = cur.size; L = next_pow2(2*n-1)
    best_cur = cur.copy()
    best_score = _score_vec(cur)
    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 1.0
    temp_factor = 1e-2
    temp_decay = 0.9995
    min_temp_factor = 1e-9 (or lower)
    it = 0
    while time.monotonic() < deadline:
        it += 1
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()
        # compute gradient via softmax weighting
        temp = max_val * max(temp_factor, min_temp_factor)
        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(0.0)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]
        # Update momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        # step size
        step = step0 / math.sqrt(it)
        # Candidate
        cand = cur - step * momentum
        # enforce symmetry
        cand = (cand + cand[::-1]) * 0.5
        # project onto simplex
        cand = _project_simplex(cand)
        # compute score
        cand_score = _score_vec(cand)
        # Accept if better else maybe accept with prob
        if cand_score < best_score:
            best_score = cand_score
            best_cur = cand.copy()
        # Simple acceptance: if cand_score < cur_score, accept; else keep cur
        cur_score = _score_vec(cur) # compute earlier? we have max_val and maybe score.
        # but we can compute cur_score = 2*n*max_val (since sum =1) -> 
        cur_score = 2.0 * n * max_val
        # Accept if cand_score <= cur_score else keep cur
        if cand_score <= cur_score:
            cur = cand
        # else keep cur unchanged

        # Temperature decay
        temp_factor = max(temp_factor * temp_decay, min_temp_factor)
        # Optionally early break if best_score < target or it >= max_iters or time left low

    return best_cur, best_score

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    def init_vector():
        if "height_sequence_1" in globals():
            a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        else:
            a0 = rng.random(2500, dtype=np.float64)
        # ensure nonnegative
        a0 = np.maximum(a0, 0.0)
        # project onto simplex (sum=1)
        return _project_simplex(a0)

    n = None
    best_vec = None
    best_score = float('inf')

    # Precompute power-of-two length for FFT
    while time.monotonic() < deadline:
        # start a new run
        cur = init_vector()
        if n is None:
            n = cur.size
            L = 1 << ((2 * n - 1).bit_length())
        # track best during run
        # optimizer state
        momentum = np.zeros_like(cur)
        beta = 0.95
        step0 = 1.0
        temp_factor = 1e-2
        temp_decay = 0.998
        min_temp_factor = 1e-9
        cur_score = _score_vec(cur)  # using own scoring
        it = 0
        no_improve = 0
        # limit per run by a small time slice
        run_deadline = time.monotonic() + 5.0  # each run max 5 sec
        while time.monotonic() < deadline and time.monotonic() < run_deadline:
            it += 1
            a_fft = np.fft.rfft(cur, n=L)

            conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
            max_val = conv_full.max()

            # Weighted gradient (softmax)
            temp = max_val * max(temp_factor, min_temp_factor)
            w = np.exp((conv_full - max_val) / temp)
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
            else:
                w.fill(0.0)
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # momentum & step
            momentum = beta * momentum + (1.0 - beta) * grad
            step = step0 / math.sqrt(it + 1.0)
            cand = cur - step * momentum

            # enforce symmetry
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

            cand_score = _score_vec(cand)
            # acceptance
            if cand_score < cur_score:
                cur, cur_score = cand, cand_score
                no_improve = 0
                if cand_score < best_score:
                    best_score, best_vec = cand_score, cand.copy()
            else:
                no_improve += 1
                # occasional random perturbation
                if no_improve % 500 == 0:
                    # swap random mass
                    i = rng.integers(0, n)
                    j = rng.integers(0, n)
                    if i != j:
                        delta = cur[i] * 0.5 * rng.random()
                        tmp = cur.copy()
                        tmp[i] -= delta
                        tmp[j] += delta
                        tmp = _project_simplex((tmp + tmp[::-1]) * 0.5)
                        tmp_score = _score_vec(tmp)
                        if tmp_score < cur_score:
                            cur, cur_score = tmp, tmp_score
                            no_improve = 0
                            if tmp_score < best_score:
                                best_score, best_vec = tmp_score, tmp.copy()
            # temperature decay
            temp_factor = max(temp_factor * temp_decay, min_temp_factor)

            # early exit if good
            if best_score < 1.50274349:
                break

        if best_score < 1.50274349:
            break

    # fallback if no vector found (should not happen)
    if best_vec is None:

    # return as Python list
    return [float(x) for x in best_vec]
