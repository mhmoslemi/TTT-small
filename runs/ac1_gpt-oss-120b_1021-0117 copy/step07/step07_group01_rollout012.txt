def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2000, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best)

    # Adam vars
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    step0 = 0.8
    it = 0
    no_improve = 0

    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # compute convolution of a with itself
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        # softmax weights
        alpha = 0.5 + min(5.0, it * 5e-4)  # start .5, increase slowly capped 5
        # compute w = exp(alpha*(conv - max_val)), then normalize
        w_raw = np.exp(alpha * (conv - max_val))
        w_sum = w_raw.sum()
        if w_sum == 0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # gradient via convolution of a and w_rev
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (np.sqrt(it) + 1.0)
        # add epsilon to denominator for safety
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        # additive candidate
        cand_add = _project_simplex(a - update)

        # multiplicative candidate (exponential)
        exp_factor = np.exp(-step * m_hat)
        cand_mul = a * exp_factor
        s = cand_mul.sum()
        if s > 0:
            cand_mul = cand_mul / s
        else:
            cand_mul = a.copy()

        # evaluate candidates
        cand_add_sc = _score(cand_add)
        cand_mul_sc = _score(cand_mul)

        if cand_add_sc < cand_mul_sc:
            cand = cand_add
            cand_sc = cand_add_sc
        else:
            cand = cand_mul
            cand_sc = cand_mul_sc

        # acceptance
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            a = cand
            no_improve = 0
        elif cand_sc < _score(a):
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - best_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = cand

        # symmetrisation
        if it % 300 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
                a = sym
                no_improve = 0

        # random perturb when stuck
        if no_improve > 2000:
            # random redistribution
            i = rng.integers(0, n)
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                a = tmp
            no_improve = 0

        if no_improve > 5000:
            # restart from combination of best and random
            rand_vec = rng.random(n)
            rand_vec = _project_simplex(rand_vec)
            a = 0.5 * best + 0.5 * rand_vec
            a = _project_simplex(a)
            no_improve = 0

        if best_sc < target:
            break

    return [float(x) for x in best]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)               # enforce sum=1
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    # Adam‑like state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.6
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution a*a -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = float(2.0 * n * max_val)

        # ----- soft‑max weighting of convolution entries -----
        # alpha grows from 0.5 up to ~5.0
        alpha = 0.5 + min(4.5, it * 0.0005)
        w_raw = np.exp(alpha * (conv - max_val))   # max entry gets weight 1
        w_sum = w_raw.sum()
        w = w_raw / w_sum if w_sum != 0 else np.full_like(w_raw, 1.0 / w_raw.size)

        # ----- gradient of weighted max‑convolution -----
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (math.sqrt(it) + 1.0)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- additive candidate (projected) -----
        cand_add = _project_simplex(a - update)

        # ----- multiplicative candidate (exponential) -----
        exp_factor = np.exp(-step * m_hat)
        cand_mul = a * exp_factor
        s = cand_mul.sum()
        if s > 0:
            cand_mul = cand_mul / s
        else:
            cand_mul = a.copy()

        # ----- evaluate candidates -----
        cand_add_sc = _score(cand_add)
        cand_mul_sc = _score(cand_mul)

        if cand_add_sc < cand_mul_sc:
            cand, cand_sc = cand_add, cand_add_sc
        else:
            cand, cand_sc = cand_mul, cand_mul_sc

        # ----- acceptance logic -----
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            a = cand
            cur_sc = cand_sc
            no_improve = 0
        elif cand_sc < cur_sc:
            a = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - best_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = cand

        # ----- occasional symmetrisation -----
        if it % 300 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
                a = sym
                cur_sc = sym_sc
                no_improve = 0

        if no_improve > 2000:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                a = tmp
                cur_sc = _score(a)
            no_improve = 0

        if no_improve > 5000:
            # restart from a blend of the best and a fresh random simplex point
            rand_vec = rng.random(n, dtype=np.float64)
            rand_vec = _project_simplex(rand_vec)
            a = 0.5 * best + 0.5 * rand_vec
            a = _project_simplex(a)
            cur_sc = _score(a)
            no_improve = 0

        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # final lightweight local search (few random swaps)
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.1
        tmp = best.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        sc = _score(tmp)
        if sc < best_sc:
            best = tmp
            best_sc = sc

    # ------------------------------------------------------------------
    return [float(x) for x in best]
