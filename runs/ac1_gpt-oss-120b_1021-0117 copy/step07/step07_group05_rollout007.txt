    minimizes the AC1 upper bound as measured by ``evaluate_sequence``.
    The routine works with any pre‑existing ``height_sequence_1``; otherwise it
    falls back to a uniform distribution of the appropriate length.
    """
    start = time.monotonic()
    # keep a safety margin for the outer harness
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ----------------------------------------------------------------------
    # Problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000                     # reasonable default if no parent is given

    # ----------------------------------------------------------------------
    # Helper: evaluate safely
    def _eval(vec):
        try:
            return evaluate_sequence(vec.tolist())
        except Exception:
            return float("inf")

    # ----------------------------------------------------------------------
    # Candidate 1: uniform distribution (sum == 1)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_vec = uniform
    best_score = _eval(uniform)

    # ----------------------------------------------------------------------
    # Candidate 2: parent sequence, if supplied
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        # Clip to the allowed range the evaluator will apply anyway
        parent = np.clip(parent, 0.0, 1000.0)
        parent_score = _eval(parent)
        if parent_score < best_score:
            best_score = parent_score
            best_vec = parent

    # ----------------------------------------------------------------------
    # Very modest stochastic refinement (optional, respects time budget)
    rng = np.random.default_rng(seed)
    cur = best_vec.copy()
    cur_score = best_score
    # Simple projection onto the simplex (sum == 1, non‑negative)
    def _project_simplex(v):
        v = np.asarray(v, dtype=np.float64)
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    # limit the number of attempts so we stay well within the time budget
    max_iters = 3000
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * 0.15   # move up to 15 % of source mass
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_score = _eval(cand)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand

    # ----------------------------------------------------------------------
    # Return the best vector found as a plain Python list of floats.
    return [float(x) for x in best_vec]
