def _score_fft(a, n, L):
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    return float(2.0 * n * conv.max() / (s * s))
```

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # tiny correction to enforce exact sum=1
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast evaluation of the AC1 upper bound (assumes a>=0)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the sandbox
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine problem size and FFT length
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2000
    L = 1 << ((2 * n - 1).bit_length())  # next power of two

    xs = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Build a diverse set of deterministic initial candidates
    candidates = []

    # 1) Uniform
    candidates.append(np.full(n, 1.0 / n))

    # 2) Symmetric triangular bump
    tri = np.minimum(xs + 1, n - xs)
    candidates.append(tri / tri.sum())

    # 3) Sine bump
    sine = np.sin(np.pi * (xs + 1) / (n + 1))
    candidates.append(sine / sine.sum())

    # 4) Raised cosine (Hann window)
    bump = (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2
    candidates.append(bump / bump.sum())

    # 5) Exponential decays (left‑hand)
    for alpha in (0.002, 0.005, 0.010):
        exp = np.exp(-alpha * xs)
        candidates.append(exp / exp.sum())

    # 6) Power laws
    for p in (0.3, 0.7, 1.2):
        pw = (xs + 1) ** (-p)
        candidates.append(pw / pw.sum())

    # 7) Symmetric Beta‑like shapes (a_i ∝ (i+1)^{a-1}(n-i)^{a-1})
    for a in (0.4, 0.7, 1.0, 1.5, 2.0):
        beta = (xs + 1) ** (a - 1) * (n - xs) ** (a - 1)
        candidates.append(beta / beta.sum())

    # 8) Random Dirichlet with varying concentration
    for conc in (0.2, 0.5, 1.0, 2.0):
        dirich = rng.dirichlet(np.full(n, conc))
        candidates.append(dirich)

    # 9) Parent if supplied
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.sum() > 0:
            candidates.append(parent / parent.sum())

    # ------------------------------------------------------------------
    # Pick the best deterministic start
    best_vec = None
    best_sc = float('inf')
    for vec in candidates:
        sc = _score_fft(vec, n, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # ------------------------------------------------------------------
    # Adam optimiser state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.7                # base learning rate
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution and active‑set mask -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
        else:
            w = mask.astype(np.float64) / cnt

        # ----- sub‑gradient of max‑convolution -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step_dir = step0 * m_hat / (np.sqrt(v_hat) + eps)

        local_best_vec = cur
        local_best_sc = cur_sc

            cand_sc = _score_fft(cand, n, L)
            if cand_sc < local_best_sc:
                local_best_sc = cand_sc
                local_best_vec = cand

        if local_best_sc < cur_sc:
            cur, cur_sc = local_best_vec, local_best_sc
            no_improve = 0
        else:

            # occasional simulated‑annealing acceptance
            prob = math.exp(-(local_best_sc - cur_sc) / (step0 + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = local_best_vec, local_best_sc
                no_improve = 0

        # ----- update global best -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()

        # ----- periodic symmetrisation -----
        if it % 150 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ----- small random mass swaps when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)

                tmp = _project_simplex(tmp)
                tmp_sc = _score_fft(tmp, n, L)

                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

        # ----- early stop if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
