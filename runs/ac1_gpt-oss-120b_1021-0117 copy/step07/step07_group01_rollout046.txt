import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n+1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.nonzero(rho_arr)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):

    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    cond = u - css / np.arange(1, n + 1) > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a):
    """Score for a vector already normalized to sum=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
                a = sym
                no_improve = 0

        # ----- random redistribution when stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                a = tmp
            no_improve = 0

        # ----- restart from a mixed random vector -----
        if no_improve > 4000:
            rand_vec = rng.random(n, dtype=np.float64)
            rand_vec = _project_simplex(rand_vec)
            a = 0.5 * best + 0.5 * rand_vec
            a = _project_simplex(a)
            no_improve = 0

        # ----- temperature schedule -----
        beta = min(beta * beta_growth, beta_max)

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    return [float(x) for x in best]
