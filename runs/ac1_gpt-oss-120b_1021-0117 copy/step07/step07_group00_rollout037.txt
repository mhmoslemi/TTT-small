import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum = 1, v >= 0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast evaluation of 2*n*max(conv) for a vector a with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin for the harness timeout
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---------- initialization ----------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback size – will be overwritten by actual length later
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # ---------- Adam optimiser ----------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2

    # temperature for soft‑max weighting of the max‑term
    temp = None
    decay = 0.997
    min_temp_factor = 1e-4

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # ----- convolution -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        w /= w_sum

        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
