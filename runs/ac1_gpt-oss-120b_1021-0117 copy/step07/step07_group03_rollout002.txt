        w = mask
    else:

        w = np.exp(alpha * (conv - max_val))

import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_grad(a, L, alpha=0.0):
    """
    Compute max of full convolution a*a and a gradient approximation.
    If alpha>0 use a soft‑max weighting; otherwise use an active‑set mask.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)

    # full convolution (real part)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    max_val = conv.max()

    if alpha <= 0.0:
        # mask of active indices (within tiny epsilon of the max)
        eps = max_val * 1e-12 + 1e-15
        w = (conv >= max_val - eps).astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
    else:
        # soft‑max weighting, shifted to avoid overflow
        w = np.exp(alpha * (conv - max_val))
        s = w.sum()
        if s > 0:
            w /= s
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

    # gradient = 2 * (a * w_rev) convolution, extracted for original indices
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

    rng = np.random.default_rng(seed)
    start = time.time()
    # small safety margin to guarantee returning before hard timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initial vector -------------------------------------------------

    L = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1

    beta2 = 0.999
    eps = 1e-12
    lr = 0.25          # base learning rate

    max_val, _ = _conv_max_and_grad(cur, L, alpha=0.0)
    best_sc = 2.0 * n * max_val
    best_vec = cur.copy()
    cur_sc = best_sc

    # algorithmic parameters
    it = 0
    stagnation = 0
    target = 1.50274349
    alpha = 0.0
    alpha_max = 1e5
    alpha_inc = 500.0          # increase per iteration until cap

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # gradually increase soft‑max temperature for a smoother → sharper transition
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha + alpha_inc)

        # gradient of current point (with current alpha)
        max_val, grad = _conv_max_and_grad(cur, L, alpha=alpha)
        cur_sc = 2.0 * n * max_val

        # Adam bias‑corrected moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # parameter update (gradient descent)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = cur - step
        cand = _project_simplex(cand)

        # enforce (optional) symmetry every 250 iterations
        if it % 250 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        cand_max, _ = _conv_max_and_grad(cand, L, alpha=alpha)
        cand_sc = 2.0 * n * cand_max

        # acceptance rule (deterministic improvement, else occasional uphill)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            stagnation = 0
        else:

            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                stagnation = 0
            else:

        # keep global best

            stagnation = 0

                stagnation = 0
            else:
                # reset momentum to escape flat region
                m.fill(0.0)
                v.fill(0.0)
                stagnation = 0

        # early exit if target achieved
        if best_sc < target:
            break

    # Return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
