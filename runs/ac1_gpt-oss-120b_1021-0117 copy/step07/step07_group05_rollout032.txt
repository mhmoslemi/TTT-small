import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of vector v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # problem size

        n = len(height_sequence_1)
    else:
        n = 2000

    rng = np.random.default_rng(seed)

    best_vec = uniform
    best_score = evaluate_sequence(best_vec.tolist())

    # try provided parent
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        # ensure not all zeros
        if parent.size and parent.sum() > 0:
            parent_score = evaluate_sequence(parent.tolist())
            if parent_score < best_score:
                best_score = parent_score
                best_vec = parent

    # helper to evaluate and possibly update best
    def _maybe_update(vec, cur_score):
        nonlocal best_vec, best_score
        if cur_score < best_score:
            best_score = cur_score
            best_vec = vec
        return cur_score

    # main search
    # start from current best
    cur = best_vec.copy()
    cur_score = _maybe_update(cur, best_score)

    # parameters
    max_total_iters = 50000  # safety cap

import time
import numpy as np

def _project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w
