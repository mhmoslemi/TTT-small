import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as measured by ``evaluate_sequence``.
    The strategy builds simple power‑law sequences a_i = (i+1)^p,
    rescales them so the maximum entry does not exceed 1000, and keeps the
    best candidate (including a uniform baseline and the supplied parent
    sequence if present).
    """
    start = time.monotonic()
    # keep a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # --------------------------------------------------------------
    # problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000                     # reasonable fallback

    best_vec = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # candidate 0: uniform distribution (sum==1, each entry = 1/n)
    try:
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        score = evaluate_sequence(uniform.tolist())
        best_vec, best_score = uniform, score
    except Exception:
        pass

    # --------------------------------------------------------------
    # candidate 1..k: power‑law shapes a_i = (i+1)^p, p = 2,…,5
    for p in (2, 3, 4, 5):
        if time.monotonic() >= deadline:
            break
        # build the raw power‑law vector
        base = np.arange(1, n + 1, dtype=np.float64) ** p
        max_val = base[-1]                    # = n**p
        # rescales so that the maximal entry is at most 1000
        scale = 1.0
        if max_val > 1000.0:
            scale = 1000.0 / max_val
        candidate = base * scale
        try:
            score = evaluate_sequence(candidate.tolist())
            if score < best_score:
                best_vec, best_score = candidate, score
        except Exception:
            # any numerical issue – just ignore this candidate
            continue

    # --------------------------------------------------------------
    # candidate parent (if supplied) – after clipping to allowed range
    if "height_sequence_1" in globals():
        try:
            parent = np.asarray(height_sequence_1, dtype=np.float64)
            parent = np.clip(parent, 0.0, 1000.0)
            score = evaluate_sequence(parent.tolist())
            if score < best_score:
                best_vec, best_score = parent, score
        except Exception:
            pass

    # --------------------------------------------------------------
    # final fallback – should never happen, but keep a safe non‑empty list
    if best_vec is None:
        best_vec = np.full(n, 1.0, dtype=np.float64)

    # return a plain Python list of floats
    return [float(x) for x in best_vec]
