def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # leave buffer

    # Initialization
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)  # fallback
    # Ensure nonnegative and project to simplex sum=1
    cur = np.maximum(cur, 0.0)
    cur = cur / np.maximum(cur.sum(), 1e-12)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())  # next power of two

    # Helper projection to simplex (exact)
    def _project_simplex(v):
        v = np.asarray(v, dtype=np.float64)
        n = v.size
        # shift to make nonnegative (v may have negative)
        # use algorithm: sort descending
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        return w

    cur = _project_simplex(cur)

    # best tracking
    best_vec = cur.copy()
    # compute initial normalized score (since sum=1)
    a_fft = np.fft.rfft(cur, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    best_norm = 2.0 * n * conv.max()
    best_score = float(best_norm)  # same as evaluate_sequence because sum=1

    # optimizer states
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.15
    t = 0

    # temperature
    temp = None
    temp_factor_initial = 0.15
    temp_decay = 0.998
    temp_min_factor = 5e-4

    # counters for stagnation
    no_improve = 0
    restart_limit = 5000  # steps without improvement before restart

    # main loop
    while time.time() < deadline:
        t += 1
        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        # update temperature
        if temp is None:
            temp = max_val * temp_factor_initial
        else:
            temp = max(temp * temp_decay, max_val * temp_min_factor)
        # softmax weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = cur - step

        cand = _project_simplex(np.maximum(cand, 0.0))
        # compute normalized score for candidate
        cand_fft = np.fft.rfft(cand, n=L)

        cand_norm = 2.0 * n * cand_conv.max()

        if cand_norm < best_norm * 1.001:  # slight tolerance for acceptance

        else:

import time, math, numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v):
    """Project a vector onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]                     # descending sort
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the AC1 bound.
    The routine runs for at most ``budget_s`` seconds and keeps the best
    feasible vector found.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # leave a small margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # --------------------------------------------------------
    # initial vector (use the provided parent if available)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(globals()["height_sequence_1"], dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())       # next power of two for FFT

    # --------------------------------------------------------
    # optimizer state (Adam)
    # --------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.4                                   # base learning rate
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # --------------------------------------------------------
    # temperature schedule for soft‑max of the convolution
    # --------------------------------------------------------
    temp = None
    temp_factor_initial = 0.30
    temp_decay = 0.999
    temp_min_factor = 1e-4

    # --------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------
    # current objective (sum of entries is 1, so denominator = 1)
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    cur_norm = 2.0 * n * conv_full[: 2 * n - 1].max()
    best_norm = cur_norm
    best_vec = cur.copy()

    no_improve = 0
    restart_limit = 8000
    target = 1.50274349
    it = 0

    # --------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- current convolution & temperature -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_norm = 2.0 * n * max_val

        if temp is None:
            temp = max_val * temp_factor_initial
        else:
            temp = max(temp * temp_decay, max_val * temp_min_factor)

        # ----- soft‑max weights -----
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:                     # numerical fallback
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of the soft‑max max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(np.maximum(cand, 0.0))

        # ----- evaluate candidate -----
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_norm = 2.0 * n * cand_conv.max()

        # ----- accept / reject -----
        if cand_norm < cur_norm:
            cur = cand
            cur_norm = cand_norm
        else:
            prob = math.exp(-(cand_norm - cur_norm) / (lr + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_norm = cand_norm

        # ----- update best -----
        if cur_norm < best_norm:
            best_norm = cur_norm
            best_vec = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- periodic symmetrisation -----
        if it % 700 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]

            if sym_norm < best_norm:
                best_norm = sym_norm

                cur = sym
                cur_norm = sym_norm
                m.fill(0.0)
                v.fill(0.0)
                it = 0

        # ----- random redistribution when stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(np.maximum(tmp, 0.0))
            tmp_fft = np.fft.rfft(tmp, n=L)
            tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
            tmp_norm = 2.0 * n * tmp_conv.max()
            if tmp_norm < best_norm:
                best_norm = tmp_norm
                best_vec = tmp.copy()
                cur = tmp
                cur_norm = tmp_norm
                m.fill(0.0)
                v.fill(0.0)
                it = 0
            no_improve = 0

        # ----- restart if no progress for a long while -----
        if no_improve > restart_limit:
            cur = rng.random(n, dtype=np.float64)
            cur = _project_simplex(np.maximum(cur, 0.0))
            m.fill(0.0)
            v.fill(0.0)
            it = 0
            no_improve = 0

        # ----- early exit if target reached -----
        if best_norm < target:
            break

    # --------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
