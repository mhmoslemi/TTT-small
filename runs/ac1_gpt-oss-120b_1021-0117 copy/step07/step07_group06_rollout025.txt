import time, math, numpy as np

def _project_simplex(v):
    """Project vector v onto the probability simplex (non‑negative, sum=1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(vec):
    """Exact score used by the evaluator (ignoring clipping/normalisation)."""
    v = np.asarray(vec, dtype=np.float64)
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
