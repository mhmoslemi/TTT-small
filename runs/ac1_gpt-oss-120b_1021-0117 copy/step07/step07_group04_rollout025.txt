def _project_simplex(v, target_sum=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - target_sum
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # adjust numeric sum to target
    excess = w.sum() - target_sum
    if abs(excess) > 1e-12:
        # distribute error to positive entries proportionally?
        # simple: w -= excess / n; clip >=0
        w -= excess / n
        w = np.maximum(w, 0.0)
        # re-scale to target sum maybe
        s = w.sum()
        if s > 0:
            w *= target_sum / s
    return w
```

def _score_fft(a, n, L):
    s = a.sum()
    if s < 1e-12:
        return np.inf

    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    max_b = conv.max()
    return float(2 * n * max_b / (s * s))

def _project_simplex(v, target_sum=1.0):
    # Euclidean projection onto simplex {x>=0, sum x = target_sum}
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - target_sum
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # ensure sum exactly target_sum (account for numerical error)
    w_sum = w.sum()
    if w_sum > 0:
        w *= target_sum / w_sum
    else:
        # all zero; distribute uniformly
        w = np.full_like(v, target_sum / n)
    return w
```

import time, math, numpy as np

def _project_simplex(v, target_sum=1.0):
    """Euclidean projection of v onto {x >= 0, sum x = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - target_sum
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum > 0:
        w *= target_sum / w_sum
    else:  # degenerate case – distribute uniformly
        w = np.full_like(v, target_sum / n)
    return w


def _conv_and_grad(a, n, L, alpha):
    """Return full linear convolution a*a and sub‑gradient of smooth max."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_c = conv_full.max()
    # smooth weights approximating the arg‑max
    w = np.exp(alpha * (conv_full - max_c))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.zeros_like(conv_full)
        w_sum = 1.0
    w /= w_sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)      # slice that yields the true gradient
    grad = 2.0 * grad_full[idx]
    return conv_full, grad


def _fast_score(conv_full, sum_target, n):
    """Score from pre‑computed convolution (avoids extra FFT)."""
    max_b = conv_full.max()
    return float(2.0 * n * max_b / (sum_target * sum_target))
