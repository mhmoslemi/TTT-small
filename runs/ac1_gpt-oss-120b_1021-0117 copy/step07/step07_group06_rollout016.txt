def propose_candidate(seed=42,budget_s=1100,**kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # small margin

    # init vector
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        # ensure nonnegative
        a0 = np.maximum(a0, 0.0)
    else:
        n = 2500
        a0 = rng.random(n, dtype=np.float64) + 1e-3

    cur = _project_simplex(a0)  # sum to 1
    
    n = cur.size
    L = 1 << ((2*n - 1).bit_length())  # next power of two

    # evaluation function using sum=1
    def _score_vec(v):
        # v sum=1 maybe not exactly due to rounding, compute sum
        s = v.sum()
        if s == 0: return np.inf
        # compute convolution via FFT
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        max_b = conv.max()
        return float(2.0 * n * max_b / (s*s))

    best = cur.copy()
    best_sc = _score_vec(best)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.8
    eps_grad = 1e-12
    it = 0
    no_improve = 0
    target = 1.50274349

    # main loop
    while time.monotonic() < deadline:
        it += 1

        # compute convolution and max
        af = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(af * af, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # Determine active indices: conv close to max
        # tolerance absolute: e.g., 1e-12 * max_val or small
        tol = max(1e-12, max_val * 1e-12)
        active_idx = np.where(conv >= max_val - tol)[0]
        # compute gradient
        grad = np.zeros_like(cur)
        # Sum contributions for each active index

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w
