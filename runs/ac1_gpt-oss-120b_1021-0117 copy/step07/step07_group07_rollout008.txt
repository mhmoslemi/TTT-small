    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # margin

        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()

    else:
        n = rng.integers(2000, 3000)
        cur_random = rng.random(n)

    n = cur.size
    L = 1 << ((2*n - 1).bit_length())

    best_sc = _score(best_vec)
    cur_sc = best_sc

    # parameters
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    eps = 1e-8
    step0 = 0.6

        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)

        delta = 1e-6 * max_val + 1e-12

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum=1, non‑negative)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Score = 2·n·max(conv(a,a)) assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialise
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = _project_simplex(np.maximum(cur, 0.0))
    else:
        n = rng.integers(2000, 3000)
        cur = _project_simplex(rng.random(n))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # hyper‑parameters
    step0 = 0.6                      # base step size
    temp = 5.0                       # initial soft‑max temperature (unused in final mask)
    temp_factor = 1.001
    temp_max = 300.0

    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ---------- convolution ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ---------- weight mask around the maximum ----------
        # take all entries whose value is within a tiny absolute tolerance of the max
        delta = 1e-12 + 1e-6 * max_val
        mask = conv >= (max_val - delta)
        if not np.any(mask):
            mask = np.ones_like(conv, dtype=bool)
        w = mask.astype(np.float64)
        w /= w.sum()          # uniform weight on the active set

        # ---------- gradient of the weighted sum ----------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)   # pick the relevant slice
        grad = 2.0 * grad_full[idx]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)

        step = step0 / (np.sqrt(v_hat) + eps)
