def propose_candidate(seed=42, budget_s=1100): 
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.2) # margin

    # init vector a
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a = rng.random(n) + 1e-3

    # project onto simplex (sum=1) ensure nonneg
    a = _project_simplex(np.maximum(a, 0))

    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())

    # evaluation function
    def score_vec(v):
        # compute conv via FFT
        fftv = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(fftv * fftv, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())

    best = a.copy()
    best_sc = score_vec(best)

    # Adam params
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1  # initial learning rate; may adjust
    it = 0

    # schedule for softmax alpha
    alpha0 = 1.0
    alpha_inc = 0.005 # per iter
    alpha_max = 200.0

    while time.monotonic() < deadline:
        it += 1

        # compute conv and max
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # compute gradient using softmax weight
        # compute alpha
        alpha = min(alpha_max, alpha0 + alpha_inc * it)
        # w = exp(alpha*(conv-max_val))
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            mask = (conv >= max_val - 1e-12)
            w = mask.astype(np.float64)
            c = w.sum()
            if c > 0:
                w /= c
        else:
            w /= w_sum

        # gradient via FFT
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n-2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # candidate
        cand = _project_simplex(a - step)
        cand_sc = score_vec(cand)

        # accept if better, else occasional acceptance
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
        # simple acceptance: accept if better, else keep old
        if cand_sc < score_vec(a):
            a = cand
        else:
            # maybe accept with prob
            # degrade temperature
            temp = 0.01 + 0.1/np.sqrt(it)
            if rng.random() < np.exp(-(cand_sc - score_vec(a))/temp):

        # periodic symmetrization

            sym = _project_simplex(sym)
            sym_sc = score_vec(sym)
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
            if sym_sc < score_vec(a):
                a = sym

        # early exit
        if best_sc < 1.50274349:
            break

    return [float(x) for x in best]

import time, math, numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level, no closures)
# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(vec):
    """Score = 2 * n * max(conv(vec,vec))  (sum(vec)=1 assumed)."""
    vec = np.asarray(vec, dtype=np.float64)
    n = vec.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _balance_permute(v):
    """Re‑order values so large entries are spread apart."""
    idx = np.argsort(v)[::-1]               # descending indices
    out = np.empty_like(v)
    left, right = 0, v.size - 1
    use_left = True
    for i in idx:
        if use_left:
            out[left] = v[i]
            left += 1
        else:
            out[right] = v[i]
            right -= 1
        use_left = not use_left
    return out

def _random_init(n, rng):
    """Generate a random non‑negative vector of length n, projected to simplex."""
    x = rng.random(n, dtype=np.float64) + 1e-6
    return _project_simplex(x)

    return [float(x) for x in best_vec]
