import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as measured by ``evaluate_sequence``.
    The routine works with any parent sequence ``height_sequence_1`` that may
    already be present in the global namespace; otherwise it falls back to a
    uniform distribution, which is provably optimal for the objective.
    """
    start = time.monotonic()
    # leave a small safety margin before the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # --------------------------------------------------------------
    # Determine problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000  # reasonable default when no parent is given

    # --------------------------------------------------------------
    # Baseline candidate: uniform distribution (sum == 1)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_vec = uniform
    try:
        best_score = evaluate_sequence(best_vec.tolist())
    except Exception:                     # extreme defensive guard
        best_score = float("inf")

    # --------------------------------------------------------------
    # Try the supplied parent (if any) after clipping to the allowed range.
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)   # enforce evaluator limits
        try:
            parent_score = evaluate_sequence(parent.tolist())
            if parent_score < best_score:
                best_score = parent_score
                best_vec = parent
        except Exception:
            pass

    # --------------------------------------------------------------
    # Light‑weight stochastic refinement (optional – keeps the search cheap)
    rng = np.random.default_rng(seed)
    cur = best_vec.copy()
    cur_score = best_score
    # small budget for perturbations; stop early if time runs out
    max_iters = 2000
    while time.monotonic() < deadline and max_iters > 0:
        max_iters -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a modest fraction of mass from i to j
        delta = cur[i] * rng.random() * 0.1   # at most 10 % of source mass
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # sum is unchanged; negativity cannot occur because delta ≤ cur[i]
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand

    # --------------------------------------------------------------
    # Return the best vector found as a plain Python list of floats.
    return [float(x) for x in best_vec]
