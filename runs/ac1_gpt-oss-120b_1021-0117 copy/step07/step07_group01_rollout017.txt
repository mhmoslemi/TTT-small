    if total == 0:

    else:
        a = a / total

    L = 1 << ((2 * n - 1).bit_length())
    # Helper functions
    def _proj(v):
        v = np.asarray(v, dtype=np.float64)
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    def _score_vec(vec):
        # sum assumed 1
        a_fft = np.fft.rfft(vec, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        maxc = conv_full[: 2 * n - 1].max()
        return float(2 * n * maxc)

    alpha = 1.0
    alpha_factor = 1.05
    max_alpha = 200.0

            rand = _proj(rand)
            a = 0.5 * best + 0.5 * rand
            a = _proj(a)
            no_improve = 0

        alpha = min(alpha * alpha_factor, max_alpha)

        # early exit
        if best_score < target:
            break

    # final fallback: ensure nonnegative and normalized, scale arbitrary (e.g., sum=1)
    final = best.copy()
    final = np.maximum(final, 0.0)
    if final.sum() == 0:
        final = np.ones_like(final) / final.size
    else:
        final = final / final.sum()
    return [float(x) for x in final]

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(v: np.ndarray) -> float:
    """Fast score for a vector assumed to have sum(v)=1."""
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(v, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    maxc = conv_full[: 2 * n - 1].max()
    return float(2 * n * maxc)
