def project_simplex(v):
    # ensures sum(v) = 1, v >=0
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_from_conv(conv, n):
    """Score when sum(a)=1: 2 * n * max(conv)."""
    return float(2.0 * n * conv.max())

def _conv_fft(a, L):
    """Return linear convolution a*a (full) using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * a.shape[0] - 1]

def _subgradient(a, conv, eps=1e-12):
    """Subgradient of max(conv) w.r.t. a (sum(a)=1)."""
    n = a.shape[0]
    max_val = conv.max()
    # tolerance relative to max_val
    tol = eps * max_val if max_val != 0 else eps
    idxs = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in idxs:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # m = lo..hi, a index = k - m
        m = np.arange(lo, hi + 1, dtype=np.int64)
        a_idx = k - m
        grad[m] += 2.0 * a[a_idx]
    if len(idxs):
        grad /= len(idxs)
    return grad

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ----- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random positive vector of length ~2500
        a = rng.random(2500, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    # evaluate initial score
    conv = _conv_fft(a, L)
    cur_sc = _score_from_conv(conv, n)
    best_a = a.copy()
    best_sc = cur_sc

    # optimizer state
    momentum = np.zeros_like(a)
    beta = 0.9
    it = 0
    no_improve = 0
    target = 1.50274349

    # ----- main loop ------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution and subgradient
        conv = _conv_fft(a, L)
        grad = _subgradient(a, conv)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule
        step = 0.6 / math.sqrt(it)

        # candidate update
        cand = _project_simplex(a - step * momentum)

        # evaluate candidate
        conv_c = _conv_fft(cand, L)

        # accept if better, otherwise probabilistic acceptance
        if cand_sc < cur_sc:
