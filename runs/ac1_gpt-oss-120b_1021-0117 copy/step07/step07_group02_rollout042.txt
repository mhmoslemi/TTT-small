def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # Initialize vector
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    # Project onto simplex (sum = 1) and ensure nonneg
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # Helper to compute score
    def _score(a):
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # Adam parameters
    m = np.zeros_like(cur)  # first moment
    v = np.zeros_like(cur)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    temp_factor = 1e-2
    temp_decay = 0.9995
    min_temp = 1e-6
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # softmax weighting
        temp = max_val * max(temp_factor, min_temp)
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(0.0)

        # gradient of weighted max conv
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        # step size schedule: diminishing
        step = 1.0 / (np.sqrt(it) + 1.0)  # simple

        cand = cur - step * (m_hat / (np.sqrt(v_hat) + eps))
        # Project onto simplex
        cand = _project_simplex(np.maximum(cand, 0.0))
        cand_sc = _score(cand)

        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

        # Accept if improvement else keep old (no stochastic acceptance)
        if cand_sc < _score(cur):
            cur = cand

        # adjust temperature
        temp_factor = max(temp_factor * temp_decay, min_temp)

        if best_sc < target:
            break

        # occasional random restart if stuck
        if it % 5000 == 0:
            # small random perturbation
            perturb = rng.normal(scale=0.01, size=n)
            cand2 = cur + perturb
            cand2 = _project_simplex(np.maximum(cand2, 0.0))
            cand2_sc = _score(cand2)
            if cand2_sc < best_sc:
                best_sc = cand2_sc
                best_vec = cand2.copy()
                cur = cand2

    return [float(x) for x in best_vec]

import time, math, numpy as np

# --------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def _project_simplex(v):
    """Project a vector onto {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# --------------------------------------------------------------
# Score assuming the vector already sums to 1 (the denominator is 1)
def _score_vec(a):
    """2 * n * max(conv(a,a)) – the exact quantity evaluated by
    evaluate_sequence after the sum‑normalisation."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Generate a Sidon‑type set (all pairwise sums distinct) – fast greedy
def _sidon_positions(n, rng):
    best = []
    best_used = set()
    # try a few random orders and keep the largest set found
    for _ in range(5):
        order = np.arange(n)
        rng.shuffle(order)
        pos = []
        used = set()
        for i in order:
            conflict = False
            if 2 * i in used:
                conflict = True
            else:
                for p in pos:
                    if (i + p) in used:
                        conflict = True
                        break
            if not conflict:
                # add new sums
                for p in pos:
                    used.add(i + p)
                used.add(2 * i)
                pos.append(i)
        if len(pos) > len(best):
            best, best_used = pos, used
    return best
