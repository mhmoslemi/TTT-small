def project_simplex(v):
    v = np.maximum(v, 0)
    if v.sum() == 0:
        # return uniform to avoid divide zero
        return np.full_like(v, 1.0 / len(v))
    # onto simplex sum=1
    # Implementation using sorting method.
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        v = v.ravel()
    # ensure non‑negative
    v = np.maximum(v, 0.0)
    if v.sum() == 0.0:
        # uniform fallback
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - css / ind > 0
    rho = np.where(cond)[0][-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    eps = 1e-8
    lr = 0.25          # base learning rate
    t = 0

    # simulated‑annealing temperature (for occasional uphill moves)
    T = 0.01 * best_sc
    cooling = 0.9995

    # soft‑max temperature for gradient approximation
    temp_factor = 1e-2
    temp_decay = 0.9995
    min_temp_factor = 1e-6

    it = 0
    no_improve = 0
    target = 1.50274349

    # ----- main optimisation loop ----------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution & weighted gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        # soft‑max weighting over lags
        temp = max_val * max(temp_factor, min_temp_factor)
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:
            w.fill(0.0)

        # gradient of the smoothed maximum
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)
        # clamp steps to keep changes moderate
        step_vec = np.clip(step_vec, -1.0, 1.0)

        cand = _project_simplex(cur - step_vec)
        cand_sc = _score(cand)

        # accept if better, otherwise occasional uphill acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (T + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
            else:
                no_improve += 1

        # periodic symmetrisation (the optimum is symmetric)
        if it % 300 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # random local redistribution when stuck
        if no_improve > 5000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0
            else:
                # full random restart
                cur = _project_simplex(rng.random(n))
                cur_sc = _score(cur)
                m.fill(0.0)
                v.fill(0.0)
                t = 0
                no_improve = 0

        # temperature schedules
        T = T * cooling
        temp_factor = max(temp_factor * temp_decay, min_temp_factor)

        # early exit if target achieved
        if best_sc < target:
            break

        # periodic time check
        if it % 1000 == 0 and time.monotonic() >= deadline:
            break

    # return best‑found coefficients as plain Python floats
    return [float(x) for x in best_vec]
