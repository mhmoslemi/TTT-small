def _score_normed(a):
    # a expected sum = 1
    # compute conv via fft
    n = a.shape[0]

    af = np.fft.rfft(a, n=L)

import time
import numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise in case of tiny numerical loss
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score_normed(a, L=None):
    """Score for a vector a that is already normalised to sum=1."""
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
