def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

            if tmp_sc < best_sc:
                best_sc = tmp_sc
                best_vec = tmp.copy()
                cur = tmp
                no_improve = 0

        # ---------- symmetrisation every few hundred steps ----------
        if t_iter % 900 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()
                cur = sym
                no_improve = 0

        # ---------- early exit ----------
        if best_sc < target_score:
            break

    # ------------------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
