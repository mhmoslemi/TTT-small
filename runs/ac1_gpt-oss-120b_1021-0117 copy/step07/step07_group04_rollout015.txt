def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0, float(budget_s) - 0.2)  # leave 0.2 sec buffer

    # Determine n
    if 'height_sequence_1' in globals():
        n = int(len(height_sequence_1))
    else:
        n = 2000
    # compute FFT length (power of two >= 2n-1)
    L = 1 << ((2 * n - 1).bit_length())

    # helper functions defined earlier (project simplex, score)
    # Build candidates
    xs = np.arange(n, dtype=np.float64)
    cands = []
    cands.append(np.full(n, 1.0 / n))
    tri = np.minimum(xs + 1, n - xs)
    cands.append(tri / tri.sum())
    sine = np.sin(np.pi * (xs + 1) / (n + 1))
    cands.append(sine / sine.sum())
    bump = (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2
    cands.append(bump / bump.sum())
    for alpha in (0.001, 0.003, 0.006):
        exp = np.exp(-alpha * xs)
        cands.append(exp / exp.sum())
    for p in (0.5, 1.0, 1.5):
        pw = (xs + 1) ** (-p)
        cands.append(pw / pw.sum())
    # random uniform
    rand = rng.random(n)
    cands.append(rand / rand.sum())
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        s = parent.sum()
        if s > 0:
            parent = parent / s
            cands.append(parent)

    # evaluate candidates using fast score
    best_sc = float('inf')
    best_vec = None
    for vec in cands:
        sc = _score_fft(vec, n, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # optimization parameters
    momentum = np.zeros_like(cur)
    beta = 0.95
    step0 = 1.0
    it = 0
    no_improve = 0
    target = 1.50274349
    sym_interval = 200
    redistrib_thresh = 3000
    factors = (0.5, 0.75, 1.0, 1.25, 1.5)

    # main loop
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # convolution and mask
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-15
        mask = (conv >= max_val - eps).astype(np.float64)
        cnt = mask.sum()
        if cnt > 0:
            mask /= cnt

        # gradient via FFT
        w_rev = mask[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size
        step = step0 / math.sqrt(it)

def _score_fft(a, n, L):
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    return float(2.0 * n * conv.max() / (s * s))

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast evaluation of the AC1 upper bound for a non‑negative vector a."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))
