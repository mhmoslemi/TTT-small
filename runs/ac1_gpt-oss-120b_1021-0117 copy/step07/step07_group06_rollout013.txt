import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a):
    """Score for a vector already normalised to sum=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())

                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            # occasional probabilistic acceptance
            temp = step * 0.5 + 1e-12
            if rng.random() < math.exp(-(cand_sc - cur_sc) / temp):
                cur, cur_sc = cand, cand_sc

        # periodic symmetrisation (helps many instances)
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # random local redistribution when stagnating
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc

        # occasional tiny perturbations for fine‑graining
        if it % 2000 == 0:
            for _ in range(10):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                eps = 0.01 * cur[i]
                if eps == 0.0:
                    continue
                tmp = cur.copy()
                tmp[i] -= eps
                tmp[j] += eps
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc

        if best_sc < target:
            break

        # keep an eye on the deadline
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # final short hill‑climbing phase with remaining budget
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = cur[i] * rng.random() * 0.3
        if delta == 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc

    # Return the best vector found (plain Python floats)
    return [float(x) for x in best_vec]
