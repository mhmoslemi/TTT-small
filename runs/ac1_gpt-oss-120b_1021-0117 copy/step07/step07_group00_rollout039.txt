    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # length of the problem
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2000

    # length for zero‑padded FFT (next power of two)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # baseline candidates
    best_vec = None
    best_sc = float('inf')

    # uniform distribution (sum = 1)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    sc = _conv_score(uniform, L)
    if sc < best_sc:
        best_sc, best_vec = sc, uniform.copy()

    # ------------------------------------------------------------------
    # geometric family a_i = (1 - r) * r^i  (r in (0,1))
    # This family is scale‑invariant for the objective, so we keep the raw form.
    # A coarse grid, denser near 1 where the optimum lives.
    r_grid = np.linspace(0.80, 0.99999, 400, dtype=np.float64)
    for r in r_grid:
        if time.time() >= deadline:
            break
        a = (1.0 - r) * np.power(r, np.arange(n, dtype=np.float64))
        sc = _conv_score(a, L)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # ------------------------------------------------------------------
    # local refinement around the best geometric parameter if we have it
    if best_vec is not None:
        nz = np.flatnonzero(best_vec)
        if nz.size >= 2:
            # estimate r from the first two positive entries
            i0, i1 = nz[0], nz[1]
            if best_vec[i0] > 0.0:
                est_r = best_vec[i1] / best_vec[i0]
                # explore a small interval around the estimate
                for delta in np.linspace(-3e-4, 3e-4, 61):
                    if time.time() >= deadline:
                        break
                    r = est_r + delta
                    if not (0.0 < r < 1.0):
                        continue
                    a = (1.0 - r) * np.power(r, np.arange(n, dtype=np.float64))
                    sc = _conv_score(a, L)
                    if sc < best_sc:
                        best_sc, best_vec = sc, a.copy()

    # ------------------------------------------------------------------
    # tiny stochastic local search around the current best vector
    rng = np.random.default_rng(seed)
    cur = best_vec.copy() if best_vec is not None else uniform.copy()
    cur_sc = best_sc
    step_factors = [0.2, 0.1, 0.05, 0.02, 0.01]
    step_i = 0

    while time.time() < deadline:
        step = step_factors[step_i]
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = cur[i] * rng.random() * step
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0.0:
            continue
        sc = _conv_score(cand, L)
        if sc < cur_sc:
            cur, cur_sc = cand, sc
            if sc < best_sc:
                best_sc, best_vec = sc, cand.copy()
            step_i = max(step_i - 1, 0)   # finer steps after improvement
        else:
            step_i = min(step_i + 1, len(step_factors) - 1)

    # ------------------------------------------------------------------
    # safety fallback
    if best_vec is None:
        best_vec = uniform

    # convert to plain Python floats
    return [float(x) for x in best_vec]
