import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # adjust tiny numerical drift
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast version of the evaluator (assumes a>=0 and already normalised)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

    for p in (0.5, 1.0, 1.5):
        pw = (xs + 1) ** (-p)
        candidates.append(pw / pw.sum())

    # Random (deterministic seed)
    rand = rng.random(n)
    candidates.append(rand / rand.sum())

    # Parent if available
    if parent is not None:
        candidates.append(parent)

    # ------------------------------------------------------------------
    # Pick the best initial candidate
    best_vec = None
    best_sc = float('inf')
    for vec in candidates:
        sc = _score_fft(vec, n, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # Enforce symmetry from the start
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)

    # ------------------------------------------------------------------
    # Optimisation state (Adam)
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.4
    tau = None
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & softmax weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        maxc = conv.max()
        if tau is None:
            tau = maxc * 0.1
        else:
            tau = max(tau * 0.999, maxc * 1e-6)

        w = np.exp((conv - maxc) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:                     # fallback – should not happen
            w = (conv == maxc).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient (softmax‑weighted) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- candidate and projection -----
        cand = cur - step
        cand = _project_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5               # enforce symmetry
        cand = _project_simplex(cand)

        # occasional tiny random jitter to escape flat regions
        if rng.random() < 0.003:
            jitter = rng.normal(scale=0.002, size=n)
            cand = np.clip(cand + jitter, 0.0, None)
            cand = _project_simplex(cand)

        cand_sc = _score_fft(cand, n, L)

        # ----- acceptance -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # simulated-annealing style acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (base_lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                no_improve = 0

        # ----- global best update -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            no_improve = 0

        # ----- periodic symmetrisation & re‑evaluation -----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ----- stagnation handling -----
        if no_improve > 4000:
            # random restart mixing best with fresh random point
            rand_vec = rng.random(n)
            rand_vec = rand_vec / rand_vec.sum()
            cur = _project_simplex(0.5 * best_vec + 0.5 * rand_vec)
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)
            cur_sc = _score_fft(cur, n, L)
            m.fill(0.0)
            v.fill(0.0)
            it = 0
            tau = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
