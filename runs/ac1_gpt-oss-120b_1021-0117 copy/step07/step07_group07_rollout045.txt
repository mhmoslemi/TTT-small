def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initialize
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        cur = _project_simplex(np.maximum(a0, 0.0))
    else:
        n = rng.integers(2000, 3000)
        cur = _project_simplex(rng.random(n))
    n = cur.size
    # optionally symmetrize
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)

    L = 1 << ((2 * n - 1).bit_length())

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.5

    # evaluate best
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    cur_score = 2.0 * n * max_val
    best_vec = cur.copy()
    best_score = cur_score

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # compute conv and mask
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        # tolerance
        tol = max_val * 1e-12 + 1e-14
        mask = (conv >= max_val - tol).astype(np.float64)
        # gradient via mask
        w_rev = mask[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)

        cand = cur - step * m_hat
        cand = _project_simplex(cand)

        # evaluate candidate score
        a_fft_cand = np.fft.rfft(cand, n=L)
        conv_full_cand = np.fft.irfft(a_fft_cand * a_fft_cand, n=L)
        max_val_cand = conv_full_cand[: 2 * n - 1].max()
        cand_score = 2.0 * n * max_val_cand

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_vec, best_score = cand.copy(), cand_score
        else:
            no_improve += 1
            if rng.random() < 0.001:
                cur, cur_score = cand, cand_score

        # periodic symmetrization
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            a_fft_sym = np.fft.rfft(sym, n=L)
            conv_full_sym = np.fft.irfft(a_fft_sym * a_fft_sym, n=L)
            max_val_sym = conv_full_sym[: 2 * n - 1].max()
            sym_score = 2.0 * n * max_val_sym
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                no_improve = 0
                if sym_score < best_score:
                    best_vec, best_score = sym.copy(), sym_score

        # random perturb when stuck
        if no_improve > 2000:
            # random mass transfer
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                # evaluate
                a_fft_tmp = np.fft.rfft(tmp, n=L)
                conv_full_tmp = np.fft.irfft(a_fft_tmp * a_fft_tmp, n=L)
                max_val_tmp = conv_full_tmp[: 2 * n - 1].max()
                tmp_score = 2.0 * n * max_val_tmp
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    no_improve = 0
                    if tmp_score < best_score:
                        best_vec, best_score = tmp.copy(), tmp_score

        if best_score < target:
            break

    return [float(x) for x in best_vec]

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
