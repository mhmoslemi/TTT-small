    start = time.monotonic()
    deadline = start + budget_s - 0.2

    # initial vector
    if 'height_sequence_1' in globals():
        vec = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        rng = np.random.default_rng(seed)
        n = 2000

    # normalize
    total = vec.sum()
    if total <= 0:
        vec[:] = 1.0/vec.size
        total = vec.sum()
    vec = vec / total   # sum to 1

    n = vec.shape[0]
    L = 1 << ((2*n-1).bit_length())

    # scoring
    def max_conv(v):
        a_fft = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        return conv.max()

    cur_max = max_conv(vec)
    cur_score = 2.0 * n * cur_max
    best_vec = vec.copy()
    best_score = cur_score

    rng = np.random.default_rng(seed+1)
    # parameters
    max_iters = 200000  # upper bound but will break on time
    # schedule
    T0 = 0.05
    alpha = 0.5  # max fraction moved
    # to keep sum constant

    for it in range(1, max_iters+1):
        if time.monotonic() >= deadline:
            break

        # choose two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1

        if vec[i] <= 0:
            continue  # nothing to move

        # move fraction of mass from i to j
        frac = rng.random() * alpha * (1.0 - it / max_iters)  # decreasing
        delta = vec[i] * frac
        cand = vec.copy()
        cand[i] -= delta
        cand[j] += delta

        # compute score
        cand_max = max_conv(cand)
        cand_score = 2.0 * n * cand_max

        # acceptance
        if cand_score < cur_score:
            vec = cand
            cur_score = cand_score
            cur_max = cand_max
            if cand_score < best_score:
                best_vec = cand.copy()
                best_score = cand_score
        else:
            # annealing probability
            T = T0 * (1 - it / max_iters)
            if T > 0:
                prob = math.exp(-(cand_score - cur_score) / T)
                if rng.random() < prob:
                    vec = cand
                    cur_score = cand_score
                    cur_max = cand_max

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a, L):
    """Return 2*n*max(conv) for a vector a (already sum‑normalised)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ------------------------------------------------------------------
    # initialise
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2100, dtype=np.float64) + 1e-4
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # optimiser state
    momentum = np.zeros_like(cur)
    beta = 0.95          # momentum coefficient
    step0 = 1.0          # initial step size
    alpha_max = 200.0
    alpha0 = 1.0

    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)

        # full convolution (linear)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # --- smooth max via soft‑max weighting (focuses on true max later) ---
        alpha = min(alpha_max, alpha0 * math.log1p(it))
        # avoid overflow / underflow
        w = np.exp(alpha * (conv_full - max_val))
        w_sum = w.sum()
        if w_sum == 0 or np.isinf(w_sum):
            # fallback: uniform weight on all maximal positions
            mask = conv_full >= max_val - max_val * 1e-12 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                cnt = 1
            w = mask.astype(np.float64) / cnt
        else:
            w /= w_sum

        # gradient of weighted max‑convolution (matrix‑free via FFT)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # map to size‑n gradient
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        step = step0 / math.sqrt(it)

        # candidate and evaluation
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score(cand, L)

        # simple greedy acceptance with occasional annealed jumps
        if cand_sc < cur_sc:

            no_improve = 0

        # periodic symmetrisation (helps many instances)
        if it % 500 == 0:

            sym_sc = _score(sym, L)

            if best_sc < target:
                break

    # ------------------------------------------------------------------
    # return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
