import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a, L):
    """Score = 2 * n * max(convolution(a,a)) for a already on simplex."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    target = 1.50274349

    # ------------------------------------------------------------------
    # initial vector (simplex, non‑negative)
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = _project_simplex(np.maximum(cur, 0.0))
    else:
        n0 = rng.integers(2000, 3000)
        cur = _project_simplex(rng.random(n0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two for FFT speed

    cur_sc = _score(cur, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # early exit if the supplied parent already beats the target
    if best_sc < target:
        return [float(x) for x in best_vec]

    # Adam‑style state
    m = np.zeros_like(cur)
    v_acc = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.7

    it = 0
    no_improve = 0

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ---------- convolution & exact max‑gradient ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_idx = int(np.argmax(conv_full))

        # one‑hot weight for the maximal entry
        w = np.zeros_like(conv_full)
        w[max_idx] = 1.0
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v_acc = beta2 * v_acc + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_acc / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)

        cand = cur - step * m_hat
        cand = _project_simplex(cand)

        cand_sc = _score(cand, L)

        # ---------- acceptance ----------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:

            # occasional probabilistic acceptance to escape shallow traps
            if rng.random() < 0.002:

        if no_improve > 2500:

                tmp[src] -= delta
                tmp[dst] += delta
                tmp = _project_simplex(tmp)

                if tmp_sc < cur_sc:

                    no_improve = 0
