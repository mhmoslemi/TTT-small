    def project_simplex(v):
        v = np.maximum(v, 0)
        # Euclidean projection onto simplex sum=1
        # implementation similar to earlier
        n = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        # ensure sum exactly 1
        s = w.sum()
        if s > 0:
            w /= s
        return w

        a_fft = np.fft.rfft(vec, n=next_pow2(2*n-1))

        return float(2.0 * n * conv.max())

        return 1 << ((x-1).bit_length())

    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w


def _next_pow2(x):
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _score(vec):
    """Score assuming sum(vec)=1 (same as evaluate_sequence)."""
    n = vec.size
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _optim_one_run(initial_vec, deadline):
    """Projected Adam‑style optimisation starting from initial_vec."""
    cur = _project_simplex(initial_vec)
    n = cur.size
    L = _next_pow2(2 * n - 1)

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 1.0

    # soft‑max parameters
    alpha_max = 120.0
    alpha_start = 1.0

    it = 0
    no_improve = 0

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution (via FFT) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weighting -----
        alpha = min(alpha_max, alpha_start * math.log1p(it))
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            mask = conv >= max_val - 1e-12
            w = mask.astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of weighted max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (math.sqrt(it) + 1.0)
        delta = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - delta)

        # ----- enforce symmetry (helps many instances) -----
        sym = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(sym)

        cand_sc = _score(cand)

        # ----- keep global best -----
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()

        # ----- accept if improvement, else occasional jump -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            if np.random.rand() < 0.001:
                cur, cur_sc = cand, cand_sc
                no_improve = 0

        # ----- random small redistribution when stuck -----
        if no_improve > 2000:
            delta = np.random.rand(n) * 0.01
            tmp = cur + delta - delta.mean()
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
            else:
                no_improve = 0  # reset to give fresh start

        # ----- early exit if target reached -----
        if best_sc < 1.50274349:
            break

    return best_vec, best_sc


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a list of non‑negative step‑function heights that (hopefully)
    achieve a low AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        init0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        init0 = rng.random(2000, dtype=np.float64) + 1e-3

    # ---- run first optimisation from the provided (or random) start -----
    best_vec, best_sc = _optim_one_run(init0, deadline)

    # ---- additional stochastic restarts while time permits -------------
    while time.monotonic() < deadline - 2.0:
        # generate a new random start; a brief low‑pass filter gives smoother shapes
        rand = rng.random(best_vec.size)
        if best_vec.size > 500:
            freq = np.fft.rfftfreq(best_vec.size)
            filt = np.exp(- (freq / 0.05) ** 2)
            rand = np.fft.irfft(np.fft.rfft(rand) * filt, n=rand.size)
            rand = np.maximum(rand, 0.0)
        rand = _project_simplex(rand)

        vec, sc = _optim_one_run(rand, deadline)
        if sc < best_sc:
            best_sc, best_vec = sc, vec
        if best_sc < 1.50274349:
            break

    # ---- final fallback (ensure non‑empty) ------------------------------
    if best_vec is None or best_vec.size == 0:
        best_vec = _project_simplex(init0)

    return [float(x) for x in best_vec]
