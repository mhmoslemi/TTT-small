def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    if "height_sequence_1" in globals():

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    base_lr = 1.0
    it = 0
    no_improve = 0

    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # compute conv and gradient
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # mask of near max
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        # gradient via reverse convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)  # map to gradient per a_i
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr / (np.sqrt(it) + 1.0)

        # candidate generation
        candidates = []
        for factor in (1.0, 0.5, 2.0):
            update = step * factor * m_hat / (np.sqrt(v_hat) + eps)
            cand = _project_simplex(a - update)
            candidates.append(cand)
        # multiplicative
        exp_factor = np.exp(-step * grad)
        cand_mul = a * exp_factor
        s = cand_mul.sum()
        if s > 0:
            cand_mul = cand_mul / s
        candidates.append(cand_mul)

            for _ in range(5):
                i = rng.integers(0, n)
                if a[i] <= 0:

                delta = a[i] * rng.random() * 0.2
                j = rng.integers(0, n)
                if j == i:
                    continue
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # No need to project because sum unchanged and nonnegativity holds
                candidates.append(cand)

        cand = None
        for c in candidates:

            if cand_sc is None or sc < cand_sc:
                cand_sc = sc
                cand = c

            best_sc = cand_sc
            a = cand
            no_improve = 0

                best_sc = sym_sc
                a = sym

            a = 0.5 * best + 0.5 * rand_vec
            a = _project_simplex(a)
            no_improve = 0

        if best_sc < target:
            break

        # check deadline occasionally
        if time.monotonic() >= deadline:
            break

    return [float(x) for x in best]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score for a vector already normalized to sum=1 (2*n*max(conv))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    base_lr = 1.0
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution and max -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- sub‑gradient (uniform over max entries) -----
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # ----- gradient of the weighted max‑convolution -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        grad = 2.0 * grad_full[idx]

        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)

        step = base_lr / (math.sqrt(it) + 1.0)
