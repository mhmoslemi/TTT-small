import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Score 2·n·max(conv(a,a)) for a sequence a with sum(a)=1."""

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early termination -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    return [float(x) for x in best]
