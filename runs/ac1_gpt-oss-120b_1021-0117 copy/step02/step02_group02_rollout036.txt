def conv_grad_at_k(a, k):
    n = len(a)
    g = np.zeros_like(a)
    low = max(0, k-(n-1))
    high = min(n-1, k)
    # iterate i from low to high, but we only need i <= k-i to avoid double
    # i <= partner => i <= k-i => 2*i <= k => i <= k//2
    end = min(high, k // 2)
    # first part i<partner
    i_range = np.arange(low, end+1) # inclusive
    partner = k - i_range
    # Determine which are i < partner
    mask = i_range < partner
    i_lt = i_range[mask]
    partner_lt = partner[mask]
    # Off-diagonal contributions
    g[i_lt] += a[partner_lt]
    g[partner_lt] += a[i_lt]
    # Diagonal (i == partner)
    diag_mask = i_range == partner
    i_diag = i_range[diag_mask]
    g[i_diag] += 2 * a[i_diag]
    return g

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_sum_shifts(a: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """Combined gradient of convolution entries indexed by ks."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in ks:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k - i
        # each term a[i]*a[partner] contributes a[partner] to grad_i
        # and a[i] to grad_partner (both for the same pair, giving factor 2)
        np.add.at(g, i, a[partner])
        np.add.at(g, partner, a[i])
    return g
