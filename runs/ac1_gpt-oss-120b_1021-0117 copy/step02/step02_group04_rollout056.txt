import math
import time

import numpy as np

# ---------- helpers ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 score using FFT (identical to the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _gen_binomial(n: int) -> np.ndarray:
    """Classic binomial step (C(2i,i)/4**i)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ---------- main entry ----------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    # give a tiny safety margin for the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- problem size -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256  # fallback size when no parent is provided

    L = _next_pow2(2 * n - 1)

    # ----- initial best candidate --------------------------------------
    if parent is not None:
        cur_best = parent.astype(np.float64).copy()
    else:
        cur_best = _gen_binomial(n).astype(np.float64)

    best_score = _ac1_score_fft(cur_best, n, L)

    # ----- simple families of sequences ---------------------------------
    indices = np.arange(n, dtype=np.float64)

    # exponential decays (r in (0,1])
    t_vals = np.geomspace(1e-6, 0.5, num=200)
    for t in t_vals:
        if time.monotonic() >= deadline:
            break
        r = 1.0 - t
        if r <= 0.0:
            continue
        cand = np.power(r, indices)
        sc = _ac1_score_fft(cand, n, L)
        if sc < best_score:
            cur_best = cand.copy()
            best_score = sc
        # reversed order (increasing shape)
        cand_rev = np.power(r, indices[::-1])
        sc = _ac1_score_fft(cand_rev, n, L)
        if sc < best_score:
            cur_best = cand_rev.copy()
            best_score = sc
        if best_score < 1.50274349:
            break

    # power‑law decays (negative exponents)
    p_vals = np.linspace(-5.0, -0.1, num=80)
    for p in p_vals:
        if time.monotonic() >= deadline:
            break
        cand = np.power(indices + 1.0, p)
        sc = _ac1_score_fft(cand, n, L)
        if sc < best_score:
            cur_best = cand.copy()
            best_score = sc
        if best_score < 1.50274349:
            break

    # linear decrease (from 1 to 0)
    cand = 1.0 - indices / (n - 1)
    sc = _ac1_score_fft(cand, n, L)
    if sc < best_score:
        cur_best = cand.copy()
        best_score = sc

    # ----- gradient‑descent refinement ----------------------------------
    cur = cur_best.copy()
    cur_score = best_score
    step_factor = 0.5            # base step multiplier
    step_decay = 0.9997           # slow decay of step size
    # occasional annealed random moves are also useful
    while time.monotonic() < deadline:
        # compute convolution and current statistics
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_b = float(conv.max())
        k_max = int(conv.argmax())
        sum_cur = float(cur.sum())
        if sum_cur < 1e-12:
            break
        # sub‑gradient of the active lag
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)

        g = np.zeros_like(cur)
        # diagonal entries (i == j)
        diag = mask & (j == i)
        g[diag] = cur[diag]
        # off‑diagonal entries
        off = mask & (j != i)
        j_off = j[off]
        g[off] = 2.0 * cur[j_off]

        C = 2.0 * max_b / sum_cur           # constant term from denominator derivative
        direction = C - g                   # descent direction (negative gradient up to scaling)

        # candidate update
        cand = cur + step_factor * direction
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = _ac1_score_fft(cand, n, L)

        # accept if improved, otherwise shrink step size
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                cur_best = cand.copy()
                best_score = cand_score
                if best_score < 1.50274349:
                    break
        else:
            step_factor *= 0.5  # aggressive reduction when not improving

        # global step schedule
        step_factor = max(step_factor * 1.01, 1e-8)  # gently increase when possible
        step_factor *= step_decay

        # small random perturbations to escape flat regions
        if rng.random() < 0.001:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.08))
            cur[start:start + width] *= factor
            np.clip(cur, 0.0, 1000.0, out=cur)

    # ----- return best found sequence ------------------------------------
    return [float(x) for x in cur_best]
