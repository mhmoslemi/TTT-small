import time, math, numpy as np

# ---------- helper utilities ----------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_self_fft(a: np.ndarray, n: int, fft_len: int) -> np.ndarray:
    """Full self‑convolution using a real FFT (length 2*n-1)."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * n - 1]

def _catalan_weights(n: int) -> np.ndarray:
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w

def _tent_shape(n: int) -> np.ndarray:
    """Symmetric linear “tent” (distance from the nearer end)."""
    idx = np.arange(n, dtype=np.float64)
    v = np.minimum(idx + 1, n - idx)
    return v / v.sum()

def _window_candidates(n: int):
    """A small palette of classic windows (normalized)."""
    cand = []
    # Hann, Hamming, Blackman, Kaiser (several betas)

        w = np.kaiser(n, beta)
        if w.sum() > 0:
            cand.append(w / w.sum())
    return cand

    # 1. parent, normalised
    if parent_arr is not None:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            candidates.append(p / s)

    # 2. Catalan
    cand = _catalan_weights(n)
    candidates.append(cand / cand.sum())

    # 3. Tent shape
    candidates.append(_tent_shape(n))

    # 4. Quadratic “parabolic” family  (i+1)*(n-i) ^ p
    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1) * (n - idx)
    for p in np.linspace(0.2, 3.0, 13):
        v = base ** p
        if v.sum() > 0:
            candidates.append(v / v.sum())

    # 5. Sin^p family
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    for p in np.linspace(0.2, 3.0, 13):
        v = sin_vals ** p
        if v.sum() > 0:
            candidates.append(v / v.sum())

    # 6. Classic windows
    candidates.extend(_window_candidates(n))

    # 7. Random Dirichlet draws (few)
    for _ in range(6):
        v = rng.random(n)
        v = v / v.sum()
        candidates.append(v)

    # ---- pick the best initial vector ----
    best_vec = None
    best_score = np.inf
    for v in candidates:
        conv = _conv_self_fft(v, n, fft_len)
        cur = 2.0 * n * float(conv.max())
        if cur < best_score:
            best_score = cur
            best_vec = v.copy()
        if best_score <= 1.50274349:
            return best_vec.tolist()

    if best_vec is None:  # fallback safety
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    a = best_vec.copy()
    best_score = 2.0 * n * float(_conv_self_fft(a, n, fft_len).max())
    best_vec = a.copy()

    # ---- iterative sub‑gradient descent (exponential update) ----
    iter_no_improve = 0
    max_no_improve = 4000
    alpha0 = 0.6          # base step magnitude
    min_alpha = 5e-5
    max_alpha = 0.5

    while time.monotonic() < deadline:
        conv = _conv_self_fft(a, n, fft_len)
        max_b = float(conv.max())
        cur_score = 2.0 * n * max_b

        # update best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            iter_no_improve = 0
            if best_score <= 1.50274349:
                break
        else:
            iter_no_improve += 1

        # active indices (within relative tolerance)
        tol = max(1e-12, max_b * 1e-9)
        active = np.where(conv >= max_b - tol)[0]

        # sub‑gradient of the active max‑term(s)
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            g[idx_range] += 2.0 * a[k - idx_range]

        g_norm2 = np.dot(g, g) + 1e-20

        # adaptive step size (Polyak‑type, bounded)
        if cur_score > best_score:
            diff = cur_score - best_score
            alpha = diff / g_norm2
        else:
            alpha = alpha0 / math.sqrt(iter_no_improve + 1)

        alpha = max(min_alpha, min(alpha, max_alpha))

        # exponential (mirror) update
        with np.errstate(over='ignore'):
            factor = np.exp(-alpha * g)
        factor = np.clip(factor, 1e-8, 1e8)
        cand = a * factor
        cand_sum = cand.sum()
        if cand_sum == 0 or not np.isfinite(cand_sum):
            cand = _project_onto_simplex(a - alpha * g)
        else:
            cand = cand / cand_sum

        # occasional jitter to escape plateaus
        if rng.random() < 0.012:
            jitter = np.exp(rng.normal(0.0, 0.018, size=n))
            cand = cand * jitter
            cand = cand / cand.sum()

        a = cand

        # restart heuristic
        if iter_no_improve > max_no_improve:
            pert = np.exp(rng.normal(0.0, 0.035, size=n))
            a = best_vec * pert
            a = a / a.sum()
            iter_no_improve = 0

    return best_vec.tolist()
