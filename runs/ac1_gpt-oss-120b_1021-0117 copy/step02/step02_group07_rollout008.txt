import numpy as np, time, math

def _initial_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i-1) / (2*i)
    return a

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 2.0)

    # determine n and initial sequence
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            a = np.clip(arr, 0.0, 1000.0)
            n = a.size
        else:
            raise Exception()
    except Exception:
        n = 3809
        a = _initial_binomial(n)

    # normalize to sum=1
    total = a.sum()
    if total == 0:
        a[0] = 1.0
        total = 1.0
    a = a / total

    L = 1 << ((2 * n - 1).bit_length())

    def conv_fft(x):
        f = np.fft.rfft(x, n=L)
        return np.fft.irfft(f * f, n=L)[: 2 * n - 1]

    conv = conv_fft(a)
    max_val = conv.max()
    cur_score = 2.0 * n * max_val  # sum=1

    best = a.copy()
    best_score = cur_score

    # parameters
    min_step = 1e-6
    max_step = 0.5

    # main loop
    while time.monotonic() < deadline:
        # compute convolution and max index
        conv = conv_fft(a)
        max_val = conv.max()
        k = int(conv.argmax())

        # gradient for active max index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        jdx = k - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[jdx]  # factor 2

        # direction with zero sum
        mean_grad = grad.mean()
        direction = -(grad - mean_grad)

        # if direction small, jitter
        if np.linalg.norm(direction) < 1e-12:
            direction = rng.normal(size=n)
            direction -= direction.mean()

        # line search
        step = max_step
        improved = False
        while step >= min_step:
            cand = a + step * direction
            np.clip(cand, 0.0, 1000.0, out=cand)
            s = cand.sum()
            if s == 0:
                # if all zero, break
                break
            cand = cand / s  # renorm to sum=1

            cand_conv = conv_fft(cand)
            cand_max = cand_conv.max()
            cand_score = 2.0 * n * cand_max

            if cand_score < cur_score - 1e-12:
                # accept
                a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
                improved = True
                break
            else:
                step *= 0.5

        if not improved:
            # occasional random perturbation to escape
            if rng.random() < 0.03:
                width = rng.integers(1, max(2, n // 40))
                start_idx = rng.integers(0, n - width + 1)
                factor = np.exp(rng.normal(0.0, 0.12))
                cand = a.copy()
                cand[start_idx:start_idx+width] *= factor
                np.clip(cand, 0.0, 1000.0, out=cand)
                cand = cand / cand.sum()
                cand_conv = conv_fft(cand)
                cand_score = 2.0 * n * cand_conv.max()
                if cand_score < cur_score:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best = cand.copy()
                        best_score = cand_score
            else:
                # small random move of single element
                idx_r = rng.integers(0, n)
                factor = np.exp(rng.normal(0.0, 0.1))
                cand = a.copy()
                cand[idx_r] *= factor
                np.clip(cand, 0.0, 1000.0, out=cand)
                cand = cand / cand.sum()
                cand_conv = conv_fft(cand)
                cand_score = 2.0 * n * cand_conv.max()
                if cand_score < cur_score:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best = cand.copy()
                        best_score = cand_score

    # return best as list of floats
    return [float(x) for x in best]

def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a
