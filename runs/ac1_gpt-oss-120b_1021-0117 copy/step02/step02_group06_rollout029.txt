        full = np.fft.irfft(B, n=fft_len)

def _conv_self_fft(a: np.ndarray, n: int, fft_len: int) -> np.ndarray:
    """Full convolution a*a (length 2*n-1) using real FFT."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * n - 1]


def _objective_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Score = 2*n*max(a*a) assuming sum(a)=1."""
    conv = _conv_self_fft(a, n, fft_len)
    return 2.0 * n * float(conv.max())


def _initial_candidates(n: int, parent_arr: np.ndarray | None):
    """Generate a modest set of diverse start vectors (each sums to 1)."""
    cand = []

    # parent if supplied
    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            cand.append(p / s)

    # Catalan‑style weights
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    cand.append(w / w.sum())

    # Symmetric tent shape a_i ∝ (i+1)*(n-i)
    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1) * (n - idx)
    cand.append(base / base.sum())

    # Sin‑power shapes
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    for p in (0.5, 1.0, 2.0):
        v = sin_vals ** p
        cand.append(v / v.sum())

    # A few random log‑normal shapes
    rng = np.random.default_rng(12345)
    for _ in range(4):
        v = np.exp(rng.normal(0.0, 0.5, size=n))
        cand.append(v / v.sum())

    return cand

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # safety margin

    # ----- obtain problem size -----------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
    except Exception:
        parent_arr = None

    n = int(parent_arr.shape[0]) if parent_arr is not None else 256
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ----- initial candidates -------------------------------------------
    best_a = None
    best_score = np.inf
    for vec in _initial_candidates(n, parent_arr):
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        sc = _objective_score(vec, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = vec.copy()
        if best_score <= 1.50274349:
            return best_a.tolist()

    # fallback to uniform if everything failed
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective_score(best_a, n, fft_len)

    # ----- first stage: LP‑based flattening -----------------------------
    cur_a = best_a.copy()
    cur_score = best_score
    max_lp_iters = 5
    lp_iter = 0

    while lp_iter < max_lp_iters and time.monotonic() < deadline:
        lp_iter += 1
        # current convolution bound
        conv_cur = _conv_self_fft(cur_a, n, fft_len)
        M = float(conv_cur.max())
        # build constraints matrix
        A = _build_conv_matrix(cur_a, n)
        # linear program: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break
        g = g0 / sum_g0                     # renormalise to sum=1
        g_score = _objective_score(g, n, fft_len)
        if g_score < cur_score:
            cur_a, cur_score = g, g_score
            if cur_score < best_score:
                best_score, best_a = cur_score, cur_a.copy()
            if best_score <= 1.50274349:
                return best_a.tolist()
        # if LP did not improve, stop early
        else:
            break

    # ----- second stage: exponentiated‑gradient descent -----------------
    iter_no_improve = 0
    alpha0 = 0.8
    min_alpha = 1e-5

    while time.monotonic() < deadline:
        # gradient of the active convolution entry
        conv = _conv_self_fft(cur_a, n, fft_len)
        max_val = float(conv.max())
        k = int(conv.argmax())
        g = np.zeros_like(cur_a)

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - idx
        mask = idx != j
        g[idx[mask]] = 2.0 * cur_a[j[mask]]
        if np.any(~mask):
            g[idx[~mask]] = cur_a[idx[~mask]]

        # step magnitude
        t = max(1, iter_no_improve + 1)
        alpha = max(min_alpha, alpha0 / np.sqrt(t))

        # multiplicative (exponentiated) update, stays on simplex
        cand = cur_a * np.exp(-alpha * g)
        # avoid complete collapse
        if cand.sum() == 0.0:
            cand = rng.random(n) + 1e-12
        cand = cand / cand.sum()

        # occasional random jitter to escape plateaus
        if rng.random() < 0.02:
            jitter = np.exp(rng.normal(0.0, 0.025, size=n))
            cand = cand * jitter
            cand = cand / cand.sum()

        cand_score = _objective_score(cand, n, fft_len)

        # accept if better, else with a tiny probability (simulated‑annealing style)
        if cand_score < cur_score:
            cur_a, cur_score = cand, cand_score
            iter_no_improve = 0
        else:
            # tiny uphill acceptance to avoid dead‑ends
            prob = np.exp(-(cand_score - cur_score) / (0.01 * cur_score + 1e-12))
            if rng.random() < prob:
                cur_a, cur_score = cand, cand_score
            iter_no_improve += 1

        if cur_score < best_score:
            best_score, best_a = cur_score, cur_a.copy()
            if best_score <= 1.50274349:
                break

        # restart from best if stuck too long
        if iter_no_improve > 1000:
            cur_a = best_a * np.exp(rng.normal(0.0, 0.03, size=n))
            cur_a = cur_a / cur_a.sum()
            iter_no_improve = 0

    return best_a.tolist()
