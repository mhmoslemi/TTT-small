def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 3.0)
    # base init
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = parent.copy()
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)
    # normalize to sum 1
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        a = np.full(n, 1.0/n, dtype=np.float64)
    else:
        a = a / s
    # best so far
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # also try uniform
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()
    # parameters
    target_score = 1.50274349
    fft_len = 1 << ((2*n - 2).bit_length())
    conv_len = 2*n - 1
    # momentum variable
    v = np.zeros_like(a)
    beta = 0.9
    eta0 = 0.5
    it = 0
    # top-k for gradient
    K = 3
    while time.monotonic() < deadline:
        it += 1
        # compute convolution via FFT
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_conv = conv.max()
        curr_score = 2.0 * n * max_conv
        if curr_score < best_score:
            best = a.copy()
            best_score = curr_score
            # optional early exit
            if best_score <= target_score:
                break
        # gradient from top K shifts
        # get top K indices
        top_idx = np.argpartition(-conv, K-1)[:K]
        g = np.zeros_like(a)
        for k in top_idx:
            start_i = max(0, k - (n-1))
            end_i = min(n-1, k)
            i_range = np.arange(start_i, end_i+1)
            g[i_range] += 2.0 * a[k - i_range]
        # average
        g /= K
        # step size schedule (sqrt decay)
        eta = eta0 / np.sqrt(it)
        # momentum update
        v = beta * v - eta * g
        a = a + v
        # project onto simplex
        a = _proj_simplex(a)
        # occasional jitter
        if it % 200 == 0:
            jitter = np.exp(rng.normal(0.0, 0.005, size=n))
            a = _proj_simplex(a * jitter)
    # final best
    return [float(v) for v in best]

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    # Translation of the classic O(n log n) algorithm.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # Degenerate case – return uniform
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
