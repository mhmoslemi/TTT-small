import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (non‑negative, sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # due to numerical noise, renormalise exactly
    s = w.sum()
    if s == 0:
        # fallback: uniform distribution
        w[:] = 1.0 / n
    else:
        w /= s
    return w


# ------------------------------------------------------------
# Analytic sqrt‑series (a*a == 1+…+x^{2n-2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Full linear convolution via FFT (returns the array)
# ------------------------------------------------------------
def _full_conv(a: np.ndarray, Lfft: int) -> np.ndarray:
    fv = np.fft.rfft(a, n=Lfft)
    conv = np.fft.irfft(fv * fv, n=Lfft)
    return conv[: 2 * a.size - 1]


# ------------------------------------------------------------
# Max of the (full) convolution a*a using FFT
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, Lfft: int) -> float:
    conv = _full_conv(a, Lfft)
    return float(conv.max())


# ------------------------------------------------------------
# Gradient of the active convolution lag k
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad

# ------------------------------------------------------------

    """Return a non‑empty sequence of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------------

    # --------------------------------------------------------------------
    if "height_sequence_1" in globals():

    else:

    n = a.shape[0]
    # Normalise to the simplex (scale does not affect the bound)
    a = np.maximum(a, 0.0)

    a = _project_simplex(a)

    # --------------------------------------------------------------------
    # bookkeeping

    Lfft = 1 << ((2 * n - 1).bit_length())  # next power of two >= 2n‑1

    best_score = 2.0 * n * cur_M

    # --------------------------------------------------------------------

    # --------------------------------------------------------------------
    return [float(x) for x in best]
