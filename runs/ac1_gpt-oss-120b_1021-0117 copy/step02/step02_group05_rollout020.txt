import numpy as np
import math, time
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(x: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound via FFT."""
    fa = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(x.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with low AC1 bound."""
    rng = np.random.default_rng(seed)

    # ---- initial sequence ------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        cur = parent.copy()
    except Exception:
        n = 256
        cur = _gen_binomial(n)

    # Ensure a minimal total mass (evaluator rejects <0.01)
    if cur.sum() < 0.01:
        cur[0] = max(cur[0], 0.01)

    L = _next_pow2(2 * n - 1)

    best = cur.copy()
    best_score = _ac1_score(best, n, L)
    cur_score = best_score

    # ---- hyper‑parameters ------------------------------------------------
    temperature = 1.0
    decay = 0.9995
    grad_eps = 0.05                # size of gradient step
    block_min = max(2, n // 200)   # smallest block for scaling moves
    block_max = max(2, n // 12)    # largest block
    transfer_max = 0.35            # max fraction moved between two positions
    lp_interval = max(200, n // 10)  # how often to invoke the LP step
    mix_ratio = 0.2                # convex‑combination weight for LP candidate
    target = 1.50274349

    iter_cnt = 0
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)  # 1 s head‑room

    while time.monotonic() < deadline:
        iter_cnt += 1
        move = rng.random()

        # --------------------------------------------------------------
        # 1) gradient‑guided mass transfer (≈50 % of moves)
        if move < 0.5:
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            kmax = int(np.argmax(conv))

            idx = np.arange(n)
            rev = kmax - idx
            mask = (rev >= 0) & (rev < n)
            g = np.zeros_like(cur)
            g[mask] = cur[rev[mask]]
            g[mask] *= 2.0                     # gradient of the max‑convolution entry
            d = g.mean() - g                    # direction that keeps total mass unchanged
            cand = cur + grad_eps * d

        # --------------------------------------------------------------
        # 2) block scaling (≈35 % of moves)
        elif move < 0.85:
            width = int(rng.integers(block_min, block_max + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start:start + width] *= factor

        # --------------------------------------------------------------
        # 3) random mass transfer (≈15 % of moves)
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * transfer_max
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = _ac1_score(cand, n, L)

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < target:
                break

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # temperature decay
        temperature *= decay

        # occasional restart from the best seen
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

        # --------------------------------------------------------------
        # 4) linear‑programming “Frank‑Wolfe” step (every lp_interval)
        if iter_cnt % lp_interval == 0:
            # recompute convolution of the current vector
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            max_b = float(conv.max())

            # active constraints: top‑few indices where conv is close to the maximum
            # keep at most 6 active rows to stay small
            top_k = 6
            if conv.size <= top_k:
                act_idx = np.arange(conv.size)
            else:
                act_idx = np.argpartition(-conv, top_k - 1)[:top_k]

            m = act_idx.size
            A = np.zeros((m, n), dtype=np.float64)
            for r, k in enumerate(act_idx):
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                cols = np.arange(lo, hi + 1, dtype=int)
                A[r, cols] = cur[k - cols]

            b = np.full(m, max_b, dtype=np.float64)
            c = -np.ones(n, dtype=np.float64)   # maximize sum(g) -> minimize -sum(g)

            # solve LP (highs is fast and accepts dense A for modest size)
            try:
                res = linprog(
                    c,
                    A_ub=A,
                    b_ub=b,
                    bounds=[(0, None)] * n,
                    method="highs",
                    options={"presolve": True, "time_limit": 5.0},
                )
            except Exception:  # pragma: no cover
                res = None

            if res is not None and res.success:
                g = res.x
                sum_cur = cur.sum()
                sum_g = g.sum()
                if sum_g > 0:
                    scale = sum_cur / sum_g
                    g = np.clip(g * scale, 0.0, 1000.0)

                    cand = (1.0 - mix_ratio) * cur + mix_ratio * g
                    cand_score = _ac1_score(cand, n, L)

                    if cand_score < best_score:
                        best = cand.copy()
                        best_score = cand_score
                        if best_score < target:
                            break

                    # accept with the usual SA rule
                    accept_lp = False
                    if cand_score < cur_score:
                        accept_lp = True
                    else:
                        prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
                        if rng.random() < prob:
                            accept_lp = True
                    if accept_lp:
                        cur = cand
                        cur_score = cand_score

    # ------------------------------------------------------------------
    return [float(v) for v in best]
