import time
import numpy as np

# ------------------------------------------------------------
# Helper: compute maximal entry of full convolution a*a
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, Lfft: int) -> float:
    fv = np.fft.rfft(a, n=Lfft)
    conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Helper: create a normalized non‑negative vector from a raw array
# ------------------------------------------------------------
def _normalise(v: np.ndarray) -> np.ndarray:
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        v[:] = 1.0 / v.size
    else:
        v = v / s
    return v


# ------------------------------------------------------------
# Parameterised families (used for quick global search)
# ------------------------------------------------------------
def _power_law(n: int, p: float) -> np.ndarray:
    # a_i ∝ (i+1)^(-p)
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return _normalise(a)


def _linear_decay(n: int, q: float) -> np.ndarray:
    # a_i ∝ (1 - i/(n-1))**q   (q>0)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    t = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    a = t ** q
    return _normalise(a)

# ------------------------------------------------------------
