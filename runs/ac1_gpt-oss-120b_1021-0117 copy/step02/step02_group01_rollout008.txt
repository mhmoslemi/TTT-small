import time
import numpy as np

# --------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x>=0, sum=1}
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

# --------------------------------------------------------------
# Helper: analytic sqrt‑series (a*a = 1+…+x^{2n‑2})
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# --------------------------------------------------------------
# Main search routine
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)  # safety margin

    # ------------------------------------------------------------------
    # Initialise sequence (use provided parent if present)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = _sqrt_series(256)
    a = _project_simplex(a)                 # enforce sum = 1, non‑negative
    n = a.size

    # FFT size: must accommodate convolution lengths up to 3n‑2
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Scoring utility (computes the AC1 upper bound)
    # ------------------------------------------------------------------
    def _score(seq: np.ndarray) -> float:
        Fa = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
        max_val = conv.max()
        s = seq.sum()
        return 2.0 * n * max_val / (s * s)

    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    step = 0.15                # initial step magnitude
    min_step = 1e-5
    decay = 0.9995
    prob_random = 0.015        # chance of a random mass‑transfer move
    alpha = 5.0                # softmax sharpness
    alpha_inc = 1.003          # gradual sharpening

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- compute full auto‑correlation and its FFT ----
        Fa = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
        max_conv = conv_full.max()

        # ---- smooth max via softmax (weights sum to 1) ----
        w_raw = np.exp(alpha * (conv_full - max_conv))
        w = w_raw / w_raw.sum()

        # ---- gradient = 2 * (a * w_rev) restricted to length n ----
        w_rev = w[::-1]
        Fw_rev = np.fft.rfft(w_rev, n=L)
        conv_grad_full = np.fft.irfft(Fa * Fw_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * conv_grad_full[n - 1 : 2 * n - 1][::-1]   # shape (n,)

        # ---- propose a gradient step and project back onto simplex ----
        cand = a - step * grad
        cand = _project_simplex(cand)

        # ---- evaluate candidate ----
        cand_score = _score(cand)

        # ---- acceptance logic ----
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            # simple back‑off when not improving
            step *= 0.7

        # ---- occasional random mass‑transfer move for exploration ----
        if rng.random() < prob_random:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.4
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                cand2_score = _score(cand2)
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()

        # ---- schedule updates ----
        step = max(min_step, step * decay)
        alpha = min(alpha * alpha_inc, 50.0)   # avoid overflow

        # ---- early exit safeguard ----
        if time.monotonic() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
