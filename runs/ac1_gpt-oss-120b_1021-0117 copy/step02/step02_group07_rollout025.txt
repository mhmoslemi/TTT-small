def _initial_binomial(n):
    a = np.empty(n,dtype=np.float64)
    a[0]=1.0
    for i in range(1,n):
        a[i]=a[i-1]*(2*i-1)/(2*i)
    return a

def _conv(seq, fft_len):
    f = np.fft.rfft(seq, n=fft_len)
    return np.fft.irfft(f*f, n=fft_len)[:2*len(seq)-1]

def _score(seq, fft_len):

import numpy as np, time, math

def _initial_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, decreasing positive."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution via FFT (real)."""
    f = np.fft.rfft(seq, n=fft_len)
    return np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]


def _score(seq: np.ndarray, fft_len: int):
    """Return (score, conv array, max entry)."""
    conv = _conv(seq, fft_len)
    max_b = float(conv.max())
    s = float(seq.sum())
    if s <= 0.0:
        return float("inf"), conv, max_b
    score = 2.0 * seq.size * max_b / (s * s)
    return score, conv, max_b
