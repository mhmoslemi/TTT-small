def _sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # produce dot reversed slice
            s = float(np.dot(a[1:k], a[k-1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a

def _shape_semicircle(n):
    i = np.arange(n, dtype=np.float64)
    t = (2 * i / (n - 1) - 1)
    a = np.sqrt(np.maximum(0.0, 1 - t * t))

    center = (n - 1) / 2.0
    a = np.exp(-np.abs(i - center) / sigma)
    return a

    a = ((i + 1) * (n - i)) ** p
    return a

import time

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(a):
    """Full linear convolution via FFT (real input)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv

def _score(a):
    """Objective: 2*n*max(conv)/sum(a)^2, assuming a non‑negative."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

    """Return (max_conv, gradient of max conv wrt a)."""
    n = a.shape[0]

    g = np.zeros_like(a)

        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
