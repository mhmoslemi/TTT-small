import time
import numpy as np

# ---------- helper utilities ----------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    Z = X * Y
    full = np.fft.irfft(Z, n=fft_len)
    return full[: len(x) + len(y) - 1]


def _initial_candidates(n: int, parent_arr: np.ndarray | None):
    """Generate a handful of plausible start vectors (already normalised)."""
    cand = []

    # parent if available
    if parent_arr is not None and parent_arr.shape == (n,):
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            cand.append(p / s)

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # sin^p shapes
    idx = np.arange(n, dtype=np.float64)
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    for pwr in (0.5, 1.0, 2.0, 3.0):
        v = sin_vals ** pwr
        cand.append(v / v.sum())

    # polynomial hill shapes (i+1)*(n-i) raised to a power
    base = (idx + 1) * (n - idx)
    for pwr in (0.5, 1.0, 1.5, 2.0):
        v = base ** pwr
        cand.append(v / v.sum())

    return cand
