    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # dot product of a[1:k] with reverse of a[1:k]
            s = np.dot(a[1:k], a[k-1:0:-1])
        a[k] = (1.0 - s) * 0.5
    # Numerical noise can produce tiny negatives; clip them.
    a[a < 0] = 0.0
    return a
