    n = len(f)
    m = len(shifts)
    # Build A_ub
    data = []
    row_ind = []
    col_ind = []
    # For each shift r, for each i in range start_i..end_i:
    # coefficient for variable i is f[k-i]
    # coefficient for t variable is -1 (col n)
    for r, k in enumerate(shifts):
        start_i = max(0, k - (n-1))
        end_i = min(n-1, k)
        # length = end_i - start_i + 1
        # fill data:

            data.append(f[k - i])
            row_ind.append(r)
            col_ind.append(i)

        col_ind.append(n)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    from scipy.sparse import csr_matrix
    from scipy.optimize import linprog

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.shape[0]
        base = parent_arr.copy()
    except Exception:
        n = 256
        base = np.ones(n) / n

    # Ensure nonnegatives, clip
    base = np.clip(base, 0.0, 1000.0)
    # Normalize sum to 1
    s = base.sum()
    if s == 0:
        base = np.ones(n) / n
        s = 1.0
    base /= s

    # Compute initial score
    def ac1_score(a):
        # Use FFT for speed
        L = 1 << ((2*n - 2).bit_length())
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        maxb = conv.max()
        sum_a = a.sum()
        if sum_a < 1e-12:
            return float('inf')
        return 2.0 * n * maxb / (sum_a * sum_a)

    best = base.copy()
    best_score = ac1_score(best)

    f_seq = best.copy()
    # Parameters
    mix_frac = 0.3
    max_outer_iters = 30
    top_k = 200  # number of shifts to enforce constraints

    while time.monotonic() < deadline and max_outer_iters > 0:
        max_outer_iters -= 1

        # compute convolutions
        L = 1 << ((2*n - 2).bit_length())
        fa = np.fft.rfft(f_seq, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        M = conv.max()
        cur_score = 2.0 * n * M  # sum=1 due to normalization

        # early exit if already good
        if cur_score < best_score:
            best = f_seq.copy()
            best_score = cur_score
            if best_score < 1.50274349:
                break

        # Choose shifts with largest conv values
        # Determine indices of top shifts (could be duplicates)
        if top_k >= conv.size:
            shifts = np.arange(conv.size, dtype=np.int64)
        else:
            # argpartition returns unsorted top; sort descending optionally
            idx = np.argpartition(-conv, top_k)[:top_k]
            # sort them descending for nicer matrix? Not needed.
            shifts = np.sort(idx)  # sorting not required but maybe consistent
        # Solve LP to find g minimizing max convolution with f_seq
        # Build the A matrix
        # Use helper function to build sparse A (must be inside loop because f_seq changes)
        n_vars = n + 1  # g plus t
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(shifts):
            start_i = max(0, k - (n - 1))
            end_i = min(n - 1, k)
            i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
            coeffs = f_seq[k - i_vals]
            rows.extend([row_idx] * i_vals.size)
            cols.extend(i_vals)
            data.extend(coeffs)
            # t coefficient -1
            rows.append(row_idx)
            cols.append(n)  # column for t
            data.append(-1.0)
        A_ub = csr_matrix((data, (rows, cols)), shape=(len(shifts), n_vars))
        b_ub = np.zeros(len(shifts))

        # Equality constraint sum(g)=1
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(0, None)]  # t >=0

        c = np.zeros(n_vars)
        c[-1] = 1.0  # minimize t

        # Solve
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds,
                          method='highs',
                          options={'presolve': True})
        except Exception:
            # fallback: break loop
            break

        if not res.success:
            # If fails, maybe try fewer constraints
            # Reduce top_k or break
            # fallback: random perturbation
            # but we break to not waste time
            break

        g_seq = res.x[:n]
        # Ensure sum close to 1 due to numeric, renorm
        g_seq = np.clip(g_seq, 0.0, None)
        g_seq_sum = g_seq.sum()
        if g_seq_sum == 0:
            # fallback to previous
            continue
        g_seq /= g_seq_sum

        # Mix with current f_seq
        f_new = (1.0 - mix_frac) * f_seq + mix_frac * g_seq

        # Clip and renorm (though mixing maintains sum approx 1)
        f_new = np.clip(f_new, 0.0, None)
        f_new_sum = f_new.sum()
        if f_new_sum == 0:
            continue
        f_new /= f_new_sum

        # Evaluate new candidate
        new_score = ac1_score(f_new)

        if new_score < best_score:
            best = f_new.copy()
            best_score = new_score

        # Accept new f_seq based on improvement or some probability (optional)
        # For simplicity, accept if new_score <= cur_score else keep old
        if new_score <= cur_score:
            f_seq = f_new
        else:
            # maybe accept with small prob to escape
            if rng.random() < 0.01:
                f_seq = f_new

        # adapt mix_frac gradually
        mix_frac = max(0.05, mix_frac * 0.95)

    # Return best as list
    return [float(x) for x in best]

import time
import math
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score_fft(a: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 score using FFT (same as the official evaluator)."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _build_A_ub(f: np.ndarray, shifts: np.ndarray) -> csr_matrix:
    """
    Build sparse matrix A for constraints (f * g)_k <= t,
    where g is the variable vector and t is the last variable.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    t_col = n  # index of the t variable
    for r, k in enumerate(shifts):
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        coeffs = f[k - i_vals]                     # f shifted
        rows.extend([r] * i_vals.size)
        cols.extend(i_vals)
        data.extend(coeffs)
        # coefficient for t (negative because we need (f*g)_k - t <= 0)
        rows.append(r)
        cols.append(t_col)
        data.append(-1.0)
    return csr_matrix((data, (rows, cols)), shape=(len(shifts), n + 1))


def _solve_min_max(f: np.ndarray, shifts: np.ndarray) -> np.ndarray | None:
    """
    Solve the linear program:
        minimize   t
        subject to (f * g)_k <= t   for k in shifts
                   sum(g) = 1
                   g >= 0, t >= 0
    Returns the optimal g (normalized) or None on failure.
    """
    n = f.shape[0]
    A_ub = _build_A_ub(f, shifts)
    b_ub = np.zeros(A_ub.shape[0])

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds: g_i >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]

    # Objective: minimize t (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs')
    except Exception:
        return None

    if not res.success:
        return None

    g = res.x[:n]
    g = np.clip(g, 0.0, None)
    s = g.sum()
    if s == 0:
        return None
    return g / s

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- initialise sequence -----
    try:

    if s == 0:

    L = _next_pow2(2 * n - 1)

    best = base.copy()

    # ----- main optimisation -----

    mix_frac = 0.3          # how far we move towards LP solution
    top_k = 200            # number of active shifts for the LP
    outer_iters = 0

    while time.monotonic() < deadline and outer_iters < 30:
        outer_iters += 1

        # current score
        cur_score = _ac1_score_fft(cur, n, L)
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score
            if best_score < 1.50274349:
                break

        # pick the most critical shifts (largest convolution values)
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        if top_k >= conv.size:
            shifts = np.arange(conv.size, dtype=np.int64)
        else:
            idx = np.argpartition(-conv, top_k)[:top_k]
            shifts = np.sort(idx)          # order not required but stable

        # solve the linear programme to reduce the max overlap
        g = _solve_min_max(cur, shifts)
        if g is None:
            # fall back to a cheap random perturbation if LP fails
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))

            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start:start + width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand = cand / cand.sum()

            # blend the LP solution with the current sequence
            cand = (1.0 - mix_frac) * cur + mix_frac * g
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand = cand / cand.sum()

        cand_score = _ac1_score_fft(cand, n, L)

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        # accept the candidate if it does not worsen the current score
        if cand_score <= cur_score or rng.random() < 0.01:
            cur = cand

        # slowly anneal the mixing fraction
        mix_frac = max(0.05, mix_frac * 0.94)

    # ----- light stochastic refinement if time remains -----
    cur2 = best.copy()
    temperature = 1.0
    temp_decay = 0.9994
    while time.monotonic() < deadline:
        # simple random block move
        width = int(rng.integers(max(1, n // 250), max(2, n // 10)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        factor = math.exp(rng.normal(0.0, 0.08))
        cand = cur2.copy()
        cand[start:start + width] *= factor
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand / cand.sum()

        cur_score = _ac1_score_fft(cur2, n, L)

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            cur2 = cand
            continue

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur2 = cand

        temperature *= temp_decay

        # early stop if we have already beaten the target comfortably
        if best_score < 1.45:
            break

    return [float(v) for v in best]
