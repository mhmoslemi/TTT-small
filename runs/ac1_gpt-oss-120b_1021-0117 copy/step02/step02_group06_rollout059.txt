The evaluator does:

```
def evaluate_sequence(sequence: list) -> float:
    # checks list nonempty and entries numeric, not nan/inf, nonnegative after clipping.
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence] # clip negativity
    sequence = [min(1000.0, x) for x in sequence] # clip max
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))
```

def _conv_self_fft(a, n):
    L = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    B = A * A
    full = np.fft.irfft(B, n=L)

    n = v.shape[0]

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, n: int, L: int) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2n‑1) using FFT of length L."""
    A = np.fft.rfft(a, n=L)
    B = A * A
    full = np.fft.irfft(B, n=L)
    return full[: 2 * n - 1]

                break

        # gradient of a smooth max (soft‑max) surrogate
        grad = _smooth_gradient(a, conv, beta, n, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        direction = m_hat / (np.sqrt(v_hat) + eps)

        lr = lr0 / np.sqrt(it)
        cand = a - lr * direction
        cand = _project_onto_simplex(cand)

        # occasional jitter to escape plateaus
        if rng.random() < 0.012:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            cand = cand * jitter
            cand = _project_onto_simplex(cand)

        a = cand
        beta = min(beta_end, beta + beta_inc)

        # safety stop if time is almost gone
        if time.monotonic() > deadline - 0.5:
            break

    # fallback to the best encountered solution
    if global_best_a is None:
        # this should never happen, but return a uniform vector just in case
        return np.full(n, 1.0 / n).tolist()
    return global_best_a.tolist()
