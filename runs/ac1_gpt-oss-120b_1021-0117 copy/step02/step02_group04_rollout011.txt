def ac1_score(a):
    n = len(a)
    L = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
    maxb = conv.max()
    s = a.sum()
    if s <= 0: return float('inf')
    return 2.0 * n * maxb / (s * s)
```

import math
import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _binomial_seq(n: int) -> np.ndarray:
    """Classic step a_i = C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, L: int, n: int) -> float:
    """Exact AC1 upper bound for a non‑negative sequence."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s <= 0.0:
        return float("inf")
    return 2.0 * n * max_b / (s * s)
