import numpy as np
import time
import math

def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1, decreasing positive."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _score(seq: np.ndarray, fft_len: int) -> float:
    """Return the AC1 upper bound score for seq."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    max_c = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf")
    return 2.0 * seq.size * max_c / (s * s)

def _conv(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution (length 2n‑1)."""
    f = np.fft.rfft(seq, n=fft_len)
    return np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
