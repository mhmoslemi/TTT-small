import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    pos = np.where(rho_arr > 0)[0]
    if pos.size:
        rho = pos[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

# ----------------------------------------------------------------------
# Full convolution via FFT (circular length L, then truncated)
# ----------------------------------------------------------------------
def _conv_full(x: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(x) - 1]
    return conv

# Analytic sqrt‑series (optional fallback initialisation)
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a
