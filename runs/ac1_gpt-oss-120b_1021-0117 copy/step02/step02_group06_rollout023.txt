import time
import numpy as np

# ----- utilities --------------------------------------------------------------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _symmetrize_half(half: np.ndarray, n: int) -> np.ndarray:
    """Mirror a half‑vector to obtain a symmetric length‑n vector."""
    if n % 2 == 0:
        return np.concatenate((half, half[::-1]))
    # odd length: central element should not be duplicated
    return np.concatenate((half, half[-2::-1]))


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a using real FFT (length 2*len(a)-1)."""
    A = np.fft.rfft(a, fft_len)
    B = A * A
    full = np.fft.irfft(B, fft_len)
    return full[: 2 * a.shape[0] - 1]


def _score_vector(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact score = 2*n*max(a*a) for sum(a)=1."""
    conv = _conv_self_fft(a, fft_len)
    return 2.0 * n * float(conv.max())


def _catalan_half_weights(m: int) -> np.ndarray:
    """Catalan‐style weights for a half‑vector of length m."""
    w = np.empty(m, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, m):
        w[i] = w[i - 1] * ((2.0 * i) * (2.0 * i - 1.0)) / (i * i * 4.0)
    return w


def _initial_candidates(n: int, parent_arr: np.ndarray | None):
    """Generate a modest list of plausible start vectors (already normalized)."""
    cand = []

    # 1. parent if present (symmetrized)
    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            cand.append(p / s)

    # 2. symmetric Catalan shape
    half_len = n // 2 if n % 2 == 0 else (n + 1) // 2
    w = _catalan_half_weights(half_len)
    cand.append(_symmetrize_half(w / w.sum(), n))

    # 3. symmetric quadratic (triangular) shape a_i ∝ (i+1)*(n-i)
    idx = np.arange(n, dtype=np.float64)
    tri = (idx + 1.0) * (n - idx)
    cand.append(tri / tri.sum())

    # 4. sine‑power shapes (already symmetric)
    theta = np.pi * (idx + 0.5) / n
    sinv = np.sin(theta)
    for p in (0.5, 1.0, 2.0):
        v = sinv ** p
        cand.append(v / v.sum())

    # 5. a few random Dirichlet draws (lightweight)
    rng = np.random.default_rng(12345)
    for _ in range(5):
        v = rng.gamma(1.0, 1.0, size=n)
        cand.append(v / v.sum())

    # ensure all are proper simplexes
    clean = []
    for v in cand:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 1e-12:
            clean.append(v / s)
    return clean


# ----- core optimisation -----------------------------------------------------

def _descent_run(start: np.ndarray, n: int, fft_len: int,
                 target: float, deadline: float, rng: np.random.Generator):
    """Projected sub‑gradient descent with momentum and occasional jitter."""
    a = start.copy()
    best_a = a.copy()
    best_score = _score_vector(a, n, fft_len)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9                    # momentum coefficient
    init_alpha = 0.8              # initial step magnitude
    it = 0

    while time.monotonic() < deadline:
        conv = _conv_self_fft(a, fft_len)
        max_b = float(conv.max())
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if best_score <= target:
                break

        # sub‑gradient of the active convolution term
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        # momentum update and step size schedule
        v = beta * v + (1.0 - beta) * g
        alpha = init_alpha / np.sqrt(it + 1.0)
        cand = a - alpha * v
        cand = np.maximum(cand, 0.0)
        cand = _project_onto_simplex(cand)

        # occasional multiplicative jitter to escape plateaus
        if rng.random() < 0.03:
            jitter = np.exp(rng.normal(0.0, 0.015, size=n))
            cand = cand * jitter
            cand = _project_onto_simplex(cand)

        a = cand
        it += 1

    return best_a, best_score
