import time

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x>=0, sum(x)=1}. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution a*a (length 2*len(a)-1) using real FFT."""
    A = np.fft.rfft(a, n=fft_len)
    full = np.fft.irfft(A * A, n=fft_len)
    n = a.shape[0]
    return full[: 2 * n - 1]


def _gradient_fft(a: np.ndarray, conv: np.ndarray, rho: float,
                  fft_len_self: int, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth approximation of max_k (a*a)[k]
    using weights w_k ∝ exp(rho*(conv_k-max(conv))).
    Returns a vector of length n.
    """
    n = a.shape[0]
    max_b = conv.max()
    diff = conv - max_b            # ≤ 0
    w = np.exp(rho * diff)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum                  # shape (2n-1,)

    # convolution of reversed a with w
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    W = np.fft.rfft(w, n=fft_len_grad)
    conv_full = np.fft.irfft(A_rev * W, n=fft_len_grad)
    start = n - 1                  # index where length‑n slice begins
    grad = 2.0 * conv_full[start:start + n]
    return grad

    cand.append(w / w.sum())

    # Simple power‑law shape (symmetric)
    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1) * (n - idx)
    for p in (0.5, 1.0, 1.5):
        v = base ** p
        cand.append(v / v.sum())

    # Sin^p shape
    theta = np.pi * (idx + 0.5) / n
    sinv = np.sin(theta)
    for p in (0.5, 1.0, 2.0):
        v = sinv ** p
        cand.append(v / v.sum())

    return cand


# ----- main optimisation -----------------------------------------------------

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that minimises
    the AC1 upper bound used by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)  # safety margin

    # ----- obtain parent if present -----------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
    except Exception:
        parent_arr = None

    n = int(parent_arr.shape[0]) if parent_arr is not None else 256

    # ----- FFT lengths -------------------------------------------------------
    fft_len_self = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    # ----- pick best initial candidate (by cheap proxy) ----------------------
    best_init = None
    best_score = np.inf
    for vec in _initial_candidates(n, parent_arr):
        conv = _conv_self_fft(vec, fft_len_self)
        max_b = float(conv.max())
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_init = vec.copy()
        if best_score <= 1.50274349:
            return best_init.tolist()

    # ----- optimisation settings ---------------------------------------------
    a = best_init.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self_fft(a, fft_len_self).max()

    rho = 1.0                     # smoothing parameter
    eta0 = 0.7                    # base step size for multiplicative update
    iter_no_improve = 0
    max_no_improve = 800
    iter_cnt = 0

    while time.monotonic() < deadline:
        conv = _conv_self_fft(a, fft_len_self)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            iter_no_improve = 0
            if best_score <= 1.50274349:
                break
        else:
            iter_no_improve += 1

        # gradient of smoothed max
        grad = _gradient_fft(a, conv, rho, fft_len_self, fft_len_grad)

        # multiplicative (exponentiated) gradient step
        eta = eta0 / np.sqrt(iter_cnt + 1)
        exp_factor = np.exp(-eta * grad)
        cand = a * exp_factor
        s = cand.sum()
        if s == 0.0:
            cand = np.full_like(a, 1.0 / n)
        else:
            cand = cand / s

        # evaluate candidate
        conv_cand = _conv_self_fft(cand, fft_len_self)
        max_b_cand = conv_cand.max()

        if max_b_cand < max_b:
            a = cand
            rho = min(rho * 1.05, 80.0)          # tighten smoothing gradually
        else:
            # fallback to additive projected sub‑gradient step
            step = eta
            cand2 = a - step * grad
            cand2 = _project_onto_simplex(cand2)
            conv_cand2 = _conv_self_fft(cand2, fft_len_self)
            if conv_cand2.max() < max_b:
                a = cand2
                rho = min(rho * 1.05, 80.0)
            else:
                # occasional random jitter to escape stagnation
                if rng.random() < 0.02:
                    jitter = np.exp(rng.normal(0.0, 0.025, size=n))
                    a = a * jitter
                    a = _project_onto_simplex(a)
                    rho = max(rho * 0.9, 1.0)

        # restart if stalled for long
        if iter_no_improve > max_no_improve:
            perturb = np.exp(rng.normal(0.0, 0.03, size=n))
            a = best_a * perturb
            a = _project_onto_simplex(a)
            rho = 1.0
            iter_no_improve = 0

        iter_cnt += 1

    return best_a.tolist()
