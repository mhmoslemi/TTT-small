def _project_simplex(v):
    # ensure sum of projection = 1, and nonnegative.
    # Works for any sum target; we will set target = 1.
    # Input: numpy array
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # use sqrt series of length 256
        def _sqrt_series(n):
            a = np.empty(n, dtype=np.float64)
            a[0] = 1.0
            for k in range(1, n):
                if k == 1:
                    s = 0.0
                else:
                    s = float(np.dot(a[1:k], a[k-1:0:-1]))
                a[k] = (1.0 - s) / 2.0
            return a
        a = _sqrt_series(256)

    # project onto simplex (sum=1)
    a = _project_simplex(a)

    n = a.shape[0]
    L = 1 << ((3*n - 2).bit_length())  # FFT size power of two

    # initial scoring
    def _score_and_fft(vec):
        A = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(A * A, n=L)[:2*n-1]
        max_val = conv.max()
        total = vec.sum()
        # avoid divide by zero
        if total == 0:
            sc = np.inf
        else:
            sc = 2.0 * n * max_val / (total * total)
        return sc, max_val, A, conv

    cur_score, cur_max, A_cur, conv_cur = _score_and_fft(a)
    best = a.copy()
    best_score = cur_score

    step = 0.2
    min_step = 1e-8
    max_step = 1.0
    decay_inc = 1.05
    decay_dec = 0.5
    alpha = 10.0
    # counters for stagnation
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() < deadline:
        # compute smooth weights
        # avoid division by zero
        if cur_max <= 0:

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex sum=1, v>=0."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

# ----------------------------------------------------------------------
# Score and FFT helpers
# ----------------------------------------------------------------------
def _score_and_fft(vec: np.ndarray, n: int, L: int):
    """Return (score, max_conv, fft(vec), full conv vec*vec)."""
    A = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = vec.sum()
    if s == 0.0:
        score = np.inf
    else:
        score = 2.0 * n * max_val / (s * s)
    return score, max_val, A, conv
