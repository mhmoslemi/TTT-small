# (coefficients of (1 - x)^{-1/2})
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # inner product a[1:k]·a[k-1:0:-1]
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty sequence of non‑negative floats with a low AC1 bound."""
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # ------------------------------------------------------------------
    # Determine a modest size to explore – the bound improves for a
    # certain finite length and then slowly rises to π/2 ≈ 1.57.
    # Length 1024 is ample and cheap; if the parent sequence exists we
    # respect its length as an upper bound.
    # ------------------------------------------------------------------
    max_n = 1024
    if "height_sequence_1" in globals():
        max_n = min(len(height_sequence_1), max_n)

    # ------------------------------------------------------------------
    # Build the analytic series up to max_n (O(max_n²) operations, tiny).
    # ------------------------------------------------------------------
    a = _sqrt_series(max_n)
    a = np.maximum(a, 0.0)          # guard against possible negative rounding

    # ------------------------------------------------------------------
    # For every prefix length n the convolution peak is exactly 1,
    # so the AC1 bound equals 2*n / (sum_{i<n} a_i)².
    # We evaluate this for all n and pick the best prefix.
    # ------------------------------------------------------------------
    cumsum = np.cumsum(a)
    n_vals = np.arange(1, max_n + 1, dtype=np.float64)
    # avoid division by zero – cumsum is strictly increasing for our series
    scores = 2.0 * n_vals / (cumsum * cumsum)

    best_idx = int(np.argmin(scores))          # zero‑based index
    best_seq = a[: best_idx + 1]               # best prefix (length = best_idx+1)

    # ------------------------------------------------------------------
    # If we ran out of time after the heavy work, fall back to a tiny
    # uniform sequence – it still satisfies the constraints.
    # ------------------------------------------------------------------
    if time.monotonic() > deadline:
        short_len = min(10, max_n)
        best_seq = np.full(short_len, 1.0 / short_len, dtype=np.float64)

    # ------------------------------------------------------------------
    # Return a plain Python list of floats.
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq]
