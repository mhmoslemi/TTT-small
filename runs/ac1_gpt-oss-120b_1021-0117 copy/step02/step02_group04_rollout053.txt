def _next_pow2(x): return 1 << (x-1).bit_length()
def _ac1_score_fft(arr, n, L):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s*s)
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.5)
    # determine length
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256
    L = _next_pow2(2*n - 1)
    # helper to produce normalized candidate
    def _norm(a):
        s = a.sum()
        if s <= 0:
            a = np.ones_like(a)
            s = a.sum()
        a = a / s
        return a
    # generate a few analytic seeds
    seeds = []
    # parent if available
    if parent is not None:
        seeds.append(_norm(parent.astype(np.float64)))
    # binomial
    binom = np.empty(n, dtype=np.float64)

    seeds.append(_norm(binom))
    # triangular (i+1)*(n-i)

import math
import time

import numpy as np

# ---------- helpers ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 bound using FFT (matches the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _norm(arr: np.ndarray) -> np.ndarray:
    """Scale to L1‑norm 1 (guard against zero sum)."""
    s = arr.sum()
    if s <= 0:
        arr = np.ones_like(arr, dtype=np.float64)
        s = arr.sum()
    return arr / s
