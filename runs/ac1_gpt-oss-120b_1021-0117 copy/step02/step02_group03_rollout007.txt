import time
import numpy as np

# ------------------------------------------------------------
# Helper: project onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Helper: maximal value of full autocorrelation a*a via FFT
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, Lfft: int) -> float:
    fv = np.fft.rfft(a, n=Lfft)
    conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Helper: gradient of a*a at lag k (sub‑gradient of the max)
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad


# ------------------------------------------------------------
# Optional analytic initialisation if no parent is supplied
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------
    # Initial sequence (parent if present, else sqrt series)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = _sqrt_series(256).astype(np.float64)

    n = a.shape[0]
    a = a / a.sum()                       # normalize to sum = 1
    a = _project_simplex(a)               # safety projection

    # --------------------------------------------------------
    # Pre‑computed FFT length (next power of two)
    # --------------------------------------------------------
    Lfft = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Bookkeeping
    # --------------------------------------------------------
    best = a.copy()
    best_score = 2.0 * n * _max_convolution(a, Lfft)
    cur_score = best_score

    # Gradient descent parameters
    eta0 = 0.8                 # initial step scale
    beta = 0.9                 # momentum decay
    velocity = np.zeros_like(a)
    iteration = 0
    target_score = 1.50274349

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # ----- compute full autocorrelation -----
        fv = np.fft.rfft(a, n=Lfft)
        conv_full = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]

        M = float(conv_full.max())
        cur_score = 2.0 * n * M

        # ----- active set: several largest lags -----
        top_k = 5
        # indices of the top_k largest entries (unsorted)
        idxs = np.argpartition(conv_full, -top_k)[-top_k:]
        # sort them descending for determinism (optional)
        idxs = idxs[np.argsort(-conv_full[idxs])]

        # ----- aggregate gradient over the active set -----
        grad = np.zeros_like(a)
        for k in idxs:
            grad += _grad_for_k(a, int(k))
        norm = np.linalg.norm(grad)
        if norm == 0.0:
            # degenerate case: take a tiny random perturbation
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = 0.001 * a[i]
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)
            continue

        direction = -grad / norm      # descent direction (unit)

        # ----- line search (simple backtracking) -----
        eta = eta0 / np.sqrt(iteration)
        cand = None
        cand_score = np.inf
        for ls_step in range(5):
            step = eta * (0.5 ** ls_step)
            trial = _project_simplex(a + step * direction)
            trial_M = _max_convolution(trial, Lfft)
            trial_score = 2.0 * n * trial_M
            if trial_score < cand_score:
                cand_score = trial_score
                cand = trial
            if trial_score < cur_score:
                break    # accept improvement early

        # ----- keep global best -----
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            # early exit if we already beat the target
            if best_score < target_score:
                break

        # ----- acceptance rule -----
        accept = cand_score < cur_score or rng.random() < 0.004
        if accept:
            # momentum update
            velocity = beta * velocity + (cand - a)
            a = cand
        else:
            # decay momentum slowly when we reject
            velocity = beta * velocity

        # ----- occasional random mass‑transfer for exploration -----
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.12
                delta = a[i] * frac
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)

        # ----- tiny random jitter to avoid stagnation -----
        if rng.random() < 0.001:
            noise = rng.normal(scale=1e-5, size=n)
            a = _project_simplex(a + noise)

    # --------------------------------------------------------
    # Return the best sequence found
    # --------------------------------------------------------
    return [float(x) for x in best]
