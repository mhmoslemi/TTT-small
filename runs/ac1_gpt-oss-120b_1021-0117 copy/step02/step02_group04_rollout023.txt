            raise ValueError
        n = int(parent.size)
        base = parent.astype(np.float64)
    except Exception:  # no parent available
        n = 512
        base = _gen_binomial(n)

    # Normalise to sum = 1 (scale invariance of the bound)
    base = np.maximum(base, 0.0)
    s = base.sum()
    if s < 1e-12:
        base = np.ones(n, dtype=np.float64) / n
    else:
        base = base / s

    L = _next_pow2(2 * n - 1)

    # ----- bookkeeping -----
    best = base.copy()
    best_score = _ac1_score_fft(best, n, L)

    cur = best.copy()
    cur_score = best_score

    temperature = 1.0
    temp_decay = 0.9995
    grad_lr = 0.20          # learning rate for gradient‑inspired step
    min_lr = 0.02
    lr_decay = 0.9997

    # ----- optimisation loop -----
    while time.monotonic() < deadline:
        # current convolution and active lag
        fa_cur = np.fft.rfft(cur, n=L)
        conv_cur = np.fft.irfft(fa_cur * fa_cur, n=L)[: 2 * n - 1]
        max_b = float(conv_cur.max())
        max_idx = int(conv_cur.argmax())
        cur_score = 2.0 * n * max_b / (cur.sum() ** 2)   # sum == 1 ⇒ denominator = 1

        # ----- gradient‑inspired move -----
        idx = np.arange(n)
        j = max_idx - idx
        mask = (j >= 0) & (j < n)
        mirror = np.zeros_like(cur)
        mirror[mask] = cur[j[mask]]
        grad = 2.0 * mirror                     # sub‑gradient of max‑convolution
        cand = cur - grad_lr * grad
        cand = np.clip(cand, 0.0, None)          # enforce non‑negativity
        s_cand = cand.sum()
        if s_cand < 1e-12:                       # avoid collapse
            cand = cur.copy()
            s_cand = cand.sum()
        cand = cand / s_cand                     # renormalise to sum = 1
        cand_score = _ac1_score_fft(cand, n, L)

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < 1.50274349:
                break

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            delta = (cur_score - cand_score) / (temperature + 1e-12)
            if delta > 700:
                prob = 1.0
            elif delta < -700:
                prob = 0.0
            else:
                prob = math.exp(delta)
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # ----- occasional random perturbations -----
        r = rng.random()
        if r < 0.15:  # block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand2 = cur.copy()
            cand2[start:start + width] *= factor
            cand2 = cand2 / cand2.sum()
            cand2_score = _ac1_score_fft(cand2, n, L)
            if cand2_score < best_score:
                best = cand2.copy()
                best_score = cand2_score
                if best_score < 1.50274349:
                    break
            # acceptance
            if cand2_score < cur_score:
                cur = cand2
                cur_score = cand2_score
            else:
                delta = (cur_score - cand2_score) / (temperature + 1e-12)
                if delta > 700:
                    prob = 1.0
                elif delta < -700:
                    prob = 0.0
                else:
                    prob = math.exp(delta)
                if rng.random() < prob:
                    cur = cand2
                    cur_score = cand2_score

        elif r < 0.20:  # mass transfer between two positions
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * 0.4
                delta = cur[i] * frac
                cand3 = cur.copy()
                cand3[i] -= delta
                cand3[j] += delta
                # sum unchanged, no need to renorm
                cand3_score = _ac1_score_fft(cand3, n, L)
                if cand3_score < best_score:
                    best = cand3.copy()
                    best_score = cand3_score
                    if best_score < 1.50274349:
                        break
                if cand3_score < cur_score:
                    cur = cand3
                    cur_score = cand3_score
                else:
                    delta = (cur_score - cand3_score) / (temperature + 1e-12)
                    if delta > 700:
                        prob = 1.0
                    elif delta < -700:
                        prob = 0.0
                    else:
                        prob = math.exp(delta)
                    if rng.random() < prob:
                        cur = cand3
                        cur_score = cand3_score

        # ----- schedule updates -----
        temperature *= temp_decay
        grad_lr = max(min_lr, grad_lr * lr_decay)

        # ----- occasional restart -----
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0
            grad_lr = 0.20

    # ----- return best candidate as plain Python floats -----
    return [float(x) for x in best]
