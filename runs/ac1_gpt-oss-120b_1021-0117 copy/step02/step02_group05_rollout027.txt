def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):

    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a step‑function minimizing the AC1 upper bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)  # reserve 1 s

    # ---- initial vector -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        init = parent.copy()
    except Exception:
        init = _gen_binomial(256)
    n = int(init.size)

    target_sum = float(init.sum())
    L = _next_pow2(2 * n - 1)

    cur = init.copy()
    cur_score = _ac1_score(cur, n, L)

    best = cur.copy()
    best_score = cur_score

    # ---- hyper‑parameters ------------------------------------------------
    temperature = 1.0
    decay = 0.9999
    restart_prob = 0.0015
    min_block = max(2, n // 200)
    max_block = max(2, n // 12)
    transfer_max = 0.5
    sigma_block = 0.13
    eps_candidates = [0.15, 0.07, 0.03, 0.01, 0.005]
    gaussian_sigma = 0.02
    target_score = 1.50274349

    # ---- main loop -------------------------------------------------------
    while time.monotonic() < deadline:
        move_type = rng.random()
        cand = None
        cand_score = None

        # 1) gradient–guided move (≈55 %)
        if move_type < 0.55:
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            kmax = int(np.argmax(conv))

            idx = np.arange(n)
            rev = kmax - idx
            mask = (rev >= 0) & (rev < n)
            g = np.zeros_like(cur)
            g[mask] = 2.0 * cur[rev[mask]]          # exact gradient of conv[kmax]
            mean_g = g.mean()
            d = mean_g - g                          # zero‑sum direction

            # back‑tracking line search (only accept improvement)
            for eps in eps_candidates:
                cand_try = cur + eps * d
                np.clip(cand_try, 0.0, 1000.0, out=cand_try)
                s = cand_try.sum()
                if s > 0:
                    cand_try *= target_sum / s
                cand_try_score = _ac1_score(cand_try, n, L)
                if cand_try_score < cur_score:
                    cand = cand_try
                    cand_score = cand_try_score
                    break

            # if none improved, keep a small uphill step (handled by SA)
            if cand is None:
                eps = eps_candidates[-1]
                cand = cur + eps * d
                np.clip(cand, 0.0, 1000.0, out=cand)
                s = cand.sum()
                if s > 0:
                    cand *= target_sum / s
                cand_score = _ac1_score(cand, n, L)

        # 2) block scaling (≈25 %)
        elif move_type < 0.80:
            width = int(rng.integers(min_block, max_block + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, sigma_block))
            cand = cur.copy()
            cand[start:start + width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)
            s = cand.sum()
            if s > 0:
                cand *= target_sum / s
            cand_score = _ac1_score(cand, n, L)

        # 3) random mass transfer (≈20 %)
        elif move_type < 0.95:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * transfer_max
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            np.clip(cand, 0.0, 1000.0, out=cand)
            s = cand.sum()
            if s > 0:
                cand *= target_sum / s
            cand_score = _ac1_score(cand, n, L)

        # 4) Gaussian mutation (≈5 %)
        else:
            noise = rng.normal(0.0, gaussian_sigma, size=n)
            cand = cur + noise
            np.clip(cand, 0.0, 1000.0, out=cand)
            s = cand.sum()
            if s > 0:
                cand *= target_sum / s
            cand_score = _ac1_score(cand, n, L)

        # ---- update global best -----------------------------------------
        if cand_score is None or not np.isfinite(cand_score):
            continue
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < target_score:
                break

        # ---- simulated‑annealing acceptance -------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= decay
        if rng.random() < restart_prob:
            cur = best.copy()
            cur_score = best_score
            temperature = max(temperature, 1.0)

    # ---- short deterministic polishing (optional) -----------------------
    polish_deadline = time.monotonic() + 0.5
    while time.monotonic() < polish_deadline:
        fa = np.fft.rfft(best, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        rev = kmax - idx
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(best)
        g[mask] = 2.0 * best[rev[mask]]
        mean_g = g.mean()
        d = mean_g - g

        improved = False
        for eps in (0.02, 0.01, 0.005):
            cand = best + eps * d
            np.clip(cand, 0.0, 1000.0, out=cand)
            s = cand.sum()
            if s > 0:
                cand *= target_sum / s
            score = _ac1_score(cand, n, L)
            if score < best_score:
                best = cand
                best_score = score
                improved = True
                break
        if not improved:
            break

    # ---- return as plain Python list ------------------------------------
    return [float(x) for x in best]
