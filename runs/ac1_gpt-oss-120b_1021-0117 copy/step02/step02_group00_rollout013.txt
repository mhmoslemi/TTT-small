import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) onto {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast full convolution via FFT (real-valued)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Objective value matching evaluate_sequence (sum assumed non‑zero)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_fft(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # Initialise: use provided parent if present, otherwise a sqrt‑series.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # analytic sqrt‑difference series (one classic construction)
        def _sqrt_series(m):
            s = np.empty(m, dtype=np.float64)
            s[0] = 1.0
            for k in range(1, m):
                s[k] = np.sqrt(k + 1) - np.sqrt(k)
            return s
        m = 256
        a = _sqrt_series(m)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / total                     # normalise to simplex (sum = 1)

    n = a.shape[0]

    # Best solution seen so far
    best_vec = a.copy()
    best_sc = _score(best_vec)

    # Current iterate
    cur = best_vec.copy()
    cur_sc = best_sc

    # --------------------------------------------------------
    # Parameters for the smoothed gradient descent
    # --------------------------------------------------------
    step0 = 0.5           # initial step size
    beta0 = 5.0           # initial smoothing factor
    beta_max = 80.0       # final smoothing factor
    beta_ramp = 2000      # iterations over which beta is increased
    it = 0
    no_improve = 0
    two_n_minus2 = 2 * n - 2
    idxs = two_n_minus2 - np.arange(n, dtype=np.int64)  # indices for gradient extraction

    while time.time() < deadline:
        it += 1

        # ----------------------------------------------------
        # Full auto‑correlation of the current iterate
        # ----------------------------------------------------
        conv = _conv_fft(cur)
        max_conv = conv.max()

        # ----------------------------------------------------
        # Smoothed weights for the max‑convolution term
        # ----------------------------------------------------
        beta = beta0 + (beta_max - beta0) * min(1.0, it / beta_ramp)
        # using (conv - max) to keep the exponent ≤ 0 and avoid overflow
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # ----------------------------------------------------
        # Gradient of the smoothed max‑convolution (approximate)
        # ----------------------------------------------------
        w_rev = w_norm[::-1]
        grad_full = np.convolve(cur, w_rev)               # length = 3n-2
        grad = 2.0 * grad_full[idxs]                     # central part gives ∂/∂a_j

        # ----------------------------------------------------
        # Projected gradient step (simple diminishing step size)
        # ----------------------------------------------------
        step = step0 / np.sqrt(it)
        cand = cur - step * grad
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # ----------------------------------------------------
        # Accept improvements, otherwise occasional uphill move
        # ----------------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            prob = np.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----------------------------------------------------
        # Random mass‑transfer move when progress stalls
        # ----------------------------------------------------
        if no_improve > 2000 and rng.random() < 0.22:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.35
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0

        # ----------------------------------------------------
        # Periodic jittered restart towards the best known point
        # ----------------------------------------------------
        if it % 4000 == 0 and rng.random() < 0.25:
            jitter = rng.normal(scale=5e-4, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        # ----------------------------------------------------
        # Early exit if target bound reached
        # ----------------------------------------------------
        if best_sc < 1.50274349:
            break

    # Return the best candidate as a plain Python list
    return [float(x) for x in best_vec]
