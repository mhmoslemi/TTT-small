def projection_simplex(v):
    # v is np array of shape (n,)
    # returns projected vector onto simplex {x >=0, sum x = 1}
    # Implementation O(n log n):
    u = np.sort(v)[::-1] # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np, time, math

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _proj_simplex(v):
    # projection onto simplex {x>=0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0)
    return w
```

import time, math
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _gen_binomial(n: int) -> np.ndarray:
    """Classic binomial step a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)   # tiny safety margin

    # ----- initialise -------------------------------------------------
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        if a.ndim != 1 or a.size == 0:
            raise ValueError
        n = a.size
    except Exception:
        n = 256
        a = _gen_binomial(n)

    s = a.sum()
    a = a / s if s > 0 else np.full_like(a, 1.0 / n)

    # ----- pre‑alloc --------------------------------------------------
    L = _next_pow2(3 * n)                # length sufficient for all FFTs

    # ----- evaluate initial best ---------------------------------------
    FA = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(FA * FA, n=L)[: 2 * n - 1]
    best = a.copy()
    best_score = 2.0 * n * conv.max()

    # ----- optimisation parameters ------------------------------------
    beta = 2.0            # start of soft‑max temperature
    beta_max = 80.0
    beta_mul = 1.004
    eta = 0.5            # step size
    eta_mul = 0.9995

    it = 0
    while time.monotonic() < deadline:
        # ----- convolution (current) ----------------------------------
        FA = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(FA * FA, n=L)[: 2 * n - 1]
        maxv = conv.max()
        cur_score = 2.0 * n * maxv
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score < 1.50274349:
                break

        # ----- smooth max weights ------------------------------------
        w = np.exp(beta * (conv - maxv))
        w /= w.sum()

        # ----- gradient of smooth max (correlation) ------------------
        FW = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(FA * np.conj(FW), n=L)[: n + len(w) - 1]
        grad = 2.0 * corr[:n]                     # ∇_a f

        # ----- gradient step + random jitter -------------------------
        a_new = a - eta * grad
        if rng.random() < 0.01:                  # occasional shake
            a_new += rng.normal(scale=0.01, size=n)

        # ----- project back onto simplex -----------------------------
        a = _proj_simplex(a_new)

        # ----- schedule updates --------------------------------------
        beta = min(beta_max, beta * beta_mul)
        eta = max(1e-7, eta * eta_mul)

        it += 1
        # occasional re‑centre on the best found so far
        if it % 3000 == 0:
            a = best + 0.001 * rng.random(n)
            a = _proj_simplex(a)
