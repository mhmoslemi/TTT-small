import numpy as np, time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 upper bound via FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes the AC1 bound.
    Returns a list of Python floats.
    """
    rng = np.random.default_rng(seed)
    # reserve a little time for returning the answer
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # initialise
    try:
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = cur.size
    except Exception:
        n = 256
        cur = _gen_binomial(n)

    # ensure a minimal total mass (evaluator requires sum >= 0.01)
    s = cur.sum()
    if s < 0.01:
        cur *= (0.01 / (s + 1e-12))

    L = _next_pow2(2 * n - 1)

    cur_score = _ac1_score(cur, n, L)
    best = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # hyper‑parameters
    temperature = 1.0
    decay = 0.9995
    target = 1.50274349
    # move probabilities
    prob_grad = 0.5          # gradient‑guided move
    prob_smooth = 0.1        # local smoothing

    # derived
    prob_rand = 1.0 - (prob_grad + prob_smooth + prob_block)

    # smoothing kernel (sum = 1, thus preserves total mass)
    smooth_kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)

    while time.monotonic() < deadline:
        r = rng.random()

        # ------------------------------------------------------------------

        if r < prob_grad:

            fa = np.fft.rfft(cur, n=L)

            kmax = int(np.argmax(conv))

            # exact gradient of conv[kmax] w.r.t. a_i : 2*a_{kmax-i} when index valid
            idx = np.arange(n)

            mask = (rev >= 0) & (rev < n)
            g = np.zeros_like(cur)
            g[mask] = 2.0 * cur[rev[mask]]

            # direction that does not change total mass
            d = g.mean() - g

            step = temperature * 0.2 * max(1.0, np.mean(np.abs(d)))
