def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)  # leave 1 sec headroom

    # get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = parent.size
    except Exception:
        parent = None
        n = 256

    # define helper functions:
    def next_pow2(x):
        return 1 << ((x-1).bit_length())

    L = next_pow2(2*n - 1)
    L2 = next_pow2(3*n)  # for gradient convolution if needed

    def score(arr):
        # compute AC1 bound via FFT
        fa = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        maxb = conv.max()
        s = arr.sum()
        if s < 1e-12:
            return np.inf
        return 2.0 * n * maxb / (s * s)

    # initial sequence
    if parent is not None:
        base = parent.astype(np.float64).copy()
    else:
        # generate decreasing exponential
        i = np.arange(n)
        base = np.exp(-0.001 * i)
    base = np.clip(base, 0.0, 1000.0)
    # ensure sum not-zero
    if base.sum() < 0.01:
        base[0] = 0.01
    # normalize to original sum (keep as is)
    init_sum = base.sum()
    # we might keep sum as is (makes sense)
    cur = base.copy()
    cur_score = score(cur)
    best = cur.copy()
    best_score = cur_score

    # parameters
    temperature = 1.0
    temp_decay = 0.9997
    lr = 0.02
    lr_decay = 0.9999
    # control periodic restart probability
    restart_prob = 0.001

    # main loop
    while time.monotonic() < deadline:
        # compute FFT of cur, conv, etc
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        maxb = float(conv.max())
        k_max = int(conv.argmax())
        cur_score = 2.0 * n * maxb / (cur.sum() * cur.sum())

        move_type = rng.random()
        # candidate placeholder
        cand = None

        if move_type < 0.55:
            # gradient move using top K lags (1 or 2)
            K = 1
            # get top K indices
            top_idxs = np.argpartition(conv, -K)[-K:]
            # compute aggregated gradient
            grad = np.zeros_like(cur)
            for k in top_idxs:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                idx = np.arange(i_start, i_end + 1)
                grad[idx] += 2.0 * cur[k - idx]  # contribution
            # step opposite gradient
            cand = cur - lr * grad
            np.clip(cand, 0.0, 1000.0, out=cand)

        elif move_type < 0.75:
            # random block scaling
            width = int(rng.integers(max(1, n // 250), max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start:start+width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)

        elif move_type < 0.9:
            # mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * 0.3
                delta = cur[i] * frac
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
