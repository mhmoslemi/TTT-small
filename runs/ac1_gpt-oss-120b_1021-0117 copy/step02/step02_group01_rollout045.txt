import time
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # Renormalize to avoid tiny numerical drift
    s = w.sum()
    if s == 0:
        return np.full_like(v, 1.0 / n)
    return w / s


# ----------------------------------------------------------------------
# Analytic sqrt‑series (coefficients a with a*a = 1+…+x^{2n‑2})
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ----------------------------------------------------------------------
# Convolution via FFT: return full convolution, max value, its index,
# and top‑K indices.
# ----------------------------------------------------------------------
def _conv_fft_top_k(a: np.ndarray, L: int, top_k: int = 1):
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(a) - 1]
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])
    if top_k == 1:
        top_idxs = np.array([max_idx], dtype=np.int64)
    else:
        # indices of the K largest values (unsorted)
        top_idxs = np.argpartition(conv, -top_k)[-top_k:]
    return conv, max_val, max_idx, top_idxs


# ----------------------------------------------------------------------
# Gradient for a set of convolution peaks.
# ----------------------------------------------------------------------
def _gradient_peaks(a: np.ndarray, peaks: np.ndarray) -> np.ndarray:
    n = a.size
    g = np.zeros_like(a)
    for p in peaks:
        lo = max(0, p - (n - 1))
        hi = min(n - 1, p)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[p - idx]
    return g
