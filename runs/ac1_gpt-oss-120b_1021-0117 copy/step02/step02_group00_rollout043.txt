def evaluate_sequence(sequence: list) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]

import time
import numpy as np

# ------------------------------------------------------------
# Simple projection onto the probability simplex (unused,
# kept for reference)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Full linear convolution using a power‑of‑two FFT
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return a*a (full linear convolution) using FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


# ------------------------------------------------------------
# Score identical to the supplied evaluator (sum normalised to 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, fft_len: int) -> float:
    conv = _conv_fft(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max() / (a.sum() ** 2))
