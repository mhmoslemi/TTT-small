import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all entries non‑positive
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic optimisation of the AC1 upper bound.
    Starts from a few simple initialisations (including the provided
    ``height_sequence_1`` when present) and runs a momentum‑accelerated
    projected sub‑gradient descent with occasional jitter.
    Returns the best coefficient vector found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ----------------------------------------------------------------------
    # 1. Determine problem size and a basic initial vector
    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = None

    if parent is not None:
        n = int(parent.size)
        a0 = np.clip(parent, 0.0, 1000.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a0 = a0 / s
    else:
        n = 256
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # ----------------------------------------------------------------------
    # 2. Build a modest collection of start points (no huge literals)
    # ----------------------------------------------------------------------
    starts = []

    # a) the (possibly) provided parent
    starts.append(a0)

    # b) uniform distribution
    starts.append(np.full(n, 1.0 / n, dtype=np.float64))

    # c) Dirichlet sample (more spread)
    starts.append(rng.dirichlet(np.ones(n)))

    # d) Random positive vector, normalised
    rnd = rng.random(n)
    rnd = np.clip(rnd, 0.0, None)
    if rnd.sum() > 0.0:
        starts.append(rnd / rnd.sum())

    # e) Simple triangular shape
    x = np.arange(n, dtype=np.float64)
    tri = 1.0 - np.abs((x - (n - 1) / 2.0) / ((n - 1) / 2.0))
    tri = np.clip(tri, 0.0, None)
    tri = tri / tri.sum()
    starts.append(tri)

    # Remove duplicates (within tiny tolerance)
    uniq = []
    for s in starts:
        s = s.astype(np.float64)
        s = np.clip(s, 0.0, 1000.0)
        total = s.sum()
        s = s / total if total >= 0.01 else np.full(n, 1.0 / n)
        if not any(np.allclose(s, u) for u in uniq):
            uniq.append(s)
    starts = uniq

    # ----------------------------------------------------------------------
    # 3. Pre‑compute FFT length for the given n (common to all runs)
    # ----------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ----------------------------------------------------------------------
    # 4. Optimisation loop over each start
    # ----------------------------------------------------------------------
    best_vec = None
    best_score = np.inf
    target_score = 1.50274349

    for start in starts:
        a = start.copy()
        m = np.zeros_like(a)          # momentum term
        it = 0
        # simple schedule parameters
        step0 = 0.8
        beta = 0.9                     # momentum coefficient

        while time.monotonic() < deadline:
            it += 1

            # ----- convolution via FFT (linear, not circular) -----
            fa = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

            # ----- current bound (sum of a is 1, so 2*n*max_conv) -----
            kmax = int(np.argmax(conv))
            max_conv = conv[kmax]
            cur_score = 2.0 * n * max_conv

            if cur_score < best_score:
                best_vec = a.copy()
                best_score = cur_score
                if best_score <= target_score:
                    return best_vec.tolist()

            # ----- sub‑gradient for the active shift -----
            L = max(0, kmax - (n - 1))
            U = min(n - 1, kmax)
            idx = np.arange(L, U + 1, dtype=np.int64)
            partner = kmax - idx
            g = np.zeros_like(a)
            g[idx] = 2.0 * a[partner]

            # ----- momentum update -----
            m = beta * m + (1.0 - beta) * g

            # ----- step size (slow decay) -----
            eta = step0 / (1.0 + it / 500.0)
            a_new = a - eta * m

            # ----- project back onto the simplex -----
            a = _proj_simplex(a_new)

            # ----- occasional multiplicative jitter to avoid stagnation -----
            if it % 100 == 0:
                jitter = np.exp(rng.normal(0.0, 0.005, size=n))
                a = _proj_simplex(a * jitter)

            # early termination of inner loop if time is almost exhausted
            if time.monotonic() + 0.1 >= deadline:
                break

    # ----------------------------------------------------------------------
    # 5. Fallback: if nothing was found (should not happen) return uniform
    # ----------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return best_vec.tolist()
