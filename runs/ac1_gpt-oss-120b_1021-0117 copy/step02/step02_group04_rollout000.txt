import math

# ---------- helper utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """Generate classic binomial step a_i = C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _score_and_conv(arr: np.ndarray, L: int):
    """Return (score, full convolution) for a non‑negative array."""
    n = arr.shape[0]
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf"), conv
    score = 2.0 * n * max_b / (s * s)
    return score, conv


def _normalize_clip(arr: np.ndarray, target_sum: float):
    """Clip to [0,1000] and renormalise to target_sum (if possible)."""
    np.clip(arr, 0.0, 1000.0, out=arr)
    cur_sum = arr.sum()
    if cur_sum <= 0:
        return arr
    arr *= (target_sum / cur_sum)
    np.clip(arr, 0.0, 1000.0, out=arr)
    return arr


def _move_targeted(cur: np.ndarray, conv_cur: np.ndarray, rng: np.random.Generator):
    """Transfer mass away from the pair that contributes most to max lag."""
    n = cur.shape[0]
    k_max = int(conv_cur.argmax())
    idx = np.arange(n)
    j = k_max - idx
    valid = (j >= 0) & (j < n)
    contrib = np.zeros_like(cur)
    contrib[valid] = cur[valid] * cur[j[valid]]
    i_max = int(contrib.argmax())
    j_max = k_max - i_max
    # fraction of each element to move (up to 30%)
    frac = rng.random() * 0.30
    delta_i = cur[i_max] * frac
    delta_j = cur[j_max] * frac
    total = delta_i + delta_j
    cand = cur.copy()
    cand[i_max] -= delta_i
    cand[j_max] -= delta_j
    # choose a destination distinct from i_max and j_max
    dst = rng.integers(0, n)
    while dst == i_max or dst == j_max:
        dst = rng.integers(0, n)
    cand[dst] += total
    return cand


def _move_block_scale(cur: np.ndarray, target_sum: float, rng: np.random.Generator):
    """Scale a random contiguous block and renormalise."""
    n = cur.shape[0]
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = math.exp(rng.normal(0.0, 0.12))
    cand = cur.copy()
    cand[start:start + width] *= factor
    _normalize_clip(cand, target_sum)
    return cand


def _move_transfer(cur: np.ndarray, target_sum: float, rng: np.random.Generator):
    """Move a random fraction of mass from one index to another."""
    n = cur.shape[0]
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    while j == i:
        j = int(rng.integers(0, n))
    frac = rng.random() * 0.40
    delta = cur[i] * frac
    cand = cur.copy()
    cand[i] -= delta
    cand[j] += delta
    _normalize_clip(cand, target_sum)
    return cand


def _move_smooth(cur: np.ndarray, target_sum: float, rng: np.random.Generator):
    """Blend with a small smoothing kernel."""
    n = cur.shape[0]
    alpha = rng.random() * 0.45  # up to 45% smoothing
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    sm = np.convolve(cur, kernel, mode="same")
    cand = (1.0 - alpha) * cur + alpha * sm
    _normalize_clip(cand, target_sum)
    return cand
