def _ac_info(arr, L):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*len(arr)-1]
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])
    s = float(arr.sum())
    if s < 1e-12: 
        return float('inf'), conv, max_idx, max_val
    score = 2.0 * len(arr) * max_val / (s * s)
    return score, conv, max_idx, max_val
```

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = parent.size
        candidates = [parent]
    except Exception:
        n = 256
        candidates = []
    # generate additional candidates
    # 1. binomial
    try:
        binom = _gen_binomial(n)
        candidates.append(binom)
    except Exception:
        pass
    # 2. powerlaw p=0.5
    pow_half = np.arange(1, n+1, dtype=np.float64) ** -0.5
    candidates.append(pow_half)
    # 3. powerlaw p=0.75
    pow75 = np.arange(1, n+1, dtype=np.float64) ** -0.75
    candidates.append(pow75)
    # 4. exponential r=0.99
    r = 0.99
    exp99 = np.power(r, np.arange(n, dtype=np.float64))
    candidates.append(exp99)
    # 5. exponential r = 1 - 1.0/n  (approx)
    r2 = 1.0 - 1.0/n
    exp_r2 = np.power(r2, np.arange(n, dtype=np.float64))
    candidates.append(exp_r2)

    L = _next_pow2(2*n - 1)
    best = None
    best_score = float('inf')
    for cand in candidates:
        # clip ensure positive
        cand = np.clip(cand, 0.0, 1000.0)
        score = _ac1_score(cand, n, L)
        if score < best_score:
            best = cand.copy()
            best_score = score

    cur = best.copy()
    cur_score = best_score

    # Hyperparameters
    temperature = 1.0
    decay = 0.9999  # slower decay
    min_block = max(2, n // 200)
    max_block = max(2, n // 12)
    step_factor0 = 0.2
    step_decay = 0.995
    step_factor = step_factor0

    target = 1.50274349

    while time.monotonic() < deadline:
        move = rng.random()
        if move < 0.55:
            # gradient move
            # compute conv and gradient
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2*n - 1]
            kmax = int(conv.argmax())
            # gradient g
            idx = np.arange(n)
            rev = kmax - idx
            mask = (rev >= 0) & (rev < n)
            g = np.zeros_like(cur)
            g[mask] = cur[rev[mask]]
            # double for distinct partners
            self_mask = mask & (rev != idx)
            g[self_mask] *= 2.0
            mean_g = g.mean()
            d = mean_g - g  # zero-sum direction
            # compute max allowable step
            negative = d < -1e-12
            if np.any(negative):
                max_alpha = np.min(cur[negative] / (-d[negative]))
                # step size scaled
                alpha = min(step_factor * max_alpha, max_alpha)
                cand = cur + alpha * d
            else:
                # no negative direction => skip
                cand = cur.copy()
        elif move < 0.85:
            width = int(rng.integers(min_block, max_block + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start:start+width] *= factor
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * 0.5
                delta = cur[i] * frac
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                continue

        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = _ac1_score(cand, n, L)

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < target:
                break

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= decay
        step_factor *= step_decay
        # occasional restart
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0
            step_factor = step_factor0

    return [float(x) for x in best]

def _gen_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr, n, L):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound via FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def _evaluate(arr: np.ndarray, n: int, L: int):
    """Return (score, conv, kmax) for a given array."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    kmax = int(conv.argmax())
    return 2.0 * n * float(conv.max()) / (float(arr.sum()) ** 2), conv, kmax
