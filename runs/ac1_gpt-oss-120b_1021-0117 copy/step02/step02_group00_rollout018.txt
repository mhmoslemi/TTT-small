import time, numpy as np

def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
    return conv

def _conv_fft_general(x, y):
    m = x.shape[0] + y.shape[0] - 1
    L = 1 << (m-1).bit_length()
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx*fy, n=L)[:m]
    return conv

def _project_simplex(v):
    # Project onto simplex sum=1, v>=0
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0)

import time

# ------------------------------------------------------------
# Simplex projection (sum to 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# FFT based full linear convolution (real inputs)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a using FFT (real inputs)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


def _conv_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution x * y for arbitrary lengths using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    L = 1 << (m - 1).bit_length()
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[:m]
    return conv
