import numpy as np

def _score_conv(seq: np.ndarray, n: int, fft_len: int):
    """Return (score, full linear convolution, max value)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    max_b = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_b
    return 2.0 * n * max_b / (s * s), conv, max_b


def _gradient(seq: np.ndarray, conv: np.ndarray, beta: float,
              n: int, fft_len_g: int):
    """
    Gradient of a soft‑max approximation of the maximal convolution entry.
    Returns a vector of length n.
    """
    max_b = np.max(conv)
    w = np.exp(beta * (conv - max_b))               # weight ≤1, max at 1
    a_rev = seq[::-1]
    a_fft = np.fft.rfft(a_rev, n=fft_len_g)
    w_fft = np.fft.rfft(w, n=fft_len_g)

    grad = 2.0 * full[n - 1: n - 1 + n]            # extract shifted part
    return grad
