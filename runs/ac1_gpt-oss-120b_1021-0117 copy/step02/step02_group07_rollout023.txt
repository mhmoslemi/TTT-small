    def score_of(seq):
        s = seq.sum()
        if s < 0.01:
            return math.inf, None, None
        f = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
        maxb = conv.max()
        score = 2.0 * n * maxb / (s * s)
        return float(score), conv, maxb

import numpy as np

def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1, decreasing and positive."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a
