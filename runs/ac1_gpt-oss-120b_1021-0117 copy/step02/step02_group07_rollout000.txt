    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5) # margin

    # Determine n
    try: arr = np.asarray(height_sequence_1, dtype=np.float64); parent = arr
    except: parent = None
    if parent is not None and parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
        a0 = np.clip(parent, 0, 1000)
    else:
        n = 3809 if parent is None else parent.shape[0]
        if parent is None:
            n = 3809 # fallback length from previous code? maybe similar
        # binomial
        a0 = np.empty(n, dtype=np.float64)
        a0[0] = 1.0
        for i in range(1, n):
            a0[i] = a0[i-1] * (2*i-1) / (2*i)
    n = a0.size
    fft_len = 1 << ((2*n - 1).bit_length())
    # Helper functions
    def conv_of(seq):
        f = np.fft.rfft(seq, n=fft_len)
        return np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    def score_of(seq):
        conv = conv_of(seq)
        maxb = conv.max()
        s = seq.sum()
        return 2.0 * n * maxb / (s * s) if s > 0 else float('inf')
    # Initialize best
    best = a0.copy()
    best_score = score_of(best)
    # If parent exists but maybe not same as a0
    if parent is not None and parent.ndim == 1 and parent.size == n:
        parent_score = score_of(parent)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score
    current = best.copy()
    current_score = best_score

    # Main loop
    step_factors = [0.0, 0.001, 0.003, 0.006, 0.012, 0.024, 0.048]
    eps = 1e-12

    while time.monotonic() < deadline:
        conv = conv_of(current)
        k = int(conv.argmax())
        # gradient
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi+1, dtype=np.int64)

        grad = np.empty_like(current)

def _initial_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, i=0..n-1 (decreasing positive)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0

        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
