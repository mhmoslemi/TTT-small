    f = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[:2*a.size - 1]

import numpy as np
import time

def _initial_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i for i=0..n-1 (positive, slowly decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _conv_fft(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution of seq with itself using an FFT of length fft_len."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    return conv

def _score(seq: np.ndarray, fft_len: int):
    """Return (score, full convolution, max value)."""
    conv = _conv_fft(seq, fft_len)
    maxb = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float("inf"), conv, maxb
    score = 2.0 * seq.size * maxb / (s * s)
    return float(score), conv, maxb

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a low AC1 upper‑bound step function."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 2.0)   # safety margin

    # Determine length and optional parent sequence
    # ------------------------------------------------------------------
    n = 3809

        arr = np.asarray(height_sequence_1, dtype=np.float64)

            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
    except Exception:
        parent = None

    # ------------------------------------------------------------------
    # Initialise best candidate (parent or binomial seed)

    best = _initial_binomial(n)
    fft_len = 1 << ((2 * n - 1).bit_length())
    best_score, best_conv, _ = _score(best, fft_len)

    if parent is not None:

    # ------------------------------------------------------------------

    fft_len_grad = 1 << ((3 * n - 2).bit_length())

        elif op == 1:        # single entry scaling
