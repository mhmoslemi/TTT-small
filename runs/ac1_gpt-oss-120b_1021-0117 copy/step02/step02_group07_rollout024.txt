        cand = None
        cand_max = None
        for _ in range(6):
            cand_tmp = a + step * direction
            np.clip(cand_tmp, 0.0, 1000.0, out=cand_tmp)
            s = cand_tmp.sum()
            if s > 0:
                cand_tmp *= (target_sum / s)   # renormalise the total mass
            np.clip(cand_tmp, 0.0, 1000.0, out=cand_tmp)
            cand_max, _ = _conv_and_max(cand_tmp, fft_len)
            if cand_max < max_b - 1e-12:
                cand = cand_tmp
                break
            step *= 0.5

        # ----------------------------------------------------------------
        # If the gradient move failed, fall back to a tiny random tweak.
        # ----------------------------------------------------------------
        if cand is None:
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.08))
            cand = a.copy()
            cand[idx] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)
            s = cand.sum()
            if s > 0:
                cand *= (target_sum / s)
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_max, _ = _conv_and_max(cand, fft_len)

        cand_score = 2.0 * n * cand_max / (target_sum * target_sum)

        # --------------------------------------------------------------
        # Acceptance test – greedy with a little simulated‑annealing.
        # --------------------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            elapsed = time.monotonic() - start
            total = deadline - start
            temp = temperature * max(0.0, 1.0 - elapsed / total)
            if temp > 0.0 and rng.random() < math.exp((cur_score - cand_score) / (temp + 1e-12)):
                accept = True
        if accept:
            a = cand

        # slowly shrink the step scale to fine‑tune at the end
        step_scale *= step_decay

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python floats).
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
