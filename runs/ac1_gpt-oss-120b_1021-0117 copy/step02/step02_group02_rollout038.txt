def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    # initial
    try: parent = np.asarray(height_sequence_1, dtype=float)
    except: parent = None
    if parent is not None and parent.ndim ==1 and parent.size>0 and np.isfinite(parent).all():
        n = parent.size
        a = np.clip(parent, 0, 1000)
    else:
        n = 256
        a = np.ones(n, dtype=float)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 0.01:
        a = np.full(n, 1.0/n, dtype=float)
    else:
        a = a / total

    # best
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # uniform candidate
    uniform = np.full(n, 1.0/n, dtype=float)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    # parameters
    target_score = 1.50274349
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    momentum = np.zeros_like(a)
    mu = 0.9
    eta0 = 0.5
    jitter_sigma = 0.01
    jitter_period = 50
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # convolution via FFT
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        # score
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target_score:
                break

        # active shift (max index)
        kmax = int(np.argmax(conv))
        # compute subgradient
        g = np.zeros_like(a)
        offset = kmax - (n - 1)
        if offset >= 0:
            idx = np.arange(offset, n)
            partner = kmax - idx
            g[idx] = 2.0 * a[partner]
        else:
            idx = np.arange(0, n + offset)  # offset negative
            partner = kmax - idx
            g[idx] = 2.0 * a[partner]

        # update with momentum & step size
        eta = eta0 / np.sqrt(it)
        momentum = mu * momentum - eta * g
        a_new = a + momentum
        # projection onto simplex
        a = _proj_simplex(a_new)

        # jitter
        if it % jitter_period == 0:
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = _proj_simplex(a * jitter)
            # optionally decay jitter_sigma
            jitter_sigma = max(jitter_sigma * 0.99, 0.001)

    return [float(v) for v in best]

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    it = 0
    eta0 = 0.5                # base step size
    mu = 0.9                  # momentum coefficient
    momentum = np.zeros_like(a)
    K = 5                     # number of top convolution entries to use
    jitter_period = 40
    jitter_sigma = 0.02
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 4. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- convolution via FFT (real) ---------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        # ---- current objective ------------------------------------------
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv          # sum(a)=1 -> exact objective
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target_score:
                break

        # ---- multi‑peak sub‑gradient ------------------------------------
        top_k = min(K, conv_len)
        if top_k == conv_len:
            top_idx = np.arange(conv_len)
        else:
            # unsorted indices of the K largest values
            top_idx = np.argpartition(-conv, top_k - 1)[:top_k]

        g = np.zeros_like(a)
        for k in top_idx:
            offset = int(k) - (n - 1)
            if offset >= 0:
                idx = np.arange(offset, n, dtype=np.int64)
                partner = k - idx
                g[idx] += 2.0 * a[partner]
            else:
                idx = np.arange(0, n + offset, dtype=np.int64)   # offset negative
                partner = k - idx
                g[idx] += 2.0 * a[partner]

        # ---- momentum update & projection --------------------------------
        eta = eta0 / np.sqrt(it)
        momentum = mu * momentum - eta * g
        a_candidate = a + momentum
        a = _proj_simplex(a_candidate)

        # ---- occasional jitter -------------------------------------------
        if it % jitter_period == 0:
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = _proj_simplex(a * jitter)
            jitter_sigma = max(jitter_sigma * 0.98, 0.001)  # slowly decay

    # ------------------------------------------------------------------
    # 5. Return best sequence (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
