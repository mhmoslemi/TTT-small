import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection requires a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic mirror‑descent optimisation for the AC1 upper bound.
    Starts from ``height_sequence_1`` when available, otherwise from a uniform
    distribution.  The iterate stays on the simplex (non‑negative, sum = 1)
    using multiplicative updates driven by the sub‑gradient of the active
    autocorrelation peak.  Occasional log‑normal jitter avoids stagnation.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
