def _initial_binomial(n: int) -> np.ndarray:
    """Generate a decreasing positive sequence a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_fft(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of seq with itself via FFT (real‑valued)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    return conv


def _score(seq: np.ndarray, n: int, fft_len: int) -> float:
    """Exact AC1 upper bound for a non‑negative sequence."""
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf")
    conv = _conv_fft(seq, fft_len)
    mx = np.max(conv)
    return 2.0 * n * mx / (s * s)


def _grad_max(seq: np.ndarray, kmax: int) -> np.ndarray:
    """Sub‑gradient of the maximal convolution entry."""
    n = seq.size
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    grad = np.zeros_like(seq)
    grad[idx] = seq[kmax - idx]
    return grad

            rand_seq *= (target_sum / np.sum(rand_seq))
            rand_score = _score(rand_seq, n, fft_len)
            if rand_score < best_score:
                best = rand_seq
                best_score = rand_score

    # Return a plain Python list so the evaluator can consume it
    return [float(x) for x in best]
```
