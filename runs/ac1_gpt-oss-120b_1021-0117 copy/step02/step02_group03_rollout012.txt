# ------------------------------------------------------------
# Simplex projection (sum = 1, x_i >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Max of full autocorrelation via FFT (linear convolution)
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> float:
    """Return max_{k} (a * a)[k] using FFT of length L."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Sub‑gradient of the active lag of the autocorrelation
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    # correct double count when the pair is (i,i)
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad


# ------------------------------------------------------------
# Analytic sqrt‑series (fallback initializer)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Score = 2 * n * max_autocorr (sum is normalised to 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int, n: int) -> float:
    return 2.0 * n * _max_convolution(a, L)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # time control – keep a small safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # --------------------------------------------------------
    # initialise – use the provided parent if available
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = _sqrt_series(256).astype(np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / a.sum()
    a = _project_simplex(a)          # enforce exact simplex

    n = a.shape[0]
    Lfft = 1 << ((2 * n - 1).bit_length())  # next power of two

    # --------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------
    best = a.copy()
    best_score = _score(a, Lfft, n)

    # hyper‑parameters
    eta0 = 0.85          # initial step magnitude
    iteration = 0

    # --------------------------------------------------------
    # optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # current autocorrelation and its active position
        fv = np.fft.rfft(a, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        M = float(conv.max())
        k = int(conv.argmax())
        cur_score = 2.0 * n * M

        # sub‑gradient at the active lag
        grad = _grad_for_k(a, k)
        gnorm = np.linalg.norm(grad)
        if gnorm > 0:
            direction = -grad / gnorm
        else:
            direction = np.zeros_like(a)

        # diminishing step size
        step = eta0 / np.sqrt(iteration)

        # ----------------------------------------------------
        # line‑search over a few step multipliers
        # ----------------------------------------------------
        best_local = a
        best_local_score = cur_score
        for factor in (0.5, 1.0, 1.5, 2.0):
            cand = a + step * factor * direction
            cand = _project_simplex(cand)
            sc = _score(cand, Lfft, n)
            if sc < best_local_score:
                best_local = cand
                best_local_score = sc

        # ----------------------------------------------------
        # occasional random jitter (to escape plateaus)
        # ----------------------------------------------------
        if rng.random() < 0.01:
            jitter = np.zeros_like(a)
            idx = rng.choice(n, size=3, replace=False)
            jitter[idx] = rng.normal(scale=0.02, size=3)
            cand = a + jitter
            cand = _project_simplex(cand)
            sc = _score(cand, Lfft, n)
            if sc < best_local_score:
                best_local = cand
                best_local_score = sc

        # ----------------------------------------------------
        # occasional mass‑transfer move
        # ----------------------------------------------------
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                j = rng.integers(0, n)
                if j == i:
                    j = (j + 1) % n
                frac = rng.random() * 0.15
                delta = a[i] * frac
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)
                sc = _score(cand, Lfft, n)
                if sc < best_local_score:
                    best_local = cand
                    best_local_score = sc

        # ----------------------------------------------------
        # accept the best local move (with a tiny chance to accept worsening)
        # ----------------------------------------------------
        if best_local_score < cur_score or rng.random() < 0.01:
            a = best_local

        # keep the global best seen so far
        if best_local_score < best_score:
            best_score = best_local_score
            best = best_local.copy()

        # early exit if we already beat the target
        if best_score < 1.50275:
            break

    # --------------------------------------------------------
    # return the best sequence as a plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best]
