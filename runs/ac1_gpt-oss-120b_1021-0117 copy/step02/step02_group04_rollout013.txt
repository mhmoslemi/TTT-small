def proj_simplex(v, s=1):
    # v is numpy array
    u = np.sort(v)[::-1]

import time
import math
import numpy as np

# ---------- utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast computation of the AC1 upper bound (identical to the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _proj_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Projection of v onto the simplex {x >= 0, sum(x) = s}."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_at_lag(f: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution value at lag `k` (full convolution index).
    For a non‑negative f the derivative of `conv[k] = Σ_i f_i f_{k-i}` equals
    `2 * f_{k-i}` whenever the partner index is in range.
    """
    n = f.size
    g = np.zeros_like(f)
    if k < n:
        # overlaps the left part of the signal
        # i = 0 .. k  =>  j = k-i  (runs k .. 0)
        g[: k + 1] = 2.0 * f[k :: -1]
    else:
        # overlaps the right part
        start = k - (n - 1)                     # smallest i with a partner
        idx = np.arange(start, n)               # i values
        g[start:] = 2.0 * f[k - idx]            # partner values
    return g

    """

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    L = _next_pow2(2 * n - 1)
