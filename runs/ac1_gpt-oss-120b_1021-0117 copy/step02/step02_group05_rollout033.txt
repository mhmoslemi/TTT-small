    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Determine n and initial sequence

        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:

        n = base.shape[0]
        cur = base.copy()
    except Exception:

        cur = np.ones(n)  # maybe start uniform? We'll use binomial, but not needed.

    # ensure nonnegative etc.
    cur = np.clip(cur, 0.0, 1000.0)

    # compute function for score
    L = 1 << ((2*n - 1).bit_length())
    def ac1_score(arr):
        fa = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        maxb = conv.max()
        s = arr.sum()
        return 2.0 * n * maxb / (s*s) if s >= 0.01 else float('inf')
    
    cur_score = ac1_score(cur)
    best = cur.copy()
    best_score = cur_score

    # parameters
    temperature = 1.0
    decay = 0.9995
    min_blk = max(2, n // 250)  # adjust
    max_blk = max(2, n // 6)
    # step size for gradient
    eps_base = 0.02
    # target (optional)
    target = 1.50274349

    # main loop
    while time.monotonic() < deadline:
        move_type = rng.random()
        cand = None
        if move_type < 0.4:
            # gradient guided move with random epsilon factor
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
            kmax = int(np.argmax(conv))
            # compute gradient g as earlier
            idx = np.arange(n)
            partner = kmax - idx
            mask = (partner >= 0) & (partner < n)
            g = np.zeros(n, dtype=np.float64)
            g[mask] = cur[partner[mask]]
            # those with partner != idx (i.e., i != kmax-i), add second contribution
            self_mask = mask & (partner != idx)
            g[self_mask] *= 2.0
            # direction to keep sum unchanged
            d = g.mean() - g
            # scale step size randomly
            eps = eps_base * (0.5 + rng.random())
            cand = cur + eps * d
        elif move_type < 0.65:
            # block scaling
            width = int(rng.integers(min_blk, max_blk+1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start:start+width] *= factor
        elif move_type < 0.8:
            # random mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * 0.5
                delta = cur[i] * frac
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
        elif move_type < 0.9:
            # swap two entries
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                cand = cur.copy()
                cand[i], cand[j] = cand[j], cand[i]
        else:

            start = int(rng.integers(0, n - width + 1))
            cand = cur.copy()
            cand[start:start+width] = cand[start:start+width][::-1]

        # if cand is None (skip), continue
        if cand is None:

        # clip to allowed range

        # evaluate

        if cand_score < best_score:

            best_score = cand_score

import math
import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic optimisation of a non‑negative step function for the AC1 bound.
    Returns a list of floats.
    """
    rng = np.random.default_rng(seed)

    # keep a short safety margin before the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # Load parent if present, otherwise fall back to a simple seed.
    parent = globals().get("height_sequence_1", None)
    cur = None
    n = None
    if parent is not None:
        try:
            arr = np.asarray(parent, dtype=np.float64)
            if arr.ndim != 1 or arr.size == 0:
                raise ValueError
            cur = arr.copy()
            n = arr.size
        except Exception:
            cur = None

    if cur is None:
        # no usable parent – generate a modest initial shape
        n = 256
        cur = _gen_binomial(n)

    # --------------------------------------------------------------
    # Convolution parameters
    L = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Fast AC1 score using FFT
    def _ac1_score(arr: np.ndarray) -> float:
        fa = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_b = float(conv.max())
        s = float(arr.sum())
        if s < 1e-12:
            return float('inf')
        return 2.0 * n * max_b / (s * s)

    cur_score = _ac1_score(cur)
    best = cur.copy()
    best_score = cur_score

    # maybe a fresh binomial is better than the parent
    if parent is None:
        alt = _gen_binomial(n)
        alt_score = _ac1_score(alt)
        if alt_score < best_score:
            best = alt.copy()
            best_score = alt_score
            cur = alt.copy()
            cur_score = alt_score

    # --------------------------------------------------------------
    # Hyper‑parameters
    temperature = 1.0
    decay = 0.9996          # a bit slower cooling
    min_block = max(2, n // 250)
    max_block = max(2, n // 6)
    eps_base = 0.02
    target = 1.50274349

    # --------------------------------------------------------------
    # Main search loop
    while time.monotonic() < deadline:
        move = rng.random()
        cand = None

        # 1) Gradient driven mass redistribution (≈40 %)
        if move < 0.40:
            # compute convolution once
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            kmax = int(np.argmax(conv))

            idx = np.arange(n)
            partner = kmax - idx
            mask = (partner >= 0) & (partner < n)
            g = np.zeros_like(cur)
            g[mask] = cur[partner[mask]]
            self_mask = mask & (partner != idx)
            g[self_mask] *= 2.0

            # direction which keeps total mass unchanged
            d = g.mean() - g

            # step size shrinks with temperature
            eps = eps_base * (0.5 + rng.random()) * temperature
            cand = cur + eps * d

        # 2) Block scaling (≈20 %)
        elif move < 0.60:
            width = int(rng.integers(min_block, max_block + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start:start + width] *= factor

        # 3) Random mass transfer (≈15 %)
        elif move < 0.75:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * 0.45
                delta = cur[i] * frac
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta

        # 4) Swap a pair of entries (≈10 %)
        elif move < 0.85:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                cand = cur.copy()
                cand[i], cand[j] = cand[j], cand[i]

        # 5) Reverse a sub‑block (≈10 %)
        else:
            width = int(rng.integers(min_block, max_block + 1))
            start = int(rng.integers(0, n - width + 1))
            cand = cur.copy()
            cand[start:start + width] = cand[start:start + width][::-1]

        if cand is None:
            continue

        # Respect evaluator’s bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Evaluate candidate
        cand_score = _ac1_score(cand)

        # Keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < target:
                break

        # Simulated‑annealing acceptance for the current state
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # Temperature decay
        temperature *= decay

        # Occasional restart from the best seen
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # --------------------------------------------------------------
    # Return the best found sequence as plain Python floats
    return [float(x) for x in best]
