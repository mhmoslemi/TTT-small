def propose_candidate(seed=42,budget_s=1100,**kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0,budget_s - 5) # small safety

    # init a
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # create uniform?
        n = 256
        a = np.ones(n, dtype=np.float64) / n

    n = a.shape[0]
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1/n)
    else:
        a = a / a_sum
    a = _project_simplex(a)

    Lfft = 1 << ((2*n - 1).bit_length())

    def conv_max(arr):
        fv = np.fft.rfft(arr, n=Lfft)
        conv = np.fft.irfft(fv*fv, n=Lfft)[:2*n-1]
        return float(conv.max()), int(conv.argmax())  # also index

    best_a = a.copy()
    best_val, _ = conv_max(best_a)
    # Multiply by 2n for score
    best_score = 2*n*best_val

    iter_count = 0
    no_improve = 0
    while time.monotonic() < deadline:
        iter_count += 1

        # compute current conv max and argmax
        cur_max, k = conv_max(a)
        cur_score = 2*n*cur_max

        # compute gradient for lag k
        grad = _grad_for_k(a, k)  # 2 * contributions etc.

        # direction: negative gradient
        d = -grad

        # line search candidate step sizes
        # We'll have list of step fractions descending
        step_candidates = [0.8, 0.4, 0.2, 0.1, 0.05, 0.01]
        best_local_a = a
        best_local_val = cur_max
        chose = False

        for step_frac in step_candidates:
            # magnitude scaling based on norm to keep step size moderate
            # compute step = step_frac * d / (||d||+eps)
            norm = np.linalg.norm(d)
            if norm == 0:
                cand = a
            else:
                step = step_frac * d / norm
                cand_raw = a + step
                cand = _project_simplex(cand_raw)
            cand_max, _ = conv_max(cand)
            if cand_max < best_local_val:
                best_local_val = cand_max
                best_local_a = cand
                chose = True
        if chose:
            a = best_local_a
            cur_max = best_local_val
            cur_score = 2*n*cur_max

        # keep best overall
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # occasional random mass transfer to escape
        if rng.random() < 0.02 or no_improve > 2000:
            # mass transfer
            i = rng.integers(n)
            if best_a[i] > 0:
                # pick j != i
                j = rng.integers(n-1)
                if j >= i:
                    j += 1
                frac = rng.random()*0.2  # up to 20% move
                delta = best_a[i] * frac
                new = best_a.copy()
                new[i] -= delta
                new[j] += delta
                new = _project_simplex(new)
                # Evaluate
                new_max,_ = conv_max(new)
                new_score = 2*n*new_max
                if new_score < best_score:
                    best_score = new_score
                    best_a = new.copy()
                a = new.copy()
                no_improve = 0

        # occasional reset to initial
        if rng.random() < 0.01:
            # reinitialize with random Dirichlet
            alpha = np.ones(n) * 0.5
            new = rng.dirichlet(alpha)
            # optionally mix with best
            blend = rng.random()
            new = blend*best_a + (1-blend)*new
            new = _project_simplex(new)
            a = new
            # Could evaluate at the end.

        # early exit if we achieve target bound < 1.5028
        if best_score < 1.5028:
            break

    return [float(x) for x in best_a]

import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Gradient of the convolution at a specific lag k
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of (a*a)[k] with respect to a."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    # central term counted twice when k is even and the center index exists
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad

# ------------------------------------------------------------
# Maximum value of the full convolution a*a (computed via FFT)
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> tuple[float, int]:
    """Return (max value, argmax index) of the linear convolution a*a."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    max_idx = int(np.argmax(conv))
    return float(conv[max_idx]), max_idx

# ------------------------------------------------------------

    if s == 0.0:
