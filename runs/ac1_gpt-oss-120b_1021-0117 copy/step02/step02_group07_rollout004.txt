def conv_score(seq, fft_len, n):
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a <= 0:
        return np.inf, conv
    score = 2.0 * n * max_b / (sum_a * sum_a)
    return score, conv
```

        max_b = np.max(c)
        s = np.sum(seq)

        cand_score, cand_conv = eval_seq(cand)

                best_score = cand_score

        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
