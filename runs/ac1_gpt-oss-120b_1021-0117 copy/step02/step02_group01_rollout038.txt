import time
import numpy as np

# ----------------------------------------------------------------------
# Project a vector onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Compute the maximal convolution entry (and its index) via FFT
# ----------------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    # leave a couple of seconds for clean‑up before the sandbox kills us
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # length of the problem (use the provided parent if present)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = int(len(height_sequence_1))
    else:
        # fallback length – still a few thousand elements to be realistic
        n = 2048

    # pre‑compute FFT length (next power of two >= 2·n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Search over simple geometric families a_i ∝ r^i (r ∈ (0,1])
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # geometric decay parameters (α = 1‑r).  We explore a log‑spaced range.
    alphas = np.logspace(np.log10(2e-4), np.log10(5e-2), num=80)

    # the index array is reused
    i_arr = np.arange(n, dtype=np.float64)

    for alpha in alphas:
        # keep an eye on the deadline
        if time.monotonic() + 0.001 > deadline:
            break
        r = 1.0 - alpha
        if r <= 0.0:
            continue
        # raw geometric terms, normalised to sum = 1
        raw = np.power(r, i_arr)
        total = raw.sum()
        a = raw / total                     # a.sum() == 1.0

        # evaluate the AC1 bound
        fv = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        max_b = float(conv.max())
        score = 2.0 * n * max_b               # denominator = 1 after normalisation

        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # ------------------------------------------------------------------
    # If the geometric sweep failed (should not happen), fall back to uniform
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * (np.convolve(best_seq, best_seq).max())

    # ------------------------------------------------------------------
    # Small projected‑gradient refinement to squeeze a little more out
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    step = 0.3
    min_step = 1e-6
    decay = 0.999
    prob_random = 0.02

    while time.monotonic() + 0.1 < deadline:
        # current convolution and its worst index
        max_val, max_idx = _conv_max_and_index(a, L)
        cur_score = 2.0 * n * max_val

        # gradient of the active convolution entry
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        g = np.zeros_like(a)
        if lo <= hi:

            g[idxs] = 2.0 * a[max_idx - idxs]

        # candidate step
        cand = a - step * g
        cand = _project_simplex(cand)

        cand_score = 2.0 * n * max_val_c

        # accept better candidates

            cur_score = cand_score
            if cand_score < best_score:

                best_seq = cand.copy()
        else:

            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-12)):
                a = cand
                cur_score = cand_score

    # ------------------------------------------------------------------
