import time
import numpy as np

# -------------------------------------------------------------
# Helper utilities (simple, memory‑light, and pure NumPy)
# -------------------------------------------------------------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, n: int, fft_len: int) -> np.ndarray:
    """Full convolution a*a using real FFT (length 2*n-1)."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * n - 1]


def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style coefficients w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _gaussian_shape(n: int, sigma_frac: float) -> np.ndarray:
    """Gaussian bump centred in the middle, sigma = sigma_frac * n."""
    sigma = max(1.0, sigma_frac * n)
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    v = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    if v.sum() == 0:
        v = np.ones_like(v)
    return v / v.sum()


def _initial_candidates(n: int, parent_arr: np.ndarray | None):
    """Generate a toolbox of plausible start vectors (all sum to 1)."""
    cand = []

    # parent, if compatible
    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            cand.append(p / s)

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Catalan shape
    w = _catalan_weights(n)
    cand.append(w / w.sum())

    # power shapes ((i+1)*(n-i))**p
    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1) * (n - idx)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        v = base ** p
        cand.append(v / v.sum())

    # sin‑power shapes
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    for p in (0.5, 1.0, 2.0):
        v = sin_vals ** p
        cand.append(v / v.sum())

    # Gaussian bumps, different spreads
    for sf in (0.05, 0.10, 0.20):
        cand.append(_gaussian_shape(n, sf))

    # few Dirichlet draws with diverse concentration
    rng = np.random.default_rng(0)
    for a in (0.3, 0.7, 1.5, 3.0):
        cand.append(rng.dirichlet(np.full(n, a)))
    return cand
