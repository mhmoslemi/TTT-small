import time, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho_arr = u - css / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0
    return w

def _conv_fft(a, L):
    # convolution full using FFT
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2*len(a)-1]
    return conv
```

import time
import numpy as np

def _normalize_nonneg(v):
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total <= 0.0:
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / total
    return v

def _conv_full_fft(a, L):
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return conv

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = np.full(256, 1.0, dtype=np.float64)

    a = _normalize_nonneg(a)
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    conv = _conv_full_fft(a, L)
    max_val = float(conv.max())
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    step = 0.5
    min_step = 1e-6
    decay = 0.997
    momentum = 0.9
    vel = np.zeros_like(a)

    rel_tol = 1e-7
    prob_move = 0.004
    no_improve = 0
    restart_thresh = 800  # iterations without improvement before perturb
    # main loop
    while time.monotonic() < deadline:
        # recompute conv and max
        conv = _conv_full_fft(a, L)
        max_val = float(conv.max())
        cur_score = 2.0 * n * max_val

        # keep best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # active set near max
        thresh = max_val - max(rel_tol * max_val, 1e-12)
        active = np.where(conv >= thresh)[0]

        # gradient from active set
        g = np.zeros_like(a)
        for idx in active:
            lo = idx - (n - 1)
            if lo < 0:
                lo = 0
            hi = idx
            if hi >= n:
                hi = n - 1
            # slice indices
            j = np.arange(lo, hi + 1, dtype=np.int64)
            g[j] += 2.0 * a[idx - j]

        # momentum
        vel = momentum * vel + (1.0 - momentum) * g

        # backtracking line search
        step_cur = step
        improved = False
        for _ in range(8):
            cand = a - step_cur * vel
            cand = _normalize_nonneg(cand)
            conv_c = _conv_full_fft(cand, L)
            cand_score = 2.0 * n * float(conv_c.max())
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                step = step_cur  # keep successful step size
                improved = True
                break
            step_cur *= 0.5
        if not improved:
            step = max(min_step, step * decay)
            # occasional random mass move
            if rng.random() < prob_move:
                i = rng.integers(0, n)
                if a[i] > 1e-12:
                    j = rng.integers(0, n - 1)
                    if j >= i:

                    frac = rng.random() * 0.35

                    cand2 = a.copy()

                    cand2[j] += delta
                    cand2 = _normalize_nonneg(cand2)
                    conv2 = _conv_full_fft(cand2, L)
                    cand2_score = 2.0 * n * float(conv2.max())
                    if cand2_score < cur_score:

                        cur_score = cand2_score

        if time.monotonic() + 0.5 > deadline:

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple non‑negative L1 normalisation (keeps sum = 1)
# ----------------------------------------------------------------------
def _normalize_nonneg(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform if everything vanished
        v.fill(1.0 / v.size)
        return v
    v /= s
    return v


# ----------------------------------------------------------------------
# Full auto‑correlation via real FFT (returns the length‑(2n‑1) array)
# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return conv

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # leave a tiny buffer for the harness shutdown
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # initial sequence (parent if provided)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = np.full(256, 1.0, dtype=np.float64)      # very simple fallback
    a = _normalize_nonneg(a)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two for FFT speed

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    conv = _conv_full_fft(a, L)
    max_val = float(conv.max())
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # optimisation parameters
    step = 0.5
    min_step = 1e-6
    decay = 0.997
    momentum = 0.9
    vel = np.zeros_like(a)

    rel_tol = 1e-7          # active‑set tolerance (relative to peak)
    prob_move = 0.004       # probability of a random mass transfer
    no_improve = 0
    restart_after = 800    # iterations without improvement before a perturb

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---------- recompute convolution and current score ----------
        conv = _conv_full_fft(a, L)
        max_val = float(conv.max())
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ---------- active set (indices whose value is almost maximal) ----------
        thresh = max_val - max(rel_tol * max_val, 1e-12)
        active = np.where(conv >= thresh)[0]

        # ---------- gradient from the active set ----------
        g = np.zeros_like(a)
        for idx in active:
            lo = idx - (n - 1)
            if lo < 0:
                lo = 0
            hi = idx
            if hi >= n:
                hi = n - 1
            j = np.arange(lo, hi + 1, dtype=np.int64)
            g[j] += 2.0 * a[idx - j]

        # ---------- momentum ----------
        vel = momentum * vel + (1.0 - momentum) * g

        # ---------- back‑tracking line search ----------
        step_cur = step
        improved = False
        for _ in range(8):
            cand = a - step_cur * vel
            cand = _normalize_nonneg(cand)
            conv_c = _conv_full_fft(cand, L)
            cand_score = 2.0 * n * float(conv_c.max())
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                step = step_cur            # keep the successful step size
                improved = True
                break
            step_cur *= 0.5

        # ---------- if no improvement: shrink step and try random move ----------
        if not improved:
            step = max(min_step, step * decay)

            if rng.random() < prob_move:
                i = rng.integers(0, n)
                if a[i] > 1e-12:
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1

                    delta = a[i] * frac
                    cand2 = a.copy()
                    cand2[i] -= delta
                    cand2[j] += delta
                    cand2 = _normalize_nonneg(cand2)
                    conv2 = _conv_full_fft(cand2, L)
                    cand2_score = 2.0 * n * float(conv2.max())

                        cur_score = cand2_score

                a = _normalize_nonneg(best + perturb)

                no_improve = 0

        # ---------- early termination safeguard ----------
        if time.monotonic() + 0.5 > deadline:
            break

    return [float(x) for x in best]
