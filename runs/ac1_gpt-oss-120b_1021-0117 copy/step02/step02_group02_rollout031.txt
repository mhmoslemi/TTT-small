    # 3. Initialise best candidate
    # ------------------------------------------------------------------ #
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    target_score = 1.50274349

    # convolution parameters
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # optimisation parameters
    eta0 = 0.30                     # initial step magnitude
    delta_rel = 0.001               # tolerance for “active” shifts
    max_active = 5                  # cap on number of active constraints
    jitter_interval = 30
    jitter_sigma = 0.018            # log‑normal jitter std. dev.

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ---- convolution via FFT (full) ---------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target_score:
                break

        # ---- active–set gradient ----------------------------------------
        # choose up to ``max_active`` largest shifts
        if conv.size <= max_active:
            active_shifts = np.arange(conv.size)
        else:
            # back‑selection gives the indices of the largest values
            active_shifts = np.argpartition(-conv, max_active)[:max_active]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active_shifts:
            partner = k - idx
            mask = (partner >= 0) & (partner < n)
            # sub‑gradient contribution (double count, safe sub‑gradient)
            g[mask] += 2.0 * a[partner[mask]]

        # ---- multiplicative mirror‑descent update ------------------------
        eta = eta0 / np.sqrt(it)
        # bound exponent to avoid under‑flow / overflow
        exp_factor = np.exp(np.clip(-eta * g, -50.0, 0.0))
        a = a * exp_factor
        total = a.sum()
        if total <= 0.0 or not np.isfinite(total):
            a = uniform.copy()
        else:
            a = a / total

        # ---- occasional jitter ------------------------------------------
        if it % jitter_interval == 0:
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = a * jitter
            a = a / a.sum()

    # ------------------------------------------------------------------ #
    # 4. Return best found sequence as plain Python floats
    # ------------------------------------------------------------------ #
    return [float(v) for v in best]
