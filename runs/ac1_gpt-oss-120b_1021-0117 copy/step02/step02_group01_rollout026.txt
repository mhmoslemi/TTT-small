        step = max(min_step, step * decay)

        # early exit if near deadline
        if time.monotonic() + 0.5 > deadline:

    return [float(x) for x in best]

import time
import numpy as np

# ----------------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the simplex. """
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic sqrt‑series (fallback initializer)
# ----------------------------------------------------------------------
def _sqrt_series(m: int) -> np.ndarray:
    a = np.empty(m, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, m):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ----------------------------------------------------------------------
# Convolution via FFT (full, linear convolution)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    return conv
