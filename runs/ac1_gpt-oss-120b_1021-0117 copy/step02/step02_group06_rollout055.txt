import time
import numpy as np

# -------------------------------------------------
# Helper utilities
# -------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution of a using real FFT (length 2*len(a)-1)."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    n = a.shape[0]
    return full[: 2 * n - 1]


def _next_pow2(x: int) -> int:
    """Small utility for the next power of two."""
    return 1 << ((x - 1).bit_length())

# Main optimisation routine
# -------------------------------------------------

    n = int(parent_arr.shape[0]) if parent_arr is not None else 256

    # ---- build a few sensible initial candidates ----
    cand = []

    cand.append(uniform)

    # parent normalised
    if parent_arr is not None:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            cand.append(p / s)

    # sinusoid power shapes
    idx = np.arange(n, dtype=np.float64)

    sin_vals = np.sin(theta)
    for pwr in (1.0, 2.0, 4.0):

    return best_a.tolist()
