import time
import numpy as np

# ------------------------------------------------------------
# Analytic sqrt‑series (a*a == 1+…+x^{2n-2})  – fallback generator
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ------------------------------------------------------------
# Full linear convolution (FFT based) – returns the first 2n‑1 values
# ------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(a, n=L)
    c = np.fft.irfft(fv * fv, n=L)
    return c[: 2 * a.size - 1]

# ------------------------------------------------------------
# Gradient of the maximal convolution entry (sub‑gradient of max)
# ------------------------------------------------------------
def _active_gradient(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        idx = np.arange(i_min, i_max + 1)
        grad[idx] += 2.0 * a[k - idx]
    return grad

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    # allow a few seconds margin for the harness to collect the answer
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # -------------------- initialise ---------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = _sqrt_series(256).astype(np.float64)

    n = a.shape[0]
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / a_sum                 # enforce sum = 1

    # FFT length (next power of two)
    Lfft = 1 << ((2 * n - 1).bit_length())

    # -------------------- bookkeeping --------------------------------
    conv = _full_conv(a, Lfft)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best = a.copy()
    best_M = M

    iteration = 0
    eta0 = 0.6                       # initial step magnitude
    no_improve = 0

    # -------------------- main optimisation loop --------------------
    while time.monotonic() < deadline:
        iteration += 1

        # recompute convolution and maximum for the current a
        conv = _full_conv(a, Lfft)
        M = float(conv.max())
        cur_score = 2.0 * n * M

        # active indices where the maximum is attained (with a tiny tolerance)
        active = np.where(conv >= M - 1e-14)[0]

        # sub‑gradient of the maximal entry
        grad = _active_gradient(a, active)

        # step size (decreasing)
        eta = eta0 / np.sqrt(iteration)
        # exponentiated‑gradient update (keeps non‑negativity)
        a_new = a * np.exp(-eta * grad)
        s = a_new.sum()

            s = a_new.sum()
        a_new /= s
