import time
import numpy as np

# ------------------------------------------------------------
# Helper: FFT based full linear convolution (a * a)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (exact same as evaluate_sequence)
# ------------------------------------------------------------
def _score_fast(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ------------------------------------------------------------
# Compute gradient of a smoothed max‑convolution objective.
# w : non‑negative weights on each convolution index, sum(w)=1
# grad = 2 * (w ⋆ a)   (cross‑correlation)
# ------------------------------------------------------------
def _grad_weighted(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    a_rev = a[::-1]
    L = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a_rev, L)
    conv = np.fft.irfft(W * A, L)
    start = n - 1
    return 2.0 * conv[start:start + n]


# ------------------------------------------------------------
# Project onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin for the sandbox
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initial sequence (use provided parent if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: decreasing exponential (length 256)
        n = 256
        i = np.arange(n, dtype=np.float64)
        lam = np.log(2.0) / n
        a = np.exp(-lam * i)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                                     # normalise to sum = 1

    n = a.size

    best = a.copy()
    best_score = _score_fast(best)

    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    beta = 1000.0                # smoothing parameter for soft‑max weights
    eta0 = 0.5                   # base step size for exponentiated update
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # ---- current convolution and score (no extra FFT)
        conv = _conv_fft(a)
        conv_max = conv.max()
        cur_score = 2.0 * n * conv_max          # sum(a) == 1 by construction

        # ---- weighted gradient (soft‑max approximation of the max)
        # avoid division by zero
        conv_max_safe = max(conv_max, 1e-15)
        w_raw = np.exp(beta * (conv / conv_max_safe - 1.0))
        w_sum = w_raw.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # fall back to pure sub‑gradient of the active maximal entry
            k = int(np.argmax(conv))
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g = np.zeros_like(a)

            break

    # return as a plain Python list (required by the harness)
    return [float(x) for x in best]
