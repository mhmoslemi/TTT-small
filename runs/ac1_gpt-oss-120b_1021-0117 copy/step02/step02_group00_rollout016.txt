# ------------------------------------------------------------
# Simplex projection (non‑negative, unit L1 norm)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex. """
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return full convolution a*a using length‑L FFT (L power of two)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # initialise: use parent if present, otherwise a simple start
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / a.size)
    else:
        # simple uniform start (any non‑empty vector works)
        n0 = 256
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    n = a.size

    # --------------------------------------------------------
    # parameters for the FFT size – we need lengths for
    #   a*a  (2n‑1) and a*w (n+2n‑2 = 3n‑2)
    # --------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2

    # --------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------
    best_vec = a.copy()
    best_conv = _conv_full_fft(best_vec, L)
    best_score = 2.0 * n * best_conv.max()   # sum(a)=1 after projection

    cur = best_vec.copy()
    cur_score = best_score

    step0 = 0.5          # initial step length
    iter_cnt = 0
    no_improve = 0

    while time.monotonic() < deadline:
        iter_cnt += 1

        # ---- FFT of current vector (re‑used for conv & correlation) ----
        a_fft = np.fft.rfft(cur, n=L)

        # ---- convolution a*a (full) ----------------------------------
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_vec = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ---- soft‑max weights over convolution values -----------------
        #   beta ramps up slowly, turning the smooth approximation into
        #   a near‑max as optimisation proceeds.
        beta = min(200.0, 0.02 * iter_cnt)        # schedule
        m = max_conv
        # stab: shift by max before exp to avoid overflow
        exp_vals = np.exp((conv - m) * beta)
        w = exp_vals / exp_vals.sum()            # length 2n‑1, non‑negative

        # ---- correlation a * w (needed for gradient) ----------------
        w_fft = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(a_fft * np.conj(w_fft), n=L)[:n]

        # ---- gradient of the smooth surrogate -----------------------
        #   grad = a + correlation(a, w)
        grad = cur + corr

        # ---- sub‑gradient step + projection onto simplex ---------------
        step = step0 / math.sqrt(iter_cnt)
        cand = cur - step * grad
        cand = _project_simplex(cand)

        # ---- evaluate candidate ----------------------------------------
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_score = 2.0 * n * cand_conv.max()

        # ---- accept if improvement, else Metropolis move ---------------
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best_vec, best_score = cand.copy(), cand_score
                no_improve = 0
        else:
            # occasional uphill move to escape flat regions
            prob = math.exp(-(cand_score - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        # ---- random mass‑transfer exploration (periodic) ---------------
        if no_improve > 2500 and rng.random() < 0.12:
            i = rng.integers(0, n)
            if cur[i] > 0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.35
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_score = 2.0 * n * tmp_conv.max()
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    if tmp_score < best_score:
                        best_vec, best_score = tmp.copy(), tmp_score
                        no_improve = 0

        # ---- jittered restart towards the best solution (occasionally) --
        if iter_cnt % 5000 == 0 and rng.random() < 0.25:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_fft = np.fft.rfft(tmp, n=L)
            tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
            tmp_score = 2.0 * n * tmp_conv.max()
            if tmp_score < cur_score:
                cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best_vec, best_score = tmp.copy(), tmp_score
                    no_improve = 0

        # ---- early exit if target reached --------------------------------
        if best_score < 1.50274349:
            break

    # return as plain Python list (scores are scale‑invariant)
    return [float(x) for x in best_vec]
