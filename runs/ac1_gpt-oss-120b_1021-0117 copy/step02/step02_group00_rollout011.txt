def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
    return conv

def _project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]

            g[idx] += 2.0 * cur[k - idx]
        g /= float(len(active))

        # decreasing step size (≈ 1/√t)
        base_alpha = init_alpha / np.sqrt(iteration)

        # line‑search on a few multipliers (including staying put)
        cand_best = cur
        cand_best_sc = cur_sc
        for mult in (0.5, 1.0, 1.5, 2.0):
            cand = cur - (base_alpha * mult) * g
            cand = _project_simplex(cand)
            cand_sc = _score(cand)
            if cand_sc < cand_best_sc:
                cand_best = cand
                cand_best_sc = cand_sc

        cur = cand_best
        cur_sc = cand_best_sc

        # record any improvement
        if cur_sc + 1e-12 < best_sc:
            best_vec = cur.copy()
            best_sc = cur_sc
            no_improve = 0
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # occasional random mass‑transfer when progress stalls
        # ------------------------------------------------------------------
        if no_improve > 2500 and rng.random() < 0.2:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < best_sc:
                    best_vec = tmp.copy()
                    best_sc = tmp_sc
                    cur = tmp
                    cur_sc = tmp_sc
                    no_improve = 0

        # ------------------------------------------------------------------
        # periodic jittered restart around the best solution
        # ------------------------------------------------------------------
        if iteration % 4000 == 0:
            jitter = rng.normal(scale=1e-4, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur = tmp
                cur_sc = tmp_sc
            if tmp_sc < best_sc:
                best_vec = tmp.copy()
                best_sc = tmp_sc
                no_improve = 0

        # ------------------------------------------------------------------
        # hard restart after a long stagnation
        # ------------------------------------------------------------------
        if no_improve > 5000:
            cur = _project_simplex(rng.random(n))
            cur_sc = _score(cur)
            no_improve = 0

        # early exit if target already reached
        if best_sc < 1.50274349:
            break

    # ----------------------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # ----------------------------------------------------------------------
    return [float(x) for x in best_vec]
