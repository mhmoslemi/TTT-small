import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes the AC1 bound.
    Starts from the supplied parent (``height_sequence_1``) or from a uniform
    vector, then repeatedly:
      * evaluates the current max autocorrelation,
      * solves a linear program that maximises the total sum under the same
        autocorrelation bound,
      * rescales the LP solution back to a unit‑sum simplex.
    The best sequence encountered is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1. Initialise a
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        a = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = None
        a = None

    if a is None or not np.isfinite(a).all():
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s                       # normalise to sum = 1

    # ------------------------------------------------------------------
    # 2. Keep the best seen sequence
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # compare with pure uniform as a cheap fallback
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_score = float(evaluate_sequence(uniform.tolist()))
    if uni_score < best_score:
        best = uniform.copy()
        best_score = uni_score
        a = uniform.copy()

    target_score = 1.50274349
    max_lp_iters = 8          # never exceed the overall time budget

    # ------------------------------------------------------------------
    # 3. Main optimisation loop – LP based improvement
    # ------------------------------------------------------------------
    lp_iter = 0
    while time.monotonic() < deadline and lp_iter < max_lp_iters:
        lp_iter += 1

        # ---- current maximal autocorrelation M ----
        conv_len = 2 * n - 1
        fft_len = 1 << ((conv_len - 1).bit_length())
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        M = conv.max()
        cur_score = 2.0 * n * M               # sum(a)=1 ⇒ denominator =1
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target_score:
                break

        # ---- build sparse constraint matrix A (size (2n‑1)×n) ----
        # each column i has the constant a[i] on rows i .. i+n‑1
        col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
        row_idx = np.tile(np.arange(n, dtype=np.int32), n) + col_idx
        data = np.repeat(a, n)
        A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

        # ---- linear programme: maximise sum(b) with A·b ≤ M, b ≥ 0 ----
        c = -np.ones(n, dtype=np.float64)               # minimise –sum(b)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n

        try:
            res = opt.linprog(
                c, A_ub=A, b_ub=b_ub, bounds=bounds,
                method='highs',
                options={'presolve': True}
            )
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b_val = np.maximum(res.x, 0.0)
        sum_b = b_val.sum()
        if sum_b <= 1.0 + 1e-12:          # no useful increase
            break

        # ---- rescale to keep sum = 1 (simplex) ----
        a = b_val / sum_b

        # occasional tiny jitter to avoid stagnation
        if lp_iter % 2 == 0:
            jitter = np.exp(rng.normal(0.0, 0.003, size=n))
            a = _proj_simplex(a * jitter)

        # renormalise exactly (numerical safety)
        a = a / a.sum()

    # ------------------------------------------------------------------
    # 4. Final evaluation and return
    # ------------------------------------------------------------------
    final_score = float(evaluate_sequence(a.tolist()))
    if final_score < best_score:
        best = a.copy()
        best_score = final_score

    return [float(v) for v in best]
