def _project_simplex(v):
    # returns projection onto simplex sum=1
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # adjust for numeric issues; if sum zero, set uniform
    s = w.sum()
    if s <= 0:
        w = np.full_like(v, 1.0/n)
    else:
        w = w / s  # ensure sum=1
    return w

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx_i = 1, x_i >= 0."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_idx = np.where(rho_arr > 0)[0]
    if rho_idx.size:
        rho = rho_idx[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s <= 0:                      # degenerate case – fall back to uniform
        w[:] = 1.0 / n
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
# Analytic sqrt‑series (fallback initializer)
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Initialise sequence (use parent when present)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = _sqrt_series(256)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)

    n = a.size
    # FFT length sufficient for linear convolutions of lengths (2n‑1) and (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())
    idx_rev = 2 * n - 2 - np.arange(n)          # indices used for gradient

    # ------------------------------------------------------------------
    # Initial score (sum is 1, so score = 2·n·max(conv))
    # ------------------------------------------------------------------
    fa = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    cur_score = 2.0 * n * conv_full.max()
    best = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    step = 0.2
    step_min, step_max = 1e-6, 2.0
    beta = 5.0                # soft‑max sharpness
    beta_inc = 1.02
    beta_max = 20000.0
    prob_random = 0.02        # probability of a random mass‑transfer move
    momentum = np.zeros_like(a)

    it = 0
    while time.monotonic() < deadline and it < 20000:
        it += 1

        # ------- full auto‑convolution -------
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ------- soft‑max weights on the convolution -------
        delta = conv - max_val                               # ≤ 0
        exps = np.exp(np.clip(beta * delta, -700, 0))
        w = exps / exps.sum()                                # Σw = 1

        # ------- gradient of weighted sum -------
        w_rev = w[::-1]
        fw_rev = np.fft.rfft(w_rev, n=L)
        conv_rev = np.fft.irfft(fa * fw_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * conv_rev[idx_rev]                       # shape (n,)

        # momentum update

        cand = _project_simplex(cand)

        # ------- acceptance -------
        improved = cand_score < cur_score

            cur_score = cand_score

                best_score = cand_score
                best = cand.copy()
        else:
            # Metropolis‑style acceptance

            if rng.random() < np.exp(-(cand_score - cur_score) / T):
                a = cand
                cur_score = cand_score

    # ------------------------------------------------------------------
    # Return the best discovered sequence (as plain Python list)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    best = _project_simplex(best)          # final safety projection
    return [float(x) for x in best]
