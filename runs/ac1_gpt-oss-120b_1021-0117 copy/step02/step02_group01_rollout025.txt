    fv = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2*len(seq)-1]
    idx = conv.argmax()
    return conv[idx], int(idx)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    # time
    deadline = time.monotonic() + max(0., float(budget_s) - 0.5) # small headroom

    rng = np.random.default_rng(seed)

    # initial vector a, fallback if height_sequence_1 not in globals
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # create uniform distribution across n (maybe 1280)
        n_init = 1024
        a = np.full(n_init, 1.0 / n_init)
    # project onto simplex (so sum=1)
    a = _project_simplex(a)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # compute current max conv & score
    max_val, max_idx = _conv_max_and_index(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # optimization parameters
    step = 0.5   # initial learning rate
    beta = 0.9   # momentum
    v = np.zeros_like(a)
    decay = 0.9997
    min_step = 1e-6
    # probability for random mass transfer moves
    prob_move = 0.02
    # temperature for simulated annealing acceptance for gradient steps worsens
    temperature = 0.1

    # loops
    iter_count = 0
    while time.monotonic() < deadline:
        # compute gradient at max_idx
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        # g shape: for j in [lo,hi], derivative = 2 * a[max_idx - j]
        # Build g array
        g = np.zeros_like(a)
        # slice assignment
        j = np.arange(lo, hi + 1)
        g[j] = 2.0 * a[max_idx - j]

        # Momentum update
        v = beta * v - step * g
        cand = a + v
        cand = _project_simplex(cand)

        # evaluate
        max_val_c, max_idx_c = _conv_max_and_index(cand, L)
        cand_score = 2.0 * n * max_val_c

        # Accept if improvement, else maybe accept by temperature
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            max_idx = max_idx_c
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            # Acceptance by simulated annealing: probability exp(-(cand-score-cur_score)/temp)
            if temperature > 0 and rng.random() < np.exp(-(cand_score - cur_score) / temperature):
                a = cand
                cur_score = cand_score
                max_idx = max_idx_c
                # note: not update best if worse
            else:
                # revert momentum (if not accepted) -> discard v? or keep? We'll keep v unchanged not to accumulate poor direction.
                # Instead, we can reset v to zero to avoid sticky momentum on bad moves
                v = np.zeros_like(a)

        # occasional random mass transfer for extra exploration
        if rng.random() < prob_move:
            # choose donor i with a_i>0
            donor_indices = np.where(a > 1e-12)[0]
            if donor_indices.size > 0:
                i = rng.choice(donor_indices)
                # choose recipient j != i
                j = rng.integers(0, n - 1)
                while j == i:
                    j = rng.integers(0, n - 1)
                # fraction up to 0.5 of donor mass
                frac = rng.random() * 0.5
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                # optionally add small noise to avoid all zeros
                cand2 = _project_simplex(cand2)
                max_val2, max_idx2 = _conv_max_and_index(cand2, L)
                cand2_score = 2.0 * n * max_val2
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    max_idx = max_idx2
                    if cand2_score < best_score:

                    if temperature > 0 and rng.random() < np.exp(-(cand2_score - cur_score)/temperature):

                        cur_score = cand2_score
                        max_idx = max_idx2

        # schedule update
        step = max(min_step, step * decay)
        temperature *= decay  # slowly cooling

        iter_count += 1

        # early break if time near end
        if time.monotonic() + 0.2 > deadline:
            break

    return [float(x) for x in best]

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (L2‑Euclidean) – keeps sum == 1, non‑negative
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Full convolution via FFT, returning max value, its index,
# and the whole array (needed for an active‑set gradient)
# ------------------------------------------------------------
def _conv_full_and_max(a: np.ndarray, L: int):
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # reserve a little head‑room before the hard timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----------------------------------------------------------------
    # initialise a – use the provided parent when available
    # ----------------------------------------------------------------

    a = _project_simplex(a)                     # sum == 1
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())          # next power of two

    # initial score (sum == 1  →  score = 2·n·max(conv))
    max_val, max_idx, conv = _conv_full_and_max(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------

    decay = 0.9994
    beta = 0.85                 # momentum

    no_improve = 0
