import time
import numpy as np

def _max_conv(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum entry of the full self‑convolution of a (a is assumed sum‑to‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(conv.max())


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Returns a probability‑vector (non‑negative, sum = 1) whose
    2·n·max(a*a) score is as low as possible.
    The search explores exponential decays of the form a_i ∝ r^i,
    performs a coarse‑then‑fine grid search and keeps the best
    candidate (including the supplied parent if present).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1.  Determine length n and a fallback uniform start.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = int(parent.size)
        parent = np.maximum(parent, 0.0)
        s = parent.sum()
        if s < 0.01:
            parent = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            parent = parent / s          # normalise to sum = 1
    except Exception:
        parent = None
        n = 256
        parent = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2.  Pre‑computations for FFT‑based convolution.
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    idx = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3.  Helper to evaluate a candidate (sum already = 1).
    # ------------------------------------------------------------------
    def score_of(a: np.ndarray) -> float:
        """Exact score used by the evaluator (2·n·max(conv))."""
        return 2.0 * n * _max_conv(a, fft_len, conv_len)

    # ------------------------------------------------------------------
    # 4.  Initialise best candidate (uniform + optional parent).
    # ------------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = score_of(best_seq)

    if parent is not None:
        parent_score = score_of(parent)
        if parent_score < best_score:
            best_score = parent_score
            best_seq = parent.copy()

    # ------------------------------------------------------------------
    # 5.  Exponential‑family search.
    # ------------------------------------------------------------------
    # Coarse sweep
    low, high = 0.95, 0.9999
    coarse_steps = 250
    best_r = None

    for r in np.linspace(low, high, coarse_steps):
        if time.monotonic() >= deadline:
            break
        if abs(r - 1.0) < 1e-14:               # avoid division by zero
            a = best_seq.copy()
        else:
            # geometric series normaliser, stable for r≈1
            if abs(1.0 - r) < 1e-12:
                denom = float(n)
            else:
                rn = r ** n
                denom = (1.0 - rn) / (1.0 - r)
            a = (r ** idx) / denom
        cur = score_of(a)
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            best_r = r
            if best_score <= 1.50274349:
