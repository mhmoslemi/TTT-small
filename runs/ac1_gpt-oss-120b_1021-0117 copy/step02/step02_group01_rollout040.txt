import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Generate a simple decreasing power‑law sequence.
    Several exponents are tried; the one with the smallest
    2·n·max(conv)/sum² (as returned by the provided evaluator) is returned.
    """
    # leave a little head‑room for the sandbox
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Length of the parent sequence if it exists, otherwise a modest default
    n = len(globals().get("height_sequence_1", []))
    if n == 0:
        n = 256

    # Base decreasing integers: n, n‑1, …, 1
    base = np.arange(n, 0, -1, dtype=np.float64)

    best_seq = None
    best_score = float("inf")

    # Try a few power‑law exponents; p=3 (cubic) is usually the best
    for p in (1, 2, 3, 4, 5):
        # Abort if time is exhausted
        if time.monotonic() > deadline:
            break

        # Power‑law weights and L¹ normalisation (sum = 1)
        a = np.power(base, p, dtype=np.float64)
        total = a.sum()
        if total == 0.0:
            continue
        a /= total

        # Evaluate via the provided scorer
        try:
            score = evaluate_sequence(a.tolist())
        except Exception:
            score = float("inf")

        if score < best_score:
            best_score = score
            best_seq = a.copy()

        # Very low scores are already optimal enough; early exit
        if best_score < 0.01:
            break

    # Fallback to a uniform distribution if something went wrong
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64) / n

    # Return plain Python floats (the evaluator expects a list)
    return [float(x) for x in best_seq]
