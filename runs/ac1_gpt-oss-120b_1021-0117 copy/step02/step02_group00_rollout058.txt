def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # get n and initial vector
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        a0 = np.ones(n) # maybe uniform
    # ensure nonnegative and sum to 1
    a0 = np.maximum(a0, 0.0)
    a0 = a0 + 1e-12  # add epsilon to avoid zero later
    a0 /= a0.sum()

    # function definitions: convolution via FFT (use caching length)
    def conv_full(a):
        n = a.shape[0]
        L = 1 << ((2*n - 1).bit_length())
        fv = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[:2*n - 1]
        return conv

    def score(a):
        # Use evaluate_sequence? But to avoid overhead, compute directly same formula
        # we can compute as 2 * n * max(conv) / (sum(a)**2). Since sum=1 we just 2*n*max(conv)
        conv = conv_full(a)
        return 2.0 * a.shape[0] * conv.max()  # since sum=1

    # define function to compute gradient subgradient for max conv
    def grad_max_conv(a):
        n = a.shape[0]
        conv = conv_full(a)
        k = int(np.argmax(conv))
        # gradient
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[k - idx]

    """Full linear convolution a*a using a power‑of‑two FFT."""
