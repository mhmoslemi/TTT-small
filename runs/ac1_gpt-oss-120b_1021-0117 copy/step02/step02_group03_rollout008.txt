import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Analytic sqrt‑series (fallback if no parent is supplied)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

            i = rng.integers(0, n)
            if cand[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.12
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta

        # enforce feasibility (non‑negative, sum = 1)
        cand = _project_simplex(cand)

        a = cand

    # final check – maybe the last a is better than best
    a_fft = np.fft.fft(a, n=L)
    conv_full = np.fft.ifft(a_fft * a_fft).real[: 2 * n - 1]
    final_score = 2.0 * n * conv_full.max()
    if final_score < best_score:
        best = a.copy()
        best_score = final_score

    # ------------------------------------------------------------
    # return result as a plain Python list
    # ------------------------------------------------------------
    return [float(x) for x in best]
