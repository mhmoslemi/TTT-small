import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Sub‑gradient of the active lag k of a*a
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad

# ------------------------------------------------------------
# Max of the (full) convolution a*a using FFT (Lfft is power of 2)
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, Lfft: int) -> float:
    fv = np.fft.rfft(a, n=Lfft)
    conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())

# Targeted mass‑transfer move using the active lag
# ------------------------------------------------------------
def _targeted_transfer(a: np.ndarray, k: int, rng: np.random.Generator) -> np.ndarray:
    n = a.shape[0]
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx

    i_local = int(np.argmax(contrib))
    i = int(idx[i_local])

    if a[i] <= 1e-12:
        return a.copy()
    # pick a destination distinct from i and its partner
    while True:
        j = rng.integers(0, n)
        if j != i and j != p:
            break
    # move a modest fraction of mass
    eps = rng.uniform(0.02, 0.10)
    delta = min(eps * a[i], a[i])
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    # renormalise onto simplex (keeps sum=1, eliminates tiny negatives)
    return _project_simplex(a2)
