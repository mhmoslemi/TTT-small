def sqrt_series_full(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1,n):
        if k == 1:
            s = 0.0
        else:
            # compute dot of a[1:k] and a[k-1:0:-1]
            # we can use np.dot for slices
            s = np.dot(a[1:k], a[k-1:0:-1])
        a[k] = (1.0 - s) / 2.0
    return a

import numpy as np

def _sqrt_series_full(n: int) -> np.ndarray:
    """Return the length‑n coefficients a satisfying a*a = 1+…+x^{2n-2}.
    The recurrence a₀=1, a_k = (1‑∑_{i=1}^{k‑1} a_i·a_{k‑i})/2
    generates the optimal sequence for the AC1 bound."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # sum_{i=1}^{k-1} a_i * a_{k-i}
            s = np.dot(a[1:k], a[k-1:0:-1])
        a[k] = (1.0 - s) / 2.0
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Produce a non‑negative step‑function sequence that attains the
    theoretical minimum of the AC1 upper bound.
    The sequence length matches the provided parent (height_sequence_1)
    when available; otherwise a default length of 256 is used.
    """
    # Length to construct
    n = len(height_sequence_1) if "height_sequence_1" in globals() else 256

    # Compute the optimal sqrt‑series coefficients
    a = _sqrt_series_full(n)

    # Safety clamp – values are already ≤1, but keep within evaluator limits
    a = np.clip(a, 0.0, 1000.0)

    # Return as ordinary Python list of floats
    return [float(x) for x in a]
