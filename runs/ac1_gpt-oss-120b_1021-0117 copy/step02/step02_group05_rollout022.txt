import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast evaluation of the AC1 bound via FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # initialise from parent if available, otherwise use a simple start
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        cur = parent.copy()
    except Exception:
        n = 256
        cur = _gen_binomial(n)

    cur = np.clip(cur, 0.0, 1000.0)
    L = _next_pow2(2 * n - 1)

    # make a quick check of the reversed version – sometimes better
    rev = cur[::-1].copy()
    cur_score_rev = _ac1_score(rev, n, L)
    cur_score = _ac1_score(cur, n, L)
    if cur_score_rev < cur_score:
        cur = rev
        cur_score = cur_score_rev

    best = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # hyper‑parameters
    temperature = 1.0
    decay = 0.9995
    min_block = max(2, n // 250)
    max_block = max(2, n // 12)
    target = 1.50274349

    # ------------------------------------------------------------------
    # main optimisation loop
    while time.monotonic() < deadline:
        move = rng.random()
        cand = None
        cand_score = None

        # --------------------------------------------------------------
        # 1) gradient‑guided descent – most frequent
        if move < 0.65:
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            kmax = int(np.argmax(conv))

            idx = np.arange(n)
            rev_idx = kmax - idx
            mask = (rev_idx >= 0) & (rev_idx < n)

            g = np.zeros_like(cur)
            g[mask] = 2.0 * cur[rev_idx[mask]]       # correct gradient of conv[kmax]

            mean_g = g.mean()
            d = mean_g - g                           # descent direction (sum(d)=0)

            # feasible step size respecting 0‑1000 bounds
            pos = d > 0
            neg = d < 0
            step_pos = np.inf
            step_neg = np.inf
            if np.any(pos):
                step_pos = ((1000.0 - cur[pos]) / d[pos]).min()
            if np.any(neg):
                step_neg = ((-cur[neg]) / d[neg]).min()
            step_max = min(step_pos, step_neg)

            if step_max > 0 and np.isfinite(step_max):
                # try a few fractions of the maximal step
                best_local = cur
                best_local_score = cur_score
                for frac in (1.0, 0.5, 0.25, 0.125, 0.0625):
                    step = step_max * frac
                    tmp = cur + step * d
                    np.clip(tmp, 0.0, 1000.0, out=tmp)
                    sc = _ac1_score(tmp, n, L)
                    if sc < best_local_score:

                        best_local_score = sc
                    if sc < cur_score:

                cand = best_local
                cand_score = best_local_score

        # --------------------------------------------------------------
        # 2) block scaling – moderate fraction

            cur_score = best_score
            temperature = 1.0

    # ------------------------------------------------------------------
    return [float(x) for x in best]
