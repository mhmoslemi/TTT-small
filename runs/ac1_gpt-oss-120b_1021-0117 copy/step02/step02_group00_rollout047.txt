def active_gradient(a, conv, eps=1e-12):
    n = a.size
    max_val = conv.max()
    act = (conv >= max_val - 1e-12).astype(float)    # float indicator
    # compute convolution of act with reversed a
    L = 1 << ((2*n - 1 + n - 1).bit_length())
    # but act and a_rev lengths: act_len = 2n-1, a_rev_len=n
    # cross-correlation length is act_len + a_rev_len - 1 = 3n-2.
    # we need conv at positions n-1 : n-1+(n-1) = 2n-2 i.e., n entries.
    L = 1 << ((3*n - 2).bit_length())
    # do FFT for real vectors: use np.fft.rfft and irfft
    # compute rfft for both zero-padded
    act_fft = np.fft.rfft(act, n=L)
    a_rev_fft = np.fft.rfft(a[::-1], n=L)
    full = np.fft.irfft(act_fft * a_rev_fft, n=L)
    # g = 2 * full[(n-1):(n-1+n)]
    g = 2.0 * full[(n-1):(n-1+n)]
    return g

def smooth_gradient(a, conv, beta):
    n = a.size
    maxv = conv.max()
    # compute weights
    w = np.exp(beta * (conv - maxv))
    # optionally cap to avoid overflow
    # w may be huge but conv - maxv <=0, so w <=1.
    # Use exponent of negative numbers -> small, fine.
    # compute convolution of w and a_rev
    L = 1 << ((2*n-1 + n -1).bit_length())
    w_fft = np.fft.rfft(w, n=L)
    a_rev_fft = np.fft.rfft(a[::-1], n=L)
    full = np.fft.irfft(w_fft * a_rev_fft, n=L)
    g = 2.0 * full[(n-1):(n-1+n)]
    return g

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (non‑negative, sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Full linear convolution via FFT (real input)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Gradient of the (smoothed) max‑convolution
# ------------------------------------------------------------
def _active_gradient(a: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """Gradient = 2 * (weight * a_rev) evaluated on the n entries."""
    n = a.shape[0]
    # convolution of weight (len 2n‑1) with a reversed (len n)
    L = 1 << ((weight.shape[0] + n - 1).bit_length())
    w_fft = np.fft.rfft(weight, n=L)
    a_rev_fft = np.fft.rfft(a[::-1], n=L)
    full = np.fft.irfft(w_fft * a_rev_fft, n=L)
    start = n - 1
    g = 2.0 * full[start:start + n]
    return g


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weighted gradient; beta controls smoothing."""
    maxv = conv.max()
    w = np.exp(beta * (conv - maxv))          # 0 ≤ w ≤ 1, max at the peak(s)
    return _active_gradient(a, w)

# Score (same formula as evaluate_sequence up to a constant factor)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2·n·max(conv) for a normalised to sum=1."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return 2.0 * n * conv.max()

                    no_improve = 0

        # early exit if target reached
        if best_sc < 1.50274349:
            break

    # ---------- final safeguard ----------
    if best_vec is None or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
