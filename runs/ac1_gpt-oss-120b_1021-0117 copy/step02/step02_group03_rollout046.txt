import time
import numpy as np

def _sqrt_series(n: int) -> np.ndarray:
    """Return the length‑n series a with a*a = 1 + x + … + x^{2n‑2}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # sum_{i=1}^{k-1} a_i * a_{k-i}
            s = np.dot(a[1:k], a[k - 1 : 0 : -1])
        a[k] = (1.0 - s) / 2.0
    return a

    else:
        n = 256

    # Build the exact square‑root series.
    a = _sqrt_series(n)

    # Clip tiny negatives that may arise from round‑off, then normalise to sum = 1.
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0.0:
        total = 1.0
    a = a / total
    # A second normalisation guarantees the sum is exactly 1 after clipping.
    a = a / a.sum()

    # Return as plain Python floats.
    return [float(x) for x in a]
