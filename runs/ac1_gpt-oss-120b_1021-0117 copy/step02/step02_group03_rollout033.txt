import time
import numpy as np

# ------------------------------------------------------------
# Simple projection onto the non‑negative simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum 1 (fallback uniform)."""
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        # uniform fallback – never happens in normal use
        v.fill(1.0 / v.size)
    else:
        v /= total
    return v


# ------------------------------------------------------------
# Gradient of the active autocorrelation lag k
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad


# ------------------------------------------------------------
# Return max autocorrelation and the full array (FFT‑based)
# ------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return conv


def _max_conv_and_active(a: np.ndarray, L: int, rel_tol: float = 1e-8):
    conv = _full_conv(a, L)
    M = float(conv.max())
    tol = max(1e-12, rel_tol * M)
    active = np.where(conv >= M - tol)[0]
    return M, active, conv


# ------------------------------------------------------------
# Score = 2 * n * max_conv (sum is normalised to 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    M = float(_full_conv(a, L).max())
    return 2.0 * a.shape[0] * M


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # tiny fall‑back: sqrt series of length 256
        def _sqrt_series(n):
            a = np.empty(n, dtype=np.float64)
            a[0] = 1.0
            for k in range(1, n):
                if k == 1:
                    s = 0.0
                else:
                    s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
                a[k] = (1.0 - s) / 2.0
            return a
        a0 = _sqrt_series(256)

    a0 = _project_simplex(a0)          # guarantee feasibility
    n = a0.shape[0]
    Lfft = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two for FFT

    # ---- bookkeeping ------------------------------------------------
    best = a0.copy()
    best_score = _score(best, Lfft)

    # ---- local optimiser -------------------------------------------
    def _local_search(start_vec, max_iters):
        nonlocal best, best_score
        a = start_vec.copy()
        cur_score = _score(a, Lfft)
        # momentum vector, initially zero
        vel = np.zeros_like(a)
        beta = 0.9          # momentum coefficient
        eta0 = 0.7          # base step size

        for it in range(1, max_iters + 1):
            if time.monotonic() >= deadline:
                break

            M, act_ks, _ = _max_conv_and_active(a, Lfft)
            # accumulate sub‑gradient from all active lags
            grad = np.zeros_like(a)
            for k in act_ks:
                grad += _grad_for_k(a, k)
            norm = np.linalg.norm(grad)
            if norm > 0.0:
                direction = -grad / norm
                vel = beta * vel + (1.0 - beta) * direction
                # normalise velocity to unit length for stable step magnitudes
                vnorm = np.linalg.norm(vel)
                if vnorm > 0.0:
                    vel = vel / vnorm
                eta = eta0 / np.sqrt(it)          # decaying step size
                cand = a + eta * vel
                cand = _project_simplex(cand)
            else:
                cand = a.copy()

            cand_score = _score(cand, Lfft)

            # keep global best
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

            # accept if improves, otherwise keep old point (small chance to accept)
            if cand_score < cur_score or rng.random() < 0.002:
                a = cand
                cur_score = cand_score

            # occasional random perturbation to escape plateaus
            if rng.random() < 0.015:
                i = rng.integers(0, n)
                if a[i] > 0.0:
                    j = rng.integers(0, n)
                    while j == i:
                        j = rng.integers(0, n)
                    frac = rng.random() * 0.12
                    delta = a[i] * frac
                    a[i] -= delta
                    a[j] += delta
                    a = _project_simplex(a)
                    cur_score = _score(a, Lfft)

            # early stop if target already reached
            if best_score < 1.5028:
                return

    # ---- primary search starting from the provided parent ----------
    _local_search(a0, max_iters=8000)

    # ---- additional exploratory starts if time permits -------------
    while time.monotonic() < deadline:
        # sample a decreasing power‑law shape
        p = rng.uniform(0.3, 3.0)
        shape = np.arange(n, 0, -1, dtype=np.float64) ** p
        start = _project_simplex(shape)
        _local_search(start, max_iters=400)

        if time.monotonic() >= deadline:
            break

        # sample a random Dirichlet‑like vector
        rand = rng.exponential(size=n)
        start = _project_simplex(rand)
        _local_search(start, max_iters=400)

        # tiny random Gaussian perturbation of the current best
        if rng.random() < 0.1:
            pert = rng.normal(scale=0.008, size=n)
            start = _project_simplex(best + pert)
            _local_search(start, max_iters=200)

        # safety break
        if best_score < 1.5028:
            break

    # ---------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ---------------------------------------------------------------
    return [float(x) for x in best]
