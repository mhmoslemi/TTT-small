# ------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, >=0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise exactly
    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w

def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)

        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[: 2 * len(vec) - 1]

        a_new = _project_simplex(a_new)

        # evaluate candidate
        maxc_new, _ = _conv_max(a_new, L)
        new_score = 2.0 * n * maxc_new

        # accept if improvement, otherwise Metropolis‑type acceptance
        improved = new_score < cur_score
        if improved:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best = a_new.copy()
                best_score = new_score
                no_improve = 0
        else:
            # occasional uphill acceptance
            if rng.random() < np.exp(-(new_score - cur_score) / max(step, 1e-8)):
                a = a_new
                cur_score = new_score

        # random mass transfer to escape flat regions
        if rng.random() < prob_random:
            # pick source with non‑trivial mass
            src_candidates = np.where(a > eps)[0]
            if src_candidates.size:
                i = rng.choice(src_candidates)
                j = rng.integers(0, n)
                if j == i:
                    j = (j + 1) % n
                frac = rng.random() * 0.3
                delta = a[i] * frac
                a_tmp = a.copy()
                a_tmp[i] -= delta
                a_tmp[j] += delta
                a_tmp = _project_simplex(a_tmp)
                maxc_tmp, _ = _conv_max(a_tmp, L)
                tmp_score = 2.0 * n * maxc_tmp
                if tmp_score < cur_score:
                    a = a_tmp
                    cur_score = tmp_score
                    if tmp_score < best_score:
                        best = a_tmp.copy()
                        best_score = tmp_score
                        no_improve = 0

        # step‑size schedule
        step = max(min_step, step * decay)

        # early‑stop if little progress for many iterations
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        if no_improve > 2000:
            # shrink step more aggressively to fine‑tune
            step = max(min_step, step * 0.5)
            no_improve = 0

        # stop if we already beat the target bound
        if best_score < 1.50274349:
            break

        # cheap time check
        if time.monotonic() + 0.1 > deadline:
            break

    # ------------------------------------------------------------
    # Return the best found sequence as plain Python list
    # ------------------------------------------------------------
    return [float(x) for x in best]
