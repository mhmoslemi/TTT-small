import time, math
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Projected smoothed‑max gradient descent for the AC1 bound.
    Starts from ``height_sequence_1`` if present, otherwise from a uniform
    distribution.  Returns the best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------------
    # 1. Initialise a
    # --------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        a = np.clip(parent, 0.0, 1000.0)
        n = a.size
        s = a.sum()
        a = a / s if s >= 0.01 else np.full(n, 1.0 / n, dtype=np.float64)
    except Exception:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------------
    # 2. Pre‑compute FFT size
    # --------------------------------------------------------------------
    conv_len = 2 * n - 1
    grad_len = 3 * n - 2
    fft_len = 1 << ((max(conv_len, grad_len) - 1).bit_length())

    # --------------------------------------------------------------------
    # 3. Parameters
    # --------------------------------------------------------------------
    eta0 = 0.6               # initial step size factor
    beta0 = 5.0              # start of smoothing
    beta_rate = 0.20         # linear increase per iteration
    max_beta = 200.0
    jitter_interval = 80
    jitter_sigma = 0.02
    target_score = 1.50274349

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    it = 0

    # --------------------------------------------------------------------
    # 4. Main optimisation loop
    # --------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        # ----- convolution (FFT) -------------------------------------------------
        a_fft = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)
        conv = conv_full[:conv_len]

        # ----- current score ----------------------------------------------------
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target_score:
                break

        # ----- smoothed max (softmax) -----------------------------------------
        beta = min(max_beta, beta0 + beta_rate * it)
        shifted = conv - max_conv                     # stability shift
        exp_vals = np.exp(beta * shifted)
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0 or not np.isfinite(sum_exp):
            p = np.zeros_like(conv)
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # ----- gradient via convolution p * a_rev ------------------------------
        a_rev = a[::-1]
        p_fft = np.fft.rfft(p, n=fft_len)
        a_rev_fft = np.fft.rfft(a_rev, n=fft_len)
        grad_full = np.fft.irfft(p_fft * a_rev_fft, n=fft_len)
        gradient = 2.0 * beta * grad_full[n - 1:n - 1 + n]

        # ----- step and projection ---------------------------------------------
        eta = eta0 / math.sqrt(it)
        a = _proj_simplex(a - eta * gradient)

        # ----- occasional jitter ------------------------------------------------
        if it % jitter_interval == 0:
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = _proj_simplex(a * jitter)

    # --------------------------------------------------------------------
    # 5. Return best candidate
    # --------------------------------------------------------------------
    return [float(v) for v in best]
