def _score_conv(seq):
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[:2*n-1]
    max_b = conv.max()
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_b
    score = 2.0 * n * max_b / (s * s)
    return score, conv, max_b
```

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _score_conv(seq: np.ndarray, fft_len: int):
    """Return (score, full convolution array, its maximum)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    max_b = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_b
    score = 2.0 * seq.size * max_b / (s * s)
    return float(score), conv, max_b


def _targeted_move(seq: np.ndarray, conv: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Reduce the contribution of the biggest pair (i,j) that forms the
    maximal convolution entry. A fraction of both values is moved to a random
    location.
    """
    n = seq.size
    kmax = int(np.argmax(conv))
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idxs = np.arange(lo, hi + 1, dtype=np.int64)
    js = kmax - idxs
    contrib = seq[idxs] * seq[js]
    pos = np.argmax(contrib)
    i = idxs[pos]
    j = js[pos]

    # fraction to move (tuned range)
    frac = rng.uniform(0.07, 0.18)

    if i == j:
        delta = seq[i] * frac
        seq[i] -= delta
        moved = delta
    else:
        delta_i = seq[i] * frac
        delta_j = seq[j] * frac
        seq[i] -= delta_i
        seq[j] -= delta_j
