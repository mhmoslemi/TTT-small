def gradient_smooth(a, alpha):
    # a: np.ndarray shape (n,)
    n = a.shape[0]
    conv = _conv_fft(a)  # length 2n-1
    # compute softmax weights
    # Use stable exponent: shift by max
    maxc = conv.max()
    # weight = exp(alpha*(conv - maxc))
    w = np.exp(alpha * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0: w = np.ones_like(w) / w.size
    else: w = w / w_sum
    # compute gradient = 2 * (w * a_rev) conv
    a_rev = a[::-1]
    L = 1 << ((3*n - 2).bit_length())
    fw = np.fft.rfft(w, n=L)
    fa = np.fft.rfft(a_rev, n=L)
    g_full = np.fft.irfft(fw * fa, n=L)
    # slice indices: we need positions corresponding to i+ (n-1) offset.
    start = n - 1
    g = 2.0 * g_full[start:start + n]
    return g

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (sum == 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Full linear convolution via FFT (real inputs)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv

# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (sum of a is 1)
# ------------------------------------------------------------
def _score_fast(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_fft(a)
    return 2.0 * n * conv.max()

# ------------------------------------------------------------
# Gradient of (smoothed) max‑convolution using all active indices
# ------------------------------------------------------------
def _active_gradient(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """gradient of the max‑convolution term averaged over all indices where mask==True"""
    n = a.shape[0]
    a_rev = a[::-1]
    L = 1 << ((3 * n - 2).bit_length())
    fw = np.fft.rfft(mask.astype(np.float64), n=L)
    fa = np.fft.rfft(a_rev, n=L)
    g_full = np.fft.irfft(fw * fa, n=L)
    start = n - 1
    slice_ = g_full[start:start + n]
    cnt = mask.sum()
    if cnt == 0:
        return np.zeros_like(a)
    return 2.0 * slice_ / cnt
