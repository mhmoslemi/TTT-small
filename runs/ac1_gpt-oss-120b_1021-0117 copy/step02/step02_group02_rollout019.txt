import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i >= 0, sum_i x_i = 1}
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Starts from several simple simplex‑feasible seeds (uniform,

    then performs a projected (momentum) sub‑gradient descent that
    minimises the maximal autocorrelation value.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 2.  Determine length

    try:

    # ------------------------------------------------------------------

        if sc < best_score:

            best = a0.copy()
    a = best.copy()                     # current iterate

    # ------------------------------------------------------------------
    # 5.  Pre‑compute FFT length (next power of two)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1

        if top_k >= conv_len:

    # ------------------------------------------------------------------
