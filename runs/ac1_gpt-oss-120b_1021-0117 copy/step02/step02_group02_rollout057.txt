import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, v>=0)."""
    if v.ndim != 1:
        raise ValueError
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_conv_fft(a: np.ndarray, fft_len: int) -> float:
    """Fast (FFT) evaluation of max(a*a) where * is linear convolution."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    # length of linear convolution = 2*len(a)-1, excess tail is zero‑padded
    return float(conv[:2 * a.size - 1].max())

    """

    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 1.  Determine length and initialise a feasible point (uniform)
    # ------------------------------------------------------------------
    try:

    # uniform simplex point (sum = 1, all entries >=0)
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # pre‑compute FFT length for the current size
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    it = 0
