def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math, random
    from scipy.signal import correlate

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0., budget_s - 5.)

    # initial a
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        a = np.sqrt(np.arange(1, n+1)) # placeholder?
    # Normalize to sum=1 and project onto simplex for safety
    a = a / a.sum()
    a = _project_simplex(a)

    n = a.shape[0]
    # Precompute FFT length
    Lfft = 1 << ((2*n - 1).bit_length())
    best = a.copy()
    best_score = 2.0 * n * _max_convolution(a, Lfft)  # Using existing helper
    # Optionally define _max_convolution similar using FFT for speed.

    # Parameters
    step0 = 0.5
    beta = 30.0  # smoothing parameter
    momentum = np.zeros_like(a)
    mu = 0.9
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1

        # compute convolution via FFT
        fv = np.fft.rfft(a, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2*n-1]
        M = float(conv.max())
        cur_score = 2.0 * n * M

        # smooth weight vector
        diff = conv - M
        w = np.exp(beta * diff)  # max is 1

        # cross-correlation of w with a (valid mode)
        # using scipy.signal.correlate (valid) gives length n
        cross = correlate(w, a, mode='valid')
        # cross shape (n,)
        grad = 2.0 * cross - w[2*np.arange(n)] * a

        norm = np.linalg.norm(grad)
        if norm > 0:
            # Nesterov-like momentum
            momentum = mu * momentum + grad / norm
            eta = step0 / np.sqrt(iteration)
            cand = a - eta * momentum
        else:
            cand = a.copy()
            momentum = np.zeros_like(a)

        # Project onto simplex
        cand = _project_simplex(cand)

        # Random mass transfer small probability
        if rng.random() < 0.01:
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.2
                delta = a[i] * frac
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)

        # Evaluate candidate
        fv_c = np.fft.rfft(cand, n=Lfft)
        conv_c = np.fft.irfft(fv_c * fv_c, n=Lfft)[:2*n-1]
        M_c = float(conv_c.max())
        cand_score = 2.0 * n * M_c

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            # early exit if good enough
            if best_score < 1.5028:
                break

        # accept improvement or occasional uphill
        if cand_score < cur_score or rng.random() < 0.005:
            a = cand

    # return best as list
    return [float(x) for x in best]

import time
import numpy as np
from scipy.signal import correlate

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# max of the (full) convolution a*a using FFT (linear time)
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty sequence of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    # leave a few seconds margin for finalisation
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------
    # initialise – use the provided parent if present, else a short sqrt series
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()

    a = _project_simplex(a)

    n = a.shape[0]
    Lfft = 1 << ((2 * n - 1).bit_length())   # next power of two for FFT
    best = a.copy()

    # --------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------
    step0 = 0.6          # initial step magnitude
    beta = 18.0          # smoothing strength for the log‑sum‑exp approximation
    mu = 0.9             # momentum decay
    momentum = np.zeros_like(a)
    it = 0

    # --------------------------------------------------------
    # main iterative improvement loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current full autocorrelation via FFT
        fv = np.fft.rfft(a, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----------------------------------------------------
        # smooth (log‑sum‑exp) weighting of all lags
        # ----------------------------------------------------

        cross = correlate(w, a, mode="valid")   # sum_i w[i+j] * a[i] , j=0..n-1

        grad = 2.0 * cross - w[2 * np.arange(n)] * a
