            return np.inf
        return 2.0 * n * max_b / (s * s)

    best_score = _score(best)

    if parent is not None:
        p_score = _score(parent)
        if p_score < best_score:
            best = parent.copy()
            best_score = p_score

    # ----------------------------------------------------------------------
    # Main optimisation loop (gradient descent + occasional random moves)
    # ----------------------------------------------------------------------
    current = best.copy()
    current_score = best_score

    # step‑size schedule parameters
    min_step_scale = 1e-4          # relative to max(current) at the end
    max_step_scale = 0.30          # relative to max(current) at the start

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----- gradient of the active convolution entry ---------------------------------
        cur_conv = _conv(current)
        kmax = int(np.argmax(cur_conv))

        grad = np.zeros_like(current)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        if lo <= hi:
            i_range = np.arange(lo, hi + 1, dtype=np.int64)
            j_range = kmax - i_range
            # ∂(max_conv)/∂a_i = 2 * a_{kmax-i} for all contributing i
            grad[i_range] = 2.0 * current[j_range]

        # direction that (approximately) reduces the maximal convolution,
        # projected onto the hyperplane sum(direction)=0
        direction = -grad
        direction += grad.mean()          # makes the sum zero

        # ----- adaptive step size ---------------------------------------------------------
        elapsed = time.monotonic() - start
        total = deadline - start
        progress = elapsed / total
        step_scale = max_step_scale * (1.0 - progress) + min_step_scale * progress
        step_base = step_scale * max(current.max(), 1e-12)

        improved = False
        for factor in (1.0, 0.5, 0.25):   # a few trial step lengths
            alpha = step_base * factor
            if alpha <= 0.0:
                continue
            cand = current + alpha * direction
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_score = _score(cand)
            if cand_score < current_score - 1e-12:
                current = cand
                current_score = cand_score
                improved = True
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
                break

        # ----- if gradient step fails, try a cheap stochastic move -----------------------
        if not improved:
            op = rng.integers(0, 5)
            cand = current.copy()
            if op == 0:                                    # block scaling
                width = rng.integers(1, max(2, n // 30))
                start_idx = rng.integers(0, n - width + 1)
                factor = np.exp(rng.normal(0.0, 0.30))
                cand[start_idx:start_idx + width] *= factor
            elif op == 1:                                  # single index scaling
                idx = rng.integers(0, n)
                factor = np.exp(rng.normal(0.0, 0.50))
                cand[idx] *= factor
            elif op == 2:                                  # smooth two neighbours
                i = rng.integers(0, n - 1)
                avg = (cand[i] + cand[i + 1]) * 0.5
                cand[i] = avg
                cand[i + 1] = avg
            elif op == 3:                                  # early‑to‑late mass transfer
                early = max(1, n // 10)
                i = rng.integers(0, early)
                j = rng.integers(n - early, n)
                frac = rng.uniform(0.02, 0.15)
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta
            else:                                          # additive Gaussian noise
                noise = rng.normal(0.0, 0.05 * cand.mean(), size=n)
                cand += noise
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_score = _score(cand)
            if cand_score < current_score:
                current = cand
                current_score = cand_score
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score

    # Return a plain Python list – the evaluator expects that form.
    return [float(x) for x in best]
```
