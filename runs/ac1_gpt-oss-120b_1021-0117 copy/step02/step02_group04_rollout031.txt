Implementation:

def simplex_projection(v):
    # project onto simplex
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n‑1 (the classic binomial step)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, sum x=1}."""
    # Duchi et al. (2008) – O(n log n) but fast enough for our sizes.
