Define fast_convolve(x, y):
    n = len(x)
    m = len(y)
    L = 1 << ((n+m-1).bit_length())
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[:n+m-1]
    return conv

    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # get a0
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
    else:
        n = 1024
        a = np.full(n, 1.0/n)  # uniform

    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.size

    # init log variables
    x = np.log(a + 1e-12)   # ensure positivity

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    t = 0

    # beta for smoothing
    beta_s = 1.0
    beta_max = 200.0
    beta_inc = 1.02  # multiplicative per iteration

    # best tracking
    b_full = _conv_fft(a)   # conv
    best_score = 2.0 * n * b_full.max()
    best_a = a.copy()
    target = 1.50274349

    # functions definitions:
    def fast_convolve(x, y):
        L = 1 << ((len(x) + len(y) - 1).bit_length())
        fx = np.fft.rfft(x, n=L)
        fy = np.fft.rfft(y, n=L)
        conv = np.fft.irfft(fx * fy, n=L)[:len(x)+len(y)-1]
        return conv

    def conv_self(a):
        # returns full convolution a*a
        # reuse fast_convolve maybe.
        L = 1 << ((2 * len(a) - 1).bit_length())
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * len(a) - 1]
        return conv

    # gradient step loop
    while time.monotonic() < deadline:
        t += 1
        # compute a from x
        a = np.exp(x)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / a.size)
            a_sum = 1.0
        a /= a_sum

        # compute full conv and score
        b = conv_self(a)   # returns length 2n-1
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if best_score < target:
                break

        # compute softmax weights for smoothing
        # compute exponentiated
        # For large beta may overflow, use stable method:
        beta_s = min(beta_max, beta_s * beta_inc)  # increase gradually
        # compute w = softmax(beta_s * b)
        # for stability subtract max:
        exp_input = beta_s * b
        max_exp = exp_input.max()
        w = np.exp(exp_input - max_exp)
        w_sum = w.sum()
        w /= w_sum

        # compute grad w.r.t a using convolution with reversed a
        a_rev = a[::-1]
        conv_grad = fast_convolve(w, a_rev)
        # slice:
        grad_a = 2.0 * conv_grad[n-1 : n-1 + n]

        # compute grad_x
        G = np.dot(a, grad_a)
        grad_x = a * (grad_a - G)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if t % 500 == 0 and rng.random() < 0.2:
            jitter = rng.normal(scale=1e-3, size=n)
            x += jitter

        # maybe random perturbation of a (via x) to escape

            # randomly swap mass between indices
            i, j = rng.integers(0, n, size=2)

            delta = a[i] * rng.random() * 0.2
            # modify x? Hard. Instead directly modify a then convert to x
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # renorm via log
            x = np.log(a_new + 1e-12)

        # continue

    # after gradient loop, refine with subgradient iteration
    # define subgradient update
    cur = best_a.copy()
    cur_sc = best_score
    # Step size schedule
    alpha0 = 0.5
    no_improve = 0
    # compute new deadline left
    while time.monotonic() < deadline - 0.05:
        # compute convolution
        conv_full = conv_self(cur)
        k = int(np.argmax(conv_full))
        # compute subgradient g
        g = np.zeros_like(cur)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) for non‑negative vectors
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    # work on a copy to avoid side‑effects
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Full linear convolution via FFT (real‑valued, matrix‑free)
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a single FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


def _fast_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[: n + m - 1]
    return conv
