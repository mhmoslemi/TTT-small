def propose_candidate(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)  # a bit earlier to be safe

    # get parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise ValueError
        n = int(parent_arr.size)
        base = parent_arr.copy()
    except Exception:
        n = 256  # fallback size
        base = np.empty(n, dtype=np.float64)
        # binomial
        base[0] = 1.0
        for i in range(1, n):
            base[i] = base[i-1] * (2*i-1) / (2*i)
    # ensure nonnegative
    base = np.maximum(base, 0.0)
    # Normalize sum to 1 (or sqrt(2n))
    sum_base = base.sum()
    if sum_base < 1e-12:
        base = np.full(n, 1.0 / n)
        sum_base = 1.0
    # scale to something; we can keep sum = 1.0
    cur = base / sum_base
    best = cur.copy()
    best_score = np.inf

    L = 1 << ((2*n - 2).bit_length())

    # record global best; ensure fallback after any error
    # store sum_target maybe 1.0.

    temp_decay = 0.9997
    step_lr = 0.2
    lr_decay = 0.9999
    # allow jitter std
    jitter_std = 0.02  # relative to current values magnitude
    # main loop
    while time.monotonic() < deadline:
        # compute conv
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        max_b = float(conv.max())
        k_max = int(conv.argmax())
        sum_cur = float(cur.sum())
        cur_score = 2.0 * n * max_b / (sum_cur**2)

        move_type = rng.random()
        cand = cur.copy()
        # gradient-based block move
        if move_type < 0.35:
            # compute mirror (gradient)
            idx = np.arange(n)
            j = k_max - idx
            mask = (j >= 0) & (j < n)
            mirror = np.zeros_like(cur)
            mirror[mask] = cur[j[mask]]
            # scaling factor C as earlier
            C = 4.0 * n / (sum_cur * sum_cur) if sum_cur != 0 else 0.0
            t = max_b / sum_cur if sum_cur != 0 else 0.0
            delta = step_lr * C * (t - mirror)
            cand = cur + delta
            np.clip(cand, 0.0, 1000.0, out=cand)
        elif move_type < 0.65:

            # compute mirror as above
            idx = np.arange(n)

            mirror = np.zeros_like(cur)
            mirror[mask] = cur[j[mask]]
            product = cur * mirror

            i_max = int(np.argmax(product))
            # find a low-value index j (random among bottom 10%)
            low_thresh = np.percentile(cur, 10)
            low_candidates = np.where(cur <= low_thresh)[0]
            if low_candidates.size == 0:
                j_target = rng.integers(0, n)
            else:

            # transfer fraction up to 30% of source mass

            delta = cur[i_max] * frac
            cand[i_max] -= delta
            cand[j_target] += delta
            np.clip(cand, 0.0, 1000.0, out=cand)
        elif move_type < 0.85:

            cand[start:start+width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)
        else:
            # jitter move
            sigma = jitter_std * (cur.mean() if cur.mean() > 0 else 1.0)
            cand += rng.normal(0.0, sigma, size=n)
            np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate candidate
        cand_score = _ac1_score_fft(cand, n, L)

        # update global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            # if beat target, break
            if best_score < 1.50274349:
                break

        # acceptance criterion (simulated annealing)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability based on temperature
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            # Optionally update step_lr? Keep same.

        temperature *= temp_decay
        step_lr = max(step_lr * lr_decay, 1e-6)

        # occasional restart to best
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 5.0
            step_lr = 0.2

    # finalize: ensure all values non-negative and <=1000; also scaling sum to not be weird but not needed.
    best = np.clip(best, 0.0, 1000.0)
    # return as ordinary list of Python floats
    return [float(x) for x in best]
