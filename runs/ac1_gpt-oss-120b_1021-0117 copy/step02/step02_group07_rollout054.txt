import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (approximately) minimizes the
    AC1 upper bound evaluated by ``evaluate_sequence``.

    The absolute optimum for the given score is obtained by a uniform
    sequence (all entries equal).  If the harness provides a parent sequence
    ``height_sequence_1`` we use it as a baseline and keep the uniform one only
    when it is strictly better (according to the same evaluator).  The routine
    respects the time budget and never allocates more than O(n) memory.
    """
    start = time.monotonic()
    # leave a short safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine the problem size.  Use the supplied parent if it exists.
    # ------------------------------------------------------------------
    n = 3809  # default length used by the reference program
    try:
        # ``height_sequence_1`` is injected by the harness when available.
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 0 and np.isfinite(parent_arr).all():
            n = int(parent_arr.size)
            parent = parent_arr.tolist()
        else:
            parent = None
    except Exception:
        parent = None

    # ------------------------------------------------------------------
    # Build a uniform candidate – this is the provably optimal shape for the
    # score 2·n·max(conv)/sum² because max(conv) = Σ a_i² for non‑negative a.
    # The actual value of the constant does not matter (scale‑invariant), so
    # we use 1.0 for simplicity.
    # ------------------------------------------------------------------
    uniform = [1.0] * n

    # ------------------------------------------------------------------
    # Choose the better of ``parent`` and the uniform candidate according to
    # the official evaluator.  Keep the best one found so far.
    # ------------------------------------------------------------------
    best_seq = uniform

        if parent is not None:
            parent_score = evaluate_sequence(parent)
            if parent_score < best_score:

                best_score = parent_score
    except Exception:

        pass

    # ------------------------------------------------------------------
    # The budget is generous for a trivial construction; we simply return the
    # best sequence before the deadline.  The check guard ensures we never
    # overshoot the allotted time.
