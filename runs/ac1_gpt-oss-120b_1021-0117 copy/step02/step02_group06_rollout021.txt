def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, n: int, fft_len: int) -> np.ndarray:
    """Full a*a convolution (length 2*n-1) via real FFT."""
    A = np.fft.rfft(a, fft_len)
    full = np.fft.irfft(A * A, fft_len)
    return full[: 2 * n - 1]


def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style coefficients w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)

    alpha_update_every = 400  # iterations
    iter_no_improve = 0
    max_no_improve = 3000

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # FFT of current vector
        A = np.fft.rfft(a, fft_len)

        # self‑convolution and max entry
        conv_full = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]
        max_b = float(conv_full.max())
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            iter_no_improve = 0
            if best_score <= target_score:
                break
        else:
            iter_no_improve += 1

        # ----- smooth gradient (log‑sum‑exp) -----
        shifted = conv_full - max_b                 # max becomes 0 (non‑positive)
        w = np.exp(alpha * shifted)                 # weights ∈ (0,1]
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # correlation a ⋆ w gives gradient /2
        W = np.fft.rfft(w, fft_len)
        corr = np.fft.irfft(A * np.conj(W), fft_len)  # linear correlation (length fft_len)
        grad = 2.0 * corr[:n]                         # gradient of smoothed objective

        # projected gradient step
        step = eta0 / np.sqrt(it)
        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate)

        # occasional jitter to escape plateaus
        if rng.random() < 0.015:
            jitter = np.exp(rng.normal(0.0, 0.018, size=n))
            a_candidate = a_candidate * jitter
            a_candidate = _project_onto_simplex(a_candidate)

        a = a_candidate

        # update smoothing parameter periodically
        if it % alpha_update_every == 0 and alpha < max_alpha:
            alpha = min(alpha * alpha_factor, max_alpha)

        # restart if stagnating for long
        if iter_no_improve >= max_no_improve:
            pert = np.exp(rng.normal(0.0, 0.03, size=n))
            a = best_vec * pert
            a = _project_onto_simplex(a)
            iter_no_improve = 0

    # ---- sub‑gradient refinement on the true max ----
    a = best_vec.copy()
    it = 0
    iter_no_improve = 0
    max_no_improve = 2500
    eta_sub = 0.25

    while time.monotonic() < deadline:
        it += 1
        A = np.fft.rfft(a, fft_len)
        conv_full = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]
        max_b = float(conv_full.max())
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            iter_no_improve = 0
            if best_score <= target_score:
                break
        else:
            iter_no_improve += 1

        # active index of the max entry
        k = int(conv_full.argmax())

        # sub‑gradient for active term
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            g[idx] = 2.0 * a[k - idx]

        step = eta_sub / np.sqrt(it)
        a_candidate = a - step * g
        a_candidate = _project_onto_simplex(a_candidate)

        if rng.random() < 0.01:
            jitter = np.exp(rng.normal(0.0, 0.016, size=n))
            a_candidate = a_candidate * jitter
            a_candidate = _project_onto_simplex(a_candidate)

        a = a_candidate

        if iter_no_improve >= max_no_improve:
            # perturb around best and continue
            pert = np.exp(rng.normal(0.0, 0.025, size=n))
            a = best_vec * pert
            a = _project_onto_simplex(a)
            iter_no_improve = 0

    # Return the best candidate found (as plain Python list)
    return best_vec.tolist()
