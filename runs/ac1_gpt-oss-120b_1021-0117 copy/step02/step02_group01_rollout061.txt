import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    # Euclidean projection onto the probability simplex (L1 norm = 1)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    # convolution a*a (full) using FFT of length L (power of two)
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(a) - 1]
    return conv

def _gradient_multi(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in idxs:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # range of j where both a[j] and a[k-j] are defined
        j = np.arange(lo, hi + 1)
        # contribution: 2 * a[k - j]
        g[lo:hi + 1] += 2.0 * a[k - j]
    return g
