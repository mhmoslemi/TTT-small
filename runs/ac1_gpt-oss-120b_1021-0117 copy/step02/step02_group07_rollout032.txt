import numpy as np, time, math

def _initial_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i-1) / (2*i)
    return a

def _conv_fft(seq, L):
    A = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(A * A, n=L)
    n = seq.shape[0]
    return conv[:2*n-1]

def _softmax_weights(conv, lam):
    scaled = lam * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum

def _gradient(a, w, L):
    A_fft = np.fft.rfft(a, n=L)
    W_fft = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), n=L)
    n = a.shape[0]
    return 2.0 * corr[:n]

def _score(seq, conv):
    s = seq.sum()
    if s <= 0.0:
        return float('inf')
    n = seq.shape[0]
    return float(2.0 * n * conv.max() / (s * s))

                conv = conv2

                best_score = score2
                best_seq = cand2.copy()
                best_conv = conv2.copy()
                if best_score <= target:

        # parameter update

        step = max(min_step, step * step_decay)

    # Final scaling (optional) to typical magnitude
    out = best_seq * math.sqrt(2.0 * n)
    out = np.clip(out, 0.0, 1000.0)
    return [float(x) for x in out]

import numpy as np
import time
import math

def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1, positive decreasing."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_fft(seq: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of seq with itself using FFT, length 2*len(seq)-1."""
    A = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(A * A, n=L)
    n = seq.shape[0]
    return conv[: 2 * n - 1]


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Softmax of lam*conv (numerically stable, returns a distribution)."""
    scaled = lam * conv
    m = np.max(scaled)
    w = np.exp(scaled - m)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _gradient(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """Gradient of the (smoothed) max‑convolution: 2 * correlation(a,w)."""
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(A * np.conj(W), n=L)
    n = a.shape[0]
    return 2.0 * corr[:n]


def _score(seq: np.ndarray, conv: np.ndarray) -> float:
    """Exact AC1 upper‑bound score."""
    s = seq.sum()
    if s <= 0.0:
        return float("inf")
    n = seq.shape[0]
    return float(2.0 * n * conv.max() / (s * s))

    def _norm(v):
