def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)

    # time

    # init a

        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        parent = np.clip(parent, 0., 1000.)
        a = parent.copy()
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)

    n = a.size
    a = np.maximum(a, 0.)
    total = a.sum()
    if total < 0.01:
        a = np.full(n, 1./n, dtype=np.float64)
    else:
        a = a/total

    # maintain best
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # also consider uniform
    uniform = np.full(n, 1./n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    target_score = 1.50274349
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # parameters
    eta0 = 0.2   # initial step for multiplicative update
    beta = 0.9   # momentum coefficient for exponential moving avg of gradient
    eps = 1e-12
    it = 0
    # store smoothed gradient for momentum if needed (optional)
    m = np.zeros_like(a)

    while time.monotonic() < deadline:
        it += 1
        # compute conv via FFT
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target_score:
                break

        # find active shifts: within epsilon of max
        # compute tolerance
        # relative tolerance
        tol = 1e-12

        # Compute gradient g

            # Determine range
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # i in [i_start, i_end]

        a_sum = a.sum()

import time
import numpy as np

def _active_shifts(conv: np.ndarray, tol: float = 1e-12) -> np.ndarray:
    """Indices where the convolution is (almost) maximal."""
    max_val = conv.max()
    return np.where(conv >= max_val - tol)[0]


def _gradient_from_shifts(a: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max {a*a} for the given active shifts."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in shifts:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1)
        partner = k - i
        g[i] += 2.0 * a[partner]
    return g


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Mirror‑descent search on the simplex for a low‑autocorrelation step
    function.  Starts from ``height_sequence_1`` if present, otherwise from
    uniform.  Returns the best sequence seen before the time limit.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Time management
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # 2. Initialise a (non‑negative, summed to 1)
    # --------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        a = parent.copy()
    except Exception:
        # fallback: uniform of moderate length
        n = 256
        a = np.ones(n, dtype=np.float64)

    n = int(a.size)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                      # ℓ₁‑normalised

    # --------------------------------------------------------------
    # 3. Keep the best solution seen so far
    # --------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # cheap uniform alternative
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    target_score = 1.50274349

    # convolution parameters (FFT)
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # --------------------------------------------------------------
    # 4. Optimisation loop (mirror descent with momentum)
    # --------------------------------------------------------------
    it = 0
    eta0 = 0.4               # initial step magnitude (multiplicative)
    beta = 0.9               # momentum smoothing
    m = np.zeros_like(a)    # momentum buffer
    no_improve = 0
    max_no_improve = 2000   # trigger a small restart
    eps = 1e-15

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution (FFT) -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ----- keep best -----
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            if best_score <= target_score:
                break
        else:
            no_improve += 1

        # ----- gradient of the active constraints -----
        active = _active_shifts(conv)
        g = _gradient_from_shifts(a, active)

        # momentum update
        m = beta * m + (1.0 - beta) * g

        # ----- mirror‑descent step (multiplicative) -----
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * m)
        a = np.maximum(a, eps)            # avoid exact zeros
        a = a / a.sum()                   # simplex projection

        # ----- occasional jitter (log‑normal noise) -----
        if it % 40 == 0:
            jitter = np.exp(rng.normal(0.0, 0.006, size=n))
            a = a * jitter
            a = a / a.sum()

        # ----- restart if stuck -----
        if no_improve >= max_no_improve:
            # blend the best so far with fresh noise
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.01, size=n))
            a = a * jitter
            a = a / a.sum()
            m.fill(0.0)
            no_improve = 0

        # leave a little margin for the final return
        if time.monotonic() + 0.5 >= deadline:
            break

    # --------------------------------------------------------------
    # 5. Return the best sequence as plain Python floats
    # --------------------------------------------------------------
    return [float(v) for v in best]
