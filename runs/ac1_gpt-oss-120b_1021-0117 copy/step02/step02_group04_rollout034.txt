    n = len(f)
    K = 2*n-1
    # Build A
    A = np.empty((K, n), dtype=np.float64)
    # Use broadcasting: each column j: put f into rows j:j+n

        A[j:j+n, j] = f

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
        base = parent.copy()
    except Exception:
        n = 256
        base = _gen_binomial(n)
    L = _next_pow2(2*n - 1)

    # initial population
    pop = []
    # add base
    base_score = _ac1_score_fft(base, n, L)
    pop.append([base.copy(), base_score])
    # add binomial
    bin_seq = _gen_binomial(n)
    bin_score = _ac1_score_fft(bin_seq, n, L)
    pop.append([bin_seq, bin_score])
    # add few random perturbed versions
    for _ in range(3):
        cand = base.copy()
        # random perturb: add Gaussian noise scaled by small factor (~0.1*mean)
        scale = np.mean(cand) * 0.1 + 1e-6
        cand += rng.normal(0, scale, size=n)
        np.clip(cand, 0, 1000, out=cand)
        s = _ac1_score_fft(cand, n, L)
        pop.append([cand, s])
    # ensure population size is 6
    pop_size = len(pop)

    # sort population
    pop.sort(key=lambda x: x[1])
    best_arr = pop[0][0].copy()
    best_score = pop[0][1]

    # mutation parameters
    width_min = max(1, n // 200)
    width_max = max(2, n // 12)
    sigma_factor = 0.12
    frac_max = 0.4

    # main loop
    iter_count = 0
    while time.monotonic() < deadline:
        iter_count += 1

        # select two parents (elitist crossover)
        # choose from top 3
        i, j = rng.choice(min(3, pop_size), size=2, replace=False)
        a = pop[i][0]
        b = pop[j][0]
        child = (a + b) * 0.5

        # mutation
        move_type = rng.random()
        if move_type < 0.5:
            # block scaling
            width = int(rng.integers(width_min, width_max))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, sigma_factor))
            child[start:start+width] *= factor
        elif move_type < 0.8:
            # mass transfer
            i_idx = int(rng.integers(0, n))
            j_idx = int(rng.integers(0, n))
            if i_idx != j_idx:
                frac = rng.random() * frac_max
                delta = child[i_idx] * frac
                child[i_idx] -= delta
                child[j_idx] += delta
        else:
            # edge injection
            if rng.random() < 0.5:
                # left edge
                width = int(rng.integers(1, width_min+1))
                delta = rng.random() * frac_max * child[width-1]  # small
                child[:width] += delta
            else:
                # right edge
                width = int(rng.integers(1, width_min+1))
                delta = rng.random() * frac_max * child[-width]
                child[-width:] += delta

        np.clip(child, 0.0, 1000.0, out=child)

        # evaluate
        score = _ac1_score_fft(child, n, L)

        # keep top individuals
        # if better than worst, replace
        if score < pop[-1][1]:
            pop[-1] = [child, score]
            pop.sort(key=lambda x: x[1])
            if score < best_score:

def _lp_step(f, M, kmax, w, rng):
    n = f.shape[0]
    k_start = max(0, kmax - w)
    k_end = min(2*n - 2, kmax + w)
    Ksize = k_end - k_start + 1
    rows_list = []
    cols_list = []
    data_list = []
    for k in range(k_start, k_end+1):
        i0 = max(0, k - (n-1))
        i1 = min(n-1, k)
        # For each i in i0..i1:
        # column j = k - i
        # row idx = k - k_start
        row_idx = k - k_start
        i_vals = np.arange(i0, i1+1, dtype=np.int64)
        j_vals = k - i_vals
        rows_list.append(np.full(i_vals.shape, row_idx, dtype=np.int64))
        cols_list.append(j_vals)
        data_list.append(f[i_vals])
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    data = np.concatenate(data_list)

    prob.solve(solver=cp.ECOS, warm_start=False, max_iters=500, verbose=False)
    if b.value is None:

import math
import time

import numpy as np

# optional linear programming
try:
    import cvxpy as cp
    HAVE_CVX = True
except Exception:  # pragma: no cover
    HAVE_CVX = False


def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n‑1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 score using FFT (matches evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _lp_optimize(f: np.ndarray, M: float, kmax: int, window: int = 200) -> np.ndarray | None:
    """Linear programme:
       maximise sum(b)  s.t.  f * b ≤ M on a window around kmax,
       b ≥ 0, b ≤ 1000.
       Returns the optimal b (numpy array) or None on failure.
    """
    n = f.shape[0]
    half = window // 2
    k_start = max(0, kmax - half)
    k_end = min(2 * n - 2, kmax + half)
    K = k_end - k_start + 1
    # build dense constraint matrix A (K × n)
    A = np.zeros((K, n), dtype=np.float64)
    for i in range(n):
        j_start = max(0, k_start - i)
        j_end = min(n - 1, k_end - i)
        if j_start > j_end:
            continue
        rows = np.arange(i + j_start, i + j_end + 1) - k_start
        A[rows, j_start : j_end + 1] = f[i]
    # solve LP
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M,
                   b <= 1000.0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=False, max_iters=500, verbose=False)
    except Exception:  # pragma: no cover
        return None
    if b.value is None:
        return None
    return np.asarray(b.value, dtype=np.float64)

    # ---- initialise -------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        base = parent.astype(np.float64)
    except Exception:  # no parent supplied
        n = 256
        base = _gen_binomial(n)

    L = _next_pow2(2 * n - 1)

    cur = base.copy()
    cur_score = _ac1_score_fft(cur, n, L)
    best = cur.copy()
    best_score = cur_score

    # ---- optimisation parameters ------------------------------------------
    temperature = 1.0
    temp_decay = 0.9997
    step_lr = 0.2
    lr_decay = 0.9999
    mix_t = 0.2                 # mixing proportion when using LP step
    lp_interval = 500          # how often we attempt an LP projection
