import numpy as np
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Analytic sqrt‑series (a*a = 1 + … + x^{2n‑2})
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ----------------------------------------------------------------------
# Convolution max via FFT (full convolution, return maximum value)
# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray) -> float:
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return float(conv.max())

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()

        a = np.maximum(a, 0.0)
    else:
        n = 1024
        a = _sqrt_series(n)

    n = a.size
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # LP parameters
    # ------------------------------------------------------------------
    t = 0.3            # mixing weight
    t_decay = 0.93
    min_t = 0.02

    # warm‑start for the LP
    prev_sol = None

    # ------------------------------------------------------------------

    while time.monotonic() < deadline:
        # current bound M = max convolution of a with itself
        M = _conv_max(a)

        # ---- build sparse convolution matrix A, where (A @ g)[k] = (a * g)[k] ----
