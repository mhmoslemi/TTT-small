import time
import numpy as np

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a list of non‑negative coefficients minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)  # a few seconds safety

    # ------------------------------------------------------------------
    # Determine problem size and possible parent sequence
    # ------------------------------------------------------------------
    n = 3809
    parent = None
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
                parent = np.clip(arr, 0.0, 1000.0)
                n = parent.size
        except Exception:
            parent = None

    # ------------------------------------------------------------------
    # Initialise best candidate
    # ------------------------------------------------------------------
    if parent is not None:
        best = parent.copy()
    else:
        best = _initial_binomial(n)

    # ------------------------------------------------------------------
    # Helper convolution utilities (FFT based, O(n log n))
    # ------------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    def _conv(seq: np.ndarray) -> np.ndarray:
        f = np.fft.rfft(seq, n=fft_len)
        return np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]

    def _score(seq: np.ndarray):
        conv = _conv(seq)
        max_b = np.max(conv)
        s = np.sum(seq)
        if s < 0.01:
            return np.inf, conv, max_b
        return 2.0 * n * max_b / (s * s), conv, max_b

    best_score, best_conv, best_max = _score(best)

    # possibly replace with a pure binomial seed if it is better
    if parent is not None:

            best_score = bin_score

            best_max = np.max(best_conv)

    current_score = best_score

        kmax = int(np.argmax(cur_conv))

        # ----- gradient of the maximal autocorrelation -----
        lo = max(0, kmax - (n - 1))

        cand = None

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
