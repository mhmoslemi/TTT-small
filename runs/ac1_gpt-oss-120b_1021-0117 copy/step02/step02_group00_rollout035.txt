import time
import numpy as np

# ------------------------------------------------------------
# Simple simplex projection (clip‑then‑renormalise)
# ------------------------------------------------------------
def _simplex_normalize(v: np.ndarray) -> np.ndarray:
    """Replace negatives by 0 and scale so that the vector sums to 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


# ------------------------------------------------------------
# Fast full linear convolution via FFT (matrix‑free)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[: 2 * a.shape[0] - 1]


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 upper‑bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # initialise from the provided parent if present,
    # otherwise use a uniform fallback.
    # --------------------------------------------------------

        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 256

    cur = _simplex_normalize(cur)
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two for FFT

    # current convolution and score (sum is 1 after normalisation)
    conv_cur = _conv_fft(cur, L)
    cur_sc = 2.0 * n * conv_cur.max()
    best_vec = cur.copy()
    best_sc = cur_sc

    # momentum variables
    velocity = np.zeros_like(cur)
    mu = 0.9                     # momentum coefficient
    alpha0 = 0.5                 # initial step length
    no_improve = 0
    it = 0
