    target_score = 1.50274349

    # Precompute FFT length for autocorr (2n-1)

    # optimizer parameters
    it = 0
    eta0 = 0.5
    beta = 10.0
    momentum = np.zeros_like(a)
    mom_decay = 0.9

    while time.monotonic() < deadline:

        # autocorrelation via FFT
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        max_conv = np.max(conv)
        cur_score = 2.0 * n * max_conv / (np.sum(a) ** 2)
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target_score:
                break

        # softmax weights for gradients
        # use stability trick
        shift = conv - max_conv  # <=0

        w = exp_vals / np.sum(exp_vals)

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Simple projected‑gradient search for the AC1 upper bound.
    Works with the optional parent ``height_sequence_1`` if present,
    otherwise falls back to a uniform start. The routine respects the
    time budget and returns the best sequence encountered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1.  Initialise a candidate sequence
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        a = np.clip(parent, 0.0, 1000.0)
    except Exception:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                     # enforce ℓ₁‑norm = 1

    # ------------------------------------------------------------------
    # 2.  Record the best solution seen so far
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # compare with a pure uniform vector as a cheap baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        a = uniform.copy()

    target_score = 1.50274349

    # FFT parameters (linear convolution via zero‑padding)
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())          # next power‑of‑two

    # ------------------------------------------------------------------
    # 3.  Optimisation loop (projected sub‑gradient with momentum)
    # ------------------------------------------------------------------
    it = 0
    eta0 = 0.5                       # initial step size
    momentum = np.zeros_like(a)
    mom_decay = 0.9
    while time.monotonic() < deadline:
        it += 1

        # --- convolution via FFT -------------------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        # --- current objective (sum is 1, so denominator drops out) ---------------
        max_conv = np.max(conv)
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target_score:
                break

        # --- sub‑gradient for the active shift -----------------------------------
        kmax = int(np.argmax(conv))
        offset = kmax - (n - 1)                     # shift relative to centre
        g = np.zeros_like(a)

        if offset >= 0:
            idx = np.arange(offset, n)
            partner = kmax - idx
            g[idx] = 2.0 * a[partner]
        else:
            idx = np.arange(0, n + offset)          # offset is negative
            partner = kmax - idx
            g[idx] = 2.0 * a[partner]

        # --- momentum‑aided step -------------------------------------------------
        eta = eta0 / np.sqrt(it)
        momentum = mom_decay * momentum - eta * g
        a_new = a + momentum
        a = _proj_simplex(a_new)

        # --- occasional jitter to escape shallow plateaus -------------------------
        if it % 100 == 0:
            jitter = np.exp(rng.normal(0.0, 0.005, size=n))
            a = _proj_simplex(a * jitter)

    # ------------------------------------------------------------------
    # 4.  Return the best sequence as plain Python floats
    # ------------------------------------------------------------------
    return [float(v) for v in best]
