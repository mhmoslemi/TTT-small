import numpy as np, time, math

def _initial_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def _conv_self(seq, fft_len):
    f = np.fft.rfft(seq, n=fft_len)
    conv_full = np.fft.irfft(f * f, n=fft_len)
    return conv_full[:2*len(seq)-1]

def _score_seq(seq, fft_len):

    max_b = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float('inf'), conv, max_b
    score = 2.0 * len(seq) * max_b / (s * s)
    return float(score), conv, max_b

import numpy as np, time, math

def _initial_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def _conv_self(seq, fft_len):
    """Full self‑convolution of a real sequence using FFT."""
    f = np.fft.rfft(seq, n=fft_len)
    conv_full = np.fft.irfft(f * f, n=fft_len)
    return conv_full[:2*len(seq)-1]

def _score_seq(seq, fft_len):
    conv = _conv_self(seq, fft_len)
    max_b = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float('inf'), conv, max_b
    score = 2.0 * len(seq) * max_b / (s * s)
    return float(score), conv, max_b

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin
    deadline = start + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Determine problem size and optional parent sequence
    # ------------------------------------------------------------------
    n = 3809
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = int(parent.size)
    except Exception:
        parent = None

    # ------------------------------------------------------------------
    # Initialise current sequence
    # ------------------------------------------------------------------
    if parent is not None:
        current = parent.copy()
    else:
        current = _initial_binomial(n)

    np.clip(current, 0.0, 1000.0, out=current)

    if current.sum() < 0.01:
        current = _initial_binomial(n)

    # ------------------------------------------------------------------
    # Pre‑compute FFT length (power of two) and initialise scores
    # ------------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    best_seq = current.copy()
    best_score, best_conv, _ = _score_seq(best_seq, fft_len)
    current_score = best_score

    iteration = 0
    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        elapsed = time.monotonic() - start
        fraction = elapsed / (deadline - start)

        # --------------------------------------------------------------
        # Gradient‑descent move (most of the time)
        # --------------------------------------------------------------
        if rng.random() < 0.78:
            # compute convolution of the current sequence
            conv = _conv_self(current, fft_len)
            kmax = int(np.argmax(conv))

            # gradient w for the active shift kmax
            w = np.zeros_like(current)
            lo = max(0, kmax - (n - 1))
            hi = min(n - 1, kmax)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            w[idx] = current[kmax - idx]

            # zero‑mean descent direction
            mean_w = w.mean()
            d = -w + mean_w

            # Largest feasible step that respects bounds
            neg = d < -1e-12
            if np.any(neg):
                alpha_max = np.min(current[neg] / (-d[neg]))
            else:
                alpha_max = np.inf

            pos = d > 1e-12
            if np.any(pos):
                alpha_max_pos = np.min((1000.0 - current[pos]) / d[pos])
                alpha_max = min(alpha_max, alpha_max_pos)

            # keep step moderate
            alpha_max = min(alpha_max, 1.0)

            # try a few fractions of the feasible step
            best_local_seq = current
            best_local_score = current_score
            for frac in (0.25, 0.5, 0.75, 1.0):

                    best_local_score = cand_score

                current_score = best_local_score
                if best_local_score < best_score:

                    best_score = best_local_score

            cand = current.copy()

                width = rng.integers(1, max(2, n // 25))

            break

    # --------------------------------------------------------------
    return [float(x) for x in best_seq]

import numpy as np, time, math

def _initial_binomial(n):
    """C(2i,i)/4^i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_self(seq, fft_len):
    """Full self‑convolution using FFT (same as np.convolve(seq, seq))."""
    f = np.fft.rfft(seq, n=fft_len)
    conv_full = np.fft.irfft(f * f, n=fft_len)
    return conv_full[: 2 * len(seq) - 1]


def _score_seq(seq, fft_len):
    """Return (score, full convolution, max value)."""
    conv = _conv_self(seq, fft_len)
    max_b = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float("inf"), conv, max_b
    score = 2.0 * len(seq) * max_b / (s * s)
    return float(score), conv, max_b
