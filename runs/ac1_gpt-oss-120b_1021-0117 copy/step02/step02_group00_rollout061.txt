def propose_candidate(seed=42, budget_s=1100, **kwargs):

    start = time.monotonic()
    deadline = start + max(0., float(budget_s)-0.5)

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        a = np.full(n, 1.0/n)  # uniform

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0:
        a[:] = 1.0/a.size
    else:
        a /= total

    best = a.copy()
    best_score = _score(best)   # _score = evaluate directly with sum=1.

    # parameters
    eta0 = 0.5
    it = 0
    no_improve = 0

    # Helper: compute conv via FFT
    def conv(a):
        n = a.shape[0]
        L = 1 << ((2 * n - 1).bit_length())
        fv = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[:2 * n - 1]
        return conv

    while time.monotonic() < deadline:
        it += 1
        c = conv(a)
        k = int(np.argmax(c))
        # subgradient
        n = a.shape[0]
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] = 2.0 * a[k - idx]

def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., float(budget_s) - 0.5)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # enforce non-negativity and normalise sum to 1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a[:] = 1.0 / a.size
    else:
        a /= s

    n = a.size
    best = a.copy()
    best_sc = _score(best)
    cur = best.copy()
    cur_sc = best_sc

    # parameters
    alpha0 = 0.5      # initial step size factor
    beta = 0.9        # momentum decay
    momentum = np.zeros_like(cur)
    it = 0
    no_imp = 0

    while time.monotonic() < deadline:
        it += 1
        # compute convolution and max location
        conv = _conv_fft(cur)
        k = int(np.argmax(conv))

        # subgradient of active max term
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(cur)
        g[idx] = 2.0 * cur[k - idx]

        # update momentum
        momentum = beta * momentum + (1.0 - beta) * g

        # step size diminishing
        step = alpha0 / np.sqrt(it)

        cand = cur - step * momentum
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_imp = 0
        else:
            # occasional Metropolis acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_imp += 1

        # random mass-transfer move when stagnating
        if no_imp > 800 and rng.random() < 0.4:
            i = rng.integers(0, n)
            if cur[i] > 0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc
                        no_imp = 0
                else:
                    # maybe accept with small prob
                    prob = math.exp(-(tmp_sc - cur_sc) / (step + 1e-12))
                    if rng.random() < prob:
                        cur, cur_sc = tmp, tmp_sc
        # jitter restart occasionally

            tmp = best + jitter
            tmp = np.maximum(tmp, 0.0)

            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                    no_imp = 0

        if best_sc < 1.50274349:
            break

    return [float(x) for x in best]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex:
#   {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear convolution via FFT (full length)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Gradient of the smooth max (log‑sum‑exp) objective.
#   w_j ∝ exp(eps * (conv_j - max(conv)))
# ------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, eps: float) -> np.ndarray:
    """Return ∇_a  ( (1/eps) log Σ_j exp(eps*conv_j) )   .
    Uses FFT to compute convolution of w with reversed a."""
    n = a.shape[0]
    # softmax weights
    m = np.max(conv)
    shifted = conv - m                     # ≤ 0
    w = np.exp(eps * shifted)
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution w * a_rev (linear)
    a_rev = a[::-1]
    L = 1 << ((w.size + n - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(W * A_rev, n=L)[: w.size + n - 1]
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ------------------------------------------------------------
# Objective value for a sequence normalised to sum == 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    return 2.0 * n * np.max(_conv_fft(a))
