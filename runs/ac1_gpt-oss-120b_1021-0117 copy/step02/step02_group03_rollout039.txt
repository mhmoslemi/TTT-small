import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (non‑negative, sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = u - css / np.arange(1, n + 1)
    rho_pos = np.where(rho > 0)[0]
    if rho_pos.size == 0:
        theta = 0.0
    else:
        rho_star = rho_pos[-1]
        theta = css[rho_star] / (rho_star + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic sqrt‑series: a*a = 1 + x + … + x^{2n‑2} (truncated)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Max of full convolution a*a using a power‑of‑two FFT length
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Gradient of the k‑th lag of the autocorrelation (a*a)[k]
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty sequence of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        a = _sqrt_series(n).astype(np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / a.sum()
    a = _project_simplex(a)

    n = a.shape[0]
    Lfft = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT length

    # ---- bookkeeping ------------------------------------------------
    cur_M = _max_convolution(a, Lfft)
    cur_score = 2.0 * n * cur_M
    best = a.copy()
    best_score = cur_score

    # step‑size control
    step = 1.0          # initial step length (in direction space)
    min_step = 1e-8
    max_step = 5.0
    decay_factor = 0.5
    grow_factor = 1.1

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # ---- current convolution & gradient at the active lag -----
        fv = np.fft.rfft(a, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = 2.0 * n * M
        k = int(conv.argmax())

        grad = _grad_for_k(a, k)
        norm = np.linalg.norm(grad)
        if norm == 0.0:
            direction = np.zeros_like(a)
        else:
            direction = -grad / norm      # descent direction

        # ---- backtracking line search ---------------------------------
        alpha = step
        improved = False
        for _ in range(6):                # at most 6 reductions
            cand = a + alpha * direction
            cand = _project_simplex(cand)
            cand_M = _max_convolution(cand, Lfft)
            cand_score = 2.0 * n * cand_M
            if cand_score < cur_score:
                improved = True
                break
            alpha *= decay_factor
            if alpha < min_step:
                break

        if improved:
            a = cand
            cur_score = cand_score
            step = min(alpha * grow_factor, max_step)
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            step = max(alpha * decay_factor, min_step)

        # ---- occasional random mass transfer to escape plateaus ------
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.12
                delta = a[i] * frac
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)

        # ---- early stopping ------------------------------------------------
        if best_score < 1.5028:
            break

    return [float(x) for x in best]
