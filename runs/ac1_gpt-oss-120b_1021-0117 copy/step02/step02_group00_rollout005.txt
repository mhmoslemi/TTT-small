def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v>=0)."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    ft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    return conv


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, t: float) -> np.ndarray:
    """
    Gradient of the smoothed max of the auto‑convolution.
    conv is the full convolution a*a (length 2*n-1).
    t is the smoothing parameter (larger => sharper max).
    """
    max_c = conv.max()
    # stable softmax on the negative side
    w = np.exp(t * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)

        w /= w_sum

    L = _next_pow2(a_rev.size + w.size - 1)
    Af = np.fft.rfft(a_rev, n=L)
    Wf = np.fft.rfft(w, n=L)

    n = a.shape[0]
    grad = 2.0 * grad_full[(n - 1) : (n - 1) + n]
