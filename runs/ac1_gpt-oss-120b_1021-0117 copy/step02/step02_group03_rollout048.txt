import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    """ Gradient of the convolution entry at lag k w.r.t. a. """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j = k // 2
        if 0 <= j < n:
            grad[j] -= a[j]
    return grad

def _max_convolution(a: np.ndarray, L: int) -> float:
    """ Maximum value of the linear convolution a*a using FFT. """
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return float(conv.max())

def _score(a: np.ndarray, L: int, n: int) -> float:
    """ 2*n*max_convolution, sum(a) is assumed to be 1. """
    return 2.0 * n * _max_convolution(a, L)
