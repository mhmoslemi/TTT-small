def gradient_smooth(a, b, gamma):
    # b = conv_self(a)
    b_max = b.max()
    # offset for numerical stability
    shift = b - b_max
    # compute exp(shift/gamma)
    exp_vals = np.exp(shift / gamma)
    p = exp_vals / exp_vals.sum()
    # compute gradient
    # a_rev = a[::-1]
    # conv = np.fft.ifft(np.fft.fft(p, L) * np.fft.fft(a_rev, L)).real
    # grad = 2 * conv[n-1: n-1+n] (since conv length = len(p)+n-1 = (2n-1)+n-1 = 3n-2)
    # Actually conv length = len(p) + n - 1 = (2n-1) + n - 1 = 3n-2
    # Index start = n-1; end = n-1 + n = 2n - 1 exclusive.
    # So slice conv[n-1:n-1+n] yields length n.
    L = 1 << ((len(p) + n - 1).bit_length())
    p_fft = np.fft.rfft(p, L)
    a_rev_fft = np.fft.rfft(a[::-1], L)
    conv = np.fft.irfft(p_fft * a_rev_fft, L)
    grad = 2 * conv[n-1:n-1+n]
    return grad

import time
import numpy as np

# ----- helper functions ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution a*a using real FFT (length 2*len(a)-1)."""
    A = np.fft.rfft(a, fft_len)
    B = A * A
    full = np.fft.irfft(B, fft_len)
    n = a.shape[0]
    return full[: 2 * n - 1]
