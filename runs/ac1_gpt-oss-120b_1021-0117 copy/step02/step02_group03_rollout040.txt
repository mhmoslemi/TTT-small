import time
import numpy as np

# ------------------------------------------------------------
# helpers
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    """Gradient of (a*a)[k] with respect to a."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1, dtype=np.int64)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad


def _clip_renorm(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def _sqrt_series(n: int) -> np.ndarray:
    """Analytic sqrt‑series used as a cheap fallback."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ----------------------------------------------------------------
    # initialise – use the provided parent if present, otherwise a fallback
    # ----------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback to a modest size if no parent is supplied
        n_small = 256
        base = _sqrt_series(n_small).astype(np.float64)

    n = base.shape[0]
    base = _clip_renorm(base)

    # ----------------------------------------------------------------
    # preparation
    # ----------------------------------------------------------------
    Lfft = 1 << ((2 * n - 1).bit_length())          # next power of two
    target = 1.50274349

    # helper to compute full convolution, its maximum and the bound
    def _conv_score(vec):
        fv = np.fft.rfft(vec, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        mx = float(conv.max())
        return conv, mx, 2.0 * n * mx

    # ----------------------------------------------------------------
    # start points (parent + a few random Dirichlet vectors)
    # ----------------------------------------------------------------
    starts = [base]
    for _ in range(3):
        rand_vec = rng.random(n)
        starts.append(_clip_renorm(rand_vec))

    best_vec = None
    best_score = float('inf')

    # algorithmic hyper‑parameters
    eta0 = 1.0            # initial step magnitude
    mu = 0.9              # momentum factor
    topk = 3              # number of top convolution lags used for gradient
    rand_move_prob = 0.04
    rand_move_frac = 0.25

    # ----------------------------------------------------------------
    # optimisation loops over each start
    # ----------------------------------------------------------------
    for start in starts:
        a = start.copy()
        velocity = np.zeros_like(a)
        it = 0
        while time.monotonic() < deadline:
            it += 1

            conv, mx, cur_score = _conv_score(a)

            # --- gradient from the largest few convolution entries
            if topk >= conv.size:
                idxs = np.arange(conv.size)
            else:
                idxs = np.argpartition(conv, -topk)[-topk:]
            grad = np.zeros_like(a)
            for k in idxs:
                grad += _grad_for_k(a, int(k))

            # --- momentum‑guided step
            norm = np.linalg.norm(grad)
            if norm > 0.0:
                eta = eta0 / np.sqrt(it)
                velocity = mu * velocity - eta * (grad / norm)
                cand = a + velocity
            else:
                cand = a.copy()

            cand = _clip_renorm(cand)

            # --- evaluate candidate
            _, _, cand_score = _conv_score(cand)

            # keep global best
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
                if best_score < target:
                    break

            # accept if it improves the current point
            if cand_score < cur_score:
                a = cand
                velocity[:] = 0.0

            # occasional random mass transfer to escape plateaus
            if rng.random() < rand_move_prob:
                i = rng.integers(0, n)
                if a[i] > 0.0:
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    frac = rng.random() * rand_move_frac
                    delta = a[i] * frac
                    a[i] -= delta
                    a[j] += delta
                    a = _clip_renorm(a)

            # early termination checks
            if best_score < target or time.monotonic() >= deadline:
                break
        if best_score < target or time.monotonic() >= deadline:
            break

    # ----------------------------------------------------------------
    # fallback if nothing was found (should never happen)
    # ----------------------------------------------------------------
    if best_vec is None:
        best_vec = _clip_renorm(np.ones(n) / n)

    # return as a plain Python list
    return [float(x) for x in best_vec]
