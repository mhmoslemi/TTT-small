def propose_candidate(...):
    import numpy as np, math, time

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        base_vals = parent.astype(np.float64).copy()
    except Exception:
        n = 256
        base_vals = _gen_binomial(n)
    n = int(base_vals.size)

    L = _next_pow2(2*n - 1) # define helper functions

    # initial candidate: random shuffle of base_vals (maybe keep original as alt)
    cur = base_vals.copy()
    rng.shuffle(cur)  # random permutation
    cur_score = _score(cur, n, L)
    best = cur.copy()
    best_score = cur_score

    # also evaluate original order and binomial if they exist; keep best.
    orig_score = _score(base_vals, n, L)
    if orig_score < best_score:
        best = base_vals.copy()
        best_score = orig_score
        cur = best.copy()
        cur_score = best_score

    # Sim annealing
    init_temp = max(best_score * 0.02, 0.001)
    temperature = init_temp
    decay = 0.99995  # per iteration
    it = 0
    check_interval = 128
    while time.monotonic() < deadline:
        # propose swap
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # swap
        cur[i], cur[j] = cur[j], cur[i]
        new_score = _score(cur, n, L)
        # acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability
            prob = math.exp((cur_score - new_score) / temperature) if temperature > 0 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            cur_score = new_score
            # update best
            if new_score < best_score:
                best = cur.copy()
                best_score = new_score
                # early exit if target met
                if best_score < 1.50274349:
                    break
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]
        it += 1
        temperature *= decay
        if it % check_interval == 0 and time.monotonic() >= deadline:
            break
    # final best
    return [float(x) for x in best]

Helper functions:

```
def _next_pow2(x): return 1 << (x-1).bit_length()
def _score(arr, n, L):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    maxb = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * maxb / (s * s)
```

import time, math
import numpy as np

# ---------- utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n‑1 (classic binomial step)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 bound via FFT (same as evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _alternating_order(vals: np.ndarray) -> np.ndarray:
    """Place largest, second‑largest, … alternating at the ends."""
    order = np.argsort(-vals, kind='mergesort')
    out = np.empty_like(vals)
    left, right = 0, len(vals) - 1
    toggle = True
    for idx in order:
        if toggle:
            out[left] = vals[idx]
            left += 1
        else:
            out[right] = vals[idx]
            right -= 1
        toggle = not toggle
    return out

    candidates = [base_vals,

    # start the stochastic search from the best deterministic variant
    cur = best.copy()

    TARGET = 1.50274349
