            raise ValueError

    n = parent_arr.shape[0] if parent_arr is not None else 256

    # define FFT length
    fft_len = 1 << ((2*n - 1).bit_length())
    
    # helper: project onto simplex
    def project_simplex(v):
        # same as _project_onto_simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(1, len(v)+1)
        cond = u - cssv / ind > 0
        if not np.any(cond):
            return np.full_like(v, 1.0/len(v))
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho+1)
        w = np.maximum(v - theta, 0)
        return w

    # conv max function
    def max_conv(a):
        # a is sum=1
        A = np.fft.rfft(a, n=fft_len)
        B = A * A
        conv = np.fft.irfft(B, n=fft_len)[:2*n-1]
        return float(conv.max())
    
    # generate initial candidates
    candidates = []

def _project_onto_simplex(v):
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import time
import numpy as np

# ---------- helpers (top level) ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (real) via FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum value of a*a."""
    return float(_full_convolution(a, fft_len).max())


def _initial_candidates(n: int, parent_arr: np.ndarray | None):
    """A small set of diverse starting vectors, each summed to 1."""
    cand = []

    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        if p.sum() > 0.0:
            cand.append(p / p.sum())

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # sin‑power shapes
    idx = np.arange(n, dtype=np.float64)
    theta = np.pi * (idx + 0.5) / n
    sv = np.sin(theta)
    for p in (0.5, 1.0, 2.0):
        v = sv ** p
        cand.append(v / v.sum())

    # exponential decay
    for lam in (0.001, 0.005, 0.01):
        v = np.exp(-lam * idx)
        cand.append(v / v.sum())

    # Gaussian envelopes
    mid = (n - 1) / 2.0
    for sigma in (0.2 * n, 0.35 * n, 0.5 * n):
        v = np.exp(-0.5 * ((idx - mid) / sigma) ** 2)
        cand.append(v / v.sum())

    # Catalan‑style weights
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    cand.append(w / w.sum())

    return cand


def _subgradient_step(a: np.ndarray, n: int, fft_len: int, rng: np.random.Generator):
    """One projected sub‑gradient step on the max‑convolution objective."""
    conv = _full_convolution(a, fft_len)
    k = int(conv.argmax())
    # gradient of the active term (2*a[k‑i] where defined)
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        g[idx] = 2.0 * a[k - idx]
    # adaptive step magnitude
    step = 0.5 / np.sqrt(rng.random() + 1e-12)  # simple diminishing magnitude
    a_new = _project_onto_simplex(a - step * g)
    return a_new


def _annealing_step(cur: np.ndarray, n: int, fft_len: int,
                    rng: np.random.Generator, sigma: float):
    """Propose a log‑normal perturbation of cur."""
    prop = cur * np.exp(rng.normal(0.0, sigma, size=n))
    prop = _project_onto_simplex(prop)
    return prop
