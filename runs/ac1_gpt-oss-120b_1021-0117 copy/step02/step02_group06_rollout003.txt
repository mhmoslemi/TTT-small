import time
import numpy as np

# ---------- helper utilities ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, Nfft: int) -> np.ndarray:
    """Linear convolution a*a using FFT, output length 2*len(a)-1."""
    A = np.fft.rfft(a, Nfft)
    B = A * A
    full = np.fft.irfft(B, Nfft)
    n = a.shape[0]
    return full[: 2 * n - 1]


def _cross_corr_fft(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """Cross‑correlation g_k = sum_j a_j * w_{j+k}, k=0..n-1, via FFT."""
    A = np.fft.rfft(a, Nfft)
    W = np.fft.rfft(w, Nfft)
    G_full = np.fft.irfft(np.conj(A) * W, Nfft)
    n = a.shape[0]
    start = n - 1
    return G_full[start:start + n]

            init_candidates.append(p / s)

    # 2. uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 3. triangular (i+1)*(n-i)
    idx = np.arange(n, dtype=np.float64)
    tri = (idx + 1) * (n - idx)
    init_candidates.append(tri / tri.sum())

    # 4. sin^p windows
    theta = np.pi * (idx + 0.5) / n
    sinv = np.sin(theta)
    for p in (0.5, 1.0, 2.0):
        v = sinv ** p
        init_candidates.append(v / v.sum())

    # 5. random Dirichlet draws (few)
    for _ in range(3):
        v = rng.random(n)
        init_candidates.append(v / v.sum())

    # ----- pick best initial candidate using cheap score (2*n*max(conv)) -----
    best_vec = None
    best_score = np.inf
    for cand in init_candidates:
        conv = _conv_self_fft(cand, Nfft)
        max_b = float(conv.max())
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_vec = cand.copy()
    if best_vec is None:      # fallback safety
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # ----- optimisation parameters -----
    a = best_vec.copy()
    beta = 0.6                # start smoothness parameter
    beta_max = 30.0
    beta_factor = 1.018
    lr0 = 0.12                # base learning rate
    no_imp_iters = 0
    restart_thresh = 1200
    target_score = 1.50274349

    # ----- main loop -----
    while time.monotonic() < deadline:
        # current convolution and true score
        conv = _conv_self_fft(a, Nfft)
        max_b = float(conv.max())
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            no_imp_iters = 0
            if best_score <= target_score:
                break
        else:
            no_imp_iters += 1

        # smooth max via softmax weights
        max_c = conv.max()
        # shift for numerical stability
        exp_beta = np.exp(beta * (conv - max_c))
        w = exp_beta / exp_beta.sum()   # sums to 1

        # gradient of smoothed objective: 2 * (a ⋆ w)
        grad = 2.0 * _cross_corr_fft(a, w, Nfft)

        # backtracking line search
        lr = lr0 / (1.0 + 0.001 * no_imp_iters)   # modest decay
        accepted = False
        for _ in range(8):
            cand = _project_onto_simplex(a - lr * grad)
            conv_cand = _conv_self_fft(cand, Nfft)
            score_cand = 2.0 * n * float(conv_cand.max())
            if score_cand <= cur_score:
                a = cand
                accepted = True
                break
            lr *= 0.55
        if not accepted:
            # tiny random jitter to escape flat regions
            jitter = np.exp(rng.normal(0.0, 0.021, size=n))
            a = _project_onto_simplex(a * jitter)

        # anneal beta
        beta = min(beta_max, beta * beta_factor)

        # occasional restart from best with perturbation
        if no_imp_iters > restart_thresh:
            pert = np.exp(rng.normal(0.0, 0.028, size=n))
            a = _project_onto_simplex(best_vec * pert)
            no_imp_iters = 0

        # occasional random shake (≈1 % chance)
        if rng.random() < 0.012:
            shake = np.exp(rng.normal(0.0, 0.017, size=n))
            a = _project_onto_simplex(a * shake)

    # ----- final fallback -----
    return best_vec.tolist()
