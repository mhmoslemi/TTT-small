import numpy as np
import time
import math

def _initial_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0]=1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def _fft_conv(seq, fft_len):
    """Compute linear convolution of seq with itself (zero-padded) using FFT."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)
    # conv length = fft_len, but only first (2*len(seq)-1) are meaningful
    return conv[: 2 * seq.size - 1]

def _score_from_conv(conv, n):
    """Score assuming sequence sum = 1."""
    return 2.0 * n * np.max(conv)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # 0.5s safety

    # Determine length n and possible parent
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent_seq = np.clip(arr, 0.0, 1000.0)
            n = parent_seq.size
        else:
            parent_seq = None
            n = 3809
    except Exception:
        parent_seq = None
        n = 3809

    # FFT length (next power of two >= 2n-1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # Prepare two candidate initializations
    # Parent (if any)
    candidates = []
    if parent_seq is not None:
        seq = parent_seq.copy()
        s = seq.sum()
        if s > 0:
            seq /= s
            candidates.append(seq)

    # Binomial seed
    bin_seq = _initial_binomial(n)
    s = bin_seq.sum()
    if s > 0:
        bin_seq = bin_seq / s
        candidates.append(bin_seq)

    # Choose best of the available seeds
    best_seq = None
    best_score = float("inf")
    best_conv = None

    for seq in candidates:
        conv = _fft_conv(seq, fft_len)
        sc = _score_from_conv(conv, n)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            best_conv = conv.copy()

    # Fallback if none available (should not happen)
    if best_seq is None:
        best_seq = _initial_binomial(n)
        best_seq /= best_seq.sum()
        best_conv = _fft_conv(best_seq, fft_len)
        best_score = _score_from_conv(best_conv, n)

    # Current state
    a = best_seq.copy()
    cur_score = best_score
    cur_conv = best_conv

    # Parameters
    top_k = 5
    step_factors = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    target = 1.50274349
    # Loop
    while time.monotonic() < deadline:
        # Identify top convolution entries
        if top_k >= cur_conv.size:
            idxs = np.arange(cur_conv.size)
        else:
            idxs = np.argpartition(cur_conv, -top_k)[-top_k:]

        # Gradient accumulation over selected lags
        grad = np.zeros_like(a)
        for k in idxs:
            start_i = max(0, k - (n - 1))
            end_i = min(k, n - 1)
            i_range = np.arange(start_i, end_i + 1, dtype=np.int64)
            j_range = k - i_range
            # add 2 * a[j] to grad[i]
            grad[i_range] += 2.0 * a[j_range]

        # Zero‑sum descent direction
        mean_g = grad.mean()
        direction = -grad + mean_g

        improved = False
        for sf in step_factors:
            cand = a + sf * direction
            np.clip(cand, 0.0, None, out=cand)
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s  # enforce sum = 1

            cand_conv = _fft_conv(cand, fft_len)
            cand_score = _score_from_conv(cand_conv, n)

            if cand_score < cur_score:
                a = cand
                cur_conv = cand_conv
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    best_conv = cand_conv.copy()
                break  # accept first improving step (largest sf first)

        if not improved:
            # Small random perturbation to escape flat regions
            width = rng.integers(1, max(2, n // 30))
            start_idx = rng.integers(0, n - width + 1)
            factor = math.exp(rng.normal(0.0, 0.07))
            a[start_idx:start_idx + width] *= factor
            np.clip(a, 0.0, None, out=a)
            a /= a.sum()
            cur_conv = _fft_conv(a, fft_len)
            cur_score = _score_from_conv(cur_conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                best_conv = cur_conv.copy()

        # early exit if target reached
        if best_score <= target:
            break

    return [float(x) for x in best_seq]

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

    """Linear convolution of seq with itself using an FFT of length fft_len."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)
    # keep only the linear (non‑circular) part
    return conv[: 2 * seq.size - 1]
