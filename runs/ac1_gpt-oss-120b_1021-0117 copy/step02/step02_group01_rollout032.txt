# Helper: linear decreasing distribution summing to 1
# a_i = 2 * (n - i) / (n * (n + 1)),   i = 0 .. n-1
# ----------------------------------------------------------------------
def _linear_decreasing(n: int) -> np.ndarray:
    """Return a length‑n non‑negative vector on the probability simplex."""
    # values n, n-1, ..., 1
    vals = np.arange(n, 0, -1, dtype=np.float64)
    # normalising factor 2/(n*(n+1)) makes the sum exactly 1
    return (2.0 * vals) / (n * (n + 1))

    # (performed only if time permits)
    # ------------------------------------------------------------------
    # projection onto the simplex (probability simplex)
    def _project_simplex(v: np.ndarray) -> np.ndarray:
        """Euclidean projection of v onto {x >= 0, sum x = 1}."""
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho_arr = u - css / np.arange(1, v.size + 1)
        rho = np.where(rho_arr > 0)[0]
        if rho.size:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        else:
            theta = 0.0
        return np.maximum(v - theta, 0.0)

    # only attempt refinement if we still have a healthy time budget
    if time.monotonic() + 1.0 < deadline:
        a = best_seq.copy()
        n = a.size
        L = 1 << ((2 * n - 1).bit_length())
        step = 0.2
        decay = 0.999
        min_step = 1e-6
        rng = np.random.default_rng(seed)

        while time.monotonic() < deadline:
            # full convolution via FFT (real‑valued)
            fv = np.fft.rfft(a, n=L)
            conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]

            max_val = conv.max()
            active = np.where(conv >= max_val - 1e-14)[0]

            # gradient of the supremum of convolution (sub‑gradient)
            g = np.zeros_like(a)
            for k in active:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1, dtype=int)
                g[idx] += 2.0 * a[k - idx]

            cand = a - step * g
            cand = _project_simplex(cand)

            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
            else:
                # occasional uphill move (simulated‑annealing style)
                if rng.random() < np.exp(-(cand_score - best_score) / max(step, 1e-8)):
                    a = cand

            # small random mass transfer every now and then
            if rng.random() < 0.02:
                i, j = rng.integers(0, n, size=2)
                if a[i] > 1e-12:
                    delta = a[i] * rng.random() * 0.3
                    tmp = a.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_score = evaluate_sequence(tmp.tolist())
                    if tmp_score < best_score:
                        best_score = tmp_score
                        best_seq = tmp.copy()
                        a = tmp

            step = max(min_step, step * decay)
            # early exit if step is tiny and no improvement is seen
            if step <= min_step:
                break

    # Return as plain Python list – the evaluator expects this format
    return [float(x) for x in best_seq]
