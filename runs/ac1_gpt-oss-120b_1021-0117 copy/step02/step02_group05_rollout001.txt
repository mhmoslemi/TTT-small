    # optional second init if parent not given

        alt = _gen_binomial(n)

        if alt_score < best_score:
            cur = alt.copy()
            best = alt.copy()
            best_score = alt_score
            cur_score = alt_score
    # set hyperparams
    start_temp = 1.0
    temp = start_temp
    decay = 0.9996
    min_block = max(2, n // 200)
    max_block = max(5, n // 10)

    it = 0

        move_type = rng.random()

import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast evaluation of the AC1 bound via FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)
