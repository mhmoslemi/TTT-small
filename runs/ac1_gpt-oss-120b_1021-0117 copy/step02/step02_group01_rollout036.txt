import time, numpy as np

def _project_simplex(v):
    # projection onto simplex {x>=0, sum=1}
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k-1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a

def _conv_full(v, L):
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*len(v)-1]
    return conv

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


def _sqrt_series(n: int) -> np.ndarray:
    """Analytic sqrt‑series (a*a = 1+…+x^{2n‑2})."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


def _conv_full(v: np.ndarray, L: int) -> np.ndarray:
    """Full convolution v*v using FFT (length 2·len(v)-1)."""
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(v) - 1]
    return conv
