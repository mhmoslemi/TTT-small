def evaluate_sequence(sequence):
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
         if isinstance(x, bool) or not isinstance(x, (int, float)):
             return np.inf

import numpy as np, time, math, random
def _next_pow2(x):
    return 1 << (x - 1).bit_length()
def _ac1_score_fft(a, L=None):
    n = a.shape[0]
    if L is None: L = _next_pow2(2*n-1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
    maxb = conv.max()
    s = a.sum()
    return 2.0 * n * maxb / (s*s) if s>1e-12 else float('inf')

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    # get parent or fallback

            for i in range(1,n):
                a[i] = a[i-1] * (2*i - 1) / (2*i)
            return a

    L = _next_pow2(2*n-1)
    best = base.copy()

    # keep a variable current
    cur = best.copy()
    cur_score = best_score
    # temperature schedule
    temperature = 1.0
    decay = 0.9995
    # iteration counter
    it = 0
    while time.monotonic() < deadline:
        move = rng.random()
        cand = None
        if move < 0.45:
            # swap two positions
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:

            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]
        elif move < 0.75:
            # block scaling
            w = int(rng.integers(max(1, n//200), max(2, n//12)))
            start = int(rng.integers(0, max(1, n-w+1)))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start:start+w] *= factor
        else:
            # mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * 0.4
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
        # clip
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = _ac1_score_fft(cand, L)
        # update best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        accept = False

            accept = True

            prob = math.exp((cur_score - cand_score) / (temperature + 1e-14))
            if rng.random() < prob:

        if accept:

import math
import time

import numpy as np

# -------------------------------------------------------------
# Helper functions (FFT based AC1 score and next power of two)
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score_fft(arr: np.ndarray, L: int) -> float:
    """Compute the AC1 upper bound using FFT (exactly as the evaluator)."""
    n = arr.shape[0]
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

# Main search routine
# -------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims
    to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    # Leave a tiny safety margin for the sandbox timeout.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # -----------------------------------------------------------------
    # 1. Load a parent if present, otherwise fall back to a binomial start.
    # -----------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        base = parent.astype(np.float64).copy()
    except Exception:                     # no parent available
        n = 256
        # classic binomial step: C(2i,i)/4**i  (i=0..n‑1)
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
        base = a

    # -----------------------------------------------------------------
    # 2. Prepare FFT size and initial candidates.
    # -----------------------------------------------------------------
    L = _next_pow2(2 * n - 1)

    # start from the best of: original, sorted‑ascending, sorted‑descending
    candidates = [base]
    candidates.append(np.sort(base))               # ascending
    candidates.append(np.sort(base)[::-1])         # descending
    best = min(candidates, key=lambda v: _ac1_score_fft(v, L))

        max_b = float(conv_cur.max())

    # -------------------------------------------------------------
    return [float(x) for x in best]
