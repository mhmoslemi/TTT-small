import time
import numpy as np

# ---------- helpers ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, n: int, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length 2*n-1)."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * n - 1]


def _conv_max_fft(a: np.ndarray, n: int, fft_len: int) -> float:
    """Maximum entry of the self‑convolution of a (sum(a) assumed = 1)."""
    full = _conv_full_fft(a, n, fft_len)
    return float(full.max())


# ---------- main search routine ----------
