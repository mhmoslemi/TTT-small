def _project_simplex(v):
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _grad_softmax(a, beta):
    n = a.shape[0]
    # compute convolution
    conv = _conv_fft(a)
    maxc = conv.max()
    # compute softmax weights
    # to avoid overflow
    exp_shift = np.exp(beta * (conv - maxc))
    w = exp_shift / exp_shift.sum()
    # convolution of a with w (full)
    L2 = 1 << ((3 * n - 2).bit_length())
    fa = np.fft.rfft(a, n=L2)
    fw = np.fft.rfft(w, n=L2)
    full = np.fft.irfft(fa * fw, n=L2)[: 3 * n - 2]
    # slice from n-1 to n-1+n
    grad = 2.0 * full[n-1 : n-1 + n]
    return grad, conv.max()  # return also max for use

import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex (non‑negative, sum = 1)."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast full linear convolution via FFT (real → real)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Objective used by the evaluator (identical to evaluate_sequence)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_fft(a)
    maxb = conv.max()
    sum_a = a.sum()
    if sum_a <= 0:
        return float("inf")
    return float(2.0 * n * maxb / (sum_a ** 2))


# ------------------------------------------------------------
# Gradient of a smooth soft‑max approximation of max_k (a*a)_k
# ------------------------------------------------------------
def _softmax_grad(a: np.ndarray, beta: float):
    """Return (grad, max_conv) where grad = ∂/∂a soft‑max(conv)."""
    n = a.shape[0]
    # full autocorrelation of a
    conv = _conv_fft(a)                 # length 2n‑1
    maxc = conv.max()
    # soft‑max weights (stable)
    w = np.exp(beta * (conv - maxc))
    w /= w.sum()
    # convolution a * w (full) using FFT
    L2 = 1 << ((3 * n - 2).bit_length())  # ≥ 3n‑2
    fa = np.fft.rfft(a, n=L2)
    fw = np.fft.rfft(w, n=L2)
    full = np.fft.irfft(fa * fw, n=L2)[: 3 * n - 2]
    # slice to obtain gradient (see derivation)
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad, maxc

# ------------------------------------------------------------

    rng = np.random.default_rng(seed)

    # leave a small safety margin for the harness

    # ------------------------------------------------------------------

    s = a.sum()
    if s <= 0.0:

    # best solution seen so far
    best = a.copy()

    # ------------------------------------------------------------

    it = 0
