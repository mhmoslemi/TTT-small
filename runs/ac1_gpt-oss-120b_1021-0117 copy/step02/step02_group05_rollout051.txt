import time, math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast evaluation of the AC1 bound via FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic optimisation of a non‑negative step function for the AC1 bound.
    Returns a list of Python floats.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Initialise from parent if available, otherwise use a binomial start.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        cur = parent.copy()
        n = int(parent.size)
    except Exception:
        n = 256
        cur = _gen_binomial(n)

    # Normalise sum to a convenient target (scale‑invariant for the score)
    target_sum = math.sqrt(2.0 * n)
    s = cur.sum()
    if s > 0:
        cur *= target_sum / s
    np.clip(cur, 0.0, 1000.0, out=cur)

    # Pre‑computed FFT length
    L = _next_pow2(2 * n - 1)

    # Initial scores
    cur_score = _ac1_score(cur, n, L)
    best = cur.copy()
    best_score = cur_score

    # --------------------------------------------------------------
    # Hyper‑parameters
    temperature = 1.0
    decay = 0.9997
    base_eps = 0.02          # maximal step size for gradient moves
    min_block = max(2, n // 200)
    max_block = max(2, n // 12)
    transfer_max = 0.4
    log_factor_sigma = 0.13
    target = 1.50274349

    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.5)
