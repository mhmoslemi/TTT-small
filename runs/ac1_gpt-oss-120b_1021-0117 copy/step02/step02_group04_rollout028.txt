    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Score = 2·n·max(conv)/sum² using an FFT‑based convolution."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound defined by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.5)

    # ------------------------------------------------------------------
    # problem size & initial vector
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256                     # a modest fallback length
    L = _next_pow2(2 * n - 1)

    # two cheap starting points: the provided parent (if any) and a binomial shape
    if parent is None:
        cur = _gen_binomial(n)
    else:
        cur = parent.astype(np.float64)
    binom = _gen_binomial(n)
    # pick the better of the two for the initial state
    if _ac1_score_fft(binom, n, L) < _ac1_score_fft(cur, n, L):
        cur = binom.copy()

    best = cur.copy()
    best_score = _ac1_score_fft(best, n, L)

    # ------------------------------------------------------------------
    # simulated‑annealing parameters (tuned for the 1 k‑second budget)
    # ------------------------------------------------------------------
    temperature = 1.0
    temp_decay = 0.9994
    step_size = 0.30                # magnitude of the gradient‑inspired move
    step_decay = 0.9998

    # ------------------------------------------------------------------
    # main optimisation loop (time‑bounded, always keeps the best seen)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- current convolution (FFT, cheap) ----
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_b = float(conv.max())
        k_max = int(conv.argmax())
        sum_cur = float(cur.sum())
        cur_score = 2.0 * n * max_b / (sum_cur * sum_cur)

        # ---- pick a move -------------------------------------------------
        move_type = rng.random()
        if move_type < 0.45:                       # mirror‑gradient push
            # target value that would equalise the active lag
            target = max_b / sum_cur
            idx = np.arange(n)
            j = k_max - idx
            mask = (j >= 0) & (j < n)
            mirror = np.zeros_like(cur)
            mirror[mask] = cur[j[mask]]
            # gradient‑like update (scale‑invariant)
            C = 4.0 * n / (sum_cur * sum_cur)
            delta = step_size * C * (target - mirror)
            cand = cur + delta
            np.clip(cand, 0.0, 1000.0, out=cand)

        elif move_type < 0.75:                    # random block scaling
            width = int(rng.integers(max(1, n // 250), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.10))
            cand = cur.copy()
            cand[start:start + width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)

        else:                                      # mass transfer between two sites
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * 0.35      # move up to 35 % of source mass
                delta = cur[i] * frac
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
                np.clip(cand, 0.0, 1000.0, out=cand)
            else:
                continue

        # ---- evaluate candidate (exact same FFT routine) ----
        cand_score = _ac1_score_fft(cand, n, L)

        # keep the global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            # early exit if we already beat the announced target
            if best_score < 1.50274349:
                break

        # ---- simulated‑annealing acceptance ----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # ---- schedule updates -------------------------------------------------
        temperature *= temp_decay
        step_size *= step_decay

        # occasional reset to the best seen (helps escape plateaux)
        if rng.random() < 0.001:
            cur = best.copy()
            temperature = 1.0
            step_size = 0.30

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
