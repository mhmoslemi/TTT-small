    except Exception:
        parent = None

    def binomial_initial(n):
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1,n):
            a[i] = a[i-1] * (2*i-1)/(2*i)
        return a

    best = parent if parent is not None else binomial_initial(n)
    best = best.copy()
    fft_len = 1 << ((2*n-1).bit_length())

    def conv_full(seq):
        f = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(f * f, n=fft_len)[: 2*n-1]
        return conv

        max_b = np.max(conv)
        s = np.sum(seq)
        return float(2.0 * n * max_b / (s * s))

    best_score = compute_score(best, best_conv)

        iter_count += 1

        # Occasionally random perturbation to escape stagnation:
        if rng.random() < 0.05:

            width = rng.integers(1, max(2, n//50))
            i0 = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            cand = current.copy()
            cand[i0:i0+width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)
        else:

            conv = conv_full(current)
            # compute softmax weights
            # shift for stability
            max_conv = conv.max()

        elapsed = time.monotonic() - start

                best_conv = cand_conv

    return [float(x) for x in best]

# Helper functions (top‑level, no closures)

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fft_len_for_conv(n: int) -> int:
    """Length of FFT that can hold a full linear convolution of size n."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution_fft(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the full linear convolution seq * seq (length 2n‑1) using FFT."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    return conv


def _score_from_conv(seq: np.ndarray, conv: np.ndarray) -> float:
    """Compute the AC1 upper‑bound score for a given sequence and its conv."""
    max_b = np.max(conv)
    s = np.sum(seq)
    return float(2.0 * seq.size * max_b / (s * s))


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Smooth approximation of the max‑operator: softmax over conv."""
    m = np.max(conv)
    # shift for numerical stability
    exp_arg = alpha * (conv - m)
    w = np.exp(exp_arg, dtype=np.float64)
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        # fallback to uniform weights if under/overflow occurs
        w = np.full_like(conv, 1.0 / conv.size, dtype=np.float64)
    else:
        w /= w_sum
    return w


def _gradient_via_softmax(seq: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """Return ∇(Σ w_k·conv_k) = 2·(seq * w) evaluated at needed indices."""
    # reverse w for correlation
    w_rev = w[::-1]
    A = np.fft.rfft(seq, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    g_full = np.fft.irfft(A * W, n=fft_len)[: 2 * seq.size - 1]
    n = seq.size
    # g_i = 2 * g_full[2n‑2‑i]  for i=0..n‑1
    g = 2.0 * g_full[2 * n - 2 : n - 2 : -1]
    return g


def _project_onto_sum(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale seq so that its L1‑norm equals target_sum (preserving sign)."""
    s = np.sum(seq)
    if s == 0:
        return seq
    seq = seq * (target_sum / s)
    # enforce bounds after scaling
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq
