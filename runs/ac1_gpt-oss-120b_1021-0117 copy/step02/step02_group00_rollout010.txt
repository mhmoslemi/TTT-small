import time

# ------------------------------------------------------------
# Project onto probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Full linear convolution via FFT (a * a)
# ------------------------------------------------------------
def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full self‑convolution a*a (length 2*len(a)-1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.size - 1]
    return conv


# ------------------------------------------------------------
# Gradient of the smooth max (log‑sum‑exp) of the convolution
# ------------------------------------------------------------
def _grad_from_weights(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Given a (length n) and a weight vector w (length 2n-1) that sums to 1,
    compute the gradient of  f(a)=∑_k w_k (a*a)_k.
    """
    n = a.size
    # reversed weights
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w_rev, n=L)
    # convolution a * w_rev (full length n + (2n-1) -1 = 3n-2)
    conv = np.fft.irfft(Af * Wf, n=L)[: 3 * n - 2]
    idx = (2 * n - 2) - np.arange(n)          # K-1 - j   with K=2n-1
    grad = 2.0 * conv[idx]
    return grad

    conv = _conv_self_fft(a, L)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)

                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.35
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_fft(tmp, n, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0

        # ------------------------------------------------------------
        # stagnation handling – jitter around the best known point
        # ------------------------------------------------------------
        if no_improve > 4000 and rng.random() < 0.4:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score_fft(tmp, n, L)
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc
                cur, cur_sc = tmp.copy(), tmp_sc
            no_improve = 0

        # ------------------------------------------------------------
        # slowly raise the smoothing parameter (sharpen the max)
        # ------------------------------------------------------------
        if it % 2500 == 0:
            beta = min(beta * 1.07, 80.0)

        # ------------------------------------------------------------
        # early exit if target reached
        # ------------------------------------------------------------
        if best_sc < 1.50274349:
            break

        # ------------------------------------------------------------
        # time guard
        # ------------------------------------------------------------
        if time.monotonic() >= deadline:
            break

    # return as plain Python list (the evaluator will re‑clamp if needed)
    return [float(x) for x in best_vec]
