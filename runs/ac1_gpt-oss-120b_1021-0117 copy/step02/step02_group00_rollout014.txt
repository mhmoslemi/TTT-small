def _grad_softmax(a, conv, gamma):
    # compute w = softmax(gamma*conv)
    maxc = conv.max()

    w = exps / exps.sum()  # shape 2n-1

import time, math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv

def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Return softmax of gamma*conv, numerically stable."""
    m = conv.max()
    ex = np.exp(gamma * (conv - m))
    return ex / ex.sum()

def _grad_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of softmax-weighted max: 2 * (w * a_rev) valid part."""
    n = a.shape[0]
    a_rev = a[::-1]
    L = 1 << ((len(w) + n - 1).bit_length())
    fw = np.fft.rfft(w, n=L)
    fa = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(fw * fa, n=L)
    start = n - 1
    return 2.0 * full[start:start + n]

def _hard_grad(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient when max convolution occurs at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g[idx] = 2.0 * a[k - idx]
    return g

def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Full linear convolution of a with itself via FFT
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Soft‑max weights for a smoothed max
# ------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    m = conv.max()
    ex = np.exp(gamma * (conv - m))
    return ex / ex.sum()


# ------------------------------------------------------------
# Gradient of the soft‑max weighted max (2·(w * a_rev) valid part)
# ------------------------------------------------------------
def _grad_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    a_rev = a[::-1]
    L = 1 << ((len(w) + n - 1).bit_length())
    fw = np.fft.rfft(w, n=L)
    fa = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(fw * fa, n=L)
    start = n - 1
    return 2.0 * full[start:start + n]


# ------------------------------------------------------------
# Sub‑gradient when the max occurs at index k
# ------------------------------------------------------------
def _hard_grad(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g[idx] = 2.0 * a[k - idx]
    return g


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function whose AC1 bound is low."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # ------------------- initial vector -----------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # simple decreasing power‑law fallback
        n = 256
        a0 = np.arange(1, n + 1, dtype=np.float64) ** -0.5
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s
    n = a0.size

    cur = a0.copy()
    conv_cur = _conv_fft(cur)
    cur_sc = 2.0 * n * conv_cur.max()
    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------- optimisation parameters -------------
    eta0 = 0.8                 # base step‑size factor
    gamma = 0.3                # start of soft‑max sharpening
    gamma_max = 20.0
    it = 0
    no_improve = 0

    # ------------------- main loop ---------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution and its max index
        conv = _conv_fft(cur)
        cur_sc = 2.0 * n * conv.max()
        k_max = int(np.argmax(conv))

        # choose gradient: soft‑max while gamma is modest, else hard
        if gamma < 5.0:
            w = _softmax_weights(conv, gamma)
            grad = _grad_softmax(cur, w)
        else:
            grad = _hard_grad(cur, k_max)

        # diminishing step size (1/√t)
        alpha = eta0 / math.sqrt(it)

        cand = cur - alpha * grad
        cand = _project_simplex(cand)

        # evaluate candidate
        conv_cand = _conv_fft(cand)
        cand_sc = 2.0 * n * conv_cand.max()

        # accept if better (or with small Metropolis probability)
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12)):
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_vec = cand.copy()
                best_sc = cand_sc
                no_improve = 0
        else:
            no_improve += 1

        # ------------------- occasional random moves -------------
        if no_improve > 3000 and rng.random() < 0.15:
            # small jitter around current point
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = cur + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            conv_tmp = _conv_fft(tmp)
            tmp_sc = 2.0 * n * conv_tmp.max()
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
            else:
                # random mass transfer move
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    delta = cur[i] * rng.random() * 0.35
                    tmp2 = cur.copy()
                    tmp2[i] -= delta
                    tmp2[j] += delta
                    tmp2 = _project_simplex(tmp2)
                    conv_tmp2 = _conv_fft(tmp2)
                    tmp2_sc = 2.0 * n * conv_tmp2.max()
                    if tmp2_sc < best_sc:
                        best_vec, best_sc = tmp2.copy(), tmp2_sc
                        cur, cur_sc = tmp2, tmp2_sc
                        no_improve = 0

        # ------------------- parameter schedules ---------------
        if gamma < gamma_max:
            gamma *= 1.001

        # periodic jitter around the best ever solution
        if it % 4000 == 0 and rng.random() < 0.22:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            conv_tmp = _conv_fft(tmp)
            tmp_sc = 2.0 * n * conv_tmp.max()
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0

        # early exit if we have already beaten the target bound
        if best_sc < 1.50274349:
            break

    # ---------------------------------------------------------
    # Return a plain Python list (the evaluator will clip & check)
    # ---------------------------------------------------------
    return [float(x) for x in best_vec]
