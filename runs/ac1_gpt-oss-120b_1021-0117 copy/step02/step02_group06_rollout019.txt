import time
import numpy as np

# ----- low‑level utilities -------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, n: int, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a via real FFT (length 2*n‑1)."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * n - 1]


def _score_from_vec(a: np.ndarray, n: int, fft_len: int) -> float:
    """Return the AC1 bound for a (assumed sum(a)=1)."""
    conv = _conv_self_fft(a, n, fft_len)
    return 2.0 * n * float(conv.max())


# ----- candidate factories -------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_inc(n: int) -> np.ndarray:
    v = np.arange(1, n + 1, dtype=np.float64)
    return v / v.sum()


def _linear_dec(n: int) -> np.ndarray:
    v = np.arange(n, 0, -1, dtype=np.float64)
    return v / v.sum()


def _exp_decay(n: int, r: float) -> np.ndarray:
    v = r ** np.arange(n, dtype=np.float64)
    return v / v.sum()


def _prod_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    w = (i * (n + 1 - i)) ** p
    return w / w.sum()


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    w = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    return w / w.sum()


def _sin_power(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    return w / w.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    vec_sum = vec.sum()
    if vec_sum == 0.0:
        vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vec /= vec_sum
    return vec
