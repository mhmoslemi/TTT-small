def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 2.0)
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = _sqrt_series(300)  # fallback
    # Normalize
    a = a.clip(min=0)
    s = a.sum()
    if s <= 0:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s

    n = a.shape[0]
    # Precompute FFT length
    L = 1 << ((2*n - 1).bit_length())
    
    # helper to compute max conv quickly
    def conv_max_vec(v):
        fv = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2*n-1]
        max_val = conv.max()
        max_idx = int(conv.argmax())
        # active set for tolerance
        eps = max_val * 1e-12 + 1e-14
        active = np.where(conv >= max_val - eps)[0]
        return max_val, max_idx, active, conv

    max_val, max_idx, active_set, conv = conv_max_vec(a)
    cur_score = 2 * n * max_val
    best_a = a.copy()
    best_score = cur_score

    # Parameters
    step = 0.5
    min_step = 1e-6
    step_decay = 0.998
    beta = 0.0  # momentum coefficient
    prev_a = a.copy()
    # random perturb probability
    prob_jump = 0.02
    # tolerance for improvement acceptance
    T = 0.001 # simulated annealing temperature initial? We'll vary.

    while time.monotonic() < deadline:
        # compute gradient from active set
        g = np.zeros_like(a)
        for k in active_set:
            lo = max(0, k-(n-1))
            hi = min(n-1, k)
            idx = np.arange(lo, hi+1)
            g[idx] += 2.0 * a[k - idx]
        # momentum
        v = a - step * g + beta * (a - prev_a)
        # projection onto simplex
        v_proj = _project_simplex(v)
        # update prev
        prev_a = a
        a_candidate = v_proj
        # compute score
        max_val_c, max_idx_c, active_set_c, _ = conv_max_vec(a_candidate)
        cand_score = 2 * n * max_val_c

        # accept if improvement, else maybe probabilistically.
        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            max_idx = max_idx_c
            active_set = active_set_c
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
        else:
            # Simulated annealing acceptance
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-8)):
                a = a_candidate
                cur_score = cand_score
                max_idx = max_idx_c
                active_set = active_set_c

        # random mass transfer (jump)
        if rng.random() < prob_jump:
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                j = rng.integers(0, n)

                    j = rng.integers(0, n)

                delta = a[i] * frac
                a_jump = a.copy()

                a_jump = _project_simplex(a_jump)
                max_val_j, max_idx_j, active_set_j, _ = conv_max_vec(a_jump)

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of vector v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Analytic sqrt‑series (fallback when no parent is given)
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ------------------------------------------------------------
# Full convolution via FFT (returns the whole array)
# ------------------------------------------------------------
def _conv_full(vec: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[: 2 * vec.size - 1]
