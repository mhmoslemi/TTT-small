import time, numpy as np

def _project_simplex(v):

def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
    return conv

def _grad_logsumexp(a, c):
    n = a.shape[0]
    conv = _conv_fft(a)  # length 2n-1
    maxc = conv.max()
    exp_term = np.exp(c * (conv - maxc))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp  # length 2n-1, sum to 1.

    w_rev = w[::-1]
    m = n + len(w_rev) - 1
    L = 1 << ((m).bit_length())
    fa = np.fft.rfft(a, n=L)
    fw = np.fft.rfft(w_rev, n=L)
    conv2 = np.fft.irfft(fa * fw, n=L)[:m]
    start = len(w_rev) - 1
    g = 2.0 * conv2[start: start + n]
    return g

def _score(a):
    n = a.shape[0]
    conv = _conv_fft(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x>=0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Full linear convolution via FFT (real-valued)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv

# ------------------------------------------------------------
# Smoothed max via log‑sum‑exp and its gradient
# ------------------------------------------------------------
def _grad_logsumexp(a: np.ndarray, c: float) -> np.ndarray:
    """Gradient of (1/c)·log Σ exp(c·(a*a)) w.r.t. a."""
    n = a.shape[0]
    # convolution a*a
    conv = _conv_fft(a)                       # length 2n-1
    maxc = conv.max()
    # soft‑max weights
    exp_term = np.exp(c * (conv - maxc))
    w = exp_term / exp_term.sum()              # length 2n-1, sum to 1

    # gradient: 2 * Σ_i w_i * a_{i-j}  (j = 0..n-1)
    # compute via convolution of w with reversed a
    a_rev = a[::-1]
    L = 1 << ((len(w) + n - 1).bit_length())   # >= 3n-2
    fw = np.fft.rfft(w, n=L)
    fa_rev = np.fft.rfft(a_rev, n=L)
    conv_rev = np.fft.irfft(fw * fa_rev, n=L)[: len(w) + n - 1]
    start = n - 1
    g = 2.0 * conv_rev[start:start + n]
    return g

# ------------------------------------------------------------
# Exact score (same as evaluate_sequence for sum=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_fft(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the provided parent if it exists, otherwise a simple decay.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 1024
        # exponential decay, then normalise to sum = 1
        a = np.exp(-np.arange(n) / (n / 8.0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s
    n = a.size

    best = a.copy()
    best_sc = _score(best)

    # ------------------------------------------------------------
    # Adam‑style projected gradient descent on a smoothed max.
    # ------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0

    # schedule of smoothing parameters (c) and iteration counts
    schedule = [(0.5, 3000), (1.0, 3000), (2.0, 3000),
                (5.0, 3000), (10.0, 3000), (20.0, 3000), (50.0, 3000)]

    for c, steps in schedule:
        for _ in range(steps):
            it += 1
            if time.monotonic() >= deadline:
                return [float(x) for x in best]

            # gradient of smoothed objective
            g = _grad_logsumexp(a, c)

            # Adam update (projected)
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = 0.04 * m_hat / (np.sqrt(v_hat) + eps)   # base LR = 0.04
            a = a - step
            a = _project_simplex(a)

            # occasional exact evaluation (every 10 it)
            if it % 10 == 0:
                sc = _score(a)
                if sc < best_sc:
                    best = a.copy()
                    best_sc = sc

            if best_sc < 1.50274349:
                return [float(x) for x in best]

    # ------------------------------------------------------------
    # Final refinement using exact max sub‑gradient (deterministic).
    # ------------------------------------------------------------
    cur = best.copy()
    cur_sc = best_sc
    no_improve = 0
    it2 = 0
    while time.monotonic() < deadline:

        # active index of the current max convolution term
        conv = _conv_fft(cur)
        k = int(np.argmax(conv))

        # sub‑gradient of max_{i}(cur*cur)_i
        g = np.zeros_like(cur)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] = 2.0 * cur[k - idx]

        step_sz = 0.12 / np.sqrt(it2)
        cand = cur - step_sz * g
        cand = _project_simplex(cand)

        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:

        else:

        if no_improve > 2500 and rng.random() < 0.15:

            if cur[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0

        # early exit upon reaching target
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------
    # Return the best candidate seen (as plain Python floats)
    # ------------------------------------------------------------
    return [float(x) for x in best]
