import numpy as np
import time, math

# ----- utility helpers -------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _score_and_conv(arr: np.ndarray, n: int, L: int):
    """Return (score, full convolution) using FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    s = arr.sum()
    if s < 1e-12:
        return float('inf'), conv
    score = 2.0 * n * conv.max() / (s * s)
    return float(score), conv


def _grad_topk(a: np.ndarray, conv: np.ndarray, topk: int) -> np.ndarray:
    """Gradient (sub‑gradient) from the top‑k convolution entries."""
    n = a.shape[0]
    idx = np.arange(n)
    g = np.zeros_like(a)
    # indices of the k largest entries (unsorted)
    top_idx = np.argpartition(conv, -topk)[-topk:]
    for k in top_idx:
        rev = k - idx
        mask = (rev >= 0) & (rev < n)
        g[mask] += a[rev[mask]]
        dbl = mask & (rev != idx)
        # each distinct pair appears twice in the convolution sum
        g[dbl] += a[rev[dbl]]
    return g


def _smooth_candidate(a: np.ndarray, alpha: float) -> np.ndarray:
    """Low‑pass smoothing that preserves total mass."""
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    sm = np.convolve(a, kernel, mode='same')
    diff = sm - a
    diff -= diff.mean()          # keep sum unchanged
    cand = a + alpha * diff
    return cand


# ----- main search ----------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimizes
    the AC1 upper bound. Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- initialise -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        cur = parent.copy()
    except Exception:
        parent = None
        n = 256
        cur = _gen_binomial(n)

    L = _next_pow2(2 * n - 1)

    # evaluate initial candidate
    cur_score, cur_conv = _score_and_conv(cur, n, L)
    best = cur.copy()
    best_score = cur_score

    # ----- parameters -------------------------------------------------
    temperature = 1.0
    decay = 0.9985                     # slower cooling for longer runs
    min_block = max(2, n // 250)
    max_block = max(2, n // 12)
    transfer_max = 0.35
    grad_eps = 0.025
    topk_grad = 3                      # use up to three peak indices
    smooth_alpha = 0.22                # strength of smoothing moves
    target = 1.50274349                # early‑stop threshold

    # ----- main loop --------------------------------------------------
    while time.monotonic() < deadline:
        move_type = rng.random()

        # 1) multi‑peak gradient descent
        if move_type < 0.45:
            g = _grad_topk(cur, cur_conv, topk_grad)
            mean_g = g.mean()
            d = mean_g - g               # direction preserving sum
            cand = cur + grad_eps * d

        # 2) block scaling
        elif move_type < 0.75:
            width = int(rng.integers(min_block, max_block + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start:start + width] *= factor

        # 3) random mass transfer
        elif move_type < 0.90:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * transfer_max
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        # 4) smoothing move
        else:
            cand = _smooth_candidate(cur, smooth_alpha)

        # enforce bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate candidate
        cand_score, cand_conv = _score_and_conv(cand, n, L)

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < target:
                break

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            cur_conv = cand_conv

        # temperature schedule
        temperature *= decay

        # occasional restart from the best seen
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            cur_conv = None          # force recompute on next iteration
            temperature = 1.0

        # recompute conv lazily when needed
        if cur_conv is None:
            cur_score, cur_conv = _score_and_conv(cur, n, L)

    # ----- final fall‑back sanity ------------------------------------
    if best.sum() < 0.01:
        best[0] = max(best[0], 0.01 - best[1:].sum())

    return [float(x) for x in best]
