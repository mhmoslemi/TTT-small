def propose_candidate(...):
    import numpy as np, time, math, random ...
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 1.0)

    # get n
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
        init_seq = parent.copy()
    except Exception:
        n = 256
        init_seq = _gen_binomial(n)
    # maybe also generate some param families and pick best between them and init_seq

    # compute L
    L = 1 << ((2*n - 1).bit_length())  # next_pow2

    # define helper score function
    def ac1_score(a):
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
        maxb = conv.max()
        s = a.sum()
        if s < 1e-12:
            return math.inf
        return 2.0 * n * maxb / (s*s)

    # generate candidate families:
    def gen_exp(tau):
        idx = np.arange(n)
        a = np.exp(-idx / tau)
        return a

    def gen_power(p):
        idx = np.arange(n) + 1
        a = idx**(-p)
        return a

    def gen_gauss(sigma):
        c = (n-1)/2
        idx = np.arange(n)
        a = np.exp(-0.5*((idx - c)/sigma)**2)
        return a

    # evaluate some param sets quickly:
    param_candidates = []
    for tau in (n/10, n/5, n/3):
        param_candidates.append(gen_exp(tau))
    for p in (0.6, 0.8, 1.0, 1.2):
        param_candidates.append(gen_power(p))
    for sigma in (n/6, n/4, n/3):
        param_candidates.append(gen_gauss(sigma))

    best = init_seq.copy()
    best_score = ac1_score(best)
    for cand in param_candidates:
        cand = np.clip(cand, 0.0, 1000.0)
        # scale? scaling irrelevant; but we may normalize sum to something, but not needed.
        s = ac1_score(cand)
        if s < best_score:
            best = cand.copy()
            best_score = s

    cur = best.copy()
    cur_score = best_score

    # start SA
    temperature = 5.0
    decay = 0.9998
    target = 1.50274349

    # step sizes
    base_eps = 0.2
    block_eps_sigma = 0.12

    # predefine move probabilities
    while time.monotonic() < deadline:
        move_rand = rng.random()
        if move_rand < 0.5:
            # gradient guided with line search
            # compute conv and kmax
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
            kmax = int(np.argmax(conv))
            # compute gradient g
            idx = np.arange(n)
            rev = kmax - idx
            mask = (rev >= 0) & (rev < n)
            g = np.zeros_like(cur)
            g[mask] = cur[rev[mask]]
            self_mask = mask & (rev != idx)
            g[self_mask] *= 2.0
            # direction preserving sum
            d = (g.mean() - g)
            # step size adapt
            eps = base_eps * temperature
            # try backtracking up to 5 attempts
            cand = None
            best_local_score = cur_score
            for _ in range(5):
                cand_try = cur + eps * d
                np.clip(cand_try, 0.0, 1000.0, out=cand_try)
                cand_score = ac1_score(cand_try)
                if cand_score < best_local_score:
                    cand = cand_try
                    best_local_score = cand_score
                    break
                eps *= 0.5
            if cand is None:
                # fall back to small step
                eps = base_eps * temperature * 0.1
                cand = cur + eps * d
                np.clip(cand, 0.0, 1000.0, out=cand)
                cand_score = ac1_score(cand)
        elif move_rand < 0.8:

            cand_score = ac1_score(cand)

            best_score = cand_score
            # early exit

            cur = best.copy()
            cur_score = best_score

def _next_pow2(x: int) -> int:
