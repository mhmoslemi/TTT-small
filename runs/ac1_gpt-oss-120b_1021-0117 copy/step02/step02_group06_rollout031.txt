        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, len(v) + 1)
        cond = u - cssv / ind > 0
        if not np.any(cond):
            # return uniform
            return np.full_like(v, 1.0/len(v))
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w
    def conv_max(a):
        A = np.fft.rfft(a, n=fft_len)
        full = np.fft.irfft(A * A, n=fft_len)
        # take first 2*n-1

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a using an FFT of length fft_len."""
    A = np.fft.rfft(a, n=fft_len)
    full = np.fft.irfft(A * A, n=fft_len)
    return full[: 2 * a.shape[0] - 1]


def _initial_candidates(n: int, parent: np.ndarray | None):
    """A compact set of diverse start points (each already summed to 1)."""
    cand = []

    # 1) parent if available
    if parent is not None and parent.shape == (n,):
        p = np.clip(parent, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            cand.append(p / s)

    # 2) uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 3) symmetric triangular hat
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    tri = 1.0 - np.abs(idx - mid) / mid
    tri = np.maximum(tri, 0.0)
    cand.append(tri / tri.sum())

    # 4) semicircle (sqrt of a disc)
    x = (idx - mid) / mid
    semi = np.sqrt(np.maximum(0.0, 1.0 - x * x))
    cand.append(semi / semi.sum())

    # 5) sin‑power shapes
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    for p in (1.0, 2.0, 3.0, 4.0):
        v = sin_vals ** p
        cand.append(v / v.sum())

    # 6) a few random Dirichlet samples
    rng = np.random.default_rng(12345)
    for _ in range(4):
        v = rng.exponential(scale=1.0, size=n)
        cand.append(v / v.sum())

    return cand


def _refine_sequence(a0: np.ndarray, n: int, fft_len: int,
                     deadline: float, best_global: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum.
    Returns the best vector found and its score.
    """
    a = a0.copy()
    best_local = a.copy()
    conv = _full_convolution(a, fft_len)
    best_score = 2.0 * n * float(conv.max())

    momentum = np.zeros_like(a)
    beta = 0.9
    it = 0
    stagnant = 0

    while time.monotonic() < deadline:
        conv = _full_convolution(a, fft_len)
        max_val = float(conv.max())
        score = 2.0 * n * max_val

        if score < best_local_score:
            best_local_score = score
            best_local = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        # gradient from (almost) active indices
        tol = max_val * 1e-12
        active = np.where(conv >= max_val - tol)[0]
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            g[idx] += 2.0 * a[k - idx]
        if active.size:
            g /= active.size

        # step size (diminishing)
        it += 1
        step = 0.6 / np.sqrt(it)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        cand = a - step * momentum
        cand = _project_onto_simplex(cand)

        # occasional jitter to escape flat regions
        if rng.random() < 0.015:
            jitter = np.exp(rng.normal(0.0, 0.012, size=n))
            cand = cand * jitter
            cand = _project_onto_simplex(cand)

        a = cand

        # early cut‑offs
        if best_local_score <= best_global:
            break
        if stagnant > 3000:
            # restart from the best local point with a light perturbation
            perturb = np.exp(rng.normal(0.0, 0.025, size=n))
            a = best_local * perturb
            a = _project_onto_simplex(a)
            stagnant = 0

    return best_local, best_local_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that
    minimizes the AC1 upper bound used by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)  # safety margin

    # ---- obtain parent if present ----
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
    except Exception:
        parent_arr = None

    n = int(parent_arr.shape[0]) if parent_arr is not None else 256
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ---- generate a small portfolio of start vectors ----
    candidates = _initial_candidates(n, parent_arr)

    best_vec = None
    best_score = np.inf

    # evaluate the pool (no need to call evaluate_sequence)
    for vec in candidates:
        conv = _full_convolution(vec, fft_len)
        score = 2.0 * n * float(conv.max())
        if score < best_score:
            best_score = score
            best_vec = vec.copy()
        if best_score <= 1.50274349:
            return best_vec.tolist()

    # ---- iterative refinement ----
    # use the best of the pool as a seed and keep improving while time permits
    while time.monotonic() < deadline:
        # start from the current best with a mild random perturbation
        start = best_vec.copy()
        jitter = np.exp(rng.normal(0.0, 0.018, size=n))
        start = start * jitter
        start = _project_onto_simplex(start)

        refined, refined_score = _refine_sequence(
            start, n, fft_len, deadline, best_score, rng
        )

        if refined_score < best_score:
            best_score = refined_score
            best_vec = refined.copy()
            if best_score <= 1.50274349:
                break

        # occasional pure random restart if stagnating
        if rng.random() < 0.07:
            random_vec = rng.exponential(scale=1.0, size=n)
            random_vec = _project_onto_simplex(random_vec)
            refined, refined_score = _refine_sequence(
                random_vec, n, fft_len, deadline, best_score, rng
            )
            if refined_score < best_score:
                best_score = refined_score
                best_vec = refined.copy()
                if best_score <= 1.50274349:
                    break

    # fallback – uniform if something went wrong
    if best_vec is None or best_vec.size == 0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return best_vec.tolist()
