import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Full linear convolution via FFT (a*a)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Weighted gradient of the soft‑maxed convolution
# ------------------------------------------------------------
def _weighted_grad(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of sum_k w_k * (a*a)_k (w is a probability vector)."""
    n = a.shape[0]
    rev_w = w[::-1]                     # reverse for correlation
    L = 1 << ((3 * n - 2).bit_length())  # enough for len(a)+len(rev_w)-1
    fa = np.fft.rfft(a, n=L)
    fw = np.fft.rfft(rev_w, n=L)
    full = np.fft.irfft(fa * fw, n=L)[: 3 * n - 2]
    idx = (2 * n - 2) - np.arange(n)      # pick the right slice
    g = 2.0 * full[idx]
    return g
