def grad_max_conv(a):
    conv = conv_self_fft(a)  # length 2*n-1
    max_val = conv.max()
    k = conv.argmax()
    g = np.zeros_like(a)
    i_start = max(0, k-(n-1))
    i_end = min(n-1, k)
    # For i from i_start to i_end: g[i] = 2*a[k-i]
    idx = np.arange(i_start, i_end+1)
    g[idx] = 2.0 * a[k - idx]
    return g, max_val, k

import time
import numpy as np

# ---------- utilities ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_and_fft(a: np.ndarray, n: int, fft_len: int):
    """Full self‑convolution of a and its FFT (real‑valued)."""
    A = np.fft.rfft(a, n=fft_len)
    full = np.fft.irfft(A * A, n=fft_len)
    conv = full[: 2 * n - 1]
    return conv, A


def _smooth_grad(a: np.ndarray, A, conv: np.ndarray, beta: float,
                 n: int, fft_len: int):
    """Gradient of the soft‑max smoothed max‑convolution objective."""
    max_c = conv.max()
    # stable soft‑max weights over the convolution entries
    exp_vals = np.exp(beta * (conv - max_c))
    p = exp_vals / exp_vals.sum()
    P = np.fft.rfft(p, n=fft_len)
    # correlation a ⋆ p (linear, not circular)
    corr = np.fft.irfft(np.conj(A) * P, n=fft_len)
    g = 2.0 * corr[:n]          # ∂/∂a_i of the smoothed objective
    return g, p


def _initial_candidates(n: int, parent_arr):
    """Generate a handful of promising start vectors (sum = 1)."""
    cand = []

    # parent if supplied
    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            cand.append(p / s)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # sine envelope
    idx = np.arange(n, dtype=np.float64)
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    cand.append(sin_vals / sin_vals.sum())

    # quadratic “tent” shape
    base = (idx + 1) * (n - idx)
    cand.append(base / base.sum())

    # exponential decay
    exp_shape = np.exp(-0.5 * idx / n)
    cand.append(exp_shape / exp_shape.sum())

    # random Dirichlet‑like start
    rng = np.random.default_rng(12345)
    rnd = rng.random(n)
    cand.append(rnd / rnd.sum())

    return cand


# ---------- optimizer ----------

def _optimize_from(a0: np.ndarray,
                   n: int,
                   fft_len: int,
                   deadline: float,
                   target: float = 1.50274349) -> tuple[np.ndarray, float]:
    """Adam‑style projected optimisation with annealed soft‑max."""
    a = a0.copy()
    conv, A = _conv_and_fft(a, n, fft_len)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # annealing for soft‑max
    beta = 5.0
    beta_factor = 1.015
    eta = 0.5

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # smooth gradient
        g, _ = _smooth_grad(a, A, conv, beta, n, fft_len)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        # recompute convolution & its FFT
        conv, A = _conv_and_fft(a, n, fft_len)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if best_score <= target:
                break

        # anneal the temperature
        beta *= beta_factor

        # occasional random jitter to avoid stagnation
        if np.random.rand() < 0.01:
            jitter = np.exp(np.random.normal(0.0, 0.02, size=n))
            a = a * jitter
            a = _project_onto_simplex(a)
            conv, A = _conv_and_fft(a, n, fft_len)

    return best_a, best_score

    best_score = np.inf
    target_score = 1.50274349

    for vec in candidates:
        if time.monotonic() >= deadline:
            break

        vec = np.asarray(vec, dtype=np.float64)
        if vec.ndim != 1 or vec.size != n:
            continue
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s < 1e-12:
            continue
        a0 = vec / s

        # run the optimiser on this start
        a_opt, score_opt = _optimize_from(a0, n, fft_len, deadline, target_score)

        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            if best_score <= target_score:
                break

    # Fallback to uniform if nothing succeeded
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return best_vec.tolist()
