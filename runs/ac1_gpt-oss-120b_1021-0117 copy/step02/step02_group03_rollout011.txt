    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _max_convolution_fft(a: np.ndarray, L: int) -> float:
    """Maximum entry of full convolution a*a using an FFT of length L."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.shape[0] - 1]
    return float(conv.max())

    n = a.shape[0]
    # FFT length sufficient for convolutions of lengths n + (2n‑2)

    # --------------------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------------------
    best = a.copy()
    best_score = 2.0 * n * _max_convolution_fft(best, L)
    cur_score = best_score

    iteration = 0
    step0 = 0.20                     # initial step magnitude
    lam = 5.0                        # soft‑max temperature (low = smooth)
    lam_max = 200.0
    target = 1.5028                  # early‑stop threshold

    # --------------------------------------------------------------------
    # main loop – smoothed gradient descent with occasional random moves
    # --------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # full autocorrelation of current a
        fv = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        M = conv.max()

        # smooth weights approximating the max (soft‑max)
        w_exp = np.exp(lam * (conv - M))
        w = w_exp / w_exp.sum()

        # gradient = 2 * (a_rev * w) restricted to n entries
        fv_rev = np.fft.rfft(a[::-1], n=L)
        w_fft = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(fv_rev * w_fft, n=L)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # projected step
        norm = np.linalg.norm(grad)
        if norm > 0.0:
            eta = step0 / np.sqrt(iteration)
            cand = a - eta * (grad / norm)
        else:
            cand = a.copy()
        cand = _project_simplex(cand)

        # evaluate candidate using the same FFT trick (sum = 1 ⇒ score = 2n·max)
        fv_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(fv_c * fv_c, n=L)[: 2 * n - 1]
        cand_score = 2.0 * n * float(conv_c.max())

        # keep best ever seen
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            if best_score < target:

        if cand_score < cur_score or rng.random() < 0.005:
            a = cand
            cur_score = cand_score

                a[j] += delta
                a = _project_simplex(a)

        # slowly raise the soft‑max temperature
        lam = min(lam_max, lam * 1.001)

    # --------------------------------------------------------------------
    # return the best sequence found
    # --------------------------------------------------------------------
    return [float(x) for x in best]
