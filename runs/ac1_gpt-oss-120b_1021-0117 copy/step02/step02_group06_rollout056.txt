def compute_score_and_conv(a, n, fft_len):
    # returns score and entire conv array
    sum_a = a.sum()
    if sum_a <= 0:
        return np.inf, None
    conv = np.fft.irfft(np.fft.rfft(a, fft_len)**2, n=fft_len)[:2*n-1]
    max_b = conv.max()
    score = 2.0 * n * max_b / (sum_a**2)
    return score, conv
```

import time
import math
import numpy as np

# ---------- helper utilities ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2*len(a)-1)."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * a.size - 1]


def _candidate_shapes(n: int, rng: np.random.Generator):
    """Yield a handful of plausible probability vectors (sum≈1)."""
    idx = np.arange(n, dtype=np.float64)

    # 1. uniform
    yield np.full(n, 1.0 / n, dtype=np.float64)

    # 2. symmetric polynomial powers (U‑shaped and bell‑shaped)
    base = (idx + 1) * (n - idx)
    for p in (-0.9, -0.7, -0.5, -0.3, -0.1, 0.0, 0.5, 1.0, 1.5, 2.0):
        w = np.power(base + 1e-12, p)
        w = np.maximum(w, 0.0)
        s = w.sum()
        if s > 0:
            yield w / s

    # 3. sin‑power shapes
    theta = np.pi * (idx + 0.5) / n
    sinv = np.sin(theta)
    for p in (0.5, 1.0, 2.0, 3.0):
        w = sinv ** p
        s = w.sum()
        if s > 0:
            yield w / s

    # 4. random Dirichlet draws (to diversify)
    for _ in range(3):
        a = rng.gamma(0.5, 1.0, size=n)  # heavy‑tailed
        a = a / a.sum()
        yield a


def _score_of(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact score 2*n*max(b)/(sum(a)^2) with sum(a) close to 1."""
    sum_a = a.sum()
    if sum_a <= 0:
        return np.inf
    conv = _conv_self_fft(a, fft_len)
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a * sum_a)


def _grad_active(a: np.ndarray, conv: np.ndarray, act_idx: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max (or of the active set) w.r.t. a."""
    n = a.size
    g = np.zeros_like(a)
    for k in act_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    return g

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # shape candidates

        sc = _score_of(cand, n, fft_len)

        if best_score <= 1.50274349:            # early exit
            return best_vec.tolist()

    # ---- if nothing reasonable, fall back to uniform ---------------------------
    if best_vec is None:

    # ---- iterative refinement --------------------------------------------------
    a = best_vec.copy()
    iter_no_improve = 0
    iteration = 0
    eta0 = 0.3          # initial learning rate for multiplicative update
    step0 = 0.12        # additive step scale

        cur_score = 2.0 * n * max_b / (a.sum() * a.sum())
        if cur_score < best_score:

            iter_no_improve = 0

        if active_k >= conv.size:

        # ---- gradient of the active max term(s) -------------------------------
        g = _grad_active(a, conv, act_idx)

        eta = eta0 / math.sqrt(iteration)

            a = a_mul
            continue

        a_add = a - step * g

            pert = np.exp(rng.normal(0.0, 0.04, size=n))
            a = best_vec * pert
            a = a / a.sum()
            iter_no_improve = 0

    # final fallback
    return best_vec.tolist()
