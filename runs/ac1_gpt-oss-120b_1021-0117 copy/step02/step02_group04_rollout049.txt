Implementation function projection_simplex(v):

   v = np.asarray(v)
   n = v.shape[0]
   u = np.sort(v)[::-1]
   cssv = np.cumsum(u) - 1
   ind = np.arange(1, n+1)
   cond = u - cssv / ind > 0
   rho = ind[cond][-1]
   theta = cssv[cond][-1] / rho
   w = np.maximum(v - theta, 0)
   return w

import time
import math
import numpy as np

# ---------- utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound (same as the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _ac1_score_fft(cand, n, L)

        # ----- keep global best -----
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < 1.50274349:
                break

        # ----- simulated‑annealing style acceptance -----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # ----- schedules & occasional reset -----
        temperature *= temp_decay
        step_lr *= step_decay
        if rng.random() < 0.001:
            cur = best.copy()
            temperature = 0.6
            step_lr = 0.35

    # return as plain Python floats
    return [float(x) for x in best]
