import numpy as np

def _ac1_score(arr: np.ndarray) -> float:
    """Exact AC‑1 score using the official evaluator's formula."""
    n = arr.size
    conv = np.convolve(arr, arr)          # full convolution, length 2n‑1
    max_b = float(conv.max())
    sum_a = float(arr.sum())
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)
