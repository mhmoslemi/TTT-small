import time

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError
    # sort v decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, u.size + 1) > 0)[0]
    if rho.size == 0:
        # all zeros -> uniform
        return np.full_like(v, 1.0 / v.size)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        # 6.1  Soft‑max weights for a smooth max
        # --------------------------------------------------------------
        w_unnorm = np.exp(alpha * (conv - max_conv))   # max weight = 1
        w = w_unnorm / w_unnorm.sum()

        # --------------------------------------------------------------
        # 6.2  Gradient of the smooth max (cross‑correlation a ⋆ w)
        # --------------------------------------------------------------
        fw = np.fft.rfft(w, n=nfft)
        corr = np.fft.irfft(np.conj(fa) * fw, n=nfft)   # linear correlation
        grad = 2.0 * corr[n - 1:n - 1 + n]              # slice length n
        grad = np.real(grad)

        # --------------------------------------------------------------
        # 6.3  Back‑tracking line search on the simplex
        # --------------------------------------------------------------
        eta = eta0 / np.sqrt(it)
        improved = False
        for _ in range(5):
            a_try = _proj_simplex(a - eta * grad)
            fa_try = np.fft.rfft(a_try, n=nfft)
            conv_try = np.fft.irfft(fa_try * fa_try, n=nfft)[:conv_len]
            score_try = 2.0 * n * conv_try.max()
            if score_try <= cur_score:
                a = a_try
                improved = True
                break
            eta *= 0.5
        if not improved:
            # fallback: keep current a (already simplex‑feasible)
            pass

        # --------------------------------------------------------------
        # 6.4  Occasional multiplicative jitter to escape plateaus
        # --------------------------------------------------------------
        if it % 80 == 0:
            jitter = np.exp(rng.normal(0.0, 0.008, size=n))
            a = _proj_simplex(a * jitter)

        # --------------------------------------------------------------
        # 6.5  Adapt soft‑max temperature if we stall
        # --------------------------------------------------------------
        if stagnation > 200:
            alpha = min(max_alpha, alpha * 1.5)
            stagnation = 0

    # ------------------------------------------------------------------
    # 7.  Return the best sequence encountered
    # ------------------------------------------------------------------
    return [float(v) for v in best]
