import time
import math

import numpy as np

# ------------------------------------------------------------
# Helper utilities (small and self‑contained)
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n‑1 (classic binomial step)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast evaluation of the AC1 bound (identical to the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


# ------------------------------------------------------------
# Main search routine
# ------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that
    aims to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ----- problem size & optional parent ----------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        parent = None
        n = 2000                     # a reasonable fallback length

    L = _next_pow2(2 * n - 1)

    # ----- generate a few deterministic starters ----------
    candidates = []

    # 1. parent (if present)
    if parent is not None:
        candidates.append(parent.copy())

    # 2. classic binomial step‐function
    candidates.append(_gen_binomial(n))

    # 3. uniform (simple baseline)
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 4. exponential family – δ = 2/n (empirically good)
    delta = 2.0 / n
    r = 1.0 - delta
    exp_seq = delta * (r ** np.arange(n, dtype=np.float64))
    candidates.append(exp_seq)

    # pick the best of the deterministic starters
    best = None
    best_score = float("inf")
    for cand in candidates:
        sc = _ac1_score_fft(cand, n, L)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # ----- optimisation loop variables ----------
    cur = best.copy()
    cur_score = best_score

    temperature = 1.0

            cur_score = best_score

    # ------------------------------------------------------------
    # Return the best sequence found (plain Python list)
    # ------------------------------------------------------------
