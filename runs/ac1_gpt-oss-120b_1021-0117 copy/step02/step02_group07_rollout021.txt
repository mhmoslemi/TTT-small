import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _initial_binomial(nn: int) -> np.ndarray:
    """Monotone decreasing positive seed a_i = C(2i,i)/4^i."""
    a = np.empty(nn, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, nn):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _score_conv(seq: np.ndarray, fft_len: int):
    """Return (score, full convolution, max(conv))."""
    f = rfft(seq, n=fft_len)
    conv = irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    max_b = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_b
    score = 2.0 * seq.size * max_b / (s * s)
    return float(score), conv, max_b


def _gradient_step(seq: np.ndarray, conv: np.ndarray, kmax: int,
                   step_factor: float) -> np.ndarray:
    """One gradient descent step on the maximal convolution entry."""
    n = seq.size
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idxs = np.arange(lo, hi + 1)
    jdx = kmax - idxs
    grad = np.zeros_like(seq)
    # Correct derivative: 2 * a_{kmax-i} for every contributing pair
    grad[idxs] = 2.0 * seq[jdx]
    # Preserve total mass (zero‑mean direction)
    direction = -grad + grad.mean()
    step = step_factor * (seq.max() + 1.0)
    cand = seq + step * direction
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
