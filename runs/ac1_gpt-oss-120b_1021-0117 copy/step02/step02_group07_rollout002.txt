def _calc_conv(seq, fft_len):
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[:2 * seq.size - 1]
    return conv

def _score(seq, fft_len):
    conv = _calc_conv(seq, fft_len)
    maxb = conv.max()
    s = seq.sum()
    if s <= 0: return np.inf, conv, maxb
    score = 2.0 * seq.size * maxb / (s * s)
    return score, conv, maxb

import numpy as np
import time

def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 (positive decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_fft(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of seq with itself via FFT (real)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    return conv


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) beats the target bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)   # a safety margin

    # --------------------------------------------------------------
    # Acquire initial sequence: parent if provided, else binomial seed.
    # --------------------------------------------------------------
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)  # may be absent
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            init = np.clip(arr, 0.0, 1000.0)
        else:
            raise ValueError
    except Exception:
        init = None

    if init is None:
        n = 3809
        init = _initial_binomial(n)
    else:
        n = init.size

    fft_len = 1 << ((2 * n - 1).bit_length())

    # Ensure a minimal sum (required by the evaluator)
    if init.sum() < 0.01:
        init = np.maximum(init, 1e-5)
        init *= (np.sqrt(2 * n) / init.sum())

    # --------------------------------------------------------------
    def _score(seq: np.ndarray):
        conv = _conv_fft(seq, fft_len)
        maxb = conv.max()
        s = seq.sum()
        if s <= 0.0:
            return np.inf, conv, maxb
        return 2.0 * n * maxb / (s * s), conv, maxb
