import time
import math
import numpy as np

# ---------- utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n‑1 (classical binomial step)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 bound using FFT (same as evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


# ---------- main optimisation ----------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    # give a small safety margin for the outer sandbox
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ----- problem size & initial sequence -----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256

    L = _next_pow2(2 * n - 1)

    # start from the better of parent / classic binomial
    if parent is not None:
        base = parent.astype(np.float64)
        parent_score = _ac1_score_fft(parent, n, L)
        binom_score = _ac1_score_fft(_gen_binomial(n), n, L)
        if parent_score > binom_score:
            base = _gen_binomial(n)
    else:
        base = _gen_binomial(n)

    best = base.copy()
    best_score = _ac1_score_fft(best, n, L)

    cur = best.copy()
    cur_score = best_score

    # ----- parameters -----
    temperature = 1.0
    temp_decay = 0.9995

    step_lr = 0.25          # for gradient‑inspired (mirror) moves
    lr_decay = 0.9997

    transfer_frac = 0.4     # for targeted pair transfer
    transfer_decay = 0.9999

    idx_arr = np.arange(n, dtype=np.int64)

    # ----- main optimisation loop -----
    while time.monotonic() < deadline:
        # ---- evaluate current state ----
        fa_cur = np.fft.rfft(cur, n=L)
        conv_cur = np.fft.irfft(fa_cur * fa_cur, n=L)[: 2 * n - 1]
        max_b = float(conv_cur.max())
        k_max = int(conv_cur.argmax())
        sum_cur = float(cur.sum())
        cur_score = 2.0 * n * max_b / (sum_cur * sum_cur)

        # keep global best
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score
            if best_score < 1.50274349:
                break

        # ---- decide which move to apply ----
        move_type = rng.random()
        cand = cur.copy()

        if move_type < 0.55:
            # ---- targeted pair transfer (active lag) ----
            j = k_max - idx_arr
            valid = (j >= 0) & (j < n)
            if valid.any():
                prod = np.empty_like(cand)
                prod.fill(0.0)
                prod[valid] = cur[valid] * cur[j[valid]]
                # indices among the valid zone
                valid_idx = idx_arr[valid]
                p_valid = prod[valid]
                i_max = valid_idx[p_valid.argmax()]
                i_min = valid_idx[p_valid.argmin()]
                delta = cur[i_max] * transfer_frac
                if delta > 0.0:
                    cand[i_max] -= delta
                    cand[i_min] += delta

        elif move_type < 0.80:
            # ---- gradient‑inspired mirror move (as in previous baseline) ----
            t = max_b / sum_cur
            j = k_max - idx_arr
            mask = (j >= 0) & (j < n)
            mirror = np.zeros_like(cur)
            mirror[mask] = cur[j[mask]]
            C = 4.0 * n / (sum_cur * sum_cur)
            delta = step_lr * C * (t - mirror)
            cand = cur + delta

        elif move_type < 0.95:
            # ---- random block scaling ----
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand[start:start + width] *= factor

        else:
            # ---- random mass transfer between two positions ----
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * 0.4
                delta = cur[i] * frac
                cand[i] -= delta
                cand[j] += delta

        # ---- enforce bounds ----
        np.clip(cand, 0.0, 1000.0, out=cand)

        # ---- evaluate candidate ----
        fa_cand = np.fft.rfft(cand, n=L)
        conv_cand = np.fft.irfft(fa_cand * fa_cand, n=L)[: 2 * n - 1]
        max_b_cand = float(conv_cand.max())
        sum_cand = float(cand.sum())
        cand_score = 2.0 * n * max_b_cand / (sum_cand * sum_cand)

        # ---- simulated‑annealing acceptance ----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score

        # ---- schedules & occasional restart ----
        temperature *= temp_decay
        step_lr *= lr_decay
        transfer_frac *= transfer_decay

        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0
            step_lr = 0.25
            transfer_frac = 0.4

    # ----- return the best found sequence -----
    return [float(x) for x in best]
