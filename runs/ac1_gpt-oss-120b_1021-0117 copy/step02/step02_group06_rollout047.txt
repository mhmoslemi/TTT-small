import time
import numpy as np

# ---------- utility functions ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Full self‑convolution a*a using real FFT (length 2*n‑1)."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * n - 1]
