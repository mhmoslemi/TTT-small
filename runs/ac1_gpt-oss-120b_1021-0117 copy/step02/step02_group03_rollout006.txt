    else:

    # Compute sqrt series
    def sqrt_series_len(nn):
        a = np.empty(nn, dtype=np.float64)
        a[0] = 1.0
        for k in range(1, nn):
            if k == 1:
                s = 0.0
            else:
                slice_ = a[1:k]
                s = float(np.dot(slice_, slice_[::-1]))
            a[k] = (1.0 - s) * 0.5
        return a

    a = sqrt_series_len(n)

    # Clip negative
    a = np.maximum(a, 0.0)

    # Normalize to sum=1
    total = a.sum()
    if total <= 0:
        # Fallback to uniform
        a = np.full(n, 1.0 / n)
        total = 1.0
    else:
        a = a / total

    # Evaluate candidate
    best_seq = a.copy()

import time
import numpy as np

# ------------------------------------------------------------
# Generate the truncated square‑root series a such that
# (a * a)_k = 1 for 0 <= k < len(a).
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """Return the length‑n sqrt series (positive coefficients)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # sum_{i=1}^{k-1} a_i * a_{k-i}
            slice_ = a[1:k]                # a[1] … a[k‑1]
            s = float(np.dot(slice_, slice_[::-1]))
        a[k] = (1.0 - s) * 0.5
    return a
