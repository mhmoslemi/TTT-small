def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0) # 3 sec headroom

    # length n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2048  # default if not provided
    # define function to get initial a
    def init_random():
        # exponential random for positivity
        v = rng.exponential(1.0, size=n)
        v = _project_simplex(v)
        # optionally random permutation
        perm = rng.permutation(n)
        v = v[perm]
        return v

    # initial candidate
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        # ensure nonnegative and normalized
        a = np.maximum(a,0.0)
        a = _project_simplex(a)
    else:
        a = init_random()
    # variables for Adam
    m = np.zeros_like(a)
    v_acc = np.zeros_like(a)  # rename to avoid confusion with convolution vector v
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    beta1_pow = 1.0
    beta2_pow = 1.0

    # compute L for FFT
    L = 1 << ((2 * n - 1).bit_length())

    def conv_full(vec):
        fv = np.fft.rfft(vec, n=L)
        return np.fft.irfft(fv * fv, n=L)[:2 * len(vec) - 1]

    # evaluate initial
    conv = conv_full(a)
    max_val = conv.max()
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    iter_no = 0
    # main loop
    while time.monotonic() < deadline:
        iter_no += 1

        # compute full conv and top indices
        conv = conv_full(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # gradient from top k indices (k=5)
        top_k = 5
        if conv.size <= top_k:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(-conv, top_k-1)[:top_k]
        # compute gradient g
        g = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            g[i_range] += 2.0 * a[k - i_range]
        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v_acc = beta2 * v_acc + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v_acc / (1.0 - beta2_pow)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step_vec
        cand = _project_simplex(cand)

        # evaluate candidate
        conv_c = conv_full(cand)
        max_val_c = conv_c.max()
        cand_score = 2.0 * n * max_val_c

        # accept if better
        if cand_score < cur_score:
            a = cand
        else:
            # simulated annealing acceptance
            temp = lr # magnitude as temperature
            if rng.random() < np.exp(-(cand_score - cur_score) / max(temp, 1e-8)):
                a = cand

        # random mass transfer occasionally
        if rng.random() < 0.1:
            # choose source with probability proportional to a
            # ensure at least some mass
            source_idx = rng.choice(n, p=a / a.sum())
            if a[source_idx] > 1e-12:
                target_idx = rng.integers(0, n)
                if target_idx == source_idx:
                    target_idx = (target_idx + 1) % n
                frac = rng.random() * 0.3
                delta = a[source_idx] * frac
                cand2 = a.copy()
                cand2[source_idx] -= delta
                cand2[target_idx] += delta
                cand2 = _project_simplex(cand2)
                conv2 = conv_full(cand2)
                score2 = 2.0 * n * conv2.max()
                if score2 < cur_score:
                    a = cand2
                # else maybe accept with probability
        # restart if stuck: after many iterations with no improvement
        # we track last improvement iteration
        # (implement later)
        # early exit if time nearing deadline
        if time.monotonic() + 0.5 > deadline:
            break

    return [float(x) for x in best]

import time
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projects a vector onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic sqrt‑series (optional fallback initializer)
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ----------------------------------------------------------------------
# Full convolution of a vector with itself via FFT (real, linear)
# ----------------------------------------------------------------------
def _self_conv(vec: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(vec, n=L)
    return np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)

        a = _sqrt_series(256).astype(np.float64)
        a = _project_simplex(a)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())            # FFT length (power of two)

    # ------------------------------------------------------------------
    # Adam‑style optimiser state
    # ------------------------------------------------------------------
    m = np.zeros_like(a)        # first moment
    v_acc = np.zeros_like(a)    # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12                    # base learning‑rate
    beta1_pow, beta2_pow = 1.0, 1.0

    # ------------------------------------------------------------------
    # Helper to generate a fresh random starting point
    # ------------------------------------------------------------------
    def _random_start():
        # exponential ensures positivity; permutation decorrelates indices
        x = rng.exponential(1.0, size=n)
        x = _project_simplex(x)
        perm = rng.permutation(n)
        return x[perm]

    # ------------------------------------------------------------------
    # Initial evaluation
    # ------------------------------------------------------------------
    conv = _self_conv(a, L)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    no_improve = 0
    stall_limit = 300          # restart after this many non‑improving steps
    top_k = 5                  # number of convolution peaks to guide the gradient
    while time.monotonic() < deadline:
        # --- recompute convolution and the current score -----------------
        conv = _self_conv(a, L)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # --- keep global best -------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

                    a = cand2

        if time.monotonic() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # Return the best sequence found as plain Python floats
    # ------------------------------------------------------------------
    return [float(x) for x in best]
