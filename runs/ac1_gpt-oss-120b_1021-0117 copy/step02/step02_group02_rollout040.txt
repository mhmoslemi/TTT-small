import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):

    Starts from the provided parent (``height_sequence_1``) or a uniform fallback,
    repeatedly takes a sub‑gradient step for the current maximal convolution entry,
    projects back onto the simplex, and keeps the best vector seen.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #

    # ------------------------------------------------------------------ #
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    # ------------------------------------------------------------------ #

    try:

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------ #

    conv_len = 2 * n - 1

    # ------------------------------------------------------------------ #
    # 5.  Adam optimiser state
    # ------------------------------------------------------------------ #
    m = np.zeros_like(a)
    v = np.zeros_like(a)

            if best_score <= target_score:
