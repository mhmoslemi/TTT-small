import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Produce a non‑negative step‑function (simplex‑normalized) that (theoretically)
    minimizes the AC1 upper bound.  The uniform distribution is the
    provably optimal choice under the evaluator's constraints, but a few
    random simplex samples are examined in case the provided parent already
    beats it.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # Time limit (leave a few seconds slack for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Determine the length of the candidate (use the parent if present)
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise ValueError
        n = int(parent_arr.size)
    except Exception:
        n = 256  # fallback length

    # ------------------------------------------------------------------
    # Start from the uniform simplex (theoretical optimum)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best = uniform.copy()
    try:
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        best_score = float('inf')

    # ------------------------------------------------------------------
    # Occasionally sample random points on the simplex – cheap exploration
    # ------------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline and it < 2000:

        try:

        except Exception:
            sc = float('inf')
        if sc < best_score:
            best = cand.copy()
            best_score = sc
            # if we ever stumble on a better bound we keep it
        it += 1

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python floats)
    # ------------------------------------------------------------------
