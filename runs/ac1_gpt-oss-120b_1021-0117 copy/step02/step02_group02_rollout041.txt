    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        a = np.exp(log_a)                     # current probabilities
        # ----- convolution via FFT ------------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv         # sum(a)=1 ⇒ score = 2 n max_conv

        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
            # early stop if we reach the advertised target
            if best_score <= 1.50274349:
                break
        else:
            no_improve += 1

        # ----- gradient of the active (max) constraints ----------------
        # indices whose convolution value is essentially maximal
        active = np.where(conv >= max_conv - 1e-12)[0]

        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            partner = k - idx
            g[idx] += 2.0 * a[partner]          # ∂/∂a_i (a*a)[k] = 2 a_{k-i}

        # ----- mirror‑descent (exponentiated‑gradient) step ----------
        eta = eta0 / np.sqrt(it)                # diminishing step
        log_a -= eta * g

        # re‑centre log‑variables to avoid overflow/underflow
        log_a -= log_a.max()
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0 or not np.isfinite(a_sum):
            # fallback to uniform if something went wrong
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum
        a = np.maximum(a, eps)                  # keep strictly positive
        log_a = np.log(a)

        # ----- occasional jitter to escape plateaus ------------------
        if it % 75 == 0:
            jitter = np.exp(rng.normal(0.0, 0.008, size=n))
            a *= jitter
            a = a / a.sum()
            a = np.maximum(a, eps)
            log_a = np.log(a)

        # ----- random restart after long stagnation -------------------
        if no_improve >= 250:
            alpha = rng.uniform(0.2, 2.0)
            gsample = rng.gamma(alpha, 1.0, size=n)
            a = gsample / gsample.sum()
            a = np.maximum(a, eps)
            log_a = np.log(a)
            no_improve = 0

    # ------------------------------------------------------------------
    # 5.  Return the best sequence (plain Python floats)
    # ------------------------------------------------------------------
    return [float(v) for v in best.tolist()]
