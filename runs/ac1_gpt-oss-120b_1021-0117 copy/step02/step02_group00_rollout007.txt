def subgradient(a, k):
   n = len(a)
   g = np.zeros_like(a)
   lo = max(0, k - (n - 1))
   hi = min(k, n - 1)
   idx = np.arange(lo, hi+1)
   g[idx] = 2.0 * a[k - idx]   # since a[k - idx] may be reversed
   # check: when k - idx out of bounds? idx range ensures within [0,n-1].
   return g

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – returns a vector with sum=1, >=0
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Convolution via FFT (full linear convolution)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Score identical to evaluate_sequence (up to scaling)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_fft(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ------------------------------------------------------------
# Gradient of the active (maximal) convolution entry
# ------------------------------------------------------------
def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g[idx] = 2.0 * a[k - idx]
    return g
