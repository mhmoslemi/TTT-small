import math, time
import numpy as np
import scipy.optimize as opt

# ----- small helpers -------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n‑1 (classic binomial step)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound (identical to official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)   # tiny head‑room

    # ----- initialise sequence ---------------------------------------
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if a.ndim != 1 or a.size == 0:
            raise ValueError
        n = int(a.size)
    except Exception:
        n = 1024
        a = _gen_binomial(n)
    a = np.clip(a, 0.0, 1000.0)

    # ----- parameters ------------------------------------------------
    L = _next_pow2(2 * n - 1)
    best = a.copy()
    best_score = _ac1_score_fft(best, n, L)

    # block structure for LP (few variables, many constraints)
    B = min(128, n)                     # number of blocks, tuneable
    block_len = (n + B - 1) // B        # ceil division
    # masks: B x n, each row is indicator of a block
    masks = np.zeros((B, n), dtype=np.float64)
    for j in range(B):
        start = j * block_len
        end = min(start + block_len, n)
        masks[j, start:end] = 1.0
    # pre‑compute FFT of masks (matrix‑free, reused every LP)
    Fmask = np.fft.rfft(masks, n=L, axis=1)          # shape (B, L//2+1)

    # mixing schedule for the LP step
    mix_t = 0.35
    mix_decay = 0.998

    # ----- optimisation loop -----------------------------------------
    while time.monotonic() < deadline:
        # current self‑convolution and current max
        Fa = np.fft.rfft(a, n=L)
        conv_self = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
        M = float(conv_self.max())

        # ---------- LP: maximise sum(b)  s.t. a * b <= M ----------

        cols = np.fft.irfft(Fa[None, :] * Fmask, n=L)[:, : 2 * n - 1]   # (B, 2n‑1)

                width = int(rng.integers(max(1, n // 250), max(2, n // 15)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                factor = math.exp(rng.normal(0.0, 0.12))
                cand = a.copy()
                cand[start:start + width] *= factor
            elif move < 0.9:                         # mass transfer
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j:
                    frac = rng.random() * 0.35
                    delta = a[i] * frac
                    cand = a.copy()
                    cand[i] -= delta
                    cand[j] += delta
                else:
                    cand = a.copy()
            else:                                    # full Gaussian perturbation
                noise = rng.normal(0.0, a.mean() * 0.03, size=n)
                cand = a + noise
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_score = _ac1_score_fft(cand, n, L)
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
            a = cand

        # occasional restart to the best so far (helps escape dead‑ends)
        if rng.random() < 0.001:
            a = best.copy()
            mix_t = 0.35

    # return as plain Python list
    return [float(x) for x in best]
