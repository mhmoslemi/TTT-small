def _gradient_softmax(a, conv, beta, top_k=20):
    # conv array length 2n-1
    n = a.shape[0]
    max_b = conv.max()
    # Get top k indices
    if top_k >= len(conv):
        idx = np.arange(len(conv))
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
    # Compute weights
    diff = conv[idx] - max_b  # <=0
    # Use exp(beta * diff) -> stable.
    w = np.exp(beta * diff)
    # Avoid all zero
    if w.sum() == 0:
        w = np.ones_like(w)
    p = w / w.sum()
    # Gradient init zero
    g = np.zeros_like(a)
    for k, pk in zip(idx, p):
        # indices for i
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # Compute slice vectorized
        # i_range = np.arange(i_start, i_end+1)
        # Corresponding a_val = a[k - i_range]
        # g[i_range] += 2 * pk * a_val
        # Use vectorized operations
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        g[i_range] += 2.0 * pk * a[k - i_range]
    return g

import time
import numpy as np

# ---------- utilities ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, n: int, fft_len: int) -> np.ndarray:
    """Full convolution a*a using real FFT (length 2*n-1)."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * n - 1]


def _catalan_weights(n: int) -> np.ndarray:
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


def _initial_candidates(n: int, parent_arr):

    # parent (if available)
    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)

        if s > 0.0:

def _gradient_softmax(a: np.ndarray, conv: np.ndarray, beta: float, top_k: int) -> np.ndarray:
    """Gradient of the smooth max (log‑sum‑exp) of the convolution."""
    n = a.shape[0]
    max_b = conv.max()
    # pick top‑k indices (or all if small)
    if top_k >= conv.size:
        idx = np.arange(conv.size)
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
    # weights proportional to exp(beta*(b_i-max))
    w = np.exp(beta * (conv[idx] - max_b))
    if w.sum() == 0.0:
        w = np.ones_like(w)
    p = w / w.sum()
    g = np.zeros_like(a)
    for k, pk in zip(idx, p):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_start > i_end:
            continue
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        g[i_range] += 2.0 * pk * a[k - i_range]
    return g
