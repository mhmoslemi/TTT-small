import numpy as np
import time
import math

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_and_score(seq: np.ndarray, fft_len: int):
    """Return (score, full convolution array)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    max_b = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv
    score = 2.0 * seq.size * max_b / (s * s)
    return float(score), conv

    # ------------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Initialise best / current candidate
    # ------------------------------------------------------------------
    best_seq = seq.copy()
    best_score, best_conv = _conv_and_score(best_seq, fft_len)
    current_seq = best_seq.copy()
    current_score = best_score

    # ------------------------------------------------------------------
    # Helper: gradient of the maximal convolution entry
    # ------------------------------------------------------------------
    def _gradient_at_max(conv: np.ndarray, a: np.ndarray):
        """Gradient of the convolution entry attaining the current maximum."""
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idxs = np.arange(lo, hi + 1, dtype=np.int64)
        js = kmax - idxs
        grad = np.zeros_like(a)
        # derivative of a_i a_{kmax-i} contributes 2 * a_{kmax-i}
        grad[idxs] = 2.0 * a[js]
        return grad, kmax

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        # current convolution & max location
        cur_score, cur_conv = _conv_and_score(current_seq, fft_len)
        # gradient of the maximal entry
        grad, kmax = _gradient_at_max(cur_conv, current_seq)

        # Preserve total mass (sum) by forcing zero‑mean direction
        mean_g = grad.mean()
        direction = -(grad - mean_g)               # sum(direction) == 0

        # Adaptive step size (decays slowly with iterations)
        base_step = 0.02 * (current_seq.max() + 1.0)
        decay = 0.5 ** (iteration / 3000.0)        # ~ halving every 3k steps
        step = base_step * decay

        cand = current_seq + step * direction
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score, _ = _conv_and_score(cand, fft_len)

        # Simple simulated‑annealing acceptance
        elapsed = time.monotonic() - start
        total = deadline - start
        temp = max(0.0, 0.08 * (1.0 - elapsed / total))  # max temperature ≈0.08
        accept = False
        if cand_score < current_score:
            accept = True
        elif temp > 0.0 and rng.random() < math.exp((current_score - cand_score) / (temp + 1e-12)):
            accept = True

        if accept:
            current_seq = cand
            current_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                best_conv = _conv_and_score(best_seq, fft_len)[1]

        # ------------------------------------------------------------------
        # Occasional random perturbation to escape plateaus
        # ------------------------------------------------------------------
        if rng.random() < 0.07:
            width = rng.integers(1, max(2, n // 25))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.09))
            pert = current_seq.copy()
            pert[start_idx:start_idx + width] *= factor
            np.clip(pert, 0.0, 1000.0, out=pert)
            pert_score, _ = _conv_and_score(pert, fft_len)
            if pert_score < best_score:
                best_seq = pert.copy()
                best_score = pert_score
                current_seq = pert
                current_score = pert_score

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
