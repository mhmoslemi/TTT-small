import numpy as np
import time

def _proj_simplex(v):
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, len(v)+1)) > 0)[0]
    if rho.size == 0:
        # fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(1, v.size + 1))
    mask = rho_arr > 0
    if not np.any(mask):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
