    best = a.copy()
    best_score = evaluate(a)
    grad = np.empty_like(a)
    for t in range(max_iter):
        if time.monotonic()> deadline: break
        # compute conv and max index
        conv = conv_fft(a)
        k = conv.argmax()
        # gradient
        grad.fill(0.0)
        if k < n:
            grad[:k+1] = 2 * a[k::-1]
        else:
            start = k - (n-1)
            length = 2*n-1 - k
            grad[start:] = 2 * a[:length][::-1]
        # step size
        eta = 5.0 / np.sqrt(t+1 + 1e-12)
        # update
        a = a * np.exp(-eta * grad)
        # avoid all zeros
        if a.sum() == 0:
            a = np.full_like(a, 1.0/a.size)
        a = a / a.sum()
        # evaluate
        score = evaluate(a)
        if score < best_score:
            best = a.copy()
            best_score = score
    return best, best_score

# Helper: simple decreasing binomial seed (fallback if no parent)

def _initial_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i for i=0..n-1, positive decreasing."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a
