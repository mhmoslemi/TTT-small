def gradient_step(a, eps, delta=0.02):
    # compute conv
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    maxv = conv.max()
    ks = np.where(conv >= maxv * (1 - delta))[0]
    g = np.zeros_like(a)
    for k in ks:
        idx = np.arange(len(a))
        rev = k - idx
        mask = (rev >= 0) & (rev < len(a))
        # contributions
        contrib = a[rev[mask]]
        # double for distinct
        same = (rev[mask] != idx[mask])
        contrib = np.where(same, contrib*2, contrib)
        g[mask] += contrib
    # average contributions
    g /= len(ks)
    # compute direction preserving sum
    d = g.mean() - g
    cand = a + eps * d
    # clip
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _gradient_move(a, eps, L, n, delta=0.02):
    # compute FFT conv and gradient contributions
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    maxb = conv.max()
    # threshold
    thresh = maxb * (1.0 - delta)
    ks = np.where(conv >= thresh)[0]
    # accumulate contributions
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in ks:
        rev = k - idx
        mask = (rev >= 0) & (rev < n)
        # contributions
        contrib = a[rev[mask]]
        # double where distinct
        distinct = rev[mask] != idx[mask]
        contrib = np.where(distinct, contrib * 2.0, contrib)
        g[mask] += contrib
    g /= len(ks)
    d = g.mean() - g
    cand = a + eps * d
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand, maxb

import numpy as np, time, math

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound via FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def _gradient_move(a: np.ndarray, eps: float, L: int, n: int, delta: float = 0.02) -> np.ndarray:
    """Move a a little opposite to the gradient of the current maximum convolution peaks."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    maxb = conv.max()
    thresh = maxb * (1.0 - delta)
    ks = np.where(conv >= thresh)[0]

    # average gradient over selected peaks
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in ks:
        rev = k - idx
        mask = (rev >= 0) & (rev < n)
        # contribution from partner entries
        contrib = a[rev[mask]]
        distinct = rev[mask] != idx[mask]
        contrib = np.where(distinct, contrib * 2.0, contrib)
        g[mask] += contrib
    g /= len(ks)

    # direction preserving total mass
    d = g.mean() - g
    cand = a + eps * d
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
