def _proj_simplex(v):
    # Project onto simplex {x >=0, sum(x)=1}
    # using algorithm from https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero?
        return np.maximum(v, 0) / max(np.sum(v), 1e-12)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import math
import time

import numpy as np

# ---------- small utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """Classic binomial step: a[i] = C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound (identical to the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


# ---------- main search ----------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    # leave a small head‑room for returning the result safely
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ----- initial sequence -----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        cur = np.maximum(parent, 0.0)
    except Exception:
        n = 256
        cur = _gen_binomial(n)

    # normalise to a simplex (sum = 1) – scaling does not affect the AC1 score
    cur_sum = cur.sum()
    if cur_sum <= 0:
        cur = np.ones_like(cur) / cur.size
    else:
        cur = cur / cur_sum

    # pre‑compute FFT length
    L = _next_pow2(2 * n - 1)

    # best so far
    best = cur.copy()
    best_score = _ac1_score_fft(best, n, L)

    # current state
    cur_score = best_score

    # annealing / step‑size schedules
    temperature = 1.0
    temp_decay = 0.9995
    lr = 0.05          # learning‑rate for the reflective move
    lr_decay = 0.9999

    # ----- optimisation loop -----
    while time.monotonic() < deadline:
        # evaluate current convolution, locate active lag
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_b = float(conv.max())
        k_max = int(conv.argmax())
        # current AC1 score (sum is 1)
        cur_score = 2.0 * n * max_b

        # ---- propose a candidate via a random move type ----
        move_type = rng.random()
        cand = None

        if move_type < 0.40:
            # reflective (gradient‑inspired) move
            t = max_b                      # because sum(cur)=1
            idx = np.arange(n)
            j = k_max - idx
            mask = (j >= 0) & (j < n)
            mirror = np.zeros_like(cur)
            mirror[mask] = cur[j[mask]]
            C = 4.0 * n                     # derivative factor (sum² = 1)
            delta = lr * C * (t - mirror)
            cand = cur + delta

        elif move_type < 0.70:
            # random block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start:start + width] *= factor

        elif move_type < 0.90:
            # targeted mass transfer (active lag)
            idx = np.arange(n)
            j = k_max - idx
            mask = (j >= 0) & (j < n)
            contrib = np.zeros_like(cur)
            contrib[mask] = cur[mask] * cur[j[mask]]
            i_max = int(np.argmax(contrib))
            i_target = int(rng.integers(0, n))
            while i_target == i_max:
                i_target = int(rng.integers(0, n))
            frac = rng.random() * 0.4
            delta = cur[i_max] * frac
            cand = cur.copy()
            cand[i_max] -= delta
            cand[i_target] += delta

        else:
            # smoothing move (tiny Gaussian blur)
            sig = rng.random() * 1.5 + 0.1          # 0.1 … 1.6
            half = int(min(3, max(1, sig * 2)))
            ks = np.arange(-half, half + 1)
            kernel = np.exp(-0.5 * (ks / sig) ** 2)
            kernel /= kernel.sum()
            cand = np.convolve(cur, kernel, mode='same')

        # ----- postprocess candidate -----
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s <= 0:
            continue
        cand = cand / s                     # re‑normalise to the simplex

        cand_score = _ac1_score_fft(cand, n, L)

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < 1.50274349:
                break

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # schedule updates
        temperature *= temp_decay
        lr *= lr_decay

        # occasional reset to the best known solution
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0
            lr = 0.05

    # ----- return the best sequence -----
    # values are already non‑negative and sum to 1, well inside [0,1000]
    return [float(x) for x in best]
