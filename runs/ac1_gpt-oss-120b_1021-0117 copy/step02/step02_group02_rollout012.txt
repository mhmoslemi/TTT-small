def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function that minimises the AC1 upper bound
    as evaluated by ``evaluate_sequence``.
    The theoretical optimum is a constant (uniform) vector, which gives a
    score of exactly 2 for any length.  The function therefore builds a uniform
    candidate of the appropriate length, compares it against the supplied
    ``height_sequence_1`` (if any), and returns the better one.
    """
    # ------------------------------------------------------------------
    # 1.  Determine length from the optional parent sequence
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
        n = int(parent_arr.size)
    except Exception:
        # fallback length – small enough to stay well within the budget
        n = 256
        parent_arr = None

    # ------------------------------------------------------------------
    # 2.  Build a uniform (constant) candidate – any constant works,
    #     we choose 1.0 and clip to the allowed range.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0, dtype=np.float64)
    uniform = np.clip(uniform, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # 3.  Evaluate uniform and (if present) the parent; keep the best.
    # ------------------------------------------------------------------
    best = uniform
