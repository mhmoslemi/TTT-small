import time
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projects v onto the simplex (non‑negative, sums to 1)."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
# Analytic sqrt‑series (coefficients a with a*a = 1+…+x^{2n‑2})
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

    else:
        # fallback: analytic sqrt‑series of a modest length
        n_fallback = 2000
        a = _sqrt_series(n_fallback)

    n = a.size
    # --------------------------------------------------------------
    # Normalise onto the simplex (sum = 1, all entries non‑negative)
    # --------------------------------------------------------------
    a = _project_simplex(a)

    # ------------------------------------------------------------------
    # parameters
    # ------------------------------------------------------------------
    step = 0.5               # step size for projected gradient
    min_step = 1e-5
    decay = 0.9995
    prob_rand = 0.02         # probability of a random mass‑transfer move
    max_rand_frac = 0.4      # how much of the source mass may be moved

    # soft‑max temperature for smoothing the max‑convolution term
    beta = 0.5
    beta_inc = 1.005
    beta_max = 12.0

    # FFT lengths (powers of two, enough for the required convolutions)
    conv_len = 2 * n - 1
    L_conv = 1 << ((conv_len - 1).bit_length())
    grad_len = 3 * n - 2
    L_grad = 1 << ((grad_len - 1).bit_length())

    # ------------------------------------------------------------------
    # bookkeeping of the best solution seen so far
    # ------------------------------------------------------------------
    # compute initial score
    fv = np.fft.rfft(a, n=L_conv)
    conv_full = np.fft.irfft(fv * fv, n=L_conv)[:conv_len]
    cur_max = float(conv_full.max())
    cur_score = 2.0 * n * cur_max
    best = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # recompute full convolution (needed for gradient & weights)
        # --------------------------------------------------------------
        fv = np.fft.rfft(a, n=L_conv)
        conv_full = np.fft.irfft(fv * fv, n=L_conv)[:conv_len]
        cur_max = float(conv_full.max())
        cur_score = 2.0 * n * cur_max

        # update best record
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            # early stop if we already beat the target
            if best_score < 1.5028:
                break

        # --------------------------------------------------------------
        # soft‑max weighting of convolution entries (smooth max)
        # --------------------------------------------------------------
        exp_vals = np.exp(beta * (conv_full - cur_max))      # shift for stability
        w = exp_vals / exp_vals.sum()                       # shape (conv_len,)

        # --------------------------------------------------------------
        # gradient of the smooth objective: g = 2 * (a_rev * w)
        # --------------------------------------------------------------
        a_rev = a[::-1]
        fr = np.fft.rfft(a_rev, n=L_grad)
        fw = np.fft.rfft(w, n=L_grad)
        rev_conv = np.fft.irfft(fr * fw, n=L_grad)[:grad_len]
        grad = 2.0 * rev_conv[n - 1 : n - 1 + n]

        # --------------------------------------------------------------
        # projected gradient step
        # --------------------------------------------------------------
        cand = a - step * grad
        cand = _project_simplex(cand)

        # evaluate candidate
        fv2 = np.fft.rfft(cand, n=L_conv)
        conv_cand = np.fft.irfft(fv2 * fv2, n=L_conv)[:conv_len]
        cand_score = 2.0 * n * float(conv_cand.max())

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
        else:
            # occasional uphill acceptance (simulated‑annealing flavour)
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-8)):
                a = cand
                cur_score = cand_score

        # --------------------------------------------------------------
        # random mass‑transfer move for extra exploration
        # --------------------------------------------------------------
        if rng.random() < prob_rand:
            i = rng.integers(0, n)
            if a[i] > 1e-14:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * max_rand_frac
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)

                fv3 = np.fft.rfft(cand2, n=L_conv)
                conv_cand2 = np.fft.irfft(fv3 * fv3, n=L_conv)[:conv_len]
                cand2_score = 2.0 * n * float(conv_cand2.max())

                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()

        # --------------------------------------------------------------
        # parameter updates
        # --------------------------------------------------------------
        step = max(min_step, step * decay)
        beta = min(beta_max, beta * beta_inc)

        # cheap early exit if time is about to expire
        if time.monotonic() + 0.5 > deadline:
            break

    # --------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # --------------------------------------------------------------
    return [float(x) for x in best]
