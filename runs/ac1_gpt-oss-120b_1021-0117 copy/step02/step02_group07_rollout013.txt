import numpy as np, time

def _initial_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def _conv_fft(seq, fft_len):
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[:2*seq.size - 1]
    return conv

def _score_fft(seq, fft_len):
    conv = _conv_fft(seq, fft_len)
    max_b = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return np.inf, conv, max_b
    return 2.0 * seq.size * max_b / (s*s), conv, max_b

def _random_move(seq, rng):
    n = seq.size
    cand = seq.copy()
    op = rng.integers(0,5)
    if op == 0:  # block scaling
        width = rng.integers(1, max(2, n//30))
        start = rng.integers(0, n - width + 1)
        factor = np.exp(rng.normal(0.0, 0.07))
        cand[start:start+width] *= factor
    elif op == 1:  # single index scaling
        idx = rng.integers(0, n)
        factor = np.exp(rng.normal(0.0, 0.2))
        cand[idx] *= factor
    elif op == 2:  # early to late transfer
        early_end = max(1, n//12)
        late_start = n - early_end
        i = rng.integers(0, early_end)
        j = rng.integers(late_start, n)
        frac = rng.uniform(0.07, 0.22)
        delta = cand[i] * frac
        cand[i] -= delta
        cand[j] += delta
    elif op == 3:  # smooth neighbours
        i = rng.integers(0, n-1)
        avg = (cand[i] + cand[i+1]) * 0.5
        cand[i] = avg
        cand[i+1] = avg
    else:  # Gaussian jitter
        scale = 0.005 * np.max(seq) + 1e-6

    s1 = np.sum(cand)

import numpy as np, time

def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _conv_fft(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution via real FFT (length = 2*len(seq)-1)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    return conv

def _score_fft(seq: np.ndarray, fft_len: int):
    """Return (score, full convolution, max(conv))."""
    conv = _conv_fft(seq, fft_len)
    max_b = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return np.inf, conv, max_b
    return 2.0 * seq.size * max_b / (s * s), conv, max_b

def _random_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Small stochastic perturbation that roughly preserves the total sum."""
    n = seq.size
    cand = seq.copy()
    op = rng.integers(0, 5)
    if op == 0:                         # block scaling
        width = rng.integers(1, max(2, n // 30))
        start = rng.integers(0, n - width + 1)
        factor = np.exp(rng.normal(0.0, 0.07))
        cand[start:start + width] *= factor
    elif op == 1:                       # single index scaling
        idx = rng.integers(0, n)
        factor = np.exp(rng.normal(0.0, 0.20))
        cand[idx] *= factor
    elif op == 2:                       # early-to-late mass transfer
        early_end = max(1, n // 12)
        late_start = n - early_end
        i = rng.integers(0, early_end)
        j = rng.integers(late_start, n)
        frac = rng.uniform(0.07, 0.22)
        delta = cand[i] * frac
        cand[i] -= delta
        cand[j] += delta
    elif op == 3:                       # smooth two neighbours
        i = rng.integers(0, n - 1)
        avg = (cand[i] + cand[i + 1]) * 0.5
        cand[i] = avg
        cand[i + 1] = avg
    else:                               # tiny Gaussian jitter
        scale = 0.005 * np.max(seq) + 1e-6
        cand += rng.normal(0.0, scale, size=n)

    # renormalise to keep the denominator unchanged
    s0 = np.sum(seq)
    s1 = np.sum(cand)
    if s1 > 0:
        cand *= s0 / s1
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
