import numpy as np

def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _score_conv(seq: np.ndarray, fft_len: int):
    """Return (score, full convolution array, max value)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    max_b = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_b
    score = 2.0 * seq.size * max_b / (s * s)
    return float(score), conv, max_b

        elif op == 1:  # single index scaling
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.20))
            cand[idx] *= factor

        elif op == 2:  # move a fraction from the pair contributing to max conv
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            i = rng.integers(lo, hi + 1)
            j = k_max - i
            if i != j:
                frac = rng.uniform(0.04, 0.15)
                di = cand[i] * frac
                dj = cand[j] * frac
                cand[i] -= di
                cand[j] -= dj
                # dump the removed mass somewhere else
                tgt = rng.integers(0, n)
                while tgt == i or tgt == j:
                    tgt = rng.integers(0, n)
                cand[tgt] += di + dj
            else:
                # if the same index, just shrink it a little
                cand[i] *= np.exp(rng.normal(0.0, 0.09))

        elif op == 3:  # smooth two neighbours (averaging)
            i = rng.integers(0, n - 1)
            avg = (cand[i] + cand[i + 1]) * 0.5
            cand[i] = avg
            cand[i + 1] = avg

        elif op == 4:  # early‑to‑late mass transfer
            early_len = max(1, n // 10)
            late_start = n - early_len
            i = rng.integers(0, early_len)
            j = rng.integers(late_start, n)
            frac = rng.uniform(0.05, 0.25)
            delta = cand[i] * frac
            cand[i] -= delta
            cand[j] += delta

        elif op == 5:  # flatten: push down max, lift up min
            # gradient for max lag
            lo_m = max(0, k_max - (n - 1))
            hi_m = min(k_max, n - 1)
            idxs_m = np.arange(lo_m, hi_m + 1, dtype=np.int64)
            js_m = k_max - idxs_m
            grad_max = np.zeros_like(cand)
            grad_max[idxs_m] = 2.0 * cand[js_m]

            # gradient for min lag
            lo_l = max(0, k_min - (n - 1))
            hi_l = min(k_min, n - 1)
            idxs_l = np.arange(lo_l, hi_l + 1, dtype=np.int64)
            js_l = k_min - idxs_l
            grad_min = np.zeros_like(cand)
            grad_min[idxs_l] = 2.0 * cand[js_l]

            direction = -grad_max + grad_min
            direction -= np.mean(direction)  # keep sum zero

            # line‑search over a few step sizes
            step_base = rng.uniform(0.001, 0.02) * (np.max(cand) + 1.0)
            best_local = None
            best_local_score = float("inf")
            for mul in (0.5, 1.0, 2.0):
                step = step_base * mul
                trial = cand + step * direction
                np.clip(trial, 0.0, 1000.0, out=trial)
                sc, _, _ = _score_conv(trial, fft_len)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = trial
            cand = best_local if best_local is not None else cand

        else:  # op == 6 : no operation, just keep current
            pass

        # enforce bounds (for ops that did not already clip)
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate candidate
        cand_score, cand_conv, _ = _score_conv(cand, fft_len)

        # simulated‑annealing acceptance (temperature linearly decreasing)
        elapsed = time.monotonic() - start
        total = deadline - start
        temp = max(0.0, 0.5 * (1.0 - elapsed / total))
        accept = False
        if cand_score < current_score:
            accept = True
        elif temp > 0.0 and rng.random() < np.exp((current_score - cand_score) / (temp + 1e-12)):
            accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                best_conv = cand_conv

        # early exit if we already beat the target
        if best_score < 1.50274349:
            break

    # final sanity clamp and return as plain list
    np.clip(best, 0.0, 1000.0, out=best)
    return [float(x) for x in best]
