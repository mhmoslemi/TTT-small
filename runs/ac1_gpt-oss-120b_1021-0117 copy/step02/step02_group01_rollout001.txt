import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

def _sqrt_series(n: int) -> np.ndarray:
    """Analytic sqrt‑series a with a*a = 1+…+x^{2n‑2} (fallback initializer)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        s = 0.0 if k == 1 else float(np.dot(a[1:k], a[k-1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a

def _conv_full(vec: np.ndarray, L: int) -> np.ndarray:
    """Full convolution of vec with itself via FFT (length 2*len-1)."""
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    return conv[: 2 * len(vec) - 1]

def _conv_max(vec: np.ndarray, L: int):
    """Maximum value and index of autocorrelation of vec."""
    conv = _conv_full(vec, L)
    idx = int(conv.argmax())
    return float(conv[idx]), idx

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # initialization
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = _sqrt_series(256)
    a = _project_simplex(a)
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # initial evaluation
    max_val, max_idx = _conv_max(a, L)
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # parameters
    # ------------------------------------------------------------------
    step = 0.3
    min_step = 1e-6
    decay = 0.9995
    active_tol_rel = 1e-4      # relative tolerance for active constraints
    max_active = 5             # limit number of active indices
    random_move_prob = 0.02
    line_factors = (0.5, 1.0, 2.0)

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # compute full autocorrelation, find max and active set
        conv_full = _conv_full(a, L)
        max_idx = int(conv_full.argmax())
        max_val = float(conv_full[max_idx])

        eps = max_val * active_tol_rel + 1e-12
        act = np.where(conv_full >= max_val - eps)[0]
        if act.size > max_active:
            # keep only the largest values among the active set
            top = np.argpartition(conv_full[act], -max_active)[-max_active:]
            act = act[top]

        # gradient (subgradient of max convolution)
        g = np.zeros_like(a)
        for k in act:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            j = np.arange(lo, hi + 1)
            g[j] += 2.0 * a[k - j]

        # if gradient vanished, break (unlikely)
        if np.allclose(g, 0):
            break

        # normalize direction to control step magnitude
        norm_g = np.linalg.norm(g)
        direction = g / norm_g if norm_g > 0 else g

        # line‑search over a few step multipliers
        cand_best = None
        cand_best_score = np.inf
        cand_best_max_idx = None
        for factor in line_factors:
            sz = max(step * factor, min_step)
            cand = a - sz * direction
            cand = _project_simplex(cand)
            max_val_c, max_idx_c = _conv_max(cand, L)
            cand_score = 2.0 * n * max_val_c
            if cand_score < cand_best_score:
                cand_best = cand
                cand_best_score = cand_score
                cand_best_max_idx = max_idx_c

        # accept improvement
        if cand_best_score < cur_score:
            a = cand_best
            cur_score = cand_best_score
            max_idx = cand_best_max_idx
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # occasional uphill move (simulated annealing style)
            if rng.random() < np.exp(-(cand_best_score - cur_score) / max(step, 1e-8)):
                a = cand_best
                cur_score = cand_best_score
                max_idx = cand_best_max_idx

        # random mass transfer to diversify search
        if rng.random() < random_move_prob:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.3
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                max_val2, max_idx2 = _conv_max(cand2, L)
                cand2_score = 2.0 * n * max_val2
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    max_idx = max_idx2
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()

        # decay step size
        step = max(min_step, step * decay)

        # early termination guard
        if time.monotonic() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # return best candidate as plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best]
