def _project_onto_simplex(v):
    # from previous code
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
    except Exception:
        parent_arr = None

    if parent_arr is not None:
        n = parent_arr.shape[0]
    else:
        n = 1024

    # define FFT length
    fft_len = 1 << ((3 * n - 2).bit_length())

    def conv_fft(x, y):
        X = np.fft.rfft(x, n=fft_len)
        Y = np.fft.rfft(y, n=fft_len)
        z = np.fft.irfft(X * Y, n=fft_len)
        return z[:len(x) + len(y) - 1]

    def objective(a):
        conv = conv_fft(a, a)
        return 2.0 * n * conv.max()

    # projection onto simplex
    def project_simplex(v):
        # Standard Euclidean projection onto simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(1, len(v) + 1)
        cond = u - cssv / ind > 0
        if not np.any(cond):
            # all non-positive
            return np.full_like(v, 1.0 / len(v))
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    # generate initial candidates
    candidates = []

    # parent
    if parent_arr is not None:
        p = np.clip(parent_arr, 0.0, 1000.0)
        total = p.sum()
        if total > 0:
            candidates.append(p / total)

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # triangular shape (i+1)*(n-i)
    idx = np.arange(n, dtype=np.float64)
    tri = (idx + 1) * (n - idx)
    candidates.append(tri / tri.sum())

    # sine powers
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    for p in (0.5, 1.0, 2.0):
        v = sin_vals ** p
        candidates.append(v / v.sum())

    # exponential shapes (geometric)
    for c in (0.5, 1.0, 1.5, 2.0):
        r = np.exp(-c / n)
        geom = r ** idx
        candidates.append(geom / geom.sum())

    # random Dirichlet samples
    for _ in range(3):
        rand_vec = rng.gamma(1.0, 1.0, size=n)
        candidates.append(rand_vec / rand_vec.sum())

    # evaluate candidates
    best_vec = None
    best_score = np.inf
    for a in candidates:
        sc = objective(a)
        if sc < best_score:
            best_score = sc
            best_vec = a.copy()
        if best_score <= 1.50274349:
            return best_vec.tolist()

    # gradient descent
    a = best_vec.copy()
    v_momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.7
    alpha_smooth = 120.0  # smoothing parameter (log-sum-exp)
    iter_count = 0
    no_improve = 0
    max_no_improve = 800

    while time.monotonic() < deadline:
        # compute convolution and objective for current a
        conv = conv_fft(a, a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            no_improve = 0
            if best_score <= 1.50274349:
                break
        else:
            no_improve += 1

        # compute smooth weights
        # stable scaling
        w_raw = np.exp(alpha_smooth * (conv - max_b))
        w = w_raw / w_raw.sum()

        # gradient of smooth objective
        w_rev = w[::-1]
        grad_full = conv_fft(a, w_rev)
        grad = 2.0 * grad_full[::-1][:n]

        # adaptive step size
        step = step0 / np.sqrt(iter_count + 1)

        # momentum update
        v_momentum = beta * v_momentum - step * grad
        cand = a + v_momentum
        cand = project_simplex(cand)

        # occasional random jitter to avoid stagnation
        if rng.random() < 0.01:
            jitter = np.exp(rng.normal(0.0, 0.015, size=n))
            cand = project_simplex(cand * jitter)

        a = cand

        # restart if stuck
        if no_improve > max_no_improve:
            # perturb best solution
            pert = np.exp(rng.normal(0.0, 0.03, size=n))
            a = project_simplex(best_vec * pert)
            v_momentum[:] = 0.0
            no_improve = 0

        iter_count += 1

    # return best found
    return best_vec.tolist()

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
