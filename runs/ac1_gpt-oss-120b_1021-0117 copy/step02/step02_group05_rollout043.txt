import time, math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 bound using FFT‑based convolution."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type starting sequence: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

            # direction that preserves total mass (mean(g) - g)
            d = g.mean() - g

            # try a few decreasing step sizes until an improvement appears
            step = grad_eps * temperature
            improved = False
            while step > min_step:
                cand = cur + step * d
                np.clip(cand, 0.0, 1000.0, out=cand)
                cand_score = _ac1_score(cand, n, L)
                if cand_score < cur_score:
                    cur = cand
                    cur_score = cand_score
                    improved = True
                    if cand_score < best_score:
                        best = cand.copy()
                        best_score = cand_score
                        if best_score <= target:
                            break
                    break
                step *= 0.5

            if not improved:
                # fall back to a small random perturbation
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j:
                    frac = rng.random() * trans_max
                    delta = cur[i] * frac
                    cand = cur.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    np.clip(cand, 0.0, 1000.0, out=cand)
                    cand_score = _ac1_score(cand, n, L)
                    if cand_score < cur_score or rng.random() < math.exp(
                        (cur_score - cand_score) / (temperature + 1e-12)
                    ):
                        cur = cand
                        cur_score = cand_score
                        if cand_score < best_score:
                            best = cand.copy()
                            best_score = cand_score

        # --------------------------------------------------------------
        # 2) block scaling (≈30 % of attempts)
        elif move_type < 0.8:
            width = int(rng.integers(block_min, block_max + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, block_sigma))
            cand = cur.copy()
            cand[start : start + width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_score = _ac1_score(cand, n, L)
            if cand_score < cur_score or rng.random() < math.exp(
                (cur_score - cand_score) / (temperature + 1e-12)
            ):
                cur = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score

        # --------------------------------------------------------------
        # 3) random mass transfer (≈15 % of attempts)
        elif move_type < 0.95:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * trans_max
                delta = cur[i] * frac
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
                np.clip(cand, 0.0, 1000.0, out=cand)
                cand_score = _ac1_score(cand, n, L)
                if cand_score < cur_score or rng.random() < math.exp(
                    (cur_score - cand_score) / (temperature + 1e-12)
                ):
                    cur = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best = cand.copy()
                        best_score = cand_score

        # --------------------------------------------------------------
        # 4) random swap of two entries (≈5 % of attempts)
        else:
            i, j = rng.choice(n, size=2, replace=False)
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]
            # swapping does not change sum, only may affect max‑convolution
            cand_score = _ac1_score(cand, n, L)
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score

        # end of iteration -------------------------------------------------
        temperature *= decay

        # occasional restart from the best solution found so far
        if rng.random() < 0.001 or temperature < 1e-4:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

        if best_score <= target:
            break

    # ----------------------------------------------------------------------
    # return the best sequence discovered (as plain Python floats)
    return [float(x) for x in best]
