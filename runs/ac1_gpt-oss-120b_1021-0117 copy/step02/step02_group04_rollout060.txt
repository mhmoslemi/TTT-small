def propose_candidate(seed=42, budget_s=1100, **kwargs):

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.shape[0]

    L = 1 << ((2*n-2).bit_length())
    # evaluate some candidate shapes
    best_seq = None
    best_score = float('inf')

    def eval_seq(seq):
        # seq normalized to sum=1? We'll not necessarily; eval will compute actual
        # using conv via FFT for speed
        s = seq.sum()
        if s < 1e-12:
            return float('inf')

        fa = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        max_b = float(conv.max())
        return 2.0 * n * max_b / (s * s)
    # try exponential shapes
    logn = math.log(n)
    for factor in [0.6, 0.8, 1.0, 1.2, 1.4]:

        # compute raw exponential

        seq = np.exp(-c * i / n)
        # we may also consider symmetric: maybe also compute reversed? Not needed
        # maybe scale to sum=1 to avoid extra scaling effect
        # The objective is scale-invariant (S unaffected by scaling), but we keep scaling to avoid overflow.
        seq = seq / seq.sum()
        score = eval_seq(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        # check early exit if time near deadline

    return 1 << ((x - 1).bit_length())

    the AC1 upper bound.  An exponential decay shape with a parameter tuned
    to the length of the sequence yields a very low bound.
