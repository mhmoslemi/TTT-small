import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    # from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv

def _score(a):
    n = a.shape[0]
    conv = _conv_fft(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

def _grad_smooth(a, gamma):
    """Gradient of smoothed max (log‑sum‑exp) of the autocorrelation."""
    n = a.shape[0]
    conv = _conv_fft(a)
    m = conv.max()
    # safe softmax
    exp_vals = np.exp(gamma * (conv - m))
    ws = exp_vals / exp_vals.sum()
    # convolution ws * rev(a)
    a_rev = a[::-1]
    L = 1 << ((len(ws) + n - 1).bit_length())
    fw = np.fft.rfft(ws, n=L)
    fa = np.fft.rfft(a_rev, n=L)
    prod = np.fft.irfft(fw * fa, n=L)[: len(ws) + n - 1]
    grad = 2.0 * prod[n - 1 : n - 1 + n]
    return grad

    else:

    best = a0.copy()

    cur_sc = best_sc

    gamma = 5.0           # start smoothing parameter
    momentum = np.zeros_like(cur)

import time

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Linear convolution via FFT (full, length 2n‑1)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv

# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_fft(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

    m = conv.max()

    prod = np.fft.irfft(fw * fa, n=L)[: len(w) + n - 1]
    grad = 2.0 * prod[n - 1 : n - 1 + n]
