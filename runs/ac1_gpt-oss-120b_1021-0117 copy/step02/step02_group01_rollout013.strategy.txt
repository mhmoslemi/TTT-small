def _project_simplex(v):
    n = v.shape[0] if len(v.shape)==1 else len(v)

def _sqrt_series(n):
    a = np.zeros(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        if i == 1:
            a[i] = 0.0
        else:
            s = np.sum(a[1:i] * a[i-1:0:-1])
            a[i] = (1.0 - s) / 2.0
    return a
