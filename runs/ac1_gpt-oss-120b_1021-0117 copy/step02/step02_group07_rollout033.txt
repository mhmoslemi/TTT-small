def _project_simplex(v):
    # Projects onto the probability simplex sum=1, v>=0.
    # Implementation based on Michelot (1986) / Duchi et al. 2008.
    # Works for any shape 1D.
    v = np.asarray(v, dtype=np.float64)

    return w

    w[w < 0] = 0.0
    return w


def _conv(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of seq with itself, using FFT."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    return conv


def _score(seq: np.ndarray, n: int, fft_len: int) -> float:
    """Exact AC1 upper bound for seq."""
    conv = _conv(seq, fft_len)
    max_b = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _random_block_perturb(seq: np.ndarray, rng: np.random.Generator, n: int) -> np.ndarray:
    """Multiply a random block by a log‑normal factor and project onto simplex."""
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    cand = seq.copy()
    cand[start:start + width] *= factor
    return _project_simplex(cand)


def _smooth_pair(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Replace two neighbours by their average (preserves the sum)."""
    n = seq.size
    if n < 2:
        return seq.copy()
    i = rng.integers(0, n - 1)
    avg = (seq[i] + seq[i + 1]) * 0.5
    cand = seq.copy()
    cand[i] = avg
    cand[i + 1] = avg
    return cand


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Problem size and optional parent sequence
    # ------------------------------------------------------------------
    n = 3809
    parent_seq = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent_seq = np.clip(arr, 0.0, 1000.0)
            n = parent_seq.size
    except Exception:
        parent_seq = None

    # ------------------------------------------------------------------
    # Initialise a few candidates and pick the best one
    # ------------------------------------------------------------------
    cand_list = [_initial_binomial(n)]
    if parent_seq is not None:
        cand_list.append(parent_seq)

    # Normalise each candidate to sum = 1 (scale does not affect the score)
    cand_list = [c / c.sum() for c in cand_list]

    fft_len = 1 << ((2 * n - 1).bit_length())
    best_a = None
    best_score = np.inf
    for c in cand_list:
        sc = _score(c, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = c.copy()
    a = best_a.copy()
    current_score = best_score
    best_seq = best_a.copy()

    # ------------------------------------------------------------------
    # Main optimisation loop – gradient descent with occasional random moves
    # ------------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        # Full convolution and current maximal entry
        conv = _conv(a, fft_len)
        max_b = conv.max()
        kmax = int(np.argmax(conv))
        current_score = 2.0 * n * max_b / (a.sum() * a.sum())
        if current_score < best_score:
            best_score = current_score
            best_seq = a.copy()
            if best_score < 1.50274349:
                break

        # --- Gradient of the few largest convolution entries -----------------
        m = 3                     # number of peaks to target
        if conv.size <= m:
            top_ks = np.arange(conv.size)
        else:
            top_ks = np.argpartition(conv, -m)[-m:]

        grad = np.zeros_like(a)
        for kk in top_ks:
            lo = max(0, kk - (n - 1))
            hi = min(kk, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += a[kk - idx]        # proportional to ∂(conv[kk])/∂a

        # Direction that tries to reduce those peaks while keeping sum zero
        mean_g = grad.mean()
        direction = -grad + mean_g

        # Step size (decreases with iteration)
        step = 1.0 / (np.sqrt(iteration + 1) + 1.0)

        improved = False
        # Simple back‑tracking line search
        for _ in range(5):
            cand = a + step * direction
            cand = _project_simplex(cand)
            cand_score = _score(cand, n, fft_len)
            if cand_score < current_score:
                a = cand
                current_score = cand_score
                improved = True
                break
            step *= 0.5

        # If gradient step fails, try a stochastic move
        if not improved:
            if rng.random() < 0.5:
                cand = _random_block_perturb(a, rng, n)
            else:
                cand = _smooth_pair(a, rng)
            cand_score = _score(cand, n, fft_len)
            if cand_score < current_score:
                a = cand
                current_score = cand_score

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence observed (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
