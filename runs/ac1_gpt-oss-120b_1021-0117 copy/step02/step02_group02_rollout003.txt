def gradient_for_max(a,kmax):
    n = len(a)
    g = np.zeros_like(a)
    offset = kmax - (n-1)
    if offset >= 0:
        # i from offset to n-1 inclusive correspond to partner = kmax - i within [0,n-1]
        i = np.arange(offset, n)
        partner = kmax - i
        g[i] = 2.0 * a[partner]
    else:
        # offset negative, i from 0 to n+offset-1
        i = np.arange(0, n + offset)
        partner = kmax - i
        g[i] = 2.0 * a[partner]
    return g

import time

# ----------------------------------------------------------------------
# Helper utilities

def _fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return the AC1 upper bound (without denominator) for a normalized a."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))


def _gradient_from_max(a: np.ndarray, kmax: int) -> np.ndarray:
    """Sub‑gradient of max‑convolution at shift kmax."""
    n = a.shape[0]
    g = np.zeros_like(a)
    offset = kmax - (n - 1)
    if offset >= 0:
        idx = np.arange(offset, n, dtype=int)
        partner = kmax - idx
        g[idx] = 2.0 * a[partner]
    else:
        idx = np.arange(0, n + offset, dtype=int)
        partner = kmax - idx
        g[idx] = 2.0 * a[partner]
    return g


def _optimize_once(a0: np.ndarray, deadline: float, fft_len: int,
                   n: int, rng: np.random.Generator):
    """Run a single exponentiated‑gradient optimisation run."""
    a = a0.copy()
    best = a.copy()
    best_score = _fast_score(a, fft_len, n)

    # Adam‑style moment buffers
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    eta0 = 1.0

    t = 0
    beta1_pow = 1.0
    beta2_pow = 1.0
    jitter_interval = 100
    jitter_std = 0.02
    target = 1.50274349

    while time.monotonic() < deadline:
        t += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        # convolution via FFT
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        kmax = int(np.argmax(conv))
        cur_score = float(2.0 * n * conv[kmax])

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            if best_score <= target:
                break

        # gradient of the active shift
        g = _gradient_from_max(a, kmax)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        eta = eta0 / np.sqrt(t)

        # multiplicative (exponentiated) update
        step = -eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(step)

        # renormalise to the simplex
        s = a.sum()
        if s <= 0.0:
            a.fill(1.0 / n)
        else:
            a /= s

        # occasional jitter to escape plateaus
        if t % jitter_interval == 0:
            noise = rng.normal(0.0, jitter_std, size=n)
            a = a * np.exp(noise)
            a /= a.sum()

    return best, best_score

    best_seq = uniform.copy()
    best_score = _fast_score(best_seq, fft_len, n)

    # evaluate the parent if it differs from uniform
    if not np.allclose(a_init, uniform):
        sc = _fast_score(a_init, fft_len, n)
        if sc < best_score:
            best_seq = a_init.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # Main multi‑restart optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline - 0.5:
        remaining = deadline - time.monotonic()
        if remaining <= 0.5:
            break

        # random Dirichlet start (alpha=0.5 gives a spread of shapes)
        init = rng.dirichlet(np.full(n, 0.5))

        # occasionally blend with the current best to focus the search
        if rng.random() < 0.2:
            blend = rng.uniform(0.0, 0.3)
            init = (1.0 - blend) * init + blend * best_seq
            init /= init.sum()

        run_deadline = time.monotonic() + min(30.0, remaining)
        local_best, local_score = _optimize_once(init, run_deadline,
                                                 fft_len, n, rng)

        if local_score < best_score:
            best_score = local_score
            best_seq = local_best.copy()
            if best_score <= 1.50274349:
                break

    # ------------------------------------------------------------------
    # Return as plain Python list of floats
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq]
