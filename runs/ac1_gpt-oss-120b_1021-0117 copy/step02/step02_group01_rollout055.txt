def weighted_gradient(a, p):
    n = a.shape[0]
    L = 1 << ((n + (2*n-1) - 1).bit_length())  # power of two >= 3n-2
    # But we can just use np.convolve (which uses O(n*m)) but n ~2500, m~5000 => O(12.5M) per iteration maybe fine if few iterations but we need many loops. Use FFT for speed.
    # Compute FFT of a (zero-padded to L) and p_rev (zero-padded).
    A = np.fft.rfft(a, n=L)
    p_rev = p[::-1]  # length 2n-1
    P = np.fft.rfft(p_rev, n=L)
    conv_rev = np.fft.irfft(A * P, n=L)[: (n + (2*n-1) -1)]  # full length n+2n-2 = 3n-2
    # take indices (2n-2) - j
    grad = 2.0 * conv_rev[(2*n-2) - np.arange(n)]
    return grad
```

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum == 1, all entries >= 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    v = v.astype(np.float64, copy=False)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0

        shifted = conv - max_conv                # max becomes 0
        exp_vals = np.exp(alpha * shifted)
        p = exp_vals / exp_vals.sum()           # p_i >=0 , sum p_i = 1

        # ----- gradient of the smoothed objective -----
        # grad = 2 * sum_i p_i * ∂(a*a)_i / ∂a
        # using convolution with the reversed weight vector
        P_fft = np.fft.rfft(p[::-1], n=L)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=L)
        base = 2 * n - 2
        grad = 2.0 * conv_rev[base - np.arange(n)]

        # ----- momentum update -----
        v = beta * v + (1.0 - beta) * grad

        # ----- candidate step -----
        cand = a - step * v
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_score, cand_max = _score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
        else:
            # occasional random mass transfer to escape plateaus
            if rng.random() < 0.02:
                i = rng.integers(n)
                if a[i] > 0.0:
                    j = rng.integers(n - 1)
                    if j >= i:
                        j += 1
                    frac = rng.random() * 0.2
                    delta = a[i] * frac
                    cand2 = a.copy()
                    cand2[i] -= delta
                    cand2[j] += delta
                    cand2 = _project_simplex(cand2)
                    cand2_score, _ = _score(cand2)
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()
                    if cand2_score < cur_score:
                        a = cand2
                        cur_score = cand2_score

        # ----- step‑size decay -----
        step = max(min_step, step * decay)

        # cheap early exit
        if time.monotonic() + 0.5 > deadline:
            break

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain Python list).
    # --------------------------------------------------------------
    return [float(x) for x in best]
