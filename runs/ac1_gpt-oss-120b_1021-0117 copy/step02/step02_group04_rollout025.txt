def target_transfer(a, k_max):
    n = len(a)
    start = max(0, k_max - (n-1))
    end = min(n-1, k_max)
    idx = np.arange(start, end+1)
    rev_idx = k_max - idx
    prod = a[idx] * a[rev_idx]
    i_max = idx[prod.argmax()]
    j_min = idx[prod.argmin()]
    # transfer fraction of mass from i_max to j_min
    delta = 0.2 * a[i_max]   # move up to 20% of source mass
    if delta <= 0:
        return a.copy()
    cand = a.copy()
    cand[i_max] -= delta
    cand[j_min] += delta
    # ensure nonneg
    cand[i_max] = max(cand[i_max], 0.0)
    # renorm
    cand_sum = cand.sum()
    if cand_sum <= 0:
        cand = np.full_like(a, 1.0/len(a))
    else:
        cand /= cand_sum
    return cand
```
