    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution a*a (length 2·len(a)-1) using real FFT."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * len(a) - 1]
