def generate_sidon_set(n, target_size):
    sums = set()
    indices = []
    for i in range(n):
        # compute new sums with i (including i+i)
        new_sums = [i+j for j in indices] + [i+i]
        # need to ensure new_sums not in sums
        if all(s not in sums for s in new_sums):
            indices.append(i)
            sums.update(new_sums)
        if len(indices) >= target_size:
            break
    return indices

def compute_ac1_score(arr):
    n = len(arr)
    L = 1 << ((2*n-2).bit_length())
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    max_b = conv.max()
    s = arr.sum()
    return 2.0 * n * max_b / (s * s)
```

import time
import math
import numpy as np

# ----- utilities -------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound, identical to the official evaluator."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _eval(seq: np.ndarray, n: int, L: int) -> float:
    """Wrapper that guarantees non‑negativity and clipping."""
    seq = np.clip(seq, 0.0, 1000.0, out=seq)
    return _ac1_score_fft(seq, n, L)


# ----- candidate generators --------------------------------------
def _geo(seq_len: int, d: float) -> np.ndarray:
    """Geometric decay a_i = (1-d)^i , then normalized."""
    r = 1.0 - d
    if r <= 0.0:
        r = 1e-12
    idx = np.arange(seq_len, dtype=np.float64)
    arr = np.power(r, idx)
    s = arr.sum()
    return arr / s if s != 0 else arr


def _power(seq_len: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)^{-p}, then normalized."""
    idx = np.arange(seq_len, dtype=np.float64) + 1.0
    arr = np.power(idx, -p)
    s = arr.sum()
    return arr / s if s != 0 else arr


def _beta(seq_len: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta shape a_i ∝ (i+1)^{a-1} (n-i)^{b-1}."""
    i = np.arange(seq_len, dtype=np.float64) + 1.0
    n_f = float(seq_len)
    arr = np.power(i, a - 1.0) * np.power(n_f - i + 1.0, b - 1.0)
    s = arr.sum()
    return arr / s if s != 0 else arr


def _sym_geo(seq_len: int, d: float) -> np.ndarray:
    """Symmetric geometric: a_i = (1-d)^{min(i,n-1-i)}."""
    r = 1.0 - d
    if r <= 0.0:
        r = 1e-12
    idx = np.arange(seq_len, dtype=np.float64)
    dist = np.minimum(idx, seq_len - 1 - idx)
    arr = np.power(r, dist)
    s = arr.sum()
    return arr / s if s != 0 else arr


# ----- main search -----------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ----- problem size -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = parent.size
    except Exception:
        parent = None
        n = 4806                      # fall‑back length used by the reference code

    L = _next_pow2(2 * n - 1)

    # ----- initialise best ------------------------------------------------
    best_seq = None
    best_score = float('inf')

    # use parent as a possible start
    if parent is not None:
        parent = np.clip(parent, 0.0, 1000.0)
        s = _eval(parent, n, L)
        best_seq = parent.copy()
        best_score = s

    # uniform baseline (helps when parent missing)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    s_uni = _eval(uniform, n, L)
    if s_uni < best_score:
        best_seq = uniform.copy()
        best_score = s_uni

    # ----- random exploration in a low‑dimensional param space ------------
    shape_fns = [_geo, _power, _beta, _sym_geo]
    # number of random trials – cheap enough to run many within the budget
    max_trials = 2000
    for _ in range(max_trials):
        if time.monotonic() >= deadline:
            break
        fn = rng.choice(shape_fns)
        if fn is _geo:
            d = rng.uniform(5e-5, 0.003)      # decay step
            cand = fn(n, d)
        elif fn is _sym_geo:
            d = rng.uniform(5e-5, 0.003)
            cand = fn(n, d)
        elif fn is _power:
            p = rng.uniform(0.05, 4.0)        # exponent
            cand = fn(n, p)
        else:  # beta
            a = rng.uniform(0.1, 3.0)
            b = rng.uniform(0.1, 3.0)
            cand = fn(n, a, b)

        score = _eval(cand, n, L)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            if best_score < 1.50274349:
                # good enough – keep a tiny buffer before hard timeout
                deadline = min(deadline, time.monotonic() + 0.5)

    # ----- deterministic refinement (mirror‑gradient descent) -------------
    cur = best_seq.copy()
    cur_score = best_score
    # algorithm parameters
    gamma = 0.6                # step size for multiplicative update
    gamma_decay = 0.9995
    temp = 1.0                # for occasional uphill moves
    temp_decay = 0.9999

    while time.monotonic() < deadline:
        # convolution and active lag
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_b = float(conv.max())
        k_max = int(conv.argmax())
        cur_score = 2.0 * n * max_b / (cur.sum() ** 2)

        # mirrored vector (elements that pair with cur to give the max lag)
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        mirror = np.zeros_like(cur)
        mirror[mask] = cur[j[mask]]

        # multiplicative mirror‑gradient step (always keeps sum = 1 after renorm)
        cand = cur * np.exp(-gamma * mirror)
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand /= cand_sum

        cand_score = _eval(cand, n, L)

        # keep global best
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            cur = cand
            cur_score = cand_score
            if best_score < 1.50274349:
                # we have reached the target – exit swiftly
                break
        else:
            # simulated‑annealing style acceptance
            accept = False
            if cand_score < cur_score:
                accept = True
            else:
                prob = math.exp((cur_score - cand_score) / (temp + 1e-12))
                if rng.random() < prob:
                    accept = True
            if accept:
                cur = cand
                cur_score = cand_score

        # schedules
        gamma *= gamma_decay
        temp *= temp_decay

        # occasional random shake – helps escaping plateaus
        if rng.random() < 0.003:
            # random block scaling
            w = int(rng.integers(max(2, n // 200), max(3, n // 20)))
            start = int(rng.integers(0, n - w + 1))
            factor = math.exp(rng.normal(0.0, 0.08))
            shake = cur.copy()
            shake[start:start + w] *= factor
            shake = np.clip(shake, 0.0, None)
            shake /= shake.sum()
            cur = shake

    return [float(x) for x in best_seq]
