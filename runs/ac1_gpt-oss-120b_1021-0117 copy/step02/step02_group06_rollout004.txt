import time
import numpy as np

# ---------- utilities ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution a*a using FFT (length 2*len(a)-1)."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    n = a.shape[0]
    return full[: 2 * n - 1]


def _catalan_weights(n: int) -> np.ndarray:
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w

    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    for p in (0.5, 1.0, 2.0):
        v = sin_vals ** p
        cand.append(v / v.sum())

    # Random Dirichlet draws with various concentration parameters
    rng = np.random.default_rng(0)
    for alpha in (0.2, 0.5, 1.0, 2.0):
        v = rng.gamma(alpha, size=n)
        if v.sum() > 0:
            cand.append(v / v.sum())

    return cand


# ---------- main routine ----------

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that minimizes the
    AC1 upper bound used by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ----- determine problem size -----
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or not np.isfinite(parent_arr).all():
            raise ValueError
    except Exception:
        parent_arr = None

    n = int(parent_arr.shape[0]) if parent_arr is not None else 256

    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two

    # ----- pick best initial candidate -----
    best_vec = None
    best_score = np.inf
    for vec in _initial_candidates(n, parent_arr):
        conv = _conv_self_fft(vec, fft_len)
        cur = 2.0 * n * float(conv.max())
        if cur < best_score:
            best_score = cur
            best_vec = vec.copy()
        if best_score <= 1.50274349:
            return best_vec.tolist()

    # ----- sub‑gradient descent with adaptive steps -----
    a = best_vec.copy()
    a_prev = a.copy()
    step0 = 0.5          # initial step magnitude
    min_step = 1e-5
    iter_no_imp = 0
    it = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # convolution and objective
        conv = _conv_self_fft(a, fft_len)
        max_b = float(conv.max())
        cur_score = 2.0 * n * max_b

        # update best solution
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_vec = a.copy()
            iter_no_imp = 0
            if best_score <= target:
                break
        else:
            iter_no_imp += 1

        # active indices (allow tiny numerical tolerance)
        tol = max_b * 1e-12 + 1e-14
        active = np.where(conv >= max_b - tol)[0]

        # gradient of the active max‑quadratic(s)
        g = np.zeros_like(a)
        for k in active:
            i_start = k - (n - 1)
            if i_start < 0:
                i_start = 0
            i_end = k if k < n else n - 1
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1, dtype=np.int64)
                g[idx] += 2.0 * a[k - idx]

        # step size schedule (global, not reset on stagnation)
        step = step0 / np.sqrt(it)
        if step < min_step:
            step = min_step

        # projected sub‑gradient step
        cand = a - step * g
        cand = _project_onto_simplex(cand)

        # occasional jitter to escape flat regions
        if rng.random() < 0.01:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            cand = cand * jitter
            cand = _project_onto_simplex(cand)

        a = cand

        # restart if no improvement for a while
        if iter_no_imp > 3000:
            perturb = np.exp(rng.normal(0.0, 0.04, size=n))
            a = best_vec * perturb
            a = _project_onto_simplex(a)
            iter_no_imp = 0

        # safety time check
        if time.monotonic() >= deadline:
            break

    return best_vec.tolist()
