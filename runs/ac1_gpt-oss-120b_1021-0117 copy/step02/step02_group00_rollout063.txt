def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)  # small margin
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 256
    idx = np.arange(n, dtype=np.float64)

    # FFT length
    Lfft = 1 << ((2*n - 1).bit_length())
    rfft = np.fft.rfft

    def conv_max_fft(a):
        fv = rfft(a, n=Lfft)
        conv = np.fft.irfft(fv*fv, n=Lfft)[:2*n-1]
        return conv.max()

    def gen_vector(L):
        # L = -log(r)
        if L <= 0:
            # uniform
            return np.full(n, 1.0/n, dtype=np.float64)
        # compute r = exp(-L)
        r = np.exp(-L)
        # compute normalization factor: C = (1-r)/(1-r**n)
        # Use expm1 for stability
        num = -np.expm1(-L)      # 1 - r
        den = -np.expm1(-L * n)  # 1 - r**n
        # If den is zero (unlikely unless L tiny and n huge)
        if den == 0:
            C = 1.0 / n
        else:
            C = num / den
        # powers = r**idx = exp(-L*idx)
        powers = np.exp(-L * idx)
        return C * powers

    # initial bounds
    L_min = 5e-5   # 0.00005
    L_max = 5e-3   # 0.005
    # coarse grid steps adapt based on time
    steps_coarse = 1500
    best_score = np.inf
    best_L = None
    best_a = None

    for t in np.linspace(0.0, 1.0, steps_coarse):
        L = L_min + (L_max - L_min) * t
        a = gen_vector(L)
        maxconv = conv_max_fft(a)
        score = 2.0 * n * maxconv   # sum =1
        if score < best_score:
            best_score = score
            best_L = L
            best_a = a
        # early break if time runs low
        if time.monotonic() > deadline:
            break

    # refine around best_L
    if best_L is not None:
        delta = (L_max - L_min) / steps_coarse * 5
        L_low = max(best_L - delta, 1e-9)
        L_high = best_L + delta
        steps_refine = 2000
        for t in np.linspace(0.0, 1.0, steps_refine):
            L = L_low + (L_high - L_low) * t
            a = gen_vector(L)
            maxconv = conv_max_fft(a)
            score = 2.0 * n * maxconv
            if score < best_score:
                best_score = score
                best_L = L
                best_a = a
            if time.monotonic() > deadline:
                break

    # Ensure best_a is defined; fallback to uniform if none.
    if best_a is None:
        best_a = np.full(n, 1.0/n, dtype=np.float64)
        best_score = 2.0 * n * conv_max_fft(best_a)

    # Perform a few local improvements with random mass moves using exact evaluator.
    # Only if time allows.
    cur = best_a.copy()
    cur_score = evaluate_sequence(cur.tolist())
    best_candidate = cur.copy()
    best_candidate_score = cur_score

    while time.monotonic() < deadline:
        # random move
        i = rng.integers(0, n)
        if cur[i] <= 1e-12:
            continue
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # transfer a fraction up to 20% of cur[i]
        delta_mass = cur[i] * rng.random() * 0.2
        cand = cur.copy()
        cand[i] -= delta_mass
        cand[j] += delta_mass
        # ensure nonnegative; already
        # renormalize to sum 1 (should stay same sum)
        # sum unchanged since we transfer mass
        candidate_score = evaluate_sequence(cand.tolist())
        if candidate_score < best_candidate_score:
            best_candidate = cand.copy()
            best_candidate_score = candidate_score
            cur = cand
            cur_score = candidate_score
        else:
            # maybe accept with small probability to escape local minima
            if rng.random() < 0.01:
                cur = cand
                cur_score = candidate_score

    # Return best candidate (list)
    return [float(x) for x in best_candidate]

    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast convolution (full linear) via FFT, returning the maximum term
# ------------------------------------------------------------
def _conv_max_fft(a: np.ndarray, Lfft: int) -> float:
    fv = np.fft.rfft(a, n=Lfft)
    conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small reserve for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # Determine length and initialise a feasible vector (sum = 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full_like(a0, 1.0 / a0.size)
    else:
        # fallback: a short decreasing exponential
        n = 256
        idx = np.arange(n, dtype=np.float64)
        L0 = 1.0 / n
        r = np.exp(-L0)
        num = -np.expm1(-L0)          # 1 - r
        den = -np.expm1(-L0 * n)      # 1 - r**n
        C = num / den if den != 0 else 1.0 / n
        a0 = C * np.exp(-L0 * idx)

    n = a0.size
    Lfft = 1 << ((2 * n - 1).bit_length())   # next power of two for FFT

    # ------------------------------------------------------------------
    # Baseline score (approximate, using FFT)
    # ------------------------------------------------------------------
    cur = a0.copy()
    cur_score = 2.0 * n * _conv_max_fft(cur, Lfft)
    best_vec = cur.copy()
    best_sc = cur_score

    # ------------------------------------------------------------------
    # Optimization parameters
    # ------------------------------------------------------------------
    alpha0 = 5.0               # initial step magnitude
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # full convolution via FFT
        fv = np.fft.rfft(cur, n=Lfft)
        conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
        maxb = float(conv.max())

        # active set (indices within tiny tolerance of the maximum)
        tol = maxb * 1e-12 + 1e-14
        active = np.where(conv >= maxb - tol)[0]

        # sub‑gradient of the active constraints
        g = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx_range = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx_range] += 2.0 * cur[k - idx_range]

        # step and projection
        alpha = alpha0 / np.sqrt(it)
        cand = cur - alpha * g
        cand = _project_simplex(cand)

        cand_sc = 2.0 * n * _conv_max_fft(cand, Lfft)

        if cand_sc < cur_score:
            cur, cur_score = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            # occasional Metropolis‑type acceptance
            prob = np.exp(-(cand_sc - cur_score) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_sc
            no_improve += 1

        # ------------------------------------------------------------
        # Random mass‑transfer move when progress stalls
        # ------------------------------------------------------------
        if no_improve > 2500 and rng.random() < 0.2:
            i = rng.integers(0, n)
            if cur[i] > 0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.35
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = 2.0 * n * _conv_max_fft(tmp, Lfft)
                if tmp_sc < cur_score:
                    cur, cur_score = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0

        # ------------------------------------------------------------
        # Small jitter restart towards the best point
        # ------------------------------------------------------------
        if it % 4000 == 0 and rng.random() < 0.3:
            jitter = rng.normal(scale=1e-3, size=n)
            tmp = best_vec + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = 2.0 * n * _conv_max_fft(tmp, Lfft)
            if tmp_sc < cur_score:
                cur, cur_score = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        # early exit if the target is reached
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Final exact evaluation (and tiny local refinement if time permits)
    # ------------------------------------------------------------------
    final_vec = best_vec
    final_score = evaluate_sequence(final_vec.tolist())

    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if final_vec[i] <= 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = final_vec[i] * rng.random() * 0.25
        cand = final_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < final_score:
            final_vec, final_score = cand, cand_score
            if cand_score < best_sc:
                best_vec, best_sc = cand.copy(), cand_score
            if final_score < 1.50274349:
                break

    return [float(x) for x in best_vec]
