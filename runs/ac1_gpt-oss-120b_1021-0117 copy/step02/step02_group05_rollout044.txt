import time, math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type seed: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound via FFT (exact same formula as evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    # reserve a small safety margin for the return path
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialise from the provided parent or a binomial seed
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        cur = parent.copy()
    except Exception:
        # fallback: moderate length seed (the parent length is usually >4000)
        n = 512
        cur = _gen_binomial(n)

    n = int(cur.size)
    L = _next_pow2(2 * n - 1)

    cur_score = _ac1_score(cur, n, L)
    best = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # hyper‑parameters
    temperature = 1.0
    decay = 0.9996
    step_base = 0.7           # L2 magnitude of a gradient step at temperature=1
    block_min = max(2, n // 300)
    block_max = max(2, n // 10)
    transfer_frac = 0.35
    target_score = 1.50274349

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        move = rng.random()
        cand = None

        # --------------------------------------------------------------
        # 1) gradient step based on the current maximal convolution entry
        if move < 0.45:
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
            kmax = int(np.argmax(conv))

            idx = np.arange(n)
            rev = kmax - idx
            mask = (rev >= 0) & (rev < n)

            g = np.zeros_like(cur)
            g[mask] = cur[rev[mask]]
            self_mask = mask & (rev != idx)
            g[self_mask] *= 2.0               # each pair counted twice

            # direction that reduces the peak while preserving total mass
            d = -(g - g.mean())
            norm = np.linalg.norm(d) + 1e-12
            step = step_base * temperature
            cand = cur + (step / norm) * d

        # --------------------------------------------------------------
        # 2) random block scaling
        elif move < 0.80:
            width = int(rng.integers(block_min, block_max + 1))
            start = int(rng.integers(0, n - width + 1))
            # log‑normal factor, magnitude reduced as temperature drops
            sigma = 0.25 * (temperature + 0.1)
            factor = math.exp(rng.normal(0.0, sigma))
            cand = cur.copy()
            cand[start:start + width] *= factor

        # --------------------------------------------------------------
        # 3) random mass transfer between two positions
        elif move < 0.95:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1
            frac = rng.random() * transfer_frac * temperature
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        # --------------------------------------------------------------
        # 4) global jitter (small additive noise)
        else:
            amp = 0.02 * temperature
            cand = cur + rng.normal(0.0, amp, size=n)

        # --------------------------------------------------------------
        # enforce evaluator limits
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate candidate
        cand_score = _ac1_score(cand, n, L)

        # record global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < target_score:
                break

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score

        # temperature schedule
        temperature *= decay
        if temperature < 0.02:
            temperature = 0.02

        # occasional restart from the best seen so far
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # ------------------------------------------------------------------
    # return the best sequence as plain Python floats
    return [float(x) for x in best]
