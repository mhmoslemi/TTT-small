def simplex_projection(v):
    # ensure nonnegative sum = 1
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho = max i s.t. u_i - (cssv_i / i) > 0
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    """Project the vector v onto the probability simplex (sum=1, each >=0)."""
    # based on Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all zero, return uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Determine length n and initial sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        n = int(parent.size)
        init = parent.astype(np.float64)
    except Exception:
        n = 256
        # generate binomial shape
        init = np.empty(n, dtype=np.float64)
        init[0] = 1.0
        for i in range(1, n):
            init[i] = init[i-1] * (2*i-1) / (2*i)
    # Normalize to 1
    init = np.maximum(init, 0.0)
    if init.sum() == 0:
        init[0] = 1.0
    a = init / init.sum()

    L = 1 << ((2*n - 2).bit_length())   # next power of two >=2n-1

    def ac1_score(x):
        fa = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        M = conv.max()
        return 2.0 * n * M

    # track best
    cur = a.copy()
    cur_score = ac1_score(cur)
    best = cur.copy()
    best_score = cur_score

    # Simplex projection
    def proj_simplex(v):
        # v is a numpy array, sum should be 1 after projection
        n_loc = v.shape[0]
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        rho_arr = np.where(u - cssv / np.arange(1, n_loc+1) > 0)[0]
        if rho_arr.size == 0:
            # return uniform distribution
            return np.full_like(v, 1.0 / n_loc)
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    it = 0
    step0 = 1.0   # initial step magnitude
    while time.monotonic() < deadline:
        it += 1
        # compute convolution, max, k_max
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        M = float(conv.max())
        k_max = int(conv.argmax())
        cur_score = 2.0 * n * M
        # gradient
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        mirror = np.zeros_like(cur)
        mirror[mask] = cur[j[mask]]
        grad = 4.0 * n * (mirror - M)

        # step size
        step = step0 / math.sqrt(it)

        # candidate via projected gradient
        cand = proj_simplex(cur - step * grad)

        cand_score = ac1_score(cand)

        # accept if improvement else keep current
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
        else:
            # occasional random perturbation to escape
            if rng.random() < 0.001:
                # add small noise
                noise = rng.normal(0.0, 0.02, size=n)
                cur = proj_simplex(cur + noise)

        # update global best
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score
            # early exit if goal reached
            if best_score < 1.50274349:
                break

        # optional step0 adaptation (if no improvement for many iters)
        if it % 5000 == 0:
            # after a block, if best hasn't improved, reduce step0
            # we can keep step0 unchanged for simplicity
            pass

    # Convert to list of python floats
    return [float(x) for x in best]

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _gen_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i-1) / (2*i)
    return a

def _ac1_score_fft(arr, L):
    n = arr.shape[0]
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    M = conv.max()
    return 2.0 * n * M

def _proj_simplex(v):
    # projection onto simplex sum=1, v non-negative after projection
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]

    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)

    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """Classic binomial step: C(2i,i)/4**i for i=0..n‑1."""
