def propose_candidate(seed=42,budget_s=1100,...):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0,float(budget_s)-2.0)

    # Determine n and initialize seq
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),0,1000)
        n = parent.size
    except Exception:
        n = 2000
        parent = None

    def initial_seq():
        if parent is not None:
            return parent.copy()
        # use binomial
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1,n):
            a[i] = a[i-1]*(2*i-1)/(2*i)
        return a

    current = initial_seq()
    fft_len = 1<<((2*n-1).bit_length())

    def conv_seq(seq):
        f = np.fft.rfft(seq, n=fft_len)
        return np.fft.irfft(f*f, n=fft_len)[:2*n-1]

    def score_of(seq):
        conv = conv_seq(seq)
        maxb = np.max(conv)
        s = np.sum(seq)
        if s <= 0:
            return np.inf, maxb
        return 2.0*n*maxb/(s*s), maxb

    best = current.copy()
    best_score, best_maxb = score_of(best)

    # main loop
    while time.monotonic() < deadline:
        # compute conv for current
        cur_conv = conv_seq(current)
        kmax = int(np.argmax(cur_conv))
        lo = max(0, kmax - (n-1))
        hi = min(kmax, n-1)
        # compute gradient for current max conv
        # grad has length n, zero elsewhere
        grad = np.zeros_like(current)
        idxs = np.arange(lo, hi+1, dtype=np.int64)
        js = kmax - idxs
        grad[idxs] = 2.0 * current[js]   # as derived
        # compute direction
        mean_grad = grad.mean()
        direction = -grad + mean_grad
        # compute step magnitude (decaying)
        elapsed = time.monotonic() - start
        # learning rate schedule: start high 0.02 and decrease to 0.001
        lr = 0.02 * (1 - elapsed/(deadline-start)) + 0.001
        step = lr * (np.max(current)+1.0) / (np.linalg.norm(direction)+1e-12)
        cand = current + step * direction
        # clip
        np.clip(cand, 0.0, 1000.0, out=cand)
        # re-balance sum by adjusting max element
        sum_c = cand.sum()
        sum_cur = current.sum()
        if sum_c != sum_cur:
            diff = sum_c - sum_cur
            # adjust element with largest value
            idx = np.argmax(cand)
            cand[idx] -= diff
            if cand[idx] < 0:
                cand[idx] = 0.0
        cand_score, _ = score_of(cand)
        # accept if improvement, else maybe random small step
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            current = cand
        else:
            # probabilistic acceptance with temperature
            temp = 0.005 * (1 - elapsed/(deadline-start))
            if temp>0 and rng.random() < np.exp((best_score - cand_score)/(temp+1e-12)):
                current = cand

        # occasional random perturbations to avoid stagnation
        if rng.random() < 0.02:
            # random scaling on random block
            w = rng.integers(1, max(2, n//30))
            start_idx = rng.integers(0, n-w+1)
            factor = np.exp(rng.normal(0.0, 0.1))
            current[start_idx:start_idx+w] *= factor
            np.clip(current, 0, 1000, out=current)

        # optionally break if score < target
        if best_score < 1.502:
            break

    return [float(x) for x in best]

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_self(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution (real) of seq, truncated to length 2*len(seq)-1."""
    f = np.fft.rfft(seq, n=fft_len)
    conv_full = np.fft.irfft(f * f, n=fft_len)
    n = seq.shape[0]
    return conv_full[: 2 * n - 1]


def _score_seq(seq: np.ndarray, fft_len: int) -> tuple[float, float]:
    """Return (score, max_convolution) for a given sequence."""
    conv = _conv_self(seq, fft_len)
    max_b = float(np.max(conv))
    s = float(np.sum(seq))
    if s <= 0.0:
        return float("inf"), max_b
    n = seq.shape[0]
    score = 2.0 * n * max_b / (s * s)
    return float(score), max_b


def _gradient_max(seq: np.ndarray, kmax: int) -> np.ndarray:
    """Gradient of the convolution entry at lag kmax w.r.t. seq."""
    n = seq.shape[0]
    grad = np.zeros_like(seq)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idxs = np.arange(lo, hi + 1, dtype=np.int64)
    js = kmax - idxs
    grad[idxs] = 2.0 * seq[js]            # derivative of a_i * a_{k-i}
    return grad
