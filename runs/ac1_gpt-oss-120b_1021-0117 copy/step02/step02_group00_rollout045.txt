def gradient_smooth(a, beta):
    n = a.size
    conv = conv_fft(a)  # length 2n-1
    # compute softmax weights
    maxv = conv.max()
    # shift to avoid overflow
    exps = np.exp(beta * (conv - maxv))
    sum_exps = exps.sum()
    w = exps / sum_exps  # length 2n-1 sum to 1
    # compute gradient g = 2 * cross-correlation(w, a)
    # compute rev_w
    rev_w = w[::-1]
    L = 1 << ((len(rev_w) + n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    fw = np.fft.rfft(rev_w, n=L)
    cross = np.fft.irfft(fa * fw, n=L)  # length L (real)
    # slice appropriate region
    start = len(w) - 1  # 2n-2
    g = 2.0 * cross[start:start+n]
    return g

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto non‑negative simplex, sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Convolution via FFT (full linear convolution)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using FFT (real‑valued)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv

# ------------------------------------------------------------
# Score (same as evaluate_sequence for sum == 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

# ------------------------------------------------------------
# Gradient of smooth max (log‑sum‑exp) of the convolution
# ------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float):
    """
    Returns (g, conv) where g = ∇_a (smooth_max_beta(a*a))
    and conv = a*a (full) for possible reuse.
    """
    n = a.shape[0]

    # softmax weights (stable)

    w = exp_shift / exp_shift.sum()         # sum(w) == 1

    start = len(w) - 1                      # 2n‑2
    g = 2.0 * cross[start:start + n]
    return g, conv
