def propose_candidate(seed=42,budget_s=1100,**kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-2.0)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # create sqrt series of length 512 maybe
        a = _sqrt_series(512)
    # project onto simplex
    a = _project_simplex(a)
    n = a.size
    # Determine FFT length (power of 2) >= 3*n-2:
    L = 1 << ((3*n - 2).bit_length())
    # initial evaluation
    max_val, max_idx = _conv_max_and_index(a, L)
    cur_score = 2.0 * n * max_val  # sum=1 so denominator=1
    best = a.copy()
    best_score = cur_score

    # parameters
    step = 0.2  # initial step
    min_step = 1e-7
    decay = 0.999
    beta = 1.0
    beta_inc = 1.05
    max_beta = 100.0
    # momentum optional
    momentum = np.zeros_like(a)
    mu = 0.9

    # helper functions defined earlier.

    while time.monotonic() < deadline:
        # compute smoothmax gradient
        # compute convolution a*a
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        # compute smoothmax weights
        # subtract max for numerical stability
        max_c = conv.max()
        # scale beta gradually: maybe increase if progress stagnant
        # compute w
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum

        # compute correlation gradient
        # need FFT of w
        fw = np.fft.rfft(w, n=L)
        # correlation = ifft(conj(fa) * fw) (since a first argument)
        corr = np.fft.irfft(np.conj(fa) * fw, n=L)[:n]  # take first n entries
        grad = 2.0 * corr

        # momentum update
        momentum = mu * momentum + (1 - mu) * grad

        # gradient step
        cand = a - step * momentum
        cand = _project_simplex(cand)

        # evaluate candidate
        max_val_c, max_idx_c = _conv_max_and_index(cand, L)
        cand_score = 2.0 * n * max_val_c

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            max_idx = max_idx_c
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
            # maybe increase beta after success
            beta = min(max_beta, beta * beta_inc)
        else:
            # Could try backtrack: reduce step size
            step *= 0.5
            # optionally keep candidate unchanged
            # also reduce momentum? maybe set momentum = 0
            momentum = np.zeros_like(a)

        # step decay
        step = max(min_step, step * decay)

        # occasionally random perturbation to escape plateau
        if rng.random() < 0.02:
            # pick random indices and shuffle some mass
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                j = rng.integers(0, n)
                if j != i:
                    amount = a[i] * rng.random() * 0.5
                    cand2 = a.copy()
                    cand2[i] -= amount
                    cand2[j] += amount
                    cand2 = _project_simplex(cand2)
                    max_val2, max_idx2 = _conv_max_and_index(cand2, L)
                    cand2_score = 2.0 * n * max_val2

import time
import numpy as np

# --------------------------------------------------------------
# Simplex projection (Euclidean)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_pos = np.where(rho_arr > 0)[0]
    if rho_pos.size:
        rho = rho_pos[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

# --------------------------------------------------------------
# Analytic sqrt series (fallback starter)
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# --------------------------------------------------------------
# Max of full autocorrelation (via FFT) and its index
# --------------------------------------------------------------
def _conv_max_and_index(vec: np.ndarray, L: int):
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

# --------------------------------------------------------------

    rng = np.random.default_rng(seed)
    # leave a safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # initial sequence (use the provided parent if present)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = _sqrt_series(512)
    # project onto the probability simplex (sum = 1, non‑negative)
    a = _project_simplex(a)

    n = a.size
    # FFT length sufficient for all convolutions (linear, not circular)
    L = 1 << ((3 * n - 2).bit_length())

    # initial evaluation
    max_val, max_idx = _conv_max_and_index(a, L)
    cur_score = 2.0 * n * max_val          # sum(a)=1 → denominator = 1
    best = a.copy()
    best_score = cur_score

    # --------------------------------------------------------------
    # optimisation hyper‑parameters
    # --------------------------------------------------------------
    step = 0.2                # base learning rate
    min_step = 1e-7
    decay = 0.9995
    beta = 1.0                # smooth‑max temperature
    beta_inc = 1.05
    beta_max = 100.0
    mu = 0.9                  # momentum coefficient
    momentum = np.zeros_like(a)

    # occasional random mass‑transfer probability
    prob_random = 0.015

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # -------------------- smooth‑max gradient --------------------
        fa = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]

        # smooth‑max weights (numerically stable)
        c_max = conv_full.max()
        w = np.exp(beta * (conv_full - c_max))
        w_sum = w.sum()
        if w_sum == 0:
