import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Gradient of a single convolution shift.
# ----------------------------------------------------------------------
def _grad_for_shift(a: np.ndarray, shift: int) -> np.ndarray:
    """Return sub‑gradient of the convolution entry with given shift."""
    n = a.size
    g = np.zeros_like(a)
    if shift >= 0:
        s = shift
        # a[i] pairs with a[i+s] (if inside)
        g[: n - s] += a[s:]
        # a[i] pairs with a[i-s] (if inside)
        g[s:] += a[: n - s]
    else:
        s = -shift
        g[s:] += a[: n - s]
        g[: n - s] += a[s:]
    return g
