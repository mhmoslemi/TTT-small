import numpy as np, time

def _initial_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def _score_conv(seq, fft_len):
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2*seq.size - 1]
    max_b = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float('inf'), conv, max_b
    score = 2.0 * seq.size * max_b / (s * s)
    return float(score), conv, max_b

import numpy as np
import time

def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _score_conv(seq: np.ndarray, fft_len: int):
    """Return (score, full convolution, max value)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * seq.size - 1]
    max_b = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_b
    score = 2.0 * seq.size * max_b / (s * s)
    return float(score), conv, max_b


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a coefficient list with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)   # tiny safety margin

        n = parent_arr.size

    else:
        current = _initial_binomial(n)

    best = current.copy()
    fft_len = 1 << ((2 * n - 1).bit_length())
    best_score, best_conv, _ = _score_conv(best, fft_len)
    current_score = best_score

    move_probs = np.array([0.45, 0.25, 0.12, 0.12, 0.06])

    while time.monotonic() < deadline:
        # recompute convolution and score for the present candidate
        cur_score, cur_conv, _ = _score_conv(current, fft_len)
        if cur_score < best_score:
