def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound via FFT (scale‑invariant)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

            frac = rng.random() * 0.5
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        # ------------------------------------------------------
        # 4) swap two entries (≈10 %)
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # enforce evaluator bounds
        np.clip(cand, 0.0, 1000.0, out=cand)
        if cand.sum() < 0.01:
            # renormalise minimally
            cand *= (0.01 / max(cand.sum(), 1e-12))

        # evaluate
        cand_score = _ac1_score(cand, n, L)

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < target:
                break

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # temperature schedule
        temperature = max(temperature * decay, 1e-8)

        # occasional restart from the best seen
        if rng.random() < restart_prob:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # --------------------------------------------------------------
    # final result – plain Python floats
    return [float(x) for x in best]
```
