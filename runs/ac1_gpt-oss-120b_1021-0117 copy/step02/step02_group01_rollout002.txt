def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback generate sqrt series length 256, then pad to some length? We can just use that.
        def _sqrt_series(n):
            a = np.empty(n, dtype=np.float64)
            a[0] = 1.0
            for k in range(1, n):
                if k == 1:
                    s = 0.0
                else:
                    s = float(np.dot(a[1:k], a[k-1:0:-1]))
                a[k] = (1.0 - s)/2.0
            return a
        a = _sqrt_series(256)

    a = _project_simplex(a)  # ensures sum=1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # compute initial score
    conv = _conv_full_fft(a, L)
    max_val = float(conv.max())
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    step = 0.2
    min_step = 1e-6
    max_step = 1.0
    eps = 0.01   # mask tolerance

    # momentum variables (optional)
    momentum = np.zeros_like(a)
    beta = 0.9

    # loop
    while time.monotonic() < deadline:
        # compute full convolution
        conv = _conv_full_fft(a, L)
        max_val = float(conv.max())
        max_idx = int(conv.argmax())

        thresh = max_val * (1.0 - eps)
        active = np.where(conv >= thresh)[0]
        # compute gradient
        g = np.zeros_like(a)
        # Vectorized loop
        for i in active:
            lo = max(0, i - (n - 1))
            hi = min(n - 1, i)
            idx = np.arange(lo, hi+1)

        # adapt eps maybe shrink over time

import time
import numpy as np

# ---------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# ---------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

# ---------------------------------------------------------------
# Full convolution via FFT (returns length 2·n‑1 array)
# ---------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Compute a*a (full) using FFT of length L (power of two)."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return conv

# ---------------------------------------------------------------
# Optional fallback: analytic sqrt series (same as previous code)
# ---------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a
