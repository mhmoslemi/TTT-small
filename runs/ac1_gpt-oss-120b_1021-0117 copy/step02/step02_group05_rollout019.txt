def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # get parent if available
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        n = parent.shape[0]
        cur = parent.copy()
    except Exception:
        n = 256
        cur = np.ones(n)  # uniform? Or binomial.
        # generate binomial:
        cur = np.empty(n, dtype=np.float64)
        cur[0] = 1.0
        for i in range(1, n):
            cur[i] = cur[i-1] * (2*i-1)/(2*i)
    L = 1 << ((2*n - 2).bit_length())  # next pow2

    # define fast score
    def ac1_score(arr):
        fa = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
        max_b = conv.max()
        s = arr.sum()
        if s < 1e-12:
            return float('inf')
        return 2.0 * n * max_b / (s*s)

    best = cur.copy()
    best_score = ac1_score(best)
    cur_score = best_score

    # parameters
    temperature = 1.0
    decay = 0.9995
    grad_eps = 0.02  # step size for gradient move; can adapt
    block_eps_sigma = 0.13
    transfer_max = 0.4
    min_block = max(2, n // 200)
    max_block = max(2, n // 12)
    target = 1.50274349
    iter_count = 0

    while time.monotonic() < deadline:
        iter_count += 1
        move_type = rng.random()
        cand = cur.copy()
        if move_type < 0.5:
            # gradient guided move
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
            kmax = int(np.argmax(conv))
            max_b = conv[kmax]
            s = cur.sum()
            # compute partner contributions
            idx = np.arange(n)
            rev = kmax - idx
            mask = (rev >= 0) & (rev < n)
            g = np.zeros_like(cur)
            g[mask] = cur[rev[mask]]
            # duplicate for distinct
            self_mask = mask & (rev != idx)
            g[self_mask] *= 2.0
            # gradient direction
            d = g * s - 2.0 * max_b
            # scale step size adaptively based on magnitude
            # Avoid too large steps: scale epsilon by 1/(||d||_inf+1)
            scale = grad_eps / (np.abs(d).max() + 1e-12)
            cand = cur - scale * d
        elif move_type < 0.85:
            # block scaling
            width = int(rng.integers(min_block, max_block+1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, block_eps_sigma))
            cand[start:start+width] *= factor
        else:
            # random mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * transfer_max
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta

        # enforce bounds [0,1000]
        np.clip(cand, 0.0, 1000.0, out=cand)
        # ensure sum > 0.01 (should be)
        cand_score = ac1_score(cand)

        # update best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < target:
                # exit early, but still ensure time left for return
                break

        # acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        temperature *= decay

        # occasional restart (with small probability)
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

        # optional keep track of time?
    # final fallback: use best as list
    return [float(x) for x in best]

import numpy as np, time, math

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    # --- initial sequence ---
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        cur = parent.copy()
        n = int(parent.size)
    except Exception:
        n = 256
        cur = np.empty(n, dtype=np.float64)
        cur[0] = 1.0
        for i in range(1, n):
            cur[i] = cur[i-1] * (2*i-1) / (2*i)
    # FFT length (next power of two for linear convolution)
    L = 1 << ((2*n - 2).bit_length())
    idx = np.arange(n)
    # fast AC1 score
    def _ac1_score(arr):
        fa = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
        maxb = conv.max()
        s = arr.sum()
        if s < 1e-12:
            return float('inf')
        return 2.0 * n * maxb / (s*s)
    best = cur.copy()
    best_score = _ac1_score(best)
    cur_score = best_score
    # ---- parameters ----
    temperature = 1.0
    decay = 0.9995
    grad_base = 0.4          # intended max update per gradient step
    block_sigma = 0.13
    transfer_max = 0.4
    min_block = max(2, n // 200)
    max_block = max(2, n // 12)
    target = 1.50274349
    # main loop
    while time.monotonic() < deadline:
        move = rng.random()
        cand = cur.copy()
        # ---------- gradient guided move ----------
        if move < 0.45:
            fa = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
            kmax = int(np.argmax(conv))
            maxb = conv[kmax]
            s = cur.sum()
            rev = kmax - idx
            mask = (rev >= 0) & (rev < n)
            g = np.zeros_like(cur)
            g[mask] = cur[rev[mask]]
            self_mask = mask & (rev != idx)
            g[self_mask] *= 2.0
            # gradient of objective (up to positive factor)
            d = g * s - 2.0 * maxb
            # scale so that largest absolute update ≈ grad_base
            scale = grad_base / (np.abs(d).max() + 1e-12)
            cand = cur - scale * d
        # ---------- block scaling ----------
        elif move < 0.80:
            width = int(rng.integers(min_block, max_block+1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, block_sigma))
            cand[start:start+width] *= factor
        # ---------- random mass transfer ----------
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * transfer_max
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta
        # enforce bounds and minimal sum
        np.clip(cand, 0.0, 1000.0, out=cand)
        if cand.sum() < 0.01:
            cand[0] += 0.01
        cand_score = _ac1_score(cand)
        # update best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            if best_score < target:
                break
        # simulated annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score
        temperature *= decay
        # occasional restart from best
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0
    return [float(x) for x in best]
