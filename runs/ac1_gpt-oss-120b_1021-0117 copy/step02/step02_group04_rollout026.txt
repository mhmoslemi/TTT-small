import math
import time

import numpy as np

# ---------- small utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """Classic binomial step a_i = C(2i,i)/4**i for i=0..n‑1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast evaluation of the AC1 bound (identical to the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


# ---------- main search ----------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to minimise
    the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin before the sandbox kills the process
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- problem size & initialisation -----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        have_parent = True
    except Exception:
        parent = None
        n = 256
        have_parent = False

    L = _next_pow2(2 * n - 1)

    # generate two simple starts and keep the better one
    start1 = (
        parent.astype(np.float64).copy()
        if have_parent
        else _gen_binomial(n)
    )
    start2 = _gen_binomial(n)
    score1 = _ac1_score_fft(start1, n, L)
    score2 = _ac1_score_fft(start2, n, L)
    cur = start1 if score1 <= score2 else start2
    cur_score = min(score1, score2)

    best = cur.copy()
    best_score = cur_score

    # ----- optimisation parameters -----
    lr = 0.20               # base learning rate for gradient step
    lr_decay = 0.9997
    temp = 1.0              # temperature for occasional uphill moves
    temp_decay = 0.9995
    # probability of performing an extra exploratory move each iteration
    extra_move_prob = 0.35

    # ----- main loop -----
    while time.monotonic() < deadline:
        # ---- current convolution and statistics ----
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_b = float(conv.max())
        k_max = int(conv.argmax())
        sum_cur = float(cur.sum())
        cur_score = 2.0 * n * max_b / (sum_cur * sum_cur)

        # ---- gradient‑inspired update (exact bound gradient) ----
        # forward and backward shifted copies (zero‑padded where out‑of‑range)
        shift_fwd = np.empty_like(cur)
        shift_fwd[: n - k_max] = cur[k_max:]
        shift_fwd[n - k_max :] = 0.0

        shift_bwd = np.empty_like(cur)
        shift_bwd[k_max:] = cur[: n - k_max]

        temp *= temp_decay

        if rng.random() < 0.001:
            cur = best.copy()
            lr = 0.20
            temp = 1.0

    # return the best sequence as plain Python floats
    return [float(x) for x in best]
