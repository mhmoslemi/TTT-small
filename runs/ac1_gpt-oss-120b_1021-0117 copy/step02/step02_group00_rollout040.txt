def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv

def _score(a):
    n = a.shape[0]
    conv = _conv_fft(a)
    return 2.0 * n * conv.max() / (a.sum()**2)

def _subgrad_topk(a, k=3):
    n = a.shape[0]
    conv = _conv_fft(a)
    # find top_k indices
    if k >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(conv, -k)[-k:]
    grad = np.zeros_like(a)
    for kk in ks:

        hi = min(kk, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx] += 2.0 * a[kk - idx]
    # average
    grad /= float(k)
    return grad, conv.max()

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Full linear convolution via FFT (real‑valued)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Score = 2 * n * max(conv) / (sum(a)^2)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_fft(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ------------------------------------------------------------
# Sub‑gradient from the k largest convolution entries (k = topk)
# ------------------------------------------------------------
def _subgrad_topk(a: np.ndarray, topk: int = 3):
    n = a.shape[0]
    conv = _conv_fft(a)
    if topk >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(conv, -topk)[-topk:]
    grad = np.zeros_like(a)
    for kk in ks:
        lo = max(0, kk - (n - 1))
        hi = min(kk, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx] += 2.0 * a[kk - idx]
    grad /= float(topk)
    return grad, conv.max()
