import math

import numpy as np

# ---------- utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n‑1 (classic binomial step)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound (identical to the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

            best = cand.copy()
            best_score = cand_score
            if best_score < 1.50274349:
                break

        # acceptance (simulated‑annealing style)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score
            sum_cur = cur.sum()

        # ---- occasional random perturbation (helps escape plateaux) ----
        if rng.random() < 0.12:                     # ~12 % of iterations
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand2 = cur.copy()
            cand2[start:start + width] *= factor
            np.clip(cand2, 0.0, 1000.0, out=cand2)
            s2 = cand2.sum()
            if s2 > 1e-12:
                cand2 *= sum_cur / s2
            cand2_score = _ac1_score_fft(cand2, n, L)
            if cand2_score < best_score:
                best = cand2.copy()
                best_score = cand2_score
                if best_score < 1.50274349:
                    break
            # accept if strictly better (no uphill here)
            if cand2_score < cur_score:
                cur = cand2
                cur_score = cand2_score
                sum_cur = cur.sum()

        # ---- schedules & occasional restart ----
        temperature *= temp_decay
        step_lr *= step_decay
        if rng.random() < 0.001:          # reset to the best found so far
            cur = best.copy()
            cur_score = best_score
            sum_cur = cur.sum()
            temperature = 1.0
            step_lr = 1.0

    # ----- return the best found sequence as plain Python floats -----
    return [float(x) for x in best]
