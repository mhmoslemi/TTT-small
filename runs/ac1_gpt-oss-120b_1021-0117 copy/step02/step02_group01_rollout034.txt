def _gradient_softmax(a, conv, alpha):
    # conv is a*a
    maxval = conv.max()
    # compute weights w = exp(alpha * (conv - maxval)) perhaps to avoid overflow.
    w = np.exp(alpha * (conv - maxval))  # shift to avoid overflow
    w = w / np.sum(w)
    # compute gradient
    a_rev = a[::-1]  # length n
    L = 1 << ((len(a_rev) + len(w) - 1).bit_length())
    # compute FFT of a_rev and w
    Fa = np.fft.rfft(a_rev, n=L)

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    idx = np.where(rho_arr > 0)[0]
    if idx.size:
        rho = idx[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # enforce exact sum=1 (tiny numerical adjustments)
    s = w.sum()
    if s > 0:
        w /= s
    return w

# Analytic sqrt‑series (coefficients a with a*a = 1+…+x^{2n‑2})
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a
