import numpy as np
import time

def _initial_binomial(n: int) -> np.ndarray:
    """Generate decreasing nonnegative coefficients a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2 * i - 1) / (2 * i)
    return a

def _conv_full(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Return full linear convolution of seq with itself (length 2n-1)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)
    return conv[: 2 * seq.size - 1]

def _score(seq: np.ndarray, n: int, fft_len: int) -> float:
    conv = _conv_full(seq, fft_len)
    max_conv = np.max(conv)
    s = seq.sum()
    if s <= 0.0:
        return float('inf')
    return 2.0 * n * max_conv / (s * s)

import numpy as np
import time

def _initial_binomial(n: int) -> np.ndarray:
    """Generate the decreasing coefficients a_i = C(2i,i)/4^i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv_full(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution of `seq` with itself (length 2·len(seq)-1)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)
    return conv[: 2 * seq.size - 1]


def _score(seq: np.ndarray, n: int, fft_len: int) -> float:
    """AC1 upper bound for `seq`."""
    conv = _conv_full(seq, fft_len)
    max_conv = np.max(conv)
    s = seq.sum()
    if s <= 0.0:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)
