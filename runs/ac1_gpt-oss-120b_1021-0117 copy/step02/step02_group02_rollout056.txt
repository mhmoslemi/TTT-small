def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the AC1 bound.
    Uses a smoothed max-convolution gradient with momentum,
    staying on the probability simplex (non‑negative, unit sum).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Time management
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 2.  Base sequence (fallback if the parent is unavailable)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        a = np.clip(parent, 0.0, None)
    except Exception:
        parent = None
        a = None

    if a is None:
        n = 1024                     # reasonable default length
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = int(a.size)

    # ------------------------------------------------------------------
    # 3.  Normalise to the simplex (sum = 1)
    # ------------------------------------------------------------------
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s

    # ------------------------------------------------------------------
    # 4.  Pre‑computations
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1          # length of a*a
    grad_len = 3 * n - 2          # length of a * w_rev
    L = 1 << ((max(conv_len, grad_len) - 1).bit_length())  # power‑of‑two FFT size

    # ------------------------------------------------------------------
    # 5.  Initialise best candidate
    # ------------------------------------------------------------------
    def _score_from_conv(conv):
        """Score = 2 * n * max(conv)  (sum a = 1)."""
        return 2.0 * n * np.max(conv)

    FA = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(FA * FA, n=L)[:conv_len]
    best = a.copy()

    conv_u = np.fft.irfft(FA_u * FA_u, n=L)[:conv_len]

    if uniform_score < best_score:

        best_score = uniform_score

    # ------------------------------------------------------------------

    target_score = 1.50274349

    mu = 0.90                   # momentum factor
    v = np.zeros_like(a)       # momentum buffer
    beta = 5.0                  # soft‑max sharpness
    max_beta = 200.0
    it = 0

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:

            best_score = cur_score
