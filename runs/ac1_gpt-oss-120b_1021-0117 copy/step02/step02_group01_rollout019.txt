            if a[i] > 0.0:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.2
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                conv2 = _self_conv(cand2)
                maxc2 = conv2.max()
                f2 = maxc2 + (1.0 / beta) * np.log(np.exp(beta * (conv2 - maxc2)).sum())
                if f2 < f_cur:
                    a, conv, maxc, f_cur = cand2, conv2, maxc2, f2
                    step = min(1.0, step * 1.02)

        # update best according to the true score
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            if best_score < 1.5027:  # close to target → keep going a bit
                pass

        # increase smoothing parameter toward the true max
        if beta < beta_max:
            beta = min(beta_max, beta * beta_mul)

        # time guard
        if time.monotonic() + 0.3 > deadline:
            break

    # ----------- final refinement: projected gradient on max index ----------
    a_ref = best.copy()
    conv_ref = _self_conv(a_ref)
    step_ref = 0.2
    min_step_ref = 1e-6
    decay_ref = 0.999

    for _ in range(500):
        if time.monotonic() >= deadline:
            break
        max_idx = int(conv_ref.argmax())
        # gradient for the current maximal convolution entry
        g_ref = np.zeros_like(a_ref)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            g_ref[idx] = 2.0 * a_ref[max_idx - idx]
        cand_ref = a_ref - step_ref * g_ref
        cand_ref = _project_simplex(cand_ref)
        cand_score = evaluate_sequence(cand_ref.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best = cand_ref.copy()
            a_ref = cand_ref
            conv_ref = _self_conv(a_ref)
        else:
            step_ref = max(min_step_ref, step_ref * decay_ref)

    # final sanity projection and return
    best = _project_simplex(best)                     # ensure simplex
    return [float(x) for x in best.tolist()]
