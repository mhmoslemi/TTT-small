import math
import time

import numpy as np

# ---------- utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Score = 2·n·max(conv) / (sum(arr)^2) using FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n‑1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _gen_semicircle(n: int) -> np.ndarray:
    """Discrete semicircle: sqrt(1‑x^2) on [-1,1]."""
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i + 1.0 - n) / n
    a = np.sqrt(np.clip(1.0 - x * x, 0.0, 1.0))
    return a


def _gen_hann(n: int) -> np.ndarray:
    """Hann window (raised cosine)."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1.0)))
    return a
