import math, time, numpy as np

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _ac1_score_fft(arr, n, L):
    # compute convolution via FFT
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
    max_b = conv.max()
    s = arr.sum()
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def _combined_mirror(arr, lags, n):
    combined = np.zeros_like(arr)
    for lag in lags:
        if lag < n:
            combined[:lag+1] += arr[lag::-1]
        else:
            start = lag - (n - 1)
            length = n - start
            combined[start:] += arr[:length][::-1]
    return combined

import math
import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast evaluation of the AC1 bound (identical to the evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _combined_mirror(arr: np.ndarray, lags: np.ndarray, n: int) -> np.ndarray:
    """Sum of the mirrored values of `arr` for each lag in `lags`."""
    combined = np.zeros_like(arr)
    for lag in lags:
        if lag < n:
            # i=0..lag  ->  arr[lag-i]
            combined[: lag + 1] += arr[lag::-1]
        else:
            # i=lag-(n-1)..n-1  ->  arr[lag-i]
            start = lag - (n - 1)
            length = n - start
            combined[start:] += arr[:length][::-1]
    return combined
