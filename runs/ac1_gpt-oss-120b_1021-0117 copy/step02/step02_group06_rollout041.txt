import time
import math
import numpy as np

# ---------- utilities ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full auto‑convolution a*a using a real FFT."""
    A = np.fft.rfft(a, n=fft_len)
    C = A * A
    full = np.fft.irfft(C, n=fft_len)
    return full[: 2 * a.shape[0] - 1]


def _score_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact score from a pre‑computed convolution."""
    max_b = float(conv.max())
    return 2.0 * n * max_b / (total * total)


def _softmax_weights(B: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max on the vector B."""
    Bmax = B.max()
    exp = np.exp(beta * (B - Bmax))
    w = exp / exp.sum()
    return w


def _grad_smooth_max(a: np.ndarray, w_rev: np.ndarray, conv_len: int) -> np.ndarray:
    """Gradient of the smoothed max (soft‑max) objective."""
    n = a.shape[0]
    A = np.fft.rfft(a, n=conv_len)
    W = np.fft.rfft(w_rev, n=conv_len)
    full = np.fft.irfft(A * W, n=conv_len)[: 3 * n - 2]
    start = 2 * n - 2
    # gradient component j uses index start‑j
    grad = 2.0 * full[start - np.arange(n)]
    return grad


def _compute_score(a: np.ndarray, fft_len_self: int, n: int) -> float:
    """Fast evaluation of the AC1 upper bound for a normalized vector."""
    conv = _conv_self_fft(a, fft_len_self)
    return _score_from_conv(conv, a.sum(), n)


# ---------- candidate generators ----------

def _initial_candidates(n: int, parent_arr: np.ndarray | None, rng: np.random.Generator):
    """Yield a stream of plausible start vectors (all sum to 1)."""
    idx = np.arange(n, dtype=np.float64)

    # 1. parent if supplied
    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            yield p / s

    # 2. uniform
    yield np.full(n, 1.0 / n)

    # 3. symmetric power shapes (i+1)*(n-i) ^ p
    base = (idx + 1) * (n - idx)
    for p in np.linspace(0.2, 4.0, 20):
        v = base ** p
        if v.sum() > 0:
            yield v / v.sum()

    # 4. sin‑power shapes
    theta = np.pi * (idx + 0.5) / n
    svals = np.sin(theta)
    for p in np.linspace(0.2, 4.0, 20):
        v = svals ** p
        if v.sum() > 0:
            yield v / v.sum()

    # 5. geometric decays r^i
    for r in np.linspace(0.8, 0.999, 20):
        v = r ** idx
        if v.sum() > 0:
            yield v / v.sum()

    # 6. exponential decays exp(-a i)
    for a in np.linspace(0.001, 0.02, 20):
        v = np.exp(-a * idx)
        if v.sum() > 0:

    # 7. random Dirichlet‑type draws

        if v.sum() > 0:
