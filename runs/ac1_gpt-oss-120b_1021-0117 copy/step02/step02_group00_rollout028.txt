import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_fft(a, L, n):
    """Return 2*n*max(conv(a,a)) using FFT of length L."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    return 2.0 * n * conv.max()

        cand1 = cur - step * grad

        if cand2_score < cur_score:

            no_improve = 0

    # normalize a tiny bit to avoid sum < 0.01 (should already be 1)
    if best.sum() < 0.01:
        best = best / best.sum()
    return [float(x) for x in best.tolist()]

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection of a vector onto the probability simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max()
