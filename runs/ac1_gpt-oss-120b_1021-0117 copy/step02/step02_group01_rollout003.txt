import time
import numpy as np

# ----------------------------------------------------------------------
# Project a vector onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the unit simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

# ----------------------------------------------------------------------
# Analytic sqrt‑series (coefficients a with a*a = 1+…+x^{2n‑2})
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a

# ----------------------------------------------------------------------
# Full convolution via FFT (returns the entire array)
# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Convolution a*a (full) using FFT of length L (power of two)."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    return conv

# ----------------------------------------------------------------------
# Max convolution value and all indices attaining it
# ----------------------------------------------------------------------
def _conv_max_and_indices(a: np.ndarray, L: int):
    conv = _conv_full_fft(a, L)
    max_val = float(conv.max())
    # use a tight tolerance to catch possible ties caused by round‑off
    idxs = np.where(np.abs(conv - max_val) <= 1e-12)[0]
    return max_val, idxs

# ----------------------------------------------------------------------
# Subgradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _subgrad_max(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    n = a.size
    g = np.zeros_like(a)
    for k in max_idxs:

        hi = min(k, n - 1)
        j = np.arange(lo, hi + 1, dtype=np.intp)
        # contribution 2 * a_{k-j}
        g[j] += 2.0 * a[k - j]
    return g
