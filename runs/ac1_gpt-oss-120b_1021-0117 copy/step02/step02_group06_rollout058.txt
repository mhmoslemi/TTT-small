import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities (project onto simplex, FFT convolution, reorder)
# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, n: int, fft_len: int) -> np.ndarray:
    """ Full (2n‑1) convolution a*a using real FFT. """
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * n - 1]


def _alternating_order(arr: np.ndarray) -> np.ndarray:
    """Place largest values alternately at the two ends (anti‑clustering)."""
    idx = np.argsort(-arr)                     # descending
    res = np.empty_like(arr)
    left, right = 0, arr.size - 1
    for rank, i in enumerate(idx):
        if (rank & 1) == 0:
            res[left] = arr[i]
            left += 1
        else:
            res[right] = arr[i]
            right -= 1
    return res
