import time
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - css / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic sqrt‑series (a*a == 1+…+x^{2n-2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# Sub‑gradient of the lag‑k linear convolution term
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad

    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    # give ourselves a small safety margin before the sandbox timeout

    # --------------------------------------------------------

    # --------------------------------------------------------

    else:
        a = _sqrt_series(256).astype(np.float64)

    a = a / a.sum()          # normalise to sum = 1
    a = _project_simplex(a)  # enforce simplex constraints

    n = a.shape[0]
    Lfft = 1 << ((2 * n - 1).bit_length())  # next power‑of‑two for FFT

    # --------------------------------------------------------
    # Bookkeeping
    # --------------------------------------------------------
    fv = np.fft.rfft(a, n=Lfft)
    conv = np.fft.irfft(fv * fv, n=Lfft)[: 2 * n - 1]
    best = a.copy()
    best_score = float(2.0 * n * conv.max())
    velocity = np.zeros_like(a)

    # --------------------------------------------------------
    # Parameters
    # --------------------------------------------------------
    eta0 = 0.9               # base step magnitude
    iteration = 0
    beta = 0.8               # momentum coefficient

    # --------------------------------------------------------
    # Optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        fv = np.fft.rfft(a, n=Lfft)

        # ----- early exit if target reached -----

    # --------------------------------------------------------
