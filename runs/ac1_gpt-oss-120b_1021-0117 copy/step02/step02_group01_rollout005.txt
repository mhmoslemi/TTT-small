import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

def _sqrt_series(n: int) -> np.ndarray:
    """Fallback analytic series (the one used in the reference code)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        a[k] = (1.0 - s) / 2.0
    return a

def _fft_convolve(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution via FFT (real input, real output)."""
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)
    return conv[: x.size + y.size - 1]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # initialise a (use parent if available)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = _sqrt_series(256)

    # project onto the simplex so that sum(a)=1
    a = _project_simplex(a)
    if a.sum() == 0.0:                     # extremely unlikely, but safe
        a = np.full_like(a, 1.0 / a.size)

    n = a.size
    L = 1 << ((3 * n - 2).bit_length())    # FFT length >= 3n-2

    # initial convolution a*a (full length 2n-1)
    conv_a = _fft_convolve(a, a, L)

    cur_score = 2.0 * n * conv_a.max()
    best_a = a.copy()
