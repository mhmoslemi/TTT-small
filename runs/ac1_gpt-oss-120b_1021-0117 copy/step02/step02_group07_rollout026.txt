def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    n determined.

    fft_len = 1 << ((2*n - 1).bit_length())

    def conv(seq):
        f = np.fft.rfft(seq, n=fft_len)
        return np.fft.irfft(f * f, n=fft_len)[:2*n-1]

    def score(seq):
        s = np.sum(seq)
        if s <= 0: return float('inf')
        max_b = np.max(conv(seq))
        return float(2.0 * n * max_b / (s * s))

    # init sequence
    init_seq = binomial(n) or parent.
    init_seq = init_seq / np.sum(init_seq)
    best_seq = init_seq.copy()
    best_score = score(best_seq)

    # gradient descent phase
    cur = best_seq.copy()
    cur_score = best_score
    while time.monotonic() < deadline * 0.6:
        # compute conv, find kmax
        conv_cur = conv(cur)
        max_b = conv_cur.max()
        kmax = int(conv_cur.argmax())

        # compute gradient for top indices (maybe topK=1)
        g = np.zeros_like(cur)
        lo = max(0, kmax-(n-1))
        hi = min(kmax, n-1)
        idxs = np.arange(lo, hi+1, dtype=np.int64)
        js = kmax - idxs
        g[idxs] = cur[js]
        diag = idxs == js
        if np.any(diag):
            g[idxs[diag]] = 2.0 * cur[idxs[diag]]

        # direction zero-mean
        d = -(g - g.mean())

        # compute feasible step size
        neg = d < 0
        if np.any(neg):
            max_step = np.min(-cur[neg] / d[neg])
            step = min(0.5 * max_step, 0.05)  # base step cap
        else:
            step = 0.05

        cand = cur + step * d
        cand = np.clip(cand, 0.0, None)
        cand = cand / np.sum(cand)

        cand_score = score(cand)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score

    cur_score = best_score

import time
import numpy as np

# -------------------------------------------------------------
# Helper functions (top level, no closures)
# -------------------------------------------------------------
def _conv_fft(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of seq with itself using FFT (real)."""
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)
    return conv[: 2 * seq.size - 1]


def _score(seq: np.ndarray, n: int, fft_len: int) -> float:
    """Exact score used by the evaluator (no clipping)."""
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf")
    max_b = float(_conv_fft(seq, fft_len).max())
    return 2.0 * n * max_b / (s * s)


def _initial_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a
