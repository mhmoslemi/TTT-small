import time, math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast evaluation of the AC1 bound via FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)

    Heuristic search for a non‑negative step function that lowers the AC1 bound.
    Returns a Python list of floats.
    """

    # ------------------------------------------------------------------

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        cur = np.clip(parent, 0.0, 1000.0)

        n = 1024
        cur = _gen_binomial(n)
        cur = np.clip(cur, 0.0, 1000.0)

    n = int(cur.size)
    L = _next_pow2(2 * n - 1)

    best = cur.copy()
    best_score = _ac1_score(best, n, L)

    alt = _gen_binomial(n)
    alt = np.clip(alt, 0.0, 1000.0)

    if alt_score < best_score:
        best = alt.copy()
        best_score = alt_score

        if mv < prob_grad:

            fa = np.fft.rfft(cur, n=L)

            cand[i] -= delta_i

            cand[t] += delta_i + delta_j
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_score = _ac1_score(cand, n, L)

        # ------------------------------------------------------------------

        if cand_score is not None and cand_score < best_score:

            best_score = cand_score
            if best_score < target:
                break

            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # ------------------------------------------------------------------
    # return the best sequence found (as plain Python floats)
    return [float(x) for x in best]
