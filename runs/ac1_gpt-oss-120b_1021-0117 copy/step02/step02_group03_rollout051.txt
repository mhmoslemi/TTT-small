import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)  # safety margin

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

        n0 = 256
        a = np.empty(n0, dtype=np.float64)
        a[0] = 1.0
        for k in range(1, n0):
            if k == 1:
                s = 0.0
            else:
                s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
            a[k] = (1.0 - s) / 2.0

    n = a.shape[0]
    a = a / max(np.sum(a), 1e-12)          # scale to sum≈1

    # ---- FFT bookkeeping ---------------------------------------------------

    def _score(vec: np.ndarray) -> float:

    best = a.copy()
