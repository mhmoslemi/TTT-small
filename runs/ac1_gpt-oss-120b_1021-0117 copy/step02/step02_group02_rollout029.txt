        it += 1

        # ----- convolution via FFT (real‑valued) --------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= 1.50274349:          # target reached
                break

        # ----- gradient built from the largest few convolutions -------
        # (sub‑gradient of the max‑norm, aggregated over top_k shifts)
        idxs = np.argpartition(-conv, min(top_k, conv_len - 1))[:top_k]
        g = np.zeros_like(a)

        for k in idxs:
            offset = k - (n - 1)
            if offset >= 0:
                idx = np.arange(offset, n)
                partner = k - idx
                g[idx] += 2.0 * a[partner]
            else:
                idx = np.arange(0, n + offset)   # offset is negative
                partner = k - idx
                g[idx] += 2.0 * a[partner]

        # ----- exponentiated‑gradient step ---------------------------
        eta = eta0 / np.sqrt(it)                # diminishing step size
        delta = -eta * g                         # always non‑positive
        # avoid overflow / extreme under‑flow in exp()
        delta = np.clip(delta, -50.0, 0.0)
        a = a * np.exp(delta)
        total = a.sum()
        if total == 0.0:                        # numerical collapse – reset
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / total

        # ----- occasional stochastic jitter to escape flat spots -----
        if it % jitter_every == 0:
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a *= jitter
            a /= a.sum()

    # --------------------------------------------------------------
    # 6.  Final safety checks and return
    # --------------------------------------------------------------
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best]
