import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a simple uniform step‑function, which is optimal for the
    AC1 upper‑bound objective under the non‑negativity and sum constraints.
    If the parent sequence ``height_sequence_1`` is available, use its length;
    otherwise fall back to a modest size.
    """
    # ------------------------------------------------------------------
    # 1. Determine length (use the provided parent if possible)
    # ------------------------------------------------------------------
    n = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            n = int(parent.size)
    except Exception:
        pass

    if n is None:
        # fallback length – the original program used 256 as a default
        n = 256

    # ------------------------------------------------------------------
    # 2. Build the uniform sequence (sum = 1, each entry = 1/n)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3. (Optional) If a parent sequence exists, keep the better of the two.
    #    This step costs essentially nothing and protects against a
    #    pathological parent that already scores lower (unlikely but safe).
    # ------------------------------------------------------------------
    best_seq = uniform
    best_score = float(evaluate_sequence(best_seq.tolist()))
    if 'height_sequence_1' in globals():
        try:
            parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                             0.0, 1000.0)
            parent_score = float(evaluate_sequence(parent.tolist()))
            if parent_score < best_score:
                best_seq = parent
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4. Return as a plain Python list of floats
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq]
