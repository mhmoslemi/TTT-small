def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # Determine n
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        n = int(parent.size)
    except Exception:
        n = 256

    best_seq = None
    best_score = float('inf')
    # try few exponents p
    exponents = [1.0, 2.0, 3.0]
    for p in exponents:
        # generate monotonic increasing power sequence
        idx = np.arange(1, n+1, dtype=np.float64)
        a = idx ** p
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a = a / a_sum  # normalize sum=1
        # evaluate
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_seq = a.copy()
            best_score = sc
        # also try decreasing (mirror)
        a_rev = a[::-1]
        sc_rev = evaluate_sequence(a_rev.tolist())
        if sc_rev < best_score:
            best_seq = a_rev.copy()
            best_score = sc_rev
        # early exit if time low
        if time.monotonic() >= deadline:
            break

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (non‑negative, sum = 1)."""
    if v.size == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
