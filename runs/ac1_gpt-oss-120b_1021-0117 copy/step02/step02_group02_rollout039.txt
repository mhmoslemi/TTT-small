import numpy as np

def _proj_simplex(v):
    # Project onto simplex sum=1, v>=0
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

# ----------------------------------------------------------------------
#  Helper: Euclidean projection onto the probability simplex (sum = 1, x>=0)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # degenerate – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    Heuristic projected gradient descent with momentum and adaptive active‑set.
    Starts from the supplied `height_sequence_1` (if present) or a uniform

    maximal convolution value.  Returns the best sequence encountered before
    the budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------

    try:

    except Exception:
        a = None

    if a is not None:
        n = int(a.size)
    else:
        n = 256
        a = np.ones(n, dtype=np.float64)

    # -----------------------------------------------------------------
    # 3.  Normalisation – fix ℓ₁‑norm to sqrt(2 n) (scale‑invariant)
    # -----------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    total = a.sum()
    if total < 0.01:                     # degenerate fallback
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / total                    # now sums to 1
    a = _proj_simplex(a) * target_sum   # enforce exact ℓ₁‑norm

    # -----------------------------------------------------------------
    # 4.  Initialise bookkeeping
    # -----------------------------------------------------------------
    # pre‑compute FFT length (next power of two)
    conv_len = 2 * n - 1

    sum_a = best.sum()

    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    best_score = 2.0 * n * conv.max() / (sum_a * sum_a)

    # momentum buffer
    momentum = np.zeros_like(a)

    # hyper‑parameters
    eta0 = 0.5                     # initial step magnitude
    beta = 0.9                     # momentum decay
    it = 0
    tol_factor = 0.01              # start tolerance for active set (1 %)

    # -----------------------------------------------------------------
    # 5.  Main optimisation loop
    # -----------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- convolution using FFT (real) ---------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() * a.sum())

        # ---- keep best ----------------------------------------------------
        if cur_score < best_score:
            best = a.copy()

        act_idxs = np.where(conv >= max_conv - eps)[0]

        g = np.zeros_like(a)

            inv = 1.0 / offsets.size
            for off in offsets:
                if off >= 0:
                    g[: n - off] += 2.0 * a[off:]
                else:
                    g[-off:] += 2.0 * a[: n + off]
            g *= inv

        norm_g = np.linalg.norm(g)

        eta = eta0 / np.sqrt(it)

    # 6.  Return the best sequence as plain Python floats
    # --------------------------------------------------------------
    return [float(v) for v in best]
