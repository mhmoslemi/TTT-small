def conv_full(a, L=None):
    n = a.shape[0]
    if L is None:
        L = 1 << ((2*n-1).bit_length())
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
    return conv
```

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum x = 1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho_idx = np.where(rho_arr > 0)[0]
    if rho_idx.size:
        rho = rho_idx[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
# Full auto‑convolution via FFT (returns the full 2n‑1 vector)
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Compute a*a using an FFT of length L (power of two)."""
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * a.size - 1]
    return conv

# ----------------------------------------------------------------------
# Fallback: simple random positive vector (used when no parent is present)
# ----------------------------------------------------------------------
def _random_simplex_vec(rng: np.random.Generator, n: int) -> np.ndarray:
    v = rng.random(n)
    return _project_simplex(v)

# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    # leave a safety margin for the wrapper
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Initialise the sequence (use provided parent if available)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = _random_simplex_vec(rng, 512)

    # Normalise onto the simplex (sum = 1, non‑negative)
    a = _project_simplex(a)
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two for FFT

    # ------------------------------------------------------------------
    # Adam optimiser state
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # ------------------------------------------------------------------
    # Keep track of the best candidate seen so far
    # ------------------------------------------------------------------
    conv = _conv_full(a, L)
    max_conv = conv.max()
    best_score = 2.0 * n * max_conv
    best = a.copy()

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    base_lr = 0.02          # initial Adam learning rate
    K = min(300, conv.size)  # number of top convolution entries to use
    prob_random = 0.02       # occasional mass‑transfer moves

    while time.monotonic() < deadline:
        t += 1
        # --------------------------------------------------------------
        # Compute convolution and the set of most offending indices
        # --------------------------------------------------------------
        conv = _conv_full(a, L)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        # indices of the K largest entries (hard max + neighbours)
        top_idx = np.argpartition(-conv, K - 1)[:K]

        # --------------------------------------------------------------
        # Gradient (weighted sum of contributions from the top entries)
        # --------------------------------------------------------------
        g = np.zeros_like(a)
        # normalising by the current maximum keeps weights in [0,1]
        if max_conv > 0:

            best = cand

    # Return as plain Python list of floats
    return [float(x) for x in best]
