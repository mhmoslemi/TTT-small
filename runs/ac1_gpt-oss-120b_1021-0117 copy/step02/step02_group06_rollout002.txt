```python
def _project_onto_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _generate_initial_vectors(n, parent):
    cand = []
    if parent is not None:
        p = np.clip(parent, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            cand.append(p / s)
    # Catalan weights
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i-1] * ((2*i)*(2*i-1)) / (i*i*4.0)
    cand.append(w / w.sum())
    # Power shape
    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1) * (n - idx)
    for pwr in (0.5, 1.0, 1.5, 2.0):
        v = base ** pwr
        cand.append(v / v.sum())
    # Sin shape
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    for pwr in (0.5, 1.0, 2.0):
        v = sin_vals ** pwr
        cand.append(v / v.sum())
    # Random Dirichlet / exponential
    rng = np.random.default_rng(12345)
    for _ in range(3):
        v = rng.exponential(1.0, size=n)
        cand.append(v / v.sum())
    # Lognormal random
    for _ in range(3):
        v = np.exp(rng.normal(0.0, 1.0, size=n))
        cand.append(v / v.sum())
    return cand

import time, numpy as np

def _project_onto_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_self_fft(a, n, fft_len):
    A = np.fft.rfft(a, fft_len)
    B = A * A
    full = np.fft.irfft(B, fft_len)
    return full[: 2 * n - 1]

def _generate_initial_vectors(n, parent):
    cand = []
    if parent is not None:
        p = np.clip(parent, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            cand.append(p / s)
    # Catalan-like weights
    w = np.empty(n, dtype=np.float64)

    cand.append(w / w.sum())
    # power shapes (i+1)*(n-i)
