def _project_simplex(v):
    """Project onto probability simplex (sum=1, >=0)."""
    # v is np.ndarray float
    n = v.shape[0]
    # sort decreasing
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Simplex projection (non‑negative, sum = 1)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    # find rho = max{i | u_i - css_i/(i+1) > 0}
    cond = u - css / np.arange(1, n + 1)
    rho = np.where(cond > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _max_convolution(a: np.ndarray, L: int) -> float:
    fv = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fv * fv, n=L)
    conv = conv[: 2 * a.shape[0] - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Gradient of the active lag k (sub‑gradient of max autocorrelation)
# ------------------------------------------------------------
def _grad_for_k(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    g[idx] = a[partner]
    grad = 2.0 * g
    # correction for the double‑counted centre when k is even
    if k % 2 == 0:
        j_center = k // 2
        if 0 <= j_center < n:
            grad[j_center] -= a[j_center]
    return grad


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ----------------------------------------------------------------
    # Initialise from parent if present, otherwise start from a Dirichlet
    # ----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 256
        a = rng.dirichlet(np.ones(n0)).astype(np.float64)

    n = a.shape[0]
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / a_sum
    a = _project_simplex(a)

    # ----------------------------------------------------------------
    # Pre‑compute FFT length and bookkeeping
    # ----------------------------------------------------------------
    Lfft = 1 << ((2 * n - 1).bit_length())          # next power of two
    best = a.copy()
    best_score = 2.0 * n * _max_convolution(a, Lfft)

    # algorithmic hyper‑parameters
    eta0 = 0.6                     # base step magnitude
    momentum_coeff = 0.9
    momentum = np.zeros_like(a)
    itr = 0
    no_imp = 0                     # iterations without improvement
    p_random = 0.02               # probability of a random mass‑transfer
    p_accept = 0.005              # occasional uphill acceptance

    while time.monotonic() < deadline:
        itr += 1

        # ----------------------------------------------------------------
        # Current autocorrelation and active lag
        # ----------------------------------------------------------------
        fv = np.fft.rfft(a, n=Lfft)
        conv_full = np.fft.irfft(fv * fv, n=Lfft)
        conv_full = conv_full[: 2 * n - 1]
        M = float(conv_full.max())
        k = int(conv_full.argmax())
        cur_score = 2.0 * n * M

        # ----------------------------------------------------------------
        # Sub‑gradient at the active lag and momentum update
        # ----------------------------------------------------------------
        grad = _grad_for_k(a, k)
        momentum = momentum_coeff * momentum + (1.0 - momentum_coeff) * grad

        # ----------------------------------------------------------------
        # Projected gradient step
        # ----------------------------------------------------------------
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            eta = eta0 / np.sqrt(itr)
            cand = a - eta * (momentum / norm)
        else:
            cand = a.copy()
        cand = _project_simplex(cand)

        # ----------------------------------------------------------------
        # Occasional random mass transfer for escaping plateaus
        # ----------------------------------------------------------------
        if rng.random() < p_random:
            i = rng.integers(0, n)
            if cand[i] > 1e-12:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.12
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)

        # ----------------------------------------------------------------
        # Evaluate candidate (objective is scale‑invariant ⇒ sum≈1)
        # ----------------------------------------------------------------
        cand_M = _max_convolution(cand, Lfft)
        cand_score = 2.0 * n * cand_M

        # ----------------------------------------------------------------
        # Keep the best ever seen
        # ----------------------------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            no_imp = 0
        else:
            no_imp += 1

        # ----------------------------------------------------------------
        # Accept if improvement or with a tiny uphill probability
        # ----------------------------------------------------------------
        if cand_score < cur_score or rng.random() < p_accept:
            a = cand
            # keep momentum (helps when we continue)

        # ----------------------------------------------------------------
        # If stagnating, restart from a fresh random Dirichlet point
        # ----------------------------------------------------------------
        if no_imp > 2500:
            a = rng.dirichlet(np.ones(n)).astype(np.float64)
            a = _project_simplex(a)
            momentum = np.zeros_like(a)
            no_imp = 0

        # ----------------------------------------------------------------
        # Early exit if target achieved
        # ----------------------------------------------------------------
        if best_score < 1.5028:
            break

    # --------------------------------------------------------------------
    # Return the best sequence as a plain Python list
    # --------------------------------------------------------------------
    return [float(x) for x in best]
