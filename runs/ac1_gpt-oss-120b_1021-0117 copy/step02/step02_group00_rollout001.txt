    def project_simplex(v):

        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1
        rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

        L = 1 << ((2*x.shape[0] - 1).bit_length())
        fx = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(fx*fx, n=L)[:2*x.shape[0]-1]

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Convolution via FFT (full linear convolution)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Gradient of weighted sum of autocorrelation entries
# ------------------------------------------------------------
def _weighted_grad(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    # gradient g_i = 2 * sum_k w_k * a_{k-i}
    n = a.size
    a_rev = a[::-1]
    L = 1 << ((len(w) + n - 1).bit_length())
    fw = np.fft.rfft(w, n=L)
    fa = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(fw * fa, n=L)[: len(w) + n - 1]
    # slice the centre part (indices n-1 … n-1+n-1)
    g = 2.0 * full[n - 1 : n - 1 + n]
    return g


# ------------------------------------------------------------
# Exact score (identical to the official evaluator)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.size
    conv = _conv_fft(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)
