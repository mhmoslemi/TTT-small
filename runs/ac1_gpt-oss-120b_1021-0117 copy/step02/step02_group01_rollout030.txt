def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # init sequence
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # generate sqrt series (but better maybe uniform)
        a = np.ones(256, dtype=np.float64)  # placeholder? need to get better.

    # Normalize onto simplex sum=1. We'll enforce positivity
    # But ensure it's not all zeros.
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        a = np.full_like(a, 1.0 / a.size)  # uniform
        total = 1.0
    a = a / total  # sum=1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    # compute initial conv
    # Helper function to compute conv array
    def full_conv(x):
        fv = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[:2*n-1]
        return conv

    conv = full_conv(a)
    max_val = conv.max()
    cur_score = 2.0 * n * max_val
    best = a.copy()
    best_score = cur_score

    step = 0.2
    min_step = 5e-5
    decay = 0.9992
    # For numeric tolerance to identify max indices
    eps = 1e-12

    while time.monotonic() < deadline:
        # compute full conv each iteration (cost acceptable)
        conv = full_conv(a)
        max_val = conv.max()
        # mask for max indices
        mask = conv >= max_val - eps
        # get indices
        max_idx_arr = np.nonzero(mask)[0]  # array of indices
        # compute gradient g
        g = np.zeros_like(a)
        # For each max index, compute contributions
        for k in max_idx_arr:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # for j in [lo, hi], contribution = 2 * a[k - j]
            # we can fill slice
            # g[j] += 2 * a[k - j]
            j = np.arange(lo, hi + 1)
            g[j] += 2.0 * a[k - j]
        # gradient descent step (subgradient for max)
        cand = a - step * g
        cand = np.maximum(cand, 0.0)
        # project onto simplex to ensure sum 1
        # implement projection
        # Use the previous _project_simplex code
        # We can reuse function defined outside, but define here.
        # Actually define function _project_simplex top-level.
        cand = _project_simplex(cand)

        # evaluate candidate
        conv_c = full_conv(cand)
        max_val_c = conv_c.max()
        cand_score = 2.0 * n * max_val_c

        # accept if improvement
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            # occasional accept with probability
            if rng.random() < np.exp(-(cand_score - cur_score) / max(step, 1e-9)):
                a = cand
                cur_score = cand_score

        # random mass transfer
        if rng.random() < 0.02:
            # pick two distinct indices
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                # Transfer random fraction of a[i] to a[j]
                frac = rng.random() * 0.5
                delta = a[i] * frac
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                conv2 = full_conv(cand2)
                cand2_score = 2.0 * n * conv2.max()
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()
        # decay step
        step = max(min_step, step * decay)

        # early exit if time near limit
        if time.monotonic() + 0.5 > deadline:
            break

    return [float(x) for x in best]

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex (sum = 1, x >= 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    # Clip negatives, then renormalise (fast, not the exact Euclidean proj,
    # but sufficient for our heuristic optimisation).
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # uniform fallback – never happens in normal flow
        v.fill(1.0 / v.size)
    else:
        v /= s
    return v


# ----------------------------------------------------------------------
# Analytic sqrt‑series (a*a = 1 + x + … + x^{2n‑2})
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ----------------------------------------------------------------------
# Full linear convolution via FFT (returns the whole 2n‑1 array)
# ----------------------------------------------------------------------
def _full_conv(x: np.ndarray, L: int) -> np.ndarray:
    fv = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * x.size - 1]
    return conv

                cand2[j] += delta
                cand2 = _project_simplex(cand2)
                conv2 = _full_conv(cand2, L)
                cand2_score = 2.0 * n * conv2.max()
                if cand2_score < cur_score:
                    a = cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best = cand2.copy()

        # ----- step‑size decay (ensures eventual convergence) -----
        step = max(min_step, step * decay)

        # cheap early stop if we are about to hit the deadline
        if time.monotonic() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # Return the best solution found (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
