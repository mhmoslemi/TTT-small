We need to define simplex projection: Given vector v ∈ ℝⁿ (maybe negative after step), we want to find x that minimizes ||x - v||_2 subject to x_i >=0 and sum_i x_i = S (desired sum). Implementation via sort and threshold, O(n). Code:

def project_onto_simplex(v, s=1.0):
    # v: np.ndarray
    # return x
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- problem size & initial candidates -----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256

    L = _next_pow2(2 * n - 1)

    # generate a few simple seeds
    seeds = []

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    seeds.append(uniform)

    binom = _gen_binomial(n).astype(np.float64)
    binom = binom / binom.sum()
    seeds.append(binom)

    if parent is not None:
        p = parent.astype(np.float64)
        p = p / p.sum()
        seeds.append(p)

    # pick the best starting point
    best = None
    best_score = float('inf')
    for cand in seeds:
        sc = _ac1_score_fft(cand, n, L)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    cur = best.copy()
    cur_score = best_score

    # ----- optimisation parameters -----
    eta = 0.2          # base step size for gradient descent
    decay = 0.95       # factor to shrink eta when stuck
    temp = 1.0
    temp_decay = 0.999

    # ----- main loop -----
    while time.monotonic() < deadline:
        # convolution & active lag
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_b = float(conv.max())
        k_max = int(conv.argmax())
        cur_score = 2.0 * n * max_b / (cur.sum() ** 2)

        # gradient of the active lag
        g = np.zeros_like(cur)
        if k_max < n:
            g[: k_max + 1] = cur[k_max :: -1]
        else:
            start = k_max - (n - 1)
            g[start:] = cur[start:][::-1]

        improved = False
        step = eta
        # try a few decreasing step sizes
        for _ in range(6):
            cand = cur - step * g
            cand = _project_simplex(cand, 1.0)
            cand_score = _ac1_score_fft(cand, n, L)
            if cand_score < cur_score - 1e-12:
                cur = cand
                cur_score = cand_score
                improved = True
                eta = min(1.0, step * 1.2)
                break
            step *= 0.5

        if not improved:
            # occasional random mass‑transfer move
            if rng.random() < 0.03:
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    delta = cur[i] * rng.random() * 0.3
                    cand = cur.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = _project_simplex(cand, 1.0)
                    cand_score = _ac1_score_fft(cand, n, L)
                    if cand_score < cur_score:
                        cur = cand
                        cur_score = cand_score
                        improved = True

            # simple simulated‑annealing acceptance
            if not improved and rng.random() < math.exp(-(cur_score - best_score) / (temp + 1e-12)):
                pass

            eta = max(eta * decay, 1e-6)

        # keep the overall best
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score
            if best_score < 1.50274349:
                break

        temp *= temp_decay

        # occasional restart from the best found so far
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            eta = 0.2
            temp = 1.0

    # final safety clamp
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
