import time, math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x (x >= 1)."""
    return 1 << (x - 1).bit_length()

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 upper bound using FFT (no clipping, assumes arr is non‑negative)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)
