def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    rng = np.random.default_rng(seed)

    deadline = time.monotonic() + budget_s - 0.5

    # Initialize a
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 256
        a = np.ones(n) / n
    # Normalize to sum=1 by projecting onto simplex (makes sum exactly 1, but we can just scale)
    # protect against negative values: set negative to 0, then project.
    a = np.maximum(a, 0)
    if np.sum(a) == 0:
        a[0] = 1.0
    a = a / np.sum(a)  # sum 1

    # optionally, project onto simplex for the case sum>0; but scaling suffices.

    n = a.shape[0]

    # predefine projection function
    def project_simplex(v):
        # returns vector with sum 1, nonneg
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho_arr = u - css / np.arange(1, len(v) + 1)
        rho = np.where(rho_arr > 0)[0]
        if rho.size > 0:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        else:
            theta = 0.0
        w = np.maximum(v - theta, 0.0)
        # due to numeric, ensure sum close to 1
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
        else:
            w[0] = 1.0
        return w

    # For evaluating: we can use evaluator function given; we can compute custom quick score using known sum 1.
    # But we will just compute conv max and compute score = 2*n*max.
    # compute full convolution via FFT:
    L = 1 << ((2 * n - 1).bit_length())  # power of 2 >= 2n-1

    def conv_max(a):
        fv = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        max_val = conv.max()
        max_idx = int(conv.argmax())
        return max_val, max_idx, conv

    # Initial evaluation
    max_val, max_idx, conv_full = conv_max(a)
    cur_score = 2.0 * n * max_val
    best_a = a.copy()
    best_score = cur_score

    # Parameter schedule
    step_size = 0.2
    min_step = 1e-7
    decay_factor = 0.99

    # tolerance to define near peak indices
    peak_rel_tol = 0.9999  # includes values >=0.9999*max

    # main loop
    while time.monotonic() < deadline:
        # compute gradient based on near-peak indices
        max_val, max_idx, conv_full = conv_max(a)  # compute convolution anew each iter
        # Determine strong peak indices
        threshold = max_val * peak_rel_tol
        peak_idxs = np.where(conv_full >= threshold)[0]
        # accumulate gradient
        g = np.zeros_like(a)
        for k in peak_idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # Grad contributions: for i in [lo, hi], g[i] += a[k-i]
            # vectorized
            i_range = np.arange(lo, hi + 1)
            g[i_range] += a[k - i_range]
        g = 2.0 * g

        # direction to reduce max: negative gradient
        # However ensure step alignment with constraints; we use line search.
        # Try several step sizes decreasing.
        improved = False
        step = step_size
        while step >= min_step:
            # propose candidate: a - step*g, then project onto simplex
            cand = a - step * g
            cand = project_simplex(cand)
            cand_max_val, _, _ = conv_max(cand)
            cand_score = 2.0 * n * cand_max_val
            if cand_score < cur_score - 1e-12:
                # accept
                a = cand
                cur_score = cand_score
                max_val = cand_max_val
                # update best if needed
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                improved = True
                break
            else:
                step *= 0.5
        # If no improvement with gradient step, do random small perturbation
        if not improved:
            # random neighbor: pick i->j transfer
            i = rng.integers(0, n)
            if a[i] > 0:
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                delta_frac = rng.random() * 0.1  # up to 10% of a[i]
                delta = a[i] * delta_frac
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = project_simplex(cand)
                cand_max_val, _, _ = conv_max(cand)
                cand_score = 2.0 * n * cand_max_val
                if cand_score < cur_score:
                    a = cand
                    cur_score = cand_score
                    # update best
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()
        # Decay step size gradually
        step_size = max(min_step, step_size * decay_factor)

        # early exit if close to target
        if best_score < 1.5027:
            break

        # safety: break if too many iterations? Allow as many.
        # inform runnable.

    # Return best_a list
    return [float(x) for x in best_a]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # initialization
    if 'height_sequence_1' in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback simple exponential decay: maybe a geometric progression.
        n_fallback = 256
        init = np.exp(-np.arange(n_fallback) / 20.0)  # decaying.
    # ensure non-neg
    init = np.maximum(init, 0.0)
    s = init.sum()
    if s == 0:
        init[0] = 1.0
        s = 1.0
    a = init / s  # sum to 1

    n = a.shape[0]

    # projection onto simplex (sum=1, nonneg)
    def project_simplex(v):
        # Duchi et al.
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho_arr = u - css / np.arange(1, len(v) + 1)
        rho = np.where(rho_arr > 0)[0]
        if rho.size:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        else:
            theta = 0.0
        w = np.maximum(v - theta, 0.0)
        return w

    # conv max via FFT
    L = 1 << ((2 * n - 1).bit_length())
    def conv_max_and_full(vec):
        fv = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        max_val = conv.max()
        max_idx = int(conv.argmax())
        return max_val, max_idx, conv

    # initial evaluation
    max_val, max_idx, conv_full = conv_max_and_full(a)
    cur_score = 2.0 * n * max_val
    best_a = a.copy()
    best_score = cur_score

    # parameters
    step_size = 0.2
    min_step = 1e-7
    decay = 0.999
    peak_rel_tol = 0.9999
    random_moves_per_iter = 2
    # temperature for occasional uphill acceptance (if wanted)
    # not used for now

    while time.monotonic() < deadline:
        # recompute convolution to get up-to-date gradient
        max_val, max_idx, conv_full = conv_max_and_full(a)

        # identify near-peak indices to smooth gradient
        thresh = max_val * peak_rel_tol

        # compute gradient g = 2 * sum_{k in peaks} a_{k-i} where defined
        g = np.zeros_like(a)
        for k in peak_idxs:

            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            g[i_range] += a[k - i_range]

        # Bound step to avoid negative entries before projection
        # We simply try reducing step until improvement
        accepted = False

        while trial_step >= min_step:
            cand = a - trial_step * g
            cand = project_simplex(cand)
            cand_max, _, _ = conv_max_and_full(cand)
            cand_score = 2.0 * n * cand_max
            if cand_score < cur_score - 1e-14:
                # accept
                a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                accepted = True
                break
            else:
                trial_step *= 0.5
        if not accepted:

            for _ in range(random_moves_per_iter):
                i = rng.integers(0, n)
                if a[i] <= 0.0:

        # also check time left (provide a margin for cleanup)
        if time.monotonic() + 0.5 >= deadline:
            break

    # Return best solution as list of floats
    return [float(x) for x in best_a]

import time
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x >= 0, sum x = 1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, len(v) + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Helper: maximum of the full self‑convolution (FFT, power‑of‑2)
# ------------------------------------------------------------
def _conv_max_full(vec: np.ndarray, L: int):
    """Return (max value, index of max, full convolution) of vec * vec."""
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * len(vec) - 1]
    max_val = float(conv.max())
    max_idx = int(conv.argmax())
    return max_val, max_idx, conv
