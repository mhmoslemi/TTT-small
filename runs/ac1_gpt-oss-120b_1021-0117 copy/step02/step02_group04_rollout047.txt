def _next_pow2(x):
    return 1 << (x-1).bit_length()

def _ac1_score_fft(arr, n, L):
    # arr assumed 1D numpy array
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast evaluation of the AC1 bound (identical to the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

    Produce a non‑negative step‑function (list of floats) with a low AC1 bound.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence found.
    """

    # -------------------------------------------------
    # problem size: use the supplied parent if present
    # -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256

    L = _next_pow2(2 * n - 1)

    best = None
    best_score = float('inf')

    # -------------------------------------------------
    # utility to evaluate a candidate and possibly store it
    # -------------------------------------------------
    def _register(arr):
        nonlocal best, best_score
        arr = np.clip(arr, 0.0, 1000.0)
        s = arr.sum()
        if s < 1e-12:
            return arr, float('inf')
        arr = arr / s               # scale‑invariant normalization (sum = 1)
        score = _ac1_score_fft(arr, n, L)
        if score < best_score:
            best_score = score
            best = arr.copy()
        return arr, score

    # -------------------------------------------------
    # initial candidates
    # -------------------------------------------------
    if parent is not None:
        _, _ = _register(parent.astype(np.float64))

    # 1. Simple exponential a_i = exp(-p * i / n)
    for p in np.linspace(0.2, 8.0, 80):
        if time.monotonic() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-p * i / n)
        _, _ = _register(a)

    # 2. Stretched exponential a_i = exp(-p * (i/(n‑1))**q)
    i_norm = np.arange(n, dtype=np.float64) / (n - 1)

            if time.monotonic() >= deadline:
                break

    # 3. Power‑law a_i = (i+1)^{-beta}

    for beta in np.linspace(0.3, 3.0, 60):

            break
        a = 1.0 / (idx ** beta)
        _, _ = _register(a)

    # fallback to uniform if everything else failed
    if best is None:

    # -------------------------------------------------
