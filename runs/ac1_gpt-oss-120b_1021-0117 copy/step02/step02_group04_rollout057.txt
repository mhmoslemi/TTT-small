def _project_to_simplex(v, z):
    # v is np.ndarray
    # returns projection onto {x >=0, sum x = z}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 bound using FFT (matches the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

    """Classic binomial step a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0

    candidates.append(bin_seq)

    # pick the better of the two as starting point
    cur = min(candidates, key=lambda a: _ac1_score_fft(a, n, L))
    cur = np.clip(cur, 0.0, 1000.0, out=cur)  # safety

    cur_score = _ac1_score_fft(cur, n, L)
    best = cur.copy()
    best_score = cur_score

    # ---- optimisation parameters ----
    avg = max(cur.mean(), 1e-12)
    step = 0.5 * avg                # initial learning rate
    step_decay = 0.9997
    min_step = 1e-8
    random_move_prob = 0.02

    # ---- main loop ----
    while time.monotonic() < deadline:
        # current convolution and its maximum
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        sum_cur = float(cur.sum())
        k_max = int(conv.argmax())

        # gradient step with simple back‑tracking line search
        grad_f = _gradient_f(cur, conv, k_max, sum_cur)
        step_try = step
        improved = False
        for _ in range(5):
            cand = cur - step_try * grad_f
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_score = _ac1_score_fft(cand, n, L)
            if cand_score < cur_score:
                improved = True
                break
            step_try *= 0.5
        if improved:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                if best_score < 1.50274349:
                    break
        else:
            # occasional uphill acceptance (tiny probability)
            if rng.random() < 0.001:
                cur = cand
                cur_score = cand_score

        # ----- occasional stochastic moves -----
        if rng.random() < random_move_prob:
            if rng.random() < 0.5:
                # random block scaling
                width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                factor = math.exp(rng.normal(0.0, 0.08))
                cand2 = cur.copy()
                cand2[start:start + width] *= factor
                np.clip(cand2, 0.0, 1000.0, out=cand2)
                cand2_score = _ac1_score_fft(cand2, n, L)
                if cand2_score < best_score:
                    best = cand2.copy()
                    best_score = cand2_score
                if cand2_score < cur_score or rng.random() < 0.01:
                    cur = cand2
                    cur_score = cand2_score
            else:
                # modest smoothing move
                w = int(rng.integers(max(3, n // 200), max(7, n // 30)))
                kernel = np.ones(w, dtype=np.float64) / w
                smooth = np.convolve(cur, kernel, mode='same')
                alpha = 0.07
                cand3 = (1.0 - alpha) * cur + alpha * smooth
                np.clip(cand3, 0.0, 1000.0, out=cand3)
                cand3_score = _ac1_score_fft(cand3, n, L)
                if cand3_score < best_score:
                    best = cand3.copy()
                    best_score = cand3_score
                if cand3_score < cur_score or rng.random() < 0.01:
                    cur = cand3
                    cur_score = cand3_score

        # ---- schedules & occasional restart ----
        step = max(step * step_decay, min_step)
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            step = 0.5 * avg

    # ----- safety fallback (should never trigger) -----
    if best_score == float('inf') or np.isnan(best_score):
        S = math.sqrt(2.0 * n)
        best = np.full(n, S / n, dtype=np.float64)

    return [float(x) for x in best]
