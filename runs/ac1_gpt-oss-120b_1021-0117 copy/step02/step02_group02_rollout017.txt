import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Exponentiated‑gradient descent on the max‑convolution objective.
    Starts from the supplied parent (``height_sequence_1``) or a uniform
    fallback, repeatedly:
      * compute a*a via FFT,
      * locate the maximal convolution entry,
      * take a multiplicative (mirror‑descent) step for that shift,
      * renormalise onto the simplex,
    with a diminishing step‑size, occasional jitter and smoothing.
    The best sequence seen before the deadline is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1.  Base sequence (parent -> uniform fallback)
    # --------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        a = parent.copy()
    except Exception:                     # parent unavailable or malformed
        n = 256
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                                   # normalise sum = 1

    # break the exact symmetry of the uniform vector
    a *= np.exp(rng.normal(0.0, 0.01, size=n))
    a /= a.sum()

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    target_score = 1.50274349

    # --------------------------------------------------------------
    # 2.  Pre‑computations for FFT‑based convolution
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two

    # --------------------------------------------------------------
    # 3.  Optimisation loop
    # --------------------------------------------------------------
    it = 0
    eta0 = 1.0                                      # initial step magnitude
    active_margin = 0.12                            # include ~top 12 % of max
    jitter_every = 80                               # jitter interval
    jitter_std = 0.015                              # jitter strength
    smooth_every = 200                              # smoothing interval
    smooth_kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution a*a (FFT) ---------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target_score:
                break

        # ----- active shifts (those close to the maximum) ----------
        thresh = max_conv * (1.0 - active_margin)
        active_ks = np.where(conv >= thresh)[0]
        # limit to a few top shifts to keep the gradient cheap
        if active_ks.size > 20:
            top = np.argsort(-conv[active_ks])[:20]
            active_ks = active_ks[top]

        # ----- aggregated gradient for the active shifts ------------
        g = np.zeros_like(a)
        for k in active_ks:
            offset = k - (n - 1)
            if offset >= 0:
                idx = np.arange(offset, n)
                partner = k - idx
                g[idx] += a[partner]
            else:
                idx = np.arange(0, n + offset)   # offset is negative
                partner = k - idx
                g[idx] += a[partner]
        g *= 2.0

        # ----- multiplicative mirror‑descent update -----------------
        eta = eta0 / np.sqrt(it)
        exponent = -eta * g
        np.clip(exponent, -50.0, 50.0, out=exponent)   # avoid overflow
        a = a * np.exp(exponent)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # ----- occasional random jitter to escape plateaus ---------
        if it % jitter_every == 0:
            jitter_factors = np.exp(rng.normal(0.0, jitter_std, size=n))
            a *= jitter_factors
            a /= a.sum()

        # ----- occasional smoothing of the distribution ------------
        if it % smooth_every == 0:
            a = np.convolve(a, smooth_kernel, mode='same')
            a = np.clip(a, 0.0, None)
            a /= a.sum()

    # final safety check – ensure the saved best really scores lower
    try:
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        pass

    return [float(v) for v in best]
