# ---------- helper utilities ----------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2*n‑1)."""
    n = a.shape[0]
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[: 2 * n - 1]


def _catalan_weights(n: int) -> np.ndarray:
    """Catalan‑style coefficients w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w


# ---------- main optimisation routine ----------

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that minimises the
    AC1 upper bound used by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)  # safety margin

    # ---- parent vector (if supplied) ----
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
    except Exception:
        parent_arr = None

    n = int(parent_arr.shape[0]) if parent_arr is not None else 256
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ---- generate a pool of plausible start vectors (sum=1) ----
    candidates = []

    # 1) parent if available
    if parent_arr is not None and parent_arr.size == n:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            candidates.append(p / s)

    # 2) Catalan shape
    candidates.append(_catalan_weights(n) / _catalan_weights(n).sum())

    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1) * (n - idx)

    # 3) Power shapes (both peaked and edge‑peaked)
    for pwr in (0.5, 1.0, 1.5, 2.0, 3.0, -0.5, -1.0, -1.5, -2.0):
        v = base ** pwr
        candidates.append(v / v.sum())

    # 4) Sin‑power shapes
    theta = np.pi * (idx + 0.5) / n
    sin_vals = np.sin(theta)
    for pwr in (0.5, 1.0, 2.0):
        v = sin_vals ** pwr
        candidates.append(v / v.sum())

    # 5) Uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # ---- pick the best among the pool (no evaluator calls) ----
    best_vec = None
    best_score = np.inf
    for vec in candidates:
        conv = _conv_self_fft(vec, fft_len)
        score = 2.0 * n * conv.max()           # sum(vec)=1 by construction
        if score < best_score:
            best_score = score
            best_vec = vec.copy()
        if best_score <= 1.50274349:
            return best_vec.tolist()

    # ---- sub‑gradient descent on f(p)=max_k (p*p)_k (with sum(p)=1) ----
    a = best_vec.copy()
    best_vec = a.copy()
    iter_no_improve = 0
    max_no_improve = 3000
    alpha0 = 0.8               # initial step magnitude
    min_alpha = 1e-6
    iteration = 0

    while time.monotonic() < deadline:
        conv = _conv_self_fft(a, fft_len)
        max_b = float(conv.max())
        k = int(conv.argmax())
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            iter_no_improve = 0
            if best_score <= 1.50274349:
                break
        else:
            iter_no_improve += 1

        # gradient of the active term (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            g[idx_range] = 2.0 * a[k - idx_range]

        # step size schedule
        t = iteration + 1
        alpha = max(min_alpha, alpha0 / np.sqrt(t))

        # additive update with projection onto simplex
        cand = a - alpha * g
        cand = np.maximum(cand, 0.0)
        cand = _project_onto_simplex(cand)

        # occasional random jitter to escape plateaus
        if rng.random() < 0.02:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            cand = cand * jitter
            cand = _project_onto_simplex(cand)

        a = cand
        iteration += 1

        # restart if stagnated
        if iter_no_improve >= max_no_improve:
            pert = np.exp(rng.normal(0.0, 0.05, size=n))
            a = best_vec * pert
            a = _project_onto_simplex(a)
            iter_no_improve = 0

    return best_vec.tolist()
