import math
import time

import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..n‑1 (classic binomial step)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score_fft(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound (same formula as the official evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic search for a non‑negative step function that reduces the AC1 bound.
    Uses a mixture of weighted mass‑transfer, random block scaling,
    gradient‑inspired updates and mild smoothing, all within a simulated‑annealing
    framework.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- initialise sequence -----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256
    L = _next_pow2(2 * n - 1)

    if parent is not None:
        base = parent.astype(np.float64).copy()
    else:
        base = _gen_binomial(n)

    # normalise to unit sum (scale does not affect the score)
    base_sum = base.sum()
    if base_sum == 0:
        base = np.full_like(base, 1.0 / n)
    else:
        base /= base_sum

    best = base.copy()
    best_score = _ac1_score_fft(best, n, L)

    # start from the better of parent / binomial if both exist
    if parent is not None:
        parent_norm = parent.astype(np.float64)
        parent_norm /= parent_norm.sum()
        parent_score = _ac1_score_fft(parent_norm, n, L)
        if parent_score < best_score:
            best = parent_norm.copy()
            best_score = parent_score

    cur = best.copy()
    cur_score = best_score

    # parameters for the stochastic walk
    temperature = 1.0
    temp_decay = 0.9995
    step_lr = 0.25
    lr_decay = 0.9999
    frac_transfer = 0.2
    smooth_alpha = 0.08
    block_sigma = 0.09

    while time.monotonic() < deadline:
        # ----- current convolution and score -----
        fa_cur = np.fft.rfft(cur, n=L)
        conv_cur = np.fft.irfft(fa_cur * fa_cur, n=L)[: 2 * n - 1]
        max_b = float(conv_cur.max())
        s_max = int(conv_cur.argmax())
        sum_cur = float(cur.sum())
        if sum_cur == 0.0:
            break
        cur_score = 2.0 * n * max_b / (sum_cur * sum_cur)

        # ----- decide move -----
        move_type = rng.random()
        cand = cur.copy()

        # 1) weighted mass transfer guided by the active lag
        if move_type < 0.35:
            idx = s_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            rev = np.zeros_like(cur)
            rev[mask] = cur[idx[mask]]
            weight = cur * rev
            if weight.max() > 0:
                i = int(weight.argmax())
                j = int(weight.argmin())
                delta = cur[i] * frac_transfer * rng.random()
                cand[i] -= delta
                cand[j] += delta
            else:
                # fallback to random transfer
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j:
                    delta = cur[i] * frac_transfer * rng.random()
                    cand[i] -= delta
                    cand[j] += delta

        # 2) random block scaling
        elif move_type < 0.6:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, block_sigma))
            cand[start:start + width] *= factor

            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(cand, kernel, mode='same')
            cand = cand * (1.0 - smooth_alpha) + smooth * smooth_alpha

            # stop early if we beat the target
            if best_score < 1.50274349:

        # ----- simulated‑annealing acceptance -----
        accept = False
        if cand_score < cur_score:

        else:

        if accept:
            cur = cand
            cur_score = cand_score

        # ----- schedule updates & occasional reset -----
        temperature *= temp_decay
        step_lr *= lr_decay
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0
            step_lr = 0.25

    return [float(x) for x in best]
