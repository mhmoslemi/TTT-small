import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (sum=1, non‑neg)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, u.size + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # all entries non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, matrix‑free)
# ----------------------------------------------------------------------
def _conv_array(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Return full convolution a*a (real) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def _score_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Compute the AC1 upper bound from a convolution array."""
    max_b = float(conv.max())
    return 2.0 * n * max_b / (total * total)
