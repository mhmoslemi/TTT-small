import numpy as np, time, math

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _gen_binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def _ac1_score(arr, n, L):
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2*n - 1]
    maxb = conv.max()
    s = arr.sum()
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * maxb / (s * s)

import numpy as np
import time
import math

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _gen_binomial(n: int) -> np.ndarray:
    """Classic Catalan‑type start: a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """Fast AC1 bound using FFT (identical to the evaluator)."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    maxb = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * maxb / (s * s)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic search for a non‑negative step function that minimises the
    AC1 upper bound.  Returns a list of Python floats.
    """
    rng = np.random.default_rng(seed)
    # keep a small safety margin before the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 1) initialise length n and a starting vector
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.size)
        cur = np.clip(parent, 0.0, 1000.0).copy()
    except Exception:
        n = 256
        cur = _gen_binomial(n)

    L = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2) evaluate a few simple parametrised families and keep the best
    best = cur.copy()
    best_score = _ac1_score(best, n, L)

    # binomial variant
    try:
        binom = _gen_binomial(n)
        s = _ac1_score(binom, n, L)
        if s < best_score:
            best = binom.copy()
            best_score = s
    except Exception:
        pass

    # power‑law with a small offset (grid search)
    i = np.arange(n, dtype=np.float64)
    for off in np.linspace(0.5, 20.0, 80):
        a = 1.0 / np.sqrt(i + off)
        a = np.clip(a, 0.0, 1000.0)
        s = _ac1_score(a, n, L)
        if s < best_score:
            best = a.copy()
            best_score = s

    cur = best.copy()
    cur_score = best_score

    target = 1.50274349
    min_block = max(2, n // 200)
    max_block = max(2, n // 12)

    # ------------------------------------------------------------------
    # 3) main stochastic optimisation loop
    while time.monotonic() < deadline:
        # ---------- gradient‑guided mass transfer (most frequent) ----------
        fa = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        kmax = int(np.argmax(conv))

        idx = kmax - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(cur)
        g[mask] = cur[idx[mask]]
        distinct = mask & (idx != np.arange(n))
        g[distinct] *= 2.0                     # each unordered pair appears twice
        mu = g.mean()
        d = mu - g                             # sum(d)=0, moves mass from high‑g to low‑g

        neg = d < 0
        lam_max = np.min(cur[neg] / (-d[neg])) if np.any(neg) else 0.0
        # aggressive step
        lam = lam_max * 0.85
        cand = cur + lam * d
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = _ac1_score(cand, n, L)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                if best_score < target:
                    break
        else:
            # back‑off step
            lam = lam_max * 0.30
            cand = cur + lam * d
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_score = _ac1_score(cand, n, L)
            if cand_score < cur_score:
                cur, cur_score = cand, cand_score
                if cand_score < best_score:
                    best, best_score = cand.copy(), cand_score
                    if best_score < target:
                        break

        # ---------- occasional block‑scaling move ----------
        if rng.random() < 0.25:
            width = int(rng.integers(min_block, max_block + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start:start + width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_score = _ac1_score(cand, n, L)
            if cand_score < cur_score:
                cur, cur_score = cand, cand_score
                if cand_score < best_score:
                    best, best_score = cand.copy(), cand_score
                    if best_score < target:
                        break

        # ---------- random mass transfer ----------
        if rng.random() < 0.30:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * 0.45
                delta = cur[i] * frac
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
                np.clip(cand, 0.0, 1000.0, out=cand)
                cand_score = _ac1_score(cand, n, L)
                if cand_score < cur_score:
                    cur, cur_score = cand, cand_score
                    if cand_score < best_score:
                        best, best_score = cand.copy(), cand_score
                        if best_score < target:
                            break

        # ---------- occasional reset to the current global best ----------
        if rng.random() < 0.001:
            cur, cur_score = best.copy(), best_score

    # ------------------------------------------------------------------
    # 4) final sanity check and return
    result = [float(x) for x in best]
    try:
        if not np.isfinite(evaluate_sequence(result)):
            result = [float(x) for x in np.clip(cur, 0.0, 1000.0)]
    except Exception:
        result = [float(x) for x in np.clip(cur, 0.0, 1000.0)]
    return result
