```python
def compute_c5(h):

    dx = 2.0 / n

    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta) * dx

def smooth_max_grad(h, beta, dx):
    """Returns gradient of smooth_max(corr, beta) * dx with respect to h"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)

    grad = np.zeros(n)
    # Only process significant weights
    thr = np.max(weights) * 1e-4
    sig = np.where(weights > thr)[0]

        w = weights[idx]
        lag = int(idx) - (n - 1)
        if lag >= 0:

                grad[:n_eff] += w * (1.0 - h[lag:])

                grad[la:] += w * (1.0 - h[:n_eff])

            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    return lambda h: smooth_max_val(h, beta, dx)

def make_jac(beta, dx):
    return lambda h: smooth_max_grad(h, beta, dx)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    dx = 2.0 / n
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

                     method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

_g_beta = 100.0
_g_dx = 0.01
_g_n = 320

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _sm_val(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s)) / len(corr)) / beta) * dx

def _sm_gr(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 500)
    w = np.exp(s)
    w /= np.sum(w)
    grad = np.zeros(n)
    thr = w.max() * 1e-3
    for idx in np.where(w > thr)[0]:
        wi = w[idx]
        lag = int(idx) - (n - 1)
        if lag >= 0:
            ne = n - lag
            if ne > 0:
                grad[lag:n] += wi * (1.0 - h[:ne])
                grad[:ne] -= wi * h[lag:n]
        else:
            la = -lag
            ne = n - la
            if ne > 0:
                grad[:ne] += wi * (1.0 - h[la:n])
                grad[la:n] -= wi * h[:ne]
    return grad * dx

def _obj(h):
    return _sm_val(h, _g_beta, _g_dx)

def _grad(h):
    return _sm_gr(h, _g_beta, _g_dx)

def _con_fun(h):
    return np.sum(h) - _g_n / 2.0

def _con_jac(h):
    return np.ones(_g_n)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    global _g_beta, _g_dx, _g_n
    _g_beta = beta
    _g_dx = 2.0 / n
    _g_n = n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        r = minimize(_obj, h0, jac=_grad, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        if r.x is not None:
            return project_h(r.x, target)
        return h0
    except Exception:
        return h0

def lag_surgery(h, target, n, n_iter, t0, budget_s):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    for it in range(n_iter):
        if time.time() - t0 > budget_s - 3:
            break
        c5_cur = compute_c5(cur)
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        worst_idx = np.argmax(corr)
        worst_lag = int(worst_idx) - (n - 1)
        improved = False
        if worst_lag >= 0:
            ne = n - worst_lag
            contribs = cur[:ne] * (1.0 - cur[worst_lag:ne + worst_lag])
            top_p = np.argsort(contribs)[::-1][:8]
            for pi in top_p:
                for (dj, dv) in [(pi, -0.04), (pi + worst_lag, 0.04)]:
                    trial = cur.copy()
                    trial[dj] += dv
                    trial = project_h(trial, target)
                    c_try = compute_c5(trial)
                    if c_try < best_c5 - 1e-15:
                        cur = trial
                        best_c5 = c_try
                        improved = True
                        break
                if improved:
                    break
        else:
            la = -worst_lag
            ne = n - la
            contribs = cur[la:ne + la] * (1.0 - cur[:ne])
            top_p = np.argsort(contribs)[::-1][:8]
            for pi in top_p:
                for (dj, dv) in [(pi + la, -0.04), (pi, 0.04)]:
                    trial = cur.copy()
                    trial[dj] += dv
                    trial = project_h(trial, target)
                    c_try = compute_c5(trial)
                    if c_try < best_c5 - 1e-15:
                        cur = trial
                        best_c5 = c_try
                        improved = True
                        break
                if improved:
                    break
        if not improved and it > 500:
            break
    return cur, best_c5

def binary_perturb(h, target, n, strength):
    b = np.zeros(n)
    ones = np.random.choice(n, size=n // 2, replace=False)
    b[ones] = 1.0
    return project_h(h * (1.0 - strength) + b * strength, target)

def coord_descent_fine(h, target, n, sweeps, t0, budget_s):
    best_c5 = compute_c5(h)
    cur = h.copy()
    for sw in range(sweeps):
        if time.time() - t0 > budget_s - 2:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget_s - 1:
                break
            for d in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                trial = cur.copy()
                trial[j] += d
                if trial[j] < -0.1 or trial[j] > 1.1:
                    continue
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-15:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved:
            break
    return cur, best_c5

def subgrad_refine(h, target, n, n_iter, t0, budget_s, step_size=0.01):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    mom = np.zeros(n)
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget_s - 3:
            break
        g = _sm_gr(cur, 500.0, 2.0 / n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        mom = 0.8 * mom - step_size * g
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            cur = trial
            best_c5 = c_try
            stall = 0
        else:
            stall += 1
            mom *= 0.5
            trial = project_h(cur + mom, target)
            c_try = compute_c5(trial)
            if c_try < best_c5 - 1e-15:
                cur = trial
                best_c5 = c_try
                stall = 0
            else:
                mom = np.zeros(n)
        if stall > 300:
            break
    return cur, best_c5

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(320, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(h_init)

    n = best_n
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    # Phase 1: SLSQP with analytical gradient, increasing beta
    cur = h_init.copy()
    for beta in [200, 500, 1500, 5000]:
        rem = budget_s - (time.time() - t0)
        if rem < 25:
            break
        cur = do_slsqp(cur, n, target, beta, bounds, 4000, 1e-15)
        c_try = compute_c5(cur)
        if c_try < best_c5:
            best_h, best_c5 = cur.copy(), c_try

    # Phase 2: Subgradient refinement
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        cur, c_try = subgrad_refine(best_h, target, n, 5000, t0, budget_s - 15, 0.02)
        if c_try < best_c5:
            best_h, best_c5 = cur, c_try

    # Phase 3: Lag surgery
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        cur, c_try = lag_surgery(best_h, target, n, 4000, t0, budget_s - 25)
        if c_try < best_c5:
            best_h, best_c5 = cur, c_try

    # Phase 4: More SLSQP at very high beta
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        for beta in [10000, 20000, 50000]:
            rem = budget_s - (time.time() - t0)
            if rem < 12:
                break
            cur = do_slsqp(best_h, n, target, beta, bounds, 5000, 1e-15)
            c_try = compute_c5(cur)
            if c_try < best_c5:
                best_h, best_c5 = cur.copy(), c_try

    # Phase 5: Multi-restart with binary perturbations
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        n_restarts = 4
        for rr in range(n_restarts):
            rem = budget_s - (time.time() - t0)
            if rem < 12:
                break
            strength = 0.25 + 0.15 * rr
            hp = binary_perturb(best_h, target, n, strength)
            hp = do_slsqp(hp, n, target, 3000, bounds, 3000, 1e-14)
            hp = do_slsqp(hp, n, target, 10000, bounds, 2000, 1e-15)
            cp = compute_c5(hp)
            if cp < best_c5:
                best_h, best_c5 = hp.copy(), cp
                rem = budget_s - (time.time() - t0)
                if rem > 10:
                    hp, cp = lag_surgery(hp, target, n, 1000, t0, budget_s - 5)
                    if cp < best_c5:
                        best_h, best_c5 = hp, cp

    # Phase 6: Lag surgery again
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        cur, c_try = lag_surgery(best_h, target, n, 3000, t0, budget_s - 8)
        if c_try < best_c5:
            best_h, best_c5 = cur, c_try

    # Phase 7: Final SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        cur = do_slsqp(best_h, n, target, 30000, bounds, 5000, 1e-15)
        c_try = compute_c5(cur)
        if c_try < best_c5:
            best_h, best_c5 = cur.copy(), c_try

    # Phase 8: Fine coordinate descent
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        best_h, best_c5 = coord_descent_fine(best_h, target, n, 40, t0, budget_s - 2)

    return best_h, best_c5, best_n
```
