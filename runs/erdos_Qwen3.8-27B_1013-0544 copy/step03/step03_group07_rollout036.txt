def project_onto_polytope(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    # Use bisection on the Lagrange multiplier
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Fix any remaining error
    diff = target_sum - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 1e-8) & (x < 1.0 - 1e-8)
        if free.sum() > 0:
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_polytope(h, target_sum):
    """Project onto {0<=x<=1, sum(x)=target_sum} via bisection"""
    if abs(h.sum() - target_sum) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 1e-10) & (x < 1.0 - 1e-10)
        if free.sum() > 0:
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    if sw < 1e-300:
        return m * dx, np.zeros(n)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s == 0:
                grad[:] += wk * (1.0 - 2.0 * h)
            else:
                grad[s:] += wk * (1.0 - h[:-s])
                grad[:n-s] += wk * (-h[s:])
        else:
            t = -s
            grad[:n-t] += wk * (1.0 - h[t:])
            grad[t:] += wk * (-h[:n-t])
    grad *= dx
    return val, grad

def fista_optimize(h, max_iters, beta, lr):
    n = len(h)
    target = n / 2.0
    x = project_polytope(h.copy(), target)
    y = x.copy()
    t_k = 1.0
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for k in range(max_iters):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = project_polytope(x + (t_k - 1.0) / t_new * (x - y), target)
        
        _, grad = smoothmax_val_grad(y, beta)
        x_new = project_polytope(y - lr * grad, target)
        
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        
        x = x_new
        t_k = t_new
    
    return best_x, best_v

def slsqp_optimize(h, beta, maxiter=500):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    
    def obj_grad(x):
        _, g = smoothmax_val_grad(x, beta)
        return g

        return x.sum() - target

        return np.ones(n)

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    if sw < 1e-300:
        return m * dx, np.zeros(n)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue

            grad[:] += wk * (1.0 - 2.0 * h)

        else:
            t = -s
            grad[:n-t] += wk * (1.0 - h[t:])
            grad[t:] += wk * (-h[:n-t])
    grad *= dx
    return val, grad

            h_best, best_c5 = h_new, c

    # Phase 2: FISTA fast refinement
    if rem() > 100:
        h_f, c_f = fista_optimize(h_best, 4000, beta=150, lr=0.3)
        if c_f < best_c5:
            h_best, best_c5 = h_f, c_f
        h_new = slsqp_optimize(h_best, 500, maxiter=300)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 3: Coordinate descent (quantized)
    if rem() > 80:
        h_cd, c_cd = coord_descent_1d(h_best, max_sweeps=30)
        if c_cd < best_c5:
            h_best, best_c5 = h_cd, c_cd
        h_new = slsqp_optimize(h_best, 800, maxiter=300)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 4: Multi-scale refinement
    for n_try in [300, 500, 600]:
        if rem() < 160:
            break
        h_i = interp_to_n(h_best, n_try)
        for beta in [100, 400, 1000]:
            if rem() < 55:
                break
            h_new = slsqp_optimize(h_i, beta, maxiter=350)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
                h_i = h_best.copy()
        if rem() > 70:
            h_f, c_f = fista_optimize(h_best, 3000, beta=250, lr=0.2)
            if c_f < best_c5:
                h_best, best_c5 = h_f, c_f
            h_new = slsqp_optimize(h_best, 800, maxiter=250)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
        n = len(h_best)

    # Phase 5: Multi-start perturbations (small, to maintain smoothness)
    n = len(h_best)
    for trial in range(14):
        if rem() < 38:
            break
        scale = 0.003 + 0.004 * (trial % 4)
        pert = np.random.randn(n) * scale
        h_p = project_polytope(h_best + pert, n / 2.0)
        h_new = slsqp_optimize(h_p, 600, maxiter=350)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c
        if rem() > 35:
            h_new2 = slsqp_optimize(h_new, 1200, maxiter=300)
            c2 = compute_c5(h_new2)
            if c2 < best_c5:
                h_best, best_c5 = h_new2, c2

    # Phase 6: Final FISTA + polish
    if rem() > 60:
        h_f, c_f = fista_optimize(h_best, 3000, beta=400, lr=0.15)
        if c_f < best_c5:
            h_best, best_c5 = h_f, c_f
    if rem() > 30:
        h_final = slsqp_optimize(h_best, 2000, maxiter=400)
        c = compute_c5(h_final)
        if c < best_c5:
            h_best, best_c5 = h_final, c

    # Final feasibility
    h_best = project_polytope(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
