import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            if diff > 0:
                idx = np.argsort(h)
            else:
                idx = np.argsort(h)[::-1]
            for i in idx:
                old = h[i]
                h[i] = np.clip(h[i] + diff, 0.0, 1.0)
                diff -= (h[i] - old)
                if abs(diff) < 1e-13:
                    break
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += -h[lag:] * dx
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad

def get_weighted_lags(h, n, dx, k_top, temp):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    center = n - 1
    idx = np.argsort(corr)[::-1][:k_top]
    vals = corr[idx]
    w = np.exp((vals - vals[0]) * temp)
    w /= w.sum()
    lags = [int(i) - center for i in idx]
    return lags, w

def descent_momentum(h, n, target, dx, max_iters, time_fn, min_time, best_state, mu_init=0.82):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    v = np.zeros(n)
    mu = mu_init
    k_top = max(2, min(5, n // 60 + 2))

    it_no_improve = 0

    for it in range(max_iters):
        if time_fn() < min_time:
            break

            lags, weights = get_weighted_lags(cur, n, dx, k_top, 80.0)

        x_look = project_h(cur + mu * v, target)
        grad = np.zeros(n)

            grad += w * grad_for_lag(x_look, lag, n, dx)

        if gnorm < 1e-14:
            break
        grad /= gnorm

        step = 0.04 * (1.0 + 0.3 * np.sin(0.05 * it))
        improved = False
        for _ in range(20):
            h_trial = project_h(x_look - step * grad, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-14:
                v = mu * v + (h_trial - cur)
                cur = h_trial
                cur_c5 = c5_trial
                improved = True
                it_no_improve = 0
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n

            step *= 0.5

        if not improved:

            mu = max(0.5, mu - 0.02)

                    cur = h_trial
                    cur_c5 = c5_trial
                    v = np.zeros(n)

                else:
                    v *= 0.5
    return cur, cur_c5, n

def smooth_max_obj(h, dx, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    shifted = np.clip(beta * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted)) / len(c)) / beta) * dx

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def slsqp_smooth(h, n, target, dx, beta, maxiter):
    bounds = Bounds(np.zeros(n), np.ones(n))
    from functools import partial
    obj = partial(smooth_max_obj, dx=dx, beta=beta)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_h(res.x, target)
    except Exception:
        return h.copy()

            step *= 0.5

    return cur, cur_c5, n
