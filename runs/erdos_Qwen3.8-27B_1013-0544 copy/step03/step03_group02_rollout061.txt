import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Convert to binary sequence with exactly n//2 ones."""
    target = n // 2
    # Use the fractional parts to decide
    idx = np.argsort(-h)[:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -50, 50)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def sml_obj(h_in, args):
    beta, dx = args
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def sum_constraint(x, target):
    return np.sum(x) - target

def sum_constraint_jac(x, target):
    return np.ones(x.shape)

def simulated_annealing_binary(h0, n, target, t0, budget, n_iter=None, max_it=50000):
    """Simulated annealing on binary sequences with swap moves."""
    b = to_binary(h0, n)
    cur_c5 = compute_c5(b)
    best_b = b.copy()
    best_c5 = cur_c5
    
    if n_iter is None:
        n_iter = max_it
    
    T_start = 0.1
    T_end = 0.001
    alpha = (T_end / T_start) ** (1.0 / n_iter)
    
    ones_pos = np.where(b == 1)[0]
    zeros_pos = np.where(b == 0)[0]
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        T = T_start * (alpha ** it)
        
        # Pick a move: swap a 1 with a 0
        i = ones_pos[np.random.randint(len(ones_pos))]
        j = zeros_pos[np.random.randint(len(zeros_pos))]
        
        b[i] = 0
        b[j] = 1
        
        new_c5 = compute_c5(b)
        delta = new_c5 - cur_c5
        
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            # Update position lists
            ones_pos = np.where(b == 1)[0]
            zeros_pos = np.where(b == 0)[0]
            if new_c5 < best_c5:
                best_b = b.copy()
                best_c5 = new_c5
        else:
            b[i] = 1
            b[j] = 0
    
    return best_b, best_c5

def block_swap_sa(h0, n, target, t0, budget, max_it=30000):
    """SA with block swap moves."""
    b = to_binary(h0, n)
    cur_c5 = compute_c5(b)
    best_b = b.copy()
    best_c5 = cur_c5
    
    T_start = 0.05
    T_end = 0.0005
    alpha = (T_end / T_start) ** (1.0 / max_it)
    
    ones_pos = np.where(b == 1)[0]
    zeros_pos = np.where(b == 0)[0]
    
    for it in range(max_it):
        if budget - (time.time() - t0) < 2:
            break
        T = T_start * (alpha ** it)
        
        # Block swap: pick a block of 1s and swap with a block of 0s
        block_size = np.random.choice([1, 1, 1, 2, 2, 3, 4, 5, 8])
        if block_size == 1:
            i = ones_pos[np.random.randint(len(ones_pos))]
            j = zeros_pos[np.random.randint(len(zeros_pos))]
            b[i] = 0; b[j] = 1
            new_c5 = compute_c5(b)
            delta = new_c5 - cur_c5
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
                cur_c5 = new_c5
                ones_pos = np.where(b == 1)[0]
                zeros_pos = np.where(b == 0)[0]
                if new_c5 < best_c5:
                    best_b = b.copy()
                    best_c5 = new_c5
            else:
                b[i] = 1; b[j] = 0
        else:
            # Try to find a block of consecutive 1s of the given size
            # Simplified: just pick block_size random 1s and swap with block_size random 0s
            if len(ones_pos) < block_size or len(zeros_pos) < block_size:
                continue
            idx_ones = np.random.choice(len(ones_pos), block_size, replace=False)
            idx_zeros = np.random.choice(len(zeros_pos), block_size, replace=False)
            ones_to_flip = ones_pos[idx_ones]
            zeros_to_flip = zeros_pos[idx_zeros]
            
            b[ones_to_flip] = 0
            b[zeros_to_flip] = 1

            delta = new_c5 - cur_c5

                zeros_pos = np.where(b == 0)[0]
                if new_c5 < best_c5:

                    best_c5 = new_c5
            else:
                b[ones_to_flip] = 1
                b[zeros_to_flip] = 0
    
    return best_b, best_c5

def continuous_optimize(h0, n, target, t0, budget):
    """SLSQP optimization with smooth max."""
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (target,)}]
    cur = project_h(h0, target)
    best = cur.copy()
    best_c5 = compute_c5(cur)
    
    for beta in [50, 100, 200, 500, 1000]:
        remaining = budget - (time.time() - t0)
        if remaining < 5:
            break
        try:
            r = minimize(sml_obj, cur, args=(beta, 2.0/n), method='SLSQP',
                        bounds=bounds, constraints=cons,
                        options={'maxiter': 5000, 'ftol': 1e-15})
            cur = project_h(r.x, target)
            c = compute_c5(cur)
            if c < best_c5:
                best = cur.copy()
                best_c5 = c
        except Exception:
            pass
    
    return best, best_c5

def coord_descent(h0, n, target, t0, budget, sweeps=30):
    best_c5 = compute_c5(h0)
    cur = h0.copy()
    deltas = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    
    for sw in range(sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-15:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)[:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -50, 50)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def sml_obj(h_in, args):
    beta, dx = args
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def sum_fn(x, target):
    return np.sum(x) - target

def sum_jac(x, target):
    return np.ones(x.shape)

def continuous_opt(h0, n, target, t0, budget, betas=(100, 300, 800)):

    cur = h0.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
