    c = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(c, beta) * dx

def obj_ps(h, dx, p):
    c = np.correlate(h, 1.0 - h, mode='full')
    return power_sum_max(c, p) * dx

def eq_fun(h, ts, n):
    return h.sum() - ts

def eq_jac(h, ts, n):
    return np.ones(n)

def adam_update(m, v, b1, b2, t, lr, grad):
    m = b1*m + (1-b1)*grad
    v = b2*v + (1-b2)*grad*grad
    mh = m / (1 - b1**t)
    vh = v / (1 - b2**t)
    return m, v, lr * mh / (np.sqrt(vh) + 1e-8)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n0 = 210
        x = np.linspace(0, 2, n0, endpoint=False)
        h0 = np.clip(0.5 + 0.3*np.sin(np.pi*x), 0, 1)

    n = len(h0)
    dx = 2.0 / n
    ts = n / 2.0
    h0 = project_feasible(h0, ts)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Adam active-set gradient with curriculum
    h = best_h.copy()
    ma = np.zeros(n)
    va = np.zeros(n)
    ta = 0
    lr_b = 0.005
    stuck = 0
    for it in range(120000):
        if rem() < 350:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        frac = min(1.0, it / 60000.0)
        K = int(3 + 15 * frac)
        temp = 25.0 * np.exp(-2.5 * frac)
        grad = grad_active(h, corr, n, K, max(temp, 1.5))
        ta += 1
        ma, va, step = adam_update(ma, va, 0.9, 0.999, ta, lr_b, grad)
        h_new = project_feasible(h - step, ts)
        c = compute_c5(h_new)
        if c < best_c5:
            h = h_new
            best_h = h_new.copy()
            best_c5 = c
            stuck = 0
        else:
            stuck += 1
            if stuck % 200 == 0:
                h = h_new
        if stuck > 800:
            h += np.random.randn(n) * 0.015
            h = project_feasible(h, ts)
            ma, va = np.zeros(n), np.zeros(n)
            ta = 0
            lr_b = max(0.001, lr_b * 0.75)
            stuck = 0

    # Phase 2: SLSQP refinement with multiple beta
    if rem() > 200:
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (ts, n)}]
        for b in [50, 200, 1000, 5000, 20000, 100000]:
            if rem() < 80:
                break
            x0 = best_h + np.random.randn(n) * 0.006
            x0 = project_feasible(x0, ts)
            try:
                res = minimize(obj_sm, x0, args=(dx, b), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-16})
                hn = project_feasible(res.x, ts)
                c5 = compute_c5(hn)
                if c5 < best_c5:
                    best_h = hn.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 3: trust-constr with power sum
    if rem() > 150:
        lc = LinearConstraint(np.ones(n), ts, ts)
        bnds = Bounds(np.zeros(n), np.ones(n))
        for p in [20, 50, 100, 200]:
            if rem() < 50:
                break
            x0 = best_h + np.random.randn(n) * 0.004
            x0 = project_feasible(x0, ts)
            try:
                res = minimize(obj_ps, x0, args=(dx, p), method='trust-constr',
                               bounds=bnds, constraints=[lc],
                               options={'maxiter': 3000, 'gtol': 1e-10, 'xtol': 1e-12})
                hn = project_feasible(res.x, ts)
                c5 = compute_c5(hn)
                if c5 < best_c5:
                    best_h = hn.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 4: Second Adam pass with oscillating parameters
    if rem() > 120:
        h = best_h.copy()
        ma, va, ta = np.zeros(n), np.zeros(n), 0
        lr_b = 0.003
        stuck = 0
        for it in range(60000):
            if rem() < 50:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            osc = (it % 2000) / 2000.0
            K = 6 + int(8 * osc)
            temp = 12.0 - 8.0 * osc
            grad = grad_active(h, corr, n, K, max(temp, 2.0))
            ta += 1
            ma, va, step = adam_update(ma, va, 0.9, 0.999, ta, lr_b, grad)
            h_new = project_feasible(h - step, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c
                stuck = 0
            else:
                stuck += 1
            if stuck > 400:
                h += np.random.randn(n) * 0.008
                h = project_feasible(h, ts)
                ma, va, ta = np.zeros(n), np.zeros(n), 0
                stuck = 0

    # Phase 5: Resolution exploration
    if rem() > 60:
        n_list = sorted(set([n*2, n-40, n+40, n+80, 300, 400]))
        for n_t in n_list:
            if n_t <= 50 or n_t > 600 or n_t == n or rem() < 25:
                continue
            xo = np.linspace(0, 2, len(best_h), endpoint=False)
            xn = np.linspace(0, 2, n_t, endpoint=False)
            ht = np.interp(xn, xo, best_h)
            ts_t = n_t / 2.0
            ht = project_feasible(ht, ts_t)
            bounds_t = Bounds(np.zeros(n_t), np.ones(n_t))
            cons_t = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (ts_t, n_t)}]
            for b in [5000, 50000]:
                if rem() < 12:
                    break
                try:
                    res = minimize(obj_sm, ht, args=(2.0/n_t, b), method='SLSQP',
                                   bounds=bounds_t, constraints=cons_t,
                                   options={'maxiter': 4000, 'ftol': 1e-14})
                    hn = project_feasible(res.x, ts_t)
                    c5 = compute_c5(hn)
                    if c5 < best_c5:
                        best_h = hn.copy()
                        best_c5 = c5
                        n, dx, ts = n_t, 2.0/n_t, n_t/2.0
                        bounds = Bounds(np.zeros(n), np.ones(n))
                        cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (ts, n)}]
                except Exception:
                    pass

    # Phase 6: Final ultra-high-beta SLSQP polish
    if rem() > 15:
        for b in [200000, 1000000]:
            if rem() < 8:
                break
            try:
                res = minimize(obj_sm, best_h, args=(dx, b), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-17})
                hn = project_feasible(res.x, ts)
                c5 = compute_c5(hn)
                if c5 < best_c5:
                    best_h = hn.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 7: Single-lag micro-polish
    if rem() > 8:
        h = best_h.copy()
        for it in range(10000):
            if rem() < 3:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_w = np.argmax(corr)
            lag = k_w - (n - 1)
            g = np.zeros(n)
            if lag >= 0:
                m = n - lag
                if m > 0:
                    g[:m] += (1.0 - h[lag:lag+m])
                    g[lag:lag+m] -= h[:m]
            else:
                la = -lag
                m = n - la
                if m > 0:
                    g[la:la+m] += (1.0 - h[:m])
                    g[:m] -= h[la:la+m]
            lr = 0.0008 / (1.0 + 0.003*it)
            h_new = project_feasible(h - lr*g, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c
            elif it > 3000:
                break

    best_h = project_feasible(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
