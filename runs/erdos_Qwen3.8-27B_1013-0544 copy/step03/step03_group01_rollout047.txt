def subgrad(h, n, k_star):
    s = k_star - n + 1
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            g[:m] += (1.0 - h[s:])
            g[s:] -= h[:m]
    else:
        t = -s
        m = n - t
        if m > 0:
            g[t:] += (1.0 - h[:m])
            g[:m] -= h[t:]
    return g

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def sgd_refine(h, n, deadline, rng, topk_count=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    no_improve = 0
    for phase in range(50):
        if time.time() > deadline:
            break
        lr = 0.02 * (0.85 ** phase)
        for _ in range(80):
            if time.time() > deadline:
                break
            c = np.correlate(best_h, 1 - best_h, mode='full')
            idx = np.argsort(c)[-topk_count:]
            g = np.zeros(n)
            for ks in idx:
                g += subgrad(best_h, n, ks)
            gn = np.linalg.norm(g)
            if gn < 1e-12:
                no_improve += 1
                break
            h_new = best_h - lr * g / gn
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
                no_improve = 0
            else:
                no_improve += 1
        if no_improve > 5:
            break
    return best_h, best_c

def local_pairs(h, n, deadline, rng, num_iter=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(num_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = (rng.rand() - 0.5) * 0.03
        hi = best_h[i] + d
        hj = best_h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            oi, oj = best_h[i], best_h[j]
            best_h[i], best_h[j] = hi, hj
            c = compute_c5(best_h, n)
            if c < best_c:
                best_c = c
            else:
                best_h[i], best_h[j] = oi, oj
    return best_h, best_c

def make_starts(n, rng, h_ref=None):
    x = np.linspace(0, 2, n)
    starts = [np.full(n, 0.5)]
    for k in range(1, 10):
        s = np.clip(0.5 + 0.3 * np.sin(k * np.pi * x) + 0.1 * np.cos(2 * k * np.pi * x), 0, 1)
        starts.append(project_feasible(s, n))
    for _ in range(12):
        starts.append(project_feasible(rng.uniform(0, 1, n), n))
    for period in [2, 3, 4, 5, 7, 11]:
        if period < n:
            pat = np.array([1.0 if (i % period) < period // 2 else 0.0 for i in range(n)])
            starts.append(project_feasible(pat, n))
    for amp in [0.15, 0.25, 0.35]:
        s = np.clip(0.5 + amp * np.sin(np.pi * x) + 0.08 * np.sin(3 * np.pi * x), 0, 1)
        starts.append(project_feasible(s, n))
    if h_ref is not None and len(h_ref) != n:
        try:
            f = interp1d(np.linspace(0, 2, len(h_ref)), h_ref, kind='cubic', fill_value='extrapolate')
            starts.append(project_feasible(np.clip(f(x), 0, 1), n))
        except Exception:
            pass
    return starts

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # SLSQP escalation
    for alpha in [10000, 100000, 500000, 2000000, 8000000]:
        if time.time() > deadline - 15:
            break
        h_new = slsqp(best_h, n, alpha, 4000, 1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new

    # SGD refinement
    if time.time() < deadline - 10:
        h_new, c_new = sgd_refine(best_h, n, min(deadline - 5, time.time() + 80), rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Final SLSQP polish
    if time.time() < deadline - 5:
        h_new = slsqp(best_h, n, 5000000, 3000, 1e-17)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 2

    try:
        h0 = project_feasible(np.array(initial_h_values, dtype=float), len(initial_h_values))
    except Exception:
        h0 = project_feasible(np.full(100, 0.5), 100)
    n0 = len(h0)

    best_h = h0.copy()
    best_c = compute_c5(h0, n0)

    # Phase 1: Heavy optimization at n0 (50% of budget)
    t1 = t0 + (deadline - t0) * 0.50
    best_h, best_c = optimize_at_n(best_h, n0, t1, rng)

    # Phase 2: SGD + pairs at n0 (15% of budget)
    t2 = t0 + (deadline - t0) * 0.65
    h_new, c_new = sgd_refine(best_h, n0, t2, rng, topk_count=5)
    if c_new < best_c:
        best_c = c_new
        best_h = h_new
    if time.time() < t2:
        best_h, best_c = local_pairs(best_h, n0, t2, rng, num_iter=100000)

    # Phase 3: Multi-n exploration (25% of budget)
    t3 = t0 + (deadline - t0) * 0.90
    other_ns = [x for x in [80, 90, 96, 110, 120, 150] if x != n0][:5]
    for nc in other_ns:
        if time.time() > t3 - 20:
            break
        starts = make_starts(nc, rng, best_h)
        for s in starts[:10]:
            if time.time() > t3 - 10:
                break
            hs = project_feasible(s, nc)
            cs = compute_c5(hs, nc)
            if cs < best_c:
                best_c = cs
                best_h = hs.copy()
            if time.time() < t3 - 15:
                h_new = slsqp(hs, nc, 200000, 2000, 1e-13)
                c_new = compute_c5(h_new, nc)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                if c_new < best_c + 0.001 and time.time() < t3 - 5:
                    h_new = slsqp(h_new, nc, 2000000, 3000, 1e-15)
                    c_new2 = compute_c5(h_new, nc)
                    if c_new2 < best_c:
                        best_c = c_new2
                        best_h = h_new

    # Phase 4: Final polishing (5% of budget)
    nf = len(best_h)
    if time.time() < deadline - 10:
        h_new = slsqp(best_h, nf, 10000000, 4000, 1e-18)
        c_new = compute_c5(h_new, nf)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 3:
        best_h, best_c = local_pairs(best_h, nf, deadline, rng, num_iter=150000)

    # Final: one last SLSQP + verify
    if time.time() < deadline - 1:
        h_new = slsqp(best_h, nf, 5000000, 2000, 1e-18)
        c_new = compute_c5(h_new, nf)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, nf)
    best_c = compute_c5(best_h, nf)
    return best_h, best_c, nf
