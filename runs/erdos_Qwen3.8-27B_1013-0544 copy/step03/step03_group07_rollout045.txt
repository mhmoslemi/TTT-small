    target = n / 2.0
    bounds = list(zip(np.zeros(n), np.ones(n)))
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    def grad(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def subgrad_avg(h, max_iters, lr0, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_avg = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr_t = lr0 / (1.0 + 0.005 * t)
        grad = topk_grad(x, top_k, dx)
        x_new = project_feasible(x - lr_t * grad, target)
        x_avg = (t * x_avg + x_new) / (t + 1)
        x = x_new
        if (t + 1) % 500 == 0:
            v = compute_c5(x)
            if v < best_v:
                best_v = v
                best_x = x.copy()
            v_avg = compute_c5(x_avg)
            if v_avg < best_v:
                best_v = v_avg
                best_x = x_avg.copy()
    v = compute_c5(x)
    if v < best_v:
        best_x, best_v = x.copy(), v
    v_avg = compute_c5(x_avg)
    if v_avg < best_v:
        best_x, best_v = x_avg.copy(), v_avg
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)

    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: L-BFGS-B quick passes at current resolution
    for beta in [80, 150, 300]:
        if tl() < 45:
            break
        h_new = lbb_refine(h_best, maxiter=300, beta=beta, penalty=5.0)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 2: Subgradient with averaging (long run)
    if tl() > 120:
        h_sg, c_sg = subgrad_avg(h_best, 12000, 0.04, top_k=4)
        if c_sg < best_c5:
            h_best, best_c5 = h_sg, c_sg

    # Phase 3: SLSQP polish
    for beta in [300, 600, 1200]:
        if tl() < 50:
            break
        h_new = slsqp_refine(h_best, maxiter=600, beta=beta)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 4: L-BFGS-B high penalty
    for beta in [600, 1200]:
        if tl() < 40:
            break
        h_new = lbb_refine(h_best, maxiter=400, beta=beta, penalty=30.0)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 5: Grid upgrade to 600
    if tl() > 250:
        h_interp = interp_to_n(h_best, 600)
        h_new = lbb_refine(h_interp, maxiter=200, beta=150, penalty=10.0)
        c = compute_c5(h_new)
        if tl() > 80:
            h_new = slsqp_refine(h_new, maxiter=400, beta=400)
            c = compute_c5(h_new)
        if tl() > 60:
            h_sg2, c_sg2 = subgrad_avg(h_new, 8000, 0.02, top_k=5)
            if c_sg2 < c:
                h_new, c = h_sg2, c_sg2
            if tl() > 40:
                h_pol = slsqp_refine(h_new, maxiter=400, beta=800)
                cp = compute_c5(h_pol)
                if cp < c:
                    h_new, c = h_pol, cp
        if c < best_c5:
            h_best, best_c5 = h_new, c
            n = len(h_best)

    # Phase 6: Grid upgrade to 800
    if tl() > 300:
        h_interp = interp_to_n(h_best, 800)
        h_new = lbb_refine(h_interp, maxiter=150, beta=100, penalty=10.0)
        c = compute_c5(h_new)
        if tl() > 80:
            h_new = slsqp_refine(h_new, maxiter=300, beta=300)
            c = compute_c5(h_new)
        if tl() > 50:
            h_sg3, c_sg3 = subgrad_avg(h_new, 6000, 0.015, top_k=5)
            if c_sg3 < c:
                h_new, c = h_sg3, c_sg3
            if tl() > 30:
                h_pol = slsqp_refine(h_new, maxiter=300, beta=600)
                cp = compute_c5(h_pol)
                if cp < c:
                    h_new, c = h_pol, cp
        if c < best_c5:
            h_best, best_c5 = h_new, c
            n = len(h_best)

    # Phase 7: Shake restarts at best resolution
    n_curr = len(h_best)
    for trial in range(8):
        if tl() < 50:
            break
        scale = 0.008 * (1.0 + 0.4 * trial)
        pert = np.random.randn(n_curr) * scale
        h_pert = project_feasible(h_best + pert, n_curr / 2.0)
        h_new = lbb_refine(h_pert, maxiter=250, beta=500, penalty=20.0)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c
            if tl() > 40:
                h_pol = slsqp_refine(h_best, maxiter=400, beta=1000)
                cp = compute_c5(h_pol)
                if cp < best_c5:
                    h_best, best_c5 = h_pol, cp

    # Phase 8: Final SLSQP polish with very high beta
    if tl() > 40:
        h_final = slsqp_refine(h_best, maxiter=800, beta=4000)
        c = compute_c5(h_final)
        if c < best_c5:
            h_best, best_c5 = h_final, c

    # Final feasibility check
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)

    return h_best, best_c5, len(h_best)
