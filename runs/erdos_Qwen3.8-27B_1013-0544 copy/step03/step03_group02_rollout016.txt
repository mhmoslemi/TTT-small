            break
        step = step_init / (1.0 + 0.05 * t_val)
        g, _ = exact_subgradient_top(y, 10)
        y = project_h(cur - step * g, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        cur = project_h(y + (t_val - 1.0) / t_new * (y - cur), target)
        f_cur = compute_c5(cur)
        if f_cur < best_f - 1e-15:
            best_f = f_cur
        t_val = t_new
    return cur, compute_c5(cur)

def exact_subgradient_top(h, n_top):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_top]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag >= 0:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
        else:
            la = -lag
            grad[la:] += (1.0 - h[:n - la])
            grad[:n - la] -= h[la:]
    grad /= float(n_top)
    return grad

def coord_descent(h, target, n_sweeps, t0, budget, steps_list):
    n = len(h)
    best_c5 = compute_c5(h)
    for sw in range(n_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in steps_list:
                trial = h.copy()
                trial[j] = trial[j] + d
                if trial[j] < 0.0 or trial[j] > 1.0:
                    continue
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    h = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(320, 0.5)

    n = len(h_init)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing smoothness (full)
    for beta in [60, 120, 300, 800, 1500]:
        if budget_s - (time.time() - t0) < 15:
            break
        h_cur = do_slsqp_full(best_h, n, target, beta, bounds, 4000)
        c = compute_c5(h_cur)
        if c < best_c5:
            best_h, best_c5 = h_cur, c

    # Phase 2: Active-set SLSQP (focus on top lags)
    if budget_s - (time.time() - t0) > 40:
        h_cur = do_slsqp_active(best_h, n, target, bounds, 6, 200, 5000)
        c = compute_c5(h_cur)
        if c < best_c5:
            best_h, best_c5 = h_cur, c

    # Phase 3: Nesterov subgradient
    if budget_s - (time.time() - t0) > 40:
        h_cur, c = nesterov_subgrad(best_h, target, 4000, 0.06, t0, budget_s)
        if c < best_c5:
            best_h, best_c5 = h_cur, c

    # Phase 4: Coordinate descent (medium steps)
    if budget_s - (time.time() - t0) > 30:
        steps_med = [0.04, -0.04, 0.02, -0.02, 0.01, -0.01]
        best_h, best_c5 = coord_descent(best_h, target, 50, t0, budget_s, steps_med)

    # Phase 5: Restart with perturbations
    for scale in [0.012, 0.006, 0.003, 0.002]:
        if budget_s - (time.time() - t0) < 40:
            break
        hp = project_h(best_h + np.random.randn(n) * scale, target)
        hp = do_slsqp_active(hp, n, target, bounds, 6, 300, 3000)
        steps_m = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
        hp, cp = coord_descent(hp, target, 30, t0, budget_s, steps_m)
        if cp < best_c5:
            best_h, best_c5 = hp, cp

    # Phase 6: Ultra-fine coordinate descent
    if budget_s - (time.time() - t0) > 15:
        steps_fine = [0.006, -0.006, 0.003, -0.003, 0.0015, -0.0015]
        best_h, best_c5 = coord_descent(best_h, target, 120, t0, budget_s, steps_fine)

    # Phase 7: Final micro-polish
    if budget_s - (time.time() - t0) > 10:
        steps_micro = [0.003, -0.003, 0.001, -0.001, 0.0005, -0.0005]
        best_h, best_c5 = coord_descent(best_h, target, 150, t0, budget_s, steps_micro)

    # Final validation
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
