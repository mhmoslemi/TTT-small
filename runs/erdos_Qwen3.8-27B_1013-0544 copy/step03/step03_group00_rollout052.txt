    return float(np.max(corr) * dx)

def project_exact(x, target_sum):
    n = len(x)
    lo = -np.max(x) - 0.1
    hi = 1.0 - np.min(x) + 0.1
    for _ in range(80):
        mid = (lo + hi) / 2
        projected = np.clip(x + mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) / 2, 0.0, 1.0)

def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    shifted = beta * (c_sel - m)
    max_shifted = np.max(shifted)
    exps = np.exp(shifted - max_shifted)
    sw = np.sum(exps)
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad

def fast_optimize(h, beta, n_iters, top_k=None, alpha_init=0.005, seed=0):
    """Projected gradient with adaptive step and restarts."""
    n = len(h)
    target = n / 2.0
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_val = compute_c5(h)
    alpha = alpha_init
    stall = 0
    
    for it in range(n_iters):
        val, grad = smoothmax_val_grad(h, beta, top_k)
        # Adaptive: reduce alpha if not making progress
        h_new = h - alpha * grad
        h_new = project_exact(h_new, target)
        c_new = compute_c5(h_new)
        if c_new < best_val - 1e-10:
            best_val = c_new
            best_h = h_new.copy()
            stall = 0
            # Slightly increase alpha
            alpha = min(alpha * 1.01, alpha_init * 2)
        else:
            stall += 1
            if stall % 100 == 0:
                alpha *= 0.7
            if stall > 2000:
                h = best_h + rng.randn(n) * 0.005
                h = project_exact(h, target)
                stall = 0
        h = h_new
    
    return best_h, best_val

def slsqp_solve(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]
    
    def objg(x):
        return smoothmax_val_grad(x, beta, top_k)[1]
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)})
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_exact(res.x, target)
    except Exception:
        return project_exact(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_exact(h_new, n_new / 2.0)

def block_perturb(h, block_len, scale, rng):
    n = len(h)
    target = n / 2.0
    h_new = h.copy()
    for _ in range(block_len):
        start = rng.randint(0, n)
        length = rng.randint(5, 30)
        shift = rng.uniform(-scale, scale)
        end = min(start + length, n)
        h_new[start:end] += shift
    return project_exact(h_new, target)
