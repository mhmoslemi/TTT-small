    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def subgrad_vec(h, weights, lags):
    n = len(h)
    grad = np.zeros(n)
    for w, lag in zip(weights, lags):
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:-lag]
        else:
            ll = -lag
            grad[:n - ll] += w * (1.0 - h[ll:])
            grad[ll:] -= w * h[:-ll]
    return grad

def compute_subgrad_data(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    top_k = min(12, n)
    top_idx = np.argsort(corr)[::-1][:top_k]
    w = np.exp(-40.0 * (max_val - corr[top_idx]) / max(1e-10, max_val))
    w /= w.sum()
    lags = (top_idx - (n - 1)).astype(int)
    return w, lags, max_val

def subgradient_phase(h, target, t0, budget_s, max_time, mom_factor=0.8, step0=0.06, decay=0.004):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    momentum = np.zeros(n)
    best_local = cur_c5
    it = 0
    while time.time() - t0 < budget_s - max_time:
        it += 1
        w, lags, _ = compute_subgrad_data(cur)
        grad = subgrad_vec(cur, w, lags)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        momentum = mom_factor * momentum + (1.0 - mom_factor) * grad / gn
        mn = np.linalg.norm(momentum)
        if mn < 1e-13:
            break
        step = step0 / (1.0 + decay * it)
        direction = -momentum / mn
        improved = False
        for trial in range(6):
            h_t = project_h(cur + step * direction, target)
            c5_t = compute_c5(h_t)
            if c5_t < cur_c5 - 1e-15:
                cur = h_t
                cur_c5 = c5_t
                improved = True
                if cur_c5 < best_local:
                    best_local = cur_c5
                break
            step *= 0.5
        if not improved:
            momentum *= 0.5
    return cur, best_local

def binary_local_search(h, target, t0, budget_s, max_time, n_rounds=300):
    n = len(h)
    h_b = (h >= 0.5).astype(float).copy()
    deficit = target - h_b.sum()
    if deficit > 0:
        zeros = np.where(h_b == 0)[0]
        order = zeros[np.argsort(h[zeros])[::-1]]
        h_b[order[:int(round(deficit))]] = 1.0
    elif deficit < 0:
        ones = np.where(h_b == 1)[0]
        order = ones[np.argsort(h[ones])]
        h_b[order[:int(round(-deficit))]] = 0.0
    c5_b = compute_c5(h_b)
    ones_idx = np.where(h_b == 1)[0]
    zeros_idx = np.where(h_b == 0)[0]
    for _ in range(n_rounds):
        if time.time() - t0 > budget_s - max_time:
            break
        improved = False
        n_try = min(200, max(10, len(ones_idx) * len(zeros_idx) // 5))
        for _ in range(n_try):
            i = ones_idx[np.random.randint(len(ones_idx))]
            j = zeros_idx[np.random.randint(len(zeros_idx))]
            h_b[i], h_b[j] = 0.0, 1.0
            c5_t = compute_c5(h_b)
            if c5_t < c5_b - 1e-12:
                c5_b = c5_t
                improved = True
                ones_idx = np.where(h_b == 1)[0]
                zeros_idx = np.where(h_b == 0)[0]
            else:
                h_b[i], h_b[j] = 1.0, 0.0
        if not improved:
            break
    return h_b, c5_b

def slsqp_refine(h, target, t0, budget_s, max_time):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(best)
    bounds_s = Bounds(np.zeros(n), np.ones(n))
    for beta in [100, 300, 800, 2000, 5000, 10000]:
        if time.time() - t0 > budget_s - max_time:
            break
        _b = beta
        _dx = dx
        _tgt = target
        n_local = n
        def _obj(x):
            c = np.correlate(x, 1.0 - x, mode="full")
            return smooth_max(c, _b) * _dx
        def _con(x):
            return np.sum(x) - _tgt
        def _con_jac(x):
            return np.ones(n_local)
        try:
            res = minimize(_obj, best, method='SLSQP', bounds=bounds_s,
                           constraints=[{'type': 'eq', 'fun': _con, 'jac': _con_jac}],
                           options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(280, 0.5)
    n_base = len(h_parent)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Subgradient at parent resolution
    target = n_base / 2.0
    h1, c5_1 = subgradient_phase(best_h, target, t0, budget_s, max_time=400,
                                  mom_factor=0.82, step0=0.07, decay=0.003)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: SLSQP at parent resolution
    if time.time() - t0 < budget_s - 200:
        h2, c5_2 = slsqp_refine(best_h, target, t0, budget_s, max_time=200)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 3: Subgradient again after SLSQP
    if time.time() - t0 < budget_s - 150:
        h3, c5_3 = subgradient_phase(best_h, target, t0, budget_s, max_time=150,
                                      mom_factor=0.85, step0=0.04, decay=0.005)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    # Phase 4: Higher resolution
    for mult in [1.5, 2.0, 2.5]:
        if time.time() - t0 > budget_s - 100:
            break
        n_r = int(n_base * mult)
        target_r = n_r / 2.0
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_r)
        h_r = project_h(np.interp(x_new, x_old, best_h), target_r)
        c5_r = compute_c5(h_r)
        h_cur, c5_cur = h_r, c5_r
        if c5_cur < best_c5:
            best_h, best_c5 = h_cur.copy(), c5_cur
        # Subgradient at new resolution
        h4, c5_4 = subgradient_phase(h_cur, target_r, t0, budget_s, max_time=80,
                                      mom_factor=0.82, step0=0.05, decay=0.004)
        if c5_4 < best_c5:
            best_h, best_c5 = h4.copy(), c5_4
        # SLSQP at new resolution
        if time.time() - t0 < budget_s - 80:
            h5, c5_5 = slsqp_refine(h4, target_r, t0, budget_s, max_time=80)
            if c5_5 < best_c5:
                best_h, best_c5 = h5.copy(), c5_5

    # Phase 5: Binary local search at best resolution
    if time.time() - t0 < budget_s - 30:
        n_b = len(best_h)
        h_b, c5_b = binary_local_search(best_h, n_b / 2.0, t0, budget_s, max_time=50, n_rounds=500)
        if c5_b < best_c5:
            best_h, best_c5 = h_b, c5_b

    # Phase 6: Final subgradient polish
    if time.time() - t0 < budget_s - 10:
        n_f = len(best_h)
        h6, c5_6 = subgradient_phase(best_h, n_f / 2.0, t0, budget_s, max_time=10,
                                      mom_factor=0.88, step0=0.02, decay=0.01)
        if c5_6 < best_c5:
            best_h, best_c5 = h6, c5_6

    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    final_target = len(best_h) / 2.0
    best_h = project_h(best_h, final_target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, int(len(best_h))
