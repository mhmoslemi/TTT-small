import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad


def slsqp_solve(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0

    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]

    def objg(x):
        return smoothmax_val_grad(x, beta, top_k)[1]

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)})
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def annealed_slsqp(h, beta_list, maxiter_list, top_k_list):
    n = len(h)
    target = n / 2.0
    for i in range(len(beta_list)):
        h = slsqp_solve(h, beta_list[i], maxiter_list[i], top_k_list[i])
        h = project_feasible(h, target)
    return h


def structured_perturb(h, rng, scale, n):
    """Perturb with structured (low-freq) + white noise."""
    target = n / 2.0
    # Low-frequency perturbation
    freqs = rng.randint(1, max(2, n // 10), size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    x = np.arange(n)
    structured = np.zeros(n)
    for f, p in zip(freqs, phases):
        structured += np.sin(2 * np.pi * f * x / n + p)
    structured /= (np.std(structured) + 1e-10)
    white = rng.randn(n)
    h_pert = h + scale * (0.7 * structured + 0.3 * white)
    return project_feasible(h_pert, target)

        h_interp = interp_to_n(h_best, n_try)
