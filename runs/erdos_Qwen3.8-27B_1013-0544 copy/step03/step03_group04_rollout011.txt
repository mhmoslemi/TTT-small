    dx = 2.0 / n
    h = h.copy()
    h = project_h(h, target)
    corr_vec = np.correlate(h, 1.0 - h, mode="full") * dx
    best_c5 = np.max(corr_vec)
    any_improved = False
    for sweep in range(max_sweeps):
        improved = False
        order = rng.permutation(n)
        for i in order:
            v_old = h[i]
            best_val = v_old
            best_c5_l = best_c5
            best_corr_l = corr_vec.copy()
            cands = [0.0, 1.0]
            if v_old > 0.05:
                cands.append(v_old * 0.5)
                cands.append(max(0.0, v_old - 0.15))
            if v_old < 0.95:
                cands.append(v_old + (1.0 - v_old) * 0.5)
                cands.append(min(1.0, v_old + 0.15))
            for vc in cands:
                if abs(vc - v_old) < 1e-10:
                    continue
                h[i] = np.clip(vc, 0, 1)
                h2 = project_h(h, target)
                cv = np.correlate(h2, 1.0 - h2, mode="full") * dx
                c5l = np.max(cv)
                if c5l < best_c5_l - 1e-14:
                    best_c5_l = c5l
                    best_val = h2.copy()
                    best_corr_l = cv.copy()
            if best_val != v_old:
                h = best_val
                corr_vec = best_corr_l
                if best_c5_l < best_c5 - 1e-14:
                    best_c5 = best_c5_l
                    improved = True
            else:
                h[i] = v_old
                h = project_h(h, target)
        if not improved:
            break
    return h, best_c5

def subgrad_step(h, top_k, step, target):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for li in top_idx:
        lag = int(li) - (n - 1)
        for j in range(n):
            i1 = j - lag
            i2 = j + lag
            if 0 <= i1 < n:
                grad[j] += (1.0 - h[i1])
            if 0 <= i2 < n:
                grad[j] -= h[i2]
    gn = np.linalg.norm(grad)
    if gn < 1e-12:
        return None
    return -grad / gn * step

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(280, 0.5)

    n_base = len(h_parent)
    target = n_base / 2.0
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Coordinate descent from parent
    for sweep_round in range(40):
        if remaining() < 60:
            break
        h_cd, c5_cd = coord_descent(best_h, target, rng, max_sweeps=1, n_cands=7)
        if c5_cd < best_c5 - 1e-15:
            best_h = h_cd
            best_c5 = c5_cd
        else:
            break

    # Phase 2: SLSQP with various objectives at base resolution
    bounds_s = Bounds(np.zeros(n_base), np.ones(n_base))
    constr = [{'type': 'eq', 'fun': lambda x: _eq_con(x, target), 'jac': lambda x: _eq_jac(x)}]

    for beta in [50, 100, 200, 500]:
        if remaining() < 30:
            break
        obj = _make_obj(target, 2.0/n_base, beta, 0)
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds_s,
                           constraints=constr, options={'maxiter': 2000, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
        except Exception:
            pass

    # Phase 3: Top-k objective SLSQP
    for k_top in [3, 5, 10]:
        if remaining() < 30:
            break
        for beta in [200, 500]:
            obj = _make_obj(target, 2.0/n_base, beta, k_top)
            try:
                res = minimize(obj, best_h, method='SLSQP', bounds=bounds_s,
                               constraints=constr, options={'maxiter': 2000, 'ftol': 1e-15})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new.copy(), c5_new
            except Exception:
                pass

    # Phase 4: Random restarts with perturbation + coordinate descent
    n_restarts = 20
    for ri in range(n_restarts):
        if remaining() < 60:
            break
        # Create perturbed version
        noise_scale = 0.05 + 0.03 * ri / max(1, n_restarts - 1)
        h_pert = best_h + noise_scale * (rng.random(n_base) - 0.5)
        h_pert = project_h(h_pert, target)

        # Coordinate descent from perturbed
        h_pd, c5_pd = coord_descent(h_pert, target, rng, max_sweeps=5, n_cands=5)
        if c5_pd < best_c5:
            best_h, best_c5 = h_pd.copy(), c5_pd

        # Quick SLSQP from perturbed
        if remaining() > 40:
            obj = _make_obj(target, 2.0/n_base, 300, 0)
            try:
                res = minimize(obj, h_pd, method='SLSQP', bounds=bounds_s,
                               constraints=constr, options={'maxiter': 1000, 'ftol': 1e-14})
                h_sl = project_h(res.x, target)
                c5_sl = compute_c5(h_sl)
                if c5_sl < best_c5:
                    best_h, best_c5 = h_sl.copy(), c5_sl
            except Exception:
                pass

    # Phase 5: Higher resolution upsample + optimize
    for n_up in [n_base * 2]:
        if remaining() < 80:
            break
        x_old = np.linspace(0, 2, n_base, endpoint=False)
        x_new = np.linspace(0, 2, n_up, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        tgt_up = n_up / 2.0
        h_up = project_h(h_up, tgt_up)

        bounds_up = Bounds(np.zeros(n_up), np.ones(n_up))
        constr_up = [{'type': 'eq', 'fun': lambda x: _eq_con(x, tgt_up), 'jac': lambda x: _eq_jac(x)}]

        for beta in [200, 500]:
            if remaining() < 20:
                break
            obj_up = _make_obj(tgt_up, 2.0/n_up, beta, 0)
            try:
                res = minimize(obj_up, h_up, method='SLSQP', bounds=bounds_up,
                               constraints=constr_up, options={'maxiter': 1500, 'ftol': 1e-14})
                h_up_new = project_h(res.x, tgt_up)
                c5_up = compute_c5(h_up_new)
                if c5_up < best_c5:
                    # Downsample back or keep higher res
                    best_h = h_up_new.copy()
                    best_c5 = c5_up
                    n_base = n_up
                    target = tgt_up
                    bounds_s = Bounds(np.zeros(n_base), np.ones(n_base))
                    constr = [{'type': 'eq', 'fun': lambda x: _eq_con(x, target), 'jac': lambda x: _eq_jac(x)}]
            except Exception:
                pass

        # Coordinate descent at higher res
        if remaining() > 30:
            h_cd2, c5_cd2 = coord_descent(best_h, target, rng, max_sweeps=3, n_cands=5)
            if c5_cd2 < best_c5:
                best_h, best_c5 = h_cd2.copy(), c5_cd2

    # Phase 6: Subgradient polishing
    if remaining() > 20:
        n_pol = len(best_h)
        tgt_pol = n_pol / 2.0
        cur = best_h.copy()
        cur_c5 = best_c5
        momentum = np.zeros(n_pol)
        for it in range(20000):
            if remaining() < 5:
                break
            ss = 0.04 / (1.0 + 0.003 * it)
            g = subgrad_step(cur, top_k=3, step=1.0, target=tgt_pol)
            if g is None:
                g = subgrad_step(cur, top_k=5, step=1.0, target=tgt_pol)
            if g is None:
                break
            momentum = 0.75 * momentum + 0.25 * g
            h_trial = project_h(cur + ss * momentum, tgt_pol)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                cur = h_trial
                cur_c5 = c5_trial
                if cur_c5 < best_c5:
                    best_h, best_c5 = cur.copy(), cur_c5
            else:
                h_trial2 = project_h(cur + ss * 0.1 * momentum, tgt_pol)
                c5_trial2 = compute_c5(h_trial2)
                if c5_trial2 < cur_c5 - 1e-15:
                    cur = h_trial2
                    cur_c5 = c5_trial2
                    if cur_c5 < best_c5:
                        best_h, best_c5 = cur.copy(), cur_c5
                else:
                    momentum *= 0.5

    # Phase 7: Final coordinate descent
    if remaining() > 10:
        h_fin, c5_fin = coord_descent(best_h, target, rng, max_sweeps=10, n_cands=7)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin.copy(), c5_fin

    n_final = len(best_h)
    return best_h, best_c5, int(n_final)
