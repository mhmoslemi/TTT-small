        grad += grad_for_lag(h, lag, n, dx)
    return grad, cmax, corr_val


def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def coord_descent(h, target, dx, max_sweeps, time_fn, min_time):
    n = len(h)
    cur_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if time_fn() < min_time:
            break
        improved = False
        for j in range(n):
            if time_fn() < min_time:
                break
            for val in (0.0, 1.0):
                if abs(h[j] - val) < 1e-10:
                    continue
                h[j] = val
                h_p = project_h(h, target)
                c5_new = compute_c5(h_p)
                if c5_new < cur_c5 - 1e-14:
                    h = h_p
                    cur_c5 = c5_new
                    improved = True
                else:
                    h = h_p
                    h[j] = project_h(h, target)[j]
        if not improved:
            break
    return h, cur_c5


def smooth_max_obj(h, dx, beta):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = c.max()
    shifted = np.clip(beta * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted)) / len(c)) / beta


def constraint_sum(x, target):
    return float(x.sum() - target)


def constraint_sum_jac(x, target):
    return np.ones(x.shape)


def subgrad_descent(h, n, target, dx, max_iters, time_fn, min_time, best_state):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    for phase in range(4):
        threshold = 0.008 * (0.5 ** phase)
        step_base = 0.05 * (0.75 ** phase)
        for it in range(max_iters // 4 + 1):
            if time_fn() < min_time:
                return cur, cur_c5, n
            grad, cmax, _ = subgradient(cur, n, dx, threshold)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = step_base / (1.0 + 0.001 * it)
            improved = False
            for _ in range(20):
                h_trial = project_h(cur - step * grad, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 - 1e-15:
                    cur = h_trial
                    cur_c5 = c5_trial
                    improved = True
                    if cur_c5 < best_state[0]:
                        best_state[0] = cur_c5
                        best_state[1] = cur.copy()
                        best_state[2] = n
                    break
                step *= 0.5
            if not improved:
                T = 0.003 * (0.5 ** phase)
                cur = project_h(cur + np.random.normal(0, T, n), target)
                cur_c5 = compute_c5(cur)
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
    return cur, cur_c5, n


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 250
        h_parent = np.full(n_parent, 0.5)

    best_state = [compute_c5(h_parent), h_parent.copy(), n_parent]

    # Phase 1: Subgradient at parent resolution
    n = n_parent
    target = n / 2.0
    dx = 2.0 / n
    h = h_parent.copy()
    h, _, _ = subgrad_descent(h, n, target, dx, 8000, time_left, 80, best_state)

    # Phase 2: Coordinate descent at parent resolution
    if time_left() > 30:
        h, c5_cd = coord_descent(best_state[1], best_state[2] / 2.0, 2.0 / best_state[2],
                                 max_sweeps=5, time_fn=time_left, min_time=15)
        if c5_cd < best_state[0]:
            best_state[0] = c5_cd
            best_state[1] = h
            best_state[2] = len(h)

    # Phase 3: Try higher resolutions
    res_list = [300, 350, 400]
    for n_new in res_list:
        if time_left() < 80:
            break
        target_new = n_new / 2.0
        dx_new = 2.0 / n_new
        h_new = project_h(resample(best_state[1], n_new), target_new)
        h_new, _, _ = subgrad_descent(h_new, n_new, target_new, dx_new,
                                       4000, time_left, 30, best_state)
        # Quick coord descent at new resolution
        if time_left() > 25:
            h_cd, c5_cd = coord_descent(best_state[1], best_state[2] / 2.0,
                                        2.0 / best_state[2],
                                        max_sweeps=3, time_fn=time_left, min_time=10)
            if c5_cd < best_state[0]:
                best_state[0] = c5_cd
                best_state[1] = h_cd
                best_state[2] = len(h_cd)

    # Phase 4: SLSQP polish on best so far
    if time_left() > 20:
        n_f = best_state[2]
        target_f = n_f / 2.0
        dx_f = 2.0 / n_f
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        con_fun = partial(constraint_sum, target=target_f)
        con_jac = partial(constraint_sum_jac, target=target_f)
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        for beta in (300, 800, 2000):
            if time_left() < 8:
                break
            obj = partial(smooth_max_obj, dx=dx_f, beta=beta)
            try:
                res = minimize(obj, best_state[1], method='SLSQP', bounds=bounds_f,
                               constraints=cons, options={'maxiter': 800, 'ftol': 1e-15})
                h_new = project_h(res.x, target_f)
                c5_new = compute_c5(h_new)
                if c5_new < best_state[0]:
                    best_state[0] = c5_new
                    best_state[1] = h_new.copy()
            except Exception:
                pass

    # Phase 5: Final coordinate descent
    if time_left() > 10:
        nf = best_state[2]
        h2, c52 = coord_descent(best_state[1], nf / 2.0, 2.0 / nf,
                                max_sweeps=5, time_fn=time_left, min_time=3)
        if c52 < best_state[0]:
            best_state[0] = c52
            best_state[1] = h2

    # Phase 6: Tiny final subgradient steps
    if time_left() > 5:
        nf = best_state[2]
        tf = nf / 2.0
        dxf = 2.0 / nf
        h3 = best_state[1].copy()
        c5_3 = compute_c5(h3)
        for it in range(5000):
            if time_left() < 2:
                break
            grad, _, _ = subgradient(h3, nf, dxf, 0.001)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            for step in (0.005, 0.002, 0.001, 0.0005):
                h_t = project_h(h3 - step * grad, tf)
                c5_t = compute_c5(h_t)
                if c5_t < c5_3 - 1e-15:
                    h3 = h_t
                    c5_3 = c5_t
                    break
        if c5_3 < best_state[0]:
            best_state[0] = c5_3
            best_state[1] = h3

    return best_state[1], best_state[0], len(best_state[1])
