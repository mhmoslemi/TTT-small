def min_norm_subgrad(h, active_lags):
    """Find the minimum norm convex combination of subgradients."""
    from scipy.optimize import linprog
    n = len(h)
    m = len(active_lags)
    if m == 1:
        return compute_subgrad(h, active_lags, [1.0])
    
    # Compute all subgradients
    grads = np.zeros((m, n))
    for i, lag in enumerate(active_lags):
        grads[i] = compute_subgrad(h, [lag], [1.0])
    
    # Minimize ||sum w_i * g_i||_2^2 subject to sum w_i = 1, w_i >= 0
    # This is a QP: min w^T (G^T G) w s.t. sum w = 1, w >= 0
    # Approximate with: min w^T (G^T G) w (convex QP)
    # Use scipy minimize
    GtG = grads.T @ grads  # m x m
    from scipy.optimize import minimize as sp_min
    
    def qobj(w):

    bounds_q = [(0, 1)] * m

    res = sp_min(qobj, w0, method='SLSQP', bounds=bounds_q, constraints=constraints,

    w_opt /= w_opt.sum()
    
    direction = grads.T @ w_opt
