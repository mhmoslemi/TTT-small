def corr_gradient(h, active_lags):
    """Compute gradient of sum of corr[lag] for given lags w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    for lag in active_lags:
        if lag == 0:
            g += (1 - 2*h)
        elif lag > 0:
            # corr = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            # d/dh[k]: (1-h[k-lag]) for k in [lag, n-1] minus h[k+lag] for k in [0, n-1-lag]
            g[lag:] += (1 - h[:-lag])
            g[:n-lag] -= h[lag:]
        else:

            # d/dh[k]: (1-h[k+m]) for k in [0, n-1-m] minus h[k-m] for k in [m, n-1]
            g[:n-m] += (1 - h[m:])
            g[m:] -= h[:n-m]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def corr_grad(h, lags):
    n = len(h)
    g = np.zeros(n)
    for lag in lags:
        if lag == 0:
            g += (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += (1.0 - h[:-lag])
            g[:n - lag] -= h[lag:]
        else:
            m = -lag
            g[:n - m] += (1.0 - h[m:])
            g[m:] -= h[:n - m]
    return g

def smooth_max_grad(corr, beta):
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    w = e / e.sum()
    return w  # normalized weights

    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    bc = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(bc)
    se = e.sum()
    val = (m + np.log(se / len(corr)) / beta) * dx
    w = e / se
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        if w[j] < 1e-11:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += w[j] * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w[j] * (1.0 - h[:-lag])
            g[:n - lag] -= w[j] * h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += w[j] * (1.0 - h[m2:])
            g[m2:] -= w[j] * h[:n - m2]
    return val, g * dx

def slsqp_optimize(h_init, n, beta, maxiter=3000, ftol=1e-16):
    target = n / 2.0
    bounds = [(0, 1)] * n

    def obj(x):
        return obj_grad(np.clip(x, 0, 1), beta)[0]

    def grad(x):
        return obj_grad(np.clip(x, 0, 1), beta)[1]

    def eq(x):
        return np.sum(x) - target

    def eqj(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eqj}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_sum(res.x, target)

def subgrad_polish(h, target, n_active, max_iter, t0, budget):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    no_improve = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        top_idx = np.argsort(corr)[::-1][:n_active]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                g += (1.0 - 2.0 * best)
            elif lag > 0:
                g[lag:] += (1.0 - best[:-lag])
                g[:n - lag] -= best[lag:]
            else:
                m2 = -lag
                g[:n - m2] += (1.0 - best[m2:])
                g[m2:] -= best[:n - m2]
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0004]:
            ht = project_sum(best - step * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                improved = True
                break
        if not improved:
            no_improve += 1
            if no_improve > 300 and it % 150 == 149:
                pert = np.random.randn(n) * 0.008
                ht = project_sum(best + pert, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    no_improve = 0
            if no_improve > 3000:
                break
        else:
            no_improve = 0
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with exact gradient, increasing beta
    n0 = len(best_h)
    for beta in [30, 100, 400, 1500, 5000, 20000]:
        if tl() < 80:
            break
        h_new = slsqp_optimize(best_h, n0, beta, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        # Quick subgradient step between SLSQP calls
        best_h, best_c5 = subgrad_polish(best_h, n0 / 2.0, 5, 500, t0, budget_s)

    # Phase 2: Resolution increase
    for n_try in [280, 360, 480]:
        if tl() < 180:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_sum(np.interp(x_new, x_old, best_h), n_try / 2.0)
        for beta in [50, 300, 1500, 8000]:
            if tl() < 80:
                break
            h_new = slsqp_optimize(h_interp, n_try, beta, maxiter=2500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_interp = h_new
        best_h, best_c5 = subgrad_polish(best_h, len(best_h) / 2.0, 7, 800, t0, budget_s)

    # Phase 3: Multi-start exploration at best resolution
    n_best = len(best_h)
    for restart in range(4):
        if tl() < 120:
            break
        if restart == 0:
            h_try = best_h.copy()
        elif restart == 1:
            h_try = project_sum(best_h + np.random.randn(n_best) * 0.05, n_best / 2.0)
        elif restart == 2:
            # Block pattern perturbation
            block = n_best // 8
            h_try = best_h.copy()
            h_try[:block] = 1.0 - h_try[:block]
            h_try = project_sum(h_try, n_best / 2.0)
        else:
            # Anti-symmetric around center
            h_try = np.zeros(n_best)
            h_try[:n_best // 2] = best_h[:n_best // 2]
            h_try[n_best // 2:] = 1.0 - best_h[:n_best // 2][::-1]
            h_try = project_sum(h_try, n_best / 2.0)
        for beta in [200, 2000]:
            if tl() < 60:
                break
            h_new = slsqp_optimize(h_try, n_best, beta, maxiter=2000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_try = h_new

    # Phase 4: Final intensive polishing
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h, best_c5 = subgrad_polish(best_h, target_final, 8, 10000, t0, budget_s)

    if tl() > 15:
        h_new = slsqp_optimize(best_h, n_final, 50000, maxiter=5000, ftol=1e-18)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Final micro-polish
    for it in range(5000):
        if tl() < 2:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:4]
        g = np.zeros(n_final)
        for idx in top_idx:
            lag = int(idx) - (n_final - 1)
            if lag == 0:
                g += (1.0 - 2.0 * best_h)
            elif lag > 0:
                g[lag:] += (1.0 - best_h[:-lag])
                g[:n_final - lag] -= best_h[lag:]
            else:
                m2 = -lag
                g[:n_final - m2] += (1.0 - best_h[m2:])
                g[m2:] -= best_h[:n_final - m2]
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        for step in [0.01, 0.005, 0.002, 0.001, 0.0003, 0.0001]:
            ht = project_sum(best_h - step * g, target_final)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best_h = ht
                best_c5 = c5t
                break

    return best_h, best_c5, len(best_h)
