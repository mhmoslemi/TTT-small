        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        j_lo1 = max(0, s)
        j_hi1 = min(n, n + s)
        if j_hi1 > j_lo1:
            grad[j_lo1:j_hi1] += wk * (1.0 - h[j_lo1 - s:j_hi1 - s])
        j_lo2 = max(0, -s)
        j_hi2 = min(n, n - s)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] += wk * (-h[j_lo2 + s:j_hi2 + s])
    grad *= dx
    return val, grad

def adam_optimize(h, beta, n_steps, lr, top_k=None, t0=None, budget_s=0, clip_norm=1.0):
    n = len(h)
    target = n / 2.0
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    h_cur = h.astype(float).copy()
    last_val = 0.0
    
    for step in range(n_steps):
        if t0 is not None and (budget_s - (time.time() - t0)) < 3:
            break
        val, grad = smoothmax_val_grad(h_cur, beta, top_k)
        last_val = val
        # Clip gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > clip_norm:
            grad = grad * clip_norm / gnorm
        t_idx = step + 1
        m_arr = 0.9 * m_arr + 0.1 * grad
        v_arr = 0.999 * v_arr + 0.001 * grad**2
        m_hat = m_arr / (1 - 0.9**t_idx)
        v_hat = v_arr / (1 - 0.999**t_idx)
        h_cur = h_cur - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_cur = project_feasible(h_cur, target)
    
    return h_cur, last_val

                h_best = h_opt.copy()
            else:
                h_best = h_opt.copy()  # Accept worse for exploration

    return h_best, best_c5
