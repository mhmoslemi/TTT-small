    target = n // 2
    b = (h > 0.5).astype(float)
    # Adjust to have exactly target ones
    n_ones = int(b.sum())
    if n_ones > target:

        # Turn off the smallest h values among those set to 1
        ones_idx = np.where(b == 1)[0]
        sorted_ones = ones_idx[np.argsort(h[ones_idx])]
        for i in range(n_ones - target):

        zeros_idx = np.where(b == 0)[0]
        sorted_zeros = zeros_idx[np.argsort(-h[zeros_idx])]
        for i in range(target - n_ones):

    zeros = np.where(x == 0)[0].tolist()

        # For efficiency, try random pairs first

        np.random.shuffle(zeros)

        for a in ones:
            if found:
                break
            for b_idx in zeros:

                x[b_idx] = 1

                    found = True

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    target_sum = target
    
    def obj_func(x):
        corr = corr_array(x)
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(beta * shifted)
        sw = np.sum(exps)
        return (m + np.log(sw) / beta) * dx
    
    def obj_grad(x):
        corr = corr_array(x)
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(beta * shifted)
        sw = np.sum(exps)
        w = exps / sw
        grad = np.zeros(n)
        lags = np.arange(len(corr)) - (n - 1)
        for k_idx, s in enumerate(lags):
            wk = w[k_idx]
            if wk < 1e-10:
                continue
            if s >= 0:
                grad[s:] += wk * (1.0 - x[:n-s])
                grad[:n-s] += wk * (-x[s:])
            elif s < 0:
                t = -s
                grad[:n-t] += wk * (1.0 - x[t:])
                grad[t:] += wk * (-x[:n-t])
            else:
                grad[:] += wk * (1.0 - 2.0 * x)
        grad *= dx
        return grad
    
    def con_func(x):
        return x.sum() - target_sum
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
    
    try:
        res = minimize(obj_func, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h.copy(), target_sum)

def project_feasible(h, target_sum):

    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)

        if abs(diff2) > 1e-10:

    """Adam optimizer for subgradient method."""
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(max_iters):
        grad = compute_grad_topk(x, top_k)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** (t + 1))

        x = x_new

Let me focus on Frank-Wolfe, which is potentially much more efficient:

The Frank-Wolfe algorithm for our problem:
- Polytope: P = {h : 0≤h_i≤1, ∑h_i = n/2}
- Objective: f(h) = max_k corr_k(h) * dx (convex)
- At each step:
  1. Compute subgradient g at current h
  2. Find s = argmin_{z∈P} ⟨g, z⟩ (this is: set z[i]=1 for n/2 smallest g[i])
  3. Update: h ← h + γ(s - h) where γ is the step size

The step size can be:
- γ_t = 2/(t+2) (standard, gives O(1/t) rate)
- Exact line search: minimize f(h + γ(s-h)) for γ∈[0,1]
- Strong step: γ_t = argmin (better rate, O(1/t²))

For the away-step variant:
- Also consider moving away from a vertex
- The "away" vertex is the one that maximizes ⟨g, z⟩ among vertices in the support

This could be significantly faster than subgradient methods because:
- No projection needed (the update stays in the polytope automatically)
- Large effective steps possible
- Better convergence rate with strong step

Actually, the away-step FW for the base polytope {h: 0≤h≤1, sum(h)=b} is well-studied. The key insight is:

The polytope P = {h: 0≤h≤1, sum(h)=b} has vertices that are 0-1 vectors with exactly b ones. Frank's vertex s is one such vertex. The away vertex is also a 0-1 vertex, specifically one that has a 1 where h has a large value (and we want to move away from it).

The combined FW-A step is:
h_new = (1-γ(1+ρ)) * h + γ * s - γρ * a

where a is the away vertex (a 0-1 vector with b ones that agrees with h on the "bad" coordinates).

Finding the optimal away vertex is NP-hard in general, but for our problem, a good heuristic is:
- a[i] = 1 for the b indices where h[i] * (1 - grad[i]/max_grad) is largest
- Or simply: a is the 0-1 vector with b ones that maximizes ⟨h, a⟩ (i.e., a agrees with h on the b largest h[i])

For the away-step FW (Jaggi, 2013; Lan & Montanari, 2015):
- The current x can be written as a convex combination of at most t+1 vertices (where t is the iteration number)
- The away step picks one of these vertices to move away from
- For our purposes, a simple approximation: the away vertex is the vertex that maximizes ⟨∇f(x), x⟩ among the vertices in the "support" of x

A practical approach:
1. Compute the FW vertex s (minimizes ⟨g, z⟩)
2. Compute the "worst" direction: the unit vector e_j that maximizes g[j] * h[j] (reward for decreasing h[j])
3. Take a combined step: h_new = h + γ(s - h) - ρ * h[j] * e_j (approximately)

Let me also add:
- **Greedy coordinate descent**: For each coordinate, find the optimal value given all others (1D optimization)
- This is exact for a single coordinate and can be done very fast

For coordinate j, the objective as a function of h[j] (with all other h[i] fixed) is:
f(h[j]) = max_k [corr_k - contribution_from_other_coords + contribution_from_j_at_lag_k] * dx

This is the max of linear functions of h[j], so it's a convex piecewise linear function. We can find its minimum by checking the breakpoints.

Actually, computing this exactly is expensive. A simpler approach: for each coordinate j, try h[j]=0, h[j]=current, h[j]=1, and a few values in between, and pick the best. This is a "randomized coordinate descent" approach.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_array(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-6) & (h < 1.0 - 1e-6)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_topk(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = corr_array(h)
    top_idx = np.argsort(corr)[-top_k:]
    lags = top_idx - (n - 1)
    grad = np.zeros(n)
    for s in lags:
        if s >= 0:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] += (-h[s:])
        elif s < 0:
            t = -s
            grad[:n-t] += (1.0 - h[t:])
            grad[t:] += (-h[:n-t])
        else:
            grad[:] += (1.0 - 2.0 * h)
    grad /= top_k
    grad *= dx
    return grad

def nesterov_subgrad(h, max_iters, lr0, decay, top_k):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    y = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        if t > 0:
            mu = (t - 1.0) / (t + 2.0)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        grad = compute_grad_topk(y, top_k)
        x_new = project_feasible(y - lr_t * grad, target)
        val = compute_c5(x_new)
        if val < best_v:
            best_v = val
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v
