    """One majorization-minimization step"""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Softmax weights
    exp_c = np.exp(c / tau)
    lam = exp_c / exp_c.sum()
    
    # Objective: maximize Σ_k lam[k] * c_k(h) (equivalent to minimizing -Σ lam[k] c_k(h])
    # c_k(h) = Σ_j h[j](1-h[j+k])
    # So Σ_k lam[k] c_k(h) = Σ_k lam[k] (Σ_j h[j] - Σ_j h[j]h[j+k])
    # = Σ_j h[j] * (Σ_k lam[k]) - Σ_k lam[k] Σ_j h[j]h[j+k]
    # = Σ_j h[j] - Σ_k lam[k] Σ_j h[j]h[j+k]   (since Σ lam = 1)
    
    # This is: a^T h - h^T Q h
    # where a[j] = 1 for all j (from the linear term)
    # and Q[i,j] = Σ_k lam[k] * indicator(|i-j| corresponds to shift k)

def subgradient_step(h, n, lr, top_k=5):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    top_idx = np.argsort(c)[-top_k:]
    
    g = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        # Use the correlation structure
        # c_k = sum_{j valid} h[j] * (1-h[j+k])
        # d c_k / d h[i] = (1-h[i+k]) if i in first factor
        #                - h[i-k] if i in second factor
        if k >= 0:
            # First factor: h[j] for j in [0, n-1-k]
            # g += (1 - h[k:]) for positions 0 to n-1-k
            g[:n-k] += (1.0 - h[k:])
            # Second factor: h[j+k] for j in [0, n-1-k], i.e., positions k to n-1
            # g -= h[:n-k] for positions k to n-1
            g[k:] -= h[:n-k]
        else:
            k_abs = -k
            # First factor: h[j] for j in [k_abs, n-1]
            g[k_abs:] += (1.0 - h[:n-k_abs])
            # Second factor: h[j+k] = h[j-k_abs] for j in [k_abs, n-1]
            # positions 0 to n-1-k_abs
            g[:n-k_abs] -= h[k_abs:]
    
    g /= top_k
    h_new = project_feasible(h - lr * g, n)
    return h_new, compute_c5(h_new, n)

                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= s * d
                            h[j] += s * d
        # Also try some random pairs
        rng = np.random.RandomState(int(time.time() * 100) % 99999)
        for _ in range(300):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for s in (1.0, -1.0):
                hi = h[i] + s * d
                hj = h[j] - s * d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h[i] = hi
                    h[j] = hj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved = True
                    else:
                        h[i] -= s * d
                        h[j] += s * d
    return h, best_c

def make_diverse_starts(n, rng, h_parent):
    x = np.linspace(0, 2, n)
    starts = []
    if h_parent is not None:
        starts.append(h_parent.copy())
    starts.append(project_feasible(np.full(n, 0.5), n))
    for k in [1, 2, 3, 4, 5, 7]:
        starts.append(project_feasible(0.5 + 0.35 * np.sin(k * np.pi * x) + 0.1 * np.sin(2 * k * np.pi * x), n))
    for _ in range(8):
        starts.append(project_feasible(rng.uniform(0.2, 0.8, n), n))
    for period in [3, 5, 7, 11]:
        pat = np.array([1.0 if (i % period) < (period + 1) // 2 else 0.0 for i in range(n)])
        starts.append(project_feasible(pat, n))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n = len(h_parent)
    except Exception:
        h_parent = None
        n = 100

    best_h = h_parent.copy() if h_parent is not None else project_feasible(np.full(n, 0.5), n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Moderate alpha SLSQP from parent (20% time)
    p1_end = t0 + (deadline - t0) * 0.20
    for alpha in [200, 1000, 5000, 20000]:
        if time.time() > p1_end:
            break
        h_new = slsqp_solve(best_h, n, alpha, 3000, 1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()

    # Phase 1b: Try diverse starts (15% time)
    p1b_end = t0 + (deadline - t0) * 0.35
    starts = make_diverse_starts(n, rng, h_parent)
    for s in starts:
        if time.time() > p1b_end:
            break
        c_s = compute_c5(s, n)
        if c_s < best_c:
            best_c = c_s
            best_h = s.copy()
        if time.time() > p1b_end:
            break
        h_new = slsqp_solve(s, n, 5000, 1500, 1e-14)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()

    # Phase 2: High alpha SLSQP (30% time)
    p2_end = t0 + (deadline - t0) * 0.65
    alphas = [50000, 200000, 1000000, 5000000, 20000000]
    for alpha in alphas:
        if time.time() > p2_end:
            break
        for _ in range(2):
            if time.time() > p2_end:
                break
            h_new = slsqp_solve(best_h, n, alpha, 4000, 1e-18)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new.copy()
            else:
                break

    # Phase 3: Subgradient descent (15% time)
    p3_end = t0 + (deadline - t0) * 0.80
    for tk in [3, 5, 8, 15]:
        if time.time() > p3_end:
            break
        h_new, c_new = subgrad_descent(best_h, n, p3_end, top_k=tk)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()

    # Phase 4: Fine pairwise local search (15% time)
    p4_end = deadline
    best_h, best_c = pairwise_local(best_h, n, p4_end, top_k=12)

    # Final SLSQP polish
    if time.time() < deadline - 2:
        h_new = slsqp_solve(best_h, n, 50000000, 2000, 1e-20)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Final local polish
    remaining = deadline - time.time()
    if remaining > 10:
        best_h, best_c = pairwise_local(best_h, n, deadline, top_k=15)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
