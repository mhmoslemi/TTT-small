                corr = np.correlate(h_hi, 1.0 - h_hi, mode="full")
                top_idx = np.argsort(corr)[::-1][:4]
                grad = np.zeros(n_hi)
                for li in top_idx:
                    lag = int(li) - (n_hi - 1)
                    for j in range(n_hi):
                        i1 = j - lag
                        i2 = j + lag
                        if 0 <= i1 < n_hi:
                            grad[j] += (1.0 - h_hi[i1])
                        if 0 <= i2 < n_hi:
                            grad[j] -= h_hi[i2]
                gn = np.linalg.norm(grad)
                if gn < 1e-14:
                    break
                grad = -grad / gn
                momentum = 0.75 * momentum + 0.25 * grad
                ss = 0.006 / (1.0 + 0.0005 * it)
                h_t = project_h(h_hi + ss * momentum, n_hi / 2.0)
                c5_t = compute_c5(h_t)
                if c5_t < c5_hi - 1e-15:
                    h_hi = h_t
                    c5_hi = c5_t

            # Downsample back if better
            if c5_hi < best_c5:
                h_back = np.interp(x_grid, x_hi, h_hi)
                h_back = project_h(h_back, target)
                c5_back = compute_c5(h_back)
                if c5_back < best_c5:
                    best_h, best_c5 = h_back.copy(), c5_back
                else:
                    best_h, best_c5 = h_hi.copy(), c5_hi

    # Phase 6: Binary rounding attempts
    if budget_s - elapsed() > 10:
        for t in np.linspace(0.2, 0.8, 25):
            hb = (best_h >= t).astype(float)
            hp = project_h(hb, target)
            c5_b = compute_c5(hp)
            if c5_b < best_c5:
                best_h, best_c5 = hp.copy(), c5_b
                # Quick subgradient polish on binary
                cur = best_h.copy()
                cur_c5 = best_c5
                momentum = np.zeros(n)
                for it in range(2000):
                    if budget_s - elapsed() < 3:
                        break
                    corr = np.correlate(cur, 1.0 - cur, mode="full")
                    top_idx = np.argsort(corr)[::-1][:3]
                    grad = np.zeros(n)
                    for li in top_idx:
                        lag = int(li) - (n - 1)
                        for j in range(n):
                            i1 = j - lag
                            i2 = j + lag
                            if 0 <= i1 < n:
                                grad[j] += (1.0 - cur[i1])
                            if 0 <= i2 < n:
                                grad[j] -= cur[i2]
                    gn = np.linalg.norm(grad)
                    if gn < 1e-14:
                        break
                    momentum = 0.7 * momentum - 0.3 * grad / gn
                    ss = 0.004
                    h_t = project_h(cur + ss * momentum, target)
                    c5_t = compute_c5(h_t)
                    if c5_t < cur_c5 - 1e-15:
                        cur = h_t
                        cur_c5 = c5_t
                        if cur_c5 < best_c5:
                            best_h, best_c5 = cur.copy(), cur_c5
                    else:
                        momentum *= 0.4

    # Final projection and verification
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, int(len(best_h))
