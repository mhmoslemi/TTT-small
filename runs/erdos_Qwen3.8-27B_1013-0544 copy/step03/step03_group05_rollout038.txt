        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5, best_h = c5, h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 150:
            mag = 0.05 + 0.15 * min(stall / 500.0, 1.0)
            n_pert = 3 + int(10 * min(stall / 300.0, 1.0))
            for _ in range(n_pert):
                k = np.random.randint(n)
                h[k] = np.clip(h[k] + np.random.uniform(-mag, mag), 0, 1)
            h = project_h(h, target)
            stall = 0

    # Phase 2: Binary SA from best continuous
    if tl() > 120:
        hb, cb = sa_binary(n, 100000, 0.004, seed, h_init=best_h)
        if cb < best_c5:
            best_c5, best_h = cb, hb
        h = best_h.copy()
        t_sub = time.time()
        while time.time() - t_sub < min(60.0, tl() * 0.2) and tl() > 80:
            h = coord_pass(h, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()

    # Phase 3: Extended CD with restarts
    if tl() > 60:
        h = best_h.copy()
        t_sub = time.time()
        sub_budget = min(tl() - 30, budget_s * 0.25)
        stall = 0
        while time.time() - t_sub < sub_budget and tl() > 30:
            h = coord_pass(h, n)
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-13:
                best_c5, best_h = c5, h.copy()
                stall = 0
            else:
                stall += 1
            if stall > 80:
                for _ in range(4):
                    k = np.random.randint(n)
                    h[k] = np.clip(h[k] + np.random.uniform(-0.08, 0.08), 0, 1)
                h = project_h(h, target)
                stall = 0

    # Phase 4: Multiple SA restarts at different temps
    if tl() > 40:
        for s in range(8):
            if tl() < 15:
                break
            temp = 0.001 + 0.003 * (s % 3) / 2.0
            hb, cb = sa_binary(n, 40000, temp, seed + 1000 + s * 13)
            if cb < best_c5:
                best_c5, best_h = cb, hb
                h = best_h.copy()
                for _ in range(30):
                    if tl() < 15:
                        break
                    h = coord_pass(h, n)
                    c5 = compute_c5(h)
                    if c5 < best_c5:
                        best_c5, best_h = c5, h.copy()

    # Phase 5: Fine CD polish
    if tl() > 15:
        h = best_h.copy()
        for _ in range(500):
            if tl() < 5:
                break
            h = coord_pass(h, n, grid_size=2001)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()

    # Final verification
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
