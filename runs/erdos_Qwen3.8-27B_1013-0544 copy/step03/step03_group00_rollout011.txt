def project_box_sum(h, target_sum):
    """Project h onto {x: 0≤x≤1, sum(x)=target_sum}"""
    # Use binary search on lambda for the constraint sum(x)=target_sum
    # x_i = clip(h_i - lambda, 0, 1)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        if x.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

import numpy as np
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    nc = len(corr)
    if top_k is not None and top_k < nc:
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(nc)
    c_sel = corr[top_idx]
    m = c_sel.max()
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wi = w[i]
        if wi < 1e-14:
            continue
        s = k - n + 1
        # dc/dh[j] = (1-h[j-s]) for max(0,s) <= j < min(n, n+s)
        #           + (-h[j+s]) for max(0,-s) <= j < min(n, n-s)
        j1_lo, j1_hi = max(0, s), min(n, n + s)
        if j1_hi > j1_lo:
            grad[j1_lo:j1_hi] += wi * (1.0 - h[j1_lo - s:j1_hi - s])
        j2_lo, j2_hi = max(0, -s), min(n, n - s)
        if j2_hi > j2_lo:
            grad[j2_lo:j2_hi] += wi * (-h[j2_lo + s:j2_hi + s])
    grad *= dx
    return val, grad


def proj_grad_optimize(h, beta, target, n_iters=500, lr_init=0.1, top_k=None):
    n = len(h)
    h = project_box_sum(h.copy(), target)
    lr = lr_init
    momentum = h.copy()
    mu = 0.8
    for it in range(n_iters):
        val, grad = smoothmax_val_grad(h, beta, top_k)
        if np.max(np.abs(grad)) < 1e-14:
            break
        # Backtracking line search
        step = lr
        for _ in range(30):
            h_new = h - step * grad
            h_new = project_box_sum(h_new, target)
            val_new, _ = smoothmax_val_grad(h_new, beta, top_k)
            if val_new < val - 1e-4 * step * np.dot(grad, h_new - h):
                break
            step *= 0.5
        else:
            break
        momentum = mu * momentum + (1 - mu) * h_new
        h = momentum
        h = project_box_sum(h, target)
        lr = max(lr * 0.995, 0.005)
    return h


def diverse_perturb(h, seed, mode=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    target = n / 2.0
    if mode == 0:

    elif mode == 1:

        return project_box_sum(h + pert, target)
    elif mode == 2:

        pert = np.zeros(n)
        pert[mask] = rng.uniform(-0.06, 0.06, mask.sum())
        return project_box_sum(h + pert, target)
    elif mode == 3:
        # Block swap perturbation
        b = max(2, n // 20)
        i = rng.randint(0, n - 2 * b)
        j = rng.randint(0, n - 2 * b)
        h2 = h.copy()
        h2[i:i+b], h2[j:j+b] = h2[j:j+b].copy(), h2[i:i+b].copy()
        return project_box_sum(h2, target)
    else:
        return project_box_sum(h + rng.randn(n) * (0.02 + 0.03 * mode), target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_box_sum(h_new, n_new / 2.0)
