import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

_CONSTRAINT_TARGET = 0.0
_OPT_BETA = 1.0
_OPT_TOPK = None

def _constraint_fun(x):
    return np.sum(x) - _CONSTRAINT_TARGET

def _constraint_jac(x):
    return np.ones(len(x))

def _obj(x):
    return smoothmax_val_grad(x, _OPT_BETA, _OPT_TOPK)[0]

def _objg(x):
    return smoothmax_val_grad(x, _OPT_BETA, _OPT_TOPK)[1]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_solve(h, beta, maxiter, top_k=None):
    global _CONSTRAINT_TARGET, _OPT_BETA, _OPT_TOPK
    n = len(h)
    _CONSTRAINT_TARGET = n / 2.0
    _OPT_BETA = beta
    _OPT_TOPK = top_k
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq', 'fun': _constraint_fun, 'jac': _constraint_jac})
    try:
        res = minimize(_obj, h, jac=_objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n / 2.0)
    except Exception:
        return project_feasible(h, n / 2.0)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def randomized_local_search(h, t0, time_limit, seed, max_sweeps=200):
    n = len(h)
    delta = 0.005
    h = h.copy()
    c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit - 5:
            break
        improved = False
        n_trials = 2000
        for _ in range(n_trials):
            if time.time() - t0 > time_limit - 5:
                break
            i, j = rng.randint(0, n, size=2)
            if i == j:
                continue
            if h[i] > delta and h[j] < 1.0 - delta:
                h[i] -= delta
                h[j] += delta
                c5_new = compute_c5(h)
                if c5_new < c5 - 1e-13:
                    c5 = c5_new
                    improved = True
                else:
                    h[i] += delta
                    h[j] -= delta
            elif h[j] > delta and h[i] < 1.0 - delta:
                h[j] -= delta
                h[i] += delta
                c5_new = compute_c5(h)
                if c5_new < c5 - 1e-13:
                    c5 = c5_new
                    improved = True
                else:
                    h[j] += delta
                    h[i] -= delta
        if not improved:
            delta *= 0.6
            if delta < 1e-8:
                break
    return h, c5

def basin_hop(h_best, best_c5, n_hops, beta_list, maxiter, noise_base,
              noise_decay, seed, t0, budget_s):
    rng = np.random.RandomState(seed)
    n = len(h_best)
    target = n / 2.0
    for hop in range(n_hops):
        if budget_s - (time.time() - t0) < 60:
            break
        noise = noise_base * (noise_decay ** hop)
        h_pert = h_best + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_local = h_pert
        for beta in beta_list:
            if budget_s - (time.time() - t0) < 40:
                break
            h_local = slsqp_solve(h_local, beta, maxiter, top_k=20)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()
    return h_best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    else:
        h_cur = np.full(200, 0.5)

    best_c5 = compute_c5(h_cur)
    h_best = h_cur.copy()

    # Phase 1: SLSQP continuation at n=200
    for beta in [50, 150, 300, 600]:
        if time.time() - t0 > budget_s * 0.12:
            break
        h_new = slsqp_solve(h_cur, beta, 400, top_k=None)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            h_best = h_new.copy()
        h_cur = h_new

    # Phase 2: Local search at n=200
    h_ls, c_ls = randomized_local_search(h_best, t0, budget_s * 0.22, seed=seed)
    if c_ls < best_c5:
        best_c5 = c_ls
        h_best = h_ls.copy()

    # Phase 3: Interpolate to n=400 and optimize
    if time.time() - t0 < budget_s * 0.28:
        h_400 = interp_to_n(h_best, 400)
        for beta in [50, 150, 300, 600, 1000]:
            if time.time() - t0 > budget_s * 0.48:
                break
            h_400 = slsqp_solve(h_400, beta, 250, top_k=None)
        c_400 = compute_c5(h_400)
        if c_400 < best_c5:
            best_c5 = c_400
            h_best = h_400.copy()
        h_cur = h_400
    else:
        h_cur = h_best.copy()

    # Phase 4: Local search at current best resolution
    h_ls, c_ls = randomized_local_search(h_cur, t0, budget_s * 0.55, seed=seed + 10)
    if c_ls < best_c5:
        best_c5 = c_ls
        h_best = h_ls.copy()

    # Phase 5: Basin hopping at best resolution
    h_best, best_c5 = basin_hop(
        h_best, best_c5, n_hops=15,

        noise_base=0.02, noise_decay=0.72, seed=seed + 100,
        t0=t0, budget_s=budget_s)

    # Phase 6: Fine local search after basin hopping
    h_ls, c_ls = randomized_local_search(h_best, t0, budget_s * 0.82, seed=seed + 200)
    if c_ls < best_c5:
        best_c5 = c_ls
        h_best = h_ls.copy()

    # Phase 7: Small basin hops
    h_best, best_c5 = basin_hop(
        h_best, best_c5, n_hops=8,
        beta_list=[400, 800], maxiter=250,
        noise_base=0.01, noise_decay=0.8, seed=seed + 300,
        t0=t0, budget_s=budget_s)

    # Phase 8: Final SLSQP polish
    if budget_s - (time.time() - t0) > 30:
        h_fin = slsqp_solve(h_best, 1000, 600, top_k=None)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin.copy()

    h_ls, c_ls = randomized_local_search(h_best, t0, budget_s - 3, seed=seed + 400)
    if c_ls < best_c5:
        best_c5 = c_ls
        h_best = h_ls.copy()

    # Final validation
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)

    return h_best, best_c5, len(h_best)
