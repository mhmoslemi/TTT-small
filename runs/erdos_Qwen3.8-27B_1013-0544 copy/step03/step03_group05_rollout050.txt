import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def smooth_max_and_weights(corr, beta):
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(x_shift)
    s = np.sum(e)
    return m + np.log(s / len(corr)) / beta, e / s

def grad_smooth_max(h, corr, w, n, dx):
    """True gradient of dx * smooth_max(correlate(h, 1-h)) w.r.t. h"""
    center = n - 1
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-15:
            continue
        L = idx - center
        absL = abs(L)
        if absL == 0:
            grad += wi * (1.0 - 2.0 * h) * dx
        else:
            g = np.zeros(n)
            g[absL:] += (1.0 - h[:n - absL])
            g[:n - absL] -= h[absL:]
            grad += wi * g * dx
    return grad

def gradient_descent(h, n, target, dx, max_iters, time_fn, time_limit, best_state, 
                     beta_start=200.0, beta_end=2000.0):
    """Gradient descent with Armijo + adaptive beta for smooth max."""
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    cur_f = cur_c5  # track smooth max value
    if cur_c5 < best_state[0]:
        best_state[0] = cur_c5
        best_state[1] = cur.copy()
        best_state[2] = n
    
    step = 0.1
    for it in range(max_iters):
        if time_fn() < time_limit:
            break
        beta = beta_start + (beta_end - beta_start) * it / max(1, max_iters)
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        f_val, w = smooth_max_and_weights(corr, beta)
        grad = grad_smooth_max(cur, corr, w, n, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Armijo backtracking
        alpha = step
        satisfied = False
        for _ in range(30):
            trial = project_h(cur - alpha * grad, target)
            trial_corr = np.correlate(trial, 1.0 - trial, mode="full")
            trial_f, _ = smooth_max_and_weights(trial_corr, beta)
            if trial_f < f_val - 1e-4 * alpha * gnorm:
                cur = trial
                f_val = trial_f
                satisfied = True
                break
            alpha *= 0.5
        if not satisfied:
            # Try a small random perturbation to escape
            if np.random.random() < 0.1:
                noise = np.random.normal(0, 0.01, n)
                cur = project_h(cur + noise, target)
                f_val = compute_c5(cur)
                step *= 0.5
            continue
        
        # Update best
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_state[0]:
            best_state[0] = cur_c5
            best_state[1] = cur.copy()
            best_state[2] = n
        
        # Adaptive step
        if satisfied and alpha > step:
            step = min(0.5, step * 1.1)
        elif alpha < step * 0.1:
            step = max(0.001, step * 0.5)
    
    return cur, cur_c5, n

def top_k_subgradient_descent(h, n, target, dx, max_iters, time_fn, time_limit, best_state):
    """Subgradient method focusing on top-k lags (fast, good for final polish)."""
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    step = 0.03
    for it in range(max_iters):
        if time_fn() < time_limit:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full") * dx
        k = min(5, 2 * n // 50 + 1)
        top_idx = np.argsort(corr)[::-1][:k]
        vals = corr[top_idx]
        temp = 500.0
        weights = np.exp((vals - vals[0]) * temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        center = n - 1
        for t_idx, wt in zip(top_idx, weights):
            L = int(t_idx) - center
            absL = abs(L)
            if absL == 0:
                grad += wt * (1.0 - 2.0 * cur)
            else:
                g = np.zeros(n)
                g[absL:] += (1.0 - cur[:n - absL])
                g[:n - absL] -= cur[absL:]
                grad += wt * g
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        # Backtracking
        alpha = step
        improved = False
        for _ in range(20):
            trial = project_h(cur - alpha * grad, target)
            trial_c5 = compute_c5(trial)
            if trial_c5 < cur_c5 - 1e-15:
                cur = trial
                cur_c5 = trial_c5
                improved = True
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n

                    best_state[1] = cur.copy()
                    best_state[2] = n

    n_candidates = sorted(set([n, max(n + 50, 300), max(n + 100, 400), 500]))
    n_candidates = [x for x in n_candidates if 150 <= x <= 600 and x != n]

    for n_trial in n_candidates:
        if time_left() < 80:
            break
        cur_new = resample(best_state[1], n_trial)
        target_new = n_trial / 2.0
        dx_new = 2.0 / n_trial
        cur_new = project_h(cur_new, target_new)
        c5_init = compute_c5(cur_new)
        if c5_init < best_state[0]:
            best_state[0] = c5_init
            best_state[1] = cur_new.copy()
            best_state[2] = n_trial
        cur_new, _, _ = gradient_descent(cur_new, n_trial, target_new, dx_new,
                                          max_iters=2500, time_fn=time_left,
                                          time_limit=80, best_state=best_state,
                                          beta_start=80, beta_end=2000)
        # Subgradient polish at this resolution
        if time_left() > 40:
            cur_new, _, _ = subgrad_polish(cur_new, n_trial, target_new, dx_new,
                                            max_iters=3000, time_fn=time_left,
                                            time_limit=40, best_state=best_state)

    # Phase 3: Try diverse initializations at parent resolution
    if time_left() > 100:
        x = np.linspace(0, 2, n_parent)
        inits = []
        for mode in [1, 2, 3]:
            h_f = 0.5 + 0.3 * np.cos(mode * np.pi * x / 2) + 0.1 * np.sin(2 * mode * np.pi * x / 3)
            inits.append(project_h(h_f, n_parent / 2.0))
        h_g = 0.5 + 0.4 * np.exp(-((x - 1.0) / 0.4) ** 2) - 0.2 * np.exp(-((x - 0.3) / 0.25) ** 2)
        inits.append(project_h(h_g, n_parent / 2.0))
        h_b = 0.5 + 0.2 * (np.exp(-((x - 0.4) / 0.2) ** 2) + np.exp(-((x - 1.6) / 0.2) ** 2))
        inits.append(project_h(h_b, n_parent / 2.0))
        for h_init in inits:
            if time_left() < 40:
                break
            h_trial = project_h(h_init, n_parent / 2.0)
            c5_t = compute_c5(h_trial)
            if c5_t < best_state[0]:
                best_state[0] = c5_t
                best_state[1] = h_trial.copy()
                best_state[2] = n_parent
            h_trial, _, _ = gradient_descent(h_trial, n_parent, n_parent / 2.0, 2.0 / n_parent,
                                              max_iters=1500, time_fn=time_left,
                                              time_limit=40, best_state=best_state,
                                              beta_start=100, beta_end=2000)

    # Phase 4: Final subgradient polish
    best_h = best_state[1].copy()
    best_n = best_state[2]
    best_target = best_n / 2.0
    best_dx = 2.0 / best_n
    best_h, best_c5, _ = subgrad_polish(best_h, best_n, best_target, best_dx,
                                         max_iters=8000, time_fn=time_left,
                                         time_limit=8, best_state=best_state)

    # Final projection
    final_h = project_h(best_state[1].copy(), best_state[2] / 2.0)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_state[0]:
        best_state[0] = final_c5
        best_state[1] = final_h.copy()

    return best_state[1], best_state[0], best_state[2]
