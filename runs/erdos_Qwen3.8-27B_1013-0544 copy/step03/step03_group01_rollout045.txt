            if not mask.any(): break
            t = h[mask].sum()
            if t < 1e-14: break
            h[mask] += (diff / t) * h[mask]

            break
        
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            h_try = project_feasible(best_h + alpha * (h_lp - best_h), n)
            c_try, c_vals_try = compute_c5_and_vals(h_try, n)
            if c_try < best_c - 1e-15:
                best_h = h_try
                best_c = c_try
                c_vals = c_vals_try
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
    
    return best_h, best_c
