    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sml_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(c)
    x_shift = np.clip(beta * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(x_shift)) / len(c)) / beta) * dx

def sum_eq(x_in, target):
    return np.sum(x_in) - target

def sum_eq_jac(x_in, target):
    return np.ones(x_in.shape)

def lag_grad(h, lag_idx):
    n = len(h)
    lag = int(lag_idx) - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[:n-lag] += (1.0 - h[lag:])
        grad[lag:] -= h[:n-lag]
    else:
        la = -lag
        grad[la:] += (1.0 - h[:n-la])
        grad[:n-la] -= h[la:]
    return grad

    for _ in range(n_iter):
        if budget - (time.time() - t0) < 1:
            break
        if ca < cb:
            hi, b, cb = b, a, ca
            a = hi - gr * (hi - lo)
            ha = set_coord_proj(h, j, a, target)
            ca = compute_c5(ha)
        else:
            lo, a, ca = a, b, cb
            b = lo + gr * (hi - lo)
            hb = set_coord_proj(h, j, b, target)
            cb = compute_c5(hb)
    mid = (lo + hi) / 2.0
    trial = set_coord_proj(h, j, mid, target)
    c = compute_c5(trial)
    c0 = compute_c5(h)
    if c < c0:
        return trial, c
    return h, c0

def golden_sweep(h, target, t0, budget, n_iter):
    cur = h.copy()
    best = compute_c5(cur)
    n = len(cur)
    for j in range(n):
        if budget - (time.time() - t0) < 2:
            break
        trial, c = golden_coord(cur, j, target, t0, budget, n_iter)
        if c < best - 1e-15:
            cur, best = trial, c
    return cur, best

def lag_grad(h, lag_idx):
    n = len(h)
    lag = int(lag_idx) - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[:n-lag] += (1.0 - h[lag:])
        grad[lag:] -= h[:n-lag]
    else:
        la = -lag
        grad[la:] += (1.0 - h[:n-la])
        grad[:n-la] -= h[la:]
    return grad

def focused_descent(h, target, n_iter, step, t0, budget, topk):
    cur = h.copy()
    best = compute_c5(cur)
    mom = np.zeros(len(cur))
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        tl = np.argsort(corr)[::-1][:topk]
        g = np.zeros(len(cur))
        for ti in tl:
            g += lag_grad(cur, ti)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        step_it = step / (1.0 + 0.001 * it)
        mom = 0.88 * mom - step_it * g
        trial = project_h(cur + mom, target)
        c = compute_c5(trial)
        if c < best - 1e-15:
            cur, best = trial, c
            stall = 0
        else:
            stall += 1
            mom *= 0.4
            trial = project_h(cur + mom * 0.5, target)
            c = compute_c5(trial)
            if c < best - 1e-15:
                cur, best = trial, c
                stall = 0
            else:
                mom = np.zeros(len(cur))
        if stall > 600:
            break
    return cur, best

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h))
    new_x = np.linspace(0.0, 2.0, n_new)
    return np.interp(new_x, old_x, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(320, 0.5)

    n = len(h0)
    target = n / 2.0
    best_h = project_h(h0, target)
    best_c5 = compute_c5(best_h)
    bounds = Bounds(np.zeros(n), np.ones(n))

    # Phase 1: SLSQP at increasing smoothness
    cur = best_h.copy()
    for beta in [500, 1500, 4000]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 15:
            break
        cur = do_slsqp(cur, n, target, beta, bounds, 3000, 1e-14)
        c = compute_c5(cur)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 2: Golden section coordinate optimization (key novelty)
    remaining = budget_s - (time.time() - t0)
    if remaining > 50:
        cur, c = golden_sweep(cur, target, t0, budget_s, n_iter=20)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 3: Focused subgradient descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        cur, c = focused_descent(cur, target, 10000, 0.012, t0, budget_s, topk=4)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 4: Second golden sweep (finer)
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        cur, c = golden_sweep(cur, target, t0, budget_s, n_iter=28)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 5: SLSQP polish with very high beta
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        for beta in [10000, 30000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            cur = do_slsqp(best_h.copy(), n, target, beta, bounds, 5000, 1e-15)
            c = compute_c5(cur)
            if c < best_c5:
                best_h, best_c5 = cur.copy(), c

    # Phase 6: Final golden + subgradient
    remaining = budget_s - (time.time() - t0)
    if remaining > 35:
        cur, c = golden_sweep(best_h.copy(), target, t0, budget_s, n_iter=30)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        cur, c = focused_descent(best_h.copy(), target, 5000, 0.005, t0, budget_s, topk=3)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 7: Perturbation + re-optimize from different starts
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        for att in range(5):
            remaining = budget_s - (time.time() - t0)
            if remaining < 12:
                break
            if att == 0:
                # Binary init: random 160 ones, 160 zeros
                perm = np.random.permutation(n)
                hp = np.zeros(n)
                hp[perm[:n//2]] = 1.0
                hp = project_h(hp, target)
            else:
                scale = 0.02 / (att + 1)
                hp = project_h(best_h + np.random.randn(n) * scale, target)
            hp = do_slsqp(hp, n, target, 2000, bounds, 1500, 1e-14)
            hp, cp = golden_sweep(hp, target, t0, budget_s, n_iter=15)
            if cp < best_c5:
                best_h, best_c5 = hp.copy(), cp

    # Phase 8: Try alternate resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 50:
        n_alt = 256
        target_alt = n_alt / 2.0
        bounds_alt = Bounds(np.zeros(n_alt), np.ones(n_alt))
        h_alt = project_h(resample_h(best_h, n_alt), target_alt)
        h_alt = do_slsqp(h_alt, n_alt, target_alt, 2000, bounds_alt, 2000, 1e-14)
        h_alt, c_alt = golden_sweep(h_alt, target_alt, t0, budget_s, n_iter=20)
        h_alt, c_alt = focused_descent(h_alt, target_alt, 3000, 0.008, t0, budget_s, topk=3)
        if c_alt < best_c5:
            best_h, best_c5, n = h_alt.copy(), c_alt, n_alt

    return best_h, best_c5, n
