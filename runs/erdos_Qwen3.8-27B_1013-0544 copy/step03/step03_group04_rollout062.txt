    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            # If all at bounds, do a more careful projection
            # Find the element closest to the bound we need to move
            if diff > 0:
                # Need to increase sum, find element closest to 1
                idx = np.argmin(1.0 - h)
                h[idx] = min(1.0, h[idx] + diff)
            else:
                # Need to decrease sum, find element closest to 0
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + diff)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

Let me also add a **vectorized coordinate descent** that's faster:

For each sweep, instead of iterating i=0 to n-1 and computing correlation each time, I can:
- Precompute the "base" correlation (without the i-th element's contribution)
- For each i, the change in correlation is a simple rank-1 update
- This makes each evaluation O(n) instead of O(n²)

The change in corr when h[i] changes by δ:
- corr[m] changes by: δ * (1 - h[i + (m - (n-1))]) - δ * h[i - (m - (n-1))])
  where the indices are bounded to [0, n-1]

So for a given i and δ, the new corr is:
new_corr[m] = corr[m] + δ * ((1 - h[i + m - n + 1]) * (0 <= i + m - n + 1 < n) - h[i - m + n - 1] * (0 <= i - m + n - 1 < n))

This is a vectorized operation! For each i, I can compute all 3 candidate values' effects in O(n) time.

def coord_descent_vec(h, target, max_sweeps=10000, t0=None, time_limit=None, beta_sm=1000.0):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(corr) * dx)
    
    # Precompute index offsets
    # For coordinate i, the corr index m corresponds to lag L = m - (n-1)
    # The two terms are at h[i+L] and h[i-L]
    # m = i + L + (n-1) for the first term
    # m = i - L + (n-1) for the second term
    # So for coordinate i:
    #   First term affects corr indices: m = i + L + n - 1 for L where 0 <= i+L < n
    #   Second term affects corr indices: m = i - L + n - 1 for L where 0 <= i-L < n
    
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        any_improved = False
        for i in range(n):
            old = h[i]
            best_new = old
            best_obj = smooth_max(corr, beta_sm) * dx + 1e4*(h.sum()-target)**2
            
            for cv in [0.0, 1.0, 0.5]:
                if cv == old:
                    continue
                delta = cv - old
                new_corr = corr.copy()
                # First term: h[i] * (1 - h[i+L]) for valid L
                # This affects corr[m] where m = (n-1) + L and i+L < n
                # i.e., L < n-i, so m < (n-1) + (n-i) = 2n-1-i
                # And L >= -(n-1) (always valid), but we need i+L >= 0, which is always true since L >= i-(n-1) >= -(n-1) and i >= 0
                # Actually L ranges from -(n-1) to (n-1), and we need 0 <= i+L < n
                # So L ranges from -i to n-1-i
                # m = (n-1) + L ranges from (n-1)-i to (n-1)+(n-1-i) = 2n-2-i
                
                L_range = np.arange(-i, n-i)  # valid L values for first term
                m_range = (n-1) + L_range  # corresponding corr indices
                valid = (i + L_range >= 0) & (i + L_range < n)
                # Since L_range already ensures i+L >= 0 and i+L < n, valid should all be True
                new_corr[m_range] += delta * (1.0 - h[i + L_range])
                
                # Second term: h[i-L] * (1 - h[i]) for valid L
                # This affects corr[m] where m = (n-1) + L and i-L >= 0
                # i.e., L <= i, and i-L < n (always true since L >= -(n-1))
                # So L ranges from -(n-1) to i
                L_range2 = np.arange(-(n-1), i+1)
                m_range2 = (n-1) + L_range2
                new_corr[m_range2] += delta * (-h[i - L_range2])
                # Wait: the term is h[i-L] * (1-h[i]), so when h[i] changes by delta,
                # the change is h[i-L] * (-delta) = -delta * h[i-L]
                # So new_corr[m_range2] += -delta * h[i-L_range2]
                # i.e., new_corr[m_range2] += delta * (-h[i-L_range2])
                
                obj = smooth_max(new_corr, beta_sm) * dx + 1e4*(h.sum()+delta-target)**2
                if obj < best_obj:
                    best_obj = obj
                    best_new = cv
                    best_corr = new_corr
            
            if best_new != old:
                h[i] = best_new
                corr = best_corr
                any_improved = True
        
        if not any_improved:
            break
    
    h = project_h(h, target)
    return h

The "lag" in our problem is k (shift of the second function). The overlap at shift k is:

Now, when h[i] changes by δ:
- In c_k, the terms involving h[i] are:

  - As part of the second factor: h[i-k] * (1-h[i]) → change: h[i-k] * (-δ) = -δ * h[i-k] (if i-k >= 0)

So: c_k(new) = c_k(old) + δ * (1-h[i+k]) * [i+k < n] - δ * h[i-k] * [i-k >= 0]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            if diff > 0:
                idx = np.argmin(1.0 - h)
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + diff)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx + 1e5 * (h.sum() - target) ** 2

    rng = np.random.default_rng(seed)
    n = len(h_init)
    h = (h_init >= 0.5).astype(float)
    diff = target - h.sum()
    if diff > 0:

        order = zeros[np.argsort(h_init[zeros])[::-1]]
        n_flip = int(np.ceil(diff))
        if n_flip < len(order):
            h[order[:n_flip]] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        order = ones[np.argsort(h_init[ones])]
        n_flip = int(np.ceil(-diff))
        if n_flip < len(order):

    diff = target - h.sum()
    if diff > 0:
        zeros = np.where(h == 0)[0]
        h[zeros[:int(diff)]] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        h[ones[:int(-diff)]] = 0.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = 0.003
    T_min = 0.00003

            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)

            best_c5 = c5_new

                h[j] = 0.0

            cur_c5 = c5_new

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            if diff > 0:
                idx = np.argmin(1.0 - h)
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + diff)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def _slsqp_obj(x, target, dx, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx + 1e5 * (h.sum() - target) ** 2

def _con_fun(x, target):
    return np.sum(x) - target

def _con_jac(x, target):
    return np.ones(x.shape)

def slsqp_optimize(h_init, target, dx, betas, maxiter_each, time_limit):
    n = len(h_init)
    h_cur = h_init.copy()
    cur_c5 = compute_c5(h_cur)
    bounds_s = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (target,)}]
    for beta in betas:
        if time.time() > time_limit:
            break
        res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds_s,
                       args=(target, dx, float(beta)),
                       constraints=constraints,
                       options={'maxiter': maxiter_each, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_h(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h_cur = h_new
            cur_c5 = c5_new
    return h_cur, cur_c5

def coord_descend(h, target, max_sweeps, time_limit, candidates):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if time.time() > time_limit:
            break
        improved = False
        for i in range(n):
            old = h[i]
            best_c5_i = best_c5
            best_val = old
            for cv in candidates:
                if abs(cv - old) < 1e-8:
                    continue
                h[i] = cv
                c5 = compute_c5(h)
                if c5 < best_c5_i - 1e-14:
                    best_c5_i = c5
                    best_val = cv
            h[i] = best_val
            if best_val != old:
                improved = True
                best_c5 = best_c5_i
        if not improved:
            break
    h = project_h(h, target)
    return h, compute_c5(h)

def binary_init(h_init, target):
    n = len(h_init)
    h = (h_init >= 0.5).astype(float)
    diff = target - h.sum()
    if diff > 0:
        zeros = np.where(h == 0)[0]
        order = zeros[np.argsort(h_init[zeros])[::-1]]
        h[order[:int(np.ceil(diff))]] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        order = ones[np.argsort(h_init[ones])]

    diff = target - h.sum()
    if diff > 0:
        zeros = np.where(h == 0)[0]
        h[zeros[:int(diff)]] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        h[ones[:int(-diff)]] = 0.0
    return h
