    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_optimize(h0, n, beta, maxiter=5000):

    dx = 2.0 / n
    def obj(x):
        h_p = np.clip(x, 0, 1)
        corr = np.correlate(h_p, 1.0 - h_p, mode="full")
        return float(smooth_max(corr, beta) * dx)
    def con(x):
        return np.sum(x) - target
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': con, 'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_h(res.x, target)

    indices = np.random.choice(len(h), size=n_kicks, replace=False)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def make_binary(h, target):
    n = len(h)
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:int(round(target))]] = 1.0
    return hb

def slsqp_step(h0, n, beta, maxiter):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        hp = np.clip(x, 0, 1)
        c = np.correlate(hp, 1.0 - hp, mode="full")
        return float(smooth_max(c, beta) * dx)
    def con(x):
        return np.sum(x) - target
    bounds = Bounds(np.zeros(n), np.ones(n))
    constr = [{'type': 'eq', 'fun': con, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=constr, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(res.x, target)
    except:
        return h0.copy()

def binary_anneal(h, target, max_evals, seed, T0=0.005, Tend=0.0001):
    rng = np.random.RandomState(seed)
    n = len(h)
    h = make_binary(h, target).copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for e in range(max_evals):
        T = T0 * (Tend / T0) ** (e / max_evals)
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        if h[i] == 0:
            i, j = j, i
        h[i], h[j] = 0, 1
        c5n = compute_c5(h)
        d = c5n - c5
        if d < 0 or rng.random() < np.exp(-d / T):
            c5 = c5n
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
    return best_h, best_c5

def binary_steepest(h, target, max_passes, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    h = make_binary(h, target).copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for p in range(max_passes):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 0
        best_ij = None
        # Sample swaps

        for _ in range(n_pairs):

            j = zeros[rng.randint(len(zeros))]

            h[i], h[j] = 1.0, 0.0
            d = c5 - c5t

                best_delta = d
                best_ij = (i, j)
        if best_ij is None:
            break
        i, j = best_ij
        h[i], h[j] = 0.0, 1.0
        c5 -= best_delta
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5
