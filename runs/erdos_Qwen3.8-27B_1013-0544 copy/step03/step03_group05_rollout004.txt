    return h


def subgrad_momentum(h, n, target, dx, max_iters, start_time, budget_s, min_time_left, best_state):
    vel = np.zeros(n)
    mu = 0.85
    for it in range(max_iters):
        if (time.time() - start_time) > budget_s - min_time_left:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        max_val = np.max(corr)
        active_mask = corr >= max_val * 0.99
        active_idx = np.where(active_mask)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(corr)])
        w = corr[active_idx]
        w = w - w.min() + 1e-10
        w /= w.sum()
        grad = np.zeros(n)
        center = n - 1
        for idx, wi in zip(active_idx, w):
            lag = int(idx) - center
            grad += wi * grad_for_lag(h, lag, n, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        lr = 0.015 / (1.0 + 0.001 * it)
        vel = mu * vel - lr * grad
        h_new = project_h(h + vel, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_state[0]:
            best_state[0] = c5_new
            best_state[1] = h_new.copy()
            best_state[2] = n
        h = h_new
    return h


def coord_polish(h, n, target, start_time, budget_s, min_time_left, best_state):
    dx = 2.0 / n
    for it in range(8000):
        if (time.time() - start_time) > budget_s - min_time_left:
            break
        i = np.random.randint(0, n - 2)
        j = i + np.random.randint(1, 4)
        hi, hj = h[i], h[j]
        for delta in np.array([-0.04, -0.02, -0.01, 0.01, 0.02, 0.04]):
            h_try = h.copy()
            h_try[i] = hi + delta
            h_try[j] = hj - delta
            if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
                continue
            h_try = project_h(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_state[0] - 1e-15:
                best_state[0] = c5_try
                best_state[1] = h_try.copy()
                best_state[2] = n
                h = h_try
            break
    return h


def slsqp_phase(h, n, target, dx, betas, start_time, budget_s, min_time_left, best_state):
    bounds = Bounds(np.zeros(n), np.ones(n))
    for beta in betas:
        if (time.time() - start_time) > budget_s - min_time_left:
            break
        def obj_fn(h_cur):
            c = np.correlate(h_cur, 1.0 - h_cur, mode="full")
            m = np.max(c)
            xs = np.clip(beta * (c - m), -500, 500)
            return (m + np.log(np.sum(np.exp(xs)) / len(xs)) / beta) * dx
        def obj_grad_fn(h_cur):
            c = np.correlate(h_cur, 1.0 - h_cur, mode="full") * dx
            m_v = np.max(c)
            xs = np.clip(beta * (c - m_v), -500, 500)
            w = np.exp(xs)
            w /= w.sum()
            g = np.zeros(len(h_cur))
            ctr = len(h_cur) - 1
            for idx in range(len(w)):
                if w[idx] < 1e-9:
                    continue
                g += w[idx] * grad_for_lag(h_cur, idx - ctr, len(h_cur), dx)
            return g
        def con_fn(h_cur):
            return np.sum(h_cur) - target
        def con_jac_fn(h_cur):
            return np.ones(len(h_cur))
        try:
            res = minimize(obj_fn, h, jac=obj_grad_fn, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con_fn, 'jac': con_jac_fn}],
                          options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_state[0]:
                best_state[0] = c5_new
                best_state[1] = h_new.copy()
                best_state[2] = n
            h = h_new
        except Exception:
            pass
    return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 250
        h_parent = np.full(n_parent, 0.5)

    best_state = [compute_c5(h_parent), h_parent.copy(), n_parent]

    n_list = sorted(set([n_parent, 200, 250, 300]))
    n_list = [n for n in n_list if 100 <= n <= 400]

    for n in n_list:
        if (time.time() - start_time) > budget_s - 80:
            break
        target = n / 2.0
        dx = 2.0 / n

        if n == n_parent:
            h_init = h_parent.copy()
        else:
            h_init = project_h(resample(h_parent, n), target)
        if best_state[2] != n:
            h_alt = project_h(resample(best_state[1], n), target)
            if compute_c5(h_alt) < compute_c5(h_init):
                h_init = h_alt

        cur = project_h(h_init, target)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_state[0]:
            best_state = [cur_c5, cur.copy(), n]

        # SLSQP with smooth max
        cur = slsqp_phase(cur, n, target, dx, [50, 100, 200, 400],
                         start_time, budget_s, 40, best_state)

        # Directional smoothing on active lags
        if (time.time() - start_time) < budget_s - 30:
            for alpha_try in [0.3, 0.15, 0.08]:
                corr = np.correlate(cur, 1.0 - cur, mode="full") * dx
                max_val = np.max(corr)
                active = np.where(corr >= max_val * 0.995)[0]
                center = n - 1
                for aidx in active[:3]:
                    lag = int(aidx) - center
                    if lag != 0 and abs(lag) < n:
                        cur2 = directional_smooth(cur, lag, n, dx, alpha_try, target)
                        c5_2 = compute_c5(cur2)
                        if c5_2 < best_state[0]:
                            best_state = [c5_2, cur2.copy(), n]
                            cur = cur2

        # Adam phase
        if (time.time() - start_time) < budget_s - 20:
            cur = adam_phase(cur, n, target, dx, 3000, start_time, budget_s, 15, best_state)

        # Subgradient with momentum
        if (time.time() - start_time) < budget_s - 10:
            cur = subgrad_momentum(cur, n, target, dx, 8000, start_time, budget_s, 5, best_state)

        # Coordinate polish
        if (time.time() - start_time) < budget_s - 5:
            cur = coord_polish(cur, n, target, start_time, budget_s, 3, best_state)

    # Final polish on best solution
    if (time.time() - start_time) < budget_s - 10:
        bh = best_state[1]
        bn = best_state[2]
        bt = bn / 2.0
        bd = 2.0 / bn
        bh = slsqp_phase(bh, bn, bt, bd, [600, 1200, 2500],
                        start_time, budget_s, 5, best_state)
        if (time.time() - start_time) < budget_s - 3:
            bh = subgrad_momentum(bh, bn, bt, bd, 15000, start_time, budget_s, 1, best_state)

    return best_state[1], best_state[0], best_state[2]
