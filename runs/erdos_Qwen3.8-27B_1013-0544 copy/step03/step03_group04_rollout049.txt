import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0, 1)
            continue
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def binary_local_search(h, max_iters=5000, top_k=10, time_limit=60.0, t_start=None):
    """Intensive binary local search with vectorized best-swap."""
    if t_start is None:
        t_start = time.time()
    n = len(h)

    h = h.copy()

    h[sorted_idx[:int(target)]] = 1.0
    
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        
        if max_corr * dx <= best_c5 * (1 - 1e-12):
            best_c5 = max_corr * dx
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Get top-k worst lags
        worst_idx = np.argsort(corr)[::-1][:top_k]

        # Precompute f and g for all positions, for the top_k lags
        # f[p, k] = (1-h[p+d_k]) if 0<=p+d_k<n, else 0
        # g[p, k] = h[p-d_k] if 0<=p-d_k<n, else 0
        f = np.zeros((n, top_k))
        g = np.zeros((n, top_k))
        for k_idx, d in enumerate(d_vals):
            valid_plus = (np.arange(n) + d >= 0) & (np.arange(n) + d < n)
            valid_minus = (np.arange(n) - d >= 0) & (np.arange(n) - d < n)
            f[np.where(valid_plus)[0], k_idx] = 1.0 - h[np.where(valid_plus)[0] + d]
            g[np.where(valid_minus)[0], k_idx] = h[np.where(valid_minus)[0] - d]
        
        # Vectorized: for all (i in zeros, j in ones)
        fz = f[zeros] - g[zeros]  # (n_zeros, top_k)
        fo = f[ones] - g[ones]    # (n_ones, top_k)
        
        # delta[i,j,k] = fz[i,k] - fo[j,k]
        # new_val[i,j,k] = R_worst[k] + delta[i,j,k]
        # new_max[i,j] = max_k new_val[i,j,k]

        # fz[:, :, k] - fo[k, :, :] ... hmm, need to be careful with shapes
        # fz: (nz, k), fo: (no, k), R_worst: (k,)

        new_max = np.max(new_val, axis=2)  # (nz, no)
        
        min_new_max = np.min(new_max)
        
        if min_new_max < max_corr - 1e-12:
            # Find the best swap
            flat_idx = np.argmin(new_max)
            i_idx = flat_idx // len(ones)
            j_idx = flat_idx % len(ones)
            h[zeros[i_idx]] = 1.0
            h[ones[j_idx]] = 0.0
            best_c5 = min(min_new_max, max_corr) * dx
        else:
            break
    
    return h, best_c5

    dx = 2.0 / n_pts
    
    def obj(x):
        h_proj = project_h(x, target)
        corr = np.correlate(h_proj, 1.0 - h_proj, mode="full")
        return smooth_max(corr, 500.0) * dx

        # Numerical gradient (or skip for SLSQP)
        return None

    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    while time.time() - t_start < time_limit:
        remaining = time_limit - (time.time() - t_start)
        if remaining < 30:
            break
        h_ls, c5_ls = binary_ls(best_h, max_iters=3000, top_k=12,
                                 t_start=t_start, time_limit=remaining * 0.3)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls
        remaining = time_limit - (time.time() - t_start)
        if remaining < 20:
            break
        h_sa, c5_sa = simulated_annealing(best_h, t_start=t_start,
                                           time_limit=remaining * 0.4,
                                           T_init=0.003, T_min=1e-5)
        if c5_sa < best_c5:
            best_h, best_c5 = h_sa, c5_sa
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        n_swap = np.random.randint(3, min(8, len(ones), len(zeros)) + 1)
        h_pert = best_h.copy()
        sz = np.random.choice(zeros, n_swap, replace=False)
        so = np.random.choice(ones, n_swap, replace=False)
        h_pert[sz] = 1.0
        h_pert[so] = 0.0
        h_ls2, c5_ls2 = binary_ls(h_pert, max_iters=2000, top_k=12,
                                   t_start=t_start, time_limit=remaining * 0.25)
        if c5_ls2 < best_c5:
            best_h, best_c5 = h_ls2, c5_ls2
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_parent)
    else:
        n_base = 280
        h_parent = np.full(n_base, 0.5)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    def elapsed():
        return time.time() - t0
    # Phase 1: SLSQP
    if budget_s - elapsed() > 150:
        h_ref = slsqp_refine(best_h, maxiter=5000)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref
    # Phase 2: Binary LS
    if budget_s - elapsed() > 100:
        h_b, c5_b = binary_ls(best_h, max_iters=10000, top_k=15,
                              t_start=t0, time_limit=60.0)
        if c5_b < best_c5:
            best_h, best_c5 = h_b, c5_b
    # Phase 3: SA
    if budget_s - elapsed() > 150:
        h_sa, c5_sa = simulated_annealing(best_h, t_start=t0,
                                           time_limit=150.0, T_init=0.008, T_min=1e-5)
        if c5_sa < best_c5:
            best_h, best_c5 = h_sa, c5_sa
    # Phase 4: ILS
    if budget_s - elapsed() > 200:
        h_ils, c5_ils = ilsearch(best_h, t_start=t0, time_limit=budget_s - elapsed() - 100)
        if c5_ils < best_c5:
            best_h, best_c5 = h_ils, c5_ils
    # Phase 5: Higher resolution
    for mult in [1.5, 2.0]:
        n_hi = int(n_base * mult)
        if n_hi > 600 or budget_s - elapsed() < 80:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_hi)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_h(h_hi, n_hi / 2.0)
        if budget_s - elapsed() > 60:
            h_hi = slsqp_refine(h_hi, maxiter=2000)
        if budget_s - elapsed() > 40:
            h_hb, c5_hb = binary_ls(h_hi, max_iters=3000, top_k=12,
                                     t_start=t0, time_limit=40.0)
            if c5_hb < best_c5:
                best_h, best_c5 = h_hb, c5_hb
    n_final = len(best_h)
    best_h = project_h(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, int(n_final)
