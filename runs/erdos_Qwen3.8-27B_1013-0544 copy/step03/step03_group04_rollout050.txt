    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

    """Vectorized subgradient of sum of top-k correlation values."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]

                grad[:n - l] += -h[l:]
    return grad

def sa_binary(h_init, target, n_iter, t_start, t_end, seed, t0, time_budget):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = (h_init >= 0.5).astype(float)
    deficit = int(round(target)) - int(h.sum())
    if deficit > 0:
        idx = np.where(h == 0)[0]
        h[rng.choice(idx, size=min(deficit, len(idx)), replace=False)] = 1.0
    elif deficit < 0:
        idx = np.where(h == 1)[0]
        h[rng.choice(idx, size=min(-deficit, len(idx)), replace=False)] = 0.0
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for it in range(n_iter):
        if time.time() - t0 > time_budget:
            break
        T = t_start * (t_end / t_start) ** (it / max(n_iter, 1))
        if rng.random() < 0.9:
            ia = rng.randint(len(ones_list))
            ib = rng.randint(len(zeros_list))
            a, b = ones_list[ia], zeros_list[ib]
            h[a], h[b] = 0.0, 1.0
            ones_list[ia], zeros_list[ib] = b, a
            new_c5 = compute_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_h, best_c5 = h.copy(), new_c5
            else:
                h[a], h[b] = 1.0, 0.0
                ones_list[ia], zeros_list[ib] = a, b
        else:
            ia, ib = rng.sample(range(len(ones_list)), 2)
            ja, jb = rng.sample(range(len(zeros_list)), 2)
            a, b = ones_list[ia], ones_list[ib]
            c, d = zeros_list[ja], zeros_list[jb]
            h[a], h[b], h[c], h[d] = 0.0, 0.0, 1.0, 1.0
            ones_list[ia], ones_list[ib] = c, d
            zeros_list[ja], zeros_list[jb] = a, b
            new_c5 = compute_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_h, best_c5 = h.copy(), new_c5
            else:
                h[a], h[b], h[c], h[d] = 1.0, 1.0, 0.0, 0.0
                ones_list[ia], ones_list[ib] = a, b
                zeros_list[ja], zeros_list[jb] = c, d
    return best_h, best_c5

def slsqp_refine(h_init, n_pts, target, max_iters, budget_remaining, t0):
    dx = 2.0 / n_pts
    h_cur = project_h(h_init, target)
    cur_c5 = compute_c5(h_cur)
    best_h, best_c5 = h_cur.copy(), cur_c5
    bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    for beta in [50, 100, 200, 500, 1000, 2000]:
        if budget_remaining() < 15:
            break
        def obj(x, b=beta):
            c = np.correlate(x, 1.0 - x, mode="full")
            return smooth_max(c, b) * dx + 1e-5 * (x.sum() - target) ** 2
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           options={'maxiter': max_iters, 'ftol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5:
                h_cur, cur_c5 = h_new, c5_new
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_parent)
    else:
        n_base = 280
        h_parent = np.full(n_base, 0.5)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    target_base = n_base / 2.0
    def elapsed(): return time.time() - t0
    def remaining(): return budget_s - elapsed()

    # Estimate SA iteration cost
    h_t = (best_h >= 0.5).astype(float)
    te = time.time()
    for _ in range(100):
        compute_c5(h_t)
    t_pi = max((time.time() - te) / 100, 1e-6)

    # Phase 1: SA from parent
    sa_budget = min(250, budget_s * 0.25)
    n_sa = int(250 / t_pi)
    h_sa, c5_sa = sa_binary(best_h, target_base, n_sa, 0.004, 0.00003, seed, t0, t0 + sa_budget)
    if c5_sa < best_c5:
        best_h, best_c5 = h_sa, c5_sa

    # Phase 2: SA from random restarts
    for r in range(4):
        if remaining() < 60: break
        h_r = (np.random.rand(n_base) < 0.5).astype(float)
        d = int(target_base) - int(h_r.sum())
        if d > 0:
            hi = np.where(h_r == 0)[0]
            h_r[np.random.choice(hi, min(d, len(hi)), replace=False)] = 1.0
        elif d < 0:
            hi = np.where(h_r == 1)[0]
            h_r[np.random.choice(hi, min(-d, len(hi)), replace=False)] = 0.0
        h_r, c5_r = sa_binary(h_r, target_base, int(40 / t_pi), 0.003, 0.0001,
                              seed + 7 * (r + 1), t0, t0 + remaining() - 50)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Phase 3: SLSQP at base resolution
    if remaining() > 60:
        h_best, c5_best = slsqp_refine(best_h, n_base, target_base, 2000, remaining, t0)
        if c5_best < best_c5:
            best_h, best_c5 = h_best, c5_best

    # Phase 4: SLSQP at 2x resolution
    n_hi = min(n_base * 2, 560)
    if n_hi > n_base and remaining() > 80:
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_hi)
        h_interp = project_h(np.interp(x_new, x_old, best_h), n_hi / 2.0)
        h_best, c5_best = slsqp_refine(h_interp, n_hi, n_hi / 2.0, 1500, remaining, t0)
        if c5_best < best_c5:
            best_h, best_c5 = h_best, c5_best

    # Phase 5: Subgradient polishing
    if remaining() > 15:
        n_pol = len(best_h)
        tgt = n_pol / 2.0
        cur = best_h.copy()
        cur_c5 = best_c5
        momentum = np.zeros(n_pol)
        for it in range(8000):
            if remaining() < 3: break
            ss = 0.025 / (1.0 + 0.0015 * it)
            grad = compute_subgrad_vec(cur, 10)
            gn = np.linalg.norm(grad)
            if gn < 1e-12: break
            momentum = 0.8 * momentum + 0.2 * (-grad / gn)
            h_trial = project_h(cur + ss * momentum, tgt)
            c5_t = compute_c5(h_trial)
            if c5_t < cur_c5 - 1e-15:
                cur, cur_c5 = h_trial, c5_t
                if cur_c5 < best_c5:
                    best_h, best_c5 = cur.copy(), cur_c5
            else:
                h_t2 = project_h(cur + 0.3 * ss * momentum, tgt)
                c5_t2 = compute_c5(h_t2)
                if c5_t2 < cur_c5 - 1e-15:
                    cur, cur_c5 = h_t2, c5_t2
                    if cur_c5 < best_c5:
                        best_h, best_c5 = cur.copy(), cur_c5
                else:
                    momentum *= 0.5

    # Final binary rounding
    if remaining() > 2:
        n_fin = len(best_h)
        tgt_f = n_fin / 2.0
        for t in np.linspace(0.3, 0.7, 9):
            hb = (best_h >= t).astype(float)
            d = int(round(tgt_f)) - int(hb.sum())
            if d > 0:
                idx = np.where(hb == 0)[0]
                ord_i = idx[np.argsort(best_h[idx])[::-1]]
                hb[ord_i[:min(d, len(ord_i))]] = 1.0
            elif d < 0:
                idx = np.where(hb == 1)[0]
                ord_i = idx[np.argsort(best_h[idx])]
                hb[ord_i[:min(-d, len(ord_i))]] = 0.0
            if abs(hb.sum() - tgt_f) < 0.5:
                c5_b = compute_c5(hb)
                if c5_b < best_c5:
                    best_h, best_c5 = hb.copy(), c5_b
    return best_h, best_c5, int(len(best_h))
