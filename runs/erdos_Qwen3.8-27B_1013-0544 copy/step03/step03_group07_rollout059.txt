def iterated_linearization(h, max_iters, top_k):

    dx = 2.0 / n

    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c = compute_c5(h_cur)
    
    for it in range(max_iters):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        top_idx = np.argsort(corr)[-top_k:]

        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]

            c_k, grad_k = corr_and_grad_lag(h_cur, s)
            constraints.append(c_k + grad_k @ (h_var - h_cur) <= t_var)

            prob.solve(solver=cp.GLOP)  # or CBC

            try:
                prob.solve(solver=cp.CBC)

                break

c_{n-1-s} = ∑_{j=0}^{n-1-s} h[j+s] * (1-h[j])

For k ≥ n-1 (let s = k-(n-1) ≥ 0):
i ranges from 0 to 2n-2-k = n-1-s.
c_k = ∑_{i=0}^{n-1-s} h[i] * (1-h[i+s])

So to summarize, with s = k-(n-1) (the "lag"):
- For s ≥ 0: c = ∑_{i=0}^{n-1-s} h[i] * (1-h[i+s])  ... (A)
- For s ≤ 0 (let t = -s ≥ 0): c = ∑_{i=0}^{n-1-t} h[i+t] * (1-h[i])  ... (B)

Now let me compute the gradients.

For s ≥ 0: c = ∑_{i=0}^{n-1-s} h[i]*(1-h[i+s])
∂c/∂h[j] = dx * [(1-h[j+s]) * 1{j ≤ n-1-s} - h[j-s] * 1{j-s ≥ 0, j-s ≤ n-1-s, j ≤ n-1-s+s=n-1}]

The first term: j ≤ n-1-s (i.e., j is a valid i)
The second term: j-s ≥ 0 means j ≥ s, and j-s ≤ n-1-s means j ≤ n-1. So j ranges from s to n-1.

∂c/∂h[j] = dx * [(1-h[j+s]) if 0≤j≤n-1-s, else 0] - dx * [h[j-s] if s≤j≤n-1, else 0]

For s < 0, t = -s: c = ∑_{i=0}^{n-1-t} h[i+t]*(1-h[i])
∂c/∂h[j] = dx * [(1-h[j-t]) * 1{j-t ≥ 0, j-t ≤ n-1-t, j-t+t=j ≤ n-1} + (-h[j+t]) * 1{0≤j≤n-1-t}]

First term: j-t ≥ 0 means j ≥ t, and j-t ≤ n-1-t means j ≤ n-1. So j ranges from t to n-1.
Second term: j ranges from 0 to n-1-t.

∂c/∂h[j] = dx * [(1-h[j-t]) if t≤j≤n-1, else 0] - dx * [h[j+t] if 0≤j≤n-1-t, else 0]

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def corr_and_grad_lag(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m <= 0:
            return 0.0, grad
        c = np.sum(h[:m] * (1.0 - h[s:])) * dx
        grad[:m] += (1.0 - h[s:]) * dx
        grad[s:] += (-h[:m]) * dx
    else:
        t = -s
        m = n - t
        if m <= 0:
            return 0.0, grad
        c = np.sum(h[t:] * (1.0 - h[:m])) * dx
        grad[t:] += (1.0 - h[:m]) * dx
        grad[:m] += (-h[t:]) * dx
    return c, grad

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        _, g = corr_and_grad_lag(h, s)
        grad += w[k] * g
    return val, grad

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    def obj(x):
        return smoothmax_val_grad(x, beta)[0]
    def objg(x):
        return smoothmax_val_grad(x, beta)[1]
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def lp_refine(h, n_iters=100, top_k=40, time_limit=60.0):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c = compute_c5(h_cur)
    t0 = time.time()
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for k in top_idx:
            s = k - (n - 1)
            c_k, g_k = corr_and_grad_lag(h_cur, s)
            cons.append(c_k + g_k @ (h_var - h_cur) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        solved = False
        for solver in [cp.GLOP, cp.CBC, cp.CLARABLE, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if h_var.value is not None:
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        # Damped step: try full, half, quarter
        for alpha in [1.0, 0.6, 0.3]:
            h_try = project_feasible((1 - alpha) * h_cur + alpha * h_lp, target)
            c_try = compute_c5(h_try)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
                h_cur = h_try
                break
        else:
            if it > 3:
                break
    return best_h, best_c

def active_set_descent(h, max_iters, top_k, lr0, decay):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / max(t, 1)
            y = x + mu * (x - x_prev)
            y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            _, g = corr_and_grad_lag(y, s)
            grad += g
        grad /= top_k
        lr_t = lr0 * (decay ** t)
        x_new = project_feasible(y - lr_t * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)
    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: SLSQP warm-start
    for beta in [100, 300]:
        if tl() < 40:
            break
        h_new = slsqp_refine(h_best, maxiter=300, beta=beta)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 2: LP-based iterated linearization (main innovation)
    if tl() > 120:
        h_lp, c_lp = lp_refine(h_best, n_iters=80, top_k=40, time_limit=min(120, tl()-30))
        if c_lp < best_c5:
            h_best, best_c5 = h_lp, c_lp
        # Second round with more iterations if improved
        if c_lp < best_c5 + 1e-6 and tl() > 100:
            h_lp2, c_lp2 = lp_refine(h_best, n_iters=60, top_k=50, time_limit=min(100, tl()-30))
            if c_lp2 < best_c5:
                h_best, best_c5 = h_lp2, c_lp2

    # Phase 3: Active-set subgradient
    for top_k in [10, 20, 30]:
        if tl() < 80:
            break
        h_sg, c_sg = active_set_descent(h_best, 4000, top_k, 0.025, 0.99985)
        if c_sg < best_c5:
            h_best, best_c5 = h_sg, c_sg

    # Phase 4: SLSQP high-beta polish
    for beta in [500, 1200]:
        if tl() < 35:
            break
        h_new = slsqp_refine(h_best, maxiter=400, beta=beta)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 5: Another LP round after SLSQP polish
    if tl() > 80:
        h_lp3, c_lp3 = lp_refine(h_best, n_iters=60, top_k=40, time_limit=min(80, tl()-20))
        if c_lp3 < best_c5:
            h_best, best_c5 = h_lp3, c_lp3

    # Phase 6: Shake and re-optimize
    for trial in range(6):
        if tl() < 50:
            break
        scale = 0.008 * (1.0 + 0.3 * trial)
        pert = np.random.randn(n) * scale
        h_pert = project_feasible(h_best + pert, n / 2.0)
        h_new = slsqp_refine(h_pert, maxiter=300, beta=600)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c
            if tl() > 40:
                h_lp4, c_lp4 = lp_refine(h_best, n_iters=40, top_k=30, time_limit=30)
                if c_lp4 < best_c5:
                    h_best, best_c5 = h_lp4, c_lp4

    # Phase 7: Grid refinement
    for n_try in [600, 800]:
        if tl() < 180:
            break
        h_interp = interp_to_n(h_best, n_try)
        for beta in [200, 600]:
            if tl() < 60:
                break
            h_new = slsqp_refine(h_interp, maxiter=250, beta=beta)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
                h_interp = h_best.copy()
        n = len(h_best)
        if tl() > 80:
            h_lp5, c5 = lp_refine(h_best, n_iters=50, top_k=35, time_limit=min(80, tl()-30))
            if c5 < best_c5:
                h_best, best_c5 = h_lp5, c5

    # Phase 8: Final polish
    if tl() > 30:
        h_final = slsqp_refine(h_best, maxiter=400, beta=2000)
        c = compute_c5(h_final)
        if c < best_c5:
            h_best, best_c5 = h_final, c
    if tl() > 20:
        h_fin2, c_fin2 = active_set_descent(h_best, 2000, 15, 0.01, 0.99995)
        if c_fin2 < best_c5:
            h_best, best_c5 = h_fin2, c_fin2

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
