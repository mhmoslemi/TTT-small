import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def weighted_top_subgrad(h, corr, n, k_top, temperature):
    top_idx = np.argsort(corr)[-k_top:]
    w = np.exp(temperature * (corr[top_idx] - corr[top_idx].max()))
    w /= w.sum()
    grad = np.zeros(n)
    for j, k in enumerate(top_idx):
        if w[j] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                idx = np.arange(m)
                grad[idx] += w[j] * (1.0 - h[idx + lag])
                grad[idx + lag] -= w[j] * h[idx]
        else:
            la = -lag
            m = n - la
            if m > 0:
                idx = np.arange(m)
                grad[idx + la] += w[j] * (1.0 - h[idx])
                grad[idx] -= w[j] * h[idx + la]
    return grad

def cons_fn(h, ts):
    return h.sum() - ts

def cons_jac(h, ts):
    return np.ones(len(h))

def slsqp_optimize(h, n_pts, beta, maxiter=6000):
    dx_t = 2.0 / n_pts
    ts_t = n_pts / 2.0
    bnds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    cons = [{'type': 'eq', 'fun': cons_fn, 'jac': cons_jac, 'args': (ts_t,)}]
    try:
        res = minimize(obj_smooth, h, args=(dx_t, beta), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, ts_t)
    except Exception:
        return h

def subgrad_phase(h, budget_fn, top_k=5, temp=12.0, lr_start=0.005):
    n_pts = len(h)
    ts = n_pts / 2.0
    h = h.copy()
    velocity = np.zeros(n_pts)
    beta_m = 0.82
    best_local = compute_c5(h)
    stall = 0
    for it in range(50000):
        if budget_fn() < 30:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        grad = weighted_top_subgrad(h, corr, n_pts, top_k, temp)
        lr = lr_start / (1.0 + 0.00015 * it)
        velocity = beta_m * velocity + (1 - beta_m) * (-lr * grad)
        h_new = project_feasible(h + velocity, ts)
        c_new = compute_c5(h_new)
        if c_new < best_local + 0.002:
            h = h_new
            if c_new < best_local:
                best_local = c_new
                stall = 0
            else:
                stall += 1
        else:
            velocity *= 0.4
            stall += 1
        if stall > 3000:
            break
    return h, best_local

def upsample(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return project_feasible(h_new, n_new / 2.0)
