import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_grad(h, beta, dx, topk):
    corr = compute_corr(h)
    n = len(h)
    k = min(topk, 2 * n - 1)
    top_idx = np.argpartition(corr, -k)[-k:]
    top_corr = corr[top_idx]
    m = np.max(top_corr)
    shifted = np.clip(beta * (top_corr - m), -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = (m + np.log(sum_exp / k) / beta) * dx
    w = exp_vals / sum_exp
    grad = np.zeros(n)
    for ii in range(k):
        if w[ii] < 1e-8:
            continue
        lag = int(top_idx[ii]) - (n - 1)
        if lag >= 0:
            end = n - 1 - lag
            if end >= 0:
                grad[:end+1] += w[ii] * (1.0 - h[lag:lag+end+1])
                grad[lag:] -= w[ii] * h[:end+1]
        else:
            a = -lag
            end = n - 1 - a
            if end >= 0:
                grad[a:] += w[ii] * (1.0 - h[:end+1])
                grad[:end+1] -= w[ii] * h[a:a+end+1]
    grad *= dx
    return obj, grad, corr

def momentum_grad(h0, target, beta, dx, topk, n_iter, t0, budget, step_init):
    n = len(h0)
    h = h0.copy()
    h_prev = h.copy()
    velocity = np.zeros(n)
    step = step_init
    best_c5 = compute_c5(h)
    momentum = 0.7
    stall = 0

    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        obj, grad, _ = lse_obj_grad(h, beta, dx, topk)
        velocity = momentum * velocity - step * grad
        x_new = project_h(h + velocity, target)
        new_c5 = compute_c5(x_new)
        if new_c5 < best_c5 - 1e-15:
            h_prev = h.copy()
            h = x_new
            best_c5 = new_c5
            stall = 0
        else:
            stall += 1
            if stall > 50:
                step *= 0.8
                velocity *= 0.3
                stall = 0
            x_try = project_h(h + 0.3 * velocity, target)
            c_try = compute_c5(x_try)
            if c_try < best_c5 - 1e-15:
                h = x_try
                best_c5 = c_try
                stall = 0
            if stall > 200:
                break
    return h, best_c5

def coord_desc(h, target, n_sweeps, t0, budget):
    n = len(h)
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    for sw in range(n_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        order = np.random.permutation(n)
        for j in order:
            if budget - (time.time() - t0) < 1:
                break
            found = False
            for sign in [1, -1]:
                for mag in [0.03, 0.015, 0.008, 0.004]:
                    delta = sign * mag
                    hv = h[j] + delta
                    if hv < -0.01 or hv > 1.01:
                        continue
                    trial = h.copy()
                    trial[j] = hv
                    trial = project_h(trial, target)
                    c_try = compute_c5(trial)
                    if c_try < best_c5 - 1e-15:
                        h = trial
                        best_c5 = c_try
                        improved = True
                        found = True
                        break
                if found:
                    break
        if not improved and sw > 2:
            break
    return h, best_c5

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new),
                     np.linspace(0.0, 2.0, len(h)), h)

def block_perturb(h, target, n_blocks, block_size, scale):
    n = len(h)
    hp = h.copy()
    for _ in range(n_blocks):
        start = np.random.randint(0, max(1, n - block_size))
        end = min(start + block_size, n)
        shift = np.random.uniform(-scale, scale, end - start)
        hp[start:end] += shift
    return project_h(hp, target)

    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    best_h = h_parent.copy()

    for n_pts in resolutions:

            break
        target = n_pts / 2.0
        dx = 2.0 / n_pts

        cur = project_h(resample_h(best_h, n_pts), target)

        for beta in [12, 25, 50, 100, 200, 400]:

            topk = min(12, 2 * n_pts - 1)

            n_iter = max(200, min(2000, int(remaining * 40)))
            cur, cur_c5 = momentum_grad(cur, target, beta, dx, topk,
                                         n_iter, t0, budget_s, step_init)

        # Coordinate descent polish
        remaining = budget_s - (time.time() - t0)
        if remaining > 10:
            cur, cur_c5 = coord_desc(cur, target, 25, t0, budget_s)

        if cur_c5 < best_c5:
            best_h, best_c5 = cur.copy(), cur_c5

        # Block perturbation to find new basin
        remaining = budget_s - (time.time() - t0)
        if remaining > 30:
            for bp in range(3):
                remaining = budget_s - (time.time() - t0)
                if remaining < 12:
                    break
                scale = 0.03 / (bp + 1)

                hp, hp_c5 = coord_desc(hp, target, 15, t0, budget_s)
                if hp_c5 < cur_c5:
                    cur, cur_c5 = hp, hp_c5
                if cur_c5 < best_c5:
                    best_h, best_c5 = cur.copy(), cur_c5

    # Final high-precision polish
    remaining = budget_s - (time.time() - t0)

        nf = len(best_h)
        tf = nf / 2.0
        dxf = 2.0 / nf
        for beta in [500, 1000, 2000]:
            remaining = budget_s - (time.time() - t0)

    nf = len(best_h)
    best_h = project_h(best_h, nf / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, nf
