import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_grad(h, lag, n):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < lag < n:
        g[lag:] += 1.0 - h[:n - lag]
        g[:n - lag] -= h[lag:]
    elif -n < lag < 0:
        l = -lag
        g[:n - l] += 1.0 - h[l:]
        g[l:] -= h[:n - l]
    return g

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def active_grad(h, top_idx, weights, n, dx):
    grad = np.zeros(n)
    center = n - 1
    for w, idx in zip(weights, top_idx):
        if w < 1e-12:
            continue
        lag = int(idx) - center
        grad += w * lag_grad(h, lag, n)
    return grad * dx

def get_top_weights(corr_val, top_k, beta_temp):
    idx = np.argsort(corr_val)[::-1][:top_k]
    vals = corr_val[idx]
    w = np.exp((vals - vals[0]) * beta_temp)
    w /= w.sum()
    return idx, w

def lbfgs_smooth(h, n, target, dx, beta, maxiter=1000):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(c, beta) * dx
    def objg(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(c)
        xs = np.clip(beta * (c - m), -500, 500)
        w = np.exp(xs)
        w /= w.sum()
        # Only compute for significant weights
        mask = w > 1e-8
        idx = np.where(mask)[0]
        wsub = w[idx]
        return active_grad(x, idx, wsub, n, dx)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    bnds = Bounds(np.zeros(n), np.ones(n))
    return minimize(obj, h, jac=objg, method='L-BFGS-B', bounds=bnds,
                    constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})

def active_descent(h, n, target, dx, max_iters, beta_temp, step0, time_fn, min_time):
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        corr_raw = np.correlate(h, 1.0 - h, mode="full")
        corr_val = corr_raw * dx
        top_k = min(5, n // 40 + 2)
        idx, w = get_top_weights(corr_val, top_k, beta_temp)
        grad = active_grad(h, idx, w, n, dx)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        step = step0 / (1.0 + 0.001 * it)
        improved = False
        for _ in range(30):
            ht = project_h(h - step * grad, target)
            c5t = compute_c5(ht)
            if c5t < compute_c5(h) - 1e-15:
                h = ht
                improved = True
                break
            step *= 0.5
        if not improved:
            T = 0.002 * np.exp(-it / 1500.0)
            if T > 1e-6:
                h = project_h(h + np.random.normal(0, T, n), target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    # Get parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(250, 0.5)
    n0 = len(h0)
    target0 = n0 / 2.0
    dx0 = 2.0 / n0
    h0 = project_h(h0, target0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B polish at native resolution
    for beta in [80, 150, 300, 600, 1200]:
        if tl() < 40:
            break
        res = lbfgs_smooth(best_h, n0, target0, dx0, beta, maxiter=1500)
        h_new = project_h(res.x, target0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Active-set projected gradient (long run)
    best_h = active_descent(best_h, n0, target0, dx0,
                           max_iters=8000, beta_temp=300.0, step0=0.04,
                           time_fn=tl, min_time=30)
    best_c5 = compute_c5(best_h)

    # Phase 3: Try higher resolution n=300
    if tl() > 120:
        nf = 300
        tf = nf / 2.0
        dxf = 2.0 / nf
        hf = project_h(resample(best_h, nf), tf)
        cf = compute_c5(hf)
        for beta in [150, 400, 800]:
            if tl() < 50:
                break
            res = lbfgs_smooth(hf, nf, tf, dxf, beta, maxiter=1500)
            h_new = project_h(res.x, tf)
            cn = compute_c5(h_new)
            if cn < cf:
                hf, cf = h_new, cn
        hf = active_descent(hf, nf, tf, dxf,
                           max_iters=5000, beta_temp=250.0, step0=0.03,
                           time_fn=tl, min_time=20)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_h, best_c5 = hf, cf

    # Phase 4: Try n=200 (sometimes lower res is better)
    if tl() > 80:
        nb = 200
        tab = nb / 2.0
        dxb = 2.0 / nb
        hb = project_h(resample(best_h, nb), tab)
        cb = compute_c5(hb)
        for beta in [200, 500]:
            if tl() < 40:
                break
            res = lbfgs_smooth(hb, nb, tab, dxb, beta, maxiter=1000)
            h_new = project_h(res.x, tab)
            cn = compute_c5(h_new)
            if cn < cb:
                hb, cb = h_new, cn
        hb = active_descent(hb, nb, tab, dxb,
                           max_iters=4000, beta_temp=200.0, step0=0.03,
                           time_fn=tl, min_time=15)
        cb = compute_c5(hb)
        if cb < best_c5:
            best_h, best_c5 = hb, cb

    # Phase 5: Ultra-fine final polish
    nf = len(best_h)
    tf = nf / 2.0
    dxf = 2.0 / nf
    for it in range(50000):
        if tl() < 3:
            break
        corr_raw = np.correlate(best_h, 1.0 - best_h, mode="full")
        corr_val = corr_raw * dxf
        idx, w = get_top_weights(corr_val, 3, 800.0)
        grad = active_grad(best_h, idx, w, nf, dxf)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        for step in [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]:
            ht = project_h(best_h - step * grad, tf)
            ct = compute_c5(ht)
            if ct < best_c5 - 1e-16:
                best_h, best_c5 = ht, ct
                break

    return best_h, best_c5, len(best_h)
