        for _ in range(50):  # Try 50 random swaps per iteration
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            if i == j:
                continue
            # Swap
            h_binary[i] = 0
            h_binary[j] = 1
            c_new = compute_c5(h_binary)
            if c_new < best_c5:

import numpy as np
from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:] += (1.0 - h[:m])
            grad[:m] -= h[lag:]
    else:
        la = -lag
        m = n - la
        if m > 0:
            grad[:m] += (1.0 - h[la:])
            grad[la:] -= h[:m]
    return grad

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum, n):
    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)
