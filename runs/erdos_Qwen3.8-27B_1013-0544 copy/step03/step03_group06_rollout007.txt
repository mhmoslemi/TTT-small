from scipy.optimize import minimize_scalar, minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    """Optimize coordinate j by minimizing smooth_max of correlations."""
    n = len(h)
    dx = 2.0 / n
    h_orig = h.copy()

        h_trial = h_orig.copy()
        h_trial[j] = t

        return smooth_max(corr, beta) * dx

    """Generate Thue-Morse-like balanced sequence."""
    tm = np.zeros(n)
    tm[0] = 1
    for i in range(1, n):
        # Count trailing 1s in (i-1), flip
        v = i - 1
        cnt = 0
        while v & 1:
            cnt += 1
            v >>= 1
        tm[i] = 1.0 if (cnt % 2 == 0) else 0.0
    return tm

import numpy as np
from scipy.optimize import minimize_scalar, minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _eval_coord_t(t, h_arr, j, beta):
    h_trial = h_arr.copy()
    h_trial[j] = t
    corr = np.correlate(h_trial, 1.0 - h_trial, mode="full")
    return smooth_max(corr, beta)

def _eval_pair_t(t, h_arr, i, j, beta, ps):
    h_trial = h_arr.copy()
    h_trial[i] = t
    vj = ps - t
    if vj < -1e-10 or vj > 1.0 + 1e-10:
        return 1e10
    h_trial[j] = vj
    corr = np.correlate(h_trial, 1.0 - h_trial, mode="full")
    return smooth_max(corr, beta)

def _make_thue_morse(n):
    tm = np.zeros(n)
    tm[0] = 1.0
    for i in range(1, n):
        v = i - 1
        cnt = 0
        while v & 1:
            cnt += 1
            v >>= 1
        tm[i] = 1.0 if (cnt % 2 == 0) else 0.0
    return tm

def _cd_sweep(h, target_sum, beta, use_pairs, rng):
    n = len(h)
    h = h.copy()
    for j in range(n):
        res = minimize_scalar(_eval_coord_t, args=(h, j, beta),

                             options={'xatol': 1e-9})
        h[j] = res.x
    if use_pairs:
        for _ in range(n):
            i = rng.randint(0, n)
            j2 = rng.randint(0, n)
            if i == j2:
                continue
            ps = h[i] + h[j2]
            lo = max(0.0, ps - 1.0)
            hi = min(1.0, ps)
            if hi - lo < 1e-6:
                continue
            res = minimize_scalar(_eval_pair_t, args=(h, i, j2, beta, ps),
                                 bounds=(lo, hi), method='bounded',
                                 options={'xatol': 1e-9})
            h[i] = res.x
            h[j2] = ps - res.x
        for _ in range(n // 3):
            i = rng.randint(0, n - 1)
            j2 = i + 1
            ps = h[i] + h[j2]
            lo = max(0.0, ps - 1.0)
            hi = min(1.0, ps)
            if hi - lo < 1e-6:
                continue
            res = minimize_scalar(_eval_pair_t, args=(h, i, j2, beta, ps),
                                 bounds=(lo, hi), method='bounded',
                                 options={'xatol': 1e-9})
            h[i] = res.x
            h[j2] = ps - res.x
    return h

        c5 = compute_c5(h)
        if c5 < best_c5:
