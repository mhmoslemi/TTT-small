def smoothmax_obj_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = c_sel.max()
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    
    grad = np.zeros(n)

        wk = w_sel[i]
        if wk < 1e-14:

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def smoothmax_vals(h, beta, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = c_sel.max()
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * (2.0 / n)
    return val, w, top_idx, corr

def smoothmax_grad(h, beta, top_k):

    lr = lr_init
    for _ in range(max_iter):
        if budget - (time.time() - t0) < 10:
            break
        val, g = sm_grad(h, beta, top_k)
        step = lr
        improved = False
        for _ in range(10):
            h_try = project_feasible(h - step * g, target)
            v_try, _ = sm_grad(h_try, beta, top_k)
            if v_try < val * (1.0 - 1e-10):
                h = h_try
                lr = min(step * 1.2, lr_init * 3.0)
                improved = True
                break
            step *= 0.5
        if improved:
            cv = compute_c5(h)
            if cv < best_v:
                best_v = cv
                best_h = h.copy()
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    return best_h, best_v


def interp_to_n(h, n_new):
    n_old = len(h)
    xo = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(xo, h, kind='cubic', fill_value='extrapolate')
    hnew = np.clip(f(xn), 0.0, 1.0)
    return project_feasible(hnew, n_new / 2.0)


def basin_hop_round(h_best, best_c5, target, n_hops, n, rng, t0, budget, noise_start, phase_label):
    for hop in range(n_hops):
        if budget - (time.time() - t0) < 50:
            break
        noise = noise_start * (0.9 ** min(hop, 25))
        mod = hop % 5
        if mod == 1:
            h_p = h_best.copy()
            bs = rng.randint(3, max(4, n // 10))
            st = rng.randint(0, n - bs)
            h_p[st:st + bs] = 1.0 - h_p[st:st + bs]
            h_p = project_feasible(h_p, target)
        elif mod == 2:
            fr = rng.uniform(0.5, 10.0)
            ph = rng.rand() * 6.2832
            h_p = h_best + noise * 2.0 * np.sin(2.0 * np.pi * fr * np.arange(n) / n + ph)
            h_p = project_feasible(h_p, target)
        elif mod == 3:
            h_p = h_best.copy()
            n_blocks = rng.randint(2, 5)
            for _ in range(n_blocks):
                bs = rng.randint(2, max(3, n // 8))
                st = rng.randint(0, n - bs)
                h_p[st:st + bs] += noise * rng.randn(bs)
            h_p = project_feasible(h_p, target)
        else:
            h_p = project_feasible(h_best + rng.randn(n) * noise, target)

        betas_phase = [(100, 30), (300, 15)] if phase_label == "mid" else [(200, 20), (500, 10)]
        for beta, tk in betas_phase:
            if budget - (time.time() - t0) < 25:
                break
            lr = 3.0 / max(1.0, beta / 50.0)
            h_p, _ = pgd(h_p, target, beta, tk, lr, 250, t0, budget)
        c = compute_c5(h_p)
        if c < best_c5:
            best_c5 = c
            h_best = h_p.copy()
    return h_best, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(200, 0.5)

    n = len(h)
    target = n / 2.0
    best_c5 = compute_c5(h)
    h_best = h.copy()

    # Phase 1: PGD continuation with increasing beta
    for beta, tk in [(15, None), (30, None), (60, 40), (120, 25), (250, 15), (400, 12)]:
        if budget_s - (time.time() - t0) < 70:
            break
        lr = 3.0 / max(1.0, beta / 50.0)
        h, _ = pgd(h, target, beta, tk, lr, 600, t0, budget_s)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            h_best = h.copy()

    # Phase 2: Basin hopping - broad exploration
    h_best, best_c5 = basin_hop_round(
        h_best, best_c5, target, 35, n, rng, t0, budget_s,
        noise_start=0.02, phase_label="mid")

    # Phase 3: Multi-resolution refinement
    for n_try in [300, 400, 250, 350, 500]:
        if budget_s - (time.time() - t0) < 130:
            break
        hi = interp_to_n(h_best, n_try)
        ti = n_try / 2.0
        for beta, tk in [(30, None), (100, 30), (300, 15), (500, 10)]:
            if budget_s - (time.time() - t0) < 35:
                break
            lr = 3.0 / max(1.0, beta / 50.0)
            hi, _ = pgd(hi, ti, beta, tk, lr, 350, t0, budget_s)
        c = compute_c5(hi)
        if c < best_c5:
            best_c5 = c
            h_best = hi.copy()

    # Phase 4: Fine basin hopping on best solution found
    n = len(h_best)
    target = n / 2.0
    h_best, best_c5 = basin_hop_round(
        h_best, best_c5, target, 20, n, rng, t0, budget_s,
        noise_start=0.008, phase_label="fine")

    # Phase 5: Independent restarts from different initializations
    if budget_s - (time.time() - t0) > 150:
        for restart in range(3):
            if budget_s - (time.time() - t0) < 60:
                break
            if restart == 0:
                h_r = h_best + rng.randn(n) * 0.05
            elif restart == 1:
                fr = rng.uniform(1, 6)
                h_r = 0.5 + 0.3 * np.sin(2 * np.pi * fr * np.arange(n) / n + rng.rand() * 6.28)
            else:
                h_r = np.zeros(n)
                n_high = n // 2
                idx = rng.choice(n, n_high, replace=False)
                h_r[idx] = 1.0
            h_r = project_feasible(h_r, target)
            for beta, tk in [(20, None), (80, 30), (300, 15), (600, 10)]:
                if budget_s - (time.time() - t0) < 25:
                    break
                lr = 3.0 / max(1.0, beta / 50.0)
                h_r, _ = pgd(h_r, target, beta, tk, lr, 400, t0, budget_s)
            c = compute_c5(h_r)
            if c < best_c5:
                best_c5 = c
                h_best = h_r.copy()

    # Phase 6: Final high-beta polish
    if budget_s - (time.time() - t0) > 30:
        lr = 2.0 / max(1.0, 800.0 / 50.0)
        h_fin, _ = pgd(h_best, target, 800, 8, lr, 500, t0, budget_s)
        c = compute_c5(h_fin)
        if c < best_c5:
            best_c5 = c
            h_best = h_fin.copy()

    # Final feasibility check
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
