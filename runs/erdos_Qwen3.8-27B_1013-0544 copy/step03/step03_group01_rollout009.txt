def project_feasible(h, n):

    # LSE value (numerically stable)
    c_max = c_vec.max()
    shifted = alpha * (c_vec - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum = c_max + np.log(exp_shifted.sum()) / alpha
    val = log_sum + mu * (h.sum() - n/2.0)**2
    
    # Weights
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient of LSE part
    n_lags = 2 * n - 1
    lags = np.arange(n_lags) - (n - 1)
    
    grad_lse = np.zeros(n)
    for j in range(n):
        # sum_k w_k * [h[j+d_k] * 1_{valid} + h[j-d_k] * 1_{valid}]
        val_j = 0.0
        for k in range(n_lags):
            d = lags[k]
            if w[k] < 1e-10:
                continue
            if 0 <= j + d < n:
                val_j += w[k] * h[j + d]
            if 0 <= j - d < n:
                val_j += w[k] * h[j - d]
        grad_lse[j] = -dx * val_j
    
    # Penalty gradient
    grad_pen = 2.0 * mu * (h.sum() - n/2.0) * np.ones(n)
    
    return val, grad_lse + grad_pen

    grad = np.zeros(n)

        idx1 = (j + lags)  # potential indices into h

        mask2 = (idx2 >= 0) & (idx2 < n)
        if np.any(mask2):
            grad[j] += -dx * np.sum(w[mask2] * h[idx2[mask2]])
    return grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13 * n:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h -= (-diff / s) * h
    return np.clip(h, 0.0, 1.0)

    val += mu * (h.sum() - n / 2.0) ** 2

    alphas = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]
