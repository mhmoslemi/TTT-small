import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def cd_optimize(h, n, max_sweeps=300, deadline=None):
    h = h.copy()
    nx = 2 * n - 1
    lags = np.arange(-(n - 1), n)
    prev_c = compute_c5(h, n)
    
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        
        for i in range(n):
            idx_lo = i - lags
            idx_hi = i + lags
            mask_lo = (idx_lo >= 0) & (idx_lo < n)
            mask_hi = (idx_hi >= 0) & (idx_hi < n)
            h_lo = np.zeros(nx)
            h_hi = np.zeros(nx)
            h_lo[mask_lo] = h[idx_lo[mask_lo]]
            h_hi[mask_hi] = h[idx_hi[mask_hi]]
            a = 1.0 - h_lo - h_hi
            
            pos = a > 1e-14
            neg = a < -1e-14
            zero = ~pos & ~neg
            
            t_lo = 0.0
            t_hi = float(np.max(c) + np.max(np.abs(a)) + 1.0)
            
            for _ in range(45):
                t_mid = (t_lo + t_hi) * 0.5
                if np.any(zero) and float(np.max(c[zero])) > t_mid + 1e-14:
                    t_lo = t_mid
                    continue
                L = 0.0
                U = 1.0
                if np.any(neg):
                    L = max(L, float(np.max((t_mid - c[neg]) / a[neg] + h[i])))
                if np.any(pos):
                    U = min(U, float(np.min((t_mid - c[pos]) / a[pos] + h[i])))
                if L <= U + 1e-14:
                    t_hi = t_mid
                else:
                    t_lo = t_mid
            
            t_opt = t_hi
            L = 0.0
            U = 1.0

            L = max(L, 0.0)
            U = min(U, 1.0)

        c_new = compute_c5(h, n)
        if c_new >= prev_c - 1e-13:
            if sweep > 5:
                break
        prev_c = c_new
    
    return h, compute_c5(h, n)

def slsqqpolish(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        s = alpha * (c - m)
        s = s[s > -50]
        return float(m + np.log(np.sum(np.exp(s))) / alpha)
    
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    
    def sum_con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def optimize_n(h_start, n, deadline, rng):

    best_h = h.copy()
    best_c = compute_c5(best_h, n)

        if c < best_c:

            best_h = h

            best_h = ht

    for r in range(n_restarts):

        hp = best_h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        h, c = cd_optimize(hp, n, max_sweeps=100, deadline=deadline - 5)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()

        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13 * n:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cd_optimize(h, n, max_sweeps=300, deadline=None):
    h = h.copy()
    nx = 2 * n - 1
    lags = np.arange(-(n - 1), n)
    prev_c = compute_c5(h, n)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        for i in range(n):
            idx_lo = i - lags
            idx_hi = i + lags
            mlo = (idx_lo >= 0) & (idx_lo < n)
            mhi = (idx_hi >= 0) & (idx_hi < n)
            h_lo = np.where(mlo, np.take(h, np.where(mlo, idx_lo, 0)), 0.0)
            h_hi = np.where(mhi, np.take(h, np.where(mhi, idx_hi, 0)), 0.0)
            a = 1.0 - h_lo - h_hi
            pos = a > 1e-14
            neg = a < -1e-14
            zero = ~pos & ~neg
            t_lo = 0.0
            t_hi = float(np.max(c) + np.max(np.abs(a)) + 1.0)
            for _ in range(45):
                t_mid = (t_lo + t_hi) * 0.5
                if np.any(zero) and float(np.max(c[zero])) > t_mid + 1e-14:
                    t_lo = t_mid
                    continue
                L = 0.0
                U = 1.0
                if np.any(neg):
                    L = max(L, float(np.max((t_mid - c[neg]) / a[neg] + h[i])))
                if np.any(pos):
                    U = min(U, float(np.min((t_mid - c[pos]) / a[pos] + h[i])))
                if L <= U + 1e-14:
                    t_hi = t_mid
                else:
                    t_lo = t_mid
            t_opt = t_hi
            L = 0.0
            U = 1.0
            if np.any(neg):
                L = max(L, float(np.max((t_opt - c[neg]) / a[neg] + h[i])))
            if np.any(pos):
                U = min(U, float(np.min((t_opt - c[pos]) / a[pos] + h[i])))
            L = max(L, 0.0)
            U = min(U, 1.0)
            if L <= U + 1e-14:
                h[i] = (L + U) * 0.5
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new >= prev_c - 1e-13:
            if sweep > 3:
                break
        prev_c = c_new
    return h, compute_c5(h, n)

def slsqp_polish(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        s = alpha * (c - m)
        s = s[s > -50]
        return float(m + np.log(np.sum(np.exp(s))) / alpha)
    def scon(x):
        return float(np.sum(x) - n / 2.0)
    def scon_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': scon, 'jac': scon_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_refine(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                for s in (1.0, -1.0):
                    hi = h[i] + s * d
                    hj = h[j] - s * d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c
                        else:
                            h[i], h[j] = hi - s * d, hj + s * d
    return h, best_c
