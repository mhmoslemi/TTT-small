```python
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_at_lag(h, lag):
    """Compute correlation at a specific lag."""
    n = len(h)
    if lag == 0:
        return np.dot(h, 1.0 - h)
    elif lag > 0:
        return np.dot(h[lag:], 1.0 - h[:n - lag])
    else:
        la = -lag
        return np.dot(h[:n - la], 1.0 - h[la:])

def corr_grad_lag(h, lag):
    """Gradient of correlation at specific lag."""
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad = 1.0 - 2.0 * h
    else:
        la = abs(lag)
        grad[la:] += (1.0 - h[:n - la])
        grad[:n - la] += -h[la:]
    return grad

def sml_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(c)
    x_shift = np.clip(beta * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(x_shift)) / len(c)) / beta) * dx

def sml_grad(h_in, beta, dx):
    n = len(h_in)
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(c)
    x_shift = np.clip(beta * (c - m), -500, 500)
    exp_shift = np.exp(x_shift)
    w = exp_shift / np.sum(exp_shift)
    # Only use top 30 for efficiency
    top_idx = np.argsort(w)[::-1][:30]
    top_w = w[top_idx]
    top_w = top_w / top_w.sum()
    grad = np.zeros(n)
    for i, idx in enumerate(top_idx):
        lag = int(idx) - (n - 1)
        la = abs(lag)
        if la == 0:
            grad += top_w[i] * (1.0 - 2.0 * h_in)
        else:
            grad[la:] += top_w[i] * (1.0 - h_in[:n - la])
            grad[:n - la] += top_w[i] * (-h_in[la:])
    return grad * dx

    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new, endpoint=False),
                     np.linspace(0.0, 2.0, len(h), endpoint=False), h)

    return np.sum(x) - target

def sum_eq_jac(x, target):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, jac=lambda x: sml_grad(x, beta, 2.0/n),
                     args=(beta, 2.0/n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def do_trust_constr(h0, n, target, beta, maxiter):

    bounds = Bounds(np.zeros(n), np.ones(n))
    lin = LinearConstraint(np.ones((1, n)), lb=target, ub=target)

def nesterov_subgrad(h, target, n_iter, step0, t0, budget, top_k=5):
    n = len(h)
    dx = 2.0 / n
    best = compute_c5(h)
    v = np.zeros(n)
    y = h.copy()
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        alpha = 2.0 / (it + 2)
        # Compute subgradient at y
        c = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(c)[::-1][:top_k]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            la = abs(lag)
            if la == 0:
                g += (1.0 - 2.0 * y)
            else:
                g[la:] += (1.0 - y[:n - la])
                g[:n - la] += -y[la:]
        g *= dx / top_k
        
        # Nesterov step

        trial = project_h(y - v, target)

            h = trial
            y = trial
            best = c_try
        else:
            y = project_h(y - 0.5 * v, target)
            if compute_c5(y) < best:
                h = y
                best = compute_c5(h)
    return h, best

def coord_descent(h, target, max_sweeps, t0, budget):
```

Here's another idea: **analogue of Frank-Wolfe (conditional gradient) method**. 

The problem can be written as:
minimize f(h) = max_k corr[k] * dx
subject to h ∈ [0,1]^n, sum(h) = n/2

The Frank-Wolfe method alternates between:
1. Linear oracle: find s = argmin_{h in simplex} <g, h> where g is a subgradient
2. Step: h_new = (1-t)*h + t*s

The simplex here is {h: 0 <= h <= 1, sum(h) = n/2}. The linear oracle on this set is: sort the gradient components, set s[i] = 1 for the smallest n/2 components, 0 for the rest. (If n/2 is not integer, one component gets a fractional value, but here n is even so it's fine.)

This is interesting because the "linear oracle" step is exact and easy. Let me add this as another method.

```python
def frank_wolfe(h, target, n_iter, t0, budget):
    n = len(h)
    dx = 2.0 / n
    best = compute_c5(h)
    for it in range(n_iter):
        if budget - (time.time() - t0) < 1:
            break
        # Subgradient
        c = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(c)[::-1][:5]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            la = abs(lag)
            if la == 0:
                g += (1.0 - 2.0 * h)
            else:
                g[la:] += (1.0 - h[:n - la])
                g[:n - la] += -h[la:]
        g *= dx / len(top_idx)
        
        # Linear oracle: s[i] = 1 for smallest g[i], 0 for largest
        # But we also need sum(s) = target
        order = np.argsort(g)
        s = np.zeros(n)
        s[order[:int(target)]] = 1.0
        if target != int(target):
            s[order[int(target)]] = target - int(target)
        
        # Step size: backtracking
        t_opt = 2.0 / (it + 2)  # or use backtracking
        trial = (1 - t_opt) * h + t_opt * s
        trial = project_h(trial, target)
        c_try = compute_c5(trial)
        if c_try < best:
            h = trial
            best = c_try
    return h, best
```

```python
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        h0 = np.array(globals()['initial_h_values'], dtype=float).copy()
    else:
        h0 = np.full(320, 0.5)

    n_base = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    resolutions = sorted(set([n_base, 256, 288, 320, 352, 384]))

    for n_pts in resolutions:
        remaining = budget_s - (time.time() - t0)
        if remaining < 50:
            break
        target = n_pts / 2.0
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
        cur = project_h(resample(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5 = cur.copy(), cur_c5

        # SLSQP with increasing sharpness
        for beta in [80, 200, 500, 1200]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 15:
                break
            cur = do_slsqp(cur, n_pts, target, beta, bounds, 5000, 1e-16)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5

        # Trust-constr
        if budget_s - (time.time() - t0) > 25:
            for beta in [400, 1200]:
                if budget_s - (time.time() - t0) < 10:
                    break
                cur = do_trust_constr(cur, n_pts, target, beta, 3000)
                cur_c5 = compute_c5(cur)
                if cur_c5 < best_c5:
                    best_h, best_c5 = cur.copy(), cur_c5

        # Frank-Wolfe (conditional gradient)
        if budget_s - (time.time() - t0) > 15:
            cur, cur_c5 = frank_wolfe(cur, target, 5000, t0, budget_s, top_k=5)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5

        # Nesterov subgradient
        if budget_s - (time.time() - t0) > 15:
            cur, cur_c5 = subgrad_nesterov(cur, target, 3000, 0.02, t0, budget_s, top_k=5)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5

        # Coordinate descent
        if budget_s - (time.time() - t0) > 10:
            cur, cur_c5 = coord_descent(cur, target, 25, t0, budget_s)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5

        # Multi-start perturbations
        remaining = budget_s - (time.time() - t0)
        if remaining > 40:
            n_restarts = min(6, int(remaining / 15))
            for att in range(n_restarts):
                remaining = budget_s - (time.time() - t0)
                if remaining < 10:
                    break
                sigma = 0.10 / (att + 1)
                hp = project_h(cur + np.random.randn(n_pts) * sigma, target)
                hp = do_slsqp(hp, n_pts, target, 300, bounds, 2000, 1e-15)
                hp = frank_wolfe(hp, target, 2000, t0, budget_s, top_k=5)[0]
                hp, hp_c5 = coord_descent(hp, target, 8, t0, budget_s)
                if hp_c5 < best_c5:
                    best_h, best_c5 = hp.copy(), hp_c5
                    cur = hp

    # Final polish at best resolution
    nf = len(best_h)
    tf = nf / 2.0
    bf = Bounds(np.zeros(nf), np.ones(nf))
    if budget_s - (time.time() - t0) > 10:
        for beta in [2000, 5000]:
            if budget_s - (time.time() - t0) < 5:
                break
            best_h = do_slsqp(best_h, nf, tf, beta, bf, 8000, 1e-16)
            best_c5 = compute_c5(best_h)
    if budget_s - (time.time() - t0) > 5:
        best_h = frank_wolfe(best_h, tf, 5000, t0, budget_s, top_k=3)[0]
        best_c5 = compute_c5(best_h)
    if budget_s - (time.time() - t0) > 5:
        best_h = subgrad_nesterov(best_h, tf, 3000, 0.005, t0, budget_s, top_k=3)[0]
        best_c5 = compute_c5(best_h)
    if budget_s - (time.time() - t0) > 3:
        best_h, best_c5 = coord_descent(best_h, tf, 50, t0, budget_s)

    return best_h, best_c5, nf
```
