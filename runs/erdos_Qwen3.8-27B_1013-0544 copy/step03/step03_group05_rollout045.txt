        h_bin = project_h((h_base > 0.5).astype(float), target)

        # Fourier-based
        x = np.linspace(0, 2, n)
        h_four = 0.5 + 0.3 * np.cos(np.pi * x) + 0.1 * np.sin(2 * np.pi * x)
        h_four = project_h(h_four, target)

        # Perturbed
        h_pert = project_h(h_base + np.random.normal(0, 0.03, n), target)

        starts = [h_base, h_bin, h_four, h_pert]

        for h0 in starts:
            if time_left() < 40:
                break
            cur = h0.copy()
            cur_c5 = compute_c5(cur)

            # Momentum subgradient descent
            momentum = np.zeros(n)
            mu = 0.6
            alpha = 0.05

            for it in range(8000):
                if time_left() < 15:
                    break
                grad, corr_val = get_subgradient(cur, n, dx, 180.0, 4)
                gnorm = np.linalg.norm(grad)
                if gnorm < 1e-14:
                    break
                grad /= gnorm
                momentum = mu * momentum + (1 - mu) * grad
                step = alpha / (1.0 + 0.0005 * it)
                h_trial = project_h(cur - step * momentum, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5:
                    cur = h_trial
                    cur_c5 = c5_trial
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = cur.copy()
                        best_n = n
                else:
                    h_trial = project_h(cur - step * 0.5 * grad, target)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < cur_c5:
                        cur = h_trial
                        cur_c5 = c5_trial
                        momentum = grad
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = cur.copy()
                            best_n = n
                    else:
                        # SA escape
                        T = 0.003 * np.exp(-it / 800.0)
                        if T > 1e-5:
                            noise = np.random.normal(0, T, n)
                            h_trial = project_h(cur + noise, target)
                            c5_trial = compute_c5(h_trial)
                            if c5_trial < cur_c5 + T * 4:
                                cur = h_trial
                                cur_c5 = c5_trial

        # Binary local search on best at this resolution
        if time_left() > 30:
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n)
            h_bs = np.interp(x_new, x_old, best_h)
            h_gs, c5_gs = binary_local_search(h_bs, n, target, dx,
                                              2000, time_left, 10)
            if c5_gs < best_c5:
                best_c5 = c5_gs
                best_h = h_gs.copy()
                best_n = n

        # SLSQP refinement
        if time_left() > 20:
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n)
            h_s = np.interp(x_new, x_old, best_h)
            h_s = project_h(h_s, target)
            bounds = Bounds(np.zeros(n), np.ones(n))
            cons = [{'type': 'eq', 'fun': lambda x: constraint_eq(x, target),
                     'jac': lambda x: constraint_jac(x, target)}]
            for beta in [80, 200, 500]:
                if time_left() < 10:
                    break
                try:
                    res = minimize(
                        lambda x: smooth_max_obj(x, dx, beta, target),
                        h_s, method='SLSQP', bounds=bounds,
                        constraints=cons,
                        options={'maxiter': 1500, 'ftol': 1e-14})
                    h_new = project_h(res.x, target)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new
                        best_n = n
                        h_s = h_new
                except Exception:
                    pass

    # Final fine-tuning with very small steps
    if time_left() > 5:
        n_f = len(best_h)
        target_f = n_f / 2.0
        dx_f = 2.0 / n_f
        cur = best_h.copy()
        cur_c5 = best_c5
        for it in range(30000):
            if time_left() < 2:
                break
            grad, _ = get_subgradient(cur, n_f, dx_f, 400.0, 2)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            step = 0.0015 * (0.999 ** it)
            improved = False
            for _ in range(10):
                h_trial = project_h(cur - step * grad, target_f)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 - 1e-16:
                    cur = h_trial
                    cur_c5 = c5_trial
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = cur.copy()
                        best_n = n_f
                    break
                step *= 0.5
            if not improved and it > 100:
                step = 0.0005
                h_trial = project_h(cur - step * grad, target_f)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 - 1e-17:
                    cur = h_trial
                    cur_c5 = c5_trial
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = cur.copy()
                        best_n = n_f

    return best_h, best_c5, len(best_h)
