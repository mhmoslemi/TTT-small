def project_h(h, target_sum):

def circ_corr(h, p):
    """Circular correlation of h with p (both length n)."""
    H = np.fft.rfft(h)
    P = np.fft.rfft(p)
    return np.fft.irfft(H * np.conj(P), n=len(h))

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def overlap_profile(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def circ_convolve(h, kernel):
    """Circular convolution of h with kernel via FFT."""
    n = len(h)
    return np.fft.irfft(np.fft.rfft(h) * np.fft.rfft(kernel), n=n)

def subgradient_one(h, lag_idx, n):
    """Subgradient of O(lag) w.r.t. h, for a single lag.
    lag_idx is the index in the full correlation (0 to 2n-2)."""
    lag = lag_idx - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        # O(lag) = sum_{j=0}^{n-1-lag} h[j](1-h[j+lag])
        # dO/dh[j] = (1-h[j+lag]) for j < n-lag
        # dO/dh[j] = -h[j-lag] for j >= lag
        g[:n-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n-lag]
    else:
        la = -lag
