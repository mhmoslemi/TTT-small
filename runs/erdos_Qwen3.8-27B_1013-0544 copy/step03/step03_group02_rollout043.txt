import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def sml_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def sum_eq(x_in, target):
    return np.sum(x_in) - target

def sum_eq_jac(x_in, target):
    return np.ones(x_in.shape)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_grad_fast(h, w_lag):
    """Gradient of sum_la w_lag[la] * corr(la) where corr(la) is bilateral."""
    n = len(h)
    # corr_hw[j] = sum_{m=j}^{n-1} h[m]*w_lag[m-j] for j=0..n-1
    corr_hw = np.correlate(h, w_lag, mode='full')  # index 0..2n-2
    # part2[j] = corr_hw[j] for j=0..n-1
    
    # conv_wh[j] = sum_{la=0}^{j} w_lag[la]*h[j-la] for j=0..n-1
    conv_wh = np.convolve(h, w_lag, mode='full')
    
    w_prefix = np.zeros(n + 1)
    w_prefix[1:] = np.cumsum(w_lag)
    
    part1 = w_prefix[1:n+1] - conv_wh[0:n]
    part2 = corr_hw[0:n]
    
    return part1 - part2

def reweighted_descent(h, target, n_iter, t0, budget):
    """Iterative reweighting to solve min-max."""
    n = len(h)
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        
        ctx = get_corr_vals(h_cur)
        c_max = np.max(ctx)
        
        # Build lag-magnitude weights using softmax
        lags = np.abs(np.arange(2*n-1) - (n-1))
        beta_w = 30.0 + 10.0 * (it / max(1, n_iter))
        w_dual = np.exp(beta_w * (ctx - c_max))
        w_dual /= w_dual.sum()
        
        # Convert to w_lag[la] = sum of w_dual[m] for m where |m-(n-1)|=la
        w_lag = np.zeros(n)
        for la in range(n):
            idx1 = n - 1 + la  # positive lag
            if la == 0:
                w_lag[la] = w_dual[idx1]
            else:
                idx2 = n - 1 - la  # negative lag
                w_lag[la] = w_dual[idx1] + w_dual[idx2]
        
        # Projected gradient steps
        step = 0.01
        for sub in range(50):
            if budget - (time.time() - t0) < 1:
                break
            grad = weighted_grad_fast(h_cur, w_lag)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            # Adaptive step size
            trial = project_h(h_cur - step * grad / gn, target)
            c_try = compute_c5(trial)
            if c_try < best_c5:
                h_cur = trial
                best_c5 = c_try
                step = min(step * 1.1, 0.1)
            else:
                step *= 0.7
                if step < 1e-5:
                    break
        # Add small weight to all lags for stability
        w_lag = 0.9 * w_lag + 0.1 * np.ones(n) / n
    
    return h_cur, best_c5

def do_slsqp(h0, n, target, beta, maxiter, ftol):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0/n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def coord_descent(cur, target, n_swaps, t0, budget):
    best_c5 = compute_c5(cur)
    n = len(cur)
    for sw in range(n_swaps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        deltas = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01]
        for j in np.random.permutation(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-15:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved:
            break
    return cur, best_c5

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Get starting point
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(320, 0.5)
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)
    
    n = best_n
    target = n / 2.0
    
    # Phase 1: Reweighted descent (new approach)
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        h1 = project_h(best_h, target)
        h1, c1 = reweighted_descent(h1, target, 200, t0, budget_s)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1
    
    # Phase 2: SLSQP polish with increasing smoothness
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        for beta in [100, 300, 1000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            h2 = do_slsqp(best_h, n, target, beta, 5000, 1e-14)
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_h, best_c5 = h2, c2
    
    # Phase 3: More reweighted descent (second round)
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        h3, c3 = reweighted_descent(best_h, target, 200, t0, budget_s)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3
    
    # Phase 4: Coordinate descent polish

    if remaining > 20:

    # Phase 5: Random restarts with structured perturbations

    return best_h, best_c5, best_n

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def sml_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def sum_eq(x_in, target):
    return np.sum(x_in) - target

def sum_eq_jac(x_in, target):
    return np.ones(x_in.shape)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_grad(h, w_dual):
    n = len(h)
    A_plus = np.zeros(n)
    A_minus = np.zeros(n)
    A_plus[1:] = w_dual[n:]
    A_minus[1:] = w_dual[:n-1][::-1]
    B = A_plus + A_minus
    prefix_ap = np.cumsum(A_plus)
    prefix_am = np.cumsum(A_minus)
    conv_Bh = np.convolve(h, B, mode='full')[0:n]
    S = np.correlate(h, B, mode='full')[n-1:2*n-1]
    return w_dual[n-1]*(1.0 - 2.0*h) + prefix_ap + prefix_am[::-1] - conv_Bh - S

def reweighted_descent(h, target, n_outer, sub_steps, t0, budget):
    n = len(h)
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    step = 0.03
    for it in range(n_outer):
        if budget - (time.time() - t0) < 3:
            break
        ctx = get_corr_vals(h_cur)
        c_max = np.max(ctx)
        beta_w = 40.0 + 20.0 * (it / max(1, n_outer))
        w_dual = np.exp(beta_w * (ctx - c_max))
        w_dual += 1e-10
        w_dual /= w_dual.sum()
        for _ in range(sub_steps):
            if budget - (time.time() - t0) < 1:
                break
            grad = weighted_grad(h_cur, w_dual)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            trial = project_h(h_cur - step * grad / gn, target)
            c_try = compute_c5(trial)
            if c_try < best_c5 - 1e-15:
                h_cur = trial
                best_c5 = c_try
                step = min(step * 1.05, 0.15)
            else:
                step *= 0.7
                if step < 1e-6:
                    break
    return h_cur, best_c5

def do_slsqp(h0, n, target, beta, maxiter, ftol):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0/n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def coord_descent(cur, target, sweeps, t0, budget):
    best_c5 = compute_c5(cur)
    n = len(cur)
    deltas = [0.04, -0.04, 0.02, -0.02, 0.008, -0.008, 0.003, -0.003]
    for sw in range(sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in np.random.permutation(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-15:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c5

def binary_push(h, target, t0, budget):
    n = len(h)
    h_b = h.copy()
    for _ in range(20):
        if budget - (time.time() - t0) < 2:
            break
        h_b = np.where(h_b > 0.6, 1.0, np.where(h_b < 0.4, 0.0, h_b))
        h_b = project_h(h_b, target)
        if compute_c5(h_b) >= compute_c5(h):
            h_b = project_h(0.5*h + 0.5*h_b, target)
            break
    return h_b
