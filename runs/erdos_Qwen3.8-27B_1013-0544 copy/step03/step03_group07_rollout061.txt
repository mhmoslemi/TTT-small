from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # Distribute among free variables
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    # Final fixup
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:

        s = k - (n - 1)
        if s >= 0:

            if s > 0:
                grad[:s] += wk * (-h[s:n])
        else:
            t = -s
            grad[:n-t] += wk * (1.0 - h[t:])
            grad[n-t:] += wk * (-h[:t])
    grad *= dx
    return val, grad

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf > 0:
            step = diff / nf
            # Check if step would violate bounds
            max_step_up = (1.0 - h[free]).min() if step > 0 else 1.0
            max_step_down = (-h[free]).min() if step < 0 else 1.0
            actual_step = min(max(step, max_step_down), max_step_up) if abs(step) > 0 else 0
            h[free] += actual_step
            h = np.clip(h, 0.0, 1.0)
            diff2 = target_sum - h.sum()
            if abs(diff2) > 1e-10:
                h += diff2 / len(h)
                h = np.clip(h, 0.0, 1.0)
    return h

def adam_projected(h, beta, max_iters, lr, betas_adam=(0.9, 0.999), eps=1e-8):
    """Adam optimizer with projection for the smooth max objective."""
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(max_iters):
        val, grad = smoothmax_val_grad(x, beta)
        m_adam = betas_adam[0] * m_adam + (1 - betas_adam[0]) * grad
        v_adam = betas_adam[1] * v_adam + (1 - betas_adam[1]) * grad**2
        m_hat = m_adam / (1 - betas_adam[0]**(t+1))
        v_hat = v_adam / (1 - betas_adam[1]**(t+1))
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x_new = project_feasible(x_new, target)
        c = compute_c5(x_new)
        if c < best_v:
            best_v = c
            best_x = x_new.copy()
        x = x_new
    return best_x, best_v

def slsqp_optimize(h, beta, maxiter=300):

    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        return smoothmax_val_grad(x, beta)[0]
    
    def grad(x):
        return smoothmax_val_grad(x, beta)[1]
    
    def con(x):
        return x.sum() - target
    
    def con_jac(x):
        return np.ones(n)

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_corr(h):
    n = len(h)
    if n <= 500:
        return np.correlate(h, 1.0 - h, mode="full")
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    return np.fft.irfft(H * np.conj(G), N)[:2*n-1]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf > 0:
            h[free] += diff / nf
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -50, 50))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[0:n-s] += wk * (1.0 - h[s:n])
            grad[s:n] += wk * (-h[0:n-s])
        else:
            t = -s
            grad[0:n-t] += wk * (1.0 - h[t:n])
            grad[t:n] += wk * (-h[0:n-t])
    grad *= dx
    return val, grad

def slsqp_optimize(h, beta, maxiter=300):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        return smoothmax_val_grad(x, beta)[0]
    def grad(x):
        return smoothmax_val_grad(x, beta)[1]
    def con(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
    try:
        res = minimize(obj, h.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def adam_projected(h, beta, max_iters, lr0, decay=0.9999):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    m_a = np.zeros(n)
    v_a = np.zeros(n)
    best_x = x.copy()
    best_v = compute_c5(x)
    b1, b2, eps = 0.85, 0.999, 1e-8
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        val, grad = smoothmax_val_grad(x, beta)
        m_a = b1 * m_a + (1 - b1) * grad
        v_a = b2 * v_a + (1 - b2) * grad**2
        m_hat = m_a / (1 - b1**(t+1))
        v_hat = v_a / (1 - b2**(t+1))
        x_new = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        x_new = project_feasible(x_new, target)
        c = compute_c5(x_new)
        if c < best_v:
            best_v = c
            best_x = x_new.copy()
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def structured_starts(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0/n
    results = []
    results.append(project_feasible(0.5 + 0.35*np.sin(np.pi*x), n/2.0))
    results.append(project_feasible(0.5 + 0.35*np.cos(np.pi*x), n/2.0))
    results.append(project_feasible(0.5 + 0.3*np.sin(2*np.pi*x), n/2.0))
    results.append(project_feasible(0.5 + 0.4*np.sin(np.pi*x)**2 - 0.2, n/2.0))
    results.append(project_feasible(np.where(x < 1.0, 0.75, 0.25), n/2.0))
    for _ in range(3):
        results.append(project_feasible(rng.rand(n), n/2.0))
    return results
