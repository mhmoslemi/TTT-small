def project_h(h, target_sum):

        h = np.clip(h, 0.0, 1.0)

            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-18})
            h_new = project_h(res.x, target)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 2: Nesterov momentum subgradient with adaptive step
    v = np.zeros(n)
    momentum = 0.7
    base_lr = 0.04
    n_active = 5
    no_improve = 0

    for it in range(50000):
        if time_left() < 8:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:n_active]
        top_lags = [int(i) - (n - 1) for i in top_idx]
        grad = subgrad_lags(best_h, top_lags)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        grad /= gnorm
        v = momentum * v - base_lr * grad
        found = False
        for mult in [1.0, 0.6, 0.3, 0.15, 0.07]:
            h_trial = project_h(best_h + mult * v, target)
            c5_trial, _ = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
                found = True
                no_improve = 0
                break
        if not found:
            no_improve += 1
            v *= 0.5
            if no_improve % 50 == 0:
                base_lr *= 0.85
            if no_improve > 300:
                # Restart with perturbation
                scale = min(0.03, 0.005 * (no_improve // 300))
                h_trial = project_h(best_h + np.random.randn(n) * scale, target)
                c5_trial, _ = compute_c5(h_trial)
                if c5_trial < best_c5:
                    best_h = h_trial.copy()
                    best_c5 = c5_trial
                no_improve = 0
                v = np.zeros(n)
                base_lr = 0.04
            if no_improve > 3000:
                break
        # Gradually increase active lags
        n_active = min(8, 3 + it // 5000)

    # Phase 3: Quantization to 3-valued (0, 0.5, 1)
    if time_left() > 20:
        # Try snapping
        for thresh in [1/3, 2/3]:
            h_q = np.where(best_h < thresh, 0.0, np.where(best_h < 1 - thresh + 0.5, 0.5, 1.0))
            h_q = project_h(h_q, target)
            c5_q, _ = compute_c5(h_q)
            if c5_q < best_c5:
                best_h = h_q.copy()
                best_c5 = c5_q
        # Try rounding to 0.5
        h_r = np.round(best_h * 2.0) / 2.0
        h_r = project_h(h_r, target)
        c5_r, _ = compute_c5(h_r)
        if c5_r < best_c5:
            best_h = h_r.copy()
            best_c5 = c5_r

    # Phase 4: Intensive local pairwise moves
    for it in range(50000):
        if time_left() < 4:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:4]
        improved = False
        for idx in top_idx:
            if time_left() < 1:
                break
            lag = int(idx) - (n - 1)
            abs_lag = abs(lag)
            if abs_lag >= n or abs_lag == 0:
                continue
            n_valid = n - abs_lag
            for _ in range(10):
                i = np.random.randint(n_valid)
                j = i + abs_lag
                for d in [0.04, 0.02, 0.01, -0.01, -0.02, -0.04, 0.08, -0.08]:
                    ni = best_h[i] + d
                    nj = best_h[j] - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h_trial = best_h.copy()
                    h_trial[i] = ni
                    h_trial[j] = nj
                    h_trial = project_h(h_trial, target)
                    c5_t, _ = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-15:
                        best_h = h_trial.copy()
                        best_c5 = c5_t
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved and it % 800 == 799:
            k1, k2 = np.random.choice(n, 2, replace=False)
            d = np.random.choice([0.05, -0.05, 0.1, -0.1])
            h_trial = best_h.copy()
            h_trial[k1] += d
            h_trial[k2] -= d
            h_trial = project_h(h_trial, target)
            c5_t, _ = compute_c5(h_trial)
            if c5_t < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_t

    # Phase 5: Final SLSQP at very high beta
    if time_left() > 12:
        for beta in [50000, 100000, 200000]:
            if time_left() < 10:
                break
            def obj2(h_in):
                c = np.correlate(h_in, 1.0 - h_in, mode="full")
                return smooth_max(c, beta) * dx
            cons2 = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                      'jac': lambda x: np.ones(x.shape)}]
            try:
                res = minimize(obj2, best_h, method='SLSQP', bounds=bounds,
                               constraints=cons2, options={'maxiter': 10000, 'ftol': 1e-18})
                h_new = project_h(res.x, target)
                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 6: Final micro subgradient
    for it in range(20000):
        if time_left() < 1:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:3]
        top_lags = [int(i) - (n - 1) for i in top_idx]
        grad = subgrad_lags(best_h, top_lags)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        for step in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_trial = project_h(best_h - step * grad, target)
            c5_trial, _ = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
                break

    # Final validation
    assert best_h.min() >= -1e-12 and best_h.max() <= 1.0 + 1e-12
    assert abs(best_h.sum() - target) < 1e-10
    best_c5, _ = compute_c5(best_h)

    return best_h, best_c5, n
