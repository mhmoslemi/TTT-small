from scipy.optimize import minimize, differential_evolution, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -100, 100)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def obj_smo(x, target, dx, beta):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm = smooth_max(corr, beta)
    penalty = 1e6 * (h.sum() - target)**2
    return sm * dx + penalty

def obj_topk(x, target, dx, top_k=5):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Sum of top-k correlations
    top = np.sort(corr)[::-1][:top_k]
    penalty = 1e6 * (h.sum() - target)**2
    return np.sum(top) * dx + penalty

    best_c5 = compute_c5(best_h)

The problem is: min_h max_L {corr_L(h)} subject to 0 ≤ h ≤ 1, Σh = n/2.

1. h ← h - η * ∇smooth_max(corr(h))
2. h ← project_h(h, n/2)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -100, 100)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

    top = np.sort(corr)[::-1][:top_k]

    vals = top - np.max(top)
    soft = np.max(top) + np.log(np.sum(np.exp(vals * 10.0))) / 10.0
    return soft * dx + pen

def coordinate_descent(h, target, dx, n_iter=200, t0=0, budget_s=1100):
    """Simple coordinate descent: optimize one coordinate at a time."""

        return budget_s - (time.time() - t0)
