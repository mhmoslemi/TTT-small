def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_val(corr, beta):
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(corr)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr(h, lag, n):
    """Gradient of correlation at given lag w.r.t. h"""
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n-lag]
    else:
        m = -lag
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:n-m]
    return g

def subgrad_topk(h, n_active=5):

    corr = compute_corr(h)
    top_idx = np.argsort(corr)[::-1][:n_active]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        grad += grad_corr(h, lag, n)
    return grad, corr

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_val(corr, beta):
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(corr)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_lag(h, lag, n):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g

def subgrad_topk(h, n_active=5):
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[::-1][:n_active]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        grad += grad_corr_lag(h, lag, n)
    return grad, corr

def lbfgsb_obj(x, n, beta, dx, mu, target):
    corr = compute_corr(x)
    return smooth_max_val(corr, beta) * dx + mu * (x.sum() - target) ** 2

def lbfgsb_grad(x, n, beta, dx, mu, target):
    corr = compute_corr(x)
    m = np.max(corr)
    xs = np.clip(beta * (corr - m), -500, 500)

        lag = k - (n - 1)
        g += wk * grad_corr_lag(x, lag, n)
    g += 2.0 * mu * (x.sum() - target)
    return g

    return smooth_max_val(compute_corr(x), beta) * dx

def slsqp_con(x, target):
    return x.sum() - target

def slsqp_jac(x, n):
    return np.ones(n)
