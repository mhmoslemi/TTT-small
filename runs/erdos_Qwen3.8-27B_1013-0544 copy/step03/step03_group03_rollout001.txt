                h_new = project_feasible(res.x, target)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_c5 = c_new
                    h_best = h_new.copy()
            except Exception:
                pass

    # Phase 5: Micro-polish targeting single worst lag
    if tl() > 10:
        h = h_best.copy()
        n = len(h)
        dx = 2.0 / n
        target = n / 2.0
        c_h = compute_c5(h)
        stall = 0
        for it in range(8000):
            if tl() < 4:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_worst = int(np.argmax(corr))
            lag = k_worst - (n - 1)
            grad = lag_gradient(h, lag, n) * dx
            lr = 0.003 / (1.0 + 0.005 * it)
            h_new = project_feasible(h - lr * grad, target)
            c_new = compute_c5(h_new)
            if c_new < c_h:
                h = h_new
                c_h = c_new
                stall = 0
                if c_new < best_c5:
                    best_c5 = c_new
                    h_best = h.copy()
            else:
                stall += 1
                if stall > 3000:
                    break

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
