    dx = 2.0 / n

    m = np.max(corr)

    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smoothmax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    
    grad = np.zeros(n)
    ns = n - 1
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - ns
        if s > 0:
            grad[:n-s] += wk * (1.0 - h[s:])
            grad[s:] += wk * (-h[:n-s])
        elif s < 0:
            t = -s
            grad[t:] += wk * (1.0 - h[:n-t])
            grad[:n-t] += wk * (-h[t:])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

def project_box_sum(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Adjust free variables
    free = (h > 1e-10) & (h < 1.0 - 1e-10)
    if free.sum() == 0:
        return h
    h[free] += diff / free.sum()
    h = np.clip(h, 0.0, 1.0)
    # Second pass if needed
    diff2 = target_sum - h.sum()
    if abs(diff2) > 1e-10:
        free2 = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free2.sum() > 0:
            h[free2] += diff2 / free2.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

        return smoothmax_grad(x, beta)[0]

        return smoothmax_grad(x, beta)[1]

                   bounds=[(0,1)]*n,
                   options={'maxiter': maxiter, 'gtol': 1e-8})

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_values(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = corr_values(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    ns = n - 1
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - ns
        if s > 0:
            grad[:n-s] += wk * (1.0 - h[s:])
            grad[s:] += wk * (-h[:n-s])
        elif s < 0:
            t = -s
            grad[t:] += wk * (1.0 - h[:n-t])
            grad[:n-t] += wk * (-h[t:])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    free = (h > 1e-10) & (h < 1.0 - 1e-10)
    if free.sum() == 0:
        return h
    h[free] += diff / free.sum()
    h = np.clip(h, 0.0, 1.0)
    diff2 = target_sum - h.sum()
    if abs(diff2) > 1e-10:
        free2 = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free2.sum() > 0:
            h[free2] += diff2 / free2.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def lbfgsb_run(h, beta, maxiter=300):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smoothmax_val_grad(x, beta)[0]
    
    def grad(x):
        return smoothmax_val_grad(x, beta)[1]
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-6, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def subgradient_method(h, max_iters, lr0, decay, top_k=3):
    """Subgradient method with top-k lag averaging."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        corr = corr_values(x)
        # Pick top-k lags
        top_idx = np.argsort(-corr)[:top_k]
        # Average their subgradients
        grad = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            if s > 0:
                grad[:n-s] += (1.0 - x[s:])
                grad[s:] += (-x[:n-s])
            elif s < 0:
                t2 = -s
                grad[t2:] += (1.0 - x[:n-t2])
                grad[:n-t2] += (-x[t2:])
            else:
                grad[:] += (1.0 - 2.0 * x)
        grad /= top_k
        grad *= dx
        
        x_new = project_feasible(x - lr_t * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x = x_new
    
    return best_x, best_v

def binary_local_search(h, max_iters=20000, rng=None):
    """Local search on binary solutions via swaps."""
    if rng is None:
        rng = np.random
    
    n = len(h)
    target = n // 2
    # Project to binary
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    
    best_val = compute_c5(b)
    best_b = b.copy()
    
    # Precompute current correlations
    corr = corr_values(b)
    
    for it in range(max_iters):
        # Find max lag
        max_k = np.argmax(corr)
        max_s = max_k - (n - 1)
        
        # Find candidate swaps that reduce the max
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try random swaps
        best_delta = 0
        best_a, best_bpos = -1, -1
        
        # Sample some candidates
        n_try = min(100, len(ones) * len(zeros))
        for _ in range(n_try):
            a = ones[rng.randint(len(ones))]
            bpos = zeros[rng.randint(len(zeros))]
            if a == bpos:
                continue
            # Compute delta at the max lag
            delta = _swap_delta(b, a, bpos, max_s, n)
            if delta < best_delta:
                best_delta = delta
                best_a, best_bpos = a, bpos
        
        if best_a < 0:
            break
        
        # Apply swap
        best_b[best_a] = 0.0
        best_b[best_bpos] = 1.0
        # Recompute correlations (expensive but necessary for correctness)
        corr = corr_values(best_b)
        val = np.max(corr) * (2.0 / n)
        if val < best_val:
            best_val = val
            best_b = best_b.copy()  # already is
            b = best_b.copy()
        else:
            # Revert
            b[best_a] = 1.0
            b[best_bpos] = 0.0
            best_b = b.copy()
    
    return best_b, best_val

def _swap_delta(b, a, bpos, s, n):
    """Compute change in corr at shift s from swapping a(1->0) and bpos(0->1)."""
    delta = 0.0
    if s >= 0:
        # Removing 1 at a
        if a + s < n:
            delta -= (1.0 - b[a + s])
        if a - s >= 0:
            delta += b[a - s]
        # Adding 1 at bpos
        if bpos + s < n:
            delta += (1.0 - b[bpos + s])
        if bpos - s >= 0:
            delta -= b[bpos - s]
    else:
        t = -s
        # Removing 1 at a
        if a + t < n:
            delta += b[a + t]
        if a - t >= 0:
            delta -= (1.0 - b[a - t])
        # Adding 1 at bpos
        if bpos + t < n:
            delta -= b[bpos + t]
        if bpos - t >= 0:
            delta += (1.0 - b[bpos - t])
    return delta

def adam_optimize(h, max_iters=5000, lr=0.01, beta1=0.9, beta2=0.999, top_k=5):
    """Adam optimizer with subgradient of top-k lags."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    best_x = x.copy()
    best_val = compute_c5(x)
    
    for t in range(max_iters):
        corr = corr_values(x)
        top_idx = np.argsort(-corr)[:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            if s > 0:
                grad[:n-s] += (1.0 - x[s:])
                grad[s:] += (-x[:n-s])
