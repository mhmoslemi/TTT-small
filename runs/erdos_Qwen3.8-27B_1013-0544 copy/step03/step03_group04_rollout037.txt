    corr = np.correlate(h, 1.0 - h, mode="full")

        j_start = max(0, k)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, target):
    hb = (h >= 0.5).astype(float)
    diff = int(round(target - hb.sum()))
    if diff > 0:
        zi = np.where(hb == 0)[0]
        if len(zi) >= diff:
            oi = zi[np.argsort(-h[zi])][:diff]
            hb[oi] = 1.0
    elif diff < 0:
        oi = np.where(hb == 1)[0]
        if len(oi) >= -diff:
            sel = oi[np.argsort(h[oi])][:-diff]
            hb[sel] = 0.0
    return hb

def swap_anneal(h, max_iters, seed, T0=0.008, T1=0.0003):

        a = ones[rng.randint(len(ones))]

def compute_grad(h, top_k=3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for li in idx:
        k = li - (n - 1)
        js = max(0, k)
        je = min(n - 1, n - 1 + k)
        if js <= je:
            grad[js:je+1] += 1.0 - h[js-k:je+1-k]
        js2 = max(0, -k)
        je2 = min(n - 1, n - 1 - k)
        if js2 <= je2:
            grad[js2:je2+1] -= h[js2+k:je2+1+k]
    return grad
