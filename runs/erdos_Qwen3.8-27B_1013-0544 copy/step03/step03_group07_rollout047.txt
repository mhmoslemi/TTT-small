import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, penalty=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    s = h.sum() - n / 2.0
    val += penalty * s * s
    grad = np.zeros(n)
    nc = len(corr)
    for k in range(nc):
        wk = w[k]
        if wk < 1e-12:
            continue
        s_k = k - (n - 1)
        if s_k >= 0:
            ms = np.arange(s_k, n)
            grad[ms] += wk * (1.0 - h[ms - s_k])
            mp = np.arange(0, n - s_k)
            grad[mp] += wk * (-h[mp + s_k])
        elif s_k < 0:
            t = -s_k
            ms = np.arange(0, n - t)
            grad[ms] += wk * (1.0 - h[ms + t])
            mp = np.arange(t, n)
            grad[mp] += wk * (-h[mp - t])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * s
    return val, grad

def lbfgs_iter(h, beta, penalty, maxiter=250):
    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta, penalty)
        return v
    def grad(x):
        _, g = smoothmax_val_grad(x, beta, penalty)
        return g
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h

def slsqp_refine(h, maxiter=300, beta=500):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta, 0.0)
        return v
    def grad(x):
        _, g = smoothmax_val_grad(x, beta, 0.0)
        return g
    def con_fun(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def topk_subgrad(h, max_iters, lr0, decay, top_k, time_limit, t0):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        lr_t = lr0 * (decay ** t)
        corr = np.correlate(x, 1.0 - x, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for ki in top_idx:
            s_k = ki - (n - 1)
            if s_k >= 0:
                ms = np.arange(s_k, n)
                grad[ms] += (1.0 - x[ms - s_k])
                mp = np.arange(0, n - s_k)
                grad[mp] += (-x[mp + s_k])
            elif s_k < 0:
                tt = -s_k
                ms = np.arange(0, n - tt)
                grad[ms] += (1.0 - x[ms + tt])
                mp = np.arange(tt, n)
                grad[mp] += (-x[mp - tt])
            else:
                grad[:] += (1.0 - 2.0 * x)
        grad = grad / top_k * dx
        x = project_feasible(x - lr_t * grad, target)
        v = compute_c5(x)
        if v < best_v:
            best_v = v
            best_x = x.copy()
    return best_x, best_v

def make_structured_init(n, freq, phase, amp):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp * np.cos(2.0 * np.pi * freq * x + phase)
    return project_feasible(np.clip(h, 0, 1), n / 2.0)

    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(400, 0.5)
    best_c5 = compute_c5(h_best)
    n = len(h_best)

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS progressive beta at current resolution
    for beta in [50, 100, 200, 400, 800, 1500]:
        if time_left() < 40:
            break
        h_new = lbfgs_iter(h_best, beta, penalty=1.0, maxiter=300)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 2: SLSQP refinement at increasing beta
    for beta in [400, 800, 1600]:
        if time_left() < 40:
            break
        h_new = slsqp_refine(h_best, maxiter=300, beta=beta)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 3: Top-k subgradient focused on worst lags
    if time_left() > 120:
        tl = min(time_left() - 30, 250)
        h_sg, c_sg = topk_subgrad(h_best, 10000, 0.03, 0.9999, 5, tl, t0)
        if c_sg < best_c5:
            h_best, best_c5 = h_sg, c_sg
        if time_left() > 35:
            h_new = slsqp_refine(h_best, maxiter=300, beta=1200)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 4: Diverse structured starts mixed with best
    n_curr = len(h_best)
    for freq in [1.0, 2.0, 3.0]:
        if time_left() < 50:
            break
        for amp in [0.15, 0.25]:
            if time_left() < 35:
                break
            h_struct = make_structured_init(n_curr, freq, 0.0, amp)
            for mix in [0.6, 0.8]:
                if time_left() < 30:
                    break
                h_mix = project_feasible(mix * h_best + (1.0 - mix) * h_struct, n_curr / 2.0)
                h_new = lbfgs_iter(h_mix, 300, penalty=1.0, maxiter=200)
                c = compute_c5(h_new)
                if c < best_c5:
                    h_best, best_c5 = h_new, c

    # Phase 5: Grid doubling to n=600
    if time_left() > 220:
        h_hi = interp_to_n(h_best, 600)
        for beta in [100, 300, 800]:
            if time_left() < 70:
                break
            h_new = lbfgs_iter(h_hi, beta, penalty=1.0, maxiter=250)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
                h_hi = h_new.copy()
        if time_left() > 60:
            h_new = slsqp_refine(h_best, maxiter=300, beta=1000)
            c = compute_c5(h_new)
            if c < best_c5:

    # Phase 6: Grid increase to n=800

            c = compute_c5(h_new)
            if c < best_c5:

                h_hi2 = h_new.copy()
        if time_left() > 80:

    # Phase 7: Final top-k + polish
    if time_left() > 100:
        tl = min(time_left() - 40, 200)
        h_sg, c_sg = topk_subgrad(h_best, 6000, 0.015, 0.99995, 3, tl, t0)

    return h_best, best_c5, len(h_best)
