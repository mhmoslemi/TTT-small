import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descend_step(h, full_corr, i, n):
    """Find optimal h[i] given full_corr (computed with current h)."""
    shifts = np.arange(-(n-1), n)
    i_s = i + shifts
    i_ms = i - shifts
    
    valid_pos = (i_s >= 0) & (i_s < n)
    valid_neg = (i_ms >= 0) & (i_ms < n)
    
    B = np.zeros(2 * n - 1)
    B[valid_pos] = 1.0 - h[i_s[valid_pos]]
    B[valid_neg] = B[valid_neg] - h[i_ms[valid_neg]]
    
    c = full_corr
    u_lo = -h[i]
    u_hi = 1.0 - h[i]
    
    if u_lo >= u_hi:
        return h[i]
    
    # Ternary search for minimum of convex function f(u) = max(c + u*B)
    for _ in range(50):
        m1 = u_lo + (u_hi - u_lo) / 3.0
        m2 = u_hi - (u_hi - u_lo) / 3.0
        if np.max(c + m1 * B) < np.max(c + m2 * B):
            u_hi = m2
        else:
            u_lo = m1
    
    return np.clip(h[i] + (u_lo + u_hi) / 2.0, 0.0, 1.0)

def coordinate_descent(h, max_sweeps=100, tol=1e-10):
    """Run coordinate descent, return improved h."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    
    for sweep in range(max_sweeps):
        full_corr = np.correlate(h, 1.0 - h, mode="full")
        max_before = np.max(full_corr)
        
        for i in range(n):
            t_opt = coord_descend_step(h, full_corr, i, n)
            if abs(t_opt - h[i]) > 1e-12:
                h[i] = t_opt
                # Recompute full_corr (expensive but necessary for correctness)
                full_corr = np.correlate(h, 1.0 - h, mode="full")
        
        max_after = np.max(full_corr)
        if max_before - max_after < tol:
            break
    
    # Project to ensure exact sum
    h = project_h(h, target)
    return h

def coordinate_descent_fast(h, max_sweeps=500, tol=1e-12):

    target = n / 2.0
    h = h.copy()

    shifts = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        max_before = np.max(full_corr)
        changed = False
        
        for i in range(n):
            i_s = i + shifts
            i_ms = i - shifts
            valid_pos = (i_s >= 0) & (i_s < n)
            valid_neg = (i_ms >= 0) & (i_ms < n)
            
            B = np.zeros(2 * n - 1)
            B[valid_pos] = 1.0 - h[i_s[valid_pos]]
            B[valid_neg] = B[valid_neg] - h[i_ms[valid_neg]]
            
            c = full_corr
            u_lo = -h[i]
            u_hi = 1.0 - h[i]
            
            if u_lo >= u_hi:
                continue
            
            for _ in range(40):
                m1 = u_lo + (u_hi - u_lo) / 3.0
                m2 = u_hi - (u_hi - u_lo) / 3.0
                if np.max(c + m1 * B) < np.max(c + m2 * B):
                    u_hi = m2
                else:
                    u_lo = m1
            
            t_opt = np.clip(h[i] + (u_lo + u_hi) / 2.0, 0.0, 1.0)
            
            if abs(t_opt - h[i]) > 1e-10:
                delta = t_opt - h[i]
                h[i] = t_opt
                full_corr += delta * B
                changed = True
        
        max_after = np.max(full_corr)
        if not changed or max_before - max_after < tol:
            break
    
    h = project_h(h, target)
    return h

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)

        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descend(h, max_sweeps, rng):
    """Fast coordinate descent with incremental correlation update."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    full_corr = np.correlate(h, 1.0 - h, mode="full")
    shifts = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        max_before = np.max(full_corr)
        any_change = False
        
        for i in range(n):
            i_s = i + shifts
            i_ms = i - shifts
            vp = (i_s >= 0) & (i_s < n)
            vn = (i_ms >= 0) & (i_ms < n)
            
            B = np.zeros(2 * n - 1)
            B[vp] = 1.0 - h[i_s[vp]]
            B[vn] -= h[i_ms[vn]]
            
            hi = h[i]
            u_lo = -hi
            u_hi = 1.0 - hi

def find_optimal_u(c, B, u_lo, u_hi, iterations=30):
    """Find u that minimizes max(c + u*B) for u in [u_lo, u_hi]."""
    for _ in range(iterations):
        u_mid = (u_lo + u_hi) / 2.0
        vals = c + u_mid * B
        idx = np.argmax(vals)
        # Check subgradient: is 0 in [min(B[active]), max(B[active])]?
        # Active = indices where vals close to max
        active = np.where(vals >= vals[idx] - 1e-12)[0]
        b_min = B[active].min()
        b_max = B[active].max()
        if b_min <= 0 <= b_max:
            return u_mid
        elif b_min > 0:
            # All active slopes positive, f increasing, go left
            u_hi = u_mid
        else:
            # All active slopes negative, f decreasing, go right
            u_lo = u_mid
    return (u_lo + u_hi) / 2.0

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descend(h, max_sweeps=2000, recompute_every=100):

    target = n / 2.0
    h = h.copy()
    full_corr = np.correlate(h, 1.0 - h, mode="full")
    shifts = np.arange(-(n - 1), n)
    two_n_1 = 2 * n - 1
    
    for sweep in range(max_sweeps):

            full_corr = np.correlate(h, 1.0 - h, mode="full")

        for i in range(n):
            i_s = i + shifts
            i_ms = i - shifts
            vp = (i_s >= 0) & (i_s < n)
            vn = (i_ms >= 0) & (i_ms < n)
            
            B = np.zeros(two_n_1)
            B[vp] = 1.0 - h[i_s[vp]]
            B[vn] -= h[i_ms[vn]]
            
            hi = h[i]
            u_lo = -hi
            u_hi = 1.0 - hi
            
            if u_lo >= u_hi - 1e-14:
                continue

            for _ in range(35):

                if np.max(full_corr + m1 * B) < np.max(full_corr + m2 * B):
                    u_hi = m2
                else:
