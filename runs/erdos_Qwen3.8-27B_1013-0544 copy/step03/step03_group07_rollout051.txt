import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def grad_topk(h, top_k):
    """Gradient of sum of top-k correlation terms (times dx)"""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    top_idx = top_idx[np.argsort(corr[top_idx][::-1])[::-1]]  # descending
    grad = np.zeros(n)
    for ki in top_idx:
        s = ki - (n - 1)
        if s >= 0:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] += (-h[s:])
        elif s < 0:
            t = -s
            grad[:n - t] += (1.0 - h[t:])
            grad[t:] += (-h[:n - t])
        else:
            grad[:] += (1.0 - 2.0 * h)
    grad /= top_k
    grad *= dx
    corr_max = np.max(corr) * dx
    return corr_max, grad, top_idx

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:

        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])

        elif s < 0:

            grad[t:] += wk * (-h[:n - t])

def nesterov_topk_opt(h, max_iters, lr_init, decay, top_k, target):
    n = len(h)
    dx = 2.0 / n
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    patience = 0
    patience_limit = 3000
    for t in range(max_iters):
        lr_t = lr_init * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = min(1.0, (t - 1) / (t + 2) * 1.5)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        _, grad, _ = grad_topk(y, top_k)
        x_new = project_feasible(y - lr_t * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
            patience = 0
        else:
            patience += 1
        x_prev = x
        x = x_new
        if patience > patience_limit and t > 500:
            break
    return best_x, best_v

def backtracking_topk_opt(h, max_iters, top_k, target, lr_init=0.05):
    """Top-k subgradient with backtracking line search"""
    n = len(h)
    dx = 2.0 / n
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    lr = lr_init
    for t in range(max_iters):
        if t == 0:
            y = x.copy()
        else:
            mu = min(1.0, 0.6 * (t / (t + 5)))
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        v_y = compute_c5(y)
        _, grad, _ = grad_topk(y, top_k)
        # Backtracking
        alpha = lr
        accepted = False
        for _ in range(12):
            x_new = project_feasible(y - alpha * grad, target)
            v_new = compute_c5(x_new)
            if v_new < v_y - 1e-10 * alpha * np.sum(grad**2):
                accepted = True
                break
            alpha *= 0.5
        if not accepted:
            x_new = project_feasible(y - alpha * grad, target)
            v_new = compute_c5(x_new)
        if v_new < best_v:
            best_v = v_new
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def slsqp_refine(h, maxiter, beta, target):
    n = len(h)
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    def obj_grad(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq',
             'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(400, 0.5)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: SLSQP warmup at moderate beta
    for beta in [80, 200, 500]:
        if remaining() < 30:
            break
        h_new = slsqp_refine(h_best, 300, beta, n / 2.0)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 2: Nesterov top-k with small k (focused)
    if remaining() > 100:
        h_sg, c_sg = nesterov_topk_opt(h_best, 30000, 0.03, 0.99992, 5, n / 2.0)
        if c_sg < best_c5:
            h_best, best_c5 = h_sg, c_sg

    # Phase 3: SLSQP high beta
    for beta in [800, 1500, 3000]:
        if remaining() < 25:
            break
        h_new = slsqp_refine(h_best, 400, beta, n / 2.0)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 4: More Nesterov with different top-k
    if remaining() > 80:
        h_sg, c_sg = nesterov_topk_opt(h_best, 25000, 0.02, 0.99995, 3, n / 2.0)
        if c_sg < best_c5:
            h_best, best_c5 = h_sg, c_sg
        if remaining() > 40:
            h_new = slsqp_refine(h_best, 400, 2000, n / 2.0)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 5: Shake and re-optimize
    for trial in range(6):
        if remaining() < 40:
            break
        scale = 0.003 + 0.004 * trial
        pert = np.random.randn(n) * scale
        h_pert = project_feasible(h_best + pert, n / 2.0)
        h_new = slsqp_refine(h_pert, 350, 800, n / 2.0)
        c1 = compute_c5(h_new)
        if c1 < best_c5:
            h_best = h_new
            best_c5 = c1
            n = len(h_best)
        if remaining() < 30:
            break
        h_sg2, c_sg2 = nesterov_topk_opt(h_best, 8000, 0.02, 0.99995, 4, n / 2.0)
        if c_sg2 < best_c5:
            h_best, best_c5 = h_sg2, c_sg2

    # Phase 6: Higher resolution n=800
    if remaining() > 200:
        n_new = 800
        h_interp = interp_to_n(h_best, n_new)
        for beta in [200, 600, 1500]:
            if remaining() < 50:
                break
            h_new = slsqp_refine(h_interp, 300, beta, n_new / 2.0)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
                h_interp = h_best.copy()
        if remaining() > 60:
            h_sg, c_sg = nesterov_topk_opt(h_best, 20000, 0.02, 0.99995, 5, 800.0)
            if c_sg < best_c5:
                h_best, best_c5 = h_sg, c_sg
        if remaining() > 30:
            h_new = slsqp_refine(h_best, 300, 3000, 800.0)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 7: Final polish at current resolution
    n_cur = len(h_best)
    if remaining() > 25:
        h_new = slsqp_refine(h_best, 500, 5000, n_cur / 2.0)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c
    if remaining() > 15:
        h_sg, c_sg = nesterov_topk_opt(h_best, 10000, 0.01, 0.99998, 3, n_cur / 2.0)
        if c_sg < best_c5:
            h_best, best_c5 = h_sg, c_sg

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
