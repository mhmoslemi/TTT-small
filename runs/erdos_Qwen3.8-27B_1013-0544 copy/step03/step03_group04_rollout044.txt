        return budget_s - elapsed()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_parent)
    else:
        n_base = 280
        h_parent = np.full(n_base, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: DE at base resolution
    n_de = n_base
    _g_target = n_de / 2.0
    _g_dx = 2.0 / n_de
    _g_beta = 100.0

    if remaining() > 250:
        bounds_de = [(0.0, 1.0)] * n_de
        try:
            de_res = differential_evolution(
                _obj, bounds_de, seed=seed, maxiter=15,
                popsize=15, tol=1e-6, mutation=(0.3, 0.9),
                recombination=0.8, polish=False, updating='immediate',
                init='latinhypercube'
            )
            h_de = project_h(de_res.x, _g_target)
            c5_de = compute_c5(h_de)
            if c5_de < best_c5:
                best_h, best_c5 = h_de.copy(), c5_de
        except Exception:
            pass

    # Phase 2: SLSQP at multiple resolutions
    res_list = [n_base]
    for mult in [1.3, 1.6, 2.0, 2.5]:
        nr = int(n_base * mult)
        if nr <= 700:
            res_list.append(nr)

    for n_pts in res_list:
        if remaining() < 50:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_pts)
        h_interp = project_h(np.interp(x_new, x_old, best_h), n_pts / 2.0)
        h_refined, c5_ref = slsqp_solve(h_interp, n_pts, remaining)
        if c5_ref < best_c5:
            best_h, best_c5 = h_refined.copy(), c5_ref

    # Phase 3: Subgradient polishing on best
    if remaining() > 50:
        h_sg, c5_sg = subgrad_polish(best_h, remaining, iters=50000)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg.copy(), c5_sg

    # Phase 4: Coordinate descent
    if remaining() > 30:
        h_cd, c5_cd = coord_descent(best_h, remaining, rounds=300)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd.copy(), c5_cd

    # Phase 5: Final SLSQP at best resolution
    if remaining() > 30:
        h_fin, c5_fin = slsqp_solve(best_h, len(best_h), remaining)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin.copy(), c5_fin

    # Phase 6: Final subgradient + coord descent
    if remaining() > 20:
        h_sg2, c5_sg2 = subgrad_polish(best_h, remaining, iters=20000)
        if c5_sg2 < best_c5:
            best_h, best_c5 = h_sg2.copy(), c5_sg2
    if remaining() > 15:
        h_cd2, c5_cd2 = coord_descent(best_h, remaining, rounds=200)
        if c5_cd2 < best_c5:
            best_h, best_c5 = h_cd2.copy(), c5_cd2

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, int(len(best_h))
