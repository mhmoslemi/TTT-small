def min_max_linear(slopes, constants, lb=0.0, ub=1.0):
    """Find min_{t in [lb,ub]} max_k (slopes[k]*t + constants[k])"""
    M = len(slopes)
    
    # Sort by slope
    order = np.argsort(slopes)
    s_sorted = slopes[order]
    c_sorted = constants[order]
    
    # Build upper envelope using convex hull trick
    # Maintain a list of (slope, intercept) that form the upper envelope
    # A new line makes previous lines obsolete if it overtakes them before they overtake the ones before
    
    # Upper envelope: for lines sorted by slope (increasing), the upper envelope
    # has breakpoints at increasing x values.
    
    env_slopes = []
    env_intercepts = []
    env_x_start = []  # x where this segment starts on the upper envelope
    
    for i in range(M):
        s = s_sorted[i]
        c = c_sorted[i]
        if not env_slopes:
            env_slopes.append(s)
            env_intercepts.append(c)
            env_x_start.append(-np.inf)
        else:
            # Find intersection with last line on envelope
            # s*x + c = env_slopes[-1]*x + env_intercepts[-1]
            # x = (env_intercepts[-1] - c) / (s - env_slopes[-1])
            if s == env_slopes[-1]:
                # Same slope, keep the one with higher intercept
                if c > env_intercepts[-1]:
                    # This line is always above the last one
                    env_intercepts[-1] = c
                    # But we need to redo the envelope... 
                    # For simplicity with M=399, let's just use the grid approach
                    pass
                continue
            
            x_int = (env_intercepts[-1] - c) / (s - env_slopes[-1])
            
            # Check if this intersection is before the last breakpoint
            while len(env_slopes) > 1 and x_int <= env_x_start[-1]:
                env_slopes.pop()
                env_intercepts.pop()
                env_x_start.pop()
                if s == env_slopes[-1]:
                    break
                x_int = (env_intercepts[-1] - c) / (s - env_slopes[-1])
            
            if s != env_slopes[-1]:
                env_slopes.append(s)
                env_intercepts.append(c)
                env_x_start.append(x_int)
    
    # Now find the minimum of the upper envelope on [lb, ub]
    # The envelope is convex, so the minimum is where the slope changes from neg to pos
    best_val = np.inf
    best_t = lb
    
    for j in range(len(env_slopes)):
        x_start = env_x_start[j]
        x_end = env_x_start[j+1] if j+1 < len(env_slopes) else np.inf
        
        # Clip to [lb, ub]
        seg_start = max(x_start, lb)
        seg_end = min(x_end, ub)
        
        if seg_start > seg_end:
            continue
        
        if env_slopes[j] < 0:
            # Decreasing, minimum is at right end
            t = seg_end
        elif env_slopes[j] > 0:
            # Increasing, minimum is at left end
            t = seg_start
        else:
            # Constant
            t = seg_start
        
        val = env_slopes[j] * t + env_intercepts[j]
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t, best_val

            seed=seed + 500, t0=t0, budget_s=budget_s, max_passes=60)

    # Phase 8: Final SLSQP + CD squeeze
    if remaining() > 40:
        h_fin = slsqp_solve(h_best, 600, 500, top_k=None)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin.copy()
        h_fin2, _ = cd_sweep(h_best, len(h_best), max_passes=80)
        h_fin2 = project_feasible(h_fin2, len(h_best) / 2.0)
        c_fin2 = compute_c5(h_fin2)
        if c_fin2 < best_c5:
            best_c5 = c_fin2
            h_best = h_fin2.copy()

    # Final feasibility check
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)

    return h_best, best_c5, len(h_best)
