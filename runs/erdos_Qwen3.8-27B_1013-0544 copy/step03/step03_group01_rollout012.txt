import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c_values(h, n):
    """Compute all c_k values (the correlation values times dx)."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def gradient_c_k(h, n, shift):
    """Gradient of c_k (for lag with given shift) with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # (1-h[m+shift]) for m+shift in [0, n-1]
    if shift >= 0:
        grad[:n-shift] += dx * (1.0 - h[shift:])
    else:
        grad[-shift:] += dx * (1.0 - h[:n+shift])
    
    # -h[m-shift] for m-shift in [0, n-1]
    if shift >= 0:
        grad[shift:] -= dx * h[:n-shift]
    else:
        grad[:n+shift] -= dx * h[-shift:]
    
    return grad

def softmax_gradient(h, n, alpha):
    """Compute gradient of the soft-max objective."""
    c_vals = compute_c_values(h, n)
    # Numerically stable softmax
    c_max = c_vals.max()
    exp_c = np.exp(alpha * (c_vals - c_max))
    w = exp_c / exp_c.sum()
    
    # Gradient = sum_k w_k * grad_c_k
    grad = np.zeros(n)
    for k in range(len(c_vals)):
        shift = k - (n - 1)
        if w[k] > 1e-10:
            grad += w[k] * gradient_c_k(h, n, shift)
    
    return grad, c_vals

def soft_max_obj(h, n, alpha):
    """Compute soft-max objective value."""
    c_vals = compute_c_values(h, n)
    c_max = c_vals.max()
    return float(c_max + np.log(exp_c.sum()) / alpha) if alpha > 0 else c_max

def soft_max_obj(h, n, alpha):

    c_max = c_vals.max()
    if alpha < 1e-6:
        return float(c_max)

def projected_gradient_step(h, n, alpha, lr_init, max_backtrack):
    """One projected gradient step with backtracking."""
    grad, c_vals = softmax_gradient(h, n, alpha)
    obj_current = soft_max_obj(h, n, alpha)
    
    lr = lr_init
    for bt in range(max_backtrack):

        h_new = project_feasible(h_new, n)

        if obj_new < obj_current:

        lr *= 0.5

    best_c = compute_c5(h, n)
    for p in range(num_passes):

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c_values(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def gradient_for_lag(h, n, shift):
    dx = 2.0 / n
    grad = np.zeros(n)
    if shift >= 0:
        grad[:n - shift] += dx * (1.0 - h[shift:])
        grad[shift:] -= dx * h[:n - shift]
    else:
        s = -shift
        grad[s:] += dx * (1.0 - h[:n - s])
        grad[:n - s] -= dx * h[s:]
    return grad

def soft_max_obj_grad(h, n, alpha):
    c_vals = compute_c_values(h, n)
    c_max = c_vals.max()
    if alpha < 1e-3:
        idx = int(np.argmax(c_vals))
        shift = idx - (n - 1)
        return c_max, gradient_for_lag(h, n, shift)
    shifted = alpha * (c_vals - c_max)
    s_max = shifted.max()
    e = np.exp(shifted - s_max)
    w = e / e.sum()
    logSum = s_max + np.log(e.sum())
    obj = c_max + logSum / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-10:
            grad += w[k] * gradient_for_lag(h, n, k - (n - 1))
    return obj, grad

        else:

    return best_h, best_obj

def upsample(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_from = np.linspace(0, 2, n_from, endpoint=False)
    x_to = np.linspace(0, 2, n_to, endpoint=False)
    idx = np.searchsorted(x_from, x_to)
    idx = np.minimum(idx, n_from - 1)
    return project_feasible(h[idx], n_to)

    if n_from == n_to:

    step = n_from / n_to
    indices = (np.arange(n_to) * step).astype(int)
    indices = np.minimum(indices, n_from - 1)
    return project_feasible(h[indices], n_to)
