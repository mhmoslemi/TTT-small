def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_binary(h, n):
    """Project to 0-1 with exactly n//2 ones."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:target]] = 1.0
    return result

def project_continuous(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def sa_binary_search(h_bin, n_steps=500000, T_start=0.02, T_end=0.001, seed=42):
    """Simulated annealing on binary strings with swap moves."""
    rng = np.random.RandomState(seed)
    n = len(h_bin)
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for step in range(n_steps):
        T = T_start * (T_end / T_start) ** (step / n_steps)
        
        oi = ones[rng.randint(len(ones))]
        zi = zeros[rng.randint(len(zeros))]
        if oi == zi:
            continue
        
        h[oi] = 0
        h[zi] = 1
        trial_c5 = compute_c5(h)
        delta = trial_c5 - cur_c5
        
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            cur_c5 = trial_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[oi] = 1
            h[zi] = 0
    
    return best_h, best_c5

                if 0 <= j_plus < n:

                pass

        # For lag > 0: c[lag] = sum_{i=0}^{n-lag-1} h[i+lag] * (1-h[i])
        #   d/dh[j]: if j >= lag: (1-h[j-lag]); if j < n-lag: -h[j+lag]
        # For lag < 0, let l = -lag: c[-l] = sum_{i=l}^{n-1} h[i] * (1-h[i-l])
        #   d/dh[j]: if j >= l: (1-h[j-l]); if j < n-l: -h[j+l]
        # For lag = 0: c[0] = sum h[i](1-h[i])
        #   d/dh[j] = 1 - 2h[j]
        
        grad = np.zeros(n)
        for t_idx, w in zip(top_idx, weights):
            lag = int(t_idx) - center
            if lag == 0:
                grad += w * (1.0 - 2.0 * h)
            elif lag > 0:
                # c = sum_{i=0}^{n-lag-1} h[i+lag]*(1-h[i])
                # d/dh[j] for j >= lag: (1-h[j-lag])
                # d/dh[j] for j < n-lag: -h[j+lag]
                if n - lag > 0:
                    grad[lag:] += w * (1.0 - h[:n-lag])
                    grad[:n-lag] += w * (-h[lag:])
            else:
                l = -lag
                # c = sum_{i=l}^{n-1} h[i]*(1-h[i-l])
                # d/dh[j] for j >= l: (1-h[j-l])
                # d/dh[j] for j < n-l: -h[j+l]
                grad[l:] += w * (1.0 - h[:n-l])
                grad[:n-l] += w * (-h[l:])
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        # Backtracking line search
        step = 0.05 / (1.0 + 0.01 * it)
        improved = False
        for _ in range(30):
            h_trial = project_continuous(h - step * grad, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                h = h_trial
                cur_c5 = c5_trial
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
            step *= 0.5

            noise = rng.normal(0, T, n)
            h_trial = project_continuous(h + noise, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 + T:
                h = h_trial
                cur_c5 = c5_trial
    
    return best_h, best_c5

For the "full" correlation: np.correlate(a, b, mode="full") has length 2n-1.
The k-th element (0-indexed) is:
out[k] = sum_i a[i] * b[i - (k - (n-1))]

where the sum is over all i such that both i and i-(k-(n-1)) are valid indices (0 to n-1).

Let d = k - (n-1). Then:
out[k] = sum_i a[i] * b[i-d]

For our case, a = h, b = 1-h:
out[k] = sum_i h[i] * (1 - h[i-d])

The valid range of i: 0 ≤ i ≤ n-1 and 0 ≤ i-d ≤ n-1, so:
max(0, d) ≤ i ≤ min(n-1, n-1+d)

If d ≥ 0: i ranges from d to n-1, and i-d ranges from 0 to n-1-d
  out[k] = sum_{i=d}^{n-1} h[i] * (1-h[i-d])
  = sum_{j=0}^{n-1-d} h[j+d] * (1-h[j]) where j = i-d

If d < 0, let l = -d > 0: i ranges from 0 to n-1-l
  out[k] = sum_{i=0}^{n-1-l} h[i] * (1-h[i+l])

For d ≥ 0 (k ≥ n-1):
  out[k] = sum_{j=0}^{n-1-d} h[j+d] * (1-h[j])

  Wait, let me be more careful.
  
  out[k] = sum_{m=0}^{n-1-d} h[m+d] * (1-h[m])
  
  d/dh[j]: 
  - h[j] appears as h[m+d] when m+d = j, i.e., m = j-d. Valid when 0 ≤ j-d ≤ n-1-d, i.e., d ≤ j ≤ n-1.

    # d_max = max(oi, zi, n-1-oi, n-1-zi)

    # h[oi-d] non-zero when h[oi-d] != 0, which is when oi-d is in [0,n-1] AND h[oi-d] != 0
    # For simplicity, let's just compute the full range
    
    # Actually, let me just compute the delta for all k in the affected range
    d_min = max(0, oi+1, zi+1) - (n-1)  # This isn't right either

    # d ranges from -(n-1) to (n-1)
    # For each d, the delta is:
    #   (h[oi-d] if 0<=oi-d<n else 0) - (h[zi-d] if 0<=zi-d<n else 0) + (h[oi+d] if 0<=oi+d<n else 0) - (h[zi+d] if 0<=zi+d<n else 0)
    
    # But I only need to update where the delta is non-zero
    # The delta is non-zero only when at least one of the indices is in range
    # And the values are non-zero (but for binary, non-zero means = 1)

    affected_d = set()
    for d_candidate in range(oi-(n-1), oi+1):
        if -(n-1) <= d_candidate <= n-1:
            affected_d.add(d_candidate)
    for d_candidate in range(zi-(n-1), zi+1):

            affected_d.add(d_candidate)
    for d_candidate in range(-oi, n-oi):

            affected_d.add(d_candidate)
    for d_candidate in range(-zi, n-zi):
        if -(n-1) <= d_candidate <= n-1:
            affected_d.add(d_candidate)

        k = (n-1) + d
        delta = 0.0
        if 0 <= oi-d < n:
            delta += h[oi-d]
        if 0 <= zi-d < n:
            delta -= h[zi-d]
        if 0 <= oi+d < n:
            delta += h[oi+d]
        if 0 <= zi+d < n:
            delta -= h[zi+d]
        corr[k] += delta
    
    # Update h
    h[oi] = 0.0
    h[zi] = 1.0
    return corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:target]] = 1.0
    return result

def project_continuous(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def sa_binary(h_bin, n_steps, T_start, T_end, rng):
