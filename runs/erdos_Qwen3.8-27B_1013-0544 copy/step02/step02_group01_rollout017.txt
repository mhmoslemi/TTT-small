def project_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:

        if free.sum() > 0 and abs(diff) <= free.sum() * 1.0:
            h[free] = h[free] + diff / free.sum()
        else:
            h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        # Final adjustment

    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)

        frac = t / max(max_iters - 1, 1)
        beta = betas[0] + (betas[1] - betas[0]) * frac

        x = x_new
    
    return best_x, best_v

    n = len(h)
    target = n / 2.0

        v, _ = smooth_max_grad(x, beta)
        return v

        _, g = smooth_max_grad(x, beta)
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))

            y = x + mu * (x - x_prev)

    if abs(diff) > 1e-12:
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:

        h = np.clip(h, 0.0, 1.0)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr_full(h)) * (2.0 / n))

def project_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
