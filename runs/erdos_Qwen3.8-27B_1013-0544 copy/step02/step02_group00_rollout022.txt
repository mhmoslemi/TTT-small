            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_fn(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def con_fn(x_in, target_sum):
    return np.sum(x_in) - target_sum

def con_jac_fn(x_in):
    return np.ones(x_in.shape)

def run_slsqp(h, target_sum, beta, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj_fn, h, args=(beta, dx), method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fn, 'args': (target_sum,), 'jac': con_jac_fn}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(res.x, target_sum)
    except Exception:
        return h

_g = {}

def _one_d(v):
    h_base = _g['h']
    j = _g['j']
    tgt = _g['tgt']
    b = _g['beta']
    h_t = h_base.copy()
    h_t[j] = v
    h_t = project_h(h_t, tgt)
    c = np.correlate(h_t, 1.0 - h_t, mode="full")
    return smooth_max(c, b)

def coord_descent(h, target_sum, n_sweeps, beta):
    n = len(h)
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            _g['h'] = h
            _g['j'] = j
            _g['tgt'] = target_sum
            _g['beta'] = beta
            f_cur = _one_d(h[j])
            try:
                res = minimize_scalar(_one_d, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-8, 'maxiter': 40})
                if res.fun < f_cur - 1e-12:
                    h[j] = res.x
                    h = project_h(h, target_sum)
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h

def resample(h_old, n_new, target_sum):
    x_old = np.linspace(0, 2, len(h_old))
    x_new = np.linspace(0, 2, n_new)
    return project_h(np.interp(x_new, x_old, h_old), target_sum)

def make_inits(n, target_sum, h_best, seed):
    rng = np.random.RandomState(seed)
    inits = []
    if h_best is not None and len(h_best) != n:
        inits.append(resample(h_best, n, target_sum))
    elif h_best is not None:
        inits.append(h_best.copy())
    for amp in [0.25, 0.35, 0.45]:
        for freq in [2, 3, 5, 7, 11, 13]:
            s = 0.5 + amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n)
            inits.append(project_h(s, target_sum))
    for period in [2, 3, 4, 5, 6, 7]:
        pat = np.zeros(period)
        pat[::2] = 0.6
        pat[1::2] = 0.4
        alt = np.tile(pat, n // period + 1)[:n]
        inits.append(project_h(alt, target_sum))
    for _ in range(4):
        r = rng.rand(n) * 0.3 + 0.35
        inits.append(project_h(r, target_sum))
    return inits

def refine(h, target_sum, time_budget):
    """Iteratively apply SLSQP + coord descent until no improvement or time out."""
    start = time.time()
    n = len(h)
    cur_c5 = compute_c5(h)
    betas = [50, 150, 400, 800, 1500]
    for beta in betas:
        if time.time() - start > time_budget:
            break
        h = run_slsqp(h, target_sum, beta, 500, 1e-15)
        new_c5 = compute_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
    for _ in range(3):
        if time.time() - start > time_budget:
            break
        h = coord_descent(h, target_sum, 5, 500)
        new_c5 = compute_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
    return h, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 140
        h_parent = np.full(n_parent, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_parent

    n_candidates = sorted(set([n_parent, 100, 120, 140, 160, 200]))

    for n in n_candidates:
        tl = budget_s - (time.time() - start_time)
        if tl < 50:
            break
        target_sum = n / 2.0
        inits = make_inits(n, target_sum, best_h, seed + n)
        for idx, h_init in enumerate(inits):
            tl = budget_s - (time.time() - start_time)
            if tl < 20:
                break
            cur = h_init.copy()
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n
            alloc = min(tl - 10, 60)
            if alloc > 10:
                cur, cur_c5 = refine(cur, target_sum, alloc)
                if cur_c5 < best_c5:
                    best_h, best_c5, best_n = cur.copy(), cur_c5, n

    # Final intensive refinement
    n_f = len(best_h)
    target_f = n_f / 2.0
    tl = budget_s - (time.time() - start_time)
    if tl > 30:
        best_h, best_c5 = refine(best_h, target_f, tl - 15)
    
    tl = budget_s - (time.time() - start_time)
    if tl > 15:
        best_h = run_slsqp(best_h, target_f, 2000, 5000, 1e-16)
        best_c5 = compute_c5(best_h)
    
    tl = budget_s - (time.time() - start_time)
    if tl > 10:
        best_h = coord_descent(best_h, target_f, 20, 2000)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
