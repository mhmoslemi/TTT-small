import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
    return np.clip(h, 0.0, 1.0)

def grad_active(h, top_k=5):
    """Compute subgradient from top-k active constraints."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:top_k]
    g = np.zeros(n)
    for m in idx:
        s = m - (n - 1)
        for i in range(n):
            if 0 <= i + s < n:
                g[i] += (1.0 - h[i + s])
            if 0 <= i - s < n:
                g[i] -= h[i - s]
    g /= len(idx)
    return g, corr, idx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Subgradient descent with simulated annealing
    alpha = 0.05  # Initial step size
    T = 0.01  # Initial temperature
    top_k = 5
    
    for it in range(50000):
        if time_left() < 300:  # Reserve 300s for refinement
            break
        
        # Compute subgradient
        g, corr, active_idx = grad_active(best_h, top_k)
        
        # Remove mean to preserve sum constraint approximately
        g -= np.mean(g)
        
        # Step
        h_new = best_h - alpha * g
        h_new = project_h(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        # Simulated annealing acceptance
        if c5_new < best_c5 or rng.random() < np.exp(-(c5_new - best_c5) / T):
            best_h = h_new
            best_c5 = c5_new
            alpha = max(0.001, alpha * 0.9999)
        else:
            alpha = max(0.001, alpha * 0.999)
        
        # Cooling
        T = max(0.0001, T * 0.9995)
        
        # Periodic large perturbation
        if it % 500 == 250:
            h_pert = best_h + rng.randn(n_points) * 0.05
            h_pert = project_h(h_pert, target_sum)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5 or rng.random() < np.exp(-(c5_pert - best_c5) / T):
                best_h = h_pert
                best_c5 = c5_pert
                T = 0.005  # Reheat
    
    # Phase 2: SLSQP refinement
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    
    def make_obj(beta):
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            m = np.max(corr)
            return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx
        return obj
    
    def cfun(x):
        return np.sum(x) - target_sum
    
    def cjac(x):
        return np.ones(x.shape[0])
    
    constraints = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    
    for beta in [50, 100, 200, 500, 1000, 2000]:
        if time_left() < 30:
            break
        obj = make_obj(beta)
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': 5000, 'ftol': 1e-15})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass
    
    # Phase 3: Multiple restarts from perturbed best
    for attempt in range(20):
        if time_left() < 20:
            break
        
        h0 = best_h + rng.randn(n_points) * 0.02
        h0 = project_h(h0, target_sum)
        
        # Quick subgradient
        for _ in range(200):
            g, corr, _ = grad_active(h0, top_k=5)
            g -= np.mean(g)
            h0 = project_h(h0 - 0.02 * g, target_sum)
        
        # SLSQP

        try:
            res = minimize(obj, h0, method='SLSQP', bounds=bounds,

                          options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass
    
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
    return np.clip(h, 0.0, 1.0)

def grad_c5(h, top_k=3):
    """Compute weighted subgradient of max correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Get top k indices
    idx = np.argsort(corr)[::-1][:top_k]
    g = np.zeros(n)
    for m in idx:
        s = m - (n - 1)
        for i in range(n):
            if 0 <= i + s < n:
                g[i] += (1.0 - h[i + s])
            if 0 <= i - s < n:
                g[i] -= h[i - s]
    g /= len(idx)
    return g, corr

def smooth_max_obj(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx

    """Optimize h[i] given all other h[j] fixed."""
    # c[m] = sum_j h[j](1-h[j+s]) where s = m - (n-1)
    # Terms involving h[i]:
    #   h[i] appears as h[j] when j=i (if i+s valid): contributes h[i]*(1-h[i+s])
    #   h[i] appears as h[j+s] when j=i-s (if i-s valid): contributes h[i-s]*(1-h[i])

    # where a[m] = c[m] with h[i] removed
    # and b[m] = coefficient of h[i]

    # Using minimize_scalar
    
    def f(x):
        return np.max(a + result_b * x)
    
    from scipy.optimize import minimize_scalar
    res = minimize_scalar(f, bounds=(0, 1), method='bounded', options={'xatol': 1e-10})
    return res.x

import numpy as np
from scipy.optimize import minimize, minimize_scalar, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
    return np.clip(h, 0.0, 1.0)

def grad_c5_vec(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:top_k]
    g = np.zeros(n)
    for m in idx:
        s = m - (n - 1)
        if s == 0:
            g += (1.0 - 2.0 * h)
        elif s > 0:
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
        else:
            g[-s:] += (1.0 - h[:n+s])
            g[:n+s] -= h[-s:]
    g /= len(idx)
    return g, corr

def optimize_coord(h, i, n):
    """Find optimal h[i] given all other h fixed."""
    b = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        val = 0.0
        if 0 <= i + s < n:
            val += 1.0 - h[i + s]
        if 0 <= i - s < n:
            val -= h[i - s]
        b[m] = val
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    a = corr - b * h[i]
    
    def f(x):
        return np.max(a + b * x)
    
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def coordinate_descent(h, n, ts, max_sweeps=50):
    """Coordinate descent with sum constraint projection."""
    h = h.copy()
    prev_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            new_hi = optimize_coord(h, i, n)
            if abs(new_hi - h[i]) > 1e-10:
                h[i] = new_hi
                h = project_h(h, ts)
                improved = True
        c5 = compute_c5(h)
        if not improved or c5 >= prev_c5:
            break
        prev_c5 = c5
    
    return h

def slsqp_objective(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx

def sum_con_fun(x, ts):
    return np.sum(x) - ts

def sum_con_jac(x, ts):
    return np.ones(x.shape[0])

    lr = 0.03
    T = 0.005
    stall = 0
    while tl() > 250:
        g, corr = grad_c5_vec(best_h, top_k=3)
        g -= np.mean(g)
        h_new = project_h(best_h - lr * g, ts)
        c5_new = compute_c5(h_new)
        if c5_new <= best_c5:
            best_h = h_new
            best_c5 = c5_new
            stall = 0
            lr = min(0.05, lr * 1.0001)
        else:
            if rng.random() < np.exp(-(c5_new - best_c5) / T):
                best_h = h_new
            lr = max(0.002, lr * 0.9999)
            stall += 1
        T = max(1e-5, T * 0.99995)
        if stall > 250:
            h_pert = best_h + rng.randn(n) * 0.05
            h_pert = project_h(h_pert, ts)
            h_pert = coord_descent(h_pert, n, ts, 15, rng)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_h, best_c5 = h_pert, c5_pert
            T = 0.004
            lr = 0.03
            stall = 0

    # Phase 3: SLSQP refinement
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (ts,)}]

    for beta in [50, 100, 200, 500, 1000, 3000, 5000]:
        if tl() < 25:
            break
        try:
            res = minimize(slsqp_objective, best_h, args=(beta, dx),
                           method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 5000, 'ftol': 1e-15})
            h_new = project_h(res.x, ts)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 4: Coordinate descent polish + multi-start
    h_cd2 = coord_descent(best_h, n, ts, 40, rng)
    c5_cd2 = compute_c5(h_cd2)
    if c5_cd2 < best_c5:
        best_h, best_c5 = h_cd2, c5_cd2

    for attempt in range(40):
        if tl() < 8:
            break
        strat = attempt % 4
        if strat == 0:
            h0 = best_h + rng.randn(n) * 0.012
        elif strat == 1:
            h0 = best_h.copy()
            idx = rng.choice(n, n // 5, replace=False)
            h0[idx] += 0.03 * rng.randn(len(idx))
        elif strat == 2:
            h0 = 0.85 * best_h + 0.15 * rng.uniform(0.2, 0.8, n)
        else:
            h0 = best_h[::-1] + rng.randn(n) * 0.01
        h0 = project_h(h0, ts)
        h0 = coord_descent(h0, n, ts, 25, rng)
        c5_0 = compute_c5(h0)
        if c5_0 < best_c5:
            best_h, best_c5 = h0, c5_0
            if tl() > 15:
                try:
                    res = minimize(slsqp_objective, best_h, args=(2000, dx),
                                   method='SLSQP', bounds=bounds,
                                   constraints=constraints,
                                   options={'maxiter': 3000, 'ftol': 1e-15})
                    h_new = project_h(res.x, ts)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new.copy(), c5
                except Exception:
                    pass
                h_cd3 = coord_descent(best_h, n, ts, 20, rng)
                c5_cd3 = compute_c5(h_cd3)
                if c5_cd3 < best_c5:
                    best_h, best_c5 = h_cd3, c5_cd3

    best_h = project_h(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
