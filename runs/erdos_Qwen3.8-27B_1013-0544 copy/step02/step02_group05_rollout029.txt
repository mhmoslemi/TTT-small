def proper_project(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = beta * (corr - m)
    es = np.exp(sh)
    s = np.sum(es)
    f_val = (m + np.log(s) / beta) * dx
    w = es / s
    grad = np.zeros(n)
    for p in range(n):
        lo1 = max(p - n + 1, -(n - 1))
        hi1 = min(p, n - 1)
        if lo1 <= hi1:
            ks = np.arange(lo1, hi1 + 1)
            grad[p] += np.dot(w[ks + (n - 1)], 1.0 - h[p - ks])
        lo2 = max(-p, -(n - 1))
        hi2 = min(n - p - 1, n - 1)
        if lo2 <= hi2:
            ks = np.arange(lo2, hi2 + 1)
            grad[p] -= np.dot(w[ks + (n - 1)], h[p + ks])
    grad *= dx
    return f_val, grad

def pgd_search(h0, beta, n_iter, step, momentum, target):
    n = len(h0)
    h = np.array(h0, dtype=float)
    y = h.copy()
    best_local = compute_c5(h)
    stale = 0
    for it in range(n_iter):
        lr = step / (1.0 + 0.003 * it)
        f, g = obj_grad(y, beta)
        h_new = proper_project(y - lr * g, target)
        y = h_new + momentum * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        if it % 500 == 0:
            c = compute_c5(h)
            if c < best_local - 1e-6:
                best_local = c
                stale = 0
            else:
                stale += 1
                if stale > 8:
                    break
    return h, compute_c5(h)

def slsqp_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    b = 300.0
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(b * (corr - m)))) / b) * dx

def slsqp_grad(h):
    return obj_grad(h, 300.0)[1]

def slsqp_confun(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_conjac(x):
    return np.ones(len(x))

def slsqp_refine(h0, max_iter=300):
    n = len(h0)
    bds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': slsqp_confun, 'jac': slsqp_conjac}]
    try:
        res = minimize(slsqp_obj, h0, jac=slsqp_grad, method='SLSQP',
                       bounds=bds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        h_new = proper_project(res.x, n / 2.0)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)

def make_starts(n, parent, rng):
    target = n / 2.0
    starts = []
    starts.append(np.full(n, 0.5))
    if parent is not None:
        if len(parent) == n:
            starts.append(parent.copy())
            starts.append(1.0 - parent)
        else:
            xo = np.linspace(0, 2, len(parent), endpoint=False)
            xn = np.linspace(0, 2, n, endpoint=False)
            pi = proper_project(np.interp(xn, xo, parent, period=2.0), target)
            starts.append(pi)
            starts.append(1.0 - pi)
        # Circular shifts
        if len(parent) == n:
            for shift in [n // 4, n // 3, n // 2, 2 * n // 3]:
                starts.append(proper_project(np.roll(parent, shift), target))
    # Random
    for _ in range(2):
        starts.append(proper_project(rng.random(n), target))
    # Block pattern with noise
    half = n // 2
    blk = np.zeros(n)
    blk[:half] = 1.0
    starts.append(proper_project(blk + rng.randn(n) * 0.15, target))
    # Triangular wave
    t = np.linspace(0, 2, n, endpoint=False)
    tri = 1.0 - 2.0 * np.abs(t - 1.0)  # peaks at t=1
    tri = 0.5 + 0.5 * tri  # map to [0.5, 1]... actually let me think
    # Actually: values in [0, 1]
    starts.append(proper_project(tri, target))
    return starts
