import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() == 0:
            mask = (h > 0.0) & (h < 1.0)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_and_grad(h, beta, n):
    """Compute smooth max of corr(h) and its gradient w.r.t. h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted -= np.max(shifted)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    sm = m + np.log(np.sum(exps)) / beta

    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        wm = w[mi]
        if wm < 1e-14:
            continue
        k = mi - (n - 1)
        if k >= 0:
            if k < n:
                grad[k:n] += wm * (1.0 - h[0:n - k])
            if n - 1 - k >= 0:
                grad[0:n - k] -= wm * h[k:n]
        else:
            kk = -k
            grad[0:n - kk] += wm * (1.0 - h[kk:n])
            grad[kk:n] -= wm * h[0:n - kk]

    return sm, (2.0 / n) * grad


def make_obj(n, beta, lam):
    def fn(h):
        sm, _ = smooth_max_and_grad(h, beta, n)
        return sm + lam * (h.sum() - n / 2.0) ** 2
    def jac(h):
        _, g = smooth_max_and_grad(h, beta, n)
        g += 2.0 * lam * (h.sum() - n / 2.0)
        return g
    return fn, jac


def lbfgsb_step(h, n, beta, lam, maxit=300):
    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n
    fn, jac = make_obj(n, beta, lam)
    r = minimize(fn, h, jac=jac, method='L-BFGS-B', bounds=bnds,
                 options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-14})
    return project_box_sum(r.x, ts)


def diverse_starts(n, rng):
    """Generate diverse initial solutions for given n."""
    starts = []
    t = np.linspace(0, 2.0, n, endpoint=False) + 1.0 / n
    # Pure sinusoids
    for freq in range(1, 6):
        for amp in [0.2, 0.3, 0.4, 0.5]:
            h = 0.5 + amp * np.sin(np.pi * freq * t + rng.uniform(0, 2 * np.pi))
            starts.append(project_box_sum(h, n / 2.0))
    # Two-tone
    for _ in range(5):
        f1 = rng.uniform(1, 4)
        f2 = rng.uniform(3, 8)
        a1 = rng.uniform(0.1, 0.35)
        a2 = rng.uniform(0.05, 0.25)
        h = 0.5 + a1 * np.sin(np.pi * f1 * t) + a2 * np.sin(np.pi * f2 * t)
        starts.append(project_box_sum(h, n / 2.0))
    # Block patterns
    for nb in range(2, 8):
        h = np.tile(rng.uniform(0.1, 0.9, nb), n // nb + 1)[:n]
        starts.append(project_box_sum(h, n / 2.0))
    # Random
    for _ in range(3):
        h = rng.uniform(0, 1, n)
        starts.append(project_box_sum(h, n / 2.0))
    return starts


def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    # Initial from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:

    bn = len(best_h)

    best_c5 = compute_c5(best_h)

    # Try multiple resolutions

    n_list = [nv for nv in n_list if nv <= 140]

    candidates = [(best_h.copy(), bn)]

    for nv in n_list:

            break

        starts = diverse_starts(nv, rng)

    # Phase 3: Diverse restarts with full optimization
    stall_count = 0
    max_stall = 15
    it = 0
    while tl() > 80:
        it += 1
        h = best_h.copy()
        bn = len(h)
        strat = it % 7

        if strat == 0:
            scale = max(0.005, 0.04 * np.exp(-it / 30.0))
            h += rng.randn(bn) * scale
        elif strat == 1:

            top = np.argsort(corr)[-4:]
            for mi in top:
                k = mi - (bn - 1)
                phase = 2 * np.pi * abs(k) * np.arange(bn) / bn
                h += 0.008 * np.sin(phase + rng.uniform(0, 2 * np.pi))
        elif strat == 2:
            blen = rng.randint(3, max(4, bn // 7))
            bs = rng.randint(0, bn - blen)
            h2 = h.copy()
            h2[bs:bs + blen] = 1.0 - h[bs:bs + blen]
            h = h2 + rng.randn(bn) * 0.004
        elif strat == 3:

            wfreq = rng.uniform(1, 5) * 2 * np.pi / bn
            h += 0.012 * np.sin(wfreq * np.arange(bn) + rng.uniform(0, 2 * np.pi))
        elif strat == 4:
            # Mixture with random
            alpha = rng.uniform(0.6, 0.9)
            h = alpha * best_h + (1 - alpha) * rng.uniform(0, 1, bn)
        elif strat == 5:

            corr = np.correlate(h, 1.0 - h, mode="full")
            k_star = int(np.argmax(corr)) - (bn - 1)
            pert = np.zeros(bn)
            for j in range(bn):
                other = (j + k_star) % bn
                pert[j] = 0.01
                pert[other] = -0.01
            h += pert
        else:
            # Random frequency sweep
            for _ in range(3):
                fw = rng.uniform(0.5, 8.0) * 2 * np.pi / bn
                h += 0.006 * np.sin(fw * np.arange(bn) + rng.uniform(0, 2 * np.pi))

        for beta, maxit in [(150, 200), (400, 300)]:
            if tl() < 25:

            lam = 500.0

                if nv_alt == bn or tl() < 40:
