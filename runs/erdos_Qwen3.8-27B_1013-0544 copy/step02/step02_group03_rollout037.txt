        cons = [{'type': 'eq', 'fun': slsqp_wrap_con, 'jac': slsqp_wrap_jac, 'args': (target,)}]
        try:
            res = minimize(slsqp_wrap_obj, best_h, args=(dx, 50000),
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
