def compute_smooth_grad(h, beta, n):
    """Compute gradient of smooth max of correlation values."""
    corr = compute_corr(h)  # length 2n-1
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= np.sum(weights)
    # Only process lags with significant weight
    threshold = 1.0 / (n * 100)  # significant if weight > 1/(100n)
    active = np.where(weights > threshold)[0]
    grad = np.zeros(n)

        w = weights[idx]

            grad[k:n] += w * (1.0 - h[0:n-k])

            grad[0:n-m_] += w * (1.0 - h[m_:n])
            grad[m_:n] += w * (-h[0:n-m_])
    return grad, m

                    best_h = h_trial
                    best_c5 = c5_trial

    best_h = project_h(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
