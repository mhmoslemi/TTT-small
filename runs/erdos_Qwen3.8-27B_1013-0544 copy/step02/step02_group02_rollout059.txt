```python
def compute_gradient(h, n):
    """Compute G[m, i] = dC[m]/dh[i] where C = correlate(h, 1-h, 'full')"""
    n_corr = 2 * n - 1
    G = np.zeros((n_corr, n))
    # s = m - (n-1), m = 0, ..., 2n-2, so s = -(n-1), ..., n-1
    for m in range(n_corr):
        s = m - (n - 1)
        # First term: (1-h[i+s]) for valid i+s
        # i+s ranges from s to n-1+s, valid when 0<=i+s<=n-1
        # So i ranges from max(0, -s) to min(n-1, n-1-s)
        i_start = max(0, -s)
        i_end = min(n-1, n-1-s)
        if i_start <= i_end:
            G[m, i_start:i_end+1] += (1.0 - h[i_start+s:i_end+s+1])
        # Second term: -h[i-s] for valid i-s
        # i-s ranges from -s to n-1-s, valid when 0<=i-s<=n-1
        # So i ranges from max(0, s) to min(n-1, n-1+s)
        i_start2 = max(0, s)
        i_end2 = min(n-1, n-1+s)
        if i_start2 <= i_end2:
            G[m, i_start2:i_end2+1] -= h[i_start2-s:i_end2-s+1]
    return G
```

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-15:
                break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-15:
                break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        return m + np.log(np.sum(np.exp(shifted))) / alpha
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
