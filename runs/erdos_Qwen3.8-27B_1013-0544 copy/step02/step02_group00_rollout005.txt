import numpy as np
from scipy.optimize import minimize, Bounds, differential_evolution
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta


def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def topk_sum_obj(h, dx, k=5):
    corr = compute_corr(h) * dx
    srt = np.sort(corr)[::-1][:k]
    return np.sum(srt)


def active_subgrad(h, n_top=3):
    corr = compute_corr(h)
    n = len(h)
    active = np.argsort(corr)[::-1][:n_top]
    grad = np.zeros(n)
    for m in active:
        d = m - (n - 1)
        for j in range(n):
            j2 = j + d
            if 0 <= j2 < n:
                grad[j] += (1.0 - h[j2])
            j1 = j - d
            if 0 <= j1 < n:
                grad[j] -= h[j1]
    return grad
