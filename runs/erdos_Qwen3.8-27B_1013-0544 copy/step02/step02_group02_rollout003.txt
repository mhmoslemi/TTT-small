import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15: break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15: break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha, lam, rho):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
    soft = m + logsum / alpha
    sv = h.sum() - n / 2.0
    return soft + lam * sv + 0.5 * rho * sv * sv

    h = h.copy()
    lam = 0.0
    rho = 1.0
    alpha = alpha_start

        if t0 is not None and time.time() - t0 > budget_s - 30:

                      method='L-BFGS-B', bounds=bounds,

        sv = h.sum() - n / 2.0

        alpha = min(alpha * 3.0, 1e8)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, n, alpha, lam, rho):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    g_max = g.max()
    shifted = alpha * (g - g_max)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    soft = g_max + np.log(w_sum) / alpha
    w /= w_sum
    # Gradient of soft-max w.r.t. h
    grad = np.zeros(n)
    n_half = n - 1
    for si in range(len(g)):
        wi = w[si]
        if wi < 1e-16:
            continue
        s = si - n_half
        if s >= 0:
            if s < n:
                vi = np.arange(n - s)
                grad[vi] += wi * dx * (1.0 - h[vi + s])
                vi2 = np.arange(s, n)
                grad[vi2] += wi * (-dx) * h[vi2 - s]
        else:
            sp = -s
            if sp < n:
                vi = np.arange(sp, n)
                grad[vi] += wi * dx * (1.0 - h[vi - sp])
                vi2 = np.arange(0, n - sp)
                grad[vi2] += wi * (-dx) * h[sp + vi2]
    sv = h.sum() - n / 2.0
    pen_val = lam * sv + 0.5 * rho * sv * sv
    pen_grad = (lam + rho * sv) * np.ones(n)
    return soft + pen_val, grad + pen_grad

def aug_lag_opt(h, n, alpha_start, n_outer, t0, budget_s, min_time_left=30):
    h = h.copy()
    lam = 0.0
    rho = 1.0
    alpha = float(alpha_start)
    bounds = [(0.0, 1.0)] * n
    best_local = h.copy()
    best_c5_local = compute_c5(h, n)
    for outer in range(n_outer):
        if time.time() - t0 > budget_s - min_time_left:
            break
        res = minimize(lambda x: smooth_obj_and_grad(x, n, alpha, lam, rho)[0],
                      h, jac=lambda x: smooth_obj_and_grad(x, n, alpha, lam, rho)[1],
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 1000, 'ftol': 1e-25, 'maxfun': 50000,
                               'gtol': 1e-14})
        h = np.clip(res.x, 0.0, 1.0)
        h = project_feasible(h, n)
        sv = h.sum() - n / 2.0
        lam += rho * sv
        if abs(sv) > 1e-6:
            rho = min(rho * 2.0, 1e10)
        else:
            rho = max(rho * 0.5, 1.0)
        alpha = min(alpha * 2.5, 1e8)
        c5 = compute_c5(h, n)
        if c5 < best_c5_local:
            best_local = h.copy()
            best_c5_local = c5
    return best_local
