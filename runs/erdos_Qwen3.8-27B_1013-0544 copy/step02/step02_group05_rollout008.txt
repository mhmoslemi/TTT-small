def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_polytope(h, target):
    """Project h onto {0 <= x <= 1, sum(x) = target} via bisection on offset."""
    n = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def corr_values(h):
    """Return all correlation values (not scaled by dx)."""
    return np.correlate(h, 1.0 - h, mode="full")

def grad_corr_k(h, k_offset):
    """Gradient of correlation at a given lag offset.
    k_offset is the index into the 'full' correlation array (0 to 2n-2).
    The actual lag is k_offset - (n-1).
    """
    n = len(h)
    lag = k_offset - (n - 1)
    # d(c_k)/d(h[i]) = (1 - h[i-k]) - h[i+k] (with OOB = 0)
    grad = np.zeros(n)
    # (1 - h[i-lag]) part: for i where 0 <= i-lag < n
    i_start = max(0, lag)
    i_end = min(n, n + lag)
    if i_start < i_end:
        idx = np.arange(i_start, i_end)
        src = idx - lag
        grad[idx] += (1.0 - h[src])
    # -h[i+lag] part: for i where 0 <= i+lag < n
    j_start = max(0, -lag)
    j_end = min(n, n - lag)
    if j_start < j_end:
        idx = np.arange(j_start, j_end)
        src = idx + lag
        grad[idx] -= h[src]
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

            h = project_polytope(h + np.random.randn(n) * 0.04, target_sum)
            no_improve = 0
            alpha = 0.02
    return best_local, best_c5_local

def slsqp_polish(h, n, target_sum, time_fn, time_limit):
    if time_fn() > time_limit:
        return h, compute_c5(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj_func(x):
        cv = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(cv, 300.0) * dx
    def con_fun(x):
        return np.sum(x) - target_sum
    def con_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj_func, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': 800, 'ftol': 1e-15})
        h_new = project_polytope(res.x, target_sum)
        c5 = compute_c5(h_new)
        if c5 < compute_c5(h):
            return h_new, c5
    except Exception:
        pass
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    def remaining():
        return budget_s - elapsed()

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 70
        h_init = np.full(n_init, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    best_n = n_init

    # Phase 1: Subgradient optimization at initial resolution
    n1 = n_init
    target1 = n1 / 2.0
    t_alloc1 = budget_s * 0.40
    h1, c5_1 = subgradient_optimize(h_init, n1, target1, 30000, elapsed, t_alloc1,
                                     w_scale=10.0, alpha_init=0.025)
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c5_1, n1

    # Phase 2: SLSQP polish at initial resolution
    if remaining() > 80:
        h1, c5_1 = slsqp_polish(best_h, n1, target1, elapsed, budget_s - 80)
        if c5_1 < best_c5:
            best_h, best_c5, best_n = h1.copy(), c5_1, n1

    # Phase 3: Multi-restart exploration
    if remaining() > 60:
        for restart in range(8):
            if remaining() < 60:
                break
            if restart == 0:
                h_r = best_h + np.random.randn(n1) * 0.015
            elif restart == 1:
                # Structured: blocks
                block = n1 // 8
                pattern = np.tile([1.0, 1.0, 0.0, 0.0], n1 // 4 + 1)[:n1]
                h_r = 0.3 + 0.4 * pattern + np.random.randn(n1) * 0.1
            elif restart == 2:
                # Sine-based
                x = np.linspace(0, 2, n1, endpoint=False)
                h_r = 0.5 + 0.4 * np.sin(2 * np.pi * x) + np.random.randn(n1) * 0.05
            else:
                h_r = best_h + np.random.randn(n1) * (0.03 + 0.02 * restart * 0.3)
            h_r = project_polytope(h_r, target1)
            t_lim = budget_s - 50
            h_r, c5_r = subgradient_optimize(h_r, n1, target1, 15000, elapsed, t_lim,
                                              w_scale=12.0, alpha_init=0.015)
            if c5_r < best_c5:
                best_h, best_c5, best_n = h_r.copy(), c5_r, n1
                # Polish
                if remaining() > 30:
                    h_r, c5_r = slsqp_polish(h_r, n1, target1, elapsed, budget_s - 30)
                    if c5_r < best_c5:
                        best_h, best_c5, best_n = h_r.copy(), c5_r, n1

    # Phase 4: Resolution expansion
    if remaining() > 100:
        n2 = n1 + 20
        target2 = n2 / 2.0
        x_old = np.linspace(0, 2, n1, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h, period=2.0)
        h2 = project_polytope(h2, target2)
        t_lim2 = budget_s - 50
        h2, c5_2 = subgradient_optimize(h2, n2, target2, 25000, elapsed, t_lim2,
                                         w_scale=10.0, alpha_init=0.018)
        if remaining() > 30:
            h2, c5_2 = slsqp_polish(h2, n2, target2, elapsed, budget_s - 30)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2.copy(), c5_2, n2

    # Phase 5: Final high-precision SLSQP
    if remaining() > 15:
        h_final, c5_final = slsqp_polish(best_h, best_n, best_n / 2.0, elapsed, budget_s - 5)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    return best_h, best_c5, best_n
