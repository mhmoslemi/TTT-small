import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    shifted -= np.max(shifted)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nfree = free.sum()
        if nfree == 0:
            break
        h[free] += diff / nfree
    return np.clip(h, 0.0, 1.0)

def obj_grad(h, beta, dx, target_sum, pen_weight):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Smooth max
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted -= np.max(shifted)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    
    F = (m + np.log(exp_shifted.sum()) / beta) * dx
    
    # Penalty for sum constraint
    sum_diff = h.sum() - target_sum
    F += 0.5 * pen_weight * sum_diff ** 2
    
    # Gradient
    grad_h = np.zeros(n)
    for offset in range(-(n-1), n):
        w = weights[offset + n - 1]
        if w < 1e-12:
            continue
        # First term: (1 - h[i+offset]) for valid i
        i_start = max(0, -offset)
        i_end = min(n, n - offset)
        if i_start < i_end:
            grad_h[i_start:i_end] += w * dx * (1.0 - h[i_start+offset:i_end+offset])
        # Second term: -h[i-offset] for valid i
        i_start2 = max(0, offset)
        i_end2 = min(n, n + offset)
        if i_start2 < i_end2:
            grad_h[i_start2:i_end2] -= w * dx * h[i_start2-offset:i_end2-offset]
    
    # Penalty gradient
    grad_h += pen_weight * sum_diff
    
    return F, grad_h

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    shifted -= np.max(shifted)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nfree = free.sum()
        if nfree == 0:
            break
        h[free] += diff / nfree
    return np.clip(h, 0.0, 1.0)

def obj_grad(h, beta, dx, target_sum, pen_weight):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted -= np.max(shifted)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    F = (m + np.log(exp_shifted.sum()) / beta) * dx
    sum_diff = h.sum() - target_sum
    F += 0.5 * pen_weight * sum_diff * sum_diff
    grad_h = np.zeros(n)
    for offset in range(-(n - 1), n):
        w = weights[offset + n - 1]
        if w < 1e-13:
            continue
        i1 = max(0, -offset)
        i2 = min(n, n - offset)
        if i1 < i2:
            grad_h[i1:i2] += w * dx * (1.0 - h[i1 + offset:i2 + offset])
        i3 = max(0, offset)
        i4 = min(n, n + offset)
        if i3 < i4:
            grad_h[i3:i4] -= w * dx * h[i3 - offset:i4 - offset]
    grad_h += pen_weight * sum_diff
    return F, grad_h

def lbfgs_refine(h, beta, dx, target_sum, pen_weight, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    def wrapper(x):
        return obj_grad(x, beta, dx, target_sum, pen_weight)
    try:
        res = minimize(wrapper, h, method='L-BFGS-B', bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                                'maxfun': maxiter * 20})
        return project_h(res.x, target_sum)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)
    pen_weight = 5000.0

    def tl():
        return budget_s - (time.time() - start_time)

    # Phase 1: L-BFGS-B with beta annealing
    betas = [5, 15, 50, 150, 500, 1500, 5000]
    for beta in betas:
        if tl() < 80:
            break
        mi = 800 if beta < 100 else 2000
        h_new = lbfgs_refine(best_h, beta, dx, target_sum, pen_weight, mi)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    # Phase 2: Coordinated coordinate descent + L-BFGS-B hybrid
    # Use coordinate descent to make fine adjustments
    def coord_pass(h, beta, dx, target_sum, pen_weight, n_rounds):
        n = len(h)
        for r in range(n_rounds):
            if tl() < 40:
                break
            for i in range(n):
                saved = h[i]
                lo = max(0.0, saved - 0.4)
                hi = min(1.0, saved + 0.4)
                def f1d(val):
                    h[i] = val
                    F, _ = obj_grad(h, beta, dx, target_sum, pen_weight)
                    return F
                try:
                    res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                          options={'xatol': 1e-12, 'maxiter': 50})
                    h[i] = res.x
                except Exception:
                    h[i] = saved
        return project_h(h, target_sum)

    # Interleave coordinate passes with L-BFGS-B
    cd_betas = [200, 500, 200, 1000, 500, 3000]
    for cb in cd_betas:
        if tl() < 60:
            break
        best_h = coord_pass(best_h, cb, dx, target_sum, pen_weight, 2)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5
        else:
            # Revert if worse
            best_h = lbfgs_refine(best_h, cb, dx, target_sum, pen_weight, 500)
            c5 = compute_c5(best_h)
            if c5 < best_c5:
                best_c5 = c5

    # Phase 3: Multi-start perturb + refine
    n_attempts = 0
    while tl() > 50:
        n_attempts += 1
        strat = n_attempts % 9
        h0 = best_h.copy()

        if strat == 0:
            scale = 0.008 + 0.025 * np.exp(-n_attempts / 20.0)
            h0 += rng.randn(n_points) * scale
        elif strat == 1:
            corr = np.correlate(h0, 1.0 - h0, mode="full")
            active = np.argmax(corr)
            offset = active - (n_points - 1)
            h0 = coord_pass(h0, 500, dx, target_sum, pen_weight, 3)
        elif strat == 2:
            i, j = rng.choice(n_points, 2, replace=False)
            d = 0.04 * rng.randn()
            h0[i] += d
            h0[j] -= d
        elif strat == 3:
            blen = rng.randint(3, 8)
            bs = rng.randint(0, n_points - blen)
            h0[bs:bs + blen] += rng.randn(blen) * 0.03
        elif strat == 4:
            h0 = 0.6 * h0 + 0.4 * h0[::-1]
            h0 += rng.randn(n_points) * 0.015
        elif strat == 5:
            t = np.arange(n_points)
            freq = rng.uniform(1.0, 8.0)
            h0 += 0.015 * np.sin(2.0 * np.pi * freq * t / n_points + rng.uniform(0, 6.28))
        elif strat == 6:
            h0 += rng.randn(n_points) * 0.05
        elif strat == 7:
            # Targeted: reduce the top-3 active correlations
            corr = np.correlate(h0, 1.0 - h0, mode="full")
            top3 = np.argsort(corr)[-3:]
            for ak in top3:
                offset = ak - (n_points - 1)
                for i in range(n_points):
                    if 0 <= i + offset < n_points:
                        h0[i] -= 0.005 * (1.0 - h0[i + offset])
                    if 0 <= i - offset < n_points:
                        h0[i] += 0.005 * h0[i - offset]
        else:
            # Sinusoidal mode perturbation
            k = rng.randint(1, 5)
            t = np.arange(n_points)
            h0 += 0.02 * np.cos(2.0 * np.pi * k * t / n_points + rng.uniform(0, 6.28))

        h0 = project_h(h0, target_sum)
        beta = [200, 500, 1000][n_attempts % 3]
        h_new = lbfgs_refine(h0, beta, dx, target_sum, pen_weight, 800)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5
            # Follow up with coordinate pass
            h_new = coord_pass(best_h, beta, dx, target_sum, pen_weight, 2)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    # Final high-precision polish
    if tl() > 30:
        for beta in [8000, 20000]:
            if tl() < 20:
                break
            h_new = lbfgs_refine(best_h, beta, dx, target_sum, pen_weight, 5000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
            h_new = coord_pass(best_h, beta, dx, target_sum, pen_weight, 3)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
