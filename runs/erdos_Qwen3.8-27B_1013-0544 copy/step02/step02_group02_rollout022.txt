def minmax_1d(alpha, beta):
    """Find t in [0,1] minimizing max_m (alpha[m]*t + beta[m]) using binary search."""
    pos = alpha > 1e-15
    neg = alpha < -1e-15
    
    val_at_0 = float(np.max(beta))
    val_at_1 = float(np.max(alpha + beta))
    v_lo = min(val_at_0, val_at_1)
    v_hi = max(val_at_0, val_at_1)
    
    if v_hi - v_lo < 1e-15:
        return 0.5
    
    for _ in range(45):
        v_mid = (v_lo + v_hi) / 2.0
        t_lo = 0.0
        t_hi = 1.0
        if np.any(pos):
            ub = (v_mid - beta[pos]) / alpha[pos]
            t_hi = min(float(np.min(ub)), 1.0)
        if np.any(neg):
            lb = (v_mid - beta[neg]) / alpha[neg]
            t_lo = max(float(np.max(lb)), 0.0)
        zero_mask = ~pos & ~neg
        if np.any(zero_mask) and float(np.max(beta[zero_mask])) > v_mid:
            t_lo = 1.0
        if t_lo <= t_hi:
            v_hi = v_mid
        else:
            v_lo = v_mid
    
    v_opt = v_hi
    t_lo = 0.0
    t_hi = 1.0
    if np.any(pos):
        ub = (v_opt - beta[pos]) / alpha[pos]
        t_hi = min(float(np.min(ub)), 1.0)
    if np.any(neg):
        lb = (v_opt - beta[neg]) / alpha[neg]
        t_lo = max(float(np.max(lb)), 0.0)
    t_opt = (t_lo + t_hi) / 2.0
    return float(max(0.0, min(1.0, t_opt)))

def cd_sweep(h, n):

    offset = np.arange(2 * n - 1) - (n - 1)

        # Compute alpha and beta
        valid1 = (j - offset >= 0) & (j - offset <= n - 1)
        valid2 = (j + offset >= 0) & (j + offset <= n - 1)
        
        alpha = np.zeros(2 * n - 1)
        if np.any(valid1):
            idx1 = j - offset[valid1]
            alpha[valid1] += (1.0 - h[idx1])
        if np.any(valid2):
            idx2 = j + offset[valid2]
            alpha[valid2] += (-h[idx2])
        
        c = np.correlate(h, 1.0 - h, mode='full')
        beta = c - alpha * h[j]
        
        t_opt = minmax_1d(alpha, beta)
        h[j] = t_opt
    
    return h

        c = compute_c5(h_cd, n)
        if c < best_c5:
            best_h, best_c5 = h_cd.copy(), c

    # Phase 6: Final SLSQP polish
    for alpha in [500000, 2000000]:
        if remaining() < 20:
            break
        ht = slsqp_optimize(best_h, n, alpha, 8000, 1e-18)
        c = compute_c5(ht, n)
        if c < best_c5:
            best_h, best_c5 = ht, c

    # Phase 7: Final CD polish
    if remaining() > 15:
        h_cd = best_h.copy()
        h_cd = cd_multi_sweep(h_cd, n, 30, t0, budget_s)
        c = compute_c5(h_cd, n)
        if c < best_c5:
            best_h, best_c5 = h_cd.copy(), c

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
