def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_max(x, beta):
    m = np.max(x)
    x_clipped = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_clipped))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-14 and s > 0:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def hadamard_row(n):
    """Generate a row of a Hadamard matrix of order n (power of 2)"""
    # Sylvester construction
    H = np.array([[1.0]])
    while H.shape[0] < n:
        H = np.block([[H, H], [H, -H]])
    # Take a non-DC row, convert to [0,1]
    row = H[n//2, :]  # A middle row
    return (row + 1) / 2  # Map {-1,1} to {0,1}

    """Project h onto {x : 0 ≤ x ≤ 1, sum(x) = target_sum}"""
    n = len(h)

        h_proj = np.clip(h - t, 0, 1)
        return h_proj.sum() - target_sum
    # Bisection
    lo, hi = h.min() - target_sum/n, h.max()

        mid = (lo + hi) / 2
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift))) / beta

def project_h(h, target_sum):
    """Correct projection onto {0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    if target_sum <= 0 or target_sum >= n:
        return np.clip(h, 0, 1)
    
    def sum_after_shift(t):
        return np.clip(h - t, 0, 1).sum()
    
    # Bisection
    lo = h.min() - target_sum / n - 1
    hi = h.max() - target_sum / n + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        if sum_after_shift(mid) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def lag_gradient(h, k):
    """Gradient of the k-th lag correlation with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        for j in range(n):
            if j <= n - 1 - k:
                grad[j] += (1.0 - h[j + k])
            if j >= k:
                grad[j] += (-h[j - k])
    else:

        for j in range(n):
            if j >= L:
                grad[j] += (1.0 - h[j - L])
            if j <= n - 1 - L:
                grad[j] += (-h[j + L])
    return grad

def subgradient_method(h, n_steps, alpha0, decay, top_k, target_sum):
    """Subgradient method for min max_k corr_k"""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()

        # Top lags
        top_idx = np.argsort(corr)[::-1][:top_k]

            k = idx - (n - 1)

        grad /= top_k
        
        h_new = project_h(h_cur - alpha * grad * dx, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h_cur = h_new
    
    return best_h, best_c5

def coordinate_descent(h, n_sweeps, target_sum):
    """Coordinate descent: try small adjustments to each coordinate"""
    n = len(h)
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    
    deltas = [0.001, -0.001, 0.005, -0.005, 0.01, -0.01, 0.02, -0.02]
    
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            old_val = h_cur[i]
            for d in deltas:
                h_cur[i] = np.clip(old_val + d, 0, 1)
                h_cur = project_h(h_cur, target_sum)
                c5 = compute_c5(h_cur)

                    improved = True
                    break

                h_cur = project_h(h_cur, target_sum)
            else:
                h_cur[i] = old_val
        if not improved:
            break
    
    return h_cur, best_c5

    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    deltas = np.array([0.002, -0.002, 0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.003, -0.003])
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            old = h_cur[i]
            for d in deltas:
                h_cur[i] = np.clip(old + d, 0.0, 1.0)
                h_cur = project_h(h_cur, target_sum)
                c5 = compute_c5(h_cur)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                    break
                h_cur[i] = old
            h_cur = project_h(h_cur, target_sum)
        if not improved:
            break
    return h_cur, best_c5

def hadamard_rows(n):
    """Generate several rows of the n×n Sylvester Hadamard matrix as [0,1] vectors"""
    H = np.array([[1.0, -1.0], [-1.0, -1.0]])
    while H.shape[0] < n:
        H = np.vstack([np.hstack([H, H]), np.hstack([H, -H])])
    rows = []
    for r in [n // 4, n // 2, 3 * n // 4, n // 8, 3 * n // 8]:
        if r < n and r > 0:
            rows.append((H[r, :] + 1.0) / 2.0)
    return rows

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def update(h, c5, n):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n

    h_init = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()

    # Phase 1: SLSQP on initial with progressive beta
    if h_init is not None:
        n = len(h_init)
        ts = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        h_work = h_init.copy()
        for beta in [30, 60, 120, 250, 500, 1000]:
            if time.time() - t_start > budget_s * 0.28:
                break
            def obj1(h):
                return smooth_max(corr_array(h), beta)
            def con1(h):
                return np.sum(h) - ts
            def jac1(h):
                return np.ones(n)
            try:
                res = minimize(obj1, h_work, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': con1, 'jac': jac1}],
                               options={'maxiter': 600, 'ftol': 1e-16})
                h_new = project_h(res.x, ts)
                c5 = compute_c5(h_new)
                update(h_new, c5, n)
                h_work = h_new
            except Exception:
                pass

    # Phase 2: Multi-n exploration with diverse starts
    n_list = []
    if h_init is not None:
        n_list.append(len(h_init))
    for nc in [64, 72, 80, 96, 112]:
        if nc not in n_list and nc % 2 == 0:
            n_list.append(nc)

    for n in n_list:
        if time.time() - t_start > budget_s * 0.72:
            break
        ts = n / 2.0
        if abs(ts - round(ts)) > 1e-9:
            continue
        ts = float(round(ts))
        bounds = Bounds(np.zeros(n), np.ones(n))

        starts = []
        if h_init is not None:
            if len(h_init) == n:
                starts.append(h_init.copy())
            else:
                xo = np.linspace(0, 2, len(h_init), endpoint=False)
                xn = np.linspace(0, 2, n, endpoint=False)
                xe = np.concatenate([xo - 2, xo, xo + 2])
                he = np.concatenate([h_init, h_init, h_init])
                starts.append(project_h(np.interp(xn, xe, he), ts))
        if n in [64, 128, 256]:
            for hr in hadamard_rows(n):
                starts.append(hr.copy())
        for _ in range(3):
            hb = np.zeros(n)
            hb[np.random.choice(n, int(ts), replace=False)] = 1.0
            starts.append(hb)
        x = np.linspace(0, 2, n, endpoint=False)
        for freq in [1.0, 2.0, 3.0]:
            hs = 0.5 + 0.3 * np.cos(np.pi * freq * x)
            starts.append(project_h(hs, ts))

        local_best = None
        local_c5 = float('inf')

        for h0 in starts:
            if time.time() - t_start > budget_s * 0.72:
                break
            h0 = project_h(h0, ts)

            # Subgradient
            h_sg, c5_sg = subgradient_method(h0, 3000, 0.005, 0.0005, 5, ts)
            if c5_sg < local_c5:
                local_c5 = c5_sg
                local_best = h_sg.copy()

            # SLSQP from subgradient result
            if time.time() - t_start < budget_s * 0.72:
                for beta in [80, 200, 500]:
                    if time.time() - t_start > budget_s * 0.75:
                        break
                    def obj2(h):
                        return smooth_max(corr_array(h), beta)
                    def con2(h):
                        return np.sum(h) - ts
                    def jac2(h):
                        return np.ones(n)
                    try:
                        res = minimize(obj2, local_best if local_best is not None else h0,
                                       method='SLSQP', bounds=bounds,
                                       constraints=[{'type': 'eq', 'fun': con2, 'jac': jac2}],
                                       options={'maxiter': 400, 'ftol': 1e-15})
                        h_new = project_h(res.x, ts)
                        c5 = compute_c5(h_new)
                        if c5 < local_c5:
                            local_c5 = c5
                            local_best = h_new.copy()
                    except Exception:
                        pass
            if local_best is not None:
                update(local_best, local_c5, n)

    # Phase 3: Coordinate descent on best
    if best_h is not None and time.time() - t_start < budget_s * 0.85:
        h_cd, c5_cd = coordinate_descent(best_h, 8, best_n / 2.0)
        update(h_cd, c5_cd, best_n)

    # Phase 4: Final high-precision polish
    if best_h is not None and time.time() - t_start < budget_s * 0.98:
        nf = best_n
        tsf = nf / 2.0
        bf = Bounds(np.zeros(nf), np.ones(nf))
        for beta in [800, 1500, 3000]:
            if time.time() - t_start > budget_s * 0.99:
                break
            def obj4(h):
                return smooth_max(corr_array(h), beta)
            def con4(h):
                return np.sum(h) - tsf
            def jac4(h):
                return np.ones(nf)
            try:
                res = minimize(obj4, best_h, method='SLSQP', bounds=bf,
                               constraints=[{'type': 'eq', 'fun': con4, 'jac': jac4}],
                               options={'maxiter': 3000, 'ftol': 1e-16})
                h_new = project_h(res.x, tsf)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass
        # Final coordinate sweep
        if time.time() - t_start < budget_s * 0.995:
            h_cd2, c5_cd2 = coordinate_descent(best_h, 3, tsf)
            update(h_cd2, c5_cd2, best_n)

    return best_h, best_c5, best_n
