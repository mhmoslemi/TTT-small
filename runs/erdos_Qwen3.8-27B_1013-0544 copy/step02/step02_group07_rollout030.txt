import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx

def smooth_max_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    M = np.max(corr)
    S = np.sum(np.exp(beta * (corr - M)))
    w = np.exp(beta * (corr - M)) / S
    m = n - 1
    grad = np.zeros(n)
    for d in range(-m, m + 1):
        wd = w[m + d] * dx
        if wd < 1e-16:
            continue
        if d >= 0:
            for i in range(n - d):
                grad[i] += wd * (1.0 - h[i + d])
                grad[i + d] -= wd * h[i]
        else:
            da = -d
            for i in range(da, n):
                grad[i] += wd * (1.0 - h[i - da])
                grad[i - da] -= wd * h[i]
    return grad

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(60):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = free.sum()
        if nf > 0:
            h[free] += (target_sum - s) / nf
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

def sum_con(x, target):
    return np.sum(x) - target

def sum_con_jac(x, target):
    return np.ones(len(x))

def solve_slqp(h0, n, target, beta, maxiter=5000):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: sum_con(x, target),
             'jac': lambda x: sum_con_jac(x, target)}]
    res = minimize(smooth_max_obj, h0, args=(beta, dx),
                   jac=lambda x: smooth_max_grad(x, beta, dx),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_h(res.x, target)

def coord_descent(h_cur, n, target, max_rounds=80):
    c5_cur = compute_c5(h_cur)
    best_h, best_c5 = h_cur.copy(), c5_cur
    for rd in range(max_rounds):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        vals = corr * (2.0 / n)
        top_idx = np.argsort(vals)[-min(8, len(vals)):]
        improved = False
        for k_idx in top_idx:
            d = k_idx - (n - 1)
            g = np.zeros(n)
            if d >= 0:
                for i in range(n - d):
                    g[i] += (1.0 - h_cur[i + d])
                    g[i + d] -= h_cur[i]
            else:
                da = -d
                for i in range(da, n):
                    g[i] += (1.0 - h_cur[i - da])
                    g[i - da] -= h_cur[i]
            g *= 2.0 / n

                    c5_cur = c5_try
