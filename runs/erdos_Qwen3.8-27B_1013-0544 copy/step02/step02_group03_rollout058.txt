import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_multi(h, k_indices, weights, n):
    grad = np.zeros(n)
    for ki, w in zip(k_indices, weights):
        lag = ki - (n - 1)
        if 0 <= lag < n:
            idx = np.arange(n - lag)
            grad[idx] += w * (1.0 - h[idx + lag])
            grad[idx + lag] -= w * h[idx]
        elif -n < lag < 0:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += w * (1.0 - h[idx])
            grad[idx] -= w * h[idx + la]
    return grad

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def obj_slsqp(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum):
    return h.sum() - target_sum

def con_jac(h, n):
    return np.ones(n)

    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]

    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-lag subgradient with backtracking
    h = best_h.copy()
    for it in range(80000):

            break

        top_k = np.argsort(corr)[-10:][::-1]
        top_vals = corr[top_k]
        max_v = top_vals[0]
        w = np.exp(4.0 * (top_vals - max_v))
        w /= w.sum()
        grad = subgrad_multi(h, top_k, w, n)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
