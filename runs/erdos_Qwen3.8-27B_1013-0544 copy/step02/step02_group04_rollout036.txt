_TS = [0.0]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        free = (h > 0) & (h < 1.0)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def obj_lagrange(x, beta, dx, target_sum, mu, lam):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    sm = m + np.log(np.sum(np.exp(shifted))) / beta
    s = np.sum(x)
    return sm * dx + lam * (s - target_sum) + 0.5 * mu * (s - target_sum) ** 2

def obj_smooth(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
