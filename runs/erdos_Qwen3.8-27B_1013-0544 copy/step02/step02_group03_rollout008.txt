import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt, 0.0, 1.0)
        h[mask if diff > 0 else (1 - (h == 0).astype(int)).astype(bool)] = np.clip(h, 0, 1)
    # Simpler projection
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def con_fun(h, target_sum):
    return h.sum() - target_sum

def con_jac(h, target_sum):
    return np.ones(len(h))

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def subgrad_top_k(h, K=3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-K:]
    c_top = corr[top_idx]
    w = np.exp(50.0 * (c_top - c_top[-1]))
    w = w / w.sum()
    grad = np.zeros(n)
    for ii, k in enumerate(top_idx):
        lag = k - (n - 1)
        wi = w[ii]
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += wi * (1.0 - h[idx + lag])
            grad[idx + lag] -= wi * h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += wi * (1.0 - h[idx])
            grad[idx] -= wi * h[idx + la]
    return grad, np.max(corr)

def binary_push(h, target_sum, strength=0.3):
    h_new = h.copy()
    above = h_new > 0.5
    h_new[above] = 1.0 - (1.0 - h_new[above]) * (1.0 - strength)
    below = ~above
    h_new[below] = h_new[below] * (1.0 - strength)
    return project_feasible(h_new, target_sum)
