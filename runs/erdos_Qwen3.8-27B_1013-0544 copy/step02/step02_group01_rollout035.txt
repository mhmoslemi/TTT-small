import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, penalty_coeff=0.0):
    """Compute smooth max value and gradient, optionally with penalty for sum constraint."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exps = np.exp(shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    if penalty_coeff > 0:
        val += penalty_coeff * (h.sum() - n/2.0)**2
    
    # Gradient
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    if penalty_coeff > 0:
        grad += 2.0 * penalty_coeff * (h.sum() - n/2.0)
    return val, grad

def lbfgs_optimize(h, beta_start, beta_end, maxiter, penalty, time_limit):
    """Run L-BFGS-B with adaptive beta."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(h)
    t0 = time.time()
    
    # We'll do multiple L-BFGS-B calls with increasing beta
    n_phases = 5
    betas = np.geomspace(beta_start, beta_end, n_phases)
    
    current_h = project_feasible(h, target)
    
    for beta in betas:
        if time.time() - t0 > time_limit:
            break
        maxiter_phase = maxiter // n_phases
        
        def obj(x):
            v, _ = smoothmax_val_grad(x, beta, penalty)
            return v
        
        def objg(x):
            _, g = smoothmax_val_grad(x, beta, penalty)
            return g
        
        try:
            res = minimize(obj, current_h, jac=objg, method='L-BFGS-B',
                          bounds=[(0,1)]*n,
                          options={'maxiter': maxiter_phase, 'ftol': 1e-16,
                                   'gtol': 1e-10, 'maxfun': maxiter_phase * 3})
            current_h = project_feasible(res.x, target)
            v = compute_c5(current_h)
            if v < best_val:
                best_val = v
                best_h = current_h.copy()
        except Exception:
            pass
    
    return best_h, best_val

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_general_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        if w[k] < 1e-11:
            continue
        s = k - (n - 1)
        js = j - s
        ms = (js >= 0) & (js < n)
        grad[ms] += w[k] * (1.0 - h[js[ms]])
        jp = j + s
        mp = (jp >= 0) & (jp < n)
        grad[mp] += w[k] * (-h[jp[mp]])
    grad *= dx
    return val, grad


def smoothmax_active_val_grad(h, active_indices, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr[active_indices]
    m = np.max(vals)
    shifted = np.clip(beta * (vals - m), -500, 500)
    exps = np.exp(shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for idx, wk in zip(active_indices, w):
        if wk < 1e-11:
            continue
        s = idx - (n - 1)
        js = j - s
        ms = (js >= 0) & (js < n)
        grad[ms] += wk * (1.0 - h[js[ms]])
        jp = j + s
        mp = (jp >= 0) & (jp < n)
        grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad


def get_top_lag_indices(h, k=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-k:]
    return sorted(top_idx.tolist())


def slqp_general(h, beta, maxiter, ftol=1e-15):
    n = len(h)
    target = n / 2.0

    def obj(x):
        return smoothmax_general_val_grad(x, beta)[0]

    def objg(x):
        return smoothmax_general_val_grad(x, beta)[1]

    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def slqp_active(h, active_indices, beta, maxiter, ftol=1e-15):
    n = len(h)
    target = n / 2.0

    def obj(x):
        return smoothmax_active_val_grad(x, active_indices, beta)[0]

    def objg(x):
        return smoothmax_active_val_grad(x, active_indices, beta)[1]

    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def lbfgs_with_penalty(h, beta, maxiter, penalty=500.0):
    n = len(h)
    target = n / 2.0

    def obj(x):
        v, _ = smoothmax_general_val_grad(x, beta)
        return v + penalty * (x.sum() - target) ** 2

    def objg(x):
        _, g = smoothmax_general_val_grad(x, beta)
        return g + 2.0 * penalty * (x.sum() - target)

    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = PchipInterpolator(x_old, h)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)

    best_c5 = compute_c5(h_best)

    # Phase 1: General SLSQP with increasing beta
    for beta in [40, 80, 150, 300]:
        if remaining() < 25:
            break
        h_tmp = slqp_general(h_best, beta, maxiter=400)
        c = compute_c5(h_tmp)
        if c < best_c5:
            h_best, best_c5 = h_tmp, c

    # Phase 2: Active-lag focused optimization (key improvement)
    for round_idx in range(4):
        if remaining() < 20:
            break
        active = get_top_lag_indices(h_best, k=5)
        h_tmp = slqp_active(h_best, active, beta=200, maxiter=300)
        c = compute_c5(h_tmp)
        if c < best_c5:
            h_best, best_c5 = h_tmp, c
        # Re-identify active lags and refine
        if remaining() > 10:
            active2 = get_top_lag_indices(h_best, k=7)
            h_tmp2 = slqp_active(h_best, active2, beta=400, maxiter=200)
            c2 = compute_c5(h_tmp2)
            if c2 < best_c5:
                h_best, best_c5 = h_tmp2, c2

    # Phase 3: L-BFGS-B sweep (faster per iteration, good for high beta)
    for beta in [500, 1000, 2000]:
        if remaining() < 15:
            break
        h_tmp = lbfgs_with_penalty(h_best, beta, maxiter=300, penalty=2000.0)
        c = compute_c5(h_tmp)
        if c < best_c5:
            h_best, best_c5 = h_tmp, c

    # Phase 4: Another round of active-lag refinement at high beta
    for round_idx in range(3):
        if remaining() < 15:
            break
        active = get_top_lag_indices(h_best, k=5)
        h_tmp = slqp_active(h_best, active, beta=800, maxiter=200, ftol=1e-16)
        c = compute_c5(h_tmp)
        if c < best_c5:
            h_best, best_c5 = h_tmp, c

    # Phase 5: Try higher resolution
    for n_try in [280, 350]:
        if remaining() < 80:
            break
        h_interp = interp_to_n(h_best, n_try)
        h_tmp = slqp_general(h_interp, 80, maxiter=300)
        c = compute_c5(h_tmp)
        if c < best_c5:
            h_best, best_c5 = h_tmp, c
        if remaining() > 30:
            active = get_top_lag_indices(h_best, k=5)
            h_tmp2 = slqp_active(h_best, active, beta=300, maxiter=200)
            c2 = compute_c5(h_tmp2)
            if c2 < best_c5:
                h_best, best_c5 = h_tmp2, c2

    # Phase 6: Multi-restart with perturbation
    n_cur = len(h_best)
    for _ in range(10):
        if remaining() < 18:
            break
        noise = np.random.randn(n_cur) * 0.02
        h_pert = project_feasible(h_best + noise, n_cur / 2.0)
        h_tmp = slqp_general(h_pert, 100, maxiter=150)
        c = compute_c5(h_tmp)
        if c < best_c5:
            h_best, best_c5 = h_tmp, c

    # Phase 7: Final polish - very high beta active-lag
    if remaining() > 20:
        active = get_top_lag_indices(h_best, k=5)
        h_tmp = slqp_active(h_best, active, beta=3000, maxiter=400, ftol=1e-16)
        c = compute_c5(h_tmp)
        if c < best_c5:
            h_best, best_c5 = h_tmp, c

    # Phase 8: Final general polish
    if remaining() > 15:
        h_tmp = slqp_general(h_best, 2000, maxiter=300, ftol=1e-16)
        c = compute_c5(h_tmp)
        if c < best_c5:
            h_best, best_c5 = h_tmp, c

    # Final projection
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
