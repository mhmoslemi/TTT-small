    active = w > 1e-8
    for k in np.where(active)[0]:
        wk = w[k]
        s = k - (n - 1)  # actual lag
        if s >= 0:

            # for i in [s, n-1]: += wk * (1 - h[i-s])
            # for i in [0, n-1-s]: += wk * (-h[i+s])
            grad[s:] += wk * (1.0 - h[:n-s])  # (1-h[i-s]) for i from s to n-1
            grad[:n-s] += wk * (-h[s:])  # -h[i+s] for i from 0 to n-1-s
        else:

            # out[-j] = sum_{m=0}^{n-1-j} h[m] * (1-h[m+j])
            # for i in [0, n-1-j]: += wk * (1 - h[i+j])
            # for i in [j, n-1]: += wk * (-h[i-j])
            grad[:n-j] += wk * (1.0 - h[j:])  # (1-h[i+j]) for i from 0 to n-1-j
            grad[j:] += wk * (-h[:n-j])  # -h[i-j] for i from j to n-1
    grad *= dx
    return val, grad

**Method: Normalize after each L-BFGS-B call**

Run L-BFGS-B without the sum constraint (only bounds), then project the result to be feasible. Repeat.

But this doesn't work well because L-BFGS-B uses the gradients to build its Hessian approximation, and if we project between calls, the gradients are inconsistent.

**Better method: Augmented Lagrangian**

The augmented Lagrangian for constraint g(h) = ∑h - n/2 = 0:
L(h, λ, ρ) = f(h) + λ*g(h) + (ρ/2)*g(h)²

Gradient: ∇L = ∇f(h) + λ*1 + ρ*g(h)*1

Update: After minimizing over h (with L-BFGS-B and bounds), update:
λ_new = λ + ρ*g(h_new)
ρ_new = ρ * 2 (if constraint not satisfied well enough)

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-6) & (h < 1.0 - 1e-6)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    scaled = beta * (corr - m)
    scaled = np.clip(scaled, -50, 0)  # avoid overflow
    exps = np.exp(scaled)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    active = np.where(w > 1e-6)[0]
    for k in active:
        wk = w[k]
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n-s])
            if n - s > s:  # overlap region
                # already handled by the two lines above
                pass
            grad[:n-s] += wk * (-h[s:])
        else:
            j = -s
            grad[:n-j] += wk * (1.0 - h[j:])
            grad[j:] += wk * (-h[:n-j])
    grad *= dx
    return val, grad

def topk_subgrad(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-top_k:]
    lags = top_idx - (n - 1)
    grad = np.zeros(n)
    for s in lags:
        if s >= 0:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] += (-h[s:])
        else:
            j = -s
            grad[:n-j] += (1.0 - h[j:])
            grad[j:] += (-h[:n-j])
    grad = grad / top_k * dx
    return float(np.max(corr) * dx), grad

def lbfgsb_auglag(h0, beta, max_outer=20, max_inner=500, rho0=10.0):
    n = len(h0)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    lam = 0.0
    rho = rho0
    best_h = h0.copy()
    best_v = compute_c5(h0)
    
    for outer in range(max_outer):
        def obj(x):
            v, g = smoothmax_val_grad(x, beta)
            g_hat = x.sum() - target
            g_full = g + lam * np.ones(n) + rho * g_hat * np.ones(n)
            return v + lam * g_hat + 0.5 * rho * g_hat * g_hat, g_full
        
        res = minimize(obj, h0, jac=True, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': max_inner, 'ftol': 1e-14, 'gtol': 1e-8})
        h0 = res.x
        g_hat = h0.sum() - target
        if abs(g_hat) < 1e-8:
            break
        lam = lam + rho * g_hat
        if abs(g_hat) > 1e-4:
            rho = rho * 2
        h0 = project_feasible(h0, target)
        v = compute_c5(h0)
        if v < best_v:
            best_v = v
            best_h = h0.copy()
    
    return best_h, best_v

def subgrad_bb(h0, max_iters, top_k=5, lr0=0.05):
    n = len(h0)
    target = n / 2.0
    x = h0.copy()
    x_prev = x.copy()
    g_prev = None
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(max_iters):
        v, g = topk_subgrad(x, top_k)
        if v < best_v:
            best_v = v
            best_x = x.copy()
        
        # BB step size
        if g_prev is not None and t > 0:
            s = x - x_prev
            y = g - g_prev
            sy = np.dot(s, y)
            if sy > 1e-10:
                ss = np.dot(s, s)
                yy = np.dot(y, y)
                lr = ss / sy  # BB1
                # Alternative: lr = sy / yy  # BB2
            else:
                lr = lr0 * (0.99 ** t)
        else:
            lr = lr0
        
        x_prev = x.copy()
        g_prev = g.copy()
        x_new = project_feasible(x - lr * g, target)
        x = x_new
    
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

    t0 = time.time()
    
    def elapsed():
        return time.time() - t0

        return budget_s - elapsed()

    else:

    n = len(h_best)

    if remaining() > 60:

        if c1 < best_c5:

    # Phase 2: Subgradient with BB step

    # Phase 3: Another L-BFGS-B round with higher beta
    if remaining() > 60:
        h3, c3 = lbfgsb_auglag(h_best, beta=150, max_outer=15, max_inner=300, rho0=100.0)
        if c3 < best_c5:
            h_best, best_c5 = h3, c3

        h_interp = interp_to_n(h_best, n_try)
        c_interp = compute_c5(h_interp)

                h_interp = h_best.copy()

from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-6) & (h < 1.0 - 1e-6)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    scaled = beta * (corr - m)
    scaled = np.clip(scaled, -50, 0)
    exps = np.exp(scaled)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    active = np.where(w > 1e-5)[0]
    for k in active:
        wk = w[k]
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n-s])
            grad[:n-s] += wk * (-h[s:])
        else:
            j = -s
            grad[:n-j] += wk * (1.0 - h[j:])
            grad[j:] += wk * (-h[:n-j])
    grad *= dx
    return val, grad

def topk_grad(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = float(np.max(corr) * dx)
    top_idx = np.argsort(corr)[-top_k:]
    lags = top_idx - (n - 1)
    grad = np.zeros(n)
    for s in lags:
        if s >= 0:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] += (-h[s:])
        else:
            j = -s
            grad[:n-j] += (1.0 - h[j:])
            grad[j:] += (-h[:n-j])
    grad /= top_k
    grad *= dx
    return val, grad

def lbfgsb_optimize(h0, beta, target, maxiter=300):
    n = len(h0)
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        v, g = smoothmax_val_grad(x, beta)
        # Add penalty for constraint violation
        g_sum = x.sum() - target
        if abs(g_sum) > 1e-6:
            penalty = 1e4 * g_sum * g_sum
            g = g + 2e4 * g_sum * np.ones(n)
            v = v + penalty
        return v, g
    
    try:
        res = minimize(obj, h0.copy(), jac=True, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except:
        return project_feasible(h0, target)

def subgrad_optimize(h0, max_iters, top_k, lr0, target, momentum=0.9):
    n = len(h0)
    x = h0.copy().astype(float)
    v_curr, g_curr = topk_grad(x, top_k)
    best_x = x.copy()
    best_v = v_curr
    vel = np.zeros(n)
    
    for t in range(max_iters):
        # Nesterov momentum
        y = x + momentum * vel
        y = project_feasible(y, target)
        v_y, g_y = topk_grad(y, top_k)
        
        vel = momentum * vel - lr0 * (0.98 ** t) * g_y
        x_new = project_feasible(x + vel, target)
        
        v_new = compute_c5(x_new)
        if v_new < best_v:
            best_v = v_new
            best_x = x_new.copy()
        
        x = x_new
    
    return best_x, best_v

def interp_to_n(h, n_new):
