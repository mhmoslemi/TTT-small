    n = len(h)

import time

def compute_c5(h):

    shifted = np.clip(beta * (corr - m), -500, 500)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:

        h = np.clip(h, 0.0, 1.0)

    return smooth_max_val(corr, beta) * dx

def slsqp_refine(h, beta, dx, target_sum, bounds, maxiter=3000, ftol=1e-14):

        return smooth_obj(x, beta, dx)
    def cfun(x):
        return x.sum() - target_sum
    def cjac(x):
        return np.ones(x.shape[0])
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_h(res.x, target_sum)
        return h_new
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(corr, beta) * dx

def slsqp_refine(h, beta, dx, target_sum, bounds, maxiter=5000, ftol=1e-15):
    def obj(x):
        return smooth_obj(x, beta, dx)
    def cfun(x):
        return x.sum() - target_sum
    def cjac(x):
        return np.ones(x.shape[0])
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(res.x, target_sum)
    except Exception:
        return h

        elif s == 3:
