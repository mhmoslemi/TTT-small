import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-14:
                break
            h += (diff / t) * room
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h += (diff / t) * h
    return np.clip(h, 0.0, 1.0)

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def _make_con(n):
    return {'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}

def _obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    ms = shifted.max()
    if ms > 400:
        return m + (ms + np.log(np.exp(shifted - ms).sum())) / alpha
    return m + np.log(np.exp(shifted).sum()) / alpha

def slsqp(h, n, alpha, maxiter):
    cons = [_make_con(n)]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def pair_search(h, n, t0, budget, max_sweeps=50):
    """Iterated pair line search until convergence or budget exhausted"""
    best_c5 = compute_c5(h, n)
    for sweep in range(max_sweeps):
        if time.time() - t0 > budget - 30:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > budget - 30:
                break
            for j in range(i + 1, n):
                # Golden section on d: h[i]+=d, h[j]-=d
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = d_lo, d_hi
                c1 = b - gr * (b - a)
                c2 = a + gr * (b - a)
                h_tmp = h.copy()
                h_tmp[i] += c1; h_tmp[j] -= c1
                f1 = compute_c5(h_tmp, n)
                h_tmp[i] = h[i]; h_tmp[i] += c2; h_tmp[j] -= c2
                f2 = compute_c5(h_tmp, n)
                for _ in range(30):
                    if f1 < f2:
                        b = c2; c2 = c1; f2 = f1
                        c1 = b - gr * (b - a)
                        h_tmp[i] = h[i] + c1; h_tmp[j] = h[j] - c1
                        f1 = compute_c5(h_tmp, n)

                        c2 = a + gr * (b - a)
                        h_tmp[i] = h[i] + c2; h_tmp[j] = h[j] - c2

                d_opt = (a + b) / 2.0

                h_new[j] -= d_opt
                h_new = np.clip(h_new, 0, 1)

                f_new = compute_c5(h_new, n)
                if f_new < best_c5 - 1e-14:

                    best_c5 = f_new
                    improved = True
        if not improved:
            break
    return h, best_c5

f(d) = max_k c_k(h + d*e_i - d*e_j) where e_i, e_j are unit vectors.

Each c_k is a quadratic function of h (bilinear form). So c_k(d) is at most quadratic in d. The max of quadratic functions is not necessarily convex.

However, for small d (near the current point), c_k(d) ≈ c_k(0) + d * (dc_k/dd)|_{d=0} + d²/2 * (d²c_k/dd²)|_{d=0}. The second derivative term is small for small d, so f(d) is approximately the max of linear functions, which is convex.

So golden section should work for small perturbations. For larger perturbations, it might fail, but we can limit d to a small range.

Let me limit the search range to [-0.1, 0.1] or so, centered at 0. This should keep us in the "convex" regime.

Actually, the natural constraint is d_lo = max(-h[i], h[j]-1) and d_hi = min(1-h[i], h[j]). If h[i] and h[j] are both around 0.5, then d_lo ≈ -0.5 and d_hi ≈ 0.5. That's a range of 1.0, which might be too large for the convexity assumption.

Let me use a smaller range: d in [-0.1, 0.1] (or limited by the bounds if h[i] or h[j] is near 0 or 1).

Actually, let me just use the full range but with fewer iterations (10 instead of 30). The golden section will still converge, just less precisely. And the outer loop (multiple sweeps) will refine it.

OK let me just write the final code. I'll keep it simple and effective:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-14:
                break
            h += (diff / t) * room
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h += (diff / t) * h
    return np.clip(h, 0.0, 1.0)

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def _obj_lse(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    ms = shifted.max()
    if ms > 400:
        return m + (ms + np.log(np.exp(shifted - ms).sum())) / alpha
    return m + np.log(np.exp(shifted).sum()) / alpha

def _slsqp(h, n, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_lse, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return h
