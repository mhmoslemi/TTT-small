import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)


def smooth_max(v, beta):
    m = np.max(v)
    return m + np.log(np.sum(np.exp(beta * (v - m)))) / beta


def obj_smooth(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(corr, beta) * 2.0 / n


def constraint_sum(x, target):
    return np.sum(x) - target


def constraint_jac(x, target):
    return np.ones(len(x))


def coord_sweep(h, n, target, rng, n_grid=50):
    order = rng.permutation(n)
    for i in order:
        h[i] = 0.0
        B = np.correlate(h, 1.0 - h, mode="full")
        h[i] = 1.0
        A1 = np.correlate(h, 1.0 - h, mode="full")
        c = A1 - B
        ts = np.linspace(0.0, 1.0, n_grid)
        vals = B[None, :] + ts[:, None] * c[None, :]
        maxes = np.max(vals, axis=1)
        idx = np.argmin(maxes)
        best_t = ts[idx]
        if 0 < idx < n_grid - 1:
            lo = ts[max(0, idx - 2)]
            hi = ts[min(n_grid - 1, idx + 2)]
            fine = np.linspace(lo, hi, 40)
            vals_f = B[None, :] + fine[:, None] * c[None, :]
            best_t = fine[np.argmin(np.max(vals_f, axis=1))]
        h[i] = best_t
    h = project_sum(h, target)
    return h


def slsqp_refine(h, n, target, beta, maxiter):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': constraint_sum, 'jac': constraint_jac, 'args': (target,)}]
    res = minimize(obj_smooth, h, args=(beta, n), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_sum(res.x, target)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    target = n / 2.0
    h = project_sum(h_init, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    def left():
        return budget_s - (time.time() - t0)

    # Phase 1: Heavy coordinate descent (use ~55% of budget)
    phase1_budget = budget_s * 0.55
    no_improve = 0

    while left() > budget_s - phase1_budget:
        h = coord_sweep(h, n, target, rng, n_grid=50)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_h = h.copy()
            best_c5 = c5
            no_improve = 0
        else:
            no_improve += 1
        sweep += 1
        if no_improve > 300 and sweep % 100 == 0:
            h = h.copy()
            n_pert = rng.randint(3, 10)
            idx = rng.choice(n, n_pert, replace=False)

            h = project_sum(h, target)

    # Phase 2: SLSQP refinement with increasing beta

                best_c5 = c5_new

            pass

            best_h = h2.copy()
            best_c5 = best_c5_2
            n = n2
            target = target2

    # Phase 5: Final SLSQP polish

            break
