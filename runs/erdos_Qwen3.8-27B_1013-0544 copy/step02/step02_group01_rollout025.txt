    sw = np.sum(exps)

    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)

        jp = j + s

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += (-h[jp[mp]])
    grad *= dx
    return val, grad


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h


def adam_smoothmax(h, max_iters, beta, lr0=0.01, decay=0.9995):
    n = len(h)
    target = n / 2.0
    x = project_feasible(h.copy(), target)
    m = np.zeros(n)
    v = np.zeros(n)
    best_x = x.copy()
    best_v = compute_c5(x)
    b1, b2, eps = 0.9, 0.999, 1e-8

    for t in range(1, max_iters + 1):
        val, grad = smoothmax_val_grad(x, beta)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        lr = lr0 * (decay ** t)
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x_new = project_feasible(x_new, target)
        cv = compute_c5(x_new)
        if cv < best_v:
            best_v = cv
            best_x = x_new.copy()
        x = x_new

    return best_x, best_v


def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v

    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def cyclic_shift_perturb(h, rng):
    n = len(h)
    target = n / 2.0
    shift = rng.randint(1, max(2, n // 10))
    h_shifted = np.roll(h, shift)
    h_blend = 0.7 * h + 0.3 * h_shifted
    return project_feasible(h_blend, target)


def scale_perturb(h, rng):
    n = len(h)
    target = n / 2.0
    noise = rng.standard_normal(n)
    h_pert = h + 0.06 * noise
    return project_feasible(h_pert, target)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)

    best_c5 = compute_c5(h_best)

    # Phase 1: Adam with moderate beta, learning rate decay
    if remaining() > 60:
        h_a, c_a = adam_smoothmax(h_best, 3000, beta=150, lr0=0.01, decay=0.9998)
        if c_a < best_c5:
            h_best, best_c5 = h_a, c_a

    # Phase 2: SLSQP with moderate beta
    if remaining() > 40:
        h_s = slsqp_refine(h_best, maxiter=500, beta=300)
        c_s = compute_c5(h_s)
        if c_s < best_c5:
            h_best, best_c5 = h_s, c_s

    # Phase 3: Higher beta Adam for sharper max approximation
    if remaining() > 50:
        h_a2, c_a2 = adam_smoothmax(h_best, 2000, beta=300, lr0=0.004, decay=0.9997)
        if c_a2 < best_c5:
            h_best, best_c5 = h_a2, c_a2

    # Phase 4: SLSQP polish with higher beta
    if remaining() > 35:
        h_s2 = slsqp_refine(h_best, maxiter=500, beta=600)
        c_s2 = compute_c5(h_s2)
        if c_s2 < best_c5:
            h_best, best_c5 = h_s2, c_s2

    # Phase 5: Random restarts with structured perturbations
    if remaining() > 80:
        n_restart = min(10, int(remaining() / 20))
        for i in range(n_restart):
            if remaining() < 25:
                break
            if i % 2 == 0:
                h_pert = cyclic_shift_perturb(h_best, rng)
            else:
                h_pert = scale_perturb(h_best, rng)
            h_r, c_r = adam_smoothmax(h_pert, 600, beta=200, lr0=0.008, decay=0.9995)
            if c_r < best_c5:
                h_best, best_c5 = h_r, c_r
                if remaining() > 20:
                    h_sr = slsqp_refine(h_best, maxiter=300, beta=500)
                    c_sr = compute_c5(h_sr)
                    if c_sr < best_c5:
                        h_best, best_c5 = h_sr, c_sr

    # Phase 6: Try at higher resolution
    if remaining() > 120:
        for n_try in [300, 400]:
            if remaining() < 80:
                break
            h_interp = interp_to_n(h_best, n_try)
            if remaining() > 50:
                h_hi, c_hi = adam_smoothmax(h_interp, 1000, beta=250, lr0=0.003, decay=0.9997)
                if c_hi < best_c5:
                    h_best, best_c5 = h_hi, c_hi
            if remaining() > 35:
                h_shi = slsqp_refine(h_best, maxiter=300, beta=500)
                c_shi = compute_c5(h_shi)
                if c_shi < best_c5:
                    h_best, best_c5 = h_shi, c_shi

    # Phase 7: Final very high beta SLSQP
    if remaining() > 20:
        h_f = slsqp_refine(h_best, maxiter=800, beta=1000)
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            h_best, best_c5 = h_f, c_f

    # Phase 8: One last Adam pass with very high beta
    if remaining() > 15:
        h_l, c_l = adam_smoothmax(h_best, 1000, beta=800, lr0=0.002, decay=0.9999)
        if c_l < best_c5:
            h_best, best_c5 = h_l, c_l

    # Final validation
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)

    return h_best, best_c5, len(h_best)
