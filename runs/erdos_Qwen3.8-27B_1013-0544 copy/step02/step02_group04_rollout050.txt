```python
import cvxpy as cp

def trust_region_step(h_center, delta, n_points, target_sum):
    """Solve the trust region subproblem."""
    n = n_points
    corr = corr_values(h_center)
    
    # Linearize each corr[m] at h_center
    # corr[m] ≈ corr[m]_center + grad[m]^T (h - h_center)
    # We want: min t s.t. corr[m]_center + grad[m]^T (h - h_center) ≤ t
    #                          0 ≤ h ≤ 1, sum(h) = target
    #                          ||h - h_center||_2 ≤ delta
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    
    # Add linearized correlation constraints
    # Only add top few to keep LP small
    top_k = 10
    top_idx = np.argsort(corr)[-top_k:]
    
    for m in top_idx:
        k = m - (n - 1)
        if k >= 0:
            js = np.arange(n - k)
            grad = np.zeros(n)
            grad[js + k] = (1.0 - h_center[js])
            grad[js] = -h_center[js + k]
        else:
            kp = -k
            js = np.arange(kp, n)
            grad = np.zeros(n)
            grad[js] = (1.0 - h_center[js - kp])
            grad[js - kp] = -h_center[js]
        
        linearized = corr[m] + np.dot(grad, h_var - h_center)
        constraints.append(linearized <= t_var)
    
    # Trust region (use L1 or L2)
    # L2 trust region makes it a QP
    constraints.append(cp.norm(h_var - h_center, 2) <= delta)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    prob.solve(solver=cp.SCS)  # or ECOS
    
    if h_var.value is not None:
        return h_var.value, t_var.value
    return h_center, corr.max()
```
