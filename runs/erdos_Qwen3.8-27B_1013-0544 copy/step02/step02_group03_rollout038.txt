- b_m = constant part of the change to C[m]

Since this is the max of linear functions of v, it's convex. The minimum over v ∈ [0,1] is at:

- v = 1

def two_coord_descent(h, max_sweeps=5000, n_pairs_per_sweep=None, seed=42):
    n = len(h)
    rng = np.random.RandomState(seed)
    if n_pairs_per_sweep is None:
        n_pairs_per_sweep = n
    
    h = h.copy()
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    lags = np.arange(2*n-1) - (n-1)
    
    for sweep in range(max_sweeps):
        max_change = 0.0
        for _ in range(n_pairs_per_sweep):
            j, k = rng.choice(n, size=2, replace=False)
            old_j, old_k = h[j], h[k]
            s = old_j + old_k
            # v range: h[j] ∈ [max(0, s-1), min(1, s)]
            v_lo = max(0.0, s - 1.0)
            v_hi = min(1.0, s)
            if v_hi - v_lo < 1e-15:
                continue  # No freedom
            
            # Compute b[m] = a_j[m] - a_k[m]
            # a_j[m] = dC[m]/dh[j], a_k[m] = dC[m]/dh[k]
            
            # dC[m]/dh[j]:
            #   j as h part: (1-h[j+lag]) if 0<=j+lag<n
            #   j as 1-h part: -h[j-lag] if 0<=j-lag<n
            a_j = np.zeros(2*n-1)
            valid = (j + lags >= 0) & (j + lags < n)
            a_j[valid] += (1.0 - h[j + lags[valid]])
            valid2 = (j - lags >= 0) & (j - lags < n)
            a_j[valid2] += (-h[j - lags[valid2]])
            
            a_k = np.zeros(2*n-1)
            valid = (k + lags >= 0) & (k + lags < n)
            a_k[valid] += (1.0 - h[k + lags[valid]])
            valid2 = (k - lags >= 0) & (k - lags < n)
            a_k[valid2] += (-h[k - lags[valid2]])
            
            b = a_j - a_k  # coefficient of (v - old_j)
            
            # f(v) = max_m (C[m] + b[m]*(v - old_j))
            # = max_m (b[m]*v + (C[m] - b[m]*old_j))
            
            # Bisection on subgradient
            lo, hi = v_lo, v_hi
            for _ in range(50):
                mid = (lo + hi) / 2
                vals = b * mid + (C - b * old_j)
                max_val = np.max(vals)
                active = vals >= max_val - 1e-13
                min_slope = np.min(b[active])
                max_slope = np.max(b[active])
                if max_slope < -1e-13:
                    hi = mid
                elif min_slope > 1e-13:
                    lo = mid
                else:
                    break
            
            v_new = (lo + hi) / 2
            # Verify
            candidates = [v_lo, v_hi, lo, hi, (lo+hi)/2]
            best_v, best_f = v_new, np.inf
            for v in candidates:
                f = np.max(b * v + (C - b * old_j))
                if f < best_f:
                    best_f = f
                    best_v = v
            
            delta = best_v - old_j
            if abs(delta) > 1e-14:
                h[j] = best_v
                h[k] = s - best_v
                C += b * delta
                max_change = max(max_change, abs(delta))
        
        if max_change < 1e-14:
            break
    
    return h
