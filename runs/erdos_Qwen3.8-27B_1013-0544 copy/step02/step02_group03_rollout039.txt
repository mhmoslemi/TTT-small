13. **Let me also think about the subgradient more carefully**:

    For a fixed lag k (say k ≥ 0), the correlation is:
    f_k(h) = sum_{i=0}^{n-1-k} h[i] * (1 - h[i+k])
    
    The subgradient (actually gradient, since f_k is differentiable) with respect to h[j] is:
    - If j appears as h[i] in the sum: contribution is (1 - h[j+k]) for i=j (if j < n-k)
    - If j appears as h[i+k] in the sum: contribution is -h[j-k] for i=j-k (if j ≥ k)
    
    So: ∂f_k/∂h[j] = (1 - h[j+k]) * 1_{j < n-k} - h[j-k] * 1_{j ≥ k}

    For the max, the subgradient is a convex combination of the gradients of the active functions.

    The "averaged subgradient" approach: take the average of the gradients of the top m active lags. This is a smoothed version of the true subgradient.

import numpy as np
from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    """Project onto {0<=h<=1, sum(h)=target_sum} via bisection on threshold."""
    h = h.copy()
    # First clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Now adjust to meet sum constraint
    s = h.sum()
    if abs(s - target_sum) < 1e-14:
        return h
    # Bisection on offset
    lo, hi = -h.max(), 1.0 - h.min()
    for _ in range(100):
        mid = (lo + hi) / 2
        h_test = np.clip(h + mid, 0.0, 1.0)
        s_test = h_test.sum()
        if s_test > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    # Fix any remaining tiny error
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 0) & (h < 1)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def avg_subgradient(h, n, top_m=5):
    """Average subgradient over top-m worst lags."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top m indices
    top_idx = np.argsort(corr)[-top_m:]
    grad = np.zeros(n)
    for k in top_idx:
        lag = int(k) - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            idx = np.arange(n - lag)
            g[idx] += (1.0 - h[idx + lag])
