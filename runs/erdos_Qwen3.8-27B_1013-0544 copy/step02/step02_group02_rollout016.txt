    """Compute dc[m]/dh[i] for all i"""
    dx = 2.0 / n
    p = m - n + 1
    grad = np.zeros(n)
    
    # First term: (1-h[i+p]) for i+p in [0, n-1]
    # i in [-p, n-1-p]
    i_start = max(0, -p)
    i_end = min(n, n - p)
    if i_start < i_end:
        grad[i_start:i_end] += (1 - h[i_start+p:i_end+p])
    
    # Second term: -h[i-p] for i-p in [0, n-1]
    # i in [p, p+n-1]
    i_start2 = max(0, p)
    i_end2 = min(n, p + n)
    if i_start2 < i_end2:
        grad[i_start2:i_end2] -= h[i_start2-p:i_end2-p]
    
    return grad * dx
