    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

    n = len(h)

    # For k > 0:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def soften_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def soften_max_weights(x, beta):
    m = np.max(x)
    e = np.exp(beta * (x - m))
    w = e / e.sum() / beta
    return w

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 0.0) & (h < 1.0)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def proj_constraint(u, n, S):
    """Project u onto {0<=x<=1, sum(x)=S} using bisection on shift."""
    lo, hi = u.min() - 1.0, u.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(u - mid, 0.0, 1.0)
        s = proj.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(u - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_grad(h, n, beta):

    dx = 2.0 / n

            blen = np.random.randint(3, max(4, n // 6))
            bs = np.random.randint(0, n - blen)
            h[bs:bs+blen] = 0.5 + 0.35 * np.random.randn(blen)
        elif strat == 3:
            w = np.random.uniform(1, 8) * np.pi
            ph = np.random.uniform(0, 2 * np.pi)
            x = np.linspace(0, 2, n, endpoint=False)
            h += 0.025 * np.sin(w * x + ph)
        elif strat == 4:
            eps = np.random.randn(n) * 0.03
            eps -= eps.mean()
            h += eps
        elif strat == 5:
            for _ in range(np.random.randint(2, 7)):
                i, j = np.random.choice(n, 2, replace=False)
                h[i], h[j] = h[j], h[i]
            h += np.random.randn(n) * 0.003
        else:
            idx = np.random.choice(n, np.random.randint(4, 10), replace=False)
            h[idx] = np.random.uniform(0.1, 0.9, len(idx))

        h = project_box_sum(h, target)

        # L-BFGS-B polish with gradient
        for beta, lam in [(400, 2000), (800, 4000)]:
            if remaining() < 20:
                break
            def obj_r(hh):
                val, _ = smooth_grad(hh, n, beta)
                return val + lam * (hh.sum() - target) ** 2
            def obj_r_grad(hh):
                val, g = smooth_grad(hh, n, beta)
                return val + lam * (hh.sum() - target) ** 2, g + 2*lam*(hh.sum()-target)
            try:
                r = minimize(obj_r, h, jac=obj_r_grad, method='L-BFGS-B',
                             bounds=bnds, options={'maxiter': 300, 'ftol': 1e-16})
                h_new = project_box_sum(r.x, target)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
                    best_snapshot = h_new.copy()
            except Exception:
                pass

    # Phase 5: Final high-beta polish
    if remaining() > 30:
        for beta in [2000, 4000]:
            if remaining() < 15:
                break
            def obj_fin(h):
                val, _ = smooth_grad(h, n, beta)
                return val + 5000.0 * (h.sum() - target) ** 2
            def obj_fin_grad(h):
                val, g = smooth_grad(h, n, beta)
                return val + 5000.0 * (h.sum() - target) ** 2, g + 10000.0*(h.sum()-target)
            try:
                r = minimize(obj_fin, best_h, jac=obj_fin_grad, method='L-BFGS-B',
                             bounds=bnds, options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-15})
                h_new = project_box_sum(r.x, target)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 6: SLSQP final with very high beta
    if remaining() > 25:
        def obj_f2(h):
            return smooth_grad(h, n, 3000)[0]
        def obj_f2_grad(h):
            return smooth_grad(h, n, 3000)
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                 'jac': lambda x: np.ones(n)}]
        try:
            r = minimize(obj_f2, best_h, jac=obj_f2_grad, method='SLSQP',
                         bounds=bnds, constraints=cons,
                         options={'maxiter': 1000, 'ftol': 1e-16})
            h_new = project_box_sum(r.x, target)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    best_h = project_box_sum(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
