    return np.ones(n)

def fourier_perturb(h, n, amp, rng):
    k = rng.randint(1, max(2, n // 4))
    ph = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    return h + amp * np.sin(2 * np.pi * k * x / n + ph)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n

    best_h = project_box_sum(h_init, ts)
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS-B with increasing beta from initial
    for beta in [40, 80, 160, 320, 640, 1280, 2560]:
        if time_left() < 30:
            break
        lam = max(50.0, beta * 0.4)
        try:
            r = minimize(objective_func, best_h, args=(n, ts, beta, lam),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 600, 'ftol': 1e-16, 'gtol': 1e-10})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Phase 2: Coordinate descent polish
    if time_left() > 50:
        best_h = coordinate_descent(best_h, n, ts, max_sweeps=80)
        best_c5 = compute_c5(best_h)

    # Phase 3: Diverse restarts
    stagnation = 0
    it = 0
    while time_left() > 60:
        it += 1
        h = best_h.copy()
        strat = it % 6

        if strat == 0:
            scale = max(0.008, 0.07 * np.exp(-it / 25.0))
            h += rng.randn(n) * scale
        elif strat == 1:
            h = fourier_perturb(h, n, 0.04, rng)
        elif strat == 2:
            corr = np.correlate(h, 1.0 - h, mode="full")
            k_star = int(np.argmax(corr)) - (n - 1)
            s = np.zeros(n)
            for j in range(n):
                v = 0.0
                i1 = j + k_star
                i2 = j - k_star
                if 0 <= i1 < n:
                    v += (1.0 - h[i1])
                if 0 <= i2 < n:
                    v -= h[i2]
                s[j] = v
            h = h - 0.015 * s
        elif strat == 3:
            blen = rng.randint(4, max(5, n // 5))
            bs = rng.randint(0, n - blen)
            h[bs:bs + blen] = 0.5 + 0.3 * rng.randn(blen)
        elif strat == 4:
            h = fourier_perturb(h, n, 0.025, rng)
            h = fourier_perturb(h, n, 0.025, rng)
        else:
            # Swap pairs
            for _ in range(rng.randint(3, 8)):
                a, b = rng.choice(n, 2, replace=False)
                h[a], h[b] = h[b], h[a]
            h += rng.randn(n) * 0.003

        h = project_box_sum(h, ts)

        improved = False
        for beta in [500, 1200, 2500]:
            if time_left() < 20:
                break
            lam = max(100.0, beta * 0.3)
            try:
                r = minimize(objective_func, h, args=(n, ts, beta, lam),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 350, 'ftol': 1e-16, 'gtol': 1e-10})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
                    improved = True
                    break
                else:
                    h = h_new
            except Exception:
                pass

        if not improved:
            stagnation += 1
            if stagnation % 8 == 0:
                # Coordinate descent on current best
                best_h = coordinate_descent(best_h, n, ts, max_sweeps=30)
                best_c5 = compute_c5(best_h)
        else:
            stagnation = 0

    # Phase 4: Heavy coordinate descent polish
    if time_left() > 30:
        best_h = coordinate_descent(best_h, n, ts, max_sweeps=200)
        best_c5 = compute_c5(best_h)

    # Phase 5: Ultra-high beta L-BFGS-B
    if time_left() > 20:
        for beta in [5000, 10000]:
            if time_left() < 12:
                break
            try:
                r = minimize(objective_func, best_h, args=(n, ts, beta, 5000.0),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 8000, 'ftol': 1e-18, 'gtol': 1e-14})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 6: Final coordinate descent
    if time_left() > 10:
        best_h = coordinate_descent(best_h, n, ts, max_sweeps=300)
        best_c5 = compute_c5(best_h)

    best_h = project_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
