    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        corr = compute_corr_full(x)
        m = np.max(corr)
        exps = np.exp(beta * (corr - m))
        val = (m + np.log(exps.sum()) / beta) * dx
        # Penalty for sum constraint
        s = x.sum() - target_sum
        val += 100.0 * s * s
        return val
    
    def grad(x):
        corr = compute_corr_full(x)
        m = np.max(corr)
        exps = np.exp(beta * (corr - m))
        w = exps / exps.sum()
        g = np.zeros(n)
        j = np.arange(n)
        for k in range(len(corr)):
            wk = w[k]
            if wk < 1e-12:
                continue
            s = k - (n - 1)
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                g[ms] += wk * (1.0 - x[js[ms]])
            if mp.any():
                g[mp] += wk * (-x[jp[mp]])
        g *= dx
        # Add penalty gradient
        s = x.sum() - target_sum
        g += 200.0 * s
        return g
    
    res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                   bounds=Bounds(np.zeros(n), np.ones(n)),
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
    return project_feasible(res.x, target_sum)

    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_obj_grad(x, beta, target_sum, pen_weight=50.0):

    exps = np.exp(beta * shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    s = x.sum() - target_sum
    val += pen_weight * s * s
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:

            i_start2 = 0

def slsqp_polish(h, beta, maxiter=300):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        return smoothmax_obj_grad(x, beta, target, 0.0)[0]
    
    def grad(x):
        return smoothmax_obj_grad(x, beta, target, 0.0)[1]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=Bounds(np.zeros(n), np.ones(n)),
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)
