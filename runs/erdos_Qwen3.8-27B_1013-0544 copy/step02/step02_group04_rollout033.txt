import numpy as np
from scipy.optimize import minimize, minimize_scalar, Bounds
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def smooth_max(x, b):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(b * (x - m)))) / b

def project(h, ts):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        fr = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not fr.any():
            break
        h[fr] += d / fr.sum()
    return np.clip(h, 0.0, 1.0)

def corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def lgrad(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag > 0:
        g[:n - lag] += 1.0 - h[lag:]
        g[lag:] -= h[:n - lag]
    elif lag < 0:
        g[-lag:] += 1.0 - h[:n + lag]
        g[:n + lag] -= h[-lag:]
    else:
        g[:] = 1.0 - 2.0 * h
    return g

def tksgrad(h, k):
    c = corr(h)
    n = len(h)
    g = np.zeros(n)
    for idx in np.argsort(c)[-k:]:
        g += lgrad(h, idx - (n - 1))
    return g / float(k)

_gt = 0.0
_gb = 100.0
_gd = 1.0

def _eqf(x):
    return np.sum(x) - _gt

def _eqj(x):
    return np.ones(len(x))

def _sf(x):
    return smooth_max(corr(x), _gb) * _gd

def _cd1d(d, i, fr, h, b):
    hn_i = h[i] + d
    if hn_i < -1e-10 or hn_i > 1.0 + 1e-10:
        return 1e10
    on = h[fr] - d / len(fr)
    if on.min() < -1e-10 or on.max() > 1.0 + 1e-10:
        return 1e10
    h2 = h.copy()
    h2[i] = hn_i
    h2[fr] = on
    return smooth_max(corr(h2), b)

def dpop(n, ts, hb, np_, rng):
    pop = np.zeros((np_, n))
    pop[0] = hb
    for i in range(1, np_):
        s = i % 5
        if s == 0:
            pop[i] = hb + rng.randn(n) * 0.03
        elif s == 1:
            p = rng.randint(3, max(4, n // 2 + 1))
            ph = rng.uniform(0, p)
            pop[i] = 0.5 + 0.2 * np.sin(2 * np.pi * (np.arange(n) + ph) / p)
        elif s == 2:
            nb = rng.randint(2, 8)
            bs = n // nb
            for b in range(nb):
                st = b * bs
                en = (b + 1) * bs if b < nb - 1 else n
                pop[i][st:en] = rng.uniform(0.05, 0.95)
        elif s == 3:
            kf = rng.randint(1, 5)
            t = np.linspace(0, 2 * np.pi * kf, n)
            pop[i] = 0.5 + 0.15 * np.cos(t + rng.uniform(0, 6.28))
        else:
            pop[i] = hb[::-1] + rng.randn(n) * 0.03
        pop[i] = project(pop[i], ts)
    return pop
