def project_proper(h, target):

def smoothmax_val_grad(h, beta):
    """Compute smooth-max value and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

        return v

        _, g = smoothmax_val_grad(x, beta)
        return g

    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad

def adam_opt(h, n_steps, lr, beta_sched, patience=500):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    b1, b2 = 0.85, 0.999
    eps = 1e-8
    best_x = x.copy()
    best_val = compute_c5(x)
    stall = 0
    
    for t in range(1, n_steps + 1):
        frac = (t - 1) / max(n_steps - 1, 1)
        # Piecewise linear interpolation of beta schedule
        if len(beta_sched) == 1:
            cur_beta = beta_sched[0]
        else:
            seg = frac * (len(beta_sched) - 1)
            i0 = int(seg)
            i1 = min(i0 + 1, len(beta_sched) - 1)
            frac_seg = seg - i0
            cur_beta = beta_sched[i0] * (1 - frac_seg) + beta_sched[i1] * frac_seg
        
        val, grad = smoothmax_val_grad(x, cur_beta)
        m_v = b1 * m_v + (1 - b1) * grad
        v_v = b2 * v_v + (1 - b2) * grad ** 2
        m_hat = m_v / (1 - b1 ** t)
        v_hat = v_v / (1 - b2 ** t)
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x_new = project_proper(x_new, target)
        x = x_new
        
        if t % 100 == 0 or t == 1:
            cv = compute_c5(x)
            if cv < best_val - 1e-8:
                best_val = cv
                best_x = x.copy()
                stall = 0
            else:
                stall += 100
            if stall >= patience:
                break
    
    cv = compute_c5(x)
    if cv < best_val:
        best_val = cv
        best_x = x.copy()
    return best_x, best_val

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        return smoothmax_val_grad(x, beta)[0]
    def objg(x):
        return smoothmax_val_grad(x, beta)[1]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_proper(res.x, target)
    except Exception:
        return project_proper(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_proper(h_new, n_new / 2.0)

def ga_explore(h_parent, n_pop, time_limit, rng):
    n = len(h_parent)
    target = n / 2.0
    t0 = time.time()
    
    pop = np.zeros((n_pop, n))
    pop[0] = h_parent
    for i in range(1, n_pop):
        r = rng.random()
        if r < 0.33:
            p = np.clip(rng.standard_normal(n) * 0.2 + 0.5, 0, 1)
        elif r < 0.66:
            p = h_parent + 0.08 * rng.standard_normal(n)
        else:
            p = h_parent.copy()
            for _ in range(rng.integers(1, 4)):
                a = rng.integers(0, n - 20)
                b = a + rng.integers(10, n - a)
                p[a:b], p[b:] = p[b:].copy(), p[:b].copy()
            p = p + 0.03 * rng.standard_normal(n)
        pop[i] = project_proper(p, target)
    
    scores = np.array([compute_c5(pop[i]) for i in range(n_pop)])
    best_i = np.argmin(scores)
    best_x = pop[best_i].copy()
    best_score = scores[best_i]
    elitism = max(3, n_pop // 8)
    
    gen = 0
    while time.time() - t0 < time_limit:
        gen += 1
        new_pop = np.zeros((n_pop, n))
        new_pop[:elitism] = pop[np.argsort(scores)[:elitism]]
        
        for i in range(elitism, n_pop):
            # Tournament of 3
            cands = rng.choice(n_pop, size=3, replace=False)
            p1 = pop[cands[np.argmin(scores[cands])]]
            cands2 = rng.choice(n_pop, size=3, replace=False)
            p2 = pop[cands2[np.argmin(scores[cands2])]]
            
            cut = rng.integers(1, n - 1)
            child = np.concatenate([p1[:cut], p2[cut:]])
            
            sigma = 0.015 + 0.02 * (1 - min(gen, 500) / 500.0)
            child = child + sigma * rng.standard_normal(n)
            child = project_proper(child, target)
            new_pop[i] = child
            new_pop[i] = child
            
        scores_new = np.array([compute_c5(new_pop[i]) for i in range(n_pop)])
        pop = new_pop
        scores = scores_new
        ci = np.argmin(scores)
        if scores[ci] < best_score:
            best_score = scores[ci]
            best_x = pop[ci].copy()
    
    return best_x, best_score

                    c_s = compute_c5(h_s)
                    if c_s < best_c5:

    # Phase 4: Multi-resolution

        h_u, c_u = adam_opt(h_up, 2000, 0.03, [100, 300, 800])

        if tl() > 30:

            c_s = compute_c5(h_s)
            if c_s < best_c5:
                h_best, best_c5 = h_s, c_s
    
    # Phase 5: Final high-beta ADAM

        c_f2 = compute_c5(h_f2)
        if c_f2 < best_c5:
            h_best, best_c5 = h_f2, c_f2
    
    return h_best, best_c5, len(h_best)

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_proper(h, target):
    n = len(h)
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt > 0:
            proj[free] += diff / cnt
        else:
            proj += diff / n
        proj = np.clip(proj, 0.0, 1.0)
    return proj


def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad


def adam_opt(h, n_steps, lr0, beta_sched, patience=600):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    b1, b2 = 0.85, 0.999
    eps = 1e-8
    best_x = x.copy()
    best_val = compute_c5(x)
    stall = 0
    for t in range(1, n_steps + 1):
        frac = (t - 1) / max(n_steps - 1, 1)
        if len(beta_sched) == 1:
            cur_beta = beta_sched[0]
        else:
            seg = frac * (len(beta_sched) - 1)
            i0 = int(seg)
            i1 = min(i0 + 1, len(beta_sched) - 1)
            fs = seg - i0
            cur_beta = beta_sched[i0] * (1 - fs) + beta_sched[i1] * fs
        lr = lr0 * (1.0 - 0.8 * frac)
        val, grad = smoothmax_val_grad(x, cur_beta)
        m_v = b1 * m_v + (1 - b1) * grad
        v_v = b2 * v_v + (1 - b2) * grad ** 2
        m_hat = m_v / (1 - b1 ** t)
        v_hat = v_v / (1 - b2 ** t)
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x_new = project_proper(x_new, target)
        x = x_new
        if t % 100 == 0 or t == 1:
            cv = compute_c5(x)
            if cv < best_val - 1e-9:
                best_val = cv
                best_x = x.copy()
                stall = 0
            else:
                stall += 100
            if stall >= patience:
                break
    cv = compute_c5(x)
    if cv < best_val:
        best_val = cv
        best_x = x.copy()
    return best_x, best_val


def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj(x):
        return smoothmax_val_grad(x, beta)[0]

    def objg(x):
        return smoothmax_val_grad(x, beta)[1]

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_proper(res.x, target)
    except Exception:
        return project_proper(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_proper(h_new, n_new / 2.0)


def ga_explore(h_parent, n_pop, time_limit, rng):
    n = len(h_parent)
    target = n / 2.0
    t0 = time.time()
    pop = np.zeros((n_pop, n))
    pop[0] = h_parent
    for i in range(1, n_pop):
        r = rng.random()
        if r < 0.3:
            p = np.clip(rng.standard_normal(n) * 0.25 + 0.5, 0, 1)
        elif r < 0.6:
            p = h_parent + 0.06 * rng.standard_normal(n)
        else:
            p = h_parent.copy()
            for _ in range(rng.integers(1, 4)):
                a = rng.integers(0, n - 30)
                b = a + rng.integers(15, n - a)
                p[a:b], p[b:] = p[b:].copy(), p[:b].copy()
            p += 0.02 * rng.standard_normal(n)
        pop[i] = project_proper(p, target)
    scores = np.array([compute_c5(pop[i]) for i in range(n_pop)])
    best_i = np.argmin(scores)
    best_x = pop[best_i].copy()
    best_score = scores[best_i]
    elitism = max(3, n_pop // 8)
    gen = 0
    while time.time() - t0 < time_limit:
        gen += 1
        new_pop = np.zeros((n_pop, n))
        top_idx = np.argsort(scores)[:elitism]
        new_pop[:elitism] = pop[top_idx]
        for i in range(elitism, n_pop):
            c1 = rng.choice(n_pop, size=3, replace=False)
            p1 = pop[c1[np.argmin(scores[c1])]]
            c2 = rng.choice(n_pop, size=3, replace=False)
            p2 = pop[c2[np.argmin(scores[c2])]]
            cut = rng.integers(1, n - 1)
            child = np.concatenate([p1[:cut], p2[cut:]])
            sigma = 0.01 + 0.025 * max(0, 1 - gen / 300.0)
            child += sigma * rng.standard_normal(n)
            new_pop[i] = project_proper(child, target)
        scores = np.array([compute_c5(new_pop[i]) for i in range(n_pop)])
        pop = new_pop
        ci = np.argmin(scores)
        if scores[ci] < best_score:
            best_score = scores[ci]
            best_x = pop[ci].copy()
    return best_x, best_score


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)

    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: ADAM with Beta annealing (full smooth-max gradient)
    if tl() > 180:
        h1, c1 = adam_opt(h_best, 4000, 0.04, [30, 100, 300, 800])
        if c1 < best_c5:
            h_best, best_c5 = h1, c1

    # Phase 2: SLSQP polish
    if tl() > 60:
        h2 = slsqp_refine(h_best, 500, 400)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            h_best, best_c5 = h2, c2

    # Phase 3: GA exploration for diversity
    if tl() > 280:
        h_ga, c_ga = ga_explore(h_best, 50, tl() * 0.3, rng)
        if c_ga < best_c5:
            h_best, best_c5 = h_ga, c_ga
            if tl() > 100:
                h_r, c_r = adam_opt(h_best, 2500, 0.025, [100, 400, 1000])
                if c_r < best_c5:
                    h_best, best_c5 = h_r, c_r
                if tl() > 40:
                    h_s = slsqp_refine(h_best, 400, 800)
                    c_s = compute_c5(h_s)
                    if c_s < best_c5:
                        h_best, best_c5 = h_s, c_s

    # Phase 4: Multi-resolution attempts
    for n_try in [150, 100, 300]:
        if tl() < 70:
            break
        h_up = interp_to_n(h_best, n_try)
        h_u, c_u = adam_opt(h_up, 2500, 0.03, [50, 200, 600])
        if c_u < best_c5:
            h_best, best_c5 = h_u, c_u
        if tl() > 30:
            h_s = slsqp_refine(h_best, 300, 600)
            c_s = compute_c5(h_s)
            if c_s < best_c5:
                h_best, best_c5 = h_s, c_s

    # Phase 5: Final high-beta ADAM
    if tl() > 50:
        h_f, c_f = adam_opt(h_best, 3000, 0.015, [300, 800, 2000])
        if c_f < best_c5:
            h_best, best_c5 = h_f, c_f

    # Phase 6: Final SLSQP
    if tl() > 25:
        h_f2 = slsqp_refine(h_best, 600, 1500)
        c_f2 = compute_c5(h_f2)
        if c_f2 < best_c5:
            h_best, best_c5 = h_f2, c_f2

    # Phase 7: One more short Adam pass
    if tl() > 20:
        h_f3, c_f3 = adam_opt(h_best, 1000, 0.008, [2000])
        if c_f3 < best_c5:
            h_best, best_c5 = h_f3, c_f3

    return h_best, best_c5, len(h_best)
