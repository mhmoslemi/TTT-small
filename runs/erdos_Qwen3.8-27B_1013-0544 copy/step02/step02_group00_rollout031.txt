    return float(np.max(corr) * dx)

    # Second factor: k=1..i, change = +1 if h[i-k]=1
    mask2 = h[:i][::-1] == 1  # h[i-k] for k=1..i → h[i-1], h[i-2], ..., h[0]
    deltas[1:i+1] += mask2.astype(float)

    # First factor: k=0..n-1-j, change = +(1-h[j+k]) = +1 if h[j+k]=0
    mask3 = h[j:n] == 0
    deltas[:n-j] += mask3.astype(float)
    
    # Second factor: k=1..j, change = -1 if h[j-k]=1
    mask4 = h[:j][::-1] == 1
    deltas[1:j+1] -= mask4.astype(float)
    
    return overlaps + deltas
