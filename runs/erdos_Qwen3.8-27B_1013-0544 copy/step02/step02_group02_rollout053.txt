import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-14:
                break
            h += (d / t) * room
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def c5_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    s = alpha * c
    m = s.max()
    if m > 500:
        return c.max() + np.log(np.sum(np.exp(s - m))) / alpha
    return np.log(np.sum(np.exp(s))) / alpha

def c5_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    s = alpha * c
    m = s.max()
    es = np.exp(s - m)
    w = es / es.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        sh = k - (n - 1)
        wk = w[k]
        if wk < 1e-10:
            continue
        # First term: (1-x[i+sh]) for valid i
        i0 = max(0, -sh)
        i1 = min(n - 1, n - 1 - sh)
        if i0 <= i1:
            grad[i0:i1+1] += wk * (1.0 - x[i0+sh:i1+sh+1])
        # Second term: -x[i-sh] for valid i
        j0 = max(0, sh)
        j1 = min(n - 1, n - 1 + sh)
        if j0 <= j1:
            grad[j0:j1+1] -= wk * x[j0-sh:j1-sh+1]
    return grad

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(c5_obj, h, args=(n, alpha), method='SLSQP',
                       jac=lambda x: c5_grad(x, n, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-14:
                break
            h += (d / t) * room
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def c5_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    s = alpha * c
    m = s.max()
    es = np.exp(s - m)
    return (m + np.log(es.sum())) / alpha

def c5_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    s = alpha * c
    m = s.max()
    es = np.exp(s - m)
    w = es / es.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        sh = k - (n - 1)
        wk = w[k]
        if wk < 1e-10:
            continue
        i0 = max(0, -sh)
        i1 = min(n - 1, n - 1 - sh)
        if i0 <= i1:
            grad[i0:i1 + 1] += wk * (1.0 - x[i0 + sh:i1 + sh + 1])
        j0 = max(0, sh)
        j1 = min(n - 1, n - 1 + sh)
        if j0 <= j1:
            grad[j0:j1 + 1] -= wk * x[j0 - sh:j1 - sh + 1]
    return grad

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(c5_obj, h, args=(n, alpha), method='SLSQP',
                       jac=c5_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def upscale(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_sin(n, freq, amp):
    x = np.linspace(0, 2, n, endpoint=False)
    return project_feasible(0.5 + amp * np.sin(np.pi * freq * x), n)

def make_cos(n, freq, amp):
    x = np.linspace(0, 2, n, endpoint=False)
    return project_feasible(0.5 + amp * np.cos(np.pi * freq * x), n)

def optimize_sequence(h, n, alphas, t0, budget_s, maxiter_base, ftol_base):
    for i, alpha in enumerate(alphas):
        if budget_s - (time.time() - t0) < 20:
            break
        mi = maxiter_base + i * 2000
        h = slsqp(h, n, alpha, mi, ftol_base)
    return h

def coord_descent(h, n, t0, budget_s, rng, max_iters):
    best_c5 = compute_c5(h, n)
    for it in range(max_iters):
        if budget_s - (time.time() - t0) < 5:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice([1.0, -1.0]) * rng.uniform(0.0005, 0.02)
        hi, hj = h[i] + d, h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            c = compute_c5(h, n)
            if c < best_c5:
                best_c5 = c
            else:
                h[i], h[j] = hi - d, hj + d
    return h, best_c5

    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
