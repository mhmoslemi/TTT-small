**Coordinate descent for this problem:**
The key insight is that we can maintain the sum constraint exactly by always modifying TWO coordinates at a time (one increases, one decreases by the same amount).

Algorithm:
1. Compute all correlations
2. Find the dominant lag k* (the one giving the max)
3. For pairs (i, j) where i and j are related by the dominant lag:

```python
def deterministic_anneal(h, n, n_steps, n_trials, step, rng):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5

        best_move = None
        best_move_c5 = cur_c5
        for _ in range(n_trials):

            if i == j:
                j = (i+1) % n
            delta = rng.uniform(-step, step)
            ni = cur[i] + delta
            nj = cur[j] - delta
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:

            cur[i], cur[j] = ni, nj
            c = compute_c5(cur)
            cur[i], cur[j] = cur[i]-delta, cur[j]+delta  # revert
            if c < best_move_c5:
                best_move_c5 = c
                best_move = (i, j, delta)
        
        if best_move is not None:
            i, j, delta = best_move
            cur[i] += delta
            cur[j] -= delta
            cur_c5 = best_move_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = cur.copy()
    
    return best, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    diffs = np.clip(beta * (x - m), -500, 0)
    return m + np.log(np.sum(np.exp(diffs))) / beta

def project_box_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        inner = (h > 1e-10) & (h < 1.0 - 1e-10)
        if inner.sum() == 0:
            break
        h[inner] += d / inner.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def polish(h, n, beta, lam, maxiter):
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(c, beta) * (2.0 / n) + lam * (x.sum() - target) ** 2
    try:
        r = minimize(obj, h, method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_box_sum(r.x, target)
    except Exception:
        return h

def sa_loop(h, n, n_steps, T_start, T_end, step_start, step_end, rng):

    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5

        frac = s / max(n_steps - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        step = step_start * (step_end / step_start) ** frac
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            j = (i + 1) % n
        delta = rng.uniform(-step, step)
        ni = cur[i] + delta
        nj = cur[j] - delta
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        cur[i], cur[j] = ni, nj
        new_c5 = compute_c5(cur)
        if new_c5 <= cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = cur.copy()
        else:
            cur[i], cur[j] = ni - delta, nj + delta
    return best, best_c5
```
