import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)


def obj_and_grad(h, beta, lam):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    ev = np.exp(beta * (corr - m))
    w = ev / ev.sum()
    smooth_val = m + np.log(ev.sum()) / beta
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        for k in range(-n + 1, n):
            wk = w[k + n - 1]
            i1 = j - k
            if 0 <= i1 < n:
                g += wk * (1.0 - h[i1])
            i2 = j + k
            if 0 <= i2 < n:
                g -= wk * h[i2]
        grad[j] = g
    grad *= dx
    pen = lam * (h.sum() - n / 2.0) ** 2
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return smooth_val * dx + pen, grad


def make_obj(h, beta, lam):
    return obj_and_grad(h, beta, lam)[0]


def make_jac(h, beta, lam):
    return obj_and_grad(h, beta, lam)[1]
