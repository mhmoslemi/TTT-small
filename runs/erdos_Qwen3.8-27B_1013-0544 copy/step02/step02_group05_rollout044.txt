        return np.max(corr) * dx

    best_h = h.copy()

        if len(ones) == 0 or len(zeros) == 0:

        j = zeros[rng.randint(len(zeros))]

        h[j] = 1

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

class ObjSM:
    def __init__(self, beta, dx):
        self.beta = beta
        self.dx = dx
    def __call__(self, h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(self.beta * (corr - m)))) / self.beta) * self.dx

class ConSum:
    def __init__(self, target_sum):
        self.ts = target_sum
    def __call__(self, h):
        return np.sum(h) - self.ts
    def jac(self, h):
        return np.ones_like(h)

def binary_sa_search(n, target_sum, max_iter, seed, time_fn, budget_threshold):

    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    ones_set = set(ones.tolist())
    zeros_set = set(range(n)) - ones_set
    ones_list = list(ones)
    zeros_list = list(zeros_set)

    def overlap_max(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return float(np.max(corr))

    cur_val = overlap_max(h)
    best_val = cur_val
    best_h = h.copy()
    T = 2.0
    T_min = 0.01
    no_improve = 0

    for it in range(max_iter):
        if time_fn() > budget_threshold:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        T = max(T_min, T * 0.9995)
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        new_val = overlap_max(h)
        delta = new_val - cur_val
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            ones_list = [x for x in ones_list if x != i]
            ones_list.append(j)
            zeros_list = [x for x in zeros_list if x != j]

                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        if no_improve > 3000:
            break
    return best_h, best_val * (2.0 / n)

def slsqp_refine(h0, n, target_sum, dx, bounds, con, beta_list, time_fn, threshold, max_iter=300):
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)

            break

                c5_cur = c5_new
