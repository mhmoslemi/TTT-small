def project_box_sum(h, target):

                h[idx] = max(0.0, h[idx] + d)
            continue

        h = np.clip(h, 0.0, 1.0)

    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def smooth_max_grad(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted - np.max(shifted))  # numerical stability
    exps = np.exp(beta * (corr - m))
    total = np.sum(exps)
    value = m + np.log(total) / beta
    weights = exps / total
    return value, weights

    """Compute gradient of sum_k w_k * corr_k w.r.t. h.
    corr_k = np.correlate(h, 1-h, mode='full')[k]
    which equals sum_i h[i]*(1-h[i+s]) where s = n-1-k
    """
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        s = n - 1 - k_idx  # shift
        if s >= 0:

            if n - 1 - s >= 0:

                grad[js] += w * (1.0 - h[js + s])
            # Second term: j from s to n-1, contributes -h[j-s]
            if s <= n - 1:
                js = np.arange(s, n)
                grad[js] -= w * h[js - s]
        else:
            m_abs = -s
            # First term: j from m_abs to n-1, contributes (1-h[j-m_abs])
            js = np.arange(m_abs, n)
            grad[js] += w * (1.0 - h[js - m_abs])
            # Second term: j from 0 to n-1-m_abs, contributes -h[j+m_abs]
            if n - 1 - m_abs >= 0:
                js = np.arange(0, n - m_abs)
                grad[js] -= w * h[js + m_abs]
    return grad

    sm = smooth_max_val(corr, beta)
    penalty = lam * (h.sum() - ts) ** 2
    return sm * (2.0 / n) + penalty

    g += 2.0 * lam * (h.sum() - ts) * np.ones(n)

The first part is linear in h. The second part is a quadratic form: sum_{i,k} p_k * h[i+k]*h[i]. We can write this as h^T Q h where Q is a specific matrix.

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h > 1e-14) & (h < 1.0 - 1e-14)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    return exps / np.sum(exps)

def corr_grad(h, n, weights):
    """Gradient of sum_k w_k * corr_k w.r.t. h. O(n^2)."""
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        s = n - 1 - k_idx
        if s >= 0:
            # j from 0 to n-1-s: +w*(1-h[j+s])
            if n - s > 0:
                grad[:n-s] += w * (1.0 - h[s:])
            # j from s to n-1: -w*h[j-s]
            if s < n:
                grad[s:] -= w * h[:n-s]
        else:
            m = -s
            # j from m to n-1: +w*(1-h[j-m])
            grad[m:] += w * (1.0 - h[:n-m])
            # j from 0 to n-1-m: -w*h[j+m]
            if n - m > 0:
                grad[:n-m] -= w * h[m:]
    return grad

def obj_fn(h, n, ts, beta, lam):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(corr, beta) * (2.0/n) + lam * (h.sum()-ts)**2

def obj_grad(h, n, ts, beta, lam):
    corr = np.correlate(h, 1.0 - h, mode="full")
    wts = smooth_max_weights(corr, beta)
    g = corr_grad(h, n, wts) * (2.0/n)
    g += 2.0 * lam * (h.sum() - ts)
    return g

                    options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol})
        return project_box_sum(r.x, ts)
    except Exception:
        return h
