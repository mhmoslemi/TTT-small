            valid = np.arange(n - la)
            grad[valid + la] += (1.0 - h[valid])
            grad[valid] -= h[valid + la]

                       0.02, 0.05, 0.1, 0.2, 0.3, 0.5])

            break

                if val < best_val:
                    best_val = val
                    best_t = t
            h[j] = best_t
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def obj_slsqp(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def con_fun(h, ts, n):

    x_old = np.linspace(0, 2, len(h_old), endpoint=False)

def subgrad_multi(h, top_m, n):

            grad[valid + lag] -= h[valid]
        else:

            for d in deltas:
                t = old + d
                if t < -1e-9 or t > 1 + 1e-9:

                for p in partners:
                    k = (j + p) % n
                    if k == j:
                        continue
                    tk = h[k] - d
                    if tk < -1e-9 or tk > 1 + 1e-9:
                        continue
                    tk = np.clip(tk, 0.0, 1.0)
                    h[j] = t
                    h[k] = tk
                    val = compute_c5(h)
                    if val < best_val:

                        best_tj = t
                        best_tk_list = [(p, tk)]
            h[j] = best_tj
            for p, tk in best_tk_list:
                k = (j + p) % n
                if k == j:
                    continue
                h[k] = tk
    return h
