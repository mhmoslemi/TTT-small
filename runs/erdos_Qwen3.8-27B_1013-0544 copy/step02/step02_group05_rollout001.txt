    """Compute gradient of corr[lag_offset] w.r.t. h.
    lag_offset is the index in the full correlation (0 to 2n-2).
    Actual lag k = lag_offset - (n-1).

        # Term 2: -h[m+k] if m+k valid
        j2 = m + k
        if 0 <= j2 < n:
            grad[m] -= h[j2]
    return grad

    weights = np.exp(beta * (top_vals - m))

    weights /= sw

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if abs(s - target_sum) < 1e-14:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        hs = np.clip(h - mid, 0.0, 1.0)
        if hs.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-13:
        mask = (h_proj > 1e-12) & (h_proj < 1.0 - 1e-12)
        if mask.any():
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def smooth_max_obj(corr, beta):
    m = corr.max()
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def lag_gradient(h, lag_idx):
    n = len(h)
    k = lag_idx - (n - 1)
    grad = np.zeros(n)
    for m in range(n):
        j1 = m - k
        if 0 <= j1 < n:
            grad[m] += (1.0 - h[j1])
        j2 = m + k
        if 0 <= j2 < n:
            grad[m] -= h[j2]
    return grad

def projected_subgrad(h, dx, top_k, beta, lr, n_steps, target_sum):
    h = h.copy()
    n = len(h)
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        sort_idx = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[sort_idx]
        m = top_vals.max()
        weights = np.exp(beta * (top_vals - m))
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, idx in zip(weights, sort_idx):
            if w > 1e-10:
                grad += w * lag_gradient(h, idx)
        lr_eff = lr / (1.0 + 0.005 * step)
        h = h - lr_eff * grad * dx
        h = project_feasible(h, target_sum)
    return h

def slsqp_step(h0, n_points, target_sum, beta, maxiter, ftol):
    dx = 2.0 / n_points
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max_obj(corr, beta) * dx

    def obj_grad(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = corr.max()
        w = np.exp(beta * (corr - m))
        w /= w.sum()
        grad = np.zeros(n_points)
        for i in range(len(corr)):
            if w[i] > 1e-10:
                grad += w[i] * lag_gradient(h, i)
        return grad * dx

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n_points)}]
    res = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent_h = np.array(initial_h_values, dtype=float).copy()
    else:
        parent_h = None

    resolutions = [70, 80, 100]

    for n_points in resolutions:
        if remaining() < 120:
            break
        t_res = time.time()
        target_sum = n_points / 2.0
        dx = 2.0 / n_points

        starts = []
        if parent_h is not None:
            if len(parent_h) == n_points:
                starts.append(parent_h.copy())
            else:
                xo = np.linspace(0, 2, len(parent_h), endpoint=False)
                xn = np.linspace(0, 2, n_points, endpoint=False)
                h_interp = np.interp(xn, xo, parent_h, period=2.0)
                starts.append(project_feasible(h_interp, target_sum))
        starts.append(np.full(n_points, 0.5))
        # A structured start: cosine-based
        xs = np.linspace(0, 2, n_points, endpoint=False)
        h_cos = 0.5 + 0.25 * np.cos(2.0 * np.pi * xs / 2.0) + 0.1 * np.cos(4.0 * np.pi * xs / 2.0)
        starts.append(project_feasible(h_cos, target_sum))

        curr_h = None
        curr_c5 = float('inf')

        for h0 in starts:
            h0 = project_feasible(h0, target_sum)
            c5_0 = compute_c5(h0)
            if c5_0 < curr_c5:
                curr_h, curr_c5 = h0.copy(), c5_0

            # SLSQP with increasing beta
            for beta in [30, 60, 120, 250]:
                if remaining() < 100:
                    break
                if time.time() - t_res > 150:
                    break
                try:
                    h_new = slsqp_step(curr_h, n_points, target_sum, beta, 300, 1e-14)
                    h_new = project_feasible(h_new, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < curr_c5:
                        curr_h, curr_c5 = h_new.copy(), c5
                except Exception:
                    pass

            # Projected subgradient
            if curr_h is not None and remaining() > 80:
                h_sub = curr_h.copy()
                for top_k in [2, 3, 5]:
                    if remaining() < 60:
                        break
                    for beta in [80, 150]:
                        for lr in [0.2, 0.5, 1.0]:
                            if remaining() < 50:
                                break
                            h_try = projected_subgrad(h_sub, dx, top_k, beta, lr,
                                                      150, target_sum)
                            c5 = compute_c5(h_try)
                            if c5 < curr_c5:
                                curr_c5 = c5
                                curr_h = h_try
                                h_sub = h_try.copy()

                # Polish after subgradient
                if remaining() > 40:
                    try:
                        h_new = slsqp_step(curr_h, n_points, target_sum, 300, 500, 1e-15)
                        h_new = project_feasible(h_new, target_sum)
                        c5 = compute_c5(h_new)
                        if c5 < curr_c5:
                            curr_h, curr_c5 = h_new.copy(), c5
                    except Exception:
                        pass

            if curr_c5 < best_c5:
                best_h, best_c5, best_n = curr_h.copy(), curr_c5, n_points

        # Informed stochastic exploration
        if best_n == n_points and best_h is not None and remaining() > 60:
            base = best_h.copy()
            for it in range(30):
                if remaining() < 40:
                    break
                corr = np.correlate(base, 1.0 - base, mode='full')
                max_idx = np.argmax(corr)
                # Gradient of the max lag
                g_max = lag_gradient(base, max_idx)
                # Perturb in negative gradient direction
                scale = 0.04 + 0.02 * np.random.random()
                h_try = base - scale * g_max
                h_try += np.random.randn(n_points) * 0.01
                h_try = project_feasible(h_try, target_sum)
                try:
                    h_new = slsqp_step(h_try, n_points, target_sum, 200, 150, 1e-13)
                    h_new = project_feasible(h_new, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new.copy(), c5
                        base = h_new.copy()
                except Exception:
                    pass

    # Final high-precision polish
    if best_h is not None and remaining() > 20:
        n_f = best_n
        ts = n_f / 2.0
        for beta in [400, 600]:
            if remaining() < 10:
                break
            try:
                h_new = slsqp_step(best_h, n_f, ts, beta, 1000, 1e-16)
                h_new = project_feasible(h_new, ts)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    return best_h, best_c5, best_n
