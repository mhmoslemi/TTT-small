    return np.correlate(h, 1.0 - h, mode="full")


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def smooth_max_grad_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50, 50)
    w = np.exp(shifted)
    w /= w.sum()
    return w


def grad_smooth_max(h, beta):
    n = len(h)
    corr = compute_corr(h)
    w = smooth_max_grad_weights(corr, beta)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            idx1 = np.arange(n - d)
            grad[idx1] += wk * (1.0 - h[idx1 + d])
            idx2 = np.arange(d, n)
            grad[idx2] -= wk * h[idx2 - d]
        else:
            da = -d
            idx1 = np.arange(da, n)
            grad[idx1] += wk * (1.0 - h[idx1 - da])
            idx2 = np.arange(n - da)
            grad[idx2] -= wk * h[idx2 + da]
    return grad


def project_h(h, target_sum):
    h = h.copy()
    for _ in range(60):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)


def make_initial(n, h_parent, variant):
    if variant == 0 and h_parent is not None and len(h_parent) == n:
        return h_parent.copy()
    if variant == 0 and h_parent is not None:
        x_old = np.linspace(0, 2, len(h_parent), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        return np.interp(x_new, x_old, h_parent)
    if variant == 1:
        # Alternating blocks
        h = np.zeros(n)
        block = max(2, n // 20)
        for i in range(n):
            h[i] = 1.0 if (i // block) % 2 == 0 else 0.0
        return h
    if variant == 2:
        # Sine wave
        x = np.linspace(0, 2, n, endpoint=False)
        return 0.5 + 0.4 * np.sin(4 * np.pi * x / 2)
    # Gaussian bump
    x = np.linspace(0, 2, n, endpoint=False)
    center = n // 3
    h = 0.3 + 0.5 * np.exp(-((x - 2.0 * center / n) ** 2) / (0.05))
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def tleft():
        return budget_s - (time.time() - start_time)

    h_parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()

    best_h = None
    best_c5 = float('inf')

    # Choose n values to try based on budget
    n_vals = [70, 64, 80, 72, 90, 96]
    if budget_s < 600:
        n_vals = [70, 64, 80]

    for n_pt in n_vals:
        if tleft() < 80:
            break
        target_sum = n_pt / 2.0

        for variant in range(3 if n_pt == len(h_parent) if h_parent is not None else 2 else 0):
            if tleft() < 50:
                break
            h_init = make_initial(n_pt, h_parent, variant)
            h_init = project_h(h_init, target_sum)
            c5_cur = compute_c5(h_init)

            if c5_cur < best_c5:
                best_h = h_init.copy()
                best_c5 = c5_cur

            # Phase 1: L-BFGS-B with penalty (multi-scale beta)
            h_cur = h_init.copy()
            bounds_list = [(0.0, 1.0)] * n_pt

            for beta in [20, 50, 100, 200, 400, 800, 1500]:
                if tleft() < 30:
                    break

                def obj_l(h, _beta=beta):
                    corr = compute_corr(h)
                    return smooth_max(corr, _beta) + 50.0 * (h.sum() - target_sum) ** 2

                def grad_l(h, _beta=beta):
                    return grad_smooth_max(h, _beta) + 100.0 * (h.sum() - target_sum)

                try:
                    res = minimize(obj_l, h_cur, jac=grad_l, method='L-BFGS-B',
                                   bounds=bounds_list,
                                   options={'maxiter': 3000, 'ftol': 1e-16,
                                            'gtol': 1e-14, 'maxfun': 100000})
                    h_new = project_h(res.x, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_h, best_c5 = h_new.copy(), c5_new
                    if c5_new <= c5_cur + 1e-10:
                        h_cur = h_new
                        c5_cur = c5_new
                except Exception:
                    pass

            # Phase 2: SLSQP with equality constraint (polish)
            if tleft() < 25:
                continue

            def obj_s(h):
                return smooth_max(compute_corr(h), 500)

            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                     'jac': lambda x: np.ones(n_pt)}]
            bnds = Bounds(np.zeros(n_pt), np.ones(n_pt))

            for beta_s in [300, 800, 2000, 5000]:
                if tleft() < 15:
                    break
                try:
                    res = minimize(obj_s, h_cur, method='SLSQP',
                                   bounds=bnds, constraints=cons,
                                   options={'maxiter': 5000, 'ftol': 1e-15})
                    h_new = project_h(res.x, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_h, best_c5 = h_new.copy(), c5_new
                    if c5_new <= c5_cur + 1e-10:
                        h_cur, c5_cur = h_new, c5_new
                except Exception:
                    pass

            # Phase 3: Coordinate descent
            if tleft() < 30:
                continue

            h_cd = h_cur.copy()
            n_active = 8
            for cd_round in range(100):
                if tleft() < 15:
                    break
                corr = compute_corr(h_cd)
                top_lags = np.argsort(corr)[-n_active:]
                improved = False
                for i in range(n_pt):
                    old_v = h_cd[i]
                    best_v = old_v
                    best_score = np.max(corr[top_lags])
                    # Try targeted values
                    candidates = [0.0, 1.0,
                                  max(0.0, old_v - 0.08), max(0.0, old_v - 0.03),
                                  min(1.0, old_v + 0.03), min(1.0, old_v + 0.08),
                                  1.0 - old_v]
                    for cv in candidates:
                        if abs(cv - old_v) < 1e-6:
                            continue
                        h_cd[i] = cv
                        c = compute_corr(h_cd)
                        s = np.max(c[top_lags])
                        if s < best_score - 1e-10:
                            best_score = s
                            best_v = cv
                    h_cd[i] = best_v
                    if best_v != old_v:
                        improved = True
                h_cd = project_h(h_cd, target_sum)
                c5_cd = compute_c5(h_cd)
                if c5_cd < best_c5:
                    best_h, best_c5 = h_cd.copy(), c5_cd
                if not improved:
                    break

            # Phase 4: Short simulated annealing
            if tleft() < 20:
                continue
            h_sa = h_cd.copy()
            T = 0.003
            for it in range(2000):
                if tleft() < 10:
                    break
                h_new = h_sa.copy()
                n_mod = rng.randint(2, 7)
                idx = rng.choice(n_pt, n_mod, replace=False)
                h_new[idx] += rng.randn(n_mod) * 0.02
                h_new = project_h(h_new, target_sum)
                c5_new = compute_c5(h_new)
                c5_cur_sa = compute_c5(h_sa)
                if c5_new < c5_cur_sa or rng.random() < np.exp(-min(80.0, (c5_new - c5_cur_sa) / T)):
                    h_sa = h_new
                    if c5_new < best_c5:
                        best_h, best_c5 = h_sa.copy(), c5_new
                T *= 0.998

    # Final verification
    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = compute_c5(best_h)

    fn = len(best_h)
    best_h = project_h(best_h, fn / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, fn
