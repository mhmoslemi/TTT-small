                            h_new[j] -= step * 0.5
                            h_new[i] += step * 0.5
        elif r < 0.55:
            n_pts = np.random.randint(2, max(3, n // 10))
            idx = np.random.choice(n, n_pts, replace=False)
            h_new[idx] += np.random.randn(n_pts) * (0.01 + 0.06 * T / T_init)
        elif r < 0.75:
            for tl in top_idx[:3]:
                td = int(tl) - (n - 1)
                ad = abs(td)
                if 0 < ad < n:
                    valid = n - ad
                    if valid > 2:
                        if td > 0:
                            contrib = h_cur[:valid] * (1.0 - h_cur[ad:ad + valid])
                        else:
                            contrib = h_cur[ad:ad + valid] * (1.0 - h_cur[:valid])
                        total = contrib.sum()
                        if total > 1e-10:
                            i = np.random.choice(valid, p=contrib / total)
                        else:
                            i = np.random.randint(valid)
                        j = i + ad
                        if j < n:
                            hi, hj = h_new[i], h_new[j]
                            if abs(hi - hj) > 0.03:
                                step = 0.02 + 0.04 * np.random.random()
                                if hi > hj:
                                    h_new[i] -= step * 0.5
                                    h_new[j] += step * 0.5
                                else:
                                    h_new[j] -= step * 0.5
                                    h_new[i] += step * 0.5
        elif r < 0.90:
            blen = np.random.randint(2, min(6, n // 6 + 1))
            i = np.random.randint(0, n - blen)
            h_new[i:i + blen] = np.roll(h_new[i:i + blen], np.random.choice([-1, 1]))
        else:
            i, j = np.random.choice(n, 2, replace=False)
            h_new[i], h_new[j] = h_new[j], h_new[i]

        h_new = project_h(h_new, target_sum)
        c5_new, corr_new = compute_c5(h_new)
        delta = c5_new - c5_cur
        if delta < 0 or (T > 1e-12 and np.random.random() < np.exp(-min(300.0, delta / T))):
            h_cur = h_new
            c5_cur = c5_new
            corr_cur = corr_new
            if c5_cur < best_c5:
                best_h = h_cur.copy()
                best_c5 = c5_cur
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        T *= cooling
        if no_imp > no_imp_thresh:
            T = max(T, T_init * 0.3)
            no_imp = 0

    return best_h, best_c5

def slsqp_refine(h, target_sum, t_left_fn):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    for beta in [60, 150, 400, 800]:
        if t_left_fn() < 12:
            break
        try:
            res = minimize(obj_func, best_h, args=(beta, dx), method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 4000, 'ftol': 1e-15})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)[0]
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass
    return best_h, best_c5

def binary_init(n):
    h = np.zeros(n)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def optimize_for_n(n, h_ref, t_left_fn, phase_budget):
    target_sum = n / 2.0
    best_h = None
    best_c5 = float('inf')

    initializations = []
    if h_ref is not None:
        if len(h_ref) == n:
            initializations.append(h_ref.copy())
        else:
            xr = np.linspace(0.5 / len(h_ref), 2.0 - 0.5 / len(h_ref), len(h_ref))
            xn = np.linspace(0.5 / n, 2.0 - 0.5 / n, n)
            initializations.append(np.interp(xn, xr, h_ref))
    initializations.append(binary_init(n))
    initializations.append(np.full(n, 0.5))

    n_inits = len(initializations)
    per_init_budget = max(10, phase_budget / n_inits)

    for h0 in initializations:
        if t_left_fn() < 20:
            break
        h0 = project_h(h0, target_sum)
        t_phase = per_init_budget * 0.5
        h1, c5_1 = targeted_sa(h0, target_sum, t_left_fn, 0.008, 0.9992, 250, t_phase)
        if t_left_fn() > 12 and c5_1 < best_c5:
            h1, c5_1 = slsqp_refine(h1, target_sum, t_left_fn)
        if t_left_fn() > 15 and c5_1 < best_c5:
            t_phase2 = min(per_init_budget * 0.3, max(5, t_left_fn() * 0.2))
            h2, c5_2 = targeted_sa(h1, target_sum, t_left_fn, 0.002, 0.9996, 150, t_phase2)
            if c5_2 < c5_1:
                h1, c5_1 = h2, c5_2
        if c5_1 < best_c5:
            best_h, best_c5 = h1.copy(), c5_1

    if best_h is None:
        best_h = project_h(initializations[0], target_sum)
        best_c5 = compute_c5(best_h)[0]
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    def t_left_fn():
        return budget_s - (time.time() - start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_ref = np.array(initial_h_values, dtype=float).copy()
    else:
        h_ref = None

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if h_ref is not None:
        n_ref = len(h_ref)
        h_ref = project_h(h_ref, n_ref / 2.0)
        best_h, best_c5, best_n = h_ref.copy(), compute_c5(h_ref)[0], n_ref

    n_values = [70, 60, 80, 50, 90, 100, 65, 75, 55, 85]
    time_fracs = [0.30, 0.12, 0.12, 0.08, 0.08, 0.07, 0.05, 0.05, 0.04, 0.04]
    total_frac = sum(time_fracs)

    for i, n in enumerate(n_values):
        if t_left_fn() < 30:
            break
        frac = time_fracs[i] / total_frac
        phase_budget = frac * budget_s * 0.7
        h_t, c5_t, _ = optimize_for_n(n, h_ref, t_left_fn, phase_budget)
        if c5_t < best_c5:
            best_h, best_c5, best_n = h_t.copy(), c5_t, n

    if best_h is not None and t_left_fn() > 25:
        ts = best_n / 2.0
        best_h, best_c5 = slsqp_refine(best_h, ts, t_left_fn)
    if best_h is not None and t_left_fn() > 25:
        ts = best_n / 2.0
        t_final = max(5, t_left_fn() * 0.4)
        best_h, best_c5 = targeted_sa(best_h, ts, t_left_fn, 0.001, 0.9997, 120, t_final)
    if best_h is not None and t_left_fn() > 15:
        ts = best_n / 2.0
        best_h, best_c5 = slsqp_refine(best_h, ts, t_left_fn)

    ts = best_n / 2.0
    best_h = project_h(best_h, ts)
    best_c5 = compute_c5(best_h)[0]
    return best_h, best_c5, best_n
