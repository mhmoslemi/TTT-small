```python
    for lag in top_lags:
        if lag == 0:
            # sum h[n]*(1-h[n]), d/dh[j] = (1-h[j]) - h[j] = 1-2h[j]
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            # sum_{n=0}^{n-1-lag} h[n+lag]*(1-h[n])
            # d/dh[j]: (1-h[j-lag]) for j in [lag, n-1], -h[j+lag] for j in [0, n-1-lag]
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n-lag] -= h[lag:]
        else:
            m = -lag
            # sum_{n=m}^{n-1} h[n-m]*(1-h[n])
            # d/dh[j]: (1-h[j+m]) for j in [0, n-1-m], -h[j-m] for j in [m, n-1]
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
    return grad
```

```python
            best_h = h_cur.copy()
            best_c5 = cur_c5
            cur_best_n = n_points

        con = make_eq_con(target_sum)
        jac = make_eq_jac((n_points,))

        for beta in [40, 80, 160, 320, 640, 1280, 2560]:
            if time_left() < 12:
                break
            obj = make_slsqp_obj(dx, beta)
            constraints = [{'type': 'eq', 'fun': con, 'jac': jac}]
            try:
                res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 1200, 'ftol': 1e-15})
                h_new = project_h(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < cur_c5 - 1e-14:
                    h_cur = h_new.copy()
                    cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = cur_c5
            except Exception:
                pass

    # Phase 3: Intensive subgradient descent with adaptive steps
    n_final = len(best_h)
    target_final = n_final / 2.0
    no_improve = 0

    for iteration in range(20000):
        if time_left() < 4:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        n_active = min(7, max(3, n_final // 20))
        top_idx = np.argsort(corr)[::-1][:n_active]
        top_lags = [int(i) - (n_final - 1) for i in top_idx]

        grad = compute_subgrad(best_h, top_lags)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-11:
            break
        grad /= gnorm

        found = False
        for step in [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]:
            h_trial = project_h(best_h - step * grad, target_final)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
                found = True
                no_improve = 0
                break

        if not found:
            no_improve += 1
            if no_improve > 200 and iteration % 100 == 99:
                perturb = np.random.randn(n_final) * 0.008
                h_trial = project_h(best_h + perturb, target_final)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    best_h = h_trial
                    best_c5 = c5_trial
                    no_improve = 0
            if no_improve > 2000:
                break

    # Phase 4: Targeted pairwise moves on critical lags
    for iteration in range(40000):
        if time_left() < 2:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:3]
        improved = False

        for idx in top_idx:
            if time_left() < 1:
                break
            lag = int(idx) - (n_final - 1)
            if lag == 0:
                j = np.argmin(np.minimum(best_h, 1.0 - best_h))
                for d in [0.02, -0.02, 0.01, -0.01]:
                    h_trial = best_h.copy()
                    h_trial[j] += d
                    if 0 <= h_trial[j] <= 1:
                        h_trial = project_h(h_trial, target_final)
                        c5_t = compute_c5(h_trial)
                        if c5_t < best_c5:
                            best_h = h_trial
                            best_c5 = c5_t
                            improved = True
                            break
            else:
                abs_lag = abs(lag)
                if abs_lag >= n_final:
                    continue
                n_valid = n_final - abs_lag
                for _ in range(5):
                    i = np.random.randint(n_valid)
                    j = i + abs_lag if lag > 0 else i - lag
                    for d in [0.025, -0.025, 0.012, -0.012]:
                        h_trial = best_h.copy()
                        h_trial[i] += d
                        h_trial[j] -= d
                        if 0 <= h_trial[i] <= 1 and 0 <= h_trial[j] <= 1:
                            if abs(h_trial.sum() - target_final) > 1e-10:
                                h_trial = project_h(h_trial, target_final)
                            c5_t = compute_c5(h_trial)
                            if c5_t < best_c5:
                                best_h = h_trial
                                best_c5 = c5_t
                                improved = True
                                break
                    if improved:
                        break
        if not improved and iteration % 500 == 499:
            perturb = np.random.randn(n_final) * 0.005
            h_trial = project_h(best_h + perturb, target_final)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial

    # Phase 5: Final high-precision polish
    if time_left() > 8:
        n_pol = len(best_h)
        target_pol = n_pol / 2.0
        dx_pol = 2.0 / n_pol
        bounds_pol = Bounds(np.zeros(n_pol), np.ones(n_pol))
        obj_pol = make_slsqp_obj(dx_pol, 4000)
        con_pol = make_eq_con(target_pol)
        jac_pol = make_eq_jac((n_pol,))
        constraints_pol = [{'type': 'eq', 'fun': con_pol, 'jac': jac_pol}]
        try:
            res = minimize(obj_pol, best_h, method='SLSQP', bounds=bounds_pol,
                           constraints=constraints_pol,
                           options={'maxiter': 8000, 'ftol': 1e-16})
            h_new = project_h(res.x, target_pol)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        except Exception:
            pass

    # Final subgradient sweep
    for iteration in range(5000):
        if time_left() < 1:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:3]
        top_lags = [int(i) - (n_final - 1) for i in top_idx]
        grad = compute_subgrad(best_h, top_lags)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad /= gnorm
        for step in [0.02, 0.01, 0.005, 0.002, 0.001]:
            h_trial = project_h(best_h - step * grad, target_final)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
                break

    return best_h, best_c5, len(best_h)
```
