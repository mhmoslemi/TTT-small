def smooth_max_all_grad(h, beta):
    """Gradient of (1/beta)*log(sum exp(beta*C[m]*dx)) over all m"""
    n = len(h)
    dx = 2.0/n
    C = np.correlate(h, 1.0-h, mode='full')
    # weights
    bc = beta * C * dx
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()  # weights for each lag
    
    grad = np.zeros(n)
    # For each lag m (index in C array, which goes from 0 to 2n-2):
    # C[m] corresponds to a specific shift
    # C[n-1] is lag 0
    # C[n-1+k] is lag +k (for k = 0, ..., n-1)
    # C[n-1-k] is lag -k (for k = 1, ..., n-1)
    
    for m in range(len(C)):
        if w[m] < 1e-10:
            continue
        # determine the lag
        lag = m - (n-1)  # can be negative
        k = abs(lag)
        if lag >= 0:
            # C[n-1+k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            for i in range(n-k):
                # h[i+k] contributes (1-h[i])
                grad[i+k] += w[m] * (1.0 - h[i])
                # h[i] contributes -h[i+k]

            if k < n:
                grad[k:] += wi * (1.0 - h[:n - k])
                grad[:n - k] += wi * (-h[k:])
        else:
            k = -lag
            if k < n:
                grad[:n - k] += wi * (1.0 - h[k:])
                grad[k:] += wi * (-h[:n - k])
    grad *= dx
    return val, grad

def subgradient_momentum(h0, n, target_sum, max_iter, time_limit, start_t):
    h = h0.copy()
    v = np.zeros(n)
    mu = 0.9
    lr0 = 0.02
    decay = 1e-5
    beta = 80.0
    top_k = 8
    best = h.copy()
    best_c5 = compute_c5(h)
    prev_val = smooth_max_grad(h, beta, top_k)[0]
    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break
        val, grad = smooth_max_grad(h, beta, top_k)
        lr = lr0 / (1.0 + it * decay)
        v = mu * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best = h_new.copy()
            best_c5 = c5
        h = h_new
        if it % 200 == 0:
            beta = min(300.0, beta * 1.05)
            mu = min(0.95, 0.9 + it * 1e-5)
    return best, best_c5

def binary_swap_search(h0, n, max_sweeps, time_limit, start_t):
    h = project_feasible(h0, n / 2.0)
    h = (h > 0.5).astype(float)
    dx = 2.0 / n
    current_max = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
    best_h = h.copy()
    best_c5 = current_max
    improved_any = True
    sweep = 0
    while improved_any and sweep < max_sweeps:
        if time.time() - start_t > time_limit:
            break
        improved_any = False
        perm = np.random.permutation(n)
        for ii in range(n):
            i = perm[ii]
            for jj in range(ii + 1, n):
                j = perm[jj]
                if h[i] == h[j]:
                    continue
                h[i], h[j] = h[j], h[i]
                c = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
                if c < current_max - 1e-14:
                    current_max = c
                    improved_any = True
                    if c < best_c5:
                        best_c5 = c
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[j], h[i]
        sweep += 1
    return best_h, best_c5

def slsqp_polish(h0, n, max_iter, time_limit, start_t):
    if time.time() - start_t > time_limit:
        return h0.copy(), compute_c5(h0)
    target_sum = n / 2.0
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    ones_vec = np.ones(n)
    def con_fun(x):
        return np.sum(x) - target_sum
    def con_jac(x):
        return ones_vec
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    best = h0.copy()
    best_c5 = compute_c5(h0)
    for beta in [100, 200, 400]:
        def obj_sm(h):
            corr = np.correlate(h, 1.0 - h, mode="full")
            m = np.max(corr)
            return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx
        try:
            res = minimize(obj_sm, best, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': max_iter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best = h_new.copy()
                best_c5 = c5
        except Exception:
            pass
        if time.time() - start_t > time_limit:
            break
    return best, best_c5

def upsample(h, n_src, n_dst):
    xo = np.linspace(0, 2, n_src, endpoint=False)
    xn = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_p = np.array(initial_h_values, dtype=float).copy()
    else:
        h_p = np.full(70, 0.5)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    resolutions = [70, 90, 100]
    time_splits = [380, 350, 280]
    
    for ri, n in enumerate(resolutions):
        t_phase_start = time.time()
        t_phase_limit = t_start + sum(time_splits[:ri + 1])
        target_sum = n / 2.0
        
        if h_p is not None and len(h_p) == n:
            h0 = h_p.copy()
        elif h_p is not None:
            h0 = upsample(h_p, len(h_p), n)
        else:
            h0 = np.full(n, 0.5)
        h0 = project_feasible(h0, target_sum)
        
        # Phase A: Subgradient with momentum
        t_lim = t_phase_limit - sum(time_splits[ri + 1:]) - 10
        h_sg, c5_sg = subgradient_momentum(h0, n, target_sum, 50000, t_lim, t_start)
        
        # Phase B: Binary swap search from several starts
        t_lim2 = t_phase_limit - sum(time_splits[ri + 1:]) - 5
        starts_b = [h_sg]
        for _ in range(3):
            starts_b.append(project_feasible(h_sg + np.random.randn(n) * 0.3, target_sum))
        
        h_bin_best = None
        c5_bin_best = float('inf')
        for sb in starts_b:
            if time.time() > t_lim2:
                break
            hb, c5b = binary_swap_search(sb, n, 200, t_lim2, t_start)
            if c5b < c5_bin_best:
                c5_bin_best = c5b
                h_bin_best = hb
            h_sg, c5_sg = hb, c5b  # update for next starts
        
        # Phase C: SLSQP polish on the best
        h_final, c5_final = slsqp_polish(h_sg, n, 500, t_lim2, t_start)
        if c5_bin_best < c5_final:
            h_final, c5_final = slsqp_polish(h_bin_best, n, 500, t_lim2, t_start)
        
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
            best_n = n
            h_p = h_final  # carry forward for next resolution
        
        if time.time() - t_start > budget_s - 60:
            break
    
    # Final polish with remaining time
    if best_h is not None and time.time() - t_start < budget_s - 20:
        h_final, c5_final = slsqp_polish(best_h, best_n, 800, budget_s - 10, t_start)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final
    
    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = 70
    
    return best_h, best_c5, best_n
