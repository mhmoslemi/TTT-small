   ```python
       global_c = 0.380923  # Start from parent's C5
       best_h = np.array(initial_h_values, dtype=float).copy()
       best_c5 = global_c * 0.98  # Conservative target

       betas = [500, 200, 100, 50]  # Decreasing beta order
       T = [4.0, 2.0, 1.0, 0.5]     # Temperature parameters
       current_beta = betas[0]
       time_remaining = budget_s

       # Track best solution
       def is_feasible(x):
           return np.all(x >= 0) and np.all(x <= 1) and np.isclose(np.sum(x) * (2/70), 1, atol=1e-5)

       # Strategy engine
       for beta_idx in range(len(betas)):
           if time_remaining <= 0:
               break
           current_beta = betas[beta_idx]
           T_val = T[beta_idx]
           max_iter = int(min(5000, time_remaining * 0.2))  # Time-based iteration limit

           # Bounds: [0,1] for all
           bounds = Bounds(np.zeros(n_points), np.ones(n_points))
           # Fixed constraints: integral and domain clipping
           # Use L-BFGS-B with projected steps

           # Priority list for repair:
           x = best_h.copy()
           if not is_feasible(x):
               best_h = project_h(x, target_sum=35)

           # Local search from best_h
           res = minimize(
               objective_single,
               best_h,
               args=beta_idx,
               bounds=bounds,
               method='L-BFGS-B',
               options={
                   'maxiter': max_iter,
                   'disp': False,
                   'eps': 1e-4
               }
           )

           # Posterior projection after local search
           candidate = project_h(res.x, target_sum=35)
           c5_ = compute_c5(candidate)
           # Check if any repair is needed
           cimprove = c5_ < best_c5
            if cimprove:
                best_c5 = c5_
                best_h = candidate
                # Self-constrained cooling schedule
                if beta_idx > 0:
                    T_val = T_val * 0.9

       return best_h, best_c5, n_points
   ```
