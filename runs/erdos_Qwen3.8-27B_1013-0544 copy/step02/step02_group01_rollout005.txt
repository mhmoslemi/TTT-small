from scipy.interpolate import interp1d
import time

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: smoothmax_val_grad(x, beta)[0], h,
                     jac=lambda x: smoothmax_val_grad(x, beta)[1],
                     method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time
import cvxpy as cp


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad


def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0

    def obj(x):
        return smoothmax_val_grad(x, beta)[0]

    def objg(x):
        return smoothmax_val_grad(x, beta)[1]

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def lp_solve(h, n_lags, eps_reg=0.0):
    """Single LP linearization step. Returns (h_new, success)."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-n_lags:]

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]

    for ki in top_idx:
        s = ki - (n - 1)
        i_start = max(0, -s)
        i_end = min(n, n - s)
        js = np.arange(i_start, i_end)
        if len(js) == 0:
            continue

        grad_f = np.zeros(n)
        grad_f[js] += (1.0 - h[js + s])
        if s != 0:
            grad_f[js + s] += (-h[js])

        f_val = np.sum(h[js] * (1.0 - h[js + s]))
        const = (f_val - np.dot(grad_f, h)) * dx

        constraints.append(cp.dot(grad_f, h_var) * dx + const <= t_var)

    if eps_reg > 0:
        obj = cp.Minimize(t_var + eps_reg * cp.sum_squares(h_var - h))
    else:
        obj = cp.Minimize(t_var)

    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=15000, reltol=1e-9, abstol=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=15000)
        except Exception:
            return h, False

    if h_var.value is None:
        return h, False

    h_new = np.clip(h_var.value, 0.0, 1.0)
    h_new = project_feasible(h_new, target)
    return h_new, True

        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new

    return best_x, best_v


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(70, 0.5)
    best_c5 = compute_c5(h_best)

    # Phase 1: Nesterov warm start to escape potential local optima
    if tl() > 80:
        h1, c1 = nesterov_subgrad(h_best, 2000, 0.04, 0.9996, top_k=8, time_limit=min(60, tl() - 80))
        if c1 < best_c5:
            h_best, best_c5 = h1, c1

    # Phase 2: LP linearization (primary method)
    if tl() > 60:
        h2, c2 = lp_loop(h_best, n_lags=35, max_iters=400, time_limit=min(200, tl() - 60))
        if c2 < best_c5:
            h_best, best_c5 = h2, c2

    # Phase 3: SLSQP polish
    if tl() > 40:
        h3 = slsqp_refine(h_best, maxiter=400, beta=150)
        c3 = compute_c5(h3)
        if c3 < best_c5:
            h_best, best_c5 = h3, c3

    # Phase 4: Another LP round with more lags
    if tl() > 60:
        h4, c4 = lp_loop(h_best, n_lags=50, max_iters=300, time_limit=min(150, tl() - 60), eps_reg=1e-4)
        if c4 < best_c5:
            h_best, best_c5 = h4, c4

    # Phase 5: SLSQP with higher beta
    if tl() > 40:
        h5 = slsqp_refine(h_best, maxiter=400, beta=250)
        c5 = compute_c5(h5)
        if c5 < best_c5:
            h_best, best_c5 = h5, c5

    # Phase 6: Higher resolution
    for n_try in [300, 400]:
        if tl() < 120:
            break
        h_interp = interp_to_n(h_best, n_try)
        # Quick Nesterov at new resolution
        if tl() > 50:
            h_a, c_a = nesterov_subgrad(h_interp, 1500, 0.02, 0.9997, top_k=8, time_limit=min(40, tl() - 50))
            if c_a < best_c5:
                h_best, best_c5 = h_a, c1 if False else c_a
                h_interp = h_best.copy()
        # LP at new resolution
        if tl() > 50:
            h_b, c_b = lp_loop(h_interp, n_lags=30, max_iters=150, time_limit=min(80, tl() - 50))
            if c_b < best_c5:
                h_best, best_c5 = h_b, c_b
        # SLSQP at new resolution
        if tl() > 30:
            h_c = slsqp_refine(h_best, maxiter=300, beta=200)
            c_c = compute_c5(h_c)
            if c_c < best_c5:
                h_best, best_c5 = h_c, c_c

    # Phase 7: Final LP + SLSQP polish at best resolution
    if tl() > 30:
        h_f, c_f = lp_loop(h_best, n_lags=50, max_iters=200, time_limit=min(60, tl() - 30))
        if c_f < best_c5:
            h_best, best_c5 = h_f, c_f

    if tl() > 20:
        h_g = slsqp_refine(h_best, maxiter=500, beta=400)
        c_g = compute_c5(h_g)
        if c_g < best_c5:
            h_best, best_c5 = h_g, c_g

    return h_best, best_c5, len(h_best)
