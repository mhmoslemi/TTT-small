        c = compute_c5(ht, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    # Phase 2: Differential Evolution global search
    pop_size = 35
    populations = [best_h.copy()]
    # Add structured diversity
    x = np.linspace(0, 2, n)
    populations.append(project_feasible(0.5 + 0.45 * np.sin(np.pi * x), n))
    populations.append(project_feasible(0.5 + 0.45 * np.sin(2 * np.pi * x + 0.5), n))
    populations.append(project_feasible(0.5 + 0.3 * np.cos(3 * np.pi * x), n))
    populations.append(project_feasible(0.5 + 0.4 * np.sin(5 * np.pi * x), n))
    # Random binary-like
    for _ in range(3):
        rb = np.where(rng.random(n) < 0.5, 1.0, 0.0)
        populations.append(project_feasible(rb, n))
    # Fill rest with perturbations
    while len(populations) < pop_size:
        p = best_h + rng.normal(0, 0.2, size=n)
        populations.append(project_feasible(p, n))

    pop = np.array(populations[:pop_size])
    fit = np.array([compute_c5(p, n) for p in pop])
    best_idx = np.argmin(fit)
    if fit[best_idx] < best_c5:
        best_h, best_c5 = pop[best_idx].copy(), fit[best_idx]

    # Run DE with parameters tuned for this problem
    for F, CR, gens in [(0.7, 0.4, 120), (0.5, 0.6, 100)]:
        if remaining() < 80:
            break
        pop, fit = de_evolve(pop, fit, n, F, CR, rng, gens, t0, budget_s,
                            polish_every=15)
        bi = np.argmin(fit)
        if fit[bi] < best_c5:
            best_h, best_c5 = pop[bi].copy(), fit[bi]

    # Phase 3: SLSQP polish after DE
    for alpha in [5000, 20000, 100000]:
        if remaining() < 15:
            break
        ht = slsqp_optimize(best_h, n, alpha, 4000, 1e-16)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    # Phase 4: Coordinate descent
    if remaining() > 60:
        best_h, best_c5 = coordinate_descent(best_h, n, t0, budget_s, n_rounds=30,
                                            deltas=[0.002, 0.005, 0.01, 0.02, 0.05])

    # Phase 5: More DE from improved point with tighter params
    if remaining() > 100:
        pop2 = [best_h.copy()]
        for _ in range(pop_size - 1):
            p = best_h + rng.normal(0, 0.05, size=n)
            pop2.append(project_feasible(p, n))
        pop2 = np.array(pop2)
        fit2 = np.array([compute_c5(p, n) for p in pop2])
        pop2, fit2 = de_evolve(pop2, fit2, n, 0.4, 0.5, rng, 80, t0, budget_s,
                              polish_every=10)
        bi2 = np.argmin(fit2)
        if fit2[bi2] < best_c5:
            best_h, best_c5 = pop2[bi2].copy(), fit2[bi2]

    # Phase 6: Final SLSQP
    for alpha in [10000, 50000, 200000]:
        if remaining() < 12:
            break
        ht = slsqp_optimize(best_h, n, alpha, 5000, 1e-17)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    # Phase 7: Fine coordinate descent
    if remaining() > 15:
        best_h, best_c5 = coordinate_descent(best_h, n, t0, budget_s, n_rounds=40,
                                            deltas=[0.0005, 0.001, 0.002, 0.005, 0.01])

    # Phase 8: Final SLSQP + coordinate
    if remaining() > 10:
        ht = slsqp_optimize(best_h, n, 500000, 6000, 1e-18)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c
        if remaining() > 8:
            best_h, best_c5 = coordinate_descent(best_h, n, t0, budget_s, n_rounds=20,
                                                deltas=[0.0003, 0.0007, 0.002, 0.005])

    # Final validation
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
