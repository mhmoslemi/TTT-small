        return np.ones(n)
    constraints = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]

    h = project_h(h_init, target_sum)
    best_c5 = compute_c5(h)

    # Phase 1: SLSQP with exact gradient, increasing beta
    betas = [30, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000]
    for beta in betas:
        if time_left() < 80:
            break
        h = slsqp_refine(h, beta, dx, target_sum, bounds, constraints, maxiter=8000)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            h_best_main = h.copy()
        else:
            h_best_main = h.copy()
        h = h_best_main

    # Phase 2: Subgradient on true max
    if time_left() > 60:
        h_sub = subgradient_polish(h, target_sum, dx, n_iter=3000, alpha0=0.005)
        c5 = compute_c5(h_sub)
        if c5 < best_c5:
            best_c5 = c5
            h = h_sub

    # Phase 3: More SLSQP rounds after subgradient
    for beta in [2000, 5000, 10000]:
        if time_left() < 50:
            break
        h = slsqp_refine(h, beta, dx, target_sum, bounds, constraints, maxiter=5000)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5

    # Phase 4: Subgradient again
    if time_left() > 40:
        h = subgradient_polish(h, target_sum, dx, n_iter=2000, alpha0=0.003)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5

    # Phase 5: Pairwise coordinate descent
    if time_left() > 30:
        h, c5 = pairwise_cd(h, target_sum, dx, n_passes=10, max_pairs=300)
        if c5 < best_c5:
            best_c5 = c5

    # Phase 6: Multi-start perturbation + SLSQP
    for s in range(15):
        if time_left() < 40:
            break
        scale = 0.005 + 0.02 * (s / 15.0)
        h0 = h + rng.randn(n) * scale
        h0 = project_h(h0, target_sum)
        for beta in [500, 2000]:
            if time_left() < 40:
                break
            h_new = slsqp_refine(h0, beta, dx, target_sum, bounds, constraints, maxiter=3000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                h = h_new
                h0 = h.copy()
        # Subgradient polish on improved candidate
        if time_left() > 40:
            h_sub = subgradient_polish(h0, target_sum, dx, n_iter=500, alpha0=0.003)
            c5 = compute_c5(h_sub)
            if c5 < best_c5:
                best_c5 = c5
                h = h_sub

    # Phase 7: Try slightly different n_points with new starts
    for n_try in [60, 65, 75, 80]:
        if time_left() < 80:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        bounds_try = Bounds(np.zeros(n_try), np.ones(n_try))
        def cfun_t(x, ts=ts_try):
            return x.sum() - ts
        def cjac_t(x, n=n_try):
            return np.ones(n)
        constr_try = [{'type': 'eq', 'fun': cfun_t, 'jac': cjac_t}]

        for attempt in range(3):
            if time_left() < 60:
                break
            # Random sinusoidal start
            k_freq = rng.randint(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.3, 0.5)
            t = np.arange(n_try)
            h_t = 0.5 + amp * np.sin(2 * np.pi * k_freq * t / n_try + phase)
            h_t = project_h(h_t, ts_try)

            for beta in [100, 500, 2000, 5000]:
                if time_left() < 50:
                    break
                h_t = slsqp_refine(h_t, beta, dx_try, ts_try, bounds_try, constr_try, maxiter=5000)
            if time_left() > 50:
                h_t = subgradient_polish(h_t, ts_try, dx_try, n_iter=1000, alpha0=0.005)
            c5_try = compute_c5(h_t)
            if c5_try < best_c5:
                best_c5 = c5_try
                h = h_t
                n = n_try
                dx = dx_try
                target_sum = ts_try
                bounds = bounds_try
                constraints = constr_try

    # Final polish
    if time_left() > 20:
        h = slsqp_refine(h, 5000, dx, target_sum, bounds, constraints, maxiter=3000)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5

    if time_left() > 15:
        h = subgradient_polish(h, target_sum, dx, n_iter=1000, alpha0=0.002)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5

    h = project_h(h, target_sum)
    best_c5 = compute_c5(h)
    return h, best_c5, len(h)
