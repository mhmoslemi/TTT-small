def compute_c5_and_grad(h, beta):
    """Compute smooth max of correlation and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Smooth max
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50, 50)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    val = (m + np.log(np.sum(exp_shifted)) / beta) * dx
    
    # Gradient
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for lag in range(0, n):  # lag from 0 to n-1
        # result[n-1+lag] = sum_{j=0}^{n-1-lag} h[j+lag] * (1-h[j])
        # result[n-1-lag] = sum_{j=0}^{n-1-lag} h[j] * (1-h[j+lag]) (for lag > 0)
        
        w_plus = weights[n-1+lag] if n-1+lag < len(weights) else 0
        w_minus = weights[n-1-lag] if n-1-lag >= 0 else 0
        
        if lag == 0:
            # result[n-1] = sum_j h[j]*(1-h[j])
            # d/dh[p] = (1-2*h[p])
            grad += w_plus * (1 - 2*h) * dx
        else:

            # ∂/∂h[p] = (1-h[p-lag]) for lag ≤ p ≤ n-1, minus h[p+lag] for 0 ≤ p ≤ n-1-lag
            # For result[n-1-lag]: sum_{j=0}^{n-1-lag} h[j]*(1-h[j+lag])
            # ∂/∂h[p] = (1-h[p+lag]) for 0 ≤ p ≤ n-1-lag, minus h[p-lag] for lag ≤ p ≤ n-1

            # For p in [lag, n-1]: contribution from result[n-1+lag] is (1-h[p-lag])
            # For p in [0, n-1-lag]: contribution from result[n-1+lag] is -h[p+lag]
            # For p in [0, n-1-lag]: contribution from result[n-1-lag] is (1-h[p+lag])

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_smooth_max_val_grad(h, beta):
    """Compute smooth max value and gradient of correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50, 50)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    
    # Zero lag
    w0 = weights[n-1]
    grad += w0 * (1.0 - 2.0 * h) * dx
    
    # Non-zero lags
    for lag in range(1, n):
        wp = weights[n-1+lag]
        wm = weights[n-1-lag]
        nl = n - lag
        # From result[n-1+lag]:
        grad[lag:] += wp * one_minus_h[:nl] * dx
        grad[:nl] += (-wp) * h[lag:] * dx
        # From result[n-1-lag]:
        grad[:nl] += wm * one_minus_h[lag:] * dx
        grad[lag:] += (-wm) * h[:nl] * dx
    
    return val, grad

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-10:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

            h0 = np.interp(xn, xo, h_parent, period=2.0)
            h0 = project_h(h0, target_sum)

        curr_h = h0.copy()
        curr_c5 = compute_c5(h0)
        con_sl = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                  'jac': lambda x: np.ones(n)}

        for beta in [20, 50, 100, 200, 400, 700, 1200]:
            if time_left() < 35:
                break
            b = float(beta)
            def obj_func(h):
                return corr_smooth_max_val_grad(h, b)[0]
            def obj_grad(h):
                return corr_smooth_max_val_grad(h, b)[1]
            try:
                mi = max(100, int(min(time_left() * 0.15, 600)))
                res = minimize(obj_func, curr_h, jac=obj_grad, method='SLSQP',
                               bounds=bounds, constraints=[con_sl],
                               options={'maxiter': mi, 'ftol': 1e-14})
                h_new = project_h(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < curr_c5:
                    curr_h, curr_c5 = h_new.copy(), c5_new
                    if curr_c5 < best_c5:
                        best_h, best_c5, best_n = curr_h.copy(), curr_c5, n
            except Exception:
                pass

        if n == np0 and time_left() > 80:
            b = 500.0
            def obj_tc(h):
                return corr_smooth_max_val_grad(h, b)[0]
            def obj_tc_g(h):
                return corr_smooth_max_val_grad(h, b)[1]
            lc = LinearConstraint(np.ones((1, n)), target_sum, target_sum)
            try:
                res = minimize(obj_tc, curr_h, jac=obj_tc_g, method='trust-constr',
                               bounds=bounds, constraints=[lc],
                               options={'maxiter': 3000, 'gtol': 1e-9, 'xtol': 1e-11})
                h_new = project_h(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < curr_c5:
                    curr_h, curr_c5 = h_new.copy(), c5_new
                    if curr_c5 < best_c5:
                        best_h, best_c5, best_n = curr_h.copy(), curr_c5, n
            except Exception:
                pass

        if n == np0 and time_left() > 50:
            for trial in range(20):
                if time_left() < 15:
                    break
                strat = trial % 3
                if strat == 0:
                    bs = np.random.randint(3, 8)
                    s = np.random.randint(0, n - bs)
                    h_p = curr_h.copy()
                    h_p[s:s+bs] += np.random.uniform(-0.04, 0.04, bs)
                elif strat == 1:
                    s = np.random.randint(0, n // 2)
                    bs = np.random.randint(2, 6)
                    h_p = curr_h.copy()
                    d = np.random.uniform(-0.03, 0.03)
                    h_p[s:s+bs] += d
                    s2 = (s + n // 2) % n
                    h_p[s2:s2+bs] -= d
                else:
                    h_p = curr_h + np.random.randn(n) * 0.02
                h_p = project_h(h_p, target_sum)
                b2 = 300.0
                def obj_p(h):
                    return corr_smooth_max_val_grad(h, b2)[0]
                def obj_p_g(h):
                    return corr_smooth_max_val_grad(h, b2)[1]
                try:
                    res = minimize(obj_p, h_p, jac=obj_p_g, method='SLSQP',
                                   bounds=bounds, constraints=[con_sl],
                                   options={'maxiter': 120, 'ftol': 1e-13})
                    h_new = project_h(res.x, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < curr_c5:
                        curr_h, curr_c5 = h_new.copy(), c5_new
                        if curr_c5 < best_c5:
                            best_h, best_c5, best_n = curr_h.copy(), curr_c5, n
                except Exception:
                    pass

    if time_left() > 12:
        nf = best_n
        ts = nf / 2.0
        bf = Bounds(np.zeros(nf), np.ones(nf))
        for beta in [1500, 3000]:
            if time_left() < 8:
                break
            b = float(beta)
            def obj_f(h):
                return corr_smooth_max_val_grad(h, b)[0]
            def obj_f_g(h):
                return corr_smooth_max_val_grad(h, b)[1]
            con_f = {'type': 'eq', 'fun': lambda x: np.sum(x) - ts,
                     'jac': lambda x: np.ones(nf)}
            try:
                res = minimize(obj_f, best_h, jac=obj_f_g, method='SLSQP',
                               bounds=bf, constraints=[con_f],
                               options={'maxiter': 800, 'ftol': 1e-16})
                h_new = project_h(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new.copy(), c5_new
            except Exception:
                pass

    best_h = project_h(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
