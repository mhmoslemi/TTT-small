                        curr_h = h_try.copy()
                        curr_c5 = c5_try
                        base = h_try.copy()
                        if c5_try < best_sa:
                            best_sa = c5_try
                    T *= 0.985

        if curr_h is not None and curr_c5 < best_c5:
            best_h, best_c5, best_n = curr_h.copy(), curr_c5, n_points

    # Final: Coordinate-wise greedy
    if best_h is not None and time.time() - start_time < budget_s - 40:
        n_f = len(best_h)
        t_f = n_f / 2.0
        base = best_h.copy()
        improved = True
        while improved and time.time() - start_time < budget_s - 25:
            improved = False
            for _pass in range(3):
                if time.time() - start_time > budget_s - 25:
                    break
                for i in range(n_f):
                    if time.time() - start_time > budget_s - 25:
                        break
                    for delta in [0.004, -0.004, 0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005]:
                        h_try = base.copy()
                        nv = base[i] + delta
                        if nv < -0.01 or nv > 1.01:
                            continue
                        h_try[i] = nv
                        h_try = project_exact(h_try, t_f)
                        c5_try, _ = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-14:
                            best_h = h_try.copy()
                            best_c5 = c5_try
                            base = h_try.copy()
                            improved = True
                            break

    # Final L-BFGS-B polish
    if best_h is not None and time.time() - start_time < budget_s - 12:
        n_f = len(best_h)
        t_f = n_f / 2.0
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        for beta in [2000, 4000]:
            if time.time() - start_time > budget_s - 10:
                break
            try:
                res = minimize(obj_smooth_max, best_h,
                               args=(beta, 50.0, t_f),
                               method='L-BFGS-B', bounds=bounds_f,
                               options={'maxiter': 800, 'ftol': 1e-18, 'gtol': 1e-15})
                h_new = project_exact(res.x, t_f)
                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new.copy(), c5_new
            except Exception:
                pass

    # Absolute final SLSQP
    if best_h is not None and time.time() - start_time < budget_s - 6:
        n_f = len(best_h)
        t_f = n_f / 2.0
        _target[0] = t_f
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        try:
            res = minimize(obj_smooth_max, best_h,
                           args=(3000.0, 20.0, t_f),
                           method='SLSQP', bounds=bounds_f,
                           constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}],
                           options={'maxiter': 300, 'ftol': 1e-18})
            h_new = project_exact(res.x, t_f)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
