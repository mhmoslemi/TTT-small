    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        return smooth_max(get_corr(x), beta) * dx
    def obj_grad(x):
        corr = get_corr(x)
        m = np.max(corr)
        w = np.exp(beta * (corr - m))
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-10:
                continue
            d = k - (n - 1)
            if d >= 0:
                for ii in range(d, n):
                    g[ii] += wk * (1.0 - x[ii - d])
                for ii in range(n - d):
                    g[ii] -= wk * x[ii + d]
            else:
                m2 = -d
                for ii in range(n - m2):
                    g[ii] += wk * (1.0 - x[ii + m2])
                for ii in range(m2, n):
                    g[ii] -= wk * x[ii - m2]
        return g * dx

                    'fun': lambda x: np.sum(x) - target_sum,
                    'jac': lambda x: np.ones(n)}]

                   options={'maxiter': maxiter, 'ftol': ftol})

            g[i] -= h[i - m]

        if tleft() < 60:
            break
        h_cur = cd_sweep(h_cur, target_sum, n_points * 2)
        c5 = compute_c5(h_cur)
        if c5 < best_c5:
            best_h = h_cur.copy()
            best_c5 = c5

    # Phase 4: Diversified restarts with CD
    for restart in range(8):
        if tleft() < 40:
            break
        h0 = best_h.copy()
        n_pert = np.random.randint(3, 10)
        idx = np.random.choice(n_points, n_pert, replace=False)
        h0[idx] += np.random.randn(n_pert) * 0.08
        h0 = project_h(h0, target_sum)
        # Quick CD from perturbed start
        for s in range(80):
            if tleft() < 30:
                break
            h0 = cd_sweep(h0, target_sum, n_points)
        c5 = compute_c5(h0)
        if c5 < best_c5:
            best_h, best_c5 = h0.copy(), c5

    # Phase 5: Final SLSQP + CD polish
    for beta in [400, 800, 1500, 3000]:
        if tleft() < 20:
            break
        try:
            h_new = slsqp_polish(best_h, target_sum, dx, beta, maxiter=3000, ftol=1e-15)
            h_new = project_h(h_new, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    if tleft() > 30:
        h_cur = best_h.copy()
        for sweep in range(200):
            if tleft() < 15:
                break
            h_cur = cd_sweep(h_cur, target_sum, n_points * 3)
            c5 = compute_c5(h_cur)
            if c5 < best_c5:
                best_h = h_cur.copy()
                best_c5 = c5

    # Phase 6: Gradient-guided fine tuning
    if tleft() > 20:
        h_cur = best_h.copy()
        ss = 0.01
        for it in range(3000):
            if tleft() < 10:
                break
            h_new = gradient_step(h_cur, target_sum, ss)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h_cur):
                h_cur = h_new
                if c5_new < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = c5_new
            else:
                ss *= 0.95
                if ss < 0.001:
                    ss = 0.01
        # Final CD
        h_cur = cd_sweep(h_cur, target_sum, n_points * 2)
        c5 = compute_c5(h_cur)
        if c5 < best_c5:
            best_h, best_c5 = h_cur.copy(), c5

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
