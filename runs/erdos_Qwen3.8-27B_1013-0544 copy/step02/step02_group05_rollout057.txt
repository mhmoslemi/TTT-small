import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12 and s > 0:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_optimal(h, i):
    """Find optimal value for coordinate i using exact 1D optimization."""
    h0 = h.copy(); h0[i] = 0.0
    h1 = h.copy(); h1[i] = 1.0
    A = np.correlate(h0, 1.0 - h0, mode="full")
    B = np.correlate(h1, 1.0 - h1, mode="full") - A
    
    # Evaluate at grid
    t_grid = np.linspace(0, 1, 256)
    vals = np.max(A[None, :] + B[None, :] * t_grid[:, None], axis=1)
    best_t = t_grid[np.argmin(vals)]
    
    # Refine
    lo = max(0.0, best_t - 0.005)
    hi = min(1.0, best_t + 0.005)
    t_fine = np.linspace(lo, hi, 128)
    vals_fine = np.max(A[None, :] + B[None, :] * t_fine[:, None], axis=1)
    best_t = t_fine[np.argmin(vals_fine)]
    
    return best_t

def coord_descent(h, target_sum, max_sweeps=30, tol=1e-8, deadline=None):
    """Coordinate descent with constraint projection."""
    n = len(h)
    h = project_h(h, target_sum)
    curr_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        total_change = 0.0
        for i in range(n):
            t_opt = coord_optimal(h, i)
            total_change += abs(t_opt - h[i])
            h[i] = t_opt
        h = project_h(h, target_sum)
        new_c5 = compute_c5(h)
        if new_c5 < curr_c5 - 1e-10:
            curr_c5 = new_c5
        if total_change < tol:
            break
    return h, curr_c5

def slsqp_polish(h, target_sum, deadline):
    """SLSQP with smooth max."""
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def con_fun(x):
        return np.sum(x) - target_sum
    def con_jac(x):
        return np.ones(n)
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    for beta in [100, 200, 400]:
        if time.time() > deadline:
            break
        def obj(h):
            corr = np.correlate(h, 1.0 - h, mode="full")
            return smooth_max(corr, float(beta)) * dx
        try:

                           constraints=constraints,
                           options={'maxiter': 500, 'ftol': 1e-14})
            h_new = project_h(res.x, target_sum)
            if compute_c5(h_new) < compute_c5(h):
                h = h_new
        except Exception:
            pass
    return h

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_target_sum_global = 0.0

def constraint_fun(x):
    return np.sum(x) - _target_sum_global

def constraint_jac(x):
    return np.ones(len(x))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_optimal(h, i):
    h0 = h.copy()
    h0[i] = 0.0
    h1 = h.copy()
    h1[i] = 1.0
    A = np.correlate(h0, 1.0 - h0, mode="full")
    B = np.correlate(h1, 1.0 - h1, mode="full") - A
    t_grid = np.linspace(0, 1, 256)
    vals = np.max(A[None, :] + B[None, :] * t_grid[:, None], axis=1)
    best_t = t_grid[np.argmin(vals)]
    lo = max(0.0, best_t - 0.003)
    hi = min(1.0, best_t + 0.003)
    t_fine = np.linspace(lo, hi, 128)
    vals_fine = np.max(A[None, :] + B[None, :] * t_fine[:, None], axis=1)
    return float(t_fine[np.argmin(vals_fine)])

def coord_descent(h, target_sum, max_sweeps=30, deadline=None, rng=None):
    global _target_sum_global
    _target_sum_global = target_sum
    n = len(h)
    h = project_h(h, target_sum)
    curr_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        total_change = 0.0
        for i in order:
            t_opt = coord_optimal(h, i)
            total_change += abs(t_opt - h[i])
            h[i] = t_opt
        h = project_h(h, target_sum)
        new_c5 = compute_c5(h)
        if new_c5 < curr_c5:
            curr_c5 = new_c5
        if total_change < 1e-9:
            break
    return h, curr_c5

def slsqp_polish(h, target_sum, deadline):
    global _target_sum_global
    _target_sum_global = target_sum
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    for beta in [80, 200, 400]:
        if time.time() > deadline:
            break
        def _obj(h):
            corr = np.correlate(h, 1.0 - h, mode="full")
            return smooth_max(corr, float(beta)) * dx
        try:
            res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-14})
            h_new = project_h(res.x, target_sum)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            if c5_new < c5_old:
                h = h_new
        except Exception:
            pass
    return h

def generate_binary_start(n, target_sum, rng):
    scores = rng.random(n)
    rb = np.zeros(n)
    k = int(target_sum)
    rb[np.argsort(scores)[-k:]] = 1.0
    return rb

def generate_structured_start(n, target_sum, period, fill_ratio, rng):
    pattern = np.zeros(n)
    fill = max(1, int(period * fill_ratio))
    offset = rng.randint(0, max(1, period - fill))
    for p in range(0, n, period):
        s = p + offset
        e = min(s + fill, n)
        if s < n:
            pattern[s:e] = 1.0
    return project_h(pattern, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 15
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.full(70, 0.5)

    n_parent = len(h_parent)
    n = n_parent
    target_sum = n / 2.0

    # Phase 1: Coordinate descent from diverse starts at parent resolution
    starts = [project_h(h_parent, target_sum)]

    for trial in range(4):
        starts.append(generate_binary_start(n, target_sum, rng))

    for period in [7, 9, 11]:
        for fr in [0.35, 0.5]:
            starts.append(generate_structured_start(n, target_sum, period, fr, rng))

    for h0 in starts:
        if time.time() > deadline - 120:
            break
        h_opt, c5 = coord_descent(h0, target_sum, max_sweeps=25, deadline=deadline - 120, rng=rng)
        h_opt = slsqp_polish(h_opt, target_sum, deadline)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5
            best_n = n

    # Phase 2: Higher resolution attempts
    for n2 in [n + 10, n + 20, n + 40]:
        if time.time() > deadline - 80:
            break
        ts2 = n2 / 2.0
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = project_h(np.interp(x_new, x_old, best_h, period=2.0), ts2)
        h_opt, c5 = coord_descent(h2, ts2, max_sweeps=15, deadline=deadline - 80, rng=rng)
        h_opt = slsqp_polish(h_opt, ts2, deadline)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5
            best_n = n2

    # Phase 3: Perturbation restarts around best solution
    if time.time() < deadline - 80:
        for trial in range(20):
            if time.time() > deadline - 40:
                break
            scale = 0.03 + 0.015 * trial
            h_p = best_h + rng.randn(best_n) * scale
            h_p = project_h(h_p, best_n / 2.0)
            h_opt, c5 = coord_descent(h_p, best_n / 2.0, max_sweeps=15, deadline=deadline - 40, rng=rng)
            h_opt = slsqp_polish(h_opt, best_n / 2.0, deadline)

            if c5 < best_c5:
                best_h = h_opt.copy()

            if time.time() > deadline - 30:
                break
            rb = generate_binary_start(best_n, best_n / 2.0, rng)
            # Snap parent solution toward binary, then refine
            h_mixed = 0.5 * best_h + 0.5 * rb
            h_mixed = project_h(h_mixed, best_n / 2.0)
            h_opt, c5 = coord_descent(h_mixed, best_n / 2.0, max_sweeps=20, deadline=deadline - 30, rng=rng)
            h_opt = slsqp_polish(h_opt, best_n / 2.0, deadline)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5

    # Final high-precision SLSQP
    if best_h is not None and time.time() < deadline:
        global _target_sum_global
        _target_sum_global = best_n / 2.0
        n_f = best_n
        dx_f = 2.0 / n_f
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        cons_f = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]

        for beta in [300, 600]:
            if time.time() > deadline:
                break
            def _obj_final(h):
                corr = np.correlate(h, 1.0 - h, mode="full")
                return smooth_max(corr, float(beta)) * dx_f
            try:
                res = minimize(_obj_final, best_h, method='SLSQP', bounds=bounds_f,
                               constraints=cons_f,
                               options={'maxiter': 1000, 'ftol': 1e-15})
                h_new = project_h(res.x, best_n / 2.0)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass

    return best_h, best_c5, best_n
