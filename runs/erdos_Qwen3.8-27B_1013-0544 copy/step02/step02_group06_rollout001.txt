def min_1d_convex(a, b, lo, hi, iters=50):
    """Minimize max_m (a[m] + b[m]*u) over u in [lo,hi] using subgradient bisection."""
    for _ in range(iters):
        mid = (lo + hi) / 2
        vals = a + b * mid
        m_star = np.argmax(vals)
        if b[m_star] > 1e-14:
            hi = mid  # function is increasing at mid, go left
        elif b[m_star] < -1e-14:
            lo = mid  # function is decreasing at mid, go right
        else:
            break
    return (lo + hi) / 2

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            if d > 0:
                # increase smallest values
                order = np.argsort(h)
                cnt = min(int(np.ceil(abs(d))), n)
                for idx in order[:cnt]:
                    add = min(d, 1.0 - h[idx])
                    h[idx] += add
                    d -= add
            else:

                cnt = min(int(np.ceil(abs(d))), n)

                    sub = min(-d, h[idx])
                    h[idx] -= sub
                    d += sub

def compute_D(h, j):

        elif b_active.max() < -1e-14:

            k = idx[2 * p + 1]
            s = h[j] + h[k]
            
            D_j = compute_D(h, j)
            D_k = compute_D(h, k)
            E = D_j - D_k
            
            d_lo = max(-h[j], h[k] - 1.0)
            d_hi = min(1.0 - h[j], h[k])
            
            if d_hi - d_lo < 1e-14:

                h[k] -= d_opt
                corr += d_opt * E
                h[j] = np.clip(h[j], 0, 1)
                h[k] = np.clip(h[k], 0, 1)
    
    return h, compute_c5(h)

        starts = make_starts(n, ts, h_parent, rng)

        max_starts = max(2, min(len(starts), int(remaining() / 40)))
        for h_start in starts[:max_starts]:
            if remaining() < 30:
                break
            h_work = h_start.copy()
            # Pair coordinate descent (multiple rounds with different rng)
            for sweep_count in [40, 30, 20]:
                h_work = pair_cd(h_work.copy(), ts, sweep_count, rng)
            c_cd = compute_c5(h_work)

            # L-BFGS-B polish
            if remaining() > 25:
                try:
                    for beta, lam in [(200, 100), (500, 300), (1000, 500)]:
                        r = minimize(smooth_obj, h_work, args=(n, ts, beta, lam),
                                     method='L-BFGS-B', bounds=bnds,
                                     options={'maxiter': 300, 'ftol': 1e-15})
                        h_new = project_h(r.x, ts)
                        c_new = compute_c5(h_new)
                        if c_new < c_cd:
                            h_work, c_cd = h_new, c_new
                except Exception:
                    pass

            # Final coordinate descent polish
            if remaining() > 15:
                h_cd2 = pair_cd(h_work.copy(), ts, 50, rng)
                c2 = compute_c5(h_cd2)
                if c2 < c_cd:
                    h_work, c_cd = h_cd2, c2

            if c_cd < best_c5:
                best_h, best_c5, best_n = h_work.copy(), c_cd, n

    # Final aggressive polish on best solution
    if best_h is not None and remaining() > 25:
        n = best_n
        ts = n / 2.0
        bnds = [(0.0, 1.0)] * n
        # Coordinate descent
        best_h = pair_cd(best_h.copy(), ts, 60, rng)
        best_c5 = compute_c5(best_h)
        # L-BFGS-B with high beta
        if remaining() > 15:
            try:
                for beta, lam in [(2000, 1000), (3000, 2000)]:
                    if remaining() < 8:
                        break
                    r = minimize(smooth_obj, best_h, args=(n, ts, beta, lam),
                                 method='L-BFGS-B', bounds=bnds,
                                 options={'maxiter': 500, 'ftol': 1e-16})
                    h_new = project_h(r.x, ts)
                    c_new = compute_c5(h_new)
                    if c_new < best_c5:
                        best_h, best_c5 = h_new, c_new
            except Exception:
                pass
        # One more CD round
        if remaining() > 5:
            h_final = pair_cd(best_h.copy(), ts, 40, rng)
            c_final = compute_c5(h_final)
            if c_final < best_c5:
                best_h, best_c5 = h_final, c_final

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = 70

    best_h = project_h(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
