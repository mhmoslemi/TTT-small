This makes sense! The lag 0 term is always n/4 * dx = 0.5 when sum(h) = n/2 and h is in [0,1]... wait no, that's only when h is uniform.

Actually, c_0 = sum_j h[j]*(1-h[j]) = sum_j h[j] - sum_j h[j]^2 = n/2 - sum_j h[j]^2.

To minimize c_0, we want to maximize sum_j h[j]^2 subject to sum_j h[j] = n/2 and 0 ≤ h[j] ≤ 1.
This is maximized when h is as "spread out" as possible: half 0s and half 1s.
Then sum h^2 = n/2, so c_0 = n/2 - n/2 = 0.

So if h is {0,1}-valued with exactly n/2 ones, c_0 = 0!

But then for other lags, c_k might be large. The trade-off is between making c_0 small (by making h binary) and making other c_k small (by making h smooth).

This is the fundamental tension in the problem. The optimal h is somewhere in between.

This suggests that a good approach might be to:
1. Start with a binary or near-binary h
2. Smooth it gradually while tracking the max lag
3. Find the sweet spot

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def compute_all_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_grad_c(h, lag_idx):
    n = len(h)
    s = lag_idx - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        if 0 <= i - s < n:
            g += (1.0 - h[i - s])
        if 0 <= i + s < n:
            g -= h[i + s]
        grad[i] = g
    return grad


def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 0.0) & (h < 1.0)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def objective_smooth(h, n, ts, beta, lam):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * (2.0 / n) + lam * (h.sum() - ts) ** 2


def slsqp_func(h_flat, n, ts, active_lags, dx):
    h = h_flat[:n]
    t = h_flat[n]
    cons = []
    for lag in active_lags:
        cons.append(np.correlate(h, 1.0 - h, mode="full")[lag] * dx - t)
    cons.append(h.sum() - ts)
    return cons


def slsqp_jac(h_flat, n, ts, active_lags, dx):
    h = h_flat[:n]
    t = h_flat[n]
    jac_rows = []
    for lag in active_lags:
        g = compute_grad_c(h, lag) * dx
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        jac_rows.append(row)
    eq_row = np.zeros(n + 1)
    eq_row[:n] = 1.0
    jac_rows.append(eq_row)
    return np.array(jac_rows)

            h_new = project_box_sum(r.x, ts)
