from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def lse_objective(h, beta, target_sum, lam):
    """Log-sum-exp based objective with augmented Lagrangian for sum constraint"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    lse = (np.log(np.sum(np.exp(beta * (corr - m)))) + beta * m) / beta
    return lse * (2.0 / n) + lam * (h.sum() - target_sum) ** 2

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    x = np.linspace(0, 2.0, n)
    a1 = rng.uniform(0.15, 0.45)
    w1 = rng.uniform(1.0, 4.0) * np.pi
    p1 = rng.uniform(0, 2 * np.pi)
    a2 = rng.uniform(-0.15, 0.15)
    w2 = rng.uniform(3.0, 8.0) * np.pi
    p2 = rng.uniform(0, 2 * np.pi)
    h = 0.5 + a1 * np.sin(w1 * x + p1) + a2 * np.sin(w2 * x + p2)
    return project_sum(np.clip(h, 0, 1), n / 2.0)


def make_block_start(n, s):
    rng = np.random.RandomState(s)
    h = np.zeros(n)
    n_blocks = rng.randint(2, max(3, n // 12))
    blen = max(2, n // (2 * n_blocks))
    for i in range(n_blocks):
        st = rng.randint(0, max(1, n - blen))
        h[st:st + blen] = 1.0
    h += rng.randn(n) * 0.08
    return project_sum(h, n / 2.0)


def local_optimize(h, n, target, bnds, t0, deadline):
    h = project_sum(h, target)
    best = compute_c5(h)
    scales = [(30, 5, 120), (100, 20, 200), (300, 50, 300), (800, 100, 400), (2000, 300, 500)]
    for beta, lam, maxiter in scales:
        if time.time() > deadline - 5:
            break
        args = (beta, target, lam)
        try:
            r = minimize(lse_obj, h, args=(args,), method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
            h_new = project_sum(r.x, target)
            c_new = compute_c5(h_new)
            if c_new < best:
                h, best = h_new, c_new
        except Exception:
            pass
    return h, best


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    best_c5 = float('inf')
    best_h = None
    best_n = 0

    has_init = ('initial_h_values' in globals() and initial_h_values is not None)

    # Prioritize n=70 (initial), then 100 (more freedom), then 50 (easier)
    n_list = [70, 100, 50]
    if has_init:
        ni = len(initial_h_values)
        if ni not in n_list:
            n_list.append(ni)
        n_list.remove(ni)
        n_list.insert(0, ni)

    # Time allocation: 50% to first n, 30% to second, 20% to rest
    fracs = [0.50, 0.30, 0.10, 0.05, 0.05]
    while len(fracs) < len(n_list):
        fracs.append(0.02)

    for ni, n in enumerate(n_list):
        deadline_n = t0 + budget_s * sum(fracs[:ni + 1])
        deadline_n = min(deadline_n, t0 + budget_s - 20)
        if time.time() > deadline_n - 10:
            continue

        target = n / 2.0
        bnds = [(0.0, 1.0)] * n

        starts = []
        if has_init:
            init_h = np.array(initial_h_values, dtype=float)
            if len(init_h) == n:
                starts.append(project_sum(init_h, target))
            else:
                starts.append(resample_h(init_h, n))

        for s in range(3):
            starts.append(make_binary_start(n, seed + s * 137))
        for s in range(3):
            starts.append(make_sinusoidal_start(n, seed + 700 + s * 137))
        for s in range(2):
            starts.append(make_block_start(n, seed + 1400 + s * 137))

        # Limit starts to fit in time budget
        max_starts = 6 if ni == 0 else 5
        starts = starts[:max_starts]

        for h_start in starts:
            if time.time() > deadline_n - 5:
                break
            # Local optimization
            h_opt, c_opt = local_optimize(h_start, n, target, bnds, t0, deadline_n - 10)

            # Equalize top lags
            if time.time() < deadline_n - 15:
                h_eq = equalize_lags(h_opt, top_k=5, n_iter=40, lr=0.02)
                c_eq = compute_c5(h_eq)
                if c_eq < c_opt:
                    h_opt, c_opt = h_eq, c_eq

            # Post-equalization polish
            if time.time() < deadline_n - 8:
                args = (1500, target, 300.0)
                try:
                    r = minimize(lse_obj, h_opt, args=(args,), method='L-BFGS-B',
                                 bounds=bnds, options={'maxiter': 600, 'ftol': 1e-16})
                    h_new = project_sum(r.x, target)
                    c_new = compute_c5(h_new)
                    if c_new < c_opt:
                        h_opt, c_opt = h_new, c_new
                except Exception:
                    pass

            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()
                best_n = n

    # Final high-precision polish
    if best_h is not None:
        n = best_n
        target = n / 2.0
        bnds = [(0.0, 1.0)] * n
        for beta in [4000, 8000]:
            if time.time() > budget_s - 10:
                break
            args = (beta, target, 500.0)
            try:
                r = minimize(lse_obj, best_h, args=(args,), method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-15})
                h_new = project_sum(r.x, target)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_h, best_c5 = h_new, c_new
            except Exception:
                pass
        # One more equalization + polish
        if time.time() < budget_s - 15:
            h_eq = equalize_lags(best_h, top_k=4, n_iter=30, lr=0.005)
            c_eq = compute_c5(h_eq)
            if c_eq < best_c5:
                best_h, best_c5 = h_eq, c_eq
            args = (10000, target, 1000.0)
            try:
                r = minimize(lse_obj, best_h, args=(args,), method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 2000, 'ftol': 1e-18})
                h_new = project_sum(r.x, target)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_h, best_c5 = h_new, c_new
            except Exception:
                pass

    best_h = project_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
