    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    # Water-filling to enforce sum
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase sum: add to elements not at 1
        mask = h < 1.0
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    else:
        # Need to decrease sum: subtract from elements not at 0
        mask = h > 0.0
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)

    Returns arrays a, b, c0 of length 2n-1.

    d = np.zeros(n)
    d[j] = 1.0
    d[k] = -1.0

    corr0 = np.correlate(h, 1.0 - h, mode="full")

    # we need to compute the derivative of correlate(h+td, 1-h-td) w.r.t. t at t=0
    
    # np.correlate(h+td, 1-h-td) = np.correlate(h,1-h) + t*(correlate(d,1-h) - correlate(h,d)) - t²*correlate(d,d)
    
    # Wait, let me verify:
    # correlate(h+td, 1-h-td, full)[m] = sum_i (h[i]+t*d[i]) * (1-h[i+k]-t*d[i+k])
    # where the index mapping depends on the lag.
    
    # Actually, by linearity of correlate:
    # correlate(h+td, 1-h-td) = correlate(h, 1-h) + t*correlate(d, 1-h) + t*correlate(h, -d) + t²*correlate(d, -d)

    a = -np.correlate(d, d, mode="full")
    
    return a, b, corr0

def optimize_1d_quadratic_max(a, b, c, t_low, t_high, n_iter=40):
    """
    Minimize max_m (a_m * t^2 + b_m * t + c_m) for t in [t_low, t_high].
    Uses bisection on the value, then finds feasible t.
    """
    def f(t):
        return np.max(a * t * t + b * t + c)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = t_low, t_high
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    fc1, fc2 = f(c1), f(c2)
    
    for _ in range(n_iter):
        if fc1 < fc2:
            hi = c2
            c2 = c1
            fc2 = fc1
            c1 = hi - gr * (hi - lo)
            fc1 = f(c1)
        else:
            lo = c1
            c1 = c2
            fc1 = fc2
            c2 = lo + gr * (hi - lo)
            fc2 = f(c2)
    
    t_opt = (lo + hi) / 2
    return t_opt, f(t_opt)

def coordinate_descent(h, n_iter=5, n_pairs_per_iter=200, seed=42):
    """Coordinate descent in the null space of the sum constraint."""
    rng = np.random.RandomState(seed)
    n = len(h)
    target_sum = n / 2.0
    
    for it in range(n_iter):
        for _ in range(n_pairs_per_iter):
            j = rng.randint(n)
            k = rng.randint(n)
            if j == k:
                continue
            
            a, b, c = corr_quadratic_coeffs(h, j, k)
            
            # Find optimal t
            # t must satisfy: 0 <= h[j]+t <= 1 and 0 <= h[k]-t <= 1
            # i.e., max(-h[j], h[k]-1) <= t <= min(1-h[j], h[k])
            t_lo = max(-h[j], h[k] - 1.0)
            t_hi = min(1.0 - h[j], h[k])
            
            if t_lo >= t_hi:
                continue
            
            t_opt, _ = optimize_1d_quadratic_max(a, b, c, t_lo, t_hi, n_iter=30)
            
            h[j] += t_opt
            h[k] -= t_opt
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[k] = np.clip(h[k], 0.0, 1.0)
    
    return h

Let me also think about whether the golden section search is appropriate. The function f(t) = max_m (a_m t² + b_m t + c_m) is the maximum of quadratics. - At lag k-j: d[k]*d[j] = -1

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    else:
        mask = h > 0.0
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def corr_along_direction(h, d):
    """Compute a, b, c arrays for c(t) = a*t^2 + b*t + c along direction d."""
    a = -np.correlate(d, d, mode="full")
    b = np.correlate(d, 1.0 - h, mode="full") - np.correlate(h, d, mode="full")
    c = np.correlate(h, 1.0 - h, mode="full")
    return a, b, c

def minmax_quadratic(a, b, c, t_lo, t_hi, n_iter=50):
    """Minimize max(a*t^2 + b*t + c) over t in [t_lo, t_hi]."""
    def f(t):
        return float(np.max(a * t * t + b * t + c))
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = t_lo, t_hi
    if hi - lo < 1e-15:
        return (lo + hi) / 2.0
    
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    fc1, fc2 = f(c1), f(c2)
    for _ in range(n_iter):
        if fc1 < fc2:
            hi, c2, fc2 = c2, c1, fc1
            c1 = hi - gr * (hi - lo)
            fc1 = f(c1)
        else:
            lo, c1, fc1 = c1, c2, fc2
            c2 = lo + gr * (hi - lo)
            fc2 = f(c2)
    return (lo + hi) / 2.0

def coord_descent(h, n_rounds=10, pairs_per_round=400, seed=42, target_sum=None):

            d[j] = 1.0
            d[k] = -1.0
            a, b, c = corr_along_direction(h, d)
            t_lo = max(-h[j], h[k] - 1.0)
            t_hi = min(1.0 - h[j], h[k])
            if t_lo >= t_hi - 1e-15:

            t_opt = minmax_quadratic(a, b, c, t_lo, t_hi)

            h[k] -= t_opt
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[k] = np.clip(h[k], 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    else:
        mask = h > 0.0
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def minmax_quadratic(a, b, c, t_lo, t_hi, n_iter=40):
    def f(t):
        return float(np.max(a * t * t + b * t + c))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = t_lo, t_hi
    if hi - lo < 1e-14:
        return (lo + hi) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    fc1, fc2 = f(c1), f(c2)
    for _ in range(n_iter):
        if fc1 < fc2:
            hi, c2, fc2 = c2, c1, fc1
            c1 = hi - gr * (hi - lo)
            fc1 = f(c1)
        else:
            lo, c1, fc1 = c1, c2, fc2
            c2 = lo + gr * (hi - lo)
            fc2 = f(c2)
    return (lo + hi) / 2.0

def coord_descent(h, n_rounds=10, pairs_per_round=400, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h)
    for r in range(n_rounds):
        for _ in range(pairs_per_round):
            j = rng.randint(n)
            k = rng.randint(n)
            if j == k:
                continue
            d = np.zeros(n)
            d[j] = 1.0
            d[k] = -1.0
            a = -np.correlate(d, d, mode="full")
            b = np.correlate(d, 1.0 - h, mode="full") - np.correlate(h, d, mode="full")
            c = np.correlate(h, 1.0 - h, mode="full")
            t_lo = max(-h[j], h[k] - 1.0)
            t_hi = min(1.0 - h[j], h[k])
            if t_lo >= t_hi - 1e-14:
                continue
            t_opt = minmax_quadratic(a, b, c, t_lo, t_hi)
            h[j] += t_opt
            h[k] -= t_opt
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[k] = np.clip(h[k], 0.0, 1.0)
    return h

def slsqp_optimize(h0, n_pts, target_sum, beta, maxiter, ftol):
    dx = 2.0 / n_pts
    bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    def obj(h):
        return smooth_max(np.correlate(h, 1.0 - h, mode="full"), beta) * dx
    def con_fun(x):
        return np.sum(x) - target_sum
    def con_jac(x):
        return np.ones(n_pts)
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    best_h, best_c5, best_n = None, float('inf'), 0

    h_parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()

    def elapsed():
        return time.time() - t_start

    for n_pts in [70, 80, 100]:
        if elapsed() > budget_s - 90:
            break
        ts = n_pts / 2.0
        starts = []
        if h_parent is not None:
            if len(h_parent) == n_pts:
                starts.append(h_parent.copy())
            else:
                x_old = np.linspace(0, 2, len(h_parent), endpoint=False)
                x_new = np.linspace(0, 2, n_pts, endpoint=False)
                hi = np.interp(x_new, x_old, h_parent, period=2.0)
                starts.append(project_feasible(hi, ts))
        starts.append(project_feasible(np.random.random(n_pts), ts))

        for h0 in starts:
            if elapsed() > budget_s - 70:
                break
            curr_h, curr_c = h0.copy(), compute_c5(h0)

            for beta in [60, 120, 250, 500]:
                if elapsed() > budget_s - 55:
                    break
                try:
                    hn = slsqp_optimize(curr_h, n_pts, ts, beta, 300, 1e-14)
                    c = compute_c5(hn)
                    if c < curr_c:
                        curr_h, curr_c = hn, c
                except Exception:
                    pass

            if elapsed() < budget_s - 45:
                hcd = coord_descent(curr_h, n_rounds=10, pairs_per_round=500, seed=seed + n_pts)
                ccd = compute_c5(hcd)
                if ccd < curr_c:
                    curr_h, curr_c = hcd, ccd

            if elapsed() < budget_s - 30:
                for beta in [400, 700]:
                    if elapsed() > budget_s - 20:
                        break
                    try:
                        hn = slsqp_optimize(curr_h, n_pts, ts, beta, 400, 1e-15)
                        c = compute_c5(hn)
                        if c < curr_c:
                            curr_h, curr_c = hn, c
                    except Exception:
                        pass

            if elapsed() < budget_s - 15:
                hcd2 = coord_descent(curr_h, n_rounds=12, pairs_per_round=600, seed=seed + 100 + n_pts)
                ccd2 = compute_c5(hcd2)
                if ccd2 < curr_c:
                    curr_h, curr_c = hcd2, ccd2

            # Binary perturbation + re-optimize
            if elapsed() < budget_s - 10 and curr_c < 0.382:
                for flip_n in [2, 4, 6]:
                    if elapsed() > budget_s - 8:
                        break
                    hbin = (curr_h > 0.5).astype(float)
                    hbin = project_feasible(hbin, ts)
                    flip_idx = np.random.choice(n_pts, size=flip_n, replace=False)
                    hbin[flip_idx] = 1.0 - hbin[flip_idx]
                    hbin = project_feasible(hbin, ts)
                    try:
                        hn = slsqp_optimize(hbin, n_pts, ts, 300, 200, 1e-13)
                        hcd3 = coord_descent(hn, n_rounds=5, pairs_per_round=300, seed=seed + 200 + flip_n)
                        c3 = compute_c5(hcd3)
                        if c3 < curr_c:
                            curr_h, curr_c = hcd3, c3
                    except Exception:
                        pass

            if curr_c < best_c5:
                best_h, best_c5, best_n = curr_h.copy(), curr_c, n_pts

    # Final polish
    if best_h is not None and elapsed() < budget_s - 5:
        for rd in [15, 20]:
            hcd = coord_descent(best_h, n_rounds=rd, pairs_per_round=800, seed=seed + rd)
            cc = compute_c5(hcd)
            if cc < best_c5:
                best_h, best_c5 = hcd, cc
        if elapsed() < budget_s - 3:
            try:
                hn = slsqp_optimize(best_h, best_n, best_n / 2.0, 800, 500, 1e-16)
                c = compute_c5(hn)
                if c < best_c5:
                    best_h, best_c5 = hn, c
            except Exception:
                pass
            hcd = coord_descent(best_h, n_rounds=10, pairs_per_round=500, seed=seed + 999)
            cc = compute_c5(hcd)
            if cc < best_c5:
                best_h, best_c5 = hcd, cc

    return best_h, best_c5, best_n
