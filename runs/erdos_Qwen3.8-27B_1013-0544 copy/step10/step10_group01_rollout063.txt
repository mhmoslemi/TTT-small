    """Compute weighted sum of gradients of overlap_k.
    w has length 2n-1, indexed by correlation index m=0..2n-2.
    Returns gradient vector of length n."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2*n-1):
        if w[m] < 1e-15:
            continue
        k = m - (n-1)  # shift
        # For this shift, overlap = sum over valid i of h[i]*(1-h[i+k])*dx
        # grad[j] += w[m] * dx * [(1-h[j+k]) if 0<=j+k<n - h[j-k] if 0<=j-k<n]
        if k >= 0:
            # First term: (1-h[j+k]) valid when 0<=j<n and 0<=j+k<n => 0<=j<n-k
            # Second term: -h[j-k] valid when 0<=j-k<n and 0<=j<n => k<=j<n
            if n-k > 0:
                grad[:n-k] += w[m] * dx * (1.0 - h[k:].copy()[:n-k])  # (1-h[j+k]) for j=0..n-k-1
            if k < n:
                grad[k:] += -w[m] * dx * h[:n-k]  # -h[j-k] for j=k..n-1

        L = alpha * 8.0 / n
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                return best_h, best_c
            c_raw = np.correlate(y, 1.0 - y, mode='full')
            c = c_raw * (2.0 / n)
            sh = alpha * (c - c.max())
            exps = np.exp(sh)
            w = exps / exps.sum()
            grad = weighted_grad(y, n, w)
            h_new = y - grad / L
            h_new = project_feasible(h_new, n)
            c_check = np.correlate(h_new, 1.0 - h_new, mode='full') * (2.0 / n)
            sh_c = alpha * (c_check - c_check.max())
            f_new = (c_check.max() + np.log(np.sum(np.exp(sh_c)))) / alpha
            f_curr = (c.max() + np.log(np.sum(exps))) / alpha
            if f_new > f_curr + 0.5 / L * np.dot(grad, grad):
                L *= 2.0
                h_new = y - grad / L
                h_new = project_feasible(h_new, n)
            t_next = (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            mom = (t_k - 1.0) / t_next
            y = h_new + mom * (h_new - y)
            y = project_feasible(y, n)
            t_k = t_next
            cc = compute_c5(h_new, n)
            if cc < best_c:
                best_c = cc
                best_h = h_new.copy()
    return best_h, best_c

def coord_descent(h_init, n, max_passes, deadline, threshold=1e-10):
    h = h_init.copy()
    best_c = compute_c5(h, n)
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    dx = 2.0 / n
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            s_lo = max(-(p), -(n - 1 - p))
            s_hi = min(p, n - 1 - p)
            ms = np.arange(s_lo + n - 1, s_hi + n)
            n_ms = len(ms)
            lin = np.zeros(n_ms)
            for idx in range(n_ms):
                s_m = ms[idx] - (n - 1)
                if s_m == 0:
                    lin[idx] = 1.0 - 2.0 * h0
                elif s_m > 0:
                    v = 0.0
                    if p + s_m < n:
                        v += (1.0 - h[p + s_m])
                    if p - s_m >= 0:
                        v -= h[p - s_m]
                    lin[idx] = v
                else:
                    sa = -s_m
                    v = 0.0
                    if p - sa >= 0:
                        v += (1.0 - h[p - sa])
                    if p + sa < n:
                        v -= h[p + sa]
                    lin[idx] = v
            quad = np.zeros(n_ms)
            di = ms == (n - 1)
            if di.any():
                quad[di] = -1.0
            c_aff = c[ms]
            unaff_max = 0.0
            if n_ms < (2 * n - 1):
                full_mask = np.ones(2 * n - 1, dtype=bool)
                full_mask[ms] = False
                if full_mask.any():
                    unaff_max = float(c[full_mask].max())
            d_lo = -h0
            d_hi = 1.0 - h0
            def f_d(d):
                vals = c_aff + dx * (lin * d + quad * d * d)
                r = float(vals.max())
                if unaff_max > r:
                    r = unaff_max
                return r
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1, f2 = f_d(d1), f_d(d2)
            for _ in range(45):
                if d_hi - d_lo < 1e-14:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = f_d(d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = f_d(d2)
            d_opt = (d_lo + d_hi) / 2.0
            if f_d(d_opt) < float(c.max()) - 1e-13:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h[h < threshold] = 0.0
        h[h > 1.0 - threshold] = 1.0
        h = project_feasible(h, n)
        cn = compute_c5(h, n)
        if cn < best_c:
            best_c = cn
    return h, min(best_c, compute_c5(h, n))
