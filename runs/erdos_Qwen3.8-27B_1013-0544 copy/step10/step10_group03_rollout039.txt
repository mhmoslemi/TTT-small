def overlap_guided_perturb(h, n, rng, strength):
    """Perturb coordinates that contribute to high-overlap regions"""
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    # Find which shifts have high overlap
    active_shifts = c > cmax * 0.9
    # For each active shift, find contributing coordinate pairs
    contrib = np.zeros(n)
    for m in np.where(active_shifts)[0]:
        m_int = int(m)
        jf_range = range(max(0, m_int - n + 1), min(n, m_int + 1))
        for i in jf_range:
            j = i + m_int - (n - 1)
            if 0 <= j < n:
                # h[i] * (1-h[j]) contributes to this overlap
                contrib[i] += (1.0 - h[j])
                contrib[j] += h[i]

def overlap_guided_perturb(h, n, rng, strength):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    if cmax < 1e-15:
        return project_feasible(h + rng.normal(0, strength, n), n)
    active = c > 0.85 * cmax
    contrib = np.zeros(n)
    for m in np.where(active)[0]:
        m_int = int(m)
        lo = max(0, m_int - n + 1)
        hi = min(n, m_int + 1)
        for i in range(lo, hi):
            j = i + m_int - (n - 1)
            if 0 <= j < n:
                contrib[i] += (1.0 - h[j])
                contrib[j] += h[i]
    cmax_contrib = contrib.max()
    if cmax_contrib < 1e-15:
        return project_feasible(h + rng.normal(0, strength, n), n)
    weight = 0.5 + 0.5 * (contrib / cmax_contrib)
    noise = rng.normal(0, strength, n) * weight
    return project_feasible(h + noise, n)

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _penalty_obj(x, n, alpha, penalty):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    base = (m + lse / alpha) * (2.0 / n)
    viol = x.sum() - n / 2.0
    return base + penalty * viol * viol

def slsqp_opt(h, n, alphas, maxiter, deadline, penalty=100.0):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(_penalty_obj, best_h, args=(n, alpha, penalty),
                           method='SLSQP', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)

def overlap_perturb(h, n, rng, strength):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    if cmax < 1e-15:
        return project_feasible(h + rng.normal(0, strength, n), n)
    active = c > 0.85 * cmax
    contrib = np.zeros(n)
    for m in np.where(active)[0]:
        mi = int(m)
        lo = max(0, mi - n + 1)
        hi = min(n, mi + 1)
        for i in range(lo, hi):
            j = i + mi - (n - 1)
            if 0 <= j < n:
                contrib[i] += (1.0 - h[j])
                contrib[j] += h[i]
    cm = contrib.max()
    if cm < 1e-15:
        return project_feasible(h + rng.normal(0, strength, n), n)
    w = 0.5 + 0.5 * (contrib / cm)
    return project_feasible(h + rng.normal(0, strength, n) * w, n)

        lh[lh >= 0.5] = np.minimum(1.0, lh[lh >= 0.5] + s)
        lh = project_feasible(lh, n_work)
        h_opt, c_opt = exact_cd(lh, n_work, 300, deadline - 35, rng)
        if c_opt < best_c - 1e-10:
            h_best, best_c = h_opt, c_opt

    # Phase 5: Final intensive CD with random order
    n_final = len(h_best)
    h_f, c_f = exact_cd(h_best, n_final, 1000, deadline, rng)
    if c_f < best_c:
        h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
