    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(60):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if xc_lo < r1 and xc_hi > r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return c
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_i = np.where(ms == (n - 1))[0]
    c[ms] += delta * a
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta
    return c

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = list(range(n))
        if rng is not None and it % 2 == 1:
            rng.shuffle(order)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def build_A(h0, n):
    A = np.zeros((2 * n - 1, n))
    c0 = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        j_min = max(0, (n - 1) - m)
        j_max = min(n - 1, 2 * (n - 1) - m)
        js = np.arange(j_min, j_max + 1)
        ks = js + m - (n - 1)
        A[m, js] += (1.0 - h0[ks])
        A[m, ks] -= h0[js]
        c0[m] = np.dot(h0[js], h0[ks])
    return A, c0

def sca_step(h0, n):
    A, c0 = build_A(h0, n)
    inv_dx = n / 2.0
    cp_h = cp.Variable(n)
    L = cp.Variable()
    cons = [cp_h >= 0, cp_h <= 1, cp.sum(cp_h) == n / 2.0]
    cons.append(A @ cp_h + c0 <= L * inv_dx)
    prob = cp.Problem(cp.Minimize(L), cons)
    for solver in [cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            prob.solve(solver=solver)
            if cp_h.value is not None:
                return project_feasible(cp_h.value, n)
        except Exception:
            continue
    return h0

def sca_loop(h, n, max_iter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        h_new = sca_step(h, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
        if c_new >= best_c - 1e-9:
            improved = False
            for alpha in [0.7, 0.5, 0.3]:
                h_b = project_feasible(alpha * h_new + (1 - alpha) * h, n)
                c_b = compute_c5(h_b, n)
                if c_b < best_c:
                    best_h, best_c = h_b, c_b
                    h = h_b
                    improved = True
                    break
            if not improved:
                break
        h = h_new
    return best_h, best_c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)
    # Phase 1: CD on initial point
    h1, c1 = exact_cd(h_best, n, 80, deadline, rng)
    if c1 < best_c:
        h_best, best_c = h1, c1
    # Phase 2: SCA + CD
    h2, c2 = sca_loop(h_best, n, 60, deadline)
    if c2 < best_c - 1e-10:
        h_best, best_c = h2, c2
    h2a, c2a = exact_cd(h_best, n, 80, deadline, rng)
    if c2a < best_c - 1e-10:
        h_best, best_c = h2a, c2a
    # Phase 3: Multi-resolution with SCA
    n_list = [600, 800, 500, 900, 700, 1000, 650, 550]
    for n_try in n_list:
        if time.time() > deadline - 180:
            break
        if n_try == len(h_best):
            continue
        h_int = interp_h(h_best, len(h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 60, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o
        h_s, c_s = sca_loop(h_best, len(h_best), 30, deadline)
        if c_s < best_c - 1e-9:
            h_best, best_c = h_s, c_s
        h_o2, c_o2 = exact_cd(h_best, len(h_best), 60, deadline, rng)
        if c_o2 < best_c - 1e-9:
            h_best, best_c = h_o2, c_o2
    # Phase 4: Diversified restarts
    n_cur = len(h_best)
    for r in range(10):
        if time.time() > deadline - 60:
            break
        if r % 3 == 0:
            h_r = project_feasible(h_best + rng.normal(0, 0.03, n_cur), n_cur)
        elif r % 3 == 1:
            h_r = project_feasible(np.clip(0.5 + 0.4 * np.sin(np.pi * np.arange(n_cur) / n_cur * 2 + rng.uniform(0, 2 * np.pi)), 0, 1), n_cur)
        else:
            h_r = project_feasible(rng.uniform(0.2, 0.8, n_cur), n_cur)
        h_o, c_o = exact_cd(h_r, n_cur, 60, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o
            h_s2, c_s2 = sca_loop(h_best, n_cur, 20, deadline)
            if c_s2 < best_c - 1e-9:
                h_best, best_c = h_s2, c_s2
    # Phase 5: Final polish
    h_f, c_f = exact_cd(h_best, len(h_best), 150, deadline, rng)
    if c_f < best_c - 1e-10:
        h_best, best_c = h_f, c_f
    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
