                r2 = (1.0 + np.sqrt(disc)) * 0.5
                lo = max(x_lo, 0.0)
                hi = min(x_hi, 1.0)
                if not (lo < r1 and hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0

def cd_pass(h, c, n, order=None):
    """One pass of coordinate descent. Returns updated h, c, and number of changes."""
    if order is None:
        order = range(n)
    improved = 0
    for p in order:
        h0 = h[p]
        if h0 < 1e-14 or h0 > 1.0 - 1e-14:
            continue
        x, success = solve_1d(c, h, p, n)
        if not success:
            continue
        if abs(x - h0) > 1e-13:
            d = x - h0
            ms = np.arange(n - 1 - p, 2 * n - 1 - p)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c[ms] += d * a
            diag_i = np.where(ms == (n - 1))[0]
            if len(diag_i) > 0:
                c[ms[diag_i[0]]] -= d * d
            h[p] = x
            improved += 1
    return h, c, improved

            c = np.correlate(h, 1.0 - h, mode='full')
    return h, c

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if alpha < 1e6:
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * (2.0 / n)
    else:
        return m * (2.0 / n)

def slsqp(h, n, alpha, maxiter, deadline):

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        return new_h
    except Exception:
        return h
