        h_int = interp_h(h_best, len(h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 80, deadline, rng)
        if c_o < best_c - 1e-11:
            h_best, best_c = h_o, c_o
            n_base = n_try

    # Phase 6: Second SLSQP pass at best n
    n_cur = len(h_best)
    h6, c6 = slsqp_opt(h_best, n_cur, [500000, 2000000], 300, deadline)
    if c6 < best_c:
        h_best, best_c = h6, c6

    # Phase 7: Final CD
    h7, c7 = exact_cd(h_best, len(h_best), 300, deadline, rng)
    if c7 < best_c:
        h_best, best_c = h7, c7

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
