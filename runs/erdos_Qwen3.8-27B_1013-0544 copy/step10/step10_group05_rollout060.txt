**Approach: Lagrangian/Saddle-point method**

Instead of directly minimizing the max, we can use the following formulation:
min_h max_t { t : c_k ≥ t for all k } is equivalent to

3. Move in the direction that reduces the imbalance

One more idea: **iterative reweighting (proximal point)**

The problem min_h max_k c_k(h) can be seen as:
min_h max_k c_k(h) = min_h sup_{p ∈ Δ} sum_k p_k c_k(h)

By minimax theorem (if we relax to convex):
= sup_{p ∈ Δ} min_h sum_k p_k c_k(h)

But min_h sum_k p_k c_k(h) is nonconvex (because c_k is bilinear). However, we can use a "relaxed" version:
- Linearize c_k around the current point: c_k(h) ≈ c_k(h_old) + ∇c_k(h_old)·(h - h_old)
- Solve the linear program: min_h sum_k p_k [c_k(h_old) + ∇c_k(h_old)·(h - h_old)]
- Update p based on which lags are most violated

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            c_diag = c_base[diag_idx[0]]
            a_d = a[diag_idx[0]]
            disc = a_d * a_d + 4.0 * (c_diag - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (a_d - sq) / 2.0
                r2 = (a_d + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a_d = a[diag_idx[0]]
        disc = a_d * a_d + 4.0 * (c_diag - L)
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (a_d - sq) / 2.0
            r2 = (a_d + sq) / 2.0
            if x_lo <= h0 + r1:
                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):

            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                m_min = max(0, n - 1 - p)
                m_max = min(2 * n - 2, 2 * n - 2 - p)
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                diag_i = np.where(ms == (n - 1))[0]

                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:

def block_flip_perturb(h, n, rng, block_len):
    h_new = h.copy()
    L = min(block_len, n - 2)
    a = rng.randint(0, n - L)
    h_new[a:a+L] = 1.0 - h_new[a:a+L]
    return project_feasible(h_new, n)

def fourier_perturb(h, n, rng, max_freq):
    h_new = h.copy()
    k = rng.randint(1, max_freq)
    amp = rng.uniform(0.15, 0.6)
    phase = rng.uniform(0, 2.0 * np.pi)
    x = np.arange(n)
    h_new += amp * np.cos(2.0 * np.pi * k * x / n + phase)
    return project_feasible(h_new, n)

def noise_perturb(h, n, rng, amp):
    return project_feasible(h + rng.normal(0, amp, n), n)

def swap_perturb(h, n, rng, n_swaps):
    h_new = h.copy()
    for _ in range(n_swaps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i != j:
            h_new[i], h_new[j] = h_new[j], h_new[i]
    return project_feasible(h_new, n)

def interpolate_h(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)
