import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_block_lp(c, h, p_start, b, n):
    """Solve LP for block [p_start, p_start+b). Returns new block values."""
    n_corr = 2 * n - 1
    x_block = h[p_start:p_start + b].copy()
    
    # Build coefficient matrix A[n_corr, b]
    A = np.zeros((n_corr, b))
    for j in range(b):
        p = p_start + j
        # Forward: h[p]*(1-h[q]), q = p + m - (n-1)
        m_lo_f = max((n - 1) - p, 0)
        m_hi_f = min((n - 1) - p + n - 1, n_corr - 1)
        if m_lo_f <= m_hi_f:
            mq = np.arange(m_lo_f, m_hi_f + 1)
            q = p + mq - (n - 1)
            A[mq, j] += (1.0 - h[q])
        # Backward: h[r]*(1-h[p]), r = p - m + (n-1)
        m_lo_b = max(p - (n - 1), 0)
        m_hi_b = min(p - (n - 1) + n - 1, n_corr - 1)
        if m_lo_b <= m_hi_b:
            mb = np.arange(m_lo_b, m_hi_b + 1)
            r = p - mb + (n - 1)
            A[mb, j] -= h[r]
    
    # LP: min t s.t. c + A@delta - t <= 0, 0 <= x_block+delta <= 1
    # Variables: [delta(0..b-1), t]
    A_lp = np.zeros((n_corr, b + 1))
    A_lp[:, :b] = A
    A_lp[:, b] = -1.0
    b_lp = -c.copy()
    
    c_obj = np.zeros(b + 1)
    c_obj[b] = 1.0
    
    bounds = list(zip(-x_block, 1.0 - x_block))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A_lp, b_ub=b_lp, bounds=bounds, method='highs')
        if res.success:
            delta = res.x[:b]
            # Limit step size to prevent issues with quadratic self-terms
            max_step = max(np.abs(delta))
            if max_step > 0.3:
                delta *= 0.3 / max_step
            return np.clip(x_block + delta, 0.0, 1.0)
    except Exception:
        pass
    return x_block

def block_cd(h, n, block_size, max_passes, deadline, step_limit=0.3):

    n_corr = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    
    for _ in range(max_passes):

            break
        improved = False

def solve_1d(c, h, p, n):
    """Exact 1D coordinate update."""
    h0 = h[p]
    if h0 < 1e-13 or h0 > 1.0 - 1e-13:
        return h0
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if (xc_lo < r1 and xc_hi > r2) or (x_lo >= r1 and x_hi <= r2):
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + (lse + 0.0) / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x, n_=n: x.sum() - n_ / 2.0,
                 'jac': lambda x, n_=n: np.ones(n_)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
