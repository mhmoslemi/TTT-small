```python
def compute_delta_c(h, i, j, n):
    """Compute delta_c for swap i:0->1, j:1->0. h is the state BEFORE swap."""
    m = np.arange(-(n-1), n)  # physical shift values
    # i+m valid: 0 <= i+m < n
    valid_im = (i + m >= 0) & (i + m < n)
    # i-m valid: 0 <= i-m < n  
    valid_im2 = (i - m >= 0) & (i - m < n)
    # j+m valid
    valid_jm = (j + m >= 0) & (j + m < n)
    # j-m valid
    valid_jm2 = (j - m >= 0) & (j - m < n)
    
    delta = np.zeros(2*n-1)
    # Use numpy indexing
    im_idx = i + m[valid_im]  # these are valid indices into h
    delta[valid_im] += (1.0 - h[im_idx])
    
    im2_idx = i - m[valid_im2]
    delta[valid_im2] -= h[im2_idx]
    
    jm_idx = j + m[valid_jm]
    delta[valid_jm] -= (1.0 - h[jm_idx])
    
    jm2_idx = j - m[valid_jm2]
    delta[valid_jm2] += h[jm2_idx]
    
    return delta
```

```python
def delta_c_fast(h, i, j, n, c):
    """Compute new max of correlation after swap i:0->1, j:1->0.
    Returns (new_c_array or just the new max value)."""
    # We only need the new max, not the full array
    # For each k, the affected n values are in {i-k, j-k, i, j}
    # k ranges from -(n-1) to n-1
    
    new_max = 0.0
    for k in range(-(n-1), n):
        # Determine affected n values
        affected = set()
        for n_val in [i-k, j-k, i, j]:
            if 0 <= n_val < n and 0 <= n_val + k < n:
                affected.add(n_val)
        
        if not affected:
            # c[k + n - 1] is unchanged
            val = c[k + n - 1]
        else:
            # Recompute the affected terms
            # The new value of output[k] = old value + sum of changes for affected terms
            # But we need to be careful: the old value already includes the old contributions
            # of these terms. So: new_val = old_val - sum(old_contrib) + sum(new_contrib)
            old_contrib = 0.0
            new_contrib = 0.0
            for n_val in affected:
                a = h[n_val + k]
                b = h[n_val]
                old_contrib += a * (1 - b)
                # New values:
                a_new = 1.0 if (n_val + k == i) else (0.0 if (n_val + k == j) else a)
                b_new = 1.0 if (n_val == i) else (0.0 if (n_val == j) else b)
                new_contrib += a_new * (1 - b_new)
            val = c[k + n - 1] - old_contrib + new_contrib
        new_max = max(new_max, val)
    
    return new_max
```
