import time
import math
import numpy as np
from scipy.optimize import minimize, differential_evolution


def compute_c5(h):
    n = len(h)
    if n <= 0:
        return 1e9
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def anti_h(x):
    x = np.asarray(x, dtype=float)
    h = np.empty(2 * x.size, dtype=float)
    h[:x.size] = x
    h[x.size:] = 1.0 - x
    return h


def c5_x(x):
    return compute_c5(anti_h(x))


def resize_full(h, n):
    h = np.asarray(h, dtype=float)
    if h.size == n:
        return h.copy()
    x0 = (np.arange(h.size) + 0.5) * (2.0 / h.size)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x1, x0, h)


def parent_to_x(parent, n):
    h = resize_full(parent, n)
    m = n // 2
    x = 0.5 * (h[:m] - h[m:] + 1.0)
    return np.clip(x, 0.0, 1.0)


def resize_x(x, m):
    x = np.asarray(x, dtype=float)
    if x.size == m:
        return x.copy()
    t0 = (np.arange(x.size) + 0.5) / x.size
    t1 = (np.arange(m) + 0.5) / m
    return np.clip(np.interp(t1, t0, x), 0.0, 1.0)


def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    diff = target - h.sum()
    if diff != 0.0:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            v = h[idx] + diff
            v = min(1.0, max(0.0, v))
        else:
            idx = int(np.argmax(h))
            v = h[idx] + diff
            v = min(1.0, max(0.0, v))
        h[idx] = v
    return h


def _add_var_linear(h, r, sg, L, n):
    nm = n - 1
    L[nm] += sg * (1.0 - 2.0 * h[r])
    lim1 = min(r, n - 1)
    for d in range(1, lim1 + 1):
        hd = h[r - d]
        L[nm + d] += sg * (1.0 - hd)
        L[nm - d] += -sg * hd
    lim2 = min(n - 1 - r, n - 1)
    for d in range(1, lim2 + 1):
        hd = h[r + d]
        L[nm + d] += -sg * hd
        L[nm - d] += sg * (1.0 - hd)


def _fval_x(c, L, d, q0, qp, qn, buf, dx):
    np.multiply(L, float(d), out=buf)
    buf += c
    dd = d * d
    buf[q0] -= 2.0 * dd
    buf[qp] += dd
    buf[qn] += dd
    return float(buf.max()) * dx


def _line_search_x(c, L, lo, hi, q0, qp, qn, buf, dx,
                   grid_pts=25, gold_iters=18):
    f0 = _fval_x(c, L, 0.0, q0, qp, qn, buf, dx)
    d_best = 0.0
    f_best = f0
    best_idx = -1
    if hi - lo < 1e-14:
        return d_best, f_best
    ts = np.linspace(lo, hi, grid_pts)
    for i in range(grid_pts):
        t = float(ts[i])
        f = _fval_x(c, L, t, q0, qp, qn, buf, dx)
        if f < f_best:
            f_best = f
            d_best = t
            best_idx = i
    if best_idx < 0:
        return 0.0, f0
    left = float(ts[best_idx - 1]) if best_idx > 0 else float(lo)
    right = float(ts[best_idx + 1]) if best_idx < grid_pts - 1 else float(hi)
    if right - left < 1e-14:
        return d_best, f_best
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = right - gr * (right - left)
    d1 = left + gr * (right - left)
    fc = _fval_x(c, L, c1, q0, qp, qn, buf, dx)
    fd = _fval_x(c, L, d1, q0, qp, qn, buf, dx)
    for _ in range(gold_iters):
        if right - left < 1e-15:
            break
        if fc < fd:
            right = d1
            d1 = c1
            fd = fc
            c1 = right - gr * (right - left)
            fc = _fval_x(c, L, c1, q0, qp, qn, buf, dx)
        else:
            left = c1
            c1 = d1
            fc = fd
            d1 = left + gr * (right - left)
            fd = _fval_x(c, L, d1, q0, qp, qn, buf, dx)
    dm = 0.5 * (left + right)
    fm = _fval_x(c, L, dm, q0, qp, qn, buf, dx)
    if fm < f_best:
        return dm, fm
    return d_best, f_best


def cd_x(x, m, passes, deadline, grid_pts=25, gold_iters=18):
    best_x = np.clip(x.copy(), 0.0, 1.0)
    best_c = c5_x(best_x)
    n = 2 * m
    dx = 1.0 / m
    nm = n - 1
    qc0 = nm
    qc1 = nm + m
    qcm = nm - m
    L = np.empty(2 * n - 1)
    buf = np.empty(2 * n - 1)

    for _ in range(passes):
        if time.time() > deadline:
            break
        h = anti_h(x)
        c = np.asarray(np.correlate(h, 1.0 - h, mode='full'), dtype=float)
        curr_c = float(c.max()) * dx
        if curr_c < best_c:
            best_c = curr_c
            best_x = x.copy()
        improved = False
        for p in range(m):
            xp = x[p]
            if xp <= 1e-13 or xp >= 1.0 - 1e-13:
                continue
            if time.time() > deadline:
                break
            L.fill(0.0)
            _add_var_linear(h, p, 1.0, L, n)
            _add_var_linear(h, m + p, -1.0, L, n)
            lo = -xp
            hi = 1.0 - xp
            d, f = _line_search_x(c, L, lo, hi, qc0, qc1, qcm, buf, dx,
                                  grid_pts, gold_iters)
            if f < curr_c - 1e-13 and abs(d) > 1e-15:
                if d < lo:
                    d = lo
                elif d > hi:
                    d = hi
                x[p] = xp + d

            break

            best_c = cc
            best_x = x.copy()

        best_c = cc
    return best_x, best_c


def _smooth_x_fg(x, m, alpha):
    n = 2 * m
    dx = 1.0 / m
    nm = n - 1
    h = np.empty(n, dtype=float)
    h[:m] = x
    h[m:] = 1.0 - x
    c = np.correlate(h, 1.0 - h, mode='full')
    maxc = float(c.max())
    sh = alpha * (c - maxc)
    ex = np.exp(sh)
    s = float(ex.sum())
    if s <= 0.0 or not math.isfinite(s):
        s = 1.0
        ex = np.zeros_like(c)
        ex[nm] = 1.0
    val = (maxc + math.log(s) / alpha) * dx
    w = ex / s
    g = w[nm] * (1.0 - 2.0 * h)
    thr = 1e-12
    for d in range(1, n):
        wp = w[nm + d]
        if wp > thr:
            g[d:] += wp * (1.0 - h[:n - d])
            g[:n - d] -= wp * h[d:]
        wn = w[nm - d]
        if wn > thr:
            g[:n - d] += wn * (1.0 - h[d:])
            g[d:] -= wn * h[:n - d]
    gx = g[:m] - g[m:]
    return val, gx


def smooth_optimize_x(x, m, deadline, alphas, maxiter):
    best_x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0).copy()
    best_c = c5_x(best_x)
    bounds = [(0.0, 1.0)] * m
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(_smooth_x_fg, best_x, args=(m, float(alpha)),
                           method='L-BFGS-B', bounds=bounds, jac=True,
                           options={'maxiter': int(maxiter),
                                    'ftol': 1e-12,
                                    'gtol': 1e-6,
                                    'maxfun': int(3 * maxiter + 100),
                                    'maxls': 30})
        except Exception:
            continue
        rx = res.x
        if rx is None:
            continue
        if not np.all(np.isfinite(rx)):
            continue
        nx = np.clip(rx, 0.0, 1.0)
        c = c5_x(nx)
        if c < best_c:
            best_x = nx.copy()
            best_c = c
    return best_x, best_c


def build_basis(m):
    t = (np.arange(m) + 0.5) / m
    cols = [
        np.ones(m),
        t - 0.5,
        (t - 0.5) * (t - 0.5),
    ]
    for f in (1, 2, 3, 4):
        cols.append(np.cos(2.0 * np.pi * f * t))
        cols.append(np.sin(2.0 * np.pi * f * t))
    return np.column_stack(cols)


def basis_bounds(nv):
    b = [(0.0, 1.0), (-1.5, 1.5), (-1.0, 1.0)]
    b.extend([(-0.9, 0.9)] * max(0, nv - 3))
    return b


def _de_objective(params, B):
    x = np.clip(params @ B.T, 0.0, 1.0)
    return c5_x(x)


def _add_start(starts, vec):
    vec = np.clip(np.asarray(vec, dtype=float), 0.0, 1.0)
    for v in starts:
        if np.sum(np.abs(v - vec)) < 1e-12:
            return
    starts.append(vec)
