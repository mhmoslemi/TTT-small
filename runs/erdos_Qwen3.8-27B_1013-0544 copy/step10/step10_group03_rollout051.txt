                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(ms.shape[0])
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                d = np.where(ms == (n - 1))[0]
                if len(d) > 0:
                    c[ms[d[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n * 0.5) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
        cur = compute_c5(h, n)
        if cur < best_local:
            best_local = cur
    return h, best_local

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def _cf(x_):
            return x_.sum() - n * 0.5
        def _cj(x_):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: SLSQP polish on initial
    remaining = deadline - time.time()
    if remaining > 200:
        alphas = [2000, 8000, 30000, 80000]
        h1, c1 = slsqp_opt(h_best, n, alphas, 300, deadline - 200)
        if c1 < best_c:
            h_best, best_c = h1, c1

    # Phase 2: CD on current best
    remaining = deadline - time.time()
    if remaining > 100:
        h2, c2 = exact_cd(h_best, n, 600, deadline - 100, rng)
        if c2 < best_c:
            h_best, best_c = h2, c1 if False else c2

    # Phase 3: Multi-n exploration
    n_list = [400, 500, 600, 700, 800, 900, 1000, 1200]
    if n == 700:
        n_list = [x for x in n_list if x != 700] + [700]
    for n_try in n_list:
        if time.time() > deadline - 300:
            break
        x_old = np.linspace(0, 2, len(h_best), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = slsqp_opt(h_int, n_try, [5000, 30000], 200, deadline - 300)
        h_o, c_o = exact_cd(h_o, n_try, 300, deadline - 300, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o

    # Phase 4: Perturbation + CD cycles with varying scale
    n = len(h_best)
    scales = [0.002, 0.005, 0.01, 0.02, 0.005, 0.01, 0.003, 0.015, 0.008, 0.012,
              0.006, 0.018, 0.004, 0.014, 0.01, 0.007]
    for si in range(len(scales)):
        if time.time() > deadline - 150:
            break
        s = scales[si]
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        h_o, c_o = slsqp_opt(hp, n, [10000, 50000], 200, deadline - 150)
        h_o, c_o = exact_cd(h_o, n, 250, deadline - 150, rng)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 5: Asymmetric perturbation (push one half differently)
    if time.time() < deadline - 100:
        half = n // 2
        for _ in range(3):
            if time.time() > deadline - 80:
                break
            pert = np.zeros(n)
            sign = rng.choice([-1, 1])
            pert[:half] = sign * rng.uniform(0.005, 0.02, half)
            pert[half:] = -sign * rng.uniform(0.005, 0.02, n - half)
            hp = project_feasible(h_best + pert, n)
            h_o, c_o = exact_cd(hp, n, 300, deadline - 80, rng)
            if c_o < best_c:
                h_best, best_c = h_o, c_o

    # Phase 6: Final intensive CD
    if time.time() < deadline - 5:
        n = len(h_best)
        h_f, c_f = exact_cd(h_best, n, 800, deadline, rng)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
