    best_h, best_n, best_c = h_p, n_p, compute_c5(h_p, n_p)
    t_phase1 = t0 + 0.18 * budget_s
    t_phase2 = t0 + 0.50 * budget_s
    t_phase3 = t0 + 0.78 * budget_s
    t_phase4 = t0 + 0.92 * budget_s
    cands = []
    # Phase 1: diverse starts
    for nt in [550, 600, 650, 700, 750, 800]:
        if time.time() > t_phase1 - 20:
            break
        ht = h_p.copy() if nt == n_p else interp_h(h_p, n_p, nt)
        ht, ct = exact_cd(ht, nt, 50, t_phase1 - 20)
        cands.append((ct, nt, ht))
    for nb in [4, 5, 6, 8, 10, 12]:
        if time.time() > t_phase1 - 15:
            break
        ht, ct = exact_cd(make_blocks(n_p, nb), n_p, 25, t_phase1 - 15)
        cands.append((ct, n_p, ht))
    for f, a in [(2, 0.3), (3, 0.3), (5, 0.2), (7, 0.25), (4, 0.35)]:
        if time.time() > t_phase1 - 10:
            break
        ht, ct = exact_cd(make_wave(n_p, f, a), n_p, 20, t_phase1 - 10)
        cands.append((ct, n_p, ht))
    for f1, f2, a1, a2 in [(2, 5, 0.25, 0.15), (3, 7, 0.2, 0.2), (1, 4, 0.3, 0.2)]:
        if time.time() > t_phase1 - 8:
            break
        ht, ct = exact_cd(make_wave2(n_p, f1, f2, a1, a2), n_p, 20, t_phase1 - 8)
        cands.append((ct, n_p, ht))
    for _ in range(4):
        if time.time() > t_phase1 - 5:
            break
        hr = project_feasible((rng.random(n_p) < 0.5).astype(float), n_p)
        ht, ct = exact_cd(hr, n_p, 15, t_phase1 - 5)
        cands.append((ct, n_p, ht))
    cands.sort(key=lambda x: x[0])
    if len(cands) > 0 and cands[0][0] < best_c:
        best_c, best_n, best_h = cands[0][0], cands[0][1], cands[0][2]
    # Phase 2: intensive refinement
    for i in range(min(4, len(cands))):
        if time.time() > t_phase2 - 50:
            break
        ci, ni, hi = cands[i]
        hi, ci = exact_cd(hi, ni, 300, t_phase2 - 50)
        if ci < best_c:
            best_c, best_n, best_h = ci, ni, hi
        for s in [0.03, 0.06, 0.10, 0.15]:
            if time.time() > t_phase2 - 30:
                break
            hp = project_feasible(hi + rng.normal(0, s, ni), ni)
            hp, cp = exact_cd(hp, ni, 100, t_phase2 - 30)
            if cp < best_c:
                best_c, best_n, best_h = cp, ni, hp
    if time.time() < t_phase2 - 10:
        best_h, best_c = exact_cd(best_h, best_n, 200, t_phase2 - 10)
    # Phase 3: adaptive perturbation
    n = best_n
    for r in range(60):
        if time.time() > t_phase3:
            break
        s = 0.002 + 0.02 * (r % 15) / 15.0
        hp = project_feasible(best_h + rng.normal(0, s, n), n)
        hp, cp = exact_cd(hp, n, 80, t_phase3)
        if cp < best_c:
            best_c, best_h = cp, hp
    # Phase 4: binary shake
    n = len(best_h)
    for _ in range(25):
        if time.time() > t_phase4:
            break
        nflip = rng.randint(n // 30, n // 8)
        idx = rng.choice(n, size=nflip, replace=False)
        hp = best_h.copy()
        hp[idx] = 1.0 - hp[idx]
        hp = project_feasible(hp, n)
        hp, cp = exact_cd(hp, n, 60, t_phase4)
        if cp < best_c:
            best_c, best_h = cp, hp
    # Phase 5: block complementation
    n = len(best_h)
    for _ in range(15):
        if time.time() > deadline - 60:
            break
        blen = rng.randint(n // 15, n // 4)
        start = rng.randint(0, n - blen)
        hp = best_h.copy()
        hp[start:start + blen] = 1.0 - hp[start:start + blen]
        hp = project_feasible(hp, n)
        hp, cp = exact_cd(hp, n, 50, deadline - 60)
        if cp < best_c:
            best_c, best_h = cp, hp
    # Phase 6: final intensive CD
    n = len(best_h)
    best_h, best_c = exact_cd(best_h, n, 2000, deadline)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
