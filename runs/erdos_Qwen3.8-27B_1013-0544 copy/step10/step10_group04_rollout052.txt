                h_bin[idx[i]] = 0.0
        elif excess < 0:
            idx = np.where(h_bin == 0.0)[0]
            for i in range(min(int(np.ceil(-excess)), len(idx))):
                h_bin[idx[i]] = 1.0
        h_bin = project_feasible(h_bin, n)
        ho, co = exact_cd(h_bin, n, 600, deadline - 40)
        if co < best_c:
            h_best, best_c, n = ho, co, n

    # Phase 7: Final heavy CD polish
    if time.time() < deadline - 5:
        hf, cf = exact_cd(h_best, n, 800, deadline)
        if cf < best_c:
            h_best, best_c = hf, cf

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)
    return h_best, best_c, n
