def solve_1d_bisect(c, h, p, n):

    m_min = n - 1 - p

    ms = np.arange(m_min, m_max + 1)

    jb = p - ms + (n - 1)

                best_b = b.copy()
        else:
            b[i], b[j] = 1.0, 0.0
    return best_b, best_c

def targeted_perturb(h, n, c, rng, scale):
    cmax = c.max()
    active = np.where(c >= cmax - 0.02 * cmax)[0]
    affected = set()
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        affected.update(range(i_min, i_max + 1))
    affected = np.array(list(affected), dtype=int)
    if len(affected) == 0:
        affected = np.arange(n)
    hp = h.copy()
    hp[affected] += rng.normal(0, scale, len(affected))
    return project_feasible(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Initial CD convergence
    h_best, best_c = exact_cd(h_best, n, 80, deadline - 900)

    # Phase 2: Binary SA + CD
    if time.time() < deadline - 600:
        b_sol, b_c = binary_sa(h_best, n, deadline - 400, rng, 35000)
        h_ref = project_feasible(b_sol, n)
        h_o, c_o = exact_cd(h_ref, n, 150, deadline - 400)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 3: More SA with different temp
    if time.time() < deadline - 450:
        b_sol2, _ = binary_sa(h_best, n, deadline - 300, rng, 25000)
        h_ref2 = project_feasible(b_sol2, n)
        h_o2, c_o2 = exact_cd(h_ref2, n, 120, deadline - 300)
        if c_o2 < best_c:
            h_best, best_c = h_o2, c_o2

    # Phase 4: Targeted perturbation + CD
    n_curr = len(h_best)
    for round_i in range(6):
        if time.time() > deadline - 250:
            break
        scale = 0.015 * (0.7 ** round_i)
        c_arr = np.correlate(h_best, 1.0 - h_best, mode='full')
        for attempt in range(2):
            if time.time() > deadline - 250:
                break
            hp = targeted_perturb(h_best, n_curr, c_arr, rng, scale)
            h_o, c_o = exact_cd(hp, n_curr, 80, deadline - 250)
            if c_o < best_c:
                h_best, best_c = h_o, c_o
                break

    # Phase 5: Multi-n exploration
    for n_try in [600, 650, 750, 850, 950, 1100]:
        if time.time() > deadline - 150:
            break
        if n_try == n_curr:
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 120, deadline - 150)
        if c_o < best_c - 1e-10:
            h_best, best_c = h_o, c_o
            n_curr = n_try

    # Phase 6: Final intensive CD
    n_final = len(h_best)
    h_best, best_c = exact_cd(h_best, n_final, 300, deadline - 10)

    # Phase 7: Tiny perturbation polish
    if time.time() < deadline - 5:
        c_arr = np.correlate(h_best, 1.0 - h_best, mode='full')
        hp = targeted_perturb(h_best, n_final, c_arr, rng, 0.002)
        h_o, c_o = exact_cd(hp, n_final, 100, deadline - 2)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
