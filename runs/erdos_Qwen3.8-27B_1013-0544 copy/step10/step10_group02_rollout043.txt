def smooth_lse(x, n, alpha):

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    if c5v < best_c:
        h_best, best_c = h5, c5v

    # Phase 6: Try structured initializations at current n
    if time.time() < deadline - 200:
        hs, cs = try_structured(n, rng, deadline)
        if hs is not None and cs < best_c:
            h_best, best_c = hs, cs

    # Phase 7: Try lower resolutions too
    for n_try in [600, 500, 400]:
        if time.time() > deadline - 100: break
        hi = interp_h(h_best, len(h_best), n_try)
        ho, co = cd_optimize(hi, n_try, 150, deadline, rng)
        if co < best_c - 1e-9:
            h_best, best_c = ho, co

    # Phase 8: Final CD polish
    n = len(h_best)
    h_f, c_f = cd_optimize(h_best, n, 300, deadline, rng)
    if c_f < best_c - 1e-12:
        h_best, best_c = h_f, c_f

    # Final SLSQP if time permits
    if time.time() < deadline - 60:
        h_s, c_s = slsqp_opt(h_best, n, [50000, 200000], 200, deadline)
        if c_s < best_c - 1e-12:
            h_best, best_c = h_s, c_s
            h_best, c_s2 = cd_optimize(h_best, n, 200, deadline, rng)
            if c_s2 < best_c:
                h_best, best_c = h_best, c_s2

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
