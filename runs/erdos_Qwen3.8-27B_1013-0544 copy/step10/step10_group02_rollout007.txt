def coeff_row(h, n, m):
    row = np.zeros(n)

def lp_refine(h, n, iters, deadline, r0=0.18):

    best = h.copy()

    c = np.correlate(h, 1.0 - h, mode='full')
    mx = float(c.max())

        ce[n] = 1.0

                          bounds=bnds, method='highs')

                continue

                    h = ht

                    ok = True
                    break

                r *= 0.6
                if r < 0.01:
                    break

            break
    return project_feasible(best, n), compute_c5(best, n)

def interp_h(h_old, n_old, n_new):
    if n_old == n_new:
        return project_feasible(h_old, n_new)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)

    idx = np.argsort(-score)
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def smooth_init(n, rng):
    m = max(8, n // 12)
    z = rng.normal(0.0, 1.0, m)
    z = (z + np.roll(z, 1) + np.roll(z, -1)) / 3.0
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    f = np.interp(x_new, x_old, z)
    sd = float(f.std())
    if sd < 1e-10:
        f = rng.normal(0.0, 1.0, n)
        sd = float(f.std()) + 1e-10
    f = (f - f.mean()) / sd
    return project_feasible(0.5 + 0.45 * np.tanh(f), n)

def make_trial(n, kind, hb, rng):
    hb = project_feasible(hb, len(hb))
    if kind == 0:
        return interp_h(hb, len(hb), n)
    if kind == 1:
        return threshold_binary(interp_h(hb, len(hb), n), n, rng, 0.06)
    if kind == 2:
        h0 = interp_h(hb, len(hb), n)
        return project_feasible(h0 + rng.normal(0.0, 0.06 + 0.1 * rng.rand(), n), n)
    return smooth_init(n, rng)

def refine_solution(h, n, cd1, cd2, lp1, deadline, rng):
    h, c = exact_cd(h, n, cd1, deadline, rng)
    if time.time() < deadline - 3 and lp1 > 0:
        h, c = lp_refine(h, n, lp1, deadline, 0.18)
    if time.time() < deadline - 3:
        h, c = exact_cd(h, n, cd2, deadline, rng)
    if time.time() < deadline - 3 and lp1 > 0:
        h, c = lp_refine(h, n, max(5, lp1 // 2), deadline, 0.05)
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        hp = np.asarray(initial_h_values, float).copy().ravel()
    except Exception:
        hp = smooth_init(600, rng)
    if hp.size % 2:
        hp = hp[:-1]
    if hp.size < 160:
        hp = smooth_init(480, rng)
    if hp.size > 1000:
        hp = interp_h(hp, hp.size, 1000)
    h = project_feasible(hp, hp.size)
    n = len(h)
    h, c = refine_solution(h, n, 55, 28, 18, min(deadline, t0 + 0.35 * budget_s), rng)
    best_h, best_c = h, c

    base = [n, n - 32, n + 32, 640, 720, 560, 480, 520]
    nl = []
    seen = set()
    for v in base:
        v = int(v)
        if v % 2 == 0 and 160 <= v <= 1000 and v not in seen:
            seen.add(v)
            nl.append(v)
    if not nl:
        nl = [512]

    kind = 1
    while time.time() < deadline - 120 and kind < 12:
        ntr = nl[kind % len(nl)]
        h0 = make_trial(ntr, kind % 4, best_h, rng)
        td = min(deadline - 60, time.time() + 45)
        hq, cq = refine_solution(h0, ntr, 16, 9, 6, td, rng)
        if cq < best_c - 1e-12:
            best_h, best_c = hq, cq
            td = min(deadline - 60, time.time() + 95)
            h2, c2 = refine_solution(hq, ntr, 45, 22, 13, td, rng)
            if c2 < best_c - 1e-12:
                best_h, best_c = h2, c2
        elif cq < best_c + 0.002 and ntr != len(best_h):
            hf = interp_h(hq, ntr, len(best_h))
            td = min(deadline - 60, time.time() + 65)
            h2, c2 = refine_solution(hf, len(best_h), 28, 14, 9, td, rng)
            if c2 < best_c - 1e-12:
                best_h, best_c = h2, c2
        kind += 1

    if time.time() < deadline - 20:
        h2, c2 = refine_solution(best_h, len(best_h), 42, 22, 16, deadline, rng)
        if c2 < best_c - 1e-12:
            best_h, best_c = h2, c2

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
