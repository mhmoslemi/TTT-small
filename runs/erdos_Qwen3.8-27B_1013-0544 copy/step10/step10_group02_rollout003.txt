```
def project_feasible(x, target):
    x = np.asarray(x, dtype=float)
    n = len(x)
    target = float(target)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = -float(np.max(x)) - 1.0
    hi = -float(np.min(x)) + 1.0
    # ensure lo < hi
    for _ in range(70):
        mid = 0.5*(lo+hi)
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    tau = 0.5*(lo+hi)
    z = np.clip(x + tau, 0.0, 1.0)
    diff = target - float(z.sum())
    if abs(diff) > 1e-12:
       mask = (z > 1e-15) & (z < 1.0 - 1e-15)
       if mask.any():
           z = np.clip(z + diff / float(mask.sum()), 0,1)
       else:
          # adjust lowest or highest
          if diff > 0:
             idx = np.argsort(z)[:int(diff)+1]
             z[idx] = 1.0
          else:
             idx = np.argsort(-(z))[:int(-diff)+1]
             z[idx] = 0.0
    # final tiny average adjust on interior? to ensure sum exact within 1e-15
    diff = target - float(z.sum())
    if abs(diff) > 1e-14:
       mask = (z > 1e-12) & (z < 1.0 - 1e-12)
       if mask.any(): z[mask] += diff / mask.sum()
    return z
```

def project_feasible(x,t):
    if t<=0: return np.zeros(len(x))
    if t>=len(x): return np.ones(len(x))
    lo=-float(np.max(x))-1.0; hi=-float(np.min(x))+1.0
    for _ in range(60):
        m=0.5*(lo+hi)
        if np.clip(x+m,0,1).sum() < t: lo=m
        else: hi=m
    z=np.clip(x+0.5*(lo+hi),0,1)
    d=t-z.sum()
    if abs(d)>1e-13:
        mask=(z>1e-15)&(z<1-1e-15)
        if mask.any(): z=np.clip(z+d/mask.sum(),0,1)
    d=t-z.sum()
    if abs(d)>1e-15:
        mask=(z>1e-12)&(z<1-1e-12)
        if mask.any(): z[mask]+=d/mask.sum()
    return z

```python
import numpy as np

_RANGE_CACHE = {}

def get_range_data(n):
    if n in _RANGE_CACHE:
        return _RANGE_CACHE[n]
    idx = np.arange(n)
    dq = np.maximum(idx, n - 1 - idx)
    starts = (n - 1 - dq).astype(np.intp)
    ends = (n - 1 + dq).astype(np.intp)
    deltas = [np.arange(int(starts[i]), int(ends[i]) + 1) - (n - 1) for i in range(n)]
    if len(_RANGE_CACHE) >= 5:
        _RANGE_CACHE.pop(next(iter(_RANGE_CACHE)))
    _RANGE_CACHE[n] = (starts, ends, deltas)
    return _RANGE_CACHE[n]

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def project_feasible(x, t):
    x = np.asarray(x, dtype=float)
    n = len(x)
    t = float(t)
    if t <= 0.0:
        return np.zeros(n)
    if t >= n:
        return np.ones(n)
    lo = -float(np.max(x)) - 1.0
    hi = -float(np.min(x)) + 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if float(np.clip(x + m, 0.0, 1.0).sum()) < t:
            lo = m
        else:
            hi = m
    z = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    d = t - float(z.sum())
    if abs(d) > 1e-13:
        mask = (z > 1e-15) & (z < 1.0 - 1e-15)
        if mask.any():
            z = np.clip(z + d / float(mask.sum()), 0.0, 1.0)
    d = t - float(z.sum())
    if abs(d) > 1e-15:
        mask = (z > 1e-12) & (z < 1.0 - 1e-12)
        if mask.any():
            z[mask] += d / float(mask.sum())
    return z

def smooth_obj_grad(h, n, alpha):
    r = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(r))
    z = alpha * (r - m)
    ex = np.exp(z)
    s = float(np.sum(ex))
    p = ex / s
    y = 1.0 - h
    L = 1
    while L < 3 * n - 2:
        L <<= 1
    f1 = np.fft.fft(p, L)
    f2 = np.fft.fft(p[::-1], L)
    f3 = np.fft.fft(y, L)
    f4 = np.fft.fft(h, L)
    a = np.fft.ifft(f1 * f3).real
    b = np.fft.ifft(f2 * f4).real
    st = n - 1
    g = (a[st:st + n] - b[st:st + n]) * (2.0 / n)
    F = (m + math.log(s) / alpha) * (2.0 / n)
    return F, g
```
