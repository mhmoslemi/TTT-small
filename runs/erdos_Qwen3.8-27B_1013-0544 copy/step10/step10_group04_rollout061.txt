    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)

        m_max = 2 * n - 2 - p

def power_mean_obj_grad(x, n, p):
    """Compute p-mean of correlations and its gradient."""
    c = np.correlate(x, 1.0 - x, mode='full')
    c_pos = np.maximum(c, 1e-30)
    obj = (np.sum(c_pos ** p) / len(c)) ** (1.0 / p) * (2.0 / n)
    # gradient: (2/n) * (1/p) * (mean(c^p))^(1/p - 1) * p * c^(p-1) * dc/dh * (1/len(c))
    # = (2/n) * (mean(c^p))^(1/p - 1) / len(c) * sum_k c_k^(p-1) * dc_k/dh
    # Simplify: weight_k = c_k^(p-1) / sum(c^p)
    # grad[p] = (2/n) * obj_base * sum_k weight_k * dc_k/dh_p
    # where obj_base = (mean(c^p))^(1/p)
    
    cp = c_pos ** (p - 1)
    w = cp / cp.sum()
    base = (np.sum(c_pos ** p) / len(c)) ** (1.0 / p) * (2.0 / n)
    
    grad = np.zeros(n)
    for i in range(n):
        m_min = n - 1 - i
        m_max = 2 * n - 2 - i
        ms = np.arange(m_min, m_max + 1)
        jf = i + ms - (n - 1)
        jb = i - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib = np.zeros(len(ms))
        if np.any(vf): contrib[vf] += (1.0 - x[jf[vf]])
        if np.any(vb): contrib[vb] -= x[jb[vb]]
        contrib[ms == (n - 1)] = (1.0 - 2.0 * x[i])
        grad[i] = base * np.dot(w[ms], contrib)
    return obj, grad

def smooth_descent(h, n, p, max_iter, deadline, lr_init=0.005):
    h = h.copy()
    best_h, best_obj = h.copy(), power_mean_obj_grad(h, n, p)[0]
    for it in range(max_iter):
        if time.time() > deadline: break
        obj, grad = power_mean_obj_grad(h, n, p)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
        gc = grad - grad.mean()
        lr = lr_init
        for _ in range(25):
            h_new = project_feasible(np.clip(h - lr * gc, 0, 1), n)
            v_new, _ = power_mean_obj_grad(h_new, n, p)
            if v_new < obj:
                h = h_new
                break
            lr *= 0.5
        else:
            break
    return best_h, compute_c5(best_h, n)
