                break
            p = pi
            q = n - 1 - p
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            T = np.zeros(2 * n - 1)
            vp = (p + ms >= 0) & (p + ms < n)
            vq = (q + ms >= 0) & (q + ms < n)
            vmp = (p - ms >= 0) & (p - ms < n)
            vmq = (q - ms >= 0) & (q - ms < n)
            T[vp] = 1.0 - h[p + ms[vp]]
            T[vq] -= 1.0 - h[q + ms[vq]]
            T[vmp] -= h[p - ms[vmp]]
            T[vmq] += h[q - ms[vmq]]
            L_lo = 0.0
            L_hi = float(np.max(c))
            for _ in range(nbisect):
                if L_hi - L_lo < 1e-12:
                    break
                Lm = (L_lo + L_hi) * 0.5
                pos = T > 1e-14
                neg = T < -1e-14
                zero = ~(pos | neg)
                ok = True
                if np.any(zero) and float(np.max(c[zero])) > Lm + 1e-13:
                    ok = False
                else:
                    dlo = -h0
                    dhi = 1.0 - h0
                    if np.any(pos):
                        dhi = min(dhi, float(np.min((Lm - c[pos]) / T[pos])))
                    if np.any(neg):
                        dlo = max(dlo, float(np.max((Lm - c[neg]) / T[neg])))
                    if dlo > dhi + 1e-12:
                        ok = False
                if ok:
                    L_hi = Lm
                else:
                    L_lo = Lm
            L = (L_lo + L_hi) * 0.5
            pos = T > 1e-14
            neg = T < -1e-14
            dlo = -h0
            dhi = 1.0 - h0
            if np.any(pos):
                dhi = min(dhi, float(np.min((L - c[pos]) / T[pos])))
            if np.any(neg):
                dlo = max(dlo, float(np.max((L - c[neg]) / T[neg])))
            if dlo > dhi + 1e-10:
                continue
            dopt = (dlo + dhi) * 0.5
            vnew = h0 + dopt
            if abs(vnew - h0) > 1e-12:
                h[p] = vnew
                h[q] = 1.0 - vnew
                improved = True
        if not improved:
            break
    h = h_to_sym(h, n)
    return h, compute_c5(h, n)

def lse_sym(x, n, alpha):
    h = 0.5 + x
    h = np.clip(h, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    lse = mx + np.log(np.sum(np.exp(sh - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_sym(h, n, alphas, maxiter, deadline):
    n_half = n // 2
    a = h_to_sym(h, n) - 0.5
    a = np.clip(a, -0.5, 0.5)
    best_a = a.copy()
    best_c = compute_c5(0.5 + best_a, n)
    bounds = [(-0.5, 0.5)] * n_half
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        def obj(a_first):
            a_full = np.zeros(n)
            a_full[:n_half] = a_first
            a_full[n_half:] = -a_first[::-1]
            if n % 2 == 1:
                a_full[n // 2] = 0.0
            return lse_sym(a_full, n, alpha)
        try:
            res = minimize(obj, best_a[:n_half] if n % 2 == 0 else best_a[:n_half],
                           method='SLSQP', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            new_a_first = res.x
            a_full = np.zeros(n)
            a_full[:n_half] = new_a_first
            a_full[n_half:] = -new_a_first[::-1]
            if n % 2 == 1:
                a_full[n // 2] = 0.0
            new_h = 0.5 + a_full
            new_h = np.clip(new_h, 0.0, 1.0)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_c = new_c
                best_a = a_full.copy()
        except Exception:
            pass
    return 0.5 + best_a, best_c

def make_initial(n, rng):
    a = np.zeros(n)
    n_half = n // 2
    a[:n_half] = rng.uniform(-0.3, 0.3, n_half)
    a[n_half:] = -a[:n_half][::-1]
    if n % 2 == 1:
        a[n // 2] = 0.0
    return np.clip(0.5 + a, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h_init)
    except Exception:
        h_init = None
        n0 = 700
    best_h = None
    best_c = 1.0
    n_opt = n0
    if h_init is not None:
        h_sym = h_to_sym(h_init, n0)
        h_sym = np.clip(h_sym, 0, 1)
        c_sym = compute_c5(h_sym, n0)
        if c_sym < best_c:
            best_h, best_c, n_opt = h_sym, c_sym, n0
    for n_try in [600, 700, 800, 500]:
        if time.time() > deadline - 100:
            break
        if n_try == n_opt and best_h is not None:
            h_start = best_h.copy()
        elif h_init is not None:
            x_new = np.linspace(0, 2, n_try)
            x_old = np.linspace(0, 2, len(h_init))
            h_start = h_to_sym(np.interp(x_new, x_old, h_init), n_try)
            h_start = np.clip(h_start, 0, 1)
        else:
            h_start = make_initial(n_try, rng)
        if time.time() > deadline - 60:
            break
        h1, c1 = cd_sym(h_start, n_try, 60, deadline, rng, 35)
        if c1 < best_c:
            best_h, best_c, n_opt = h1, c1, n_try
        if time.time() > deadline - 60:
            break
        h2, c2 = slsqp_sym(best_h if best_h is not None else h1, n_opt, [500, 2000, 5000], 200, deadline)
        if c2 < best_c:
            best_h, best_c, n_opt = h2, c2, n_opt
        h3, c3 = cd_sym(best_h, n_opt, 80, deadline, rng, 40)
        if c3 < best_c:
            best_h, best_c, n_opt = h3, c3, n_opt
    if time.time() < deadline - 80:
        n_cur = n_opt
        for restart in range(5):
            if time.time() > deadline - 40:
                break
            h_r = best_h + rng.normal(0, 0.05, len(best_h))
            h_r = h_to_sym(h_r, len(best_h))
            h_r = np.clip(h_r, 0, 1)
            h_r, _ = cd_sym(h_r, len(best_h), 30, deadline, rng, 30)
            cr = compute_c5(h_r, len(best_h))
            if cr < best_c:
                best_h, best_c, n_opt = h_r, cr, len(best_h)
    if time.time() < deadline - 30:
        h_f, c_f = slsqp_sym(best_h, n_opt, [10000, 30000], 300, deadline)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        hp, cp = cd_sym(best_h, n_opt, 100, deadline, rng, 45)
        if cp < best_c:
            best_h, best_c = hp, cp
    best_h = h_to_sym(best_h, len(best_h))
    best_h = np.clip(best_h, 0, 1)
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
