    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(700, 0.5)
        n = 700
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Intensive CD with corrected diagonal (forward order)
    if time.time() < deadline - 350:
        h1, c1 = coord_descent(best_h, n, 100, deadline - 350, gs_iters=70)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: CD with reverse order to break symmetry
    if time.time() < deadline - 300:
        h2, c2 = coord_descent(best_h, n, 60, deadline - 300, gs_iters=65,
                               ordering=np.arange(n - 1, -1, -1))
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: SLSQP on smooth objective
    if time.time() < deadline - 250:
        h3, c3 = slsqp_optimize(best_h, n, [5000, 20000, 80000], 400)
        if c3 < best_c:
            best_h, best_c = h3, c3
        h3, c3 = coord_descent(best_h, n, 50, deadline - 250, gs_iters=70)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Multi-restart with diverse perturbations
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 80
        n_restarts = max(8, int(remaining / 35))
        for r in range(n_restarts):
            if time.time() > deadline - 80:
                break
            if r % 6 == 0:
                scale = 0.015 + 0.04 * (r / max(1, n_restarts - 1))
                hp = project_feasible(best_h + rng.normal(0, scale, n_cur), n_cur)
            elif r % 6 == 1:
                x = np.linspace(0, 2, n_cur)
                freqs = rng.choice([3, 5, 7, 11, 13, 17], size=3, replace=False)
                pert = np.zeros(n_cur)
                for f in freqs:
                    pert += rng.uniform(0.01, 0.05) * np.sin(np.pi * f * x + rng.uniform(0, 2 * np.pi))
                hp = project_feasible(best_h + pert, n_cur)
            elif r % 6 == 2:
                block = n_cur // 6
                start = rng.randint(0, max(1, n_cur - block))
                hp = best_h.copy()
                hp[start:start + block] = 1.0 - hp[start:start + block]
                hp = project_feasible(hp, n_cur)
            elif r % 6 == 3:
                b = n_cur // 5
                s1 = rng.randint(0, max(1, n_cur - 2 * b))
                s2 = (s1 + b + rng.randint(1, max(1, n_cur - 2 * b))) % (n_cur - b)
                hp = best_h.copy()
                tmp = hp[s1:s1 + b].copy()
                hp[s1:s1 + b] = hp[s2:s2 + b]
                hp[s2:s2 + b] = tmp
                hp = project_feasible(hp, n_cur)
            elif r % 6 == 4:
                hp = project_feasible(best_h * (1 + 0.025 * rng.randn(n_cur)), n_cur)
            else:
                # Random coordinate swap perturbation
                k = max(1, n_cur // 20)
                hp = best_h.copy()
                for _ in range(k):
                    i, j = rng.choice(n_cur, 2, replace=False)
                    hp[i], hp[j] = hp[j], hp[i]
                hp = project_feasible(hp, n_cur)
            h_o, c_o = coord_descent(hp, n_cur, 40, deadline - 80, gs_iters=60)
            if c_o < best_c:
                h_o, c_o = slsqp_optimize(h_o, n_cur, [10000, 50000], 200)
                h_o, c_o = coord_descent(h_o, n_cur, 30, deadline - 80, gs_iters=65)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 5: Try different resolutions
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 550, 800, 1000, 1200]:
        if time.time() > deadline - 60:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = coord_descent(h_int, n_try, 50, deadline - 60, gs_iters=65)
        if c_o < best_c:
            h_o, c_o = slsqp_optimize(h_o, n_try, [20000], 200)
            h_o, c_o = coord_descent(h_o, n_try, 30, deadline - 60, gs_iters=65)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = np.linspace(0, 2, n_try)

    # Phase 6: Final polish with multiple CD orderings
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        h_f, c_f = coord_descent(best_h, n_cur, 80, deadline - 10, gs_iters=75)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = coord_descent(best_h, n_cur, 40, deadline - 10, gs_iters=70,
                                 ordering=np.arange(n_cur - 1, -1, -1))
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = slsqp_optimize(best_h, n_cur, [50000, 200000], 300)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = coord_descent(best_h, n_cur, 40, deadline - 10, gs_iters=75)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
