def cd_pass(h, n, order, c):
    """One pass of coordinate descent, returns (h, c, improved)."""
    imp = False
    for p in order:
        h0 = h[p]
        if h0 < 1e-12 or h0 > 1.0 - 1e-12:
            continue
        xn = solve_1d(c, h, p, n)
        if abs(xn - h0) > 1e-12:
            d = xn - h0
            m0 = n - 1 - p
            m1 = 2 * n - 2 - p
            ms = np.arange(m0, m1 + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb): a[vb] -= h[jb[vb]]
            di = np.where(ms == (n - 1))[0]
            c[ms] += d * a
            if len(di) > 0: c[ms[di[0]]] -= d * d
            h[p] = xn
            imp = True
    return imp

def exact_cd(h, n, max_passes, deadline, rng=None, n_random=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline: break
        if rng is not None and (n_random is None or it >= n_random):
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        imp = cd_pass(h, n, order, c)
        if not imp: break
        h = project_feasible(h, n)
        if abs(h.sum() - n/2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def slsqp(h, n, alpha, maxiter, deadline):
    if time.time() > deadline: return h, compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        old_c = compute_c5(h, n)
        if new_c < old_c:
            return new_h, new_c
    except:
        pass
    return h, compute_c5(h, n)

**New idea: Direct C-targeting with active set method**

The idea is:
1. Start with current best h and compute c = correlate(h, 1-h)
2. Identify the "active" shifts (those where c[k]*dx is close to the maximum)
3. Try to reduce only the active shifts while keeping inactive ones from increasing
4. This is like an active set LP approach

More specifically:
- Let M = c.max() * dx (current C₅)
- Active set: A = {k : c[k] * dx > M - ε} for small ε
- We want to find δh such that for k in A: (c + δc)_k decreases, and for k not in A: (c + δc)_k doesn't increase too much
- δc is bilinear in h and δh, but for small δh, it's approximately linear: δc[k] ≈ ∑ h[j]δh[j+k] + (1-h[j+k])δh[j] (approximately)

Actually, the derivative of c[k] with respect to h[p] is:
- If p is in the forward part: dc[k]/dh[p] = (1 - h[p - k + (n-1)]) for valid indices
- If p is in the backward part: dc[k]/dh[p] = -h[p + k - (n-1)] for valid indices
- At the diagonal: dc[k]/dh[p] = 1 - 2h[p]

So the gradient of c[k] with respect to h is a sparse vector. We can compute the gradient of the max (or smooth max) and do gradient descent.

But this is essentially what the smooth SLSQP is already doing. The issue is that SLSQP with SLSQP's maxiter might not converge well.

Actually, let me try a different approach: **minimize the sum of the top-k overlaps** instead of the max. This gives a smoother objective that's easier to optimize.

The feasibility problem is:

This is a non-convex feasibility problem (bilinear constraints). But we can try:
- Linearize around current h: c[k] + ∇c[k]·δh ≤ C/dx
- Solve the LP
- Check if the solution actually satisfies the nonlinear constraints
- If not, update linearization and repeat

This is a **successive linear programming** approach.

def slp_feasibility(h, n, C_target, max_iter, deadline):
    """Successive linear programming to find h with C₅ ≤ C_target."""
    dx = 2.0 / n
    C_max = C_target / dx  # max allowed correlation value
    h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline: return h, False
        c = np.correlate(h, 1.0 - h, mode='full')
        # Check if already feasible
        if c.max() <= C_max + 1e-10:
            return h, True
        # Build LP approximation
        # c[k] + sum_p dc[k]/dh[p] * (h_new[p] - h[p]) <= C_max for all k
        # This is: sum_p dc[k]/dh[p] * h_new[p] <= C_max - c[k] + sum_p dc[k]/dh[p] * h[p]
        # But sum_p dc[k]/dh[p] * h[p] is not simply related to c[k]...
        # Actually, by Euler's theorem for homogeneous functions... but c is bilinear, not linear.
        # Let's just do: c_approx[k] = c[k] + grad_c_k . (h_new - h)
        # where grad_c_k = dc[k]/dh evaluated at h
        # So constraint: c[k] + grad_c_k . (h_new - h) <= C_max
        # i.e., grad_c_k . h_new <= C_max - c[k] + grad_c_k . h
        
        # Compute gradients efficiently
        # dc[k]/dh[p] for each k, p
        # This is a (2n-1) x n matrix, which is too large for n=700 (1399 x 700 = ~1M entries)
        # But it's sparse! Each row has at most 2 non-zero entries per p... wait no.
        # For a given k, dc[k]/dh[p] is non-zero when:
        #   p + k - (n-1) is valid (forward) OR p - k + (n-1) is valid (backward)
        # For a given p, dc[k]/dh[p] is non-zero for k in [n-1-p, 2n-2-p]
        
        # Actually, let's think of it the other way. For the LP, we need the constraint matrix.
        # Let G be the (2n-1) x n matrix where G[k, p] = dc[k]/dh[p] at current h.
        # Constraint: G @ h_new <= C_max - c + G @ h
        # Additional: sum(h_new) = n/2, 0 <= h_new <= 1
        
        # For n=700, G has 1399 rows and 700 cols = ~1M entries. Not too bad.
        # But building G explicitly is O(n^2) which is ~500K operations. Fine.
        
        # Let's build G
        G = np.zeros((2*n-1, n))
        for p in range(n):
            m0 = n - 1 - p
            m1 = 2 * n - 2 - p
            ms = np.arange(m0, m1 + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                G[ms[vf], p] += (1.0 - h[jf[vf]])
            if np.any(vb):
                G[ms[vb], p] -= h[jb[vb]]
            # Diagonal correction
            di = ms == (n - 1)
            if np.any(di):
                G[ms[di], p] = (1.0 - 2.0 * h[p])
        
        # RHS for LP constraints
        c_cur = np.correlate(h, 1.0 - h, mode='full')
        rhs = C_max - c_cur + G @ h
        
        # Solve LP: minimize 0 subject to G @ x <= rhs, sum(x) = n/2, 0 <= x <= 1
        # Use scipy linprog
        from scipy.optimize import linprog
        try:
            result = linprog(
                c=np.zeros(n),
                A_ub=G, b_ub=rhs,
                A_eq=np.ones((1, n)), b_eq=np.array([n/2.0]),
                bounds=[(0, 1)] * n,
                method='highs'
            )
            if result.success:
                h_new = project_feasible(result.x, n)
                c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
                if c_new.max() < c_cur.max():
                    h = h_new
                else:
                    # SLP didn't help, damp
                    h = 0.5 * h + 0.5 * result.x
                    h = project_feasible(h, n)
            else:
                break
        except:
            break
    return h, c.max() <= C_max + 1e-10

        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if (xlo < r1 and xhi > r2) or (xlo >= r1 and xhi <= r2):
                if h0 - r1 <= r2 - h0: xhi = min(xhi, r1)
                else: xlo = max(xlo, r2)
    if xlo > xhi: return h0
    return float(np.clip((xlo + xhi) / 2.0, 0.0, 1.0))

def cd_update(h, c, p, n, xn):
    d = xn - h[p]
    m0 = n - 1 - p
    m1 = 2 * n - 2 - p
    ms = np.arange(m0, m1 + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb): a[vb] -= h[jb[vb]]
    di = np.where(ms == (n - 1))[0]
    c[ms] += d * a
    if len(di) > 0: c[ms[di[0]]] -= d * d
    h[p] = xn

def exact_cd(h, n, max_passes, deadline, rng=None, rand_from=0):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline: break
        if rng is not None and it >= rand_from:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        imp = False
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12: continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                cd_update(h, c, p, n, xn)
                imp = True
        if not imp: break
        h = project_feasible(h, n)
        if abs(h.sum() - n/2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def build_gradient_matrix(h, n):
    G = np.zeros((2*n-1, n))
    for p in range(n):
        m0 = n - 1 - p
        m1 = 2 * n - 2 - p
        ms = np.arange(m0, m1 + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        if np.any(vf): G[ms[vf], p] += (1.0 - h[jf[vf]])
        if np.any(vb): G[ms[vb], p] -= h[jb[vb]]
        di = ms == (n - 1)
        if np.any(di): G[ms[di], p] = (1.0 - 2.0 * h[p])
    return G

def slp_feasibility(h_start, n, C_target, max_iter, deadline, step_frac=0.3):
    dx = 2.0 / n
    C_max = C_target / dx
    h = h_start.copy()
    for it in range(max_iter):
        if time.time() > deadline: break
        c = np.correlate(h, 1.0 - h, mode='full')
        if c.max() <= C_max + 1e-10:
            return h, True
        G = build_gradient_matrix(h, n)
        rhs = C_max - c + G @ h
        try:
            res = linprog(c=np.zeros(n), A_ub=G, b_ub=rhs,
                          A_eq=np.ones((1, n)), b_eq=np.array([n/2.0]),
                          bounds=[(0.0, 1.0)]*n, method='highs')
            if not res.success: break
            h_lp = res.x
            h_new = h + step_frac * (h_lp - h)
            h_new = project_feasible(h_new, n)
            c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            if c_new.max() < c.max():
                h = h_new
            else:
                step_frac *= 0.5
                if step_frac < 0.01: break
        except:
            break
    c_final = np.correlate(h, 1.0 - h, mode='full')
    return h, c_final.max() <= C_max + 1e-8

def slsqp(h, n, alpha, maxiter, deadline):
    if time.time() > deadline: return h, compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    old_c = compute_c5(h, n)
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < old_c: return new_h, new_c
    except: pass
    return h, old_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: CD polish on initial
    h_best, best_c = exact_cd(h_best, n, 100, deadline, rng, 30)

    # Phase 2: SLP targeting progressively lower C
    for red in [0.00003, 0.00006, 0.0001, 0.00015, 0.0002, 0.0003, 0.0005]:
        if time.time() > deadline - 150: break
        C_t = best_c - red
        if C_t < 0.36: break
        h_s, ok = slp_feasibility(h_best, n, C_t, 60, deadline)
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            h_best, best_c = h_s, c_s
            h_best, best_c = exact_cd(h_best, n, 50, deadline, rng, 5)

    # Phase 3: SLSQP at multiple smoothness
    for alpha in [8000, 30000, 100000, 400000]:
        if time.time() > deadline - 200: break
        h2, c2 = slsqp(h_best, n, alpha, 200, deadline)
        if c2 < best_c: h_best, best_c = h2, c2
        h3, c3 = exact_cd(h_best, n, 80, deadline, rng, 10)
        if c3 < best_c: h_best, best_c = h3, c3

    # Phase 4: SLP again after SLSQP
    for red in [0.00005, 0.0001, 0.0002]:
        if time.time() > deadline - 150: break
        C_t = best_c - red
        h_s, _ = slp_feasibility(h_best, n, C_t, 40, deadline)
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            h_best, best_c = h_s, c_s
            h_best, best_c = exact_cd(h_best, n, 50, deadline, rng, 5)

    # Phase 5: Multi-n exploration
    n_base = len(h_best)
    for n_try in [n_base - 100, n_base + 100, 400, 600, 800]:
        if time.time() > deadline - 200 or n_try <= 100: continue
        if n_try == n_base: continue
        x_old = np.linspace(0, 2, n_base)
        x_new = np.linspace(0, 2, n_try)
        h_t = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_t, _ = slsqp(h_t, n_try, 30000, 100, deadline)
        h_t, c_t = exact_cd(h_t, n_try, 100, deadline, rng, 20)
        if c_t < best_c - 1e-7:
            h_best, best_c, n_base = h_t, c_t, n_try

    # Phase 6: Random restarts
    for _ in range(15):
        if time.time() > deadline - 60: break
        ns = 0.01 + 0.04 * rng.rand()
        h_r = project_feasible(h_best + rng.normal(0, ns, n_base), n_base)
        h_r, c_r = exact_cd(h_r, n_base, 80, deadline, rng, 5)
        if c_r < best_c:
            h_best, best_c = h_r, c_r
            # Try SLP from new best
            C_t = best_c - 0.0001
            h_s, _ = slp_feasibility(h_best, n_base, C_t, 30, deadline)
            c_s = compute_c5(h_s, n_base)
            if c_s < best_c:
                h_best, best_c = h_s, c_s

    # Phase 7: Final SLP + CD polish
    if time.time() < deadline - 20:
        C_t = best_c - 0.0001
        h_s, _ = slp_feasibility(h_best, n_base, C_t, 40, deadline)
        c_s = compute_c5(h_s, n_base)
        if c_s < best_c: h_best, best_c = h_s, c_s
        h_best, best_c = exact_cd(h_best, n_base, 100, deadline, rng, 10)

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
