```python
def lp_step(h, n, top_lags, t_current):
    """Solve LP relaxation: minimize t s.t. linearized c_k ≤ t for k in top_lags."""
    from scipy.optimize import linprog
    # Variables: h[0], ..., h[n-1], t
    # Objective: min t
    # Constraints:
    #   For each k in top_lags: sum_p (dc_k/dh_p) * h_p + c_k(h) - t ≤ 0  (linearized)
    #   0 ≤ h_p ≤ 1
    #   sum h_p = n/2
    
    c = np.correlate(h, 1.0 - h, mode='full')
    
    # Build constraint matrix
    m = len(top_lags)
    A_ub = np.zeros((m + n + 1, n + 1))  # +1 for t, +n for h≤1
    b_ub = np.zeros(m + n + 1)
    
    # Linearized: c_k(h) ≈ c_k(h*) + sum_p grad_k[p] * (h_p - h*_p) ≤ t
    # => sum_p grad_k[p] * h_p - t ≤ c_k(h*) - c_k(h*)_linear_part = 0 (by construction)
    # Actually: c_k(h) ≈ c_k(h*) + sum_p grad_k[p] * (h_p - h*_p)
    # We want: c_k(h*) + sum_p grad_k[p] * (h_p - h*_p) ≤ t
    # => sum_p grad_k[p] * h_p - t ≤ c_k(h*) - sum_p grad_k[p] * h*_p
    
    for i, k in enumerate(top_lags):
        shift = k - (n - 1)
        grad = np.zeros(n)
        for p in range(n):
            # Forward term: h[p] * (1-h[p+shift])
            pf = p + shift
            if 0 <= pf < n:
                grad[p] += (1.0 - h[pf])
            # Backward term: (1-h[p]) * h[p-shift] contributes to h[p] as -h[p-shift]
            pb = p - shift
            if 0 <= pb < n:
                grad[p] -= h[pb]
        A_ub[i, :n] = grad
        A_ub[i, n] = -1.0
        rhs = c[k] - np.dot(grad, h)
        b_ub[i] = rhs
    
    # h_p ≤ 1
    for p in range(n):
        A_ub[m + p, p] = 1.0
        b_ub[m + p] = 1.0
    
    # sum(h) = n/2 (as equality, need to handle differently)
    # linprog doesn't handle equality constraints well with the default method
    # Let's use the 'highs' method which supports both
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    bounds_list = [(0.0, 1.0)] * n + [(0.0, 2.0)]  # t can be anything positive
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_list, method='highs')
    if res.success:
        new_h = res.x[:n]
        return project_feasible(new_h, n)
    return h
```

    m = len(top_lags)

    # c[k] = sum_i h[i] * (1 - h[i + k - n + 1])
    # dc[k]/dh[p] = (1 - h[p + k - n + 1]) - h[p - (k - n + 1)]

    b_ub = np.zeros(m + n)

        for p in range(n):
            g = 0.0
            pf = p + shift
            if 0 <= pf < n:
                g += (1.0 - h[pf])
            pb = p - shift
            if 0 <= pb < n:

        b_ub[ii] = c[k] - np.dot(A_ub[ii, :n], h)
    
    for p in range(n):
        A_ub[m + p, p] = 1.0
        b_ub[m + p] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(0.0, 1.0)] * n + [(0.0, None)], method='highs')
    if res.success:
        return project_feasible(res.x[:n], n), res.x[n]
    return h, t_val

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _grad_c(h, n, k):
    """Gradient of c[k] w.r.t. h."""
    shift = k - (n - 1)
    g = np.zeros(n)
    pf = shift  # forward: h[p] * (1-h[p+shift])
    for p in range(n):
        g[p] = 0.0
        f = p + shift
        if 0 <= f < n:
            g[p] += (1.0 - h[f])
        b = p - shift
        if 0 <= b < n:
            g[p] -= h[b]
    return g

def lp_step(h, n, top_lags):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = len(top_lags)
    A_ub = np.zeros((m + n, n + 1))
    b_ub = np.zeros(m + n)
    for ii, k in enumerate(top_lags):
        g = _grad_c(h, n, k)
        A_ub[ii, :n] = g
        A_ub[ii, n] = -1.0
        b_ub[ii] = c[k] - np.dot(g, h)
    for p in range(n):
        A_ub[m+p, p] = 1.0
        b_ub[m+p] = 1.0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0

                      bounds=[(0.0,1.0)]*n + [(0.0,None)], method='highs',
                      options={'presolve': True})
        if res.success:
            return project_feasible(res.x[:n], n)
    except Exception:
        pass
    return h

    for it in range(max_iters):

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2*n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            diag_pos = np.where(ms == (n-1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff = max(unaff, float(c[m_hi+1:].max()))
            
            def fval(d):
                vals = c_aff + d * a
                if has_diag:
                    vals[diag_idx] = c_aff[diag_idx] + d*(1.0 - h0 - d)
                return max(float(np.max(vals)), unaff)
            
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr*(d_hi - d_lo)
            d2 = d_lo + gr*(d_hi - d_lo)
            f1, f2 = fval(d1), fval(d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-14:

                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr*(d_hi - d_lo)
                    f1 = fval(d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr*(d_hi - d_lo)
                    f2 = fval(d2)
            d_opt = (d_lo + d_hi) / 2.0
            if fval(d_opt) < fval(0.0) - 1e-13:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    return h, min(best_c, compute_c5(h, n))

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _grad_c_vec(h, n, k):
    shift = k - (n - 1)
    idx = np.arange(n)
    g = np.zeros(n)
    pf = idx + shift
    vf = (pf >= 0) & (pf < n)
    g[vf] += (1.0 - h[pf[vf]])
    pb = idx - shift
    vb = (pb >= 0) & (pb < n)
    g[vb] -= h[pb[vb]]
    return g

def lp_step(h, n, top_lags):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = len(top_lags)
    A_ub = np.zeros((m + n, n + 1))
    b_ub = np.zeros(m + n)
    for ii, k in enumerate(top_lags):
        g = _grad_c_vec(h, n, int(k))
        A_ub[ii, :n] = g
        A_ub[ii, n] = -1.0
        b_ub[ii] = c[int(k)] - np.dot(g, h)
    for p in range(n):
        A_ub[m + p, p] = 1.0
        b_ub[m + p] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0.0, 1.0)] * n + [(0.0, None)], method='highs',
                      options={'presolve': True})
        if res.success:
            return project_feasible(res.x[:n], n)
    except Exception:
        pass
    return h

def lp_optimize(h, n, top_k, max_iters, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_lags = np.argsort(c)[::-1][:top_k]
        new_h = lp_step(best_h, n, top_lags)
        new_c = compute_c5(new_h, n)
        if new_c < best_c - 1e-15:
            best_h, best_c = new_h, new_c
            stall = 0
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1, 0.05]:
                trial = project_feasible(best_h + frac * (new_h - best_h), n)
                tc = compute_c5(trial, n)
                if tc < best_c - 1e-15:
                    best_h, best_c = trial, tc
                    improved = True
                    stall = 0
                    break
            if not improved:
                stall += 1
                if stall > 30:
                    break
    return best_h, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-14:
                    break
                if f1 < f2:
                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d1)
                else:
                    d_lo = d1; d1 = d2; f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d_opt)
            f_cur = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, 0.0)
            if f_opt < f_cur - 1e-13:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    final_c = compute_c5(h, n)
    return h, min(best_c, final_c)

def block_cd2(h, n, max_passes, deadline):

    best_c = compute_c5(h, n)
    for _ in range(max_passes):

        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(0, n - 1, 2):

                f1 = compute_c5(h, n)
