```
def proj(x,n):
    x=np.asarray(x,dtype=float)
    T=0.5*n
    y=np.clip(x,0.0,1.0)
    diff=T-y.sum()
    if abs(diff)>1e-8:
       lam_lo=-float(np.max(x))-2.0
       lam_hi=1.0-float(np.min(x))+2.0
       for _ in range(60):
          lam=0.5*(lam_lo+lam_hi)
          s=np.sum(np.clip(x+lam,0.0,1.0))
          if s<T: lam_lo=lam
          else: lam_hi=lam
       y=np.clip(x+0.5*(lam_lo+lam_hi),0.0,1.0)
       diff=T-y.sum()
    if abs(diff)>1e-12:
       # proportional up to 1000
       for _ in range(1000):
          diff=T-y.sum()
          if abs(diff)<=1e-13: break
          if diff>0:
             m=y<1.0-1e-12
             c=int(m.sum());
             if c==0: break
             add=diff/c
             y[m]=np.minimum(1.0,y[m]+add)
          else:
             m=y>1e-12
             c=int(m.sum())
             if c==0: break
             sub=diff/c
             y[m]=np.maximum(0.0,y[m]+sub)
    diff=T-y.sum()
    if abs(diff)>1e-13:
       if diff>0:
          idx=np.where(y<1.0-1e-10)[0]
          if idx.size: i=idx[-1]; y[i]=min(1.0, y[i]+diff)
       else:
          idx=np.where(y>1e-10)[0]
          if idx.size: i=idx[-1]; y[i]=max(0.0, y[i]+diff)
    return y
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def proj(x, n):
    y = np.clip(x, 0.0, 1.0)
    T = 0.5 * n
    for _ in range(10000):
        diff = T - y.sum()
        if abs(diff) <= 1e-12:
            break
        if diff > 0.0:
            m = y < 1.0 - 1e-15
            c = int(m.sum())
            if c == 0:
                break
            y[m] = np.minimum(1.0, y[m] + diff / c)
        else:
            m = y > 1e-15
            c = int(m.sum())
            if c == 0:
                break
            y[m] = np.maximum(0.0, y[m] + diff / c)
    diff = T - y.sum()
    if abs(diff) > 1e-12:
        if diff > 0.0:
            idx = np.where(y < 1.0 - 1e-12)[0]
            if idx.size:
                y[idx[-1]] = float(np.clip(y[idx[-1]] + diff, 0.0, 1.0))
        else:
            idx = np.where(y > 1e-12)[0]
            if idx.size:
                y[idx[-1]] = float(np.clip(y[idx[-1]] + diff, 0.0, 1.0))
    return y

def calc_c(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_val(h, n):
    return float(calc_c(h).max() * (2.0 / n))

def smooth_count(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def smooth_obj(x, alpha):
    return smooth_count(calc_c(x), alpha)

def smooth_jac(x, alpha):
    n = len(x)
    c = calc_c(x)
    m = float(np.max(c))
    s = alpha * (c - m)
    w = np.exp(s)
    z = float(np.sum(w))
    if z > 0.0:
        w /= z
    else:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    g1 = np.convolve(1.0 - x, w[::-1], mode='full')[n - 1:n - 1 + n]
    g2 = np.convolve(x, w, mode='full')[n - 1:n - 1 + n]
    return g1 - g2

def sum_con(x):
    return float(x.sum() - 0.5 * len(x))

def sum_jac(x):
    return np.ones(len(x))

def pg_refine(h, n, deadline, alphas, maxit, seed=1):
    rng = np.random.RandomState(seed)
    best = np.array(h, dtype=float).copy()
    n = len(best)
    bc = c5_val(best, n)
    for alpha in alphas:
        cur = best.copy()
        cc = calc_c(cur)
        cur_sm = smooth_count(cc, alpha)
        best_sm = cur_sm
        cb = float(cc.max() * (2.0 / n))
        step = 0.018
        stall = 0
        for _ in range(maxit):
            if time.time() > deadline:
                break
            g = smooth_jac(cur, alpha)
            g = g - g.mean()
            norm = float(np.linalg.norm(g))
            if norm < 1e-12:
                break
            g = g / norm
            improved = False
            for f in (1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.01):
                cand = proj(cur - step * f * g, n)
                cc2 = calc_c(cand)
                sm2 = smooth_count(cc2, alpha)
                val2 = float(cc2.max() * (2.0 / n))
                if (sm2 < cur_sm - 1e-9 and val2 <= bc + 1e-4) or val2 < bc - 1e-10:
                    cur = cand
                    cur_sm = sm2
                    improved = True
                    if val2 < bc:
                        bc = val2
                        best = cand.copy()
                    if sm2 < best_sm and val2 <= bc + 1e-7:
                        best_sm = sm2
                    break
            if improved:
                step = min(0.2, step * 1.08)
                stall = 0
            else:
                step *= 0.5
                stall += 1
                if stall > 6:
                    cur = proj(cur + rng.normal(0.0, 0.02, n), n)
                    cc = calc_c(cur)
                    cur_sm = smooth_count(cc, alpha)
                    step = 0.018
                    stall = 0
                    if cur_sm < best_sm:
                        best_sm = cur_sm
                    if c5_val(cur, n) < bc:
                        bc = c5_val(cur, n)
                        best = cur.copy()
    cc = calc_c(best)
    return proj(best, n), float(min(bc, cc.max() * (2.0 / n)))

def slsqp_polish(h, n, deadline, alphas):
    best = np.array(h, dtype=float).copy()
    n = len(best)
    bc = c5_val(best, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
            res = minimize(smooth_obj, best, args=(alpha,), jac=smooth_jac,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 25, 'ftol': 1e-12})
            if hasattr(res, 'x') and res.x is not None:
                y = proj(res.x, n)
                v = c5_val(y, n)
                if v < bc - 1e-10:
                    best, bc = y, v
        except Exception:
            pass
    return best, bc

def binary_from(h, n):
    a = np.zeros(n, dtype=float)
    if n % 2:
        n = n - 1
    t = n // 2
    if t <= 0:
        return a[:n]
    if t >= n:
        a[:] = 1.0
        return a
    idx = np.argsort(-np.asarray(h, dtype=float))[:t]
    a[np.asarray(idx)] = 1.0
    return a

def binary_random(n, rng):
    a = np.zeros(n, dtype=float)
    t = n // 2
    if t > 0:
        a[rng.choice(n, t, replace=False)] = 1.0
    return a

def apply_flip(c, a, p, new):
    old = float(a[p])
    delta = float(new) - old
    if abs(delta) < 1e-15:
        return
    n = len(a)
    s = n - 1 - int(p)
    c[s:s + n] += delta * (1.0 - a)
    c[int(p):int(p) + n] += -delta * a[::-1]
    c[n - 1] -= delta * delta
    a[int(p)] = float(new)

def binary_cost(c):
    return float(c.max()) * 100.0 + float(np.dot(c, c)) * 1e-6

def opp_pairs(a, k):
    n = len(a)
    d = n - 1 - int(k)
    if d >= 0:
        if d >= n:
            return np.empty(0), np.empty(0)
        valid = (a[:n - d] < 0.5) & (a[d:] > 0.5)
        idx = np.nonzero(valid)[0]
        return d + idx, idx
    else:
        ad = -d
        if ad >= n:
            return np.empty(0), np.empty(0)
        valid = (a[ad:] < 0.5) & (a[:n - ad] > 0.5)
        idx = np.nonzero(valid)[0]
        return idx, ad + idx

def local_binary(a, n, deadline, seed=1):
    rng = np.random.RandomState(seed)
    cur_a = np.array(a, dtype=float).copy()
    cur_c = calc_c(cur_a)
    best_a = cur_a.copy()
    best_c = cur_c.copy()
    best_cost = binary_cost(best_c)
    cur_cost = best_cost
    no_improve = 0
    for _ in range(100000):
        if time.time() > deadline:
            break
        L = len(cur_c)
        top = min(12, L)
        if top < L:
            order = np.argpartition(cur_c, -top)[-top:]
        else:
            order = np.arange(L)
        order = order[np.argsort(cur_c[order])[::-1]]
        improved = False
        for kk in order:
            p_arr, q_arr = opp_pairs(cur_a, int(kk))
            if p_arr.size == 0:
                continue
            if p_arr.size > 140:
                sel = rng.choice(p_arr.size, 140, replace=False)
                p_arr = p_arr[sel]
                q_arr = q_arr[sel]
            for ii in range(p_arr.size):
                p = int(p_arr[ii])
                q = int(q_arr[ii])
                if p == q or cur_a[p] <= 0.5 or cur_a[q] >= 0.5:
                    continue
                a2 = cur_a.copy()
                c2 = cur_c.copy()
                apply_flip(c2, a2, p, 0.0)
                apply_flip(c2, a2, q, 1.0)
                cost = binary_cost(c2)
                if cost < best_cost - 1e-8:
                    best_a, best_c, best_cost = a2, c2, cost
                if cost < cur_cost - 1e-8:
                    cur_a, cur_c, cur_cost = a2, c2, cost
                    improved = True
                    break
            if improved:
                break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                cur_a = best_a.copy()
                cur_c = best_c.copy()
                cur_cost = best_cost
                for _ in range(4):
                    ones = np.nonzero(cur_a > 0.5)[0]
                    zeros = np.nonzero(cur_a < 0.5)[0]
                    if not ones.size or not zeros.size:
                        break
                    p = int(ones[rng.randint(ones.size)])
                    q = int(zeros[rng.randint(zeros.size)])
                    if p == q:
                        continue
                    a2 = cur_a.copy()
                    c2 = cur_c.copy()
                    apply_flip(c2, a2, p, 0.0)
                    apply_flip(c2, a2, q, 1.0)
                    cur_a, cur_c = a2, c2
                    cur_cost = binary_cost(c2)
                no_improve = 0
    return best_a, float(best_c.max() * (2.0 / n))

def interp_h(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return np.array(h, dtype=float).copy()
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return proj(np.interp(xn, xo, np.asarray(h, dtype=float)), n_new)

def initial_plain(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    y = (0.55 * np.sin(2.0 * np.pi * 2.0 * x + rng.rand() * 6.283)
         + 0.35 * np.sin(2.0 * np.pi * 5.0 * x + rng.rand() * 6.283)
         + 0.18 * (rng.rand(n) - 0.5))
    return proj(0.5 + 0.38 * np.tanh(y), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s) - 4.0)
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
        if n > 950:
            h = interp_h(h, 850)
            n = len(h)
    except Exception:
        n = 700
        h = initial_plain(n, seed)
    h = proj(h, n)
    best_h = h.copy()
    best_n = n
    best_c = c5_val(best_h, best_n)

    remain = deadline - t0
    t1 = max(t0, min(deadline - 1.0, t0 + max(5.0, min(180.0, 0.25 * remain))))
    if t1 > t0 + 1.0:
        h1, c1 = pg_refine(best_h, best_n, t1, (1.0, 10.0, 100.0, 5000.0), 350, seed + 1)
        if c1 < best_c:
            best_h, best_c, best_n = h1, c1, len(h1)

    t2 = max(t1, min(deadline - 1.0, t1 + max(5.0, min(140.0, 0.15 * remain))))
    if t2 > t1 + 1.0:
        h2, c2 = slsqp_polish(best_h, best_n, t2, (10.0, 200.0, 5000.0))
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, len(h2)

    t3 = max(t2, min(deadline - 15.0, t2 + max(10.0, min(320.0, 0.55 * remain))))
    if best_n % 2 == 0 and t3 > t2 + 2.0:
        ab, cb = local_binary(binary_from(best_h, best_n), best_n, t3, seed + 70)
        if cb < best_c:
            best_h, best_c, best_n = ab, cb, len(ab)
        if time.time() < t3 - 20.0 and best_c > 0.3807:
            hp = proj(ab, best_n)
            h3, c3 = pg_refine(hp, best_n, t3 - 20.0, (100.0, 5000.0), 120, seed + 80)
            if c3 < best_c:
                best_h, best_c, best_n = h3, c3, len(h3)

    small_ns = [650, 600, 550, 500, 450, 400, 350, 300, 256, 200]
    for nn in small_ns:
        if time.time() > deadline - 20.0:
            break
        if nn % 2 or nn == best_n:
            continue
        slot = min(deadline - 8.0, time.time() + 70.0)
        hi = interp_h(best_h, nn)
        if slot - time.time() > 35.0:
            hi2, ci2 = pg_refine(hi, nn, slot - 15.0, (10.0, 500.0, 10000.0), 150, seed + 1000 + nn)
            if ci2 < best_c:
                best_h, best_c, best_n = hi2, ci2, nn
        start_h = best_h if best_n == nn else interp_h(best_h, nn)
        ab = binary_from(start_h, nn)
        ab2, cb2 = local_binary(ab, nn, min(deadline - 8.0, time.time() + 45.0), seed + 200000 + nn)
        if cb2 < best_c:
            best_h, best_c, best_n = ab2, cb2, nn
        rcount = 2 if nn <= 350 else 1
        for rs in range(rcount):
            if time.time() > deadline - 12.0:
                break
            ar = binary_random(nn, rng)
            ar2, cr2 = local_binary(ar, nn, min(deadline - 8.0, time.time() + 22.0),
                                    seed + 300000 + 3 * nn + rs)
            if cr2 < best_c:
                best_h, best_c, best_n = ar2, cr2, nn

    best_h = proj(best_h, best_n)
    for _ in range(5):
        diff = 0.5 * best_n - best_h.sum()
        if abs(diff) <= 1e-14:
            break
        m = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        if not np.any(m):
            break
        i = int(np.nonzero(m)[0][-1])
        best_h[i] = float(np.clip(best_h[i] + diff, 0.0, 1.0))
    best_c = c5_val(best_h, best_n)
    return best_h, float(best_c), int(best_n)
```
