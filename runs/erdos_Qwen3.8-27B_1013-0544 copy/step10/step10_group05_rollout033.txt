import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-14
            if not mask.any():
                break
            h[mask] += diff / h[mask].sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _cons_fun(x):
    return x.sum() - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def solve_coord(c, h, p, n):

    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = max(0, n - 1 - p)
                m_max = min(2 * n - 2, 2 * n - 2 - p)
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                diag_i = np.where(ms == (n - 1))[0]
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def slsqp_opt(h0, n, alphas, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def freq_perturb(h, n, rng, n_modes=5, amp=0.05):
    """Perturb in frequency domain for structured changes."""
    H = np.fft.rfft(h)
    pert = np.zeros_like(H)
    for _ in range(n_modes):
        k = rng.randint(1, len(H) // 2)
        pert[k] = amp * rng.normal()
        if k < len(H) - 1:
            pert[len(H) - 1 - k] = pert[k].conjugate()
    h_new = np.fft.irfft(H + pert, n=n)
    return project_feasible(h_new, n)

def block_swap(h, n, rng, block_size=10):
    """Swap two random blocks."""
    h = h.copy()
    i = rng.randint(0, n - 2 * block_size)
    j = rng.randint(0, n - 2 * block_size)
    if abs(i - j) < block_size:
        return h
    h[i:i+block_size], h[j:j+block_size] = h[j:j+block_size].copy(), h[i:i+block_size].copy()
    return project_feasible(h, n)

            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a_d = a[diag_idx[0]]
        disc = a_d * a_d + 4.0 * (c_diag - L)
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (a_d - sq) / 2.0
            r2 = (a_d + sq) / 2.0
            if x_lo <= h0 + r1:
                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = max(0, n - 1 - p)
                m_max = min(2 * n - 2, 2 * n - 2 - p)
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                diag_i = np.where(ms == (n - 1))[0]
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def lp_iterate(h, n, max_iters=30, deadline=None, active_frac=0.02):
    h = h.copy()
    for _ in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        max_c = float(c.max())
        thresh = max_c * (1.0 - active_frac)
        active = c >= thresh
        if active.sum() == 0:
            active = c >= max_c - 0.001
        if active.sum() == 0:
            break
        active_idx = np.where(active)[0]
        A_rows = []
        b_vals = []
        for m in active_idx:
            l = m - (n - 1)
            g = np.zeros(n)
            p1s, p1e = max(0, l), min(n, l + n)
            if p1e > p1s:
                g[p1s:p1e] += (1.0 - h[p1s - l:p1e - l])
            p2s, p2e = max(0, -l), min(n, n - l)
            if p2e > p2s:
                g[p2s:p2e] -= h[p2s + l:p2e + l]
            A_rows.append(g)
            b_vals.append(c[m] - g @ h)
        A_ub = np.column_stack([np.array(A_rows), -np.ones(len(A_rows))])
        b_ub = np.array(b_vals)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break
        h_lp = project_feasible(res.x[:n], n)
        c_lp = compute_c5(h_lp, n)
        if c_lp <= max_c + 1e-9:
            h = h_lp
        else:
            moved = False
            for alpha in [0.5, 0.25, 0.1, 0.05]:
                h_try = project_feasible(h + alpha * (h_lp - h), n)
                if compute_c5(h_try, n) <= max_c + 1e-8:
                    h = h_try
                    moved = True
                    break
            if not moved:
                break
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _cons_fun(x):
    return x.sum() - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def slsqp_opt(h0, n, alphas, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def freq_perturb(h, n, rng, n_modes, amp):
    H = np.fft.rfft(h)
    pert = np.zeros_like(H)
    for _ in range(n_modes):
        k = rng.randint(1, len(H) // 2)
        pert[k] = amp * rng.normal()
        if k < len(H) - 1:
            pert[len(H) - 1 - k] = pert[k].conjugate()
    return project_feasible(np.fft.irfft(H + pert, n=n), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: CD convergence
    h_best, best_c = exact_cd(h_best, n, 80, deadline)

    # Phase 2: LP linearization
    h_best, best_c = lp_iterate(h_best, n, max_iters=40, deadline=deadline, active_frac=0.02)

    # Phase 3: SLSQP polish
    h_best, best_c = slsqp_opt(h_best, n, [30000, 300000, 3000000], 300, deadline)
    h_best, best_c = exact_cd(h_best, n, 50, deadline)

    # Phase 4: LP again after SLSQP
    h_best, best_c = lp_iterate(h_best, n, max_iters=30, deadline=deadline, active_frac=0.01)
    h_best, best_c = exact_cd(h_best, n, 40, deadline)

    # Phase 5: Multi-start with perturbations
    for r in range(25):
        if time.time() > deadline - 80:
            break
        temp = 0.01 + 0.025 * (r / 8.0)
        strat = r % 4
        if strat == 0:
            hp = project_feasible(h_best + rng.normal(0, temp, n), n)
        elif strat == 1:
            hp = freq_perturb(h_best, n, rng, 3 + 2 * (r % 5), temp * 0.4)
        elif strat == 2:
            hp = h_best.copy()
            idx = rng.choice(n, size=max(1, n // 8), replace=False)
            hp[idx] = np.clip(hp[idx] + rng.normal(0, temp * 2, len(idx)), 0, 1)
            hp = project_feasible(hp, n)
        else:
            hp = h_best.copy()
            bs = 8 + 4 * (r % 4)
            i1 = rng.randint(0, n - bs)
            i2 = rng.randint(0, n - bs)
            if abs(i1 - i2) > bs:
                hp[i1:i1+bs], hp[i2:i2+bs] = hp[i2:i2+bs].copy(), hp[i1:i1+bs].copy()
                hp = project_feasible(hp, n)
            else:
                hp = project_feasible(h_best + rng.normal(0, temp, n), n)
        h_o, c_o = exact_cd(hp, n, 40, deadline)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o
            if time.time() < deadline - 150:
                h_best, best_c = lp_iterate(h_best, n, 20, deadline, active_frac=0.01)
                h_best, best_c = exact_cd(h_best, n, 30, deadline)

    # Phase 6: Final SLSQP + CD
    if time.time() < deadline - 60:
        h_best, best_c = slsqp_opt(h_best, n, [500000, 5000000, 50000000], 200, deadline)
        h_best, best_c = exact_cd(h_best, n, 60, deadline)

    # Phase 7: Final LP + CD
    if time.time() < deadline - 20:
        h_best, best_c = lp_iterate(h_best, n, 20, deadline, active_frac=0.005)
        h_best, best_c = exact_cd(h_best, n, 50, deadline)

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
