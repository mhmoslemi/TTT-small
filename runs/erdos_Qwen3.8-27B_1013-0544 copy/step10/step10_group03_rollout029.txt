        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def pnorm_c5(h, n, p=16):
    c = np.correlate(h, 1.0 - h, mode='full')
    return (np.sum(c**p))**(1.0/p) * (2.0/n)

def pnorm_grad(h, n, p=16):
    c = np.correlate(h, 1.0 - h, mode='full')
    cp = c ** p
    s = np.sum(cp)
    if s < 1e-30:
        return np.zeros(n), 0.0
    coeff = (p / s) * c ** (p-1) * (2.0/n)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if abs(coeff[k]) < 1e-30:
            continue
        m = k
        m_min = max(0, m - (n-1))
        m_max = min(n-1, m + (n-1))
        # c[k] = sum_i h[i] * (1-h[i+k-(n-1)])

            jb = p_idx - k + (n-1)
            dc_dp = 0.0
            if 0 <= jf < n:
                dc_dp += (1.0 - h[jf])
            if 0 <= jb < n:
                dc_dp -= h[jb]
            grad[p_idx] += coeff[k] * dc_dp
    return grad, (s ** (1.0/p)) * (2.0/n)

        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def _obj(x, a=alpha, n=n):
            c = np.correlate(x, 1.0 - x, mode='full')
            if use_pnorm:
                return pnorm_max(c, a) * (2.0/n)
            smooth = smooth_max(c, a)
            return smooth * (2.0/n)
        def _cons_fun(x):
            return x.sum() - n / 2.0
        def _cons_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            res = minimize(_obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def binary_push(h, n, strength):

    lh[mask_low] = np.maximum(0.0, lh[mask_low] - strength)
    lh[mask_high] = np.minimum(1.0, lh[mask_high] + strength)
    return project_feasible(lh, n)

            return x.sum() - n / 2.0
        def _cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
        try:
            res = minimize(_obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def binary_push(h, n, strength):
    lh = h.copy()
    m_low = lh < 0.5
    m_high = lh >= 0.5
    lh[m_low] = np.maximum(0.0, lh[m_low] - strength)
    lh[m_high] = np.minimum(1.0, lh[m_high] + strength)
    return project_feasible(lh, n)

def segment_flip(h, n, rng):
    h2 = h.copy()
    seg_len = rng.randint(4, max(5, n // 12))
    start = rng.randint(0, n - seg_len)
    h2[start:start + seg_len] = 1.0 - h2[start:start + seg_len]
    return project_feasible(h2, n)

def local_search_perturb(h, n, rng, scale):
    perturb = rng.randint(0, n)
    width = rng.randint(3, max(4, n // 20))
    lo = max(0, perturb - width)
    hi = min(n, perturb + width)
    h2 = h.copy()
    h2[lo:hi] = h2[lo:hi] + rng.normal(0, scale, hi - lo)
    return project_feasible(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: SLSQP with geometric alpha progression
    if time.time() < deadline - 350:
        alphas = [3000, 15000, 75000, 375000]
        h1, c1 = slsqp_opt(h_best, n, alphas, 500, deadline - 350)
        if c1 < best_c:
            h_best, best_c = h1, c1

    # Phase 2: Bidirectional exact CD
    if time.time() < deadline - 280:
        h2, c2 = exact_cd(h_best, n, 400, deadline - 280, bidirectional=True)
        if c2 < best_c:
            h_best, best_c = h2, c1 if c1 < best_c else h2, c2
            best_c = c2 if c2 < best_c else best_c
        h_best, best_c = h2, c2 if c2 < best_c else (h_best, best_c)

    # Phase 3: Multi-n exploration
    n_candidates = [350, 450, 500, 550, 600, 650, 750, 850, 950]
    for n_try in n_candidates:
        if time.time() > deadline - 250:
            break
        if abs(n_try - len(h_best)) < 40:
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = slsqp_opt(h_int, n_try, [10000, 50000], 200, deadline - 250)
        h_o, c_o = exact_cd(h_o, n_try, 200, deadline - 250, bidirectional=True)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o

    # Phase 4: Diversified local search with CD
    n = len(h_best)
    for it in range(30):
        if time.time() > deadline - 180:
            break
        pertype = rng.randint(5)
        if pertype == 0:
            s = 0.004 + 0.012 * (it // 5)
            hp = project_feasible(h_best + rng.normal(0, s, n), n)
        elif pertype == 1:
            hp = local_search_perturb(h_best, n, rng, 0.03 + 0.02 * (it // 8))
        elif pertype == 2:
            hp = segment_flip(h_best, n, rng)
        elif pertype == 3:
            hp = binary_push(h_best, n, 0.02 + 0.02 * (it // 10))
        else:
            hp = binary_push(h_best, n, 0.05 + 0.03 * (it // 10))
        h_o, c_o = exact_cd(hp, n, 120, deadline - 180, bidirectional=True)
        if c_o < best_c:
            h_best, best_c, n = h_o, c_o, len(h_o)

    # Phase 5: Aggressive binary push + long CD
    if time.time() < deadline - 120:
        for bp in range(8):
            if time.time() > deadline - 100:
                break
            strength = 0.02 + 0.03 * bp
            hb = binary_push(h_best, n, strength)
            h_o, c_o = exact_cd(hb, n, 200, deadline - 100, bidirectional=True)
            if c_o < best_c:
                h_best, best_c = h_o, c_o

    # Phase 6: Final intensive bidirectional CD
    if time.time() < deadline - 5:
        n = len(h_best)
        h_f, c_f = exact_cd(h_best, n, 600, deadline, bidirectional=True)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
