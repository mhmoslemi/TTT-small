        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo < r1 and x_hi > r2:
                if r1 - x_lo > x_hi - r2:
                    x_lo = r2
                else:
                    x_hi = r1
            elif x_lo < r1:
                x_hi = min(x_hi, r1)
            elif x_hi > r2:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(c.max())
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if improved:
            h = project_feasible(h, n)
            if abs(h.sum() - n / 2.0) > 1e-12:
                c = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(c.max())
            if new_max < best_c:
                best_c = new_max
        else:
            break
    return h, float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def smooth_obj_func(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def cons_eq_fun(x):
    return x.sum() - x.size / 2.0

def cons_eq_jac(x):
    return np.ones(x.size)

def slsqp_refine(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
        try:
            res = minimize(smooth_obj_func, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def harmonic_perturb(h, n, rng, k, amp):
    x = np.linspace(0, 2, n, endpoint=False)
    pert = h + amp * np.sin(2.0 * np.pi * k * x / 2.0)
    return project_feasible(pert, n)

def block_perturb(h, n, rng, amp):
    bp = h.copy()
    b0 = rng.randint(0, max(1, n - n // 5))
    blen = rng.randint(n // 15, n // 4)
    bp[b0:b0 + blen] += rng.normal(0, amp, blen)
    return project_feasible(bp, n)

def anti_reflect(h, n):
    return project_feasible(1.0 - h[::-1].copy(), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Intensive CD from parent
    h_best, best_c = exact_cd(h_best, n, 300, deadline - 650)

    # Phase 2: Diverse perturbations + CD
    perturb_starts = []
    # Gaussian with varying sigma
    for s in [0.15, 0.08, 0.04, 0.02, 0.01, 0.005]:
        perturb_starts.append(project_feasible(h_best + rng.normal(0, s, n), n))
    # Harmonic perturbations
    for k in [1, 2, 3, 5]:
        for amp in [0.1, 0.05]:
            perturb_starts.append(harmonic_perturb(h_best, n, rng, k, amp))
    # Block perturbations
    for _ in range(6):
        perturb_starts.append(block_perturb(h_best, n, rng, 0.06))
    # Anti-reflect
    perturb_starts.append(anti_reflect(h_best, n))
    # Anti-reflect + small noise
    ar = anti_reflect(h_best, n)
    perturb_starts.append(project_feasible(ar + rng.normal(0, 0.02, n), n))

    t_phase2_end = deadline - 350
    per_start = max(20, int((t_phase2_end - time.time()) / max(1, len(perturb_starts))))
    for hp in perturb_starts:
        if time.time() > t_phase2_end:
            break
        ho, co = exact_cd(hp, n, per_start, t_phase2_end)
        if co < best_c:
            h_best, best_c = ho, co

    # Phase 3: Multi-n exploration
    t_phase3_end = deadline - 150
    n_list = [500, 600, 800, 950, 1100]
    per_n = max(30, int((t_phase3_end - time.time()) / max(1, len(n_list))))
    for n_try in n_list:
        if time.time() > t_phase3_end:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, len(h_best), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        ho, co = exact_cd(h_int, n_try, per_n, t_phase3_end)
        if co < best_c:
            h_best, best_c = ho, co
            n = n_try

    # Phase 4: SLSQP on best
    if time.time() < deadline - 80:
        n = len(h_best)
        h_s, c_s = slsqp_refine(h_best, n, [20000, 80000], 300, deadline - 60)
        if c_s < best_c:
            h_best, best_c = h_s, c_s

    # Phase 5: Final intensive CD
    n = len(h_best)
    h_best, best_c = exact_cd(h_best, n, 500, deadline)

    # Final projection and validation
    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
