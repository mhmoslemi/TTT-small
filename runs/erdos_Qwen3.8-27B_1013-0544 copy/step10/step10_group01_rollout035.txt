    h = h.copy()
    for it in range(iterations):
        # Compute autocorrelation
        H = np.fft.rfft(h)
        power = H * np.conj(H)
        autocorr = np.fft.irfft(power, n)
        # autocorr[k] = sum_i h[i]*h[i+k] (circular)
        # We want to increase the minimum
        min_ac = autocorr[1:].min()  # exclude lag 0
        max_ac = autocorr[1:].max()
        # Identify lags needing improvement
        weak_lags = np.where(autocorr < (min_ac + max_ac)/2)[0]
