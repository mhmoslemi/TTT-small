import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _obj(x, n, alpha, penalty):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    base = (m + lse / alpha) * (2.0 / n)
    cv = (x.sum() - n / 2.0) ** 2
    return base + penalty * cv

def _grad(x, n, alpha, penalty):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    # Gradient of base = (2/n) * (alpha * w @ dc/dx)
    # where dc/dx is the Jacobian of c w.r.t. x
    # dc[m]/dx[p] = (1-x[p+m-n+1])*I[0<=p+m-n+1<n] - x[p-m+n-1]*I[0<=p-m+n+1<n]
    
    dx = 2.0 / n
    grad_c = np.zeros(n)
    one_minus_x = 1.0 - x
    for p in range(n):
        # Forward: m such that 0 <= p+m-(n-1) < n → n-1-p <= m <= 2n-2-p
        m_lo = max(0, n - 1 - p)
        m_hi = min(2 * n - 2, 2 * n - 2 - p)
        if m_hi >= m_lo:
            idx = p + np.arange(m_lo, m_hi + 1) - (n - 1)
            grad_c[p] += w[m_lo:m_hi + 1] @ one_minus_x[idx]
        # Backward: m such that 0 <= p-m+(n-1) < n → p <= m <= p+n-1
        m_lo2 = max(0, p)
        m_hi2 = min(2 * n - 2, p + n - 1)
        if m_hi2 >= m_lo2:
            idx2 = p - np.arange(m_lo2, m_hi2 + 1) + (n - 1)
            grad_c[p] -= w[m_lo2:m_hi2 + 1] @ x[idx2]
    
    grad = dx * alpha * grad_c
    # Gradient of penalty term
    grad += 2.0 * penalty * (x.sum() - n / 2.0) * np.ones(n)
    return grad

def lbfgs_optimize(h, n, alphas, penalties, maxiter_each=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        pen = penalties[min(i, len(penalties) - 1)]
        try:
            res = minimize(_obj, best_h, jac=_grad, args=(n, alpha, pen),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter_each, 'ftol': 1e-14, 'gtol': 1e-8})
            if res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
