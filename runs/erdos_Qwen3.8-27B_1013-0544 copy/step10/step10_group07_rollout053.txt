            ok = False
        if ok and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2:
                x_hi = min(x_hi, r1) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
                x_lo = min(x_lo, r2) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                x_hi = min(x_hi, r1) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, order=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        idxs = order if order is not None else range(n)
        for p in idxs:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def subgrad_step(y, n):
    c = np.correlate(y, 1.0 - y, mode='full')
    j = int(np.argmax(c))
    m = j - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -m)
    i_hi = min(n - 1, n - 1 - m)
    p_lo1 = max(0, i_lo + m)
    p_hi1 = min(n - 1, i_hi + m)
    if p_lo1 <= p_hi1:
        pr = np.arange(p_lo1, p_hi1 + 1)
        grad[pr] += (1.0 - y[pr - m])
    if i_lo <= i_hi:
        pr = np.arange(i_lo, i_hi + 1)
        grad[pr] -= y[pr + m]
    return grad, c

def nesterov_subgrad(h, n, max_iter, deadline, lr_base=0.3):
    h_prev = h.copy()
    h_t = h.copy()
    y = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    L = dx * n * 2.0
    for t in range(1, max_iter + 1):
        if time.time() > deadline:
            break
        g, c = subgrad_step(y, n)
        lr = lr_base / np.sqrt(t)
        h_new = project_feasible(y - lr * g, n)
        beta = (t - 1) / (t + 2)
        y = project_feasible(h_new + beta * (h_new - h_prev), n)
        h_prev = h_new
        c_t = compute_c5(h_t, n)
        if c_t < best_c - 1e-15:
            best_c = c_t
            best_h = h_t.copy()
    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            def _cf(z, n_=n):
                return z.sum() - n_ / 2.0
            def _cj(z, n_=n):
                return np.ones(n_)
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                           options={'maxiter': maxiter, 'ftol': 1e-20})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def greedy_order(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    worst = int(np.argmax(c))
    m = worst - (n - 1)
    impact = np.zeros(n)
    for p in range(n):
        i1 = p - m
        i2 = p
        if 0 <= i1 < n and 0 <= i2 < n:
            impact[p] = abs(1.0 - h[i1]) + abs(h[i2])
    return np.argsort(-impact)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(400, 0.5)
        n = 400
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Quick CD polish with greedy ordering
    order = greedy_order(h_best, n)
    h1, c1 = exact_cd(h_best, n, 80, deadline, order=order)
    if c1 < best_c:
        h_best, best_c = h1, c1
    h1, c1 = exact_cd(h_best, n, 80, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Nesterov subgradient (global moves)
    if time.time() < deadline - 150:
        h2, c2 = nesterov_subgrad(h_best, n, 800, deadline, lr_base=0.2)
        if c2 < best_c:
            h_best, best_c = h2, c2
            h2b, c2b = exact_cd(h_best, n, 100, deadline)
            if c2b < best_c:
                h_best, best_c = h2b, c2b

    # Phase 3: SLSQP with smooth max
    if time.time() < deadline - 120:
        h3, c3 = slsqp_opt(h_best, n, [3000, 15000, 60000, 200000], 600, deadline)
        if c3 < best_c:
            h_best, best_c = h3, c3
            h3b, c3b = exact_cd(h_best, n, 150, deadline)
            if c3b < best_c:
                h_best, best_c = h3b, c3b

    # Phase 4: More CD with greedy ordering
    order = greedy_order(h_best, n)
    h4, c4 = exact_cd(h_best, n, 500, deadline, order=order)
    if c4 < best_c:
        h_best, best_c = h4, c4
    h4, c4 = exact_cd(h_best, n, 300, deadline)
    if c4 < best_c:
        h_best, best_c = h4, c4

    # Phase 5: Try nearby n values
    n_list = [n - 20, n - 10, n + 10, n + 20, n + 50, n - 50]
    for n_try in n_list:
        if time.time() > deadline - 120:
            break
        if n_try < 100 or n_try == n:
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 200, deadline)
        order_o = greedy_order(h_o, n_try)
        h_o2, c_o2 = exact_cd(h_o, n_try, 200, deadline, order=order_o)
        if c_o2 < c_o:
            h_o, c_o = h_o2, c_o2
        if c_o < best_c - 1e-7:
            h_best, best_c, n = h_o, c_o, n_try

    # Phase 6: Shake and settle
    for r in range(8):
        if time.time() > deadline - 60:
            break
        s = 0.003 + 0.004 * r
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        h_o, c_o = exact_cd(hp, n, 200, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 7: Final SLSQP + CD
    if time.time() < deadline - 40:
        h7, c7 = slsqp_opt(h_best, n, [100000, 500000], 400, deadline)
        if c7 < best_c:
            h_best, best_c = h7, c7
    if time.time() < deadline - 5:
        h_f, c_f = exact_cd(h_best, n, 200, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
