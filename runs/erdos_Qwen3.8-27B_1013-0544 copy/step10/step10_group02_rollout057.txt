def project_feasible(h, n):

def pnorm_obj(h, n, p):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.sum(np.abs(c) ** p)) ** (1.0 / p) * (2.0 / n)

def pnorm_opt(h, n, p_values, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for p in p_values:
        if time.time() > deadline:
            break
        def obj(x):
            return pnorm_obj(x, n, p)
        def cons_fun(x):
            return x.sum() - n / 2.0
        def cons_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def lse_obj(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_opt(h, n, obj_func, params, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for param in params:
        if time.time() > deadline:
            break
        cons = [{'type': 'eq', 'fun': _cons_f, 'jac': _cons_j, 'args': (n,)}]
        try:
            res = minimize(obj_func, best_h, args=(n, param), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c - 1e-12:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)

def binary_init(n, rng):
    m = n // 2
    phi = 0.618033988749895
    vals = (np.arange(n) * phi) % 1.0
    h = np.zeros(n)
    h[np.argsort(vals)[-m:]] = 1.0
    h += rng.uniform(0.02, 0.06, n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(350, 0.5)
        n = 350
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: CD on initial
    h1, c1 = exact_cd(h_best, n, 250, deadline, rng)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: LSE smooth optimization
    h2, c2 = smooth_opt(h_best, n, _lse_obj, [5000, 20000, 80000, 300000], 400, deadline)
    if c2 < best_c - 1e-12:
        h_best, best_c = h2, c2

    # Phase 3: P-norm smooth optimization (new)
    h3, c3 = smooth_opt(h_best, n, _pnorm_obj, [30, 80, 200, 500], 350, deadline)
    if c3 < best_c - 1e-12:
        h_best, best_c = h3, c3

    # Phase 4: CD after smooth
    h4, c4 = exact_cd(h_best, n, 250, deadline, rng)
    if c4 < best_c - 1e-12:
        h_best, best_c = h4, c4

    # Phase 5: Multi-resolution
    n_cur = len(h_best)
    n_list = [550, 650, 750, 450, 850, 400, 500, 950]
    if len(h_best) not in n_list:
        n_list.insert(0, len(h_best))
    for n_try in n_list:
        if time.time() > deadline - 180:
            break
        h_int = interp_h(h_best, n_cur, n_try)
        h_o, c_o = exact_cd(h_int, n_try, 150, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c, n_cur = h_o, c_o, n_try
        if time.time() < deadline - 80:
            h_o2, c_o2 = smooth_opt(h_o, n_try, _lse_obj, [30000, 150000], 250, deadline)
            if c_o2 < best_c - 1e-9:
                h_best, best_c, n_cur = h_o2, c_o2, n_try
            h_o3, c_o3 = smooth_opt(h_best, n_cur, _pnorm_obj, [50, 200], 200, deadline)
            if c_o3 < best_c - 1e-9:
                h_best, best_c, n_cur = h_o3, c_o3, n_try
            h_o4, c_o4 = exact_cd(h_best, n_cur, 100, deadline, rng)
            if c_o4 < best_c - 1e-9:
                h_best, best_c, n_cur = h_o4, c_o4, n_try

    # Phase 6: Annealed restarts
    for r in range(25):
        if time.time() > deadline - 50:
            break
        scale = 0.04 * (1.0 - r / 25.0) + 0.006
        hp = project_feasible(h_best + rng.normal(0, scale, n_cur), n_cur)
        h_o, c_o = exact_cd(hp, n_cur, 80, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o

    # Phase 7: Binary-style init探索
    for n_try in [200, 300, 400, 500, 600]:
        if time.time() > deadline - 30:
            break
        h_bin = binary_init(n_try, rng)
        h_o, c_o = exact_cd(h_bin, n_try, 100, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c, n_cur = h_o, c_o, n_try
        if time.time() < deadline - 25:
            h_o2, c_o2 = smooth_opt(h_o, n_try, _lse_obj, [10000, 100000], 200, deadline)
            if c_o2 < best_c - 1e-9:
                h_best, best_c, n_cur = h_o2, c_o2, n_try
            h_o3, c_o3 = exact_cd(h_best, n_cur, 80, deadline, rng)
            if c_o3 < best_c - 1e-9:
                h_best, best_c, n_cur = h_o3, c_o3, n_try

    # Phase 8: Final polish
    h_f, c_f = exact_cd(h_best, n_cur, 300, deadline, rng)
    if c_f < best_c - 1e-12:
        h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
