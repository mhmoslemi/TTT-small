**New approach: Adam-based optimization on a smoothed objective**

Instead of coordinate descent (which is one-directional), use a first-order method with adaptive learning rates (Adam) on a smooth approximation of the max. This explores the neighborhood more efficiently.

The smooth approximation: f_α(h) = (1/α) * log(∑_k exp(α * c_k / dx))  (log-sum-exp)

The gradient of this is: (∑_k exp(α*c_k/dx) * ∂c_k/∂h) / (∑_k exp(α*c_k/dx)) * (1/dx)

This is a weighted average of the gradients of the individual c_k, with weights concentrated on the largest c_k.

With Adam, this should converge faster than plain gradient descent and handle the "non-smooth" nature of max better.

But computing the full gradient requires O(n²) work per step (since there are 2n-1 correlation values, each depending on O(n) components of h). For n=700, that's about 10^6 operations per gradient, which is fine.

Actually, let me think about the gradient more carefully.

c_s = ∑_i h[i] * (1-h[i+s]) for valid i,s

∂c_s/∂h[j] = 
- If j is in the "first" position (h[i] where i=j): (1-h[j+s]) if j+s is valid
- If j is in the "second" position (h[i+s] where i+s=j, so i=j-s): -h[j-s] if j-s is valid
- If s=0 and j is both: (1-h[j]) - h[j] = 1-2h[j]

So the gradient of c_s with respect to h is:
∂c_s/∂h[j] = (1-h[j+s]) * 1_{j+s valid and j valid} - h[j-s] * 1_{j-s valid and j valid} (for s≠0)
And for s=0: 1-2h[j]

The gradient of the log-sum-exp: ∇f_α = (1/α) * ∑_k p_k * ∇c_k where p_k = exp(α*c_k/dx) / ∑_l exp(α*c_l/dx)

This is a weighted sum of the individual gradients. For large α, it's dominated by the max-correlation lags.

Implementation:
- For each step, compute c (O(n²) via correlate)
- Compute p (softmax of c)
- Compute ∇f_α = ∑_k p_k * ∇c_k (this is also O(n²) if done naively, but can be done in O(n) using the fact that ∇c_k is a shift of h)

Actually, ∇c_s is: (1-h shifted by -s) - (h shifted by s), masked to valid range.

So ∇f_α[j] = ∑_s p_s * [(1-h)[j+s_valid] - h[j-s_valid]]

This is like a convolution of p with (1-h) and h! I can compute it efficiently using FFT or just a sum over the significant p_s values.

For large α, only a few p_s are significant (those at the max), so I can just sum over those.

    h = h.copy()
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        t += 1
        c = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
        
        # Softmax weights
        c_max = c.max()
        p = np.exp((c - c_max) * alpha)
        p /= p.sum()
        
        # Gradient: sum_s p_s * grad_c_s
        # grad_c_s[j] ≈ (1-h)[j + (s - (n-1))] - h[j - (s - (n-1))]
        # Only need to consider s where p_s is significant
        grad = np.zeros(n)
        # For efficiency, only use top p_s
        top_k = 200  # number of top lags to consider
        top_idx = np.argsort(-p)[:top_k]
        
        for s_idx in top_idx:
            if p[s_idx] < 1e-10:
                break
            s = s_idx - (n-1)  # actual lag
            # grad_c_s[j] = (1-h[j+s]) * 1_{valid} - h[j-s] * 1_{valid}
            # For j+s valid: 0 <= j+s < n, and j valid: 0 <= j < n
            # (1-h[j+s]) for j in [max(0,-s), min(n, n-s))
            lo1 = max(0, -s)
            hi1 = min(n, n - s)
            if lo1 < hi1:
                grad[lo1:hi1] += p[s_idx] * (1.0 - h[lo1+s:hi1+s])
            # -h[j-s] for j in [max(0,s), min(n, n+s))
            lo2 = max(0, s)
            hi2 = min(n, n + s)
            if lo2 < hi2:
                grad[lo2:hi2] -= p[s_idx] * h[lo2-s:hi2-s]
        
        # Adam update
        m = betas[0] * m + (1-betas[0]) * grad
        v = betas[1] * v + (1-betas[1]) * grad**2
        m_hat = m / (1 - betas[0]**t)
        v_hat = v / (1 - betas[1]**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb): a[vb] -= h[jb[vb]]
            dp = np.where(ms == (n - 1))[0]
            has_d = len(dp) > 0
            di = int(dp[0]) if has_d else -1
            unaff = 0.0
            if m_lo > 0: unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2*n-2: unaff = max(unaff, float(c[m_hi+1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr*(d_hi - d_lo)
            d2 = d_lo + gr*(d_hi - d_lo)
            f1 = _cd_fval(c[ms], a, di, has_d, unaff, h0, d1)
            f2 = _cd_fval(c[ms], a, di, has_d, unaff, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15: break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr*(d_hi - d_lo)
                    f1 = _cd_fval(c[ms], a, di, has_d, unaff, h0, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr*(d_hi - d_lo)
                    f2 = _cd_fval(c[ms], a, di, has_d, unaff, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c[ms], a, di, has_d, unaff, h0, d_opt)
            f_curr = _cd_fval(c[ms], a, di, has_d, unaff, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved: break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    p = np.exp((c - cm) * alpha)
    p /= p.sum()
    grad = np.zeros(n)
    top_idx = np.argsort(-p)
    for si in top_idx:
        if p[si] < 1e-12: break
        s = si - (n - 1)
        if s >= 0:
            lo1, hi1 = 0, n - s
            lo2, hi2 = s, n
        else:
            lo1, hi1 = -s, n
            lo2, hi2 = 0, n + s
        grad[lo1:hi1] += p[si] * (1.0 - h[lo1+s:hi1+s])
        grad[lo2:hi2] -= p[si] * h[lo2-s:hi2-s]
    return grad

def adam_smooth(h, n, deadline, alpha, lr):
    h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    while time.time() < deadline:
        t += 1
        lr_t = lr / (1.0 + 0.001 * t)
        g = smooth_gradient(h, n, alpha)
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        mh = m / (1.0 - b1**t)
        vh = v / (1.0 - b2**t)
        h_new = h - lr_t * mh / (np.sqrt(vh) + eps)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def block_flip_cd(h, n, max_flips, deadline, rng):

    for freq in [2, 3, 4, 5, 6, 7, 8]:
        if time.time() > deadline: break
        for amp in [0.3, 0.4, 0.45]:
            if time.time() > deadline: break
            h = 0.5 + amp * np.sin(np.pi * freq * x)
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            results.append((h, c))
    return results

def two_block_cd(h, n, max_passes, deadline):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline: break
        c_all = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(0, n - 1, 2):
            if time.time() > deadline: break
            h0a, h0b = h[p], h[p+1]
            if h0a <= 1e-13 or h0a >= 1-1e-13 or h0b <= 1e-13 or h0b >= 1-1e-13:
                continue
            m_lo = max(0, n-1-p)
            m_hi = min(2*n-2, n-1+p)
            m_lo_b = max(0, n-2-p)
            m_hi_b = min(2*n-2, n-2+p)
            mlo = min(m_lo, m_lo_b)
            mhi = max(m_hi, m_hi_b)
            ms = np.arange(mlo, mhi+1)
            unaff = 0.0
            if mlo > 0: unaff = max(unaff, float(c_all[:mlo].max()))
            if mhi < 2*n-2: unaff = max(unaff, float(c_all[mhi+1:].max()))
            aff_a = np.zeros(len(ms))
            aff_b = np.zeros(len(ms))
            for j, aff in [(p, aff_a), (p+1, aff_b)]:
                jf = j + ms - (n-1)
                jb = j - ms + (n-1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                if np.any(vf): aff[vf] += (1.0 - h[jf[vf]])
                if np.any(vb): aff[vb] -= h[jb[vb]]
            da_lo, da_hi = -h0a, 1.0 - h0a
            db_lo, db_hi = -h0b, 1.0 - h0b
            best_f = float(np.max(c_all))
            for _gs in range(30):
                if (da_hi-da_lo)*(db_hi-db_lo) < 1e-20: break
                das = [(da_hi-gr*(da_hi-da_lo)), (da_lo+gr*(da_hi-da_lo))]
                dbs = [(db_hi-gr*(db_hi-db_lo)), (db_lo+gr*(db_hi-db_lo))]
                best_move = None
                best_mv = best_f
                for da in das:
                    for db in dbs:
                        vals = c_all[ms] + da*aff_a + db*aff_b
                        dp_a = np.where(ms == n-1)[0]
                        dp_b = np.where(ms == n-2)[0]
                        if len(dp_a)>0:
                            vals[int(dp_a[0])] = c_all[int(dp_a[0])] + da*(1-h0a-da) + db*(1-h0b-db)
                        elif len(dp_b)>0:
                            vals[int(dp_b[0])] = c_all[int(dp_b[0])] + da*(1-h0a-da) + db*(1-h0b-db)
                        fv = max(float(np.max(vals)), unaff)
                        if fv < best_mv:
                            best_mv = fv
                            best_move = (da, db)
                if best_move is None: break
                da, db = best_move
                if da < da_lo - 1e-15: da_lo = da
                elif da > da_hi + 1e-15: da_hi = da
                if db < db_lo - 1e-15: db_lo = db
                elif db > db_hi + 1e-15: db_hi = db
            if best_move is not None:
                da, db = best_move
                h[p] = h0a + da
                h[p+1] = h0b + db
                improved = True
        if not improved: break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 500
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: CD polish
    d1 = t0 + 0.12 * budget_s
    h1, c1 = coord_descent(best_h, n, 60, d1, gs_iters=55)
    if c1 < best_c: best_h, best_c = h1, c1

    # Phase 2: Adam smooth max exploration
    d2 = t0 + 0.30 * budget_s
    for a_val, lr_val in [(30, 0.008), (80, 0.004), (200, 0.002)]:
        if time.time() > d2: break
        h2, c2 = adam_smooth(best_h, n, min(time.time() + (d2 - time.time())/3, d2), a_val, lr_val)
        h2, c2 = coord_descent(h2, n, 25, d2, gs_iters=45)
        if c2 < best_c: best_h, best_c = h2, c2

    # Phase 3: Two-block CD
    d3 = t0 + 0.45 * budget_s
    h3, c3 = two_block_cd(best_h, n, 25, d3)
    if c3 < best_c: best_h, best_c = h3, c3
    h3, c3 = coord_descent(best_h, n, 20, d3, gs_iters=50)
    if c3 < best_c: best_h, best_c = h3, c3

    # Phase 4: Block flip perturbations
    d4 = t0 + 0.70 * budget_s
    h4, c4 = block_flip_cd(best_h, n, 20, d4, rng)
    if c4 < best_c: best_h, best_c = h4, c4
    h4, c4 = coord_descent(best_h, n, 30, d4, gs_iters=50)
    if c4 < best_c: best_h, best_c = h4, c4

    # Phase 5: Sinusoid-based starts + CD
    d5 = t0 + 0.85 * budget_s
    starts = sinusoid_starts(n, d5 - 10)
    for hs, cs in starts:
        if time.time() > d5: break
        if cs < best_c + 0.05:
            ho, co = coord_descent(hs, n, 40, d5, gs_iters=50)
            if co < best_c: best_h, best_c = ho, co

    # Phase 6: More Adam with different params on best
    d6 = t0 + 0.93 * budget_s
    for a_val, lr_val in [(50, 0.006), (150, 0.003)]:
        if time.time() > d6: break
        h6, c6 = adam_smooth(best_h, n, min(time.time() + (d6-time.time())/2, d6), a_val, lr_val)
        h6, c6 = coord_descent(h6, n, 25, d6, gs_iters=50)
        if c6 < best_c: best_h, best_c = h6, c6

    # Phase 7: Final CD + two-block polish
    h7, c7 = coord_descent(best_h, n, 40, deadline, gs_iters=60)
    if c7 < best_c: best_h, best_c = h7, c7
    h7, c7 = two_block_cd(best_h, n, 10, deadline)
    if c7 < best_c: best_h, best_c = h7, c7

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
