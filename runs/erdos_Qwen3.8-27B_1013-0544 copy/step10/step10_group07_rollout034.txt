        if j1 == i:  # diagonal
            grad[i] -= h[i]
        if 0 <= j2 < n:
            grad[i] -= h[j2]
        if j2 == i:  # diagonal

def smooth_obj(x, n, alpha):

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    elif d > 0:
        # j+d < n means j < n-d
        grad[:n-d] += (1.0 - h[d:])
        # j-d >= 0 means j >= d
        grad[d:] -= h[:-d]
    else:  # d < 0
        dd = -d
        # j+d >= 0 means j >= dd
        grad[dd:] += (1.0 - h[:n-dd])
        # j-d < n means j < n+dd, always true for j < n
        grad[:] -= h[dd:]
    return project_feasible(h - step_size * grad, n)

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def cd_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = max(unaff, float(np.max(c[:m_min])))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(45):
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        ok = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-15):
            ok = False
        if ok and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if ok and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            ok = False
        if ok and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2:
                x_hi = min(x_hi, r1) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
                x_lo = min(x_lo, r2) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                x_hi = min(x_hi, r1) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for p in order:

            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:

                m_min = n - 1 - p

                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])

                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def subgrad_step(h, n, step_size):

    k_star = int(np.argmax(c))
    d = k_star - (n - 1)

            if new_c < best_c - 1e-12:

            pass

The key insight is that R(k) = dx * (∑h - ∑h[i]*h[i+k]) = dx * (n/2 - autocorr_h(k)).

So minimizing max_k R(k) is equivalent to maximizing min_k autocorr_h(k) (over all valid k).

The auto-correlation of h is related to its Fourier transform: autocorr_h(k) = (1/2π) ∫ |H(ω)|² e^{iωk} dω (in the continuous case).

In the discrete case, the DFT gives us: the auto-correlation is the inverse DFT of |H|².

So we want to maximize the minimum of the inverse DFT of |H|², subject to constraints on h.

This is related to the "minimum peak sidelobe" problem in signal processing! The goal is to make the auto-correlation as flat as possible (high minimum value), which is like designing a signal with low peak sidelobe levels.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def cd_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = max(unaff, float(np.max(c[:m_min])))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(45):
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        ok = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-15):
            ok = False
        if ok and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if ok and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            ok = False
        if ok and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2:
                x_hi = min(x_hi, r1) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
                x_lo = min(x_lo, r2) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                x_hi = min(x_hi, r1) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def subgrad_soft(h, n, step_size, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    w = np.exp(alpha * (c - m))
    wsum = w.sum()
    if wsum < 1e-30:
        return h
    w /= wsum
    top = np.argsort(c)[-10:]
    grad = np.zeros(n)
    for k_s in top:
        wi = w[k_s]
        if wi < 1e-8:
            continue
        d = k_s - (n - 1)
        if d == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n - d] += wi * (1.0 - h[d:])
            grad[d:] -= wi * h[:-d]
        else:
            dd = -d
            grad[dd:] += wi * (1.0 - h[:n - dd])
            grad[:] -= wi * h[dd:]
    return project_feasible(h - step_size * grad, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _cf(z):
    return z.sum() - len(z) / 2.0

def _cf_jac(z):
    return np.ones(len(z))

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cf_jac}],
                           options={'maxiter': maxiter, 'ftol': 1e-20})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-13:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
