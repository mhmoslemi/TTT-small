def smooth_f(x):
    n=int(_G_N); alpha=float(_G_ALPHA)
    c=np.correlate(x,1.0-x,mode='full')
    m=float(np.max(c))
    shifted=alpha*(c-m)
    S=float(np.sum(np.exp(shifted)))
    return (2.0/n)*(m + np.log(S)/alpha)

def smooth_jac(x):
    n=int(_G_N); alpha=float(_G_ALPHA)
    c=np.correlate(x,1.0-x,mode='full')
    m=float(np.max(c))
    shifted=alpha*(c-m)
    u=np.exp(shifted)
    S=float(np.sum(u))
    if S <= 0: return np.zeros(n)
    T1=np.convolve(1.0-x, u, mode='full')[n-1:2*n-1]
    T2=np.convolve(x, u[::-1], mode='full')[n-1:2*n-1]
    return ((2.0/n)/S)*(T1-T2)

```python
import numpy as np
import time
from scipy.optimize import minimize

_G_N = 0
_G_ALPHA = 1.0


def compute_c5_from_c(c, n):
    return float(np.max(c) * (2.0 / float(n)))


def compute_c5(h, n):
    return compute_c5_from_c(np.correlate(h, 1.0 - h, mode="full"), n)


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = int(n)
    if n <= 0:
        return h
    target = 0.5 * float(n)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    return np.clip(h - tau, 0.0, 1.0)


def interp_h(h_old, n_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    n_old = int(n_old)
    n_new = int(n_new)
    if n_old == n_new:
        return project_feasible(h_old, n_new)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, left=float(h_old[0]), right=float(h_old[-1]))
    return project_feasible(h_new, n_new)


def coord_data(h, p, n):
    kp = n - 1 - p
    kmin = p if p < kp else kp
    kmax1 = n - 1 + p
    kmax2 = 2 * n - 2 - p
    kmax = kmax1 if kmax1 >= kmax2 else kmax2
    ks = np.arange(kmin, kmax + 1)
    d = ks - (n - 1)
    a = np.zeros(len(ks), dtype=float)

    i1 = p - d
    v1 = (i1 >= 0) & (i1 < n)
    if np.any(v1):
        a[v1] += 1.0 - h[i1[v1]]

    i2 = p + d
    v2 = (i2 >= 0) & (i2 < n)
    if np.any(v2):
        a[v2] -= h[i2[v2]]

    diag = (n - 1) - kmin
    a_lin = a.copy()
    a_lin[diag] = 0.0
    return ks, a, a_lin, diag, kmin, kmax


def solve_1d(c, c_base, a_lin, h0, n, kmin, kmax, diag):
    K = 2 * n - 1
    unaff = 0.0
    if kmin > 0:
        unaff = float(np.max(c[:kmin]))
    if kmax < K - 1:
        u2 = float(np.max(c[kmax + 1:]))
        if u2 > unaff:
            unaff = u2

    L_lo = unaff
    L_hi = float(np.max(c))
    if L_hi < L_lo:
        L_hi = L_lo
    if L_hi <= L_lo:
        L_hi = L_lo + 1e-12

    eps = 1e-13
    zero = np.abs(a_lin) <= eps
    neg = a_lin < -eps
    pos = a_lin > eps

    zany = bool(np.any(zero))
    nzmax = float(np.max(c_base[zero])) if zany else -np.inf
    anyneg = bool(np.any(neg))
    anypos = bool(np.any(pos))

    cneg = c_base[neg] if anyneg else None
    aneg = a_lin[neg] if anyneg else None
    cpos = c_base[pos] if anypos else None
    apos = a_lin[pos] if anypos else None

    for _ in range(42):
        if L_hi - L_lo <= 1e-11:
            break
        L = 0.5 * (L_lo + L_hi)
        feasible = True

        if zany and nzmax > L + 1e-10:
            feasible = False

        xl = 0.0
        xh = 1.0
        if feasible and anyneg:
            t = float(np.max((L - cneg) / aneg))
            if t > xl:
                xl = t
        if feasible and anypos:
            t = float(np.min((L - cpos) / apos))
            if t < xh:
                xh = t

        if xl < 0.0:
            xl = 0.0
        if xh > 1.0:
            xh = 1.0
        if xl > xh + 1e-10:
            feasible = False

        if feasible:
            Cq = L - c_base[diag] + h0 - h0 * h0
            if Cq < 0.25 - 1e-12:
                disc = 1.0 - 4.0 * Cq
                if disc < 0.0:
                    disc = 0.0
                r1 = 0.5 * (1.0 - np.sqrt(disc))
                r2 = 1.0 - r1
                if xl > r1 + 1e-10 and xh < r2 - 1e-10:
                    feasible = False

        if feasible:
            L_hi = L
        else:
            L_lo = L

    L = L_hi

    if zany and nzmax > L + 1e-8:
        return h0

    xl = 0.0
    xh = 1.0
    if anyneg:
        t = float(np.max((L - cneg) / aneg))
        if t > xl:
            xl = t
    if anypos:
        t = float(np.min((L - cpos) / apos))
        if t < xh:
            xh = t

    if xl < 0.0:
        xl = 0.0
    if xh > 1.0:
        xh = 1.0
    if xl > xh + 1e-9:
        return h0

    Cq = L - c_base[diag] + h0 - h0 * h0
    if Cq >= 0.25 - 1e-11:
        x = 0.5 * (xl + xh)
    else:
        if Cq < -1e-9:
            return h0
        disc = 1.0 - 4.0 * Cq
        if disc < 0.0:
            disc = 0.0
        r1 = 0.5 * (1.0 - np.sqrt(disc))
        r2 = 1.0 - r1

        left_ok = xl < r1 + 1e-10
        right_ok = xh > r2 - 1e-10
        left_hi = r1 if r1 < xh else xh
        right_lo = r2 if r2 > xl else xl
        left_len = left_hi - xl if left_ok else -1.0
        right_len = xh - right_lo if right_ok else -1.0

        if left_len < 0.0 and right_len < 0.0:
            return h0
        if right_len < 0.0:
            x = 0.5 * (xl + left_hi)
        elif left_len < 0.0:
            x = 0.5 * (right_lo + xh)
        else:
            lm = 0.5 * (xl + left_hi)
            rm = 0.5 * (right_lo + xh)
            x = lm if abs(lm - h0) <= abs(rm - h0) else rm

    if x < 0.0:
        x = 0.0
    elif x > 1.0:
        x = 1.0
    if not np.isfinite(x):
        return h0
    return float(x)


def exact_cd(h, n, max_passes, deadline, rng):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    n = int(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    best_h = h.copy()
    best_c = compute_c5_from_c(c, n)
    order = list(range(n))

    for it in range(max_passes):
        if time.time() > deadline:
            break
        if it % 2 == 1 and rng is not None:
            rng.shuffle(order)

        improved = 0
        for p in order:
            if time.time() > deadline:
                break
            old = float(h[p])
            ks, a_full, a_lin, diag, kmin, kmax = coord_data(h, p, n)
            c_base = c[ks]
            xn = solve_1d(c, c_base, a_lin, old, n, kmin, kmax, diag)
            delta = xn - old
            if abs(delta) > 1e-11:
                c[ks] += delta * a_full
                c[ks[diag]] -= delta * delta
                h[p] = xn
                improved += 1

        h2 = project_feasible(h, n)
        changed = bool(np.any(np.abs(h2 - h) > 1e-14))
        h = h2
        if changed or (it > 0 and it % 25 == 0):
            c = np.correlate(h, 1.0 - h, mode="full")

        curr = compute_c5_from_c(c, n)
        if curr < best_c:
            best_c = curr
            best_h = h.copy()
        if improved == 0:
            break

    return best_h, best_c


def smooth_f(x):
    n = int(_G_N)
    alpha = float(_G_ALPHA)
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(c))
    shifted = np.minimum(0.0, alpha * (c - m))
    S = float(np.sum(np.exp(shifted)))
    return float((2.0 / float(n)) * (m + np.log(S) / alpha))


def smooth_jac(x):
    n = int(_G_N)
    alpha = float(_G_ALPHA)
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(c))
    shifted = np.minimum(0.0, alpha * (c - m))
    u = np.exp(shifted)
    S = float(np.sum(u))
    if S <= 0.0:
        return np.zeros(n, dtype=float)
    t1 = np.convolve(1.0 - x, u, mode="full")[n - 1:2 * n - 1]
    t2 = np.convolve(x, u[::-1], mode="full")[n - 1:2 * n - 1]
    return ((2.0 / float(n)) / S) * (t1 - t2)


def sum_eq(x):
    return float(np.sum(x) - 0.5 * float(_G_N))


def sum_jac(x):
    return np.ones(int(_G_N), dtype=float)


def slsqp_polish(h, n, alphas, maxiter, deadline):
    global _G_N, _G_ALPHA
    h = project_feasible(h, n)
    n = int(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_eq, "jac": sum_jac}]
    maxit = int(maxiter)
    if n > 900:
        maxit = max(25, int(0.6 * maxit))

    for alpha in alphas:
        if time.time() > deadline:
            break
        _G_N = n
        _G_ALPHA = float(alpha)
        x0 = project_feasible(best_h, n)
        try:
            res = minimize(
                smooth_f,
                x0,
                jac=smooth_jac,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": maxit, "ftol": 1e-14},
            )
        except Exception:
            res = None

        if res is not None:
            try:
                new_h = project_feasible(res.x, n)
            except Exception:
                new_h = best_h
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_c = new_c
                best_h = new_h

    return best_h, best_c


def to_binary(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    idx = np.argsort(h)
    b = np.zeros(n, dtype=float)
    b[idx[n // 2:]] = 1.0
    return b


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s) - 6.0)
    rng = np.random.RandomState(seed)

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_best = np.full(700, 0.5, dtype=float)

    n_best = int(len(h_best))
    if n_best <= 1:
        h_best = np.full(700, 0.5, dtype=float)
        n_best = 700
    if n_best % 2 == 1:
        h_best = interp_h(h_best, n_best, 700)
        n_best = 700

    h_best = project_feasible(h_best, n_best)
    best_c = compute_c5(h_best, n_best)

    if time.time() < deadline - 90:
        h1, c1 = exact_cd(h_best, n_best, 70, deadline, rng)
        if c1 < best_c - 1e-12:
            h_best, best_c = h1, c1
            n_best = int(len(h_best))

    if time.time() < deadline - 170:
        h2, c2 = slsqp_polish(h_best, n_best, [1000.0, 5000.0, 20000.0], 70, deadline)
        if c2 < best_c - 1e-9:
            h_best, best_c = h2, c2
            n_best = int(len(h_best))

    if time.time() < deadline - 150:
        h3, c3 = exact_cd(h_best, n_best, 45, deadline, rng)
        if c3 < best_c - 1e-12:
            h_best, best_c = h3, c3
            n_best = int(len(h_best))

    for n_try in (900, 1000, 850, 950, 800):
        if time.time() > deadline - 260:
            break
        n_try = int(n_try)
        h_int = interp_h(h_best, n_best, n_try)
        h1, c1 = exact_cd(h_int, n_try, 55, deadline, rng)
        if c1 < best_c - 1e-9:
            h_best, best_c = h1, c1
            n_best = int(len(h_best))

        if time.time() > deadline - 200:
            continue
        if n_best <= 850:
            h2, c2 = slsqp_polish(h_best, n_best, [1000.0, 5000.0, 20000.0], 55, deadline)
        else:
            h2, c2 = slsqp_polish(h_best, n_best, [2000.0, 10000.0, 30000.0], 45, deadline)
        if c2 < best_c - 1e-9:
            h_best, best_c = h2, c2
            n_best = int(len(h_best))

        if time.time() > deadline - 150:
            continue
        h3, c3 = exact_cd(h_best, n_best, 60, deadline, rng)
        if c3 < best_c - 1e-12:
            h_best, best_c = h3, c3
            n_best = int(len(h_best))

    n_cur = int(len(h_best))

    if time.time() < deadline - 90:
        hp = to_binary(h_best, n_cur)
        hp = project_feasible(hp, n_cur)
        ho, co = exact_cd(hp, n_cur, 35, deadline, rng)
        if co < best_c - 1e-9:
            h_best, best_c = ho, co
            n_cur = int(len(h_best))

    max_rest = 5 if n_cur > 850 else 9
    for r in range(max_rest):
        if time.time() > deadline - 70:
            break
        scale = 0.025 * (1.0 - float(r) / float(max_rest)) + 0.006
        hp = project_feasible(h_best + rng.normal(0.0, scale, n_cur), n_cur)
        ho, co = exact_cd(hp, n_cur, 30, deadline, rng)
        if co < best_c - 1e-9:
            h_best, best_c = ho, co
            n_cur = int(len(h_best))

    if time.time() < deadline - 30:
        h_best, best_c = exact_cd(h_best, n_cur, 110, deadline, rng)
        n_cur = int(len(h_best))

    h_best = project_feasible(h_best, n_cur)
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, n_cur)
    best_c = compute_c5(h_best, n_cur)
    return h_best, best_c, int(n_cur)
```
