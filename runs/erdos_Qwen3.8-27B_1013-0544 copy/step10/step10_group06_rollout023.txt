def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s2 = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s2 < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(25):
        diff = target - float(h.sum())
        if abs(diff) < 1e-12:

        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-15:

        else:
            tot = float(h.sum())

    return h

def anti_coord_descent(v, max_passes, deadline, rng, random_order=True, bisect=30):
    v = np.clip(np.asarray(v, dtype=float), 0.0, 1.0).copy()
    m = v.size
    n = 2 * m
    h = anti_full(v)
    dx = 2.0 / n
    best_c = c_value_h(h)
    best_v = v.copy()
    best_h = h.copy()

    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode="full")
        cur = float(np.max(c)) * dx
        if cur < best_c - 1e-13:
            best_c = cur
            best_v = v.copy()
            best_h = h.copy()
        improved = False
        if random_order:
            order = rng.permutation(m)
        else:
            order = np.arange(m)
        for ii in order:
            if time.time() > deadline:
                break
            if _anti_coord_improve(int(ii), v, h, c, n, bisect):
                improved = True
        if not improved:
            break

    c = np.correlate(h, 1.0 - h, mode="full")
    cur = float(np.max(c)) * dx
    if cur < best_c - 1e-13:
        best_c = cur
        best_v = v.copy()
        best_h = h.copy()
    return best_v, best_c, best_h

        v_hi = main_v if 2 * len(main_v) == 700 else v_from_h(anti_full(main_v), 700)
        v_fun, c_fun, h_fun = anti_coord_descent(v_hi, 30, deadline - 1.0, rng, True, 35)
        if c_fun < c_best - 1e-13:
            c_best = c_fun
            h_best = h_fun
            n_best = 2 * len(v_fun)
            v_best = v_fun.copy()

    h_best = np.clip(project_feasible(h_best, n_best), 0.0, 1.0)
    c_best = c_value_h(h_best)
    return h_best, c_best, n_best
