        n_cur = len(best_h)
        hf, cf = slsqp_polish(best_h, n_cur, [5000, 20000, 100000], 500)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = coord_descent(best_h, n_cur, 100, deadline - 10, gs_iters=90, tol=1e-15)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
