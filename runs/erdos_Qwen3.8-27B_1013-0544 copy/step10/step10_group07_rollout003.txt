def obj_lbfgs(x, n, beta, gamma):
    dx = 2.0 / n
    h = np.clip(x,0.0,1.0) # L-BFGS bounds but safety
    c = np.correlate(h, 1.0-h, 'full') * dx
    m = float(np.max(c))
    shifted = beta * (c - m)
    vals = np.exp(shifted)
    s = float(np.sum(vals))
    if s <= 0: weights = np.ones_like(c)/len(c)
    else: weights = vals / s
    F = m + np.log(s) / beta
    # gradient
    M = 2*n-1
    pref = np.empty(M+1); pref[0]=0; pref[1:]=np.cumsum(weights)
    p = np.arange(n)
    start = n-1-p
    end = 2*n-1-p
    A = pref[end] - pref[start]
    B = np.convolve(h, weights, 'full')[n-1:2*n-2]
    wrev = weights[::-1]
    C = np.convolve(h, wrev, 'full')[n-1:2*n-2]
    g = dx * (A - B - C)
    diff = float(np.sum(h) - n/2.0)
    F += gamma * diff * diff
    if gamma != 0.0:
       g += 2.0*gamma*diff
    return F, g
```

def intersect_quad_intervals(intervals, A, B, CmL):
    out=[]
    for a,b in intervals:
        if b < a - 1e-12: continue
        if abs(A) < 1e-14:
            if abs(B) < 1e-14:
                if CmL <= 1e-12: out.append((a,b))
            elif B > 0:
                ub = -CmL / B
                nb = min(b, ub)
                if nb >= a - 1e-12: out.append((a, nb))
            else:
                lb = -CmL / B
                na = max(a, lb)
                if na <= b + 1e-12: out.append((na, b))
            continue
        D = B*B - 4.0*A*CmL
        if D < -1e-12:
            if A < 0.0: out.append((a,b))
            continue
        if D < 0: D=0.0
        sr = np.sqrt(D)
        r1 = (-B - sr) / (2.0*A)
        r2 = (-B + sr) / (2.0*A)
        if r1 > r2: r1,r2=r2,r1
        if A > 0:
            na=max(a,r1); nb=min(b,r2)
            if nb >= na - 1e-12: out.append((na,nb))
        else:
            if b >= r1 - 1e-12:
                nb=min(b,r1)
                if nb >= a - 1e-12: out.append((a,nb))
            if a <= r2 + 1e-12:
                na=max(a,r2)
                if na <= b + 1e-12: out.append((na,b))
    if not out: return []
    out.sort()
    merged=[]
    for a,b in out:
        if a > 1.0 + 1e-12 or b < -1e-12: continue
        a=max(0.0,a); b=min(1.0,b)
        if b < a - 1e-12: continue
        if merged and a <= merged[-1][1] + 1e-10:
            if b > merged[-1][1]: merged[-1][1]=b
        else:
            merged.append([a,b])
    return [(a,b) for a,b in merged]

import numpy as np
import time
from scipy.optimize import minimize


def compute_c_array(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_c5(h, n):
    return float(np.max(compute_c_array(h, n)))


def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(2000):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = float(h.sum())
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    diff = target - float(h.sum())
    if diff > 1e-12:
        idx = int(np.argmin(h))
        add = min(diff, 1.0 - h[idx])
        h[idx] += add
    elif diff < -1e-12:
        idx = int(np.argmax(h))
        sub = min(-diff, h[idx])
        h[idx] -= sub
    return h


def obj_lbfgs(x, n, beta, gamma):
    dx = 2.0 / float(n)
    h = np.clip(x, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(np.max(c))
    sh = beta * (c - m)
    vals = np.exp(sh)
    svals = float(np.sum(vals))
    if svals <= 0.0:
        weights = np.full_like(c, 1.0 / len(c))
    else:
        weights = vals / svals
    val = m + np.log(svals) / beta

    M = 2 * n - 1
    pref = np.empty(M + 1)
    pref[0] = 0.0
    pref[1:] = np.cumsum(weights)
    p = np.arange(n)
    start = n - 1 - p
    end = 2 * n - 1 - p
    A = pref[end] - pref[start]
    B = np.convolve(h, weights, mode="full")[n - 1:2 * n - 1]
    C = np.convolve(h, weights[::-1], mode="full")[n - 1:2 * n - 1]
    grad = dx * (A - B - C)

    diff = float(h.sum()) - float(n) / 2.0
    if gamma != 0.0:
        val += gamma * diff * diff
        grad = grad + 2.0 * gamma * diff
    return val, grad


def lbfgs_polish(h, n, deadline, rng):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

    for beta, gamma, maxit in [(120, 80, 80), (450, 300, 80), (1500, 1000, 60)]:
        if time.time() > deadline:
            break
        try:
            res = minimize(obj_lbfgs, best_h, args=(n, beta, gamma),
                           method="L-BFGS-B", bounds=bounds,
                           options={"maxiter": maxit, "ftol": 1e-12, "disp": 0})
        except Exception:
            continue
        nh = project_feasible(res.x, n)
        nc = compute_c5(nh, n)
        if nc < best_c - 1e-12:
            best_h, best_c = nh, nc

    for sigma in (0.03, 0.08, 0.16):
        if time.time() > deadline - 40:

        hp = project_feasible(best_h + rng.normal(0.0, sigma, size=n), n)

def intersect_quad_intervals(intervals, A, B, CmL):
    out = []
    for a, b in intervals:
        if b < a - 1e-12:
            continue
        if abs(A) < 1e-14:
            if abs(B) < 1e-14:
                if CmL <= 1e-12:
                    out.append((a, b))
            elif B > 0.0:
                ub = -CmL / B
                nb = min(b, ub)
                if nb >= a - 1e-12:
                    out.append((a, nb))
            else:
                lb = -CmL / B
                na = max(a, lb)
                if na <= b + 1e-12:
                    out.append((na, b))
            continue
        D = B * B - 4.0 * A * CmL
        if D < -1e-12:
            if A < 0.0:
                out.append((a, b))
            continue
        if D < 0.0:
            D = 0.0
        sr = np.sqrt(D)
        r1 = (-B - sr) / (2.0 * A)
        r2 = (-B + sr) / (2.0 * A)
        if r1 > r2:
            r1, r2 = r2, r1
        if A > 0.0:
            na = max(a, r1)
            nb = min(b, r2)
            if nb >= na - 1e-12:
                out.append((na, nb))
        else:
            if b >= r1 - 1e-12:
                nb = min(b, r1)
                if nb >= a - 1e-12:
                    out.append((a, nb))
            if a <= r2 + 1e-12:
                na = max(a, r2)
                if na <= b + 1e-12:
                    out.append((na, b))
    if not out:
        return []
    out.sort()
    merged = []
    for a, b in out:
        if a > 1.0 + 1e-12 or b < -1e-12:
            continue
        a = max(0.0, a)
        b = min(1.0, b)
        if b < a - 1e-12:
            continue
        if merged and a <= merged[-1][1] + 1e-10:
            if b > merged[-1][1]:
                merged[-1][1] = b
        else:
            merged.append([a, b])
    return [(a, b) for a, b in merged]


def pair_check(L, xlo, xhi, const, pos_a1, pos_const,
               neg_a1, neg_const, zero_const, qA, qB, qC):
    lo = xlo
    hi = xhi
    if pos_a1.size:
        hi = min(hi, float(np.min((L - pos_const) / pos_a1)))
    if neg_a1.size:
        lo = max(lo, float(np.max((L - neg_const) / neg_a1)))
    if hi < lo - 1e-12:
        return False, []
    if zero_const.size:
        if float(np.max(zero_const)) > L + 1e-11:
            return False, []
    lo = max(xlo, lo)
    hi = min(xhi, hi)
    if hi < lo - 1e-12:
        return False, []
    intervals = [(lo, hi)]
    for i in range(qA.size):
        intervals = intersect_quad_intervals(intervals, float(qA[i]),
                                             float(qB[i]), float(qC[i] - L))
        if not intervals:
            return False, []
    return True, intervals

    line = np.abs(ca2) < 1e-13
    pos_idx = np.where(line & (ca1 > 1e-13))[0]
    neg_idx = np.where(line & (ca1 < -1e-13))[0]
    zero_idx = np.where(line & (np.abs(ca1) <= 1e-13))[0]
    quad_idx = np.where(~line)[0]

    pos_a1 = ca1[pos_idx]
    pos_const = const[pos_idx]
    neg_a1 = ca1[neg_idx]
    neg_const = const[neg_idx]
    zero_const = const[zero_idx]
    qA = ca2[quad_idx]
    qB = ca1[quad_idx]
    qC = const[quad_idx]

    Llo = max(0.0, max_unaff)
    Lhi = Ccur + 1e-12
    if Llo > Lhi:
        Llo = 0.0
        Lhi = float(np.max(np.concatenate((c[aff], np.array([max_unaff]))))) + 1e-10

    ok, _ = pair_check(Lhi, xlo, xhi, const, pos_a1, pos_const,
                       neg_a1, neg_const, zero_const, qA, qB, qC)
    if not ok:
        Lhi += 1e-8
        ok, _ = pair_check(Lhi, xlo, xhi, const, pos_a1, pos_const,
                           neg_a1, neg_const, zero_const, qA, qB, qC)
        if not ok:
            return False

    for _ in range(48):
        Lm = 0.5 * (Llo + Lhi)
        ok, _ = pair_check(Lm, xlo, xhi, const, pos_a1, pos_const,
                           neg_a1, neg_const, zero_const, qA, qB, qC)
        if ok:
            Lhi = Lm
        else:
            Llo = Lm

    ok, intervals = pair_check(Lhi + 1e-12, xlo, xhi, const, pos_a1, pos_const,
                               neg_a1, neg_const, zero_const, qA, qB, qC)
    if not ok or not intervals:
        return False

    best_dist = 1e100
    best_a = xlo
    best_b = xhi
    for a, b in intervals:
        if a <= x0 <= b:
            dist = 0.0
        else:
            dist = min(abs(x0 - a), abs(x0 - b))
        if dist < best_dist:
            best_dist = dist
            best_a, best_b = a, b

    a, b = best_a, best_b
    if x0 < a:
        xc = a
    elif x0 > b:
        xc = b
    else:
        xc = x0
    mid = 0.5 * (a + b)
    x_new = float(np.clip(0.5 * (xc + mid), xlo, xhi))
    if abs(x_new - x0) < 1e-12:
        return False

    new_aff = base + ca0 + ca1 * x_new + ca2 * x_new * x_new
    new_global = float(np.max(new_aff))
    if max_unaff > new_global:
        new_global = max_unaff
    if new_global > Ccur + 1e-10:
        return False
    if new_global >= Ccur - 1e-12:
        return False

    c[aff] = new_aff
    h[p] = x_new
    h[q] = s - x_new
    return True


def star_polish(h, n, deadline, rng):
    h = project_feasible(h.copy(), n)
    c = compute_c_array(h, n)
    dx = 2.0 / float(n)
    hubs = [n - 1, 0, n // 2]
    max_updates = 100 if n <= 800 else 50

    for hub in hubs:
        active_tol = 1e-6
        prev_c = float(np.max(c))
        passes = 0
        while time.time() < deadline - 25 and passes < 80:
            passes += 1
            C = float(np.max(c))
            if C < prev_c - 1e-12:
                prev_c = C
                active_tol = 1e-6
            else:
                active_tol *= 2.0
                if active_tol > 1e-3:
                    break

            hot = np.zeros(n, dtype=bool)
            act = np.where(c >= C - active_tol)[0]
            for rr in act:
                sa = abs(int(rr - (n - 1)))
                if sa == 0:
                    hot.fill(True)
                else:
                    hot[0:n - sa] = True
                    hot[sa:n] = True
            if not np.any(hot):
                hot.fill(True)
            hub_hot = bool(hot[hub])

            order = np.arange(n)
            rng.shuffle(order)
            updates = 0
            for probe, p in enumerate(order):
                if time.time() > deadline - 25:
                    break
                if p == hub:
                    continue
                if not (hot[p] or hub_hot):
                    continue
                a = min(p, hub)
                b = max(p, hub)
                st = n - 1 - b
                en = 2 * n - 2 - a
                if en < st:
                    continue
                if float(np.max(c[st:en + 1])) < C - active_tol:
                    continue
                if pair_cd(h, c, n, p, hub, dx):
                    updates += 1
                    if updates % 20 == 0:
                        C = float(np.max(c))
                if updates >= max_updates:
                    break

            if passes % 10 == 0:
                if abs(float(h.sum()) - float(n) / 2.0) > 1e-10:
                    h = project_feasible(h, n)
                    c = compute_c_array(h, n)

    h = project_feasible(h, n)
    c = compute_c_array(h, n)
    return h, float(np.max(c))


def interp_resample(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 10.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
        if n0 < 2:
            raise RuntimeError
    except Exception:
        n0 = 600
        h0 = np.full(n0, 0.5)

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0

    try:
        smooth_days = min(260, max(60, int(0.24 * budget_s)))
        smooth_dl = t0 + smooth_days
        h1, c1 = lbfgs_polish(best_h, best_n, smooth_dl, rng)
        if c1 < best_c - 1e-12:
            best_h, best_c, best_n = h1, c1, best_n

        main_dl = deadline - 220.0
        h2, c2 = star_polish(best_h, best_n, main_dl, rng)
        if c2 < best_c - 1e-12:
            best_h, best_c, best_n = h2, c2, best_n

        for n_try in (800, 1000):
            if time.time() > deadline - 170:
                break
            if n_try == best_n:
                continue
            h_try = interp_resample(best_h, best_n, n_try)
            if time.time() < deadline - 130:
                h_try, c_try = lbfgs_polish(h_try, n_try, min(deadline - 100, time.time() + 90), rng)
            else:
                c_try = compute_c5(h_try, n_try)
            if time.time() < deadline - 45:
                h_try, c_try = star_polish(h_try, n_try, deadline - 40, rng)
            if c_try < best_c - 1e-12:
                best_h, best_c, best_n = h_try, c_try, n_try

        if time.time() < deadline - 8:
            hf, cf = star_polish(best_h, best_n, deadline - 3, rng)
            if cf < best_c - 1e-12:
                best_h, best_c, best_n = hf, cf, best_n
    except Exception:
        pass

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
