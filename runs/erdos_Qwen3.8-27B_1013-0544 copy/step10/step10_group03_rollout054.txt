def power_mean_obj(h, n, p):
    """Smooth approximation of max using power mean."""
    c = np.correlate(h, 1.0 - h, mode='full')

    # But c can be negative, so use shifted version
    m = c.max()
    shifted = c - m + 1.0  # ensure positive

    return pm * (2.0 / n)

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):
    # Same as before - this is the key CD subroutine
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        zero_mask = np.abs(a) <= 1e-15
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        neg_mask = a < -1e-15
        pos_mask = a > 1e-15
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc < 0:
                feasible = False
            else:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if xc_lo > xc_hi:
                    feasible = False
                elif not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if h0 - r1 <= r2 - h0:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def _cf(x):
            return x.sum() - n / 2.0
        def _cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        zero_mask = np.abs(a) <= 1e-15
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        neg_mask = a < -1e-15
        pos_mask = a > 1e-15
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc < 0:
                feasible = False
            else:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if xc_lo > xc_hi:
                    feasible = False
                elif not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if h0 - r1 <= r2 - h0:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def _cf(x):
            return x.sum() - n / 2.0
        def _cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Initial CD to converge
    if time.time() < deadline - 550:
        h1, c1 = exact_cd(h_best, n, 250, deadline - 550)
        if c1 < best_c:
            h_best, best_c = h1, c1

    # Phase 2: SLSQP with increasing smoothness
    if time.time() < deadline - 400:
        h2, c2 = slsqp_opt(h_best, n, [5000, 20000, 80000, 250000], 600, deadline - 400)
        if c2 < best_c:
            h_best, best_c = h2, c2

    # Phase 3: CD refinement
    if time.time() < deadline - 300:
        h3, c3 = exact_cd(h_best, n, 300, deadline - 300)
        if c3 < best_c:
            h_best, best_c = h3, c2 if False else h3, c3
        n = len(h_best)

    # Phase 4: Coarsen-refine (multigrid)
    for cf in [2, 3]:
        if time.time() > deadline - 200:
            break
        nc = n // cf
        if nc < 100:
            continue
        hc = np.array([h_best[i*cf:(i+1)*cf].mean() for i in range(nc)])
        hc = project_feasible(hc, nc)
        hc, cc = slsqp_opt(hc, nc, [5000, 20000], 200, deadline - 200)
        hc, cc = exact_cd(hc, nc, 100, deadline - 200)
        x_c = np.linspace(0, 2, nc)
        x_f = np.linspace(0, 2, n)
        h_r = project_feasible(np.interp(x_f, x_c, hc), n)
        h_r, c_r = exact_cd(h_r, n, 200, deadline - 200)
        if c_r < best_c:
            h_best, best_c = h_r, c_r

    # Phase 5: Multi-n exploration
    n_list = [500, 550, 600, 650, 750, 800, 900, 1000, 1100]
    for n_try in n_list:
        if time.time() > deadline - 150:
            break
        if n_try == n or n_try < 200:
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = slsqp_opt(h_int, n_try, [8000, 30000], 200, deadline - 150)
        h_o, c_o = exact_cd(h_o, n_try, 150, deadline - 150)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o
            n = n_try

    # Phase 6: Structured perturbations + optimization
    n = len(h_best)
    for r in range(30):
        if time.time() > deadline - 50:
            break
        strategy = r % 6
        if strategy == 0:
            s = 0.003 + 0.008 * (r // 6)
            hp = project_feasible(h_best + rng.normal(0, s, n), n)
        elif strategy == 1:
            bs = rng.randint(0, max(1, n - 35))
            bl = rng.randint(8, 35)
            hp = h_best.copy()
            hp[bs:bs+bl] = 1.0 - hp[bs:bs+bl]
            hp = project_feasible(hp, n)
        elif strategy == 2:
            hp = h_best.copy()
            for _ in range(rng.randint(2, 5)):
                s_ = rng.randint(0, max(1, n - 15))
                l_ = rng.randint(5, 15)
                hp[s_:s_+l_] = np.clip(rng.uniform(0, 1), 0, 1)
            hp = project_feasible(hp, n)
        elif strategy == 3:
            shift = rng.randint(3, 20)
            w = rng.uniform(0.3, 0.7)
            hp = w * h_best + (1 - w) * np.roll(h_best, shift)
            hp = project_feasible(hp, n)
        elif strategy == 4:
            # Push towards binary
            hp = h_best.copy()
            th = 0.5
            hp[hp < th] = np.maximum(0, hp[hp < th] - 0.15)
            hp[hp >= th] = np.minimum(1, hp[hp >= th] + 0.15)
            hp = project_feasible(hp, n)
        else:
            # Regional inversion with smooth blending
            center = rng.randint(0, n)
            radius = rng.randint(15, 40)
            dist = np.abs(np.arange(n) - center).astype(float) / radius
            weight = np.exp(-dist * dist)
            hp = h_best + weight * (1.0 - 2.0 * h_best) * rng.uniform(0.5, 1.0)
            hp = project_feasible(hp, n)

        h_o, c_o = slsqp_opt(hp, n, [10000, 50000], 120, deadline - 50)
        h_o, c_o = exact_cd(h_o, n, 100, deadline - 50)
        if c_o < best_c:
            h_best, best_c = h_o, c_o
            n = len(h_best)

    # Phase 7: Final intensive CD
    if time.time() < deadline - 2:
        h_f, c_f = exact_cd(h_best, n, 400, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
