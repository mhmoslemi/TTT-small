    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if xc_lo < r1 and xc_hi > r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return c
    ms, a, has_diag, diag_idx = get_coefficients(h, p, n)
    c[ms] += delta * a
    if has_diag:
        c[ms[diag_idx[0]]] -= delta * delta
    return c

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = list(range(n))
        if rng is not None and it % 3 == 1:
            rng.shuffle(order)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def cf(x):
            return x.sum() - n / 2.0
        def cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def structured_init_h(n, freq, phase, amp):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.sin(np.pi * freq * x + phase)
    return project_feasible(h, n)

def random_binary_init(n, rng):
    h = rng.randint(0, 2, n).astype(float)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Initial CD
    h1, c1 = exact_cd(h_best, n, 150, deadline, rng)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: SLSQP refinement
    if time.time() < deadline - 200:
        h2, c2 = slsqp_opt(h_best, n, [5000, 20000, 60000], 250, deadline)
        if c2 < best_c - 1e-12:
            h_best, best_c = h2, c2
        h3, c3 = exact_cd(h_best, n, 150, deadline, rng)
        if c3 < best_c - 1e-12:
            h_best, best_c = h3, c3

    # Phase 3: Multi-resolution search
    n_list = [n, 650, 550, 500, 750, 450, 850, 600, 900, 400]
    for n_try in n_list:
        if time.time() > deadline - 200:
            break
        h_int = interp_h(h_best, len(h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 120, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o
        if time.time() < deadline - 100:
            h_o2, c_o2 = slsqp_opt(h_o, n_try, [20000, 80000], 150, deadline)
            if c_o2 < best_c - 1e-9:
                h_best, best_c = h_o2, c_o2
            h_o3, c_o3 = exact_cd(h_o2, n_try, 100, deadline, rng)
            if c_o3 < best_c - 1e-9:
                h_best, best_c = h_o3, c_o3

    # Phase 4: Diverse structured initializations
    n_base = len(h_best)
    for (freq, amp) in [(2, 0.2), (3, 0.15), (1, 0.3), (4, 0.12), (2, 0.1), (3, 0.25)]:
        if time.time() > deadline - 100:
            break
        h_s = structured_init_h(n_base, freq, rng.uniform(0, 2 * np.pi), amp)
        h_o, c_o = exact_cd(h_s, n_base, 80, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o

    # Phase 5: Annealed restarts with varying noise levels
    for r in range(40):
        if time.time() > deadline - 80:
            break
        scale = 0.025 * (1.0 - r / 40.0) + 0.003
        hp = project_feasible(h_best + rng.normal(0, scale, n_base), n_base)
        h_o, c_o = exact_cd(hp, n_base, 80, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o
            # If improvement found, do SLSQP polish
            if time.time() < deadline - 60:
                h_p, c_p = slsqp_opt(h_best, n_base, [30000], 100, deadline)
                if c_p < best_c - 1e-12:
                    h_best, best_c = h_p, c_p

    # Phase 6: Final polish with long CD
    n_final = len(h_best)
    h_f, c_f = exact_cd(h_best, n_final, 300, deadline, rng)
    if c_f < best_c - 1e-12:
        h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
