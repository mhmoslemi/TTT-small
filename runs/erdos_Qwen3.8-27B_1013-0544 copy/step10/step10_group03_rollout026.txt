    c = np.correlate(x, 1.0 - x, mode='full')

    if m <= 0:

    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        a[diag_idx] = 0.0
        c_diag = c_base[diag_idx[0]]
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    L_lo, L_hi = unaff, cur_max
    for _ in range(60):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        zero_mask = np.abs(a) <= 1e-15
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible:
            neg_mask = a < -1e-15
            pos_mask = a > 1e-15
            if np.any(neg_mask):
                x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
            if np.any(pos_mask):
                x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if h0 - r1 <= r2 - h0:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    if m <= 1e-15:
        return 0.0
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x, nn=n: x.sum() - nn / 2.0,
                 'jac': lambda x, nn=n: np.ones(nn)}]
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def shifted_perturb(h, n, rng, shift_amt):
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n)
    h_shifted = np.interp(x_new - shift_amt, x_old, h, left=h[0], right=h[-1])
    return project_feasible(h_shifted, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: SLSQP warm-up on parent
    if time.time() < deadline - 500:
        h1, c1 = slsqp_opt(h_best, n, [3000, 15000, 80000], 400, deadline - 500)
        if c1 < best_c:
            h_best, best_c = h1, c1

    # Phase 2: CD refinement
    if time.time() < deadline - 400:
        h2, c2 = exact_cd(h_best, n, 300, deadline - 400)
        if c2 < best_c:
            h_best, best_c = h2, c2

    # Phase 3: Multi-n exploration with generous time
    n_list = [400, 500, 600, 700, 800, 900, 1000, 1100, 1300]
    for n_try in n_list:
        if time.time() > deadline - 300:
            break
        if n_try == len(h_best):
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        # Quick SLSQP + CD
        h_o, c_o = slsqp_opt(h_int, n_try, [5000, 30000], 150, deadline - 300)
        h_o, c_o = exact_cd(h_o, n_try, 200, deadline - 300)
        if c_o < best_c - 1e-10:
            h_best, best_c = h_o, c_o

    # Phase 4: Shift perturbations + CD
    n = len(h_best)
    for r in range(12):
        if time.time() > deadline - 200:
            break
        shift = (rng.rand() - 0.5) * 0.1
        hp = shifted_perturb(h_best, n, rng, shift)
        h_o, c_o = exact_cd(hp, n, 150, deadline - 200)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 5: Gaussian perturbations with increasing scale
    for r in range(15):
        if time.time() > deadline - 150:
            break
        s = 0.002 + 0.012 * r
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        h_o, c_o = slsqp_opt(hp, n, [20000], 100, deadline - 150)
        h_o, c_o = exact_cd(h_o, n, 120, deadline - 150)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 6: Binary push with varying strengths
    n = len(h_best)
    for strength in [0.02, 0.04, 0.06, 0.10, 0.15]:
        if time.time() > deadline - 100:
            break
        hb = h_best.copy()
        mask = hb < 0.5
        hb[mask] = np.maximum(0.0, hb[mask] - strength)
        hb[~mask] = np.minimum(1.0, hb[~mask] + strength)
        hb = project_feasible(hb, n)
        h_o, c_o = exact_cd(hb, n, 200, deadline - 100)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 7: Final intensive CD
    if time.time() < deadline - 5:
        n = len(h_best)
        h_f, c_f = exact_cd(h_best, n, 500, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
