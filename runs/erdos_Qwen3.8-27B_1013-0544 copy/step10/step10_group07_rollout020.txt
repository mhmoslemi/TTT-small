def project_sum(h, target):
    h=np.clip(h,0.0,1.0)
    target=float(target)
    for _ in range(30):
        s=float(np.sum(h))
        diff=target-s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room=1.0-h
            tot=float(room.sum())
            if tot <= 1e-15: break
            h = h + diff * room / tot
            np.clip(h,0,1,out=h)
        else:
            tot=float(h.sum())
            if tot <= 1e-15: break
            h = h + diff * h / tot
            np.clip(h,0,1,out=h)
    # final exact-ish adjust
    s=float(np.sum(h)); diff=target-s
    if abs(diff)>1e-15:
        idx=np.where((h>1e-12)&(h<1-1e-12))[0]
        if len(idx): h[idx[0]] += diff
    return h
```

    M=max(p,n-1-p)
    k=np.arange(-M,M+1)
    idx=n-1+k
    a=np.zeros(M*2+1)
    fwd = (k >= p-(n-1)) & (k <= p)
    if np.any(fwd):
        a[fwd] += 1.0 - h[p - k[fwd]]
    bwd = (k >= -p) & (k <= n-1-p)
    if np.any(bwd):

    dpos = int((M)) # k=0 at index M

import numpy as np
import time


def fast_c(h):
    return np.correlate(h, 1.0 - h, mode="full")


def c5_value(cnt, n):
    return float(cnt) * (2.0 / float(n))


def project_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = float(target)
    for _ in range(60):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot <= 1e-18:
                break
            h += diff * room / tot
            np.clip(h, 0.0, 1.0, out=h)
        else:
            tot = float(h.sum())
            if tot <= 1e-18:
                break
            h += diff * (h / tot)
            np.clip(h, 0.0, 1.0, out=h)
    return h


def resample_solution(h, n):
    old = len(h)
    if old == n:
        return project_sum(h.copy(), 0.5 * n)
    xo = np.linspace(0.0, 2.0, old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n, endpoint=False)
    return project_sum(np.interp(xn, xo, h), 0.5 * n)

def merge_ivals(ivals):
    if not ivals:
        return []
    ivals = sorted(ivals)
    out = [[ivals[0][0], ivals[0][1]]]
    for l, r in ivals[1:]:
        if l <= out[-1][1] + 1e-12:

    res = []

        ll = lin[linm]
        cc = cst[linm]
        pos = ll > 1e-14
        if np.any(pos):
            hi = min(hi, float(np.min((L - cc[pos]) / ll[pos])))
        neg = ll < -1e-14
        if np.any(neg):
            lo = max(lo, float(np.max((L - cc[neg]) / ll[neg])))
        zero = np.abs(ll) <= 1e-14
        if np.any(zero) and np.any(cc[zero] > L + 1e-11):
            return []

    conv = q > 1e-12
    if np.any(conv):
        qq = q[conv]
        ll = lin[conv]
        cc = cst[conv]
        D = ll * ll - 4.0 * qq * (cc - L)
        if float(np.min(D)) < -1e-10:
            return []
        sq = np.sqrt(np.clip(D, 0.0, None))
        r1 = (-ll - sq) / (2.0 * qq)
        r2 = (-ll + sq) / (2.0 * qq)
        lo = max(lo, float(np.max(r1)))
        hi = min(hi, float(np.min(r2)))

    if lo > hi + 1e-12:
        return []

    new = []
    for l, h in regs:
        ll = max(l, lo)
        hh = min(h, hi)
        if hh >= ll - 1e-12:
            new.append((ll, hh))
    return merge_ivals(new)


def pair_update(c, h, n, p, q, iters=24):
    if p == q or n < 2:
        return False
    a0 = float(h[p])
    b0 = float(h[q])
    s = a0 + b0
    lo0 = max(0.0, s - 1.0)
    hi0 = min(1.0, s)
    if hi0 - lo0 < 1e-12:
        return False

    Mp = max(p, n - 1 - p)
    Mq = max(q, n - 1 - q)
    M = max(Mp, Mq)
    Lk = 2 * M + 1
    k = np.arange(-M, M + 1)
    idx = (n - 1) + k

    A = np.zeros(Lk)
    B = np.zeros(Lk)
    G = np.zeros(Lk)
    selfQ = np.zeros(Lk)

    kl = max(-M, p - (n - 1))
    kr = min(M, p)
    if kl <= kr:
        ks = np.arange(kl, kr + 1)
        pos = ks + M
        j = p - ks
        selfp = (j == p)
        cross = (j == q)
        fix = ~selfp & ~cross
        if np.any(fix):
            A[pos[fix]] += 1.0 - h[j[fix]]
        if np.any(selfp):
            A[pos[selfp]] += 1.0 - 2.0 * a0
            selfQ[pos[selfp]] -= 1.0
        if np.any(cross):
            A[pos[cross]] += 1.0
            G[pos[cross]] -= 1.0

    kl = max(-M, -p)
    kr = min(M, n - 1 - p)
    if kl <= kr:
        ks = np.arange(kl, kr + 1)
        pos = ks + M
        i = p + ks
        fix = (i != p) & (i != q)
        if np.any(fix):
            A[pos[fix]] += -h[i[fix]]

    kl = max(-M, q - (n - 1))
    kr = min(M, q)
    if kl <= kr:
        ks = np.arange(kl, kr + 1)
        pos = ks + M
        j = q - ks
        selfq = (j == q)
        cross = (j == p)
        fix = ~selfq & ~cross
        if np.any(fix):
            B[pos[fix]] += 1.0 - h[j[fix]]
        if np.any(selfq):
            B[pos[selfq]] += 1.0 - 2.0 * b0
            selfQ[pos[selfq]] -= 1.0
        if np.any(cross):
            B[pos[cross]] += 1.0
            G[pos[cross]] -= 1.0

    kl = max(-M, -q)
    kr = min(M, n - 1 - q)
    if kl <= kr:
        ks = np.arange(kl, kr + 1)
        pos = ks + M
        i = q + ks
        fix = (i != p) & (i != q)
        if np.any(fix):
            B[pos[fix]] += -h[i[fix]]

    qcoef = -G + selfQ
    lincoef = A - B + G * s - 2.0 * selfQ * a0
    cbase = c[idx] - (A - B) * a0 - G * a0 * b0 + selfQ * a0 * a0

    left = n - 1 - M
    Llo = 0.0
    if left > 0:
        Llo = max(Llo, float(np.max(c[:left])))
    rs = n + M
    if rs <= 2 * n - 2:
        Llo = max(Llo, float(np.max(c[rs:])))
    Lhi = float(np.max(c))
    if Lhi < Llo:
        Lhi = Llo

    for _ in range(iters):
        L = 0.5 * (Llo + Lhi)
        if pair_regions(L, lo0, hi0, qcoef, lincoef, cbase):
            Lhi = L
        else:
            Llo = L

    L = Lhi + 1e-10 * max(1.0, Lhi) + 1e-12
    regs = pair_regions(L, lo0, hi0, qcoef, lincoef, cbase)
    if not regs:
        return False

    besta = None
    bestd = 1e100
    for l, r in regs:
        x = min(max(a0, l), r)
        d = abs(x - a0)
        if d < bestd:
            bestd = d
            besta = x
    if besta is None:
        return False
    a = float(np.clip(besta, lo0, hi0))
    if abs(a - a0) < 1e-13:
        return False
    nz = s - a
    val = cbase + lincoef * a + qcoef * a * a
    c[idx] = np.maximum(val, 0.0)
    h[p] = a
    h[q] = nz
    return True


def pair_pass(h, c, n, rng, deadline, iters=22, max_pairs=None):
    if n < 2:
        return
    if max_pairs is None:
        if n <= 360:
            max_pairs = n - 1
        else:
            max_pairs = max(40, n // 2)
    max_pairs = min(int(max_pairs), n - 1)
    if max_pairs <= 0:
        return
    perm = rng.permutation(n - 1)
    for i in range(max_pairs):
        if time.time() > deadline:
            return
        p = int(perm[i])
        q = p + 1
        pair_update(c, h, n, p, q, iters)


def optimize_instance(h, n, rng, deadline, iters=26, pair_iters=22, max_stag=7, do_1d=True):
    target = 0.5 * float(n)
    h = project_sum(h, target)
    c = fast_c(h)
    best_h = h.copy()
    best_counts = float(c.max())
    stag = 0

    if do_1d and time.time() < deadline - 30:
        c = fast_c(h)
        order = rng.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            coordinate_update(c, h, n, p, iters=iters)
        h = project_sum(h, target)
        c = fast_c(h)
        cur = float(c.max())
        if cur < best_counts - 1e-12:
            best_counts = cur
            best_h = h.copy()
            stag = 0
        elif cur > best_counts + 1e-7:
            h = best_h.copy()
            c = fast_c(h)
            best_counts = float(c.max())

    while time.time() < deadline:
        if time.time() > deadline - 8:
            break
        pair_pass(h, c, n, rng, min(deadline, time.time() + max(10.0, 0.02 * n)), iters=pair_iters)
        h = project_sum(h, target)
        c = fast_c(h)
        cur = float(c.max())
        if cur < best_counts - 1e-12:
            best_counts = cur
            best_h = h.copy()
            stag = 0
        else:
            if cur > best_counts + 1e-7:
                h = best_h.copy()
                c = fast_c(h)
                best_counts = float(c.max())
            stag += 1
            if stag >= max_stag:
                break

    h = project_sum(best_h, target)
    c = fast_c(h)
    return h, float(c.max())


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        budget_s = max(1.0, float(budget_s))
    except Exception:
        budget_s = 1100.0
    deadline = t0 + budget_s - 10.0
    rng = np.random.RandomState(int(seed))

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    h0 = project_sum(h0, 0.5 * n0)
    c0 = fast_c(h0)
    best_h = h0.copy()
    best_n = n0
    best_c = c5_value(float(c0.max()), n0)

    # Initial deep polish on the supplied construction.
    if time.time() < deadline - 60:
        limit = min(420.0, deadline - time.time() - 120)
        if limit > 40:
            hh, cnt = optimize_instance(h0, n0, rng, time.time() + limit)
            val = c5_value(cnt, n0)
            if val + 1e-12 < best_c:
                best_h, best_n, best_c = hh, n0, val

    # Resolution sweep: coarser grids are much easier to polish and can beat the parent grid.
    ns = []
    if n0 > 0:
        ns.extend([n0, int(round(0.75 * n0)), int(round(0.55 * n0)), int(round(0.35 * n0)), 240, 160, 120])
    ns = sorted(set(max(24, int(x)) for x in ns if x > 24), reverse=True)
    for ntry in ns:
        rem = deadline - time.time()
        if rem < 70:
            break
        limit = min(80.0, rem - 35)
        if limit < 30:
            break
        hstart = resample_solution(best_h, ntry)
        hh, cnt = optimize_instance(hstart, ntry, rng, time.time() + limit, iters=24, pair_iters=20, max_stag=6)
        val = c5_value(cnt, ntry)
        if val + 1e-12 < best_c:
            best_h, best_n, best_c = hh, ntry, val

    # Diversification: small, algorithmically generated starts.
    for r in range(14):
        rem = deadline - time.time()
        if rem < 55:
            break
        limit = min(45.0, rem - 25)
        if limit < 22:
            break
        if r % 3 == 0:
            ntry = best_n if best_n > 0 else 240
            hstart = resample_solution(best_h, ntry)
            hstart = project_sum(hstart + rng.normal(0.0, 0.008 + 0.002 * r, ntry), 0.5 * ntry)
        elif r % 3 == 1:
            ntry = 160 + 32 * (r % 4)
            hstart = project_sum((rng.rand(ntry) < 0.5).astype(float), 0.5 * ntry)
        else:
            ntry = 180 + 24 * (r % 5)
            x = np.linspace(0.0, 2.0, ntry, endpoint=False)
            f = rng.uniform(1.0, 6.0)
            ph = rng.uniform(0.0, 6.2831853)
            hstart = np.clip(0.5 + 0.5 * np.sin(2.0 * 3.14159265 * f * x + ph), 0.0, 1.0)
            hstart = project_sum(hstart, 0.5 * ntry)
        hh, cnt = optimize_instance(hstart, ntry, rng, time.time() + limit, iters=22, pair_iters=18, max_stag=5, do_1d=True)
        val = c5_value(cnt, ntry)
        if val + 1e-12 < best_c:
            best_h, best_n, best_c = hh, ntry, val

    # Final polish on the current best resolution.
    rem = deadline - time.time()
    if rem > 40:
        hh, cnt = optimize_instance(best_h, best_n, rng, deadline - 2, iters=26, pair_iters=22, max_stag=8, do_1d=False)
        val = c5_value(cnt, best_n)
        if val + 1e-12 < best_c:
            best_h, best_c = hh, val

    best_h = project_sum(best_h, 0.5 * best_n)
    c = fast_c(best_h)
    return best_h, float(c.max()) * (2.0 / float(best_n)), int(best_n)
