def cd_full(h, n, max_passes, deadline, rng, damping=1.0):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        improved = False
        order = range(n) if rng is None else rng.permutation(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            delta = (xn - h0) * damping
            if abs(delta) > 1e-13:
                update_c(c, h, p, delta, n)
                h[p] = h0 + delta
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def subgradient_dir(h, n, rng):
    c = np.correlate(h, 1.0 - h, mode='full')
    max_val = c.max()
    active = np.where(c >= max_val - 1e-8)[0]
    k_idx = active[rng.randint(len(active))]
    k = k_idx - (n - 1)
    grad = np.zeros(n)
    kk = abs(k)
    for i in range(n - kk):
        grad[i] += (1.0 - h[i + kk])
        grad[i + kk] -= h[i]
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return np.zeros(n)
    return -grad / gn

def subgrad_descent(h, n, deadline, rng, max_iters=500):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.005
    for i in range(max_iters):
        if time.time() > deadline:
            break
        d = subgradient_dir(h, n, rng)
        h_try = project_feasible(h + step * d, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-12:
            h = h_try
            best_h = h_try.copy()
            best_c = c_try
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.9
        if i % 30 == 29:
            h_cd, c_cd = cd_full(h, n, 15, deadline, rng)
            if c_cd < best_c:
                h = h_cd
                best_h = h.copy()
                best_c = c_cd
    return best_h, best_c

def lse_obj_fn(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def lse_con_fun(x, n_holder):
    return x.sum() - n_holder[0] / 2.0

def lse_con_jac(x, n_holder):
    return np.ones(len(x))

def slsqp_opt(h0, n, alphas, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    n_holder = [n]
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lse_con_fun, 'jac': lse_con_jac, 'args': (n_holder,)}]
        try:
            res = minimize(lse_obj_fn, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interpolate_h(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)
    if time.time() > deadline:
        return h_best, best_c, n

    # Phase 1: CD warmup
    h_cd, c_cd = cd_full(h_best, n, 20, deadline, rng)
    if c_cd < best_c:
        h_best, best_c = h_cd, c_cd

    # Phase 2: SLSQP multiple alphas
    h_s, c_s = slsqp_opt(h_best, n, [2000, 10000, 50000, 200000], 300, deadline)
    if c_s < best_c:
        h_best, best_c = h_s, c_s

    # Phase 3: CD refinement
    h_cd, c_cd = cd_full(h_best, n, 100, deadline, rng)
    if c_cd < best_c:
        h_best, best_c = h_cd, c_cd

    # Phase 4: Subgradient descent to escape local min
    h_sg, c_sg = subgrad_descent(h_best, n, deadline, rng, max_iters=300)
    if c_sg < best_c:
        h_best, best_c = h_sg, c_sg

    # Phase 5: CD + SLSQP interleave
    if time.time() < deadline - 100:
        h_cd, c_cd = cd_full(h_best, n, 50, deadline, rng)
        if c_cd < best_c:
            h_best, best_c = h_cd, c_cd
        h_s, c_s = slsqp_opt(h_best, n, [100000, 500000, 2000000], 250, deadline)
        if c_s < best_c:
            h_best, best_c = h_s, c_s

    # Phase 6: Structured restarts
    for r in range(25):
        if time.time() > deadline - 60:
            break
        if r % 4 == 0:
            hp = project_feasible(h_best + rng.normal(0, 0.015, n), n)
        elif r % 4 == 1:
            bp = rng.randint(15, 60)
            st = rng.randint(0, n - bp)
            hp = h_best.copy()
            hp[st:st+bp] = 1.0 - hp[st:st+bp]
            hp = project_feasible(hp, n)
        elif r % 4 == 2:
            h_rand = project_feasible(rng.uniform(0.1, 0.9, n), n)
            hp = project_feasible(0.85 * h_best + 0.15 * h_rand, n)
        else:
            hp = project_feasible(h_best + 0.01 * rng.choice([-1, 1], n).astype(float), n)
        h_o, c_o = cd_full(hp, n, 40, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o

    # Phase 7: Try nearby n
    n_cur = len(h_best)
    for n_try in [n_cur - 10, n_cur + 10, n_cur - 20, n_cur + 20]:
        if time.time() > deadline - 50:
            break
        if n_try < 100:
            continue
        h_int = interpolate_h(h_best, n_cur, n_try)
        h_o, c_o = cd_full(h_int, n_try, 80, deadline, rng)
        h_o, c_o = slsqp_opt(h_o, n_try, [20000, 100000], 150, deadline)
        h_o, c_o = cd_full(h_o, n_try, 50, deadline, rng)
        if c_o < best_c - 1e-8:
            h_best, best_c = h_o, c_o
            n_cur = n_try

    # Phase 8: Final subgradient + CD
    if time.time() < deadline - 15:
        h_sg, c_sg = subgrad_descent(h_best, len(h_best), deadline, rng, max_iters=200)
        if c_sg < best_c:
            h_best, best_c = h_sg, c_sg

    # Phase 9: Final CD polish
    if time.time() < deadline - 3:
        h_f, c_f = cd_full(h_best, len(h_best), 200, deadline, rng)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
