def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    ac = alpha * c
    ac -= ac.max()
    w = np.exp(ac)
    w /= w.sum()

    # Actually: log(sum(exp(alpha*c))) / alpha = max(c) + log(sum(exp(alpha*(c-max(c))))) / alpha
    # = max(c) + log(sum(w)) / alpha ... wait
    # sum(exp(alpha*c)) = exp(alpha*max_c) * sum(exp(alpha*(c-max_c)))
    # log(sum(exp(alpha*c))) / alpha = max_c + log(sum(exp(alpha*(c-max_c)))) / alpha
    # = max_c + log(sum(exp(shifted))) / alpha where shifted = alpha*(c-max_c) ≤ 0
    # Since w = exp(shifted)/sum(exp(shifted)), we have sum(exp(shifted)) = sum(w)*sum(exp(shifted))/sum(w) = ...
    # Let me just compute it directly:
    ac2 = alpha * c
    ac2_max = ac2.max()
    lse = ac2_max + np.log(np.sum(np.exp(ac2 - ac2_max)))
    obj = lse / alpha * dx
    
    # penalty
    s = x.sum() - n / 2.0
    obj += penalty * s * s
    
    # gradient
    # d(obj)/dx = dx/alpha * sum_k exp(alpha*c_k) * dc_k/dx / sum_j exp(alpha*c_j)
    #           = dx * sum_k w_k * dc_k/dx  (where w_k = exp(alpha*c_k)/sum)

        grad[p] = dx * (np.dot(a, seg1) - np.dot(x, seg2))
    grad += 2.0 * penalty * s * np.ones(n)
    
    return obj, grad

            break

            new_c = compute_c5(new_h, n)
            if new_c < best_c:

        except Exception:
            pass
    return best_h, best_c

        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _obj_and_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    ac = alpha * c
    ac_max = ac.max()
    e = np.exp(ac - ac_max)
    lse = ac_max + np.log(e.sum())
    obj = lse / alpha * dx
    s = x.sum() - n / 2.0
    obj += penalty * s * s
    
    w = e / e.sum()
    grad = np.zeros(n)
    for p in range(n):
        seg1 = w[n-1+p : p-1 : -1]
        seg2 = w[n-1-p : 2*n-1-p]
        grad[p] = dx * (np.dot(1.0-x, seg1) - np.dot(x, seg2))
    grad += 2.0 * penalty * s
    return obj, grad

def _wrap_lbfgs(x, n, alpha, penalty):
    obj, grad = _obj_and_grad(x, n, alpha, penalty)
    return obj, grad

def lbfgs_opt(h, n, alphas, penalty, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(_wrap_lbfgs, best_h, args=(n, alpha, penalty),
                          jac=True, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-15,
                                   'gtol': 1e-10, 'maxls': 50})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def cd_solve_1d(c, h, p, n):
    """Solve for the optimal value of h[p] given others fixed, targeting min max c."""
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    # Diagonal term (where ms == n-1, i.e., offset = 0):
    # c[n-1] includes h[p]*(1-h[p]) = h[p] - h[p]^2
    # So when we change h[p] to x, the diagonal term changes from h0-h0^2 to x-x^2
    # The linear part a already accounts for (1-h[p]) and -h[p] contributions
    # Wait: at offset 0 (k=n-1): dc/dh[p] = (1-h[0]) - h[0] = 1-2*h[p]
    # But in our formula: a gets (1-h[p]) from vf (since jf=p when ms=n-1) and -h[p] from vb (since jb=p when ms=n-1)
    # So a[diag] = (1-h[p]) - h[p] = 1-2*h[p] ✓
    # But c[n-1] also has the quadratic term: h[p]*(1-h[p]) contributes to c[n-1]
    # Actually no! c[n-1] = sum_n h[n] * (1-h[n]) = sum_n h[n] - sum_n h[n]^2
    # The term for index p is h[p] - h[p]^2.
    # When we write c[n-1] = (constant) + h[p] - h[p]^2, the linear coefficient is 1 (not 1-2h[p])
    # and there's a -h[p]^2 term.
    # 
    # Wait, I think the issue is: in the correlation c[k] = sum_n h[n+offset]*(1-h[n]),
    # when offset=0: c[n-1] = sum_n h[n]*(1-h[n])
    # The contribution of h[p] to this is h[p]*(1-h[p]) = h[p] - h[p]^2
    # When we change h[p] to x: new contribution is x - x^2
    # Change: (x-x^2) - (h0-h0^2) = (x-h0) - (x^2-h0^2) = (x-h0)(1-x-h0)
    # 
    # In our linear model c[ms] ≈ c_base[ms] + a[ms] * (x - h0), we're missing the quadratic term.
    # The diagonal k=n-1 has: c[n-1] = c_base[n-1 - m_min] + (x-h0) - (x^2-h0^2)
    # = c_base[...] + (1-2*h0)*(x-h0) - (x-h0)^2 ... let me expand:
    # (x-x^2) - (h0-h0^2) = x - x^2 - h0 + h0^2
    # = (x-h0) - (x-h0)(x+h0) = (x-h0)(1 - x - h0)
    # = (x-h0)(1-h0) - (x-h0)*x
    # Hmm, this isn't linear in (x-h0).
    #
    # Actually: let δ = x - h0.
    # (x-x^2) - (h0-h0^2) = (h0+δ) - (h0+δ)^2 - h0 + h0^2
    # = h0+δ - h0^2 - 2*h0*δ - δ^2 - h0 + h0^2
    # = δ - 2*h0*δ - δ^2 = δ(1-2*h0) - δ^2
    #
    # So c[n-1] = c_base[diag] + (1-2*h0)*δ - δ^2
    # The linear part is (1-2*h0)*δ which matches a[diag]*δ where a[diag] = 1-2*h0 ✓
    # The quadratic part is -δ^2 which we handle separately.
    
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = a[diag_idx]  # stays the same (1-2*h0)
    
    # Unaffected correlations (outside [m_min, m_max])
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    
    # Bisection on L: find minimum L such that there exists x ∈ [0,1] with
    # all affected c values ≤ L and the quadratic constraint satisfied
    L_lo, L_hi = unaff, float(np.max(c))
    
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    
    for _ in range(60):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        possible = True
        x_lo, x_hi = 0.0, 1.0
        
        # For zero coefficient: c_base must be ≤ L
        if possible and np.any(zero_mask):
            if np.any(c_base[zero_mask] > L):
                possible = False
        
        # For negative a: c_base + a*δ ≤ L → δ ≥ (L-c_base)/a (since a<0, this is an upper bound on δ)
        # Wait: a*δ ≤ L - c_base → if a < 0: δ ≥ (L-c_base)/a (dividing by negative flips)
        # δ = x - h0, so x ≥ h0 + (L-c_base)/a
        # Since a < 0, (L-c_base)/a > 0 when L > c_base, < 0 when L < c_base
        # x ≥ h0 + (L-c_base)/a
        if possible and np.any(neg_mask):
            bounds_lo = (L - c_base[neg_mask]) / a[neg_mask] + h0
            x_lo = max(x_lo, float(np.max(bounds_lo)))
        
        # For positive a: a*δ ≤ L - c_base → δ ≤ (L-c_base)/a → x ≤ h0 + (L-c_base)/a
        if possible and np.any(pos_mask):
            bounds_hi = (L - c_base[pos_mask]) / a[pos_mask] + h0
            x_hi = min(x_hi, float(np.min(bounds_hi)))
        
        # Quadratic constraint: c_diag + (1-2*h0)*δ - δ^2 ≤ L
        # → δ^2 - (1-2*h0)*δ + (L - c_diag) ≥ 0
        # → (δ - r1)(δ - r2) ≤ 0 where r1,r2 are roots (for the ≤ 0 version)
        # Actually: δ^2 - (1-2h0)δ + (L-c_diag) ≥ 0
        # This means δ ≤ r1 or δ ≥ r2 where r1 ≤ r2 are roots of δ^2 - (1-2h0)δ + (L-c_diag) = 0
        if possible and has_diag:
            Cq = L - c_diag
            # δ^2 - (1-2h0)δ + Cq ≥ 0
            disc = (1-2*h0)**2 - 4*Cq
            if disc < 0:
                # Always positive (since leading coeff > 0), no constraint
                pass
            else:
                r1 = ((1-2*h0) - np.sqrt(disc)) / 2.0
                r2 = ((1-2*h0) + np.sqrt(disc)) / 2.0
                # Need δ ≤ r1 or δ ≥ r2
                # i.e., x ≤ h0+r1 or x ≥ h0+r2
                x_lo_quad = h0 + r2  # lower bound for the "upper" region
                x_hi_quad = h0 + r1  # upper bound for the "lower" region
                # Check if [x_lo, x_hi] intersects (-∞, x_hi_quad] ∪ [x_lo_quad, ∞)
                if x_hi > x_hi_quad and x_lo < x_lo_quad:
                    # Interval spans the gap (x_hi_quad, x_lo_quad)
                    # Need to check if it reaches either side
                    if x_lo <= x_hi_quad or x_hi >= x_lo_quad:
                        # It does reach one side, but we need to split
                        # For bisection feasibility, just check if the full interval works
                        # Actually for feasibility, we need [x_lo, x_hi] ⊆ (-∞, x_hi_quad] ∪ [x_lo_quad, ∞)
                        # This fails if x_lo > x_hi_quad AND x_hi < x_lo_quad
                        pass
                    else:
                        possible = False
                # More precise: the constraint is satisfied if x_lo ≥ x_lo_quad or x_hi ≤ x_hi_quad
                # For the bisection, we just need to check if there EXISTS x in [x_lo, x_hi] satisfying the constraint
                # This is true iff [x_lo, min(x_hi, x_hi_quad)] is non-empty OR [max(x_lo, x_lo_quad), x_hi] is non-empty
                if x_hi > x_hi_quad and x_lo < x_lo_quad:
                    # Check left side: x_lo ≤ x_hi_quad → need x_lo ≤ min(x_hi, x_hi_quad)
                    # x_lo ≤ x_hi_quad is true (we're in this branch)
                    # So left side is feasible if x_lo ≤ x_hi_quad, which is true
                    # But we also need the point to be in [x_lo, x_hi]
                    # The feasible set is [x_lo, x_hi_quad] ∪ [x_lo_quad, x_hi] (intersected with [x_lo, x_hi])
                    # Since x_lo ≤ x_hi_quad and x_hi_quad < x_lo_quad < x_hi (in this branch)
                    # Left part: [x_lo, x_hi_quad] - non-empty ✓
                    # So it's feasible
                    pass
                # Actually I realize the bisection just needs a yes/no answer.
                # The constraint δ ≤ r1 or δ ≥ r2 is violated iff r1 < δ < r2
                # i.e., h0+r1 < x < h0+r2
                # So the forbidden interval for x is (h0+r1, h0+r2)
                # Feasibility: [x_lo, x_hi] is NOT entirely contained in (h0+r1, h0+r2)
                # i.e., x_lo ≤ h0+r1 OR x_hi ≥ h0+r2
                if not (x_lo <= h0 + r1 or x_hi >= h0 + r2):
                    possible = False
        
        if possible:
            L_hi = L
        else:
            L_lo = L
    
    L = L_hi
    # Find x
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        bounds_lo = (L - c_base[neg_mask]) / a[neg_mask] + h0
        x_lo = max(x_lo, float(np.max(bounds_lo)))
    if np.any(pos_mask):
        bounds_hi = (L - c_base[pos_mask]) / a[pos_mask] + h0
        x_hi = min(x_hi, float(np.min(bounds_hi)))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    
    if has_diag:
        Cq = L - c_diag
        disc = (1-2*h0)**2 - 4*Cq
        if disc >= 0:
            r1 = ((1-2*h0) - np.sqrt(disc)) / 2.0
            r2 = ((1-2*h0) + np.sqrt(disc)) / 2.0
            # Forbidden: h0+r1 < x < h0+r2
            # Move x to nearest boundary
            if x_lo > h0 + r1 and x_hi < h0 + r2:
                # Entire feasible interval is forbidden - shouldn't happen if bisection worked
                mid = (h0 + r1 + h0 + r2) / 2
                if h0 - (h0+r1) < (h0+r2) - h0:
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
            elif x_hi > h0 + r1 and x_lo < h0 + r2:
                # Interval overlaps forbidden zone - pick the side closer to h0
                if x_lo <= h0 + r1:
                    x_hi = min(x_hi, h0 + r1)
                elif x_hi >= h0 + r2:
                    x_lo = max(x_lo, h0 + r2)
                else:
                    # x_lo > h0+r1 and x_hi < h0+r2: pick nearest
                    if h0 <= (h0+r1 + h0+r2)/2:
                        x_hi = min(x_hi, h0+r1)
                    else:
                        x_lo = max(x_lo, h0+r2)
    
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))
