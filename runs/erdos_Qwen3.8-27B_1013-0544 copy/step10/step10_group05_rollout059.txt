    return float(np.clip((x_lo + x_hi)/2.0, 0.0, 1.0))

def cd_pass(h, n, deadline, rng=None, skip_boundary=True):
    """One pass of coordinate descent."""
    c = np.correlate(h, 1.0-h, mode='full')
    improved = False
    order = rng.permutation(n) if rng is not None else np.arange(n)
    for p in order:
        if time.time() > deadline: break
        h0 = h[p]
        if skip_boundary and (h0 < 1e-10 or h0 > 1.0 - 1e-10):
            continue
        xn = solve_coord(c, h, p, n)
        if abs(xn - h0) > 1e-13:
            delta = xn - h0
            m_min = max(0, n-1-p)
            m_max = min(2*n-2, 2*n-2-p)
            ms = np.arange(m_min, m_max+1)
            jf = p + ms - (n-1)
            jb = p - ms + (n-1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb): a[vb] -= h[jb[vb]]
            c[ms] += delta * a
            di = np.where(ms == (n-1))[0]
            if len(di) > 0:
                c[ms[di[0]]] -= delta*delta
            h[p] = xn
            improved = True
    return h, c, improved

def cd_polish(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0-h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline: break
        h, c, improved = cd_pass(h, n, deadline, rng)
        if not improved: break
        h = project_feasible(h, n)
        if abs(h.sum() - n/2.0) > 1e-10:
            c = np.correlate(h, 1.0-h, mode='full')
    return h, compute_c5(h, n)

def slsqp_obj(x, n, alpha):
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse/alpha) * (2.0/n)

def slsqp_cons_fun(x):
    return x.sum() - len(x)/2.0

def slsqp_cons_jac(x):
    return np.ones(len(x))

def slsqp_step(h0, n, alpha, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_cons_fun, 'jac': slsqp_cons_jac}]
    try:
        res = minimize(slsqp_obj, best_h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    except:
        pass
    return best_h, best_c

def interpolate_h(h_src, n_src, n_tgt):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(xt, xs, h_src), n_tgt)
