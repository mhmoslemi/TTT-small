import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def lsq_obj_and_grad(h, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    exp_c = np.exp(alpha * c - alpha * m)
    Z = exp_c.sum()
    w = exp_c / Z
    val = (m + np.log(Z) / alpha) + mu * (h.sum() - n / 2.0) ** 2
    # Gradient
    # grad[i] = sum_k w[k] * dc[k]/dh[i] + 2*mu*(sum(h)-n/2)
    # dc[k]/dh[i] where c[k] = dx * sum_j h[j+k-n+1]*(1-h[j])
    # = dx * [(1-h[i-(k-n+1)]) if 0<=i-(k-n+1)<n - h[i+(k-n+1)] if 0<=i+(k-n+1)<n]
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        m_k = k - (n - 1)
        # Positive part: (1-h[i-m_k]) for i in [max(0,m_k), min(n, n+m_k))
        ilo = max(0, m_k)
        ihi = min(n, n + m_k) if m_k < 0 else min(n, n)  # need to be careful
        # Actually: 0 <= i-m_k < n => m_k <= i < m_k+n
        # So i in [max(0, m_k), min(n, m_k+n))
        ilo = max(0, m_k)
        ihi = min(n, m_k + n)
        if ilo < ihi:
            grad[ilo:ihi] += wk * (1.0 - h[ilo - m_k:ihi - m_k])
        # Negative part: -h[i+m_k] for 0 <= i+m_k < n => -m_k <= i < n-m_k
        ilo2 = max(0, -m_k)
        ihi2 = min(n, n - m_k)
        if ilo2 < ihi2:
            grad[ilo2:ihi2] -= wk * h[ilo2 + m_k:ihi2 + m_k]
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_ls(h, n, alphas, mus, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        for mu in mus:
            if time.time() > deadline:
                return best_h, best_c
            bounds = [(0.0, 1.0)] * n
            def _.fun(x):
                v, _ = lsq_obj_and_grad(x, n, alpha, mu)
                return v
            def _grad(x):
                _, g = lsq_obj_and_grad(x, n, alpha, mu)
                return g
            try:
                res = minimize(_fun, best_h, jac=_grad, method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
            except Exception:
                pass
    return best_h, best_c
