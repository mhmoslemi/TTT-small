import numpy as np
import time

def compute_counts(h):
    return np.correlate(h, 1.0 - h, mode="full")[::-1]

def compute_c5(h, n):
    return float(np.max(compute_counts(h)) * (2.0 / n))

def project_feasible(x, n):
    x = np.asarray(x, dtype=float)
    x = np.clip(x, 0.0, 1.0)
    target = 0.5 * n
    if abs(float(x.sum()) - target) <= 1e-12:
        return x
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    mid = 0.5 * (lo + hi)
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - mid, 0.0, 1.0)
    for _ in range(25):
        diff = target - y.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - y
            tot = room.sum()
            if tot < 1e-15:
                break
            y = np.clip(y + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = y.sum()
            if tot < 1e-15:
                break
            y = np.clip(y + (diff / tot) * y, 0.0, 1.0)
    return y

def weighted_grad(p, h, n):
    L = n - 1
    A = np.zeros(n)
    B = np.zeros(n)
    C = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx]
        if w <= 1e-12:
            continue
        d = idx - L
        if d == 0:
            A += w
            B += w * h
            C += w * h
        elif d > 0:
            nd = n - d
            A[:nd] += w
            B[:nd] += w * h[d:]
            C[d:] += w * h[:nd]
        else:
            e = -d
            A[e:] += w
            B[e:] += w * h[:n - e]
            C[:n - e] += w * h[e:]
    return A - B - C

def shift_descent(h, n, max_iter, deadline, K):
    h = project_feasible(np.asarray(h, dtype=float).copy(), n)
    c = compute_c5(h, n)
    best_h = h.copy()
    best_c = c
    k = int(min(K, 2 * n - 1))
    steps = (0.004, 0.01, 0.03, 0.1, 0.3, 1.0, 3.0, 8.0)
    no_improve = 0
    for _ in range(max_iter):
        if time.time() > deadline - 2.0:
            break
        m = compute_counts(h)
        if k >= m.size:
            idx = np.arange(m.size)
        else:
            idx = np.argpartition(-m, k - 1)[:k]
        p = np.zeros(m.size)
        p[idx] = 1.0 / k
        g = weighted_grad(p, h, n)
        gmax = float(np.max(np.abs(g)))
        if gmax < 1e-13:
            break
        d = g / gmax
        best_trial = None
        best_trial_c = c
        for s in steps:
            trial = project_feasible(h - s * d, n)
            tc = compute_c5(trial, n)

            no_improve = 0
            h = best_trial
            c = best_trial_c
            if c < best_c:

    for i in range(1, 97):
        t = lo + (hi - lo) * (i / 96.0)
        trial = (1.0 - t) * h + t * u
        c = compute_c5(trial, n)
        if c < best_c:
            best_c = c
            best_h = trial.copy()
    return best_h, best_c

def pair_local_search(h, n, max_trials, deadline, rng):
    h = project_feasible(np.asarray(h, dtype=float).copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_trials):
        if time.time() > deadline - 2.0:
            break
        i = int(rng.randint(0, n))
        j = int(rng.randint(0, n))
        if i == j:
            continue
        hi_i = h[i]
        hj = h[j]
        lo = max(-hi_i, hj - 1.0)
        hi = min(1.0 - hi_i, hj)
        if hi - lo < 1e-12:
            continue
        oldc = best_c
        best_d = 0.0
        best_local = oldc
        ds = (lo, hi, 0.5 * (lo + hi),
              lo + (hi - lo) * rng.random(),
              lo + (hi - lo) * rng.random(),
              lo + (hi - lo) * rng.random())
        for dd in ds:
            dd = float(np.clip(dd, lo, hi))
            h[i] = hi_i + dd
            h[j] = hj - dd
            c = compute_c5(h, n)
            if c < best_local - 1e-14:
                best_local = c
                best_d = dd
            h[i] = hi_i
            h[j] = hj
        if best_local < oldc - 1e-14:
            h[i] = hi_i + best_d
            h[j] = hj - best_d
            oldc = best_local
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(int(seed) if seed is not None else 42)
    try:
        h = np.asarray(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 700
        x = np.linspace(0.0, 2.0, n)
        h = 0.5 + 0.25 * np.sin(2.0 * np.pi * x) + 0.15 * np.sin(4.0 * np.pi * x)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    if time.time() < deadline - 60:
        h2, c2 = coord_descent(best_h, n, 2, deadline - 10, True, rng, 26, 0.1)
        if c2 < best_c:
            best_h, best_c = h2, c2

    dir_end = min(deadline - 120.0, t0 + 230.0)
    for K in (1, 2, 4, 8, 16, 32, 64, 128, 256):
        if time.time() > dir_end - 20:
            break
        h2, c2 = shift_descent(best_h, n, 70, dir_end, K)
        if c2 < best_c:
            best_h, best_c = h2, c2
            if time.time() < deadline - 80:
                h3, c3 = coord_descent(best_h, n, 1, deadline - 10, False, rng, 24, 0.06)
                if c3 < best_c:
                    best_h, best_c = h3, c3

    if time.time() < deadline - 80:
        h2, c2 = coord_descent(best_h, n, 3, min(deadline - 20, time.time() + 140.0), True, rng, 28, 0.08)
        if c2 < best_c:
            best_h, best_c = h2, c2

    if time.time() < deadline - 100:
        refs = [1.0 - best_h, best_h[::-1], (1.0 - best_h)[::-1], 1.0 - best_h[::-1]]
        for u in refs:
            if time.time() > deadline - 60:
                break
            nu, cu = line_search_ref(best_h, u, n)
            if cu < best_c:
                best_h, best_c = nu, cu
                if time.time() < deadline - 50:
                    h2, c2 = shift_descent(best_h, n, 50, deadline - 10, 16)
                    if c2 < best_c:
                        best_h, best_c = h2, c2
                if time.time() < deadline - 30:
                    h2, c2 = coord_descent(best_h, n, 1, deadline - 10, False, rng, 24, 0.05)
                    if c2 < best_c:
                        best_h, best_c = h2, c2

    if time.time() < deadline - 90:
        h2, c2 = pair_local_search(best_h, n, 350, deadline - 10, rng)
        if c2 < best_c:
            best_h, best_c = h2, c2
            if time.time() < deadline - 30:
                h3, c3 = coord_descent(best_h, n, 1, deadline - 10, False, rng, 22, 0.05)
                if c3 < best_c:
                    best_h, best_c = h3, c3

    if time.time() < deadline - 180:
        for sigma in (0.02, 0.05, 0.1, 0.2):
            if time.time() > deadline - 80:
                break
            h2 = project_feasible(best_h + sigma * rng.normal(0.0, 1.0, n), n)
            h2, c2 = shift_descent(h2, n, 40, deadline - 20, 32)
            h2, c2 = coord_descent(h2, n, 1, deadline - 10, True, rng, 20, 0.06)
            if c2 < best_c:
                best_h, best_c = h2, c2
            span = int(max(5, min(n - 5, int(n * (0.05 + 0.35 * rng.rand())))))
            if span >= n:
                continue
            st = int(rng.randint(0, n - span + 1))
            h3 = best_h.copy()
            h3[st:st + span] = 1.0 - h3[st:st + span]
            h3 = project_feasible(h3, n)
            h3, c3 = shift_descent(h3, n, 30, deadline - 20, 32)
            h3, c3 = coord_descent(h3, n, 1, deadline - 10, True, rng, 18, 0.06)
            if c3 < best_c:
                best_h, best_c = h3, c3

    if time.time() < deadline - 220:
        candidates = []
        if n != 1000:
            candidates.append(1000)
        if n != 1400 and time.time() < deadline - 380:
            candidates.append(1400)
        if n != 560:
            candidates.append(560)
        for n2 in candidates:
            if time.time() > deadline - 80:
                break
            if n2 <= 0 or n2 >= 3000:
                continue
            xo = np.linspace(0.0, 2.0, n)
            xn = np.linspace(0.0, 2.0, n2)
            hh = project_feasible(np.interp(xn, xo, best_h), n2)
            hh, cc = shift_descent(hh, n2, 50, deadline - 20, 16)
            hh, cc = shift_descent(hh, n2, 40, deadline - 20, 48)
            if n2 <= 800:
                hh, cc = coord_descent(hh, n2, 1, deadline - 10, True, rng, 20, 0.07)
            hh, cc = pair_local_search(hh, n2, 80, deadline - 5, rng)
            if cc < best_c:
                best_h, best_c, n = hh, cc, n2
                break

    if time.time() < deadline - 25:
        h2, c2 = coord_descent(best_h, n, 1, deadline - 5, False, rng, 30, 0.04)
        if c2 < best_c:
            best_h, best_c = h2, c2

    if time.time() < deadline - 8:
        h2, c2 = pair_local_search(best_h, n, 60, deadline - 1, rng)
        if c2 < best_c:
            best_h, best_c = h2, c2

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, float(best_c), int(n)
