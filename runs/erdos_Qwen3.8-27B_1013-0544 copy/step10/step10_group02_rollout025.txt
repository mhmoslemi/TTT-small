    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return c
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_i = np.where(ms == (n - 1))[0]
    c[ms] += delta * a
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta
    return c

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    prev_c = compute_c5(h, n)
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = list(range(n))
        if rng is not None and it % 3 == 1:
            rng.shuffle(order)
        elif rng is not None and it % 3 == 2:
            order.reverse()
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
        if it % 10 == 9:
            cur = compute_c5(h, n)
            if cur < prev_c - 1e-12:
                prev_c = cur
    return h, compute_c5(h, n)

def water_fill(h, n, max_iters, deadline, eta=0.01):
    """Gradient ascent on minimum correlation (water-filling)."""
    h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_min = int(np.argmin(c[:n + 1]))
        g = np.zeros(n)
        for j in range(n):
            s = 0.0
            i2 = j + k_min
            if i2 < n:
                s += h[j + k_min]
            i3 = j - k_min
            if i3 >= 0:
                s += h[j - k_min]
            g[j] = s
        g = g / (np.max(np.abs(g)) + 1e-15)
        h_new = np.clip(h + eta * g, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        c_old = compute_c5(h, n)
        if c_new < c_old:
            h = h_new
            eta = min(eta * 1.1, 0.05)
        else:
            for j in range(n):
                i2 = j - k_min
                if i2 < 0:
                    continue
                g[j] = h[j - k_min]
            g = g / (np.max(np.abs(g)) + 1e-15)
            h_new = np.clip(h - eta * g, 0.0, 1.0)
            h_new = project_feasible(h_new, n)
            if compute_c5(h_new, n) < c_old:
                h = h_new
            eta = max(eta * 0.8, 0.001)
    return h, compute_c5(h, n)

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    def smooth_lse(x, alpha):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * (2.0 / n)
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def cf(x):
            return x.sum() - n / 2.0
        def cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        try:
            res = minimize(smooth_lse, best_h, args=(alpha,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: CD on parent
    h1, c1 = exact_cd(h_best, n, 80, deadline, rng)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: SLSQP smooth refinement on parent
    h2, c2 = slsqp_opt(h_best, n, [3000, 15000, 60000], 300, deadline)
    if c2 < best_c - 1e-12:
        h_best, best_c = h2, c2
    h2b, c2b = exact_cd(h_best, n, 50, deadline, rng)
    if c2b < best_c:
        h_best, best_c = h2b, c2b

    # Phase 3: Multi-resolution exploration (higher n for tighter bound)
    n_current = len(h_best)
    n_list = [1200, 1400, 1050, 900, 1100]
    for n_try in n_list:
        if time.time() > deadline - 300:
            break
        if n_try <= n_current:
            continue
        h_int = interp_h(h_best, n_current, n_try)
        h_o, c_o = exact_cd(h_int, n_try, 250, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c, n_current = h_o, c_o, n_try
        elif time.time() < deadline - 150:
            h_o2, c_o2 = slsqp_opt(h_o, n_try, [20000, 80000], 200, deadline)
            if c_o2 < best_c - 1e-9:
                h_best, best_c, n_current = h_o2, c_o2, n_try
                h_o3, c_o3 = exact_cd(h_o2, n_try, 100, deadline, rng)
                if c_o3 < best_c:
                    h_best, best_c = h_o3, c_o3

    # Phase 4: Water-filling on current best
    n_cur = len(h_best)
    h_wf, c_wf = water_fill(h_best, n_cur, 300, deadline, eta=0.005)
    if c_wf < best_c - 1e-12:
        h_best, best_c = h_wf, c_wf
    h_wf2, c_wf2 = exact_cd(h_best, n_cur, 100, deadline, rng)
    if c_wf2 < best_c:
        h_best, best_c = h_wf2, c_wf2

    # Phase 5: Structured perturbations (sinusoidal + block)
    n_cur = len(h_best)
    freqs = [1, 2, 3, 5, 7, 11, 13, 17]
    amps = [0.015, 0.03, 0.05, 0.08]
    for f in freqs:
        if time.time() > deadline - 100:
            break
        for amp in amps[:2]:
            if time.time() > deadline - 50:
                break
            x = np.arange(n_cur)
            phase = rng.uniform(0, 2 * np.pi)
            hp = np.clip(h_best + amp * np.sin(2 * np.pi * f * x / n_cur + phase), 0, 1)
            hp = project_feasible(hp, n_cur)
            h_o, c_o = exact_cd(hp, n_cur, 80, deadline, rng)
            if c_o < best_c - 1e-9:
                h_best, best_c = h_o, c_o

    # Phase 6: Gaussian adaptive restarts
    n_cur = len(h_best)
    for r in range(20):
        if time.time() > deadline - 60:
            break
        scale = 0.025 * (1.0 - r / 20.0) + 0.004
        hp = project_feasible(h_best + rng.normal(0, scale, n_cur), n_cur)
        h_o, c_o = exact_cd(hp, n_cur, 80, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o

    # Phase 7: Final long CD polish
    n_cur = len(h_best)
    h_f, c_f = exact_cd(h_best, n_cur, 400, deadline, rng)
    if c_f < best_c - 1e-12:
        h_best, best_c = h_f, c_f

    # Final SLSQP micro-refinement
    h_s, c_s = slsqp_opt(h_best, n_cur, [100000, 500000], 150, deadline)
    if c_s < best_c - 1e-12:
        h_best, best_c = h_s, c_s
    h_f2, c_f2 = exact_cd(h_best, n_cur, 200, deadline, rng)
    if c_f2 < best_c - 1e-12:
        h_best, best_c = h_f2, c_f2

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
