import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

def solve_coord(c, h, p, n):
    h0 = h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(60):
        if L_hi - L_lo < 1e-16:
            break

        x_lo, x_hi = 0.0, 1.0

            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):

        if x_lo > x_hi + 1e-12:

                r1 = (a_d - sq) / 2.0
                r2 = (a_d + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi

    if np.any(neg_mask):

    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)

def cd_pass(h, c, n, order, deadline):
    improved = 0
    for p in order:
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-13 or h0 > 1.0 - 1e-13:
            continue
        xn = solve_coord(c, h, p, n)
        if abs(xn - h0) > 1e-13:
            delta = xn - h0
            m_min = max(0, n - 1 - p)
            m_max = min(2 * n - 2, 2 * n - 2 - p)
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c[ms] += delta * a
            diag_i = np.where(ms == (n - 1))[0]
            if len(diag_i) > 0:
                c[ms[diag_i[0]]] -= delta * delta
            h[p] = xn
            improved += 1
    return h, c, improved

def slsqp_step(h, n, alpha, maxiter, deadline):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * (2.0 / n)
    def obj_jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w /= w.sum()
        n2 = len(x)
        # dc[mes]/dx[p] for each mes
        # c[mes] = sum_i x[i]*(1-x[i+mes-(n2-1)])
        # dc[mes]/dx[p] = (1-x[p+mes-(n2-1)]) - x[p-mes+(n2-1)] (valid indices)
        g = np.zeros(n2)
        for p in range(n2):
            m_min = max(0, n2 - 1 - p)
            m_max = min(2 * n2 - 2, 2 * n2 - 2 - p)
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n2 - 1)
            jb = p - ms + (n2 - 1)
            vf = (jf >= 0) & (jf < n2)
            vb = (jb >= 0) & (jb < n2)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - x[jf[vf]])
            if np.any(vb):
                a[vb] -= x[jb[vb]]
            g[p] = np.sum(w[ms] * a)
        return (2.0 / n) * g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        return new_h
    except Exception:
        return h
