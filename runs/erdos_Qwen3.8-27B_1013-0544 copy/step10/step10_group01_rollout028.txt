import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals = vals.copy()
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50, threshold=1e-5):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h[np.where(h < threshold)[0]] = 0.0
            h[np.where(h > 1.0 - threshold)[0]] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    final_c = compute_c5(h, n)
    return h, min(best_c, final_c)

**Idea: Use a warm-start for the golden section.**

**Idea: Skip coordinates that are "far" from being active.**

    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    W = exp_c / exp_c.sum()  # length 2n-1
    
    Ar = W[::-1]  # reversed
    
    # Compute convolutions via FFT
    # (Ar * (1-h))[n-1+j] for j = 0,...,n-1
    # (A * h)[n-1+j] for j = 0,...,n-1
    
    # Use np.convolve for simplicity (or FFT for speed)

    conv1 = np.convolve(Ar, B, mode='full')  # length 3n-2
    conv2 = np.convolve(W, h, mode='full')  # length 3n-2
    
    first_sum = conv1[n-1:n-1+n]  # indices n-1 to 2n-2
    second_sum = conv2[n-1:n-1+n]
    
    grad = first_sum - second_sum
    return grad * (2.0/n)  # multiply by dx

```python
def compute_gradient_fast(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    W = exp_c / exp_c.sum()
    dx = 2.0 / n
    
    # k_m = m - (n-1) for m = 0, ..., 2n-2
    k = np.arange(-(n-1), n)  # length 2n-1
    
    # grad[j] = dx * sum_m W[m] * [(1-h[j-k[m]])*I(0<=j-k[m]<n) - h[j+k[m]]*I(0<=j+k[m]<n)]
    
    # For all j and m:
    # j-k[m] is valid: 0 <= j-k[m] < n
    # j+k[m] is valid: 0 <= j+k[m] < n
    
    # Create index matrices
    # J[j, m] = j - k[m], valid when 0 <= J < n
    # K[j, m] = j + k[m], valid when 0 <= K < n
    
    # This is O(n²) memory and computation, which for n=700 is 490,000 elements. Fine.
    
    j_idx = np.arange(n)[:, None]  # (n, 1)
    k_rows = k[None, :]  # (1, 2n-1)
    
    jm = j_idx - k_rows  # (n, 2n-1): j - k[m]
    jp = j_idx + k_rows  # (n, 2n-1): j + k[m]
    
    valid_m = (jm >= 0) & (jm < n)
    valid_p = (jp >= 0) & (jp < n)
    
    # term1[j] = sum_m W[m] * (1-h[j-k[m]]) * valid_m
    # term2[j] = sum_m W[m] * h[j+k[m]] * valid_p
    
    # Clip indices for safe access
    jm_clipped = np.clip(jm, 0, n-1)
    jp_clipped = np.clip(jp, 0, n-1)
    
    h_vals_m = (1.0 - h[jm_clipped])  # (n, 2n-1)
    h_vals_p = h[jp_clipped]  # (n, 2n-1)
    
    W_row = W[None, :]  # (1, 2n-1)
    
    term1 = (W_row * h_vals_m * valid_m).sum(axis=1)  # (n,)
    term2 = (W_row * h_vals_p * valid_p).sum(axis=1)  # (n,)
    
    grad = (term1 - term2) * dx
    return grad
```

This is O(n * (2n-1)) = O(n²) which for n=700 is about 1 million operations. Very fast in numpy.

Now let me put together the full algorithm:

```python
def subgradient_optimize(h, n, alpha, lr_init, max_iter, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        grad = compute_gradient_fast(h, n, alpha)
        
        # Take a step
        h_new = h - lr * grad
        
        # Project to feasible set
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            h = h_new
            best_c = c_new
            # Keep step size or increase slightly
            lr = lr * 1.01
        else:
            # Decrease step size
            lr = lr * 0.8
    
    return h, best_c
```

But wait, this is a non-convex problem, so the subgradient method might not work well. The "gradient" I computed is the gradient of the SMOOTH approximation (log-sum-exp), not the true subgradient. So it should be more stable.

Actually, the gradient of the smooth objective f(h) = (m + (1/α)log(Σ exp(α(c_k - m)))) * dx is:

∂f/∂h[j] = dx * Σ_k (w_k * ∂c_k/∂h[j])

where w_k = exp(α(c_k - m)) / Σ exp(α(c_k - m))

This is exactly what I computed above! So the subgradient method on the smooth objective is a proper gradient descent.

The issue is that for non-convex functions, gradient descent can get stuck. But with adaptive step size and multiple restarts, it should work.

Let me also think about the **line search** for the gradient step. Instead of fixed step size, I can do a backtracking line search:

```python
def gradient_step(h, n, alpha, lr, deadline):
    grad = compute_gradient_fast(h, n, alpha)
    c_curr = compute_c5(h, n)
    
    # Backtracking line search
    for _ in range(30):
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < c_curr - 1e-12:
            return h_new, c_new, lr
        lr *= 0.5
        if lr < 1e-10:
            break
    return h, c_curr, lr
```

Now let me think about the overall algorithm structure:

1. Start from initial_h_values
2. Phase 1: Coordinate descent (proven to work well)
3. Phase 2: Gradient descent with log-sum-exp smoothing (different from before)
4. Phase 3: More coordinate descent
5. Phase 4: Try at smaller n, then upsample
6. Phase 5: Final polish

The key difference from the previous code is the use of gradient descent (Phase 2) instead of SLSQP. SLSQP is a general-purpose optimizer that uses a QP subproblem at each step, which is expensive. The gradient method is cheaper per-iteration and can take many more steps in the same time.

Also, I'll use a better alpha schedule: start with small alpha (soft approximation) and gradually increase.

Let me also add a **momentum** to the gradient step:
v = β*v + (1-β)*grad
h = h - lr*v

This can help escape shallow local minima.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals = vals.copy()
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50, threshold=1e-5):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h[np.where(h < threshold)[0]] = 0.0
            h[np.where(h > 1.0 - threshold)[0]] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    final_c = compute_c5(h, n)
    return h, min(best_c, final_c)

def compute_gradient_fast(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    W = exp_c / exp_c.sum()
    dx = 2.0 / n
    
    k = np.arange(-(n-1), n)
    j_idx = np.arange(n)[:, None]
    k_rows = k[None, :]
    
    jm = j_idx - k_rows
    jp = j_idx + k_rows
    
    valid_m = (jm >= 0) & (jm < n)
    valid_p = (jp >= 0) & (jp < n)
    
    jm_clipped = np.clip(jm, 0, n-1)
    jp_clipped = np.clip(jp, 0, n-1)
    
    h_vals_m = (1.0 - h[jm_clipped])
    h_vals_p = h[jp_clipped]
    
    W_row = W[None, :]
    
    term1 = (W_row * h_vals_m * valid_m).sum(axis=1)
    term2 = (W_row * h_vals_p * valid_p).sum(axis=1)
    
    return (term1 - term2) * dx

def gradient_descent(h, n, alphas, max_iter_each=500, deadline=None, momentum=0.9):
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if deadline and time.time() > deadline:
            break
        v = np.zeros(n)
        lr = 0.01
        for it in range(max_iter_each):
            if deadline and time.time() > deadline:
                break
            grad = compute_gradient_fast(h, n, alpha)
            v = momentum * v + (1 - momentum) * grad
            h_new = project_feasible(h - lr * v, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                h = h_new
                best_c = c_new
                lr = min(lr * 1.05, 0.5)
            else:
                lr *= 0.7
                v = 0.5 * v
                if lr < 1e-8:
                    break
    return h, best_c
