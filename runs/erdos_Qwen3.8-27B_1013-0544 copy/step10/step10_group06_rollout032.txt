import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def lse_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        wt = w[k] * dx
        if sh > 0:
            grad[:n-sh] += wt * (1.0 - x[sh:n])
            grad[sh:n] += wt * (-x[:n-sh])
        elif sh < 0:
            s2 = -sh
            grad[s2:n] += wt * (1.0 - x[:n-s2])
            grad[:n-s2] += wt * (-x[s2:n])
        else:
            grad[:] += wt * (1.0 - 2.0 * x)
    obj = lse_obj(x, n, alpha)
    return obj, grad

def sum_fun(x, n):
    return x.sum() - n / 2.0

def sum_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (n,)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() > deadline:
            return h, compute_c5(h, n)
        nh = project_feasible(res.x, n)
        return nh, compute_c5(nh, n)
    except Exception:
        return h, compute_c5(h, n)

        for p in rng.permutation(n):

            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:

            hi_a = min(2 * n - 2, n - 1 + p)
            lo_b = max(0, p)
            hi_b = min(2 * n - 2, p + n - 1)
            m_lo, m_hi = min(lo_a, lo_b), max(hi_a, hi_b)
            ms = np.arange(m_lo, m_hi + 1)
            jf, jb = p + ms - (n - 1), p - ms + (n - 1)
            vf, vb = (jf >= 0) & (jf < n), (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb): a[vb] -= h[jb[vb]]

def proj_grad_descent(h, n, alpha, max_iter, deadline):

        obj, grad = lse_grad(h, n, alpha)

        if prev_grad is not None:
            s = h - prev_x
            y = grad - prev_grad
            sy = s @ y
            yy = y @ y
            if yy > 1e-16 and sy > 1e-16:

            else:

        if np.linalg.norm(h_new - h) < 1e-14:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def lse_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        wt = w[k] * dx
        if sh > 0:
            grad[:n - sh] += wt * (1.0 - x[sh:n])
            grad[sh:n] += wt * (-x[:n - sh])
        elif sh < 0:
            s2 = -sh
            grad[s2:n] += wt * (1.0 - x[:n - s2])
            grad[:n - s2] += wt * (-x[s2:n])
        else:
            grad[:] += wt * (1.0 - 2.0 * x)
    return lse_obj(x, n, alpha), grad

def sum_fun(x, n):
    return x.sum() - n / 2.0

def sum_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(lse_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() > deadline:
            return h, compute_c5(h, n)
        nh = project_feasible(res.x, n)
        return nh, compute_c5(nh, n)
    except Exception:
        return h, compute_c5(h, n)

def pgd(h, n, alpha, max_iter, deadline):
    h = h.copy()
    pg, px = None, None
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        _, g = lse_grad(h, n, alpha)
        if pg is not None:
            s = h - px
            y = g - pg
            sy = s @ y
            yy = y @ y
            step = sy / yy if (yy > 1e-16 and sy > 1e-16) else 1.0 / (np.linalg.norm(g) + 1e-16)
        else:
            step = 1.0 / (np.linalg.norm(g) + 1e-16)
        step = min(step, 0.5)
        nh = project_feasible(h - step * g, n)
        if np.linalg.norm(nh - h) < 1e-14:
            break
        pg, px = g.copy(), h.copy()
        h = nh
    return h, compute_c5(h, n)

def coord_descent(h, n, max_passes, deadline, rng, nbisect=40):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in rng.permutation(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            lo_a, hi_a = max(0, n - 1 - p), min(2 * n - 2, n - 1 + p)
            lo_b, hi_b = max(0, p), min(2 * n - 2, p + n - 1)
            m_lo, m_hi = min(lo_a, lo_b), max(hi_a, hi_b)
            ms = np.arange(m_lo, m_hi + 1)
            jf, jb = p + ms - (n - 1), p - ms + (n - 1)
            vf, vb = (jf >= 0) & (jf < n), (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            cb = c[ms].copy()
            di = np.where(ms == (n - 1))[0]
            hd = len(di) > 0
            cd = cb[di[0]] if hd else 0.0
            if hd:
                a[di] = 0.0
            nd = np.ones(len(ms), dtype=bool)
            if hd:
                nd[di] = False
            a_nd, c_nd = a[nd], cb[nd]
            um = 0.0
            if m_lo > 0:
                um = max(um, float(np.max(c[:m_lo])))
            if m_hi < 2 * n - 2:
                um = max(um, float(np.max(c[m_hi + 1:])))
            Llo, Lhi = um, float(np.max(c))
            for _ in range(nbisect):
                if Lhi - Llo < 1e-14:
                    break
                T = (Llo + Lhi) / 2.0
                ok = True
                if um > T + 1e-15:
                    ok = False
                elif ok and len(a_nd) > 0:
                    xl, xh = 0.0, 1.0
                    neg, pos = a_nd < -1e-13, a_nd > 1e-13
                    if np.any(neg):
                        xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                    if xl > xh + 1e-14:
                        ok = False
                if ok and hd:
                    Cq = T - cd + h0 - h0 * h0
                    disc = 1.0 - 4.0 * Cq
                    if disc >= 0:
                        r1 = (1.0 - np.sqrt(disc)) / 2.0
                        r2 = (1.0 + np.sqrt(disc)) / 2.0
                        if len(a_nd) > 0:
                            neg, pos = a_nd < -1e-13, a_nd > 1e-13
                            xl, xh = 0.0, 1.0
                            if np.any(neg):
                                xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                            if np.any(pos):
                                xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                            xl, xh = max(xl, 0.0), min(xh, 1.0)
                            if xl > xh + 1e-14:
                                ok = False
                            elif xl >= r1 and xh <= r2:
                                ok = False
                if ok:
                    Lhi = T
                else:
                    Llo = T
            T = (Llo + Lhi) / 2.0
            xl, xh = 0.0, 1.0
            if len(a_nd) > 0:
                neg, pos = a_nd < -1e-13, a_nd > 1e-13
                if np.any(neg):
                    xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                if np.any(pos):
                    xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            xl, xh = max(xl, 0.0), min(xh, 1.0)
            if hd:
                Cq = T - cd + h0 - h0 * h0
                disc = 1.0 - 4.0 * Cq
                if disc >= 0:
                    r1 = (1.0 - np.sqrt(disc)) / 2.0
                    r2 = (1.0 + np.sqrt(disc)) / 2.0
                    if xl < r1:
                        xh = min(xh, r1)
                    elif xh > r2:
                        xl = max(xl, r2)
            xl, xh = max(xl, 0.0), min(xh, 1.0)
            if xl > xh + 1e-10:
                nx = h0
            else:
                nx = float(np.clip((xl + xh) / 2.0, 0.0, 1.0))
            if abs(nx - h0) > 1e-13:
                h[p] = nx
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)
