import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n=len(h)
    if n==0: return float('inf')
    c=np.correlate(h, 1.0-h, mode='full')
    return float(np.max(c) * (2.0/n))

def project_feasible(h):
    h=np.asarray(h,dtype=float).copy()
    n=len(h); target=0.5*n
    h=np.clip(h,0.0,1.0)
    lo=-float(h.max()) if n else 0.0
    hi=1.0-float(h.min()) if n else 0.0
    for _ in range(60):
       mid=0.5*(lo+hi)
       s=np.clip(h+mid,0.0,1.0).sum()
       if s < target: lo=mid
       else: hi=mid
    h=np.clip(h+0.5*(lo+hi),0.0,1.0)
    for _ in range(8):
       diff=target - h.sum()
       if abs(diff) < 1e-12: break
       if diff > 0:
          idx=int(np.argmax(1.0-h))
          add=min(diff, 1.0-h[idx])
          h[idx]+=add
       else:
          idx=int(np.argmin(h))
          add=max(diff, h[idx])
          h[idx]+=add
    return h

def smooth_obj(x,n,alpha):
    c=np.correlate(x,1.0-x,mode='full')
    m=float(c.max())
    sh=alpha*(c-m)
    mx=sh.max()
    vals=np.exp(sh-mx)
    s=vals.sum()
    if s<=0: lse=0.0
    else: lse=mx+float(np.log(s))
    return (m + lse/alpha)*(2.0/n)

def smooth_grad(x,n,alpha):
    c=np.correlate(x,1.0-x,mode='full')
    m=float(c.max())
    sh=alpha*(c-m)
    mx=sh.max()
    w=np.exp(sh-mx); s=w.sum()
    if s>0 and np.isfinite(s): w/=s
    g=np.zeros(n)
    w0=w[n-1]
    if w0>0: g += w0*(1.0-2.0*x)
    tol=1e-15
    for q in range(1,n):
       wp=w[n-1+q]
       if wp>tol:
          g[:-q] += wp*(1.0 - x[q:])
          g[q:] -= wp*x[:n-q]
       wn=w[n-1-q]
       if wn>tol:
          g[q:] += wn*(1.0 - x[:n-q])
          g[:-q] -= wn*x[q:]
    return (2.0/n)/alpha * g

def thue_morse(n):
    if n<=0: return np.array([], dtype=float)
    arr=np.zeros(n, dtype=float)
    for i in range(1,n):
       if i & 1:
          arr[i]=1.0-arr[i>>1]
       else:
          arr[i]=arr[i>>1]
    return arr

def make_start_k(n, rng, kind):

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    if n == 0:
        return float("inf")
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))


def project_feasible(h):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = 0.5 * n
    if n == 0:
        return h
    lo = -float(h.max())
    hi = 1.0 - float(h.min())
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.clip(h + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(8):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            h[i] += min(diff, 1.0 - h[i])
        else:
            i = int(np.argmin(h))
            h[i] += max(diff, h[i])
    return h


def grid_centers(n):
    return (np.arange(n, dtype=float) + 0.5) * (2.0 / n)


def interpolate_h(h, n_target):
    if len(h) == n_target:
        return project_feasible(h)
    x1 = grid_centers(len(h))
    x2 = grid_centers(n_target)
    return project_feasible(np.interp(x2, x1, h))


def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    e = np.exp(sh - mx)
    s = float(e.sum())
    lse = mx + (float(np.log(s)) if s > 0 else 0.0)
    return (alpha * m + lse) * (2.0 / n)


def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    w = np.exp(sh - mx)
    s = float(w.sum())
    if s > 0 and np.isfinite(s):
        w /= s
    g = np.zeros(n)
    g += w[n - 1] * (1.0 - 2.0 * x)
    tol = 1e-15
    for q in range(1, n):
        wp = w[n - 1 + q]
        if wp > tol:
            g[:-q] += wp * (1.0 - x[q:])
            g[q:] -= wp * x[:n - q]
        wn = w[n - 1 - q]
        if wn > tol:
            g[q:] += wn * (1.0 - x[:n - q])
            g[:-q] -= wn * x[q:]
    return (2.0 / n) * g


def _sum_eq(x, n):
    return x.sum() - 0.5 * n


def _sum_eq_jac(x, n):
    return np.ones_like(x)


def slsqp_optimize(h, n, alphas, maxiter_each, deadline):
    h = project_feasible(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    if n == 0:
        return best_h, best_c
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _sum_eq, "jac": _sum_eq_jac, "args": (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(
                smooth_obj,
                best_h,
                args=(n, float(alpha)),
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                jac=smooth_grad,
                options={"maxiter": int(maxiter_each), "ftol": 1e-10, "disp": False},
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                nh = project_feasible(res.x)
                nc = compute_c5(nh)
                if nc < best_c:
                    best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c


def _cd_fval(c_aff, a, diag_idx, unaff_max, d):
    vals = c_aff + d * a
    vals[diag_idx] = c_aff[diag_idx] + d * a[diag_idx] - d * d
    r = float(vals.max())
    if unaff_max > r:
        r = unaff_max
    return r


def coord_descent(start_h, n, max_passes, deadline, golden=35):
    h = project_feasible(start_h)
    best_h = h.copy()
    best_c = compute_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    stagnant = 0
    for pass_idx in range(int(max_passes)):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode="full")
        stale = False
        improved = False
        if pass_idx % 2 == 0:

        else:

        for p in order:

            if stale:

            m_lo = p if p < n - 1 - p else n - 1 - p

            hi2 = n - 1 + p

            if m_hi > 2 * n - 2:

            if m_lo > m_hi:

            c_aff = c[m_lo:m_hi + 1]
            a = np.zeros(len(c_aff), dtype=float)

            f_hi = min(m_hi, 2 * n - 2 - p)
            if f_lo <= f_hi:

            b_lo = max(m_lo, p)

            if b_lo <= b_hi:

            diag_idx = n - 1 - m_lo

            for _ in range(int(golden)):

                if f1 < f2:
                    d_hi = d2
                    f2 = f1
                    d2 = d1
                    d1 = d_hi - gr * (d_hi - d_lo)

                else:
                    d_lo = d1
                    f1 = f2
                    d1 = d2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, unaff, d2)
            d_opt = 0.5 * (d_lo + d_hi)
            best_d = d_opt
            best_f = _cd_fval(c_aff, a, diag_idx, unaff, d_opt)
            for cand in (d_lo, d_hi, 0.0):
                if cand < d_lo - 1e-15 or cand > d_hi + 1e-15:
                    continue

                elif cand > d_hi:
                    cand = d_hi
                fc = _cd_fval(c_aff, a, diag_idx, unaff, cand)

            h[h > 1.0 - snap] = 1.0

                best_c = cn
                best_h = h.copy()
                stagnant = 0
            else:
                stagnant += 1
                h = best_h.copy()
                if stagnant >= 3:
                    break
        else:
            break
    return best_h, best_c


def thue_morse(n):
    a = np.zeros(n, dtype=float)
    for i in range(1, n):
        if i & 1:
            a[i] = 1.0 - a[i >> 1]
        else:
            a[i] = a[i >> 1]
    return a


def make_start_k(n, rng, kind):
    x = grid_centers(n)
    if kind == 0:
        amp = 0.25 + 0.35 * rng.rand()
        f = 1.0 + 2.0 * rng.rand()
        ph = 2.0 * np.pi * rng.rand()
        h = 0.5 + amp * np.sin(2.0 * np.pi * f * x + ph)
    elif kind == 1:
        amp = 0.2 + 0.3 * rng.rand()
        f = 1.0 + 3.0 * rng.rand()
        ph = 2.0 * np.pi * rng.rand()
        h = 0.5 + amp * np.cos(2.0 * np.pi * f * x + ph) + 0.2 * np.sin(4.0 * np.pi * x + rng.rand())
    else:
        y = rng.normal(0.0, 1.0, n)
        k = max(5, n // 40)
        kern = np.ones(k)
        kern /= kern.sum()
        y = np.convolve(y, kern, mode="same")
        y = (y - y.mean()) / (y.std() + 1e-12)
        h = 0.5 + 0.5 * np.tanh(y)
    return project_feasible(h)


def refine_candidate(h, n, deadline, alphas, maxiter, cd_passes, golden):
    h = project_feasible(h)
    c = compute_c5(h)
    if time.time() < deadline - 15:
        h2, c2 = slsqp_optimize(h, n, alphas, maxiter, deadline)
        if c2 < c:

        h3, c3 = coord_descent(h, n, cd_passes, deadline, golden)
        if c3 < c:
            h, c = h3, c3
    h = project_feasible(h)
    return h, compute_c5(h)

        parent = make_start_k(700, rng, 2)
    best_h = project_feasible(parent)
    best_n = len(best_h)
    best_c = compute_c5(best_h)

    d = deadline - 500
    if d > time.time() + 10:
        h2, c2 = slsqp_optimize(best_h, best_n, [500, 5000, 30000], 60, d)
        if c2 < best_c:
            best_h, best_c = h2, c2

    d = deadline - 350
    h2, c2 = coord_descent(best_h, best_n, 60, d, 32)
    if c2 < best_c:
        best_h, best_c = h2, c2

    d = deadline - 250
    if d > time.time() + 10:
        h2, c2 = slsqp_optimize(best_h, best_n, [200000], 35, d)
        if c2 < best_c:
            best_h, best_c = h2, c2

    h2, c2 = coord_descent(best_h, best_n, 40, deadline - 150, 45)
    if c2 < best_c:
        best_h, best_c = h2, c2

    if time.time() < deadline - 220:
        best_grid = grid_centers(best_n)
        for ns in (max(120, best_n // 4), max(240, best_n // 2)):
            if time.time() > deadline - 220:
                break
            if ns >= best_n or ns < 60:
                continue
            low_grid = grid_centers(ns)
            cand_h = project_feasible(np.interp(low_grid, best_grid, best_h))
            candidates = [
                cand_h,
                make_start_k(ns, rng, 0),
                make_start_k(ns, rng, 1),
                project_feasible(thue_morse(ns).astype(float)),
            ]
            for hl in candidates:
                if time.time() > deadline - 220:
                    break
                d_end = min(deadline - 200, time.time() + 55)
                if d_end <= time.time() + 5:
                    break
                hl, cl = refine_candidate(hl, ns, d_end, [500, 8000], 35, 18, 25)
                hb = project_feasible(np.interp(best_grid, low_grid, hl))
                hb, cb = coord_descent(hb, best_n, 12, min(deadline - 120, time.time() + 40), 25)
                if cb < best_c:
                    best_h, best_c = hb, cb

    if time.time() < deadline - 200:
        ncur = best_n
        x = grid_centers(ncur)
        remaining = max(0.0, deadline - 180 - time.time())
        n_restarts = max(1, min(5, int(remaining // 80)))
        for r in range(n_restarts):
            if time.time() > deadline - 180:
                break
            mode = r % 4
            if mode == 0:
                hp = best_h + rng.normal(0.0, 0.02, ncur)
            elif mode == 1:
                f = 1.0 + 2.0 * rng.rand()
                hp = best_h + 0.04 * np.sin(2.0 * np.pi * f * x + 2.0 * np.pi * rng.rand())
            elif mode == 2:
                mask = rng.rand(ncur) < 0.12
                hp = best_h + 0.3 * mask - 0.15
            else:
                hp = best_h + (rng.rand(ncur) - 0.5) * 0.05
            hp = project_feasible(hp)
            d_end = min(deadline - 150, time.time() + 65)
            if d_end <= time.time() + 5:
                break
            hp, ch = refine_candidate(hp, ncur, d_end, [2000, 20000], 35, 18, 30)
            if ch < best_c:
                best_h, best_c = hp, ch

    if time.time() < deadline - 120 and best_n < 1000:
        hh = interpolate_h(best_h, 1000)
        if time.time() < deadline - 150:
            hh, ch = slsqp_optimize(hh, 1000, [3000, 30000], 30, min(deadline - 80, time.time() + 55))
        else:
            ch = compute_c5(hh)
        hh, ch = coord_descent(hh, 1000, 18, min(deadline - 25, time.time() + 75), 32)
        if ch < best_c:
            best_h, best_n, best_c = hh, 1000, ch

    if time.time() < deadline - 30 and best_n == 1000:
        hh, ch = coord_descent(best_h, 1000, 20, deadline - 10, 40)
        if ch < best_c:
            best_h, best_c = hh, ch

    if time.time() < deadline - 10:
        h2, c2 = coord_descent(best_h, best_n, 40, deadline - 3, 50)
        if c2 < best_c:
            best_h, best_c = h2, c2

    best_h = project_feasible(best_h)
    best_c = compute_c5(best_h)
    return best_h, float(best_c), int(len(best_h))
