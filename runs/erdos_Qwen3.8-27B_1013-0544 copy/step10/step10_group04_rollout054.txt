    h_init = project_feasible(h_init, n)
    h, c = exact_cd(h_init, n, passes, deadline, rng, False)
    h, c = exact_cd(h, n, max(passes // 3, 30), deadline, rng, True)
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5); n0 = 500
    h_best = project_feasible(h_best, n0)
    best_c = compute_c5(h_best, n0)

    # Quick polish
    h_best, best_c = exact_cd(h_best, n0, 20, deadline, rng, False)

    # Phase A: diverse n values with interpolated start
    for n_try in [300, 350, 400, 450, 500, 550]:
        if time.time() > deadline - 300: break
        xo = np.linspace(0, 2, n0)
        xn = np.linspace(0, 2, n_try)
        h_try = project_feasible(np.interp(xn, xo, h_best), n_try)
        h_try, c_try = try_candidate(h_try, n_try, deadline, rng, 500)
        if c_try < best_c:
            h_best, best_c, n0 = h_try, c_try, n_try

    # Phase B: diverse initializations at promising n values
    for n_try in [350, 400, 450, 500]:
        if time.time() > deadline - 200: break
        for init_fn in [make_sine_init, make_block_init, make_antisym_init]:
            if time.time() > deadline - 100: break
            h_try = init_fn(n_try, rng)
            h_try, c_try = try_candidate(h_try, n_try, deadline, rng, 600)
            if c_try < best_c:
                h_best, best_c, n0 = h_try, c_try, n_try

    # Phase C: more diverse starts at the best n
    if time.time() < deadline - 150:
        for _ in range(8):
            if time.time() > deadline - 60: break
            init_type = rng.randint(3)
            if init_type == 0:
                h_try = make_sine_init(n0, rng)
            elif init_type == 1:
                h_try = make_block_init(n0, rng)
            elif init_type == 2:
                h_try = make_antisym_init(n0, rng)
            else:
                h_try = project_feasible(h_best + rng.normal(0, 0.1, n0), n0)
            h_try, c_try = try_candidate(h_try, n0, deadline, rng, 400)
            if c_try < best_c:
                h_best, best_c = h_try, c_try

    # Phase D: noise restarts from best
    for nl in [0.03, 0.05, 0.08, 0.12, 0.04, 0.06]:
        if time.time() > deadline - 40: break
        hp = project_feasible(h_best + rng.normal(0, nl, n0), n0)
        hp, cp = try_candidate(hp, n0, deadline, rng, 300)
        if cp < best_c:
            h_best, best_c = hp, cp

    # Phase E: final intensive polish
    nf = len(h_best)
    h_best, best_c = exact_cd(h_best, nf, 800, deadline, rng, False)
    h_best, best_c = exact_cd(h_best, nf, 300, deadline, rng, True)

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
