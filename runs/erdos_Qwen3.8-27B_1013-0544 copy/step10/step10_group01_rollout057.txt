```python
        # h[j-lag] for valid j-lag in [0,n-1]
        # j-lag in [0,n-1] => j in [lag, n-1+lag]
        for j in range(n):
            term = 0.0
            if 0 <= j+lag < n:
                term += (1.0 - h[j+lag])
            if 0 <= j-lag < n:
                term -= h[j-lag]
            grad[j] += p[k] * term
    grad *= (2.0/n)
    return grad
```

But this is O(n^2) per gradient, and with the inner loop over k and j, it's actually O(n * 2n) = O(n^2) which for n=700 is about 1M operations per gradient. That's fine.

But wait, I can vectorize this! For a given lag, the contribution to the gradient is:
- For term (1-h[j+lag]): this adds to grad[j] the value p[k]*(1-h[j+lag]) for j in the valid range
- For term -h[j-lag]: this adds to grad[j] the value -p[k]*h[j-lag] for j in the valid range

These are just shifted versions of (1-h) and h weighted by p[k]. So I can use numpy's slice operations:

```python
def compute_gradient(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_shifted = alpha * (c - c.max())
    p = np.exp(c_shifted)
    p /= p.sum()
    grad = np.zeros(n)
    for k in range(2*n-1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lag = k - (n-1)
        # Contribution from (1-h[j+lag])
        if lag >= 0:
            j_range = range(0, n-lag)  # j+lag < n => j < n-lag
            grad[:n-lag] += pk * (1.0 - h[lag:])
        else:
            j_range = range(-lag, n)  # j+lag >= 0 => j >= -lag
            grad[-lag:] += pk * (1.0 - h[:n+lag])
        # Contribution from -h[j-lag]
        if lag >= 0:
            j_range = range(lag, n)  # j-lag >= 0 => j >= lag
            grad[lag:] -= pk * h[:n-lag]
        else:
            j_range = range(0, n+lag)  # j-lag < n => j < n+lag
            grad[:n+lag] -= pk * h[-lag:]
    grad *= (2.0/n)
    return grad
```

This is O(n * 2n) = O(n^2) but with vectorized inner operations, so it's fast in practice.

For n=700, that's about 1400 vector operations of length ~700, which is very fast.

But the projection onto {0≤h≤1, sum(h)=n/2} is not a simple projection (it's not separable due to the sum constraint). However, my `project_feasible` function handles this.

```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, n, alpha):
    """Compute smooth max objective and its gradient."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    p = np.exp(shifted - shifted.max())
    p /= p.sum()
    lse = shifted.max() + np.log(p.sum())  # = shifted.max() + log(sum(exp(shifted - shifted.max())))
    obj = (m + lse / alpha) * (2.0 / n)
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        # (1-h[j+lag]) contribution
        if lag >= 0:
            grad[:n - lag] += pk * (1.0 - h[lag:])
        else:
            grad[-lag:] += pk * (1.0 - h[:n + lag])
        # -h[j-lag] contribution
        if lag >= 0:
            grad[lag:] -= pk * h[:n - lag]
        else:
            grad[:n + lag] -= pk * h[-lag:]
    grad *= (2.0 / n)
    return obj, grad

def adam_optimize(h, n, deadline, alpha_schedule, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with projected gradient."""
    h = h.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0
    total_steps = len(alpha_schedule)
    
    for alpha in alpha_schedule:
        n_inner = max(20, int((deadline - time.time()) * 2))
        for step in range(n_inner):
            if time.time() > deadline:
                break
            obj, grad = smooth_obj_and_grad(h, n, alpha)
            
            # Adam update
            m_v = beta1 * m_v + (1 - beta1) * grad
            v_v = beta2 * v_v + (1 - beta2) * grad ** 2
            m_hat = m_v / (1 - beta1 ** (step + 1))
            v_hat = v_v / (1 - beta2 ** (step + 1))
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project
            h_new = project_feasible(h_new, n)
            
            # Check improvement
            c = compute_c5(h_new, n)
            if c < best_c:
                best_c = c
                best_h = h_new.copy()
            
            h = h_new
    
    return best_h, best_c

def golden_section_1d(f, a, b, tol=1e-10, max_iter=50):
    """Minimize f on [a, b] using golden section search."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return f((a + b) / 2.0)

def block_cd(h, n, block_size, deadline, max_passes=30):
    """Block coordinate descent."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for pass_idx in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for start in range(0, n - block_size + 1, block_size):
            if time.time() > deadline:
                break
            end = start + block_size
            x0 = h[start:end].copy()
            
            def block_obj(x):
                h_trial = h.copy()
                h_trial[start:end] = x
                return compute_c5(h_trial, n)
            
            # Use Nelder-Mead for small blocks
            try:
                res = minimize(block_obj, x0, method='Nelder-Mead',
                             bounds=[(0.0, 1.0)] * block_size,
                             options={'maxiter': 200, 'xatol': 1e-10, 'fatol': 1e-12})
                if res.fun < block_obj(x0) - 1e-14:
                    h[start:end] = np.clip(res.x, 0.0, 1.0)
                    improved = True
            except Exception:
                pass
        
        if improved:
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
    
    return h, best_c

def single_cd(h, n, deadline, max_passes=50, gs_iters=60):

    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

        improved = False

            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:

            # c_k = sum_i h[i](1-h[i+lag_k])
            # d c_k / d h[p] = (1-h[p+lag_k]) - h[p-lag_k] for valid indices
            
            # For 1D optimization, c as a function of h[p] = h0 + d:
            # c_k(d) = c_k(0) + d * [(1-(h0+lag... )) - ...]  -- this is linear in d for the non-diagonal term
            # Actually c_k is quadratic in h when p appears in both h[i] and h[i+lag]
            # Specifically, if lag_k = 0 (diagonal), then c_k has a term h[p](1-h[p]) which is quadratic.
            # For lag_k != 0, c_k is linear in h[p].

            # where b_k is nonzero only for the diagonal term (lag=0)

            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)

            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]

                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))

                vals = c_aff + d * a
                if has_diag:
                    vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - 2.0 * h0) + d * h0 - d**2
                    # Wait, let me recompute. If h[p] changes by d:
                    # The diagonal term in c_{n-1} is h[p](1-h[p])
                    # Originally: h0*(1-h0)
                    # New: (h0+d)*(1-h0-d) = h0+h0*d + d + d^2 - h0^2 - 2*h0*d - h0*d^2 - d^2
                    # Hmm, let me just do it directly
                    # Actually, the diagonal term contributes h[p]*(1-h[p]) to c_{n-1}
                    # But c_aff[diag_idx] already includes the contribution of h0*(1-h0) to that correlation
                    # When we change h[p] to h0+d, the diagonal term changes from h0*(1-h0) to (h0+d)*(1-h0-d)
                    # Delta = (h0+d)*(1-h0-d) - h0*(1-h0) = h0 + d - h0^2 - h0*d - h0*d - d^2 - h0 + h0^2
                    # = d - 2*h0*d - d^2
                    # So c_{n-1}(d) = c_{n-1}(0) + d*(1-2*h0) - d^2
                    vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - 2.0 * h0) - d**2
                result = float(np.max(vals))
                if unaff_max > result:
                    result = unaff_max
                return result
        
            # Golden section search on [d_lo, d_hi] = [-h0, 1-h0]
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = eval_c(d1)
            f2 = eval_c(d2)
            
            for _ in range(gs_iters):

                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = eval_c(d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = eval_c(d2)
            
            d_opt = (d_lo + d_hi) / 2.0

            f_curr = eval_c(0.0)
            
            if f_opt < f_curr - 1e-13:

                improved = True
        
        if improved:
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
        else:
            break
    
    return h, best_c
```

```python
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    p = np.exp(shifted - shifted.max())
    p /= p.sum()
    obj = (m + np.log(p.sum()) / alpha) * (2.0 / n)
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += pk * (1.0 - h[lag:])
            grad[lag:] -= pk * h[:n - lag]
        else:
            grad[-lag:] += pk * (1.0 - h[:n + lag])
            grad[:n + lag] -= pk * h[-lag:]
    grad *= (2.0 / n)
    return obj, grad, p

def adam_phase(h0, n, deadline, alphas, lr=0.02):
    h = h0.copy()
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    t = 0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for alpha in alphas:
        n_steps = max(30, int((deadline - time.time()) / 0.05))
        for _ in range(n_steps):
            if time.time() > deadline:
                break
            t += 1
            obj, grad, _ = smooth_obj_grad(h, n, alpha)
            m1 = b1 * m1 + (1 - b1) * grad
            m2 = b2 * m2 + (1 - b2) * grad * grad
            m1_hat = m1 / (1 - b1**t)
            m2_hat = m2 / (1 - b2**t)
            h_new = h - lr * m1_hat / (np.sqrt(m2_hat) + eps)
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c:
                best_c = c
                best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def subgrad_phase(h0, n, deadline, n_steps=5000, eta0=0.01):
    """Projected subgradient with heavy ball momentum."""
    h = h0.copy()
    h_prev = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    momentum = 0.0
    
    for t in range(n_steps):
        if time.time() > deadline:
            break
        eta = eta0 / (np.sqrt(t + 1) + 1.0)
        
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = c.max()
        # Active set: lags within 1% of max
        tol = 0.01 * max(c_max, 1e-10)
        active = np.where(c >= c_max - tol)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Compute subgradient (average over active)
        grad = np.zeros(n)
        for k in active:
            lag = k - (n - 1)
            if lag >= 0:
                grad[:n - lag] += (1.0 - h[lag:])
                grad[lag:] -= h[:n - lag]
            else:
                grad[-lag:] += (1.0 - h[:n + lag])
                grad[:n + lag] -= h[-lag:]
        grad *= (2.0 / n) / len(active)
        
        # Heavy ball update
        y = h + momentum * (h - h_prev)
        h_new = y - eta * grad
        h_new = project_feasible(h_new, n)
        
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        
        momentum = 0.5 * (1.0 + momentum) * 0.999  # slowly decaying
        h_prev = h
        h = h_new
    
    return best_h, best_c
```
