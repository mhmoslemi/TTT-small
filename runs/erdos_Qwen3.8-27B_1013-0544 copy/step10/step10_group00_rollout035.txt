import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    return (m + mx / alpha + np.log(np.sum(exp_s)) / alpha) * dx

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    weights = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for m_idx in range(len(c)):
        w = weights[m_idx] * dx
        if w < 1e-15:
            continue
        s = m_idx - (n - 1)
        if s >= 0:
            if s < n:
                grad[:n - s] += w * (1.0 - h[s:])
                grad[s:] -= w * h[:n - s]
        else:
            t = -s
            if t < n:
                grad[t:] += w * (1.0 - h[:n - t])
                grad[:n - t] -= w * h[t:]
    return grad

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_smooth(h, n, alpha, deadline, maxiter=400):
    def obj(x):
        return smooth_obj(x, n, alpha)
    def grad(x):
        return smooth_grad(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    try:
        res = minimize(obj, best_h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    except Exception:
        pass
    if time.time() > deadline:
        return best_h, best_c
    return best_h, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            dp = np.where(ms == (n - 1))[0]
            has_diag = len(dp) > 0
            diag_idx = int(dp[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1; d1 = d2; f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgradient_step(h, n, step_size, momentum_arr, beta=0.8):
    c = np.correlate(h, 1.0 - h, mode='full')
    m_star = int(np.argmax(c))
    s = m_star - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1.0 - h[s:])
        if s > 0:
            grad[s:] -= h[:n - s]
    else:
        t = -s
        grad[t:] += (1.0 - h[:n - t])
        if t > 0:
            grad[:n - t] -= h[t:]
    if s == 0:
        grad = 1.0 - 2.0 * h
    momentum_arr = beta * momentum_arr + (1.0 - beta) * grad
    h_new = project_feasible(h - step_size * momentum_arr, n)
    return h_new, momentum_arr

    for n_try in [500, 900, 1200, 1500]:
        if time.time() > deadline - 100:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = coord_descent(h_int, n_try, 40, deadline - 100, gs_iters=50)
        if c_o < best_c:
            h_o, c_o = slsqp_smooth(h_o, n_try, 5000, deadline - 100, maxiter=150)
            h_o, c_o = coord_descent(h_o, n_try, 30, deadline - 100, gs_iters=50)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = np.linspace(0, 2, n_try)

    # Phase 5: Multi-restart
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        n_restarts = 8
        for r in range(n_restarts):
            if time.time() > deadline - 60:
                break
            scale = 0.005 + 0.02 * (r / max(1, n_restarts - 1))
            if r % 5 == 0:
                hp = project_feasible(best_h + rng.normal(0, scale, n_cur), n_cur)
            elif r % 5 == 1:
                x = np.linspace(0, 2, n_cur)
                hp = project_feasible(best_h + scale * 3 * np.sin(np.pi * (2 + r) * x), n_cur)
            elif r % 5 == 2:
                hp = project_feasible(best_h + scale * rng.randn(n_cur), n_cur)
            elif r % 5 == 3:
                block = max(10, n_cur // 10)
                hp = best_h.copy()
                start = int((r * 0.11 * n_cur) % max(1, n_cur - block))
                hp[start:start + block] = 1.0 - hp[start:start + block]
                hp = project_feasible(hp, n_cur)
            else:
                x = np.linspace(0, 2, n_cur)
                freq = 1.0 + 0.5 * r
                hp = project_feasible(best_h + scale * 2 * np.cos(2 * np.pi * freq * x), n_cur)
            h_o, c_o = coord_descent(hp, n_cur, 25, deadline - 60, gs_iters=40)
            if c_o < best_c:
                h_o, c_o = slsqp_smooth(h_o, n_cur, 5000, deadline - 60, maxiter=100)
                h_o, c_o = coord_descent(h_o, n_cur, 20, deadline - 60, gs_iters=45)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 6: Final polish
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        h_f, c_f = slsqp_smooth(best_h, n_cur, 50000, deadline - 10, maxiter=300)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = coord_descent(best_h, n_cur, 60, deadline - 10, gs_iters=60)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
