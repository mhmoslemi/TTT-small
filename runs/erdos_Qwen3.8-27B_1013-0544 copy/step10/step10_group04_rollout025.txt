import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def compute_gradient(h, w, n):
    """Compute combined gradient g[j] = sum_m w[m] * dc[m]/dh[j]"""
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        # Forward: (1-h[j+m-n+1]) for valid j
        lo_f = max(0, n - 1 - m)
        hi_f = min(n - 1, 2 * n - 2 - m)
        if lo_f <= hi_f:
            idx_f = lo_f + m - (n - 1)
            g[lo_f:hi_f + 1] += wm * (1.0 - h[idx_f:idx_f + (hi_f - lo_f + 1)])
        # Backward: -h[j-m+n-1] for valid j
        lo_b = max(0, m - n + 1)
        hi_b = min(n - 1, m)
        if lo_b <= hi_b:
            idx_b = lo_b - m + (n - 1)
            g[lo_b:hi_b + 1] -= wm * h[idx_b:idx_b + (hi_b - lo_b + 1)]
    return g

def fw_vertex(g, n):
    """Find the FW vertex: argmin <g, s> over {s: 0<=s<=1, sum(s)=n/2}"""
    s = np.zeros(n)
    s[np.argsort(g)[:n // 2]] = 1.0
    return s

def fw_line_search(h, s, n, max_iter=30):
    """Line search: minimize f((1-t)*h + t*s) for t in [0,1]"""
    # Use golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c_ = b - gr * (b - a)
    d_ = a + gr * (b - a)
    fc = compute_c5((1 - c_) * h + c_ * s, n)
    fd = compute_c5((1 - d_) * h + d_ * s, n)
    for _ in range(max_iter):
        if b - a < 1e-10:
            break
        if fc < fd:
            b = d_
            d_ = c_
            fd = fc
            c_ = b - gr * (b - a)
            fc = compute_c5((1 - c_) * h + c_ * s, n)
        else:
            a = c_
            c_ = d_
            fc = fd
            d_ = a + gr * (b - a)
            fd = compute_c5((1 - d_) * h + d_ * s, n)
    t = (a + b) / 2.0
    return t

def fw_method(h, n, n_iter, eta, deadline, rng):
    """Frank-Wolfe with exponential weights."""
    h = h.copy()
    n_lags = 2 * n - 1
    w = np.ones(n_lags) / n_lags
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Compute gradient
        g = compute_gradient(h, w, n)
        # Find FW vertex
        s = fw_vertex(g, n)
        # Line search
        t = fw_line_search(h, s, n)
        # Update
        h = (1 - t) * h + t * s
        h = project_feasible(h, n)
        # Update weights
        c = np.correlate(h, 1.0 - h, mode='full')
        w *= np.exp(eta * c)
        w /= w.sum()
        # Track best
        cur_c = float(c.max() * 2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c

Let me also think about whether the FW method I described is actually correct for this problem.

The problem is: min_h max_m c(h, m) subject to h in S (hypersimplex).

The FW method solves: at each step, find s = argmin_{v in S} <∇f(h), v> where f(h) = max_m c(h,m).

But ∇f(h) is not well-defined when multiple m's achieve the max. Instead, we use a subgradient: any convex combination of the gradients of the active functions.

The exponential weights approach handles this: w[m] is proportional to exp(eta * c[m]), which puts more weight on lags with high c[m]. Then we use g = sum_m w[m] * ∇c[m] as a (smoothed) subgradient direction.

This is the "multiplicative weights + conditional gradient" method, which is a well-known approach for min-max problems. It should work well here.

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min, m_max = n - 1 - p, 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf, vb = (jf >= 0) & (jf < n), (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask, pos_mask, zero_mask = a < -1e-15, a > 1e-15, np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi, feasible = 0.0, 1.0, True
        if feasible and np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1, r2 = (1.0 - np.sqrt(disc)) / 2.0, (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1, r2 = (1.0 - np.sqrt(disc)) / 2.0, (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if (xc_lo < r1 and xc_hi > r2) or (x_lo >= r1 and x_hi <= r2):
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, random_order=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n) if (random_order and rng is not None) else np.arange(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf, jb = p + ms - (n - 1), p - ms + (n - 1)
                vf, vb = (jf >= 0) & (jf < n), (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def compute_grad(h, w, n):
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        lo_f, hi_f = max(0, n - 1 - m), min(n - 1, 2 * n - 2 - m)
        if lo_f <= hi_f:
            idx_f = lo_f + m - (n - 1)
            g[lo_f:hi_f + 1] += wm * (1.0 - h[idx_f:idx_f + (hi_f - lo_f + 1)])
        lo_b, hi_b = max(0, m - n + 1), min(n - 1, m)
        if lo_b <= hi_b:
            idx_b = lo_b - m + (n - 1)
            g[lo_b:hi_b + 1] -= wm * h[idx_b:idx_b + (hi_b - lo_b + 1)]
    return g

def fw_optimize(h, n, n_iter, eta, deadline, use_line_search=False):
    h = h.copy()
    n_lags = 2 * n - 1
    w = np.ones(n_lags) / n_lags
    best_h, best_c = h.copy(), compute_c5(h, n)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        g = compute_grad(h, w, n)
        s = np.zeros(n)
        s[np.argsort(g)[:n // 2]] = 1.0
        if use_line_search:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_, b_ = 0.0, 1.0
            c_ = b_ - gr * (b_ - a_)
            d_ = a_ + gr * (b_ - a_)
            fc = compute_c5((1 - c_) * h + c_ * s, n)
            fd = compute_c5((1 - d_) * h + d_ * s, n)
            for _ in range(20):
                if b_ - a_ < 1e-10:
                    break
                if fc < fd:
                    b_, d_, fd = d_, c_, fc
                    c_ = b_ - gr * (b_ - a_)
                    fc = compute_c5((1 - c_) * h + c_ * s, n)
                else:
                    a_, c_, fc = c_, d_, fd
                    d_ = a_ + gr * (b_ - a_)
                    fd = compute_c5((1 - d_) * h + d_ * s, n)
            t = (a_ + b_) / 2.0
        else:
            t = 2.0 / (it + 2)
        h = (1 - t) * h + t * s
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        w *= np.exp(eta * c)
        w /= w.sum()
        cur_c = float(c.max() * 2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x, n_=n: x.sum() - n_ / 2.0,
                 'jac': lambda x, n_=n: np.ones(n_)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    h1, c1 = fw_optimize(h_best, n, 3000, eta=5.0 / n, deadline=deadline, use_line_search=False)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: CD polish
    h2, c2 = exact_cd(h_best, n, 200, deadline)
    if c2 < best_c:
        h_best, best_c = h2, c2

    # Phase 3: FW with different eta and line search
    h3, c3 = fw_optimize(h_best, n, 2000, eta=10.0 / n, deadline=deadline, use_line_search=True)
    if c3 < best_c:
        h_best, best_c = h3, c3
    h3b, c3b = exact_cd(h_best, n, 200, deadline)
    if c3b < best_c:
        h_best, best_c = h3b, c3b

    # Phase 4: Random-order CD
    h4, c4 = exact_cd(h_best, n, 300, deadline, random_order=True, rng=rng)
    if c4 < best_c:
        h_best, best_c = h4, c4

    # Phase 5: SLSQP with smooth objective
    h5, c5 = slsqp_opt(h_best, n, [10000, 50000, 200000], 200, deadline)
    if c5 < best_c:
        h_best, best_c = h5, c5
    h5b, c5b = exact_cd(h_best, n, 200, deadline)
    if c5b < best_c:
        h_best, best_c = h5b, c5b

    # Phase 6: Diverse restarts with FW+CD
    for s_noise in [0.01, 0.03, 0.05, 0.08, 0.12]:
        if time.time() > deadline - 100:
            break
        hp = project_feasible(h_best + rng.normal(0, s_noise, n), n)
        hw, cw = fw_optimize(hp, n, 1500, eta=8.0 / n, deadline=deadline, use_line_search=False)
        ho, co = exact_cd(hw, n, 200, deadline)
        if co < best_c:
            h_best, best_c = ho, co

    # Phase 7: Multi-n exploration
    for n_try in [500, 600, 400, 800]:
        if time.time() > deadline - 120:
            break
        if n_try == n:
            continue
        x_old, x_new = np.linspace(0, 2, len(h_best)), np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        ho, co = fw_optimize(h_int, n_try, 1000, eta=5.0 / n_try, deadline=deadline, use_line_search=False)
        ho, co = exact_cd(ho, n_try, 200, deadline)
        if co < best_c - 1e-7:
            h_best, best_c, n = ho, co, n_try

    # Phase 8: Binary-projected CD
    if time.time() < deadline - 60:
        n = len(h_best)
        h_bin = (h_best > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n)
        ho, co = exact_cd(h_bin, n, 300, deadline)
        if co < best_c:
            h_best, best_c = ho, co

    # Phase 9: Final polish
    if time.time() < deadline - 5:
        n = len(h_best)
        hf, cf = exact_cd(h_best, n, 500, deadline)
        if cf < best_c:
            h_best, best_c = hf, cf

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
