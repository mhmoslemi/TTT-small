def power_smooth(c, n, p):
    """Power-sum smooth approximation to max(c)"""
    c_shifted = c - c.max()
    c_abs = np.abs(c_shifted)

def power_smooth_grad(c, n, p):
    """Gradient of power-sum smooth w.r.t. c"""
    c_pos = np.maximum(c, 0)
    m = c_pos.max()
    if m < 1e-15:
        return np.zeros_like(c)
    scaled = c_pos / m
    # d/dc_k (sum c_j^p)^(1/p) = c_k^(p-1) / (sum c_j^p)^(1/p)

        order = list(range(n))
        if rng is not None and it % 3 == 1:

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_lse(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            def cf(x):
                return x.sum() - n / 2.0
            def cj(x):
                return np.ones(n)
            cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
            res = minimize(lse_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgrad_step(h, n, alpha, eps):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    active = np.where(c > c_max - eps)[0]
    K = len(active)
    if K == 0:
        return h
    grad = np.zeros(n)
    for m in active:
        mo = m - (n - 1)
        i1 = max(0, mo)
        i2 = min(n, n + mo)
        if i1 < i2:
            grad[i1:i2] += (1.0 - h[i1 - mo:i2 - mo])
        i3 = max(0, -mo)
        i4 = min(n, n - mo)
        if i3 < i4:
            grad[i3:i4] -= h[i3 + mo:i4 + mo]
    grad /= K
    return project_feasible(h - alpha * grad, n)

    mt = int(rng.randint(6))
    if mt == 0:
        w = int(rng.randint(max(2, n // 25), max(4, n // 8)))
        shift = int(rng.randint(-8, 9))
        s = int(rng.randint(0, max(1, n - w - abs(shift))))
        seg = h_new[s:s + w].copy()
        h_new[s:s + w] = 0.5
        if 0 <= s + shift and s + shift + w <= n:
            h_new[s + shift:s + shift + w] = seg
    elif mt == 1:
        ctr = int(rng.randint(n // 10, 9 * n // 10))
        w = int(rng.randint(3, max(4, n // 12)))
        amp = float(h[ctr])
        for off in range(w):
            if 0 <= ctr + off < n:
                h_new[ctr + off] = max(h_new[ctr + off], amp * (1.0 - off / w))
            if 0 <= ctr - off < n:
                h_new[ctr - off] = max(h_new[ctr - off], amp * (1.0 - off / w))
    elif mt == 2:
        ctr = int(rng.randint(n // 10, 9 * n // 10))
        w = int(rng.randint(3, max(4, n // 12)))
        for off in range(w):
            if 0 <= ctr + off < n:
                h_new[ctr + off] *= (0.4 + 0.6 * off / w)
            if 0 <= ctr - off < n:
                h_new[ctr - off] *= (0.4 + 0.6 * off / w)
    elif mt == 3:
        w = int(rng.randint(4, max(5, n // 12)))
        i = int(rng.randint(0, max(1, n - 2 * w)))
        j = int(rng.randint(i + w, max(i + w + 1, n - w)))
        h_new[i:i + w], h_new[j:j + w] = h_new[j:j + w].copy(), h_new[i:i + w].copy()
    elif mt == 4:
        ctr = int(rng.randint(n // 8, 7 * n // 8))
        sig = float(rng.uniform(n / 40, n / 10))
        amp = float(rng.uniform(-0.08, 0.08))
        x = np.arange(n, dtype=float)
        h_new += amp * np.exp(-0.5 * ((x - ctr) / sig) ** 2)
    else:
        scale = float(rng.uniform(0.01, 0.04))
        mask = rng.rand(n) > 0.7
        h_new[mask] += rng.normal(0, scale, mask.sum())
    return project_feasible(h_new, n)

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(600, 0.5)
        n = 600
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Heavy exact CD
    h1, c1 = exact_cd(h_best, n, 250, deadline, rng)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: SLSQP log-sum-exp with multiple smoothing levels
    h2, c2 = slsqp_lse(h_best, len(h_best), [3000, 10000, 50000, 200000, 500000], 500, deadline)
    if c2 < best_c - 1e-12:
        h_best, best_c = h2, c2

    # Phase 3: CD refinement
    h3, c3 = exact_cd(h_best, len(h_best), 250, deadline, rng)
    if c3 < best_c - 1e-12:
        h_best, best_c = h3, c3

    # Phase 4: Multi-resolution exploration
    n_list = [650, 750, 800, 700, 600, 550, 500, 450, 900, 1000, 850, 780]
    for n_try in n_list:
        if time.time() > deadline - 250:
            break
        h_int = interp_h(h_best, len(h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 150, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o
        if time.time() < deadline - 120:
            h_s, c_s = slsqp_lse(h_o, n_try, [50000, 200000], 300, deadline)
            if c_s < best_c - 1e-9:
                h_best, best_c = h_s, c_s
            h_o2, c_o2 = exact_cd(h_best, len(h_best), 120, deadline, rng)
            if c_o2 < best_c - 1e-9:
                h_best, best_c = h_o2, c_o2

    # Phase 5: Subgradient exploration + CD
    n = len(h_best)
    for r in range(25):
        if time.time() > deadline - 80:
            break
        alpha_sg = 0.15 * (1.0 - r / 25.0) + 0.02
        eps = 0.5 * (1.0 - r / 25.0) + 0.1
        h_sg = h_best.copy()
        for step in range(30):
            h_sg = subgrad_step(h_sg, n, alpha_sg, eps)
        h_o, c_o = exact_cd(h_sg, n, 100, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o

    # Phase 6: Annealed restarts (mix of structured and Gaussian)
    for r in range(25):
        if time.time() > deadline - 60:
            break
        if r % 3 == 0:
            scale = 0.03 * (1.0 - r / 25.0) + 0.005
            hp = project_feasible(h_best + rng.normal(0, scale, n), n)
        elif r % 3 == 1:
            hp = structured_perturb(h_best, n, rng)
        else:
            hp = structured_perturb(project_feasible(h_best + rng.normal(0, 0.02, n), n), n, rng)
        h_o, c_o = exact_cd(hp, n, 100, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o

    # Phase 7: Final polish - long CD + SLSQP
    n = len(h_best)
    h_f, c_f = exact_cd(h_best, n, 400, deadline, rng)
    if c_f < best_c - 1e-12:
        h_best, best_c = h_f, c_f
    if time.time() < deadline - 15:
        h_f2, c_f2 = slsqp_lse(h_best, n, [500000, 1000000], 300, deadline)
        if c_f2 < best_c - 1e-12:
            h_best, best_c = h_f2, c_f2
        h_f3, c_f3 = exact_cd(h_best, n, 200, deadline, rng)
        if c_f3 < best_c - 1e-12:
            h_best, best_c = h_f3, c_f3

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
