def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * dx

    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:

            pass
    return best_h, best_c

            a = 1.0 - hs_f - hs_b
            b = c[ms].copy()
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            f0 = max(float(np.max(b)), unaff_max)
            if _feasible_t(a, b, unaff_max, d_lo, d_hi, unaff_max):
                t_opt = unaff_max
            else:
                t_lo = unaff_max
                t_hi = f0
                for _ in range(50):
                    t_mid = (t_lo + t_hi) / 2.0
                    if _feasible_t(a, b, t_mid, d_lo, d_hi, unaff_max):
                        t_hi = t_mid
                    else:
                        t_lo = t_mid
                t_opt = t_hi
            d_opt = _find_d_opt(a, b, t_opt, d_lo, d_hi)
            f_new = max(float(np.max(b + d_opt * a)), unaff_max)
            if f_new < f0 - 1e-15:
                h[p] = h0 + d_opt
                c[ms] = b + d_opt * a
                improved = True
        if not improved:
            break
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * dx

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgrad_step(h, n, step, mom, beta=0.7):
    c = np.correlate(h, 1.0 - h, mode='full')
    m_star = int(np.argmax(c))
    s = m_star - (n - 1)
    grad = np.zeros(n)
    jf = np.arange(n) + s
    jb = np.arange(n) - s
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    grad[vf] += (1.0 - h[jf[vf]])
    grad[vb] -= h[jb[vb]]
    if s == 0:
        grad[:] = 1.0 - 2.0 * h
    mom = beta * mom + (1.0 - beta) * grad
    h_new = project_feasible(h - step * mom, n)
    return h_new, mom

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(700, 0.5)
        n = 700
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Exact CD
    if time.time() < deadline - 300:
        h1, c1 = exact_1d_cd(best_h, n, 80, deadline - 300)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: SLSQP polish
    if time.time() < deadline - 250:
        h2, c2 = slsqp_optimize(best_h, n, [5000, 20000, 80000], 200)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2, c2 = exact_1d_cd(best_h, n, 50, deadline - 250)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Subgradient escape + CD
    if time.time() < deadline - 200:
        mom = np.zeros(n)
        step = 0.04
        for it in range(300):
            if time.time() > deadline - 200:
                break
            step = max(0.003, 0.04 / (1.0 + 0.02 * it))
            h_new, mom = subgrad_step(best_h, n, step, mom)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_h, best_c = h_new, c_new
            best_h, best_c = exact_1d_cd(best_h, n, 10, deadline - 200)

    # Phase 4: Multi-restart
    remaining = deadline - time.time() - 120
    n_restarts = max(3, int(remaining / 50))
    for r in range(n_restarts):
        if time.time() > deadline - 120:
            break
        if r == 0:
            hp = project_feasible(best_h + rng.normal(0, 0.015, n), n)
        elif r == 1:
            x = np.linspace(0, 2, n)
            hp = project_feasible(best_h + 0.025 * np.sin(2 * np.pi * (3 + r) * x), n)
        elif r == 2:
            block = n // 4
            hp = best_h.copy()
            start = (r * block) % max(1, n - block)
            hp[start:start + block] = 1.0 - hp[start:start + block]
            hp = project_feasible(hp, n)
        elif r == 3:
            x = np.linspace(0, 2, n)
            hp = project_feasible(best_h + 0.02 * np.cos(4 * np.pi * x), n)
        else:
            scale = 0.01 + 0.02 * (r % 5)
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
        ho, co = exact_1d_cd(hp, n, 40, deadline - 120)
        if co < best_c:
            ho, co = slsqp_optimize(ho, n, [20000, 80000], 150)
            ho, co = exact_1d_cd(ho, n, 40, deadline - 120)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 5: Try different n
    for n_try in [500, 800, 1000]:
        if time.time() > deadline - 80:
            break
        if n_try == len(best_h):
            continue
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        ho, co = exact_1d_cd(h_int, n_try, 50, deadline - 80)
        if co < best_c:
            ho, co = slsqp_optimize(ho, n_try, [20000, 80000], 150)
            ho, co = exact_1d_cd(ho, n_try, 40, deadline - 80)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 6: Final polish
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        h_f, c_f = exact_1d_cd(best_h, n_cur, 60, deadline - 3)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = slsqp_optimize(best_h, n_cur, [50000, 200000], 300)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = exact_1d_cd(best_h, n_cur, 40, deadline)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
