def compute_c5(h, n):

    return h

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def fast_cd(h, n, max_passes, deadline, bisect_iters=30):
    h = h.copy()
    best_c = compute_c5(h, n)

    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(c.max())
        improved = False

        for p in range(n):
            if time.time() > deadline:
                return h, best_c
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue

            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)

            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)

            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]

            c_aff = c[ms]

            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))

            d_lo = -h0
            d_hi = 1.0 - h0
            for _ in range(bisect_iters):
                d_mid = 0.5 * (d_lo + d_hi)
                vals = c_aff + d_mid * a
                f_val = float(vals.max())
                active = np.where(vals >= f_val - 1e-9)[0]
                amin = float(a[active].min())
                amax = float(a[active].max())
                if amin > 1e-7:
                    d_hi = d_mid
                elif amax < -1e-7:
                    d_lo = d_mid
                else:
                    break

            d_opt = 0.5 * (d_lo + d_hi)
            new_max = max(float((c_aff + d_opt * a).max()), unaff_max)

            if new_max < c_max - 1e-10:
                h[p] = h0 + d_opt
                c[ms] = c_aff + d_opt * a
                c_max = new_max
                improved = True

        if improved:
            h[h < 1e-12] = 0.0
            h[h > 1.0 - 1e-12] = 1.0
            h = project_feasible(h, n)
            nc = compute_c5(h, n)
            if nc < best_c:
                best_c = nc
        else:
            break

    return h, best_c

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqq_polish(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            if res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
