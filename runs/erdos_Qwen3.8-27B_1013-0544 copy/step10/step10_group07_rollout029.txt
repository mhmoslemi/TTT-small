    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = c.max()
    s = alpha * (c - mc)
    sm = s.max()
    lse = mc + (sm + np.log(np.exp(s - sm).sum())) / alpha
    return lse * (2.0 / n)

def lse_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = c.max()
    s = alpha * (c - mc)
    sm = s.max()
    e = np.exp(s - sm)
    p = e / e.sum()
    cum = np.concatenate([[0.0], np.cumsum(p)])
    idx = np.arange(n)
    lo = np.maximum(0, n - 1 - idx)
    hi = np.minimum(2 * n - 2, n - 1 + idx)
    W = cum[hi + 1] - cum[lo]
    cp = np.correlate(x, p, mode='full')
    C1 = cp[3 * n - 3 - idx]
    T2 = cp[n - 1 + idx]
    return (2.0 / n) * (W - C1 - T2)

def slsqp_analytic(h0, n, alphas, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    bounds = [(0.0, 1.0)] * n
    sum_eq = lambda z: z.sum() - n / 2.0
    sum_jac = lambda z: np.ones(n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(lse_obj, best_h.copy(), args=(n, alpha),
                          method='SLSQP', jac=lambda x, *a: lse_grad(x, *a),
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
