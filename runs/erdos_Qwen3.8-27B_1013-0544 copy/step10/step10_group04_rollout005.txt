def project_l2(y, n):
    target = 0.5 * float(n)
    y = np.asarray(y, dtype=float).copy()
    lo = -float(np.max(y))

import numpy as np
import time


def corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")


def c5_of(c, n):
    return float(np.max(c) * (2.0 / float(n)))


def project_l2(y, n):
    target = 0.5 * float(n)
    y = np.asarray(y, dtype=float).copy()
    lo = -float(np.max(y))
    hi = 1.0 - float(np.min(y))
    if lo > hi:
        lo, hi = hi, lo
    best_mid = 0.5 * (lo + hi)
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y + mid, 0.0, 1.0)))
        best_mid = mid
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y + best_mid, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-14:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            add = min(diff, max(0.0, 1.0 - h[idx]))
            h[idx] += add
        else:
            idx = int(np.argmax(h))
            sub = min(-diff, h[idx])
            h[idx] -= sub
    return h


def enforce_exact_sum(h, n):
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 0.0:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
        else:
            idx = int(np.argmax(h))
            h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h


def make_off(n):
    off = np.ones(2 * n - 1, dtype=bool)
    off[n - 1] = False
    return off


def solve_coord(c, h, p, n, off, diag, iters=42, tie_alpha=6.0):

    pos_b = c_no[pos] if pos_flag else None

    for _ in range(iters):

def cd_polish(h, n, deadline, off, order=None, max_passes=200, reserve=1.0,
              iters=42, tie_alpha=6.0):
    if n <= 1:
        hh = np.array([0.5], dtype=float)
        return hh, 0.5

    h = project_l2(h, n)
    enforce_exact_sum(h, n)
    c = corr_full(h)
    best_val = c5_of(c, n)
    best_h = h.copy()
    cur_h = h.copy()
    cur_c = c.copy()
    cur_val = best_val
    if order is None:
        order = np.arange(n)
    diag = n - 1
    no_improve = 0

    for _ in range(max_passes):
        if time.time() > deadline - reserve:
            break

        h_w = cur_h.copy()
        c_w = cur_c.copy()
        completed = True
        stop_t = deadline - 0.5

        for p in order:
            if time.time() > stop_t:
                completed = False
                break
            x_new, a = solve_coord(c_w, h_w, p, n, off, diag, iters, tie_alpha)
            delta = x_new - h_w[p]
            if abs(delta) > 1e-14:
                c_w += delta * a
                c_w[diag] += delta * (1.0 - 2.0 * h_w[p]) - delta * delta
                h_w[p] = x_new

        if not completed:
            break

        h_w = project_l2(h_w, n)
        enforce_exact_sum(h_w, n)
        c_w = corr_full(h_w)
        val = c5_of(c_w, n)
        dh = float(np.max(np.abs(h_w - cur_h)))

        if val <= cur_val + 1e-9:
            cur_h, cur_c, cur_val = h_w, c_w, float(val)
        else:
            no_improve += 2

        if val < best_val - 1e-13:
            best_val = val
            best_h = h_w.copy()
            no_improve = 0
        else:
            no_improve += 1

        if (dh < 1e-13 and no_improve > 0) or no_improve >= 6:
            break

    return best_h, best_val


def interp_center(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old, dtype=float) + 0.5) / float(n_old)
    x_new = (np.arange(n_new, dtype=float) + 0.5) / float(n_new)
    return np.interp(x_new, x_old, h_old)

                h = rb.copy()

    if time.time() < deadline - 160:
        for nt in (550, 800, 450):
            if time.time() > deadline - 90:
                break
            h_int = interp_center(h, n, nt)
            h_int = project_l2(h_int, nt)
            enforce_exact_sum(h_int, nt)
            off_t = make_off(nt)
            h_try, v_try = cd_polish(
                h_int, nt, deadline, off_t,
                order=np.arange(nt), max_passes=35, iters=40
            )
            if v_try < best - 1e-7:
                best = float(v_try)
                h = h_try.copy()
                n = nt
                off = off_t

    if time.time() < deadline - 5:
        h, best = cd_polish(h, n, deadline, off, order=np.arange(n), max_passes=180, iters=50)

    h = project_l2(h, n)
    enforce_exact_sum(h, n)
    best = c5_of(corr_full(h), n)
    return h, best, n
