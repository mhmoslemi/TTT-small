def smooth_obj_and_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * c
    mx = shifted.max()
    exp_c = np.exp(shifted - mx)
    w = exp_c / exp_c.sum()  # softmax weights
    
    # Gradient: sum_k w[k] * dc[k]/dh[i]
    # dc[k]/dh[i] for k != n-1: (1-h[i+k-(n-1)]) - h[i+(n-1)-k]
    # For k = n-1: 1 - 2*h[i]
    
    # Let m = k - (n-1), ranges from -(n-1) to (n-1)
    # dc[k]/dh[i] = (1-h[i+m]) * I(0<=i+m<n) - h[i-m] * I(0<=i-m<n) for m != 0
    # For m = 0: 1 - 2*h[i]
    
    # sum_m w[m+(n-1)] * [(1-h[i+m]) * I(0<=i+m<n) - h[i-m] * I(0<=i-m<n)]
    # = sum_m w[m+(n-1)] * I(0<=i+m<n) - sum_m w[m+(n-1)] * h[i+m] * I(0<=i+m<n)
    #   - sum_m w[m+(n-1)] * h[i-m] * I(0<=i-m<n)
    
    # Term 1: sum_m w[m+(n-1)] * I(0<=i+m<n)
    # Term 2: - sum_m w[m+(n-1)] * h[i+m] * I(0<=i+m<n) = -(w * h)[i] (cross-corr)
    # Term 3: - sum_m w[m+(n-1)] * h[i-m] * I(0<=i-m<n) = -(w ⋆ h)[i] (cross-corr, other direction)
    
    # For term 1: it's a "box filter" type thing
    # sum_{m: 0<=i+m<n} w[m+(n-1)] = sum of w over the range of m
    
    # Let me just compute it directly for clarity (n=700 is small enough)
    # Actually, for n=700, O(n²) = 490000 per gradient eval is fine.
    
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for k in range(2*n-1):
            m = k - (n-1)
            jf = i + m
            jb = i - m
            contrib = 0.0
            if m == 0:
                contrib = 1.0 - 2.0*h[i]
            else:
                if 0 <= jf < n:
                    contrib += (1.0 - h[jf])
                if 0 <= jb < n:
                    contrib -= h[jb]
            s += w[k] * contrib
        grad[i] = s
    
    obj = (mx + np.log(exp_c.sum())) / alpha
    return obj, grad

    jb = ii - mm  # (n, 2n-1)

    vb = (jb >= 0) & (jb < n)

    # But we can't index h with jf directly when jf is out of bounds.

    h_padded[n-1:n-1+n] = h  # center h in the padded array

    # In the padded array, this corresponds to indices jf + (n-1)

    jb_padded = jb + (n - 1)

    h_jb = h_padded[jb_padded] * vb
    # Wait, h_padded has h at positions n-1 to 2n-2.
    # jf_padded = i + m + (n-1) = i + (k-(n-1)) + (n-1) = i + k
    # For i in [0,n-1] and k in [0,2n-2]: jf_padded ranges from 0 to n-1+2n-2 = 3n-3
    # But h_padded has length n + 2n - 1 = 3n - 1. OK that works.
    # h_padded[jf_padded] is h[i+k-(n-1)] when 0<=i+k-(n-1)<n, else 0.
    # That's exactly what we want!
    
    dc_matrix = (1.0 - h_jf) * vf - h_jb * vb  # (n, 2n-1)
    
    # But wait, for the diagonal (k=n-1, m=0): jf = i, jb = i, both valid
    # dc = (1-h[i]) - h[i] = 1 - 2*h[i]. Correct!
    
    grad = dc_matrix @ w  # (n,) = sum_k dc_matrix[i,k] * w[k]
    
    obj = (mx + np.log(np.sum(exp_c))) / alpha
    return obj, grad
