def project_feasible(h, n):
    # Similar to before but faster
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15: break
            h = np.clip(h + diff/s*room, 0, 1)
        else:
            s = h.sum()
            if s < 1e-15: break
            h = np.clip(h + diff/s*h, 0, 1)
    return h

def subgradient_of_max_corr(h, n, top_k=5):
    """Compute subgradient of max_k c_k(h), averaging over top_k correlations."""
    c = np.correlate(h, 1.0 - h, mode='full')
    c_sorted = np.sort(c)[::-1]
    threshold = c_sorted[min(top_k, len(c_sorted))-1]
    active = np.where(c >= threshold)[0]
    
    grad = np.zeros(n)
    for k in active:
        # ∂c[k]/∂h[j] = (1-h[j-k])*1_{0≤j-k<n} - h[j+k]*1_{0≤j+k<n}
        # For each j:
        # Term 1: if 0 ≤ j-k < n, add (1-h[j-k])
        # Term 2: if 0 ≤ j+k < n, subtract h[j+k]
        if k >= 0:

            j_start = k
            j_end = n

            if j_start < j_end:
                grad[j_start:j_end] += (1.0 - h[0:j_end-j_start])
            # j+k in [0, n-1] means j in [-k, n-1-k], intersect [0, n-1] = [0, n-1-k]
            j_start2 = 0
            j_end2 = n - k
            if j_start2 < j_end2:

    t = 1.0

    best_c = compute_c5(h, n)

```python
def subgradient_of_max_corr(h, n, top_k=5):
    c = np.correlate(h, 1.0 - h, mode='full')
    # Find top_k active correlations
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    
    grad = np.zeros(n)
    for k in top_indices:
        if k >= n - 1:
            s = k - (n - 1)  # shift >= 0
            if s == 0:
                grad += (1.0 - h) - h  # (1-h[j]) - h[j] for all j
            else:
                # For j in [s, n-1]: add (1-h[j-s])
                grad[s:] += (1.0 - h[:n-s])
                # For j in [0, n-1-s]: subtract h[j+s]
                grad[:n-s] -= h[s:]
        else:
            s = (n - 1) - k  # s > 0, shift = -s
            # For j in [0, n-1-s]: add (1-h[j+s])
            grad[:n-s] += (1.0 - h[s:])
            # For j in [s, n-1]: subtract h[j-s]
            grad[s:] -= h[:n-s]
    
    grad /= len(top_indices)
    return grad, c.max(), c
```

This is O(top_k * n) per evaluation, which is fast.

Now for the FISTA method, I need to be careful about step size. The subgradient method with fixed step size converges slowly. FISTA with step size 1/L (where L is the Lipschitz constant of the gradient) would be better if the function were smooth, but for a max of smooth functions, it's only subdifferentiable.

Actually, for the max of smooth functions, we can use a "smooth max" approximation:
f_α(h) = (1/α) * log(Σ exp(α*c_k))

This is smooth with Lipschitz gradient, and as α→∞, it approaches the max. The gradient is:
∇f_α(h) = (1/α) * Σ_k (α*exp(α*c_k) / Σ exp(α*c_j)) * ∇c_k
= Σ_k p_k * ∇c_k

where p_k = exp(α*c_k) / Σ exp(α*c_j) is a softmax weight.

This is a weighted average of the gradients, with weights concentrated on the maximum terms. This is actually a better subgradient than just averaging over the top-k!

Let me use this approach:

```python
def smooth_subgradient(h, n, alpha=100):
    c = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    p = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    # Weight sum by p
    # Only iterate over significant p's
    threshold = 0.01
    active = np.where(p > threshold)[0]
    
    for k in active:
        pk = p[k]
        if k >= n - 1:
            s = k - (n - 1)
            if s == 0:
                grad += pk * ((1.0 - h) - h)
            else:
                grad[s:] += pk * (1.0 - h[:n-s])
                grad[:n-s] -= pk * h[s:]
        else:
            s = (n - 1) - k
            grad[:n-s] += pk * (1.0 - h[s:])
            grad[s:] -= pk * h[:n-s]
    
    return grad, c.max() * (2.0/n)
```

Then I can use gradient descent with backtracking or fixed step size on this smooth approximation.

For the step size, the Lipschitz constant of ∇f_α is α * (max eigenvalue of Hessian of c_k) which is bounded by α * 4 (roughly, since c_k is bilinear). So step size ~ 1/(4α) would be safe but slow.

In practice, I'll use a moderate α (like 100-1000) and a step size of maybe 0.01-0.1, with backtracking.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(150):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = np.clip(h + diff / s * room, 0, 1)
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = np.clip(h + diff / s * h, 0, 1)
    return h

def _corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_corr_k(h, n, k, out):
    """Add gradient of c[k] to out."""
    if k >= n - 1:
        s = k - (n - 1)
        if s == 0:
            out += (1.0 - 2.0 * h)
        else:
            out[s:] += (1.0 - h[:n - s])
            out[:n - s] -= h[s:]
    else:
        s = (n - 1) - k
        out[:n - s] += (1.0 - h[s:])
        out[s:] -= h[:n - s]

def smooth_max_gradient(h, n, alpha):
    c = _corr(h, n)
    cmax = c.max()
    exp_c = np.exp(alpha * (c - cmax))
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    thresh = 1.0 / (len(c) * 5)
    active = np.where(p > thresh)[0]
    for k in active:
        _grad_corr_k(h, n, k, grad)
        grad *= p[k]
    return grad, float(cmax * 2.0 / n)

def sum_sq_gradient(h, n):
    c = _corr(h, n)
    grad = np.zeros(n)
    for k in range(len(c)):
        _grad_corr_k(h, n, k, grad)
        grad *= 2.0 * c[k]
    return grad, float(np.sum(c * c) * (2.0 / n) ** 2)

def fista_smooth(h, n, alpha, max_iter, deadline, mode='max'):
    h = h.copy().astype(float)
    y = h.copy()
    t = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5
    dx = 2.0 / n
    stuck = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if mode == 'max':
            g, _ = smooth_max_gradient(y, n, alpha)
        else:
            g, _ = sum_sq_gradient(y, n)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        z = y - lr * g / gnorm * 0.1
        z = project_feasible(z, n)
        c = compute_c5(z, n)
        if c < best_c:
            best_h = z.copy()
            best_c = c
            stuck = 0
            lr = min(lr * 1.05, 1.0)
        else:
            stuck += 1
            if stuck > 20:
                lr *= 0.9
                stuck = 0
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = z + (t / t_new - 1.0) * (z - h)
        h = z.copy()
        t = t_new
    return best_h, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = _corr(h, n)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = c.max() if unaff_max < c.max() else unaff_max
            if f_opt < f_curr - 1e-13:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h[np.where(h < 1e-6)[0]] = 0.0
            h[np.where(h > 1.0 - 1e-6)[0]] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    return h, min(best_c, compute_c5(h, n))

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def smooth_obj(x, n, alpha):
    c = _corr(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            if res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def binary_sa(h, n, deadline, max_iters=8000):
    rng = np.random.RandomState(123)
    b = (h > 0.5).astype(float)
    num_ones = int(b.sum())
    target = n // 2
    if num_ones > target:
        ones = np.where(b == 1)[0]
        for i in np.random.choice(ones, num_ones - target, replace=False):
            b[i] = 0
    elif num_ones < target:
        zeros = np.where(b == 0)[0]
        for i in np.random.choice(zeros, target - num_ones, replace=False):
            b[i] = 1
    best_b = b.copy()
    best_c = compute_c5(b, n)
    T = 0.02
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, 2, replace=False)
        if b[i] == b[j]:
            continue
        b[i], b[j] = b[j], b[i]
        c_new = compute_c5(b, n)
        delta = c_new - best_c
        T *= 0.9997
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            if c_new < best_c:
                best_b = b.copy()
                best_c = c_new
        else:
            b[i], b[j] = b[j], b[i]
    return best_b, best_c

    # Phase 3: Sum-of-squares gradient
    if time.time() < deadline - 150:
        hf, cf = fista_smooth(best_h, n, 0, 300, deadline, mode='sq')
        if cf < best_c:
            best_h, best_c = hf, cf
        h3, c3 = coord_descent(hf, n, 10, deadline)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Multi-restart
    if time.time() < deadline - 150:
        x = np.linspace(0, 2, n)
        n_restarts = max(4, int((deadline - time.time() - 80) / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 60:
                break
            if r % 3 == 0:
                hp = project_feasible(best_h + rng.normal(0, 0.01, n), n)
            elif r % 3 == 1:
                freq = 2.0 + 2.0 * r
                hp = project_feasible(best_h + 0.03 * np.sin(np.pi * freq * x), n)
            else:
                mask = (rng.rand(n) < 0.08).astype(float)
                hp = project_feasible(best_h + 0.4 * mask - 0.2, n)
            ho, co = fista_smooth(hp, n, 500, 150, deadline, mode='max')
            ho, co = coord_descent(ho, n, 15, deadline)
            if co < best_c:
                best_h, best_c = ho, co
    # Phase 5: Binary SA + relaxation
    if time.time() < deadline - 200:
        bb, cb = binary_sa(best_h, n, deadline, 6000)
        if cb < best_c + 0.01:
            h_rel = project_feasible(bb + 0.05 * rng.rand(n), n)
            h_rel, c_rel = coord_descent(h_rel, n, 20, deadline)
            hf2, cf2 = fista_smooth(h_rel, n, 1000, 200, deadline, mode='max')
            hf2, cf2 = coord_descent(hf2, n, 10, deadline)
            if cf2 < best_c:
                best_h, best_c = hf2, cf2
    # Phase 6: SLSQP polish
    if time.time() < deadline - 50:
        hs, cs = slsqp_optimize(best_h, n, [5000, 20000], 400)
        if cs < best_c:
            best_h, best_c = hs, cs
    # Phase 7: Final CD
    hf_final, cf_final = coord_descent(best_h, n, 25, deadline, gs_iters=60)
    if cf_final < best_c:
        best_h, best_c = hf_final, cf_final
    # Final SLSQP if time allows
    if time.time() < deadline - 5:
        fs, cs = slsqp_optimize(best_h, n, [100000], 500)
        if cs < best_c:
            best_h, best_c = fs, cs
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c, n
