import time
import numpy as np


def project_sum_box(z, n):
    z = np.asarray(z, dtype=float)
    target = 0.5 * n
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    out = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(out))
    if abs(res) > 1e-14:
        if res > 0.0:
            idx = int(np.argmax(1.0 - out))
            out[idx] = min(1.0, out[idx] + res)
        else:
            idx = int(np.argmax(out))
            out[idx] = max(0.0, out[idx] + res)
    return out


def c5_of(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * (2.0 / n)


def compute_b(p, h, n):
    m0 = n - 1
    b = np.zeros(2 * n - 1)
    b[m0] = 1.0 - 2.0 * h[p]
    right = n - 1 - p
    if right > 0:
        seg = h[p + 1:n]
        b[m0 + 1:m0 + right + 1] += 1.0 - seg
        b[m0 - right:m0] -= seg[::-1]
    left = p
    if left > 0:
        seg = h[p - left:p]
        b[m0 - left:m0] += 1.0 - seg
        b[m0 + 1:m0 + left + 1] -= seg[::-1]
    return b


def eval_pair_move(c, a, a0, c0, m0, m1, m2, d):

        val = float(np.max(c0 + a0 * d))
    else:

    if v > val:
        val = v
    v = c[m2] + a[m2] * d + d2
    if v > val:
        val = v
    return val


def line_search_pair(c, a, m0, m1, m2, dlo, dhi, iters):

def pair_move(h, c, p, q, n, iters):

        return h, c, False
    hp = h[p]
    hq = h[q]
    dlo = max(-hp, hq - 1.0)

        d = -hp
    if hp + d > 1.0:
        d = 1.0 - hp
    if hq - d < 0.0:
        d = hq
    if hq - d > 1.0:
        d = 1.0 - hq
    if abs(d) < 1e-15:
        return h, c, False
    c += a * d
    d2 = d * d
    c[m0] -= 2.0 * d2
    c[m1] += d2
    c[m2] += d2
    h[p] = hp + d
    h[q] = hq - d
    if h[p] < 0.0:
        h[p] = 0.0
    elif h[p] > 1.0:
        h[p] = 1.0
    if h[q] < 0.0:
        h[q] = 0.0
    elif h[q] > 1.0:
        h[q] = 1.0
    return h, c, True


def reservoir_cd(h, c, n, deadline, iters=50):
    target = 0.5 * n
    best_h = h.copy()
    best_count = float(np.max(c))
    rng = np.random.RandomState(11)
    bad = 0
    passno = 0
    while time.time() < deadline:
        dist = np.abs(h - 0.5)
        cand = np.argsort(dist)
        k = min(10, n)
        r = int(cand[int(rng.randint(k))])
        order = rng.permutation(n)
        improved = 0
        for px in order:
            if time.time() > deadline:
                break
            p = int(px)
            if p == r:
                continue
            h, c, imp = pair_move(h, c, p, r, n, iters)
            if imp:
                improved += 1
        c = np.correlate(h, 1.0 - h, mode="full")
        if abs(float(np.sum(h)) - target) > 1e-9:
            h = project_sum_box(h, n)
            c = np.correlate(h, 1.0 - h, mode="full")
        cur_count = float(np.max(c))
        if cur_count < best_count - 1e-8:
            best_count = cur_count
            best_h = h.copy()
        if cur_count > best_count + 1e-6:
            h = best_h.copy()
            c = np.correlate(h, 1.0 - h, mode="full")
            cur_count = float(np.max(c))
        if improved == 0:
            bad += 1
        else:
            bad = 0
        passno += 1
        if bad >= 4:
            break
    return best_h, best_count * (2.0 / n)


def gradient_linear(w, h, n):
    m0 = n - 1
    mlen = 2 * n - 1
    cum = np.empty(mlen + 1)
    cum[0] = 0.0
    cum[1:] = np.cumsum(w)
    idx = np.arange(n)
    A = cum[mlen - idx] - cum[m0 - idx]
    v = w + w[::-1]
    B = np.convolve(h, v[::-1], mode="full")[m0:m0 + n]
    return A - B


def grad_polish(h, c, n, deadline):
    best_h = h.copy()
    best_count = float(np.max(c))
    step_sizes = (0.18, 0.09, 0.045, 0.0225, 0.01125)
    for beta in (80.0, 350.0, 1200.0):
        for _ in range(45):
            if time.time() > deadline - 2:
                break
            c = np.correlate(h, 1.0 - h, mode="full")
            cm = float(np.max(c))
            z = beta * (c - cm)
            s = float(np.sum(np.exp(z)))
            if s <= 0.0:
                break
            w = np.exp(z) / s
            g = gradient_linear(w, h, n)
            ng = float(np.linalg.norm(g))
            if ng < 1e-12:
                break
            v = -g / ng
            accepted = None
            for st in step_sizes:
                hp = project_sum_box(h + st * v, n)
                cp = np.correlate(hp, 1.0 - hp, mode="full")
                pc = float(np.max(cp))
                if pc < cm - 1e-9:
                    accepted = (hp, cp, pc)
                    break
            if accepted is None:
                break
            h, c, pc = accepted
            if pc < best_count:
                best_count = pc
                best_h = h.copy()
    return best_h, best_count * (2.0 / n)


def interp_to(h, n2):
    n1 = len(h)
    if n1 == n2:
        return h.copy()
    x1 = np.linspace(0.0, 2.0, n1)
    x2 = np.linspace(0.0, 2.0, n2)
    return np.interp(x2, x1, h)

    best_h = project_sum_box(best_h, best_n)
