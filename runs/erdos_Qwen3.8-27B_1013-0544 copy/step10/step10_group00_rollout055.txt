import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _lse_val(h, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    shifted = alpha * (c - cmax) - alpha * (c - cmax).max()
    val = (cmax + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    s = h.sum() - n / 2.0
    val += mu * s * s
    return val

def _lse_grad(h, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    shifted = alpha * (c - cmax)
    shifted = shifted - shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(-(n-1), -j)
        s_hi = min(n-1, n-1-j)
        if s_hi >= s_lo:
            idx = np.arange(s_lo, s_hi + 1) + (n - 1)
            hj = j + np.arange(s_lo, s_hi + 1)
            grad[j] += np.dot(w[idx], 1.0 - h[hj])
        s2_lo = max(-(n-1), j-(n-1))
        s2_hi = min(n-1, j)
        if s2_hi >= s2_lo:
            idx2 = np.arange(s2_lo, s2_hi + 1) + (n - 1)
            hj2 = j - np.arange(s2_lo, s2_hi + 1)
            grad[j] -= np.dot(w[idx2], h[hj2])
    grad *= dx
    s = h.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def lbfghs_optimize(h, n, alphas, mu, deadline, maxiter=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

        if time.time() > deadline - 30:

        new_c = compute_c5(new_h, n)
        if new_c < best_c:

    return best_h, best_c

    for _ in range(max_passes):
        if time.time() > deadline:

        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False

            if time.time() > deadline:

            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:

            m_lo = max(0, n - 1 - p)

            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):

                result = float(np.max(vals))
                if unaff_max > result:

                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _lse_obj(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    cmax = c.max()
    sh = alpha * (c - cmax)
    sh -= sh.max()
    val = (cmax + np.log(np.sum(np.exp(sh))) / alpha) * dx
    s = x.sum() - n / 2.0
    val += mu * s * s
    return val

def _lse_grad(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    cmax = c.max()
    sh = alpha * (c - cmax)
    sh -= sh.max()
    w = np.exp(sh)
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(-(n-1), -j)
        s_hi = min(n-1, n-1-j)
        if s_hi >= s_lo:
            idx = np.arange(s_lo, s_hi+1) + (n-1)
            hj = j + np.arange(s_lo, s_hi+1)
            grad[j] += np.dot(w[idx], 1.0 - x[hj])
        s2_lo = max(-(n-1), j-(n-1))
        s2_hi = min(n-1, j)
        if s2_hi >= s2_lo:
            idx2 = np.arange(s2_lo, s2_hi+1) + (n-1)
            hj2 = j - np.arange(s2_lo, s2_hi+1)
            grad[j] -= np.dot(w[idx2], x[hj2])
    grad *= dx
    s = x.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals = vals.copy()
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def lbfghs_phase(h, n, deadline, mu=200.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [40, 80, 160, 320, 640, 1280, 2560, 5120]
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        try:
            res = minimize(_lse_obj, best_h, jac=_lse_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-12})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            dpos = np.where(ms == (n-1))[0]
            has_d = len(dpos) > 0
            di = int(dpos[0]) if has_d else -1
            c_aff = c[ms]
            umax = 0.0
            if m_lo > 0:
                umax = max(umax, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                umax = max(umax, float(c[m_hi+1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr*(d_hi - d_lo)
            d2 = d_lo + gr*(d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, di, has_d, umax, h0, d1)
            f2 = _cd_fval(c_aff, a, di, has_d, umax, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr*(d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, di, has_d, umax, h0, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr*(d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, di, has_d, umax, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            if _cd_fval(c_aff, a, di, has_d, umax, h0, d_opt) < \
               _cd_fval(c_aff, a, di, has_d, umax, h0, 0.0) - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick CD to establish local optimum
    if time.time() < deadline - 400:
        h1, c1 = coord_descent(best_h, n, 40, deadline - 400, gs_iters=50)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: L-BFGS-B with log-sum-exp (key new component)
    if time.time() < deadline - 250:
        h2, c2 = lbfghs_phase(best_h, n, deadline - 250, mu=200.0)
        if c2 < best_c:
            best_h, best_c = h2, c2
        # CD polish after L-BFGS-B
        if time.time() < deadline - 200:
            h2b, c2b = coord_descent(best_h, n, 30, deadline - 200, gs_iters=55)
            if c2b < best_c:
                best_h, best_c = h2b, c2b

    # Phase 3: Second L-BFGS-B round with higher mu
    if time.time() < deadline - 150:
        h3, c3 = lbfghs_phase(best_h, n, deadline - 150, mu=1000.0)
        if c3 < best_c:
            best_h, best_c = h3, c3
        if time.time() < deadline - 100:

            if c3b < best_c:
                best_h, best_c = h3b, c3b

    # Phase 4: Resolution adaptation
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [500, 800, 1000, 600]:
        if time.time() > deadline - 100:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = coord_descent(h_int, n_try, 50, deadline - 100, gs_iters=50)
        if c_o < best_c and time.time() < deadline - 60:
            h_o, c_o = lbfghs_phase(h_o, n_try, deadline - 60, mu=500.0)
            h_o, c_o = coord_descent(h_o, n_try, 30, deadline - 60, gs_iters=50)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = np.linspace(0, 2, n_try)

    # Phase 5: Diverse restarts
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 60
        n_restarts = max(6, int(remaining / 35))
        for r in range(n_restarts):
            if time.time() > deadline - 60:
                break
            mode = r % 5
            if mode == 0:
                scale = 0.02 + 0.04 * (r / max(1, n_restarts - 1))
                hp = project_feasible(best_h + rng.normal(0, scale, n_cur), n_cur)
            elif mode == 1:
                x = np.linspace(0, 2, n_cur)
                freq = 2 + 3 * r
                hp = project_feasible(best_h + 0.05 * np.sin(2 * np.pi * freq * x), n_cur)
            elif mode == 2:
                hp = project_feasible(best_h + 0.03 * rng.randn(n_cur), n_cur)
            elif mode == 3:
                hp = project_feasible(1.0 - best_h, n_cur)
                hp = project_feasible(hp + 0.02 * rng.randn(n_cur), n_cur)
            else:
                block = max(10, n_cur // 5)
                hp = best_h.copy()
                start = (r * block) % max(1, n_cur - block)
                hp[start:start+block] = 1.0 - hp[start:start+block]
                hp = project_feasible(hp, n_cur)
            h_o, c_o = coord_descent(hp, n_cur, 25, deadline - 60, gs_iters=45)
            if c_o < best_c and time.time() < deadline - 30:
                h_o, c_o = lbfghs_phase(h_o, n_cur, deadline - 30, mu=500.0)
                h_o, c_o = coord_descent(h_o, n_cur, 20, deadline - 30, gs_iters=50)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 6: Final polish
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        h_f, c_f = lbfghs_phase(best_h, n_cur, deadline - 10, mu=5000.0)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = coord_descent(best_h, n_cur, 50, deadline - 5, gs_iters=60)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
