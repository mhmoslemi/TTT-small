    if x_lo > x_hi: return h0
    return float(np.clip((x_lo+x_hi)/2.0, 0.0, 1.0))

def update_c(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-15: return
    m_min = max(0, n-1-p)
    m_max = min(2*n-2, 2*n-2-p)
    ms = np.arange(m_min, m_max+1)
    jf = p + ms - (n-1)
    jb = p - ms + (n-1)
    a = np.zeros(len(ms))
    vf = (jf >= 0) & (jf < n)
    a[vf] += (1.0 - h[jf[vf]])
    vb = (jb >= 0) & (jb < n)
    a[vb] -= h[jb[vb]]
    c[ms] += delta * a
    di = np.where(ms == (n-1))[0]
    if len(di) > 0:
        c[ms[di[0]]] -= delta*delta

def cd_pass(h, n, c, max_passes, deadline, order_fn=None, skip_bounds=True):
    improved = True
    passes = 0
    while improved and passes < max_passes and time.time() < deadline:
        improved = False
        passes += 1
        if order_fn:
            order = order_fn(n)
        else:
            order = range(n)
        for p in order:
            h0 = h[p]
            if skip_bounds and (h0 < 1e-12 or h0 > 1.0-1e-12):
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                update_c(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if improved:
            h = project_feasible(h, n)
            if abs(h.sum() - n/2.0) > 1e-9:
                c = np.correlate(h, 1.0-h, mode='full')
    return h, c

def quick_c5(h, n):
    c = np.correlate(h, 1.0-h, mode='full')
    return float(np.max(c) * 2.0/n)

def gen_perturbations(h, n, rng, M, deadline):
    """Generate M diverse perturbations"""
    cands = []
    for i in range(M):
        if time.time() > deadline: break
        t = i / M
        # Mix of perturbation types
        r = rng.rand()
        if r < 0.3:
            # Global Gaussian
            hp = h + rng.normal(0, 0.01 + 0.05*t, n)
        elif r < 0.5:
            # Block perturbation
            bp = n // 10
            idx = rng.randint(0, n-bp)
            hp = h.copy()
            hp[idx:idx+bp] = rng.uniform(0, 1, bp)
        elif r < 0.7:
            # Swap-based
            hp = h.copy()
            for _ in range(n//20):
                i, j = rng.choice(n, 2, replace=False)
                hp[[i,j]] = hp[[j,i]]
        elif r < 0.85:
            # Flip-based (complement a block)
            bp = n // 15
            idx = rng.randint(0, n-bp)
            hp = h.copy()
            hp[idx:idx+bp] = 1.0 - hp[idx:idx+bp]
        else:
            # Multi-scale: large perturbation on subset
            subset = rng.choice(n, n//5, replace=False)
            hp = h.copy()
            hp[subset] = rng.uniform(0, 1, len(subset))
        hp = project_feasible(hp, n)
        c5 = quick_c5(hp, n)
        cands.append((c5, hp))
    cands.sort(key=lambda x: x[0])
    return cands

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff/tot)*room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff/tot)*h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    m_min = max(0, n-1-p)
    m_max = min(2*n-2, 2*n-2-p)
    ms = np.arange(m_min, m_max+1)
    jf = p + ms - (n-1)
    jb = p - ms + (n-1)
    a = np.zeros(len(ms))
    vf = (jf >= 0) & (jf < n)
    a[vf] += (1.0 - h[jf[vf]])
    vb = (jb >= 0) & (jb < n)
    a[vb] -= h[jb[vb]]
    c_base = c[ms]
    has_diag = (n-1) >= m_min and (n-1) <= m_max
    unaff = 0.0
    if m_min > 0: unaff = float(np.max(c[:m_min]))
    if m_max < 2*n-2: unaff = max(unaff, float(np.max(c[m_max+1:])))
    cur_max = float(np.max(c))
    neg_m = a < -1e-14
    pos_m = a > 1e-14
    zero_m = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(45):
        if L_hi - L_lo < 1e-13: break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        ok = True
        if np.any(zero_m) and np.any(c_base[zero_m] > L + 1e-13): ok = False
        if ok and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if ok and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
        if x_lo > x_hi + 1e-11: ok = False
        if ok and has_diag:
            di = np.where(ms == n-1)[0]
            if len(di) > 0:
                cd, ad = c_base[di[0]], a[di[0]]
                disc = ad*ad + 4.0*(cd - L)
                if disc >= 0:
                    sq = np.sqrt(disc)
                    if x_lo > h0 + (ad-sq)/2.0 and x_hi < h0 + (ad+sq)/2.0: ok = False
        L_hi = L if ok else L_hi
        if not ok: L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_m): x_lo = max(x_lo, float(np.max((L - c_base[neg_m])/a[neg_m])))
    if np.any(pos_m): x_hi = min(x_hi, float(np.min((L - c_base[pos_m])/a[pos_m])))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if has_diag:
        di = np.where(ms == n-1)[0]
        if len(di) > 0:
            cd, ad = c_base[di[0]], a[di[0]]
            disc = ad*ad + 4.0*(cd - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1, r2 = (ad-sq)/2.0, (ad+sq)/2.0
                if x_lo <= h0+r1: x_hi = min(x_hi, h0+r1)
                elif x_hi >= h0+r2: x_lo = max(x_lo, h0+r2)
                else:
                    if abs(x_lo-(h0+r1)) <= abs(x_hi-(h0+r2)): x_hi = min(x_hi, h0+r1)
                    else: x_lo = max(x_lo, h0+r2)
    if x_lo > x_hi: return h0
    return float(np.clip((x_lo+x_hi)/2.0, 0.0, 1.0))

def update_c(c, h, p, old, new, n):
    d = new - old
    if abs(d) < 1e-15: return
    m_min, m_max = max(0, n-1-p), min(2*n-2, 2*n-2-p)
    ms = np.arange(m_min, m_max+1)
    jf, jb = p+ms-(n-1), p-ms+(n-1)
    a = np.zeros(len(ms))
    vf = (jf >= 0) & (jf < n)
    a[vf] += (1.0 - h[jf[vf]])
    vb = (jb >= 0) & (jb < n)
    a[vb] -= h[jb[vb]]
    c[ms] += d * a
    di = np.where(ms == n-1)[0]
    if len(di) > 0: c[ms[di[0]]] -= d*d

def _cons_fun(x):
    return x.sum() - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h0, n, alphas, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    for alpha in alphas:
        if time.time() > deadline: break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=[(0.0, 1.0)]*n,
                          constraints=[{'type':'eq','fun':_cons_fun,'jac':_cons_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c: best_h, best_c = nh, nc
        except Exception: pass
    return best_h, best_c
