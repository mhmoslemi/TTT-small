    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

            if tot < 1e-14:

def smooth_grad(h, n, alpha):
    """Compute smooth max value and gradient."""
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    shifted = alpha * (c - c.max())
    es = np.exp(shifted)
    ws = es / es.sum()
    logsum = c.max() + np.log(es.sum()) / alpha

    second = np.convolve(h, ws, mode='full')[n-1:2*n-1]
    grad = (first - second) * dx
    return val, grad

def exact_cd(h, n, max_passes, deadline):
    """Exact coordinate descent."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            # Solve 1D subproblem: minimize max_k (c_k + delta * a_k - delta^2 * d_k)
            # where a_k is the linear coefficient and d_k = 1 if k = n-1 (diagonal)
            m_min = n - 1 - p
            m_max = 2 * n - 2 - p
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            c_base = c[ms].copy()
            if has_diag:
                c_diag = c_base[diag_idx[0]]
                a[diag_idx] = 0.0
            # Unaffected correlations
            unaff = 0.0
            if m_min > 0:
                unaff = float(np.max(c[:m_min]))
            if m_max < 2 * n - 2:
                unaff = max(unaff, float(np.max(c[m_max + 1:])))
            cur_max = float(np.max(c))
            # Bisection on L
            neg_mask = a < -1e-15
            pos_mask = a > 1e-15
            zero_mask = np.abs(a) <= 1e-15
            L_lo, L_hi = unaff, cur_max
            for _ in range(50):
                if L_hi - L_lo < 1e-13:
                    break
                L = (L_lo + L_hi) / 2.0
                x_lo, x_hi = 0.0, 1.0
                feasible = True
                if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
                    feasible = False
                if feasible and np.any(neg_mask):
                    x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
                if feasible and np.any(pos_mask):
                    x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
                if x_lo > x_hi + 1e-12:
                    feasible = False
                if feasible and has_diag:
                    Cq = L - c_diag + h0 - h0 * h0
                    disc = 1.0 - 4.0 * Cq
                    if disc >= 0:
                        r1 = (1.0 - np.sqrt(disc)) / 2.0
                        r2 = (1.0 + np.sqrt(disc)) / 2.0
                        xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                        if not (xc_lo < r1 and xc_hi > r2):
                            feasible = False
                if feasible:
                    L_hi = L
                else:
                    L_lo = L
            L = L_hi
            x_lo, x_hi = 0.0, 1.0
            if np.any(neg_mask):
                x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
            if np.any(pos_mask):
                x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
            x_lo = max(x_lo, 0.0)
            x_hi = min(x_hi, 1.0)
            if has_diag:
                Cq = L - c_diag + h0 - h0 * h0
                disc = 1.0 - 4.0 * Cq
                if disc >= 0:
                    r1 = (1.0 - np.sqrt(disc)) / 2.0
                    r2 = (1.0 + np.sqrt(disc)) / 2.0
                    if h0 - r1 <= r2 - h0:
                        x_hi = min(x_hi, r1)
                    else:
                        x_lo = max(x_lo, r2)
            if x_lo > x_hi:
                xn = h0
            else:
                xn = float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                if np.any(vf):
                    c[ms[vf]] += delta * (1.0 - h[jf[vf]])
                if np.any(vb):
                    c[ms[vb]] -= delta * h[jb[vb]]
                if has_diag:
                    c[ms[diag_idx[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)
