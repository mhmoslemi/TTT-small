def get_orientation():

def direct_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def internal_c5(c, n):
    return float(np.max(c) * 2.0 / n)

def project_feasible(v, n):
    t = 0.5 * float(n)
    v = np.asarray(v, dtype=float)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > t:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = t - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 0.0) & (y < 1.0)
        if free.any():
            y[free] += diff / float(free.sum())
            y = np.clip(y, 0.0, 1.0)
            diff = t - y.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                room = 1.0 - y
                tot = room.sum()
                if tot > 0:
                    y = y + diff / tot * room
            else:
                room = y
                tot = room.sum()
                if tot > 0:
                    y = y + diff / tot * room
    return np.clip(y, 0.0, 1.0)

def make_affine(p, h, n):
    kmin = min(p - (n - 1), -p)
    kmax = max(p, n - 1 - p)
    L = kmax - kmin + 1
    A = np.zeros(L)
    a1lo = max(kmin, p - (n - 1))
    a1hi = min(kmax, p)
    if a1lo <= a1hi:
        s = a1lo - kmin
        ln = a1hi - a1lo + 1
        sj = p - a1hi

        A[s:s + ln] += 1.0 - h[sj:ej][::-1]

        A[s:s + ln] -= h[p + a2lo:p + a2hi + 1]

            hi = min(hi, float(np.min((L - cs[pos]) / A[pos])))
        if ok and neg.any():
            lo = max(lo, float(np.max((L - cs[neg]) / A[neg])))
        if ok and lo > hi + 1e-12:
            ok = False
        if ok:
            D = a1 * a1 - 4.0 * (L - cd)
            if D >= 0:
                r = math.sqrt(D)
                r1 = 0.5 * (a1 - r)
                r2 = 0.5 * (a1 + r)
                if lo > r1 and hi < r2:
                    ok = False
        if ok:
            hiL = L
        else:
            loL = L
        if hiL - loL < 1e-10 * max(1.0, abs(hiL)):
            break
    lo = dm
    hi = dp
    if z.any() and cs[z].max() > hiL + 1e-10:
        return 0.0
    if pos.any():
        hi = min(hi, float(np.min((hiL - cs[pos]) / A[pos])))
    if neg.any():
        lo = max(lo, float(np.max((hiL - cs[neg]) / A[neg])))
    if hi < lo:
        if lo - hi > 1e-9:
            return 0.0
        m = 0.5 * (lo + hi)
        lo = hi = m
    lo = max(lo, dm)
    hi = min(hi, dp)
    D = a1 * a1 - 4.0 * (hiL - cd)
    forb = False
    r1 = r2 = 0.0
    if D >= 0:
        r = math.sqrt(D)
        r1 = 0.5 * (a1 - r)
        r2 = 0.5 * (a1 + r)
        forb = True
    cl = [dm, dp, 0.0, lo, hi]
    if forb:
        cl.append(r1)
        cl.append(r2)
        cl.append(lo + 0.25 * (hi - lo))
        cl.append(lo + 0.5 * (hi - lo))
        cl.append(lo + 0.75 * (hi - lo))
    best_v = f0
    best_d = 0.0
    for dd in cl:
        if dd < dm - 1e-12 or dd > dp + 1e-12:
            continue
        dd = min(max(dd, dm), dp)
        if forb and dd > r1 and dd < r2:
            continue
        v = cs + A * dd
        v[dg] -= dd * dd
        val = max(U, float(np.max(v)))
        if val < best_v:
            best_v = val
            best_d = dd
    if abs(best_d) > 1e-14 and best_v < f0 - 1e-8:
        c[st:en] += A * best_d
        c[st + dg] -= best_d * best_d
        h[p] = h0 + best_d
        return best_d
    return 0.0

def cd_polish(h, n, orient, deadline, step=0.25, drift_lim=2.0, rng=None):
    h = project_feasible(h.copy(), n)
    c = sync_c(h, n, orient)
    best_h = h.copy()
    best_c = internal_c5(c, n)
    step = max(1e-4, float(step))
    cnt = 0
    while time.time() < deadline:
        if rng is not None and cnt % 3 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        drift = 0.0
        for p in order:
            if time.time() > deadline:
                break
            d = coord_update(h, c, p, n, step, 35)
            if d:
                drift += d
                if abs(drift) > drift_lim:
                    h = project_feasible(h, n)
                    c = sync_c(h, n, orient)
                    drift = 0.0
                    cv = internal_c5(c, n)
                    if cv < best_c:
                        best_c = cv
                        best_h = h.copy()
        if cnt % 3 == 0:
            c = sync_c(h, n, orient)
        if abs(h.sum() - 0.5 * float(n)) > 1e-7:
            h = project_feasible(h, n)
            c = sync_c(h, n, orient)
        cv = internal_c5(c, n)
        if cv < best_c - 1e-12:
            best_c = cv
            best_h = h.copy()
        elif cv > best_c + 1e-8:
            step *= 0.7
            if step < 1e-3:
                break
            h = best_h.copy()
            c = sync_c(h, n, orient)
        cnt += 1
    return best_h, best_c

def gradient_weights(h, idx, w, n):
    g = np.zeros(n)
    for mm, ww in zip(idx, w):
        if ww < 1e-9:
            continue
        k = int(mm) - (n - 1)
        if k >= 0:
            L = n - k
            if L <= 0:
                continue
            g[k:] += ww * (1.0 - h[:L])
            g[:L] -= ww * h[k:]
        else:
            q = -k
            L = n - q
            if L <= 0:
                continue
            g[q:] += ww * (1.0 - h[:L])
            g[:L] -= ww * h[q:]
    return g

def active_polish(h, n, orient, deadline, rng):
    h = project_feasible(h.copy(), n)
    c = sync_c(h, n, orient)
    best_h = h.copy()
    best_c = internal_c5(c, n)
    Ks = (16, 48, 96, 192)
    Ts = (0.3, 1.0, 3.0, 8.0)
    etas = (0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625)
    it = 0
    while time.time() < deadline:
        for K in Ks:
            if time.time() > deadline:
                break
            c = sync_c(h, n, orient)
            idx = np.argsort(c)[-K:]
            for temp in Ts:
                if time.time() > deadline:
                    break
                vals = c[idx]
                maxv = float(vals.max())
                w = np.exp((vals - maxv) / temp)
                s = float(w.sum())
                if s <= 0:
                    continue
                w = w / s
                g = gradient_weights(h, idx, w, n)
                gn = math.sqrt(float(np.dot(g, g)))
                if gn < 1e-12:
                    continue
                g = g / gn
                acc = False
                for eta in etas:
                    h1 = project_feasible(h - eta * g, n)
                    c1 = direct_c5(h1)
                    if c1 < best_c - 1e-12:
                        h = h1
                        c = sync_c(h, n, orient)
                        best_h = h.copy()
                        best_c = c1
                        acc = True
                        break
                if acc:
                    break
        it += 1
        if it % 8 == 0 and rng is not None and time.time() < deadline:
            h = project_feasible(h + rng.normal(0.0, 0.02, n), n)
            c = sync_c(h, n, orient)
    return best_h, best_c

def interp_h(h, n_old, n_new):
    if n_new == len(h):
        return project_feasible(h.copy(), n_new)
    x0 = np.linspace(0.0, 2.0, len(h), endpoint=False)
    x1 = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return project_feasible(np.interp(x1, x0, h), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(1.0, float(budget_s))
    deadline = t0 + max(0.2, total - 1.0)
    rng = np.random.RandomState(seed)

    try:
        orient = get_orientation()
    except Exception:
        orient = 1

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
        if n < 300:
            h = interp_h(h, n, 700)
            n = 700
        elif n > 900:
            h = interp_h(h, n, 700)
            n = 700
    except Exception:
        n = 700
        h = np.full(n, 0.5)

    try:
        h = project_feasible(h, n)
        best_h = h.copy()
        best_c = direct_c5(h)
        best_n = n

        dl = min(deadline, t0 + 0.55 * total)
        if dl > t0 + 5:
            hh, cc = cd_polish(best_h, best_n, orient, dl, 0.35, 2.0, rng)
            if cc < best_c:
                best_h, best_c = hh, cc

        dl = min(deadline, t0 + 0.75 * total)
        if dl > time.time() + 5:
            hh, cc = active_polish(best_h, best_n, orient, dl, rng)
            if cc < best_c:
                best_h, best_c = hh, cc

        dl = min(deadline, t0 + 0.90 * total)
        if dl > time.time() + 5:
            hh, cc = cd_polish(best_h, best_n, orient, dl, 0.06, 1.0, rng)
            if cc < best_c:
                best_h, best_c = hh, cc

        rem = deadline - time.time()
        alts = []
        if best_n > 400:
            alts.append(max(350, best_n // 2))
        if best_n < 900:
            alts.append(min(1000, best_n + 150))

        for ntr in alts:
            if rem < 90:
                break
            local_dl = min(deadline, time.time() + min(rem * 0.45, 220))
            if local_dl <= time.time() + 5:
                continue
            ht = interp_h(best_h, best_n, ntr)
            hh, cc = cd_polish(ht, ntr, orient, local_dl, 0.20, 1.5, rng)
            if time.time() < local_dl - 20:
                hh2, cc2 = active_polish(hh, ntr, orient, local_dl, rng)
                if cc2 < cc:
                    hh, cc = hh2, cc2
            if cc < best_c:
                best_h, best_c, best_n = hh, cc, ntr
            rem = deadline - time.time()

        cnt = 0
        while time.time() < deadline - 40 and cnt < 30:
            scale = 0.025 * (1.0 - cnt / 30.0) + 0.004
            hp = project_feasible(best_h + rng.normal(0.0, scale, best_n), best_n)
            local_dl = min(deadline, time.time() + 35)
            if local_dl <= time.time() + 5:
                break
            hh, cc = cd_polish(hp, best_n, orient, local_dl, 0.08, 1.0, rng)
            if cc < best_c:
                best_h, best_c = hh, cc
            cnt += 1

        if time.time() < deadline - 5:
            hh, cc = cd_polish(best_h, best_n, orient, deadline, 0.02, 0.8, rng)
            if cc < best_c:
                best_h, best_c = hh, cc
    except Exception:
        pass

    best_h = project_feasible(best_h, best_n)
    bc = direct_c5(best_h)
    return best_h, bc, best_n
