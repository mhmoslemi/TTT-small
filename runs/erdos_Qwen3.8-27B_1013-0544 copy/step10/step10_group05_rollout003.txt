def obj_soft(x,n,alpha,pen):
    x=np.asarray(x,float)
    c=np.correlate(x,1.0-x,'full')
    m=float(c.max())
    dx=2.0/n

       e=np.exp(z); s=e.sum()
       val=dx*(m + float(np.log(s))/alpha)

def obj_soft(x,n,alpha,pen):
    c=np.correlate(x,1.0-x,'full')
    m=float(c.max())
    dx=2.0/n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def enforce_sum(h, n):
    target = 0.5 * n
    for _ in range(30):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        idx = np.where((h > 1e-13) & (h < 1.0 - 1e-13))[0]
        if idx.size:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
        else:
            h = np.clip(h + d / float(h.size), 0.0, 1.0)
    return h

def project_box_sum(x, n):
    x = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(x)):
        return np.full(int(n), 0.5)
    target = 0.5 * n
    if x.size and x.min() >= -1e-12 and x.max() <= 1.0 + 1e-12 and abs(x.sum() - target) < 1e-12:
        return enforce_sum(np.clip(x, 0.0, 1.0), n)
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return enforce_sum(np.clip(x - 0.5 * (lo + hi), 0.0, 1.0), n)

def obj_soft(x, n, alpha, pen):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    dx = 2.0 / n
    if alpha > 1e-9:
        z = alpha * (c - m)
        z = np.minimum(z, 0.0)
        e = np.exp(z)
        s = float(e.sum())
        val = dx * (m + float(np.log(s)) / alpha)
    else:
        val = dx * m
    if pen:
        d = float(x.sum() - 0.5 * n)
        val += float(pen) * d * d
    return float(val)

def grad_soft(x, n, alpha, pen):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    dx = 2.0 / n
    if alpha > 1e-9:
        z = alpha * (c - m)
        z = np.minimum(z, 0.0)
        e = np.exp(z)
        s = float(e.sum())
        w = e / s
    else:
        w = np.zeros_like(c)
        mask = (c == m)
        w[mask] = 1.0 / float(max(1, int(mask.sum())))
    first = np.convolve(1.0 - x, w[::-1], mode='full')[n - 1:2 * n - 1]
    second = np.convolve(x, w, mode='full')[n - 1:2 * n - 1]
    g = dx * (first - second)
    if pen:
        g += float(2.0 * pen) * float(x.sum() - 0.5 * n)
    return g

def cons_eq(x):
    return float(x.sum() - 0.5 * len(x))

def cons_eq_jac(x):
    return np.ones(len(x))

def lbfgs_polish(h0, n, alphas, pens, maxiter, deadline):
    h = project_box_sum(np.asarray(h0, dtype=float), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if n <= 0:
        return best_h, best_c
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        for pen in pens:
            if time.time() > deadline:
                break
            try:
                res = minimize(obj_soft, h, jac=grad_soft, args=(n, alpha, pen),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': maxiter, 'ftol': 1e-14,
                                        'gtol': 1e-8, 'maxfun': maxiter * 5})
                if not np.all(np.isfinite(res.x)):
                    continue
                nh = project_box_sum(res.x, n)
                nc = compute_c5(nh, n)
                if nc < best_c - 1e-15:
                    best_h, best_c, h = nh, nc, nh
            except Exception:
                pass
        if time.time() > deadline:
            break
    return best_h, best_c

def slsqp_polish(h0, n, alphas, maxiter, deadline):
    h = project_box_sum(np.asarray(h0, dtype=float), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if n <= 0:
        return best_h, best_c
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(obj_soft, h, jac=grad_soft, args=(n, alpha, 0.0),
                           method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            if not np.all(np.isfinite(res.x)):
                continue
            nh = project_box_sum(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c - 1e-15:
                best_h, best_c, h = nh, nc, nh
        except Exception:
            pass
    return best_h, best_c

def coord_update(c, h, p, n):
    h0 = float(h[p])
    if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
        return False
    m0 = n - 1
    amin = max(0, m0 - p)
    amax = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(amin, amax + 1)
    off = ms - m0
    jf = p + off
    jb = p - off
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms), dtype=float)
    if np.any(vf):
        a[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    cb = c[ms]
    unaff = 0.0
    if amin > 0:
        unaff = max(unaff, float(np.max(c[:amin])))
    if amax < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[amax + 1:])))
    cur = float(np.max(c))
    dlo, dhi = -h0, 1.0 - h0
    pos = (a > 1e-14)
    neg = (a < -1e-14)
    zero = ~(pos | neg)
    zmax = float(np.max(cb[zero])) if np.any(zero) else -1.0
    pa, cp = a[pos], cb[pos]
    na, cn = a[neg], cb[neg]
    hasd = (amin <= m0 <= amax)
    di = np.where(ms == m0)[0] if hasd else np.array([], dtype=int)
    Llo = max(unaff, zmax)
    Lhi = cur
    for _ in range(32):
        if Lhi - Llo < 1e-13:
            break
        L = 0.5 * (Llo + Lhi)
        ok = (L >= unaff and zmax <= L)
        Ld, Hd = dlo, dhi
        if ok and pa.size:
            Hd = min(Hd, float(np.min((L - cp) / pa)))
        if ok and Ld <= Hd and na.size:
            Ld = max(Ld, float(np.max((L - cn) / na)))
        if ok and Ld > Hd + 1e-13:
            ok = False
        if ok and hasd:
            ad, cd = a[di[0]], cb[di[0]]
            disc = ad * ad - 4.0 * (L - cd)
            if disc >= 0.0:
                sq = float(np.sqrt(disc))
                r1, r2 = 0.5 * (ad - sq), 0.5 * (ad + sq)
                if Ld > r1 + 1e-14 and Hd < r2 - 1e-14:
                    ok = False
        if ok:
            Lhi = L
        else:
            Llo = L
    L = Lhi
    Ld, Hd = dlo, dhi
    if pa.size:
        Hd = min(Hd, float(np.min((L - cp) / pa)))
    if Ld <= Hd and na.size:
        Ld = max(Ld, float(np.max((L - cn) / na)))
    Ld, Hd = max(Ld, dlo), min(Hd, dhi)
    if Ld > Hd + 1e-13:
        return False
    cand = [Ld, Hd, 0.5 * (Ld + Hd), 0.0]
    if hasd:
        ad, cd = a[di[0]], cb[di[0]]
        disc = ad * ad - 4.0 * (L - cd)
        if disc >= 0.0:
            sq = float(np.sqrt(disc))
            r1, r2 = 0.5 * (ad - sq), 0.5 * (ad + sq)
            for rr in (r1, r2):
                if Ld - 1e-12 <= rr <= Hd + 1e-12:
                    cand.append(float(rr))
    best_x, best_val = h0, cur
    for gd in cand:
        x = h0 + gd
        if x < -1e-13 or x > 1.0 + 1e-13:
            continue
        x = float(np.clip(x, 0.0, 1.0))
        delta = x - h0
        if abs(delta) < 1e-14:
            continue
        nv = cb + delta * a
        if hasd:
            nv[di[0]] -= delta * delta
        val = float(np.max(nv))
        if unaff > val:
            val = unaff
        if val < best_val - 1e-14:
            best_x, best_val = x, val
    if best_x == h0:
        return False
    delta = best_x - h0
    c[ms] = c[ms] + delta * a
    if hasd:
        c[ms[di[0]]] -= delta * delta
    h[p] = best_x
    return True

def cd_exact(h0, n, max_passes, deadline, random=False, rng=None):
    h = project_box_sum(np.asarray(h0, dtype=float), n)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_h = h.copy()
    best_c = float(c.max() * dx)
    if random and rng is None:
        rng = np.random.RandomState(1)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        order = rng.permutation(n) if random else np.arange(n)
        ch = 0
        for idx, p in enumerate(order):
            if (idx & 63) == 0 and time.time() > deadline:
                break
            if coord_update(c, h, int(p), n):
                ch += 1
        if ch == 0:
            break
        nh = project_box_sum(h, n)
        if np.max(np.abs(nh - h)) > 1e-15:
            h = nh
            c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max() * dx)
        if cur < best_c - 1e-15:
            best_c, best_h = cur, h.copy()
    best_h = project_box_sum(best_h, n)
    return best_h, compute_c5(best_h, n)

def fourier_init(n, K, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        amp = 0.45 / (0.5 + k)
        h += amp * np.cos(2.0 * np.pi * k * x / 2.0 + rng.uniform(0.0, 2.0 * np.pi))
    return project_box_sum(h, n)

def block_init(n, rng):
    h = np.empty(n, dtype=float)
    pos = 0
    while pos < n:
        L = max(1, int(rng.exponential(max(3.0, float(n) / 25.0))))
        v = 0.0 if rng.random() < 0.5 else 1.0
        h[pos:pos + L] = v
        pos += L
    return project_box_sum(h, n)

def resize_h(h, n, m):
    n, m = int(n), int(m)
    if m <= 0:
        return np.array([], dtype=float)
    if m == n:
        return project_box_sum(h, n)
    if n == 1:
        return project_box_sum(np.full(m, float(h[0])), m)
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    xt = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_box_sum(np.interp(xt, xs, h), m)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 6.0
    rng = np.random.RandomState(seed)
    try:
        try:
            h = np.asarray(initial_h_values, dtype=float).copy()
            n = int(len(h))
        except Exception:
            n = 700
            h = fourier_init(n, 12, rng)
        if n < 10:
            n = 700
            h = fourier_init(n, 12, rng)
        h = project_box_sum(h, n)
        best_h, best_c, best_n = h, compute_c5(h, n), n
        orig_n = best_n

        if time.time() < deadline - 500:
            hh, cc = cd_exact(best_h, best_n, 6, deadline - 450, False)
            if cc < best_c:
                best_h, best_c = hh, cc

        if time.time() < deadline - 450:
            hh, cc = lbfgs_polish(best_h, best_n, [10, 100, 1000, 10000, 100000],
                                  [0.1, 1.0], 55, deadline - 380)
            if cc < best_c:
                best_h, best_c = hh, cc
            hh, cc = cd_exact(best_h, best_n, 8, deadline - 300, True, rng)
            if cc < best_c:
                best_h, best_c = hh, cc

        if time.time() < deadline - 280:
            hh, cc = slsqp_polish(best_h, best_n, [20000, 200000], 70, deadline - 220)
            if cc < best_c:
                best_h, best_c = hh, cc
            hh, cc = cd_exact(best_h, best_n, 8, deadline - 180, False)
            if cc < best_c:
                best_h, best_c = hh, cc

        if time.time() < deadline - 180:
            hh, cc = cd_exact(best_h, best_n, 16, deadline - 120, True, rng)
            if cc < best_c:
                best_h, best_c = hh, cc

        tried = set()
        for m0 in (orig_n // 2, orig_n // 3, 300, 200, 150):
            m0 = 2 * int(m0 // 2)
            if m0 < 50 or m0 in tried:
                continue
            tried.add(m0)
            if time.time() > deadline - 180:
                break
            hm = resize_h(best_h, best_n, m0)
            hm, cm = lbfgs_polish(hm, m0, [50, 500, 5000, 50000], [1.0], 45, deadline - 110)
            hm, cm = cd_exact(hm, m0, 8, deadline - 70, True, rng)
            if cm < best_c:
                best_h, best_c, best_n = hm, cm, int(m0)
            if time.time() < deadline - 50 and best_n != m0:
                hup = resize_h(hm, m0, best_n)
                hup, cmp2 = cd_exact(hup, best_n, 5, deadline - 35, False)
                if cmp2 < best_c:
                    best_h, best_c, best_n = hup, cmp2, len(hup)

        if best_n < orig_n and time.time() < deadline - 160:
            hup = resize_h(best_h, best_n, orig_n)
            hup, cmp2 = lbfgs_polish(hup, orig_n, [100, 1000, 10000], [1.0], 45, deadline - 90)
            hup, cmp2 = cd_exact(hup, orig_n, 10, deadline - 50, False)
            if cmp2 < best_c:
                best_h, best_c, best_n = hup, cmp2, orig_n

        for r in range(7):
            if time.time() > deadline - 160:
                break
            if r % 2 == 0:
                init = fourier_init(best_n, 8 + r, rng)
            else:
                init = block_init(best_n, rng)
            init, ci = lbfgs_polish(init, best_n, [200, 2000, 20000], [0.5], 35, deadline - 90)
            init, ci = cd_exact(init, best_n, 6, deadline - 50, True, rng)
            if ci < best_c:
                best_h, best_c = init, ci

        if time.time() < deadline - 30:
            hh, cc = cd_exact(best_h, best_n, 30, deadline - 20, False)
            if cc < best_c:
                best_h, best_c = hh, cc
        if time.time() < deadline - 10:
            hh, cc = cd_exact(best_h, best_n, 20, deadline - 8, True, rng)
            if cc < best_c:
                best_h, best_c = hh, cc

        best_h = project_box_sum(best_h, best_n)
        best_c = compute_c5(best_h, best_n)
        return best_h, float(best_c), int(best_n)
    except Exception:
        try:
            h = np.asarray(initial_h_values, dtype=float).copy()
            n = int(len(h))
            h = project_box_sum(h, n)
            return h, compute_c5(h, n), n
        except Exception:
            h = np.full(700, 0.5)
            return h, compute_c5(h, 700), 700
