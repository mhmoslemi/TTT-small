import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / max(tot, 1e-15) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / max(tot, 1e-15) * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n, nbisect=28):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo_a = max(0, n - 1 - p)
    hi_a = min(2 * n - 2, n - 1 + p)
    lo_b = max(0, p)
    hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(lo_a, lo_b)
    m_hi = max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    non_diag = np.ones(len(ms), dtype=bool)
    if has_diag:
        non_diag[diag_idx] = False
    a_nd = a[non_diag]
    c_nd = c_base[non_diag]
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(c[m_hi + 1:])))
    cur_max = float(np.max(c))
    L_lo = unaff_max
    L_hi = cur_max
    for _ in range(nbisect):
        if L_hi - L_lo < 1e-12:
            break
        T = (L_lo + L_hi) / 2.0
        feasible = True
        if unaff_max > T + 1e-14:
            feasible = False
        elif feasible and len(a_nd) > 0:
            x_lo_c = 0.0
            x_hi_c = 1.0
            neg = a_nd < -1e-13
            pos = a_nd > 1e-13
            if np.any(neg):
                x_lo_c = max(x_lo_c, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                x_hi_c = min(x_hi_c, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            if x_lo_c > x_hi_c + 1e-13:
                feasible = False
        if feasible and has_diag:
            Cq = T - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if len(a_nd) > 0:
                    neg = a_nd < -1e-13
                    pos = a_nd > 1e-13
                    xl = 0.0
                    xh = 1.0
                    if np.any(neg):
                        xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                    xl = max(xl, 0.0)
                    xh = min(xh, 1.0)
                    if xl > xh + 1e-13:
                        feasible = False
                    elif xl >= r1 and xh <= r2:
                        feasible = False
                else:
                    if (0.0 >= r1 and 1.0 <= r2) or (0.0 < r1 and 1.0 > r2):
                        feasible = False
        if feasible:
            L_hi = T
        else:
            L_lo = T
    T = (L_lo + L_hi) / 2.0
    x_lo = 0.0
    x_hi = 1.0
    if len(a_nd) > 0:
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo < r1:
                x_hi = min(x_hi, r1)
            elif x_hi > r2:
                x_lo = max(x_lo, r2)
            else:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1 + 1e-8)
                else:
                    x_lo = max(x_lo, r2 - 1e-8)
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng=None, nbisect=28, order='seq'):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if order == 'random' and rng is not None:
            ord_arr = rng.permutation(n)
        elif order == 'rev':
            ord_arr = np.arange(n - 1, -1, -1)
        else:
            ord_arr = np.arange(n)
        for p in ord_arr:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n, nbisect)
            if abs(x_new - h0) > 1e-12:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_phase(h, n, deadline, alphas):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline - 30:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 120, 'ftol': 1e-13})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def symmetrize(h, n):
    x = np.linspace(0, 2, n)
    h_rev = np.interp(x, x[::-1], h)
    h_sym = 0.5 * (h + 1.0 - h_rev)
    return project_feasible(h_sym, n)

    best_c = compute_c5(best_h, n)

    # Phase 1: Sequential CD (main workhorse)
    h1, c1 = coord_descent(best_h, n, 60, deadline, rng, 28, 'seq')
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Reverse order CD
    h2, c2 = coord_descent(best_h, n, 40, deadline, rng, 28, 'rev')
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: SLSQP refinement
    h3, c3 = slsqp_phase(best_h, n, deadline, [5000, 15000, 40000])
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: More CD
    h4, c4 = coord_descent(best_h, n, 80, deadline, rng, 30, 'seq')
    if c4 < best_c:
        best_h, best_c = h4, c4

    # Phase 5: Symmetrize and optimize
    h_sym = symmetrize(best_h, n)
    c_sym = compute_c5(h_sym, n)
    if c_sym < best_c:
        best_h, best_c = h_sym, c_sym
        h_sym2, c_sym2 = coord_descent(best_h, n, 40, deadline, rng, 28, 'seq')
        if c_sym2 < best_c:
            best_h, best_c = h_sym2, c_sym2

    # Phase 6: Random order CD
    h6, c6 = coord_descent(best_h, n, 50, deadline, rng, 28, 'random')
    if c6 < best_c:
        best_h, best_c = h6, c6

    # Phase 7: SLSQP again
    h7, c7 = slsqp_phase(best_h, n, deadline, [15000, 50000])
    if c7 < best_c:
        best_h, best_c = h7, c7

    # Phase 8: Perturbation diversification
    if time.time() < deadline - 120:
        for trial in range(10):
            if time.time() > deadline - 60:
                break
            scale = 0.02 + 0.015 * trial
            noise = rng.randn(n) * scale
            h_p = project_feasible(best_h + noise, n)
            h_p, c_p = coord_descent(h_p, n, 20, deadline, rng, 25, 'random')
            if c_p < best_c:
                best_h, best_c = h_p, c_p

    # Phase 9: Try different n
    if time.time() < deadline - 150:
        x_orig = np.linspace(0, 2, n)
        for n_try in [600, 800]:
            if time.time() > deadline - 80:
                break
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
            h_opt, c_opt = coord_descent(h_int, n_try, 30, deadline, rng, 25, 'seq')
            h_opt, c_opt = slsqp_phase(h_opt, n_try, deadline, [10000])
            h_opt, c_opt = coord_descent(h_opt, n_try, 30, deadline, rng, 28, 'seq')
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
                n = n_try
                x_orig = x_new

    # Phase 10: Final intensive CD polish
    remaining = deadline - time.time()
    if remaining > 15:
        n_cur = len(best_h)
        max_p = min(300, int(remaining / 0.25))
        h_f, c_f = coord_descent(best_h, n_cur, max_p, deadline, None, 35, 'seq')
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f2, c_f2 = coord_descent(best_h, len(best_h), 30, deadline, rng, 35, 'random')
        if c_f2 < best_c:
            best_h, best_c = h_f2, c_f2

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
