from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def compute_c5_corr(h, n):

    dx = 2.0 / n

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            if res.x is not None:

                new_c = compute_c5(new_h, n)

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def water_filling(h, n, deadline, max_iters=100):
    """Try to equalize the top correlation values."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = corr.max()
        # Find the top lags
        top_thresh = m * 0.999
        top_lags = np.where(corr >= top_thresh)[0]
        
        if len(top_lags) <= 1:
            break
        
        # For each top lag, the corresponding coordinate should be decreased
        # This is a heuristic - try to reduce values at positions that contribute
        # most to the top correlations
        reduced = False
        for lag in top_lags:
            if time.time() > deadline:
                break
            # lag corresponds to offset k = lag - (n-1)
            k = lag - (n - 1)
            # c(k) = sum_i h(i) * (1 - h(i+k))
            # To reduce c(k), we want to increase h where (1-h(i+k)) is large
            # or decrease h where (1-h(i+k)) is large
            if k >= 0:
                idx = np.arange(0, n - k)
                partner = idx + k
            else:
                idx = np.arange(-k, n)
                partner = idx + k
            
            # Gradient of c(k) w.r.t. h(idx) is (1-h(partner))
            # Gradient of c(k) w.r.t. h(partner) is -h(idx)
            # To minimize max, decrease h(idx) where (1-h(partner)) * h(idx) is largest
            contrib = h[idx] * (1.0 - h[partner])
            # Find the coordinate with highest contribution
            if len(idx) > 0 and contrib.sum() > 0:
                p = idx[np.argmax(contrib)]
                # Try small decrease
                step = 0.01
                h_new = h.copy()
                h_new[p] -= step
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-10:
                    h = h_new
                    best_c = c_new
                    reduced = True
        if not reduced:
            break
    return h, best_c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    return x.sum() - n / 2.0

    return np.ones(n)

            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',

            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-15:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h[h < th] = 0.0
            h[h > 1.0 - th] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    final_c = compute_c5(h, n)
    return h, min(best_c, final_c)

def binary_refine(h, n, deadline, max_rounds=30):
    """After CD, try snapping near-binary coords and re-optimizing free ones."""
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        # Identify near-binary coordinates
        snap0 = h < 0.02
        snap1 = h > 0.98
        n_free = (~snap0) & (~snap1)
        if n_free.sum() == 0:
            break
        h[snap0] = 0.0
        h[snap1] = 1.0
        h = project_feasible(h, n)
        # CD only on free coordinates (approximation: full CD but binary ones won't move much)
        h2, c2 = coord_descent(h, n, 30, deadline, gs_iters=80, threshold=1e-8)
        if c2 < best_c:
            best_c = c2
            h = h2
        else:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(350, 0.5)
        n = 350

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP warm start
    if time.time() < deadline - 500:
        h1, c1 = slsqp_optimize(best_h, n, [500, 2000, 8000], 250)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Aggressive CD
    if time.time() < deadline - 400:
        h2, c2 = coord_descent(best_h, n, 150, deadline, gs_iters=95, threshold=4e-6)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Binary refinement
    if time.time() < deadline - 350:
        h3, c3 = binary_refine(best_h, n, deadline, max_rounds=20)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Multi-resolution search
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 500, 600, 350]:
        if time.time() > deadline - 250:
            break
        x_s = np.linspace(0, 2, n_try)
        h_s = project_feasible(np.interp(x_s, x_orig, best_h), n_try)
        if time.time() < deadline - 300:
            h_s, c_s = slsqp_optimize(h_s, n_try, [3000, 15000], 200)
        if time.time() < deadline - 250:
            h_s, c_s = coord_descent(h_s, n_try, 100, deadline, gs_iters=85)
        if c_s < best_c:
            best_h, best_c = h_s, c3 if False else c_s
            x_orig = np.linspace(0, 2, n_try)
        # Try interpolating back to original n
        if time.time() < deadline - 200:
            n_orig = len(x_orig)
            h_back = project_feasible(np.interp(x_orig, np.linspace(0,2,n_try), h_s), n_orig)
            h_back, c_back = coord_descent(h_back, n_orig, 50, deadline, gs_iters=85)
            if c_back < best_c:
                best_h, best_c = h_back, c_back

    # Phase 5: Multi-restart with diverse perturbations
    if time.time() < deadline - 100:
        n_cur = len(best_h)
        x = np.linspace(0, 2, n_cur)
        remaining = deadline - time.time() - 80
        n_restarts = max(4, int(remaining / 70))
        for r in range(n_restarts):
            if time.time() > deadline - 60:
                break
            if r % 6 == 0:
                size = 0.02 + 0.02 * (r % 3)
                hp = project_feasible(best_h + rng.normal(0, size, n_cur), n_cur)
            elif r % 6 == 1:
                freq = 0.3 + 0.7 * rng.rand()
                hp = project_feasible(best_h + 0.04 * np.sin(2*np.pi*freq*x/2.0 + rng.rand()*2*np.pi), n_cur)
            elif r % 6 == 2:
                # Flip a random subset
                n_flip = max(3, n_cur // 30)
                idx = rng.choice(n_cur, size=n_flip, replace=False)
                hp = best_h.copy()
                hp[idx] = 1.0 - hp[idx]
                hp = project_feasible(hp, n_cur)
            elif r % 6 == 3:
                # Perturb mid-range values more
                mask = (best_h > 0.15) & (best_h < 0.85)
                hp = best_h.copy()
                hp[mask] += rng.normal(0, 0.06, mask.sum())
                hp = project_feasible(hp, n_cur)
            elif r % 6 == 4:
                # Block swap
                bs = max(8, n_cur // 15)
                s1, s2 = sorted(rng.choice(n_cur - bs, size=2, replace=True))
                hp = best_h.copy()
                hp[s1:s1+bs], hp[s2:s2+bs] = hp[s2:s2+bs].copy(), hp[s1:s1+bs].copy()
                hp = project_feasible(hp, n_cur)
            else:
                # Sparse large perturbation
                n_sp = max(5, n_cur // 20)
                idx = rng.choice(n_cur, size=n_sp, replace=False)
                hp = best_h.copy()
                hp[idx] = rng.rand(n_sp)
                hp = project_feasible(hp, n_cur)
            h_o, c_o = coord_descent(hp, n_cur, 70, deadline, gs_iters=80, threshold=4e-6)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 6: Final CD polish
    if time.time() < deadline - 20:
        n_cur = len(best_h)
        h_f, c_f = coord_descent(best_h, n_cur, 100, deadline, gs_iters=100, threshold=1e-8)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Phase 7: Final SLSQP
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        h_f, c_f = slsqp_optimize(best_h, n_cur, [100000, 500000], 400)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        best_h = project_feasible(best_h, n_cur)
        best_c = compute_c5(best_h, n_cur)

    return best_h, best_c, len(best_h)
