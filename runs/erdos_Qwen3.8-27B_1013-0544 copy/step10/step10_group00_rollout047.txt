import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def cd_eval(c_aff, aff, has_diag, diag_idx, unaff_max, d):
    """Evaluate the objective for a change of d at one position."""
    vals = c_aff + d * aff
    if has_diag:
        vals[diag_idx] -= d * d
    m = float(np.max(vals)) if len(vals) > 0 else 0.0
    if unaff_max > m:
        m = unaff_max
    return m

def fast_cd(h, n, max_passes, deadline, gs_iters=40):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2*n-2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n-1)
            jb = p - ms + (n-1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            aff = np.zeros(len(ms))
            if np.any(vf):
                aff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                aff[vb] -= h[jb[vb]]
            has_diag = (m_lo <= n-1 <= m_hi)
            diag_idx = (n-1) - m_lo if has_diag else -1
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff = max(unaff, float(c[m_hi+1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr*(d_hi - d_lo)
            d2 = d_lo + gr*(d_hi - d_lo)
            c_aff = c[ms]
            f1 = cd_eval(c_aff, aff, has_diag, diag_idx, unaff, d1)
            f2 = cd_eval(c_aff, aff, has_diag, diag_idx, unaff, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-14:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr*(d_hi - d_lo)
                    f1 = cd_eval(c_aff, aff, has_diag, diag_idx, unaff, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr*(d_hi - d_lo)
                    f2 = cd_eval(c_aff, aff, has_diag, diag_idx, unaff, d2)
            d_opt = (d_lo + d_hi) / 2.0
            if cd_eval(c_aff, aff, has_diag, diag_idx, unaff, d_opt) < cd_eval(c_aff, aff, has_diag, diag_idx, unaff, 0.0) - 1e-13:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
    return h, compute_c5(h, n)

def fourier_init(n, m1=3, m2=7, m3=11, a1=0.45, a2=0.15, a3=0.08, ph1=0.0, ph2=1.2, ph3=2.5):
    """Generate a Fourier-based initial candidate."""
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + a1*np.sin(np.pi*m1*x + ph1) + a2*np.sin(np.pi*m2*x + ph2) + a3*np.sin(np.pi*m3*x + ph3)
    return project_feasible(h, n)

def block_cd(h, n, block_size, max_passes, deadline, gs_iters=40):
    """Coordinate descent where we optimize blocks of positions simultaneously."""
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for start in range(0, n, block_size):
            if time.time() > deadline:
                break
            end = min(start + block_size, n)
            block = h[start:end]
            if np.all(block < 1e-10) or np.all(block > 1.0 - 1e-10):
                continue
            # For simplicity, just do individual CD within the block
            for p in range(start, end):
                h0 = h[p]
                if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                    continue
                m_lo = max(0, n - 1 - p)
                m_hi = min(2*n-2, n - 1 + p)
                ms = np.arange(m_lo, m_hi + 1)
                jf = p + ms - (n-1)
                jb = p - ms + (n-1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                aff = np.zeros(len(ms))
                if np.any(vf):
                    aff[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    aff[vb] -= h[jb[vb]]
                has_diag = (m_lo <= n-1 <= m_hi)
                diag_idx = (n-1) - m_lo if has_diag else -1
                unaff = 0.0
                if m_lo > 0:
                    unaff = max(unaff, float(c[:m_lo].max()))
                if m_hi < 2*n-2:
                    unaff = max(unaff, float(c[m_hi+1:].max()))
                d_lo, d_hi = -h0, 1.0 - h0
                d1 = d_hi - gr*(d_hi - d_lo)
                d2 = d_lo + gr*(d_hi - d_lo)
                c_aff = c[ms]
                f1 = cd_eval(c_aff, aff, has_diag, diag_idx, unaff, d1)
                f2 = cd_eval(c_aff, aff, has_diag, diag_idx, unaff, d2)
                for _ in range(gs_iters):
                    if d_hi - d_lo < 1e-14:
                        break
                    if f1 < f2:
                        d_hi, d2, f2 = d2, d1, f1
                        d1 = d_hi - gr*(d_hi - d_lo)
                        f1 = cd_eval(c_aff, aff, has_diag, diag_idx, unaff, d1)
                    else:
                        d_lo, d1, f1 = d1, d2, f2
                        d2 = d_lo + gr*(d_hi - d_lo)
                        f2 = cd_eval(c_aff, aff, has_diag, diag_idx, unaff, d2)
                d_opt = (d_lo + d_hi) / 2.0
                if cd_eval(c_aff, aff, has_diag, diag_idx, unaff, d_opt) < cd_eval(c_aff, aff, has_diag, diag_idx, unaff, 0.0) - 1e-13:
                    h[p] = h0 + d_opt
                    improved = True
        if not improved:
            break
    return h, compute_c5(h, n)
