```python
import numpy as np
import time


def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def full_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_feasible(x, n):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    s = float(x.sum())
    if abs(s - target) < 1e-13:
        return x
    if s > target:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0
    for _ in range(48):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        ys = float(y.sum())
        if ys > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(x - tau, 0.0, 1.0)
    diff = target - float(y.sum())
    for _ in range(60):
        if abs(diff) < 1e-14:
            break
        mask = (y > 1e-13) & (y < 1.0 - 1e-13)
        if not np.any(mask):
            break
        if diff > 0:
            room = float((1.0 - y[mask]).sum())
            if room < 1e-15:
                break
            y[mask] = np.clip(y[mask] + diff * (1.0 - y[mask]) / room, 0.0, 1.0)
        else:
            tot = float(y[mask].sum())
            if tot < 1e-15:
                break
            y[mask] = y[mask] + diff * y[mask] / tot
        diff = target - float(y.sum())
    return y


def interp_h(h_old, n_old, n_new):
    if n_old == n_new:
        return project_feasible(h_old.copy(), n_new)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h = np.interp(x_new, x_old, h_old)
    return project_feasible(h, n_new)


def active_gradient(h, n, c, alpha):
    diag = n - 1
    maxv = float(c.max())
    w = np.exp(alpha * (c - maxv))
    idx = np.where(w > 1e-6)[0]
    if idx.size == 0:
        return np.zeros(n), 0
    if idx.size > 48:
        ord = np.argsort(w[idx])[-48:]
        idx = idx[ord]
    wsum = float(w[idx].sum())
    if wsum <= 0.0:
        return np.zeros(n), 0
    g = np.zeros(n)
    for m in idx:
        wg = float(w[m]) / wsum
        d = int(m - diag)
        if d >= 0:
            g[:n - d] += wg * (1.0 - h[d:])
            g[d:] -= wg * h[:n - d]
        else:
            e = -d
            g[e:] += wg * (1.0 - h[:n - e])
            g[:n - e] -= wg * h[e:]
    return g, int(idx.size)


def value_pairs(h, n, k, rng):
    k = int(min(k, max(0, n // 2)))
    if k <= 0:
        return []
    hi = np.argsort(h)[-k:]
    lo = np.argsort(h)[:k]
    m = min(int(hi.size), int(lo.size))
    if m <= 0:
        return []
    perm = rng.permutation(m)
    pairs = []
    for i in perm:
        p = int(hi[i])
        q = int(lo[i])
        if p != q:
            pairs.append((p, q))
    return pairs


def pair_step(h, c, n, p, q, samples):
    if p == q:
        return 0.0
    hp = float(h[p])
    hq = float(h[q])
    tmax = min(hp, 1.0 - hq)
    if tmax <= 1e-12:
        return 0.0
    M = 2 * n - 1
    diag = n - 1
    a = np.zeros(M)
    omh = 1.0 - h
    rev = h[::-1]
    a[n - 1 - p : 2 * n - 1 - p] -= omh
    a[p : p + n] += rev
    a[n - 1 - q : 2 * n - 1 - q] += omh
    a[q : q + n] -= rev
    left = float(c.max())
    samples = max(16, int(samples))
    T = np.linspace(0.0, tmax, samples)
    TT = T * T
    vals = c[None, :] + a[None, :] * T[:, None]
    cr1 = diag + (p - q)
    cr2 = diag + (q - p)
    vals[:, diag] -= 2.0 * TT
    vals[:, cr1] += TT
    vals[:, cr2] += TT
    f = vals.max(axis=1)
    bi = int(np.argmin(f))
    if bi == 0 or f[bi] >= left - 1e-11:
        return 0.0
    if bi == len(T) - 1:
        lo, hi_t = float(T[-2]), tmax
    elif bi == 1:
        lo, hi_t = 0.0, float(T[2])
    else:
        lo, hi_t = float(T[bi - 1]), float(T[bi + 1])
    T2 = np.linspace(lo, hi_t, 32)
    TT2 = T2 * T2
    vals2 = c[None, :] + a[None, :] * T2[:, None]
    vals2[:, diag] -= 2.0 * TT2
    vals2[:, cr1] += TT2
    vals2[:, cr2] += TT2
    f2 = vals2.max(axis=1)
    bj = int(np.argmin(f2))
    if f2[bj] >= left - 1e-11:
        return 0.0
    t = float(T2[bj])
    if t <= 1e-12:
        return 0.0
    h[p] = hp - t
    h[q] = hq + t
    c += a * t
    c[diag] -= 2.0 * t * t
    c[cr1] += t * t
    c[cr2] += t * t
    return t


def pair_polish(h, n, deadline, rng, samples=64, k=8):
    h = project_feasible(h.copy(), n)
    c = full_corr(h)
    best_h = h.copy()
    best_raw = float(c.max())
    no = 0
    it = 0
    while time.time() < deadline - 1.0 and no < 60 and it < 60000:
        if no < 5:
            alpha = 10.0
        elif no < 15:
            alpha = 50.0
        elif no < 30:
            alpha = 200.0
        else:
            alpha = 800.0
        g, cnt = active_gradient(h, n, c, alpha)
        pairs = []
        if cnt > 0:
            gm = g - g.mean()
            scale = float(np.max(np.abs(gm)))
            if scale > 1e-14:
                pos = np.where(gm > scale * 1e-3)[0]
                neg = np.where(gm < -scale * 1e-3)[0]
                if pos.size and neg.size:
                    kp = min(k, int(pos.size))
                    kn = min(k, int(neg.size))
                    pos_sel = pos[np.argsort(gm[pos])[-kp:]]
                    neg_sel = neg[np.argsort(gm[neg])[:kn]]
                    m = min(int(pos_sel.size), int(neg_sel.size))
                    if m > 0:
                        perm = rng.permutation(m)
                        for ii in perm:
                            pairs.append((int(pos_sel[ii]), int(neg_sel[ii])))
        if it % 3 == 0:
            vp = value_pairs(h, n, k // 2 + 1, rng)
            pairs.extend(vp[:4])
        updated = False
        for p, q in pairs:
            if time.time() > deadline - 2.0:
                break
            t = pair_step(h, c, n, p, q, samples)
            if t > 1e-12:
                updated = True
                if it % 20 == 19:
                    c = full_corr(h)
                    cur = float(c.max())
                    if cur < best_raw - 1e-13:
                        best_raw = cur
                        best_h = h.copy()
                    elif cur > best_raw + 1e-8:
                        h = best_h.copy()
                        c = full_corr(h)
        if not updated:
            for _ in range(min(5, k)):
                if time.time() > deadline - 2.0:
                    break
                p = int(rng.randint(n))
                q = int(rng.randint(n))
                if p == q:
                    continue
                if h[p] >= h[q]:
                    src, dst = p, q
                else:
                    src, dst = q, p
                if pair_step(h, c, n, src, dst, samples) > 1e-12:
                    updated = True
        if updated:
            no = 0
        else:
            no += 1
        if it % 8 == 7:
            c = full_corr(h)
            cur = float(c.max())
            if cur < best_raw - 1e-13:
                best_raw = cur
                best_h = h.copy()
            elif cur > best_raw + 1e-8:
                h = best_h.copy()
                c = full_corr(h)
        it += 1
    c = full_corr(h)
    cur = float(c.max())
    if cur < best_raw - 1e-13:
        best_h = h.copy()
        best_raw = cur
    return best_h, float(best_raw) * (2.0 / n)


def insert_candidate(tops, c, n, h):
    c = float(c)
    n = int(n)
    if len(tops) < 3 or c < tops[-1][0]:
        item = (c, n, h)
        if len(tops) < 3:
            tops.append(item)
            i = len(tops) - 1
        else:
            tops[-1] = item
            i = len(tops) - 1
        while i > 0 and tops[i][0] < tops[i - 1][0]:
            tops[i], tops[i - 1] = tops[i - 1], tops[i]
            i -= 1


def scan_candidates(base_h, base_n, best_c, deadline, rng):
    tops = []
    start_n = max(300, base_n - 200)
    end_n = min(1000, base_n + 300)
    seen = set()
    ns = []
    for nn in range(start_n, end_n + 1, 4):
        if nn % 2 == 0 and nn not in seen:
            seen.add(nn)
            ns.append(nn)
    for nn in (base_n, 500, 600, 800, 900):
        if nn % 2 == 0 and start_n <= nn <= end_n and nn not in seen:
            seen.add(nn)
            ns.append(nn)
    tested = 0
    for nn in ns:
        if time.time() > deadline - 35.0:
            break
        if tested >= 130:
            break
        tested += 1
        h = interp_h(base_h, base_n, nn)
        c = compute_c5(h, nn)
        insert_candidate(tops, c, nn, h)
    return tops


def alternative_starts(h_best, n, best_c, rng):
    out = []
    x = (np.arange(n) + 0.5) / n
    candidates = []
    candidates.append(1.0 - x)
    candidates.append(np.clip(1.0 - x + 0.18 * np.cos(6.0 * np.pi * x), 0.0, 1.0))
    candidates.append(np.clip(0.5 + 0.45 * (1.0 - x) + 0.12 * np.sin(7.0 * np.pi * x), 0.0, 1.0))
    trig = 0.5 + 0.40 * np.cos(2.0 * np.pi * (x + 0.17)) + rng.normal(0.0, 0.03, n)
    candidates.append(np.clip(trig, 0.0, 1.0))
    noisy = h_best + rng.normal(0.0, 0.04, n)
    candidates.append(np.clip(noisy, 0.0, 1.0))
    for hh in candidates:
        h = project_feasible(hh, n)
        c = compute_c5(h, n)
        if c < best_c + 0.02:
            out.append(h)
    return out


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 4.0
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 700
        x = (np.arange(n) + 0.5) / n
        h = 1.0 - x

    if n < 2:
        n = 700
        x = (np.arange(n) + 0.5) / x_max if False else (np.arange(n) + 0.5) / n
        h = 1.0 - x

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    best_n = n

    try:
        left = deadline - time.time()
        if left > 90.0:
            sub = min(deadline - max(120.0, 0.2 * left), time.time() + max(90.0, 0.5 * left))
            h1, c1 = pair_polish(best_h, best_n, sub, rng, samples=72, k=8)
            if c1 < best_c - 1e-12:
                best_h, best_c, best_n = h1.copy(), float(c1), int(best_n if False else len(h1))

        left = deadline - time.time()
        if left > 150.0:
            sub = min(deadline - 100.0, time.time() + min(220.0, 0.35 * left))
            h1, c1 = pair_polish(best_h, best_n, sub, rng, samples=64, k=8)
            if c1 < best_c - 1e-12:
                best_h, best_c, best_n = h1.copy(), float(c1), int(len(h1))

        if time.time() < deadline - 160.0:
            starts = alternative_starts(best_h, best_n, best_c, rng)
            for hs in starts:
                if time.time() > deadline - 80.0:
                    break
                alloc = min(120.0, max(30.0, (deadline - time.time() - 50.0) * 0.5))
                sub = min(deadline - 50.0, time.time() + alloc)
                hr, cr = pair_polish(hs, len(hs), sub, rng, samples=56, k=6)
                if cr < best_c - 1e-12:
                    best_h, best_c, best_n = hr.copy(), float(cr), int(len(hr))

        if time.time() < deadline - 160.0:
            tops = scan_candidates(best_h, best_n, best_c, deadline, rng)
            for idx, item in enumerate(tops):
                if time.time() > deadline - 40.0:
                    break
                c0, nn, hc = item
                rem = max(1, len(tops) - idx)
                alloc = min(120.0, max(40.0, (deadline - time.time() - 50.0) / float(rem)))
                sub = min(deadline - 50.0, time.time() + alloc)
                hr, cr = pair_polish(hc, nn, sub, rng, samples=56, k=6)
                if cr < best_c - 1e-12:
                    best_h, best_c, best_n = hr.copy(), float(cr), int(nn)

        if time.time() < deadline - 50.0:
            scales = (0.018, 0.006, 0.002)
            for r, sc in enumerate(scales):
                if time.time() > deadline - 15.0:
                    break
                hp = project_feasible(best_h + rng.normal(0.0, sc, best_n), best_n)
                sub = min(deadline - 10.0, time.time() + max(30.0, 70.0 - 10.0 * r))
                hr, cr = pair_polish(hp, best_n, sub, rng, samples=48, k=5)
                if cr < best_c - 1e-12:
                    best_h, best_c, best_n = hr.copy(), float(cr), int(len(hr))
    except Exception:
        pass

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
```
