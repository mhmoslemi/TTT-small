import numpy as np
import time

def max_c_unscaled(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def compute_c5(h, n):
    return max_c_unscaled(h) * (2.0 / float(n))

def project_box_sum(z, target):
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    if abs(z.sum() - target) < 1e-13:
        return z
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        v = np.clip(z - mid, 0.0, 1.0)
        if v.sum() > target:
            lo = mid
        else:
            hi = mid
    v = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        if diff > 0.0:
            room = 1.0 - v
            tot = room.sum()
            if tot > 1e-15:
                v = np.clip(v + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = v.sum()
            if tot > 1e-15:
                v = np.clip(v + (diff / tot) * v, 0.0, 1.0)
        diff = target - v.sum()
        if abs(diff) > 1e-13:
            idx = int(np.argmax(1.0 - v)) if diff > 0.0 else int(np.argmax(v))
            v[idx] = float(np.clip(v[idx] + diff, 0.0, 1.0))
    return v

def project_feasible(h, n):
    return project_box_sum(h, 0.5 * float(n))

def interpolate_h(h_src, n_src, n_tgt):
    if n_src == n_tgt:
        return project_feasible(h_src, n_tgt)
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_tgt, endpoint=False)
    return project_feasible(np.interp(xt, xs, h_src), n_tgt)

def fourier_init(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n, dtype=float)
    for k in range(1, 11):
        amp = rng.normal(0.0, 1.0 / np.sqrt(float(k)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(np.pi * k * x + ph)
    h = 0.5 + 0.35 * h
    return project_feasible(h, n)

def build_coord_info(n):
    infos = []
    for p in range(n):
        lo1, hi1 = p, p + n - 1
        lo2, hi2 = n - 1 - p, 2 * n - 2 - p
        ms = np.arange(min(lo1, lo2), max(hi1, hi2) + 1, dtype=np.int64)
        r1 = p - (ms - (n - 1))
        r2 = p + (ms - (n - 1))
        v1 = (r1 >= 0) & (r1 < n)
        v2 = (r2 >= 0) & (r2 < n)
        idx1 = np.nonzero(v1)[0]
        idx2 = np.nonzero(v2)[0]
        r1v = r1[v1]
        r2v = r2[v2]
        dp = int(np.searchsorted(ms, n - 1))
        msk = np.ones(len(ms), dtype=bool)
        msk[dp] = False
        infos.append((ms, idx1, r1v, idx2, r2v, dp, msk))
    return infos

def coord_intervals(L, U, D, bp, dp, bn, dn, dz, eps=1e-10):
    if U > L + eps or dz > L + eps:
        return []
    lo, hi = 0.0, 1.0
    if bp.size:
        hi = min(hi, float(np.min((L - dp) / bp)))
    if bn.size:
        lo = max(lo, float(np.max((L - dn) / bn)))
    lo = max(lo, 0.0)
    hi = min(hi, 1.0)
    if lo > hi + eps:
        return []
    T = L - D
    if T < -eps:
        return []
    if T < 0.0:
        T = 0.0
    out = []
    if T < 0.25 - eps:
        disc = max(0.0, 1.0 - 4.0 * T)
        sq = np.sqrt(disc)
        r1 = 0.5 * (1.0 - sq)
        r2 = 0.5 * (1.0 + sq)
        l1, h1 = max(lo, 0.0), min(hi, r1)
        l2, h2 = max(lo, r2), min(hi, 1.0)
        if l1 <= h1 + eps:
            out.append((l1, h1))
        if l2 <= h2 + eps:
            out.append((l2, h2))
    else:
        out.append((lo, hi))
    res = []
    for a, b in out:
        a = float(np.clip(a, 0.0, 1.0))
        b = float(np.clip(b, 0.0, 1.0))
        if a <= b + eps:
            res.append((a, b))
    return res

def choose_x(intervals, x_ref):
    x_ref = float(np.clip(x_ref, 0.0, 1.0))
    best = x_ref
    bestd = 1e100
    for lo, hi in intervals:
        lo = float(np.clip(lo, 0.0, 1.0))
        hi = float(np.clip(hi, 0.0, 1.0))
        if hi < lo:
            continue
        cands = [lo, hi, 0.5 * (lo + hi)]
        if lo <= x_ref <= hi:
            cands.append(x_ref)
        for cand in cands:
            cand = float(np.clip(cand, 0.0, 1.0))
            dd = (cand - x_ref) * (cand - x_ref)
            if dd < bestd:
                bestd = dd
                best = cand
    return best

def cd_pass(h, n, c, order, deadline, infos):
    h = np.array(h, dtype=float, copy=True)
    c = np.array(c, dtype=float, copy=True)
    target = 0.5 * float(n)
    sum_h = float(h.sum())
    current_max = float(np.max(c))
    eps = 1e-10
    for step, p in enumerate(order):
        if time.time() > deadline:
            break
        ms, idx1, r1v, idx2, r2v, dp, msk = infos[int(p)]
        x0 = float(h[int(p)])
        b = np.zeros(len(ms), dtype=float)
        if idx1.size:
            b[idx1] += 1.0 - h[r1v]
        if idx2.size:
            b[idx2] -= h[r2v]
        c_ms = c[ms]
        start = int(ms[0])
        end = int(ms[-1])
        U = 0.0
        if start > 0:
            U = max(U, float(np.max(c[:start])))
        if end < len(c) - 1:
            U = max(U, float(np.max(c[end + 1:])))
        D = c[n - 1] - x0 + x0 * x0
        b_non = b[msk]
        d_non = (c_ms - b * x0)[msk]
        pos = b_non > 1e-12
        neg = b_non < -1e-12
        zero = ~(pos | neg)
        dz = float(np.max(d_non[zero])) if np.any(zero) else -1.0
        bp = b_non[pos]
        dparr = d_non[pos]
        bn = b_non[neg]
        dn = d_non[neg]
        L_lo = U
        if bp.size:
            L_lo = max(L_lo, float(np.min(np.minimum(dparr, dparr + bp))))
        if bn.size:
            L_lo = max(L_lo, float(np.min(np.minimum(dn, dn + bn))))
        L_lo = max(L_lo, dz, D)
        L_hi = max(current_max, L_lo)
        for _ in range(35):

            if coord_intervals(mid, U, D, bp, dparr, bn, dn, dz, eps):
                L_hi = mid
            else:
                L_lo = mid
            if L_hi - L_lo < 1e-9:
                break
        L = L_hi
        intervals = coord_intervals(L, U, D, bp, dparr, bn, dn, dz, 1e-9)
        if not intervals:
            intervals = coord_intervals(current_max, U, D, bp, dparr, bn, dn, dz, 1e-8)
        if not intervals:
            x = x0
        else:
            x_ref = x0
            dev = target - sum_h
            if abs(dev) > 1e-4:

            delta = x - x0
            c[ms] += delta * b
            c[n - 1] -= delta * delta
            h[int(p)] = x
            sum_h += delta
            current_max = max(U, float(np.max(c[ms])))
    h = project_box_sum(h, target)
    c = np.correlate(h, 1.0 - h, mode='full')
    return h, c, compute_c5(h, n)

def cd_sequence(h, n, max_passes, deadline, rng=None, infos=None):

def feasible_alpha(h, d):
    lo, hi = -1e18, 1e18
    eps = 1e-14
    pos = d > eps
    neg = d < -eps
    if np.any(pos):
        lo = max(lo, float(np.max(-h[pos] / d[pos])))
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max((1.0 - h[neg]) / d[neg])))
        hi = min(hi, float(np.min(-h[neg] / d[neg])))
    if hi > 1.0:
        hi = 1.0
    if lo < -1.0:
        lo = -1.0
    if not (lo < hi):
        return 0.0, 0.0
    return lo, hi

def line_search_1d(h, d, a_lo, a_hi, target, n, best_m, grid=25, refine=15):
    span = a_hi - a_lo
    if span <= 1e-10:
        return 0.0, best_m, h
    a0 = a_lo + 1e-7 * span
    a1 = a_hi - 1e-7 * span
    if a1 <= a0:
        return 0.0, best_m, h
    best_a = 0.0
    best_val = best_m
    alphas = np.linspace(a0, a1, int(grid))
    for aa in alphas:
        val = max_c_unscaled(h + float(aa) * d)
        if val < best_val - 1e-12:
            best_val = val
            best_a = float(aa)
    if best_a != 0.0:
        idx = int(np.argmin(np.abs(alphas - best_a)))
        if idx == 0:
            lo, hi = a0, (alphas[1] if alphas.size > 1 else a1)
        elif idx == alphas.size - 1:
            lo, hi = alphas[-2], a1
        else:
            lo, hi = alphas[idx - 1], alphas[idx + 1]
        for aa in np.linspace(lo, hi, int(refine)):
            val = max_c_unscaled(h + float(aa) * d)
            if val < best_val - 1e-12:
                best_val = val
                best_a = float(aa)
        ht = np.clip(h + best_a * d, 0.0, 1.0)
        ht = project_box_sum(ht, target)
        val = max_c_unscaled(ht)
        if val < best_m - 1e-11:
            return best_a, val, ht
    return 0.0, best_m, h

def active_gradient(c, h):
    n = len(h)
    M = float(np.max(c))
    tol = 1e-8 * max(1.0, M)
    act = c >= M - tol
    cnt = max(1, int(np.count_nonzero(act)))
    w = np.zeros(len(c), dtype=float)
    w[act] = 1.0 / cnt
    q = 1.0 - h
    A = np.convolve(w, q, mode='full')[n - 1:n - 1 + n]
    B = np.convolve(w[::-1], h, mode='full')[n - 1:n - 1 + n]
    return A - B

def random_direction(n, rng):
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    d = np.zeros(n, dtype=float)
    for k in range(1, 11):
        amp = rng.normal(0.0, 1.0 / np.sqrt(float(k)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        d += amp * np.cos(2.0 * np.pi * k * x + ph)
    d -= d.mean()
    norm = float(np.linalg.norm(d))
    if norm < 1e-12:
        d = rng.normal(0.0, 1.0, n)
        d -= d.mean()
        norm = float(np.linalg.norm(d))
    if norm < 1e-12:
        return np.zeros(n, dtype=float)
    return d / norm

def symmetry_direction(h):
    d = (1.0 - h[::-1]) - h
    d -= d.mean()
    return d

def gradient_line_search(h, n, best_c_scaled, deadline, rng, max_iters=70):
    target = 0.5 * float(n)
    best_m = best_c_scaled * (0.5 * float(n))
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(int(max_iters)):
        if time.time() > deadline:
            break
        if c is None:
            c = np.correlate(h, 1.0 - h, mode='full')
        M = float(np.max(c))
        if M < best_m:
            best_m = M
        if it == 0:
            d = symmetry_direction(h)
        elif it % 7 == 6:
            d = random_direction(n, rng)
        else:
            g = active_gradient(c, h)
            d = -g
            d -= d.mean()
        norm = float(np.linalg.norm(d))
        if norm < 1e-12:
            d = random_direction(n, rng)
            norm = float(np.linalg.norm(d))
        if norm < 1e-12:
            continue
        d = d / norm
        a_lo, a_hi = feasible_alpha(h, d)
        if a_hi - a_lo <= 1e-7:
            continue
        _, val, ht = line_search_1d(h, d, a_lo, a_hi, target, n, best_m, grid=25, refine=15)
        if val < best_m - 1e-10:
            h = ht
            best_m = val
            c = None

def targeted_pair_search(h, n, best_c_scaled, deadline, rng, max_iters=70):

            hr, cr = gradient_line_search(hr, best_n, cr, cand_dead2, rng, 30)
        if cr < best_c:
            best_h, best_c = hr, cr
            best_n = len(best_h)
            infos = build_coord_info(best_n)
        else:
            break

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, float(best_c), int(len(best_h))
