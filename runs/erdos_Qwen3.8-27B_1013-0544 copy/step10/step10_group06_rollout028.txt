def point_eta_precomp(T, unaff, d_zero, d_pos, b_pos, d_neg, b_neg, Aq, Bq, Dq, An, Bn, Dn, tol=1e-9):
    if unaff > T + tol:
        return None
    L=0.0; U=1.0
    if d_zero.size:
        if float(np.max(d_zero)) > T + tol:
            return None
    if b_pos.size:
        vals = (T - d_pos) / b_pos
        up = float(np.min(vals))
        if up < L - tol:
            return None
        U = up if up < U else U
    if b_neg.size:
        vals = (T - d_neg) / b_neg
        low = float(np.max(vals))
        if low > U + tol:
            return None
        L = low if low > L else L
    if Aq.size:
        disc = Bq*Bq - 4.0*Aq*(Dq - T)
        if float(np.min(disc)) < -tol:
            return None
        disc = np.maximum(disc, 0.0)
        sqrtD = np.sqrt(disc)
        r1 = (-Bq - sqrtD)/(2.0*Aq)
        r2 = (-Bq + sqrtD)/(2.0*Aq)
        low = float(np.max(r1)); high = float(np.min(r2))
        if low > U + tol: return None
        if high < L - tol: return None
        L = low if low > L else L
        U = high if high < U else U
    if L < 0.0: L=0.0
    if U > 1.0: U=1.0
    if L > U + tol: return None
    if An.size:
        disc = Bn*Bn - 4.0*An*(Dn - T)
        good = disc >= -tol
        if np.any(good):
            dg = np.maximum(disc[good], 0.0)
            sqrtD = np.sqrt(dg)
            ag = An[good]; bg = Bn[good]
            r1 = (-bg - sqrtD)/(2.0*ag)
            r2 = (-bg + sqrtD)/(2.0*ag)
            lo = np.minimum(r1, r2)
            hi = np.maximum(r1, r2)
            lo_c = np.maximum(lo, L); hi_c = np.minimum(hi, U)
            keep = hi_c > lo_c + tol
            if np.any(keep):
                lo_k = lo_c[keep]; hi_k = hi_c[keep]
                order = np.argsort(lo_k)
                lo_s = lo_k[order]; hi_s = hi_k[order]
                cur = L
                for i in range(len(lo_s)):
                    if hi_s[i] <= cur + tol:
                        continue
                    if lo_s[i] > cur + tol:
                        # gap
                        if lo_s[i] - cur > tol:

                        else:

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, target=None):
    n = len(h)
    h = np.asarray(h, dtype=float).copy()
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = 0.5 * (lo + hi)
        s = np.clip(h - lam, 0.0, 1.0).sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(h - lo, 0.0, 1.0)
    d = target - x.sum()
    if abs(d) > 1e-12:
        idx = (x > 1e-9) & (x < 1.0 - 1e-9)
        if idx.sum() > 0:
            x[idx] += d / float(idx.sum())
        else:
            x = np.clip(h, 0.0, 1.0)
    return np.clip(x, 0.0, 1.0)

def force_antisym(h, n=None):
    h = np.asarray(h, dtype=float).copy()
    if n is None:
        n = len(h)
    elif len(h) != n:
        h = np.interp(np.linspace(0.0, 2.0, n), np.linspace(0.0, 2.0, len(h)), h)
    m = n // 2
    for i in range(m):
        v = 0.5 * (h[i] + 1.0 - h[n - 1 - i])
        if v < 0.0:
            v = 0.0
        elif v > 1.0:
            v = 1.0
        h[i] = v
        h[n - 1 - i] = 1.0 - v
    if n % 2:
        h[n // 2] = 0.5
    return h

def eval_F(base, A, B, C, unaff, vs):
    vs = np.asarray(vs, dtype=float).ravel()
    v2 = vs * vs
    vals = base[None, :] + C[None, :] + B[None, :] * vs[:, None] + A[None, :] * v2[:, None]
    return np.maximum(vals.max(axis=1), unaff)

def point_eta_precomp(T, unaff, d_zero, d_pos, b_pos, d_neg, b_neg,
                      Aq, Bq, Dq, An, Bn, Dn, tol=1e-9):
    if unaff > T + tol:
        return None
    L = 0.0
    U = 1.0
    if d_zero.size:
        if float(np.max(d_zero)) > T + tol:
            return None
    if b_pos.size:
        up = float(np.min((T - d_pos) / b_pos))
        U = min(U, up)
    if b_neg.size:
        low = float(np.max((T - d_neg) / b_neg))
        L = max(L, low)
    if Aq.size:
        disc = Bq * Bq - 4.0 * Aq * (Dq - T)
        if float(np.min(disc)) < -tol:
            return None
        disc = np.maximum(disc, 0.0)
        s = np.sqrt(disc)
        r1 = (-Bq - s) / (2.0 * Aq)
        r2 = (-Bq + s) / (2.0 * Aq)
        L = max(L, float(np.max(r1)))
        U = min(U, float(np.min(r2)))
    if L < 0.0:
        L = 0.0
    if U > 1.0:
        U = 1.0
    if L > U + tol:
        return None
    if An.size:
        disc = Bn * Bn - 4.0 * An * (Dn - T)
        good = disc >= -tol
        if np.any(good):
            dg = np.maximum(disc[good], 0.0)
            s = np.sqrt(dg)
            ag = An[good]
            bg = Bn[good]
            r1 = (-bg - s) / (2.0 * ag)
            r2 = (-bg + s) / (2.0 * ag)
            lo = np.minimum(r1, r2)
            hi = np.maximum(r1, r2)
            lo = np.maximum(lo, L)
            hi = np.minimum(hi, U)
            keep = hi > lo + tol
            if np.any(keep):
                lok = lo[keep]
                hik = hi[keep]
                ord_ = np.argsort(lok)
                lok = lok[ord_]
                hik = hik[ord_]
                cur = L
                for i in range(len(lok)):
                    if hik[i] <= cur + tol:
                        continue
                    if lok[i] > cur + tol:
                        return float(0.5 * (cur + lok[i]) if lok[i] - cur > tol else cur)
                    if hik[i] >= U - tol:
                        return None
                    cur = hik[i]
                if cur >= U - tol:
                    return None
                return float(0.5 * (cur + U))
    return float(0.5 * (L + U))

def update_antisym_coord(h, c, p, n, all_idx, bisect=30, atol=1e-8, tol=1e-9):
    mid = n - 1
    q = n - 1 - p
    v0 = float(h[p])
    Lc = 2 * n - 1
    Aarr = np.zeros(Lc)
    Barr = np.zeros(Lc)
    Carr = np.zeros(Lc)
    old = np.zeros(Lc)
    mask = np.zeros(Lc, dtype=bool)

    b1 = all_idx
    cb = np.where(b1 == p, 1.0, np.where(b1 == q, -1.0, 0.0))
    db = np.where(b1 == p, 0.0, np.where(b1 == q, 1.0, h[b1]))

    m1 = mid + p - b1
    A1 = -cb
    B1 = 1.0 - db
    old1 = v0 * (1.0 - (cb * v0 + db))
    Aarr[m1] += A1
    Barr[m1] += B1
    old[m1] += old1
    mask[m1] = True

    m2 = mid + q - b1
    A2 = cb
    B2 = db - 1.0 - cb
    C2 = 1.0 - db
    old2 = (1.0 - v0) * (1.0 - (cb * v0 + db))
    Aarr[m2] += A2
    Barr[m2] += B2
    Carr[m2] += C2
    old[m2] += old2
    mask[m2] = True

    excl = (b1 != p) & (b1 != q)
    a3 = b1[excl]
    if a3.size:
        ha3 = h[a3]

        m3 = mid + a3 - p
        B3 = -ha3
        C3 = ha3
        old3 = ha3 * (1.0 - v0)
        Barr[m3] += B3
        Carr[m3] += C3
        old[m3] += old3
        mask[m3] = True

        m4 = mid + a3 - q
        B4 = ha3
        old4 = ha3 * v0
        Barr[m4] += B4
        old[m4] += old4
        mask[m4] = True

    aff = np.flatnonzero(mask)
    if aff.size == 0:
        return False, 0.0
    if bool(mask.all()):
        unaff = -1.0
    else:
        unaff = float(np.max(c[~mask]))
    base = c[aff] - old[aff]
    Aa = Aarr[aff]
    Ba = Barr[aff]
    Ca = Carr[aff]
    cur_max = float(np.max(c))

    tolA = 1e-8
    neg_q = Aa < -tolA
    pos_q = Aa > tolA
    lin = ~(neg_q | pos_q)

    d_zero = np.empty(0)
    d_pos = np.empty(0)
    b_pos = np.empty(0)
    d_neg = np.empty(0)
    b_neg = np.empty(0)
    if bool(lin.any()):
        D = base[lin] + Ca[lin]
        B = Ba[lin]
        zero = np.abs(B) <= 1e-14
        if bool(zero.any()):
            d_zero = D[zero]
        bp = B > 1e-14
        if bool(bp.any()):
            d_pos = D[bp]
            b_pos = B[bp]
        bn = B < -1e-14
        if bool(bn.any()):
            d_neg = D[bn]
            b_neg = B[bn]

    if bool(pos_q.any()):
        Aq = Aa[pos_q]
        Bq = Ba[pos_q]
        Dq = base[pos_q] + Ca[pos_q]
    else:
        Aq = Bq = Dq = np.empty(0)
    if bool(neg_q.any()):
        An = Aa[neg_q]
        Bn = Ba[neg_q]
        Dn = base[neg_q] + Ca[neg_q]
    else:
        An = Bn = Dn = np.empty(0)

    L0 = max(0.0, unaff)
    H = cur_max + 1e-8
    if point_eta_precomp(H, unaff, d_zero, d_pos, b_pos, d_neg, b_neg,
                         Aq, Bq, Dq, An, Bn, Dn, tol) is None:
        fv0 = float(np.max(base + Aa * v0 * v0 + Ba * v0 + Ca))
        if fv0 > cur_max:
            cur_max = fv0
        H = max(H, fv0) + 1e-7
        if point_eta_precomp(H, unaff, d_zero, d_pos, b_pos, d_neg, b_neg,
                             Aq, Bq, Dq, An, Bn, Dn, tol) is None:
            return False, cur_max
    if L0 > H:
        L0 = H

    for _ in range(bisect):
        T = 0.5 * (L0 + H)
        if point_eta_precomp(T, unaff, d_zero, d_pos, b_pos, d_neg, b_neg,
                             Aq, Bq, Dq, An, Bn, Dn, tol) is not None:
            H = T
        else:
            L0 = T

    vp = point_eta_precomp(H + 1e-7, unaff, d_zero, d_pos, b_pos, d_neg, b_neg,
                           Aq, Bq, Dq, An, Bn, Dn, tol)
    if vp is not None and np.isfinite(vp):
        cand = np.array([0.0, 1.0, v0, 0.5, float(vp)])
    else:
        cand = np.linspace(0.0, 1.0, 121)
        vp = None
    cand = np.clip(cand, 0.0, 1.0)
    vals = eval_F(base, Aa, Ba, Ca, unaff, cand)
    bi = int(np.argmin(vals))
    best_v = float(cand[bi])
    best_val = float(vals[bi])

    if vp is not None and best_val > cur_max - 1e-5:
        full = np.linspace(0.0, 1.0, 121)
        vf = eval_F(base, Aa, Ba, Ca, unaff, full)
        fi = int(np.argmin(vf))
        if vf[fi] < best_val:
            best_v = float(full[fi])
            best_val = float(vf[fi])

    lo = max(0.0, best_v - 0.03)
    hi = min(1.0, best_v + 0.03)
    if hi > lo:
        loc = np.linspace(lo, hi, 41)
        lv = eval_F(base, Aa, Ba, Ca, unaff, loc)
        li = int(np.argmin(lv))
        if lv[li] < best_val:
            best_v = float(loc[li])
            best_val = float(lv[li])

    if best_val < cur_max - atol:
        h[p] = best_v
        h[q] = 1.0 - best_v
        c[aff] = base + Aa * best_v * best_v + Ba * best_v + Ca
        return True, float(best_val)
    return False, cur_max

def antisym_cd(h, n, deadline, rng, bisect=30, atol=1e-8):
    h = force_antisym(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = float(np.max(c)) * dx
    all_idx = np.arange(n)
    m = n // 2
    prev_c5 = best_c5

    while time.time() < deadline and m > 0:
        improved = False
        order = rng.permutation(m)
        for p in order:
            if time.time() > deadline:
                break
            ok, _ = update_antisym_coord(h, c, p, n, all_idx, bisect, atol, 1e-9)
            if ok:
                improved = True

        c = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(c)) * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()

        if not improved or c5 > prev_c5 - 1e-9:
            break
        if c5 > best_c5 + 1e-6:
            h = best_h.copy()
            c = np.correlate(h, 1.0 - h, mode='full')
            break
        prev_c5 = c5

    return best_h, best_c5

def random_antisym_start(n, rng, base=None, sigma=0.1):
    x = np.linspace(0.0, 2.0, n)
    if base is not None:
        hb = np.asarray(base, dtype=float)
        hb = np.interp(x, np.linspace(0.0, 2.0, len(hb)), hb)
    else:
        hb = np.full(n, 0.5)
    for _ in range(4):
        f = rng.uniform(1.0, 6.0)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        am = rng.uniform(0.05, 0.2)
        hb += am * np.sin(2.0 * np.pi * f * x + ph)
    hb += sigma * rng.normal(0.0, 1.0, n)
    return force_antisym(np.clip(hb, 0.0, 1.0), n)
