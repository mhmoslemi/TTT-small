    """Smooth approximation of max_k c_k with emphasis on top values.
    alpha: temperature for log-sum-exp
    beta: weight on the raw max (for stability)
    """

    lse = mx + np.log(np.sum(np.exp(shifted - mx)))

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals = vals.copy()
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def quick_cd(h, n, passes, deadline, gs_iters=30):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            dp = np.where(ms == (n - 1))[0]
            has_d = len(dp) > 0
            di = int(dp[0]) if has_d else -1

                unaff = max(unaff, float(c[m_hi + 1:].max()))

            d1 = dhi - gr * (dhi - dlo)
            d2 = dlo + gr * (dhi - dlo)
            f1 = _cd_fval(c_aff, a, di, has_d, unaff, h0, d1)
            f2 = _cd_fval(c_aff, a, di, has_d, unaff, h0, d2)
            for _ in range(gs_iters):
                if dhi - dlo < 1e-14:
                    break
                if f1 < f2:
                    dhi, d2, f2 = d2, d1, f1
                    d1 = dhi - gr * (dhi - dlo)
                    f1 = _cd_fval(c_aff, a, di, has_d, unaff, h0, d1)
                else:
                    dlo, d1, f1 = d1, d2, f2
                    d2 = dlo + gr * (dhi - dlo)
                    f2 = _cd_fval(c_aff, a, di, has_d, unaff, h0, d2)

                h[p] = h0 + dopt
                improved = True

        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def mutate_individual(h, n, rng, stage):
    """Mutate a solution. stage in [0,1] controls mutation strength."""
    h_m = h.copy()
    block = 100
    u = rng.rand()
    if u < 0.25:

        sigma = 0.05 * (1.0 - 0.5 * stage)
        h_m += rng.normal(0, sigma, n)

        start = rng.randint(0, n - block)
        h_m[start:start + block] = rng.rand(block)
    elif u < 0.60:

        x = np.linspace(0, 2, n)

        start = rng.randint(0, n - block)

        for _ in range(3):

            h_m[start:start + 50] += rng.normal(0, 0.05, 50)

    """Uniform crossover."""
    mask = rng.rand(n) < 0.5
    child = np.where(mask, h1, h2)
    return project_feasible(child, n)

    """Evolutionary strategy."""
    pop_size = 30
    # Initialize population
    pop = [h_init.copy()]
    fitness = [c_init]
    for i in range(1, pop_size):
        if rng.rand() < 0.5:
            h = mutate_individual(h_init, n, rng, 0.0)
        else:
            h = project_feasible(h_init + rng.normal(0, 0.05, n), n)
        c = compute_c5(h, n)
        pop.append(h)
        fitness.append(c)
    
    best_h = h_init.copy()
    best_c = c_init
    gen = 0
    max_gen = 200
    
    while time.time() < deadline and gen < max_gen:
        gen += 1
        # Sort by fitness
        idx = np.argsort(fitness)
        pop = [pop[i] for i in idx]
        fitness = [fitness[i] for i in idx]
        
        if fitness[0] < best_c:
            best_c = fitness[0]
            best_h = pop[0].copy()
        
        # Elitism: keep top 3
        new_pop = [pop[0].copy(), pop[1].copy(), pop[2].copy()]
        new_fit = [fitness[0], fitness[1], fitness[2]]
        
        # Generate offspring
        stage = gen / max_gen
        while len(new_pop) < pop_size:
            if time.time() > deadline:
                break
            r = rng.rand()
            # Select parents (tournament of 3)
            t1, t2, t3 = rng.randint(0, min(10, len(pop)), 3)
            p1_idx = min(t1, t2, t3)
            t1, t2, t3 = rng.randint(0, min(10, len(pop)), 3)
            p2_idx = min(t1, t2, t3)
            
            if r < 0.6:
                # Mutation
                parent = pop[p1_idx]
                child = mutate_individual(parent, n, rng, stage)
            else:
                # Crossover + mutation
                child = crossover(pop[p1_idx], pop[p2_idx], n, rng)
                if rng.rand() < 0.7:
                    child = mutate_individual(child, n, rng, stage)
            
            c = compute_c5(child, n)
            new_pop.append(child)
            new_fit.append(c)
        
        pop = new_pop
        fitness = new_fit
    
    return best_h, best_c

```python
def directed_mutate(h, n, rng, strength=0.1):
    """Mutate h in a direction that reduces the worst correlation."""
    c = np.correlate(h, 1.0 - h, mode='full')
    s_star = int(np.argmax(c)) - (n - 1)
    h_m = h.copy()
    # For the worst shift s*, we want to reduce c(s*)
    # c(s*) = sum_i h[i]*(1-h[i+s*])
    # Gradient w.r.t h[i]: (1-h[i+s*]) - h[i-s*] (when valid)
    # To REDUCE c(s*), move h in the NEGATIVE gradient direction
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        if 0 <= i + s_star < n:
            g += (1.0 - h[i + s_star])
        if 0 <= i - s_star < n:
            g -= h[i - s_star]
        grad[i] = g
    # Move against gradient (to reduce c(s*))
    h_m -= strength * grad
    return project_feasible(h_m, n)
```

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h

            if t < 1e-15:

    v = c_aff + d * a
    if hd:
        v = v.copy()
        v[di] = c_aff[di] + d * (1.0 - h0 - d)
    r = float(np.max(v))
    return r if r > um else um

        if 0 <= i + s < n:
            v += 1.0 - h[i + s]

        hm[st:st + b] = rng.rand(b)
    elif u < 0.36:
        f = rng.randint(2, 8)
        ph = rng.rand() * 6.2832
        x = np.linspace(0, 2, n)
        hm += 0.08 * (1.0 - 0.5 * stage) * np.sin(f * np.pi * x + ph)
    elif u < 0.46:
        b1 = rng.randint(0, n - b)
        b2 = rng.randint(0, n - b)
        if abs(b1 - b2) > b:
            tmp = hm[b1:b1 + b].copy()
            hm[b1:b1 + b] = hm[b2:b2 + b]
            hm[b2:b2 + b] = tmp
    elif u < 0.55:
        st = rng.randint(0, n - b)
        hm[st:st + b] = 1.0 - hm[st:st + b]
    elif u < 0.72:
        return directed_mut(h, n, rng, k=rng.randint(1, 4),
                            strength=0.04 + 0.06 * (1.0 - stage))
    elif u < 0.82:
        i1, i2 = rng.choice(n, 2, replace=False)
        hm[i1], hm[i2] = hm[i2], hm[i1]
    elif u < 0.92:
        st = rng.randint(0, n - b)
        hm[st:st + b] += rng.normal(0, 0.05 * (1.0 - 0.3 * stage), b)
    else:
        for _ in range(3):
            st = rng.randint(0, n - 40)
            hm[st:st + 40] += rng.normal(0, 0.04, 40)
    return project_feasible(hm, n)

def _xover(h1, h2, n, rng):
    m = rng.rand(n) < 0.5
    return project_feasible(np.where(m, h1, h2), n)

def evolve(h0, n, dl, rng):
    ps = 25
    pop = [h0.copy()]
    fit = [compute_c5(h0, n)]
    for i in range(1, ps):
        if time.time() > dl:
            break
        h = _mutate(h0, n, rng, 0.0)
        pop.append(h)
        fit.append(compute_c5(h, n))
    best_h = h0.copy()
    best_c = min(fit)
    best_g = 0
    for gen in range(20000):
        if time.time() > dl:
            break
        idx = np.argsort(fit)
        pop = [pop[i] for i in idx]
        fit = [fit[i] for i in idx]
        if fit[0] < best_c:
            best_c = fit[0]
            best_h = pop[0].copy()
            best_g = gen
        if gen - best_g > 150 and gen % 25 == 0:
            for i in range(ps - 3, ps):
                if rng.rand() < 0.5:
                    pop[i] = _mutate(best_h, n, rng, 0.0)
                else:
                    pop[i] = project_feasible(best_h + rng.normal(0, 0.12, n), n)
                fit[i] = compute_c5(pop[i], n)
        stage = min(1.0, gen / 800.0)
        npop = [pop[0].copy(), pop[1].copy()]
        nfit = [fit[0], fit[1]]
        while len(npop) < ps:
            if time.time() > dl:
                break
            r = rng.rand()
            t = rng.randint(0, min(8, len(pop)), 3)
            p1 = pop[int(min(t))]
            t = rng.randint(0, min(8, len(pop)), 3)
            p2 = pop[int(min(t))]
            if r < 0.45:
                ch = _mutate(p1, n, rng, stage)
            elif r < 0.70:
                ch = _xover(p1, p2, n, rng)
                if rng.rand() < 0.4:
                    ch = _mutate(ch, n, rng, stage)
            elif r < 0.85:
                ch = directed_mut(p1, n, rng, k=rng.randint(1, 4),
                                  strength=0.03 + 0.05 * (1.0 - stage))
            else:
                ch = project_feasible(p1 * 0.6 + p2 * 0.4 + rng.normal(0, 0.02, n), n)
            npop.append(ch)
            nfit.append(compute_c5(ch, n))
        pop, fit = npop, nfit
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    dl = t0 + budget_s - 10
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 500
        h0 = np.full(n, 0.5)
    bh = project_feasible(h0, n)
    bc = compute_c5(bh, n)
    t1 = t0 + 0.06 * budget_s
    bh, bc = cd(bh, n, 40, t1, gi=40)
    t2 = t0 + 0.52 * budget_s
    eh, ec = evolve(bh, n, t2, rng)
    if ec < bc:
        bh, bc = eh, ec
    t3 = t0 + 0.74 * budget_s
    bh, bc = cd(bh, n, 50, t3, gi=45)
    nc = max(200, n // 2)
    if nc < n and time.time() < dl - 80:
        xo = np.linspace(0, 2, len(bh))
        xn = np.linspace(0, 2, nc)
        hc = project_feasible(np.interp(xn, xo, bh), nc)
        hc, cc = cd(hc, nc, 20, dl - 80, gi=30)
        hc, cc = evolve(hc, nc, dl - 50, rng)
        xc = np.linspace(0, 2, nc)
        h_new = project_feasible(np.interp(np.linspace(0, 2, n), xc, hc), n)
        h_new, c_new = cd(h_new, n, 30, dl - 20, gi=45)
        if c_new < bc:
            bh, bc = h_new, c_new
    if time.time() < dl:
        bh, bc = cd(bh, n, 40, dl, gi=50)
    bh = project_feasible(bh, len(bh))
    bc = compute_c5(bh, len(bh))
    return bh, bc, len(bh)
