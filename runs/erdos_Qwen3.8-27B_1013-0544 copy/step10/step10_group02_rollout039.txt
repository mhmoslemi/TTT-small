        if abs(xn - h0) > 1e-12:
            update_corr(c, h, p, h0, xn, n)
            h[p] = xn
            count += 1
    return count

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:

        else:

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_i = np.where(ms == (n - 1))[0]
    c[ms] += delta * a
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta

def cd_pass(h, c, n, order):

    for p in order:
        h0 = h[p]
        if h0 < 1e-13 or h0 > 1.0 - 1e-13:

        if abs(xn - h0) > 1e-12:
            update_corr(c, h, p, h0, xn, n)
            h[p] = xn
            count += 1
    return count

def targeted_perturb(h, n, c, rng, strength):
    """Perturb coordinates that contribute to the maximum correlation shifts."""
    c_max = np.max(c)
    n_active = 200  # number of shifts to consider
    active_shifts = np.argsort(c)[-n_active:]
    # Find coordinates appearing in many active shifts
    # Coordinate p appears in shift k if n-1-p <= k <= 2n-2-p
    score = np.zeros(n)
    for k in active_shifts:
        p_lo = max(0, n - 1 - k)
        p_hi = min(n - 1, 2 * n - 2 - k)
        score[p_lo:p_hi + 1] += 1
    # Sample coordinates weighted by score
    total = score.sum()
    if total < 1e-10:
        probs = np.ones(n) / n
    else:
        probs = score / total
    n_perturb = max(3, min(15, n // 50))
    indices = rng.choice(n, size=n_perturb, replace=False, p=probs)
    h_new = h.copy()
    for p in indices:
        h_new[p] = np.clip(h_new[p] + rng.normal(0, strength), 0.0, 1.0)
    return project_feasible(h_new, n)

def swap_perturb(h, n, c, rng):
    """Swap two random coordinates (preserves sum automatically)."""
    p, q = rng.choice(n, size=2, replace=False)
    h_new = h.copy()
    h_new[p], h_new[q] = h_new[q], h_new[p]
    return h_new

def smooth_lse(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def cf(x):
            return x.sum() - n / 2.0
        def cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        try:
            res = minimize(smooth_lse, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
