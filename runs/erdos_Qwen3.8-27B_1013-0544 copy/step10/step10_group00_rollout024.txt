import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def c5_cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50):

            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:

            m_hi = min(2*n-2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n-1)
            jb = p - ms + (n-1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]

            has_diag = len(dp) > 0
            diag_idx = int(dp[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr*(d_hi - d_lo)
            d2 = d_lo + gr*(d_hi - d_lo)
            f1 = c5_cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = c5_cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr*(d_hi - d_lo)
                    f1 = c5_cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:

def topk_subgrad_step(h, n, k_top, step, mom, beta=0.7):
    c = np.correlate(h, 1.0 - h, mode='full')
    k_idx = np.argsort(c)[-k_top:]
    grad = np.zeros(n)
    for ki in k_idx:
        s = ki - (n - 1)
        if s == 0:
            grad += 1.0 - 2.0 * h
        elif s > 0:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] -= h[s:]
        else:
            sp = -s
            grad[:n-sp] += (1.0 - h[sp:])
            grad[sp:] -= h[:n-sp]
    mom = beta * mom + (1.0 - beta) * grad
    h_new = project_feasible(h - step * mom, n)
    return h_new, mom

    return best_h, best_c

def smooth_obj_anti(u, n, alpha):
    half = n // 2
    h = np.empty(n)
    h[:half] = u; h[half:] = 1.0 - u
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return m + mx / alpha + np.log(np.sum(np.exp(shifted - mx))) / alpha
