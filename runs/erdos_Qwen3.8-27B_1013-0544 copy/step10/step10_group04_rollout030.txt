def logsumexp_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def lse_gradient(x, n, alpha):
    """Compute gradient of log-sum-exp approximation."""
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    grad = np.zeros(n)
    for p in range(n):
        p_min = n - 1 - p
        p_max = 2 * n - 2 - p
        ms = np.arange(p_min, p_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib = np.zeros(len(ms))
        if np.any(vf):
            contrib[vf] += (1.0 - x[jf[vf]])
        if np.any(vb):
            contrib[vb] -= x[jb[vb]]
        diag_mask = (ms == (n - 1))
        contrib[diag_mask] = (1.0 - 2.0 * x[p])
        grad[p] = np.sum(weights[ms] * contrib[ms])
    return grad * (2.0 / n)

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(40):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if (xc_lo < r1 and xc_hi > r2) or (x_lo >= r1 and x_hi <= r2):
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, random_order=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = compute_c5(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n) if (random_order and rng is not None) else np.arange(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        cur = compute_c5(h, n)
        if cur < best_c:
            best_c = cur
        if abs(h.sum() - n / 2.0) > 1e-9:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, best_c

def lse_gradient(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    grad = np.zeros(n)
    for p in range(n):
        p_min = n - 1 - p
        p_max = 2 * n - 2 - p
        ms = np.arange(p_min, p_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib = np.zeros(len(ms))
        if np.any(vf):
            contrib[vf] += (1.0 - x[jf[vf]])
        if np.any(vb):
            contrib[vb] -= x[jb[vb]]
        diag_mask = (ms == (n - 1))
        contrib[diag_mask] = (1.0 - 2.0 * x[p])
        grad[p] = np.dot(weights[ms], contrib)
    return grad * (2.0 / n)

def adam_lse(h, n, alphas, steps, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        x = best_h.copy()
        m_v = np.zeros(n)
        v_v = np.zeros(n)
        t = 0
        lr = 0.005
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        for _ in range(steps):
            if time.time() > deadline:
                break
            t += 1
            g = lse_gradient(x, n, alpha)
            m_v = beta1 * m_v + (1.0 - beta1) * g
            v_v = beta2 * v_v + (1.0 - beta2) * g * g
            m_hat = m_v / (1.0 - beta1 ** t)
            v_hat = v_v / (1.0 - beta2 ** t)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
            x = project_feasible(x, n)
            if t % 40 == 0:
                cur = compute_c5(x, n)
                if cur < best_c:
                    best_c = cur
                    best_h = x.copy()
    return best_h, best_c

def perturb_h(h, n, rng, scale=1.0):
    method = rng.randint(5)
    if method == 0:
        shift = rng.randint(1, max(2, n // 40)) * int(scale)
        h_new = np.roll(h, shift)
    elif method == 1:
        ks = max(3, n // 80)
        kernel = np.exp(-0.5 * (np.arange(-ks, ks + 1) / (ks // 2 + 1)) ** 2)
        kernel /= kernel.sum()
        h_smooth = np.convolve(h, kernel, mode='same')
        h_new = h + 0.4 * (h_smooth - h)
    elif method == 2:
        factor = 1.0 + rng.uniform(-0.04, 0.04) * scale
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2.0 / factor, n)
        x_new = np.clip(x_new, 0, 2)
        h_new = np.interp(x_new, x_old, h)
    elif method == 3:
        i = rng.randint(n)
        length = rng.randint(n // 8, n // 3)
        j = min(n, i + length)
        h_new = h.copy()
        h_new[i:j] = 1.0 - h_new[i:j]
    else:
        sigma = 0.01 * scale
        h_new = h + rng.normal(0, sigma, n)
    return project_feasible(h_new, n)
