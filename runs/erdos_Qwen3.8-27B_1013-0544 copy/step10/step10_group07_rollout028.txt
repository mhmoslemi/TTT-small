    shifted = alpha * (c - m)
    mx = shifted.max()

    grad = np.zeros(n)

    # is a shifted version of h
    # Forward: for k > 0, grad[:n-k] += p * (1 - h[k:])
    # Backward: for k > 0, grad[k:] -= p * h[:n-k]
    # For k < 0 (k = -kk): grad[kk:] += p * (1 - h[:n-kk]), grad[:n-kk] -= p * h[kk:]
    # For k = 0: grad += p * (1 - 2*h)

    pos_mask = k_all > 0
    neg_mask = k_all < 0
    zero_mask = k_all == 0

    # = sum over k>0, j+k<n of p_k*(1-h[j+k]) - sum over k>0, j>=k of p_k*h[j-k]

        k = k_all[ki]
        pi = p[ki]
        if pi < 1e-14:

def subgradient_polish(h, n, max_iter, deadline):

    c_lower = 0.3  # lower bound
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_c5 = c.max() * (2.0 / n)
        k_star = int(c.argmax()) - (n - 1)
        # Compute subgradient
        grad = np.zeros(n)
        if k_star == 0:
            grad[:] = 1.0 - 2.0 * h
        elif k_star > 0:
            grad[:n-k_star] = 1.0 - h[k_star:]
            grad[k_star:] -= h[:n-k_star]
        else:
            k = -k_star
            grad[k:] = 1.0 - h[:n-k]
            grad[:n-k] -= h[k:]
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:
            break

        obj_val = cur_c5 * n / 2.0  # in units of sum(c)*dx
        eta = (obj_val - c_lower * n / 2.0) / (g_norm * g_norm)
        eta = min(eta, 0.1)  # cap step size

            h_new = project_feasible(h - eta * grad * (2.0/n), n)
            if compute_c5(h_new, n) < cur_c5:
                h = h_new
            else:
                break

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def obj_grad_smooth(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    ev = np.exp(shifted - mx)
    p = ev / ev.sum()
    obj = (m + (mx + np.log(ev.sum())) / alpha) * (2.0 / n)
    k_all = np.arange(len(c)) - (n - 1)
    grad = np.zeros(n)
    zm = (k_all == 0)
    pm = (k_all > 0)
    nm = (k_all < 0)
    if np.any(zm):
        grad[:] += p[zm] * (1.0 - 2.0 * h)
    for ki in np.where(pm)[0]:
        k = k_all[ki]
        pi = p[ki]
        if pi < 1e-14:
            continue
        grad[:n-k] += pi * (1.0 - h[k:])
        grad[k:] -= pi * h[:n-k]
    for ki in np.where(nm)[0]:
        k = -k_all[ki]
        pi = p[ki]
        if pi < 1e-14:
            continue
        grad[k:] += pi * (1.0 - h[:n-k])
        grad[:n-k] -= pi * h[k:]
    grad *= (2.0 / n)
    return obj, grad

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            def f(z):
                return obj_grad_smooth(z, n, alpha)[0]
            def g(z):
                return obj_grad_smooth(z, n, alpha)[1]
            def ceq(z):
                return z.sum() - n / 2.0
            res = minimize(f, best_h, jac=g, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': ceq}],
                           options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-14:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def obj_grad_smooth(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    ev = np.exp(shifted - mx)
    p = ev / ev.sum()
    obj = (m + (mx + np.log(ev.sum())) / alpha) * (2.0 / n)
    k_all = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for ki in range(len(k_all)):
        pi = p[ki]
        if pi < 1e-14:
            continue
        k = k_all[ki]
        if k == 0:
            grad[:] += pi * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += pi * (1.0 - h[k:])
            grad[k:] -= pi * h[:n - k]
        else:
            kk = -k
            grad[kk:] += pi * (1.0 - h[:n - kk])
            grad[:n - kk] -= pi * h[kk:]
    grad *= (2.0 / n)
    return obj, grad

def _sf(z, n, alpha):
    return obj_grad_smooth(z, n, alpha)[0]

def _sg(z, n, alpha):
    return obj_grad_smooth(z, n, alpha)[1]

def _se(z, n):
    return z.sum() - n / 2.0

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            f = partial(_sf, n=n, alpha=alpha)
            g = partial(_sg, n=n, alpha=alpha)
            c = partial(_se, n=n)
            res = minimize(f, best_h, jac=g, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': c}],
                           options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-14:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def cd_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = max(unaff, float(np.max(c[:m_min])))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(45):
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        ok = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-15):
            ok = False
        if ok and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if ok and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            ok = False
        if ok and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, order=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        idxs = order if order is not None else range(n)
        for p in idxs:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def subgrad_polish(h, n, max_iter, deadline):
    h = h.copy()
    c_lower = 0.33
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_c5 = float(c.max()) * (2.0 / n)
        ki = int(c.argmax())
        k_star = ki - (n - 1)
        grad = np.zeros(n)
        if k_star == 0:
            grad[:] = 1.0 - 2.0 * h
        elif k_star > 0:
            grad[:n - k_star] = 1.0 - h[k_star:]
            grad[k_star:] = -h[:n - k_star]
        else:
            kk = -k_star
            grad[kk:] = 1.0 - h[:n - kk]
            grad[:n - kk] = -h[kk:]
        g_norm_sq = float(np.dot(grad, grad))
        if g_norm_sq < 1e-20:
            break
        obj_v = cur_c5 * n / 2.0
        eta = (obj_v - c_lower * n / 2.0) / (g_norm_sq + 1e-20)
        eta = min(eta, 0.05, 1.0 / (it + 1) * 0.5)
        for factor in [1.0, 0.5, 0.25, 0.1]:
            h_new = project_feasible(h - factor * eta * grad * (2.0 / n), n)
            if compute_c5(h_new, n) < cur_c5:
                h = h_new
                break
    return h, compute_c5(h, n)
