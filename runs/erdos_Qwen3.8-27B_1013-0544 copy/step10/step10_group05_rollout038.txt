                else:

def cd_pass(h, n, L, max_passes, deadline, rng=None, rand_order=False):
    """Coordinate descent with fixed level L. Returns (h, converged)."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n) if (rand_order and rng) else range(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            ms, a = get_coord_info(h, n, p)
            c_base = c[ms]
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            
            neg_mask = a < -1e-14
            pos_mask = a > 1e-14
            zero_mask = np.abs(a) <= 1e-14
            
            x_lo, x_hi = 0.0, 1.0
            if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
                continue  # Can't reduce this coordinate to help
            if np.any(neg_mask):
                x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
            if np.any(pos_mask):
                x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
            x_lo = max(x_lo, 0.0)
            x_hi = min(x_hi, 1.0)
            if has_diag:
                ci = c_base[diag_idx[0]]
                ad = a[diag_idx[0]]
                disc = ad * ad + 4.0 * (ci - L)
                if disc >= 0:
                    sq = np.sqrt(disc)
                    r1 = (ad - sq) / 2.0
                    r2 = (ad + sq) / 2.0
                    if x_lo <= h0 + r1:
                        x_hi = min(x_hi, h0 + r1)
                    elif x_hi >= h0 + r2:
                        x_lo = max(x_lo, h0 + r2)
                    else:

                            x_hi = min(x_hi, h0 + r1)
                        else:
                            x_lo = max(x_lo, h0 + r2)
            if x_lo > x_hi + 1e-12:
                continue
            xn = float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                c[ms] += delta * a
                if has_diag:
                    c[ms[diag_idx[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        # Periodic reprojection
        if it % 10 == 9:
            h = project_feasible(h, n)
            if abs(h.sum() - n / 2.0) > 1e-10:
                c = np.correlate(h, 1.0 - h, mode='full')
    return h, improved

def subgradient_step(h, n, f_star, step_alpha):
    """One subgradient step with Polyak step size."""
    c = np.correlate(h, 1.0 - h, mode='full')
    max_c = c.max()
    f = max_c * (2.0/n)
    active = np.where(c >= max_c - 1e-8)[0]
    g = np.zeros(n)
    for m in active:
        i_lo = max(0, m - n + 1)
        i_hi = min(n - 1, m)
        for i in range(i_lo, i_hi + 1):
            j = m - i
            g[i] += (1.0 - 2.0 * h[j])
    g /= len(active)
    # Polyak step size
    g_norm_sq = np.dot(g, g)
    if g_norm_sq < 1e-20:
        return h
    alpha = (f - f_star) / g_norm_sq * step_alpha
    h_new = h - alpha * g
    return project_feasible(h_new, n)

= dx · (∇max_c + ∑_i softmax_i · ∇c_i)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    m = c.max()

        bounds = [(0.0, 1.0)] * n

            res = minimize(smooth_obj, best_h, args=(n, alpha), jac=lambda x, nn, aa: smooth_grad(x, nn, aa),
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interpolate_h(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)

def fourier_init(n, K, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        amp = rng.normal(0, 0.3 / np.sqrt(k))
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * x / 2.0 + phase)
    return project_feasible(h, n)
