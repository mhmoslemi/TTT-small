def soft_max_obj(h, n, alpha):
    """Compute soft max of correlations and its gradient."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    shifted = shifted - shifted.max()  # for numerical stability
    weights = np.exp(shifted)
    weights /= weights.sum()
    obj = (m + np.log(weights.sum()) / alpha) * (2.0 / n)
    return obj, weights, c

    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_coord_fast(c, h, p, n, nbisect=35):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo_a = max(0, n - 1 - p)
    hi_a = min(2 * n - 2, n - 1 + p)
    lo_b = max(0, p)
    hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(lo_a, lo_b)
    m_hi = max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    non_diag = np.ones(len(ms), dtype=bool)
    if has_diag:
        non_diag[diag_idx] = False
    a_nd = a[non_diag]
    c_nd = c_base[non_diag]
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(c[m_hi + 1:])))
    L_lo = unaff_max
    L_hi = float(np.max(c))
    for _ in range(nbisect):
        if L_hi - L_lo < 1e-14:
            break
        T = (L_lo + L_hi) / 2.0
        feasible = True
        if unaff_max > T + 1e-15:
            feasible = False
        elif feasible and len(a_nd) > 0:
            x_lo_c, x_hi_c = 0.0, 1.0
            neg = a_nd < -1e-13
            pos = a_nd > 1e-13
            if np.any(neg):
                x_lo_c = max(x_lo_c, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                x_hi_c = min(x_hi_c, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            if x_lo_c > x_hi_c + 1e-14:
                feasible = False
