def cd_active_set(h, n, max_passes, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_passes):
        if time.time() > deadline: break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12: continue
            # Find which lags are affected by changing h[p]
            m_lo = max(0, n - 1 - p)
            m_hi = min(2*n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            
            # Linear and quadratic coefficients for each affected lag
            # g_k(h) changes by: d * (sum of affected terms) - d^2 (self term if k=0 relative)
            # Actually: g_k = sum_i h_i * (1-h_{i+k})
            # When h[p] changes by d:
            # - If p is in the "h" position: contributes d * (1-h_{p+k}) or d * (1-h_{p-k})
            # - If p is in the "(1-h)" position: contributes -d * h_{p-k} or -d * h_{p+k}
            # - Self term: when the same index appears in both, we get -d^2
            
            a = np.zeros(len(ms))
            has_self = np.zeros(len(ms), dtype=bool)
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            
            # Self term: when both jf and jb point to p
            # jf[i] == p means p + ms[i] - (n-1) == p, so ms[i] == n-1
            # jb[i] == p means p - ms[i] + (n-1) == p, so ms[i] == n-1
            # So the self term is at ms == n-1
            self_mask = (ms == (n-1))
            if np.any(self_mask):
                has_self[self_mask] = True
                # The coefficient a already includes the linear term
                # The quadratic term is -d^2 (since h_p*(1-h_p) -> h_p*(1-h_p) + d*(1-h_p) + d*(-h_p) - d^2)

                # So the change is d*(1-2*h_p) - d^2
                # But in our formulation, a includes (1-h_p) from valid_f and -h_p from valid_b
                # So a at self position = (1-h_p) - h_p = 1 - 2*h_p
                # And the quadratic term is -d^2

            # Unaffected max
            unaff_max = 0.0
            if m_lo > 0: unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n-2: unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            
            # Now optimize f(d) = max(c_aff + d*a - d^2*has_self, unaff_max)
            # subject to -h0 <= d <= 1-h0
            
            # The function is max of (parabolas and a constant)
            # Each parabola: c_aff[i] + d*a[i] - d^2*has_self[i]
            # For has_self[i]: concave parabola, vertex at d = a[i]/2
            # For not has_self[i]: linear function
            # Plus the constant unaff_max
            
            # Find the optimal d by checking:
            # 1. Vertex of each concave parabola (clipped to bounds)
            # 2. Intersection points where the max changes
            # 3. Bounds
            
            d_lo, d_hi = -h0, 1.0 - h0
            best_d = 0.0
            best_f = float('inf')
            
            # Candidate points:
            candidates = [d_lo, d_hi, 0.0]
            for i in range(len(ms)):
                if has_self[i]:
                    v = a[i] / 2.0  # vertex
                    if d_lo <= v <= d_hi:
                        candidates.append(v)
                # Intersections with unaff_max:
                # c_aff[i] + d*a[i] - d^2*has_self[i] = unaff_max
                if has_self[i]:
                    # -d^2 + a[i]*d + (c_aff[i] - unaff_max) = 0
                    # d^2 - a[i]*d - (c_aff[i] - unaff_max) = 0
                    disc = a[i]**2 + 4*(c_aff[i] - unaff_max)
                    if disc >= 0:
                        sqrt_disc = np.sqrt(disc)
                        d1 = (a[i] + sqrt_disc) / 2.0
                        d2 = (a[i] - sqrt_disc) / 2.0
                        if d_lo <= d1 <= d_hi: candidates.append(d1)
                        if d_lo <= d2 <= d_hi: candidates.append(d2)
                else:
                    # Linear: c_aff[i] + d*a[i] = unaff_max
                    if abs(a[i]) > 1e-15:
                        d_int = (unaff_max - c_aff[i]) / a[i]
                        if d_lo <= d_int <= d_hi: candidates.append(d_int)
            
            # Also check points where two parabolas intersect
            # (This could be many, so let's just sample)
            n_samples = 20
            for ds in np.linspace(d_lo, d_hi, n_samples):
                candidates.append(ds)

                if d < d_lo or d > d_hi: continue

                if f_val < best_f - 1e-15:

                    best_d = d

            lo = max(d_lo, best_d - delta)
            hi = min(d_hi, best_d + delta)

                vals = c_aff + d * a - d**2 * has_self
                return max(float(np.max(vals)), unaff_max)
            # Golden section
            d1 = hi - gr * (hi - lo)
            d2 = lo + gr * (hi - lo)
            f1, f2 = f_of_d(d1), f_of_d(d2)
            for _ in range(40):

                if f1 < f2:
                    hi, d2, f2 = d2, d1, f1
                    d1 = hi - gr * (hi - lo)
                    f1 = f_of_d(d1)
                else:
                    lo, d1, f1 = d1, d2, f2
                    d2 = lo + gr * (hi - lo)
                    f2 = f_of_d(d2)
            d_star = (lo + hi) / 2.0
            f_star = f_of_d(d_star)
            
            f_curr = f_of_d(0.0)
            if f_star < f_curr - 1e-14:
                h[p] = h0 + d_star
                improved = True
        
        if improved:
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c: best_c = c_new
        else:
            break
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > time.time():
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            if res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

        if time.time() > deadline:

        improved = False

            if time.time() > deadline:

            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:

            ms = np.arange(m_lo, m_hi + 1)

            jf = p + kf
            jb = p - kf
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):

            has_self = (ms == (n - 1)).astype(float)

            unaff_max = 0.0
