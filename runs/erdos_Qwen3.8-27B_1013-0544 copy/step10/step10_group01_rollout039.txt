                nc = compute_c5(nh, n)
                if nc < best_c:
                    best, best_c = nh, nc
        except Exception:
            pass
    return best, best_c

def _cd_inner(h, n, p, c, gs_iters=70):
    h0 = h[p]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    dp = np.where(ms == (n - 1))[0]
    has_d = len(dp) > 0
    di = int(dp[0]) if has_d else -1
    ca = c[ms]
    um = 0.0
    if m_lo > 0:
        um = max(um, float(c[:m_lo].max()))
    if m_hi < 2 * n - 2:
        um = max(um, float(c[m_hi + 1:].max()))

    def fv(d):
        v = ca + d * a
        if has_d:
            v[di] = ca[di] + d * (1.0 - h0 - d)
        return max(float(np.max(v)), um)

    dlo, dhi = -h0, 1.0 - h0
    d1 = dhi - gr * (dhi - dlo)
    d2 = dlo + gr * (dhi - dlo)
    f1, f2 = fv(d1), fv(d2)
    for _ in range(gs_iters):
        if dhi - dlo < 1e-16:
            break
        if f1 < f2:
            dhi, d2, f2 = d2, d1, f1
            d1 = dhi - gr * (dhi - dlo)
            f1 = fv(d1)
        else:
            dlo, d1, f1 = d1, d2, f2
            d2 = dlo + gr * (dhi - dlo)
            f2 = fv(d2)
    dopt = (dlo + dhi) / 2.0
    return dopt, fv(dopt), fv(0.0)

def cd_optimize(h, n, max_passes, deadline, thresh=5e-6, gs_iters=70):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            if h[p] <= 1e-12 or h[p] >= 1.0 - 1e-12:
                continue
            dopt, fok, fcur = _cd_inner(h, n, p, c, gs_iters)
            if fok < fcur - 1e-14:
                h[p] = h[p] + dopt
                improved = True
        if improved:
            h[h < thresh] = 0.0
            h[h > 1.0 - thresh] = 1.0
            h = project_feasible(h, n)
            cnew = compute_c5(h, n)
            if cnew < best_c:
                best_c = cnew
        else:
            break
    return h, min(best_c, compute_c5(h, n))

def focused_cd(h, n, max_passes, deadline, gs_iters=70):
    """CD that only updates coords affecting near-max correlations."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = c.max()
        tol = 0.005 * dx
        active_idx = np.where(c >= cmax - tol)[0]
        active_coords = set()
        for ki in active_idx:
            lag = ki - (n - 1)
            lo = max(0, -lag)
            hi = min(n, n - lag)
            active_coords.update(range(lo, hi))
        improved = False
        for p in sorted(active_coords):
            if time.time() > deadline:
                break
            if h[p] <= 1e-12 or h[p] >= 1.0 - 1e-12:
                continue
            dopt, fok, fcur = _cd_inner(h, n, p, c, gs_iters)
            if fok < fcur - 1e-14:
                h[p] = h[p] + dopt
                improved = True
        if improved:
            h = project_feasible(h, n)
            cnew = compute_c5(h, n)
            if cnew < best_c:
                best_c = cnew
        else:
            break
    return h, min(best_c, compute_c5(h, n))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(200, 0.5)
        n = 200
    h = project_feasible(h, n)
    best_h, best_c = h.copy(), compute_c5(h, n)

    # Phase 1: SLSQP warm start
    h1, c1 = slsqp(h, n, [300, 3000, 30000], 350)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Full CD
    h2, c2 = cd_optimize(best_h, n, 80, deadline, thresh=5e-6, gs_iters=70)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Focused CD on active constraints
    h3, c3 = focused_cd(best_h, n, 60, deadline, gs_iters=80)
    if c3 < best_c:
        best_h, best_c = h3, c2 if False else c3
    best_h = h3 if c3 < best_c else best_h
    if c3 < best_c:
        best_c = c3

    # Phase 4: Multi-scale
    x_big = np.linspace(0, 2, len(best_h))
    for n_try in [300, 450, 550]:
        if time.time() > deadline - 350:
            break
        if n_try >= n:
            continue
        x_s = np.linspace(0, 2, n_try)
        hs = project_feasible(np.interp(x_s, x_big, best_h), n_try)
        hs, cs = slsqp(hs, n_try, [3000, 30000], 250)
        hs, cs = cd_optimize(hs, n_try, 60, deadline, gs_iters=60)
        hs, cs = focused_cd(hs, n_try, 40, deadline, gs_iters=70)
        hb = project_feasible(np.interp(x_big, x_s, hs), n)
        hb, cb = cd_optimize(hb, n, 50, deadline, gs_iters=60)
        hb, cb = focused_cd(hb, n, 40, deadline, gs_iters=75)
        if cb < best_c:
            best_h, best_c = hb, cb

    # Phase 5: Diverse restarts
    n_cur = len(best_h)
    x = np.linspace(0, 2, n_cur)
    while time.time() < deadline - 120:
        r = rng.randint(8)
        if r == 0:
            hp = project_feasible(best_h + rng.normal(0, 0.008, n_cur), n_cur)
        elif r == 1:
            freq = rng.uniform(0.5, 6.0)
            hp = project_feasible(best_h + 0.015 * np.sin(2 * np.pi * freq * x), n_cur)
        elif r == 2:
            idx = rng.choice(n_cur, size=max(5, n_cur // 40), replace=False)
            hp = best_h.copy()
            hp[idx] = rng.uniform(0.0, 1.0, len(idx))
            hp = project_feasible(hp, n_cur)
        elif r == 3:
            p = rng.randint(0, max(1, n_cur - 25))
            hp = best_h.copy()
            hp[p:p + 25] = 1.0 - hp[p:p + 25]
            hp = project_feasible(hp, n_cur)
        elif r == 4:
            # Target active correlations
            c = np.correlate(best_h, 1.0 - best_h, mode='full')
            cmax = c.max()
            active = np.where(c >= cmax - 0.002 * (2.0 / n_cur))[0]
            hp = best_h.copy()
            for ki in active[:5]:
                lag = ki - (n_cur - 1)
                lo = max(0, -lag)
                hi = min(n_cur, n_cur - lag)
                pts = rng.randint(lo, hi, size=3)
                hp[pts] += rng.uniform(-0.08, 0.08, 3)
            hp = np.clip(hp, 0, 1)
            hp = project_feasible(hp, n_cur)
        elif r == 5:
            # Smooth perturbation
            kernel = np.exp(-0.5 * ((np.arange(n_cur) - n_cur // 2) / 20.0) ** 2)
            hp = project_feasible(best_h + 0.03 * kernel * rng.randn(), n_cur)
        elif r == 6:
            # Low-freq mode
            m = rng.randint(1, 4)
            phase = rng.uniform(0, 2 * np.pi)
            hp = project_feasible(best_h + 0.02 * np.cos(2 * np.pi * m * x + phase), n_cur)
        else:
            # Mirror perturbation
            hp = best_h.copy()
            half = n_cur // 2
            hp[half:] = 1.0 - hp[half:][::-1] + rng.normal(0, 0.01, n_cur - half)
            hp = project_feasible(hp, n_cur)

        ho, co = slsqp(hp, n_cur, [3000, 30000], 180)
        ho, co = cd_optimize(ho, n_cur, 40, deadline, gs_iters=55)
        ho, co = focused_cd(ho, n_cur, 30, deadline, gs_iters=65)
        if co < best_c:
            best_h, best_c = ho, co

    # Phase 6: Final polish - alternating CD and focused CD
    if time.time() < deadline - 15:
        for _ in range(3):
            if time.time() > deadline - 10:
                break
            hf, cf = cd_optimize(best_h, n_cur, 50, deadline, thresh=3e-7, gs_iters=80)
            if cf < best_c:
                best_h, best_c = hf, cf
            hf, cf = focused_cd(best_h, n_cur, 40, deadline, gs_iters=85)
            if cf < best_c:
                best_h, best_c = hf, cf

    # Phase 7: Final SLSQP with very high alpha
    if time.time() < deadline - 5:
        hf, cf = slsqp(best_h, n_cur, [500000, 2000000], 400)
        if cf < best_c:
            best_h = project_feasible(hf, n_cur)
            best_c = compute_c5(best_h, n_cur)
        else:
            best_h = project_feasible(best_h, n_cur)
            best_c = compute_c5(best_h, n_cur)

    return best_h, best_c, len(best_h)
