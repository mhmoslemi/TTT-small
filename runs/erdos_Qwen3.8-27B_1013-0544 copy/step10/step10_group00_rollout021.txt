def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def coord_descent(h, n, max_passes, deadline, gs_iters=50):

            break

        improved = False

            if time.time() > deadline:
                break

            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue

            m_hi = min(2 * n - 2, n - 1 + p)

            jb = p - ms + (n - 1)

            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]

                return result

def slsqp_optimize(h, n, alphas, maxiter_each=200, deadline=None):

        dx = 2.0 / n
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * dx
    
    def _sum_eq(x):
        return x.sum() - n / 2.0
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': lambda x: np.ones(n), 'args': ()}]
    for alpha in alphas:

        try:
            res = minimize(smooth_obj, best_h, args=(alpha,), method='SLSQP',
                           bounds=bounds, constraints=cons,

            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    best_c = compute_c5(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    return h, best_c

def quasi_periodic_h(n, alpha, beta=0.0):
    x = np.arange(n, dtype=float)
    vals = alpha * x + beta
    frac = vals - np.floor(vals)
    h = (frac < 0.5).astype(float)
    return project_feasible(h, n)

def _slsqp_sum_eq(x, n):
    return x.sum() - n / 2.0

def _slsqp_sum_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=200, deadline=None):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            def obj(x):
                c = np.correlate(x, 1.0 - x, mode='full')
                m = c.max()
                shifted = alpha * (c - m)
                mx = shifted.max()
                lse = mx + np.log(np.sum(np.exp(shifted - mx)))
                return float((m + lse / alpha) * dx)
            res = minimize(obj, best_h, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _slsqp_sum_eq,
                                         'jac': _slsqp_sum_jac, 'args': (n,)}],
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def swap_phase(h, n, max_passes, deadline, seed=42):
    h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        c_curr = np.correlate(h, 1.0 - h, mode='full')
        max_val = float(c_curr.max())
        n_attempts = n
        for _ in range(n_attempts):
            if time.time() > deadline:
                break
            p = rng.randint(n)
            q = rng.randint(n)
            if p == q:
                continue
            hp, hq = h[p], h[q]
            if abs(hp - hq) < 1e-12:
                continue
            h[p], h[q] = hq, hp
            c_new = np.correlate(h, 1.0 - h, mode='full')
            if float(c_new.max()) < max_val - 1e-14:
                improved = True
                c_curr = c_new
                max_val = float(c_new.max())
            else:
                h[p], h[q] = hp, hq
        if improved:
            h = project_feasible(h, n)
            c_check = compute_c5(h, n)
            if c_check < best_c:
                best_c = c_check
        else:
            break
    return h, best_c

def perturbation_cycle(h, n, deadline, seed=42):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    remaining = deadline - time.time()
    n_cycles = max(3, int(remaining / 30))
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        scale = 0.05 * (0.7 ** (cyc / max(1, n_cycles // 3)))
        t = cyc % 5
        if t == 0:
            hp = best_h + rng.normal(0, scale, n)
        elif t == 1:
            hp = best_h.copy()
            idx = rng.choice(n, size=max(1, n // 6), replace=False)
            hp[idx] = 1.0 - hp[idx]
        elif t == 2:
            hp = best_h.copy()
            block = max(1, n // 8)
            start = rng.randint(0, max(1, n - block))
            hp[start:start + block] = 1.0 - hp[start:start + block]
        elif t == 3:
            x = np.linspace(0, 2, n)
            freq = rng.uniform(0.5, 8.0)
            phase = rng.uniform(0, 2 * np.pi)
            hp = best_h + scale * 0.5 * np.sin(2 * np.pi * freq * x + phase)
        else:
            hp = 1.0 - best_h + rng.normal(0, scale * 0.5, n)
        hp = project_feasible(hp, n)
        h_opt, c_opt = coord_descent(hp, n, 25, deadline, gs_iters=45)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    return best_h, best_c
