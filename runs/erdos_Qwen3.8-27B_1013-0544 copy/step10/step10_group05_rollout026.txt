def project_feasible(h, n):

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    sm = np.max(shifted)
    lse = sm + np.log(np.sum(np.exp(shifted - sm)))
    return (m + lse / alpha) * (2.0 / n)

    return x.sum() - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def slsqp_opt(h0, n, alphas, maxiter, deadline):

    for alpha in alphas:

            if new_c < best_c:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):

    m_max = min(2 * n - 2, 2 * n - 2 - p)

    jb = p - ms + (n - 1)

    vb = (jb >= 0) & (jb < n)

def cd_pass(h, c, n, order, deadline):
    improved = False
    for p in order:
        h0 = h[p]
        if h0 < 1e-13 or h0 > 1.0 - 1e-13:
            continue
        xn = solve_coord(c, h, p, n)
        if abs(xn - h0) > 1e-14:
            delta = xn - h0
            m_min = max(0, n - 1 - p)
            m_max = min(2 * n - 2, 2 * n - 2 - p)
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c[ms] += delta * a
            di = np.where(ms == (n - 1))[0]
            if len(di) > 0:
                c[ms[di[0]]] -= delta * delta
            h[p] = xn
            improved = True
    return improved

def cd_solve(h, n, max_passes, deadline, rng=None, sequential=True):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        if sequential:
            order = np.arange(n)
        else:
            order = rng.permutation(n)
        imp = cd_pass(h, c, n, order, deadline)
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
        if not imp:
            break
    return h, compute_c5(h, n)

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    lse = sm + np.log(np.sum(np.exp(shifted - sm)))
    return (m + lse / alpha) * (2.0 / n)

def _cons_fun(x):
    return x.sum() - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def slsqp_opt(h0, n, alphas, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            res = minimize(smooth_max_obj, best_h, args=(n, alpha),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interp_h(h_src, n_src, n_tgt):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(xt, xs, h_src), n_tgt)
