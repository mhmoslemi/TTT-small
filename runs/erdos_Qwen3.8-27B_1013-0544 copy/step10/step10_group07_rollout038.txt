    lse = mx + np.log(np.sum(np.exp(shifted - mx)))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def cd_1d(h, n, max_passes, deadline):

def cd_pair(h, n, p):

    s = n / 2.0 - h[p] - h[p2]
    a_lo = max(0.0, s - 1.0)
    a_hi = min(1.0, s)
    if a_hi - a_lo < 1e-13:
        return

    ho[p] = 0.0
    ho[p2] = 0.0
    c_base = np.correlate(ho, 1.0 - ho, mode='full')
    k_all = np.arange(2 * n - 1) - (n - 1)
    pk = p - k_all
    pk2 = p + k_all
    pkv = (pk >= 0) & (pk < n)
    pk2v = (pk2 >= 0) & (pk2 < n)
    hp = np.zeros(2 * n - 1)
    hp2 = np.zeros(2 * n - 1)
    if np.any(pkv):
        hp[pkv] = ho[pk[pkv]]
    if np.any(pk2v):
        hp2[pk2v] = ho[pk2[pk2v]]
    q_p = p + 1 - k_all
    q_p2 = p + 1 + k_all
    q1v = (q_p >= 0) & (q_p < n)
    q2v = (q_p2 >= 0) & (q_p2 < n)
    hq1 = np.zeros(2 * n - 1)
    hq2 = np.zeros(2 * n - 1)
    if np.any(q1v):
        hq1[q1v] = ho[q_p[q1v]]
    if np.any(q2v):
        hq2[q2v] = ho[q_p2[q2v]]
    A = 1.0 - hp2 - hp
    B = 1.0 - hq2 - hq1
    A0 = hp2
    B0 = hq2
    cross_lin = np.zeros(2 * n - 1)
    cross_quad = np.zeros(2 * n - 1)
    cross_quad[n] = 1.0
    cross_lin[n] = -(s + 1.0)
    cross_quad[n - 2] = 1.0
    cross_lin[n - 2] = (1.0 - s)
    Q0 = c_base + A0 + B0 + s * B
    Q1 = A - B + cross_lin
    Q2 = cross_quad

    vals = Q0[None, :] + Q1[None, :] * grid[:, None] + Q2[None, :] * grid[:, None] ** 2

    i0 = int(fvals.argmin())

    h[p] = a_star
    h[p2] = s - a_star

def cd_pair_sweep(h, n, max_passes, deadline):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(0, n - 1, 2):
            old = (h[p], h[p + 1])
            cd_pair(h, n, p)
            if abs(h[p] - old[0]) > 1e-13 or abs(h[p + 1] - old[1]) > 1e-13:
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': lambda z: z.sum() - n / 2.0,
                                         'jac': lambda z: np.ones(len(z))}],
                           options={'maxiter': maxiter, 'ftol': 1e-20})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-13:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def fourier_init(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, 20):
        amp = rng.uniform(0.1, 0.5) / (k ** 0.7)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * x / 2.0 + phase)
    return project_feasible(h, n)

def periodic_init(n, period, duty):
    x = np.linspace(0, 2, n, endpoint=False)
    frac = (x * period / 2.0) % 1.0
    h = (frac < duty).astype(float)
    return project_feasible(h, n)
