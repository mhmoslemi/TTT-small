    if kf_lo <= kf_hi:
        idx = np.arange(kf_lo, kf_hi + 1)
        a[idx] += (1.0 - h[p + idx - (n - 1)])
    kb_lo = max(0, p)
    kb_hi = min(2 * n - 2, p + n - 1)
    if kb_lo <= kb_hi:
        idx = np.arange(kb_lo, kb_hi + 1)
        a[idx] -= h[p - idx + (n - 1)]
    lo_d = -h0
    hi_d = 1.0 - h0
    pos = a > 1e-15
    neg = a < -1e-15
    zero = ~(pos | neg)
    T_lo = 0.0
    T_hi = float(np.max(c))
    if np.any(zero):
        T_lo = max(T_lo, float(np.max(c[zero])))
    for _ in range(nbisect):
        T_mid = (T_lo + T_hi) / 2.0
        feas = True
        if np.any(zero) and T_mid < c[zero].max():
            feas = False
        dl, dh = lo_d, hi_d
        if feas and np.any(pos):
            dh = min(dh, float(np.min((T_mid - c[pos]) / a[pos])))
        if feas and np.any(neg):
            dl = max(dl, float(np.max((T_mid - c[neg]) / a[neg])))
        if dl > dh + 1e-14:
            feas = False
        if feas:
            T_hi = T_mid
        else:
            T_lo = T_mid
    T = (T_lo + T_hi) / 2.0
    dl, dh = lo_d, hi_d
    if np.any(pos):
        dh = min(dh, float(np.min((T - c[pos]) / a[pos])))
    if np.any(neg):
        dl = max(dl, float(np.max((T - c[neg]) / a[neg])))
    dl = max(dl, lo_d)
    dh = min(dh, hi_d)
    if dl > dh + 1e-14:
        return h0
    return float(np.clip((dl + dh) / 2.0, 0.0, 1.0))

def fast_cd(h, n, max_passes, deadline, rng=None, random_order=True):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            x_new = solve_point(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def fourier_project(h, n, J=4):
    x = np.linspace(0, 2, n)
    coeffs = np.zeros(2 * J)
    for j in range(1, J + 1):
        coeffs[2*(j-1)] = np.mean(h * np.cos(j * np.pi * x)) * 2
        coeffs[2*(j-1)+1] = np.mean(h * np.sin(j * np.pi * x)) * 2
    h_f = np.full(n, 0.5)
    for j in range(1, J + 1):
        h_f += coeffs[2*(j-1)] * np.cos(j * np.pi * x)
        h_f += coeffs[2*(j-1)+1] * np.sin(j * np.pi * x)
    return project_feasible(h_f, n)

def smooth_restart(h, n, rng, amp_range=(0.02, 0.06)):
    x = np.linspace(0, 2, n)
    freq = rng.randint(1, 5)
    amp = rng.uniform(*amp_range)
    phase = rng.uniform(0, 2 * np.pi)
    perturbed = h + amp * np.sin(freq * np.pi * x + phase)
    return project_feasible(perturbed, n)

def binarize_push(h, n, threshold=0.03):
    h = h.copy()
    mask_low = h < threshold
    mask_high = h > 1.0 - threshold
    h[mask_low] = 0.0
    h[mask_high] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        x = np.linspace(0, 2, 700)
        h_init = 0.5 + 0.25 * np.sin(np.pi * x) + 0.08 * np.sin(3 * np.pi * x)
        h_init = project_feasible(np.clip(h_init, 0, 1), len(h_init))
        n_init = len(h_init)

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    # Phase 1: CD at original resolution
    h1, c1 = fast_cd(best_h, n_init, 40, deadline, rng, True)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Try multiple resolutions
    n_candidates = [300, 400, 500, 600, 700, 800, 900]
    x_best = np.linspace(0, 2, len(best_h))
    for n_try in n_candidates:
        if time.time() > deadline - 120:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_try = project_feasible(np.interp(x_new, x_best, best_h), n_try)
        h_opt, c_opt = fast_cd(h_try, n_try, 35, deadline, rng, True)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_best = x_new

    # Phase 3: Fourier projection + CD (different starting landscape)
    if time.time() < deadline - 150:
        n_f = len(best_h)
        h_f = fourier_project(best_h, n_f, J=4)
        h_f2, c_f2 = fast_cd(h_f, n_f, 30, deadline, rng, True)
        if c_f2 < best_c:
            best_h, best_c = h_f2, c_f

    # Phase 4: Multi-restart with smooth perturbations
    n_best = len(best_h)
    for restart in range(30):
        if time.time() > deadline - 60:
            break
        h_pert = smooth_restart(best_h, n_best, rng)
        h_opt, c_opt = fast_cd(h_pert, n_best, 25, deadline, rng, True)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 5: Binarization push + CD
    if time.time() < deadline - 80:
        for thresh in [0.02, 0.05, 0.1]:
            if time.time() > deadline - 40:
                break
            h_bin = binarize_push(best_h, n_best, thresh)
            h_opt, c_opt = fast_cd(h_bin, n_best, 20, deadline, rng, True)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

    # Phase 6: Final refinement (deterministic order for convergence)
    remaining = deadline - time.time()
    if remaining > 10:
        max_p = max(15, int(remaining / 1.0))
        h_final, c_final = fast_cd(best_h, n_best, max_p, deadline, rng, False)
        if c_final < best_c:
            best_h, best_c = h_final, c_final

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
