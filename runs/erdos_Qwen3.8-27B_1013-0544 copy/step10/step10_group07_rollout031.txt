    weights = np.exp(shifted)
    weights /= weights.sum()

        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d_fast(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(400, 0.5)
        n = 400
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: CD warmup
    h1, c1 = exact_cd(h_best, n, 80, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Momentum projected gradient
    if time.time() < deadline - 300:
        h2, c2 = projected_momentum_grad(h_best, n, [300, 1000, 5000, 20000, 80000], 150, deadline)
        if c2 < best_c:
            h_best, best_c = h2, c2

    # Phase 3: CD polish
    if time.time() < deadline - 150:
        h3, c3 = exact_cd(h_best, n, 400, deadline)
        if c3 < best_c:
            h_best, best_c = h3, c3

    # Phase 4: Multi-n exploration
    n_list = [500, 600, 800, 1000, 1200]
    for n_try in n_list:
        if time.time() > deadline - 200:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 200, deadline)
        if time.time() < deadline - 100:
            h_o, c_o2 = projected_momentum_grad(h_o, n_try, [1000, 5000, 20000], 80, deadline)
            if c_o2 < c_o:
                h_o, c_o = h_o, c_o2
        if c_o < best_c - 1e-8:
            h_best, best_c, n = h_o, c_o, n_try

    # Phase 5: Diverse shake and settle
    for r in range(20):
        if time.time() > deadline - 50:
            break
        s = 0.002 + 0.003 * r
        hp = project_feasible(h_best + rng.normal(0, s, len(h_best)), len(h_best))
        h_o, c_o = exact_cd(hp, len(h_best), 150, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 6: Final CD polish
    if time.time() < deadline - 5:
        h_f, c_f = exact_cd(h_best, len(h_best), 300, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
