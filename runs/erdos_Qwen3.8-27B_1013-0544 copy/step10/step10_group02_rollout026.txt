from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    if m_min > m_max:
        return h0
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    L_lo, L_hi = unaff, cur_max
    for _ in range(60):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        neg = a < -1e-15
        pos = a > 1e-15
        zero = np.abs(a) <= 1e-15
        if np.any(zero) and np.any(c_base[zero] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg):
            x_lo = max(x_lo, float(np.max((L - c_base[neg]) / a[neg])))
        if feasible and np.any(pos):
            x_hi = min(x_hi, float(np.min((L - c_base[pos]) / a[pos])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (x_lo < r1 and x_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    neg = a < -1e-15
    pos = a > 1e-15
    if np.any(neg):
        x_lo = max(x_lo, float(np.max((L - c_base[neg]) / a[neg])))
    if np.any(pos):
        x_hi = min(x_hi, float(np.min((L - c_base[pos]) / a[pos])))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if has_diag:

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c[ms] += delta * a
    diag_i = (ms == (n - 1))
    if np.any(diag_i):
        c[ms[diag_i]] -= delta * delta

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = list(range(n))
        if rng is not None and it % 2 == 1:
            rng.shuffle(order)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def pair_cd(h, n, max_passes, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Pick random pairs
        for _ in range(n // 2):
            p = rng.randint(n)
            q = rng.randint(n)
            if p == q:
                continue
            s = h[p] + h[q]
            if s < 1e-12 or s > 2.0 - 1e-12:
                continue
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            if t_hi - t_lo < 1e-12:
                continue
            # Try a grid of t values
            best_t = h[p]
            best_val = float('inf')
            ts = np.linspace(t_lo, t_hi, 17)
            for t in ts:
                h[p] = t
                h[q] = s - t
                mx = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                if mx < best_val:
                    best_val = mx
                    best_t = t
            h[p] = best_t
            h[q] = s - best_t
            if abs(best_t - h[p]) > 1e-10 or abs((s - best_t) - h[q]) > 1e-10:
                improved = True
                c = np.correlate(h, 1.0 - h, mode='full')
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def pair_coeffs(h, p, q, n):
    """Compute quadratic coefficients for c(m) change when h[p]=t, h[q]=s-t."""
    s = h[p] + h[q]
    # For each m, compute alpha*m (t^2 coef), beta*m (t coef), gamma*m (const)
    # such that delta_c(m) = alpha*m * t^2 + beta*m * t + gamma*m
    nms = 2 * n - 1
    alphas = np.zeros(nms)
    betas = np.zeros(nms)
    gammas = np.zeros(nms)
    
    for m in range(nms):
        shift = m - (n - 1)
        terms = set()
        
        # Term (i=p, j=p+shift) if valid
        if 0 <= p + shift < n:
            j = p + shift
            terms.add(('fwd_p', j))
            if j == q:
                # t*(1-(s-t)) - hp*(1-hq) = t^2 - st + t - hp + hp*hq
                alphas[m] += 1.0
                betas[m] += 1.0 - s
                gammas[m] += -h[p] * (1.0 - h[q])
            else:
                betas[m] += (1.0 - h[j])
                gammas[m] += -h[p] * (1.0 - h[j])
        
        # Term (i=p-shift, j=p) if valid
        if 0 <= p - shift < n:
            i = p - shift
            # Check if this is already counted as (i=q, j=q+shift) i.e. q = p-shift and q+shift = p
            # That would mean the same term
            if i == q and (0 <= i + shift < n) and (i + shift == p):
                # Same term as fwd_q, skip here to avoid double count
                pass
            else:
                # h[i]*(1-h[p]) -> h[i]*(1-t): change = h[i]*(h[p]-t)
                betas[m] += -h[i]
                gammas[m] += h[i] * h[p]
        
        # Term (i=q, j=q+shift) if valid
        if 0 <= q + shift < n:
            j = q + shift

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h; tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p; m_max = 2*n - 2 - p
    if m_min > m_max: return h0
    ms = np.arange(m_min, m_max + 1)

    a = np.zeros(len(ms))
    if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb): a[vb] -= h[jb[vb]]

    unaff = 0.0

    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    neg = a < -1e-15; pos = a > 1e-15

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14: return
    m_min = n-1-p; m_max = 2*n-2-p
    ms = np.arange(m_min, m_max+1)
    jf = p + ms - (n-1); jb = p - ms + (n-1)
    vf = (jf >= 0) & (jf < n); vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb): a[vb] -= h[jb[vb]]
    c[ms] += delta * a
    di = (ms == (n-1))
    if np.any(di): c[ms[di]] -= delta*delta

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0-h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline: break
        improved = False
        order = list(range(n))
        if rng is not None and it % 2 == 1: rng.shuffle(order)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0-1e-13: continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn; improved = True
        if not improved: break
        h = project_feasible(h, n)
        if abs(h.sum() - n/2.0) > 1e-10:
            c = np.correlate(h, 1.0-h, mode='full')
    return h, compute_c5(h, n)

def pair_cd(h, n, max_passes, deadline, rng, n_pairs=None):
    h = h.copy()
    c = np.correlate(h, 1.0-h, mode='full')
    if n_pairs is None: n_pairs = n // 3
    nms = 2*n - 1

        improved = False

            if p == q: continue

            s = hp + hq
            if s < 1e-12 or s > 2.0-1e-12: continue

            if t_hi - t_lo < 1e-6: continue

            for m in range(nms):

                seen = set()
                for i,j in [(p, p+sh), (p-sh, p), (q, q+sh), (q-sh, q)]:
                    if (i,j) in seen: continue
                    seen.add((i,j))
                    if not (0 <= i < n and 0 <= j < n or i < 0 or j < 0):
                        pass
                    if 0 <= i < n and 0 <= j < n and i != j:
                        pass
                    if 0 <= i < n and 0 <= j < n:
                        if i == p and j == q:

                        elif i == q and j == p:

                        elif i == p:

                        elif j == p:

                        elif i == q:

            ts = np.linspace(t_lo, t_hi, 21)

                c = np.correlate(h, 1.0-h, mode='full')
                improved = True
        if not improved: break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_lse(x, n, alpha):
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max(); shifted = alpha*(c-m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted-mx)))
    return (m + lse/alpha)*(2.0/n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline: break
        try:
            res = minimize(smooth_lse, best_h, args=(n, alpha), method='SLSQP',
                           bounds=[(0.0,1.0)]*n,
                           constraints=[{'type':'eq','fun':lambda x: x.sum()-n/2.0,
                                         'jac':lambda x: np.ones(n)}],
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12: best_h, best_c = new_h, new_c
        except Exception: pass
    return best_h, best_c

    x_new = np.linspace(0, 2, n_new, endpoint=False)
