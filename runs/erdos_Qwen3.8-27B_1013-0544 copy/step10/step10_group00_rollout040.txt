import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def lse_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = alpha * (c - m)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * dx

def lse_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    w = np.exp(alpha * (c - m))
    w = w / w.sum()
    grad = np.full(n, 1.0)
    for k in range(len(c)):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        j_lo = max(0, s)
        j_hi = min(n, n + s)
        if j_lo < j_hi:
            grad[j_lo:j_hi] -= w[k] * h[j_lo - s:j_hi - s]
        j_lo2 = max(0, -s)
        j_hi2 = min(n, n - s)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= w[k] * h[j_lo2 + s:j_hi2 + s]
    return grad * dx

def coord_descent(h, n, max_passes, deadline, gs_iters=45):

    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            s_vals = ms - (n - 1)
            a = np.empty(len(ms))
            has_diag = False
            diag_idx = -1
            for idx in range(len(ms)):
                sv = s_vals[idx]
                if sv == 0:
                    a[idx] = 1.0 - 2.0 * h0
                    has_diag = True
                    diag_idx = idx
                else:
                    val = 1.0
                    if 0 <= p - sv < n:
                        val -= h[p - sv]
                    if 0 <= p + sv < n:
                        val -= h[p + sv]
                    a[idx] = val
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)

            def fval(d):
                vals = c[ms] + d * a
                if has_diag:
                    vals = vals.copy()
                    vals[diag_idx] -= d * d
                return max(float(np.max(vals)), unaff_max)

            f1 = fval(d1)
            f2 = fval(d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-14:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = fval(d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = fval(d2)
            d_opt = (d_lo + d_hi) * 0.5
            if fval(d_opt) < fval(0.0) - 1e-13:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
    return h, best_c

        h1, c1 = proj_grad_phase(best_h, n, deadline - 450,
                                 alphas=[100, 300, 1000, 5000], its_per_alpha=250)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Block SLSQP
    if time.time() < deadline - 350:
        h2, c2 = block_slsqp(best_h, n, block_size=35, max_passes=8, deadline=deadline - 350)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Coordinate descent
    if time.time() < deadline - 220:
        h3, c3 = coord_descent(best_h, n, 70, deadline - 220, gs_iters=50)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Multi-resolution
    x_cur = np.linspace(0, 2, len(best_h))
    for n_try in [400, 500, 900, 1200]:
        if time.time() > deadline - 160:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_cur, best_h), n_try)
        h_o, c_o = coord_descent(h_int, n_try, 50, deadline - 160, gs_iters=45)
        if c_o < best_c:
            h_o, c_o = proj_grad_phase(h_o, n_try, deadline - 160,
                                       alphas=[200, 1000, 5000], its_per_alpha=100)
            h_o, c_o = coord_descent(h_o, n_try, 40, deadline - 160, gs_iters=50)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
                x_cur = np.linspace(0, 2, n_try)

    # Phase 5: Multi-restart
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        remaining = max(10, deadline - time.time() - 80)
        n_restarts = max(4, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 80:
                break
            scale = 0.003 + 0.025 * (r / max(1, n_restarts))
            if r % 4 == 0:
                hp = project_feasible(best_h + rng.normal(0, scale, n_cur), n_cur)
            elif r % 4 == 1:
                x = np.linspace(0, 2, n_cur)
                hp = project_feasible(best_h + scale * np.sin(2 * np.pi * (3 + r) * x), n_cur)
            elif r % 4 == 2:
                hp = project_feasible(best_h + 0.01 * scale * rng.randn(n_cur), n_cur)
            else:
                blk = max(10, n_cur // 10)
                hp = best_h.copy()
                st = (r * blk) % max(1, n_cur - blk)
                hp[st:st + blk] = 1.0 - hp[st:st + blk]
                hp = project_feasible(hp, n_cur)
            h_o, c_o = coord_descent(hp, n_cur, 30, deadline - 80, gs_iters=40)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
            h_o, c_o = coord_descent(best_h, n_cur, 20, deadline - 80, gs_iters=50)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 6: Final polish
    if time.time() < deadline - 10:
        h_f, c_f = coord_descent(best_h, len(best_h), 60, deadline - 10, gs_iters=60)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
