    h=base.copy()
    v=variant%6
    if v==0: h += 0.02*rng.normal(size=n)
    elif v==1: h += 0.05*rng.normal(size=n)
    elif v==2:

def perturb_h(base,n,rng,variant):
    h=np.array(base,dtype=float).copy()
    v=int(variant) % 6
    if v==0: h += 0.015*rng.normal(size=n)
    elif v==1: h += 0.04*rng.normal(size=n)
    elif v==2:
       x=np.linspace(0.0,2.0,n)
       h += 0.04*np.sin(np.pi*(2+v)*x + rng.uniform(0.0,2*np.pi))
    elif v==3:
       h = 0.5*(h + h[::-1]) + 0.02*rng.normal(size=n)
    elif v==4:
       h = 0.5*(h - h[::-1]) + 0.5 + 0.02*rng.normal(size=n)
    else:
       bs=max(1,n//8)
       if n>bs+1:
          s=int(rng.randint(0,n-bs))
          h[s:s+bs] = 1.0 - h[s:s+bs]
       h += 0.01*rng.normal(size=n)
    return project_exact(h,0.5*float(n))

import numpy as np
import time
from scipy.optimize import minimize


def project_exact(y, target):
    y = np.asarray(y, dtype=float)
    if y.size == 0:
        return y
    low = -float(np.max(y)) - 1.0
    high = 1.0 - float(np.min(y))
    for _ in range(50):
        mid = 0.5 * (low + high)
        s = float(np.clip(y + mid, 0.0, 1.0).sum())
        if s < target:
            low = mid
        else:
            high = mid
    p = 0.5 * (low + high)
    return np.clip(y + p, 0.0, 1.0)


def exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def make_fft_cache(n):
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    R = np.fft.rfft(np.ones(n), L)
    return L, R


def fft_convolve(h, n, L, R):
    H = np.fft.rfft(h, L)
    Hr = np.fft.rfft(h[::-1], L)
    c = np.fft.irfft(H * (R - Hr), L)
    return c[:2 * n - 1], H


def weighted_gradient(h, w, n, L, H):
    G = np.zeros(L, dtype=float)
    G[:n] = w[n - 1::-1]
    if n > 1:
        G[L - (n - 1):] = w[2 * n - 2:n - 1:-1]
    pref = np.empty(2 * n, dtype=float)
    pref[0] = 0.0
    pref[1:] = np.cumsum(w)
    P = pref[n:2 * n] - pref[:n]
    Gf = np.fft.rfft(G, L)
    T2 = np.fft.irfft(Gf * H, L)[:n]
    T1 = np.fft.irfft(np.conjugate(Gf) * H, L)[:n]
    return P - T1 - T2


def _smooth_fgrad(x, n, alpha, rho, L, R):
    h = np.clip(x, 0.0, 1.0)
    c, H = fft_convolve(h, n, L, R)
    cmax = float(np.max(c))
    z = alpha * (c - cmax)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        f = cmax
        w = np.zeros(2 * n - 1, dtype=float)
        w[int(np.argmax(c))] = 1.0
    else:
        f = cmax + np.log(s) / alpha
        w = e / s
    g = weighted_gradient(h, w, n, L, H)
    if rho != 0.0:
        diff = float(np.sum(h) - 0.5 * n)
        f += rho * diff * diff
        g = g + (2.0 * rho * diff)
    return f, g


def _cd_max(c_arr, a_arr, diag_base, diag_a, has_diag, unaff, d):
    if c_arr.size:
        m = float(np.max(c_arr + d * a_arr))
    else:
        m = 0.0
    if has_diag:
        dv = diag_base + d * diag_a - d * d
        if dv > m:
            m = dv
    if unaff > m:
        m = unaff
    return m


def coord_descent(h_in, n, deadline, max_passes, gs_iters, best_c, force_all=False):
    target = 0.5 * float(n)
    h = project_exact(np.asarray(h_in, dtype=float), target)
    c_cur = exact_c5(h)
    if c_cur < best_c:
        best_c = c_cur
    best_h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    no_imp = 0
    for _ in range(int(max_passes)):
        if time.time() > deadline:
            break
        h = project_exact(h, target)
        c = np.correlate(h, 1.0 - h, mode="full")
        n1 = n - 1
        if force_all:
            min_count = 1
        else:
            min_count = int(np.ceil(best_c * n / 2.0 - 1e-12))
            if min_count < 1:
                min_count = 1
            if min_count > n:
                min_count = n
        s_max = n - min_count
        r_lo = n1 - s_max
        r_hi = n1 + s_max
        if r_lo < 0:
            r_lo = 0
        if r_hi > 2 * n - 2:
            r_hi = 2 * n - 2
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = float(h[p])
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            m_lo = max(n1 - p, r_lo)
            m_hi = min(n1 + p, r_hi)
            if m_lo > m_hi:
                continue
            ms = np.arange(m_lo, m_hi + 1)
            s = ms - n1
            jf = p - s
            jb = p + s
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms), dtype=float)
            if np.any(vf):
                a[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > r_lo:
                u = c[r_lo:m_lo]
                if u.size:
                    unaff = max(unaff, float(np.max(u)))
            if m_hi < r_hi:
                u = c[m_hi + 1:r_hi + 1]
                if u.size:
                    unaff = max(unaff, float(np.max(u)))
            di = n1 - m_lo if (n1 >= m_lo and n1 <= m_hi) else -1
            if di >= 0:
                if di == 0:
                    c_nd, a_nd = c_aff[1:], a[1:]
                elif di == c_aff.size - 1:
                    c_nd, a_nd = c_aff[:-1], a[:-1]
                else:
                    c_nd = np.concatenate((c_aff[:di], c_aff[di + 1:]))
                    a_nd = np.concatenate((a[:di], a[di + 1:]))
                diag_base, diag_a, has_diag = float(c_aff[di]), float(a[di]), True
            else:
                c_nd, a_nd, diag_base, diag_a, has_diag = c_aff, a, 0.0, 0.0, False
            f_curr = _cd_max(c_nd, a_nd, diag_base, diag_a, has_diag, unaff, 0.0)
            d_lo, d_hi = -h0, 1.0 - h0
            if d_hi - d_lo < 1e-15:
                continue
            gpts = np.linspace(d_lo, d_hi, 15)
            fvals = np.empty(15, dtype=float)
            best_d, best_f, best_i = 0.0, f_curr, 0
            for i, dd in enumerate(gpts):
                fvals[i] = _cd_max(c_nd, a_nd, diag_base, diag_a, has_diag, unaff, dd)
                if fvals[i] < best_f:
                    best_f, best_d, best_i = fvals[i], float(dd), i
            lo_r = float(gpts[max(0, best_i - 1)])
            hi_r = float(gpts[min(14, best_i + 1)])
            if hi_r > lo_r + 1e-16:
                c1 = lo_r + gr * (hi_r - lo_r)
                c2 = hi_r - gr * (hi_r - lo_r)
                f1 = _cd_max(c_nd, a_nd, diag_base, diag_a, has_diag, unaff, c1)
                f2 = _cd_max(c_nd, a_nd, diag_base, diag_a, has_diag, unaff, c2)
                for _ in range(int(gs_iters)):
                    if hi_r - lo_r < 1e-15:
                        break
                    if f1 < f2:
                        hi_r, c2, f2 = c2, c1, f1
                        c1 = lo_r + gr * (hi_r - lo_r)
                        f1 = _cd_max(c_nd, a_nd, diag_base, diag_a, has_diag, unaff, c1)
                    else:
                        lo_r, c1, f1 = c1, c2, f2
                        c2 = lo_r + gr * (hi_r - lo_r)
                        f2 = _cd_max(c_nd, a_nd, diag_base, diag_a, has_diag, unaff, c2)
                dm = 0.5 * (lo_r + hi_r)
                fm = _cd_max(c_nd, a_nd, diag_base, diag_a, has_diag, unaff, dm)
                if fm < best_f:
                    best_f, best_d = fm, dm
            if best_f < f_curr - 1e-12:
                d = float(best_d)
                h[p] = float(np.clip(h0 + d, 0.0, 1.0))
                inc = d * a
                if di >= 0:
                    inc[di] -= d * d
                c[ms] += inc
        h = project_exact(h, target)
        c_cur = exact_c5(h)
        if c_cur < best_c - 1e-13:
            best_c = c_cur
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 2 and c_cur > best_c:
                h = best_h.copy()
            no_imp = 0
    best_h = project_exact(best_h, target)
    cb = exact_c5(best_h)
    if cb < best_c:
        best_c = cb
    return best_h, best_c


def optimize_at(h_in, n, deadline, alphas, it_each, cd_passes, gs_iters, best_c):
    target = 0.5 * float(n)
    h = project_exact(np.asarray(h_in, dtype=float), target)
    c_last = exact_c5(h)
    if c_last < best_c:
        best_c = c_last
    best_h = h.copy()
    start_h = h.copy()
    if alphas and time.time() < deadline - 80:
        L, R = make_fft_cache(n)
        bounds = [(0.0, 1.0)] * n
        reserve = 90 if cd_passes > 0 else 25
        for alpha in alphas:
            if time.time() > deadline - reserve:
                break
            try:
                res = minimize(
                    _smooth_fgrad, h, args=(n, float(alpha), 50.0, L, R),
                    method="L-BFGS-B", jac=True, bounds=bounds,
                    options={"maxiter": int(it_each), "ftol": 1e-12,
                             "gtol": 1e-6, "maxls": 25, "iprint": 0}
                )
                h = project_exact(res.x, target)
            except Exception:
                h = project_exact(h, target)
            c_last = exact_c5(h)
            if c_last < best_c:
                best_c = c_last
                best_h = h.copy()
                start_h = h.copy()
            elif c_last <= best_c + 1e-8:
                start_h = h.copy()
            else:
                start_h = best_h.copy()
        h = start_h
    if cd_passes > 0 and time.time() < deadline - 30:
        h2, c2 = coord_descent(h, n, deadline - 30, int(cd_passes), int(gs_iters), best_c)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    return best_h, best_c, n


def interp_to(h, n):
    m = len(h)
    if m == n:
        return project_exact(np.asarray(h, dtype=float), 0.5 * float(n))
    x_old = np.linspace(0.0, 2.0, m)
    x_new = np.linspace(0.0, 2.0, n)
    y = np.interp(x_new, x_old, np.asarray(h, dtype=float))
    return project_exact(y, 0.5 * float(n))


def perturb_h(base, n, rng, variant):
    h = np.array(base, dtype=float).copy()
    v = int(variant) % 6
    if v == 0:
        h += 0.015 * rng.normal(size=n)
    elif v == 1:
        h += 0.04 * rng.normal(size=n)
    elif v == 2:
        x = np.linspace(0.0, 2.0, n)
        h += 0.04 * np.sin(np.pi * (2 + v) * x + rng.uniform(0.0, 2.0 * np.pi))
    elif v == 3:
        h = 0.5 * (h + h[::-1]) + 0.02 * rng.normal(size=n)
    elif v == 4:
        h = 0.5 * (h - h[::-1]) + 0.5 + 0.02 * rng.normal(size=n)
    else:
        bs = max(1, n // 8)
        if n > bs + 1:
            s = int(rng.randint(0, n - bs))
            h[s:s + bs] = 1.0 - h[s:s + bs]
        h += 0.01 * rng.normal(size=n)
    return project_exact(h, 0.5 * float(n))


def finalize_area(h):
    target = 0.5 * float(len(h))
    h = project_exact(h, target)
    diff = target - float(h.sum())
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
        diff2 = target - float(h.sum())
        if abs(diff2) > 1e-14:
            for i in range(len(h)):
                if 0.0 < h[i] < 1.0:
                    h[i] = float(np.clip(h[i] + diff2, 0.0, 1.0))
                    break
    return h

    if time.time() < deadline - 170 and n_try != int(len(best_h)):

        hb, cb, nb = optimize_at(interp_to(best_h, n_try), n_try, ph, alphas_main[:4], 45, 14, 28, best_c)
        if cb < best_c:
            best_h, best_c, nb = hb, cb, nb
    n_cur = int(len(best_h))
    if time.time() < deadline - 140:
        n_pol = 700 if n_cur <= 500 else 1000
        if n_pol != n_cur:
            ph = min(deadline, time.time() + 150.0)
            hb, cb, nb = optimize_at(interp_to(best_h, n_pol), n_pol, ph, alphas_main[1:], 25, 7, 28, best_c)
            if cb < best_c:
                best_h, best_c, nb = hb, cb, nb
    al_short = [alphas_main[-2], alphas_main[-1]]
    for r in range(6):
        if time.time() > deadline - 80:
            break
        n_cur = int(len(best_h))
        hp = perturb_h(best_h, n_cur, rng, r)
        dur = min(120.0, max(40.0, deadline - 80.0 - time.time()))
        cdp = 4 if dur > 90 else 2
        hb, cb, nb = optimize_at(hp, n_cur, time.time() + dur, al_short, 25, cdp, 25, best_c)
        if cb < best_c:
            best_h, best_c, nb = hb, cb, nb
    if time.time() < deadline - 5:
        n_cur = int(len(best_h))
        hb, cb = coord_descent(best_h, n_cur, deadline, 2, 32, best_c, True)
        if cb < best_c:
            best_h, best_c = hb, cb
    best_h = finalize_area(best_h)
    best_c = exact_c5(best_h)
    return best_h, float(best_c), int(len(best_h))
