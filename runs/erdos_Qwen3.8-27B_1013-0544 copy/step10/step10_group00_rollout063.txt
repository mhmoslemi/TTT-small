import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def lse_objective(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(beta * (c - m))
    s = e.sum()
    return (m + np.log(s) / beta) * dx

def lse_gradient(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(beta * (c - m))
    p = e / e.sum()
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        w = p[n-1+s]
        if w < 1e-12:
            continue
        if s >= 0:
            if n - s > 0:
                grad[:n-s] += w * (1.0 - h[s:])
                grad[s:] -= w * h[:n-s]
        else:
            t = -s
            if n - t > 0:
                grad[t:] += w * (1.0 - h[:n-t])
                grad[:n-t] -= w * h[t:]
    return grad * dx

def lse_obj_grad(x, n, beta):
    return lse_objective(x, n, beta), lse_gradient(x, n, beta)

def lbfgs_smooth(h, n, beta, max_iter=200, deadline=None):
    """Optimize smoothed objective with L-BFGS-B + projection."""
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c = compute_c5(h_cur, n)
    
    def fun(x):
        v, g = lse_obj_grad(x, n, beta)
        return v
    
    def fun_and_grad(x):
        return lse_obj_grad(x, n, beta)
    
    bounds = [(0.0, 1.0)] * n
    best_lbfgs_c = lse_objective(h_cur, n, beta)
    
    for attempt in range(3):
        if deadline and time.time() > deadline:
            break
        res = minimize(fun_and_grad, h_cur, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
        h_cur = h_new
        if res.success and res.status == 0:
            break
    
    return best_h, best_c

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1; d1 = d2; f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def simulated_annealing(h, n, deadline, n_restarts=20, cd_passes=15):

    rng = np.random.RandomState(1234)
    temp = 0.005
    cool = 0.97
    x = np.linspace(0, 2, n)
    
    for it in range(5000):
        if time.time() > deadline:
            break
        # Propose perturbation
        r = rng.random()
        if r < 0.3:

            phase = rng.uniform(0, 2*np.pi)
            amp = temp * 2
            hp = h + amp * np.sin(freq * np.pi * x + phase)
        elif r < 0.5:

            block_len = rng.randint(n//20, n//5)
            start = rng.randint(0, n - block_len)
            hp = h.copy()
            hp[start:start+block_len] = 1.0 - hp[start:start+block_len]
        elif r < 0.7:

        elif r < 0.85:
            # Block shift

            start = rng.randint(0, n - block_len)
            shift = rng.randint(-5, 6)
            hp = h.copy()
            segment = hp[start:start+block_len].copy()
            hp[start:start+block_len] = np.roll(segment, shift)
        else:
            # Swap blocks
            bl1 = rng.randint(n//20, n//5)
            bl2 = rng.randint(n//20, n//5)
            s1 = rng.randint(0, n - bl1)
            s2 = rng.randint(0, n - bl2)
            if abs(s1-s2) > max(bl1, bl2):
                hp = h.copy()
                hp[s1:s1+bl1], hp[s2:s2+bl2] = h[s2:s2+bl2].copy(), h[s1:s1+bl1].copy()
            else:
                hp = h + rng.normal(0, temp, n)
        
        hp = project_feasible(hp, n)
        c_new = compute_c5(hp, n)
        c_cur = compute_c5(h, n)
        
        if c_new < c_cur or rng.random() < np.exp(-(c_new - c_cur) / max(temp, 1e-10)):
            h = hp
            if c_new < best_c:
                best_h, best_c = h.copy(), c_new
        
        temp *= cool
        
        # Periodic CD refinement
        if (it + 1) % 100 == 0:
            h_cd, c_cd = coord_descent(h, n, cd_passes, min(deadline, time.time() + 60))
            if c_cd < best_c:

            h = h_cd
            temp = max(temp, 0.002)
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            dp = np.where(ms == (n - 1))[0]
            has_d = len(dp) > 0
            di = int(dp[0]) if has_d else -1
            c_aff = c[ms]
            um = 0.0
            if m_lo > 0:
                um = max(um, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                um = max(um, float(c[m_hi + 1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, di, has_d, um, h0, d1)
            f2 = _cd_fval(c_aff, a, di, has_d, um, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, di, has_d, um, h0, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, di, has_d, um, h0, d2)

            if f_opt < f_cur - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_obj_grad(x, n, beta, rho):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(beta * (c - m))
    p = e / e.sum()

                grad[t:] += w * (1.0 - x[:n - t])
                grad[:n - t] -= x[t:]
    grad *= dx
    grad += 2.0 * rho * (x.sum() - n / 2.0)
    return val, grad
