        def block_obj(xb):
            hx = best_h.copy()
            hx[idx] = xb
            return compute_c5(hx, n)
        def block_eq(xb):
            return xb.sum() - remaining_sum
        def block_eq_jac(xb):
            return np.ones(block_size)
        cons = [{'type': 'eq', 'fun': block_eq, 'jac': block_eq_jac}]
        try:
            res = minimize(block_obj, x0, method='SLSQP', bounds=bounds_b,
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-14})
            if res.x is not None and res.fun < best_c:
                best_h[idx] = res.x
                best_h = project_feasible(best_h, n)
                best_c = compute_c5(best_h, n)
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP warm-up
    if time.time() < deadline - 400:
        h1, c1 = slsqp_opt(best_h, n, [300, 2000, 10000], 400, deadline)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Heavy coordinate descent
    if time.time() < deadline - 300:
        h2, c2 = coord_descent(best_h, n, 200, deadline, gs_iters=70, randomize=False, rng=rng)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Multi-resolution global search
    if time.time() < deadline - 250:
        x_f = np.linspace(0, 2, n)
        for n_try in [120, 200, 350]:
            if time.time() > deadline - 120:
                break
            if n_try >= n:
                continue
            x_s = np.linspace(0, 2, n_try)
            h_s = project_feasible(np.interp(x_s, x_f, best_h), n_try)
            h_s, _ = slsqp_opt(h_s, n_try, [1000, 10000], 300, deadline)
            h_s, _ = coord_descent(h_s, n_try, 100, deadline, gs_iters=65)
            h_back = project_feasible(np.interp(x_f, x_s, h_s), n)
            h_back, c_back = coord_descent(h_back, n, 80, deadline, gs_iters=70)
            if c_back < best_c:
                best_h, best_c = h_back, c_back

    # Phase 4: Block optimization
    if time.time() < deadline - 200:
        h4, c4 = block_optimize(best_h, n, 4, deadline, max_blocks=30)
        if c4 < best_c:
            best_h, best_c = h4, c4
        h4, c4 = block_optimize(best_h, n, 3, deadline, max_blocks=40)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Multi-restart with diverse strategies
    if time.time() < deadline - 100:
        c_prof = np.correlate(best_h, 1.0 - best_h, mode='full')
        n_restarts = max(3, int((deadline - time.time() - 60) / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            if r % 5 == 0:
                hp = project_feasible(best_h + rng.normal(0, 0.012, n), n)
            elif r % 5 == 1:
                freq = 2.0 + 1.5 * (r % 3)
                hp = project_feasible(best_h + 0.02 * np.sin(np.pi * freq * np.linspace(0, 2, n)), n)
            elif r % 5 == 2:
                k_active = int(np.argmax(c_prof)) - (n - 1)
                perturb = np.zeros(n)
                for p in range(n):
                    pf, pb = p + k_active, p - k_active
                    if 0 <= pf < n:
                        perturb[p] += (1.0 - best_h[pf])
                    if 0 <= pb < n:
                        perturb[p] -= best_h[pb]
                norm_p = np.max(np.abs(perturb))
                if norm_p > 1e-10:
                    hp = project_feasible(best_h - 0.008 * perturb / norm_p, n)
                else:
                    hp = project_feasible(best_h + rng.normal(0, 0.01, n), n)
            elif r % 5 == 3:
                mask = (rng.rand(n) < 0.08).astype(float)
                hp = project_feasible(best_h + 0.25 * mask - 0.12, n)
            else:
                hp = project_feasible(best_h * (1.0 + 0.015 * np.sin(np.pi * 2 * np.linspace(0, 2, n))), n)

            h_o, c_o = coord_descent(hp, n, 50, deadline, gs_iters=60, randomize=True, rng=rng)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
                c_prof = np.correlate(best_h, 1.0 - best_h, mode='full')

    # Phase 6: Interleaved SLSQP + CD polish
    if time.time() < deadline - 30:
        h6, c6 = slsqp_opt(best_h, n, [50000, 200000], 600, deadline)
        if c6 < best_c:
            best_h, best_c = h6, c6
        h6, c6 = coord_descent(best_h, n, 150, deadline, gs_iters=75)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Phase 7: Final precise CD
    if time.time() < deadline - 3:
        h7, c7 = coord_descent(best_h, n, 300, deadline, gs_iters=80)
        if c7 < best_c:
            best_h, best_c = h7, c7

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
