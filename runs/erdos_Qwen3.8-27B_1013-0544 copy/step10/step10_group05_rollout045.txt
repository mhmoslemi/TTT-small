    lse = mx + np.log(np.sum(np.exp(shifted - mx)))

            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            c_diag = c_base[diag_idx[0]]
            a_d = a[diag_idx[0]]
            disc = a_d * a_d + 4.0 * (c_diag - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (a_d - sq) / 2.0
                r2 = (a_d + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a_d = a[diag_idx[0]]
        disc = a_d * a_d + 4.0 * (c_diag - L)
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (a_d - sq) / 2.0
            r2 = (a_d + sq) / 2.0
            if x_lo <= h0 + r1:
                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def update_c(c, h, p, xn, n):
    delta = xn - h[p]
    if abs(delta) < 1e-15:
        return delta
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c[ms] += delta * a
    diag_i = np.where(ms == (n - 1))[0]
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta
    h[p] = xn
    return delta

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                update_c(c, h, p, xn, n)
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def power_smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    c_pos = np.maximum(c, 1e-30)
    val = np.power(np.sum(np.power(c_pos, alpha)), 1.0 / alpha)
    return val * (2.0 / n)

def lse_smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _cons_fun(x):
    n = len(x)
    return x.sum() - n / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def slsqp_opt(h0, n, alphas, maxiter, deadline, use_power=False):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        func = power_smooth_obj if use_power else lse_smooth_obj
        try:
            res = minimize(func, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def random_pairs_cd(h, n, max_passes, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                update_c(c, h, p, xn, n)
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def dual_focus_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        c_max = c.max()
        threshold = c_max - 0.02 * c_max
        top_lags = np.where(c > threshold)[0]
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                update_c(c, h, p, xn, n)
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def interpolate_h(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(600, 0.5)
        n = 600
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: CD from parent
    h1, c1 = exact_cd(h_best, n, 150, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: SLSQP with log-sum-exp
    h2, c2 = slsqp_opt(h_best, n, [10000, 100000, 1000000, 10000000], 300, deadline)
    if c2 < best_c:
        h_best, best_c = h2, c2

    # Phase 3: CD again
    h3, c3 = exact_cd(h_best, n, 100, deadline)
    if c3 < best_c:
        h_best, best_c = h3, c1

    # Phase 4: Random-order CD
    h4, c4 = random_pairs_cd(h_best, n, 100, deadline, rng)
    if c4 < best_c:
        h_best, best_c = h4, c4

    # Phase 5: Multi-scale perturbation + CD + SLSQP
    for it in range(25):
        if time.time() > deadline - 60:
            break
        frac = 0.03 + 0.15 * (it / 25.0)
        scale = 0.01 + 0.08 * (it / 25.0)
        k = max(2, int(n * frac))
        idx = rng.choice(n, size=k, replace=False)
        hp = h_best.copy()
        hp[idx] += rng.normal(0, scale, k)
        hp = project_feasible(hp, n)
        h_o, c_o = exact_cd(hp, n, 80, deadline)
        h_o, c_o = slsqp_opt(h_o, n, [200000, 2000000], 150, deadline)
        h_o, c_o = exact_cd(h_o, n, 60, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 6: SLSQP with power smoothing (different landscape)
    if time.time() < deadline - 100:
        h_p, c_p = slsqp_opt(h_best, n, [64, 256, 1024, 4096], 200, deadline, use_power=True)
        h_p, c_p = exact_cd(h_p, n, 100, deadline)
        if c_p < best_c:
            h_best, best_c = h_p, c_p

    # Phase 7: Try nearby n values
    for n_try in [n - 5, n + 5, n - 15, n + 15]:
        if time.time() > deadline - 80:
            break
        if n_try < 100 or n_try > 1200 or n_try == n:
            continue
        h_int = interpolate_h(h_best, n, n_try)
        h_o, c_o = exact_cd(h_int, n_try, 100, deadline)
        h_o, c_o = slsqp_opt(h_o, n_try, [100000, 1000000], 150, deadline)
        h_o, c_o = exact_cd(h_o, n_try, 80, deadline)
        if c_o < best_c - 1e-8:
            h_best, best_c, n = h_o, c_o, n_try

    # Phase 8: Aggressive final polish
    if time.time() < deadline - 20:
        h_f, c_f = slsqp_opt(h_best, n, [5000000, 50000000], 400, deadline)
        h_f, c_f = exact_cd(h_f, n, 200, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    # Final: ensure feasibility
    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
