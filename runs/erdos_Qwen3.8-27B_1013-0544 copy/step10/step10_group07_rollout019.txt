import numpy as np
import time
from scipy.optimize import minimize


def corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5(h):
    n = len(h)
    return float(np.max(corr(h)) * (2.0 / n))


def proj(h, n):
    t = 0.5 * n
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(120):
        d = t - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-14:
                break
            h = np.clip(h + d * r / s, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + d * h / s, 0.0, 1.0)
    return h


def topidx(c, q):
    m = len(c)
    if q <= 0 or q >= m:
        return np.arange(m)
    return np.argpartition(c, -q)[-q:]


def actdata(c, q):
    idx = topidx(c, q)
    m = len(c)
    if len(idx) >= m:
        return idx, 0.0
    mk = np.ones(m, dtype=bool)
    mk[idx] = False
    return idx, (float(c[mk].max()) if mk.any() else 0.0)


def choose_x(h, c, idx, imax, n, p, C):
    x0 = h[p]
    k = idx - (n - 1)
    ca = c[idx]
    a = np.zeros(len(k), dtype=float)

    pp = p + k
    m1 = (pp >= 0) & (pp < n)
    a[m1] += 1.0 - h[pp[m1]]
    pm = p - k
    m2 = (pm >= 0) & (pm < n)
    a[m2] -= h[pm[m2]]

    d = (k == 0)
    nz = (np.abs(a) > 1e-14) & (~d)
    ind = (~nz) & (~d)

    lo = float(imax)
    if ind.any():
        v = float(ca[ind].max())
        if v > lo:
            lo = v

    hd = False
    b0 = 0.0
    if d.any():
        b0 = float(ca[d][0]) - x0 + x0 * x0
        hd = True

    if not nz.any() and not hd:
        return float(x0)

    an = a[nz]
    bn = ca[nz] - an * x0
    pos = an > 1e-14
    neg = an < -1e-14
    hi = float(C)
    if hi < lo:
        hi = lo

    for _ in range(30):
        L = 0.5 * (lo + hi)
        xl, xh, ok = 0.0, 1.0, True
        if pos.any():
            xh = min(xh, float(np.min((L - bn[pos]) / an[pos])))
        if ok and neg.any():
            xl = max(xl, float(np.max((L - bn[neg]) / an[neg])))
        if ok and (xh < -1e-12 or xl > 1.0 + 1e-12 or xl > xh + 1e-12):
            ok = False
        if ok and hd:
            Cq = L - b0
            if Cq < -1e-12:
                ok = False
            elif Cq < 0.25 - 1e-12:
                dd = max(0.0, 1.0 - 4.0 * Cq)
                s2 = np.sqrt(dd)
                r1 = 0.5 * (1.0 - s2)
                r2 = 0.5 * (1.0 + s2)
                if xl > r1 + 1e-10 and xh < r2 - 1e-10:
                    ok = False
        if ok:
            hi = L
        else:
            lo = L

    L = hi
    xl, xh = 0.0, 1.0
    if pos.any():
        xh = min(xh, float(np.min((L - bn[pos]) / an[pos])))
    if neg.any():
        xl = max(xl, float(np.max((L - bn[neg]) / an[neg])))
    if xh < -1e-10 or xl > 1.0 + 1e-10 or xl > xh + 1e-8:
        return float(x0)

    x = float(np.clip(x0, xl, xh))
    if hd:

        if Cq < 0.25 - 1e-12:
            dd = max(0.0, 1.0 - 4.0 * Cq)
            s2 = np.sqrt(dd)
            r1 = 0.5 * (1.0 - s2)
            r2 = 0.5 * (1.0 + s2)

            bd = 1e100

            if best is None:
                return float(x0)
            x = best

        x = 0.0
    elif x > 1.0:
        x = 1.0
    return x


def trial(h, n, p, x):
    h2 = h.copy()
    h2[p] = x
    h2 = proj(h2, n)
    c2 = corr(h2)
    return h2, c2, float(c2.max())


def gpass(h, c, n, q, deadline):
    C = float(c.max())
    idx, imax = actdata(c, q)
    imp = 0
    since = 0
    for p in range(n):
        if time.time() > deadline:
            break
        if since >= 30:
            idx, imax = actdata(c, q)
            since = 0
        x0 = float(h[p])
        xs = choose_x(h, c, idx, imax, n, p, C)
        dx = xs - x0
        if abs(dx) <= 1e-13:
            continue
        h2, c2, Cn = trial(h, n, p, xs)
        if Cn < C - 1e-13:
            h, c, C = h2, c2, Cn
            imp += 1
            since += 1
        elif abs(dx) > 0.01:
            for tt in (0.5, 0.75):
                x = x0 + tt * dx
                if x < 0.0 or x > 1.0:
                    continue
                h2, c2, Cn = trial(h, n, p, x)
                if Cn < C - 1e-13:
                    h, c, C = h2, c2, Cn
                    imp += 1
                    since += 1
                    break
    return h, c, C, imp


def gcd(h, n, qs, ps, deadline):
    h = proj(h, n)
    c = corr(h)
    dx = 2.0 / n
    bc = float(c.max()) * dx
    for q in qs:
        for _ in range(ps):
            if time.time() > deadline:
                break
            h, c, C, im = gpass(h, c, n, q, deadline)
            if C * dx < bc:
                bc = C * dx
            if im == 0:
                break
    h = proj(h, n)
    return h, c5(h)


def sobj(x, n, b):
    c = corr(x)
    m = float(c.max())
    e = np.exp(b * (c - m))
    return m + float(np.log(e.sum()) / b)


def sgn(x, n, b):
    c = corr(x)
    m = float(c.max())
    w = np.exp(b * (c - m))
    w /= w.sum()
    g = np.zeros(n, dtype=float)
    for i in np.nonzero(w > 1e-12)[0]:
        ww = float(w[i])
        k = int(i) - (n - 1)
        if k == 0:
            g += ww * (1.0 - 2.0 * x)
        elif k > 0:
            g[:-k] += ww * (1.0 - x[k:])
            g[k:] -= ww * x[:-k]
        else:
            r = -k
            g[r:] += ww * (1.0 - x[:-r])
            g[:-r] -= ww * x[r:]
    return g


def pobj(x, n, b, r):
    s = float(x.sum() - 0.5 * n)
    return sobj(x, n, b) + 0.5 * r * s * s


def pgn(x, n, b, r):
    s = float(x.sum() - 0.5 * n)
    return sgn(x, n, b) + r * s


def minopt(h, n, betas, rho, mit, deadline):
    best = proj(h, n)
    bc = c5(best)
    bounds = [(0.0, 1.0)] * n
    for b in betas:
        if time.time() > deadline - 30:
            break
        try:
            res = minimize(pobj, best, args=(n, b, rho), jac=pgn,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': mit, 'ftol': 1e-15,
                                    'gtol': 1e-8, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if np.isfinite(x).all():
                x = proj(x, n)
                cv = c5(x)
                if cv < bc:
                    best, bc = x, cv
        except Exception:
            pass
    return best, bc


def resamp(h, n):
    o = len(h)
    xo = np.linspace(0.0, 2.0, o)
    xn = np.linspace(0.0, 2.0, n)
    return proj(np.interp(xn, xo, h), n)


def randbin(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n, dtype=float)
    h[:n // 2] = 1.0
    rng.shuffle(h)
    return proj(h, n)


def symtry(h, n):
    h = proj(h, n)
    c0 = c5(h)
    hs = proj(0.5 * (h + h[::-1]), n)
    c1 = c5(hs)
    if c1 < c0:
        return hs, c1
    return h, c0
