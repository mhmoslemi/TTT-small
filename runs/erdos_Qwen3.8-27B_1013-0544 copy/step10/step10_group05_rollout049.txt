    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    # Same as before - exact coordinate update
    h0 = h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:

    neg_mask = a < -1e-14
    pos_mask = a > 1e-14

    L_lo, L_hi = unaff, cur_max
    for _ in range(60):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            c_diag = c_base[diag_idx[0]]
            a_d = a[diag_idx[0]]
            disc = a_d * a_d + 4.0 * (c_diag - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (a_d - sq) / 2.0
                r2 = (a_d + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a_d = a[diag_idx[0]]
        disc = a_d * a_d + 4.0 * (c_diag - L)
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (a_d - sq) / 2.0
            r2 = (a_d + sq) / 2.0
            if x_lo <= h0 + r1:
                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_optimize(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                m_min = max(0, n - 1 - p)
                m_max = min(2 * n - 2, 2 * n - 2 - p)
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                diag_i = np.where(ms == (n - 1))[0]
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)

            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def nesterov_smooth(h, n, deadline, alpha_start, alpha_end, max_iter=5000):
    # Nesterov-accelerated projected gradient on the smooth max
    h = h.copy()
    m = h.copy()  # momentum variable
    beta = 0.9  # momentum parameter
    step = 1.0 / (alpha_start + n)  # initial step size
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Anneal alpha
        frac = it / max_iter
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        # Compute gradient of smooth objective
        c = np.correlate(h, 1.0 - h, mode='full')
        z = alpha * c
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        w = exp_z / exp_z.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-8:
                continue
            s = k - n + 1
            p_lo = max(0, -s)
            p_hi = min(n, n - s)
            if p_lo < p_hi:
                grad[p_lo:p_hi] += wk * (1.0 - h[p_lo + s:p_hi + s])
            p_lo2 = max(0, s)
            p_hi2 = min(n, n + s)
            if p_lo2 < p_hi2:
                grad[p_lo2:p_hi2] -= wk * h[p_lo2 - s:p_hi2 - s]
        grad *= 2.0 / n
        
        # Nesterov step
        m = project_feasible(h + beta * (m - h), n)
        lr = step / (1.0 + it * 0.01)  # decreasing step
        h_new = project_feasible(m - lr * grad, n)
        
        # Adaptive step: if objective increased, halve step
        c_new = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * (2.0 / n)
        c_old = np.max(c) * (2.0 / n)
        if c_new > c_old * 1.001:
            lr /= 2.0
            h_new = project_feasible(m - lr * grad, n)
        
        h = h_new
        beta = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * beta * beta))
    
    return h

def interpolate_h(h_src, n_src, n_tgt):

    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)
