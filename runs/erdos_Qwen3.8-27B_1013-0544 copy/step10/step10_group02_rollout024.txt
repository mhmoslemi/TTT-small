        hp = project_feasible(h_best + rng.normal(0, scale, n), n)
        h_o, c_o = exact_cd(hp, n, 60, deadline, rng)
        if c_o < best_c - 1e-11:
            h_best, best_c = h_o, c_o
            h_o2, c_o2 = slsqp_refine(h_best, n, deadline,
                                      alphas=[50000, 200000],
                                      topk_list=[3, 2])
            if c_o2 < best_c - 1e-12:
                h_best, best_c = h_o2, c_o2

    # Phase 6: Final long CD polish
    h_best, best_c = exact_cd(h_best, len(h_best), 300, deadline, rng)
    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
