    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(55):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        lo, hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            lo = max(lo, float(np.max(h0 + (L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            hi = min(hi, float(np.min(h0 + (L - c_base[pos_mask]) / a[pos_mask])))
        if lo > hi + 1e-14:
            feasible = False
        if feasible and has_diag:
            ad = a[diag_idx[0]]
            cd = c_base[diag_idx[0]]
            disc = ad * ad - 4.0 * (L - cd)
            if disc >= 0:
                r1 = (ad - np.sqrt(disc)) / 2.0
                r2 = (ad + np.sqrt(disc)) / 2.0
                if lo > h0 + r1 and hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    lo, hi = 0.0, 1.0
    if np.any(neg_mask):
        lo = max(lo, float(np.max(h0 + (L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        hi = min(hi, float(np.min(h0 + (L - c_base[pos_mask]) / a[pos_mask])))
    lo = max(lo, 0.0)
    hi = min(hi, 1.0)
    if has_diag:
        ad = a[diag_idx[0]]
        cd = c_base[diag_idx[0]]
        disc = ad * ad - 4.0 * (L - cd)
        if disc >= 0:
            r1 = (ad - np.sqrt(disc)) / 2.0
            r2 = (ad + np.sqrt(disc)) / 2.0
            if lo <= h0 + r1:
                hi = min(hi, h0 + r1)
            elif hi >= h0 + r2:
                lo = max(lo, h0 + r2)
            else:
                return h0
    if lo > hi:
        return h0
    return float(np.clip((lo + hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, random_order=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n) if (random_order and rng is not None) else np.arange(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def _slsqp_cons_fun(x, n):
    return x.sum() - n / 2.0

def _slsqp_cons_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac,
                 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgrad_step(h, n, lr):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    active = c > cmax - 0.02 * cmax
    n_active = active.sum()
    if n_active == 0:
        return h
    grad = np.zeros(n)
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib = np.zeros(len(ms))
        if np.any(vf):
            contrib[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            contrib[vb] -= h[jb[vb]]
        contrib[ms == (n - 1)] = (1.0 - 2.0 * h[p])
        mask = active[ms]
        if mask.any():
            grad[p] = np.sum(contrib[mask]) / mask.sum()
    h = np.clip(h - lr * grad, 0.0, 1.0)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: SLSQP with increasing smoothness
    h1, c1 = slsqp_opt(h_best, n, [3000, 15000, 75000, 300000, 600000], 250, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Exact CD fixed order
    h2, c2 = exact_cd(h_best, n, 80, deadline)
    if c2 < best_c:
        h_best, best_c = h2, c2

    # Phase 3: Random-order CD for escape
    h3, c3 = exact_cd(h_best, n, 400, deadline, random_order=True, rng=rng)
    if c3 < best_c:
        h_best, best_c = h3, c3

    # Phase 4: Subgradient + CD
    if time.time() < deadline - 200:
        for lr in [0.01, 0.005, 0.002]:
            if time.time() > deadline - 100:
                break
            hs = h_best.copy()
            for _ in range(30):
                if time.time() > deadline - 80:
                    break
                hs = subgrad_step(hs, n, lr)
            ho, co = exact_cd(hs, n, 200, deadline)
            if co < best_c:
                h_best, best_c = ho, co

    # Phase 5: Multi-n with interpolation
    n_list = [650, 750, 550, 800, 600, 700, 850]
    n_cur = len(h_best)
    for n_try in n_list:
        if time.time() > deadline - 120:
            break
        if n_try == n_cur:
            continue
        x_old = np.linspace(0, 2, n_cur)
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        ho, co = slsqp_opt(h_int, n_try, [20000, 100000, 400000], 120, deadline)
        ho, co = exact_cd(ho, n_try, 250, deadline)
        if co < best_c - 1e-8:
            h_best, best_c, n_cur = ho, co, n_try

    # Phase 6: Structured Fourier perturbations
    for freq in [1, 2, 3, 4, 6]:
        if time.time() > deadline - 60:
            break
        x = np.linspace(0, 2, len(h_best))
        for amp in [0.03, 0.06, 0.1, 0.15]:
            if time.time() > deadline - 40:
                break
            pert = project_feasible(h_best + amp * np.sin(np.pi * freq * x + rng.uniform(0, 2 * np.pi)), len(h_best))
            ho, co = exact_cd(pert, len(h_best), 150, deadline)
            if co < best_c:
                h_best, best_c = ho, co

    # Phase 7: Random restarts with moderate noise
    for s in [0.02, 0.05, 0.08, 0.12]:
        if time.time() > deadline - 40:
            break
        hp = project_feasible(h_best + rng.normal(0, s, len(h_best)), len(h_best))
        ho, co = slsqp_opt(hp, len(h_best), [50000, 200000], 80, deadline)
        ho, co = exact_cd(ho, len(h_best), 200, deadline)
        if co < best_c:
            h_best, best_c = ho, co

    # Phase 8: Final intensive CD
    if time.time() < deadline - 5:
        hf, cf = exact_cd(h_best, len(h_best), 300, deadline)
        if cf < best_c:
            h_best, best_c = hf, cf

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
