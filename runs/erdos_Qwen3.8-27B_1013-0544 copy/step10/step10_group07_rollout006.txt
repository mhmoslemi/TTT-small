                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2:
                x_hi = min(x_hi, r1) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
                x_lo = min(x_lo, r2) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                x_hi = min(x_hi, r1) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, random_order=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = list(range(n))
        if random_order and rng is not None:
            rng.shuffle(order)
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def lbfgs_smooth(h, n, alpha, deadline, maxiter=500):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    c_cons = n / 2.0

    def obj(z):
        c = np.correlate(z, 1.0 - z, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * dx

    def jac(z):
        c = np.correlate(z, 1.0 - z, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        weights = np.exp(shifted - mx)
        weights /= weights.sum()
        obj_val = (m + np.log(weights.sum()) ) * dx
        # grad of m term + grad of lse/alpha term
        k_all = np.arange(2 * n - 1) - (n - 1)
        g = np.zeros(n)
        for p in range(n):
            g[p] = 0.0
        # Use vectorized approach
        for p in range(n):
            jf = p + k_all
            jb = p - k_all
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            contrib = np.zeros(2 * n - 1)
            if np.any(vf):
                contrib[vf] += (1.0 - z[jf[vf]])
            if np.any(vb):
                contrib[vb] -= z[jb[vb]]
            if (k_all == 0).any():
                k0_idx = np.where(k_all == 0)[0]
                contrib[k0_idx] = 1.0 - 2.0 * z[p]
            dm_dz_p = np.sum(np.isclose(c, m, atol=1e-10) * contrib) * dx
            lse_grad_p = np.sum(weights * contrib) * dx
            g[p] = dm_dz_p + lse_grad_p
        return g

    try:
        res = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda z: z.sum() - c_cons,
                                     'jac': lambda z: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def subgradient_step(h, n, step, active_frac=0.1):
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = c.max()
    thresh = m - 0.1 * m * active_frac
    active = c >= thresh
    k_all = np.arange(2 * n - 1) - (n - 1)
    g = np.zeros(n)
    for p in range(n):
        jf = p + k_all
        jb = p - k_all
        vf = (jf >= 0) & (jf < n) & active
        vb = (jb >= 0) & (jb < n) & active
        contrib = np.zeros(2 * n - 1)
        if np.any(vf):
            contrib[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            contrib[vb] -= h[jb[vb]]
        k0_idx = np.where((k_all == 0) & active)[0]
        if len(k0_idx) > 0:
            contrib[k0_idx] = 1.0 - 2.0 * h[p]
        g[p] = np.sum(contrib)
    g -= np.mean(g)
    h_new = project_feasible(h - step * g / np.linalg.norm(g), n)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500

    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Cyclic CD
    if time.time() < deadline - 300:
        h1, c1 = exact_cd(h_best, n, 200, deadline)
        if c1 < best_c:
            h_best, best_c = h1, c1

    # Phase 2: L-BFGS-B with increasing alpha
    alphas = [5000, 20000, 80000, 300000, 1000000]
    for alpha in alphas:
        if time.time() > deadline - 300:
            break
        h_a, c_a = lbfgs_smooth(h_best, n, alpha, deadline, maxiter=200)
        h_a, c_a2 = exact_cd(h_a, n, 100, deadline)
        if c_a2 < best_c:
            h_best, best_c = h_a, c_a2

    # Phase 3: Random-order CD for escape
    for r in range(8):
        if time.time() > deadline - 200:
            break
        h_r, c_r = exact_cd(h_best, n, 150, deadline, random_order=True, rng=rng)
        if c_r < best_c:
            h_best, best_c = h_r, c_r

    # Phase 4: Subgradient + CD hybrid
    for sg_iter in range(30):
        if time.time() > deadline - 200:
            break
        step = 0.01 / (1.0 + 0.1 * sg_iter)
        h_s = subgradient_step(h_best, n, step, active_frac=0.05)
        h_s, c_s = exact_cd(h_s, n, 80, deadline)
        if c_s < best_c:
            h_best, best_c = h_s, c_s

    # Phase 5: Multi-n exploration (focused)
    x_old = np.linspace(0, 2, len(h_best))
    for n_try in [500, 600, 700, 800, 900, 1000]:
        if time.time() > deadline - 100:
            break
        if n_try == n:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_n, c_n = exact_cd(h_int, n_try, 300, deadline)
        h_n, c_n2 = exact_cd(h_n, n_try, 200, deadline, random_order=True, rng=rng)
        if c_n2 < best_c - 1e-8:
            h_best, best_c, n = h_n, c_n2, n_try
            x_old = np.linspace(0, 2, n)

    # Phase 6: Final aggressive polish
    if time.time() < deadline - 30:
        h_f, c_f = exact_cd(h_best, n, 500, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    # Phase 7: One more L-BFGS with very high alpha
    if time.time() < deadline - 20:
        h_g, c_g = lbfgs_smooth(h_best, n, 2000000, deadline, maxiter=300)
        h_g, c_g2 = exact_cd(h_g, n, 200, deadline)
        if c_g2 < best_c:
            h_best, best_c = h_g, c_g2

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
