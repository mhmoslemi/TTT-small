import time, math
import numpy as np

def next_pow2(m):
    p = 1
    while p < max(1, int(m)):
        p <<= 1
    return p

def corr_our(h, n):
    v = 1.0 - h
    N = next_pow2(3 * n - 2)
    Fh = np.fft.fft(h, N)
    Fv = np.fft.fft(v, N)
    cor = np.fft.ifft(Fh * np.conj(Fv)).real
    c = np.empty(2 * n - 1, dtype=float)
    L = n - 1
    c[L:L + n] = cor[0:n]
    if L > 0:
        c[:L] = cor[N - L:N]
    c[c < 0.0] = 0.0
    return c

def c5_fast(h, n):
    return float(corr_our(h, n).max() * (2.0 / n))

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * n
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    lo, hi, mid = -1.0, 1.0, 0.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        if np.clip(h + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + mid, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            msk = h < 1.0 - 1e-12
        else:
            msk = h > 1e-12
        cnt = msk.sum()
        if cnt:
            h[msk] = np.clip(h[msk] + d / cnt, 0.0, 1.0)
        else:
            h = np.clip(h + d / n, 0.0, 1.0)
    np.clip(h, 0.0, 1.0, out=h)
    d = target - h.sum()
    if abs(d) > 1e-13:
        if d > 0:
            msk = h < 1.0 - 1e-9
        else:
            msk = h > 1e-9
        cnt = msk.sum()
        if cnt:

    return h

def choose_x_interval(xlo, xhi, x0, Cq, r1, r2):

        return float(x0)
    xlo = max(0.0, float(xlo))
    xhi = min(1.0, float(xhi))
    if xlo > xhi:
        xlo, xhi = xhi, xlo
    cands = [0.5 * (xlo + xhi), xlo, xhi]
    if Cq < 0.25:
        cands.extend([r1, r2])
    best = None
    for x in cands:
        if x < xlo - 1e-12 or x > xhi + 1e-12:
            continue
        if x < 0.0:
            x = 0.0
        if x > 1.0:
            x = 1.0
        if Cq < 0.25 and x > r1 + 1e-12 and x < r2 - 1e-12:
            continue
        d = abs(x - x0)
        if best is None or d < best[1]:
            best = (x, d)
    if best is None:
        x = 0.5 * (xlo + xhi)
        if Cq < 0.25 and x > r1 and x < r2:
            x = r1 if x0 <= 0.5 * (r1 + r2) else r2
        x = max(xlo, min(xhi, x))
        best = (x, 0.0)
    return float(best[0])

def solve_1d(c, h, p, n):
    h0 = float(h[p])
    L = n - 1
    d1 = L - p
    d2 = L + p
    t_min = p if p < d1 else d1
    t_max = L + (p if p > d1 else d1)
    ts = np.arange(t_min, t_max + 1, dtype=int)
    jf = ts - d1
    jb = d2 - ts
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(ts.size, dtype=float)
    if vf.any():
        a[vf] += 1.0 - h[jf[vf]]
    if vb.any():
        a[vb] -= h[jb[vb]]
    c_base = c[ts]
    diag_mask = np.zeros(ts.size, dtype=bool)
    dpos = np.where(ts == L)[0]
    has_diag = dpos.size > 0
    if has_diag:
        dpos = int(dpos[0])
        diag_mask[dpos] = True
        c_diag = float(c_base[dpos])
        a[dpos] = 0.0
    else:
        c_diag = 0.0
    zero_mask = (np.abs(a) <= 1e-15) & (~diag_mask)
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    unaff = 0.0
    if t_min > 0:
        unaff = float(np.max(c[:t_min]))

            lo = mv

            hi = mv

    if zero_mask.any() and float(np.max(c_base[zero_mask])) > target + 1e-12:
        return h0
    if neg_mask.any():

    xlo = max(0.0, xlo)

    return choose_x_interval(xlo, xhi, h0, Cq, r1, r2)

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return c
    L = n - 1
    d1 = L - p
    d2 = L + p
    t_min = p if p < d1 else d1
    t_max = L + (p if p > d1 else d1)
    ts = np.arange(t_min, t_max + 1, dtype=int)
    jf = ts - d1
    jb = d2 - ts
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(ts.size, dtype=float)
    if vf.any():
        a[vf] += 1.0 - h[jf[vf]]
    if vb.any():
        a[vb] -= h[jb[vb]]
    c[ts] += delta * a
    c[L] -= delta * delta
    return c

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = corr_our(h, n)
    for it in range(max_passes):
        if time.time() > deadline:
            break
        order = list(range(n))
        if rng is not None and it % 2 == 1:
            rng.shuffle(order)
        changed = False
        for p in order:
            if time.time() > deadline:
                break
            h0 = float(h[p])
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-11:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                changed = True
        if not changed:
            break
        h2 = project_feasible(h, n)
        if np.max(np.abs(h2 - h)) > 1e-14:
            h = h2
            c = corr_our(h, n)
        else:
            h = h2
    return h, c5_fast(h, n)

def smooth_fg(h, n, alpha, beta=0.0):
    v = 1.0 - h
    N = next_pow2(3 * n - 2)
    Fh = np.fft.fft(h, N)
    Fv = np.fft.fft(v, N)
    cor = np.fft.ifft(Fh * np.conj(Fv)).real
    c = np.empty(2 * n - 1, dtype=float)
    L = n - 1
    c[L:L + n] = cor[0:n]
    if L > 0:
        c[:L] = cor[N - L:N]
    c[c < 0.0] = 0.0
    m = float(c.max())
    sh = alpha * (c - m)
    s = np.exp(sh)
    ss = float(s.sum())
    if ss <= 0.0:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = s / ss
    mean_c = 0.0
    if beta > 0.0:
        mean_c = float(c.mean())
        w = w + beta * (c - mean_c)
    FU = np.fft.fft(w, N)
    Fvr = np.fft.fft(v[::-1], N)
    convF = np.fft.ifft(Fvr * FU).real
    convS = np.fft.ifft(Fh * FU).real
    idx = np.arange(n)
    grad = (convF[2 * L - idx] - convS[L + idx]) * (2.0 / n)
    obj = (m + math.log(ss) / alpha) * (2.0 / n)
    if beta > 0.0:
        obj += 0.5 * beta * float(np.sum((c - mean_c) ** 2)) * (2.0 / n)
    return obj, grad

def smooth_projected(h, n, alpha_beta, max_iter, deadline):

    best_h = h_cur.copy()
    best_c = c5_fast(h_cur, n)
    for alpha, beta in alpha_beta:
        if time.time() > deadline:
            break
        h_cur = best_h.copy()
        lr = 50.0
        stale = 0
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            f, g = smooth_fg(h_cur, n, alpha, beta)
            gn2 = float(np.dot(g, g))
            if gn2 < 1e-20:
                break
            accepted = False
            step = lr
            for __ in range(8):
                if step <= 1e-10:
                    break
                cand = project_feasible(h_cur - step * g, n)
                f_new, _ = smooth_fg(cand, n, alpha, beta)
                if f_new < f - 1e-14:
                    accepted = True
                    h_cur = cand
                    break
                step *= 0.5
            if accepted:
                lr = min(500.0, max(1e-5, 1.15 * (lr if step > 0.8 * lr else step)))
                stale = 0
            else:
                lr = max(1e-6, 0.5 * step)
                stale += 1
                if stale > 8:
                    break
        cc = c5_fast(h_cur, n)
        if cc < best_c - 1e-13:
            best_c = cc
            best_h = h_cur.copy()
    return best_h, c5_fast(best_h, n)

def interp_h(h_old, n_old, n_new):
    if n_new == n_old:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def binary_local(h, n, deadline, rng):
    target = n // 2
    h_cur = np.zeros(n)
    h_cur[np.argsort(h)[::-1][:target]] = 1.0
    c = corr_our(h_cur, n)
    curc = float(c.max())
    dx = 2.0 / n
    best_h = h_cur.copy()
    best_c5 = curc * dx
    T = 1.0
    noimp = 0
    K = 10 if n <= 800 else 7
    while time.time() < deadline and noimp < 450:
        ones = np.nonzero(h_cur)[0]
        zeros = np.nonzero(h_cur == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        c0 = c
        best_move, best_tri, best_val = None, None, curc + 1e9
        for _ in range(K):
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            if i == j:
                continue
            ct = c0.copy()
            ht = h_cur.copy()
            update_corr(ct, ht, i, 1.0, 0.0, n)
            ht[i] = 0.0
            update_corr(ct, ht, j, 0.0, 1.0, n)
            ht[j] = 1.0
            val = float(ct.max())
            if val < best_val:
                best_val, best_tri, best_move = val, ct, ht
        if best_move is None:
            break
        if best_val < curc - 1e-12:
            h_cur, c, curc = best_move, best_tri, best_val
            if curc * dx < best_c5 - 1e-13:
                best_c5 = curc * dx
                best_h = h_cur.copy()
            noimp = 0
            T = max(0.2, 0.98 * T)
        else:
            noimp += 1
            delta = best_val - curc
            if delta <= 1e-12 or (rng.rand() < math.exp(-max(0.0, delta) / T)):
                h_cur, c, curc = best_move, best_tri, best_val
                T = max(0.05, 0.95 * T)
            else:
                T = max(0.05, 0.98 * T)
    cc = c5_fast(best_h, n)
    if cc < best_c5:
        best_c5 = cc
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(30.0, float(budget_s)) - 4.0
    rng = np.random.RandomState(seed)
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_best = np.full(700, 0.5)
    n_best = len(h_best)
    h_best = project_feasible(h_best, n_best)
    best_c = compute_c5(h_best, n_best)

    t = time.time()
    if deadline - t > 240:
        lim = min(deadline - 700.0, t + 190.0)
        h1, c1 = exact_cd(h_best, n_best, 5000, lim, rng)
        if c1 < best_c - 1e-10:
            cc = compute_c5(h1, len(h1))
            if cc < best_c - 1e-13:
                h_best, n_best, best_c = h1, len(h1), cc

    t = time.time()
    if deadline - t > 120:
        lim = min(deadline - 450.0, t + 90.0)
        h2, c2 = smooth_projected(h_best, n_best, [(20.0, 0.0), (200.0, 0.0), (2000.0, 1e-7)], 500, lim)
        if c2 < best_c - 1e-10:
            cc = compute_c5(h2, len(h2))
            if cc < best_c - 1e-13:
                h_best, n_best, best_c = h2, len(h2), cc

    t = time.time()
    if deadline - t > 110:
        lim = min(deadline - 350.0, t + 90.0)
        h3, c3 = exact_cd(h_best, n_best, 5000, lim, rng)
        if c3 < best_c - 1e-10:
            cc = compute_c5(h3, len(h3))
            if cc < best_c - 1e-13:
                h_best, n_best, best_c = h3, len(h3), cc

    t = time.time()
    if deadline - t > 420 and n_best < 1000:
        n_try = 1000
        h4 = interp_h(h_best, n_best, n_try)
        lim_phase = min(deadline - 220.0, t + 320.0)
        lim = min(lim_phase - 70.0, t + 190.0)
        h4, c4 = exact_cd(h4, n_try, 5000, lim, rng)
        t = time.time()
        lim = min(lim_phase - 20.0, t + 70.0)
        if lim > t + 5:
            h4s, c4s = smooth_projected(h4, n_try, [(50.0, 0.0), (500.0, 1e-7), (5000.0, 1e-6)], 350, lim)
            if c4s < c4 - 1e-12:
                h4, c4 = h4s, c4s
        lim = min(lim_phase, t + 90.0)
        h4e, c4e = exact_cd(h4, n_try, 5000, lim, rng)
        if c4e < c4 - 1e-12:
            h4, c4 = h4e, c4e
        if c4 < best_c - 1e-10:
            cc = compute_c5(h4, len(h4))
            if cc < best_c - 1e-13:
                h_best, n_best, best_c = h4, len(h4), cc

    t = time.time()
    if deadline - t > 220:
        n_bin = n_best if n_best % 2 == 0 else n_best - 1
        if n_bin >= 4:
            hbin = interp_h(h_best, n_best, n_bin) if n_bin != n_best else h_best.copy()
            lim = min(deadline - 120.0, t + 130.0)
            hb, cb = binary_local(hbin, n_bin, lim, rng)
            t = time.time()
            lim = min(lim + 10.0, deadline - 110.0)
            if lim > t + 5:
                hb2, cb2 = exact_cd(hb, n_bin, 5000, lim, rng)
                if cb2 < cb - 1e-12:
                    hb, cb = hb2, cb2
            if cb < best_c - 1e-10:
                cc = compute_c5(hb, n_bin)
                if cc < best_c - 1e-13:
                    h_best, n_best, best_c = hb, n_bin, cc

    t = time.time()
    for sc in (0.05, 0.03, 0.015, 0.008):
        if deadline - t < 70:
            break
        hp = project_feasible(h_best + rng.normal(0.0, sc, n_best), n_best)
        lim = min(deadline - 30.0, t + 40.0)
        hp, cp = exact_cd(hp, n_best, 800, lim, rng)
        if cp < best_c - 1e-10:
            cc = compute_c5(hp, n_best)
            if cc < best_c - 1e-13:
                h_best, best_c = hp, cc

    if deadline - time.time() > 20:
        hf, cf = exact_cd(h_best, n_best, 100000, deadline - 0.5, rng)
        if cf < best_c - 1e-13:
            h_best = hf

    h_best = project_feasible(h_best, n_best)
    best_c = compute_c5(h_best, n_best)
    return h_best, best_c, n_best
