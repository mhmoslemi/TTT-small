def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_max_val(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = c.max()
    shifted = alpha * (c - mx)
    sm = shifted.max()
    lse = sm + np.log(np.sum(np.exp(shifted - sm)))
    return float(mx + lse / alpha)

def smooth_max_grad(h, n, alpha):

```python
def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    shifted = alpha * (c - cmax)
    sm = shifted.max()
    p = np.exp(shifted - sm)
    p /= p.sum()  # softmax weights, length 2n-1
    
    # p[k] corresponds to shift s = k-(n-1)
    # grad[j] = dx * sum_k p[k] * [(1-h[j+s])*1_{0<=j+s<n} - h[j-s]*1_{0<=j-s<n}]
    # where s = k-(n-1)
    
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        for k in range(2*n-1):
            if p[k] < 1e-18:
                continue
            s = k - (n-1)
            if 0 <= j+s < n:
                val += p[k] * (1.0 - h[j+s])
            if 0 <= j-s < n:
                val -= p[k] * h[j-s]
        grad[j] = dx * val
    return grad
```
