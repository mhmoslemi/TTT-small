import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_feasible(z, n):
    target = 0.5 * n
    z = np.asarray(z, dtype=float)
    h = np.clip(z, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-15:
        return h

    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(75):
        tau = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(z - tau, 0.0, 1.0)))
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    h = np.clip(z - tau, 0.0, 1.0)

    diff = target - float(h.sum())
    mask = (h > 1e-13) & (h < 1.0 - 1e-13)
    if mask.any():
        h[mask] += diff / float(mask.sum())
        h = np.clip(h, 0.0, 1.0)

    for _ in range(120):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = float(h.sum())
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h


def smooth_core(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(c))
    z = alpha * (c - m)
    ez = np.exp(z)
    s = float(ez.sum())
    if s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ez / s
    val = m + np.log(max(s, 1e-300)) / alpha
    return val, w


def smooth_value(x, n, alpha, mu):
    val, _ = smooth_core(x, n, alpha)
    d = float(x.sum()) - 0.5 * n
    return val + mu * d * d


def add_grad_weights(w, h, n, grad):
    idx = np.flatnonzero(w > 1e-12)
    for ii in idx:
        wi = float(w[ii])
        m = int(ii - (n - 1))
        if m >= 0:
            d = m
            ln = n - d
            if ln <= 0:
                continue
            grad[d:] += wi * (1.0 - h[:ln])
            grad[:ln] -= wi * h[d:]
        else:
            d = -m
            ln = n - d
            if ln <= 0:
                continue
            grad[:ln] += wi * (1.0 - h[d:])
            grad[d:] -= wi * h[:ln]


def smooth_grad(x, n, alpha, mu):
    _, w = smooth_core(x, n, alpha)
    grad = np.zeros(n, dtype=float)
    add_grad_weights(w, x, n, grad)
    if mu != 0.0:
        d = float(x.sum()) - 0.5 * n
        grad += 2.0 * mu * d
    return grad


def llbfgs_opt(h, n, alphas, maxiter, mu, deadline):
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(
                smooth_value,
                best_h,
                args=(n, alpha, mu),
                method="L-BFGS-B",
                jac=smooth_grad,
                bounds=bounds,
                options={"maxiter": maxiter, "maxfun": max(30, 8 * maxiter),
                         "ftol": 1e-11, "gtol": 1e-5},
            )
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            x = project_feasible(x, n)
            c = compute_c5(x, n)
            if c < best_c - 1e-14:
                best_h, best_c = x, c
        except Exception:
            pass
    return best_h, best_c


def cd_update_p(h, c, p, n):
    h0 = float(h[p])
    m1, m2 = p - (n - 1), p
    m3, m4 = -p, n - 1 - p
    m_min = m1 if m1 < m3 else m3
    m_max = m4 if m4 > m2 else m2
    start = n - 1 + m_min
    end = n - 1 + m_max
    L = end - start + 1
    if L <= 0:
        return 0.0

    m = np.arange(m_min, m_max + 1, dtype=int)

    md = (m >= p - (n - 1)) & (m <= p)
    if np.any(md):
        b[md] += 1.0 - h[p - m[md]]

    A = 0.0
    cdiag = 0.0
    z = m == 0

    cseg = c[start:end + 1]

    lin = np.ones(L, dtype=bool)

    pos_idx = np.flatnonzero(lin & (b > 1e-14))

    pos_c = cseg[pos_idx]
    neg_b = b[neg_idx]
    neg_c = cseg[neg_idx]

    low = max(0.0, out_max)
    high = float(np.max(c))
    if high <= low + 1e-13:
        return 0.0

    eps = 1e-12
    for _ in range(30):
        mid = 0.5 * (low + high)
        feas = True

        if feas and zero_idx.size and float(np.max(zero_c)) > mid + eps:
            feas = False
        if feas and pos_idx.size:
            ub = float(np.max((mid - pos_c) / pos_b))
            if ub < hi:
                hi = ub
        if feas and neg_idx.size:
            lb = float(np.min((mid - neg_c) / neg_b))
            if lb > lo:
                lo = lb
        if feas and lo > hi + 1e-11:
            feas = False
        if feas and dpos is not None:
            disc = A * A - 4.0 * (mid - cdiag)
            if disc >= -1e-13:
                if disc < 0.0:
                    disc = 0.0
                r = np.sqrt(disc)
                r1 = 0.5 * (A - r)
                r2 = 0.5 * (A + r)
                if not (lo <= r1 + 1e-11 or hi >= r2 - 1e-11):
                    feas = False
        if feas:
            high = mid
        else:
            low = mid
        if high - low < 1e-11:
            break

    Lval = high
    lo, hi = -h0, 1.0 - h0
    if out_max > Lval + 1e-10:
        return 0.0
    if zero_idx.size and float(np.max(zero_c)) > Lval + 1e-10:
        return 0.0
    if pos_idx.size:
        hi = min(hi, float(np.max((Lval - pos_c) / pos_b)))
    if neg_idx.size:
        lo = max(lo, float(np.min((Lval - neg_c) / neg_b)))
    if lo > hi + 1e-10:
        return 0.0

    forbidden1 = None
    forbidden2 = None
    disc = -1.0
    if dpos is not None:
        disc = A * A - 4.0 * (Lval - cdiag)
        if disc >= 0.0:
            r = np.sqrt(disc)
            forbidden1 = 0.5 * (A - r)
            forbidden2 = 0.5 * (A + r)

    cands = [lo, hi, 0.5 * (lo + hi)]
    if forbidden1 is not None:
        cands += [forbidden1, forbidden2]
        if lo + 1e-12 < forbidden1 < hi - 1e-12:
            cands.append(0.5 * (lo + forbidden1))
        if lo + 1e-12 < forbidden2 < hi - 1e-12:
            cands.append(0.5 * (forbidden2 + hi))

    best_y = 0.0
    best_val = np.inf
    for y in cands:
        y = float(np.clip(y, lo, hi))
        if forbidden1 is not None and forbidden1 + 1e-11 < y < forbidden2 - 1e-11:
            continue
        vals = cseg + b * y
        if dpos is not None:
            vals[dpos] = cdiag + A * y - y * y
        val = float(max(np.max(vals), out_max))
        if val < best_val - 1e-10 or (abs(val - best_val) <= 1e-10 and abs(y) < abs(best_y)):
            best_val = val
            best_y = y

    if best_val > Lval + 1e-7:
        return 0.0
    y = float(np.clip(best_y, -h0, 1.0 - h0))
    if abs(y) <= 1e-13:
        return 0.0

    c[start:end + 1] += b * y
    if dpos is not None:
        c[start + dpos] += A * y - y * y
    h[p] = h0 + y
    return y


def cd_polish(h, n, max_passes, deadline, random_order=False, rng=None):
    h = project_feasible(h, n).copy()
    c = np.correlate(h, 1.0 - h, mode="full")
    best_c = compute_c5(h, n)
    best_h = h.copy()
    best_c_arr = c.copy()
    cur = 0
    stall = 0

    for _ in range(max_passes):
        if time.time() > deadline:
            break
        order = rng.permutation(n) if (random_order and rng is not None) else np.arange(n)
        changed = False
        for pp in order:
            if time.time() > deadline:
                break
            y = cd_update_p(h, c, int(pp), n)
            if abs(y) > 1e-13:
                changed = True

        h2 = project_feasible(h, n)
        c2 = np.correlate(h2, 1.0 - h2, mode="full")

        if c2v < best_c - 1e-12:

            h, c = h2, c2
        else:
            h, c = best_h.copy(), best_c_arr.copy()

        if not changed or stall > 3:
            break

    return best_h, best_c


def resample_h(h, n_new):
