            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')
    C = R * dx
    Cmax = C.max()
    exp_vals = np.exp(alpha * (C - Cmax))
    wsum = exp_vals.sum()
    obj = Cmax + np.log(wsum) / alpha
    w = exp_vals / wsum
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-18:
            continue
        s = k - (n - 1)
        if s >= 0:
            g[:n-s] += wk * (1.0 - h[s:])
            g[s:] -= wk * h[:n-s]
        else:
            g[-s:] += wk * (1.0 - h[:n+s])
            g[:n+s] -= wk * h[-s:]
    return obj, dx * g

def smooth_descent(h, n, alpha, max_iters, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 1.0
    momentum = np.zeros(n)
    beta = 0.7
    for it in range(max_iters):
        if time.time() > deadline:
            break
        obj, grad = smooth_obj_and_grad(h, n, alpha)
        momentum = beta * momentum + (1.0 - beta) * grad
        for _ in range(30):
            h_new = project_feasible(h - step * momentum, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c + 0.001:  # allow small increase
                break
            step *= 0.5
        else:
            break
        h = h_new
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        step = min(step * 1.1, 5.0)
        momentum = beta * momentum
    return best_h, best_c

def simulated_annealing(h, n, deadline, max_iters=5000, T0=0.01, T1=0.0001):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = T0
    n_iter = max_iters
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (T1/T0) ** (it / n_iter)
        # Random perturbation
        h_new = h.copy()
        # Pick a random block to perturb
        block_size = np.random.randint(1, 6)
        start = np.random.randint(0, n - block_size + 1)
        h_new[start:start+block_size] = np.clip(
            h_new[start:start+block_size] + np.random.randn(block_size) * 0.3, 0, 1)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        delta = c_new - best_c
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c
