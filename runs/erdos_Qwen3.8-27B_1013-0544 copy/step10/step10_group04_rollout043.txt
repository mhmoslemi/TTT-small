def solve_1d_correct(c, h, p, n, max_bisect=35):
    h0 = h[p]
    if p < (n-1)/2.0:
        k_min = p
        k_max = 2*n-2-p
    else:
        k_min = n-1-p
        k_max = p+n-1
    
    ks = np.arange(k_min, k_max+1)
    # For each k, determine the coefficient a[k] and whether there's a -delta² term
    # a[k] = (1-h[iA]) - h[iB] where:
    #   iA = p-k+(n-1) (from case 1, the index in (1-h)[i])
    #   iB = p+k-(n-1) (from case 2, the index in h[i+k-(n-1)])
    # Diagonal when iA = iB = p, i.e., k = n-1
    
    iA = p - ks + (n-1)  # index in (1-h) for case 1
    iB = p + ks - (n-1)  # index in h for case 2
    
    vA = (iA >= 0) & (iA < n)
    vB = (iB >= 0) & (iB < n)
    
    a = np.zeros(len(ks))
    c_base = c[ks].copy()
    
    # Case 1 contribution: delta * (1 - h[iA])
    if np.any(vA):
        a[vA] += (1.0 - h[iA[vA]])
    # Case 2 contribution: -delta * h[iB]
    if np.any(vB):
        a[vB] -= h[iB[vB]]
    
    # Diagonal: where iA = iB = p, i.e., k = n-1
    diag_mask = (ks == (n-1))
    has_diag = np.any(diag_mask)
    if has_diag:
        # For diagonal, the actual change is delta*(1-2*h0) - delta²
        # The linear part from our formula gives: (1-h[p]) - h[p] = 1-2*h0 ✓
        # But we need to subtract delta² 
        # So we mark this and handle it in the bisection
        c_diag = c_base[diag_mask[0]]
    
    # Unaffected lags
    unaff = 0.0
    if k_min > 0:
        unaff = float(np.max(c[:k_min]))
    if k_max < 2*n-2:
        unaff = max(unaff, float(np.max(c[k_max+1:])))

    L_hi = cur_max

    for _ in range(max_bisect):
        if L_hi - L_lo < 1e-12:

    for iteration in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n) if (random_order and rng is not None) else np.arange(n)

                    k_max = p+n-1

                iB = p + ks - (n-1)
                vA = (iA >= 0) & (iA < n)
                vB = (iB >= 0) & (iB < n)
                a = np.zeros(len(ks))
                if np.any(vA):
                    a[vA] += (1.0 - h[iA[vA]])
                if np.any(vB):
                    a[vB] -= h[iB[vB]]
                c[ks] += delta * a
                # Subtract delta² for diagonal
                diag_idx = np.where(ks == (n-1))[0]
                if len(diag_idx) > 0:
                    c[ks[diag_idx[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n/2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(np.max(c)) * (2.0/n)
        if cur < best_c:
            best_c = cur
    return h, best_c
