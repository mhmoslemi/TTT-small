def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def cd_1d(c, h, p, n):

    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)

    jb = p - ms + (n - 1)

    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]

        a[diag_idx] = 0.0
    unaff = 0.0

        unaff = max(unaff, float(np.max(c[:m_min])))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max

        if ok and np.any(pos_mask):

        if x_lo > x_hi + 1e-14:
            ok = False

            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq

                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0

    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq

    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def subgrad_explore(h, n, n_steps, step_scale, deadline, rng, momentum=0.9):
    """Random subgradient descent with momentum for exploration."""
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_c = compute_c5(h_cur, n)
    dx = 2.0 / n
    for step in range(n_steps):
        if time.time() > deadline:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        max_c = c.max()
        active = np.where(c >= max_c - 1e-3)[0]
        k_star = rng.choice(active)
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        # dc/dh[j] = (1-h[j+lag]) if 0<=j+lag<n, plus (-h[j-lag]) if 0<=j-lag<n
        if lag == 0:
            grad[:] = 1.0 - 2.0 * h_cur
        elif lag > 0:
            # j+lag valid: j < n-lag
            valid_fwd = np.arange(0, n - lag)
            grad[valid_fwd] += (1.0 - h_cur[valid_fwd + lag])
            # j-lag valid: j >= lag
            valid_bwd = np.arange(lag, n)
            grad[valid_bwd] += (-h_cur[valid_bwd - lag])
        else:
            lag_abs = -lag
            valid_fwd = np.arange(0, n - lag_abs)
            grad[valid_fwd] += (1.0 - h_cur[valid_fwd + lag_abs])
            valid_bwd = np.arange(lag_abs, n)
            grad[valid_bwd] += (-h_cur[valid_bwd - lag_abs])
        grad *= dx
        # Step size schedule

        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def cd_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = max(unaff, float(np.max(c[:m_min])))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(45):
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        ok = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-15):
            ok = False
        if ok and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if ok and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            ok = False
        if ok and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2:
                x_hi = min(x_hi, r1) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
                x_lo = min(x_lo, r2) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                x_hi = min(x_hi, r1) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def cd_pair(h, n, p):
    p2 = p + 1
    s = n / 2.0 - h[p] - h[p2]
    a_lo = max(0.0, s - 1.0)
    a_hi = min(1.0, s)
    if a_hi - a_lo < 1e-12:
        return
    h_orig = h.copy()
    h_orig[p] = 0.0
    h_orig[p2] = 0.0
    c_base = np.correlate(h_orig, 1.0 - h_orig, mode='full')
    k_all = np.arange(2 * n - 1) - (n - 1)
    pk = p - k_all
    pk2 = p + k_all
    pkv = (pk >= 0) & (pk < n)
    pk2v = (pk2 >= 0) & (pk2 < n)
    hp = np.zeros(2 * n - 1)
    hp2 = np.zeros(2 * n - 1)
    if np.any(pkv):
        hp[pkv] = h_orig[pk[pkv]]
    if np.any(pk2v):
        hp2[pk2v] = h_orig[pk2[pk2v]]
    q_p = p + 1 - k_all
    q_p2 = p + 1 + k_all
    qp1v = (q_p >= 0) & (q_p < n)
    qp2v = (q_p2 >= 0) & (q_p2 < n)
    hq1 = np.zeros(2 * n - 1)
    hq2 = np.zeros(2 * n - 1)
    if np.any(qp1v):
        hq1[qp1v] = h_orig[q_p[qp1v]]
    if np.any(qp2v):
        hq2[qp2v] = h_orig[q_p2[qp2v]]
    A = 1.0 - hp2 - hp
    B = 1.0 - hq2 - hq1
    A0 = hp2
    B0 = hq2
    cross_lin = np.zeros(2 * n - 1)
    cross_quad = np.zeros(2 * n - 1)
    cross_quad[n] = 1.0
    cross_lin[n] = -(s + 1.0)
    cross_quad[n - 2] = 1.0
    cross_lin[n - 2] = (1.0 - s)
    Q0 = c_base + A0 + B0 + s * B
    Q1 = A - B + cross_lin
    Q2 = cross_quad
    grid = np.linspace(a_lo, a_hi, 150)
    vals = Q0[None, :] + Q1[None, :] * grid[:, None] + Q2[None, :] * grid[:, None] ** 2
    fvals = vals.max(axis=1)
    i0 = int(fvals.argmin())
    lo = grid[max(0, i0 - 1)]
    hi = grid[min(len(grid) - 1, i0 + 1)]
    phi = (1.0 + np.sqrt(5.0)) / 1.0
    a1 = hi - (hi - lo) / phi
    a2 = lo + (hi - lo) / phi
    f1 = np.max(Q0 + Q1 * a1 + Q2 * a1 ** 2)
    f2 = np.max(Q0 + Q1 * a2 + Q2 * a2 ** 2)
    for _ in range(50):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, a2, f2 = a2, a1, f1
            a1 = hi - (hi - lo) / phi
            f1 = np.max(Q0 + Q1 * a1 + Q2 * a1 ** 2)
        else:
            lo, a1, f1 = a1, a2, f2
            a2 = lo + (hi - lo) / phi
            f2 = np.max(Q0 + Q1 * a2 + Q2 * a2 ** 2)
    a_star = float(np.clip((lo + hi) / 2.0, a_lo, a_hi))
    h[p] = a_star
    h[p2] = s - a_star

def exact_cd_pair(h, n, max_passes, deadline):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(0, n - 1, 2):
            h_old = (h[p], h[p + 1])
            cd_pair(h, n, p)
            h = project_feasible(h, n)
            if abs(h[p] - h_old[0]) > 1e-12 or abs(h[p + 1] - h_old[1]) > 1e-12:
                improved = True
        if not improved:
            break
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': lambda z: z.sum() - n / 2.0,
                                         'jac': lambda z: np.ones(len(z))}],
                           options={'maxiter': maxiter, 'ftol': 1e-22})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-14:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgrad_explore(h, n, n_steps, step_scale, deadline, rng, momentum=0.85):
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_c = compute_c5(h_cur, n)
    dx = 2.0 / n
    for step in range(n_steps):
        if time.time() > deadline:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        max_c = c.max()
        active = np.where(c >= max_c - 2e-3)[0]
        k_star = rng.choice(active)
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag == 0:
            grad[:] = 1.0 - 2.0 * h_cur
        else:
            la = abs(lag)
            fwd = np.arange(0, n - la)
            grad[fwd] += (1.0 - h_cur[fwd + la])
            bwd = np.arange(la, n)
            grad[bwd] += (-h_cur[bwd - la])
        grad *= dx
        step_size = step_scale / (1.0 + 0.05 * step)
        v = momentum * v + grad
        h_new = h_cur - step_size * v
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_c
