def project_feasible(h, n):

        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            if tot < 1e-15:

def block_slsqp(h, n, start, end, alpha, deadline):
    B = end - start
    if B < 2 or B > 40:
        return h.copy()
    h_fixed = h.copy()
    x0 = h_fixed[start:end].copy()
    fixed_sum = h_fixed.sum() - x0.sum()
    target_sum = n / 2.0 - fixed_sum

    def obj(xb):
        ht = h_fixed.copy()
        ht[start:end] = xb
        c = np.correlate(ht, 1.0 - ht, mode='full')
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        lse = mx + np.log(np.sum(np.exp(s - mx)))
        return (m + lse / alpha) * (2.0 / n)

    def obj_grad(xb):
        ht = h_fixed.copy()
        ht[start:end] = xb
        c = np.correlate(ht, 1.0 - ht, mode='full')
        dx = 2.0 / n
        g = np.zeros(B)
        eps = 1e-7
        for j in range(B):
            ht[start+j] = xb[j] + eps
            c2 = np.correlate(ht, 1.0 - ht, mode='full')
            m2 = c2.max()
            s2 = alpha * (c2 - m2)
            mx2 = s2.max()
            lse2 = mx2 + np.log(np.sum(np.exp(s2 - mx2)))
            g[j] = ((m2 + lse2 / alpha) * dx - obj(xb)) / eps
            ht[start+j] = xb[j]
        return g

    bounds = [(0.0, 1.0)] * B
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(B)}]

        h_new = h.copy()
        h_new[start:end] = res.x
        return h_new
    except Exception:

        if delta_c < 0 or (T > 1e-6 and rng.rand() < np.exp(-delta_c / T)):
            h_best, best_c = h_t, c_t
        T *= T_cool

    # Phase 4: Block SLSQP refinement
    block_deadline = deadline - 100
    for B in [5, 10, 15, 20]:
        if time.time() > block_deadline:
            break
        attempts = 0
        while time.time() < block_deadline and attempts < 60:
            attempts += 1
            start = rng.randint(0, max(1, n - B))
            h_new = block_slsqp(h_best, n, start, start + B, 3000, deadline)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-10:
                h_best, best_c = h_new, c_new

    # Phase 5: Multi-n exploration with structured starts
    n_list = [600, 800, 900, 1100, 1400]
    for n_try in n_list:
        if time.time() > deadline - 80:
            break
        if n_try == n:
            continue
        # Interpolate current best
        x_old = np.linspace(0, 2, len(h_best), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = np.interp(x_new, x_old, h_best)
        h_int = project_feasible(h_int, n_try)
        h_t, c_t = exact_cd(h_int, n_try, 40, deadline - 80, rng=None)
        if c_t < best_c - 1e-9:
            h_best, best_c, n = h_t, c_t, n_try

    # Phase 6: Final intensive CD with random order
    n = len(h_best)
    h_best, best_c = exact_cd(h_best, n, 60, deadline, rng=rng)
    # One more forward pass
    h_best, best_c = exact_cd(h_best, n, 30, deadline, rng=None)

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
