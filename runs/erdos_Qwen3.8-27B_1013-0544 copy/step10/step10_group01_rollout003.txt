def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(v, n):
    v = np.asarray(v, dtype=float).copy()
    v = np.clip(v, 0.0, 1.0)
    target = 0.5 * n
    s = float(v.sum())
    if abs(s - target) <= 1e-13:
        return v
    lo, hi = -1.0, 1.0
    for _ in range(70):
        tau = 0.5 * (lo + hi)
        val = float(np.clip(v - tau, 0.0, 1.0).sum())
        if val > target:
            lo = tau
        else:
            hi = tau
    out = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(out.sum())
    if abs(diff) > 1e-12:
        can = (out > 1e-9) & (out < 1.0 - 1e-9)
        if np.any(can):
            out[can] += diff / float(np.sum(can))
            out = np.clip(out, 0.0, 1.0)
    return out

def sym_feasible(h, n):
    h = np.asarray(h, dtype=float)
    h = 0.5 * (h + h[::-1])
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - 0.5*n) > 1e-10:
        h = project_feasible(h, n)
    return h

def interp_to(h, nsrc, ndst):
    if ndst == nsrc:

def max_pref_suff(c):
    L = len(c)
    pref = np.empty(L + 1, dtype=float)
    pref[0] = -1.0
    pref[1:] = np.maximum.accumulate(c)
    suff = np.empty(L + 1, dtype=float)
    suff[L] = -1.0
    suff[:L] = np.maximum.accumulate(c[::-1])[::-1]

def build_coord_info(n):
    info = []
    d0 = n - 1
    d1 = 2 * n - 2
    for p in range(n):
        a = min(p, d0 - p)
        b = max(d0 + p, d1 - p)
        ms = np.arange(a, b + 1, dtype=np.int32)
        plus_idx = p + d0 - ms
        minus_idx = p + ms - d0
        va = (plus_idx >= 0) & (plus_idx < n)
        vb = (minus_idx >= 0) & (minus_idx < n)
        ia = np.nonzero(va)[0].astype(np.int32)
        ja = plus_idx[ia].astype(np.int32)
        ib = np.nonzero(vb)[0].astype(np.int32)
        jb = minus_idx[ib].astype(np.int32)
        info.append((a,b,d0-a,ms,ia,ja,ib,jb))
    return info

import numpy as np
import time

COORD_CACHE = {}
GR = 0.6180339887498949

def c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / float(n)))

def project(v, n):
    v = np.asarray(v, dtype=float).copy()
    v = np.clip(v, 0.0, 1.0)
    target = 0.5 * float(n)
    s = float(v.sum())
    if abs(s - target) <= 1e-13:
        return v
    lo, hi = -1.0, 1.0
    for _ in range(70):
        tau = 0.5 * (lo + hi)
        val = float(np.clip(v - tau, 0.0, 1.0).sum())
        if val > target:
            lo = tau
        else:
            hi = tau
    out = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(out.sum())
    if abs(diff) > 1e-12:
        can = (out > 1e-9) & (out < 1.0 - 1e-9)
        if np.any(can):
            out[can] += diff / float(np.sum(can))
            out = np.clip(out, 0.0, 1.0)
    return out

def sym_feasible(h, n):
    h = np.asarray(h, dtype=float)
    h = 0.5 * (h + h[::-1])
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - 0.5 * float(n)) > 1e-10:
        h = project(h, n)
    return h

def interp(h, n0, n1):
    if n0 == n1:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n1) + 0.5) * (2.0 / float(n1))
    return sym_feasible(np.interp(x1, x0, h), n1)

def build_info(n):
    info = []
    d0 = n - 1
    d1 = 2 * n - 2
    for p in range(n):
        m_lo = min(p, d0 - p)
        m_hi = max(d0 + p, d1 - p)
        ms = np.arange(m_lo, m_hi + 1, dtype=np.int32)
        plus_idx = p + d0 - ms
        minus_idx = p + ms - d0
        va = (plus_idx >= 0) & (plus_idx < n)
        vb = (minus_idx >= 0) & (minus_idx < n)
        ia = np.nonzero(va)[0].astype(np.int32)
        ja = plus_idx[ia].astype(np.int32)
        ib = np.nonzero(vb)[0].astype(np.int32)
        jb = minus_idx[ib].astype(np.int32)
        info.append((m_lo, m_hi, d0 - m_lo, ms, ia, ja, ib, jb))
    return info

def get_info(n):
    if n not in COORD_CACHE:
        COORD_CACHE[n] = build_info(n)
    return COORD_CACHE[n]

def _cd_eval(ca, a, di, h0, d, bm):
    vals = ca + d * a
    m = float(np.max(vals))
    dv = ca[di] + (1.0 - 2.0 * h0) * d - d * d
    if dv > m:
        m = dv
    if bm > m:
        m = bm
    return m

def _line_min(ca, a, di, h0, lo, hi, bm, ns, gi):
    w = hi - lo
    if w <= 1e-15:
        f = float(np.max(ca))
        if bm > f:
            f = bm
        return 0.0, f
    bd = 0.0
    bf = _cd_eval(ca, a, di, h0, 0.0, bm)
    f = _cd_eval(ca, a, di, h0, lo, bm)
    if f < bf:
        bf = f
        bd = lo
    f = _cd_eval(ca, a, di, h0, hi, bm)
    if f < bf:
        bf = f
        bd = hi
    ns = max(5, int(ns))
    bj = -1
    for j in range(1, ns):
        d = lo + w * (float(j) / float(ns))
        f = _cd_eval(ca, a, di, h0, d, bm)
        if f < bf:
            bf = f
            bd = d
            bj = j
    if bd <= lo + 1e-15:
        bj = 0
    elif bd >= hi - 1e-15:
        bj = ns
    if bj == -1:
        idx0 = int(np.floor((0.0 - lo) / w * float(ns)))
        if idx0 < 1:
            bj = 1
        elif idx0 >= ns - 1:
            bj = ns - 1
        else:
            bj = idx0
    if bj <= 1:
        b1 = lo
        b2 = lo + 2.0 * w / float(ns)
    elif bj >= ns - 1:
        b1 = lo + (ns - 2) * w / float(ns)
        b2 = hi
    else:
        b1 = lo + (bj - 1) * w / float(ns)
        b2 = lo + (bj + 1) * w / float(ns)
    if gi > 0 and b2 > b1 + 1e-15:
        d1 = b1 + GR * (b2 - b1)
        d2 = b2 - GR * (b2 - b1)
        f1 = _cd_eval(ca, a, di, h0, d1, bm)
        f2 = _cd_eval(ca, a, di, h0, d2, bm)
        for _ in range(int(gi)):
            if b2 - b1 <= 1e-14 + 1e-13 * abs(b1 - b2):
                break
            if f1 < f2:
                b2 = d2
                d2 = d1
                f2 = f1
                d1 = b1 + GR * (b2 - b1)
                f1 = _cd_eval(ca, a, di, h0, d1, bm)
            else:
                b1 = d1
                d1 = d2
                f1 = f2
                d2 = b1 + GR * (b2 - b1)
                f2 = _cd_eval(ca, a, di, h0, d2, bm)
        dm = 0.5 * (b1 + b2)
        fm = _cd_eval(ca, a, di, h0, dm, bm)
        if fm < bf:
            bf = fm
            bd = dm
    return bd, bf

def cd_opt(h, n, passes, deadline, ns=8, gi=28, keep_tol=1e-4):
    cur = project(h, n)
    best = sym_feasible(cur, n)
    bc = c5(best, n)
    info = get_info(n)
    miss = 0
    for _ in range(int(passes)):
        if time.time() > deadline:
            break
        c = np.correlate(cur, 1.0 - cur, mode='full')
        L = len(c)
        improved = False
        since = 0
        for p in range(n):
            mlo, mhi, di, ms, ia, ja, ib, jb = info[p]
            h0 = cur[p]
            a = np.zeros(len(ms), dtype=float)
            if ia.size:
                a[ia] += 1.0 - cur[ja]
            if ib.size:
                a[ib] -= cur[jb]
            ca = c[ms]
            base = -1.0
            if mlo > 0:
                base = float(c[:mlo].max())
            if mhi + 1 < L:
                t = float(c[mhi + 1:].max())
                if t > base:
                    base = t
            f0 = float(np.max(ca))
            if base > f0:
                f0 = base
            d, fo = _line_min(ca, a, di, h0, -h0, 1.0 - h0, base, ns, gi)
            if fo < f0 - 1e-12:
                v = h0 + d
                if v < 0.0:
                    v = 0.0
                    d = -h0
                elif v > 1.0:
                    v = 1.0
                    d = 1.0 - h0
                if abs(d) > 1e-16:
                    nv = ca + d * a
                    nv[di] = ca[di] + (1.0 - 2.0 * h0) * d - d * d
                    c[ms] = nv
                    cur[p] = v
                    improved = True
                    since += 1
                    if since >= 50:
                        cur = project(cur, n)
                        c = np.correlate(cur, 1.0 - cur, mode='full')
                        L = len(c)
                        since = 0
        if not improved:
            miss += 1
            if miss > 8:
                break
            continue
        nh = project(cur, n)
        cn = c5(nh, n)
        sh = sym_feasible(nh, n)
        cs = c5(sh, n)
        if cs < bc - 1e-12:
            best = sh
            bc = cs
            cur = best.copy()
            miss = 0
        elif cn < bc - 1e-12:
            best = nh
            bc = cn
            cur = best.copy()
            miss = 0
        elif cn <= bc + keep_tol:
            cur = nh.copy()
            miss += 1
            if miss > 14:
                cur = best.copy()
        else:
            cur = best.copy()
            miss += 3
            if miss > 14:
                break
    fh = sym_feasible(project(cur, n), n)
    fc = c5(fh, n)
    if fc < bc:
        best, bc = fh, fc
    return best, bc

def grad_top(C, h, K=180):
    n = len(h)
    L = len(C)
    K = max(1, min(K, L))
    m = float(C.max())
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(-C, K - 1)[:K]
    e = np.exp(C[idx] - m)
    w = e / float(e.sum())
    g = np.zeros(n, dtype=float)
    d0 = n - 1
    for ii in range(len(idx)):
        wt = float(w[ii])
        if wt <= 1e-14:
            continue
        k = int(idx[ii]) - d0
        if k >= 0:
            mm = n - k
            if mm > 0:
                g[k:] += wt * (1.0 - h[:mm])
                g[:mm] -= wt * h[k:]
        else:
            s = -k
            mm = n - s
            if mm > 0:
                g[:mm] += wt * (1.0 - h[s:])
                g[s:] -= wt * h[:mm]
    return g

def smooth_opt(h, n, deadline, betas, steps=140, K=180, lr=0.04):
    cur = sym_feasible(h, n)
    best = cur.copy()
    bc = c5(best, n)
    for beta in betas:
        m = np.zeros(n, dtype=float)
        v = np.zeros(n, dtype=float)
        step = 0
        stuck = 0
        ccur = bc
        for _ in range(int(steps)):
            if time.time() > deadline:
                break
            step += 1
            g = grad_top(np.correlate(cur, 1.0 - cur, mode='full'), cur, K) / float(beta)
            if not np.all(np.isfinite(g)):
                break
            b1 = 0.90
            b2 = 0.999
            m = b1 * m + (1.0 - b1) * g
            v = b2 * v + (1.0 - b2) * g * g
            b1t = 1.0 - b1 ** step
            b2t = 1.0 - b2 ** step
            upd = (lr / (1.0 + 0.02 * step)) * (m / b1t) / (np.sqrt(v / b2t) + 1e-8)
            cand = project(cur - upd, n)
            cand = sym_feasible(cand, n)
            cc = c5(cand, n)
            if cc < bc - 1e-12:
                best = cand
                bc = cc
                cur = best.copy()
                ccur = cc
                stuck = 0
            elif cc < ccur + 2e-4 and cc < bc + 2e-3:
                cur = cand
                ccur = cc
            else:
                cur = best.copy()
                ccur = bc
                stuck += 1
                if stuck > 35:
                    break
        cur = best.copy()
    return best, bc

def perturb(h, n, rng, kind):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    if kind % 4 == 0:
        return project(h + rng.normal(0.0, 0.018, n), n)
    if kind % 4 == 1:
        fr = float(rng.randint(1, 12))
        amp = float(rng.uniform(0.03, 0.12))
        return project(h + amp * np.sin(np.pi * fr * x) + 0.5 * amp * np.cos(2.0 * np.pi * fr * x), n)
    if kind % 4 == 2:
        hp = h.copy()
        s = int(rng.randint(0, n))
        l = int(rng.randint(max(2, n // 25), max(3, n // 4)))
        e = min(n, s + l)
        hp[s:e] = 1.0 - hp[s:e]
        return project(hp, n)
    mask = rng.rand(n) < 0.25
    sh = float(rng.uniform(-0.25, 0.25))
    return project(h + sh * mask, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = float(max(60.0, float(budget_s)))
    dl = t0 + B - 4.0
    rng = np.random.RandomState(seed)
    try:
        arr = np.array(initial_h_values, dtype=float)
    except Exception:
        arr = np.full(500, 0.5)
    n0 = len(arr)
    h = sym_feasible(project(arr, n0), n0)
    bc = c5(h, n0)
    bn = n0
    high_n = min(max(n0, 500), 900)
    work = interp(h, bn, high_n) if bn != high_n else h.copy()
    work = sym_feasible(work, high_n)
    wc = c5(work, high_n)
    if wc < bc:
        h, bc, bn = work, wc, high_n

    e1 = min(dl, t0 + 0.10 * B)
    work, wc = cd_opt(work, high_n, 45, e1, 8, 28, 1e-4)
    if wc < bc:
        h, bc, bn = work, wc, high_n
    work = interp(h, bn, high_n) if bn != high_n else h.copy()
    work = sym_feasible(work, high_n)
    wc = c5(work, high_n)
    if wc < bc:
        h, bc, bn = work, wc, high_n

    e2 = min(dl, t0 + 0.34 * B)
    for nr in (120, 240, 400):
        if time.time() > e2 - 30:
            break
        if nr >= high_n:
            continue
        se = min(e2, time.time() + 0.14 * B)
        r = interp(h, bn, nr)
        r, rc = cd_opt(r, nr, 1200, se, 6, 20, 1e-4)
        if rc < bc:
            h, bc, bn = r, rc, nr
        b = interp(r, nr, high_n)
        b, cb = cd_opt(b, high_n, 70, se, 8, 25, 1e-4)
        if cb < bc:
            h, bc, bn = b, cb, high_n
        work = interp(h, bn, high_n) if bn != high_n else h.copy()
        work = sym_feasible(work, high_n)
        wc = c5(work, high_n)
        if wc < bc:
            h, bc, bn = work, wc, high_n

    e3 = min(dl, t0 + 0.56 * B)
    if time.time() < e3 - 20:
        work = interp(h, bn, high_n) if bn != high_n else h.copy()
        work, wc = smooth_opt(work, high_n, e3, (30, 100, 400, 1500), 150, 180, 0.04)
        if wc < bc:
            h, bc, bn = work, wc, high_n
        work = interp(h, bn, high_n) if bn != high_n else h.copy()
        work = sym_feasible(work, high_n)
        wc = c5(work, high_n)
        if wc < bc:
            h, bc, bn = work, wc, high_n

    e4 = min(dl, t0 + 0.78 * B)
    work, wc = cd_opt(work, high_n, 55, e4, 9, 31, 1e-4)
    if wc < bc:
        h, bc, bn = work, wc, high_n
    work = interp(h, bn, high_n) if bn != high_n else h.copy()
    work = sym_feasible(work, high_n)
    wc = c5(work, high_n)
    if wc < bc:
        h, bc, bn = work, wc, high_n

    tries = 0
    while time.time() < dl - 130 and tries < 8:
        tries += 1
        p = perturb(h, bn, rng, tries)
        pn = bn
        if pn > 900:
            p = interp(p, pn, high_n)
            pn = high_n
        p, pc = cd_opt(p, pn, 24, dl - 110, 7, 23, 1e-4)
        if pc < bc:
            h, bc, bn = p, pc, pn
        work = interp(h, bn, high_n) if bn != high_n else h.copy()
        work = sym_feasible(work, high_n)
        wc = c5(work, high_n)
        if wc < bc:
            h, bc, bn = work, wc, high_n

    if time.time() < dl - 8:
        fn = high_n if high_n <= 900 else bn
        fh = interp(h, bn, fn) if bn != fn else h.copy()
        fh, fc = cd_opt(fh, fn, 90, dl - 1, 10, 35, 0.0)
        if fc < bc:
            h, bc, bn = fh, fc, fn
    sf = sym_feasible(h, bn)
    sfc = c5(sf, bn)
    if sfc < bc:
        h, bc = sf, sfc
    h = project(h, bn)
    bc = c5(h, bn)
    return h, bc, bn
