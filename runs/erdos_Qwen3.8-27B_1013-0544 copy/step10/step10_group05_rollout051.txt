import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            c_diag = c_base[diag_idx[0]]
            a_d = a[diag_idx[0]]
            disc = a_d * a_d + 4.0 * (c_diag - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (a_d - sq) / 2.0
                r2 = (a_d + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:

                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:

                    x_hi = min(x_hi, h0 + r1)

    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                m_min = max(0, n - 1 - p)
                m_max = min(2 * n - 2, 2 * n - 2 - p)
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                diag_i = np.where(ms == (n - 1))[0]
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def multi_lag_subgrad(h, n, deadline, k_lags=10, temperature=5.0, max_iter=500):
    h = h.copy()
    lr = 0.5 / n
    y = h.copy()
    momentum = 0.5
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(y, 1.0 - y, mode='full')
        c_max = c.max()
        idx = np.argsort(c)[-k_lags:]
        active_vals = c[idx]
        weights = np.exp(temperature * (active_vals - c_max))
        weights /= weights.sum()
        grad = np.zeros(n)
        for i, lag in enumerate(idx):
            w = weights[i] * (2.0 / n)
            offset = lag - (n - 1)
            if offset >= 0:
                if n - offset > 0:
                    grad[:n - offset] += w * (1.0 - y[offset:])
                if offset < n:
                    grad[offset:] += w * (-y[:n - offset])
            else:
                d = -offset
                if n - d > 0:
                    grad[d:] += w * (1.0 - y[:n - d])
                if d < n:
                    grad[:n - d] += w * (-y[d:])
        new_momentum = momentum
        y_new = project_feasible(y - new_momentum * grad, n)
        c_new = compute_c5(y_new, n)
        if c_new < compute_c5(h, n):
            h = y_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            lr *= 0.7
            momentum = min(0.9, momentum * 1.02)
    return best_h, best_c

def fourier_perturb(h, n, rng, n_modes=5):
    x = np.arange(n)
    h_p = h.copy()
    for _ in range(n_modes):
        k = rng.randint(1, n // 2)
        amp = rng.uniform(0.01, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        h_p += amp * np.sin(2 * np.pi * k * x / n + phase)
    return project_feasible(h_p, n)

def interpolate_h(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)

def lse_smooth_opt(h, n, alpha, maxiter, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 2.0 / (n * alpha)
    for _ in range(maxiter):

        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w /= w.sum()
        grad = np.zeros(n)
        top_k = np.argsort(w)[-20:]
        for i in top_k:
            if w[i] < 1e-8:

            offset = i - (n - 1)

                if n - offset > 0:
                    grad[:n - offset] += wi * (1.0 - h[offset:])

                    grad[offset:] += wi * (-h[:n - offset])
            else:
                d = -offset
                if n - d > 0:
                    grad[d:] += wi * (1.0 - h[:n - d])
                if d < n:
                    grad[:n - d] += wi * (-h[d:])
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            h = h_new
            best_c = c_new
        else:
            lr *= 0.8
    return h, best_c
