```python
def smooth_gradient(h, n, alpha):
    """Compute gradient of smooth max of correlation values."""
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    scaled = alpha * c * dx
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    weights = exp_scaled / exp_scaled.sum()
    
    grad = np.zeros(n)
    for p in range(n):
        # ∂c_k/∂h[p] for all k
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        dc = np.zeros(len(ms))
        if np.any(vf):
            dc[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            dc[vb] -= h[jb[vb]]
        diag = ms == (n-1)
        dc[diag] = 1.0 - 2.0 * h[p]  # diagonal: ∂(h(1-h))/∂h = 1-2h
        grad[p] = np.dot(weights[ms], dc) * dx
    return grad
```

```python
        kb = p - jb_range + (n-1)
        vb = (kb >= 0) & (kb <= 2*n-2)
        contrib_b = -np.sum(w[kb[vb]] * h[jb_range[vb]])
        
        grad[p] = dx * (contrib_f + contrib_b)
    return grad
```
