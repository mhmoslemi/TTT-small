def project_feasible(y,n):
    y=np.asarray(y,dtype=float)
    if y.size != n:
        y=np.resize(y,n)
    y[~np.isfinite(y)] = 0.5
    y=np.clip(y,0.0,1.0)
    target=0.5*n
    s=float(y.sum())
    if abs(s-target) < 1e-12:
        return y.copy()
    lo=-1.0; hi=1.0
    # ensure bounds (with clipped y, -1 => 0, 1=>n)
    for _ in range(70):
        t=0.5*(lo+hi)
        h=t + y # broadcasting
        h=np.clip(h,0.0,1.0)
        if float(h.sum()) < target:
            lo=t
        else:
            hi=t
    h=np.clip(y+0.5*(lo+hi),0.0,1.0)
    diff=target-float(h.sum())
    if abs(diff) > 1e-13:
        idx=np.where((h>1e-10)&(h<1.0-1e-10))[0]
        if len(idx)>0:
            h[idx[0]] += diff
        else:
            h=np.clip(h+diff/n,0.0,1.0)
    return h

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

_EPS = 1e-13


def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_fast(y, n):
    y = np.asarray(y, dtype=float)
    y = np.clip(y, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(20):
        s = float(y.sum())
        diff = target - s
        if abs(diff) < 1e-12:
            return y
        if diff > 0.0:
            room = 1.0 - y
            tot = float(room.sum())
            if tot < 1e-15:
                break
            y = np.clip(y + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = s
            if tot < 1e-15:
                break
            y = np.clip(y + (diff / tot) * y, 0.0, 1.0)
    return y


def project_feasible(y, n):
    y = np.asarray(y, dtype=float)
    if y.size != n:
        y = np.resize(y, n)
    y[~np.isfinite(y)] = 0.5
    y = np.clip(y, 0.0, 1.0)
    target = 0.5 * n
    if abs(float(y.sum()) - target) < 1e-12:
        return y.copy()
    lo, hi = -1.0, 1.0
    for _ in range(60):
        t = 0.5 * (lo + hi)
        h = np.clip(y + t, 0.0, 1.0)
        if float(h.sum()) < target:
            lo = t
        else:
            hi = t
    h = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        idx = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(idx) > 0:
            h[idx[0]] += diff
        else:
            h = np.clip(h + diff / float(n), 0.0, 1.0)
    return h


def resample_h(h, n_to):
    n_from = len(h)
    if n_from == n_to:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_from, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_to, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h), n_to)


def weighted_c_grad(h, w, n):
    pref = np.concatenate((np.zeros(1), np.cumsum(w)))
    p = np.arange(n)
    lo = n - 1 - p
    hi = 2 * n - 2 - p
    c1 = pref[hi + 1] - pref[lo]
    corr = np.correlate(h, w, mode="full")
    s1 = corr[3 * n - 3:2 * n - 3:-1]
    corr2 = np.correlate(h[::-1], w, mode="full")
    s2 = corr2[2 * n - 2:3 * n - 2]
    return c1 - s1 - s2


def constraint_eq(x):
    return float(x.sum())


def constraint_jac(x):
    return np.ones_like(x)


def soft_fun(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    b = float(c.max())
    s = float(np.exp(alpha * (c - b)).sum())
    if s <= 0.0:
        s = 1.0
    return (b + np.log(s) / alpha) * (2.0 / float(n))


def soft_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    b = float(c.max())
    e = np.exp(alpha * (c - b))
    s = float(e.sum())
    if s <= 0.0:
        e = e / max(1.0, float(np.max(e)))
        s = float(e.sum())
    w = e / s
    val = (b + np.log(s) / alpha) * (2.0 / float(n))
    grad = (2.0 / float(n)) * weighted_c_grad(x, w, n)
    return val, grad


def slsqp_opt(h, n, alphas, deadline, maxiter=40):
    best_h = project_feasible(h, n).copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                soft_fun,
                best_h,
                args=(n, alpha),
                method="SLSQP",
                jac=soft_jac,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint_eq, "jac": constraint_jac},
                options={"maxiter": maxiter, "ftol": 1e-12},
            )
            h2 = project_feasible(res.x, n)
            c2 = compute_c5(h2, n)
            if np.isfinite(c2) and c2 < best_c:
                best_h, best_c = h2, c2
        except Exception:
            pass
    return best_h, best_c


def solve_coordinate(c, h, p, n):
    M = p if p >= n - 1 - p else n - 1 - p
    L = 2 * M + 1
    lo = n - 1 - M
    hi = n - 1 + M
    idx0 = M
    h0 = float(h[p])

    a = np.zeros(L)
    lof = max(-M, -p)
    hif = min(M, n - 1 - p)
    if lof <= hif:
        a[lof + M:hif + M + 1] += 1.0 - h[p + lof:p + hif + 1]
    lob = max(-M, p - (n - 1))
    hib = min(M, p)
    if lob <= hib:
        a[lob + M:hib + M + 1] -= h[p - lob:p - hib - 1:-1]

    c_aff = c[lo:hi + 1]
    a_lin = a.copy()
    a_lin[idx0] = 0.0

    neg = a_lin < -_EPS
    pos = a_lin > _EPS
    zero = np.abs(a_lin) <= _EPS
    zero[idx0] = False

    if neg.any():
        aneg = a_lin[neg]
        cneg = c_aff[neg]
    else:
        aneg = cneg = None
    if pos.any():
        apos = a_lin[pos]
        cpos = c_aff[pos]
    else:
        apos = cpos = None
    zero_max = float(c_aff[zero].max()) if zero.any() else -np.inf

    unaff = 0.0
    if lo > 0:
        unaff = max(unaff, float(c[:lo].max()))
    if hi < 2 * n - 2:
        unaff = max(unaff, float(c[hi + 1:].max()))

    c_center = float(c[n - 1])
    h0q = h0 - h0 * h0
    Llo = unaff
    Lhi = max(unaff, float(c_aff.max()))

    for _ in range(35):
        if Lhi - Llo < 1e-10:
            break
        Lm = 0.5 * (Llo + Lhi)
        feasible = Lm >= unaff - 1e-12
        xl, xh = 0.0, 1.0
        if feasible:
            if aneg is not None:
                xl = max(xl, float(np.max(h0 + (Lm - cneg) / aneg)))
            if apos is not None:
                xh = min(xh, float(np.min(h0 + (Lm - cpos) / apos)))
            xl = max(0.0, xl)
            xh = min(1.0, xh)
            if xl > xh + 1e-12:
                feasible = False
            elif zero_max > Lm + 1e-12:
                feasible = False
            else:
                Cq = Lm - c_center + h0q
                if Cq < -1e-12:
                    feasible = False
                elif Cq < 0.25 - 1e-14:
                    disc = max(0.0, 1.0 - 4.0 * Cq)
                    r1 = 0.5 * (1.0 - np.sqrt(disc))
                    r2 = 1.0 - r1
                    if not (xl <= xh + 1e-12 and (xl <= r1 + 1e-12 or r2 <= xh + 1e-12)):
                        feasible = False
        if feasible:
            Lhi = Lm
        else:
            Llo = Lm

    L = Lhi
    xl, xh = 0.0, 1.0
    if aneg is not None:
        xl = max(xl, float(np.max(h0 + (L - cneg) / aneg)))
    if apos is not None:
        xh = min(xh, float(np.min(h0 + (L - cpos) / apos)))
    xl = max(0.0, xl)
    xh = min(1.0, xh)

    if xl > xh + 1e-10 or zero_max > L + 1e-10:
        return h0, a, lo, hi

    Cq = L - c_center + h0q
    if Cq < -1e-10:
        return h0, a, lo, hi

    if Cq < 0.25 - 1e-14:
        disc = max(0.0, 1.0 - 4.0 * Cq)
        r1 = 0.5 * (1.0 - np.sqrt(disc))
        r2 = 1.0 - r1
        first = xl <= r1 + 1e-12 and xl <= xh + 1e-12
        second = r2 <= xh + 1e-12 and xl <= xh + 1e-12
        if not first and not second:
            return h0, a, lo, hi
        best_x = h0
        best_d = 1e100
        if first:
            lo1 = xl
            hi1 = min(xh, r1)
            if hi1 >= lo1 - 1e-12:
                x1 = float(np.clip(h0, lo1, hi1))
                d1 = abs(x1 - h0)
                if d1 < best_d:
                    best_x, best_d = x1, d1
        if second:
            lo2 = max(xl, r2)
            hi2 = xh
            if hi2 >= lo2 - 1e-12:
                x2 = float(np.clip(h0, lo2, hi2))
                d2 = abs(x2 - h0)
                if d2 < best_d:
                    best_x = x2
        x = best_x
    else:
        x = float(np.clip(h0, xl, xh))

    if not np.isfinite(x):
        x = h0
    x = float(np.clip(x, 0.0, 1.0))
    return x, a, lo, hi


def cd_polish(h, n, max_passes, deadline, random_order=False, rng=None):
    h = project_feasible(h, n).copy()
    c = np.correlate(h, 1.0 - h, mode="full")
    cur_raw = float(c.max())
    no_improve = 0

    for _ in range(max_passes):
        if time.time() > deadline:
            break
        if random_order:
            if rng is None:
                rng = np.random.RandomState(12345)
            order = rng.permutation(n)
        else:
            order = range(n)
        moved = False
        for p in order:
            if time.time() > deadline:
                break
            x, a, lo, hi = solve_coordinate(c, h, p, n)
            if abs(x - h[p]) > 1e-12:
                delta = x - h[p]
                c[lo:hi + 1] += delta * a
                c[n - 1] -= delta * delta
                h[p] = x
                moved = True
        if not moved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode="full")
        new_raw = float(c.max())
        if new_raw < cur_raw - 1e-11:
            cur_raw = new_raw
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                break

    h = project_feasible(h, n)
    return h, compute_c5(h, n)


def polish_candidate(h, n, deadline, best_h, best_c, cd_passes=80, rng=None, use_slsqp=True):

        if c2 <= c + 1e-12:

def de_objective(y, n):
    h = project_fast(y, n)
    c = compute_c5(h, n)
    diff = abs(float(h.sum()) - 0.5 * n) / float(n)
    return c + 10.0 * diff


def de_search(n, deadline, rng):
    if n < 24 or n > 120 or time.time() > deadline:
        return None
    bounds = [(0.0, 1.0)] * n
    try:
        res = differential_evolution(
            de_objective,
            bounds,
            args=(n,),
            seed=int(rng.randint(2**31 - 1)),
            maxiter=24,
            popsize=12,
            tol=1e-5,
            polish=False,
        )
        if getattr(res, "x", None) is not None:
            h = project_feasible(res.x, n)
            if np.isfinite(compute_c5(h, n)):
                return h
    except Exception:
        pass
    return None
