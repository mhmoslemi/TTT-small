def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    if abs(h.sum() - target) <= 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.clip(h + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(25):
        res = target - h.sum()
        if abs(res) < 1e-13:
            break
        if res > 0.0:
            free = np.where(h < 1.0 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if free.size == 0:
            break
        h[free] = np.clip(h[free] + res / free.size, 0.0, 1.0)
    return h

def coord_step(c, h, p, n, nbisect):
    h0 = float(h[p])
    tot = 2 * n - 1
    la = n - 1 - p
    la = la if la > 0 else 0
    ha = n - 1 + p
    ha = ha if ha < tot else tot - 1
    lb = p
    hb = p + n - 1
    hb = hb if hb < tot else tot - 1
    ml = la if la < lb else lb
    mh = ha if ha > hb else hb
    ln = mh - ml + 1
    if ln <= 0:
        return False
    ms = np.arange(ml, mh + 1)
    a = np.zeros(ln)
    cb = c[ms]
    jf = p + ms - (n - 1)
    vf = (jf >= 0) & (jf < n)

    vb = (jb >= 0) & (jb < n)

def cd_improve(h, n, rng, deadline, nbisect, random_order=True, max_passes=1000):
    target = 0.5 * n
    hh = project_feasible(h, n).copy()
    c = np.correlate(hh, 1.0 - hh, mode="full")
    bh = hh.copy()
    bc = float(np.max(c) * (2.0 / n))
    passes = 0
    while passes < max_passes and time.time() < deadline:
        imp = 0
        order = rng.permutation(n) if random_order else np.arange(n)
        for idx, p in enumerate(order):
            if (idx & 7) == 0 and time.time() >= deadline:
                break
            if coord_step(c, hh, p, n, nbisect):
                imp += 1
        passes += 1
        if imp == 0:
            break
        if abs(hh.sum() - target) > 1e-9:
            hh = project_feasible(hh, n)
            c = np.correlate(hh, 1.0 - hh, mode="full")
        cv = float(np.max(c) * (2.0 / n))
        if cv < bc - 1e-12:
            bh = hh.copy()
            bc = cv
    hh = project_feasible(hh, n)
    c = np.correlate(hh, 1.0 - hh, mode="full")
    cv = float(np.max(c) * (2.0 / n))
    if cv < bc:
        bh = hh.copy()
        bc = cv
    return bh, bc

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return np.array(h, dtype=float).copy()
    x0 = np.linspace(0.0, 2.0, n_from)
    x1 = np.linspace(0.0, 2.0, n_to)
    return np.interp(x1, x0, h)

def make_even(m):

        m -= 1

        m = 1000

def fourier_build(p, n, J, x):
    h = 0.5 * np.ones(n)
    for j in range(J):
        a = p[2 * j]
        if abs(a) > 1e-14:
            h += a * np.cos((j + 1) * np.pi * x + p[2 * j + 1])
    return h

def fourier_obj(p, n, J, alpha, x):
    h = fourier_build(p, n, J, x)
    vm = 0.0
    mind = float(h.min())
    maxd = float(h.max())
    if mind < 0.0:
        vm += mind * mind
    if maxd > 1.0:
        vm += (maxd - 1.0) * (maxd - 1.0)
    pa = p[0::2]
    over = np.abs(pa) - 0.45
    if np.any(over > 0.0):
        vm += float(np.sum(over[over > 0.0] ** 2))
    hc = np.clip(h, 0.0, 1.0)
    area = hc.sum() - 0.5 * n
    c = np.correlate(hc, 1.0 - hc, mode="full")
    m = float(np.max(c))
    s = alpha * (c - m)
    mx = float(np.max(s))
    lse = mx + float(np.log(np.sum(np.exp(s - mx))))
    return (m + lse / alpha) * (2.0 / n) + 60.0 * vm + 8.0 * area * area

def fourier_optimize(n, rng, deadline, J, amp):
    x = np.linspace(0.0, 2.0, n)
    bh = None
    bc = 1e100
    for _ in range(3):
        if time.time() >= deadline:
            break
        p = np.zeros(2 * J)
        for j in range(J):
            p[2 * j] = rng.uniform(-amp, amp) * (1.0 / (0.65 + 0.35 * j))
            p[2 * j + 1] = rng.uniform(0.0, 2.0 * np.pi)
        if minimize is None:
            res_x = p
        else:
            try:
                res = minimize(fourier_obj, p, args=(n, J, 260.0, x), method="Nelder-Mead",
                               options={"maxiter": 450, "xatol": 1e-7, "fatol": 1e-9})
                res_x = res.x
            except Exception:
                res_x = p
        h = project_feasible(fourier_build(res_x, n, J, x), n)
        cv = compute_c5(h, n)
        if cv < bc:
            bh = h.copy()
            bc = cv
    if bh is None:
        bh = project_feasible(0.5 * np.ones(n), n)
        bc = compute_c5(bh, n)
    return bh, bc

def make_sine_seed(n, rng, J, amp):
