import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def get_correlate(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def update_correlate(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c[ms] += delta * a
    diag_i = np.where(ms == (n - 1))[0]
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta

def solve_coord(c, h, p, n):
    h0 = h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        um = float(np.max(c[m_max + 1:]))
        unaff = max(unaff, um)
    cur_max = float(np.max(c))
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(45):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            c_d = c_base[diag_idx[0]]
            a_d = a[diag_idx[0]]
            disc = a_d * a_d + 4.0 * (c_d - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (a_d - sq) / 2.0
                r2 = (a_d + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if has_diag:
        c_d = c_base[diag_idx[0]]
        a_d = a[diag_idx[0]]
        disc = a_d * a_d + 4.0 * (c_d - L)
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (a_d - sq) / 2.0
            r2 = (a_d + sq) / 2.0
            if x_lo <= h0 + r1:
                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_passes(h, n, max_passes, deadline, rng=None, order_mode='seq'):
    h = h.copy()
    c = get_correlate(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if order_mode == 'rand' and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                update_correlate(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = get_correlate(h, n)
    return h, compute_c5(h, n)

def pair_cd(h, n, max_pairs, deadline, rng):
    h = h.copy()
    c = get_correlate(h, n)
    best_c = float(np.max(c))
    for _ in range(max_pairs):
        if time.time() > deadline:
            break
        p = rng.randint(n)
        q = rng.randint(n)
        if p == q:
            continue
        h0p, h0q = h[p], h[q]
        if h0p < 1e-4 or h0p > 1 - 1e-4 or h0q < 1e-4 or h0q > 1 - 1e-4:
            continue
        best_xp, best_xq, best_val = h0p, h0q, best_c
        for xp_n in np.linspace(max(0.01, h0p - 0.3), min(0.99, h0p + 0.3), 7):
            for xq_n in np.linspace(max(0.01, h0q - 0.3), min(0.99, h0q + 0.3), 7):
                test_h = h.copy()
                test_h[p] = xp_n
                test_h[q] = xq_n
                tc = c.copy()
                delta_p = xp_n - h0p
                delta_q = xq_n - h0q
                for pp, dd in [(p, delta_p), (q, delta_q)]:
                    m_min = max(0, n - 1 - pp)
                    m_max = min(2 * n - 2, 2 * n - 2 - pp)
                    ms = np.arange(m_min, m_max + 1)
                    jf = pp + ms - (n - 1)
                    jb = pp - ms + (n - 1)
                    vf = (jf >= 0) & (jf < n)
                    vb = (jb >= 0) & (jb < n)
                    aa = np.zeros(len(ms))
                    if np.any(vf):
                        aa[vf] += (1.0 - test_h[jf[vf]])
                    if np.any(vb):
                        aa[vb] -= test_h[jb[vb]]
                    tc[ms] += dd * aa
                    di = np.where(ms == (n - 1))[0]
                    if len(di) > 0:
                        tc[ms[di[0]]] -= dd * dd
                val = float(np.max(tc))
                if val < best_val - 1e-7:
                    best_val = val
                    best_xp, best_xq = xp_n, xq_n
        if best_val < best_c - 1e-7:
            update_correlate(c, h, p, h0p, best_xp, n)
            update_correlate(c, h, q, h0q, best_xq, n)
            h[p], h[q] = best_xp, best_xq
            best_c = best_val
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_obj_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    grad = np.zeros(2 * n - 1)
    for k_idx in range(len(c)):
        if w[k_idx] < 1e-15:
            continue
        k = k_idx - (n - 1)
        for i in range(max(0, -k), min(n, n - k)):
            j = i + k
            grad[k_idx] += w[k_idx] * (1.0 - x[j])
        for i in range(max(0, -k), min(n, n - k)):
            j = i + k
            grad[k_idx] -= w[k_idx] * x[i]
    gf = np.zeros(n)
    for p in range(n):
        m_min = max(0, n - 1 - p)
        m_max = min(2 * n - 2, 2 * n - 2 - p)
        gf[p] = (grad[m_min:m_max + 1]).sum()
        m = n - 1
        gf[p] -= 1.0 - 2.0 * x[p] * 0  # diagonal handled in grad already
    return gf * (2.0 / n)

def slsqp_opt(h0, n, alphas, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(len(x))}]
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(slsqp_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interpolate_h(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)

def fourier_init(n, K, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    for k in range(1, K + 1):
        amp = rng.normal(0, 1.0 / np.sqrt(k))
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * x / 2.0 + phase)
    h = 0.5 + 0.4 * h
    return project_feasible(h, n)
