    lse = mx + np.log(np.sum(np.exp(shifted - mx)))

            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interpolate_h(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)

def targeted_perturb(h, n, c, rng, magnitude=0.05):
    dx = 2.0 / n
    active = np.where(c > c.max() - 0.1 * dx)[0]
    hp = h.copy()
    for m in active:
        if m < n:
            lo = max(0, m - (n - 1))
            hi = min(n, m - (n - 1) + n)
            idx = rng.choice(range(lo, hi), size=min(3, hi - lo), replace=False)
            hp[idx] += rng.choice([-1, 1]) * magnitude * rng.uniform(0.5, 1.5)
        else:
            lo = max(0, m - 2 * (n - 1))
            hi = min(n, n - (n - 1 - (m - (n - 1))))
            if hi > lo:
                idx = rng.choice(range(lo, hi), size=min(3, hi - lo), replace=False)
                hp[idx] += rng.choice([-1, 1]) * magnitude * rng.uniform(0.5, 1.5)
    return project_feasible(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(600, 0.5)
        n = 600
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: SLSQP refinement
    h1, c1 = slsqp_opt(h_best, n, [5000, 50000, 500000], 300, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Exact CD (long)
    h2, c2 = exact_cd(h_best, n, 500, deadline, tol=1e-11)
    if c2 < best_c:
        h_best, best_c = h2, c2

    # Phase 3: Pair CD for local refinement
    if time.time() < deadline - 200:
        h3, c3 = pair_cd(h_best, n, 30, deadline, rng)
        if c3 < best_c:
            h_best, best_c = h3, c2
            h_best = project_feasible(h_best, n)
            best_c = compute_c5(h_best, n)
        h3, c3 = exact_cd(h3, n, 200, deadline, tol=1e-11)
        if c3 < best_c:
            h_best, best_c = h3, c3

    # Phase 4: Targeted perturbations + CD
    n = len(h_best)
    c = np.correlate(h_best, 1.0 - h_best, mode='full')
    for attempt in range(30):
        if time.time() > deadline - 150:
            break
        hp = targeted_perturb(h_best, n, c, rng, magnitude=0.03 + 0.02 * attempt)
        ho, co = exact_cd(hp, n, 300, deadline, tol=1e-11)
        if co < best_c:
            h_best, best_c = ho, co
            c = np.correlate(h_best, 1.0 - h_best, mode='full')

    # Phase 5: Random Fourier restarts
    for r in range(10):
        if time.time() > deadline - 200:
            break
        hp = fourier_init(n, 5 + 3 * r, rng)
        ho, co = slsqp_opt(hp, n, [100000, 1000000], 150, deadline)
        ho, co = exact_cd(ho, n, 300, deadline, tol=1e-11)
        if co < best_c:
            h_best, best_c = ho, co

    # Phase 6: Try nearby n values
    n = len(h_best)
    for n_try in [n - 20, n + 20, 650, 750]:
        if time.time() > deadline - 150:
            break
        if n_try < 300 or n_try > 1000 or n_try == n:
            continue
        h_int = interpolate_h(h_best, n, n_try)
        ho, co = slsqp_opt(h_int, n_try, [50000, 500000], 150, deadline)
        ho, co = exact_cd(ho, n_try, 300, deadline, tol=1e-11)
        if co < best_c - 1e-7:
            h_best, best_c, n = ho, co, n_try

    # Phase 7: Final SLSQP + CD polish
    if time.time() < deadline - 30:
        for alpha in [1000000, 5000000, 20000000]:
            if time.time() > deadline - 15:
                break
            ho, co = slsqp_opt(h_best, n, [alpha], 200, deadline)
            ho, co = exact_cd(ho, n, 300, deadline, tol=1e-12)
            if co < best_c:
                h_best, best_c = ho, co

    # Phase 8: Final exact CD
    if time.time() < deadline - 3:
        h_f, c_f = exact_cd(h_best, n, 500, deadline, tol=1e-12)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
