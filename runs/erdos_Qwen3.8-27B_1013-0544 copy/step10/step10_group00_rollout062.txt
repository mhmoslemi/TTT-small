            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _fv(ca, a, di, hd, um, h0, d):
    v = ca + d * a
    if hd:
        v[di] = ca[di] + d * (1.0 - 2.0 * h0) - d * d
    r = float(np.max(v))
    return max(r, um)

def cd(h, n, passes, deadline, gi=35):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        imp = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            lo = max(0, n - 1 - p)
            hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(lo, hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            dp = np.where(ms == n - 1)
            hd = len(dp[0]) > 0
            di = int(dp[0][0]) if hd else -1
            ca = c[ms]
            um = 0.0
            if lo > 0:
                um = max(um, float(c[:lo].max()))
            if hi < 2 * n - 2:
                um = max(um, float(c[hi + 1:].max()))
            dl, dh = -h0, 1.0 - h0
            d1 = dh - gr * (dh - dl)
            d2 = dl + gr * (dh - dl)
            f1 = _fv(ca, a, di, hd, um, h0, d1)
            f2 = _fv(ca, a, di, hd, um, h0, d2)
            for _ in range(gi):
                if dh - dl < 1e-13:
                    break
                if f1 < f2:
                    dh = d2; d2 = d1; f2 = f1
                    d1 = dh - gr * (dh - dl)
                    f1 = _fv(ca, a, di, hd, um, h0, d1)
                else:
                    dl = d1; d1 = d2; f1 = f2
                    d2 = dl + gr * (dh - dl)
                    f2 = _fv(ca, a, di, hd, um, h0, d2)
            do = (dl + dh) / 2.0
            fo = _fv(ca, a, di, hd, um, h0, do)
            fc = max(float(np.max(ca)), um)
            if fo < fc - 1e-13:
                h[p] = h0 + do
                imp = True
        if not imp:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def _sobj(x, n, al):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = al * (c - m)
    mx = s.max()
    return (m + mx + np.log(np.sum(np.exp(s - mx)))) / al * dx

def _seq(x, n):
    return x.sum() - n / 2.0

def _sej(x, n):
    return np.ones(n)

def slsqp(h, n, alphas, mi=300):
    bh = h.copy()
    bc = compute_c5(h, n)
    bd = [(0.0, 1.0)] * n
    cn = [{'type': 'eq', 'fun': _seq, 'jac': _sej, 'args': (n,)}]
    for al in alphas:
        try:
            r = minimize(_sobj, bh, args=(n, al), method='SLSQP',
                         bounds=bd, constraints=cn,
                         options={'maxiter': mi, 'ftol': 1e-15})
            nh = project_feasible(r.x, n)
            nc = compute_c5(nh, n)
            if nc < bc:
                bh, bc = nh, nc
        except:
            pass
    return bh, bc

def _make_restart(bh, n, r, rng):
    sc = 0.02 + 0.03 * (r % 10) / 10.0
    if r % 7 == 0:
        hp = project_feasible(bh + rng.normal(0, sc, n), n)
    elif r % 7 == 1:
        sh = rng.randint(1, max(2, n // 6))
        hp = project_feasible(np.roll(bh, sh), n)
    elif r % 7 == 2:
        bl = max(1, n // 8)
        st = rng.randint(0, max(1, n - bl))
        hp = bh.copy()
        hp[st:st + bl] = 1.0 - hp[st:st + bl]
        hp = project_feasible(hp, n)
    elif r % 7 == 3:
        x = np.linspace(0, 2, n)
        fr = rng.uniform(3, 12)
        ph = rng.uniform(0, 6.28)
        hp = project_feasible(bh + sc * 2.5 * np.sin(fr * x + ph), n)
    elif r % 7 == 4:
        bl = max(1, n // 16)
        st = rng.randint(0, max(1, n - bl))
        hp = bh.copy()
        hp[st:st + bl] = 1.0 - hp[st:st + bl]
        hp = project_feasible(hp, n)
    elif r % 7 == 5:
        hp = project_feasible(bh * (1.0 + 0.03 * rng.randn(n)), n)
    else:
        bl = max(1, n // 10)
        s1 = rng.randint(0, max(1, n - 2 * bl))
        s2 = s1 + bl + rng.randint(1, max(1, n - 2 * bl - bl))
        if s2 + bl > n:
            s2 = n - bl
        hp = bh.copy()
        tmp = hp[s1:s1 + bl].copy()
        hp[s1:s1 + bl] = hp[s2:s2 + bl]
        hp[s2:s2 + bl] = tmp
        hp = project_feasible(hp, n)
    return hp

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    dl = t0 + budget_s - 10
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except:
        h0 = np.full(500, 0.5)
        n = 500
    bh = project_feasible(h0, n)
    bc = compute_c5(bh, n)

    # Phase 1: Initial CD with corrected diagonal
    if time.time() < dl - 400:
        h, c = cd(bh, n, 60, dl - 400, gi=35)
        if c < bc:
            bh, bc = h, c

    # Phase 2: SLSQP continuation
    if time.time() < dl - 350:
        h, c = slsqp(bh, n, [400, 1500, 8000, 40000], 250)
        if c < bc:
            bh, bc = h, c
            h, c = cd(bh, n, 30, dl - 350, gi=35)
            if c < bc:
                bh, bc = h, c

    # Phase 3: Structured multi-restart
    rem = dl - time.time() - 280
    nre = max(8, int(rem / 30))
    for r in range(nre):
        if time.time() > dl - 280:
            break
        hp = _make_restart(bh, n, r, rng)
        h, c = cd(hp, n, 25, dl - 280, gi=30)
        if c < bc:
            bh, bc = h, c
            if time.time() < dl - 280:
                h, c = slsqp(h, n, [3000, 15000], 200)
                if c < bc:
                    bh, bc = h, c
                    h, c = cd(bh, n, 20, dl - 280, gi=35)
                    if c < bc:
                        bh, bc = h, c

    # Phase 4: Grid size exploration
    xo = np.linspace(0, 2, n)
    grid_opts = [400, 550, 800, 1000, 1300]
    if n not in grid_opts:
        grid_opts.append(n)
    grid_opts = [g for g in grid_opts if g != n]
    for nt in grid_opts:
        if time.time() > dl - 120:
            break
        xn = np.linspace(0, 2, nt)
        hi = project_feasible(np.interp(xn, xo, bh), nt)
        h, c = cd(hi, nt, 35, dl - 120, gi=30)
        if c < bc:
            h, c = slsqp(h, nt, [2000, 10000], 200)
            if c < bc:
                h, c = cd(h, nt, 20, dl - 120, gi=35)
                if c < bc:
                    bh, bc, n = h, c, nt
                    xo = xn

    # Phase 5: Second round of restarts on best
    rem2 = dl - time.time() - 100
    nre2 = max(4, int(rem2 / 30))
    for r in range(nre2):
        if time.time() > dl - 100:
            break
        hp = _make_restart(bh, n, r + 50, rng)
        h, c = cd(hp, n, 30, dl - 100, gi=35)
        if c < bc:
            bh, bc = h, c
            h, c = slsqp(h, n, [5000, 20000], 200)
            if c < bc:
                bh, bc = h, c
                h, c = cd(bh, n, 20, dl - 100, gi=35)
                if c < bc:
                    bh, bc = h, c

    # Phase 6: Final polish
    if time.time() < dl - 5:
        h, c = slsqp(bh, n, [5000, 20000, 100000], 400)
        if c < bc:
            bh, bc = h, c
        h, c = cd(bh, n, 60, dl - 5, gi=40)
        if c < bc:
            bh, bc = h, c

    bh = project_feasible(bh, n)
    bc = compute_c5(bh, n)
    return bh, bc, n
