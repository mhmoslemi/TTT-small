    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n, nb=40):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = float(c_base[diag_idx[0]])
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(nb):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) * 0.5
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0.0:
                r1 = (1.0 - np.sqrt(disc)) * 0.5
                r2 = (1.0 + np.sqrt(disc)) * 0.5
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0.0:
            r1 = (1.0 - np.sqrt(disc)) * 0.5
            r2 = (1.0 + np.sqrt(disc)) * 0.5
            if h0 - r1 <= r2 - h0:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) * 0.5, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, nb=40, random_order=True, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    order = np.arange(n) if not random_order or rng is None else None
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        if random_order and rng is not None:
            order = rng.permutation(n)
        improved = False
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_1d_bisect(c, h, p, n, nb)
            if abs(xn - h0) > 1e-11:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n * 0.5) > 1e-11:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def perturbation(h, n, strategy, rng, round_num):
    x = np.linspace(0, 2, n)
    if strategy == 0:
        sigma = 0.02 + 0.02 * min(round_num, 5)
        return project_feasible(h + rng.normal(0, sigma, n), n)
    elif strategy == 1:
        freq = rng.randint(2, 40)
        phase = rng.uniform(0, 2 * np.pi)
        amp = 0.04 + 0.03 * min(round_num, 4)
        return project_feasible(h + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase), n)
    elif strategy == 2:
        s = 0.04 + 0.04 * min(round_num, 3)
        lh = h.copy()
        m = lh < 0.5
        lh[m] = np.maximum(0.0, lh[m] - s)
        lh[~m] = np.minimum(1.0, lh[~m] + s)
        return project_feasible(lh, n)
    elif strategy == 3:
        bl = rng.randint(30, max(50, n // 5))
        st = rng.randint(0, n - bl)
        hp = h.copy()
        hp[st:st + bl] = 1.0 - hp[st:st + bl]
        return project_feasible(hp, n)
    else:
        n2 = n + rng.choice([-150, -80, 80, 150])
        n2 = max(350, min(1500, n2))
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n2)
        return project_feasible(np.interp(x_new, x_old, h), n2)

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_refine(h, n, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in [8000, 40000]:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def _cf(_x):
            return _x.sum() - n * 0.5
        def _cj(_x):
            return np.ones(n)
        try:
            res = minimize(smooth_max_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                           options={'maxiter': 200, 'ftol': 1e-14})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: CD to convergence with random ordering
    h_best, best_c = exact_cd(h_best, n, 120, deadline - 350, nb=40, random_order=True, rng=rng)

    # Phase 2: SLSQP refinement
    if time.time() < deadline - 250:
        hr, cr = slsqp_refine(h_best, n, deadline - 200)
        if cr < best_c:
            h_best, best_c = hr, cr
        h_best, best_c = exact_cd(h_best, n, 80, deadline - 200, nb=40, random_order=True, rng=rng)

    # Phase 3: Diversification loop
    n = len(h_best)
    n_rounds = 0
    while time.time() < deadline - 80 and n_rounds < 25:
        n_rounds += 1
        strat = n_rounds % 5
        hp = perturbation(h_best, n, strat, rng, n_rounds)
        np2 = len(hp)
        # Quick CD then check
        hp2, c2 = exact_cd(hp, np2, 60, deadline - 80, nb=35, random_order=True, rng=rng)
        if c2 < best_c - 1e-9:
            h_best, best_c = hp2, c2
            n = np2
        elif n_rounds > 10:
            # Occasionally try SLSQP on perturbed
            if time.time() < deadline - 120:
                hs, cs = slsqp_refine(hp, np2, deadline - 100)
                hs, cs = exact_cd(hs, np2, 50, deadline - 80, nb=35, random_order=True, rng=rng)
                if cs < best_c - 1e-9:
                    h_best, best_c = hs, cs
                    n = len(h_best)

    # Phase 4: Multi-n exploration
    for n_try in [550, 600, 650, 750, 800, 850, 950, 1050]:
        if time.time() > deadline - 60:
            break
        if n_try == len(h_best):
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 80, deadline - 60, nb=35, random_order=True, rng=rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o

    # Phase 5: Final intensive CD
    if time.time() < deadline - 3:
        h_best, best_c = exact_cd(h_best, len(h_best), 200, deadline, nb=45,
                                  random_order=True, rng=rng)

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
