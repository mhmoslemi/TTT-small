    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Quick SLSQP + CD
    for alpha in [2000, 10000, 50000]:
        if time.time() > deadline - 100:
            break
        best_h = slsqp_step(best_h, n, alpha, 300)
        best_h, best_c = coord_descend(best_h, n, 30, deadline)
    
    # Phase 2: Lag-targeted diverse search
    x = np.linspace(0, 2, n)
    n_restarts = int((deadline - time.time() - 60) / 25)
    n_restarts = max(10, min(n_restarts, 200))
    
    for r in range(n_restarts):
        if time.time() > deadline - 55:
            break
        c = np.correlate(best_h, 1.0 - h, mode='full')
        if time.time() > deadline - 55:
            break
        
        # Get active lags
        c_all = np.correlate(best_h, 1.0 - best_h, mode='full')
        c_max = c_all.max()
        active = np.where(c_all > c_max - 0.001)[0]
        
        pert_type = r % 6
        if pert_type == 0:
            # Small Gaussian
            hp = best_h + rng.normal(0, 0.01, n)
        elif pert_type == 1:
            # Lag-targeted: perturb to reduce active lag correlations
            # Gradient of c[k] w.r.t. h[p] is roughly (1-h[p+k-n+1]) - h[p-k+n-1]
            grad = np.zeros(n)
            for k_idx in active[:5]:
                k = int(k_idx)
                for p in range(n):
                    jf = p + k - (n - 1)
                    jb = p - k + (n - 1)
                    if 0 <= jf < n:
                        grad[p] += (1.0 - best_h[jf])
                    if 0 <= jb < n:
                        grad[p] -= best_h[jb]
            # Normalize and apply
            gn = np.linalg.norm(grad)
            if gn > 1e-10:
                hp = best_h - 0.03 * grad / gn
        elif pert_type == 2:
            # Frequency perturbation
            freq = 1.0 + 2.0 * rng.rand()
            amp = 0.02 + 0.03 * rng.rand()
            hp = best_h + amp * np.sin(np.pi * freq * x + rng.rand() * 2 * np.pi)
        elif pert_type == 3:
            # Asymmetric mass shift
            shift = rng.uniform(-0.1, 0.1)
            hp = np.roll(best_h, int(round(shift * n)))
        elif pert_type == 4:
            # Block perturbation
            block_size = max(10, n // 14)
            start = rng.randint(0, n - block_size)
            hp = best_h.copy()
            hp[start:start+block_size] += rng.uniform(-0.1, 0.1, block_size)
        else:
            # Amplitude modulation
            freq = rng.randint(1, 5)
            hp = best_h * (1.0 + 0.03 * np.sin(np.pi * freq * x))
        
        hp = project_feasible(hp, n)
        # Optimize this perturbation
        hp, c_o = coord_descend(hp, n, 50, deadline, gs_iters=40)
        if c_o < best_c:
            best_h, best_c = hp, c_o
    
    # Phase 3: SLSQP polish on best
    if time.time() < deadline - 30:
        for alpha in [100000, 500000]:
            if time.time() > deadline - 15:
                break
            best_h = slsqp_step(best_h, n, alpha, 400)
            best_h, best_c = coord_descend(best_h, n, 50, deadline, gs_iters=50)
    
    # Phase 4: Final CD with high precision
    if time.time() < deadline - 3:
        best_h, best_c = coord_descend(best_h, n, 80, deadline, gs_iters=60)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
