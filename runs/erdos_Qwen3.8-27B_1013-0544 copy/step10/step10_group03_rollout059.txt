def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):

    vb = (jb >= 0) & (jb < n)

def exact_cd(h, n, max_passes, deadline, random_order=False):

        if time.time() > deadline:
            break
        improved = False
        order = np.random.permutation(n) if random_order else np.arange(n)

                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]

def lp_iter(h, n, deadline, max_iters=50, alpha=0.5):
    """One round of LP-based alternating method."""
    best_h, best_c = h.copy(), compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = 1.0 - h
        h_var = cp.Variable(n)
        L_var = cp.Variable()
        cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        for m in range(2 * n - 1):
            shift = m - (n - 1)
            i_min = max(0, -shift)
            i_max = min(n, n - shift)
            if i_max > i_min:
                c_m = cp.sum(v[i_min:i_max] * h_var[i_min + shift:i_max + shift])
                cons.append(c_m <= L_var)
        prob = cp.Problem(cp.Minimize(L_var), cons)
        try:
            prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        except:
            break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, n)
        h_new = (1.0 - alpha) * h + alpha * h_lp
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
        h = h_new
    return best_h, best_c

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-13 or h0 > 1.0 - 1e-13:
        return h0
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    c_diag = c_base[diag_idx[0]] if has_diag else 0.0
    a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            L_lo = L
            continue
        if np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            L_lo = L
            continue
        if has_diag:
            D = L - c_diag - h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * D
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                # Infeasible only if entire interval is strictly inside (r1, r2)
                if xc_lo > r1 and xc_hi < r2:
                    L_lo = L
                    continue
        L_hi = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        D = L - c_diag - h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * D
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            left_lo, left_hi = x_lo, min(x_hi, r1)
            right_lo, right_hi = max(x_lo, r2), x_hi
            left_feas = left_lo <= left_hi + 1e-14
            right_feas = right_lo <= right_hi + 1e-14
            if left_feas and right_feas:
                if abs(h0 - (left_lo + left_hi) / 2.0) <= abs(h0 - (right_lo + right_hi) / 2.0):
                    x_lo, x_hi = left_lo, left_hi
                else:
                    x_lo, x_hi = right_lo, right_hi
            elif left_feas:
                x_lo, x_hi = left_lo, left_hi
            elif right_feas:
                x_lo, x_hi = right_lo, right_hi
            else:
                return h0
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').copy()
    best_c_local = compute_c5(h, n)
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                diag_i = np.where(ms == (n - 1))[0]
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full').copy()
        c_now = np.max(c) * (2.0 / n)
        if c_now < best_c_local:
            best_c_local = c_now
    return h, compute_c5(h, n)

def lp_alternating(h, n, deadline, max_iters=20, alpha=0.6):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = 1.0 - h
        h_var = cp.Variable(n)
        L_var = cp.Variable()
        cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        for m in range(2 * n - 1):
            shift = m - (n - 1)
            i_min = max(0, -shift)
            i_max = min(n, n - shift)
            if i_max > i_min:
                c_m = cp.sum(cp.multiply(v[i_min:i_max], h_var[i_min + shift:i_max + shift]))
                cons.append(c_m <= L_var)
        prob = cp.Problem(cp.Minimize(L_var), cons)
        try:
            prob.solve(solver=cp.SCS, max_iters=3000, verbose=False)
        except Exception:
            break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, n)
        h_new = (1.0 - alpha) * h + alpha * h_lp
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
        h = h_new
    return best_h, best_c

def structured_h(n, variant):
    x = np.linspace(0, 2, n, endpoint=False)
    if variant == 0:
        h = np.full(n, 0.5)
    elif variant == 1:
        h = 0.5 + 0.35 * np.cos(np.pi * x)
    elif variant == 2:
        h = 0.5 + 0.35 * np.cos(2.0 * np.pi * x)
    elif variant == 3:
        h = 0.5 + 0.4 * np.sin(np.pi * x)
    elif variant == 4:
        h = np.where(x < 1.0, 0.7, 0.3)
    elif variant == 5:
        h = 0.5 + 0.3 * np.cos(np.pi * x + 0.7)
    elif variant == 6:
        h = 0.5 + 0.25 * np.cos(3.0 * np.pi * x)
    else:
        h = np.full(n, 0.5)
    return project_feasible(np.clip(h, 0.0, 1.0), n)

def binary_push(h, n, strength):
    lh = h.copy()
    mask_low = lh < 0.5
    mask_high = lh >= 0.5
    lh[mask_low] = np.maximum(0.0, lh[mask_low] - strength)
    lh[mask_high] = np.minimum(1.0, lh[mask_high] + strength)
    return project_feasible(lh, n)
