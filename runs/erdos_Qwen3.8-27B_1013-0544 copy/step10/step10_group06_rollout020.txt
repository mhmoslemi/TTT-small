    """Solve LP to find direction that reduces max correlation."""
    c = np.correlate(h, 1.0 - h, mode='full')
    # Compute gradients for active lags
    # grad[k, p] = ∂c[k]/∂h[p]
    grads = np.zeros((len(active_indices), n))
    for idx, k in enumerate(active_indices):
        d = k - (n - 1)
        for p in range(n):
            g = 0.0
            if 0 <= p + d < n:
                g += (1.0 - h[p + d])
            if 0 <= p - d < n:
                g -= h[p - d]
            if d == 0:
                g += (1.0 - 2.0 * h[p])
            grads[idx, p] = g
    
    # LP: min t
    # s.t. grad[k,:] @ delta - t <= -c[k]  for each active k
    #      -1 <= h[p] + delta[p] <= 1  => bounds on delta
    #      sum(delta) = 0
    #      |delta[p]| <= step_cap  (for stability, optional)
    
    # Variables: delta[0..n-1], t  => n+1 variables
    # c_obj = [0, 0, ..., 0, 1] (minimize t)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active k: grad[k,:] @ delta - t <= -c[k]
    A_ub = np.zeros((len(active_indices), n + 1))
    A_ub[:, :n] = grads
    A_ub[:, -1] = -1.0
    b_ub = -c[active_indices]
    
    # Bounds: -h[p] <= delta[p] <= 1-h[p], and -step_cap <= delta[p] <= step_cap
    # Combined: max(-h[p], -step_cap) <= delta[p] <= min(1-h[p], step_cap)
    lb = np.maximum(-h, -step_cap)
    ub = np.minimum(1 - h, step_cap)
    bounds = list(zip(lb, ub)) + [(None, None)]  # t is unbounded
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:
        delta = res.x[:n]
        t_opt = res.x[-1]
        return delta, t_opt
    return None, None

def compute_c5(h, n):

def coord_descent(h, n, max_passes, deadline, rng, random_order=True, nbisect=40):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]

            x_new = solve_coord(c, h, p, n, nbisect)

                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def compute_grads(h, n, active_idxs):
    """Compute gradients of c[active_idxs] w.r.t. h for all p."""
    grads = np.zeros((len(active_idxs), n))
    for idx, k in enumerate(active_idxs):
        d = k - (n - 1)
        for p in range(n):
            g = 0.0
            pd = p + d
            pb = p - d
            if 0 <= pd < n:
                g += (1.0 - h[pd])
            if 0 <= pb < n:
                g -= h[pb]
            if d == 0:
                g += (1.0 - 2.0 * h[p])
            grads[idx, p] = g
    return grads

def lp_direction(h, n, active_idxs, step_cap):
    """Find direction via LP to reduce max correlation."""
    c = np.correlate(h, 1.0 - h, mode='full')
    grads = compute_grads(h, n, active_idxs)
    c_active = c[active_idxs]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((len(active_idxs), n + 1))
    A_ub[:, :n] = grads
    A_ub[:, -1] = -1.0
    b_ub = -c_active
    lb = np.maximum(-h, -step_cap)
    ub = np.minimum(1.0 - h, step_cap)
    bounds = list(zip(lb, ub)) + [(None, None)]

    return None, None

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def lse_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        d = k - (n - 1)
        for i in range(n):
            pd = i + d
            pb = i - d
            if 0 <= pd < n:
                grad[i] += w * (1.0 - x[pd]) * dx
            if 0 <= pb < n:
                grad[i] -= w * x[pb] * dx
            if d == 0:
                grad[i] += w * (1.0 - 2.0 * x[i]) * dx
    return lse_obj(x, n, alpha), grad

    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), jac=lse_grad,
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n / 2.0,
                                         'jac': lambda n=n: np.ones(n)}],
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:

        except Exception:

    return best_h, best_c

    h3, c3 = lp_minmax_iter(best_h, n_init, deadline, max_iters=150, K=30)
    if c3 < best_c:
        best_h, best_c = h3, c3
    # Phase 4: CD refinement
    h4, c4 = coord_descent(best_h, n_init, 60, deadline, rng, True, 40)
    if c4 < best_c:
        best_h, best_c = h4, c3 if c4 < c3 else c4
        best_c = min(best_c, c4)
    # Phase 5: LP with smaller K (focus on top correlations)
    if time.time() < deadline - 100:
        h5, c5v = lp_minmax_iter(best_h, n_init, deadline, max_iters=100, K=15)
        if c5v < best_c:
            best_h, best_c = h5, c5v
    # Phase 6: SLSQP with high alpha
    if time.time() < deadline - 80:
        h6, c6 = slsqp_run(best_h, n_init, [5000, 15000, 50000], 200, deadline)
        if c6 < best_c:
            best_h, best_c = h6, c6
    # Phase 7: Try different grid sizes
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [840, 560, 1000]:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, 30, deadline, rng, True, 35)
        if c_opt < best_c:
            h_opt, c_opt = lp_minmax_iter(h_opt, n_try, deadline, max_iters=50, K=25)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
                x_orig = x_new
    # Phase 8: Final LP + CD polish
    if time.time() < deadline - 30:
        h_f, c_f = lp_minmax_iter(best_h, len(best_h), deadline, max_iters=50, K=20)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    if time.time() < deadline - 10:
        remaining = deadline - time.time() - 3
        max_p = max(10, int(remaining / 1.0))
        h_pf, c_pf = coord_descent(best_h, len(best_h), max_p, deadline, rng, False, 45)
        if c_pf < best_c:
            best_h, best_c = h_pf, c_pf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
