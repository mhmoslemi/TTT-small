import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_i = np.where(ms == (n - 1))[0]
    has_diag = len(diag_i) > 0
    if has_diag:
        c_diag = c_base[diag_i[0]]
        a[diag_i] = 0.0
    
    # Unaffected correlations
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    
    cur_max = float(np.max(c))
    L_lo = unaff
    L_hi = cur_max + 0.3  # Give room above current max
    
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    
    for _ in range(60):
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        
        if np.any(zero_mask):
            if float(np.max(c_base[zero_mask])) > L:
                feasible = False
        
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        
        if x_lo > x_hi + 1e-12:
            feasible = False
        
        if feasible and has_diag:
            # Diagonal: new_val = c_diag + x*(1-x) - h0*(1-h0) ≤ L
            # => x - x² ≤ L - c_diag + h0 - h0²
            # => x² - x + (L - c_diag + h0 - h0²) ≥ 0
            K = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * K
            if disc < 0:
                # x² - x + K < 0 for all x means the parabola is always negative
                # which means x - x² > K for all x, so the constraint is violated
                feasible = False
            else:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo = max(x_lo, 0.0)
                xc_hi = min(x_hi, 1.0)
                # Need a point in [xc_lo, xc_hi] that is < r1 or > r2
                if xc_lo > xc_hi:
                    feasible = False
                elif not (xc_lo < r1 or xc_hi > r2):
                    feasible = False
        
        if feasible:
            L_hi = L
        else:
            L_lo = L
    
    L = L_hi
    # Find best x
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    
    if has_diag:
        K = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * K
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            # Pick x in [x_lo, x_hi] that is outside [r1, r2]
            # Prefer the point that minimizes x*(1-x) (i.e., closest to 0 or 1)
            candidates = []
            if x_lo < r1:
                candidates.append(max(x_lo, 0.0))
            if x_hi > r2:
                candidates.append(min(x_hi, 1.0))
            if not candidates:
                # Fallback: pick boundary closest to outside
                x_lo_c = max(x_lo, 0.0)
                x_hi_c = min(x_hi, 1.0)
                if x_lo_c <= x_hi_c:
                    # Pick the one with smaller x*(1-x)
                    if x_lo_c * (1 - x_lo_c) <= x_hi_c * (1 - x_hi_c):
                        candidates.append(x_lo_c)
                    else:
                        candidates.append(x_hi_c)
            if candidates:
                # Pick the one that minimizes x*(1-x)
                x = min(candidates, key=lambda v: v * (1 - v))
            else:
                x = np.clip(h0, 0, 1)
        else:
            x = np.clip(h0, 0, 1)
    else:
        x_lo = max(x_lo, 0.0)
        x_hi = min(x_hi, 1.0)
        if x_lo <= x_hi:
            x = (x_lo + x_hi) / 2.0
        else:
            x = np.clip(h0, 0, 1)
    
    return float(np.clip(x, 0.0, 1.0))

    c = np.correlate(h, 1.0 - h, mode='full')
    
    for _pass in range(max_passes):

            break
        improved = False

                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        # Project to maintain feasibility
        h = project_feasible(h, n)

        if abs(h.sum() - n / 2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
        else:
            # Check if projection changed h
            # (We can't easily track this without storing the old h, so just recompute periodically)
            pass
    
    c5 = compute_c5(h, n)
    return h, c5

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_i = np.where(ms == (n - 1))[0]
    has_diag = len(diag_i) > 0
    if has_diag:
        c_diag = c_base[diag_i[0]]
        a[diag_i] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    L_lo = unaff
    L_hi = cur_max
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    for _ in range(55):
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and float(np.max(c_base[zero_mask])) > L + 1e-16:
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            K = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * K
            if disc < -1e-14:
                feasible = False
            else:
                disc = max(disc, 0.0)
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo = max(x_lo, 0.0)
                xc_hi = min(x_hi, 1.0)
                if xc_lo > xc_hi + 1e-14:
                    feasible = False
                elif not (xc_lo < r1 - 1e-14 or xc_hi > r2 + 1e-14):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    if has_diag:
        K = L - c_diag + h0 - h0 * h0
        disc = max(1.0 - 4.0 * K, 0.0)
        r1 = (1.0 - np.sqrt(disc)) / 2.0
        r2 = (1.0 + np.sqrt(disc)) / 2.0
        xc_lo = max(x_lo, 0.0)
        xc_hi = min(x_hi, 1.0)
        cands = []
        if xc_lo < r1:
            cands.append(xc_lo)
        if xc_hi > r2:
            cands.append(xc_hi)
        if not cands:
            if xc_lo <= xc_hi:
                cands.append(xc_lo if xc_lo * (1 - xc_lo) <= xc_hi * (1 - xc_hi) else xc_hi)
            else:
                cands.append(h0)
        x = min(cands, key=lambda v: v * (1.0 - v))
    else:
        x_lo = max(x_lo, 0.0)
        x_hi = min(x_hi, 1.0)
        x = (x_lo + x_hi) / 2.0 if x_lo <= x_hi else h0
    return float(np.clip(x, 0.0, 1.0))

def exact_cd(h, n, max_passes=200, deadline=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').copy()
    for p in range(n):
        h0 = h[p]
        if h0 < 1e-14 or h0 > 1.0 - 1e-14:
            continue
        xn = solve_1d(c, h, p, n)
        if abs(xn - h0) > 1e-13:
            delta = xn - h0
            m_min = n - 1 - p
            m_max = 2 * n - 2 - p
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            diag_i = np.where(ms == (n - 1))[0]
            c[ms] += delta * a
            if len(diag_i) > 0:
                c[ms[diag_i[0]]] -= delta * delta
            h[p] = xn
    h = project_feasible(h, n)
    c5 = compute_c5(h, n)
    return h, c5

def cd_full(h, n, max_passes, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        h, c5 = exact_cd(h, n, 1, None)
        if c5 < best_c - 1e-15:
            best_c = c5
        elif c5 >= best_c - 1e-15:
            # Check if we've converged
            h_old = h.copy()
            h, c5b = exact_cd(h, n, 1, None)
            if np.max(np.abs(h - h_old)) < 1e-14:
                break
    return h, best_c
