```python
import cvxpy as cp

def sca_step(h, n, solver='ECOS'):
    """One SCA iteration: linearize and solve LP."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute gradient of c[m] w.r.t. h for each m
    # c[m] = sum_i h[i] * (1-h[i+m-(n-1)]) for valid pairs
    # grad[m, j] = (1-h[j+s]) if valid - h[j-s] if valid, where s = m-(n-1)
    
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2.0]
    
    # For each lag m, add constraint: linearized_c[m] <= t
    # linearized_c[m] = c[m] + sum_j grad[m,j] * (h_var[j] - h[j])
    # = (c[m] - sum_j grad[m,j]*h[j]) + grad[m,:] @ h_var
    
    for m in range(2*n - 1):
        s = m - (n-1)
        # grad[m, j] for each j
        grad = np.zeros(n)
        # forward: (1-h[j+s]) for 0<=j<n and 0<=j+s<n
        jf = np.arange(n)
        valid_f = (jf + s >= 0) & (jf + s < n)
        grad[valid_f] += (1.0 - h[jf[valid_f] + s])
        # backward: -h[j-s] for 0<=j-s<n
        valid_b = (jf - s >= 0) & (jf - s < n)
        grad[valid_b] -= h[jf[valid_b] - s]
        
        # linearized_c[m] = c[m] - grad @ h + grad @ h_var
        const = c[m] - grad @ h
        constraints.append(const + grad @ h_var <= t)
    
    obj = cp.Minimize(t * dx)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=solver)
    
    if h_var.value is not None:
        h_new = np.clip(h_var.value, 0, 1)
        h_new = project_feasible(h_new, n)
        return h_new
    return h
```

```python
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

    # where const[m] = c[m] - grad[m,:] @ h

    # grad[m, j] for m in range(2n-1), j in range(n)
```

For lag m with shift s = m - (n-1):
grad[m, j] = (1 - h[j+s]) * I(0≤j+s<n) - h[j-s] * I(0≤j-s<n)

This can be written as:

where f_forward(j, s) = 1 - h[j+s] if 0≤j+s<n, else 0
and f_backward(j, s) = h[j-s] if 0≤j-s<n, else 0

For fixed s, this is just a shifted version of h (or 1-h). So grad[m, :] is:
- The vector (1-h[s: n]) padded with zeros on the left (for j such that j+s < n, i.e., j < n-s when s>0)
  Minus
- The vector h[:n-s] shifted right by s (for j such that j-s ≥ 0, i.e., j ≥ s when s>0)

This can be computed efficiently using numpy slicing.

```python
    for m in range(2*n-1):
        s = m - (n-1)
        if s >= 0:
            # forward: 1-h[j+s] for j in [0, n-s-1]
            grad_mat[m, :n-s] += (1.0 - h[s:])
            # backward: -h[j-s] for j in [s, n-1]
            grad_mat[m, s:] -= h[:n-s]
        else:
            # s < 0, let t = -s
            t = -s
            # forward: 1-h[j-t] for j in [t, n-1]
            grad_mat[m, t:] += (1.0 - h[:-t])
            # backward: -h[j+t] for j in [0, n-t-1]
            grad_mat[m, :n-t] -= h[t:]
    const_vec = c - grad_mat @ h
    return grad_mat, const_vec
```

```python
            scale = 0.005 + 0.015 * (r / max(1, n_restarts - 1))
            if r % 3 == 0:
                hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            elif r % 3 == 1:
                x = np.linspace(0, 2, n)
                hp = project_feasible(best_h + scale * np.sin(np.pi * (2 + r) * x), n)
            else:
                block = max(1, n // 5)
                hp = best_h.copy()
                start = (r * block) % max(1, n - block)
                hp[start:start + block] = 1.0 - hp[start:start + block]
                hp = project_feasible(hp, n)
            h_o, c_o = coord_descent(hp, n, 60, deadline - 60, gs_iters=50)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
                # SCA polish on the new best
                h_new = sca_lp_step(best_h, n)
                if h_new is not None:
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c:
                        best_h, best_c = h_new, c_new
                    best_h, best_c = coord_descent(best_h, n, 50, deadline - 60, gs_iters=55)

    # Phase 6: Final polish
    if time.time() < deadline - 10:
        h_f, c_f = coord_descent(best_h, n, 150, deadline - 5, gs_iters=60)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_new = sca_lp_step(best_h, n)
        if h_new is not None:
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_h, best_c = h_new, c_new
                best_h, best_c = coord_descent(best_h, n, 50, deadline - 5, gs_iters=60)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
