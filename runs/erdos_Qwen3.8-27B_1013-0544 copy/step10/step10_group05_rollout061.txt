            tot = room.sum()

            tot = h.sum()

                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

def smooth_obj_and_grad(x, n, alpha):

    c = np.correlate(x, 1.0 - x, mode='full')
    c_max = c.max()
    shifted = alpha * (c - c_max)
    shifted -= shifted.max()
    e = np.exp(shifted)
    lse = c_max + (shifted.max() + np.log(e.sum())) / alpha
    obj = lse * dx
    softmax = e / e.sum()

        m_max = min(2 * n - 2, 2 * n - 2 - p)

        jb = p - ms + (n - 1)

        vb = (jb >= 0) & (jb < n)

        if np.any(vf):

    JB = P - M + (n - 1)
    VF = (JF >= 0) & (JF < n)
    VB = (JB >= 0) & (JB < n)
    HJF = np.where(VF, np.clip(JF, 0, n - 1), 0)
    HJB = np.where(VB, np.clip(JB, 0, n - 1), 0)
    return (1.0 - h[HJF]) * VF - h[HJB] * VB

def smooth_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    c_max = c.max()
    shifted = alpha * (c - c_max)
    shifted_max = shifted.max()
    e = np.exp(shifted - shifted_max)
    lse = c_max + (shifted_max + np.log(e.sum())) / alpha
    obj = lse * dx
    w = e / e.sum()
    A = build_A(x, n)
    grad = dx * (A @ w)
    return obj, grad
