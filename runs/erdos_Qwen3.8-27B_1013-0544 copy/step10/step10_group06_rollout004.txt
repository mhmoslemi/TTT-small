import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = s
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h


def anti_symmetrize(h, n):
    h = 0.5 * (h + 1.0 - h[::-1])
    if n % 2:
        h = h.copy()
        h[n // 2] = 0.5
    return np.clip(h, 0.0, 1.0)


def build_anti(z, n):
    z = np.clip(np.asarray(z, dtype=float), 0.0, 1.0)
    h = np.empty(n)
    m = n // 2
    h[:m] = z
    if n % 2:
        h[m] = 0.5
        h[m + 1:] = 1.0 - z[::-1]
    else:
        h[m:] = 1.0 - z[::-1]
    return h


def resample_h(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return project_feasible(np.interp(x_new, x_old, h), n_new)


def single_shift_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    a = max(0, k)
    b = min(n, n + k)
    if a < b:
        g[a:b] += 1.0 - h[a - k:b - k]
    a = max(0, -k)
    b = min(n, n - k)
    if a < b:
        g[a:b] -= h[a + k:b + k]
    return g


def smooth_obj_grad_h(h, n, alpha):
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(c))
    shifted = alpha * (c - m)
    raw = np.exp(shifted)
    sraw = float(np.sum(raw))
    dx = 2.0 / n
    f = dx * (m + np.log(sraw) / alpha)
    off = n - 1
    w = raw / sraw
    sig = np.where(w > 1e-10)[0]
    if sig.size == 0:
        idx = int(np.argmax(c))
        g = single_shift_grad(h, idx - off)
    elif sig.size <= 30:
        ws = w[sig].copy()
        ws /= ws.sum()
        g = np.zeros(n)
        for j in range(sig.size):
            g += ws[j] * single_shift_grad(h, int(sig[j]) - off)
    else:
        cv = np.convolve(w, 1.0 - h, mode='full')
        ch = np.convolve(w[::-1], h, mode='full')
        g = cv[off:off + n] - ch[off:off + n]
    return f, dx * g


def anti_obj(z, n, alpha):
    h = build_anti(z, n)
    f, g = smooth_obj_grad_h(h, n, alpha)
    m = n // 2
    if n % 2:
        gz = g[:m] - g[m + 1:][::-1]
    else:
        gz = g[:m] - g[m:][::-1]
    return f, gz


def optimize_anti(h, n, alphas, maxiter, deadline):
    h = anti_symmetrize(project_feasible(h, n), n)
    best_h = h.copy()
    best_c = compute_c5(h)
    m = n // 2
    if m == 0:
        return best_h, best_c
    z = np.clip(h[:m], 0.0, 1.0).copy()
    best_z = z.copy()
    bounds = [(0.0, 1.0)] * m
    for al in alphas:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(anti_obj, z, args=(n, al), method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
            if not np.all(np.isfinite(res.x)):
                zc = z.copy()
            else:
                zc = np.clip(res.x, 0.0, 1.0).copy()
        except Exception:
            zc = z.copy()
        hc = build_anti(zc, n)
        cc = compute_c5(hc)
        if cc < best_c:
            best_h = hc.copy()
            best_c = cc
            best_z = zc.copy()
            z = zc
        else:
            z = zc if cc < best_c + 2e-5 else best_z.copy()
    return best_h, best_c

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=True, nbis=18, sym=False):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h)
    stag = 0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n) if (random_order and rng is not None) else np.arange(n)
        improved = False
        for p in order:
            if time.time() > deadline:
                break
            x0 = h[p]
            nx = solve_coord(c, h, p, n, nbis)
            if abs(nx - x0) > 1e-14:
                h[p] = nx
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if sym:
            h = anti_symmetrize(h, n)
        cn = compute_c5(h)
        if cn < best_c - 1e-13:
            best_c = cn
            best_h = h.copy()
            h = best_h.copy()
            stag = 0
        else:
            h = best_h.copy()
            stag += 1
            if stag > 12:
                break
    return best_h, best_c


def optimize_candidate(h, n, alloc, rng, main):
    ts = time.time()

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h)

    d = min(end, ts + 0.32 * alloc)
    if d > time.time() + 2:
        h1, c1 = optimize_anti(best_h, n, [20, 80, 300, 1200, 5000], 180, d)
        if c1 < best_c:
            best_h, best_c = h1, c1

    d = min(end, ts + 0.65 * alloc)
    if d > time.time() + 2:
        h2, c2 = coord_descent(best_h, n, 10000, d, rng, True, 18, False)
        if c2 < best_c:
            best_h, best_c = h2, c2

    d = min(end, ts + 0.82 * alloc)
    if d > time.time() + 2:
        hs = anti_symmetrize(best_h, n)
        cs = compute_c5(hs)
        if cs < best_c:
            best_h, best_c = hs, cs
        h3, c3 = coord_descent(best_h, n, 5000, d, rng, False, 20, True)
        if c3 < best_c:
            best_h, best_c = h3, c3

    d = min(end, ts + 0.96 * alloc)
    if d > time.time() + 2:
        h4, c4 = coord_descent(best_h, n, 5000, d, rng, True, 18, False)
        if c4 < best_c:
            best_h, best_c = h4, c4

    return best_h, best_c


def random_perturb(h, rng, amp):
    n = len(h)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    y = np.zeros(n)
    a = rng.uniform(0.02, amp)
    for f in (1, 2, 3, 5, 8):
        y += a * rng.uniform(-1.0, 1.0) * np.sin(np.pi * f * x + rng.uniform(0.0, 2.0 * np.pi))
    return project_feasible(h + y, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)

    try:
        hp = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        nf = 700
        xf = (np.arange(nf) + 0.5) * (2.0 / nf)
        hp = np.clip(0.5 + 0.25 * np.sin(np.pi * xf) + 0.1 * np.sin(3.0 * np.pi * xf), 0.0, 1.0)

    hp = project_feasible(hp, len(hp))
    best_h = hp.copy()
    best_c = compute_c5(best_h)
    ha = anti_symmetrize(hp, len(hp))
    ca = compute_c5(ha)
    if ca < best_c:
        best_h, best_c = ha, ca

    n0 = len(best_h)
    rem = deadline - time.time()
    if rem > 30:
        alloc = min(350.0, max(60.0, rem - 50.0), rem - 5.0)
        h1, c1 = optimize_candidate(best_h, n0, alloc, rng, True)
        if c1 < best_c:
            best_h, best_c = h1, c1

    rem = deadline - time.time()
    if rem > 40:
        cands = []
        seen = set()
        for nn in (n0 - 50, n0 - 100, n0 + 50, n0 + 100, int(0.85 * n0), int(1.15 * n0), 560, 640, 760, 840):
            if 240 <= nn <= 1000 and nn not in seen and abs(nn - len(best_h)) >= 10:
                seen.add(nn)
                cands.append(nn)
        for nn in cands:
            rem = deadline - time.time()
            if rem < 35:
                break
            if nn == len(best_h):
                continue
            alloc = min(80.0, max(25.0, rem * 0.20), rem - 10.0)
            h_try = resample_h(best_h, nn)
            h2, c2 = optimize_candidate(h_try, nn, alloc, rng, False)
            if c2 < best_c:
                best_h, best_c = h2, c2

    rem = deadline - time.time()
    if rem > 30:
        ncur = len(best_h)
        for _ in range(3):
            rem = deadline - time.time()
            if rem < 25:
                break
            ep = min(deadline - 5, time.time() + min(25.0, rem * 0.25))
            h_p = random_perturb(best_h, rng, 0.08)
            h3, c3 = coord_descent(h_p, ncur, 1000, ep, rng, True, 16, False)
            if c3 < best_c:
                best_h, best_c = h3, c3
            hs = anti_symmetrize(h3, ncur)
            c4 = compute_c5(hs)
            if c4 < best_c:
                best_h, best_c = hs, c4
            h5, c5 = coord_descent(hs, ncur, 500, deadline - 5, rng, False, 16, True)
            if c5 < best_c:
                best_h, best_c = h5, c5
            if deadline - time.time() < 15:
                break

    if deadline - time.time() > 20:
        h6, c6 = coord_descent(best_h, len(best_h), 2000, deadline - 3, rng, True, 20, False)
        if c6 < best_c:
            best_h, best_c = h6, c6
        hs = anti_symmetrize(best_h, len(best_h))
        c7 = compute_c5(hs)
        if c7 < best_c:
            best_h, best_c = hs, c7
        if deadline - time.time() > 10:
            h8, c8 = coord_descent(best_h, len(best_h), 1000, deadline - 2, rng, False, 20, True)
            if c8 < best_c:
                best_h, best_c = h8, c8

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h)
    return best_h, best_c, len(best_h)
