        disc = a[di]**2 - 4*(L - c_base[di])
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (a[di] - sq) / 2
            r2 = (a[di] + sq) / 2
            if xl <= r1:
                xh = min(xh, r1)
            elif xh >= r2:
                xl = max(xl, r2)
            else:
                if abs(xl - r1) <= abs(xh - r2):
                    xh = r1
                else:
                    xl = r2
    if xl > xh:
        return h0
    return float(np.clip((xl + xh) / 2.0, 0.0, 1.0))

def cd_update(c, h, p, n, delta):
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c[ms] += delta * a
    diag_i = np.where(ms == (n - 1))[0]
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta

        sg[valid_f] += (1.0 - h[fwd[valid_f]])

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_alpha_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    sh = sh - sh.max()
    return (m + np.log(np.sum(np.exp(sh))) / alpha) * (2.0 / n)

def smooth_alpha_grad(x, n, alpha):

    m = c.max()
    sh = alpha * (c - m)
    sh = sh - sh.max()

    idx = np.arange(n)
    grad = np.zeros(n)
