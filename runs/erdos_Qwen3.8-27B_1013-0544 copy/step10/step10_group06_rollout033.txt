def coord_descent(h, n, max_passes, deadline, rng=None, random_order=True, nbisect=40):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord_fast(c, h, p, n, nbisect)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)


def adam_optimize(h, n, alpha, steps, lr, deadline, rng):
    """Adam optimizer with LSE-smoothed objective."""
    h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    for t in range(1, steps + 1):
        if time.time() > deadline:
            break
        val, grad = lse_objective_and_grad(h, n, alpha)
        grad -= np.mean(grad)  # maintain sum constraint approximately
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n)
    
    return h

def lse_gradient_fast(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    # For each lag ell = k - (n-1), we need:
    # term1[ell]: (1-h) shifted so that index i gets (1-h[i-ell])
    # term2[ell]: h shifted so that index i gets h[i+ell]
    
    # Build using numpy roll with padding
    grad = np.zeros(n)
    
    for k in range(n_lags):
        if weights[k] < 1e-15:
            continue
        ell = k - (n - 1)
        w = weights[k] * dx
        
        # term1: (1-h[i-ell]) for valid i-ell
        if ell >= 0:
            # i-ell >= 0 → i >= ell
            idx = np.arange(ell, n)
            src = np.arange(0, n - ell)
            grad[idx] += w * (1.0 - x[src])
        else:
            # i-ell < n → i < n+ell = n-|ell|
            idx = np.arange(0, n + ell)
            src = np.arange(-ell, n)
            grad[idx] += w * (1.0 - x[src])
        
        # term2: -h[i+ell] for valid i+ell
        if ell >= 0:
            # i+ell < n → i < n-ell
            idx = np.arange(0, n - ell)
            src = np.arange(ell, n)
            grad[idx] -= w * x[src]
        else:
            # i+ell >= 0 → i >= -ell = |ell|
            idx = np.arange(-ell, n)
            src = np.arange(0, n + ell)
            grad[idx] -= w * x[src]
    
    return grad

import time

def project_feasible(h, n):

        else:
            tot = h.sum()
            if tot < 1e-15:

def lse_obj_and_grad(x, n, alpha):

    m = float(c.max())

    mx = shifted.max()

    weights = exps / wsum
    val = (m + (mx + np.log(wsum)) / alpha) * (2.0 / n)
    dx = 2.0 / n

        w = weights[k] * dx
        if ell >= 0:
            grad[ell:] += w * (1.0 - x[:n - ell])

        else:
            e = -ell
            grad[:n - e] += w * (1.0 - x[e:])
            grad[e:] -= w * x[:n - e]
    return val, grad

def sum_con(x, n):

def solve_coord_fast(c, h, p, n, nbisect=40):

        j2 = p + ell

                c_base[idx] -= h0 * (1.0 - h0)

                c_base[idx] -= h0 * (1.0 - h[j1])
                a[idx] += (1.0 - h[j1])
        if 0 <= j2 < n and j2 != p:
            c_base[idx] += h[j2] * h0
            a[idx] -= h[j2]
    affected = np.zeros(2 * n - 1, dtype=bool)

def coord_descent(h, n, max_passes, deadline, rng, random_order=True, nbisect=40):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n) if (random_order and rng is not None) else np.arange(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord_fast(c, h, p, n, nbisect)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def adam_optimize(h, n, alpha, steps, lr, deadline):
    h = h.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    for t in range(1, steps + 1):
        if time.time() > deadline:
            break
        val, grad = lse_obj_and_grad(h, n, alpha)
        grad -= np.mean(grad)
        m_v = b1 * m_v + (1.0 - b1) * grad
        v_v = b2 * v_v + (1.0 - b2) * grad * grad
        m_hat = m_v / (1.0 - b1 ** t)
        v_hat = v_v / (1.0 - b2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n)
    return h

def slsqp_optimize(h, n, alphas, maxiter_each, deadline):

                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    remaining = deadline - time.time() - 2
    if remaining > 5:
        max_p = max(10, int(remaining / 0.4))
        h_f, c_f = coord_descent(best_h, n_cur, max_p, deadline, rng, False, 45)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
