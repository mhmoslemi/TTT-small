from scipy.fft import fft, ifft
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_box_sum(v, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(64):
        tau = (lo + hi) / 2.0
        h = np.clip(v - tau, 0.0, 1.0)
        if h.sum() > target:
            lo = tau
        else:
            hi = tau
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def fft_convolve(a, b):
    N = 1
    while N < len(a) + len(b) - 1:
        N *= 2
    fa = fft(a, N)
    fb = fft(b, N)
    return ifft(fa * fb).real[:len(a) + len(b) - 1]

def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    sm = alpha * c - alpha * c.max()
    w = np.exp(sm)
    w /= w.sum()
    Wp = w[n-1:]
    A = 1.0 - h
    conv1 = fft_convolve(Wp[::-1], A)
    conv2 = fft_convolve(Wp, h)
    return (conv1[n-1:2*n-1] - conv2[:n]) * (2.0 / n)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_box_sum(v, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(64):
        tau = (lo + hi) / 2.0
        s = np.clip(v - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def fft_conv(a, b):
    sz = 1
    while sz < len(a) + len(b) - 1:
        sz *= 2
    return ifft(fft(a, sz) * fft(b, sz)).real[:len(a) + len(b) - 1]

def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    sm = alpha * (c - c.max())
    w = np.exp(sm)
    w /= w.sum()
    Wp = w[n - 1:]
    conv1 = fft_conv(Wp[::-1], 1.0 - h)
    conv2 = fft_conv(Wp, h)
    return (conv1[n - 1:2 * n - 1] - conv2[:n]) * (2.0 / n)
