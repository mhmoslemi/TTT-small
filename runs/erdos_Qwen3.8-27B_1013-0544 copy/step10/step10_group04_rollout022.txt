        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if (xc_lo < r1 and xc_hi > r2) or (x_lo >= r1 and x_hi <= r2):
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, rng=None, random_order=True):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def binary_snap(h, n, threshold):
    h = h.copy()
    h[h < threshold] = 0.0
    h[h > 1.0 - threshold] = 1.0
    return project_feasible(h, n)

def make_init(n, kind, rng):
    h = np.zeros(n)
    if n % 2 == 0:
        half = n // 2
    else:
        half = n // 2
    if kind == 'block':
        h[:half] = 1.0
    elif kind == 'alt':
        h[::2] = 1.0
        if h.sum() > half:
            h[-1] = 0.0
    elif kind == 'periodic5':
        for i in range(n):
            h[i] = 1.0 if (i % 5) < 2 else 0.0
    elif kind == 'fourier2':
        x = np.linspace(0, 2, n, endpoint=False)
        h[:] = (np.sin(np.pi * x) > 0).astype(float)
    elif kind == 'random':
        h[:half] = 1.0
        h = h[rng.permutation(n)]
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        n = 700
        h_best = make_init(n, 'random', rng)

    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: CD on given initial
    h1, c1 = exact_cd(h_best, n, 250, deadline, rng=rng)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Binary snap with annealing thresholds (key innovation)
    thresholds = [0.25, 0.20, 0.15, 0.12, 0.08, 0.05, 0.03]
    for thresh in thresholds:
        if time.time() > deadline - 80:
            break
        hs = binary_snap(h_best, n, thresh)
        ho, co = exact_cd(hs, n, 150, deadline, rng=rng)
        if co < best_c:
            h_best, best_c = ho, co

    # Phase 3: Aggressive binary + CD (force near-binary)
    if time.time() < deadline - 60:
        # Try to find the best pure binary pattern via CD from binary starts
        for init_type in ['random', 'random', 'periodic5', 'block', 'alt']:
            if time.time() > deadline - 40:
                break
            hb = make_init(n, init_type, rng)
            ho, co = exact_cd(hb, n, 120, deadline, rng=rng)
            if co < best_c:
                h_best, best_c = ho, co

    # Phase 4: Noise perturbation at multiple scales
    for noise in [0.15, 0.08, 0.04, 0.02]:
        if time.time() > deadline - 40:
            break
        for trial in range(2):
            if time.time() > deadline - 20:
                break
            hp = project_feasible(h_best + rng.normal(0, noise, n), n)
            ho, co = exact_cd(hp, n, 150, deadline, rng=rng)
            if co < best_c:
                h_best, best_c = ho, co

    # Phase 5: Interpolation to different n and back
    if time.time() < deadline - 120:
        for n_try in [n - 50, n + 50]:
            if n_try < 100 or n_try > 1500:
                continue
            if time.time() > deadline - 60:
                break
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_int = project_feasible(np.interp(x_new, x_old, h_best, period=2), n_try)
            ho, co = exact_cd(h_int, n_try, 100, deadline, rng=rng)
            if co < best_c - 1e-7:
                h_best, best_c, n = ho, co, n_try

    # Phase 6: Final long CD polish
    n = len(h_best)
    if time.time() < deadline - 10:
        hf, cf = exact_cd(h_best, n, 400, deadline, rng=rng)
        if cf < best_c:
            h_best, best_c = hf, cf

    # Phase 7: One more snap+CD cycle
    if time.time() < deadline - 15:
        for thresh in [0.06, 0.04]:
            if time.time() > deadline - 5:
                break
            hs = binary_snap(h_best, n, thresh)
            ho, co = exact_cd(hs, n, 100, deadline, rng=rng)
            if co < best_c:
                h_best, best_c = ho, co

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
