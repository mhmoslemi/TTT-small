def smooth_c_grad(x,n,alpha):
    y = 1.0 - x
    c = np.correlate(x, y, mode='full')
    cmax = float(c.max())
    # if alpha small? 
    shift = alpha * (c - cmax)
    maxs = float(shift.max())
    exps = np.exp(shift - maxs)
    s = float(exps.sum())
    val = cmax + np.log(s) / alpha
    w = exps / s
    Wf = w
    T1 = np.convolve(y, Wf, mode='full')
    T2 = np.convolve(x, Wf[::-1], mode='full')
    grad = T1[n-1:2*n-1] - T2[n-1:2*n-1]
    return val, grad
```

import numpy as np

def project_box_sum(x, n):
    target = 0.5 * float(n)
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, -2.0, 3.0)
    lo = -float(np.max(x)) - 1e-12
    hi = 1.0 - float(np.min(x)) + 1e-12
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + t, 0.0, 1.0)))
        if s < target:
            lo = t
        else:
            hi = t
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if diff != 0.0:
        cand = [n // 2, 0, n - 1]
        for i in cand:
            if h[i] > 1e-12 and h[i] < 1.0 - 1e-12:
                h[i] = h[i] + diff
                break
        else:
            h += diff / float(n)
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_count(c, n):
    return float(np.max(c)) * (2.0 / float(n))

def solve_coord_core(c0, a, L, h0, od, Llo, Hhi, iters):
    if Hhi <= Llo + 1e-13:
        return h0
    if Llo < 0.0:
        Llo = 0.0
    t_lo = -h0
    t_hi = 1.0 - h0
    pos = a > 1e-14
    neg = a < -1e-14
    zero = ~(pos | neg)
    c_zero_max = float(np.max(c0[zero])) if np.any(zero) else -1e100
    if c_zero_max > Llo:
        Llo = c_zero_max
        if Llo < 0.0: Llo=0.0
    if Llo >= Hhi - 1e-13:
        return h0
    cpos = c0[pos]; apos = a[pos]
    cneg = c0[neg]; aneg = a[neg]
    hp = cpos.size > 0; hn = cneg.size > 0
    ad = float(a[od]); cd = float(c0[od])
    lo = Llo; hi = Hhi
    for _ in range(iters):
        v = 0.5 * (lo + hi)
        feasible = c_zero_max <= v + 1e-13
        if feasible:
            tl = t_lo; th = t_hi
            if hn:
                tl = max(tl, float(np.max((v - cneg) / aneg)))
            if hp:
                th = min(th, float(np.min((v - cpos) / apos)))
            if tl <= th + 1e-13:
                disc = ad * ad + 4.0 * (cd - v)
                if disc >= 0.0:
                    sq = math.sqrt(disc)
                    r1 = 0.5 * (ad - sq)
                    r2 = 0.5 * (ad + sq)
                    if not (tl <= r1 + 1e-12 or th >= r2 - 1e-12):
                        feasible = False
        if feasible:
            hi = v
        else:
            lo = v
        if hi - lo < 1e-11:
            break
    v = hi
    tl = t_lo; th = t_hi
    if hn:
        tl = max(tl, float(np.max((v - cneg) / aneg)))
    if hp:
        th = min(th, float(np.min((v - cpos) / apos)))
    if tl > th:
        v2 = v + 1e-9
        tl = t_lo; th = t_hi
        if hn:
            tl = max(tl, float(np.max((v2 - cneg) / aneg)))
        if hp:
            th = min(th, float(np.min((v2 - cpos) / apos)))
        if tl > th:
            return h0
        v = v2
    cands = []
    disc = ad * ad + 4.0 * (cd - v)
    if disc >= 0.0:
        sq = math.sqrt(disc); r1=0.5*(ad-sq); r2=0.5*(ad+sq)
        if tl <= r1:
            le = min(th, r1)
            if tl <= le:
                cands.append(0.5*(tl+le))
        if th >= r2:
            lr = max(tl, r2)
            if lr <= th:
                cands.append(0.5*(lr+th))
        if not cands:
            center=0.5*(tl+th)
            cands.append(r1 if abs(center-r1) <= abs(center-r2) else r2)
    else:
        cands.append(0.5*(tl+th))
    best_t = cands[0]
    best_val = 1e100
    for t in cands:
        t = float(np.clip(t, t_lo, t_hi))
        vals = c0 + a * t
        vals[od] -= t * t
        val = float(np.max(vals))
        if val < best_val:
            best_val = val; best_t = t
    x = float(np.clip(h0 + best_t, 0.0, 1.0))
    if abs(x - h0) <= 1e-13:
        return h0
    return x

def cd_pass(h,c,n,order,deadline,iters=30):
    om_rev = 1.0 - h[::-1]
    a_buf = np.zeros(2*n - 1)
    improved = False
    total = 2*n - 1
    for p in order:
        if time.time() >= deadline:
            break
        h0 = float(h[p])
        low1 = p - n + 1
        low2 = -p
        d_start = low1 if low1 < low2 else low2
        high1 = p
        high2 = n - 1 - p
        d_end = high1 if high1 > high2 else high2
        L = d_end - d_start + 1
        start = d_start + n - 1
        stop = start + L
        a = a_buf[:L]
        a.fill(0.0)
        sf = p - n + 1 - d_start
        a[sf:sf+n] += om_rev
        sb = -p - d_start
        a[sb:sb+n] -= h
        c0 = c[start:stop]
        c0max = float(np.max(c0))
        if start > 0:
            out = float(np.max(c[:start]))
        else:

def smooth_c_fg(x,n,alpha):
    y=1.0-x
    c=np.correlate(x,y,mode='full')
    cmax=float(np.max(c))
    shift=alpha*(c-cmax)
    smax=float(np.max(shift))
    exps=np.exp(shift-smax)
    s=float(np.sum(exps))
    val=cmax + math.log(s)/alpha
    w=exps/s
    T1=np.convolve(y,w,mode='full')
    T2=np.convolve(x,w[::-1],mode='full')
    grad=T1[n-1:2*n-1] - T2[n-1:2*n-1]
    grad=np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    if not math.isfinite(val): val=cmax
    return val, grad
```

import numpy as np
import time
import math
from scipy.optimize import minimize


def compute_c_counts(h):
    return np.correlate(h, 1.0 - h, mode="full")


def c5_count(c, n):
    return float(np.max(c)) * (2.0 / float(n))


def project_box_sum(x, n):
    target = 0.5 * float(n)
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, -2.0, 3.0)
    lo = -float(np.max(x)) - 1e-12
    hi = 1.0 - float(np.min(x)) + 1e-12
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + t, 0.0, 1.0)))
        if s < target:
            lo = t
        else:
            hi = t
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if diff != 0.0:
        done = False
        for i in (n // 2, 0, n - 1):
            if h[i] > 1e-12 and h[i] < 1.0 - 1e-12:
                h[i] = h[i] + diff
                done = True
                break
        if not done:
            h = np.clip(h + diff / float(n), 0.0, 1.0)
    return h


def solve_coord_core(c0, a, L, h0, od, Llo, Hhi, iters):
    if Hhi <= Llo + 1e-13:
        return h0
    if Llo < 0.0:
        Llo = 0.0
    t_lo = -h0
    t_hi = 1.0 - h0
    pos = a > 1e-14
    neg = a < -1e-14
    zero = ~(pos | neg)
    c_zero_max = float(np.max(c0[zero])) if np.any(zero) else -1e100
    if c_zero_max > Llo:
        Llo = max(c_zero_max, 0.0)
    if Llo >= Hhi - 1e-13:
        return h0

    cpos = c0[pos]
    apos = a[pos]
    cneg = c0[neg]
    aneg = a[neg]
    has_pos = cpos.size > 0
    has_neg = cneg.size > 0
    ad = float(a[od])
    cd = float(c0[od])

    lo, hi = Llo, Hhi
    for _ in range(iters):
        v = 0.5 * (lo + hi)
        feasible = c_zero_max <= v + 1e-13
        if feasible:
            tl, th = t_lo, t_hi
            if has_neg:
                tl = max(tl, float(np.max((v - cneg) / aneg)))
            if has_pos:
                th = min(th, float(np.min((v - cpos) / apos)))
            if tl <= th + 1e-13:
                disc = ad * ad + 4.0 * (cd - v)
                if disc >= 0.0:
                    sq = math.sqrt(disc)
                    r1 = 0.5 * (ad - sq)
                    r2 = 0.5 * (ad + sq)
                    if not (tl <= r1 + 1e-12 or th >= r2 - 1e-12):
                        feasible = False
        if feasible:
            hi = v
        else:
            lo = v
        if hi - lo < 1e-11:
            break

    v = hi
    tl, th = t_lo, t_hi
    if has_neg:
        tl = max(tl, float(np.max((v - cneg) / aneg)))
    if has_pos:
        th = min(th, float(np.min((v - cpos) / apos)))
    if tl > th:
        v2 = v + 1e-9
        tl, th = t_lo, t_hi
        if has_neg:
            tl = max(tl, float(np.max((v2 - cneg) / aneg)))
        if has_pos:
            th = min(th, float(np.min((v2 - cpos) / apos)))
        if tl > th:
            return h0
        v = v2

    cands = []
    disc = ad * ad + 4.0 * (cd - v)
    if disc >= 0.0:
        sq = math.sqrt(disc)
        r1 = 0.5 * (ad - sq)
        r2 = 0.5 * (ad + sq)
        if tl <= r1:
            le = min(th, r1)
            if tl <= le:
                cands.append(0.5 * (tl + le))
        if th >= r2:
            lr = max(tl, r2)
            if lr <= th:
                cands.append(0.5 * (lr + th))
        if not cands:
            center = 0.5 * (tl + th)
            cands.append(r1 if abs(center - r1) <= abs(center - r2) else r2)
    else:
        cands.append(0.5 * (tl + th))

    best_t = cands[0]
    best_val = 1e100
    for t in cands:
        t = float(np.clip(t, t_lo, t_hi))
        vals = c0 + a * t
        vals[od] -= t * t
        val = float(np.max(vals))
        if val < best_val:
            best_val = val
            best_t = t

    x = float(np.clip(h0 + best_t, 0.0, 1.0))
    if abs(x - h0) <= 1e-13:
        return h0
    return x


def cd_pass(h, c, n, order, deadline, iters=30):
    om_rev = 1.0 - h[::-1]
    a_buf = np.zeros(2 * n - 1)
    total = 2 * n - 1
    improved = False

    for p in order:
        if time.time() >= deadline:
            break
        h0 = float(h[p])

        low1 = p - n + 1
        low2 = -p
        d_start = low1 if low1 < low2 else low2

        high1 = p
        high2 = n - 1 - p
        d_end = high1 if high1 > high2 else high2

        L = d_end - d_start + 1
        start = d_start + n - 1
        stop = start + L

        a = a_buf[:L]
        a.fill(0.0)
        sf = p - n + 1 - d_start
        a[sf:sf + n] += om_rev
        sb = -p - d_start
        a[sb:sb + n] -= h

        c0 = c[start:stop]
        c0max = float(np.max(c0))

        out = 0.0
        if start > 0:
            out = float(np.max(c[:start]))
        if stop < total:
            rr = float(np.max(c[stop:]))
            if rr > out:
                out = rr

        if out > c0max + 1e-11:
            continue

        Llo = max(out, 0.0)
        xn = solve_coord_core(c0, a, L, h0, -d_start, Llo, c0max, iters)

        if abs(xn - h0) > 1e-12:
            delta = xn - h0
            c[start:stop] += delta * a
            c[start - d_start] -= delta * delta
            h[p] = xn
            om_rev[n - 1 - p] = 1.0 - xn
            improved = True

    return h, c, improved


def smooth_c_fg(x, n, alpha):
    y = 1.0 - x
    c = np.correlate(x, y, mode="full")
    cmax = float(np.max(c))
    shift = alpha * (c - cmax)
    smax = float(np.max(shift))
    exps = np.exp(shift - smax)
    s = float(np.sum(exps))
    val = cmax + math.log(s) / alpha
    w = exps / s

    t1 = np.convolve(y, w, mode="full")
    t2 = np.convolve(x, w[::-1], mode="full")
    grad = t1[n - 1:2 * n - 1] - t2[n - 1:2 * n - 1]
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    if not math.isfinite(val):
        val = cmax
    return val, grad


def lbfgs_fg(x, n, alpha, beta):
    val, grad = smooth_c_fg(x, n, alpha)
    d = float(np.sum(x)) - 0.5 * float(n)
    val += beta * d * d
    grad = grad + (2.0 * beta * d)
    return val, grad


def lbfgs_opt(h, n, alpha, beta, maxiter, bounds):
    h = np.asarray(h, dtype=float).copy()
    opts = {
        "maxiter": int(maxiter),
        "ftol": 1e-12,
        "gtol": 1e-8,
        "maxfun": int(min(250, 8 * int(maxiter) + 20)),
    }
    try:
        res = minimize(
            lbfgs_fg,
            h,
            args=(n, alpha, beta),
            method="L-BFGS-B",
            bounds=bounds,
            jac=True,
            options=opts,
        )
        x = project_box_sum(res.x, n)
    except Exception:
        x = project_box_sum(h, n)
    return x


def fourier_init(n, K, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        amp = 0.28 * float(rng.normal(0.0, 1.0 / np.sqrt(k)))
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        h += amp * np.cos(2.0 * math.pi * k * x / 2.0 + ph)
    return project_box_sum(h, n)


def block_init(n, period, rng):
    period = max(4, int(period))
    half = period // 2
    off = float(rng.uniform(0.0, period))
    idx = np.arange(n)
    h = (np.floor((idx + off) / half) % 2).astype(float)
    return project_box_sum(h, n)


def topk_binary(h, n):
    k = int(n // 2)
    if k <= 0:
        return np.zeros(n)
    if k >= n:
        return np.ones(n)
    idx = np.argpartition(h, -k)[-k:]
    out = np.zeros(n)
    out[idx] = 1.0
    return out


def interp_h(h, n_src, n_tgt):
    if n_tgt == n_src:
        return project_box_sum(h, n_src)
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_tgt, endpoint=False)
    return project_box_sum(np.interp(xt, xs, h), n_tgt)

                if p5 < best_c5 - 1e-12:
                    best_h, best_c, best_c5 = hp.copy(), cp.copy(), p5
                if not imp:
                    break

    while time.time() < deadline - 220:
        hw, cw = best_h.copy(), best_c.copy()
        hw, cw, imp = cd_pass(hw, cw, n, range(n), deadline - 220, 32)
        hp = project_box_sum(hw, n)
        cp = compute_c_counts(hp)
        p5 = c5_count(cp, n)
        if p5 < best_c5 - 1e-13:
            best_h, best_c, best_c5 = hp.copy(), cp.copy(), p5
        if not imp:
            break

    stall = 0
    passes = 0
    while time.time() < deadline - 120 and passes < 60:
        order = rng.permutation(n)
        hw, cw = best_h.copy(), best_c.copy()
        hw, cw, imp = cd_pass(hw, cw, n, order, deadline - 120, 28)
        hp = project_box_sum(hw, n)
        cp = compute_c_counts(hp)
        p5 = c5_count(cp, n)
        passes += 1
        if p5 < best_c5 - 1e-13:
            best_h, best_c, best_c5 = hp.copy(), cp.copy(), p5
            stall = 0
        else:
            stall += 1
        if stall >= 18 and not imp:
            break

    if time.time() < deadline - 70:
        inits = [
            fourier_init(n, 6, rng),
            fourier_init(n, 12, rng),
            block_init(n, 6, rng),
            block_init(n, 10, rng),
            topk_binary(best_h + rng.normal(0.0, 0.01, n), n),
        ]
        for hi in inits:
            if time.time() > deadline - 60:
                break
            hi = project_box_sum(hi, n)
            cs = compute_c_counts(hi)
            s5 = c5_count(cs, n)
            if s5 < best_c5 + 0.08:
                hs = lbfgs_opt(hi, n, 5000, 30, 25, bounds)
                cs = compute_c_counts(hs)
                s5 = c5_count(cs, n)
                hw, cw = hs.copy(), cs.copy()
                for _ in range(4):
                    if time.time() > deadline - 60:
                        break
                    hw, cw, imp = cd_pass(hw, cw, n, range(n), deadline - 60, 18)
                    hp = project_box_sum(hw, n)
                    cp = compute_c_counts(hp)
                    p5 = c5_count(cp, n)
                    hw, cw = hp, cp
                    if p5 < best_c5 - 1e-12:
                        best_h, best_c, best_c5 = hp.copy(), cp.copy(), p5
                    if not imp:
                        break

    for r in range(3):
        if time.time() > deadline - 45:
            break
        hp = project_box_sum(best_h + rng.normal(0.0, 0.003 + 0.005 * r, n), n)
        hs = lbfgs_opt(hp, n, 20000, 80, 18, bounds)
        cs = compute_c_counts(hs)
        p5 = c5_count(cs, n)
        hw, cw = hs.copy(), cs.copy()
        for _ in range(2):
            if time.time() > deadline - 45:
                break
            hw, cw, imp = cd_pass(hw, cw, n, range(n), deadline - 45, 16)
            hp2 = project_box_sum(hw, n)
            cp2 = compute_c_counts(hp2)
            p52 = c5_count(cp2, n)
            if p52 < best_c5 - 1e-12:
                best_h, best_c, best_c5 = hp2.copy(), cp2.copy(), p52
            if not imp:
                break

    if time.time() < deadline - 30:
        seen = set()
        base_n = n
        cand_list = [base_n + d for d in (-40, -20, -10, 10, 20, 40)]
        cand_list += [300, 400, 520, 640, 760, 900]
        for nty in cand_list:
            if time.time() > deadline - 30:
                break
            nty = int(nty)
            if nty in seen:
                continue
            seen.add(nty)
            if nty < 128 or nty > 1000 or nty % 2 or nty == n:
                continue
            htry = interp_h(best_h, len(best_h), nty)
            ctry = compute_c_counts(htry)
            t5 = c5_count(ctry, nty)
            if time.time() < deadline - 25:
                bounds_t = [(0.0, 1.0)] * nty
                htemp = lbfgs_opt(htry, nty, 8000, 30, 18, bounds_t)
                ctemp = compute_c_counts(htemp)
                t5t = c5_count(ctemp, nty)
                if t5t < t5:
                    htry, ctry, t5 = htemp, ctemp, t5t
                hw, cw = htry.copy(), ctry.copy()
                for _ in range(3):
                    if time.time() > deadline - 25:
                        break
                    hw, cw, imp = cd_pass(hw, cw, nty, range(nty), deadline - 25, 18)
                    hp = project_box_sum(hw, nty)
                    cp = compute_c_counts(hp)
                    p5 = c5_count(cp, nty)
                    hw, cw = hp, cp
                    if p5 < t5:
                        t5 = p5
                    if not imp:
                        break
            if t5 < best_c5 - 1e-10:
                best_h, best_c, best_c5 = htry.copy(), ctry.copy(), t5
                n = nty
                bounds = [(0.0, 1.0)] * n

    while time.time() < deadline - 2:
        nfit = len(best_h)
        hw, cw = best_h.copy(), best_c.copy()
        hw, cw, imp = cd_pass(hw, cw, nfit, range(nfit), deadline - 2, 26)
        hp = project_box_sum(hw, nfit)
        cp = compute_c_counts(hp)
        p5 = c5_count(cp, nfit)
        if p5 < best_c5 - 1e-13:
            best_h, best_c, best_c5 = hp.copy(), cp.copy(), p5
        if not imp:
            break

    final_h = project_box_sum(best_h, len(best_h))
    final_c = compute_c_counts(final_h)
    final_c5 = c5_count(final_c, len(final_h))
    return final_h, final_c5, len(final_h)
