∂c[k]/∂h[p] = (1-h[p+k-(n-1)]) - h[p-k+(n-1)] (with bounds, and for k=n-1 (the diagonal), it's (1-2h[p]))

So the subgradient descent update is:
h[p] ← clip(h[p] - η * (1/N_active) * Σ_{k in active} ∂c[k]/∂h[p], 0, 1)

Then project to satisfy the sum constraint.

def precompute_tables(n):
    tables = []
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        diag_i = np.where(ms == (n - 1))[0]
        tables.append({
            'ms': ms, 'vf': vf, 'jb': jb,
            'jf': jf[vf] if np.any(vf) else np.array([], dtype=int),
            'jb_v': jb[vb] if np.any(vb) else np.array([], dtype=int),
            'diag': int(diag_i[0]) if len(diag_i) > 0 else -1,
            'm_min': m_min, 'm_max': m_max
        })
    return tables

def solve_1d(c, h, p, n, T):

    h0 = h[p]
    a = np.zeros(len(ms))
    if len(jf) > 0:
        a[vf] += (1.0 - h[jf])
    if len(jb_v) > 0:
        a[vb] -= h[jb_v]
    c_base = c[ms]
    has_diag = diag >= 0

        a = a.copy()
        a[diag] = 0.0

        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:

    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    feasible = False
        L_hi = L if feasible else L_lo
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))

        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))

def lse_objective_and_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    probs = np.exp(shifted - mx)
    probs /= probs.sum()
    val = (m + mx / alpha + np.log(probs.sum()) / alpha) * (2.0 / n)
    # Gradient: dx * sum_m probs[m] * dc[m]/dx[p]
    # dc[m]/dx[p] = (1-x[p+m-(n-1)]) - x[p-m+(n-1)] (with bounds, and diagonal special case)
    # For the diagonal (m=n-1): dc[n-1]/dx[p] = (1-x[p]) - x[p] = 1-2x[p]... 
    # Wait, c[n-1] = sum_i x[i]*(1-x[i]), so dc[n-1]/dx[p] = (1-x[p]) - x[p] = 1-2x[p]
    # For off-diagonal: dc[m]/dx[p] = (1-x[p+m-(n-1)]) - x[p-m+(n-1)] (with OOB=0 terms)
    
    # Compute gradient efficiently
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(probs > 1e-12)[0]
    for m_i in active:
        m = int(m_i)
        w = probs[m]
        offset = m - (n - 1)
        # Forward: (1-x[p+offset]) for 0<=p+offset<n
        lo = max(0, -offset)
        hi = min(n, n - offset)
        if lo < hi:
            idx = np.arange(lo + max(offset, 0), hi + max(offset, 0))
            grad[lo:hi] += w * (1.0 - x[idx])
        # Backward: -x[p-offset] for 0<=p-offset<n
        lo2 = max(0, offset)
        hi2 = min(n, n + offset)
        if lo2 < hi2:
            idx2 = np.arange(lo2 - offset, hi2 - offset)
            grad[lo2:hi2] -= w * x[idx2]
    grad *= dx
    return val, grad

def nesterov_lse(h, n, alpha, max_iter, deadline, lr=None):
    h_curr = h.copy()
    h_prev = h.copy()
    if lr is None:
        lr = 0.01
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Nesterov preview
        h_mom = h_curr + (h_curr - h_prev) * 0.9
        h_mom = np.clip(h_mom, 0.0, 1.0)

        h_new = h_mom - lr * grad
        h_new = project_feasible(h_new, n)
        h_prev = h_curr
        h_curr = h_new
        lr = lr * 0.999  # slow decay
    return h_curr, compute_c5(h_curr, n)
