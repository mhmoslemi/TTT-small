def coord_move(c,h,p,n,prefer):
    h0=float(h[p])
    q0=n-1-p
    qs=np.arange(q0,q0+n)
    lags=qs-(n-1)

    vb=(jb>=0)&(jb<n)
    a=np.zeros(n,dtype=float)
    if np.any(vf): a[vf] += 1.0 - h[jf[vf]]

def single_pass(h,n,prefer,deadline):

def project_values(v, n):

def smooth_score(n, rng, modes):
    modes = int(max(1, min(modes, max(1, n // 2))))
    x = (np.arange(n) + 0.5) * (2.0 / n)
    y = np.zeros(n)

            ok = False
        if ok:
            Cq = L - base_d + h0 * (1.0 - h0)
            if Cq < -1e-11:
                ok = False
            elif Cq < 0.25 - 1e-12:
                disc = 1.0 - 4.0 * Cq
                if disc < 0.0:
                    ok = False
                else:
                    r1 = 0.5 * (1.0 - np.sqrt(disc))
                    r2 = 0.5 * (1.0 + np.sqrt(disc))
                    if xlo > r1 + 1e-11 and xhi < r2 - 1e-11:
                        ok = False
        if ok:
            Lhi = L
        else:
            Llo = L
    L = Lhi
    xlo, xhi = 0.0, 1.0
    if bn_neg.size:
        xlo = max(xlo, float(np.max((L - bn_neg) / an_neg)))
    if bn_pos.size:
        xhi = min(xhi, float(np.min((L - bn_pos) / an_pos)))
    xlo = max(0.0, xlo)
    xhi = min(1.0, xhi)
    segs = []
    if xlo <= xhi + 1e-12:
        Cq = L - base_d + h0 * (1.0 - h0)
        if Cq >= -1e-11:
            if Cq >= 0.25 - 1e-12:
                segs.append((xlo, xhi))
            else:
                disc = max(0.0, 1.0 - 4.0 * Cq)
                r1 = 0.5 * (1.0 - np.sqrt(disc))
                r2 = 0.5 * (1.0 + np.sqrt(disc))
                if min(xhi, r1) >= xlo - 1e-12:
                    segs.append((xlo, min(xhi, r1)))
                if max(xlo, r2) <= xhi + 1e-12:
                    segs.append((max(xlo, r2), xhi))
    if not segs:
        return h0, None
    target = (1.0 if h0 >= 0.5 else 0.0) if prefer == "bin" else h0
    best_x = target
    best_key = 1e30
    for A, B in segs:
        if B < A:
            continue
        A = max(0.0, A)
        B = min(1.0, B)
        if A <= target <= B:
            cand = target
        elif abs(target - A) <= abs(target - B):
            cand = A
        else:
            cand = B
        key = abs(cand - target)
        if key < best_key - 1e-14:
            best_key = key
            best_x = cand
    best_x = float(np.clip(best_x, 0.0, 1.0))
    if abs(best_x - h0) <= 1e-14:
        return h0, None
    return best_x, (best_x - h0, qs, a)

def single_pass(h, n, prefer, deadline):
    h = project_values(h, n)
    c = np.correlate(h, 1.0 - h, mode="full")
    changed = 0
    for p in range(n):
        if time.time() > deadline:
            break
        x, data = coord_move(c, h, p, n, prefer)
        if data is not None:
            delta, qs, a = data
            c[qs] += delta * a
            c[n - 1] -= delta * delta
            h[p] = x
            changed += 1
    h = project_values(h, n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return h, float(np.max(c) * (2.0 / n)), changed

def coord_improve(h, n, passes, deadline, prefer):
    h = project_values(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for _ in range(passes):
        if time.time() > deadline:
            break
        new_h, new_c, changed = single_pass(best_h, n, prefer, deadline)
        if new_c < best_c - 1e-13:
            best_h, best_c = new_h, new_c
        else:
            break
        if changed == 0:
            break
    return best_h, best_c

def fourier_h_params(params, n, modes):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    y = 0.5 * np.ones(n)
    for kk in range(1, modes + 1):
        aa = float(params[2 * (kk - 1)])
        bb = float(params[2 * (kk - 1) + 1])
        y += aa * np.cos(np.pi * kk * x) + bb * np.sin(np.pi * kk * x)
    return project_values(y, n)

def fourier_obj(params, n, modes, alpha):
    h = fourier_h_params(params, n, modes)
    c = np.correlate(h, 1.0 - h, mode="full")
    cm = float(np.max(c))
    sh = alpha * (c - cm)
    mx = float(np.max(sh))
    val = cm + (mx + float(np.log(np.sum(np.exp(sh - mx))))) / alpha
    return float(val) * (2.0 / n)

def optimize_fourier(n, deadline, modes, rng):
    best_h = None
    best_c = 1e30
    pcount = 2 * modes
    bounds = [(-1.2, 1.2)] * pcount
    for rr in range(3):
        if time.time() > deadline:
            break
        if rr == 0:
            x0 = np.zeros(pcount)
            x0[0] = 0.8
            x0[1] = rng.uniform(-0.8, 0.8)
        else:
            x0 = rng.uniform(-0.7, 0.7, pcount)
        try:
            res = minimize(fourier_obj, x0, args=(n, modes, 40.0),
                           method="L-BFGS-B", bounds=bounds,
                           options={"maxiter": 90, "ftol": 1e-12})
            params = res.x
        except Exception:
            params = x0
        h = fourier_h_params(params, n, modes)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h, c
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(30.0, float(budget_s))
    end = t0 + total - 5.0
    rng = np.random.RandomState(seed)
    parent_arr = None
    try:
        parent_arr = np.asarray(initial_h_values, dtype=float).reshape(-1)
        if parent_arr.size < 4:
            parent_arr = None
    except Exception:
        parent_arr = None
    if parent_arr is not None:
        n_par = int(parent_arr.size)
        if n_par % 2:
            n_par -= 1
        if n_par < 16:
            n_par = 16
        h0 = project_values(parent_arr, n_par) if len(parent_arr) == n_par else project_values(parent_arr, n_par)
    else:
        n_par = 240 if total > 200.0 else 96
        h0 = smooth_random_h(n_par, rng, None, 8)
    best_h = h0
    best_c = compute_c5(best_h)
    n_best = len(best_h)
    d_parent = min(end, t0 + 0.18 * total)
    h, c = coord_improve(best_h, n_best, 200, d_parent, "close")
    if c < best_c:
        best_h, best_c, n_best = h.copy(), c, len(h)
    parent_phase = min(end, t0 + 0.45 * total)
    for _ in range(4):
        if time.time() > parent_phase:
            break
        h2, c2, chg = single_pass(best_h, n_best, "bin", min(parent_phase, time.time() + 25.0))
        if chg <= 0:
            break
        if c2 < best_c - 1e-12:
            best_h, best_c, n_best = h2.copy(), c2, len(h2)
        if c2 <= best_c + 1e-8:
            h3, c3 = coord_improve(h2, n_best, 40, min(parent_phase, time.time() + 35.0), "close")
            if c3 < best_c:
                best_h, best_c, n_best = h3.copy(), c3, len(h3)
    resample_phase = min(end, t0 + 0.65 * total)
    cand_ns = []
    for vv in [n_best, n_par, 512, 384, 320, 256, 224, 192, 160, 144, 128, 112, 96, 80, 64, 56, 48, 40, 32, 28]:
        vv = int(vv)
        if vv < 16 or vv > 1000:
            continue
        if vv % 2:
            vv -= 1
        if vv not in cand_ns:
            cand_ns.append(vv)
    for nt in cand_ns:
        if time.time() > resample_phase:
            break
        if nt == n_best:
            continue
        lim = min(55.0, max(8.0, 0.045 * total))
        d = min(resample_phase, time.time() + lim)
        if d <= time.time() + 2.0:
            break
        x = np.interp((np.arange(nt) + 0.5) * (2.0 / nt),
                      (np.arange(n_best) + 0.5) * (2.0 / n_best), best_h)
        hstart = project_values(x, nt)
        h, c = coord_improve(hstart, nt, 90, d, "close")
        if c < best_c:
            best_h, best_c, n_best = h.copy(), c, len(h)
    random_phase = min(end, t0 + 0.82 * total)
    for nt in cand_ns:
        if time.time() > random_phase:
            break
        for kind in range(3):
            if time.time() > random_phase:
                break
            lim = min(35.0, max(6.0, 0.028 * total))
            d = min(random_phase, time.time() + lim)
            if d <= time.time() + 2.0:
                break
            hstart = make_start(nt, rng, best_h, kind)
            h, c = coord_improve(hstart, nt, 70, d, "close")
            if c < best_c:
                best_h, best_c, n_best = h.copy(), c, len(h)
    fourier_phase = min(end, t0 + 0.92 * total)
    for nf in [192, 128, 96]:
        if time.time() > fourier_phase:
            break
        d = min(fourier_phase, time.time() + min(55.0, 0.05 * total))
        if d <= time.time() + 5.0:
            break
        hstart, cf = optimize_fourier(nf, d, 3, rng)
        if hstart is not None and cf < best_c + 0.02:
            h, c = coord_improve(hstart, nf, 70, min(fourier_phase, time.time() + 25.0), "close")
            if c < best_c:
                best_h, best_c, n_best = h.copy(), c, len(h)
    n_best = len(best_h)
    h, c = coord_improve(best_h, n_best, 300, end, "close")
    if c < best_c:
        best_h, best_c, n_best = h.copy(), c, len(h)
    for _ in range(3):
        if time.time() > end - 2.0:
            break
        h2, c2, chg = single_pass(best_h, n_best, "bin", end)
        if c2 < best_c - 1e-12:
            best_h, best_c, n_best = h2.copy(), c2, len(h2)
        elif chg > 0 and c2 <= best_c + 1e-8:
            h3, c3 = coord_improve(h2, n_best, 25, end, "close")
            if c3 < best_c:
                best_h, best_c, n_best = h3.copy(), c3, len(h3)
    best_h = project_values(best_h, len(best_h))
    best_c = compute_c5(best_h)
    return best_h, best_c, len(best_h)
