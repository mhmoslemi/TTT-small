def project_feasible(x, n):
    x = np.asarray(x, dtype=float)
    target = 0.5 * n
    lo = -float(np.max(x))
    hi = 1.0 - float(np.min(x))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        h = np.clip(x + mid, 0.0, 1.0)
        if float(h.sum()) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot > 1e-15:
                h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = float(h.sum())
            if tot > 1e-15:
                h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-14:

            h[idx] += diff
        else:
            for k in range(n):

                    break
    return h


def full_c(h, n):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h, n):
    return float(np.max(full_c(h, n))) * (2.0 / n)


def interpolate_to(h, n_out):
    n = len(h)
    if n == n_out:
        return project_feasible(h.copy(), n_out)
    x_in = np.linspace(0.0, 2.0, n)
    x_out = np.linspace(0.0, 2.0, n_out)
    return project_feasible(np.interp(x_out, x_in, h), n_out)


def pair_coefficients(a, b, h, n, m_lo, m_hi):
    L = m_hi - m_lo + 1
    s = m_lo + np.arange(L, dtype=int) - (n - 1)
    coef = np.zeros(L, dtype=float)
    m1 = (a - s >= 0) & (a - s < n)
    if np.any(m1):
        coef[m1] += 1.0 - h[a - s[m1]]
    m2 = (a + s >= 0) & (a + s < n)
    if np.any(m2):
        coef[m2] -= h[a + s[m2]]
    m1 = (b - s >= 0) & (b - s < n)
    if np.any(m1):
        coef[m1] -= 1.0 - h[b - s[m1]]
    m2 = (b + s >= 0) & (b + s < n)
    if np.any(m2):
        coef[m2] += h[b + s[m2]]
    return coef


def pair_fval(d, seg, coef, c0, Kpair, unaff, cpos):

    vals[cpos] = c0 + Kpair * d - 2.0 * d * d
    m = float(np.max(vals))
    if unaff > m:
        m = unaff
    return m


def pair_lval(d, seg, coef, unaff):
    vals = seg + coef * d
    m = float(np.max(vals))
    if unaff > m:

    return m

    Kpair = 2.0 * (hb - ha)
    cpos = n - 1 - m_lo
    d = choose_pair_d(seg, coef, c0, Kpair, unaff, cpos, lo, hi)
    if abs(d) <= 1e-15:
        return 0.0, -1.0
    newmax = pair_fval(d, seg, coef, c0, Kpair, unaff, cpos)
    if newmax >= max(seg_max, unaff) - 1e-10:
        return 0.0, -1.0
    c[m_lo:m_hi + 1] += coef * d
    c[n - 1] -= 2.0 * d * d
    na = ha + d
    nb = hb - d
    if na < 0.0:
        na = 0.0
    elif na > 1.0:
        na = 1.0
    if nb < 0.0:
        nb = 0.0
    elif nb > 1.0:
        nb = 1.0
    h[a] = na
    h[b] = nb
    return d, newmax


def pair_coord_descent(h, c, n, max_passes, deadline, seed=99):
    dx = 2.0 / n
    h = project_feasible(np.array(h, dtype=float), n)
    c = full_c(h, n)
    best_h = h.copy()
    best_c = float(np.max(c)) * dx
    rng = np.random.RandomState(seed)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        h = project_feasible(h, n)
        c = full_c(h, n)
        curc = float(np.max(c)) * dx
        if curc < best_c:
            best_c = curc
            best_h = h.copy()
        improved_any = False
        order = list(range(n - 1))
        rng.shuffle(order)
        for i in order:
            if time.time() > deadline:
                break
            oldmax = float(np.max(c))
            d, newmax = pair_update_once(i, i + 1, h, c, n)
            if d != 0.0 and newmax >= 0.0 and newmax < oldmax - 1e-10:
                improved_any = True
                nc = newmax * dx
                if nc < best_c:
                    best_c = nc
                    best_h = h.copy()
        if n > 1 and time.time() < deadline:
            oldmax = float(np.max(c))
            d, newmax = pair_update_once(n - 1, 0, h, c, n)
            if d != 0.0 and newmax >= 0.0 and newmax < oldmax - 1e-10:
                improved_any = True
                nc = newmax * dx
                if nc < best_c:
                    best_c = nc
                    best_h = h.copy()
        extra = int(min(60, n // 8))
        for _ in range(extra):
            if time.time() > deadline:
                break
            i = int(rng.randint(0, n))
            j = int(rng.randint(0, n))
            if i == j:
                continue
            oldmax = float(np.max(c))
            d, newmax = pair_update_once(i, j, h, c, n)
            if d != 0.0 and newmax >= 0.0 and newmax < oldmax - 1e-10:
                improved_any = True
                nc = newmax * dx
                if nc < best_c:
                    best_c = nc
                    best_h = h.copy()
        if not improved_any:
            break
    return best_h, best_c


def weighted_gradient(h, n, m_idx, w):
    grad = np.zeros(n, dtype=float)
    for i in range(len(m_idx)):
        m = int(m_idx[i])
        wt = float(w[i])
        if wt == 0.0:
            continue
        s = m - (n - 1)
        if s >= 0:
            L = n - s
            if L > 0:
                grad[s:n] += wt * (1.0 - h[:L])
                grad[:L] -= wt * h[s:n]
        else:
            r = -s
            L = n - r
            if L > 0:
                grad[:L] += wt * (1.0 - h[r:n])
                grad[r:n] -= wt * h[:L]
    return grad


def active_gradient_descent(h, n, max_iters, deadline, seed=11):
    h = project_feasible(np.array(h, dtype=float), n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c = float(np.max(full_c(h, n))) * dx
    rng = np.random.RandomState(seed)
    step = 0.15
    K = 12
    beta = 500.0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = full_c(h, n)
        curc = float(np.max(c)) * dx
        if curc < best_c:
            best_c = curc
            best_h = h.copy()
        kc = min(K, len(c) - 1)
        if kc <= 0:
            break
        if kc < len(c):
            top = np.argpartition(-c, kc)[:kc]
        else:
            top = np.arange(len(c))
        w = np.exp(beta * (c[top] - float(np.max(c[top]))))
        ws = float(np.sum(w))
        if ws <= 1e-300 or not np.isfinite(ws):
            w = np.ones(len(top), dtype=float)
            w /= float(len(w))
        else:
            w /= ws
        grad = weighted_gradient(h, n, top, w)
        grad -= grad.mean()
        norm = float(np.linalg.norm(grad))
        if norm > 1e-13:
            direction = -grad / norm
            improved = False
            for t in (step, 0.5 * step):
                hn = project_feasible(h + t * direction, n)
                cn = float(np.max(full_c(hn, n))) * dx
                if cn < best_c - 1e-12:
                    h = hn
                    best_c = cn
                    best_h = hn.copy()
                    improved = True
                    break
            if improved:
                step = min(0.4, step * 1.1)
            else:
                step *= 0.7
                if step < 5e-4:
                    kick = np.zeros(n, dtype=float)
                    ki = int(rng.randint(0, n))
                    kj = int(rng.randint(0, n))
                    ki2 = kj if ki == kj else kj
                    kick[ki] = 1.0
                    kick[ki2] = -1.0
                    kick -= kick.mean()
                    kn = float(np.linalg.norm(kick))
                    if kn > 1e-12:
                        hn = project_feasible(h + 0.03 * kick / kn, n)
                        cn = float(np.max(full_c(hn, n))) * dx
                        if cn < best_c:
                            h = hn
                            best_c = cn
                            best_h = hn.copy()
                            step = 0.1
                        else:
                            h = project_feasible(h + 0.02 * rng.randn(n), n)
        if it % 25 == 24:
            K = min(60, K + 5)
            beta *= 1.3
    return best_h, best_c


def roll_search(h, n, deadline, use_complement=True):
    h = project_feasible(np.array(h, dtype=float), n)
    dx = 2.0 / n
    best = h.copy()
    best_c = float(np.max(full_c(best, n))) * dx
    bases = [h]
    if use_complement:
        bases.append(1.0 - h)
    for base in bases:
        base = project_feasible(base, n)
        if n <= 1000:
            for off in range(n):
                if time.time() > deadline:
                    break
                arr = np.roll(base, off)
                cc = float(np.max(full_c(arr, n))) * dx
                if cc < best_c:
                    best_c = cc
                    best = arr.copy()
        else:
            cc = float(np.max(full_c(base, n))) * dx
            if cc < best_c:
                best_c = cc
                best = base.copy()
    return project_feasible(best, n), best_c

            ccur = full_c(project_feasible(best_h, best_n), best_n)
            best_h, best_c = pair_coord_descent(best_h, ccur, best_n, 14, parent_end)
    if best_n > 240 and time.time() < deadline - 180:
        n2 = best_n // 2
        if n2 >= 120:
            t2 = deadline - 90
            h2 = interpolate_to(best_h, n2)
            h2, c2 = roll_search(h2, n2, t2, True)

                c2arr = full_c(project_feasible(h2, n2), n2)
                h2, c2 = pair_coord_descent(h2, c2arr, n2, 10, t2)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
