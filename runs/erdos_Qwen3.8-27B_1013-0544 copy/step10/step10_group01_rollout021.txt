import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - 2.0 * h0) - d * d
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def cd_optimize(h, n, max_passes, deadline, gs_iters=65, orders=None):

            break

        improved = False
        if order == 'forward':

            idx_range = range(n - 1, -1, -1)

            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    return x.sum() - n / 2.0

    return np.ones(n)

            h_o, c_o = cd_optimize(hp, n_cur, 40, deadline, gs_iters=55, orders=ord2)
            h_o, c_o = subspace_optimize(h_o, n_cur, deadline)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 7: Final polish
    if time.time() < deadline - 8:
        n_cur = len(best_h)
        h_f, c_f = cd_optimize(best_h, n_cur, 50, deadline, gs_iters=80, orders=ord1)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = slsqp_opt(best_h, n_cur, [1000000, 5000000], 600)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = cd_optimize(best_h, len(best_h), 30, deadline, gs_iters=80, orders=ord2)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        best_h = project_feasible(best_h, len(best_h))
        best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
