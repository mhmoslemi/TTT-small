def smooth_parts(h,n,alpha):
    c=np.correlate(h,1.0-h,mode='full')
    dx=2.0/n
    m=c.max()
    z=alpha*(c-m)
    mx=z.max()
    s=np.exp(z-mx)
    w=s/s.sum()
    val=dx*(m + (mx+np.log(s.sum()))/alpha)
    # gradient
    u=1.0-h
    first=np.convolve(u,w,mode='full')[n-1:2*n-1]
    second=np.correlate(w,h,mode='full')[n-1:2*n-1][::-1]
    g=dx*(first-second)
    return val,w,g
```

def smooth_parts(h,n,alpha):
    h=np.asarray(h,dtype=float)
    one_m=1.0-h
    c=np.correlate(h,one_m,mode='full')
    if not np.all(np.isfinite(c)):
       return 1e3, np.zeros(2*n-1), np.zeros(n)
    dx=2.0/n
    m=float(np.max(c))
    if alpha <= 0.0:
       active=c>=m-1e-12
       w=np.zeros_like(c)
       cnt=float(np.count_nonzero(active))
       if cnt<=0: w[int(np.argmax(c))]=1
       else: w[active]=1.0/cnt
       val=dx*m
    else:
       z=alpha*(c-m)
       mx=float(np.max(z))
       e=np.exp(z-mx)
       den=float(np.sum(e))
       if den <= 0.0 or not np.isfinite(den):
           w=np.zeros_like(c); w[int(np.argmax(c))]=1.0; val=dx*m
       else:
           w=e/den
           val=dx*(m+(mx+np.log(den))/alpha)
    u=one_m
    first=np.correlate(w,u,mode='full')[n-1:2*n-1][::-1]
    second=np.convolve(h,w,mode='full')[n-1:2*n-1]
    g=dx*(first-second)
    if not np.all(np.isfinite(g)):
       g=np.zeros(n)
    return float(val),w,g

    for _ in range(max_passes):

       c=np.correlate(h,1.0-h,mode='full')
       improved=False

          if time.time() > deadline: break
          h0=h[p]
          if h0 <= 1e-13 or h0 >= 1.0-1e-13: continue
          m_lo=max(0,n-1-p); m_hi=min(2*n-2,n-1+p)
          ms=np.arange(m_lo,m_hi+1)
          jf=p+ms-(n-1); jb=p-ms+(n-1)
          vf=(jf>=0)&(jf<n); vb=(jb>=0)&(jb<n)
          a=np.zeros(len(ms))
          if np.any(vf): a[vf] += (1.0-h[jf[vf]])

    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(12):
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
        diff = target - h.sum()
    return h

def _sum_eq(x, n):
    return float(np.sum(x) - 0.5 * n)

def _sum_eq_jac(x, n):
    return np.ones(n)

def smooth_parts(h, n, alpha):
    h = np.asarray(h, dtype=float)
    om = 1.0 - h
    c = np.correlate(h, om, mode='full')
    if not np.all(np.isfinite(c)):
        return 1e3, np.zeros(2 * n - 1), np.zeros(n)
    dx = 2.0 / n
    m = float(np.max(c))
    if alpha <= 0.0:
        act = c >= m - 1e-12
        w = np.zeros_like(c)
        cnt = float(np.count_nonzero(act))
        if cnt <= 0.0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[act] = 1.0 / cnt
        val = dx * m
    else:
        z = alpha * (c - m)
        mx = float(np.max(z))
        e = np.exp(z - mx)
        den = float(np.sum(e))
        if den <= 0.0 or not np.isfinite(den):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            val = dx * m
        else:
            w = e / den
            val = dx * (m + (mx + np.log(den)) / alpha)
    first = np.correlate(w, om, mode='full')[n - 1:2 * n - 1][::-1]
    second = np.convolve(h, w, mode='full')[n - 1:2 * n - 1]
    g = dx * (first - second)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n)
    return float(val), w, g

def obj_jac(x, n, alpha):
    v, w, g = smooth_parts(x, n, alpha)
    return v, g

def slsqp_alpha(h, n, alpha, maxiter, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h)
    if time.time() > deadline - 5.0:
        return h, best_c
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(obj_jac, h, args=(n, alpha), method='SLSQP', jac=True,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None:
            nh = project_feasible(res.x, n)
            c = compute_c5(nh)
            if c < best_c:
                return nh, c
    except Exception:
        pass
    return h, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    r = float(np.max(vals))
    if unaff_max > r:
        r = unaff_max
    return r

def coord_descent(h, n, max_passes, deadline, g_iters=35):
    h = np.array(h, dtype=float).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            pos = np.where(ms == n - 1)[0]
            has_diag = len(pos) > 0
            diag_idx = int(pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            if d_hi - d_lo <= 1e-15:
                continue
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d2)
            for __ in range(g_iters):
                if d_hi - d_lo <= 1e-14:
                    break
                if f1 < f2:
                    d_hi = d2
                    f2 = f1
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d1)
                else:
                    d_lo = d1
                    f1 = f2
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d2)
            d_opt = 0.5 * (d_lo + d_hi)
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d_opt)
            for dc in (-h0, 1.0 - h0):
                ft = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, dc)
                if ft < f_opt:
                    f_opt = ft
                    d_opt = dc
            f_cur = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, 0.0)
            if f_opt < f_cur - 1e-14:
                h[p] = h0 + d_opt
                delta = d_opt * a
                if has_diag:
                    delta[diag_idx] -= d_opt * d_opt
                c[ms] += delta
                improved = True
        if improved:
            h[h < 1e-6] = 0.0
            h[h > 1.0 - 1e-6] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h.copy()
            else:
                h = best_h.copy()
                break
        else:
            break
    c_final = compute_c5(best_h)
    if c_final < best_c:
        best_c = c_final
    return best_h, best_c

def gradient_polish(h, n, deadline, max_steps=25, step0=0.015):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for alpha in (4000.0, 20000.0):
        for _ in range(max_steps):
            if time.time() > deadline:
                break
            v, w, g = smooth_parts(h, n, alpha)
            gn = float(np.linalg.norm(g))
            if gn < 1e-12:
                break
            cur = step
            accepted = False
            for __ in range(6):
                trial = project_feasible(h - (cur / gn) * g, n)
                c = compute_c5(trial)
                if c < best_c - 1e-13:
                    best_h = trial.copy()
                    best_c = c
                    accepted = True
                    break
                cur *= 0.5
            if accepted:
                step = min(0.05, step * 1.2)
            else:
                step *= 0.4
                if step < 1e-5:
                    break
        if step < 1e-5:
            break
    return best_h, best_c

def cd_params(n):
    if n <= 64:
        return 60, 42
    if n <= 128:
        return 42, 36
    if n <= 256:
        return 30, 30
    if n <= 512:
        return 20, 28
    return 14, 25

def local_refine(h, n, deadline, allow_slsqp=True):
    h = project_feasible(h, n)
    best_c = compute_c5(h)
    best_h = h.copy()
    if time.time() >= deadline:
        return best_h, best_c
    t_start = time.time()
    total = max(1.0, deadline - t_start)
    cd_dead = min(deadline, t_start + max(2.0, 0.5 * total))
    passes, gits = cd_params(n)
    h2, c2 = coord_descent(best_h, n, passes, cd_dead, g_iters=gits)
    if c2 < best_c:
        best_h, best_c = h2, c2
    if allow_slsqp and deadline - time.time() > 18.0:
        if n <= 80:
            alphas = [(2000.0, 110), (20000.0, 80)]
        elif n <= 200:
            alphas = [(4000.0, 80), (20000.0, 60)]
        elif n <= 350:
            alphas = [(6000.0, 50)]
        else:
            alphas = []
        for a, mx in alphas:
            if deadline - time.time() < 10.0:
                break
            h2, c2 = slsqp_alpha(best_h, n, a, mx, deadline)
            if c2 < best_c:
                best_h, best_c = h2, c2
        if deadline - time.time() > 7.0:
            h2, c2 = coord_descent(best_h, n, 6, deadline, g_iters=26)
            if c2 < best_c:
                best_h, best_c = h2, c2
    if n > 350 and n <= 800 and deadline - time.time() > 160.0:
        h2, c2 = slsqp_alpha(best_h, n, 12000.0, 18, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2
    if deadline - time.time() > 9.0:
        h2, c2 = gradient_polish(best_h, n, deadline - 2.0, max_steps=18, step0=0.013)
        if c2 < best_c:
            best_h, best_c = h2, c2
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src, n_dst)
    x0 = np.linspace(0.0, 2.0, n_src)
    x1 = np.linspace(0.0, 2.0, n_dst)
    return project_feasible(np.interp(x1, x0, h_src), n_dst)

def fourier_start(n, rng, amp):
    x = np.linspace(0.0, 2.0, n)
    h = np.full(n, 0.5)
    modes = np.arange(1, 6, dtype=float)
    coeff = rng.rand(5) * amp / modes
    ph = rng.rand(5) * 2.0 * np.pi
    for j in range(5):
        h += coeff[j] * np.cos(np.pi * modes[j] * x + ph[j])
    h += 0.04 * rng.randn(n)
    return project_feasible(h, n)

def _raw_start(n, rng, idx, base):
    if base is None:
        base_h = np.full(n, 0.5)
    elif len(base) == n:
        base_h = base
    else:
        base_h = interpolate_to(base, n)
    x = np.linspace(0.0, 2.0, n)
    mode = idx % 6
    if mode == 0:
        f = 1.0 + (idx % 5)
        amp = 0.10 + 0.03 * (idx % 4)
        h = 0.5 + amp * np.sin(np.pi * f * x + 0.37 * idx) + 0.03 * rng.randn(n)
        return project_feasible(h, n)
    if mode == 1:
        return fourier_start(n, rng, 0.18 + 0.03 * (idx % 3))
    if mode == 2:
        length = max(3, n // 12)
        block = np.clip(0.5 + 0.25 * rng.normal(0.0, 1.0, length), 0.0, 1.0)
        h = np.tile(block, (n // length) + 1)[:n]
        h += 0.02 * rng.randn(n)
        return project_feasible(h, n)
    if mode == 3:
        v = rng.rand(n)
        h = np.where(v > np.median(v), 1.0, 0.0)
        h += 0.02 * rng.randn(n)
        return project_feasible(h, n)
    if mode == 4:
        h = base_h + (0.07 + 0.02 * (idx % 3)) * rng.randn(n)
        return project_feasible(h, n)
    alpha = 0.6180339887 + 0.09 * ((idx % 4) + 1)
    h = np.where(np.mod(np.arange(n) * alpha, 1.0) < 0.5, 1.0, 0.0)
    h += 0.02 * rng.randn(n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.RandomState(seed)
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_parent = np.full(700, 0.5)
    n_parent = len(h_parent)
    best_h = project_feasible(h_parent, n_parent)
    best_c = compute_c5(best_h)
    best_n = n_parent

    if budget_s > 70.0:
        p_dead = min(deadline, t0 + min(120.0, 0.12 * budget_s))
        h2, c2 = local_refine(best_h, len(best_h), p_dead, True)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, len(h2)

    max_n = max(n_parent, len(best_h))
    fixed = [24, 36, 48, 64, 96, 128, 192, 288, 448, 640]
    scales = [s for s in fixed if s < max_n]
    if 20 <= max_n <= 1200 and max_n not in scales:
        scales.append(max_n)
    scales = sorted(set(scales))
    if not scales:
        scales = [min(24, max_n)]

    for n in scales:
        if time.time() > deadline - 45.0:
            break
        rem0 = deadline - time.time()
        if rem0 < 25.0:
            break
        if n <= 30:
            tl = 90.0
        elif n <= 60:
            tl = 80.0
        elif n <= 130:
            tl = 60.0
        elif n <= 260:
            tl = 45.0
        elif n <= 510:
            tl = 35.0
        else:
            tl = 60.0
        tl = min(tl, 0.85 * rem0)
        scale_dead = min(deadline - 10.0, time.time() + tl)
        base = interpolate_to(best_h, n)

        if n <= 32:
            cand = min(8, max(3, int(tl / 12.0)))
        elif n <= 72:
            cand = min(6, max(2, int(tl / 18.0)))
        elif n <= 140:
            cand = min(4, max(2, int(tl / 22.0)))
        elif n <= 280:
            cand = min(3, max(1, int(tl / 28.0)))
        else:
            cand = 1
        total_cand = cand + 1
        for k in range(total_cand):
            if time.time() > scale_dead:
                break
            left_cand = total_cand - k
            cand_dead = min(scale_dead, time.time() + max(8.0, (scale_dead - time.time()) / float(left_cand)))
            if k == 0:
                h = base
            else:
                h = _raw_start(n, rng, k, base)
            h, c = local_refine(h, n, cand_dead, True)
            if c < best_c:
                best_h, best_c, best_n = h, c, len(h)
                base = interpolate_to(best_h, n)

        if time.time() < scale_dead:
            h2, c2 = local_refine(best_h if len(best_h) == n else interpolate_to(best_h, n), n, scale_dead, True)
            if c2 < best_c:
                best_h, best_c, best_n = h2, c2, len(h2)

    n_cur = len(best_h)
    if time.time() < deadline - 30.0:
        h2, c2 = local_refine(best_h, n_cur, deadline, True)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, len(h2)

    best_n_final = len(best_h)
    max_n = max(best_n_final, n_parent)
    r = 0
    while time.time() < deadline - 25.0:
        if r == 0:
            n_try = best_n_final
        elif r % 5 == 4 and best_n_final > 100:
            n_try = max(64, best_n_final - 32)
        elif r % 7 == 3:
            n_try = min(max_n, best_n_final + 32)
        else:
            n_try = best_n_final
        base_try = interpolate_to(best_h, n_try)
        if r % 3 == 0:
            h = project_feasible(base_try + (0.008 + 0.002 * (r % 6)) * rng.normal(0.0, 1.0, n_try), n_try)
        elif r % 3 == 1:
            x = np.linspace(0.0, 2.0, n_try)
            h = project_feasible(base_try + 0.018 * np.sin(np.pi * (1.0 + 0.4 * (r % 4)) * x), n_try)
        else:
            h = _raw_start(n_try, rng, r + 64, base_try)
        left = deadline - time.time()
        limit = min(deadline - 5.0, time.time() + max(12.0, min(35.0, 0.4 * left)))
        h2, c2 = local_refine(h, n_try, limit, True)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, len(h2)
            best_n_final = best_n
            max_n = max(max_n, best_n)
        r += 1

    if time.time() < deadline - 1.0:
        h2, c2 = coord_descent(best_h, len(best_h), 15, deadline, g_iters=30)
        if c2 < best_c:
            best_h, best_c = h2, c2
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h)
    return best_h, float(best_c), int(len(best_h))
