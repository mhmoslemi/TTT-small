For the gradient of the max correlation:
- Find active lags (within ε of max)
- For each active lag k, compute the gradient
- Average them

The gradient of c_k = sum_i h_i(1-h_{i+k}) w.r.t. h_j:
= (1 - h_{j+k}) * [0 ≤ j+k < n] - h_{j-k} * [0 ≤ j-k < n]

Using numpy:
```python
def max_correlation_gradient(h, n, active_lags=None, epsilon=0.01):
    c = np.correlate(h, 1.0 - h, mode='full')
    if active_lags is None:
        c_max = c.max()
        active = np.where(c >= c_max - epsilon)[0]
    else:
        active = active_lags
    grad = np.zeros(n)
    for k_idx in active:
        k = int(k_idx - (n - 1))
        if k >= 0:
            # h_{j+k} valid for j = 0..n-1-k
            grad[:n-k] += (1.0 - h[k:])
            # h_{j-k} valid for j = k..n-1
            grad[k:] -= h[:n-k]
        else:
            kk = -k
            # h_{j+k} = h_{j-kk} valid for j = 0..n-1-kk
            grad[:n-kk] += (1.0 - kk_offset(h, kk))  # need h[j-kk] for j >= kk
            # Actually let me be more careful
            # c_k = sum_{i} h_i (1 - h_{i+k}) where i+k >= 0 and i+k < n
            # If k < 0, let kk = -k, then i-kk >= 0 means i >= kk
            # c_k = sum_{i=kk}^{n-1} h_i (1 - h_{i-kk})
            # d(c_k)/d(h_j) = (1-h_{j-kk})*[j-kk valid] - h_{j+kk}*[j+kk valid]
            # For j >= kk: (1 - h_{j-kk})
            # For j + kk < n, i.e., j < n-kk: -h_{j+kk}
            grad[kk:n] += (1.0 - h[:n-kk])
            grad[:n-kk] -= h[kk:n]
    grad /= len(active)
    return grad
```

Wait, let me double-check the gradient formula.

c_k = sum_{i: valid} h_i * (1 - h_{i+k})

where valid means 0 ≤ i < n and 0 ≤ i+k < n.

d(c_k)/d(h_j) = sum_i [d/d(h_j)(h_i * (1-h_{i+k}))]
= sum_i [δ_{i,j} * (1-h_{i+k}) + h_i * (-δ_{i+k,j})]
= (1-h_{j+k}) * [j+k valid] - h_{j-k} * [j-k valid]

For k ≥ 0:
- j+k valid means 0 ≤ j+k < n, i.e., 0 ≤ j < n-k
- j-k valid means 0 ≤ j-k < n, i.e., k ≤ j < n+k, but j < n so k ≤ j < n

So:
- For j = 0, ..., n-k-1: contribution from first term = (1-h_{j+k})
- For j = k, ..., n-1: contribution from second term = -h_{j-k}

For k < 0, let kk = -k > 0:
- j+k = j-kk valid means 0 ≤ j-kk < n, i.e., kk ≤ j < n+kk, but j < n so kk ≤ j < n
- j-k = j+kk valid means 0 ≤ j+kk < n, i.e., 0 ≤ j < n-kk

So:
- For j = kk, ..., n-1: contribution from first term = (1-h_{j-kk})
- For j = 0, ..., n-kk-1: contribution from second term = -h_{j+kk}

This matches what I had. Good.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_eq(x, n):
    return x.sum() - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            if res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50, threshold=1e-5, seed=42):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    rng = np.random.RandomState(seed)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h[np.where(h < threshold)[0]] = 0.0
            h[np.where(h > 1.0 - threshold)[0]] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    final_c = compute_c5(h, n)
    return h, min(best_c, final_c)

def max_corr_gradient(h, n, epsilon=0.02):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    active = np.where(c >= c_max - epsilon)[0]
    grad = np.zeros(n)
    for k_idx in active:
        k = int(k_idx - (n - 1))
        if k >= 0:
            grad[:n - k] += (1.0 - h[k:])
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[kk:n] += (1.0 - h[:n - kk])
            grad[:n - kk] -= h[kk:n]
    grad /= max(len(active), 1)
    return grad

def flatten_obj(x, n, alpha, beta):
    c = np.correlate(x, 1.0 - x, mode='full')
    c_max = c.max()
    shifted = alpha * (c - c_max)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    top_k = min(20, len(c))
    top_vals = np.sort(c)[-top_k]
    spread = top_vals.max() - top_vals.min()
    return ((c_max + lse / alpha) * (2.0 / n) + beta * spread * (2.0 / n))

def run(seed=42, budget_s=1100, **kwargs):
```
