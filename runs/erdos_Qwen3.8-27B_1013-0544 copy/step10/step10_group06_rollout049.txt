And the original is ∫h(x)(1-h(x+k)) dx.

These are related but not the same in general. However, if we also shift: ∫(1-h(x)) h(x+k) dx vs ∫h(x)(1-h(x+k)) dx.
Let y = x+k: ∫(1-h(y-k)) h(y) dy = ∫h(y)dy - ∫h(y)h(y-k)dy = 1 - ∫h(y)h(y-k)dy

And the original: ∫h(x)(1-h(x+k))dx = ∫h(x)dx - ∫h(x)h(x+k)dx = 1 - ∫h(x)h(x+k)dx

So C₅(1-h) = max_k (1 - ∫h(y)h(y-k)dy) and C₅(h) = max_k (1 - ∫h(x)h(x+k)dx).

Since the max is over all k (positive and negative), these are the same! Because {-(k)} = {k} as k ranges over all integers.

So C₅(1-h) = C₅(h). Great, so the complement is always as good.

Another idea: **circular shifts**. If we shift h cyclically, does C₅ change? In the continuous setting, a shift of h by s gives:
∫h(x-s)(1-h(x+k-s)) dx = ∫h(u)(1-h(u+k)) du (substituting u = x-s)

So circular shifts preserve C₅! This means we have a lot of symmetry.

Given these symmetries, the search space is still large but we can use them for validation.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n, nbisect=40):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo_a = max(0, n - 1 - p)
    hi_a = min(2 * n - 2, n - 1 + p)
    lo_b = max(0, p)
    hi_b = min(2 * n - 2, p + n - 1)

    m_hi = max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)

    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))

        a[vf] += (1.0 - h[jf[vf]])

    c_diag = 0.0

        a[diag_idx] = 0.0

    if has_diag:
        non_diag[diag_idx] = False

def coord_descent(h, n, max_passes, deadline, rng, random_order=True, nbisect=40):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n, nbisect)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_loss(x, n, alpha):
    """Smooth approximation to max correlation using log-sum-exp."""
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def lse_into_h(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    return c, weights

def lse_grad(x, n, alpha):
    c, weights = lse_into_h(x, n, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        kf = k - (n - 1)
        for i in range(n):
            jf = i + kf
            if 0 <= jf < n:
                grad[i] += w * (1.0 - x[jf])
            jb = i - kf
            if 0 <= jb < n:
                grad[i] -= w * x[jb]
    grad *= dx
    return lse_loss(x, n, alpha), grad

def lse_opt(h, n, alphas, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        bounds = [(0.0, 1.0)] * n
        sum_con = lambda x: x.sum() - n / 2.0
        sum_jac = lambda x: np.ones(n)
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
        try:
            res = minimize(lse_loss, best_h, args=(n, alpha), jac=lambda x: lse_grad(x, n, alpha)[1],
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    dx = 2.0 / n
    # grad_p = dx * sum_k w_k * [(1-h_{p-k}) * 1[0≤p-k<n] - h_{p+k} * 1[0≤p+k<n]]
    # Forward: sum_k w_k * (1-h_{p-k}) where 0≤p-k<n
    # This is like a correlation of (1-h) with w, but truncated
    # Let's compute it as: for each p, the sum is over k from max(-(p), -(n-1)) to min(n-1-p, n-1)
    # Actually, let's use the full correlation and mask
    # np.correlate(1-h, w, mode='full') doesn't quite work because w has length 2n-1
    
    # Alternative: use the direct formula
    # forward_p = sum_{k=-(n-1)}^{n-1} w[k+n-1] * (1-h[p-k]) * 1[0≤p-k<n]
    # = sum_{j=0}^{n-1} w[p-j+n-1] * (1-h[j]) * 1[0≤p-j+n-1<2n-1]
    # = sum_{j=0}^{n-1} w[n-1+(p-j)] * (1-h[j])
    
    # This is a convolution of (1-h) with w, evaluated at specific points
    # np.convolve(1-h, w, mode='full') has length n + (2n-1) - 1 = 3n-2
    # The element at position p + (n-1) in the full convolution gives:
    # sum_j (1-h[j]) * w[n-1+p-j]
    
    full_conv_fw = np.convolve(1.0 - x, weights, mode='full')
    # full_conv_fw[i] = sum_j (1-x[j]) * weights[i-j]
    # We want: sum_j (1-x[j]) * weights[n-1+p-j] for p = 0,...,n-1
    # = full_conv_fw[n-1+p]
    forward = full_conv_fw[n-1:n-1+n]
    
    # backward_p = sum_{k=-(n-1)}^{n-1} w[k+n-1] * h[p+k] * 1[0≤p+k<n]
    # = sum_{j=0}^{n-1} w[n-1+p+j-n] ... hmm this is getting confusing
    
    # Let me just do it directly but vectorized
    # For each p, backward_p = sum_{k: 0≤p+k<n} w[k+n-1] * h[p+k]
    # Let j = p+k, so k = j-p, and 0≤j<n, 0≤j-p<2n-1 → -p≤k<2n-1-p → 0≤k≤n-1-p (since k≤n-1)
    # Also k≥-(n-1) is always satisfied since k = j-p ≥ -p ≥ -(n-1)
    # So backward_p = sum_{j=p}^{n-1} w[j+n-1-p] * h[j]
    
    # This is also a convolution: sum_j h[j] * w[n-1+(j-p)]
    # = (h * w)[n-1-p + n-1]... let me just use cross-correlation
    
    # Actually, simpler: backward is the "reverse" of forward
    # backward_p = sum_{j=p}^{n-1} w[n-1+j-p] * h[j]
    # Let i = j-p: backward_p = sum_{i=0}^{n-1-p} w[n-1+i] * h[p+i]
    
    # This is a cross-correlation of h with w[n-1:], evaluated at lag p
    # Actually it's more like: for each p, dot(h[p:], w[n-1:n-1+(n-p)])
    
    # Let me just use a loop over p but with numpy operations
    backward = np.zeros(n)
    for p in range(n):
        backward[p] = np.dot(x[p:], weights[n-1:n-1+n-p])
    
    grad = dx * (forward - backward)
    obj = (m + (mx + np.log(weights.sum())) / alpha) * dx
    return obj, grad
