import time
import numpy as np
from scipy.optimize import minimize

def corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5(h, n):
    return float(np.max(corr(h)) * (2.0 / n))

def proj(h, n):
    h = np.clip(np.array(h, dtype=float, copy=True), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(60):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s <= 1e-15:
                break
            h = np.clip(h + d * r / s, 0.0, 1.0)
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h = np.clip(h + d * h / s, 0.0, 1.0)
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = np.flatnonzero(h < 1.0)
            if m.size == 0:
                break
            h[m] += d / m.size
            h[np.clip(h - 1.0, 0.0, None)] = 1.0
        else:
            m = np.flatnonzero(h > 0.0)
            if m.size == 0:
                break
            h[m] += d / m.size
            h[np.clip(h, 0.0, 1.0)] = h
    return h

def soft_jac(x, T, K):
    n = x.size
    c = corr(x)
    f = c.size
    if K > 0 and K < f:
        idx = np.argpartition(c, -K)[-K:]
        ct = c[idx]
        mx = float(ct.max())
        e = np.exp((ct - mx) / T)
        se = float(e.sum())
        w = np.zeros(f)
        w[idx] = e / se
        val = (mx + T * np.log(se)) * (2.0 / n)
    else:
        mx = float(c.max())
        e = np.exp((c - mx) / T)
        se = float(e.sum())
        w = e / se
        val = (mx + T * np.log(se)) * (2.0 / n)
    cw = np.cumsum(w)
    p = np.arange(n)
    sf = cw[2 * n - 1 - p] - cw[n - 1 - p]
    sb = cw[p + n] - cw[p]
    g = (2.0 / n) * ((1.0 - x) * sf - x * sb)
    return val, g

def pen_obj(x, T, K, lam):
    v, _ = soft_jac(x, T, K)
    e = x.sum() - 0.5 * x.size
    return v + 0.5 * lam * e * e

def pen_jac(x, T, K, lam):
    _, g = soft_jac(x, T, K)
    e = x.sum() - 0.5 * x.size
    return g + lam * e * np.ones_like(x)

def lbfgs(h, n, T, K, lam, it):
    try:
        res = minimize(pen_obj, h, args=(T, K, lam), jac=pen_jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': it, 'ftol': 1e-15, 'gtol': 1e-10,
                                'maxfun': max(100, 8 * it)})
        h = res.x
    except Exception:
        pass
    return proj(h, n)

def solvedir(h, c, p, n):
    fl = 2 * n - 1
    kf0 = n - 1 - p
    kf1 = 2 * n - 2 - p
    kb0 = p
    kb1 = p + n - 1
    kmin = min(kf0, kb0)
    kmax = max(kf1, kb1)
    L = kmax - kmin + 1
    cb = c[kmin:kmax + 1]
    a = np.zeros(L)
    s0 = kf0 - kmin
    a[s0:s0 + n] += 1.0 - h
    s1 = kb0 - kmin
    a[s1:s1 + n] -= h[::-1]
    d = n - 1 - kmin
    un = 0.0
    if kmin > 0:
        un = max(un, float(c[:kmin].max()))
    if kmax < fl - 1:
        un = max(un, float(c[kmax + 1:].max()))
    cm = max(un, float(cb.max()))
    x0 = h[p]
    if cm <= un + 1e-13:
        return x0, kmin, kmax, a
    pos = a > 1e-14
    neg = a < -1e-14
    zero = ~(pos | neg)
    ad = a[d]
    cd0 = cb[d]
    lo, hi = un, cm
    for _ in range(45):
        mv = 0.5 * (lo + hi)
        xl, xh = 0.0, 1.0
        ok = True
        if np.any(zero) and np.any(cb[zero] > mv + 1e-12):
            ok = False
        if ok and np.any(pos):
            xh = min(xh, float(np.min(x0 + (mv - cb[pos]) / a[pos])))
        if ok and np.any(neg):
            xl = max(xl, float(np.max(x0 + (mv - cb[neg]) / a[neg])))
        if ok and xl > xh:
            ok = False
        if ok:
            disc = ad * ad + 4.0 * (cd0 - mv)
            if disc > 1e-12:
                sq = np.sqrt(disc)
                L1 = x0 + (ad - sq) / 2.0
                R1 = x0 + (ad + sq) / 2.0
                if xl > L1 and xh < R1:
                    ok = False
        if ok:
            hi = mv
        else:
            lo = mv
    mv = hi
    xl, xh = 0.0, 1.0
    if np.any(pos):
        xh = min(xh, float(np.min(x0 + (mv - cb[pos]) / a[pos])))
    if np.any(neg):
        xl = max(xl, float(np.max(x0 + (mv - cb[neg]) / a[neg])))
    xl = max(0.0, min(xl, 1.0))
    xh = min(1.0, max(xh, 0.0))
    if xl > xh:
        return x0, kmin, kmax, a
    intervals = []
    disc = ad * ad + 4.0 * (cd0 - mv)
    if disc > 1e-12:
        sq = np.sqrt(disc)
        L1 = x0 + (ad - sq) / 2.0
        R1 = x0 + (ad + sq) / 2.0
        if xl <= L1:
            intervals.append((xl, min(xh, L1)))
        if xh >= R1:
            intervals.append((max(xl, R1), xh))
    else:
        intervals.append((xl, xh))
    bestx = x0
    bests = (cm, float('inf'), -1.0)
    cands = []
    for A, B in intervals:
        if B < A:
            continue
        for q in (A, B, 0.5 * (A + B)):
            cands.append(float(np.clip(q, 0.0, 1.0)))
    for q in cands:
        dl = q - x0
        cn = cb + a * dl
        cn[d] -= dl * dl
        pm = max(un, float(cn.max()))
        if pm < cm + 1e-12:
            act = cb > cm - 0.5
            sec = float(cn[act].sum()) if np.any(act) else float(cn.max())
            sc = (pm, sec, abs(dl))
            if sc < bests:
                bests = sc
                bestx = q
    return bestx, kmin, kmax, a

def cd_polish(h, n, passes, stop, random=False, rng=None):
    h = h.copy()
    c = corr(h)
    best = c5(h, n)
    best_h = h.copy()
    for _ in range(passes):
        if time.time() > stop:
            break
        order = rng.permutation(n) if random else np.arange(n)
        imp = False
        for p in order:
            x0 = h[p]
            xn, k0, k1, a = solvedir(h, c, p, n)
            d = xn - x0
            if abs(d) > 1e-12:
                c[k0:k1 + 1] += a * d
                c[n - 1] -= d * d
                h[p] = xn
                imp = True
        h = proj(h, n)
        c = corr(h)
        cc = float(np.max(c) * (2.0 / n))
        if cc < best:
            best = cc
            best_h = h.copy()
        if not imp:
            break
    return best_h, best

def pair_polish(h, stop, passes=120):
    n = len(h)
    h = h.copy()
    best = c5(h, n)
    for _ in range(passes):
        if time.time() > stop:
            break
        c = corr(h)
        if c.size >= 8:
            topc = np.argpartition(c, c.size - 8)[-8:][::-1]
        else:
            topc = np.argsort(c)[::-1]
        done = False
        for k in topc:
            if time.time() > stop:
                break
            s = k - (n - 1)
            i0 = max(0, -s)
            i1 = n - max(0, s)
            if i1 <= i0:
                continue
            ii = np.arange(i0, i1)
            jj = ii + s
            tv = h[ii] * (1.0 - h[jj])
            if tv.size >= 2:
                tp = np.argpartition(tv, tv.size - 2)[-2:]
            else:
                tp = np.argsort(tv)[::-1]
            for oi in tp[::-1]:
                i = int(ii[oi])
                j = int(jj[oi])
                dm = min(h[i], 1.0 - h[j])
                if dm < 1e-10:
                    continue
                bv = best
                bh = h
                for fr in (0.5, 1.0, 0.25, 0.75, 0.1):
                    d = fr * dm
                    d = min(d, h[i], 1.0 - h[j])
                    if d < 1e-12:
                        continue
                    nh = h.copy()
                    nh[i] -= d
                    nh[j] += d
                    v = c5(nh, n)
                    if v < bv:
                        bv = v
                        bh = nh
                if bv < best - 1e-12:
                    h = bh
                    best = bv
                    done = True
                    break
            if done:
                break
        if not done:
            break
    return h, best

def f_init(n, rng, K=12):
    x = 2.0 * np.arange(n) / n
    h = 0.5 * np.ones(n)
    for k in range(1, K + 1):
        h += rng.normal(0.0, 0.25 / np.sqrt(k)) * np.cos(2.0 * np.pi * k * x / 2.0 + rng.uniform(0.0, 2.0 * np.pi))
    return proj(h, n)

def resample(h, nt):
    a = np.linspace(0.0, 2.0, len(h), endpoint=False)
    b = np.linspace(0.0, 2.0, nt, endpoint=False)
    return proj(np.interp(b, a, h), nt)
