def project_feasible(h,n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5*n
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-14: break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not mask.any(): break
            add = diff / mask.sum()
            h[mask] = np.clip(h[mask] + add, 0.0, 1.0)
        else:
            mask = h > 1e-14
            if not mask.any(): break
            add = diff / mask.sum()
            h[mask] = np.clip(h[mask] + add, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
       if diff>0:
          order=np.argsort(1.0-h)
          for idx in order:
             if diff <= 1e-15: break
             room = 1.0 - h[idx]
             take = room if room < diff else diff
             h[idx] += take; diff -= take
       else:
          order=np.argsort(h)
          for idx in order:
             if -diff <= 1e-15: break
             room = h[idx]
             take = room if room < -diff else -diff
             h[idx] -= take; diff += take
    return h

import numpy as np
import time
import math

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = 0.5 * n
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not mask.any():
                break
            add = diff / mask.sum()
            h[mask] = np.clip(h[mask] + add, 0.0, 1.0)
        else:
            mask = h > 1e-14
            if not mask.any():
                break
            add = diff / mask.sum()
            h[mask] = np.clip(h[mask] + add, 0.0, 1.0)

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                if diff <= 1e-15:
                    break
                room = 1.0 - h[idx]
                take = room if room < diff else diff
                h[idx] += take
                diff -= take
        else:
            order = np.argsort(h)
            for idx in order:
                if -diff <= 1e-15:
                    break
                room = h[idx]
                take = room if room < -diff else -diff
                h[idx] -= take
                diff += take
    return h

def coord_range(p, n):
    return min(p, n - 1 - p), max(p + n - 1, 2 * n - 2 - p)

def add_coeff(seg, h, p, n, coef, lo, hi):
    a_lo = p
    a_hi = p + n - 1
    l = max(lo, a_lo)
    r = min(hi, a_hi)
    if l <= r:
        length = r - l + 1
        off = l - lo
        start = p - l + n - 1
        lower = start - length + 1
        if lower <= 0:
            rev = h[start::-1]
            if rev.size > length:
                rev = rev[:length]
        else:
            rev = h[start:start - length:-1]
        seg[off:off + length] += coef * (1.0 - rev)

    v_lo = n - 1 - p
    v_hi = 2 * n - 2 - p
    l = max(lo, v_lo)
    r = min(hi, v_hi)
    if l <= r:
        length = r - l + 1
        off = l - lo
        start = p + l - n + 1
        seg[off:off + length] -= coef * h[start:start + length]

def pair_solve(corr, h, p, q, n, dx, gmax, iters):
    if p == q:
        return None
    x = float(h[p])
    y = float(h[q])
    tl = max(-x, y - 1.0)
    th = min(1.0 - x, y)
    if th - tl < 1e-14:
        return None

    lo1, hi1 = coord_range(p, n)
    lo2, hi2 = coord_range(q, n)
    lo = lo1 if lo1 < lo2 else lo2
    hi = hi1 if hi1 > hi2 else hi2
    L = hi - lo + 1
    M = 2 * n - 1

    U = -1.0
    if lo > 0:
        U = float(np.max(corr[:lo]))
    if hi + 1 < M:
        v = float(np.max(corr[hi + 1:]))
        if v > U:
            U = v

    alpha = np.zeros(L)
    add_coeff(alpha, h, p, n, 1.0, lo, hi)
    add_coeff(alpha, h, q, n, -1.0, lo, hi)
    alpha *= dx

    cseg = corr[lo:hi + 1]
    center = n - 1 - lo
    cross_pos = []
    d = abs(p - q)
    m1 = n - 1 + d
    if lo <= m1 <= hi:
        cross_pos.append(m1 - lo)
    m2 = n - 1 - d
    if lo <= m2 <= hi:
        ip2 = m2 - lo
        if ip2 != (cross_pos[-1] if cross_pos else -1):
            cross_pos.append(ip2)

    lin = np.ones(L, dtype=bool)
    lin[center] = False
    for ip in cross_pos:
        lin[ip] = False

    eps = 1e-14
    posm = lin & (alpha > eps)
    negm = lin & (alpha < -eps)
    zerm = lin & ~(posm | negm)

    low = U
    if posm.any():
        low = max(low, float(np.max(cseg[posm] + alpha[posm] * tl)))
    if negm.any():
        low = max(low, float(np.max(cseg[negm] + alpha[negm] * th)))
    if zerm.any():
        low = max(low, float(np.max(cseg[zerm])))

    if cross_pos:
        cc = cseg[cross_pos]
        ac = alpha[cross_pos]
        tstar = np.clip(-ac / (2.0 * dx), tl, th)
        low = max(low, float(np.max(cc + ac * tstar + dx * tstar * tstar)))
    else:
        cc = np.empty(0)
        ac = np.empty(0)

    c0 = float(cseg[center])
    a0 = float(alpha[center])
    low = max(low, min(c0 + a0 * tl - 2.0 * dx * tl * tl,
                       c0 + a0 * th - 2.0 * dx * th * th))

    if low >= gmax - 1e-12:
        return None
    high = gmax
    if high - low <= 1e-12:
        return None

    cp = cseg[posm]
    ap = alpha[posm]
    cn = cseg[negm]
    an = alpha[negm]
    constmax = float(np.max(cseg[zerm])) if zerm.any() else -1e300
    tol = 1e-12

    for _ in range(iters):
        if high - low <= tol:
            break
        mid = 0.5 * (low + high)
        feasible = True
        tlo = tl
        thi = th

        if constmax > mid + tol:
            feasible = False
        if feasible and ap.size:
            up = float(np.min((mid - cp) / ap))
            if up < thi:
                thi = up
        if feasible and an.size:
            dn = float(np.max((mid - cn) / an))
            if dn > tlo:
                tlo = dn
        if feasible and tlo > thi + tol:
            feasible = False

        if feasible:
            for i in range(cc.size):
                aa = float(ac[i])
                cci = float(cc[i])
                D = aa * aa - 4.0 * dx * (cci - mid)
                if D < -1e-12:
                    feasible = False
                    break
                if D < 0.0:
                    D = 0.0
                s = math.sqrt(D)
                r1 = (-aa - s) / (2.0 * dx)
                r2 = (-aa + s) / (2.0 * dx)
                if r1 > tlo:
                    tlo = r1
                if r2 < thi:
                    thi = r2
                if tlo > thi + tol:
                    feasible = False
                    break

        if feasible:
            D = a0 * a0 - 8.0 * dx * (mid - c0)
            if D >= 0.0:
                if D < 0.0:
                    D = 0.0
                s = math.sqrt(D)
                r1 = (a0 - s) / (4.0 * dx)
                r2 = (a0 + s) / (4.0 * dx)
                if not (tlo <= r1 + tol or thi >= r2 - tol):
                    feasible = False

        if feasible:
            high = mid
        else:
            low = mid

    Lstar = high
    cands = [tl, th, 0.5 * (tl + th)]
    tlo = tl
    thi = th
    if ap.size:
        up = float(np.min((Lstar - cp) / ap))
        if up < thi:
            thi = up
    if an.size:
        dn = float(np.max((Lstar - cn) / an))
        if dn > tlo:
            tlo = dn
    cands.extend([tlo, thi])

    for i in range(cc.size):
        aa = float(ac[i])
        cci = float(cc[i])
        D = aa * aa - 4.0 * dx * (cci - Lstar)
        if D >= -1e-12:
            if D < 0.0:
                D = 0.0
            s = math.sqrt(D)
            cands.append((-aa - s) / (2.0 * dx))
            cands.append((-aa + s) / (2.0 * dx))

    Dc = a0 * a0 - 8.0 * dx * (Lstar - c0)
    if Dc >= -1e-12:
        if Dc < 0.0:
            Dc = 0.0
        s = math.sqrt(Dc)
        cr1 = (a0 - s) / (4.0 * dx)
        cr2 = (a0 + s) / (4.0 * dx)
        cands.extend([cr1, cr2])
        if tlo <= cr1 + tol:
            right = min(thi, cr1)
            if right > tlo:
                cands.extend([tlo, right, 0.5 * (tlo + right)])
        if thi >= cr2 - tol:
            left = max(tlo, cr2)
            if thi > left:
                cands.extend([left, thi, 0.5 * (left + thi)])
    else:
        if thi > tlo:
            cands.append(0.5 * (tlo + thi))

    best_t = 0.0
    bestF = gmax
    for tt in cands:
        if tt < tl - 1e-12 or tt > th + 1e-12:
            continue
        if tt < tl:
            tt = tl
        elif tt > th:
            tt = th
        t2 = tt * tt
        vals = cseg + alpha * tt
        vals[center] -= 2.0 * dx * t2
        if cross_pos:
            vals[cross_pos] += dx * t2
        vm = float(np.max(vals))
        F = U if U > vm else vm
        if F < bestF:
            bestF = F
            best_t = tt

    if bestF >= gmax - 1e-12:
        return None

    hp = x + best_t
    hq = y - best_t
    need_recompute = False
    if hp < 0.0:
        hp = 0.0
        need_recompute = True
    elif hp > 1.0:
        hp = 1.0
        need_recompute = True
    if hq < 0.0:
        hq = 0.0
        need_recompute = True
    elif hq > 1.0:
        hq = 1.0
        need_recompute = True
    if abs(hp - (x + best_t)) > 1e-13 or abs(hq - (y - best_t)) > 1e-13:
        need_recompute = True

    h[p] = hp
    h[q] = hq

    if need_recompute:
        corr[:] = np.correlate(h, 1.0 - h, mode="full") * dx
        return float(corr.max())

    vals = cseg + alpha * best_t
    vals[center] -= 2.0 * dx * best_t * best_t
    if cross_pos:
        vals[cross_pos] += dx * best_t * best_t
    newseg = np.maximum(vals, 0.0)
    corr[lo:hi + 1] = newseg
    nm = float(np.max(newseg))
    return U if U > nm else nm

def optimize_pairs(h0, deadline, rng, iters=18):
    h = project_feasible(np.asarray(h0, dtype=float).copy(), len(h0))
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    gmax = float(corr.max())
    best_h = h.copy()
    best_c = gmax
    no_improve = 0

    while time.time() < deadline:
        pass_h = h.copy()
        pass_corr = corr.copy()
        pass_g = gmax
        improved = False
        idx = rng.permutation(n)

        for k in range(0, n - 1, 2):
            if time.time() >= deadline:
                break
            p = int(idx[k])
            q = int(idx[k + 1])
            res = pair_solve(corr, h, p, q, n, dx, gmax, iters)
            if res is not None:
                gmax = res
                improved = True

        if abs(h.sum() - 0.5 * n) > 1e-10:
            h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        gmax = float(corr.max())

        if gmax > pass_g + 1e-9:
            h = pass_h
            corr = pass_corr
            gmax = pass_g
        elif gmax < pass_g - 1e-10:
            improved = True

        if gmax < best_c - 1e-15:
            best_c = gmax
            best_h = h.copy()

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

    return best_h, float(best_c)

def resize_h(h, n_new):
