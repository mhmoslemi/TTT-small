        viol = np.sum(np.maximum(-h, 0)**2) + np.sum(np.maximum(h-1, 0)**2)
        return 1.0 + 100.0 * viol
    h = project_feasible(h, n)
    return compute_c5(h, n)

def solve_coord_fast(c, h, p, n, nbisect=35):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo_a = max(0, n - 1 - p)
    hi_a = min(2 * n - 2, n - 1 + p)
    lo_b = max(0, p)
    hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(lo_a, lo_b)
    m_hi = max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    non_diag = np.ones(len(ms), dtype=bool)
    if has_diag:
        non_diag[diag_idx] = False
    a_nd = a[non_diag]
    c_nd = c_base[non_diag]
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(c[m_hi + 1:])))
    L_lo = unaff_max
    L_hi = float(np.max(c))
    for _ in range(nbisect):
        if L_hi - L_lo < 1e-14:
            break
        T = (L_lo + L_hi) / 2.0
        feasible = True
        if unaff_max > T + 1e-15:
            feasible = False
        elif feasible and len(a_nd) > 0:
            x_lo_c, x_hi_c = 0.0, 1.0
            neg = a_nd < -1e-13
            pos = a_nd > 1e-13
            if np.any(neg):
                x_lo_c = max(x_lo_c, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                x_hi_c = min(x_hi_c, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            if x_lo_c > x_hi_c + 1e-14:
                feasible = False
        if feasible and has_diag:
            Cq = T - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if len(a_nd) > 0:
                    neg = a_nd < -1e-13
                    pos = a_nd > 1e-13
                    xl, xh = 0.0, 1.0
                    if np.any(neg):
                        xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                    xl, xh = max(xl, 0.0), min(xh, 1.0)
                    if xl > xh + 1e-14 or (xl >= r1 and xh <= r2):
                        feasible = False
                elif 0.0 >= r1 and 1.0 <= r2:
                    feasible = False
        L_hi = T if feasible else L_lo
        if not feasible:
            L_lo = T
    T = (L_lo + L_hi) / 2.0
    x_lo, x_hi = 0.0, 1.0
    if len(a_nd) > 0:
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo < r1:
                x_hi = min(x_hi, r1)
            elif x_hi > r2:
                x_lo = max(x_lo, r2)
            else:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1 + 1e-9)
                else:
                    x_lo = max(x_lo, r2 - 1e-9)
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng=None, nbisect=35):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = np.arange(n) if rng is None else rng.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord_fast(c, h, p, n, nbisect)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_phase(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n/2.0, 'jac': lambda v: np.ones(n)}]
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 700
        x = np.linspace(0, 2, n)
        h_init = np.clip(0.5 + 0.2*np.sin(np.pi*x), 0, 1)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    x = np.linspace(0, 2, n)
    # Phase 1: CD baseline
    h1, c1 = coord_descent(best_h, n, 20, deadline, rng, 30)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Fourier Nelder-Mead with perturbations
    for J in [3, 5, 7]:
        if time.time() > deadline - 120:
            break
        params0 = h_to_fourier(best_h, x, J)
        for trial in range(3):
            if time.time() > deadline - 30:
                break
            scale = 0.015 * (trial + 1)
            params_init = params0 + rng.randn(2*J) * scale
            try:
                res = minimize(fourier_cost, params_init, args=(x, J, n),
                               method='Nelder-Mead',
                               options={'maxiter': 400, 'xatol': 1e-9, 'fatol': 1e-11})
                h_f = project_feasible(fourier_build(res.x, x, J), n)
                c_f = compute_c5(h_f, n)
                if c_f < best_c:
                    best_h, best_c = h_f, c_f
            except Exception:
                pass
    # Phase 3: CD refinement
    h3, c3 = coord_descent(best_h, n, 60, deadline, rng, 35)
    if c3 < best_c:
        best_h, best_c = h3, c3
    # Phase 4: SLSQP on grid
    if time.time() < deadline - 100:
        h4, c4 = slsqp_phase(best_h, n, [500, 2000, 8000], 300, deadline)
        if c4 < best_c:
            best_h, best_c = h4, c4
    # Phase 5: Multi-resolution
    for n_try in [840, 1000, 560]:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x, best_h), n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, 40, deadline, rng, 30)
        if c_opt < best_c:
            best_h, best_c, n, x = h_opt, c_opt, n_try, x_new
    # Phase 6: Fourier shake + CD
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        x_cur = np.linspace(0, 2, n_cur)
        for J in [4, 6]:
            if time.time() > deadline - 20:
                break
            p0 = h_to_fourier(best_h, x_cur, J)
            for trial in range(4):
                if time.time() > deadline - 10:
                    break
                p_w = p0 + rng.randn(2*J) * 0.008 * (trial+1)
                h_w = project_feasible(fourier_build(p_w, x_cur, J), n_cur)
                h_w, c_w = coord_descent(h_w, n_cur, 25, deadline, rng, 30)
                if c_w < best_c:
                    best_h, best_c = h_w, c_w
    # Phase 7: Final polish
    h_f, c_f = coord_descent(best_h, len(best_h), 30, deadline, None, 40)
    if c_f < best_c:
        best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
