    """Take a subgradient step to reduce the max correlation."""
    dx = 2.0 / n
    c_max = c.max()

    tol = c_max - 1e-10
    active = np.where(c >= tol)[0]
    if len(active) == 0:
        return h
    # Compute subgradient: average of dc/dh over active shifts
    g = np.zeros(n)
    for k in active:
        s = k - (n - 1)  # shift
        if s >= 0:
            # c = sum_{i=s}^{n-1} h[i]*(1-h[i-s])
            for p in range(s, n):
                g[p] += (1.0 - h[p - s])
            for p in range(0, n - s):
                g[p] -= h[p + s]
        else:
            sp = -s
            for p in range(0, n - sp):
                g[p] += (1.0 - h[p + sp])
            for p in range(sp, n):
                g[p] -= h[p - sp]
    g /= len(active)
    # Step in negative gradient direction
    h_new = h - eta * g
    h_new = project_feasible(h_new, n)
    c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
    return h_new, c_new

        h7, c7, v7 = subgradient_polish(h7, 500, 80, min(t0 + budget_s * 0.94, deadline), 0.005)
        if v7 < best_c:
            h_best, best_c = h7, v7

    # Phase 8: Final polish
    n_final = len(h_best)
    if time.time() < deadline - 5:
        h8, c8, v8 = cd_optimize(h_best, n_final, 50, deadline, None, False)
        if v8 < best_c:
            h_best, best_c = h8, v8
        h9, c9, v9 = subgradient_polish(h_best, n_final, 50, deadline, 0.003)
        if v9 < best_c:
            h_best, best_c = h9, v9

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
