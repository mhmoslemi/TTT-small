def subgradient_step(h, n, target_c, lr, deadline):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

            continue

def smooth_obj(trust_alpha):
    def obj(h, n):
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        shifted = trust_alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / trust_alpha) * (2.0 / n)
    return obj

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(90):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if xc_lo < r1 and xc_hi > r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return c
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_i = np.where(ms == (n - 1))[0]
    c[ms] += delta * a
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta
    return c

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = list(range(n))
        if rng is not None:
            if it % 4 == 1:
                rng.shuffle(order)
            elif it % 4 == 2:
                order = order[::-1]
            elif it % 4 == 3:
                order = list(range(n - 1, -1, -1))
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        if it % 8 == 7:
            h = project_feasible(h, n)
            if abs(h.sum() - n / 2.0) > 1e-10:
                c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj_alpha(x):
    def obj(h, n):
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        shifted = x * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / x) * (2.0 / n)
    return obj

def slsqp_refine(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def cf(x):
            return x.sum() - n / 2.0
        def cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        obj_fn = smooth_obj_alpha(alpha)
        try:
            res = minimize(obj_fn, best_h, args=(n,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-20})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

    # Phase 2: SLSQP smooth refinement with progressive sharpness
    h2, c2 = slsqp_refine(h_best, n, [2000, 8000, 30000, 100000, 500000], 500, deadline)
    if c2 < best_c - 1e-12:
        h_best, best_c = h2, c2

    # Phase 3: CD after SLSQP
    h3, c3 = exact_cd(h_best, n, 400, deadline, rng)
    if c3 < best_c - 1e-12:
        h_best, best_c = h3, c3

    # Phase 4: Explore nearby n values
    n_list = [n - 4, n - 2, n + 2, n + 4, n - 8, n + 8, n - 16, n + 16]
    n_list = sorted(set(x for x in n_list if 200 <= x <= 2000))
    for n_try in n_list:
        if time.time() > deadline - 180:
            break
        h_int = interp_h(h_best, len(h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 250, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o
        if time.time() < deadline - 80:
            h_o2, c_o2 = slsqp_refine(h_o, n_try, [50000, 200000], 250, deadline)
            if c_o2 < best_c - 1e-9:
                h_best, best_c = h_o2, c_o2
            h_o3, c_o3 = exact_cd(h_o2, n_try, 200, deadline, rng)
            if c_o3 < best_c - 1e-9:
                h_best, best_c = h_o3, c_o3

    # Phase 5: Structured perturbations (decreasing scale)
    n = len(h_best)
    scales = [0.06, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    for scale in scales:
        if time.time() > deadline - 50:
            break
        for r in range(4):
            if time.time() > deadline - 20:
                break
            hp = project_feasible(h_best + rng.normal(0, scale, n), n)
            h_o, c_o = exact_cd(hp, n, 150, deadline, rng)
            if c_o < best_c - 1e-9:
                h_best, best_c = h_o, c_o

    # Phase 6: Final long CD
    n = len(h_best)
    h_f, c_f = exact_cd(h_best, n, 500, deadline, rng)
    if c_f < best_c - 1e-12:
        h_best, best_c = h_f, c_f

    # Phase 7: Final SLSQP + CD polish
    if time.time() < deadline - 40:
        h_p, c_p = slsqp_refine(h_best, n, [200000, 1000000], 300, deadline)
        if c_p < best_c - 1e-12:
            h_best, best_c = h_p, c_p
        h_g, c_g = exact_cd(h_best, n, 300, deadline, rng)
        if c_g < best_c - 1e-12:
            h_best, best_c = h_g, c_g

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
