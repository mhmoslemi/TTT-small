import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx


def c_from_corr(c, n):
    return float(np.max(c)) * (2.0 / n)


def project_feasible(v, n):
    x = np.asarray(v, dtype=float).reshape(-1).copy()
    target = n / 2.0
    np.clip(x, 0.0, 1.0, out=x)
    if abs(float(x.sum()) - target) < 1e-15:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

        diff = target - float(y.sum())
        if abs(diff) < 1e-14:

        if diff > 0.0:

def default_h(n):
    x = np.linspace(0.0, 2.0, n)
    g = np.cos(math.pi * (x - 1.0))
    return project_feasible(0.5 + 0.35 * g, n)


def resample(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    xo = np.linspace(0.0, 2.0, n_old)
    xn = np.linspace(0.0, 2.0, n_new)
    return project_feasible(np.interp(xn, xo, h_old), n_new)


def pair_intervals(L, lo, hi, c0max, c_pos, A_pos, c_neg, A_neg, sA, sQ, sC):
    if c0max > L + 1e-13:
        return []
    a = lo
    b = hi
    if c_pos.size:
        ub = float(np.min((L - c_pos) / A_pos))
        if ub < b:
            b = ub
    if c_neg.size:
        lb = float(np.max((L - c_neg) / A_neg))
        if lb > a:
            a = lb
    if a > b + 1e-12:
        return []
    if a < lo:
        a = lo
    if b > hi:
        b = hi
    if a > b + 1e-12:
        return []

    inters = [(a, b)]
    for i in range(3):
        Qi = sQ[i]
        As = sA[i]
        cs = sC[i]
        if Qi > 0.0:
            disc = As * As - 4.0 * Qi * (cs - L)
            if disc < -1e-12:
                return []
            if disc < 0.0:
                disc = 0.0
            sq = math.sqrt(disc)
            r1 = (-As - sq) / (2.0 * Qi)
            r2 = (-As + sq) / (2.0 * Qi)
            if r1 > r2:
                r1, r2 = r2, r1
            new = []
            for aa, bb in inters:
                na = aa if aa > r1 else r1
                nb = bb if bb < r2 else r2
                if na <= nb + 1e-12:
                    new.append((na, nb))
            inters = new
            if not inters:
                return inters
        else:
            disc = As * As - 4.0 * Qi * (cs - L)
            if disc <= 1e-12:
                continue
            sq = math.sqrt(disc)
            r1 = (-As - sq) / (2.0 * Qi)

            if r1 > r2:

                    na = aa if aa > r2 else r2
                    if na <= bb + 1e-12:
                        new.append((na, bb))

def pair_solve(h, c, p, q, n, A):
    hp = h[p]
    hq = h[q]
    s = hp + hq

    lo = -hp
    t = hq - 1.0
    if t > lo:
        lo = t
    hi = 1.0 - hp
    if hq < hi:
        hi = hq
    if hi <= lo + 1e-13:
        return None

    L = 2 * n - 1
    A.fill(0.0)
    hfirst = 1.0 - h[::-1]

    A[p:p + n] += hfirst
    sp = n - 1 - p
    A[sp:sp + n] -= h

    A[q:q + n] -= hfirst
    sq = n - 1 - q
    A[sq:sq + n] += h

    r0 = n - 1
    r1 = p + n - 1 - q
    r2 = q + n - 1 - p

    not_sp = np.ones(L, dtype=bool)
    not_sp[r0] = False
    not_sp[r1] = False
    not_sp[r2] = False

    tolA = 1e-13
    pos = (A > tolA) & not_sp
    neg = (A < -tolA) & not_sp
    zero = (~pos) & (~neg) & not_sp

    if np.any(zero):
        c0max = float(c[zero].max())
    else:
        c0max = 0.0

    c_pos = c[pos]
    A_pos = A[pos]
    c_neg = c[neg]
    A_neg = A[neg]

    maxc = float(np.max(c))
    if maxc < 0.0:
        maxc = 0.0

    lb = c0max
    if c_pos.size:
        v = float(np.max(c_pos + A_pos * lo))
        if v > lb:
            lb = v
    if c_neg.size:
        v = float(np.max(c_neg + A_neg * hi))
        if v > lb:
            lb = v

    for idx, Q in ((r0, -2.0), (r1, 1.0), (r2, 1.0)):
        As = A[idx]
        cs = c[idx]
        f0 = cs + As * lo + Q * lo * lo
        f1 = cs + As * hi + Q * hi * hi
        v = f0 if f0 < f1 else f1
        if Q > 0.0:
            vv = -As / (2.0 * Q)
            if lo <= vv <= hi:
                fv = cs + As * vv + Q * vv * vv
                if fv < v:
                    v = fv
        if v > lb:
            lb = v

    if lb > maxc:
        lb = maxc
    if lb < 0.0:
        lb = 0.0

    for _ in range(32):

            Lhi = L
        else:
            Llo = L

    L = Lhi
    inters = pair_intervals(L, lo, hi, c0max, c_pos, A_pos, c_neg, A_neg, sA, sQ, sC)
    if not inters:
        L += 1e-10
        inters = pair_intervals(L, lo, hi, c0max, c_pos, A_pos, c_neg, A_neg, sA, sQ, sC)
    if not inters:
        return None

    best_d = 0.0
    bd = 1e30
    for aa, bb in inters:
        if aa <= 1e-14 and bb >= -1e-14:
            return None
        ca = aa if abs(aa) < abs(bb) else bb
        if abs(ca) < bd:
            bd = abs(ca)
            best_d = ca

    if abs(best_d) < 1e-12:
        return None
    return float(best_d), r0, r1, r2


def make_pair_move(h, c, A, p, q, n):
    res = pair_solve(h, c, p, q, n, A)
    if res is None:
        return None
    delta, r0, r1, r2 = res

    hp = h[p]
    hq = h[q]
    s = hp + hq

    newp = hp + delta
    if newp < 0.0:
        newp = 0.0
    elif newp > 1.0:
        newp = 1.0
    newq = s - newp
    if newq < 0.0 or newq > 1.0:
        newq = min(max(newq, 0.0), 1.0)
        newp = min(max(s - newq, 0.0), 1.0)

    delta = newp - hp
    if abs(delta) < 1e-13:
        return None

    h[p] = newp
    h[q] = newq
    d2 = delta * delta
    c += A * delta
    c[r0] -= 2.0 * d2
    c[r1] += d2
    c[r2] += d2
    return hp, hq, delta, r0, r1, r2


def pair_cd(h, n, deadline, max_passes=20, mode="adj", rng=None):
    if mode != "adj" and rng is None:
        mode = "adj"

    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode="full").astype(float)
    C = c_from_corr(c, n)
    best_h = h.copy()
    best_c = C
    A = np.zeros(2 * n - 1, dtype=float)
    inv_dx = 2.0 / n
    stall = 0

    for _ in range(max_passes):
        if time.time() > deadline:
            break

        if mode == "rand":
            pairs = []
            for i in range(n):
                q = int(rng.randint(n))
                if q == i:
                    q = (i + 1) % n
                pairs.append((i, q))
        elif mode == "mix":
            pairs = []
            for i in range(n):
                if rng.rand() < 0.5:
                    q = int(rng.randint(n))
                    if q == i:
                        q = (i + 1) % n
                else:
                    q = (i + 1) % n
                pairs.append((i, q))
        else:
            pairs = [(i, (i + 1) % n) for i in range(n)]

        improved = 0
        for idx, (p, q) in enumerate(pairs):
            if (idx & 127) == 0 and time.time() > deadline:
                break
            mv = make_pair_move(h, c, A, p, q, n)
            if mv is not None:
                oldp, oldq, delta, r0, r1, r2 = mv
                newC = float(np.max(c)) * inv_dx
                if newC < C - 1e-15:
                    C = newC
                    improved += 1
                else:
                    h[p] = oldp
                    h[q] = oldq
                    d2 = delta * delta
                    c -= A * delta
                    c[r0] += 2.0 * d2
                    c[r1] -= d2
                    c[r2] -= d2

        if improved == 0:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0

        c = np.correlate(h, 1.0 - h, mode="full").astype(float)
        C = c_from_corr(c, n)
        if C < best_c - 1e-14:
            best_c = C
            best_h = h.copy()
        elif C > best_c + 1e-12:
            h = best_h.copy()
            c = np.correlate(h, 1.0 - h, mode="full").astype(float)
            C = best_c

    return best_h, best_c


def optimize_blocks(h, n, deadline, blocks, rng):
    C = compute_c5(h, n)
    for mode, passes in blocks:
        if time.time() >= deadline:
            break
        h, C = pair_cd(h, n, deadline, passes, mode, rng)
    return h, C


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s)) - 5.0
    rng = np.random.RandomState(seed)

    try:
        h_best = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if h_best.size < 2:
            raise ValueError
    except Exception:
        h_best = default_h(400)

    n_best = len(h_best)
    h_best = project_feasible(h_best, n_best)
    best_c = compute_c5(h_best, n_best)

    main_deadline = min(deadline, t0 + 350.0)
    h_best, best_c = optimize_blocks(
        h_best, n_best, main_deadline,
        [("adj", 25), ("rand", 35), ("mix", 25)],
        rng
    )

    candidates = [600, 500, 800, 400, 300, 250, 200, 150, 120, 100, 900, 180]
    for idx, nt in enumerate(candidates):
        if time.time() > deadline - 150.0:
            break
        if nt < 4 or nt == n_best:
            continue
        rem = deadline - time.time() - 150.0
        if rem <= 0.0:
            break
        alloc = min(80.0, rem / max(1, len(candidates) - idx))
        if alloc < 25.0:
            break

        h_try = resample(h_best, n_best, nt)
        passes = 25 if nt < 450 else 15
        blocks = [("adj", passes), ("rand", passes), ("mix", passes)]
        ht, ct = optimize_blocks(h_try, nt, time.time() + alloc, blocks, rng)
        if ct < best_c:
            h_best, best_c = ht, ct
            n_best = nt

    n_best = len(h_best)
    scales = [0.01, 0.02, 0.04, 0.08, 0.015, 0.05, 0.10]
    for s in scales:
        if time.time() > deadline - 100.0:
            break
        rem = deadline - time.time() - 100.0
        if rem <= 0.0:
            break
        alloc = min(70.0, rem / 2.0)
        if alloc < 25.0:
            break

        h_try = project_feasible(h_best + rng.normal(0.0, s, n_best), n_best)
        ht, ct = optimize_blocks(
            h_try, n_best, time.time() + alloc,
            [("rand", 20), ("mix", 20)],
            rng
        )
        if ct < best_c:
            h_best, best_c = ht, ct

    n_best = len(h_best)
    if time.time() < deadline - 80.0:
        h_try = project_feasible(0.5 * (h_best + h_best[::-1]), n_best)
        alloc = min(60.0, max(0.0, deadline - time.time() - 40.0))
        if alloc > 25.0:
            ht, ct = optimize_blocks(
                h_try, n_best, time.time() + alloc,
                [("adj", 15), ("rand", 15)],
                rng
            )
            if ct < best_c:
                h_best, best_c = ht, ct

    n_best = len(h_best)
    if time.time() < deadline - 5.0:
        h_best, best_c = optimize_blocks(
            h_best, n_best, deadline,
            [("adj", 30), ("rand", 30)],
            rng
        )

    n_best = len(h_best)
    h_best = project_feasible(h_best, n_best)
    best_c = compute_c5(h_best, n_best)
    return h_best, best_c, n_best
