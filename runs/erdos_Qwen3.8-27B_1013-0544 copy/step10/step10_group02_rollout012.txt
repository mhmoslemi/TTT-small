def project_box_mean(v, n):
    v = np.asarray(v, dtype=float)
    target = 0.5 * float(n)
    if v.min() >= 0.0 and v.max() <= 1.0 and abs(v.sum() - target) < 1e-12:
        return make_exact(np.clip(v, 0.0, 1.0), n)
    lo = float(np.min(v)) - 1.0

    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v + mid, 0.0, 1.0).sum())
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    return make_exact(h, n)

def make_exact(h, n):
    target = 0.5 * float(n)
    for _ in range(30):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt == 0:
            h = np.clip(h + diff / float(n), 0.0, 1.0)
        else:
            h = h.copy()
            h[mask] += diff / float(cnt)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c(h, n):

    return e / s

                    continue
                dec = [i]
                if mi != i and mi != j:
                    dec.append(mi)
                inc = [j]
                if mj != j and mj != i:
                    inc.append(mj)
                if any(x in inc for x in dec):
                    stale += 1
                    if stale > 8000:
                        break
                    continue
            else:
                dec = [i]
                inc = [j]
            low = -1.0
            high = 1.0
            for idx in dec:
                o = float(h[idx])
                if o - 1.0 > low:
                    low = o - 1.0
                if o < high:
                    high = o
            for idx in inc:
                o = float(h[idx])
                if -o > low:
                    low = -o
                if 1.0 - o < high:
                    high = 1.0 - o
            if high <= low + 1e-14:
                stale += 1
                if stale > 8000:
                    break
                continue
            d0 = float(rng.uniform(-step, step))
            if d0 < low:
                d = low
            elif d0 > high:
                d = high
            else:
                d = d0
            if abs(d) < 1e-9:
                stale += 1
                if stale > 8000:
                    break
                continue
            updates = []
            ok = True
            for idx in dec:
                o = float(h[idx])
                nw = o - d
                if nw < -1e-12 or nw > 1.0 + 1e-12:
                    ok = False
                    break
                updates.append((idx, o, nw))
            if ok:
                for idx in inc:
                    o = float(h[idx])
                    nw = o + d
                    if nw < -1e-12 or nw > 1.0 + 1e-12:
                        ok = False
                        break
                    updates.append((idx, o, nw))
            if not ok or len(updates) == 0:
                stale += 1
                if stale > 8000:
                    break
                continue
            old_cur = cur
            for idx, o, nw in updates:
                update_corr(c, h, idx, o, nw, n)
                h[idx] = nw
            new_cur = float(c.max())
            if not np.isfinite(new_cur):
                for idx, o, nw in reversed(updates):
                    update_corr(c, h, idx, nw, o, n)
                    h[idx] = o
                c = np.correlate(h, 1.0 - h, mode='full')
                cur = float(c.max())
                stale += 1
                if stale > 8000:
                    break
                continue
            oldC = old_cur * dx
            newC = new_cur * dx
            accept = False
            if newC < oldC - 1e-15:
                accept = True
            else:
                deltaC = newC - oldC
                if deltaC > 0.0 and deltaC < T and float(rng.rand()) < float(np.exp(-deltaC / T)):
                    accept = True
            if accept:
                cur = new_cur
                stale = 0
                if newC < best_c:
                    best_c = newC
                    best_h = h.copy()
            else:
                for idx, o, nw in reversed(updates):
                    update_corr(c, h, idx, nw, o, n)
                    h[idx] = o
                stale += 1
                if stale > 8000:
                    break
    return best_h, best_c

def make_cos_basis(n, M):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    B = np.empty((n, M), dtype=float)
    for m in range(1, M + 1):
        B[:, m - 1] = np.cos(np.pi * float(m) * (x - 1.0))
    return B

def fourier_h(a, B, n):
    a = np.asarray(a, dtype=float)
    s = float(np.clip(a[0], 0.0, 1.0))
    f = a[1:] @ B
    am = float(np.max(np.abs(f))) if f.size else 0.0
    if not np.isfinite(am) or am < 1e-12:
        f = np.zeros(n, dtype=float)
    else:
        f = 0.99 * f / am
    return project_box_mean(0.5 + s * f, n)

def fourier_obj(a, B, n):
    try:
        if not np.all(np.isfinite(a)):
            return 1e100
        return compute_c(fourier_h(a, B, n), n)
    except Exception:
        return 1e100

def get_fourier_coeff_from_basis(h, B, n, M):
    y = h - 0.5
    G = B.T @ B
    G += 1e-8 * np.eye(M)
    coef = np.linalg.solve(G, B.T @ y)
    f = B @ coef
    fmax = float(np.max(np.abs(f))) if f.size else 0.0
    ymax = float(np.max(np.abs(y))) if y.size else 0.0
    s = 0.0 if fmax < 1e-12 else min(1.0, ymax / fmax)
    coef = np.clip(coef, -5.0, 5.0)
    return np.concatenate(([s], coef))

def fourier_search(h, n, M, limit, rng):
    h = project_box_mean(h, n)
    B = make_cos_basis(n, M)
    starts = [get_fourier_coeff_from_basis(h, B, n, M)]
    for _ in range(3):
        a = np.zeros(M + 1, dtype=float)
        a[0] = float(rng.uniform(0.2, 1.0))
        a[1:] = 0.45 * rng.standard_normal(M)
        starts.append(a)
    best_h = h.copy()
    best_c = compute_c(h, n)
    for a0 in starts:
        if time.time() > limit - 5:
            break
        try:
            res = minimize(fourier_obj, a0, args=(B, n), method='Nelder-Mead',
                           options={'maxiter': 550, 'xatol': 1e-6, 'fatol': 1e-10})
            ax = res.x if res.x is not None else a0
            hx = fourier_h(ax, B, n)
            cx = compute_c(hx, n)
            if cx < best_c - 1e-12:
                best_c = cx
                best_h = hx
        except Exception:
            pass
    return best_h, best_c

def sum_obj(x):
    return float(np.sum(x) - 0.5 * float(len(x)))

def sum_jac(x):
    return np.ones(len(x), dtype=float)

def soft_lse_value(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    a = alpha * (c - m)
    ma = float(a.max())
    s = float(np.exp(a - ma).sum())
    if s <= 0.0 or not np.isfinite(s):
        return m * (2.0 / float(n))
    return (m + (ma + float(np.log(s))) / float(alpha)) * (2.0 / float(n))

def soft_lse_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    p = softmax_weights(c, alpha)
    return grad_weights(x, p, n)

def slsqp_polish(h, n, alphas, limit, maxiter=18):
    h = project_box_mean(np.asarray(h, dtype=float).copy(), n)
    best_h = h.copy()
    best_c = compute_c(h, n)
    if time.time() >= limit:
        return best_h, best_c
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_obj, 'jac': sum_jac}]
    for alpha in alphas:
        if time.time() > limit:
            break
        try:
            res = minimize(soft_lse_value, best_h, args=(n, alpha), method='SLSQP',
                           jac=soft_lse_grad, bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            nh = project_box_mean(res.x, n)
            nc = compute_c(nh, n)
            if nc < best_c - 1e-13:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def is_symmetric(h, tol=1e-6):
    h = np.asarray(h, dtype=float)
    return bool(np.allclose(h, h[::-1], rtol=0.0, atol=tol))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
        if n0 <= 0:
            raise ValueError
    except Exception:
        n0 = 350
        h0 = np.full(n0, 0.5, dtype=float)
    h0 = project_box_mean(h0, n0)
    best_h = h0.copy()
    best_n = n0
    best_c = compute_c(best_h, best_n)

    scan_h = best_h.copy()
    scan_n = best_n
    scan_ns = set(range(max(200, scan_n - 40), min(950, scan_n + 40) + 1))
    for m in (300, 350, 400, 450, 500, 550, 600, 650, 700, 750, 800, 850, 900):
        scan_ns.add(m)
    for m in sorted(scan_ns):
        try:
            ht = interp_h(scan_h, scan_n, m)
            ct = compute_c(ht, m)
            if ct < best_c - 1e-12:
                best_h, best_n, best_c = ht, m, ct
        except Exception:
            pass

    d1 = min(deadline - 90.0, t0 + max(25.0, 0.22 * float(budget_s)))
    if time.time() < d1:
        hh, cc = gradient_polish(best_h, best_n, d1, sym=False, rng=rng)
        if cc < best_c - 1e-13:
            best_h, best_n, best_c = hh, len(hh), cc

    d2 = min(deadline - 80.0, t0 + max(35.0, 0.42 * float(budget_s)))
    if time.time() < d2:
        hs = sym_project(best_h, best_n)
        hh, cc = gradient_polish(hs, best_n, d2, sym=True, rng=rng)
        if cc < best_c - 1e-13:
            best_h, best_n, best_c = hh, len(hh), cc

    d3 = min(deadline - 70.0, t0 + max(40.0, 0.62 * float(budget_s)))
    if deadline - time.time() > 150.0:
        n_four = best_n if best_n <= 500 else 450
        h_four = interp_h(best_h, best_n, n_four) if best_n != n_four else best_h.copy()
        try:
            hf, cf = fourier_search(h_four, n_four, 9, d3, rng)
            if cf < best_c - 1e-12:
                best_h, best_n, best_c = hf, n_four, cf
            if time.time() < d3 - 25.0:
                lim = min(d3 - 15.0, deadline - 55.0)
                hh, cc = gradient_polish(best_h, best_n, lim,
                                         sym=is_symmetric(best_h, 1e-5), rng=rng)
                if cc < best_c - 1e-13:
                    best_h, best_n, best_c = hh, len(hh), cc
        except Exception:
            pass

    d4 = min(deadline - 45.0, t0 + max(50.0, 0.85 * float(budget_s)))
    if time.time() < d4 - 30.0:
        hh, cc = pair_polish(best_h, best_n, d4 - 30.0,
                             sym=is_symmetric(best_h, 1e-6), rng=rng,
                             steps=(0.05, 0.02, 0.01, 0.005, 0.002))
        if cc < best_c - 1e-13:
            best_h, best_n, best_c = hh, len(hh), cc
    if time.time() < d4 - 5.0:
        hh, cc = pair_polish(sym_project(best_h, best_n), best_n, d4,
                             sym=True, rng=rng,
                             steps=(0.03, 0.015, 0.008, 0.004))
        if cc < best_c - 1e-13:
            best_h, best_n, best_c = hh, len(hh), cc

    if deadline - time.time() > 170.0:
        n_small = int(min(180, max(140, best_n // 4)))
        try:
            hs = interp_h(best_h, best_n, n_small)
            hh, cc = slsqp_polish(hs, n_small, (1200.0, 6000.0), deadline - 90.0, maxiter=14)
            if cc < best_c - 1e-12:
                best_h, best_n, best_c = hh, n_small, cc
        except Exception:
            pass

    if time.time() < deadline - 35.0:
        hh, cc = gradient_polish(best_h, best_n, deadline - 25.0,
                                 sym=is_symmetric(best_h, 1e-6), rng=rng)
        if cc < best_c - 1e-13:
            best_h, best_n, best_c = hh, len(hh), cc

    if time.time() < deadline - 3.0:
        hh, cc = pair_polish(best_h, best_n, deadline - 2.0,
                             sym=is_symmetric(best_h, 1e-6), rng=rng,
                             steps=(0.008, 0.004, 0.002, 0.001))
        if cc < best_c - 1e-13:
            best_h, best_n, best_c = hh, len(hh), cc

    best_h = project_box_mean(best_h, best_n)
    best_c = compute_c(best_h, best_n)
    return best_h, float(best_c), int(best_n)
