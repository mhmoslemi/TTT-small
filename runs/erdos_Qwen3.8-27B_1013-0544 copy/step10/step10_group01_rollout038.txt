    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:

            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

            if res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1; d1 = d2; f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h[h < 1e-7] = 0.0
            h[h > 1.0 - 1e-7] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    return h, min(best_c, compute_c5(h, n))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(600, 0.5)
        n = 600

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: CD polish
    t_end1 = t0 + 100
    h1, c1 = coord_descent(best_h, n, 30, t_end1, gs_iters=55)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: SLSQP
    t_end2 = t0 + 280
    if time.time() < t_end2:
        h2, c2 = slsqp_optimize(best_h, n, [2000, 10000, 50000, 200000], 350)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Multi-n exploration
    n_base = len(best_h)
    x_base = np.linspace(0, 2, n_base)
    t_end3 = t0 + 550
    for n_try in [400, 500, 550, 600, 650, 700, 750, 800, 900, 1000, 1200]:
        if time.time() > t_end3:
            break
        x_s = np.linspace(0, 2, n_try)
        h_s = project_feasible(np.interp(x_s, x_base, best_h), n_try)
        h_s, c_s = coord_descent(h_s, n_try, 25, min(time.time() + 25, t_end3), gs_iters=45)
        if c_s < best_c:
            best_h = h_s
            best_c = c_s
            n_base = n_try
            x_base = np.linspace(0, 2, n_base)

    # Phase 4: Structured perturbations + CD
    t_end4 = t0 + 820
    n_cur = len(best_h)
    x = np.linspace(0, 2, n_cur)
    perturbations = []
    # Fourier modes
    for m in range(1, 15):
        perturbations.append(0.03 * np.sin(np.pi * m * x))
        perturbations.append(0.03 * np.cos(np.pi * m * x))
    # Block flips
    for blk in [10, 20, 40, n_cur // 7, n_cur // 4]:
        start = rng.randint(0, max(1, n_cur - blk))
        mask = np.zeros(n_cur)
        mask[start:start + blk] = 1.0
        perturbations.append(0.4 * mask - 0.2)
    # Random perturbations
    for scale in [0.01, 0.02, 0.04, 0.06]:
        perturbations.append(rng.normal(0, scale, n_cur))
    # Binary-like
    for _ in range(3):
        b = (rng.rand(n_cur) < 0.5).astype(float) * 0.3 - 0.15
        perturbations.append(b)

    time_per_pert = max(5, (t_end4 - time.time()) / max(1, len(perturbations) + 2))
    for pert in perturbations:
        if time.time() > t_end4:
            break
        hp = project_feasible(best_h + pert, n_cur)
        h_p, c_p = coord_descent(hp, n_cur, 20, time.time() + time_per_pert, gs_iters=45)
        if c_p < best_c:
            best_h, best_c = h_p, c_p
            n_cur = len(best_h)
            x = np.linspace(0, 2, n_cur)

    # Phase 5: SLSQP again with high alpha
    t_end5 = t0 + 950
    if time.time() < t_end5:
        n_cur = len(best_h)
        h5, c5 = slsqp_optimize(best_h, n_cur, [100000, 500000, 2000000], 500)
        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 6: Final tight CD polish
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        h6, c6 = coord_descent(best_h, n_cur, 40, deadline - 5, gs_iters=70)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Final verification
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
