Looking at the previous code, I identified a key bug in the correlation update during coordinate descent: the diagonal term `delta*(1-2*h0)` is missing from the `c` vector update. This causes `c` to drift from its true value during a pass, leading to suboptimal convergence. I'll fix this and add Nesterov-accelerated subgradient for global search, plus a multi-scale approach.

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_full(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def update_c_diag(c, h, p, n, delta, h0):
    """Update correlation vector after changing h[p] by delta from h0."""
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_pos = np.where(ms == (n - 1))[0]
    if len(diag_pos) > 0:
        a[diag_pos] = 0.0
    c[ms] += delta * a
    if len(diag_pos) > 0:
        c[n - 1] += delta * (1.0 - 2.0 * h0) - delta * delta

def solve_coord(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-12 or h0 > 1.0 - 1e-12:
        return h0
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    c_diag = c_base[diag_idx[0]] if has_diag else 0.0
    a[diag_idx] = 0.0 if has_diag else 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    L_lo, L_hi = unaff, cur_max
    if L_hi - L_lo < 1e-14:
        return h0
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    for _ in range(45):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag
            b_coeff = 1.0 - 2.0 * h0
            disc = b_coeff * b_coeff - 4.0 * Cq
            if disc >= 0:
                r1 = (b_coeff - np.sqrt(disc)) / 2.0
                r2 = (b_coeff + np.sqrt(disc)) / 2.0
                x1_lo = max(x_lo, 0.0)
                x1_hi = min(x_hi, h0 + r1)
                x2_lo = max(x_lo, h0 + r2)
                x2_hi = min(x_hi, 1.0)
                if not ((x1_lo <= x1_hi) or (x2_lo <= x2_hi)):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    if has_diag:
        Cq = L - c_diag
        b_coeff = 1.0 - 2.0 * h0
        disc = b_coeff * b_coeff - 4.0 * Cq
        if disc >= 0:
            r1 = (b_coeff - np.sqrt(disc)) / 2.0
            r2 = (b_coeff + np.sqrt(disc)) / 2.0
            if abs(r1) <= abs(r2):
                x_hi = min(x_hi, h0 + r1)
            else:
                x_lo = max(x_lo, h0 + r2)
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-13:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, order=None):
    h = h.copy()
    c = corr_full(h, n)
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        idxs = list(range(n)) if order is None else order
        for p in idxs:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            delta = xn - h0
            if abs(delta) > 1e-12:
                update_c_diag(c, h, p, n, delta, h0)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = corr_full(h, n)
    return h, compute_c5(h, n)

def subgradient_step(h, n, tol_frac=0.02):
    c = corr_full(h, n)
    cmax = c.max()
    active = c > cmax - tol_frac * cmax
    grad = np.zeros(n)
    for m in np.where(active)[0]:
        k = m - (n - 1)
        for p in range(n):
            jf = p + k
            jb = p - k
            g = 0.0
            if 0 <= jf < n:
                g += (1.0 - h[jf])
            if 0 <= jb < n:
                g -= h[jb]
            if k == 0:
                g = 1.0 - 2.0 * h[p]
            grad[p] += g
    n_active = max(1, np.sum(active))
    grad /= n_active
    return grad

def nesterov_subgrad(h, n, max_iter, deadline, scale=0.003):
    x = h.copy()
    y = h.copy()
    xv = h.copy()
    t = 1.0
    best = x.copy()
    best_c = compute_c5(x, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = subgradient_step(y, n, tol_frac=0.02)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = x + beta * (x - xv)
        step = scale * t / (t + 50.0)
        x_new = np.clip(y - step * g, 0.0, 1.0)
        x_new = project_feasible(x_new, n)
        xv = x
        x = x_new
        t = t_new
        c_val = compute_c5(x, n)
        if c_val < best_c:
            best_c = c_val
            best = x.copy()
    return best, best_c
