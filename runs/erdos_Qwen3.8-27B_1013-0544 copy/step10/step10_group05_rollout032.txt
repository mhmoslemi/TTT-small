    # Build LP
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0  # minimize t

    b_ub = []

            feas = False
        if feas and np.any(neg_mask):
            x_lo = max(x_lo, float(((L - c_base[neg_mask]) / a[neg_mask]).max()))
        if feas and np.any(pos_mask):
            x_hi = min(x_hi, float(((L - c_base[pos_mask]) / a[pos_mask]).min()))
        if x_lo > x_hi + 1e-12:
            feas = False
        if feas and has_diag:
            cd = c_base[diag_idx[0]]
            ad = a[diag_idx[0]]
            disc = ad * ad + 4.0 * (cd - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1, r2 = (ad - sq) / 2.0, (ad + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feas = False
        if feas:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(((L - c_base[neg_mask]) / a[neg_mask]).max()))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(((L - c_base[pos_mask]) / a[pos_mask]).min()))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if has_diag:
        cd, ad = c_base[diag_idx[0]], a[diag_idx[0]]
        disc = ad * ad + 4.0 * (cd - L)
        if disc >= 0:
            sq = np.sqrt(disc)
            r1, r2 = (ad - sq) / 2.0, (ad + sq) / 2.0
            if x_lo <= h0 + r1:
                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_pass(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        imp = False
        for p in range(n):
            if h[p] < 1e-13 or h[p] > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h[p]) > 1e-13:
                d = xn - h[p]
                m_min, m_max = max(0, n - 1 - p), min(2 * n - 2, 2 * n - 2 - p)
                ms = np.arange(m_min, m_max + 1)
                jf, jb = p + ms - (n - 1), p - ms + (n - 1)
                vf, vb = (jf >= 0) & (jf < n), (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += 1.0 - h[jf[vf]]
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += d * a
                di = np.where(ms == n - 1)[0]
                if len(di) > 0:
                    c[ms[di[0]]] -= d * d
                h[p] = xn
                imp = True
        if not imp:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def lp_refine(h, n, c, K, deadline):
    h = h.copy()
    c = c.copy()
    best = float(c.max())
    for it in range(8000):
        if time.time() > deadline:
            break
        top = np.argpartition(c, -K)[-K:]
        active = sorted(top.tolist())
        c_obj = np.zeros(n + 1)
        c_obj[0] = 1.0
        A = np.zeros((len(active), n + 1))
        b = np.zeros(len(active))
        for i, m in enumerate(active):
            g = lag_grad(h, n, m)
            A[i, 0] = -1.0
            A[i, 1:] = g
            b[i] = -(c[m] - g @ h)
        Aeq = np.zeros((1, n + 1))
        Aeq[0, 1:] = 1.0
        bounds = [(None, None)] + [(0.0, 1.0)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=[n / 2.0],
                      bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            break
        h_lp = res.x[1:]
        d = h_lp - h
        dmax = np.abs(d).max()
        if dmax > 0.5:
            d *= 0.5 / dmax
        imp = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15]:
            ht = project_feasible(h + alpha * d, n)
            ct = np.correlate(ht, 1.0 - ht, mode='full')
            if ct.max() < best - 1e-14:
                h, c, best = ht, ct, float(ct.max())
                imp = True
                break
        if not imp:
            K = min(K + 10, 2 * n - 1)
            if K > 80:
                break
    return h, c, best

def interp_h(h_src, n_src, n_tgt):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(xt, xs, h_src), n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n_best = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n_best = 700
    h_best = project_feasible(h_best, n_best)
    c_best = compute_c5(h_best, n_best)

    # Phase 1: CD polish from initial
    h1, c1 = cd_pass(h_best, n_best, 300, deadline)
    if c1 < c_best:
        h_best, c_best = h1, c1

    # Phase 2: LP-guided refinement with increasing K
    n = len(h_best)
    c = np.correlate(h_best, 1.0 - h_best, mode='full')
    for K in [5, 10, 20, 30, 50]:
        if time.time() > deadline - 120:
            break
        h2, c2, best2 = lp_refine(h_best, n, c, K, min(deadline, time.time() + 120))
        if best2 < c_best:
            h_best, c_best, c = h2, best2, c2
        else:
            c = np.correlate(h_best, 1.0 - h_best, mode='full')

    # Phase 3: CD polish after LP
    h3, c3 = cd_pass(h_best, n, 300, deadline)
    if c3 < c_best:
        h_best, c_best = h3, c3

    # Phase 4: More LP with perturbation restarts
    if time.time() < deadline - 60:
        for restart in range(5):
            if time.time() > deadline - 30:
                break
            temp = 0.005 + 0.01 * restart
            hp = project_feasible(h_best + rng.normal(0, temp, n), n)
            c = np.correlate(hp, 1.0 - hp, mode='full')
            h4, c4, best4 = lp_refine(hp, n, c, 15, min(deadline, time.time() + 80))
            if best4 < c_best:
                h_best, c_best = h4, best4
                c = c4
            h4b, c4b = cd_pass(h_best, n, 100, deadline)
            if c4b < c_best:
                h_best, c_best = h4b, c4b

    # Phase 5: Final CD
    h5, c5 = cd_pass(h_best, n, 200, deadline)
    if c5 < c_best:
        h_best, c_best = h5, c5

    # Phase 6: Try nearby n values
    for n_try in [n - 15, n + 15, n - 30, n + 30]:
        if time.time() > deadline - 40:
            break
        if n_try < 200 or n_try > 1500 or n_try == n:
            continue
        hi = interp_h(h_best, len(h_best), n_try)
        hi2, ci2 = cd_pass(hi, n_try, 150, deadline)
        if ci2 < c_best - 1e-7:
            h_best, c_best = hi2, ci2

    h_best = project_feasible(h_best, len(h_best))
    c_best = compute_c5(h_best, len(h_best))
    return h_best, c_best, len(h_best)
