    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):

def update_corr(c, h, p, delta, n):
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c[ms] += delta * a
    diag_i = np.where(ms == (n - 1))[0]
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta

def cd_pass(h, n, c, deadline):
    improved = False
    for p in range(n):
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-13 or h0 > 1.0 - 1e-13:

        xn = solve_1d(c, h, p, n)
        if abs(xn - h0) > 1e-13:
            delta = xn - h0
            update_corr(c, h, p, delta, n)
            h[p] = xn
            improved = True
    return improved
