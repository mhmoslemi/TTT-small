def subgrad_descent(h, n, steps, deadline, M=1):
    """Subgradient descent focusing on top M lags."""
    h = h.copy()
    dx = 2.0 / n
    # Adaptive step size
    grad_sq = np.zeros(n)
    base_lr = 0.1
    
    for s in range(steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        # Get top M indices
        top_idx = np.argsort(c)[::-1][:M]
        top_vals = c[top_idx]
        
        if M == 1:
            w_top = np.ones(1)
        else:
            mx = top_vals.max()
            w_top = np.exp(top_vals - mx)
            w_top /= w_top.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        for m, k in enumerate(top_idx):
            d = k - (n - 1)
            wm = w_top[m]
            lo1 = max(0, d)
            hi1 = min(n, n + d)
            if lo1 < hi1:
                grad[lo1:hi1] += wm * (1.0 - h[lo1 - d:hi1 - d])
            lo2 = max(0, -d)
            hi2 = min(n, n - d)
            if lo2 < hi2:
                grad[lo2:hi2] -= wm * h[lo2 + d:hi2 + d]
        grad *= dx
        grad -= np.mean(grad)  # Center to maintain sum
        
        # Adaptive step size (AdaGrad-like)
        grad_sq += grad ** 2
        lr = base_lr / np.sqrt(grad_sq + 1e-8)
        
        h_new = project_feasible(h - lr * grad, n)
        # Accept if it improves
        new_c = compute_c5(h_new, n)
        old_c = compute_c5(h, n)
        if new_c <= old_c + 1e-10:
            h = h_new
        else:
            # Reduce step size
            base_lr *= 0.9
    
    return h

            if has_diag:
                Cq = T - c_diag + h0 - h0 * h0
                disc = 1.0 - 4.0 * Cq
                if disc >= 0:
                    r1 = (1.0 - np.sqrt(disc)) / 2.0
                    r2 = (1.0 + np.sqrt(disc)) / 2.0
                    xl2 = max(x_lo, 0.0)
                    xh2 = min(x_hi, 1.0)
                    if xl2 < r1 and xh2 > r2:
                        if h0 - r1 <= r2 - h0:
                            x_hi = min(x_hi, r1)
                        else:
                            x_lo = max(x_lo, r2)
                    elif xl2 < r1:
                        x_hi = min(x_hi, r1)
                    elif xh2 > r2:
                        x_lo = max(x_lo, r2)
            x_lo = max(x_lo, 0.0)
            x_hi = min(x_hi, 1.0)
            if x_lo > x_hi + 1e-10:
                continue
            x_new = float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))
            if abs(x_new - h0) > 1e-12:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def fourier_construct(coeffs, n):
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for j in range(len(coeffs) // 2):
        a_j = coeffs[2 * j]
        b_j = coeffs[2 * j + 1]
        h += a_j * np.cos((j + 1) * np.pi * x / 2.0) + b_j * np.sin((j + 1) * np.pi * x / 2.0)
    return project_feasible(np.clip(h, 0, 1), n)

def fourier_obj(coeffs, n):
    h = fourier_construct(coeffs, n)
    return compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except:
        n = 700
        x = np.linspace(0, 2, n)
        h_init = 0.5 + 0.15 * np.sin(np.pi * x / 2.0) + 0.08 * np.cos(np.pi * x)
        h_init = np.clip(h_init, 0, 1)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP moderate
    if time.time() < deadline - 200:
        h1 = slsqp_run(best_h, n, 500, 250, deadline)
        c1 = compute_c5(h1, n)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Subgradient hard focus M=1
    if time.time() < deadline - 200:
        h2 = subgrad_opt(best_h, n, 800, deadline, M=1, lr0=0.15, focus_beta=1.0)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: CD
    if time.time() < deadline - 100:
        rem = deadline - time.time()
        h3, c3 = coord_descent(best_h, n, max(5, int(rem * 0.3 / 1.2)), deadline, rng, 38)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: SLSQP high alpha
    if time.time() < deadline - 120:
        h4 = slsqp_run(best_h, n, 5000, 300, deadline)
        c4 = compute_c5(h4, n)
        if c4 < best_c:
            best_h, best_c = h4, c4
        h4b, c4b = coord_descent(best_h, n, 15, deadline, rng, 40)
        if c4b < best_c:
            best_h, best_c = h4b, c4b
    # Phase 5: Subgrad M=5 smooth
    if time.time() < deadline - 80:
        h5 = subgrad_opt(best_h, n, 600, deadline, M=5, lr0=0.1, focus_beta=5.0)
        c5 = compute_c5(h5, n)
        if c5 < best_c:
            best_h, best_c = h5, c5
            h5b, c5b = coord_descent(h5, len(h5), 15, deadline, rng, 40)
            if c5b < best_c:
                best_h, best_c = h5b, c5b
    # Phase 6: Try n//2
    if time.time() < deadline - 100:
        n_s = len(best_h) // 2
        x_o = np.linspace(0, 2, len(best_h))
        x_n = np.linspace(0, 2, n_s)
        h_s = project_feasible(np.interp(x_n, x_o, best_h), n_s)
        h_s = slsqp_run(h_s, n_s, 2000, 200, deadline)
        h_s, c_s = coord_descent(h_s, n_s, 20, deadline, rng, 38)
        if c_s < best_c:
            best_h, best_c = h_s, c_s
    # Phase 7: Symmetrize
    if time.time() < deadline - 40:
        nc = len(best_h)
        x = np.linspace(0, 2, nc)
        h_rev = np.interp(x, x[::-1], best_h)
        h_sym = project_feasible(0.5 * (best_h + 1.0 - h_rev), nc)
        c_sym = compute_c5(h_sym, nc)
        if c_sym < best_c:
            best_h, best_c = h_sym, c_sym
            h_sym2, c_sym2 = coord_descent(h_sym, nc, 15, deadline, rng, 40)
            if c_sym2 < best_c:
                best_h, best_c = h_sym2, c_sym2
    # Phase 8: Random restart + re-optimize
    if time.time() < deadline - 60:
        nc = len(best_h)
        for trial in range(3):
            if time.time() > deadline - 30:
                break
            noise = rng.randn(nc) * 0.02
            h_r = project_feasible(best_h + noise, nc)
            h_r = subgrad_opt(h_r, nc, 200, deadline, M=3, lr0=0.1, focus_beta=10.0)
            h_r, c_r = coord_descent(h_r, nc, 10, deadline, rng, 35)
            if c_r < best_c:
                best_h, best_c = h_r, c_r
    # Phase 9: Final CD polish
    if time.time() < deadline - 10:
        nc = len(best_h)
        h_f, c_f = coord_descent(best_h, nc, 20, deadline, rng, 42)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
