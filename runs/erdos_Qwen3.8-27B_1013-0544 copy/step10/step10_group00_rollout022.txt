- Autocorrelation (f*f)(k) is as "flat" (uniformly positive) as possible

This is related to the theory of "flat autocorrelation" sequences! In coding theory, sequences with flat autocorrelation are called "perfect autocorrelation" sequences or are related to Barker sequences, Zadoff-Chu sequences, etc.

For the continuous case, a function with perfectly flat autocorrelation would be ideal. This is related to "radial basis functions" or "constant amplitude zero autocorrelation" (CAZAC) sequences.

In the discrete case, a CAZAC sequence of length n would have:
|sum_i s[i] * conj(s[i+k])| = constant for all k != 0

This is exactly what we want (for the f part)! CAZAC sequences include:
- Zadoff-Chu sequences: s[k] = exp(-j*pi*k*(2k+1)/n) for n odd
- Chirp sequences: s[k] = exp(j*pi*k²/n)

But we need a real-valued function in [0, 1], not a complex sequence. However, we can use the real part of a CAZAC sequence as a starting point.

For a chirp: f[i] = sin(2*pi*i²/n) or cos(2*pi*i²/n). The autocorrelation of cos(2*pi*i²/n) is approximately flat (it's the basis for the Zak transform / time-frequency analysis).

import numpy as np
from scipy.optimize import minimize
import time

def _fft_corr_full(h, n):
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.fft(h, N)
    V = np.fft.fft(1.0 - h, N)
    r = np.real(np.fft.ifft(H * np.conj(V)))
    idx = (n - 1 - np.arange(2 * n - 1)) % N
    return r[idx]

def _fft_corr_max(h, n):
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.fft(h, N)
    V = np.fft.fft(1.0 - h, N)
    r = np.real(np.fft.ifft(H * np.conj(V)))
    return float(np.max(r))

def compute_c5(h, n):
    return _fft_corr_max(h, n) * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_max(c, alpha):
    m = c.max()
    if alpha < 1e-4:
        return m
    sh = alpha * (c - m)
    return m + np.log(np.sum(np.exp(sh))) / alpha

def anticomm_full(hh):
    return np.concatenate([hh, 1.0 - hh])

def anticomm_obj(x, n, alpha):
    h = anticomm_full(x)
    c = _fft_corr_full(h, n)
    return smooth_max(c, alpha) * (2.0 / n)

def lbfgs_anticomm(n, x0, alphas, deadline):
    hh = x0.copy()
    best_c = compute_c5(anticomm_full(hh), n)
    bounds = [(0.0, 1.0)] * (n // 2)
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(anticomm_obj, hh, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-10})
            new_hh = np.clip(res.x, 0.0, 1.0)
            h_full = anticomm_full(new_hh)
            new_c = compute_c5(h_full, n)
            if new_c < best_c:
                hh, best_c = new_hh, new_c
        except Exception:
            pass
    return hh, best_c

def full_obj(x, n, alpha):
    c = _fft_corr_full(x, n)
    return smooth_max(c, alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_full(h0, n, alphas, deadline, maxiter=200):
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(full_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def cd_fft(h, n, max_passes, deadline, gs_iters=40):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    N = 1
    while N < 2 * n - 1:
        N *= 2
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        H = np.fft.fft(h, N)
        V = np.fft.fft(1.0 - h, N)
        r = np.real(np.fft.ifft(H * np.conj(V)))
        idx = (n - 1 - np.arange(2 * n - 1)) % N
        c = r[idx]
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            fw_lo = n - 1 - p
            fw_hi = 2 * n - 2 - p
            bw_lo = p
            bw_hi = p + n - 1
            m_lo = min(fw_lo, bw_lo)
            m_hi = max(fw_hi, bw_hi)
            ms = np.arange(m_lo, m_hi + 1)
            mp = ms - (n - 1)
            a = np.zeros(len(ms))
            fp = p + mp
            vf = (fp >= 0) & (fp < n)
            a[vf] += (1.0 - h[fp[vf]])
            bp = p - mp
            vb = (bp >= 0) & (bp < n)
            a[vb] -= h[bp[vb]]
            dp = np.where(mp == 0)[0]
            has_d = len(dp) > 0
            di = int(dp[0]) if has_d else -1
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            def fv(d):
                vals = c_aff + d * a
                if has_d:
                    vals[di] = c_aff[di] + d * (1.0 - 2.0 * h0) - d * d
                return max(float(np.max(vals)), unaff)
            f1, f2 = fv(d1), fv(d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-14:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = fv(d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = fv(d2)
            d_opt = (d_lo + d_hi) / 2.0
            if fv(d_opt) < fv(0.0) - 1e-13:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def chirp_hh(n, freq=1.0, phase=0.0):
    nh = n // 2
    i = np.arange(nh)
    sig = np.sin(2.0 * np.pi * freq * i * i / n + phase)
    hh = 0.5 + 0.35 * sig
    return np.clip(hh, 0.0, 1.0)

def sine_hh(n, m=3):
    nh = n // 2
    i = np.arange(nh)
    sig = np.sin(2.0 * np.pi * m * i / nh)
    hh = 0.5 + 0.35 * sig
    return np.clip(hh, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    best_h = None
    best_c = 1.0

    try:
        h_par = np.array(initial_h_values, dtype=float).copy()

    except Exception:
        h_par, n_par = None, 700

    n_list = sorted(set([350, 400, 500, 550, 600, 650, 700, 750, 800]) & set(range(2, 2000)), reverse=False)

        if time.time() > deadline - 30:

        t_rem = deadline - time.time() - 30
        if t_rem < 8:
            break
        phase_time = t_rem / max(1, len([x for x in n_list if x % 2 == 0 and x >= n_try]))
        phase_time = min(phase_time, 60.0)
        d1 = time.time() + phase_time

        inits_hh = []
        inits_hh.append(chirp_hh(n_try, 1.0, 0.0))
        inits_hh.append(chirp_hh(n_try, 2.0, np.pi / 4))
        inits_hh.append(sine_hh(n_try, 3))
        inits_hh.append(sine_hh(n_try, 5))
        inits_hh.append(0.5 + 0.1 * rng.randn(n_try // 2))
        if h_par is not None and n_try == n_par:

        for hh_init in inits_hh:

            if c_val < best_c:

            hh2, c2 = cd_fft(anticomm_full(hh), n_try, 15, d1 - 1, gs_iters=30)
            if c2 < best_c:

    if best_h is not None and time.time() < deadline - 40:

    if best_h is not None and time.time() < deadline - 10:

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
