import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if h0 - r1 <= r2 - h0:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    order = np.arange(n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        if rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-13:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def multi_resolution_escape(h, n, deadline, rng):
    best_h, best_c = h.copy(), compute_c5(h, n)
    # Try different resolutions
    for n_coarse in [n // 2, n // 3, n // 4]:
        if time.time() > deadline:
            break
        if n_coarse < 50:
            continue
        # Downsample
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_coarse)
        h_coarse = np.interp(x_new, x_old, h)
        h_coarse = project_feasible(h_coarse, n_coarse)
        # Add random perturbation

        h_coarse, c_coarse = exact_cd(h_coarse, n_coarse, 200, deadline - 30)

        h_fine = np.interp(x_old, x_new, h_coarse)
        h_fine = project_feasible(h_fine, n)

        h_fine, c_fine = exact_cd(h_fine, n, 100, deadline - 10)

At the optimum, we expect K_active constraints to be active (c_k = c_max for K_active values of k). The optimal h satisfies:

1. Compute c and gradient g (subgradient of max)
2. Take a step: h_new = h - alpha * g
3. Project to feasible set
4. Repeat

This is the **subgradient method**, which is simple but slow.

To make it faster, use **accelerated subgradient** (Nesterov's method):
1. y = h + (beta - 1) / beta * (h - h_prev) where beta = 2 / (t + 1)
2. Compute subgradient at y
3. h_new = h - alpha * g_y
4. alpha = 1 / (L * sqrt(t)) where L is a Lipschitz constant

This converges in O(1/t) iterations, which is still slow but better than O(1/sqrt(t)).

However, for non-convex problems, the convergence guarantees don't hold. But in practice, accelerated subgradient can still help.

```python
def compute_subgradient(h, n, epsilon=0.01):
    """Compute a subgradient of max_k c_k at h."""
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    active = c >= cmax - epsilon * cmax
    n_active = np.sum(active)
    
    grad = np.zeros(n)
    # For each active k, compute the gradient contribution
    # c_k = sum_i h[i] * (1 - h[i + k - n + 1])
    # dc_k/dh[p] = (1 - h[p + k - n + 1]) - h[p - k + n - 1]
    
    # For efficiency, use convolution:
    # grad[p] = (1/n_active) * sum_{k active} [(1-h[p+m_k]) - h[p-m_k]]
    # where m_k = k - n + 1
    
    # Let A[m] = 1/n_active if c[n-1+m] is active, else 0
    # Then grad = A * (1-h) - A * h (with appropriate indexing)
    
    # Actually, let's just compute it directly for the active k's
    ms = np.where(active)[0] - (n - 1)  # shifts
    for m in ms:
        # For shift m, the valid range of p is:
        # p + m in [0, n-1] AND p - m in [0, n-1]
        # => max(0, -m) <= p <= min(n-1, n-1-m)
        p_lo = max(0, -m)
        p_hi = min(n - 1, n - 1 - m)
        if p_lo > p_hi:
            # Only one term is valid
            if 0 <= m < n:
                # h[p+m] is valid for p in [0, n-1-m]
                p2_lo, p2_hi = 0, n - 1 - m
                grad[p2_lo:p2_hi+1] += (1.0 - h[p2_lo+m:p2_hi+1+m])
