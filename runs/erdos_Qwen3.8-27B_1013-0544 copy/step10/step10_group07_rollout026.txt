def subgrad_step(h, n, c, step_size, active_tol=0.05):
    """One subgradient step with averaged active gradients."""
    cur_max = c.max()
    active = np.where(c > cur_max * (1 - active_tol))[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    # Compute averaged gradient
    grad = np.zeros(n)
    for m in active:
        offset = m - (n - 1)
        lo1 = max(0, -offset)
        hi1 = min(n, n - offset)
        grad[lo1:hi1] += (1.0 - h[lo1 + offset:hi1 + offset])
        lo2 = max(0, offset)
        hi2 = min(n, n + offset)
        grad[lo2:hi2] -= h[lo2 - offset:hi2 - offset]
    grad /= len(active)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return h
    # Step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

                m = c.max()
                shifted = alpha * (c - m)
                mx = shifted.max()
                lse = mx + np.log(np.sum(np.exp(shifted - mx)))
                return (m + lse / alpha) * (2.0 / n)
            def cf(z):
                return z.sum() - n / 2.0
            def cj(z):
                return np.ones(n)

                          options={'maxiter': maxiter, 'ftol': 1e-20})

            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
