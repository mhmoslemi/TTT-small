            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_cd_coord(c, h, p, n, nb=35):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = max(unaff, float(np.max(c[:m_min])))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(nb):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if h0 - r1 <= r2 - h0:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_pass(h, n, deadline, nb=35):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    improved = False
    for p in range(n):
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-13 or h0 > 1.0 - 1e-13:
            continue
        xn = solve_cd_coord(c, h, p, n, nb)
        if abs(xn - h0) > 1e-11:
            delta = xn - h0
            ms = np.arange(n - 1 - p, 2 * n - 1 - p)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            diag_i = np.where(ms == (n - 1))[0]
            c[ms] += delta * a
            if len(diag_i) > 0:
                c[ms[diag_i[0]]] -= delta * delta
            h[p] = xn
            improved = True
    h = project_feasible(h, n)
    if abs(h.sum() - n / 2.0) > 1e-12:
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n), improved

def exact_cd(h, n, max_passes, deadline, nb=35):
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        h, c5, improved = cd_pass(h, n, deadline, nb)
        if not improved:
            break
    return h, c5

def fourier_init(n, rng, nh=4):
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for k in range(1, nh + 1):
        freq = k + rng.randint(0, 2)
        amp = rng.uniform(0.04, 0.15) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(np.pi * freq * x + phase)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def plateau_init(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    start = rng.randint(0, n - n_ones)
    h[start:start + n_ones] = 1.0
    noise = rng.normal(0, 0.1, n)
    h = np.clip(h + noise, 0, 1)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = fourier_init(700, rng, 5)
        n = len(h_best)
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Quick CD stabilization
    if time.time() < deadline - 200:
        h_best, best_c = exact_cd(h_best, n, 25, deadline - 200, nb=30)

    # Phase 2: Diverse multi-start with CD refinement
    n_starts = 14
    for i in range(n_starts):
        if time.time() > deadline - 350:
            break
        elapsed = time.time() - t0
        remain = deadline - time.time() - 350
        if remain < 25:
            break
        alloc = min(remain / max(1, n_starts - i), 80)
        npass = max(5, int(alloc / 2.5))

        if i % 4 == 0:
            scale = 0.008 + 0.004 * i
            hp = h_best + rng.normal(0, scale, n)
        elif i % 4 == 1:
            hp = fourier_init(n, rng, 3 + (i % 5))
        elif i % 4 == 2:
            hp = plateau_init(n, rng)
        else:
            hp = h_best.copy()
            hp = np.roll(hp, rng.randint(-10, 10))
            hp = project_feasible(hp, n)
            mask = rng.random(n) < 0.25
            hp[mask] += rng.normal(0, 0.15, mask.sum())

        hp = project_feasible(hp, n)
        h_try, c_try = exact_cd(hp, n, npass, time.time() + alloc, nb=30)
        if c_try < best_c:
            h_best, best_c = h_try, c_try

    # Phase 3: Multi-n exploration
    for n_try in [550, 650, 850, 950]:
        if time.time() > deadline - 250:
            break
        if n_try == n or n_try > 1100:
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        nm = len(h_int)
        remain = deadline - time.time() - 250
        if remain < 30:
            break
        alloc = min(remain * 0.5, 120)
        h_try, c_try = exact_cd(h_int, nm, max(10, int(alloc / 2.5)), time.time() + alloc, nb=30)
        if c_try < best_c - 1e-9:
            h_best, best_c, n = h_try, c_try, nm

    # Phase 4: Final intensive CD with higher bisection precision
    n = len(h_best)
    h_best, best_c = exact_cd(h_best, n, 500, deadline, nb=50)

    # Phase 5: One more round of small perturbations + CD
    if time.time() < deadline - 10:
        for _ in range(3):
            if time.time() > deadline - 5:
                break
            hp = h_best + rng.normal(0, 0.005, n)
            hp = project_feasible(hp, n)
            h_try, c_try = exact_cd(hp, n, 50, deadline, nb=45)
            if c_try < best_c:
                h_best, best_c = h_try, c_try

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
