            m_min = n - 1 - p
            m_max = 2 * n - 2 - p
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c_base = c[ms].copy()
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            a_diag_val = a[diag_idx[0]] if has_diag else 0.0
            if has_diag:
                a[diag_idx] = 0.0
            c_diag = c_base[diag_idx[0]] if has_diag else 0.0
            unaff = 0.0
            if m_min > 0:
                unaff = max(unaff, float(np.max(c[:m_min])))
            if m_max < 2 * n - 2:
                unaff = max(unaff, float(np.max(c[m_max + 1:])))
            cur_max = float(np.max(c))
            neg_mask = a < -1e-15
            pos_mask = a > 1e-15
            zero_mask = np.abs(a) <= 1e-15
            L_lo, L_hi = unaff, cur_max
            for _ in range(50):
                if L_hi - L_lo < 1e-15:
                    break
                L = (L_lo + L_hi) / 2.0
                x_lo, x_hi = 0.0, 1.0
                feasible = True
                if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
                    feasible = False
                if feasible and np.any(neg_mask):
                    x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
                if feasible and np.any(pos_mask):
                    x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
                if x_lo > x_hi + 1e-14:
                    feasible = False
                if feasible and has_diag:
                    D = L - c_diag + h0 - h0 * h0
                    disc = 1.0 - 4.0 * D
                    if disc >= 0:
                        r1 = (1.0 - np.sqrt(disc)) / 2.0
                        r2 = (1.0 + np.sqrt(disc)) / 2.0
                        xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                        if xc_hi > r1 and xc_lo < r2:
                            feasible = False
                if feasible:
                    L_hi = L
                else:
                    L_lo = L
            L = L_hi
            x_lo, x_hi = 0.0, 1.0
            if np.any(neg_mask):
                x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
            if np.any(pos_mask):
                x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
            x_lo = max(x_lo, 0.0)
            x_hi = min(x_hi, 1.0)
            if has_diag:
                D = L - c_diag + h0 - h0 * h0
                disc = max(0.0, 1.0 - 4.0 * D)
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if xc_hi > r1 and xc_lo < r2:
                    if h0 - r1 <= r2 - h0:
                        x_hi = min(x_hi, r1)
                    else:
                        x_lo = max(x_lo, r2)
            if x_lo > x_hi:
                x_new = h0
            else:
                x_new = float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))
            if abs(x_new - h0) > 1e-13:
                delta = x_new - h0
                c[ms] += delta * a
                if has_diag:
                    c[ms[diag_idx[0]]] += delta * a_diag_val - delta * delta
                h[p] = x_new
                improved += 1
        if improved == 0:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def cons_fun(x, n_=n):
            return x.sum() - n_ / 2.0
        def cons_jac(x, n_=n):
            return np.ones(n_)
        cons = [{'type': 'eq', 'fun': lambda x, n_=n: x.sum() - n_ / 2.0,
                 'jac': lambda x, n_=n: np.ones(n_)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgrad_momentum(h, n, lr, momentum, deadline, max_steps):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    active = c > cmax - 0.05 * cmax
    n_active = max(1, active.sum())
    grad = np.zeros(n)
    for p in range(n):
        m_min, m_max = n - 1 - p, 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        contrib = np.zeros(len(ms))
        if np.any(jf >= 0) and np.any(jf < n):
            vf = (jf >= 0) & (jf < n)
            contrib[vf] += (1.0 - h[jf[vf]])
        vb = (jb >= 0) & (jb < n)
        if np.any(vb):
            contrib[vb] -= h[jb[vb]]
        di = (ms == (n - 1))
        if np.any(di):
            contrib[di] = (1.0 - 2.0 * h[p])
        mask = active[ms]
        if mask.any():
            grad[p] = np.sum(contrib[mask]) / mask.sum()
    h = np.clip(h - lr * grad, 0.0, 1.0)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Fixed-order corrected CD
    h1, c1 = exact_cd(h_best, n, 800, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: SLSQP with multiple alphas
    if time.time() < deadline - 300:
        h2, c2 = slsqp_opt(h_best, n, [3000, 10000, 50000, 200000], 400, deadline)
        if c2 < best_c:
            h_best, best_c = h2, c2

    # Phase 3: CD again after SLSQP
    h3, c3 = exact_cd(h_best, n, 800, deadline)
    if c3 < best_c:
        h_best, best_c = h3, c3

    # Phase 4: Random-order CD
    h4, c4 = exact_cd(h_best, n, 500, deadline, random_order=True, rng=rng)
    if c4 < best_c:
        h_best, best_c = h4, c4

    # Phase 5: Subgradient + CD
    if time.time() < deadline - 250:
        for lr in [0.03, 0.015, 0.008, 0.004]:
            if time.time() > deadline - 100:
                break
            hs = h_best.copy()
            for _ in range(30):
                if time.time() > deadline - 100:
                    break
                hs = subgrad_momentum(hs, n, lr, 0.9, deadline, 30)
            ho, co = exact_cd(hs, n, 400, deadline)
            if co < best_c:
                h_best, best_c = ho, co

    # Phase 6: Try different n values
    n_list = [600, 750, 850, 550, 950]
    for n_try in n_list:
        if time.time() > deadline - 150:
            break
        n_cur = len(h_best)
        x_old = np.linspace(0, 2, n_cur)
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        ho, co = exact_cd(h_int, n_try, 400, deadline)
        if co < best_c - 1e-8:
            h_best, best_c = ho, co
        ho, co = slsqp_opt(h_int, n_try, [10000, 50000], 200, deadline)
        ho, co = exact_cd(ho, n_try, 300, deadline)
        if co < best_c - 1e-8:
            h_best, best_c = ho, co

    # Phase 7: Noise perturbation + CD
    n = len(h_best)
    for s in [0.01, 0.02, 0.04, 0.08, 0.005]:
        if time.time() > deadline - 30:
            break
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        ho, co = exact_cd(hp, n, 400, deadline)
        if co < best_c:
            h_best, best_c = ho, co

    # Phase 8: Final polish
    if time.time() < deadline - 5:
        hf, cf = exact_cd(h_best, len(h_best), 500, deadline)
        if cf < best_c:
            h_best, best_c = hf, cf

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
