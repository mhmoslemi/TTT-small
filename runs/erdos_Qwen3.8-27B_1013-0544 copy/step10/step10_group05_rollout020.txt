def make_exact_sum(h,n):
    h = np.clip(h,0.0,1.0)
    target = n/2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13: break
        if diff > 0:
            idx = np.nonzero(h < 1.0 - 1e-15)[0]
            if idx.size == 0: break
            room = 1.0 - h[idx]
            # load into largest room until diff
            order = idx[np.argsort(-room)]
            for i in order:
                if diff <= 0: break
                inc = min(1.0 - h[i], diff)
                h[i] += inc
                diff -= inc
        else:
            idx = np.nonzero(h > 1e-15)[0]
            if idx.size == 0: break
            order = idx[np.argsort(h)]
            for i in order:
                if diff >= 0: break
                dec = min(h[i], -diff)
                h[i] -= dec
                diff += dec
    # If still off due fp, distribute to first interior
    if abs(target - h.sum()) > 1e-12:
        idx = np.nonzero((h > 1e-12) & (h < 1 - 1e-12))[0]

                    dec = min(h[i], -diff); h[i]-=dec; diff+=dec

def compute_c5(h,n):
    dx=2.0/n
    return float(np.max(np.correlate(h,1.0-h,mode='full'))*dx)

def project_feasible(h,n):
    h=np.clip(h.astype(float),0.0,1.0)
    target=n/2.0
    for _ in range(200):
        diff=target-h.sum()
        if abs(diff)<1e-13: break
        if diff>0:
            room=1.0-h
            tot=room.sum()
            if tot<=1e-15: break
            h=np.clip(h + (diff/tot)*room,0.0,1.0)
        else:
            tot=h.sum()
            if tot<=1e-15: break
            h=np.clip(h + (diff/tot)*h,0.0,1.0)
    diff=target-h.sum()
    if abs(diff)>1e-12:
        idx=np.where((h>1e-14)&(h<1.0-1e-14))[0]
        rem=diff
        for i in idx:
            if rem>0:
                inc=min(1.0-h[i], rem)
                h[i]+=inc; rem-=inc
            else:
                dec=min(h[i], -rem)
                h[i]-=dec; rem+=dec
            if abs(rem)<1e-14: break
    return h

def fix_sum_exact(h,n):

def subgradient(c,h,n,alpha=3000.0,top=80):
    m=float(np.max(c))
    z=alpha*(c-m)
    p=np.exp(z)
    S=p.sum()
    if S <= 0.0:
        p=np.zeros_like(c); p[int(np.argmax(c))]=1.0
    else:
        p=p/S
    if 0 < top < p.size:
        idx=np.argpartition(p, -top)[-top:]
        idx=idx[np.argsort(p[idx])[::-1]]
    else:
        idx=np.nonzero(p>1e-12)[0]
    if idx.size==0:
        idx=np.array([int(np.argmax(c))], dtype=int)
    g=np.zeros(n,float)
    mid=n-1
    for k in idx:
        w=float(p[k])
        if w <= 1e-14: continue
        d=int(k)-mid
        if d>=0:
            L=n-d
            i=np.arange(L)
            g[i] += w*(1.0-h[i+d])
            g[i+d] -= w*h[i]
        else:
            e=-d
            L=n-e
            i=np.arange(L)
            g[i+e] += w*(1.0-h[i])
            g[i] -= w*h[i+e]
    g -= g.mean()
    return g

def harmonic_init(n,K,rng,scale=0.85):
    x=np.linspace(0.0,2.0,n,endpoint=False)
    u=np.zeros(n)

import numpy as np
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(250):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot <= 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot <= 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        rem = diff
        for i in idx:
            if rem > 0:
                inc = min(1.0 - h[i], rem)
                h[i] += inc
                rem -= inc
            else:
                dec = min(h[i], -rem)
                h[i] -= dec
                rem += dec
            if abs(rem) < 1e-14:
                break
    return h


def fix_sum_exact(h, n):
    h = project_feasible(h, n)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
            if idx.size:
                idx = idx[np.argsort(-(1.0 - h[idx]))]
                for i in idx:
                    inc = min(1.0 - h[i], diff)
                    if inc <= 0:
                        continue
                    h[i] += inc
                    diff -= inc
                    if abs(diff) < 1e-14:

        else:
            idx = np.where(h > 1e-15)[0]

                    if dec <= 0:

def interpolate_h(h_src, n_src, n_tgt):

    x_tgt = np.linspace(0.0, 2.0, n_tgt, endpoint=False)

def apply_coordinate(c, h, p, n, maxit=20, bias=0.0):

    last = 2 * n - 2
    m_min = max(0, n - 1 - p)

    jb = p - d

    a = np.zeros(ms.size, float)

    if np.any(vf):

    if np.any(vb):

        xhi = h0

            if np.any(pos):
                xhi = min(xhi, float(h0 + np.min((L - c_base[pos]) / a[pos])))
            xlo = max(0.0, xlo)
            xhi = min(1.0, xhi)
            if xlo > xhi + 1e-11:
                feasible = False

        if feasible and has_diag:
            disc = a_d * a_d + 4.0 * (c_diag - L)
            if disc >= 0.0:
                sq = np.sqrt(max(0.0, disc))
                r1 = 0.5 * (a_d - sq)

                A = xlo
                B = xhi
                if not (A <= min(B, h0 + r1) + 1e-11 or max(A, h0 + r2) <= B + 1e-11):

            Lhi = L
        else:
            Llo = L

    L = Lhi + 1e-12
    xlo = h0
    xhi = h0

            cands.append(float(np.clip(h0 + r2, 0.0, 1.0)))

    best_x = h0
    best_val = None
    best_dist = 1e30
    best_score = -1e30

    for x in cands:
        xc = float(np.clip(x, 0.0, 1.0))
        delta = xc - h0
        if best_val is not None and abs(delta) < 1e-13:
            continue
        val = c_base + a * delta
        if has_diag:
            val[diag_i] -= delta * delta
        nm = float(np.max(val))
        if unaff > nm:
            nm = unaff
        sc = (xc - h0) * bias
        better = False
        if best_val is None:
            better = True
        elif nm < best_val - 1e-12:
            better = True

            best_x = xc

    if best_val is None or best_val >= cur_max - 1e-13:
        return False

    delta = best_x - h0
    if abs(delta) < 1e-13:
        return False

    c[ms] += delta * a
    if has_diag:
        c[ms[diag_i]] -= delta * delta
    h[p] = best_x
    return True


def subgradient(c, h, n, alpha=3000.0, top=80):
    m = float(np.max(c))
    z = alpha * (c - m)
    p = np.exp(z)
    S = p.sum()
    if S <= 0.0:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = p / S

    if 0 < top < p.size:
        idx = np.argpartition(p, -top)[-top:]
        idx = idx[np.argsort(p[idx])[::-1]]
    else:
        idx = np.nonzero(p > 1e-12)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))], dtype=int)

    g = np.zeros(n, float)
    mid = n - 1
    for k in idx:
        w = float(p[k])
        if w <= 1e-14:
            continue
        d = int(k) - mid
        if d >= 0:
            L = n - d
            i = np.arange(L)
            g[i] += w * (1.0 - h[i + d])
            g[i + d] -= w * h[i]
        else:
            e = -d
            L = n - e
            i = np.arange(L)
            g[i + e] += w * (1.0 - h[i])
            g[i] -= w * h[i + e]

    g -= g.mean()
    return g


def cd_loop(h, n, max_passes, deadline, rng=None, use_active=True, use_random=False):
    h = project_feasible(h.copy(), n)
    c = np.correlate(h, 1.0 - h, mode="full")
    cur_c = compute_c5(h, n)
    best_h = h.copy()
    best_c = cur_c
    stagn = 0

    for _ in range(max_passes):
        if time.time() > deadline:
            break

        bias = n / 2.0 - h.sum()

        if use_random and rng is not None:
            order = rng.permutation(n)
        elif use_active:
            g = subgradient(c, h, n, 3000.0, 80)
            gn = float(np.dot(g, g))
            if gn < 1e-20:
                order = np.arange(n)
            else:
                order = np.argsort(-np.abs(g))
        else:
            order = np.arange(n)

        improved = 0
        for p in order:
            if apply_coordinate(c, h, p, n, 20, bias):
                improved += 1

        if improved == 0:
            break

        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode="full")
        cur_c = compute_c5(h, n)

        if cur_c < best_c - 1e-14:
            best_h = h.copy()
            best_c = cur_c
            stagn = 0
        else:
            stagn += 1
            if cur_c > best_c + 1e-10:
                h = best_h.copy()
                c = np.correlate(h, 1.0 - h, mode="full")
            if stagn >= 5:
                break

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def harmonic_init(n, K, rng, scale=0.85):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    u = np.zeros(n)
    tot = 0.0
    for k in range(1, K + 1):
        amp = rng.uniform(0.3, 1.0) / np.sqrt(k)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        u += amp * np.cos(np.pi * k * x + ph)
        tot += amp
    u -= u.mean()
    if tot > 0:
        u = (u / tot) * scale
    else:
        u *= scale
    h = 0.5 + u
    return project_feasible(h, n)


def wavelet_init(n, rng):
    h = np.full(n, 0.5)
    blocks = max(4, n // 12)
    for i in range(0, n, blocks):
        h[i:min(i + blocks, n)] = rng.choice([0.05, 0.5, 0.95])
    return project_feasible(h, n)

            d = -g / norm
            accepted = False
            s = step
            for __ in range(8):
                h2 = project_feasible(h + s * d, n)
                v = compute_c5(h2, n)
                if v < cur - 1e-12:
                    h = h2
                    cur = v
                    accepted = True
                    break
                s *= 0.5
            if not accepted:
                step *= 0.6
                if step < 1e-4:
                    break
            if cur < best_c - 1e-14:
                best = h.copy()
                best_c = cur

    return best, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_best = np.full(700, 0.5)

    n_best = len(h_best)
    h_best = project_feasible(h_best, n_best)
    best_c = compute_c5(h_best, n_best)

    if time.time() < deadline:
        h1, c1 = cd_loop(h_best, n_best, 120, deadline, rng=rng, use_active=True)
        if c1 < best_c:
            h_best, best_c = h1, c1

    if time.time() < deadline - 60:
        h1, c1 = cd_loop(h_best, n_best, 60, deadline, rng=rng, use_active=False, use_random=True)
        if c1 < best_c:
            h_best, best_c = h1, c1

    if time.time() < deadline - 80:
        h0 = harmonic_init(n_best, 3, rng, 0.8)
        h1, c1 = cd_loop(h0, n_best, 45, deadline, rng=rng, use_active=True)
        if c1 < best_c:
            h_best, best_c = h1, c1

    if time.time() < deadline - 80:
        h0 = harmonic_init(n_best, 6, rng, 0.9)
        h1, c1 = cd_loop(h0, n_best, 45, deadline, rng=rng, use_active=True)
        if c1 < best_c:
            h_best, best_c = h1, c1

    if time.time() < deadline - 80:
        h0 = wavelet_init(n_best, rng)
        h1, c1 = cd_loop(h0, n_best, 45, deadline, rng=rng, use_active=True)
        if c1 < best_c:
            h_best, best_c = h1, c1

    for temp in (0.003, 0.008, 0.02, 0.05):
        if time.time() < deadline - 80:
            hp = project_feasible(h_best + rng.normal(0.0, temp, n_best), n_best)
            h1, c1 = cd_loop(hp, n_best, 35, deadline, rng=rng, use_active=True)
            if c1 < best_c:
                h_best, best_c = h1, c1

    tried = set([n_best])
    for ntry in (n_best + 50, n_best + 100, 800, 600, 1000, n_best - 50):
        ntry = int(ntry)
        if ntry in tried or ntry < 200 or ntry > 1200:
            continue
        tried.add(ntry)
        if time.time() < deadline - 120:
            hi = interpolate_h(h_best, n_best, ntry)
            h1, c1 = cd_loop(hi, ntry, 45, deadline, rng=rng, use_active=True)
            if c1 < best_c - 1e-7:
                h_best, best_c, n_best = h1, c1, ntry

    if time.time() < deadline - 20:
        h1, c1 = basin_polish(h_best, n_best, deadline - 5)
        if c1 < best_c:
            h_best, best_c = h1, c1

    if time.time() < deadline - 3:
        h1, c1 = cd_loop(h_best, n_best, 30, deadline - 2, rng=rng, use_active=True)
        if c1 < best_c:
            h_best, best_c = h1, c1

    h_best = fix_sum_exact(h_best, n_best)
    best_c = compute_c5(h_best, n_best)
    return h_best, float(best_c), int(n_best)
