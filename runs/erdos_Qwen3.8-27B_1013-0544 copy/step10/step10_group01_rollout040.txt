```python
def active_set_lp(h, n, active_set, t_bound):

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    c = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute gradients of c_k with respect to h at current point
    # For shift s = k - (n-1):
    # d c_k / d h[j] = (1-h[j+s]) if j+s in [0,n) and s != 0
    #                 + (-h[j-s]) if j-s in [0,n) and s != 0

    #           if s=0 and j is both: (1-2h[j])
    
    constraints = []
    for k in active_set:
        s = k - (n + 1 - 2)  # s = k - (n-1)

            else:
                if 0 <= j + s < n:
                    grad[j] += (1 - h[j + s])
                if 0 <= j - s < n:
                    grad[j] += (-h[j - s])
        constraints.append(c[k] + grad @ (h_var - h) <= t_var)
    
    constraints.append(cp.sum(h_var) == n / 2.0)
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    constraints.append(t_var <= t_bound)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    prob.solve(solver=cp.SCS, max_iters=10000)
    
    if h_var.value is not None:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            if res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _cd_fval(c_aff, a, unaff_max, d):
    vals = c_aff + d * a
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def golden_section_1d(c_aff, a, unaff_max, h0, n_iter=45):
    """Find optimal d in [-h0, 1-h0] minimizing max(c_aff + d*a, unaff_max)."""
    d_lo, d_hi = -h0, 1.0 - h0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    d1 = d_hi - gr * (d_hi - d_lo)
    d2 = d_lo + gr * (d_hi - d_lo)
    f1 = _cd_fval(c_aff, a, unaff_max, d1)
    f2 = _cd_fval(c_aff, a, unaff_max, d2)

            break
        if f1 < f2:

            d1 = d_hi - gr * (d_hi - d_lo)

        else:

            d2 = d_lo + gr * (d_hi - d_lo)
            f2 = _cd_fval(c_aff, a, unaff_max, d2)
    return (d_lo + d_hi) / 2.0

def single_cd_pass(h, n, order, gs_iters):
    """One pass of CD over coordinates in given order. Returns (h, improved)."""
    h = h.copy()
    improved = False
    for idx in order:
        h0 = h[idx]
        if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
            continue
        c = compute_corr(h, n)
        m_lo = max(0, n - 1 - idx)
        m_hi = min(2*n - 2, n - 1 + idx)
        ms = np.arange(m_lo, m_hi + 1)
        s_vals = ms - (n - 1)
        a = np.zeros(len(ms))
        for i, s in enumerate(s_vals):
            if s == 0:
                a[i] += (1.0 - 2.0 * h0)
            else:
                if 0 <= idx + s < n:
                    a[i] += (1.0 - h[idx + s])
                if 0 <= idx - s < n:
                    a[i] += (-h[idx - s])
        c_aff = c[ms]
        unaff_max = 0.0

        if abs(d_opt) > 1e-14:

            improved = True

    """Active-set-aware coordinate descent."""

        rng = np.random.RandomState(0)
    h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if time.time() > deadline:
            break
        c = compute_corr(h, n)
        cmax = c.max()
        active = np.where(c >= cmax - eps)[0]
        # Priority: coordinates in active constraints
        priority = np.zeros(n)
        for k in active:

            hi = min(n, n - abs(int(s)))

            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    return h, best_c

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            if res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
```

```python
def _gs_with_diag(c_aff, a, unaff_max, h0, has_diag, diag_pos, n_iter):
    """Golden section with quadratic correction at diagonal position."""
    d_lo, d_hi = -h0, 1.0 - h0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    def f(d):
        vals = c_aff + d * a
        if has_diag:
            vals[diag_pos] = c_aff[diag_pos] + d * (1.0 - 2.0 * h0) - d * d
        result = float(np.max(vals))
        if unaff_max > result:
            result = unaff_max
        return result
    
    d1 = d_hi - gr * (d_hi - d_lo)
    d2 = d_lo + gr * (d_hi - d_lo)
    f1 = f(d1)
    f2 = f(d2)
    for _ in range(n_iter):

            d1 = d_hi - gr * (d_hi - d_lo)

        else:
            d_lo, d1, f1 = d1, d2, f2
            d2 = d_lo + gr * (d_hi - d_lo)
            f2 = f(d2)
    return (d_lo + d_hi) / 2.0
```

- Or h(x) = sigmoid(α(x - x0)) + ...

**Alternative: Use the duality**

The problem min_h max_k c_k(h) can be written as:
min_h max_k c_k(h) = min_h Σ_k p_k c_k(h) over p_k ≥ 0, Σ p_k = 1

(by minimax theorem, if the feasible set is convex... but it's not, since c_k is bilinear)

So duality doesn't directly apply. But we can still use the "weighted average of c_k" as a smooth objective and optimize it, with p_k adapted based on which c_k are largest.

This is essentially what the smooth_obj function with log-sum-exp does. The alpha parameter controls how "peaked" the weight is on the maximum.

```python
def nesterov_smooth(h, n, alpha, n_steps, lr, deadline):
    h = h.copy()
    v = np.zeros(n)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Compute gradient of smooth objective
        c = compute_corr(h, n)
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        weights = np.exp(shifted - mx) / np.exp(shifted - mx).sum()
        
        # Gradient of weighted sum of c_k
        grad = np.zeros(n)
        for k in range(len(c)):
            if weights[k] < 1e-10:
                continue
            s = k - (n - 1)
            for j in range(n):
                if s == 0:
                    if 0 <= j < n:
                        grad[j] += weights[k] * (1 - 2*h[j]) * (2.0/n)
                else:
                    if 0 <= j + s < n:
                        grad[j] += weights[k] * (1 - h[j+s]) * (2.0/n)
                    if 0 <= j - s < n:
                        grad[j] += weights[k] * (-h[j-s]) * (2.0/n)
        
        # Nesterov step
        v = 0.9 * v - lr * grad
        h_new = np.clip(h + v, 0, 1)
        h_new = project_feasible(h_new, n)
        h = h_new
    
    return h
```

For a given weight vector w (over lags), the gradient is:
grad[j] = (2/n) * [sum_{s: j+s < n} w[j+s relative] * (1-h[j+s]) 
                     - sum_{s: j-s >= 0} w[j-s relative] * h[j-s]
                     + w[n-1] * (1-2h[j])]  (for s=0)

- w_neg[s] = w[n-1-s] for s = 1, ..., n-1 (negative shifts)  
- w_zero = w[n-1]

Then:
grad[j] = (2/n) * [(w_pos convolved with (1-h))[j] + (w_neg convolved with h)[j] + w_zero*(1-2h[j])]

Where the convolution is appropriately truncated.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _gs_eval(c_aff, a, unaff_max, h0, has_diag, diag_pos, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_pos] = c_aff[diag_pos] + d * (1.0 - 2.0 * h0) - d * d
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def _gs_search(c_aff, a, unaff_max, h0, has_diag, diag_pos, n_iter):
    d_lo, d_hi = -h0, 1.0 - h0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    d1 = d_hi - gr * (d_hi - d_lo)
    d2 = d_lo + gr * (d_hi - d_lo)
    f1 = _gs_eval(c_aff, a, unaff_max, h0, has_diag, diag_pos, d1)
    f2 = _gs_eval(c_aff, a, unaff_max, h0, has_diag, diag_pos, d2)
    for _ in range(n_iter):
        if d_hi - d_lo < 1e-15:
            break
        if f1 < f2:
            d_hi, d2, f2 = d2, d1, f1
            d1 = d_hi - gr * (d_hi - d_lo)
            f1 = _gs_eval(c_aff, a, unaff_max, h0, has_diag, diag_pos, d1)
        else:
            d_lo, d1, f1 = d1, d2, f2
            d2 = d_lo + gr * (d_hi - d_lo)
            f2 = _gs_eval(c_aff, a, unaff_max, h0, has_diag, diag_pos, d2)
    return (d_lo + d_hi) / 2.0

def cd_pass(h, n, order, gs_iters):
    h = h.copy()
    c = compute_corr(h, n)
    improved = False
    for idx in order:
        h0 = h[idx]
        if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
            continue
        m_lo = max(0, n - 1 - idx)
        m_hi = min(2 * n - 2, n - 1 + idx)
        ms = np.arange(m_lo, m_hi + 1)
        s_vals = ms - (n - 1)
        a = np.zeros(len(ms))
        has_diag = False
        diag_pos = -1
        for i in range(len(ms)):
            s = int(s_vals[i])
            if s == 0:
                a[i] = 1.0 - 2.0 * h0
                has_diag = True
                diag_pos = i
            else:
                if 0 <= idx + s < n:
                    a[i] += (1.0 - h[idx + s])
                if 0 <= idx - s < n:
                    a[i] += (-h[idx - s])
        c_aff = c[m_lo:m_hi + 1].copy()
        unaff_max = 0.0
        if m_lo > 0:
            unaff_max = max(unaff_max, float(c[:m_lo].max()))
        if m_hi < 2 * n - 2:
            unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
        d_opt = _gs_search(c_aff, a, unaff_max, h0, has_diag, diag_pos, gs_iters)
        if abs(d_opt) > 1e-14:
            h[idx] = h0 + d_opt
            for i in range(len(ms)):
                s = int(s_vals[i])
                if s == 0:
                    c[m_lo + i] += d_opt * (1.0 - 2.0 * h0) - d_opt * d_opt
                else:
                    c[m_lo + i] += d_opt * a[i]
            improved = True
    return h, improved

def active_cd(h, n, max_passes, deadline, eps=0.001, gs_iters=50, rng=None, random_mix=0.05):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if time.time() > deadline:
            break
        c = compute_corr(h, n)
        cmax = c.max()
        active = np.where(c >= cmax - eps)[0]
        priority = np.zeros(n)
        for k in active:
            s = int(k - (n - 1))
            lo = max(0, -s)
            hi = min(n, n - abs(s))
            priority[lo:hi] += 1.0
        if random_mix > 0:
            order = np.argsort(-priority + rng.rand(n) * random_mix)
        else:
            order = np.argsort(-priority)
        h_new, improved = cd_pass(h, n, order, gs_iters)
        if improved:
            h = project_feasible(h_new, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    return h, best_c

def slsqp_opt(h, n, alphas, maxiter_each=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            if res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def anneal_cd(h, n, t_start, t_end, n_iters, deadline, gs_iters=45, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (it / max(1, n_iters - 1))
        n_pert = rng.randint(5, 20)
        idxs = rng.choice(n, n_pert, replace=False)
        h_pert = h.copy()
        h_pert[idxs] += rng.normal(0, T, n_pert)
        h_pert = project_feasible(h_pert, n)
        h_cd, c_cd = active_cd(h_pert, n, 15, deadline, eps=0.002, gs_iters=gs_iters,
                                rng=rng, random_mix=0.1)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd.copy()
            h = h_cd.copy()
        else:
            delta = c_cd - best_c
            if T > 1e-8 and rng.rand() < np.exp(-delta / (T * 0.02)):
                h = h_cd.copy()
        h_cd2, c_cd2 = active_cd(h, n, 8, deadline, eps=0.001, gs_iters=gs_iters,
                                  rng=rng, random_mix=0.02)
        if c_cd2 < best_c:
            best_c = c_cd2
            best_h = h_cd2.copy()
            h = h_cd2.copy()
    return best_h, best_c

def multi_res(h_best, n_best, deadline, rng):
    x_orig = np.linspace(0, 2, n_best)
    best_h, best_c = h_best.copy(), compute_c5(h_best, n_best)
    for n_try in [150, 250, 400]:
        if time.time() > deadline:
            break
        if n_try >= n_best:
            continue
        x_s = np.linspace(0, 2, n_try)
        h_s = project_feasible(np.interp(x_s, x_orig, best_h), n_try)
        h_s, c_s = active_cd(h_s, n_try, 60, deadline, eps=0.002, gs_iters=45, rng=rng)
        h_back = project_feasible(np.interp(x_orig, x_s, h_s), n_best)
        h_back, c_back = active_cd(h_back, n_best, 30, deadline, eps=0.001, gs_iters=50, rng=rng)
        if c_back < best_c:
            best_h, best_c = h_back, c_back
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: Quick SLSQP warmup
    if time.time() < deadline - 500:
        h1, c1 = slsqp_opt(best_h, n, [3000, 15000], 250)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Main active CD
    h2, c2 = active_cd(best_h, n, 120, deadline - 350, eps=0.001, gs_iters=55, rng=rng)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Simulated annealing with CD
    if time.time() < deadline - 200:
        t_rem = deadline - time.time() - 200
        n_iter = max(25, int(t_rem / 12))
        h3, c3 = anneal_cd(best_h, n, 0.04, 0.002, n_iter,
                           deadline - 150, gs_iters=45, rng=rng)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Multi-resolution
    if time.time() < deadline - 100:
        h4, c4 = multi_res(best_h, n, deadline - 50, rng)
        if c4 < best_c:
            best_h, best_c = h4, c4
    # Phase 5: Final tight CD
    if time.time() < deadline - 15:
        h5, c5 = active_cd(best_h, n, 60, deadline - 5, eps=0.0003, gs_iters=65, rng=rng)
        if c5 < best_c:
            best_h, best_c = h5, c5
    # Phase 6: Final SLSQP polish
    if time.time() < deadline - 2:
        n_cur = len(best_h)
        h6, c6 = slsqp_opt(best_h, n_cur, [100000, 500000], 400)
        if c6 < best_c:
            best_h, best_c = h6, c6
        best_h = project_feasible(best_h, n_cur)
        best_c = compute_c5(best_h, n_cur)
    return best_h, best_c, len(best_h)
```
