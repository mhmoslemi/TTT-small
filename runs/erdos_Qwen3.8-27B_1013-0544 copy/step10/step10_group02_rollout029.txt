        if np.any(zm) and np.any(c_base[zm] > L + 1e-14):
            feasible = False
        if feasible and np.any(nm):
            x_lo = max(x_lo, float(np.max((L - c_base[nm]) / a[nm])))
        if feasible and np.any(pm):
            x_hi = min(x_hi, float(np.min((L - c_base[pm]) / a[pm])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    nm = a < -1e-15
    pm = a > 1e-15
    if np.any(nm):
        x_lo = max(x_lo, float(np.max((L - c_base[nm]) / a[nm])))
    if np.any(pm):
        x_hi = min(x_hi, float(np.min((L - c_base[pm]) / a[pm])))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_i = np.where(ms == (n - 1))[0]
    c[ms] += delta * a
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = list(range(n))
        if rng is not None and it % 3 == 0:
            rng.shuffle(order)
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def softmax_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * (c - c.max())
    w = np.exp(shifted)
    w /= w.sum()
    a = 1.0 - h
    w_rev = w[::-1]
    T1 = np.correlate(a, w, 'full')
    T2 = np.correlate(h, w_rev, 'full')
    return T1[2 * n - 2 : n - 2 : -1] - T2[n - 1 : -1 : -1]

def subgradient_step(h, n, alpha, lr, max_iters, deadline, rng):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad = softmax_grad(h, n, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        for shrink in range(10):
            h_try = project_feasible(h - lr * grad / gn, n)
            c_try = compute_c5(h_try, n)
            if c_try <= best_c + 0.0005:
                break
            lr *= 0.5
        else:
            break
        if c_try < best_c - 1e-12:
            best_h, best_c = h_try.copy(), c_try
            h = h_try
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.7
    return best_h, best_c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    h1, c1 = exact_cd(h_best, n, 80, deadline, rng)
    if c1 < best_c:
        h_best, best_c = h1, c1

    for alpha in [30, 80, 200, 500, 1500]:
        if time.time() > deadline - 80:
            break
        lr0 = 0.05 / (1.0 + alpha * 0.005)
        h2, c2 = subgradient_step(h_best, n, alpha, lr0, 150, deadline, rng)
        if c2 < best_c - 1e-10:
            h_best, best_c = h2, c2
        if time.time() < deadline - 40:
            h3, c3 = exact_cd(h_best, n, 60, deadline, rng)
            if c3 < best_c - 1e-10:
                h_best, best_c = h3, c3

    h4, c4 = exact_cd(h_best, n, 200, deadline, rng)
    if c4 < best_c - 1e-10:
        h_best, best_c = h4, c4

    for n_try in [600, 800, 900, 500, 1000, 400]:
        if time.time() > deadline - 50:
            break
        if n_try == n:
            continue
        h_int = interp_h(h_best, len(h_best), n_try)
        h5, c5 = exact_cd(h_int, n_try, 80, deadline, rng)
        if c5 < best_c - 1e-9:
            h_best, best_c = h5, c5
        if time.time() > deadline - 20:
            continue
        h6, c6 = subgradient_step(h5, n_try, 300, 0.03, 80, deadline, rng)
        if c6 < best_c - 1e-9:
            h_best, best_c = h6, c6
        h7, c7 = exact_cd(h6, n_try, 60, deadline, rng)
        if c7 < best_c - 1e-9:
            h_best, best_c = h7, c7

    n = len(h_best)
    for r in range(12):
        if time.time() > deadline - 30:
            break
        scale = 0.015 * (1.0 - r / 12.0) + 0.004
        hp = project_feasible(h_best + rng.normal(0, scale, n), n)
        h8, c8 = exact_cd(hp, n, 60, deadline, rng)
        if c8 < best_c - 1e-9:
            h_best, best_c = h8, c8
        if time.time() > deadline - 15:
            break
        h9, c9 = subgradient_step(h8, n, 800, 0.02, 40, deadline, rng)
        if c9 < best_c - 1e-9:
            h_best, best_c = h9, c9

    n = len(h_best)
    h_f, c_f = exact_cd(h_best, n, 300, deadline, rng)
    if c_f < best_c - 1e-12:
        h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
