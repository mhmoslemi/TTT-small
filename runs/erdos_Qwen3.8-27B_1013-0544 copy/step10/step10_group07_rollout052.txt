    a2 = lo + (hi - lo) / phi
    f1 = float(np.max(Q0 + Q1 * a1 + Q2 * a1 ** 2))
    f2 = float(np.max(Q0 + Q1 * a2 + Q2 * a2 ** 2))
    for _ in range(50):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, a2, f2 = a2, a1, f1
            a1 = hi - (hi - lo) / phi
            f1 = float(np.max(Q0 + Q1 * a1 + Q2 * a1 ** 2))
        else:
            lo, a1, f1 = a1, a2, f2
            a2 = lo + (hi - lo) / phi
            f2 = float(np.max(Q0 + Q1 * a2 + Q2 * a2 ** 2))
    a_star = float(np.clip((lo + hi) / 2.0, a_lo, a_hi))
    h[p] = a_star
    h[p2] = s - a_star

def exact_cd_pair(h, n, max_passes, deadline):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(0, n - 1, 2):
            h_old = (h[p], h[p + 1])
            cd_pair(h, n, p, deadline)
            if abs(h[p] - h_old[0]) > 1e-12 or abs(h[p + 1] - h_old[1]) > 1e-12:
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def gradient_polish(h, n, max_iter, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = c.max()
    best_h = h.copy()
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_max = c.max()
        active = np.where(c >= c_max - 0.02 * c_max - 1e-8)[0]
        w = np.exp(c[active] - c_max)
        w /= w.sum()
        grad = np.zeros(n)
        for idx, m in enumerate(active):
            k = m - (n - 1)
            if k == 0:
                grad += w[idx] * (1.0 - 2.0 * h)
            else:
                jf = np.arange(n) + k
                vf = (jf >= 0) & (jf < n)
                jb = np.arange(n) - k
                vb = (jb >= 0) & (jb < n)
                tmp = np.zeros(n)
                if np.any(vf):
                    tmp[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    tmp[vb] -= h[jb[vb]]
                grad += w[idx] * tmp
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        eta = min(0.3 / gnorm, 10.0 * c_max / (gnorm * gnorm + 1e-10))
        h_new = project_feasible(h - eta * grad, n)
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        if c_new.max() < c_max - 1e-12:
            h, c = h_new, c_new
            if c_new.max() < best_c:
                best_c = c_new.max()
                best_h = h.copy()
    return best_h, float(best_c * dx)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        def _cf(z):
            return z.sum() - n / 2.0
        def _cj(z):
            return np.ones(n)
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)
    # Phase 1: SLSQP warmup
    h1, c1 = slsqp_opt(h_best, n, [3000, 10000, 50000], 200, min(deadline, t0 + 120))
    if c1 < best_c:
        h_best, best_c = h1, c1
    # Phase 2: Main 1D CD (bulk of time)
    h2, c2 = exact_cd(h_best, n, 2000, deadline)
    if c2 < best_c:
        h_best, best_c = h2, c2
    # Phase 3: Pair CD
    if time.time() < deadline - 100:
        h3, c3 = exact_cd_pair(h_best, n, 300, deadline)
        if c3 < best_c:
            h_best, best_c = h3, c3
    # Phase 4: Gradient polish
    if time.time() < deadline - 60:
        h4, c4 = gradient_polish(h_best, n, 200, deadline)
        if c4 < best_c:
            h_best, best_c = h4, c4
    # Phase 5: More 1D CD after gradient polish
    if time.time() < deadline - 30:
        h5, c5 = exact_cd(h_best, n, 1000, deadline)
        if c5 < best_c:
            h_best, best_c = h5, c5
    # Phase 6: Shake and settle
    for r in range(15):
        if time.time() > deadline - 40:
            break
        s = 0.003 + 0.003 * r
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        ho, co = exact_cd(hp, n, 500, deadline)
        if time.time() < deadline - 20:
            ho, co2 = exact_cd_pair(ho, n, 50, deadline)
            if co2 < co:
                ho, co = ho, co2
        if co < best_c:
            h_best, best_c = ho, co
    # Phase 7: Multi-n exploration
    for n_try in [500, 840, 1050]:
        if time.time() > deadline - 80:
            break
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        ho, co = exact_cd(h_int, n_try, 600, deadline)
        if time.time() < deadline - 40:
            ho, co2 = exact_cd_pair(ho, n_try, 100, deadline)
            if co2 < co:
                ho, co = ho, co2
        if co < best_c - 1e-7:
            h_best, best_c = ho, co
    # Phase 8: Final polish
    n = len(h_best)
    if time.time() < deadline - 5:
        hf, cf = exact_cd(h_best, n, 500, deadline)
        if cf < best_c:
            h_best, best_c = hf, cf
    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
