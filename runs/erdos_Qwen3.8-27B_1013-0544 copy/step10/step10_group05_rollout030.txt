import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def subgrad_c(h, n, m):
    """Subgradient of c[m] with respect to h."""
    shift = m - (n - 1)
    g = np.zeros(n)
    # Forward: h[i] * (1-h[i+shift]) contributes (1-h[i+shift]) to dh[i]
    valid_f = np.arange(max(0, -shift), min(n, n - shift))
    g[valid_f] += (1.0 - h[valid_f + shift])
    # Backward: h[i+shift] appears as -h[i] * h[i+shift], contributes -h[i] to dh[i+shift]
    valid_b = np.arange(max(0, shift), min(n, n + shift))
    g[valid_b] -= h[valid_b - shift]
    return g * (2.0 / n)

def subgrad_max(h, n):
    """Subgradient of max(c) * (2/n)."""
    c = compute_c(h, n)
    m_max = np.argmax(c)
    return subgrad_c(h, n, m_max)

def bisection_coord(c, h, p, n):

    m_max = min(2 * n - 2, 2 * n - 2 - p)

    jb = p - ms + (n - 1)

def update_c(h, c, delta, p, n):
    """Update correlation c after changing h[p] by delta."""
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c[ms] += delta * a
    diag_i = np.where(ms == (n - 1))[0]
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = compute_c(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = bisection_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                update_c(h, c, delta, p, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = compute_c(h, n)
    return h, compute_c5(h, n)

def momentum_cd(h, n, deadline, lr_init=0.05, mu=0.9, max_iter=500):
    """Subgradient method with Nesterov momentum for the max function."""
    h = h.copy()
    v = np.zeros(n)
    best_h, best_c = h.copy(), compute_c5(h, n)
    lr = lr_init
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Nesterov step
        y = project_feasible(h + mu * v, n)
        g = subgrad_max(y, n)
        v = mu * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new
            # Keep momentum
        else:
            # Reduce learning rate
            lr *= 0.95
            h_new = h  # Don't move
            v *= 0.5
        h = h_new
        if lr < 1e-6:
            break
    return best_h, best_c

def block_cd(h, n, block_size, max_passes, deadline, rng):

    c = compute_c(h, n)

                h_trial = project_feasible(h_trial, n)
                c_trial = compute_c5(h_trial, n)

                    best_block_h = h_trial[block].copy()
                    best_block_c = c_trial
                    improved = True

                improved = False  # reset per-start flag
        if not improved:
            break
        h = project_feasible(h, n)
        c = compute_c(h, n)
    return h, compute_c5(h, n)

def targeted_perturb(h, n, rng, deadline):
    """Perturb coordinates that contribute to bottleneck lags."""
    h = h.copy()
    c = compute_c(h, n)
    m_bottleneck = np.argmax(c)
    shift = m_bottleneck - (n - 1)
    
    # Find coordinates that contribute most to the bottleneck lag
    # c[m] = sum h[i]*(1-h[i+shift]) for valid i
    valid_f = np.arange(max(0, -shift), min(n, n - shift))
    contrib = h[valid_f] * (1.0 - h[valid_f + shift])
    
    # Focus on high-contribution coordinates
    top_k = min(50, len(contrib))
    top_idx = valid_f[np.argsort(contrib)[-top_k:]]
    
    # Perturb these coordinates
    for _ in range(20):
        if time.time() > deadline:
            break
        h_trial = h.copy()
        targets = top_idx[rng.choice(top_k, size=min(10, top_k), replace=False)]
        h_trial[targets] = np.clip(h[targets] + rng.normal(0, 0.1, len(targets)), 0, 1)
        h_trial = project_feasible(h_trial, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial < compute_c5(h, n):
            h = h_trial
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _cons_fun(x):
    return x.sum() - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def slsqp_opt(h0, n, alphas, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def fourier_init(n, K, rng):

    h = np.zeros(n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:

            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

def solve_coord(c, h, p, n):

    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)

    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))

        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):

    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:

    cur_max = float(np.max(c))
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True

        pos_mask = a > 1e-14

                r2 = (a_d + sq) / 2.0

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _cons_fun(x):
    return x.sum() - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def slsqp_opt(h0, n, alphas, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interpolate_h(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: CD from initial
    h1, c1 = exact_cd(h_best, n, 80, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Swap rescue (new technique)
    if time.time() < deadline - 200:
        h2, c2 = swap_rescue(h_best, n, min(deadline - 100, t0 + 200), rng, max_rounds=100)
        if c2 < best_c:
            h_best, best_c = h2, c2
            h2, c2 = exact_cd(h_best, n, 30, deadline)
            if c2 < best_c:
                h_best, best_c = h2, c2

    # Phase 3: Block shift search (new technique)
    if time.time() < deadline - 150:
        h3, c3 = block_shift_search(h_best, n, min(deadline - 80, t0 + 350), rng, max_rounds=50)
        if c3 < best_c:
            h_best, best_c = h3, c3
            h3, c3 = exact_cd(h_best, n, 30, deadline)
            if c3 < best_c:
                h_best, best_c = h3, c3

    # Phase 4: SLSQP
    if time.time() < deadline - 200:
        h4, c4 = slsqp_opt(h_best, n, [2000, 20000, 200000, 2000000], 300, deadline - 50)
        if c4 < best_c:
            h_best, best_c = h4, c4
        h4, c4 = exact_cd(h_best, n, 60, deadline - 20)
        if c4 < best_c:
            h_best, best_c = h4, c4

    # Phase 5: Another swap pass after SLSQP
    if time.time() < deadline - 100:
        h5, c5 = swap_rescue(h_best, n, min(deadline - 50, t0 + 900), rng, max_rounds=60)
        if c5 < best_c:
            h_best, best_c = h5, c5
            h5, c5 = exact_cd(h_best, n, 30, deadline - 10)
            if c5 < best_c:
                h_best, best_c = h5, c5

    # Phase 6: Diverse initializations
    inits = [fourier_init(n, 10, rng), fourier_init(n, 22, rng),
             combinatorial_init(n, rng), combinatorial_init(n, rng)]
    for h_init in inits:
        if time.time() > deadline - 130:
            break
        h_o, c_o = exact_cd(h_init, n, 40, deadline - 40)
        h_o, c_o = swap_rescue(h_o, n, deadline - 20, rng, max_rounds=30)
        if c_o < best_c:
            h_best, best_c = h_o, c_o
            h_o, c_o = exact_cd(h_best, n, 20, deadline - 10)
            if c_o < best_c:
                h_best, best_c = h_o, c_o

    # Phase 7: Different n values
    for n_try in [n + 20, n - 20, n + 50, 600, 800, 650, 750]:
        if time.time() > deadline - 80:
            break
        if n_try <= 200 or n_try > 1200 or n_try == n:
            continue
        h_int = interpolate_h(h_best, len(h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 30, deadline - 30)
        h_o, c_o = swap_rescue(h_o, n_try, deadline - 15, rng, max_rounds=20)
        if c_o < best_c - 1e-7:
            h_best, best_c = h_o, c_o
            h_o, c_o = exact_cd(h_best, len(h_best), 20, deadline - 5)
            if c_o < best_c:
                h_best, best_c = h_o, c_o

    # Phase 8: Perturbation restarts
    for r in range(15):
        if time.time() > deadline - 30:
            break
        temp = 0.004 + 0.008 * r
        hp = project_feasible(h_best + rng.normal(0, temp, len(h_best)), len(h_best))
        h_o, c_o = exact_cd(hp, len(h_best), 25, deadline - 10)
        h_o, c_o = swap_rescue(h_o, len(h_best), deadline - 5, rng, max_rounds=15)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 9: Final polish
    if time.time() < deadline - 3:
        h_f, c_f = exact_cd(h_best, len(h_best), 80, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
