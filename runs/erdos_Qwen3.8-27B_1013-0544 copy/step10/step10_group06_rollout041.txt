def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:
            break

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)

def compute_grad_matrix(h, n):
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        s = k - (n - 1)
        if s == 0:
            G[k, :] = 1.0 - 2.0 * h
        else:
            g = np.ones(n)
            if s > 0:
                g[:n - s] -= h[s:]
                g[s:] -= h[:n - s]
            else:
                sn = -s
                g[sn:] -= h[:n - sn]
                g[:n - sn] -= h[sn:]
            G[k, :] = g
    return G

def solve_lp(h, n, reg_lambda=0.0):

    G = compute_grad_matrix(h, n)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = c - G @ h
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds_lp = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs', options={'maxiter': 10000})
        if res.success:
            h_new = project_feasible(res.x[:n], n)
            return h_new
    except Exception:
        pass
    return None
