Actually, let me simplify. Let me use:
`f(x) = (1/alpha) * log(sum_k exp(alpha * (c_k * dx)))`

This is the log-sum-exp of the scaled correlations. It's smooth and its gradient is:
`df/dx_i = dx * sum_k (exp(alpha * c_k * dx) / sum_j exp(alpha * c_j * dx)) * dc_k/dx_i`

And `dc_k/dx_i` can be computed from the correlation structure.

For the correlation c_k = sum_j x[j] * (1 - x[j + k - (n-1)]) (with valid indices):
dc_k/dx_i = sum over terms where x[i] appears:
- When j = i: d/dx_i [x[i] * (1 - x[i+k-(n-1)])] = (1 - x[i+k-(n-1)])

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):

def softmax_obj_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    vals = alpha * c * dx
    m = vals.max()
    exp_vals = np.exp(vals - m)
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    f = (m + np.log(sum_exp)) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        d = k - (n - 1)
        if d == 0:
            grad += dx * w[k] * (1.0 - 2.0 * x)
        else:
            dcdx = np.zeros(n)
            if d > 0:
                dcdx[:n - d] = 1.0 - x[d:]
                dcdx[d:] -= x[:n - d]
            else:
                dcdx[-d:] = 1.0 - x[:n + d]
                dcdx[:n + d] -= x[-d:]
            grad += dx * w[k] * dcdx
    return f, grad

def slsqp_softmax(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq',
                 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(
                lambda x: softmax_obj_grad(x, n, alpha)[0],
                lambda x: softmax_obj_grad(x, n, alpha)[1],
                best_h, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-16}
            )
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
