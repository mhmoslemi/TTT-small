def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

        else:
            tot = h.sum()
            if tot < 1e-15:

            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

def exact_cd(h, n, max_passes, deadline, gs_iters=80):

    for p in range(max_passes):
