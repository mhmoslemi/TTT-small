    pert = np.zeros(n)

    for k in range(len(c)):
        if not active[k]:
            continue

        # To decrease c[k], we want to increase h where h[i+k] is low and h[i] is high

    best_c = compute_c5(h_best, n)

    # Phase 1: CD polish
    if time.time() < deadline - 300:
        h1, c1 = exact_cd(h_best, n, 300, deadline)
        if c1 < best_c:
            h_best, best_c = h1, c1

    # Phase 2: Pair CD
    if time.time() < deadline - 200:
        h2, c2 = pair_cd(h_best, n, 2000, deadline, rng)
        if c2 < best_c:
            h_best, best_c = h2, c2
        h2b, c2b = exact_cd(h_best, n, 200, deadline)
        if c2b < best_c:
            h_best, best_c = h2b, c2b

    # Phase 3: Multi-start with lag-targeted perturbation
    if time.time() < deadline - 150:
        h3, c3 = multi_start_cd(h_best, n, deadline, rng, max_restarts=6)
        if c3 < best_c:
            h_best, best_c = h3, c3

    # Phase 4: Try different n values
    n_list = [700, 699, 701, 707, 697, 713, 691]
    for n_try in n_list:
        if time.time() > deadline - 100:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        ho, co = exact_cd(h_int, n_try, 200, deadline)
        if co < best_c - 1e-7:
            h_best, best_c, n = ho, co, n_try

    # Phase 5: More aggressive pair CD + CD
    if time.time() < deadline - 60:
        h5, c5 = pair_cd(h_best, n, 3000, deadline, rng)
        h5b, c5b = exact_cd(h5, n, 300, deadline)
        if c5b < best_c:
            h_best, best_c = h5b, c5b

    # Phase 6: Final polish with random-order CD
    if time.time() < deadline - 10:
        h6, c6 = exact_cd(h_best, n, 300, deadline, random_order=True, rng=rng)
        if c6 < best_c:
            h_best, best_c = h6, c6
        h6b, c6b = exact_cd(h_best, n, 200, deadline)
        if c6b < best_c:
            h_best, best_c = h6b, c6b

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
