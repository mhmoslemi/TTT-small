        hp = np.array(initial_h_values, dtype=float).copy()
        if hp.size > 0:
            parent_n = int(hp.size)
            hp = project_feasible(hp)
            c = compute_c5(hp)
            if c < best_c:
                best_h = hp
                best_c = float(c)
                best_n = parent_n
    except Exception:
        pass

    for n in candidate_ns(parent_n):
        if time.time() > deadline:
            break
        try:
            ht = triangle_h(n)
            c = compute_c5(ht)
            if c < best_c:
                best_h = ht
                best_c = float(c)
                best_n = int(len(ht))
        except Exception:
            pass

    if best_h is None:
        best_h = triangle_h(700)
        best_c = compute_c5(best_h)
        best_n = int(len(best_h))

    best_h = project_feasible(best_h)
    best_c = compute_c5(best_h)

    return best_h, float(best_c), int(best_n)
