        x_lo, x_hi = 0.0, 1.0
        ok = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-15):
            ok = False
        if ok and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if ok and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            ok = False
        if ok:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    Cq = L - c_diag + h0 - h0 * h0
    disc = 1.0 - 4.0 * Cq
    if disc >= 0:
        r1 = (1.0 - np.sqrt(disc)) / 2.0
        r2 = (1.0 + np.sqrt(disc)) / 2.0
        if max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2:
            if h0 - r1 < r2 - h0:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    s = n - 1
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = max(0, s - p)
                m_max = min(2 * n - 2, s + p)
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - s
                jb = p - ms + s
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_pos = s - m_min
                a[diag_pos] = 0.0
                c[ms] += delta * a
                c[s] += delta * (1.0 - 2.0 * h0 - delta)
                h[p] = xn
                improved = True
        if not improved:
            break
    return h, compute_c5(h, n)

def anti_periodic_cd(g, m, max_passes, deadline):
    """CD on anti-periodic solution: h = [g, 1-g], length 2m."""
    g = g.copy()
    n = 2 * m
    h = np.empty(n)
    h[:m] = g
    h[m:] = 1.0 - g
    c = np.correlate(h, 1.0 - h, mode='full')
    s = n - 1
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(m):
            gp = g[p]
            if gp < 1e-13 or gp > 1.0 - 1e-13:
                continue
            h_temp = h.copy()
            xn = cd_1d(c, h_temp, p, n)
            if abs(xn - gp) > 1e-12:
                delta = xn - gp
                # Update c for changing h[p]
                m_min = max(0, s - p)
                m_max = min(2 * n - 2, s + p)
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - s
                jb = p - ms + s
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h_temp[jf[vf]])
                if np.any(vb):
                    a[vb] -= h_temp[jb[vb]]
                a[s - m_min] = 0.0
                c[ms] += delta * a
                c[s] += delta * (1.0 - 2.0 * gp - delta)
                h[p] = xn
                g[p] = xn
                improved = True
        if not improved:
            break
    h_out = np.empty(n)
    h_out[:m] = g
    h_out[m:] = 1.0 - g
    return h_out, compute_c5(h_out, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    def _cf(z, n_=n):
        return z.sum() - n_ / 2.0
    def _cj(z, n_=n):
        return np.ones(n_)
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Quick SLSQP warmup
    if time.time() < deadline - 100:
        h1, c1 = slsqp_opt(h_best, n, [5000, 20000, 80000], 200, deadline)
        if c1 < best_c:
            h_best, best_c = h1, c1

    # Phase 2: Aggressive 1D CD
    h2, c2 = exact_cd(h_best, n, 2000, deadline)
    if c2 < best_c:
        h_best, best_c = h2, c2

    # Phase 3: Anti-periodic optimization at same n (if even)
    if n % 2 == 0 and time.time() < deadline - 60:
        m = n // 2
        g_init = h_best[:m].copy()
        h3, c3 = anti_periodic_cd(g_init, m, 1500, deadline)
        if c3 < best_c:
            h_best, best_c = h3, c3
            n = 2 * m

    # Phase 4: Multi-n exploration with focus on smaller n (faster CD)
    n_list = [300, 400, 500, 600, 700, 800]
    for n_try in n_list:
        if time.time() > deadline - 80:
            break
        x_old = np.linspace(0, 2, len(h_best), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        # CD on this n
        h_o, c_o = exact_cd(h_int, n_try, 3000, deadline)
        if c_o < best_c - 1e-8:
            h_best, best_c, n = h_o, c_o, n_try
        # Anti-periodic on this n
        if n_try % 2 == 0 and time.time() < deadline - 50:
            m2 = n_try // 2
            g2 = h_int[:m2].copy()
            h_ap, c_ap = anti_periodic_cd(g2, m2, 1000, deadline)
            if c_ap < best_c - 1e-8:
                h_best, best_c, n = h_ap, c_ap, n_try

    # Phase 5: SLSQP polish on best
    n = len(h_best)
    if time.time() < deadline - 60:
        h5, c5 = slsqp_opt(h_best, n, [50000, 200000], 200, deadline)
        if c5 < best_c:
            h_best, best_c = h5, c5

    # Phase 6: Structured shake
    for r in range(15):
        if time.time() > deadline - 40:
            break
        n = len(h_best)
        # Various perturbation types
        if r % 3 == 0:
            sig = 0.003 + 0.003 * r
            hp = project_feasible(h_best + rng.normal(0, sig, n), n)
        elif r % 3 == 1:
            # Sinusoidal perturbation
            x = np.linspace(0, 4 * np.pi, n)
            amp = 0.01 + 0.002 * r
            hp = project_feasible(h_best + amp * np.sin(x + r * 0.7), n)
        else:
            # Block perturbation
            hp = h_best.copy()
            for _ in range(3):
                idx = rng.randint(0, n - 10)
                hp[idx:idx+10] += rng.normal(0, 0.02)
            hp = project_feasible(hp, n)
        h_o, c_o = exact_cd(hp, n, 500, deadline)
        if c_o < best_c:
            h_best, best_c, n = h_o, c_o, len(h_o)

    # Phase 7: Final CD polish
    if time.time() < deadline - 5:
        h_f, c_f = exact_cd(h_best, len(h_best), 1000, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
