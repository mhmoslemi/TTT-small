                hj_new = hj - d
                if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
                    continue
                h[i] = hi_new
                h[j] = hj_new
                c_new = np.max(np.correlate(h, 1.0 - h, mode='full'))
                if c_new < best_val - 1e-14:
                    best_val = c_new
                    best_d = d
                h[i] = hi
                h[j] = hj
            if best_d is not None:
                h[i] = h[i] + best_d
                h[j] = h[j] - best_d
                improved = True
                T_raw = best_val
        if improved:
            h = project_feasible(h, n)
            cur_c = compute_c5(h, n)
        else:
            break
    return h, compute_c5(h, n)

def slsqp_optimize(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(lse_objective_smooth, best_h,
                           args=(n, alpha, 0.0), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def symmetrize(h, n):
    h_rev = h[::-1].copy()
    h_sym = 0.5 * (h + 1.0 - h_rev)
    return project_feasible(h_sym, n)

def nesterov_subgrad(h, n, steps, lr, deadline):
    h = h.copy()
    v = np.zeros(n)
    y = h.copy()
    momentum = 0.85
    for s in range(steps):
        if time.time() > deadline:
            break
        grad = lse_grad(y, n, 500.0)
        grad -= np.mean(grad)
        momentum = 0.9 + 0.05 * np.sin(s * 0.01)
        v = momentum * v - lr * grad
        y = project_feasible(y + v, n)
        if s % 10 == 0:
            h = y.copy()
    return y

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 700
        x = np.linspace(0, 2, n_init)
        h_init = 0.5 + 0.3 * np.sin(np.pi * x) + 0.1 * np.sin(2 * np.pi * x)
        h_init = np.clip(h_init, 0, 1)
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    h1, c1 = coordinate_descent(best_h, n_init, 30, deadline, rng, 25)
    if c1 < best_c:
        best_h, best_c = h1, c1

    h2, c2 = slsqp_optimize(best_h, n_init, [500, 2000, 5000], 300, deadline)
    if c2 < best_c:
        best_h, best_c = h2, c2

    h3, c3 = coordinate_descent(best_h, n_init, 50, deadline, rng, 25)
    if c3 < best_c:
        best_h, best_c = h3, c1

    h4, c4 = slsqp_optimize(best_h, n_init, [10000, 30000], 400, deadline)
    if c4 < best_c:
        best_h, best_c = h4, c4

    h5, c5 = coordinate_descent(best_h, n_init, 60, deadline, rng, 30)
    if c5 < best_c:
        best_h, best_c = h5, c5

    # Try different grid sizes
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [500, 800, 1000, 600]:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coordinate_descent(h_int, n_try, 30, deadline, rng, 20)
        h_opt, c_opt = slsqp_optimize(h_opt, n_try, [2000, 10000], 250, deadline)
        h_opt, c_opt = coordinate_descent(h_opt, n_try, 30, deadline, rng, 25)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new

    # Symmetrize
    h_sym = symmetrize(best_h, len(best_h))
    c_sym = compute_c5(h_sym, len(best_h))
    if c_sym < best_c:
        best_h, best_c = h_sym, c_sym
        hs, cs = coordinate_descent(best_h, len(best_h), 30, deadline, rng, 25)
        if cs < best_c:
            best_h, best_c = hs, cs

    # Pair local search
    if time.time() < deadline - 60:
        np_, c_np = pair_local_search(best_h, len(best_h), 5, deadline, rng)
        if c_np < best_c:
            best_h, best_c = np_, c_np

    # Nesterov subgradient for diversity
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        h_div = nesterov_subgrad(best_h, n_cur, 100, 0.01, deadline)
        h_div, c_div = coordinate_descent(h_div, n_cur, 30, deadline, rng, 25)
        if c_div < best_c:
            best_h, best_c = h_div, c_div

    # Final polish
    if time.time() < deadline - 20:
        n_cur = len(best_h)
        hf, cf = slsqp_optimize(best_h, n_cur, [50000], 300, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
        remaining = deadline - time.time() - 3
        if remaining > 5:
            max_p = max(10, int(remaining / 0.3))
            hp, cp = coordinate_descent(best_h, n_cur, max_p, deadline, rng, 35)
            if cp < best_c:
                best_h, best_c = hp, cp

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
