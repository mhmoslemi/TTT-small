from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def compute_subgrad(h, n, thresh=0.01):
    """Compute subgradient of max_k c[k] at h."""
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    active = c > cmax - thresh * max(cmax, 1e-10)
    grad = np.zeros(n)
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib = np.zeros(len(ms))
        if np.any(vf):
            contrib[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            contrib[vb] -= h[jb[vb]]
        # Diagonal correction for m = n-1
        diag_i = ms == (n - 1)
        if np.any(diag_i):
            contrib[diag_i] = (1.0 - 2.0 * h[p])
        mask = active[ms]
        if mask.any():
            grad[p] = np.sum(contrib[mask])
    return grad, cmax * (2.0 / n)

def fw_step(h, n, gamma):
    """Frank-Wolfe step: compute subgradient, solve linear minimization, take step."""
    grad, obj = compute_subgrad(h, n)
    # Linear minimization: min grad·y s.t. 0<=y<=1, sum(y)=n/2
    # Solution: y[p]=1 for the n/2 indices with smallest grad[p]
    idx = np.argsort(grad)
    y = np.zeros(n)
    y[idx[:n//2]] = 1.0
    # Handle the case where n is odd (n/2 might not be integer)
    # Actually n/2 should be integer since sum(h)*dx = 1 means sum(h) = n/2
    # So n must be even
    new_h = (1 - gamma) * h + gamma * y
    return new_h, obj

def fw_with_momentum(h, n, max_iters, deadline):
    """Frank-Wolfe with momentum and adaptive step size."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    
    for t in range(max_iters):
        if time.time() > deadline:
            break
        gamma = 2.0 / (t + 2)  # standard FW step size
        grad, obj = compute_subgrad(h_curr, n)
        # Linear minimization
        idx = np.argsort(grad)
        y = np.zeros(n)
        y[idx[:n//2]] = 1.0
        
        # Momentum update
        v = mu * v + (1 - mu) * (y - h_curr)
        h_new = h_curr + gamma * v
        h_new = project_feasible(h_new, n)
        
        h_curr = h_new
        c_curr = compute_c5(h_curr, n)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr.copy()
    
    return best_h, best_c

= 1 - ∫₀² h(x)h(x+k) dx

So minimizing max_k ∫h(x)(1-h(x+k)) dx is equivalent to minimizing max_k (1 - ∫h(x)h(x+k) dx), which is equivalent to maximizing min_k ∫h(x)h(x+k) dx.

So we want to maximize the minimum autocorrelation of h!

This is a nice reformulation. The autocorrelation at lag 0 is ∫h² dx, and we want all other autocorrelations to be as large as possible (relative to the max overlap).

Wait, let me re-derive. The overlap at shift k is:
∫h(x)(1-h(x+k))dx = ∫h(x)dx - ∫h(x)h(x+k)dx = 1 - R(k)

where R(k) = ∫h(x)h(x+k)dx is the autocorrelation.

So C₅ = max_k (1 - R(k)) = 1 - min_k R(k)

We want to minimize C₅, which means maximizing min_k R(k).

But R(0) = ∫h² dx, which is at least (∫h)²/2 = 1/2 by Cauchy-Schwarz (since the domain has measure 2). And R(0) can be as large as 1 (if h is 0-1 valued and ∫h=1, then ∫h² = ∫h = 1).

So we want to maximize the minimum of R(k) over all k ≠ 0 (since R(0) is always the largest).

Actually wait, the problem includes k=0. Let me re-check.

The correlation np.correlate(h, 1-h, mode='full') at index n-1 (the center) is:
Σ_j h[j] * (1-h[j]) = Σ h[j] - Σ h[j]² = n/2 - Σ h[j]²

This corresponds to shift 0. So C₅ includes the "no shift" case.

For shift 0: overlap = ∫h(1-h) dx = 1 - ∫h² dx

For h ∈ [0,1] with ∫h = 1 over a domain of measure 2:
- ∫h² ≥ (∫h)²/∫1 = 1/2 (by Cauchy-Schwarz)
- So the shift-0 overlap = 1 - ∫h² ≤ 1/2

For a binary h with n/2 ones: ∫h² = n/2 * (2/n) = 1, so shift-0 overlap = 0.
For a uniform h = 1/2: ∫h² = 1/4 * 2 = 1/2, so shift-0 overlap = 1/2.

So for binary h, the shift-0 overlap is 0, and we need to worry only about non-zero shifts.
For non-binary h, the shift-0 overlap can be up to 1/2.

This means: if we restrict to binary h, the problem reduces to maximizing min_{k≠0} R(k), where R(k) is the autocorrelation. And C₅ = 1 - min_{k≠0} R(k).

For the continuous case, we need to balance the shift-0 term (which pushes towards binary) against the non-zero shift terms.

This is a really interesting insight! The optimal solution is a tradeoff between:
- Making h binary (to minimize shift-0 overlap)
- Making h have large minimum autocorrelation (to minimize non-zero shift overlap)

Now, for the binary case, the problem is purely combinatorial: find a subset S of {0,...,n-1} of size n/2 that maximizes min_{k≠0} |S ∩ (S+k)| (with periodic boundary, or with the specific boundary of the problem).

Wait, I need to be careful about the boundary. The correlation is computed on the finite array, not periodically. So there's no wrap-around.

For the finite array (no wrap-around), the autocorrelation at lag d is:
R[d] = Σ_j h[j] * h[j+d] for valid j (where both j and j+d are in [0,n))

This is different from the periodic case. The number of valid pairs decreases as |d| increases.

    """Local search for binary h using 2-swaps."""
    h = h_bin.copy().astype(float)
    c_arr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_max = c_arr.max()
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    # Precompute ms
    ms_all = np.arange(2 * n - 1)
    
    evals = 0
    while evals < max_evals and time.time() < deadline:
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        p = ones_list[rng.randint(len(ones_list))]
        q = zeros_list[rng.randint(len(zeros_list))]
        
        # Compute delta_c vectorized
        jf_p = p + ms_all - (n - 1)
        jb_p = p - ms_all + (n - 1)
        jf_q = q + ms_all - (n - 1)
        jb_q = q - ms_all + (n - 1)
        
        # Build lookup: use h with 0 outside [0,n)
        def safe_idx(idx):
            return np.where((idx >= 0) & (idx < n), idx, 0)
        
        hp_f = h[safe_idx(jf_p)] * ((jf_p >= 0) & (jf_p < n)).astype(float)
        hp_b = h[safe_idx(jb_p)] * ((jb_p >= 0) & (jb_p < n)).astype(float)
        hq_f = h[safe_idx(jf_q)] * ((jf_q >= 0) & (jf_q < n)).astype(float)
        hq_b = h[safe_idx(jb_q)] * ((jb_q >= 0) & (jb_q < n)).astype(float)
        
        delta_c = -(1.0 - hp_f) + (1.0 - hq_f) + hp_b - hq_b
        
        new_max = (c_arr + delta_c).max()
        evals += 1
        
        if new_max < best_max - 1e-12:
            # Accept
            h[p] = 0.0
            h[q] = 1.0
            c_arr = c_arr + delta_c
            best_max = new_max
            ones_list.remove(p)
            ones_list.append(q)
            zeros_list.remove(q)
            zeros_list.append(p)
    
    return h, best_max * (2.0 / n)

        hq_b = np.where(mask_q_b, h[np.clip(jb_q, 0, n - 1)], 0.0)
        delta_c = -(1.0 - hp_f) + (1.0 - hq_f) + hp_b - hq_b
        new_max = (c_arr + delta_c).max()
        evals += 1
        if new_max < best_max - 1e-12:
            h[p] = 0.0
            h[q] = 1.0
            c_arr = c_arr + delta_c
            best_max = new_max
            best_h = h.copy()
            ones.remove(p)
            ones.append(q)
            zeros.remove(q)
            zeros.append(p)
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 50000:
            stagnant = 0
    return best_h, best_max * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        n = 700
        h_best = np.zeros(n)
        h_best[rng.choice(n, size=n // 2, replace=False)] = 1.0
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)
    # Phase 1: Binary local search from binarized parent
    if time.time() < deadline - 200:
        h_bin = (h_best > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n)
        h_bs, c_bs = binary_local_search(h_bin, n, deadline - 200, rng, max_evals=300000)
        if c_bs < best_c:
            best_c = c_bs
            h_best = h_bs.copy()
    # Phase 2: CD from best binary/continuous
    if time.time() < deadline - 150:
        h_c = h_best.copy()
        h_c = project_feasible(h_c + 0.05 * rng.randn(n), n)
        h2, c2 = exact_cd(h_c, n, 300, deadline - 150)
        if c2 < best_c:
            h_best, best_c = h2, c2
    # Phase 3: FW + CD cycles
    if time.time() < deadline - 150:
        for cycle in range(8):
            if time.time() > deadline - 100:
                break
            for _ in range(200):
                if time.time() > deadline - 120:
                    break
                gamma = 2.0 / (50 + cycle * 30)
                h_best = fw_step(h_best, n, gamma, 0.05)
                c_tmp = compute_c5(h_best, n)
                if c_tmp < best_c:
                    best_c = c_tmp
            h3, c3 = exact_cd(h_best, n, 150, deadline - 100)
            if c3 < best_c:
                h_best, best_c = h3, c3
    # Phase 4: Multi-n exploration
    n_list = [500, 600, 750, 900, 1050]
    for n_try in n_list:
        if time.time() > deadline - 120:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h4, c4 = exact_cd(h_int, n_try, 200, deadline - 120)
        if c4 < best_c - 1e-9:
            h_best, best_c = h4, c4
            n = n_try
    # Phase 5: Perturbation + CD
    n = len(h_best)
    for r in range(20):
        if time.time() > deadline - 60:
            break
        s = 0.002 + 0.012 * r
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        h5, c5 = exact_cd(hp, n, 150, deadline - 60)
        if c5 < best_c:
            h_best, best_c = h5, c5
    # Phase 6: Binary push + CD
    if time.time() < deadline - 60:
        for _ in range(5):
            if time.time() > deadline - 50:
                break
            hb = h_best.copy()
            hb[hb < 0.5] -= 0.1
            hb[hb >= 0.5] += 0.1
            hb = project_feasible(hb, n)
            h6, c6 = exact_cd(hb, n, 150, deadline - 50)
            if c6 < best_c:
                h_best, best_c = h6, c5 if False else (h6, c6)
    # Phase 7: Final intensive CD
    n = len(h_best)
    if time.time() < deadline - 3:
        h7, c7 = exact_cd(h_best, n, 400, deadline)
        if c7 < best_c:
            h_best, best_c = h7, c7
    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
