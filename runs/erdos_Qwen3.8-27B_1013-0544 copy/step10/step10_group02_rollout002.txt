def project_sum(y,n):
    target=0.5*n
    y=np.asarray(y,dtype=float)
    y=np.clip(y,0.0,1.0)
    s=y.sum()
    if abs(s-target) < 1e-13: return y
    lo=-1.0; hi=1.0
    for _ in range(70):
       mid=0.5*(lo+hi)
       if np.clip(y+mid,0,1).sum() < target: lo=mid
       else: hi=mid
    x=np.clip(y+0.5*(lo+hi),0,1)
    # final tiny correction
    r=target-x.sum()
    if abs(r)>1e-12:
       free=(x>1e-15)&(x<1-1e-15)
       cnt=free.sum()
       if cnt:
          x[free] += r/cnt
          x=np.clip(x,0,1)
    return x

import numpy as np

def project_sum(y, n):
    target = 0.5 * float(n)
    y = np.asarray(y, dtype=float)
    y = np.clip(y, 0.0, 1.0)
    if abs(float(y.sum()) - target) <= 1e-13:
        return y
    lo, hi = -1.0, 1.0
    for _ in range(72):
        mid = 0.5 * (lo + hi)
        if float(np.clip(y + mid, 0.0, 1.0).sum()) < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    r = target - float(x.sum())
    if abs(r) > 1e-13:
        free = (x > 1e-14) & (x < 1.0 - 1e-14)
        cnt = int(free.sum())
        if cnt:
            x[free] += r / float(cnt)
            x = np.clip(x, 0.0, 1.0)
    return x


def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / float(n)))


def lag_gradient(h, m, n):
    g = np.zeros(n, dtype=float)
    e = int(m) - (n - 1)
    if e >= 0:
        if e < n:
            g[:n - e] += 1.0 - h[e:]
            g[e:] -= h[:n - e]
    else:
        ee = -e
        if ee < n:
            g[ee:] += 1.0 - h[:n - ee]
            g[:n - ee] -= h[ee:]
    return g


def top_gradient(h, c, n, K):
    L = 2 * n - 1
    if K <= 0 or L == 0:
        return np.zeros(n, dtype=float)
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(-c, K)[:K]
    g = np.zeros(n, dtype=float)
    cnt = 0
    for m in idx:
        g += lag_gradient(h, int(m), n)
        cnt += 1
    if cnt:
        g /= float(cnt)
    return g


def grad_single(h, p, n):
    total = 2 * n - 1
    a = np.zeros(total, dtype=float)
    start = n - 1 - p
    a[start:start + n] += 1.0 - h
    left = max(0, 2 * p - n + 1)
    right = min(2 * p, n - 1)
    if left <= right:
        a[start + left:start + right + 1] -= h[2 * p - right:2 * p - left + 1][::-1]
    return a

def update_corr(c, h, p, old, new, n):
    delta = new - old
    if abs(delta) < 1e-14:
        return c
    start = n - 1 - p
    a = np.zeros(n, dtype=float)
    a[:] = 1.0 - h
    left = max(0, 2 * p - n + 1)
    right = min(2 * p, n - 1)
    if left <= right:
        a[left:right + 1] -= h[2 * p - right:2 * p - left + 1][::-1]
    a[p] = 0.0
    c[start:start + n] += delta * a
    c[n - 1] += (new - new * new) - (old - old * old)
    return c


def exact_cd(h, n, stop_time, max_passes, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode="full")
    best_c = float(np.max(c)) * (2.0 / float(n))
    best_h = h.copy()

    for it in range(max_passes):
        if time.time() > stop_time:
            break
        order = list(range(n))
        if it % 2 == 1:
            rng.shuffle(order)
        improved = False
        for p in order:
            if time.time() > stop_time:
                break
            old = float(h[p])
            if old <= 1e-13 or old >= 1.0 - 1e-13:
                continue
            nv = solve_1d(c, h, p, n)
            if abs(nv - old) > 1e-12:
                update_corr(c, h, p, old, nv, n)
                h[p] = nv
                improved = True

        h = project_sum(h, n)
        c = np.correlate(h, 1.0 - h, mode="full")
        c5 = float(np.max(c)) * (2.0 / float(n))
        if c5 < best_c - 1e-14:
            best_c = c5
            best_h = h.copy()
        if not improved:
            break

    return best_h, best_c


def gradient_search(h, n, stop_time, rng):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for K in (4, 8, 16, 32):
        if time.time() > stop_time:
            break
        cur = best_h.copy()
        mom = np.zeros(n, dtype=float)
        var = np.zeros(n, dtype=float)
        lr = 0.025
        bad = 0
        while time.time() < stop_time - 2.0:
            c = np.correlate(cur, 1.0 - cur, mode="full")
            cmax = float(np.max(c))
            c5 = cmax * (2.0 / float(n))
            if c5 < best_c - 1e-14:
                best_c = c5
                best_h = cur.copy()

            g = top_gradient(cur, c, n, K)
            if not np.any(g):
                break
            mom = 0.9 * mom + 0.1 * g
            var = 0.999 * var + 0.001 * (g * g)
            step = lr * mom / (np.sqrt(var) + 1e-8)

            trial = project_sum(np.clip(cur - step, 0.0, 1.0), n)
            tc5 = compute_c5(trial, n)
            if tc5 <= c5 + 1e-12:
                cur = trial
                lr = min(0.12, lr * 1.05)
                bad = 0
            else:
                half = project_sum(np.clip(cur - 0.5 * step, 0.0, 1.0), n)
                hc5 = compute_c5(half, n)
                if hc5 <= c5 + 1e-12:
                    cur = half
                else:
                    lr = max(0.002, lr * 0.72)
                    bad += 1
                    if bad > 70:
                        cur = project_sum(cur + rng.normal(0.0, 0.012, n), n)
                        mom.fill(0.0)
                        var.fill(0.0)
                        lr = 0.02
                        bad = 0
                    elif bad > 180:
                        break
    return best_h, best_c


def downsample(h, n_old, n_new):
    if n_old == n_new:
        return project_sum(h, n_new)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return project_sum(np.interp(x_new, x_old, h), n_new)


def perturb_restart(h, n, rng, scale):
    return project_sum(h + rng.normal(0.0, scale, size=n), n)


def binary_restart(h, n, rng):
    b = np.zeros(n, dtype=float)
    order = np.argsort(-h)
    kmax = n // 2
    b[order[:kmax]] = 1.0
    rem = 0.5 * float(n) - float(b.sum())
    if kmax < n and rem > 1e-12:
        b[order[kmax]] = min(1.0, rem)
    return b


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 3.0
    rng = np.random.RandomState(seed)

    try:
        h = np.asarray(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 700
        h = (np.arange(n) % 2).astype(float)

    if n < 2:
        n = 700
        h = (np.arange(n) % 2).astype(float)

    if n > 1000:
        h = downsample(h, n, 700)
        n = 700

    h = project_sum(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    t1 = t0 + min(0.34 * float(budget_s), max(30.0, max(0.0, float(budget_s) - 520.0)))
    if t1 < t0 + 25.0:
        t1 = t0 + 25.0
    t1 = min(t1, deadline - 250.0)
    if t1 > time.time() + 10.0:
        a, b = pair_search(best_h, n, t1, rng)
        if b < best_c:
            best_h, best_c = a, b

    t2 = max(t1 + 25.0, t0 + 0.50 * float(budget_s))
    t2 = min(t2, deadline - 220.0)
    if t2 > time.time() + 10.0:
        a, b = exact_cd(best_h, n, t2, 120, rng)
        if b < best_c:
            best_h, best_c = a, b

    t3 = max(t2 + 20.0, t0 + 0.62 * float(budget_s))
    t3 = min(t3, deadline - 180.0)
    if t3 > time.time() + 10.0:
        a, b = gradient_search(best_h, n, t3, rng)
        if b < best_c:
            best_h, best_c = a, b

    t4 = max(t3 + 45.0, t0 + 0.84 * float(budget_s))
    t4 = min(t4, deadline - 150.0)

    sub = 45.0
    for scale in (0.055, 0.030, 0.014):
        if time.time() > t4 - sub - 10.0:
            break
        hp = perturb_restart(best_h, n, rng, scale)
        stop = min(t4 - sub, time.time() + sub)
        if stop > time.time() + 8.0:
            a, b = pair_search(hp, n, stop, rng)
            if b < best_c:
                best_h, best_c = a, b
                stop2 = min(t4 - 20.0, time.time() + 25.0)
                a2, b2 = exact_cd(best_h, n, stop2, 40, rng)
                if b2 < best_c:
                    best_h, best_c = a2, b2

    if time.time() < t4 - 40.0:
        hb = binary_restart(best_h, n, rng)
        stop = min(t4, time.time() + 45.0)
        if stop > time.time() + 8.0:
            a, b = pair_search(hb, n, stop, rng)
            if b < best_c:
                best_h, best_c = a, b
                stop2 = min(deadline - 120.0, time.time() + 25.0)
                a2, b2 = exact_cd(best_h, n, stop2, 40, rng)
                if b2 < best_c:
                    best_h, best_c = a2, b2

    for nc in (350, 250, 450):
        if time.time() > deadline - 170.0:
            break
        if nc <= 0 or nc >= n:
            continue
        hc = downsample(best_h, n, nc)
        stopc = min(deadline - 160.0, time.time() + 30.0)
        if stopc > time.time() + 8.0:
            hc, cc = pair_search(hc, nc, stopc, rng)
            hu = downsample(hc, nc, n)
            stopu = min(deadline - 140.0, time.time() + 45.0)
            if stopu > time.time() + 8.0:
                hu, cu = pair_search(hu, n, stopu, rng)
                if cu < best_c:
                    best_h, best_c = hu, cu
                    stopd = min(deadline - 120.0, time.time() + 25.0)
                    hd, cd = exact_cd(best_h, n, stopd, 30, rng)
                    if cd < best_c:
                        best_h, best_c = hd, cd

    tfp = deadline - 35.0
    if tfp > time.time() + 8.0:
        a, b = pair_search(best_h, n, tfp, rng)
        if b < best_c:
            best_h, best_c = a, b

    tcd = deadline - 2.0
    if tcd > time.time() + 8.0:
        a, b = exact_cd(best_h, n, tcd, 30, rng)
        if b < best_c:
            best_h, best_c = a, b

    best_h = project_sum(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
