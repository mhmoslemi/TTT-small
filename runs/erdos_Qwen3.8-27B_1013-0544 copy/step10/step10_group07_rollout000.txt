    m=float(c.max())

    C=np.correlate(h,1.0-h,'full'); best=float(C.max()*2.0/n)
    bounds=[(0.0,1.0)]*n

def pair_polish(h,n,stop,rs=None,adj_passes=10,rand_factor=5):
    h=project_feasible(h,n).copy()
    C=np.correlate(h,1.0-h,'full')
    cmax=float(C.max())
    if n < 2: return h,C,cmax*2.0/n
    for p in range(adj_passes):
       if time.time() >= stop: break
       improved=0
       order=np.arange(n-1)
       if rs is not None and p>0: rs.shuffle(order)
       elif p%2==1: order=order[::-1]
       for idx in order:
           if time.time() >= stop: break
           i=int(idx)
           res=optimize_two(h,n,C,cmax,i,i+1)
           if res is not None:
              h,C,cmax=res; improved+=1
       if improved==0: break
       if p % 3 == 2:
           C=np.correlate(h,1.0-h,'full'); cmax=float(C.max())
    if rs is not None:
       limit=max(500, int(rand_factor*n))
       no=0; trials=0
       while time.time() < stop:
          i=rs.randint(n); j=rs.randint(n)
          if i==j: continue
          if i>j: i,j=j,i
          res=optimize_two(h,n,C,cmax,i,j)
          trials += 1
          if res is not None:
             h,C,cmax=res; no=0
          else:
             no += 1
          if trials % 2000 == 0:
             C=np.correlate(h,1.0-h,'full'); cmax=float(C.max())
          if no >= limit:
             break
    C=np.correlate(h,1.0-h,'full')
    h=project_feasible(h,n)
    return h,C,float(C.max()*2.0/n)

import numpy as np
import time
import math
from scipy.optimize import minimize


def project_feasible(y, n):
    y = np.asarray(y, dtype=float).copy()
    t = 0.5 * float(n)
    if (np.all(y >= -1e-14) and np.all(y <= 1.0 + 1e-14)
            and abs(y.sum() - t) < 1e-11):
        return np.clip(y, 0.0, 1.0)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(y - m, 0.0, 1.0)) > t:
            lo = m
        else:
            hi = m
    out = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    d = t - out.sum()
    if d > 0.0:
        w = 1.0 - out
        s = w.sum()
        if s > 0.0:
            out = np.clip(out + d * w / s, 0.0, 1.0)
    elif d < 0.0:
        w = out
        s = w.sum()
        if s > 0.0:
            out = np.clip(out + d * w / s, 0.0, 1.0)
    return out


def eval_h(h, n):
    h = project_feasible(h, n)
    C = np.correlate(h, 1.0 - h, 'full')
    return h, C, float(C.max() * (2.0 / float(n)))


def soft_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, 'full')
    m = float(c.max())
    z = np.clip(alpha * (c - m), -700.0, 0.0)
    return (2.0 / float(n)) * (m + float(np.log(np.sum(np.exp(z)))) / alpha)


def soft_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, 'full')
    m = float(c.max())
    z = np.clip(alpha * (c - m), -700.0, 0.0)
    ez = np.exp(z)
    w = ez / ez.sum()
    g = np.zeros(n)
    ce = n - 1
    w0 = w[ce]
    if w0 > 1e-18:
        g += w0 * (1.0 - 2.0 * x)
    for d in range(1, n):
        wa = w[ce + d]
        wb = w[ce - d]
        if wa > 1e-18:
            g[d:] += wa * (1.0 - x[:-d])
            g[:-d] -= wa * x[d:]
        if wb > 1e-18:
            g[:-d] += wb * (1.0 - x[d:])
            g[d:] -= wb * x[:-d]
    return (2.0 / float(n)) * g


def soft_obj_pen(x, n, alpha, mu):
    return soft_obj(x, n, alpha) + mu * (x.sum() - 0.5 * float(n)) ** 2


def soft_grad_pen(x, n, alpha, mu):
    g = soft_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - 0.5 * float(n))
    return g


def lbfgs_warm(h, n, stop):
    h = project_feasible(h, n).copy()
    C = np.correlate(h, 1.0 - h, 'full')
    best = float(C.max() * (2.0 / float(n)))
    bounds = [(0.0, 1.0)] * n
    for alpha in (150.0, 1000.0):
        if time.time() > stop:
            break
        try:
            res = minimize(
                soft_obj_pen, h, args=(n, alpha, 1.0e5),
                method='L-BFGS-B', jac=soft_grad_pen, bounds=bounds,
                options={'maxiter': 40, 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 20}
            )
        except Exception:
            continue
        h1 = project_feasible(res.x, n)
        C1 = np.correlate(h1, 1.0 - h1, 'full')
        c1 = float(C1.max() * (2.0 / float(n)))
        if c1 < best - 1e-12:
            h = h1
            C = C1
            best = c1
    return h, C, best


def poly_max_scalar(Q0, Q1, Q2, a):
    return float(np.max(Q0 + Q1 * a + Q2 * a * a))


def poly_max_grid(Q0, Q1, Q2, grid):
    aa = np.asarray(grid, dtype=float)
    vals = Q0[None, :] + Q1[None, :] * aa[:, None] + Q2[None, :] * (aa[None, :] ** 2)
    return np.max(vals, axis=1)


def optimize_two(h, n, C, cmax, i, j):
    if i == j:
        return None
    a = float(h[i])
    b = float(h[j])
    s = a + b
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo <= 1e-12:
        return None

    L = 2 * n - 1
    ce = n - 1
    Q0 = C.copy()
    Q1 = np.zeros(L)
    Q2 = np.zeros(L)

    old = a * (1.0 - a)
    Q0[ce] -= old
    Q1[ce] += 1.0
    Q2[ce] -= 1.0

    old = b * (1.0 - b)
    Q0[ce] += s - s * s - old
    Q1[ce] += 2.0 * s - 1.0
    Q2[ce] -= 1.0

    for k in range(n):
        if k == i or k == j:
            continue
        hk = float(h[k])
        om = 1.0 - hk

        idx = ce + i - k
        Q0[idx] -= a * om
        Q1[idx] += om

        idx = ce + j - k
        Q0[idx] += a * om
        Q1[idx] -= om

        idx = ce + k - i
        Q0[idx] += hk * a
        Q1[idx] -= hk

        idx = ce + k - j
        Q0[idx] -= hk * a
        Q1[idx] += hk

    d = j - i
    idx = ce + d
    old = b * (1.0 - a)
    Q0[idx] += s - old
    Q1[idx] += -(s + 1.0)
    Q2[idx] += 1.0

    idx = ce - d
    old = a * (1.0 - b)
    Q0[idx] += -old
    Q1[idx] += 1.0 - s
    Q2[idx] += 1.0

    cur = Q0 + Q1 * a + Q2 * a * a
    curmax = float(cur.max())
    if (not np.isfinite(curmax)) or abs(curmax - cmax) > (1e-5 + 1e-6 * abs(cmax)):
        return None

    grid = np.linspace(lo, hi, 24)
    grid = np.unique(np.concatenate((grid, (a, lo, hi))))
    f = poly_max_grid(Q0, Q1, Q2, grid)
    ib = int(np.argmin(f))
    best_a = float(grid[ib])
    best_f = float(f[ib])

    if 0 < ib < len(grid) - 1 and best_f < cmax:
        glo = float(grid[ib - 1])
        ghi = float(grid[ib + 1])
        phi = 1.618033988749895
        for _ in range(25):
            if ghi - glo < 1e-11:
                break
            a1 = ghi - (ghi - glo) / phi
            a2 = glo + (ghi - glo) / phi
            f1 = poly_max_scalar(Q0, Q1, Q2, a1)
            f2 = poly_max_scalar(Q0, Q1, Q2, a2)
            if f1 < f2:
                ghi = a2
                a2 = a1
                f2 = f1
            else:
                glo = a1
                a1 = a2
                f1 = f2
        a_star = 0.5 * (glo + ghi)
        f_star = poly_max_scalar(Q0, Q1, Q2, a_star)
        if f_star < best_f:
            best_f = f_star
            best_a = a_star

    if best_f >= cmax - 1e-9:
        return None

    best_a = float(np.clip(best_a, lo, hi))
    newC = Q0 + Q1 * best_a + Q2 * best_a * best_a
    newmax = float(newC.max())
    if newmax >= cmax - 1e-9:
        return None

    h[i] = best_a
    h[j] = s - best_a
    return h, newC, newmax


def pair_polish(h, n, stop, rs=None, adj_passes=10, rand_factor=5):
    h = project_feasible(h, n).copy()
    C = np.correlate(h, 1.0 - h, 'full')
    cmax = float(C.max())
    if n < 2:
        return h, C, cmax * (2.0 / float(n))

    for p in range(adj_passes):
        if time.time() >= stop:
            break
        improved = 0
        order = np.arange(n - 1)
        if rs is not None and p > 0:
            rs.shuffle(order)
        elif p % 2 == 1:
            order = order[::-1]
        for idx in order:
            if time.time() >= stop:
                break
            i = int(idx)
            res = optimize_two(h, n, C, cmax, i, i + 1)
            if res is not None:
                h, C, cmax = res
                improved += 1
        if improved == 0:
            break
        if p % 3 == 2:
            C = np.correlate(h, 1.0 - h, 'full')
            cmax = float(C.max())

    if rs is not None:
        limit = max(500, int(rand_factor * n))
        no = 0
        trials = 0
        while time.time() < stop:
            i = rs.randint(n)
            j = rs.randint(n)
            if i == j:
                continue
            if i > j:
                i, j = j, i
            res = optimize_two(h, n, C, cmax, i, j)
            trials += 1
            if res is not None:
                h, C, cmax = res
                no = 0
            else:
                no += 1
            if trials % 2000 == 0:
                C = np.correlate(h, 1.0 - h, 'full')
                cmax = float(C.max())
            if no >= limit:
                break

    C = np.correlate(h, 1.0 - h, 'full')
    h = project_feasible(h, n)
    return h, C, float(C.max() * (2.0 / float(n)))
