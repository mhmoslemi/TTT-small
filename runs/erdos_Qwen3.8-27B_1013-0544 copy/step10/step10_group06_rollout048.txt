            if abs(x_new - h0) > 1e-12:
                h[p] = x_new
                # Update c
                # c changes for indices m where p is in the affected range
                m_lo = max(0, n - 1 - p)
                m_hi = min(2 * n - 2, n - 1 + p)
                # Actually, the affected indices are those where p or p±(shift) is valid
                # Let me just recompute the affected indices
                idx_lo = max(0, p)  # from jb = p - ms + (n-1) >= 0 => ms <= p + n - 1
                # ms = k where c[k] involves h[p]
                # c[k] = sum_n h[n+m](1-h[n]) where m = k-(n-1)
                # h[p] appears if p = n+m (for some valid n) or p = n
                # p = n+m => m = p-n, k = m+n-1 = p-1... this is getting complex
                # Let me just recompute c for the affected range
                # Affected k: those where p is in [0,m-1] as n or in [0,m-1] as n+m
                # p as n: m = k-(n_pts-1), need p+m >= 0 and p+m < n_pts => k in [p, p+n_pts-1] ∩ [0, 2n_pts-2]
                # p as n+m: n = p-m, need n >= 0 and n < n_pts => m <= p and m > p-n_pts => k in [p-(n_pts-1), p] ∩ [0, 2n_pts-2]
                k_lo = max(0, p - (n - 1))
                k_hi = min(2 * n - 2, p + n - 1)
                # Recompute c[k] for k in [k_lo, k_hi]
                for k in range(k_lo, k_hi + 1):
                    m = k - (n - 1)
                    if m >= 0:
                        c[k] = np.sum(h[m:m+n-m] * (1.0 - h[:n-m]))  # h[n+m]*(1-h[n]) for n=0..n-1-m
                    elif m < 0:
                        p_ = -m
                        c[k] = np.sum(h[:n-p_] * (1.0 - h[p_:n]))  # h[n-p_]*(1-h[n]) for n=p_..n-1
                    else:
                        c[k] = np.sum(h * (1.0 - h))
                improved = True
            else:
                # Even if h[p] doesn't change, c doesn't change
                pass
        if not improved:
            break
        h = project_feasible(h, n)
        if _pass % 5 == 4:  # Recompute c periodically for numerical stability
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)
