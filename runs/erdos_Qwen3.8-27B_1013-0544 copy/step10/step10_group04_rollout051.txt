    h = np.full_like(x, 0.5)
    for k in range(len(coeffs) // 2):

        h += coeffs[2*k+1] * np.sin(np.pi * (k+1) * x)
    return np.clip(h, 0.0, 1.0)

    m = c.max()

    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

                x_grid = np.linspace(0, 2, n)

    # Phase 7: Final intensive CD
    if time.time() < deadline - 5:
        h_best, best_c = exact_cd(h_best, len(h_best), 500, deadline)

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
