def _fullc(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5v(c, n):
    return float(np.max(c) * (2.0 / n))

def _proj(h, target):
    h = np.asarray(h, float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    lo, hi = -4.0, 4.0
    for _ in range(220):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-16:
            break
    out = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    fr = 1 - (target - out.sum())
    inl = (out > 1e-9) & (out < 1.0 - 1e-9)
    if inl.any():
        out[inl] = out[inl] * fr
        out = np.clip(out, 0.0, 1.0)
    return out

import time as _tm
import numpy as np


def _fullc(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5v(c, n):
    return float(np.max(c) * (2.0 / n))


def _c5(h):
    return _c5v(_fullc(h), len(h))


def _proj(h, target):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    if abs(h.sum() - target) < 1e-13:
        return h
    lo, hi = -4.0, 4.0
    for _ in range(220):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
        if hi - lo <= 1e-15:
            break
    out = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    return out


def _testshape(h, n):
    if n % 2:
        return False
    m = n // 2
    return bool(np.array_equal(h[:m], h[m:]))


def _resample(hs, ns, nt):
    xs = np.linspace(0.0, 1.0, ns)
    xt = np.linspace(0.0, 1.0, nt)
    return _proj(np.interp(xt, xs, hs), 0.5 * nt)

def _lse_cg(h, al, n):
    c = _fullc(h)
    mx = float(c.max())
    w = np.exp(al * (c - mx))
    sw = float(w.sum())
    S = np.zeros(2 * n)
    S[1:] = np.cumsum(w)
    ia = np.arange(n)
    head = S[n + ia] - S[ia]
    P1 = np.convolve(w, h, 'full')[n - 1:2 * n - 1]
    P2 = np.correlate(h, w, 'full')[n - 1:2 * n - 1]
    g = (head - P1 - P2) / (al * sw)
    return c, g

    tli = 0.0
    list_ok = True
    LM = np.abs(A) <= 1e-14
    for _ in range(46):
        t = 0.5 * (tid + tli)
        d = (B * B - 4.0 * A * (t - C))
        dp = A > 1e-14
        dm = A < -1e-14
        dl = LM
        list_ok = True
        Lf = wl
        Hf = wh
        if np.any(dm & (d < -1e-9)):
            list_ok = False
        if table:
            rA = B[dp]
            rB = d[dp]
            rA = np.sqrt(rB)
            ra = (B[dp] + rA) / (2.0 * A[dp])
            rb = (B[dp] - rA) / (2.0 * A[dp])
            Lf = max(Lf, float(np.max(np.minimum(ra, rb))))
            Hf = min(Hf, float(np.min(np.maximum(ra, rb))))
        if np.any(dp & (d < -1e-9)):
            list_ok = False
        if np.any(dm):
            rA = np.sqrt(np.maximum(d[dm], 0.0))
            ra = (B[dm] + rA) / (2.0 * A[dm])
            rb = (B[dm] - rA) / (2.0 * A[dm])
            ra = np.minimum(ra, rb)
            rb = np.maximum(ra, rb)
            Lf = max(Lf, float(np.max(ra)))
            if np.any(rb < Hf - 1e-12):
                pass
        if np.any(dl):
            bl = B[dl]
            cl = C[dl]
            pos = bl > 1e-15
            neg = bl < -1e-15
            zer = ~pos & ~neg
            if np.any(pos):
                Hf = min(Hf, float(np.min((t - cl[pos]) / bl[pos])))
            if np.any(neg):
                Lf = max(Lf, float(np.max((t - cl[neg]) / bl[neg])))
            if np.any(zer) and float(np.max(cl[zer])) > t:
                list_ok = False
        Lf = max(Lf, wl)
        Hf = min(Hf, wh)
        if list_ok and Lf > Hf:
            list_ok = False
        if list_ok:
            tid = t
        else:
            tli = t
    if not list_ok:
        return None
    return 0.5 * (wl + wh)


def _coord_cd(h, n, target, passes, dl):
    for _ in range(passes):
        if _tm.time() > dl:
            break
        h0 = h.copy()
        c0 = _fullc(h)
        c50 = _c5v(c0, n)
        for p in range(n):
            if _tm.time() > dl:
                break
            hz = h.copy()
            for zp in (h[p], h0[p]):
                pass
            zs = [1.25, h0[p], 0.975]
            besth = _evalcad(np.array(zs, dtype=float), h0, p, c50)
            if besth is not None:
                h[p] = besth
        h = _proj(h, target)
        cnow = _fullc(h)
        if _c5v(cnow, n) < c50 - 1e-13:
            h0 = h.copy()
        else:
            h = h0
    return h, _fullc(h)


def _pair_cd(h, n, fams, dl):
    for (dt, st) in fams:
        for p in range(0, n - dt, st):
            if _tm.time() > dl:
                break
            q = p + dt
            res = _adapt_step(h, n, p, q)
            if res is not None:
                h[p], h[q] = res
    return h, _c5(h)

def _refine(h, n):
