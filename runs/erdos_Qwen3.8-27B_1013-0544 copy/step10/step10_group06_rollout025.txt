import time

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def lse_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        wk = w[k]
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            # forward: (1-h[j+lag]) for valid j
            if lag > 0:
                grad[:n - lag] += wk * (1.0 - h[lag:])
                grad[lag:] -= wk * h[:n - lag]
            else:  # lag < 0
                nl = -lag
                grad[nl:] += wk * (1.0 - h[:n - nl])
                grad[:n - nl] -= wk * h[nl:]
    return lse_value(c, alpha) * dx, grad * dx

    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0

    lo_a = max(0, n - 1 - p)

    lo_b = max(0, p)
    hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(lo_a, lo_b)
    m_hi = max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)

    vb = (jb >= 0) & (jb < n)

        # = c_{n-1} + x - x^2 - h0 + h0^2

        v = beta * v + (1 - beta) * g
        lr = lr0 / (1 + s * 0.01)
        h_new = project_feasible(h - lr * v, n)
        f_new, _ = lse_grad_computed(h_new, n, alpha)
        if f_new <= f + 1e-15:
            h = h_new
        else:
            h = project_feasible(h - lr * 0.3 * v, n)
    return h

def perturb_and_refine(h, n, rng, deadline, scales, cd_passes=25, nbisect=35):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for scale in scales:
        if time.time() > deadline - 30:
            break
        h_p = project_feasible(h + rng.randn(n) * scale, n)
        h_opt, c_opt = coord_descent(h_p, n, cd_passes, deadline, rng, nbisect)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
        if time.time() > deadline - 30:
            break
        h_s = project_symmetric(h + rng.randn(n) * scale, n)
        h_opt2, c_opt2 = coord_descent(h_s, n, cd_passes, deadline, rng, nbisect)
        if c_opt2 < best_c:
            best_h, best_c = h_opt2, c_opt2
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 700
        x = np.linspace(0, 2, n)
        h0 = 0.5 + 0.12 * np.sin(np.pi * x) + 0.04 * np.sin(3 * np.pi * x)
        h0 = np.clip(h0, 0, 1)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)

    # Phase 1: Initial CD
    h1, c1 = coord_descent(h0, n, 25, deadline, rng, 40)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: SLSQP with moderate alpha
    alphas_phase2 = [300, 1000, 3000]
    for alpha in alphas_phase2:
        if time.time() > deadline - 200:
            break
        try:
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
            res = minimize(lse_obj_wrapper, best_h.copy(), args=(n, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 400, 'ftol': 1e-14})
            h2 = project_feasible(res.x, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_h, best_c = h2, c2
        except Exception:
            pass

    # Phase 3: Ordered CD (target top lags first)
    h3, c3 = cd_ordered(best_h, n, 40, deadline, rng, 40)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: Gradient momentum with high alpha
    if time.time() < deadline - 150:
        h4 = grad_momentum(best_h, n, 5000, 150, deadline - 60, lr0=0.02)
        h4, c4 = coord_descent(h4, n, 30, deadline, rng, 40)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: SLSQP high alpha
    for alpha in [10000, 30000]:
        if time.time() > deadline - 120:
            break
        try:
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
            res = minimize(lse_obj_wrapper, best_h.copy(), args=(n, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-14})
            h5 = project_feasible(res.x, n)
            c5v = compute_c5(h5, n)
            if c5v < best_c:
                best_h, best_c = h5, c5v
        except Exception:
            pass

    # Phase 6: More CD
    h6, c6 = coord_descent(best_h, n, 50, deadline, rng, 45)
    if c6 < best_c:
        best_h, best_c = h6, c6

    # Phase 7: Perturbation restarts
    if time.time() < deadline - 100:
        scales = [0.3 / np.sqrt(n), 0.15 / np.sqrt(n), 0.08 / np.sqrt(n)]
        h7, c7 = perturb_and_refine(best_h, n, rng, deadline, scales, 30, 35)
        if c7 < best_c:
            best_h, best_c = h7, c7

    # Phase 8: Try different n
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [600, 800, 900, 500]:
        if time.time() > deadline - 60:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, 40, deadline, rng, 35)
        if time.time() > deadline - 30:
            break
        h_opt2, c_opt2 = cd_ordered(h_opt, n_try, 20, deadline, rng, 40)
        if c_opt2 < best_c:
            best_h, best_c = h_opt2, c_opt2
            x_orig = x_new
        elif c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new

    # Phase 9: Symmetric variant
    if time.time() < deadline - 40:
        n_cur = len(best_h)
        h_sym = project_symmetric(best_h, n_cur)
        h_sym2, c_sym = coord_descent(h_sym, n_cur, 30, deadline, rng, 40)
        if c_sym < best_c:
            best_h, best_c = h_sym2, c_sym

    # Phase 10: Final polish
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        h_final, c_final = coord_descent(best_h, n_cur, 40, deadline, rng, 50)
        if c_final < best_c:
            best_h, best_c = h_final, c_final

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
