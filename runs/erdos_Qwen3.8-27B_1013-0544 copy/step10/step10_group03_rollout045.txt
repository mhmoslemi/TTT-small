9.  **Deriving the Gradient**:
    Let $u_k = 1 - R_k = \sum_i h_i (1 - h_{i+k})$. (Ignoring dx for a moment, it scales the gradient).
    $J(h) = \sum_k w_k u_k$, where $w_k = \alpha e^{\alpha u_k}$ (or similar softmax weights).
    Actually, simpler: $J(h) \approx \max u_k + \frac{1}{\alpha} \ln \sum e^{\alpha u_k}$.
    Gradient $\nabla J = \sum_k P_k \nabla u_k$, where $P_k = \frac{e^{\alpha u_k}}{\sum_j e^{\alpha u_j}}$.
    $u_k = \sum_i h_i - \sum_i h_i h_{i+k} = C - R_k$.
    $\nabla u_k$ with respect to $h_p$:
    $u_k = \sum_i h_i (1 - h_{i+k})$.
    $\frac{\partial u_k}{\partial h_p} = 1 - h_{p+k} - h_{p-k} \dots$ (summing over valid indices).
    Specifically, for a fixed $k$:
    Term 1: $\sum_i h_i$. Derivative is 1.
    Term 2: $-\sum_i h_i h_{i+k}$.
    Derivative w.r.t $h_p$:
    $h_p$ appears in $h_i$ when $i=p$: term is $-h_{p+k}$.
    $h_p$ appears in $h_{i+k}$ when $i+k=p \Rightarrow i=p-k$: term is $-h_{p-k}$.
    So $\frac{\partial u_k}{\partial h_p} = 1 - h_{p+k} - h_{p-k}$.
    (Indices out of bounds are treated as 0 contribution).
    
    This is extremely fast to compute!
    We can do Gradient Descent with this.
    
    *Algorithm*:
    1.  Compute $u_k$ for all $k$.
    2.  Compute probabilities $P_k$.
    3.  Compute gradient $\nabla J_p = \sum_k P_k (1 - h_{p+k} - h_{p-k})$.
    4.  Step: $h \leftarrow h - \eta \nabla J$.
    5.  Project to feasible set.
    
    This avoids the expensive Hessian approximations of SLSQP.

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

# Global state for helper functions
_GLOBAL_N = 700
_GLOBAL_H_COARSE = None

def compute_c5(h, n):
    """Compute the Erdős overlap constant for a given step function h."""
    dx = 2.0 / n
    # Overlap integral for shift k: sum h[i]*(1-h[i+k]) * dx
    # We want the max over k
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    """Project h onto the feasible set: 0<=h<=1 and sum(h)*dx=1 => sum(h)=n/2."""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            # Distribute 'diff' proportional to available room
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            # Reduce h proportional to current value
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def fourier_to_grid(coeffs, n):
    """Reconstruct h from Fourier coefficients on a grid of size n."""
    x = np.linspace(0, 2, n)
    # coeffs: [c0, a1, b1, a2, b2, ...]
    t = np.full(n, coeffs[0])
    num_harm = (len(coeffs) - 1) // 2
    for i in range(num_harm):
        a = coeffs[2 * i + 1]
        b = coeffs[2 * i + 2]
        t += a * np.cos(np.pi * (i + 1) * x) + b * np.sin(np.pi * (i + 1) * x)
    # Sigmoid to map to [0, 1]
    h = 1.0 / (1.0 + np.exp(-t))
    return project_feasible(h, n)

def fourier_obj(coeffs):
    """Objective for Differential Evolution in Fourier space."""
    n = _GLOBAL_N
    h = fourier_to_grid(coeffs, n)
    return compute_c5(h, n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    """SLSQP optimization with smooth max objective."""
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        
        def smooth_obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            # Log-sum-exp for numerical stability
            # We minimize max(c)
            shifted = alpha * (c - m)
            mx = shifted.max()
            lse = mx + np.log(np.sum(np.exp(shifted - mx)))
            return (m + lse / alpha) * (2.0 / n)

        def cons_fun(x):
            return x.sum() - n / 2.0

        def cons_jac(x):
            return np.ones(n)

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        
        try:
            res = minimize(smooth_obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def gradient_descent(h, n, alpha_start, alpha_end, max_iter, deadline):
    """
    Fast gradient descent on the grid using the analytical gradient of the smooth max.
    """
    h = h.copy()
    c_curr = compute_c5(h, n)
    u_vec = np.correlate(h, 1.0 - h, mode='full')
    
    for i in range(max_iter):
        if time.time() > deadline:
            break
        # Interpolate alpha
        frac = i / max_iter
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        # Compute probabilities P_k
        m = u_vec.max()
        # exp(alpha * (u_k - m))
        exp_term = np.exp(alpha * (u_vec - m))
        P_k = exp_term / np.sum(exp_term)
        
        # Compute gradient

        # We can compute the weighted sum of shifts
        
        grad = np.ones(n) # From the sum P_k * 1
        
        # Efficient convolution/correlation for the second terms
        # v_p = sum_k P_k h_{p+k} and w_p = sum_k P_k h_{p-k}
        # P_k is defined for k in [-(n-1), n-1]
        # We can use FFT for speed if n is large, but loop is ok for n=700 if optimized
        # Let's use np.roll for simplicity, it's O(n^2) total but vectorized inner loop
        # Actually, direct matrix multiplication R @ h is O(n^2)
        
        # Construct matrix R where R[p, k_idx] corresponds to P_k * h_{p+k}
        # Too expensive to build matrix. Use loops over significant k?
        # P_k decays fast if alpha is high.
        
        # Simple vectorized approach using correlate
        # v = correlate(P, h)? 
        # P is symmetric if u is symmetric (which it is for real h? No u_k != u_-k generally? 
        # Wait, u_k = sum h_i (1-h_{i+k}). u_-k = sum h_i (1-h_{i-k}) = sum h_{j+k} (1-h_j) = u_k.
        # Yes, u is symmetric. So P_k is symmetric.
        
        # Let's just compute v and w
        v = np.zeros(n)
        w = np.zeros(n)
        
        # Only iterate over k where P_k > threshold
        thresh = 1e-8
        active_indices = np.where(P_k > thresh)[0]
        
        for k_idx in active_indices:
            k = k_idx - (n - 1) # actual shift
            p_val = P_k[k_idx]
            if k >= 0:
                # h[p+k] -> shift left by k? no.

                # w_p = h_{p-k}
                if k < n:
                    v[:n-k] += p_val * h[k:]   # v[0..n-k-1] gets h[k..n-1]
                    w[k:] += p_val * h[:n-k]   # w[k..n-1] gets h[0..n-k-1]
            else:
                kk = -k
                v[kk:] += p_val * h[:n-kk]
                w[:n-kk] += p_val * h[kk:]
                
        grad -= (v + w)
        
        # Gradient step with Armijo line search
        # h_new = h - eta * grad
        # We need to minimize f(h)
        
        best_eta = 0.01
        h_loc = h.copy()
        f_loc = c_curr
        
        # Simple backtracking
        eta = best_eta
        for _ in range(10):
            h_trial = h_loc - eta * grad
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < f_loc:
                h = h_trial
                c_curr = c_trial
                break
            eta *= 0.5
        
        # Update u_vec
        u_vec = np.correlate(h, 1.0 - h, mode='full')

    return h, c_curr
