    c = fft_corr(x, 1.0 - x, n)

    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))

        best_h, best_c = h1, c1
    # Phase 2: SLSQP sweep with FFT gradients
    alphas = [300, 800, 2000, 5000, 12000, 30000]
    for alpha in alphas:
        if time.time() > deadline - 80:
            break
        h2, c2 = slsqp_run(best_h, n, alpha, 400, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h3, c3 = coord_descent(best_h, n, 25, deadline, rng, True, 40)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 3: Multi-restart with perturbations
    for restart in range(5):
        if time.time() > deadline - 100:
            break
        sigma = 0.05 / (restart + 1)
        h_r, c_r = perturb_and_optimize(best_h, n, rng, deadline, sigma, 200, 20)
        if c_r < best_c:
            best_h, best_c = h_r, c_r
    # Phase 4: High-alpha SLSQP + CD polish
    if time.time() < deadline - 60:
        h4, c4 = slsqp_run(best_h, n, 50000, 300, deadline)
        if c4 < best_c:
            best_h, best_c = h4, c4
        h5, c5 = coord_descent(best_h, n, 60, deadline, rng, False, 45)
        if c5 < best_c:
            best_h, best_c = h5, c5
    # Phase 5: Try slightly different n
    for n_try in [640, 750]:
        if time.time() > deadline - 60:
            break
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_t, c_t = slsqp_run(h_int, n_try, 3000, 200, deadline)
        h_t, c_t = coord_descent(h_t, n_try, 30, deadline, rng, True, 35)
        if c_t < best_c:
            best_h, best_c = h_t, c_t
    # Final polish
    if time.time() < deadline - 10:
        h_f, c_f = coord_descent(best_h, len(best_h), 40, deadline, rng, False, 50)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
