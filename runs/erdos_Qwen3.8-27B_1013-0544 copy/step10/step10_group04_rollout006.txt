def lp_descent(h,n,margin,deadline):
    if linprog is None: return None
    c=corr(h); cmax=float(np.max(c))
    active=np.where(c >= cmax - margin)[0]
    if active.size==0: active=np.array([np.argmax(c)])
    mlen=active.size
    G=np.zeros((mlen,n))
    center=n-1
    for r,idx in enumerate(active):
       S=idx-center
       if S>=0:
          mm=n-S
          if mm>0:
             G[r,:mm] += 1.0 - h[S:S+mm]
             G[r,S:S+mm] -= h[:mm]
       else:
          t=-S; mm=n-t
          G[r,t:t+mm] += 1.0 - h[:mm]
          G[r,:mm] -= h[t:t+mm]
    h0=h.copy()
    rhs=G.dot(h0) - c[active]
    A=np.zeros((mlen,n+1))
    A[:,0]=-1.0
    A[:,1:]=G
    Aeq=np.zeros((1,n+1)); Aeq[0,1:]=1
    f=np.zeros(n+1); f[0]=1
    bounds=[(0.0,None)] + [(0.0,1.0)] for _ in range(n)  # cannot use comprehension with _? okay list
    try:
      res=linprog(f, A_ub=A,b_ub=rhs,A_eq=Aeq,b_eq=np.array([n/2.0]),bounds=bounds,method='highs', options={'time_limit': max(1.0, deadline-time.time()-2.0)})
    except Exception: return None
    if res.status==0:
       new_h=project_box_sum(res.x[1:],n)
       new_cmax=float(np.max(corr(new_h)))
       if new_cmax < cmax - 1e-10:
          return new_h,new_cmax

def exact_cd_proj(h, n, max_passes, deadline, random_order=False, rng=None):

    for sc in (0.03, 0.08):
        sub_end = min(pert_end, time.time() + max(20.0, 0.08 * budget_s))
        if time.time() >= sub_end:
            continue
        hp = project_box_sum(best_h + rng.normal(0.0, sc, best_n), best_n)
        h1, m1 = exact_cd_proj(hp, best_n, 35, sub_end, False, rng)
        best_h, best_mx, best_n = better_cand(h1, m1, best_n, best_h, best_mx, best_n)

    bin_end = min(deadline, t0 + 0.90 * budget_s)
    if time.time() < bin_end - 15:
        res = binary_search(best_h, best_n, bin_end, rng)
        if res is not None:
            h1, m1 = res
            best_h, best_mx, best_n = better_cand(h1, m1, best_n, best_h, best_mx, best_n)

    rand_bin_end = min(deadline, t0 + 0.95 * budget_s)
    if best_n % 2 == 0 and time.time() < rand_bin_end - 15:
        hr = np.zeros(best_n, dtype=float)
        hr[rng.permutation(best_n)[: best_n // 2]] = 1.0
        res = binary_search(hr, best_n, rand_bin_end, rng)
        if res is not None:
            h1, m1 = res
            best_h, best_mx, best_n = better_cand(h1, m1, best_n, best_h, best_mx, best_n)

    per_end = min(deadline, t0 + 0.98 * budget_s)
    if time.time() < per_end - 30:
        res = periodic_search(best_h, per_end, rng)
        if res is not None:
            h1, m1, n1 = res
            best_h, best_mx, best_n = better_cand(h1, m1, n1, best_h, best_mx, best_n)

    if time.time() < deadline - 2.0:
        h1, m1 = exact_cd_proj(best_h, best_n, 40, deadline, False, rng)
        best_h, best_mx, best_n = better_cand(h1, m1, best_n, best_h, best_mx, best_n)

    best_h = project_box_sum(best_h, best_n)
    best_mx = float(np.max(corr(best_h)))
    return best_h, best_mx * (2.0 / float(best_n)), best_n
