def nesterov_subgrad(h0, n, lr0, top_k, max_iter, deadline):
    h = h0.copy()
    y = h0.copy()
    h_prev = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(1, max_iter + 1):
        if time.time() > deadline:
            break
        # Nesterov momentum
        alpha = (1 + np.sqrt(1 + 4 * (t-1)**2)) / 2  # or simpler: 2/(t+2)
        beta = (t - 1) / (t + 2)  # momentum coefficient
        y = h + beta * (h - h_prev)
        y = np.clip(y, 0, 1)
        # Subgradient at y
        c = np.correlate(y, 1.0 - y, mode='full')
        idx = np.argpartition(c, -top_k)[-top_k:]
        vals = c[idx]
        w = np.exp(vals - vals.max())
        w /= w.sum()
        grad = np.zeros(n)
        for k_idx in range(top_k):
            s = idx[k_idx] - (n - 1)
            lo1 = max(0, -s)
            hi1 = min(n, n - s)
            if lo1 < hi1:
                grad[lo1:hi1] += w[k_idx] * (1.0 - y[lo1+s:hi1+s])
            lo2 = max(0, s)
            hi2 = min(n, s + n)
            if lo2 < hi2:
                grad[lo2:hi2] -= w[k_idx] * y[lo2-s:hi2-s]
        lr = lr0 / np.sqrt(t)
        h_new = y - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_prev = h
        h = h_new
    return best_h, best_c

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:

            tot = room.sum()
            if tot < 1e-15:

        else:

            if tot < 1e-15:

            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

    """Smoothed exponential bottleneck + penalty."""
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n) + lam * (x.sum() - n / 2.0) ** 2

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _seb_val(x, n, alpha, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n) + lam * (x.sum() - n / 2.0) ** 2

def _seb_grad(x, n, alpha, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for mi in range(len(w)):
        wi = w[mi]
        if wi < 1e-12:
            continue
        s = mi - (n - 1)
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - x[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, s + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * x[lo2 - s:hi2 - s]
    grad *= (2.0 / n)
    grad += 2.0 * lam * (x.sum() - n / 2.0)
    return grad

def lbfgsb_opt(h0, n, alpha, lam, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_seb_val, h0, args=(n, alpha, lam), jac=_seb_grad,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

def solve_coord(c, h, p, n):
    h0 = h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    if m_min > m_max:
        return h0
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            c_diag = c_base[diag_idx[0]]
            a_d = a[diag_idx[0]]
            disc = a_d * a_d + 4.0 * (c_diag - L)

                r2 = (a_d + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:

        else:

                    x_lo = max(x_lo, h0 + r2)

    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):

                m_min = max(0, n - 1 - p)
                m_max = min(2 * n - 2, 2 * n - 2 - p)
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])

                    a[vb] -= h[jb[vb]]

                diag_i = np.where(ms == (n - 1))[0]

                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True

            break

        if abs(h.sum() - n / 2.0) > 1e-10:

def periodic_init(n, p, a):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i % p) < a else 0.0
    return project_feasible(h, n)

def fourier_init(n, K, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    for k in range(1, K + 1):
        amp = rng.normal(0, 1.0 / np.sqrt(k))
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * x / 2.0 + phase)
    scale = np.abs(h).max()
    if scale > 0:
        h = 0.5 + 0.3 * h / scale
    return project_feasible(h, n)

    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
