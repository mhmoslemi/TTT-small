def subgradient_step(h, n, step_size):
    c = np.correlate(h, 1.0 - h, mode='full')
    max_val = c.max()
    active = np.where(c >= max_val - 1e-6)[0]
    grad = np.zeros(n)
    for lag_idx in active:
        lag = lag_idx - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                grad[i] += (1.0 - h[i + lag])
                grad[i + lag] -= h[i]
        else:
            ll = -lag
            for i in range(n - ll):
                grad[i] += (1.0 - h[i - ll + ll])
                grad[i + ll] -= h[i - ll]
    grad = -grad / len(active)
    h_new = h + step_size * grad
    return project_feasible(h_new, n)

    return rng.normal(0, 1.0, n)

def interpolate_h(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700

    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Tight CD to local optimum
    h_best, best_c = coord_descent(h_best, n, 500, deadline)

    # Phase 2: ILS with diverse perturbations
    n_methods = 5
    for it in range(300):
        if time.time() > deadline - 50:
            break
        method = it % n_methods
        perturb = make_perturbation(n, rng, method)
        temp = 0.015 * (1.0 + 0.3 * np.sin(0.05 * it))
        hp = project_feasible(h_best + temp * perturb, n)
        if method == 2 and it % 6 == 2:
            # Subgradient step as an additional perturbation
            hp = subgradient_step(hp, n, 0.003)
        h_try, c_try = coord_descent(hp, n, 250, deadline)
        if c_try < best_c - 1e-15:
            h_best, best_c = h_try, c_try

    # Phase 3: Try different n values
    n_current = len(h_best)
    n_candidates = [n_current - 10, n_current + 10, n_current - 30, n_current + 30,
                    n_current - 50, n_current + 50, 600, 800]
    for n_try in n_candidates:
        if time.time() > deadline - 40:
            break
        if n_try < 200 or n_try > 1200 or n_try == n_current:
            continue
        h_int = interpolate_h(h_best, len(h_best), n_try)
        h_try, c_try = coord_descent(h_int, n_try, 400, deadline)
        if c_try < best_c - 1e-10:
            h_best, best_c = h_try, c_try
            n_current = n_try

    # Phase 4: Second ILS round with smaller steps on best n
    n = len(h_best)
    for it in range(150):
        if time.time() > deadline - 20:
            break
        method = rng.randint(5)
        perturb = make_perturbation(n, rng, method)
        temp = 0.005 + 0.005 * rng.random()
        hp = project_feasible(h_best + temp * perturb, n)
        h_try, c_try = coord_descent(hp, n, 200, deadline)
        if c_try < best_c - 1e-15:
            h_best, best_c = h_try, c_try

    # Phase 5: Final tight CD polish
    for _ in range(5):
        if time.time() > deadline - 2:
            break
        h_best, best_c = coord_descent(h_best, len(h_best), 100, deadline)

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
