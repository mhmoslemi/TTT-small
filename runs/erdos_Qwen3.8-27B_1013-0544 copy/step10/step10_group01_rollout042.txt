import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=40, threshold=1e-5, random_order=False, rng=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        improved = False
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _it in range(gs_iters):
                if d_hi - d_lo < 1e-14:
                    break
                if f1 < f2:
                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1; d1 = d2; f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-13:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h[np.where(h < threshold)[0]] = 0.0
            h[np.where(h > 1.0 - threshold)[0]] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    final_c = compute_c5(h, n)
    return h, min(best_c, final_c)

def lp_objective(h, n, p=100):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.sum(c**p)**(1.0/p) * (2.0/n))

def lp_grad(h, n, p=100):
    c = np.correlate(h, 1.0 - h, mode='full')
    cp = c ** (p-1)
    # Gradient of sum(c^p) w.r.t. h[i]
    # c[m] = sum_k h[k]*(1-h[k-m+n-1])
    # d(c^p)/d h[i] = p*c[m]^(p-1) * dc[m]/d h[i]
    # dc[m]/d h[i] = (1-h[i-m+n-1]) if valid - h[i+m-n+1] if valid
    grad = np.zeros(n)
    for m in range(2*n-1):
        mlo = max(0, m - n + 1)
        mhi = min(n-1, m)
        for i in range(mlo, mhi+1):
            j = i - m + n - 1
            if 0 <= j < n:
                grad[i] += p * cp[m] * (1.0 - h[j])
            k = i + m - n + 1
            if 0 <= k < n:
                grad[i] += p * cp[m] * (-h[k])
    return grad

def subgradient_step(h, n, step_size):
    """One step of projected subgradient descent on max correlation."""
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    # Compute gradient of c[k_star] w.r.t. h
    m = k_star
    mlo = max(0, m - n + 1)
    mhi = min(n - 1, m)
    grad = np.zeros(n)
    j = np.arange(mlo, mhi + 1)
    j_sec = j - m + (n - 1)
    grad[j] += (1.0 - h[j_sec])
    j_rev = j + m - (n - 1)
    valid = (j_rev >= 0) & (j_rev < n)
    grad[j[valid]] -= h[j_rev[valid]]
    # Take step
    h_new = h - step_size * grad
    return project_feasible(h_new, n)

def lse_objective(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(lse_objective, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            if res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0

        if diff > 0:

            tot = room.sum()
            if tot < 1e-14: break

        else:
            tot = h.sum()
            if tot < 1e-14: break

def coord_descent_incr(h, n, max_passes, deadline, gs_iters=30, rng=None, random_order=False):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(np.max(c)) * (2.0 / n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        n_improved = 0
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_rel = (n - 1) - m_lo  # relative index of diagonal in ms
            has_diag = 0 <= diag_rel < len(ms)
            diag_rel = diag_rel if has_diag else -1
            c_aff = c[ms].copy()
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_rel, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_rel, has_diag, unaff_max, h0, d2)
            for _it in range(gs_iters):
                if d_hi - d_lo < 1e-13:
                    break
                if f1 < f2:
                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_rel, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1; d1 = d2; f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_rel, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_rel, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_rel, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-12:
                h[p] = h0 + d_opt
                # Update c incrementally
                c[ms] += d_opt * a
                if has_diag:
                    c[ms[diag_rel]] -= d_opt * d_opt  # quadratic correction
                n_improved += 1
        if n_improved > 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            c_new = float(np.max(c)) * (2.0 / n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    final_c = compute_c5(h, n)
    return h, min(best_c, final_c)
