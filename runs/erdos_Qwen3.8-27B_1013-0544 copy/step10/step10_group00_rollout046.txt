from scipy.optimize import minimize
import time

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()

            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)

            if tot < 1e-15:

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:

        if not improved:

def binary_sa(h, n, deadline, rng, max_iters=5000):
    """Simulated annealing on binary structure with incremental correlation."""
    binary = (h > 0.5).astype(float)
    ones_count = binary.sum()
    # Fix to exactly n//2 ones
    if ones_count > n // 2:
        excess = int(ones_count - n // 2)
        idx = np.where(binary == 1.0)[0]
        worst = idx[np.argsort(h[idx])[:excess]]
        binary[worst] = 0.0
    elif ones_count < n // 2:
        deficit = int(n // 2 - ones_count)
        idx = np.where(binary == 0.0)[0]
        best = idx[np.argsort(-h[idx])[:deficit]]
        binary[best] = 1.0
    
    c = np.correlate(binary, 1.0 - binary, mode='full')
    best_c = float(c.max())
    s_vec = np.arange(2 * n - 1) - (n - 1)
    
    best_binary = binary.copy()
    
    T = 2.0
    cooling = 0.995
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T *= cooling
        if T < 0.01:
            T = 0.01
        
        p = rng.randint(n)
        delta = 1.0 - 2.0 * binary[p]
        
        # Feasibility: maintain n//2 ones
        if binary[p] == 0.0 and binary.sum() >= n // 2:
            continue
        if binary[p] == 1.0 and binary.sum() <= n // 2:
            continue
        
        valid_f = (p + s_vec >= 0) & (p + s_vec < n)
        valid_b = (p - s_vec >= 0) & (p - s_vec < n)
        delta_c = np.zeros(2 * n - 1)
        delta_c[valid_f] += delta * (1.0 - binary[p + s_vec[valid_f]])
        delta_c[valid_b] -= delta * binary[p - s_vec[valid_b]]
        new_c = c + delta_c
        new_max = float(new_c.max())
        old_max = best_c  # Actually should track current max, not best
        
        # Use current max for acceptance
        cur_max = float(c.max())
        delta_obj = new_max - cur_max
        
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            binary[p] += delta
            c = new_c
            if new_max < best_c:

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:

def coord_descent(h, n, max_passes, deadline, gs_iters=50):

            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]

            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break

                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)

            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

    mx = sh.max()
    lse = mx + np.log(np.sum(np.exp(sh - mx)))
