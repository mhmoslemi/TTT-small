import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max() * dx)


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = 0.5 * n
    if abs(h.sum() - target) < 1e-14:
        return h
    lo, hi = float(h.min()), float(h.max())
    for _ in range(70):
        tau = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - tau, 0.0, 1.0)))
        if s > target:
            lo = tau
        else:
            hi = tau
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h


def fix_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    mask = (h > 1e-12) & (h < 1.0 - 1e-12)
    if mask.any():
        h[mask] += diff / float(mask.sum())
        h = np.clip(h, 0.0, 1.0)
    if abs(target - h.sum()) > 1e-10:
        h = project_feasible(h, n)
    return h

        if gg > n // 2:
            gg = n - gg
        if 0 < gg and gg not in seen:
            seen.add(gg)
            gaps.append(gg)
    for f in fracs:
        gg = int(round(n * f)) % n
        if gg > n // 2:
            gg = n - gg
        if 0 < gg and gg not in seen:
            seen.add(gg)
            gaps.append(gg)
    for _ in range(4):
        gg = int(rng.randint(1, n))
        if gg > n // 2:
            gg = n - gg
        if 0 < gg and gg not in seen:
            seen.add(gg)
            gaps.append(gg)
    if not gaps:
        gaps = [1]
    rng.shuffle(gaps)
    return gaps


def make_binary_anti(n, seed):
    n = 2 * (n // 2)
    if n < 8:
        n = 8
    m = n // 2
    rng = np.random.RandomState(seed)
    idx = np.arange(m)
    rng.shuffle(idx)
    a = np.zeros(m)
    a[idx[:m // 2]] = 1.0
    tm = ((np.bitwise_xor(np.arange(m), np.arange(m) >> 1) & 1)).astype(float)
    a = 0.65 * a + 0.35 * tm
    h = np.empty(n)
    h[:m] = a
    h[m:] = 1.0 - a
    return project_feasible(h, n)


def interpolate_h(h, n0, n1):
    if n0 == n1:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, n0)
    x1 = np.linspace(0.0, 2.0, n1)
    return project_feasible(np.interp(x1, x0, h), n1)


def eval_pair_eps(c, A, tmp, diag, idx1, idx2, eps):
    np.add(c, eps * A, out=tmp)
    e2 = eps * eps
    tmp[diag] -= 2.0 * e2
    tmp[idx1] += e2
    tmp[idx2] += e2
    return float(np.max(tmp))


def pair_cd(h, n, stop_time, gaps, seed=0, gs_iters=20, max_stalls=3):
    h = np.clip(h.copy(), 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode="full")
    L = 2 * n - 1
    diag = n - 1
    dx = 2.0 / n
    one_minus = 1.0 - h
    rev = h[::-1]
    tmp = np.empty(L)
    A = np.empty(L)
    cmax = float(c.max())
    cur_gaps = list(gaps)
    stalls = 0
    pass_no = 0
    gr = 0.6180339887498949
    fracs = (0.0, 0.25, 0.5, 0.75, 1.0)
    ev = eval_pair_eps

    while time.time() < stop_time and pass_no < 1000000:
        pass_no += 1
        if stalls == 0 and pass_no % 7 == 0:
            np.random.RandomState(seed + pass_no).shuffle(cur_gaps)
        improved = False
        for gap in cur_gaps:
            g = int(gap) % n
            if g > n // 2:
                g = n - g
            if g == 0:
                continue
            p_end = n // 2 if (n % 2 == 0 and 2 * g == n) else n
            for p in range(p_end):
                if time.time() > stop_time:
                    break
                q = p + g
                if q >= n:
                    q -= n
                hp = h[p]
                hq = h[q]
                lo = -hp
                t = hq - 1.0
                if t > lo:
                    lo = t
                hi = 1.0 - hp
                if hq < hi:
                    hi = hq
                if hi - lo < 1e-13:
                    continue

                A.fill(0.0)
                s = n - 1 - p
                A[s:2 * n - 1 - p] += one_minus
                A[p:n + p] -= rev
                sq = n - 1 - q
                A[sq:2 * n - 1 - q] -= one_minus
                A[q:n + q] += rev

                d = q - p
                idx1 = diag + d
                idx2 = diag - d
                lo0 = lo
                hi0 = hi
                wd = hi0 - lo0
                a = hi0 - gr * wd
                b = lo0 + gr * wd
                fa = ev(c, A, tmp, diag, idx1, idx2, a)
                fb = ev(c, A, tmp, diag, idx1, idx2, b)
                loG = lo0
                hiG = hi0
                for _ in range(gs_iters):
                    if fa < fb:
                        hiG = b
                        b = a
                        fb = fa
                        a = hiG - gr * (hiG - loG)
                        fa = ev(c, A, tmp, diag, idx1, idx2, a)
                    else:
                        loG = a
                        a = b
                        fa = fb
                        b = loG + gr * (hiG - loG)
                        fb = ev(c, A, tmp, diag, idx1, idx2, b)

                dopt = 0.5 * (loG + hiG)
                fopt = ev(c, A, tmp, diag, idx1, idx2, dopt)
                bestf = fopt
                bestd = dopt
                for fr in fracs:
                    dd = lo0 + fr * wd
                    ff = ev(c, A, tmp, diag, idx1, idx2, dd)
                    if ff < bestf:
                        bestf = ff
                        bestd = dd

                if bestf < cmax - 1e-12:
                    if bestd < lo:
                        bestd = lo
                    elif bestd > hi:
                        bestd = hi
                    np.add(c, bestd * A, out=c)
                    e2 = bestd * bestd
                    c[diag] -= 2.0 * e2
                    c[idx1] += e2
                    c[idx2] += e2
                    np = hp + bestd
                    nq = hq - bestd
                    if np < 0.0:
                        np = 0.0
                    elif np > 1.0:
                        np = 1.0
                    if nq < 0.0:
                        nq = 0.0
                    elif nq > 1.0:
                        nq = 1.0
                    h[p] = np
                    h[q] = nq
                    one_minus[p] = 1.0 - np
                    one_minus[q] = 1.0 - nq
                    cmax = bestf
                    improved = True

        if not improved:
            stalls += 1
            if stalls > max_stalls:
                break
            cur_gaps = make_gaps(n, seed + pass_no)
        else:
            c = np.correlate(h, 1.0 - h, mode="full")
            cmax = float(c.max())

    c = np.correlate(h, 1.0 - h, mode="full")
    return h, float(c.max() * dx)


def make_starts(base, n, seed):
    rng = np.random.RandomState(seed)
    out = [base.copy()]
    out.append(project_feasible(base + 0.08 * rng.normal(size=n), n))
    x = np.linspace(0.0, 2.0, n)
    sm = np.zeros(n)
    for f in (1, 2, 3, 5, 8, 13):
        sm += (1.0 / f) * rng.normal(0.0, 1.0) * np.sin(
            2.0 * np.pi * f * x + rng.uniform(0.0, 6.283185307179586)
        )
    out.append(project_feasible(base + 0.28 * sm, n))
    out.append(project_feasible(1.0 - base, n))
    hba = make_binary_anti(n, seed + 7)
    out.append(project_feasible(0.7 * base + 0.3 * hba, n))
    hf = (
        0.5
        + 0.35 * np.sin(2.0 * np.pi * x + rng.uniform(0.0, 6.283185307179586))
        + 0.20 * np.sin(4.0 * np.pi * x + rng.uniform(0.0, 6.283185307179586))
    )
    out.append(project_feasible(hf, n))
    block = max(10, n // 5)
    start = int(rng.randint(0, max(1, n - block)))
    hp = base.copy()
    hp[start:start + block] = 1.0 - hp[start:start + block]
    out.append(project_feasible(hp, n))
    return out


def is_better(c1, n1, c2, n2):
    return c1 < c2 - 1e-12 or (abs(c1 - c2) <= 1e-12 and n1 < n2)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 3.0

    try:
        base = np.asarray(initial_h_values, dtype=float).copy()
        n = int(len(base))
    except Exception:
        n = 600
        base = make_binary_anti(n, seed)

    if n < 50:
        n = 500
        base = np.full(n, 0.5)
    if n % 2 != 0:
        x0 = np.linspace(0.0, 2.0, n)
        x1 = np.linspace(0.0, 2.0, n + 1)
        base = np.interp(x1, x0, base)
        n += 1

    base = project_feasible(base, n)
    best_h = base.copy()
    best_c = compute_c5(best_h, n)
    best_n = n

    if time.time() > deadline - 20.0:
        return fix_feasible(best_h, best_n), best_c, int(best_n)

    gaps = make_gaps(n, seed)
    base_stop = min(deadline - 320.0, t0 + max(120.0, 0.42 * budget_s))
    if time.time() < base_stop:
        h, c = pair_cd(best_h, best_n, base_stop, gaps, seed, gs_iters=22)
        if is_better(c, n, best_c, best_n):
            best_h, best_c, best_n = h, c, n

    if time.time() < deadline - 220.0:
        starts = make_starts(best_h, best_n, seed + 101)
        for idx in range(1, len(starts)):
            if time.time() > deadline - 200.0:
                break
            hp = starts[idx]
            stop = min(deadline - 190.0, time.time() + 95.0)
            if time.time() >= stop:
                break
            h, c = pair_cd(
                hp,
                best_n,
                stop,
                make_gaps(best_n, seed + 137 + idx),
                seed + 200 + idx,
                gs_iters=20,
            )
            if is_better(c, best_n, best_c, best_n):
                best_h, best_c, best_n = h, c, best_n

    for n_try in (500, 350, 800, 600):
        if time.time() > deadline - 170.0:
            break
        if n_try == best_n or n_try < 120:
            continue
        hp = interpolate_h(best_h, best_n, n_try)
        stop = min(deadline - 150.0, time.time() + 105.0)
        if time.time() >= stop:
            break
        h, c = pair_cd(
            hp, n_try, stop, make_gaps(n_try, seed + n_try), seed + n_try, gs_iters=20
        )
        if is_better(c, n_try, best_c, best_n):
            best_h, best_c, best_n = h, c, n_try

        if time.time() < deadline - 130.0:
            hba = make_binary_anti(n_try, seed + n_try + 5)
            hp2 = project_feasible(0.65 * hp + 0.35 * hba, n_try)
            stop2 = min(deadline - 120.0, time.time() + 65.0)
            if time.time() < stop2:
                h, c = pair_cd(
                    hp2,
                    n_try,
                    stop2,
                    make_gaps(n_try, seed + n_try + 11),
                    seed + n_try + 23,
                    gs_iters=20,
                )
                if is_better(c, n_try, best_c, best_n):
                    best_h, best_c, best_n = h, c, n_try

    if time.time() < deadline - 10.0:
        stop = min(deadline - 5.0, time.time() + 130.0)
        if stop > time.time():
            h, c = pair_cd(
                best_h, best_n, stop, make_gaps(best_n, seed + 977), seed + 977, gs_iters=24
            )
            if is_better(c, best_n, best_c, best_n):
                best_h, best_c, best_n = h, c, best_n

    best_h = fix_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
