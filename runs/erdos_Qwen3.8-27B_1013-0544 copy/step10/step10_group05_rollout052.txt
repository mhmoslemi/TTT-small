```python
    grad = np.zeros(n)
    if d >= 0:
        # h[i] * (1-h[i+d]) for i in [0, n-1-d]
        grad[:n-d] += (1 - h[d:])      # forward: d(h[i]*(1-h[i+d]))/dh[i] = 1-h[i+d]

        grad[d_neg:] += (1 - h[:-d_neg])   # forward
        grad[:-d_neg] += (-h[d_neg:])       # backward
    return grad
```

Given current h and c = correlate(h, 1-h), and we want to find the optimal h[p] = x.

The new correlation at lag m (after changing h[p] to x) is:
c_new[m] = c[m] + delta * a[m] - delta^2 * I(m == n-1)

where delta = x - h[p], and a[m] is the sensitivity.

The sensitivity a[m] is computed as:
- For forward direction (jf = p + ms - (n-1)): if jf is valid, a[ms] += (1 - h[jf])
  This is because h[p] appears as h[i] in the term h[i]*(1-h[i+d]), contributing (1-h[i+d]) = (1-h[jf])
- For backward direction (jb = p - ms + (n-1)): if jb is valid, a[ms] -= h[jb]
  This is because h[p] appears as h[i+d] in the term h[i]*(1-h[i+d]), contributing -h[i] = -h[jb]

And the quadratic term -delta^2 only applies for m = n-1 (the "diagonal" lag where d=0).

The subproblem is:
minimize max_m (c[m] + delta*a[m] - delta^2 * I(m==n-1))
subject to 0 ≤ delta + h[p] ≤ 1

Equivalently, find the minimum L such that there exists x ∈ [0,1] with:
c[m] + (x-h[p])*a[m] - (x-h[p])^2 * I(m==n-1) ≤ L for all m

The binary search on L checks feasibility:
- For linear constraints (m ≠ n-1): c[m] + (x-h[p])*a[m] ≤ L → x ≤ (L-c[m])/a[m] if a[m]>0, or x ≥ (L-c[m])/a[m] if a[m]<0
- For the quadratic constraint (m = n-1): c[m] + (x-h[p])*a[m] - (x-h[p])^2 ≤ L → (x-h[p])^2 - a[m]*(x-h[p]) + (L-c[m]) ≥ 0

The quadratic in (x-h[p]): let y = x - h[p]. Then y^2 - a*y + (L-c) ≥ 0.
This is satisfied when y is outside the roots of y^2 - a*y + (L-c) = 0.
Roots: y = (a ± sqrt(a^2 - 4(L-c))) / 2

So the constraint is: y ≤ (a - sqrt(a^2-4(L-c)))/2 OR y ≥ (a + sqrt(a^2-4(L-c)))/2
Which means: x ≤ h[p] + (a-sqrt(...))/2 OR x ≥ h[p] + (a+sqrt(...))/2

In the code, r1 = (a_d - sq)/2 and r2 = (a_d + sq)/2.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _cons_fun(x):
    return x.sum() - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def solve_coord(c, h, p, n):
    h0 = h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            c_diag = c_base[diag_idx[0]]
            a_d = a[diag_idx[0]]
            disc = a_d * a_d + 4.0 * (c_diag - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (a_d - sq) / 2.0
                r2 = (a_d + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a_d = a[diag_idx[0]]
        disc = a_d * a_d + 4.0 * (c_diag - L)
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (a_d - sq) / 2.0
            r2 = (a_d + sq) / 2.0
            if x_lo <= h0 + r1:
                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def _cd_update(c, h, p, xn, n):
    delta = xn - h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c[ms] += delta * a
    diag_i = np.where(ms == (n - 1))[0]
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta
    h[p] = xn

def exact_cd(h, n, max_passes, deadline, order_mode='seq', K=3, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if order_mode == 'seq':
            order = list(range(n))
        elif order_mode == 'rand' and rng is not None:
            order = list(rng.permutation(n))
        else:
            cmax = c.max()
            top_lags = np.where(c > cmax - 0.005)[0]
            crit = set()
            for m in top_lags:
                d = m - (n - 1)
                if d >= 0:
                    crit.update(range(n - d))
                    crit.update(range(d, n))
                else:
                    crit.update(range(-d, n))
                    crit.update(range(-d + d, n))  # same as range(0, n-d) via negative d
                    crit.update(range(n + d, n))
            crit_list = [p for p in range(n) if p in crit]
            other_list = [p for p in range(n) if p not in crit]
            if rng is not None:
                rng.shuffle(other_list)
            order = crit_list + other_list
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                _cd_update(c, h, p, xn, n)
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def slsqp_opt(h0, n, alphas, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interpolate_h(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)

def fourier_init(n, K, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    for k in range(1, K + 1):
        amp = rng.normal(0, 1.0 / np.sqrt(k))
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * x / 2.0 + phase)
    h = 0.5 + 0.4 * h
    return project_feasible(h, n)

            h_best, best_c = h_b, c_b
        elif c_a < best_c:
            h_best, best_c = h_a, c_a

    # Phase 7: Structured multi-start perturbations
    n = len(h_best)
    x = np.linspace(0, 2, n, endpoint=False)
    c_arr = np.correlate(h_best, 1.0 - h_best, mode='full')
    m_star = int(c_arr.argmax())
    d_star = abs(m_star - (n - 1))
    if d_star < 1:
        d_star = 1
    for r in range(8):
        if time.time() > deadline - 60:
            break
        eps = 0.02 + 0.01 * r
        freq = np.pi * n / (2.0 * d_star)
        phase = rng.uniform(0, 2 * np.pi)
        pert = eps * np.cos(freq * x + phase)
        hp = project_feasible(h_best + pert, n)
        h_o, c_o = exact_cd(hp, n, 50, deadline, order_mode='seq')
        h_o, c_o = exact_cd(h_o, n, 40, deadline, order_mode='crit', rng=rng)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 8: Try nearby n values
    n_best = len(h_best)
    for n_try in [n_best - 20, n_best - 10, n_best + 10, n_best + 20]:
        if time.time() > deadline - 80:
            break
        if n_try < 100:
            continue
        h_int = interpolate_h(h_best, n_best, n_try)
        h_o, c_o = exact_cd(h_int, n_try, 60, deadline, order_mode='seq')
        h_o, c_o = exact_cd(h_o, n_try, 40, deadline, order_mode='crit', rng=rng)
        if c_o < best_c - 1e-8:
            h_best, best_c = h_o, c_o

    # Phase 9: Final SLSQP + CD polish
    n = len(h_best)
    if time.time() < deadline - 30:
        h_f, c_f = slsqp_opt(h_best, n, [10000000], 150, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f
    if time.time() < deadline - 3:
        h_g, c_g = exact_cd(h_best, n, 80, deadline, order_mode='seq')
        if c_g < best_c:
            h_best, best_c = h_g, c_g

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
```
