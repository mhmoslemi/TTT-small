    if T < unaff - 1e-11 or T < zero_max - 1e-11:
        return None
    L = d_lo
    U = d_hi
    if bp.size:

def coordinate_descent(h, n, max_passes, deadline, bisect_iters=24, use_random=True, rng=None):

    for _ in range(max_passes):
        if time.time() > deadline: break
        if use_random and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        improved=False
        for p in order:
            if time.time() > deadline: break

            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            af[vf] += 1.0 - h[jf[vf]]
            af[vb] -= h[jb[vb]]
            bf = c[ml:mh + 1]
            b0 = float(bf[q])
            bnz = np.concatenate((bf[:q], bf[q + 1:]))
            anz = np.concatenate((af[:q], af[q + 1:]))
            ua = 0.0
            if ml > 0:
                ua = float(c[:ml].max())
            if mh < fl:
                v = float(c[mh + 1:].max())
                if v > ua:
                    ua = v
            am = float(bf.max())
            fc = am if am > ua else ua
            dlo = -h0
            dhi = 1.0 - h0
            if dhi - dlo > 1e-14:
                do, fo = min1d(dlo, dhi, bnz, anz, b0, 1.0 - 2.0 * h0, ua, fc, it)
            else:
                do, fo = 0.0, fc
            if do != 0.0 and fo < fc - 1e-12:
                nb = bf + af * do
                nb[q] = b0 + do * (1.0 - 2.0 * h0 - do)
                c[ml:mh + 1] = nb
                h[p] = h0 + do
                imp = True
        if not imp:
            break
        h = np.clip(h, 0.0, 1.0)
        c = np.correlate(h, 1.0 - h, mode='full')
        cc = float(np.max(c)) * t
        if cc < bc:
            bc = cc
            bh = h.copy()
    return bh, bc

def res(h, n):
    m = len(h)
    if m == n:
        return h.copy()
    xo = np.linspace(0.0, 2.0, m, endpoint=False)
    xn = np.linspace(0.0, 2.0, n, endpoint=False)
    return proj(np.interp(xn, xo, h), n)

def pert(h, rng, mode, scale):
    n = len(h)
    if mode == 0:
        return proj(h + scale * rng.normal(0.0, 1.0, n), n)
    if mode == 1:
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        f = 1.0 + 4.0 * rng.random()
        ph = 2.0 * np.pi * rng.random()
        return proj(h + scale * (1.0 + rng.random()) * np.sin(2.0 * np.pi * f * x + ph), n)
    if mode == 2:
        bl = max(1, min(n - 1, n // 5))
        st = int(rng.randint(0, max(1, n - bl)))
        g = h.copy()
        g[st:st + bl] = 1.0 - g[st:st + bl]
        return proj(g, n)
    if mode == 3:
        tr = 0.3 + 0.4 * rng.random()
        g = 0.5 * h + 0.5 * ((h > tr).astype(float))
        return proj(g, n)
    small = max(20, n // 4)
    return proj(0.3 * h + 0.7 * res(res(h, small), n), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = float(budget_s)
    hard = t0 + max(1.0, B - 3.0)
    dl = hard - 10.0
    if dl < t0 + 2.0:
        dl = hard
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).ravel().copy()
        n0 = int(h0.size)
    except Exception:
        h0 = np.full(350, 0.5)
        n0 = 350
    if n0 < 8:
        h0 = np.full(100, 0.5)
        n0 = 100
    bh = proj(h0, n0)
    bc = c5val(bh)
    fl = np.flip(bh)
    for cand in (bh, fl, 1.0 - bh, 1.0 - fl, 0.5 * (bh + fl)):
        hh = proj(cand, n0)
        cc = c5val(hh)
        if cc < bc:
            bc = cc
            bh = hh
    if B <= 35.0:
        hh, cc = cd(bh, n0, 20, dl, 18, rng)
        if cc < bc:
            bc = cc
            bh = hh
        bh = proj(bh, n0)
        bc = c5val(bh)
        return bh, bc, n0
    md = t0 + 0.45 * B
    if md > dl - 60.0:
        md = dl - 60.0
    if md < t0 + 5.0:
        md = t0 + 5.0
    hh, cc = cd(bh, n0, 500, md, 26, rng)
    if cc < bc:
        bc = cc
        bh = hh
    nb = n0
    rd = t0 + 0.70 * B
    if rd > dl - 90.0:
        rd = dl - 90.0
    if rd < md + 10.0:
        rd = max(md + 20.0, dl - 30.0)
        if rd > dl - 10.0:
            rd = dl - 10.0
    tns = [200, 300, 450, 600, 750, 900]
    tns = [m for m in tns if abs(m - nb) > 5]
    tns = sorted(set(tns))
    for i, m in enumerate(tns):
        if time.time() > rd - 20.0:
            break
        rem = rd - time.time()
        if rem < 30.0:
            break
        sub = max(25.0, min(120.0, rem / max(1.0, float(len(tns) - i))))
        lm = time.time() + sub
        if lm > rd - 5.0:
            lm = rd - 5.0
        hy = res(bh, m)
        hh, cc = cd(hy, m, 250, lm, 24 if m <= 800 else 20, rng)
        if cc < bc:
            bc = cc
            bh = hh
            nb = m
    pd = dl - 40.0
    if pd < rd + 10.0:
        pd = min(dl - 10.0, rd + 30.0)
    rr = 0
    while time.time() < pd - 15.0:
        rem = pd - time.time()
        if rem < 35.0:
            break
        lm = min(time.time() + 60.0, pd - 5.0)
        mo = rr % 5
        sc = 0.02 + 0.10 * ((rr % 4) + 1.0) / 4.0
        hp = pert(bh, rng, mo, sc)
        hh, cc = cd(hp, nb, 100, lm, 18, rng)
        if cc < bc:
            bc = cc
            bh = hh
        rr += 1
    if time.time() < dl - 3.0:
        hh, cc = cd(bh, nb, 500, dl, 30, rng)
        if cc < bc:
            bc = cc
            bh = hh
    bh = proj(bh, nb)
    bc = c5val(bh)
    return bh, bc, nb
