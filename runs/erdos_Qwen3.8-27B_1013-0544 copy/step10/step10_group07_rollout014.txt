def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    bounds = [(0.0, 1.0)] * n

            if new_c < best_c - 1e-12:

        except Exception:
            pass

        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 600, deadline)
        if c_o < best_c - 1e-7:
            h_best, best_c, n = h_o, c_o, n_try

    # Phase 4: SLSQP on best
    n = len(h_best)
    if time.time() < deadline - 80:
        h5, c5 = slsqp_opt(h_best, n, [50000, 200000], 150, deadline)
        if c5 < best_c:
            h_best, best_c = h5, c5

    # Phase 5: Aggressive annealing shake-and-settle
    n_shakes = 30
    for r in range(n_shakes):
        if time.time() > deadline - 20:
            break
        t_frac = r / float(n_shakes)
        base_noise = 0.02 * (1.0 - t_frac) + 0.002
        strategy = r % 5
        if strategy == 0:
            hp = project_feasible(h_best + rng.normal(0, base_noise, n), n)
        elif strategy == 1:
            hp = h_best.copy()
            n_flip = max(5, n // 40)
            idx = rng.choice(n, n_flip, replace=False)
            hp[idx] = rng.uniform(0, 1, n_flip)
            hp = project_feasible(hp, n)
        elif strategy == 2:
            hb = (h_best > 0.5).astype(float)
            n_flip = max(3, n // 80)
            idx = rng.choice(n, n_flip, replace=False)
            hb[idx] = 1.0 - hb[idx]
            hp = project_feasible(hb, n)
        elif strategy == 3:
            blen = max(5, n // 30)
            start = rng.randint(0, max(1, n - blen))
            hp = h_best.copy()
            hp[start:start + blen] = 1.0 - hp[start:start + blen]
            hp = project_feasible(hp, n)
        else:
            hp = h_best.copy()
            n_flip = max(3, n // 60)
            idx = rng.choice(n, n_flip, replace=False)
            hp[idx] = 1.0 - hp[idx]
            hp = project_feasible(hp, n)
        passes = 400 if t_frac < 0.5 else 200
        h_o, c_o = exact_cd(hp, n, passes, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 6: Final polish
    if time.time() < deadline - 3:
        h_f, c_f = exact_cd(h_best, n, 800, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
