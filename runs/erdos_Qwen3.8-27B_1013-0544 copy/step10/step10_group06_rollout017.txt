def project_feasible(h,n):

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_feasible(h, n):
    target = 0.5 * n
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(120):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff * room / tot, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff * h / tot, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
        if len(idx):
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
        else:
            h = np.clip(h + diff / n, 0.0, 1.0)
    return h


def project_fast(h, n, nit=10):
    target = 0.5 * n
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(nit):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + diff * room / tot)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + diff * h / tot)
    return h


def default_h(n=700):
    x = np.linspace(0.0, 2.0, n)
    h = (0.5
         + 0.32 * np.sin(np.pi * x)
         - 0.06 * np.sin(2.0 * np.pi * x)
         + 0.04 * np.sin(3.0 * np.pi * x)
         - 0.02 * np.sin(4.0 * np.pi * x))
    return project_feasible(np.clip(h, 0.0, 1.0), n)


def is_reversed_c(c, h, n):
    if n <= 1:
        return False
    v0 = float(c[0])
    ours = float(h[n - 1] * (1.0 - h[0]))
    rev = float(h[0] * (1.0 - h[n - 1]))
    if abs(v0 - rev) < abs(v0 - ours) and rev > 0.0:
        return True
    return False


def get_q_ours(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    if n > 1 and is_reversed_c(c, h, n):
        c = c[::-1]
    return c * (2.0 / n)


def h_from_y(y, n):
    target = 0.5 * n
    mu = float(np.mean(y))
    for _ in range(12):
        x = y - mu
        h = expit(x)
        err = float(h.sum() - target)
        if abs(err) < 1e-12:
            return h
        s = h * (1.0 - h)
        d = float(s.sum())
        if d < 1e-12:
            break
        step = err / d
        if step > 20.0:
            step = 20.0
        elif step < -20.0:
            step = -20.0
        mu += step
    h = expit(y - mu)
    if abs(float(h.sum() - target)) > 1e-8:
        lo = float(np.min(y)) - 60.0
        hi = float(np.max(y)) + 60.0
        for _ in range(60):
            mid = 0.5 * (lo + hi)
            fm = float(expit(y - mid).sum() - target)
            if fm > 0:
                lo = mid
            else:
                hi = mid
        h = expit(y - 0.5 * (lo + hi))
    return h


def weighted_grad_h(w, h, n):
    m = 2 * n - 1
    grad = np.zeros(n, dtype=float)
    dx = 2.0 / n
    for idx in range(m):
        wl = float(w[idx])
        if wl <= 1e-15:
            continue
        s = idx - (n - 1)
        if s >= 0:
            L = n - s
            grad[:L] += wl * (1.0 - h[s:s + L])
            grad[s:] -= wl * h[:L]
        else:
            t = -s
            L = n - t
            grad[t:t + L] += wl * (1.0 - h[:L])
            grad[:L] -= wl * h[t:t + L]
    return grad * dx


def y_value(y, n, alpha):
    z = y - float(np.mean(y))
    h = h_from_y(z, n)
    c = np.correlate(h, 1.0 - h, mode="full")
    q = c * (2.0 / n)
    m = float(q.max())
    zlog = alpha * (q - m)
    se = np.exp(zlog)
    return float(m + np.log(float(se.sum())) / alpha + 1e-10 * float(z @ z))


def y_grad(y, n, alpha):
    z = y - float(np.mean(y))
    h = h_from_y(z, n)
    c = np.correlate(h, 1.0 - h, mode="full")
    q = c * (2.0 / n)
    m = float(q.max())
    zlog = alpha * (q - m)
    se = np.exp(zlog)
    w = se / float(se.sum())
    if n > 1 and is_reversed_c(c, h, n):
        w = w[::-1]
    grad_h = weighted_grad_h(w, h, n)
    s = h * (1.0 - h)
    d = float(s.sum())
    if d < 1e-12:
        return 2e-10 * z
    s_grad = float(np.dot(grad_h, s))
    gz = s * (grad_h - s_grad / d)
    gz += 2e-10 * z
    return gz


def optimize_y(h0, n, deadline, alphas, maxiters=35):

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    y0 = np.log(np.clip(h0, 1e-7, 1.0 - 1e-7) / np.clip(1.0 - h0, 1e-7, 1.0))
    y0 -= float(np.mean(y0))
    bounds = [(-40.0, 40.0)] * n
    maxit = max(8, min(maxiters, max(10, 25000 // n)))
    for alpha in alphas:
        if time.time() > deadline - 25:
            break
        try:

                           options={"maxiter": maxit, "maxfun": max(80, 6 * maxit),
                                    "ftol": 1e-12, "gtol": 1e-9, "maxls": 20})
            yk = np.asarray(res.x, dtype=float).copy()
        except Exception:
            break
        if not np.all(np.isfinite(yk)):
            break
        yk -= float(np.mean(yk))
        hi = project_feasible(h_from_y(yk, n), n)
        ci = compute_c5(hi, n)
        if ci < best_c - 1e-15:

        y0 = np.clip(yk, -40.0, 40.0)
        if np.allclose(y0, y0, atol=0.0, rtol=0.0) is False:

    return best_h, best_c

                        r2 = 0.5 * (1.0 + np.sqrt(disc))
                        ok2 = False
                        if r1 >= 0.0:
                            if max(0.0, lo_a) <= min(1.0, hi_a, r1) + 1e-14:
                                ok2 = True
                        if (not ok2) and r2 <= 1.0:
                            if max(0.0, lo_a, r2) <= min(1.0, hi_a) + 1e-14:
                                ok2 = True
                        ok = ok2

                if ok:
                    hiT = T
                else:
                    loT = T

            T = hiT
            lo_a = 0.0
            hi_a = 1.0
            if len(ap):
                tmp = (T - bp) / ap
                hi_a = min(1.0, float(tmp.min()))
            if len(an):
                tmp = (T - bn) / an
                lo_a = max(0.0, float(tmp.max()))
            if lo_a > hi_a + 1e-12:
                continue

            D = (T - b0) / dx
            if D < -1e-12:
                continue

            intervals = []
            if D >= 0.25:
                if lo_a <= hi_a + 1e-14:
                    intervals.append((max(0.0, lo_a), min(1.0, hi_a)))
            else:
                disc = 1.0 - 4.0 * max(0.0, D)
                if disc < 0.0:
                    disc = 0.0
                r1 = 0.5 * (1.0 - np.sqrt(disc))
                r2 = 0.5 * (1.0 + np.sqrt(disc))
                if r1 >= 0.0:
                    a1 = max(0.0, lo_a)
                    b1 = min(1.0, hi_a, r1)
                    if a1 <= b1 + 1e-14:
                        intervals.append((a1, b1))
                if r2 <= 1.0:
                    a2 = max(0.0, lo_a, r2)
                    b2 = min(1.0, hi_a)
                    if a2 <= b2 + 1e-14:
                        intervals.append((a2, b2))
            if not intervals:
                continue

            best_int = None
            best_score = 1e30
            for a1, b1 in intervals:
                mid = 0.5 * (a1 + b1)
                sc = abs(mid - h0)
                if sc < best_score:
                    best_score = sc
                    best_int = (a1, b1)
            if best_int is None:
                continue
            a_new = float(np.clip(0.5 * (best_int[0] + best_int[1]), 0.0, 1.0))
            if abs(a_new - h0) < 1e-13:
                continue

            delta = a_new - h0
            q_diag = q[diag]
            q_new = q + alpha * delta
            q_new[diag] = q_diag + dx * (delta - (a_new * a_new - h0 * h0))
            new_max = float(q_new.max())
            if new_max <= cur_max + 1e-14:
                h[p] = a_new
                q = q_new
                cur_max = new_max

        h_try = project_feasible(h, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-15:
            best_h = h_try
            best_c = c_try
            h = best_h.copy()
            q = get_q_ours(h, n)
            best_q = q.copy()
        else:
            h = best_h.copy()
            q = best_q.copy()

    return best_h, best_c


def interp_h(h, n):
    m = len(h)
    if m == n:
        return h.copy()
    xo = np.linspace(0.0, 2.0, m)
    xn = np.linspace(0.0, 2.0, n)
    return np.interp(xn, xo, h)


def fourier_objective(params, basis, n):
    h = 0.5 + np.asarray(params, dtype=float) @ basis.T
    h = project_fast(h, n, 8)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))


def fourier_from_params(params, basis):
    return 0.5 + np.asarray(params, dtype=float) @ basis.T


def fourier_search(h_best, c_best, n_best, deadline, rng):
    best_h, best_c, best_n = h_best.copy(), c_best, n_best
    sizes = [int(best_n), 600, 430, 280]
    if best_n < 450:
        sizes.append(max(160, int(best_n // 2)))
    seen = set()
    for nf in sizes:
        nf = max(120, int(nf // 2) * 2)
        if nf in seen or time.time() > deadline - 110:
            continue
        seen.add(nf)
        J = 5 if nf >= 350 else 4
        x = np.linspace(0.0, 2.0, nf)
        cols = []
        for j in range(1, J + 1):
            cols.append(np.cos(j * np.pi * x))
            cols.append(np.sin(j * np.pi * x))
        basis = np.vstack(cols)

        fit_h = project_feasible(interp_h(best_h, nf), nf)
        try:
            params, *_ = np.linalg.lstsq(basis, fit_h - 0.5, rcond=None)
            params = np.clip(params, -0.6, 0.6)
        except Exception:
            params = np.zeros(J)

        h_try = project_feasible(fourier_from_params(params, basis), nf)
        c_try = compute_c5(h_try, nf)
        if c_try < best_c:
            best_h, best_c, best_n = h_try, c_try, nf

        if time.time() < deadline - 90:
            try:
                res = minimize(fourier_objective, params.copy(), args=(basis, nf),
                               method="Nelder-Mead",
                               options={"maxiter": 250, "adaptive": True, "fatol": 1e-9})
                h_try = project_feasible(fourier_from_params(res.x, basis), nf)
                c_try = compute_c5(h_try, nf)
                if c_try < best_c:
                    best_h, best_c, best_n = h_try, c_try, nf
            except Exception:
                pass

        if time.time() < deadline - 110:
            try:
                bounds = [(-0.5, 0.5)] * J
                seed = int(rng.randint(1, 2147483646))
                res = differential_evolution(fourier_objective, bounds, args=(basis, nf),
                                             maxiter=28, popsize=12, seed=seed,
                                             tol=1e-7, polish=False)
                h_try = project_feasible(fourier_from_params(res.x, basis), nf)
                c_try = compute_c5(h_try, nf)
                if c_try < best_c:
                    best_h, best_c, best_n = h_try, c_try, nf
            except Exception:
                pass
    return best_h, best_c, best_n


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(5.0, float(budget_s) - 8.0)
    rng = np.random.RandomState(int(seed) % (2 ** 31 - 1))

    try:
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size < 20:
            raise ValueError
    except Exception:
        h0 = default_h(700)

    n0 = int(h0.size)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)

    variants = [
        1.0 - best_h,
        best_h[::-1],
        0.5 * (best_h + best_h[::-1]),
        0.5 * (best_h + 1.0 - best_h[::-1]),
    ]
    for vv in variants:
        hv = project_feasible(vv, n0)
        cv = compute_c5(hv, n0)
        if cv < best_c:
            best_h, best_c = hv, cv

    if time.time() < deadline - 120:
        h1, c1 = coord_descent_q(best_h, n0, 5, deadline, rng, True, 20)
        if c1 < best_c:
            best_h, best_c = h1, c1

    if time.time() < deadline - 220:
        h2, c2 = optimize_y(best_h, n0, deadline, [20.0, 80.0, 300.0, 900.0], 32)
        if c2 < best_c:
            best_h, best_c = h2, c2

    if time.time() < deadline - 220:
        h3, c3, n3 = fourier_search(best_h, best_c, len(best_h), deadline, rng)
        if c3 < best_c:
            best_h, best_c, n0 = h3, c3, n3
        else:
            n0 = len(best_h)

    tried = set()
    for nt in [len(best_h), 620, 450, 300, 200]:
        nt = max(120, int(nt // 2) * 2)
        if nt in tried or time.time() > deadline - 110:
            continue
        tried.add(nt)
        hnt = project_feasible(interp_h(best_h, nt), nt)
        ctry = compute_c5(hnt, nt)
        if ctry < best_c:
            best_h, best_c = hnt, ctry
        if time.time() < deadline - 80:
            hy, cy = optimize_y(hnt, nt, deadline, [60.0, 220.0], 28)
            if cy < best_c:
                best_h, best_c = hy, ctry if ctry < best_c else cy
                if cy < best_c:
                    best_c = cy
        if time.time() < deadline - 35:
            ht, ct = coord_descent_q(best_h, nt, 4, deadline, rng, True, 20)
            if ct < best_c:
                best_h, best_c = ht, ct

    if time.time() < deadline - 60:
        nm = len(best_h)
        hf, cf = optimize_y(best_h, nm, deadline, [250.0, 800.0], 24)
        if cf < best_c:
            best_h, best_c = hf, cf

    if time.time() < deadline - 12:
        nt = len(best_h)
        hp, cp = coord_descent_q(best_h, nt, 100, deadline, rng, False, 24)
        if cp < best_c:
            best_h, best_c = hp, cp

    best_h = project_feasible(best_h, int(len(best_h)))
    best_c = compute_c5(best_h, int(len(best_h)))
    return best_h, float(best_c), int(len(best_h))
