def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

def softmax_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max(c, alpha) * (2.0 / n)

def lbfgs_opt(h, n, alpha, maxiter, deadline):

    try:

        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    except Exception:
        pass
    return best_h, best_c

def block_init(n, rng):
    nb = rng.randint(3, max(4, n // 30))
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = max(1, rng.randint(1, max(2, n // nb)))
        h[pos:min(pos + blen, n)] = rng.choice([0.0, 1.0, 0.3, 0.7])
        pos += blen
    return project_feasible(h, n)

def sine_init(n, rng, nh=5):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(nh):
        f = rng.randint(1, 12)
        ph = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.08, 0.3)
        h += amp * np.sin(2 * np.pi * f * x + ph)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: L-BFGS with increasing sharpness
    for alpha in [3000, 15000, 80000, 300000]:
        if time.time() > deadline - 350:
            break
        h1, c1 = lbfgs_opt(h_best, n, alpha, 250, deadline - 350)
        if c1 < best_c:
            h_best, best_c = h1, c1

    # Phase 2: CD refinement
    if time.time() < deadline - 250:
        h2, c2 = exact_cd(h_best, n, 200, deadline - 250)
        if c2 < best_c:
            h_best, best_c = h2, c2

    # Phase 3: Multi-n with diverse starts
    n_list = [500, 600, 650, 750, 850, 950, 1100, 1300]
    for n_try in n_list:
        if time.time() > deadline - 200:
            break
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        starts = [h_int]
        if time.time() < deadline - 200:
            starts.append(project_feasible(h_int + rng.normal(0, 0.08, n_try), n_try))
        for hs in starts:
            if time.time() > deadline - 190:
                break
            ho, co = lbfgs_opt(hs, n_try, 50000, 150, deadline - 190)
            ho, co = exact_cd(ho, n_try, 100, deadline - 190)
            if co < best_c - 1e-9:
                h_best, best_c = ho, co
                n = n_try

    # Phase 4: Diverse initializations at best n
    n = len(h_best)
    if time.time() < deadline - 150:
        div_starts = []
        for _ in range(4):
            div_starts.append(block_init(n, rng))
        for _ in range(3):
            div_starts.append(sine_init(n, rng))
        for hs in div_starts:
            if time.time() > deadline - 120:
                break
            cs = compute_c5(hs, n)
            if cs > best_c + 0.08:
                continue
            ho, co = lbfgs_opt(hs, n, 50000, 100, deadline - 120)
            ho, co = exact_cd(ho, n, 80, deadline - 120)
            if co < best_c:
                h_best, best_c = ho, co

    # Phase 5: Perturbation + refinement loop
    n = len(h_best)
    for r in range(25):
        if time.time() > deadline - 80:
            break
        s = 0.002 + 0.004 * r
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        ho, co = lbfgs_opt(hp, n, 80000, 100, deadline - 80)
        ho, co = exact_cd(ho, n, 80, deadline - 80)
        if co < best_c:
            h_best, best_c = ho, co

    # Phase 6: Binary push
    if time.time() < deadline - 60:
        for bp in range(6):
            if time.time() > deadline - 50:
                break
            lh = h_best.copy()
            strength = 0.02 + 0.015 * bp
            lh[lh < 0.45] = np.maximum(0.0, lh[lh < 0.45] - strength)
            lh[lh > 0.55] = np.minimum(1.0, lh[lh > 0.55] + strength)
            lh = project_feasible(lh, n)
            ho, co = exact_cd(lh, n, 120, deadline - 50)
            if co < best_c:
                h_best, best_c = ho, co

    # Phase 7: Final intensive CD
    n = len(h_best)
    h_f, c_f = exact_cd(h_best, n, 400, deadline)
    if c_f < best_c:
        h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
