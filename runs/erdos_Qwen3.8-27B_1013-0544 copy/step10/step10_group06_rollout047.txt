def project_feasible(h, n):
    target = n / 2.0

def softmax_obj_and_grad(h, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exps = np.exp(shifted - mx)
    ws = exps / exps.sum()
    obj = (m + (mx + np.log(exps.sum())) / alpha) * dx + penalty * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    w0 = ws[n - 1]
    if w0 > 1e-15:
        grad += w0 * (1.0 - 2.0 * h) * dx
    for L in range(1, n):
        wk = ws[n - 1 + L]
        if wk > 1e-15:
            grad[:n - L] += wk * (1.0 - h[L:]) * dx
            grad[L:] += -wk * h[:n - L] * dx
        wl = ws[n - 1 - L]
        if wl > 1e-15:
            grad[L:] += wl * (1.0 - h[:n - L]) * dx
            grad[:n - L] += -wl * h[L:] * dx
    grad += 2.0 * penalty * (h.sum() - n / 2.0)

            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
            h_int, c_int = lbfgsb_opt(h_int, n_try, 5000, 100.0, 2000)
            h_int, c_int = coord_descent(h_int, n_try, 30, deadline, rng, True, 40)
            if c_int < best_c:
                best_h, best_c = h_int, c_int

    # Phase 7: Final polish CD
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        h_f, c_f = coord_descent(best_h, n_cur, 100, deadline, rng, False, 50)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
