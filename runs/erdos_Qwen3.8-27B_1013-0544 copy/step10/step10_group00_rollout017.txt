import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_correlations(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def coord_descent(h, n, max_passes, deadline):

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False

            jb = p - ms + (n - 1)

                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))

            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)

                    d2 = d_lo + gr * (d_hi - d_lo)

            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def top_k_subgradient(h, n, k, step_size, momentum, beta=0.7):
    c = compute_correlations(h, n)
    # Get top k lags
    top_indices = np.argsort(c)[-k:]
    grad = np.zeros(n)
    for s_idx in top_indices:
        s = s_idx - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j + s < n:
                g += (1.0 - h[j + s])
            if 0 <= j - s < n:
                g -= h[j - s]
            if s == 0:
                g = (1.0 - 2.0 * h[j])
            grad[j] += g / k
    momentum = beta * momentum + (1.0 - beta) * grad
    h_new = project_feasible(h - step_size * momentum, n)
    return h_new, momentum
