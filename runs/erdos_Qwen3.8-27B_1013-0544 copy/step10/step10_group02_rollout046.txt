    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_d = len(di) > 0
    if has_d:
        c_di_val = c_base[di[0]]
        c_base[di] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    L_lo, L_hi = max(unaff, 0.0), cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        fe = True
        zm = np.abs(a) <= 1e-15
        if np.any(zm) and np.any(c_base[zm] > L + 1e-14):
            fe = False
        if fe:
            nm = a < -1e-15
            pm = a > 1e-15
            if np.any(nm):
                x_lo = max(x_lo, float(np.max((L - c_base[nm]) / a[nm])))
            if np.any(pm):
                x_hi = min(x_hi, float(np.min((L - c_base[pm]) / a[pm])))
            if x_lo > x_hi + 1e-13:
                fe = False
        if fe and has_d:
            Cq = L - c_di_val + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    fe = False
        L_hi = L if fe else L_lo
        if fe:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    nm = a < -1e-15
    pm = a > 1e-15
    if np.any(nm):
        x_lo = max(x_lo, float(np.max((L - c_base[nm]) / a[nm])))
    if np.any(pm):
        x_hi = min(x_hi, float(np.min((L - c_base[pm]) / a[pm])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_d:
        Cq = L - c_di_val + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo < r1 and x_hi > r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_optimize(h, n, passes, deadline, rng=None, vf_order=False):
    h = h.copy()
    c = get_c(h, n)
    for it in range(passes):
        if time.time() > deadline:
            break
        imp = 0
        if vf_order and c.max() > 0:
            cm = float(np.max(c))
            scores = np.zeros(n)
            for p in range(n):
                if h[p] < 1e-13 or h[p] > 1.0 - 1e-13:
                    continue
                lo = max(0, n - 1 - p)
                hi = min(2 * n - 2, 2 * n - 2 - p)
                if hi >= lo:
                    scores[p] = cm - c[lo:hi+1].min()
            order = np.argsort(-scores).tolist()
        else:
            order = list(range(n))
            if rng is not None and it % 4 == 3:
                rng.shuffle(order)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                update_c(c, h, p, h0, xn, n)
                h[p] = xn
                imp += 1
        if imp == 0:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = get_c(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return (m + mx + np.log(np.sum(np.exp(sh - mx)))) / alpha * (2.0 / n)

def lfbfgs(h, n, alphas, maxiter, deadline):
    bh, bc = h.copy(), compute_c5(h, n)
    for al in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, bh, args=(n, al), method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < bc - 1e-12:
                bh, bc = nh, nc
        except Exception:
            pass
    return bh, bc

def slsqp(h, n, alphas, maxiter, deadline):
    bh, bc = h.copy(), compute_c5(h, n)
    for al in alphas:
        if time.time() > deadline:
            break
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(smooth_obj, bh, args=(n, al), method='SLSQP',
                           bounds=[(0.0, 1.0)] * n, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < bc - 1e-12:
                bh, bc = nh, nc
        except Exception:
            pass
    return bh, bc

def interp(ho, no, nn):
    xo = np.linspace(0, 2, no, endpoint=False)
    xn = np.linspace(0, 2, nn, endpoint=False)
    return project_feasible(np.interp(xn, xo, ho), nn)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    dl = t0 + budget_s - 3
    try:
        hb = np.array(initial_h_values, dtype=float).copy()
        n = len(hb)
    except Exception:
        hb = np.full(500, 0.5)
        n = 500
    hb = project_feasible(hb, n)
    bc = compute_c5(hb, n)

    # Phase 1: Quick CD cleanup
    h1, c1 = cd_optimize(hb, n, 40, dl, rng)
    if c1 < bc:
        hb, bc = h1, c1

    # Phase 2: Aggressive CD with violation-first
    h2, c2 = cd_optimize(hb, n, 150, dl, rng, vf_order=True)
    if c2 < bc:
        hb, bc = h2, c2

    # Phase 3: L-BFGS smooth
    if time.time() < dl - 300:
        h3, c3 = lfbfgs(hb, n, [8000, 30000, 100000], 250, dl)
        if c3 < bc - 1e-12:
            hb, bc = h3, c3

    # Phase 4: CD after smooth
    h4, c4 = cd_optimize(hb, n, 100, dl, rng, vf_order=True)
    if c4 < bc - 1e-12:
        hb, bc = h4, c4

    # Phase 5: Multi-resolution exploration
    n_list = [550, 750, 850, 450, 650, 950, 500, 400, 1000, 600]
    for nt in n_list:
        if time.time() > dl - 180:
            break
        if nt == len(hb):
            continue
        hi = interp(hb, len(hb), nt)
        ho, co = cd_optimize(hi, nt, 80, dl, rng)
        if co < bc - 1e-9:
            hb, bc = ho, co
            if time.time() < dl - 100:
                h2o, c2o = lfbfgs(ho, nt, [30000, 100000], 150, dl)
                if c2o < bc - 1e-9:
                    hb, bc = h2o, c2o
                h3o, c3o = cd_optimize(hb, len(hb), 80, dl, rng, vf_order=True)
                if c3o < bc - 1e-9:
                    hb, bc = h3o, c3o

    # Phase 6: SLSQP with sum constraint
    if time.time() < dl - 120:
        ns = len(hb)
        hs, cs = slsqp(hb, ns, [30000, 100000], 150, dl)
        if cs < bc - 1e-12:
            hb, bc = hs, cs

    # Phase 7: Stochastic restarts
    n = len(hb)
    for r in range(12):
        if time.time() > dl - 50:
            break
        sc = 0.04 * (1.0 - r / 12.0) + 0.01
        hr = project_feasible(hb + rng.normal(0, sc, n), n)
        ho, co = cd_optimize(hr, n, 60, dl, rng, vf_order=True)
        if co < bc - 1e-9:
            hb, bc = ho, co

    # Phase 8: Final polish
    n = len(hb)
    hf, cf = cd_optimize(hb, n, 250, dl, rng)
    if cf < bc - 1e-12:
        hb, bc = hf, cf

    hb = project_feasible(hb, len(hb))
    bc = compute_c5(hb, len(hb))
    return hb, bc, len(hb)
