    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h - diff / tot * h, 0.0, 1.0)

def solve_coord_quick(c, h, p, n, nbisect=40):
    """Solve one coordinate with bisection on the max constraint."""
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    
    # Compute coefficients for the linear in x_new part
    # c_k(x_new) = c_k_base + a_k * x_new + b_k * x_new^2 (only for k where p is involved)
    # For k = n-1 (lag 0): c_0 contains h[p](1-h[p]) term
    # For other lags: c_k contains linear terms in h[p]
    
    lo_a = max(0, n - 1 - p)
    hi_a = min(2 * n - 2, n - 1 + p)
    lo_b = max(0, p)
    hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(lo_a, lo_b)
    m_hi = max(hi_a, hi_b)
    
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)  # for forward correlation
    jb = p - ms + (n - 1)  # for backward correlation
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    
    c_base = c[ms].copy()
    # The diagonal term (lag 0) contributes h[p](1-h[p]) = h0 - h0^2
    # When we change h[p] to x, this becomes x(1-x) = x - x^2
    # So the change is (x - x^2) - (h0 - h0^2) = (x-h0) - (x^2-h0^2) = (x-h0)(1-x-h0)

    if has_diag:
        c_diag = float(c_base[diag_idx[0]])
        a[diag_idx] = 0.0  # The quadratic term is handled separately
    
    # For non-diagonal terms: c_k(x) = c_base[k] + a[k]*x (linear in x)
    # For diagonal: c_0(x) = c_diag - h0 + h0^2 + x - x^2 = c_diag + (x - x^2) - (h0 - h0^2)
    
    # Unaffected max
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(c[m_hi + 1:])))
    
    cur_max = float(np.max(c))
    L_lo = unaff_max
    L_hi = cur_max
    
    nd = len(ms)
    non_diag_mask = np.ones(nd, dtype=bool)
    if has_diag:
        non_diag_mask[diag_idx] = False
    a_nd = a[non_diag_mask]
    c_nd = c_base[non_diag_mask]
    
    for _ in range(nbisect):
        if L_hi - L_lo < 1e-15:
            break
        T = (L_lo + L_hi) / 2.0
        feasible = True
        
        if unaff_max > T + 1e-15:
            feasible = False
        
        if feasible:
            # Check: exists x in [0,1] such that c_nd[i] + a_nd[i]*x <= T for all i
            # AND (if has_diag) c_diag + (x-x^2) - (h0-h0^2) <= T
            # Non-diag constraints: x <= (T - c_nd[i]) / a_nd[i] if a_nd[i] > 0
            #                      x >= (T - c_nd[i]) / a_nd[i] if a_nd[i] < 0
            x_lo = 0.0
            x_hi = 1.0
            if len(a_nd) > 0:
                neg = a_nd < -1e-13
                pos = a_nd > 1e-13
                if np.any(neg):
                    x_lo = max(x_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                if np.any(pos):
                    x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            x_lo = max(x_lo, 0.0)
            x_hi = min(x_hi, 1.0)
            
            if has_diag:
                # c_diag + x - x^2 - h0 + h0^2 <= T
                # x^2 - x + (T - c_diag + h0 - h0^2) >= 0
                # (x - 0.5)^2 >= 0.25 - (T - c_diag + h0 - h0^2)
                # Let D = T - c_diag + h0 - h0^2
                # Need x^2 - x + D >= 0, i.e., x <= (1-sqrt(1-4D))/2 or x >= (1+sqrt(1-4D))/2
                D = T - c_diag + h0 - h0 * h0
                if 1 - 4 * D >= 0:

                    r2 = (1 + np.sqrt(1 - 4 * D)) / 2
                    # Forbidden: r1 < x < r2 (within [0,1])
                    if x_lo < r1:
                        x_hi = min(x_hi, r1)
                    elif x_hi > r2:
                        x_lo = max(x_lo, r2)
                    else:
                        # x_lo >= r1 and x_hi <= r2, entirely in forbidden zone
                        feasible = False
            
            if x_lo > x_hi + 1e-14:
                feasible = False
        
        if feasible:
            L_hi = T
        else:
            L_lo = T
    
    T = (L_lo + L_hi) / 2.0
    x_lo = 0.0
    x_hi = 1.0
    if len(a_nd) > 0:
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    
    if has_diag:
        D = T - c_diag + h0 - h0 * h0
        if 1 - 4 * D >= 0:
            r1 = (1 - np.sqrt(1 - 4 * D)) / 2
            r2 = (1 + np.sqrt(1 - 4 * D)) / 2
            if x_lo < r1:
                x_hi = min(x_hi, r1 + 1e-9)
            elif x_hi > r2:
                x_lo = max(x_lo, r2 - 1e-9)
            else:
                # Pick the endpoint closer to h0
                if abs(h0 - r1) <= abs(h0 - r2):
                    x_hi = min(x_hi, r1 + 1e-9)
                else:
                    x_lo = max(x_lo, r2 - 1e-9)
    
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng, random_order=True, nbisect=40):

            break
        improved = False

        else:

                break
            x_new = solve_coord_quick(c, h, p, n, nbisect)
            if abs(x_new - h[p]) > 1e-14:

                # Update correlation incrementally... actually just recompute at end
                c = np.correlate(h, 1.0 - h, mode='full')
                improved = True
        if not improved:
            break
    return h, compute_c5(h, n)

            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h - diff / tot * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n, nbisect=50):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo_a = max(0, n - 1 - p)
    hi_a = min(2 * n - 2, n - 1 + p)
    lo_b = max(0, p)
    hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(lo_a, lo_b)
    m_hi = max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = float(c_base[diag_idx[0]])
        a[diag_idx] = 0.0
    non_diag = np.ones(len(ms), dtype=bool)
    if has_diag:
        non_diag[diag_idx] = False
    a_nd = a[non_diag]
    c_nd = c_base[non_diag]
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(c[m_hi + 1:])))
    L_lo = unaff_max
    L_hi = float(np.max(c))
    for _ in range(nbisect):
        if L_hi - L_lo < 1e-15:
            break
        T = (L_lo + L_hi) / 2.0
        feasible = True
        if unaff_max > T + 1e-15:
            feasible = False
        if feasible and len(a_nd) > 0:
            neg = a_nd < -1e-13
            pos = a_nd > 1e-13
            xl = 0.0
            xh = 1.0
            if np.any(neg):
                xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            if xl > xh + 1e-14:
                feasible = False
        if feasible and has_diag:
            D = T - c_diag + h0 - h0 * h0
            if 1 - 4 * D >= 0:
                r1 = (1 - np.sqrt(1 - 4 * D)) / 2
                r2 = (1 + np.sqrt(1 - 4 * D)) / 2
                if len(a_nd) > 0:
                    neg = a_nd < -1e-13
                    pos = a_nd > 1e-13
                    xl = 0.0
                    xh = 1.0
                    if np.any(neg):
                        xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                    xl = max(xl, 0.0)
                    xh = min(xh, 1.0)
                    if xl > xh + 1e-14:
                        feasible = False
                    elif xl >= r1 and xh <= r2:
                        feasible = False
                else:
                    if r1 <= 0 and r2 >= 1:
                        feasible = False
        if feasible:
            L_hi = T
        else:
            L_lo = T
    T = (L_lo + L_hi) / 2.0
    xl = 0.0
    xh = 1.0
    if len(a_nd) > 0:
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if np.any(neg):
            xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    xl = max(xl, 0.0)
    xh = min(xh, 1.0)
    if has_diag:
        D = T - c_diag + h0 - h0 * h0
        if 1 - 4 * D >= 0:
            r1 = (1 - np.sqrt(1 - 4 * D)) / 2
            r2 = (1 + np.sqrt(1 - 4 * D)) / 2
            if xl < r1:
                xh = min(xh, r1 + 1e-10)
            elif xh > r2:
                xl = max(xl, r2 - 1e-10)
            else:
                if abs(h0 - r1) <= abs(h0 - r2):
                    xh = min(xh, r1 + 1e-10)
                else:
                    xl = max(xl, r2 - 1e-10)
    if xl > xh + 1e-10:
        return h0
    return float(np.clip((xl + xh) / 2.0, 0.0, 1.0))

def coord_descend(h, n, max_passes, deadline, rng, nbisect=50):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            x_new = solve_coord(c, h, p, n, nbisect)
            if abs(x_new - h[p]) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_smoothed_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    ws = w.sum()
    w /= ws
    obj = (m + mx / alpha + np.log(ws) / alpha) * dx
    grad = np.zeros(n)
    sig = w > 1e-12
    for k in np.where(sig)[0]:
        s = k - (n - 1)
        wk = w[k]
        if s == 0:
            grad += wk * (1.0 - 2.0 * x)
        else:
            ss = abs(s)
            grad[:n - ss] += wk * (1.0 - x[:n - ss])
            grad[ss:] += wk * (-x[:n - ss])
    grad *= dx
    return obj, grad

def projected_grad_descent(h, n, deadline, rng, alphas, iters_per_alpha=300):
    h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 30:
            break
        lr = 1.0 / (alpha * 2.0 / n)
        for it in range(iters_per_alpha):
            if time.time() > deadline - 30:
                break
            _, grad = lse_smoothed_grad(h, n, alpha)
            grad -= np.mean(grad)
            for _lr in range(20):
                h_new = np.clip(h - lr * grad, 0.0, 1.0)
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                _, g_new = lse_smoothed_grad(h_new, n, alpha)
                if g_new < _ + 0.5 * np.mean(grad ** 2) * lr:
                    break
                lr *= 0.5
            h = h_new
            c_cur = compute_c5(h, n)
            if c_cur < best_c:
                best_c = c_cur
        h = project_feasible(h, n)
    return h, best_c

def try_fourier_init(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)
    h = 0.5 * np.ones(n)
    for m in range(1, 8):
        amp = rng.uniform(0.05, 0.3) / m
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(m * np.pi * x + phase)
    h += 0.25 * np.tanh(2.0 * (x - 1.0))
    return project_feasible(np.clip(h, 0, 1), n)

def symmetrize(h, n):
    h_sym = 0.5 * (h + 1.0 - h[::-1])
    return project_feasible(h_sym, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 700
        h_init = try_fourier_init(n_init, seed)
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: symmetrize + coordinate descent
    h_sym = symmetrize(best_h, n_init)
    c_sym = compute_c5(h_sym, n_init)
    if c_sym < best_c:
        best_h, best_c = h_sym, c_sym
    h1, c1 = coord_descend(best_h, n_init, 30, deadline, rng, 50)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: gradient descent with log-sum-exp
    if time.time() < deadline - 200:
        alphas = [200, 500, 1000, 2000]
        h2, c2 = projected_grad_descent(best_h, n_init, deadline, rng, alphas, 200)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: CD refinement
    h3, c3 = coord_descend(best_h, n_init, 60, deadline, rng, 55)
    if c3 < best_c:
        best_h, best_c = h3, c3
    # Phase 4: perturbation + re-optimize
    for restart in range(3):
        if time.time() > deadline - 100:
            break
        h_pert = best_h + rng.normal(0, 0.02, n_init)
        h_pert = symmetrize(h_pert, n_init)
        hp, cp = coord_descend(h_pert, n_init, 40, deadline, rng, 50)
        if cp < best_c:
            best_h, best_c = hp, cp
    # Phase 5: try different n
    n_cur = len(best_h)
    n_cands = [n_cur - 100, n_cur + 100, n_cur - 50, n_cur + 50]
    n_cands = [v for v in n_cands if 300 <= v <= 1200]
    x_orig = np.linspace(0, 2, n_cur)
    for n_try in n_cands:
        if time.time() > deadline - 80:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_int = symmetrize(h_int, n_try)
        hi, ci = coord_descend(h_int, n_try, 40, deadline, rng, 50)
        if ci < best_c:
            best_h, best_c = hi, ci
            x_orig = x_new
    # Phase 6: final polish
    if time.time() < deadline - 20:
        max_p = max(20, int((deadline - time.time() - 5) / 2.0))
        hf, cf = coord_descend(best_h, len(best_h), max_p, deadline, rng, 60)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
