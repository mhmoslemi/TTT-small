def cd_pass(h, n, c, order, nbisect):
    improved = 0
    for p in order:
        h0 = h[p]
        if h0 < 1e-14 or h0 > 1.0 - 1e-14:
            continue
        x_new = solve_coord(c, h, p, n, nbisect)
        delta = x_new - h0
        if abs(delta) > 1e-14:
            h[p] = x_new
            # Incremental update of correlation
            # For m where p+m valid: c[m+n-1] += delta*(1-h[p+m])
            # For m where p-m valid: c[m+n-1] += -delta*h[p-m]
            # For m=0: use delta*(1-h0-x_new) instead
            m_fwd_lo = -(p)
            m_fwd_hi = (n - 1 - p)
            m_bwd_lo = (p - (n - 1))
            m_bwd_hi = p
            all_lo = min(m_fwd_lo, m_bwd_lo)
            all_hi = max(m_fwd_hi, m_bwd_hi)
            delta_c = np.zeros(all_hi - all_lo + 1)
            for m in range(all_lo, all_hi + 1):
                contrib = 0.0
                if m_fwd_lo <= m <= m_fwd_hi:
                    if m == 0:
                        continue
                    contrib += delta * (1.0 - h[p + m])
                if m_bwd_lo <= m <= m_bwd_hi:
                    if m == 0:
                        continue
                    contrib -= delta * h[p - m]
                if m == 0:
                    contrib = delta * (1.0 - h0 - x_new)
                delta_c[m - all_lo] += contrib
            c[all_lo + n - 1: all_hi + n - 1 + 1] += delta_c
            improved += 1
    return improved

def coord_descent(h, n, max_passes, deadline, rng, random_order=True, nbisect=45):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        n_imp = cd_pass(h, n, c, order, nbisect)
        if n_imp == 0:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full').copy()
    return h, compute_c5(h, n)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    if c_cur < best_c:
        best_h, best_c = h_cur, c_cur
    
    # Phase 2: SLSQP refinement
    h2, c2 = slsqp_opt(best_h, n, [1000, 5000, 20000], 200, deadline)
    if c2 < best_c:
        best_h, best_c = h2, c_cur
    
    # Phase 3: More CD
    h3, c3 = coord_descent(best_h, n, 150, deadline, rng, False, 45)
    if c3 < best_c:
        best_h, best_c = h3, c_cur
    
    # Phase 4: Structured perturbations + re-optimize
    for _ in range(15):
        if time.time() > deadline - 60:
            break
        sigma = 0.03 + 0.02 * rng.rand()
        h_p = random_perturb(best_h, n, sigma, rng)
        h_p = symmetrize(h_p, n)
        h_p, c_p = coord_descent(h_p, n, 40, deadline, rng, True, 40)
        if c_p < best_c:
            best_h, best_c = h_p, c_p
    
    # Phase 5: Swing perturbations
    for _ in range(10):
        if time.time() > deadline - 40:
            break
        k = rng.randint(5, 30)
        h_s = structured_perturb(best_h, n, rng, k)
        h_s = symmetrize(h_s, n)
        h_s, c_s = coord_descent(h_s, n, 30, deadline, rng, True, 40)
        if c_s < best_c:
            best_h, best_c = h_s, c_s
    
    # Phase 6: Try different n
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [720, 680, 750, 650]:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_int = symmetrize(h_int, n_try)
        h_t, c_t = coord_descent(h_int, n_try, 60, deadline, rng, True, 40)
        if c_t < best_c:
            best_h, best_c = h_t, c_t
            x_orig = x_new
    
    # Phase 7: Final SLSQP with high alpha
    n_cur = len(best_h)
    h_f, c_f = slsqp_opt(best_h, n_cur, [50000, 100000], 200, deadline)
    if c_f < best_c:
        best_h, best_c = h_f, c_f
    
    # Phase 8: Final CD polish
    remaining = deadline - time.time() - 3
    if remaining > 5:
        max_p = max(30, int(remaining * 2))
        h_pf, c_pf = coord_descent(best_h, n_cur, max_p, deadline, rng, False, 50)
        if c_pf < best_c:
            best_h, best_c = h_pf, c_pf
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
