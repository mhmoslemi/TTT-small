    f_hat = np.fft.rfft(f, size)
    return np.fft.irfft(t_hat * f_hat, size)[:n]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def toeplitz_mv(p, f, transpose=False):
    n = len(f)
    if not transpose:
        first_col = p[:n][::-1]
        first_row = p[n-1:]
    else:
        first_col = p[n-1:]
        first_row = p[:n][::-1]
    t = np.concatenate([first_col, first_row[1:][::-1]])
    size = 2 * n
    t_hat = np.fft.rfft(t, size)
    f_hat = np.fft.rfft(f, size)
    return np.fft.irfft(t_hat * f_hat, size)[:n]

def smooth_grad(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    scaled = beta * (c - c_max)
    p = np.exp(scaled)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros_like(c)
        p[c.argmax()] = 1.0
    else:
        p /= s
    S1 = toeplitz_mv(p, h, transpose=False)
    S2 = toeplitz_mv(p, h, transpose=True)
    return dx * (1.0 - S1 - S2)

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50, threshold=1e-5):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h[h < threshold] = 0.0
            h[h > 1.0 - threshold] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    return h, min(best_c, compute_c5(h, n))

def gradient_phase(h, n, deadline, max_iters=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_cur = h.copy()
    for it in range(max_iters):
        if time.time() > deadline - 60:
            break
        betas = [100, 300, 1000, 3000, 10000]
        beta = betas[min(it // 40, len(betas) - 1)]
        g = smooth_grad(h_cur, n, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        best_alpha = None
        best_val = best_c
        for alpha in [n/eta for eta in [beta, beta*0.5, beta*0.25, beta*2, beta*4]]:
            if alpha <= 0 or alpha > 10*n:
                continue
            h_trial = project_feasible(h_cur - alpha * g, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_val - 1e-12:
                best_val = c_trial
                best_alpha = alpha
                best_h = h_trial.copy()
        if best_alpha is not None:
            h_cur = best_h
            h_cur, c_cd = coord_descent(h_cur, n, 30, deadline, gs_iters=45)
            if c_cd < best_c:
                best_c = c_cd
                best_h = h_cur.copy()
        else:
            break
    return best_h, best_c

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        a = 5000.0
        shifted = a * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / a) * dx

    def eq_con(x):
        return x.sum() - n / 2.0

    cons = [{'type': 'eq', 'fun': eq_con}]
    for alpha in alphas:
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_each, 'ftol': 1e-14})
            if res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
