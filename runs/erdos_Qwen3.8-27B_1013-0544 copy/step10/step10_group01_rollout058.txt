import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def compute_subgradient(h, n):
    """Compute subgradient of max_k c[k] at h."""
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    m = k_star - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        idx1 = np.arange(0, n - m)
        g[idx1] += (1.0 - h[idx1 + m])
        if m > 0:
            idx2 = np.arange(m, n)
            g[idx2] -= h[idx2 - m]
    else:
        mp = -m
        idx1 = np.arange(mp, n)
        g[idx1] += (1.0 - h[idx1 + m])
        idx2 = np.arange(0, n - mp)
        g[idx2] -= h[idx2 + mp]
    return g, c, k_star

def fw_optimize(h, n, max_iter, deadline):
    """Frank-Wolfe method."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_large_mom = h.copy()
    y = h.copy()
    for t in range(max_iter):
        if time.time() > deadline:
            break
        g, c, k_star = compute_subgradient(y, n)
        # Find vertex: n/2 smallest g entries
        v = np.zeros(n)
        v[np.argsort(g)[:n // 2]] = 1.0
        # Step size
        gamma = 2.0 / (t + 2)
        h_new = (1 - gamma) * h + gamma * v
        h = project_feasible(h_new, n)
        # Nesterov
        beta = (1.0 - np.sqrt(1.0 + 4.0 / (t + 1) ** 2)) / 2.0
        y = h + beta * (h - h_large_mom)
        h_large_mom = h
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_h, best_c = h.copy(), c_new
    return best_h, best_c

def subgrad_optimize(h, n, max_iter, deadline, lr0=0.02):
    """Subgradient with averaging."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_avg = np.zeros(n)
    lr = lr0
    for t in range(max_iter):
        if time.time() > deadline:
            break
        g, c, k_star = compute_subgradient(h, n)
        h[0]  # ensure contiguous
        h_new = project_feasible(h - lr * g, n)
        h_avg = (t / (t + 1.0)) * h_avg + (1.0 / (t + 1.0)) * h_new
        if t % 100 == 99:
            lr *= 0.7
            c_avg = compute_c5(h_avg, n)
            if c_avg < best_c:
                best_h, best_c = h_avg.copy(), c_avg
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_h, best_c = h_new.copy(), c_new
        h = h_new
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_h, best_c = h.copy(), c_final
    return best_h, best_c

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    """Coordinate descent with golden section search."""
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            
            def fval(d):
                vals = c_aff + d * a
                if has_diag:
                    vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
                result = float(np.max(vals))
                if unaff_max > result:
                    result = unaff_max
                return result
            
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = fval(d1)
            f2 = fval(d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = fval(d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = fval(d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = fval(d_opt)
            f_curr = fval(0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            threshold = 1e-6
            h[np.where(h < threshold)[0]] = 0.0
            h[np.where(h > 1.0 - threshold)[0]] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    final_c = compute_c5(h, n)
    return h, min(best_c, final_c)

def binary_refine(h, n, deadline, max_passes=50, n_tries=500):
    """Round to binary and do local search via swaps."""
    rng = np.random.RandomState(12345)
    # Round: pick top n/2 values
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    best_c = compute_c5(h_bin, n)
    best_bin = h_bin.copy()
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        k_worst = int(np.argmax(c))
        m = k_worst - (n - 1)
        
        # Score each position for swapping
        # For a 1-position p: score_p = h[p+m] + h[p-m] (want small)
        # For a 0-position q: score_q = h[q+m] + h[q-m] (want large)
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Compute scores
        def valid_score(idx):
            s = np.zeros(len(idx))
            pm = idx + m
            pmm = idx - m
            valid_pm = (pm >= 0) & (pm < n)
            valid_pmm = (pmm >= 0) & (pmm < n)
            s[valid_pm] += h_bin[pm[valid_pm]]
            s[valid_pmm] += h_bin[pmm[valid_pmm]]
            return s
        
        scores_ones = valid_score(ones)
        scores_zeros = valid_score(zeros)
        
        # Get top candidates
        n_cand = min(20, len(ones))
        best_ones = ones[np.argsort(scores_ones)[:n_cand]]
        best_zeros = zeros[np.argsort(-scores_zeros)[:n_cand]]
        
        improved = False
        for p in best_ones:
            if time.time() > deadline:
                break
            for q in best_zeros:
                if p == q:
                    continue
                new_h = h_bin.copy()
                new_h[p] = 0.0
                new_h[q] = 1.0
                new_c = compute_c5(new_h, n)
                if new_c < best_c - 1e-12:
                    h_bin = new_h
                    best_c = new_c
                    best_bin = new_h.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try random swaps as fallback
            for _ in range(n_tries // 10):
                if time.time() > deadline:
                    break
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                if p == q:
                    continue
                new_h = h_bin.copy()
                new_h[p] = 0.0
                new_h[q] = 1.0
                new_c = compute_c5(new_h, n)
                if new_c < best_c - 1e-12:
                    h_bin = new_h
                    best_c = new_c
                    best_bin = new_h.copy()
                    improved = True
                    break
            if not improved:
                break
    
    return best_bin, best_c
