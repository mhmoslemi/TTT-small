```
def topk_gradient(h, c, alpha, K):
    n = len(h)
    L = len(c)
    m = float(c.max())
    if K >= L:
       idx = np.arange(L)
    else:
       idx = np.argpartition(c, -K)[-K:]
    cm = c[idx]
    u = alpha * (cm - m)
    u_max = u.max()
    w = np.exp(u - u_max)
    wsum = w.sum()
    if wsum <= 0: return np.zeros(n)
    w /= wsum
    g = np.zeros(n)
    for ii in range(len(idx)):
       ww = float(w[ii])
       if ww <= 1e-15: continue
       s = int(idx[ii]) - (n - 1)
       if s >= 0:
          nn = n - s
          g[s:] += ww * (1.0 - h[:nn])
          g[:nn] -= ww * h[s:]
       else:
          t = -s
          nn = n - t
          g[:nn] += ww * (1.0 - h[t:])
          g[t:] -= ww * h[:nn]
    return g
```

```
def make_r_order(n, mode, rng):
```

```
def topk_gradient(h,c,alpha,K):
 n=len(h); L=len(c); K=min(int(max(1,K)),L)
 m=max(c)
 idx=np.argpartition(c,-K)[-K:] if K<L else np.arange(L)
 cm=c[idx]; u=alpha*(cm-m); w=np.exp(u-u.max()); wsum=w.sum()
 if wsum<=0: return np.zeros(n)
 w/=wsum; g=np.zeros(n)
 for i in range(len(idx)):
  ww=float(w[i]); 
  if ww<1e-15: continue
  s=int(idx[i])-(n-1)
  if s>=0:
   nn=n-s
   g[s:] += ww*(1.0-h[:nn])
   g[:nn] -= ww*h[s:]
  else:
   t=-s; nn=n-t
   g[:nn] += ww*(1.0-h[t:])
   g[t:] -= ww*h[:nn]
 return g
```

```python
import numpy as np
import time


def project_box_sum(v, target):
    v = np.clip(np.asarray(v, dtype=float), 0.0, 1.0)
    if abs(float(v.sum()) - float(target)) < 1e-15:
        return v
    lo, hi = -1.5, 1.5
    for _ in range(55):
        mu = 0.5 * (lo + hi)
        s = float(np.clip(v - mu, 0.0, 1.0).sum())
        if s > target:
            lo = mu
        else:
            hi = mu
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)


def corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def c5_of(h):
    return float(corr(h).max() * (2.0 / len(h)))


def deriv_row(p, h, s):
    A = np.zeros(len(s), dtype=float)
    idx = p - s
    m = (idx >= 0) & (idx < len(h))
    if np.any(m):
        A[m] += 1.0 - h[idx[m]]
    idx = p + s
    m = (idx >= 0) & (idx < len(h))
    if np.any(m):
        A[m] -= h[idx[m]]
    return A


def pair_f(d, c, A, diag, m1, m2):
    if d == 0.0:
        return float(c.max())
    vals = c + d * A
    q = d * d
    vals[diag] -= 2.0 * q
    vals[m1] += q
    vals[m2] += q
    return float(vals.max())


def pair_grid(c, A, diag, m1, m2, ds):
    ds = np.asarray(ds, dtype=float)
    if ds.size == 1:
        return np.array([pair_f(float(ds[0]), c, A, diag, m1, m2)])
    vals = c.reshape(1, -1) + np.outer(ds, A)
    q = ds * ds
    vals[:, diag] -= 2.0 * q
    vals[:, m1] += q
    vals[:, m2] += q
    return vals.max(axis=1)


def pair_refine(c, A, diag, m1, m2, lo, hi, its=30):
    if hi - lo <= 1e-8:
        d = 0.5 * (lo + hi)
        return d, pair_f(d, c, A, diag, m1, m2)
    g = 0.6180339887498949
    x1 = hi - g * (hi - lo)
    x2 = lo + g * (hi - lo)
    f1 = pair_f(x1, c, A, diag, m1, m2)
    f2 = pair_f(x2, c, A, diag, m1, m2)
    for _ in range(its):
        if hi - lo < 1e-7:
            break
        if f1 <= f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - g * (hi - lo)
            f1 = pair_f(x1, c, A, diag, m1, m2)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + g * (hi - lo)
            f2 = pair_f(x2, c, A, diag, m1, m2)
    d = 0.5 * (lo + hi)
    return d, pair_f(d, c, A, diag, m1, m2)


def pair_cd_pass(h, n, ro, s, grid, tol, deadline, recompute_every=40):
    h = np.array(h, dtype=float, copy=True)
    c = corr(h)
    D = n - 1
    cm = float(c.max())
    upd = 0
    for p in range(n):
        if time.time() > deadline:
            break
        r = int(ro[p]) % n
        if r == p:
            continue
        hp = float(h[p])
        hr = float(h[r])
        lo = max(-hp, hr - 1.0)
        hi = min(1.0 - hp, hr)
        if hi - lo < 1e-12:
            continue
        A = deriv_row(p, h, s)
        A -= deriv_row(r, h, s)
        m1 = D + r - p
        m2 = D + p - r
        ds = np.linspace(lo, hi, int(max(4, grid)))
        if not np.any(np.abs(ds) < 1e-14):
            ds = np.concatenate((ds, np.array([0.0])))
            ds.sort()
        fs = pair_grid(c, A, D, m1, m2, ds)
        ix = int(np.argmin(fs))
        bf = float(fs[ix])
        if bf >= cm - tol:
            continue
        if ix == 0:
            lb, ub = float(lo), float(ds[1])
        elif ix == len(ds) - 1:
            lb, ub = float(ds[-2]), float(hi)
        else:
            lb, ub = float(ds[ix - 1]), float(ds[ix + 1])
        d, fd = pair_refine(c, A, D, m1, m2, lb, ub)
        if fd < cm - tol:
            h[p] = float(np.clip(hp + d, 0.0, 1.0))
            h[r] = float(np.clip(hr - d, 0.0, 1.0))
            q = d * d
            c += d * A
            c[D] -= 2.0 * q
            c[m1] += q
            c[m2] += q
            cm = float(c.max())
            upd += 1
            if recompute_every and upd % recompute_every == 0:
                c = corr(h)
                cm = float(c.max())
    if upd:
        c = corr(h)
        cm = float(c.max())
    return h, float(cm * (2.0 / n)), upd


def topk_gradient(h, c, alpha, K):
    n = len(h)
    L = len(c)
    K = int(min(max(1, K), L))
    m = float(c.max())
    if K < L:
        idx = np.argpartition(c, -K)[-K:]
    else:
        idx = np.arange(L)
    cm = c[idx]
    u = alpha * (cm - m)
    w = np.exp(u - u.max())
    ws = float(w.sum())
    if ws <= 0.0:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n, dtype=float)
    for i in range(len(idx)):
        ww = float(w[i])
        if ww <= 1e-15:
            continue
        s = int(idx[i]) - (n - 1)
        if s >= 0:
            nn = n - s
            g[s:] += ww * (1.0 - h[:nn])
            g[:nn] -= ww * h[s:]
        else:
            t = -s
            nn = n - t
            g[:nn] += ww * (1.0 - h[t:])
            g[t:] -= ww * h[:nn]
    return g


def smooth_subgrad(h, n, deadline, iters, alpha, K, eta, beta, rng=None):
    h = np.array(h, dtype=float, copy=True)
    t = 0.5 * n
    c = corr(h)
    cm = float(c.max())
    best = h.copy()
    bc = cm
    v = np.zeros(n)
    e = float(eta)
    stuck = 0
    for _ in range(int(max(1, iters))):
        if time.time() > deadline:
            break
        g = topk_gradient(h, c, alpha, K)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        vn = float(np.linalg.norm(v))
        if vn < 1e-14:
            break
        v /= vn
        cur = cm
        moved = 0
        for fac in (1.0, 0.5, 0.25, 0.125, 0.0625):
            tr = project_box_sum(h - e * fac * v, t)
            ct = corr(tr)
            tm = float(ct.max())
            if tm < cur - 1e-11:
                h = tr
                c = ct
                cm = tm
                moved = 1
                if tm < bc:
                    best = tr.copy()
                    bc = tm
                    stuck = 0
                else:
                    stuck += 1
                break
        if moved:
            e = min(0.10, e * 1.02)
        else:
            e *= 0.7
            stuck += 1
            if stuck > 60 and rng is not None:
                h = project_box_sum(best + rng.normal(0.0, 0.02, n), t)
                c = corr(h)
                cm = float(c.max())
                if cm > bc + 0.5:
                    h = best.copy()
                    c = corr(h)
                    cm = bc
                v = np.zeros(n)
                e = 0.05
                stuck = 0
    return best, float(bc * (2.0 / n))


def make_r_order(n, mode, rng):
    a = np.arange(n)
    if mode == 0:
        return (a + 1) % n
    if mode == 1:
        return (a - 1) % n
    if mode == 2:
        return (a + n // 2) % n
    if mode == 3:
        shift = int(rng.randint(1, n))
        if shift == n:
            shift = 1
        return (a + shift) % n
    if mode == 4:
        r = rng.permutation(n)
        bad = r == a
        if np.any(bad):
            r[bad] = (r[bad] + 1) % n
        return r
    r = rng.randint(0, n, n)
    bad = r == a
    if np.any(bad):
        r[bad] = (r[bad] + 1) % n
    return r


def optimize_cd(h, n, s, deadline, max_passes, grid, mode0=0, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    if grid is None:
        grid = 18 if n <= 900 else 12
    h = np.array(h, dtype=float, copy=True)
    best = h.copy()
    bc = c5_of(best)
    cur = h.copy()
    stag = 0
    noupd = 0
    for i in range(int(max(1, max_passes))):
        if time.time() > deadline:
            break
        if i % 5 == 0:
            room = np.minimum(cur, 1.0 - cur) + 0.02 * rng.rand(n)
            r = int(np.argmax(room))
            ro = np.full(n, r, dtype=int)
        else:
            ro = make_r_order(n, (i + int(mode0)) % 5, rng)
        h2, c2, up = pair_cd_pass(cur, n, ro, s, grid, 1e-9, deadline, 35)
        if c2 < bc:
            best = h2.copy()
            bc = c2
            cur = best.copy()
            stag = 0
        else:
            cur = best.copy()
            stag += 1
        if up == 0:
            noupd += 1
        else:
            noupd = 0
        if stag >= 8 or noupd >= 5:
            break
    return best, bc


def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    t = 0.5 * n_new
    if n_old == n_new:
        return project_box_sum(h_old, t)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h_old)
    return project_box_sum(y, t)


def gen_binary(n, rng, kind):
    if kind:
        t = np.zeros(n, dtype=np.int8)
        for i in range(1, n):
            t[i] = t[i >> 1] ^ (i & 1)
        v = t.astype(float)
    else:
        v = (rng.rand(n) < 0.5).astype(float)
    return project_box_sum(v, 0.5 * n)


def fix_sum(h, target):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    d = float(target - h.sum())
    if abs(d) > 1e-12:
        if abs(d) < 1e-3:
            i = int(np.argmax(np.minimum(h, 1.0 - h)))
            h[i] = float(np.clip(h[i] + d, 0.0, 1.0))
            if abs(float(target - h.sum())) > 1e-10:
                h = project_box_sum(h, target)
        else:
            h = project_box_sum(h, target)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 700
        h = np.full(n, 0.5)
    if n < 2:
        n = 2
        h = np.array([0.5, 0.5])

    h = project_box_sum(h, 0.5 * n)
    best = h.copy()
    bc = c5_of(best)
    s = np.arange(-(n - 1), n)

    best, bc = optimize_cd(best, n, s, dl - 380, 100000, None, 0, rng)

    si = 0
    while time.time() < dl - 180:
        alphas = (30.0, 80.0, 160.0, 300.0)
        kset = (60, 100, 140, 180)
        h2, c2 = smooth_subgrad(best, n, dl - 160, 90,
                                alphas[si % 4], kset[si % 4],
                                0.03 + 0.01 * (si % 3), 0.65, rng)
        if c2 < bc:
            best = h2.copy()
            bc = c2
        cur = h2 if c2 < bc + 0.004 else best
        if time.time() < dl - 160:
            cur, ct = optimize_cd(cur, n, s, dl - 160, 4, 16, si % 5, rng)
            if ct < bc:
                best = cur.copy()
                bc = ct
        si += 1

    for nt in (1000, 500, 1400, 350):
        if time.time() > dl - 90:
            break
        limit = min(time.time() + 80.0, dl - 70.0)
        if limit <= time.time():
            continue
        ht = interpolate_h(best, nt)
        st = np.arange(-(nt - 1), nt)
        ht, ct = optimize_cd(ht, nt, st, limit, 20, None, nt % 5, rng)
        if ct < bc:
            best = ht.copy()
            bc = ct
            n = len(best)
            s = np.arange(-(n - 1), n)

    for kind in (1, 0):
        if time.time() > dl - 50:
            break
        limit = min(time.time() + 35.0, dl - 35.0)
        if limit <= time.time():
            continue
        st0 = gen_binary(n, rng, kind)
        st0, ct = optimize_cd(st0, n, s, limit, 10, 14, kind, rng)
        if ct < bc:
            best = st0.copy()
            bc = ct

    if time.time() < dl - 35:
        limit = min(time.time() + 30.0, dl - 25.0)
        if limit > time.time():
            st0 = project_box_sum(best + 0.03 * rng.normal(size=n), 0.5 * n)
            st0, ct = optimize_cd(st0, n, s, limit, 8, 12, 2, rng)
            if ct < bc:
                best = st0.copy()
                bc = ct

    best, bc = optimize_cd(best, n, s, dl - 5, 20, 16, 0, rng)

    best = fix_sum(best, 0.5 * len(best))
    bc = c5_of(best)
    return best, bc, len(best)
```
