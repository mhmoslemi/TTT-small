    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()

    c_vals = c_raw * dx
    cmax = c_vals.max()
    
    # Step size schedule
    step0 = 0.1
    decay = 0.999
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Active set: bins within 5% of max
        threshold = cmax - 0.05 * cmax
        active = c_vals > threshold
        
        # Compute gradient: average over active bins
        # grad[p] = (1/n_active) * sum_{m in active} dc[m]/dh[p]
        grad = np.zeros(n)
        for k in range(-(n-1), n):
            m = k + (n-1)
            if m < 0 or m >= 2*n-1 or not active[m]:
                continue
            if k == 0:
                grad += (1.0 - 2.0 * h)
            elif k > 0:
                grad[:n-k] += (1.0 - h[k:n])
                grad[k:n] -= h[:n-k]
            else:
                kk = -k
                grad[kk:n] += (1.0 - h[:n-kk])
                grad[:n-kk] -= h[kk:n]
        grad /= active.sum()
        
        # Step
        step = step0 * (decay ** min(it, 2000))
        h = np.clip(h - step * grad, 0, 1)
        h = project_feasible(h, n)
        
        # Update
        c_raw = np.correlate(h, 1.0 - h, mode='full')
        c_vals = c_raw * dx
        cmax = c_vals.max()
        if cmax < best_c:
            best_c = cmax
            best_h = h.copy()
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb): a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0: unaff = max(unaff, float(np.max(c[:m_min])))
    if m_max < 2 * n - 2: unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15

    L_lo, L_hi = unaff, cur_max
    for _ in range(80):
        if L_hi - L_lo < 1e-15: break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L): feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):

                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2): feasible = False
        if feasible: L_hi = L

    L = L_hi
    x_lo, x_hi = 0.0, 1.0

    for _ in range(max_passes):
        if time.time() > deadline: break
        improved = False
        if mode == 'seq': order = np.arange(n)
        elif mode == 'rev': order = np.arange(n - 1, -1, -1)
        else: order = rng.permutation(n)
        for p in order:
            h0 = h[p]

                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb): a[vb] -= h[jb[vb]]

        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def binary_flip_search(h_init, n, deadline):
    h = np.where(h_init > 0.5, 1.0, 0.0)
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        oi = np.where(h == 1)[0]
        for i in oi[np.argsort(h[oi])][:n_ones - target]: h[i] = 0
    elif n_ones < target:
        zi = np.where(h == 0)[0]
        for i in zi[np.argsort(-h[zi])][:target - n_ones]: h[i] = 1
    c_cur = compute_c5(h, n)
    best_h, best_c = h.copy(), c_cur
    rng = np.random.RandomState(123)
    for it in range(500000):
        if time.time() > deadline: break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0: break
        if it % 5000 == 0 and len(ones) * len(zeros) > 40000:
            si = rng.choice(ones, min(60, len(ones)), replace=False)
            sj = rng.choice(zeros, min(60, len(zeros)), replace=False)
        else:
            si, sj = ones, zeros
        best_flip, best_nc = None, c_cur
        for i in si:
            for j in sj:
                h[i], h[j] = 0, 1
                cn = compute_c5(h, n)
                if cn < best_nc - 1e-15:
                    best_nc, best_flip = cn, (i, j)
                h[i], h[j] = 1, 0
        if best_flip is not None:
            i, j = best_flip
            h[i], h[j] = 0, 1
            c_cur = best_nc
            if c_cur < best_c: best_c, best_h = c_cur, h.copy()
        else:
            temp = 0.003 * max(0.01, 1.0 - it / 200000.0)
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i], h[j] = 0, 1
            cn = compute_c5(h, n)
            if cn < c_cur or rng.random() < np.exp(-(cn - c_cur) / max(temp, 1e-10)):
                c_cur = cn
                if c_cur < best_c: best_c, best_h = c_cur, h.copy()
            else:
                h[i], h[j] = 1, 0
    return best_h, best_c

def lse_subgrad(h, n, deadline, max_iters=3000):
    h = h.copy()
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = 0.05
    for it in range(max_iters):
        if time.time() > deadline: break
        c_raw = np.correlate(h, 1.0 - h, mode='full')
        cmax = c_raw.max()
        shifted = 200.0 * (c_raw - cmax)
        mx = shifted.max()
        weights = np.exp(shifted - mx)
        wsum = weights.sum()
        if wsum < 1e-10: wsum = 1.0
        weights /= wsum
        grad = np.zeros(n)
        for k in range(-(n - 1), n):
            m = k + (n - 1)
            if m < 0 or m >= 2 * n - 1: continue
            w = weights[m]
            if w < 1e-8: continue
            if k == 0:
                grad += w * (1.0 - 2.0 * h)
            elif k > 0:
                grad[:n - k] += w * (1.0 - h[k:n])
                grad[k:n] -= w * h[:n - k]
            else:
                kk = -k
                grad[kk:n] += w * (1.0 - h[:n - kk])
                grad[:n - kk] -= w * h[kk:n]
        h = np.clip(h - alpha * grad, 0.0, 1.0)
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < best_c:
            best_c, best_h = nc, h.copy()
        alpha = max(0.001, alpha * 0.9995)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: CD polish
    h1, c1 = exact_cd(h_best, n, 400, deadline, 'seq')
    if c1 < best_c: h_best, best_c = h1, c1
    h1, c1 = exact_cd(h_best, n, 200, deadline, 'rev')
    if c1 < best_c: h_best, best_c = h1, c1

    # Phase 2: Binary flip search from current solution
    if time.time() < deadline - 300:
        hb, cb = binary_flip_search(h_best, n, min(deadline, t0 + 400))
        if cb < best_c:
            h_best, best_c = hb, cb
        # Polish binary result with CD
        hp, cp = exact_cd(hb, n, 300, deadline, 'seq')
        if cp < best_c: h_best, best_c = hp, cp

    # Phase 3: Log-sum-exp subgradient (smooth max optimization)
    if time.time() < deadline - 200:
        hs, cs = lse_subgrad(h_best, n, deadline)
        if cs < best_c: h_best, best_c = hs, cs
        hp, cp = exact_cd(hs, n, 200, deadline, 'seq')
        if cp < best_c: h_best, best_c = hp, cp

    # Phase 4: Shake + CD
    n = len(h_best)
    for sigma in [0.008, 0.015, 0.03, 0.005, 0.02]:
        if time.time() > deadline - 60: break
        hp = project_feasible(h_best + np.random.normal(0, sigma, n), n)
        ho, co = exact_cd(hp, n, 250, deadline, 'seq')
        if co < best_c: h_best, best_c = ho, co
        ho, co = exact_cd(h_best, n, 100, deadline, 'rand')
        if co < best_c: h_best, best_c = ho, co

    # Phase 5: Try smaller n
    for n_try in [500, 400, 350]:
        if time.time() > deadline - 80: break
        xo = np.linspace(0, 2, len(h_best))
        xn = np.linspace(0, 2, n_try)
        hi = project_feasible(np.interp(xn, xo, h_best), n_try)
        ho, co = exact_cd(hi, n_try, 400, deadline, 'seq')
        if co < best_c:
            h_best, best_c, n = ho, co, n_try

    # Phase 6: Final CD polish
    n = len(h_best)
    if time.time() < deadline - 5:
        hf, cf = exact_cd(h_best, n, 500, deadline, 'seq')
        if cf < best_c: h_best, best_c = hf, cf
        hf, cf = exact_cd(h_best, n, 200, deadline, 'rev')
        if cf < best_c: h_best, best_c = hf, cf

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
