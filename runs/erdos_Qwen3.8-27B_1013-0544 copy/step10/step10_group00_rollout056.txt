import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h


def _eval_cd(c_aff, a, diag_idx, has_diag, unaff, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] -= d * d
    r = float(vals.max())
    return r if r > unaff else unaff


def cd_pass(h, n, gs_iters, c):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    improved = False
    for p in range(n):
        h0 = h[p]
        if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
            continue
        m_lo = max(0, n - 1 - p)
        m_hi = min(2 * n - 2, n - 1 + p)
        ms = np.arange(m_lo, m_hi + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        a = np.zeros(len(ms))
        if np.any(vf):
            a[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            a[vb] -= h[jb[vb]]
        dp_arr = ms == (n - 1)
        has_diag = bool(dp_arr.any())
        diag_idx = int(np.where(dp_arr)[0][0]) if has_diag else -1
        unaff = 0.0
        if m_lo > 0:
            unaff = float(c[:m_lo].max())
        if m_hi < 2 * n - 2:
            unaff = max(unaff, float(c[m_hi + 1:].max()))
        c_aff = c[ms]
        d_lo, d_hi = -h0, 1.0 - h0
        d1 = d_hi - gr * (d_hi - d_lo)
        d2 = d_lo + gr * (d_hi - d_lo)
        f1 = _eval_cd(c_aff, a, diag_idx, has_diag, unaff, d1)
        f2 = _eval_cd(c_aff, a, diag_idx, has_diag, unaff, d2)
        for _ in range(gs_iters):
            if d_hi - d_lo < 1e-14:
                break
            if f1 < f2:
                d_hi, d2, f2 = d2, d1, f1
                d1 = d_hi - gr * (d_hi - d_lo)
                f1 = _eval_cd(c_aff, a, diag_idx, has_diag, unaff, d1)
            else:
                d_lo, d1, f1 = d1, d2, f2
                d2 = d_lo + gr * (d_hi - d_lo)
                f2 = _eval_cd(c_aff, a, diag_idx, has_diag, unaff, d2)
        d_opt = (d_lo + d_hi) * 0.5
        f_opt = _eval_cd(c_aff, a, diag_idx, has_diag, unaff, d_opt)
        f_cur = _eval_cd(c_aff, a, diag_idx, has_diag, unaff, 0.0)
        if f_opt < f_cur - 1e-13:
            h[p] = h0 + d_opt
            c[ms] += d_opt * a
            if has_diag:
                c[ms][diag_idx] -= d_opt * d_opt
            improved = True
    return improved


def cd_loop(h, n, max_passes, deadline, gs_iters=55):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = cd_pass(h, n, gs_iters, c)
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full').copy()
    return h, compute_c5(h, n)


def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    if alpha < 1e6:
        shifted = alpha * (c - m)
        mx = float(shifted.max())
        lse = mx + float(np.log(np.sum(np.exp(shifted - mx))))
        return (m + lse / alpha) * dx
    return m * dx


def _sum_eq(x, n):
    return x.sum() - n / 2.0


def _sum_eq_jac(x, n):
    return np.ones(n)


def slsqp_opt(h, n, alphas, maxiter_each=200, timeout=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        if timeout is not None and time.time() > timeout:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
