import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):

    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))

def update_c(c, h, p, xn, n):
    h0 = h[p]
    delta = xn - h0
    if abs(delta) < 1e-15:
        return
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)

def binary_round(h, n):
    h_bin = (h > 0.5).astype(float)
    diff = int(round(n / 2 - h_bin.sum()))
    if diff > 0:
        zeros = np.where(h_bin == 0)[0]
        order = zeros[np.argsort(-h[zeros])]
        for i in range(min(diff, len(order))):
            h_bin[order[i]] = 1.0
    elif diff < 0:
        ones = np.where(h_bin == 1)[0]
        order = ones[np.argsort(h[ones])]
        for i in range(min(-diff, len(order))):
            h_bin[order[i]] = 0.0
    return h_bin

def binarize_pressure(h, n, beta):
    h_new = 0.5 + 0.5 * np.sign(h - 0.5) * np.abs(2.0 * (h - 0.5)) ** beta
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n)

def binary_2swap(h, n, max_iters, deadline, rng):
    h = h.copy()
    cmax = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c_new = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
        if c_new > cmax:
            h[i], h[j] = 1.0, 0.0
        else:
            cmax = c_new
            ones.remove(i)
            ones.append(j)
            zeros.append(i)
            zeros.remove(j)
    return h, cmax * (2.0 / n)

        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
