def project_h(h, n):

    target = 0.5 * n
    s = h.sum()
    if abs(s-target) <= 1e-13:
        # still exact adjust? no
        return h
    lo = -1.0
    hi = 1.0
    # if s<target, need increase -> λ>0; bisection generic
    for _ in range(60):
       mid = 0.5*(lo+hi)
       hs = h + mid
       np.clip(hs, 0.0, 1.0, out=hs)
       ss = hs.sum()
       if ss < target:
           lo = mid
       else:
           hi = mid
    h = h + 0.5*(lo+hi)
    np.clip(h,0,1,out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
       # small distribution to non-saturated
       if diff > 0:
          mask = h < 1.0
          cnt = mask.sum()
          if cnt:
             add = diff/cnt
             h[mask] = np.minimum(1.0, h[mask]+add)
       else:
          mask = h > 0.0
          cnt = mask.sum()
          if cnt:
             sub = -diff/cnt
             h[mask] = np.maximum(0.0, h[mask]-sub)
    # very small exact one element to reduce float
    diff = target - h.sum()
    if diff > 0:
       idx = int(np.argmin(h)); h[idx] = min(1.0, h[idx]+diff)
    elif diff < 0:
       idx = int(np.argmax(h)); h[idx] = max(0.0, h[idx]+diff)

def project_h(h, n):
    h = np.clip(h.astype(float, copy=False), 0.0, 1.0).copy()
    target = 0.5*n
    s = h.sum()
    if abs(s-target) > 1e-13:

          mid=0.5*(lo+hi)
          sh = h + mid
          np.clip(sh, 0.0, 1.0, out=sh)
          if sh.sum() < target: lo=mid

    diff=target-h.sum()
    if abs(diff)>1e-14:
       if diff>0:
          mask=h<1.0
          if mask.any(): h[mask]=np.minimum(1.0,h[mask]+diff/mask.sum())
       else:
          mask=h>0.0

import time
import numpy as np


def project_h(h, n):
    h = np.clip(h.astype(float, copy=False), 0.0, 1.0).copy()
    target = 0.5 * n
    if abs(h.sum() - target) > 1e-13:
        lo, hi = -1.0, 1.0
        for _ in range(60):
            mid = 0.5 * (lo + hi)
            sh = h + mid
            np.clip(sh, 0.0, 1.0, out=sh)
            if sh.sum() < target:
                lo = mid
            else:
                hi = mid
        h = h + 0.5 * (lo + hi)
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
            if mask.any():
                h[mask] = np.minimum(1.0, h[mask] + diff / mask.sum())
        else:
            mask = h > 0.0
            if mask.any():
                h[mask] = np.maximum(0.0, h[mask] + diff / mask.sum())
    diff = target - h.sum()
    if diff > 0:
        idx = int(np.argmin(h))
        h[idx] = min(1.0, h[idx] + diff)
    elif diff < 0:
        idx = int(np.argmax(h))
        h[idx] = max(0.0, h[idx] + diff)
    return h


def project_f(f, n):
    return 2.0 * project_h(0.5 * (f + 1.0), n) - 1.0


def compute_f_vals(f, n):
    if n < 2:
        return np.zeros(0), np.zeros(0), 0.5, 0.5
    ac = np.correlate(f, f, mode="full")
    k0 = float(ac[n - 1])
    z0 = (n - k0) / (2.0 * n)
    m = np.arange(1, n)
    k = ac[n:2 * n - 1]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(f, out=pref[1:])
    d = pref[m] - (pref[n] - pref[n - m])
    inv = 1.0 / (2.0 * n)
    zp = (n - m - k + d) * inv
    zm = (n - m - k - d) * inv
    cur = max(z0, float(np.max(zp)), float(np.max(zm)))
    return zp, zm, z0, cur


def eval_f(f, n):
    f = project_f(f, n)
    h = 0.5 * (f + 1.0)
    h = project_h(h, n)
    f = 2.0 * h - 1.0
    c = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))
    return f, h, c


def make_split(a, b):
    eps = 1e-15
    pos = a > eps
    neg = a < -eps
    ap = a[pos]
    bp = b[pos]
    an = a[neg]
    bn = b[neg]
    zero = ~(pos | neg)
    zmax = float(np.max(b[zero])) if np.any(zero) else -np.inf
    return ap, bp, an, bn, zmax


def split_low_bound(sp):
    ap, bp, an, bn, zmax = sp
    lb = zmax
    if ap.size:
        v = float(np.min(bp - ap))
        if v > lb:
            lb = v
    if an.size:
        v = float(np.min(bn + an))
        if v > lb:
            lb = v
    return lb if np.isfinite(lb) else -np.inf


def interval_feasible(t, sp1, sp2, base0, fp, two_n):
    ylo, yhi = -1.0, 1.0
    ap, bp, an, bn, zmax = sp1
    if ap.size:
        v = float(np.min((t - bp) / ap))
        if v < yhi:
            yhi = v
    if an.size:
        v = float(np.max((t - bn) / an))
        if v > ylo:
            ylo = v
    if zmax > t + 1e-12:
        return -1, ylo, yhi
    ap, bp, an, bn, zmax = sp2
    if ap.size:
        v = float(np.min((t - bp) / ap))
        if v < yhi:
            yhi = v
    if an.size:
        v = float(np.max((t - bn) / an))
        if v > ylo:
            ylo = v
    if zmax > t + 1e-12:
        return -1, ylo, yhi
    if ylo > yhi + 1e-12:
        return -1, ylo, yhi
    r = fp * fp + two_n * (t - base0)
    if r < -1e-14:
        return -1, ylo, yhi
    if r < 1.0:
        if r < 0.0:
            r = 0.0
        s = float(np.sqrt(r))
        if -s > ylo:
            ylo = -s
        if s < yhi:
            yhi = s
        if ylo > yhi + 1e-12:
            return -1, ylo, yhi
    return 1, ylo, yhi


def solve_f_split(fp, base0, cur, sp1, sp2, n, iters):
    two_n = 2.0 * n
    low = base0 - (fp * fp) / two_n
    if low < 0.0:
        low = 0.0
    v = split_low_bound(sp1)
    if v > low:
        low = v
    v = split_low_bound(sp2)
    if v > low:
        low = v
    if low >= cur - 1e-13:
        return fp, cur
    lo, hi = low, cur
    for _ in range(iters):
        t = 0.5 * (lo + hi)
        ok, ylo, yhi = interval_feasible(t, sp1, sp2, base0, fp, two_n)
        if ok > 0:
            hi = t
        else:
            lo = t
        if hi - lo < 1e-13:
            break
    ok, ylo, yhi = interval_feasible(hi, sp1, sp2, base0, fp, two_n)
    if ok <= 0 or ylo > yhi + 1e-10:
        return fp, cur
    if fp < ylo:
        y = ylo
    elif fp > yhi:
        y = yhi
    else:
        y = fp
    if y < -1.0:
        y = -1.0
    elif y > 1.0:
        y = 1.0
    return float(y), float(hi)


def cd_f(f, n, max_passes, deadline, iters=22, rng=None):
    f = project_f(f, n)
    zp, zm, z0, cur = compute_f_vals(f, n)
    best_f, best_c = f.copy(), cur
    plateau = 0
    mlen = n - 1
    inv = 1.0 / (2.0 * n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        start_c = cur
        order = rng.permutation(n) if rng is not None else range(n)
        drift = 0.0
        coord_count = 0
        for p in order:
            if time.time() > deadline:
                break
            fp = f[p]
            tmat = np.zeros(mlen)
            ll = min(p, mlen)
            if ll > 0:
                stop = p - ll - 1
                if stop >= 0:
                    tmat[:ll] += f[p - 1:stop:-1]
                else:
                    tmat[:ll] += f[p - 1::-1]
            rl = min(mlen, n - 1 - p)
            if rl > 0:
                tmat[:rl] += f[p + 1:p + rl + 1]
            dm = np.zeros(mlen)
            if p + 1 < n:
                dm[p + 1:] += 1.0
            if n - p >= 1:
                dm[n - p:] -= 1.0
            a1 = (-tmat + dm) * inv
            a2 = (-tmat - dm) * inv
            sp1 = make_split(a1, zp)
            sp2 = make_split(a2, zm)
            y, newcur = solve_f_split(fp, z0, cur, sp1, sp2, n, iters)
            if abs(y - fp) > 1e-13:
                delta = y - fp
                z0 += (y * y - fp * fp) * inv
                zp += a1 * delta
                zm += a2 * delta
                f[p] = y
                drift += delta
                if newcur < cur:
                    cur = newcur
                if z0 > cur:
                    cur = z0
                if abs(drift) > 0.5:
                    f = project_f(f, n)
                    zp, zm, z0, cur = compute_f_vals(f, n)
                    drift = 0.0
            else:
                if newcur < cur:
                    cur = newcur
            coord_count += 1
            if coord_count % 64 == 0:
                cur = max(z0, float(np.max(zp)), float(np.max(zm)))
        f = project_f(f, n)
        zp, zm, z0, cur = compute_f_vals(f, n)
        if cur < best_c - 1e-14:
            best_c, best_f = cur, f.copy()
        if cur < start_c - 1e-12:
            plateau = 0
        else:
            plateau += 1
        if plateau >= 3:
            break
    return best_f, best_c


def interp_f(f, nsrc, ntgt):
    if ntgt == nsrc:
        return project_f(f, ntgt)
    xs = np.linspace(0.0, 1.0, nsrc, endpoint=False)
    xt = np.linspace(0.0, 1.0, ntgt, endpoint=False)
    return project_f(np.interp(xt, xs, f), ntgt)


def fourier_start(n, kmax, rng):
    i = np.arange(n)
    x = np.pi * (2.0 * i - (n - 1)) / n
    f = np.zeros(n)
    for k in range(1, kmax + 1):
        f += rng.normal(0.0, 1.0 / np.sqrt(float(k))) * np.cos(k * x)
    f = 0.5 * (f + f[::-1])
    f -= f.mean()
    mx = float(np.max(np.abs(f)))
    if mx > 1e-12:
        f /= mx
    else:
        f = rng.normal(0.0, 1.0, n)
        f -= f.mean()
    f = np.tanh(2.0 * f)
    return project_f(f, n)


def sin_start(n):
    x = (np.arange(n) + 0.5) / n
    f = np.sin(np.pi * x)
    f -= f.mean()
    mx = float(np.max(np.abs(f)))
    if mx > 1e-12:
        f /= mx
    f = np.tanh(2.5 * f)
    return project_f(f, n)


def bits_start(n, rng):

                break
            fs = fourier_start(ns, 8, rng)
            ft, _ = cd_f(fs, ns, 300, min(t_lowres - 5.0, hard - 60.0), iters=16)
            fh = interp_f(ft, ns, best_n)
            if time.time() < t_lowres - 5:
                fh, _ = cd_f(fh, best_n, 30, t_lowres, iters=18)
            fh, hh, ch = eval_f(fh, best_n)
            if ch < best_c:
                best_f, best_h, best_c, best_n = fh, hh, ch, best_n

    if time.time() < t_long - 20:
        ft, _ = cd_f(best_f, best_n, 1000000, t_long, iters=22, rng=None)
        ft, ht, ct = eval_f(ft, best_n)
        if ct < best_c:
            best_f, best_h, best_c, best_n = ft, ht, ct, best_n

    for r in range(16):
        if time.time() > t_pert - 35:
            break
        dcur = min(t_pert - 5.0, time.time() + 70.0)
        if r % 4 == 0:
            temp = 0.08 + 0.035 * (r // 4)
            ft = project_f(best_f + rng.normal(0.0, temp, best_n), best_n)
        elif r % 4 == 1:
            ft = project_f(np.tanh((1.5 + 0.8 * r) * best_f), best_n)
        elif r % 4 == 2:
            ft = project_f(0.55 * best_f + 0.45 * bits_start(best_n, rng), best_n)
        else:
            fsym = project_f(0.5 * (best_f + best_f[::-1]), best_n)
            ft = project_f(fsym + rng.normal(0.0, 0.05, best_n), best_n)
        ft, _ = cd_f(ft, best_n, 35, dcur, iters=18, rng=rng if r % 2 else None)
        ft, ht, ct = eval_f(ft, best_n)
        if ct < best_c:
            best_f, best_h, best_c, best_n = ft, ht, ct, best_n
        if r == 6 and time.time() < t_pert - 60:
            nt = best_n - 40 if best_n > 240 else best_n + 40
            if nt % 2 == 0 and 100 <= nt <= 1200:
                ft2 = interp_f(best_f, best_n, nt)
                ft2, _ = cd_f(ft2, nt, 70, min(t_pert - 5.0, time.time() + 90.0), iters=18)
                ft2, ht2, ct2 = eval_f(ft2, nt)
                if ct2 < best_c:
                    best_f, best_h, best_c, best_n = ft2, ht2, ct2, nt

    if time.time() < hard - 4:
        ft, _ = cd_f(best_f, best_n, 1000000, hard - 2.0, iters=24, rng=None)
        ft, ht, ct = eval_f(ft, best_n)
        if ct < best_c:
            best_f, best_h, best_c, best_n = ft, ht, ct, best_n

    best_h = project_h(best_h, best_n)
    best_c = float(np.max(np.correlate(best_h, 1.0 - best_h, mode="full")) * (2.0 / best_n))
    if not np.all((best_h >= 0.0) & (best_h <= 1.0)):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c = float(np.max(np.correlate(best_h, 1.0 - best_h, mode="full")) * (2.0 / best_n))
    return best_h, best_c, best_n
