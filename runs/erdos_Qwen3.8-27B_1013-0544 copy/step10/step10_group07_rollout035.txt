    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(np.max(c))
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
        cur = float(np.max(c))
        if cur < best_c:
            best_c = cur
    return h, float(np.max(c) * 2.0 / n)

    # correlate(g, w, 'full')[j + (n-1)] = sum_i g[i] * w[i - (j+(n-1)) + (len(w)-1)]

    grad = np.zeros(n)

        sw = softmax[k]
        if sw < 1e-15:
            continue
        # Part A: (1-h[p+d]) for p in [max(0,-d), min(n-1, n-1-d)]
        p_lo = max(0, -d)
        p_hi = min(n-1, n-1-d)
        if p_lo <= p_hi:
            grad[p_lo:p_hi+1] += sw * (1 - h[p_lo+d:p_hi+d+1])
        # Part B: -h[p-d] for p in [max(0,d), min(n-1, n-1+d)]
        # Wait: p-dk[k] = p-d, need 0<=p-d<n, so d<=p<=d+n-1
        p_lo2 = max(0, d)
        p_hi2 = min(n-1, d+n-1)
        if p_lo2 <= p_hi2:
            grad[p_lo2:p_hi2+1] -= sw * h[p_lo2-d:p_hi2-d+1]
    grad *= dx
    return val, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            r = 1.0 - h; t = r.sum()
            if t < 1e-15: break
            h = np.clip(h + (d / t) * r, 0.0, 1.0)
        else:
            t = h.sum()
            if t < 1e-15: break
            h = np.clip(h + (d / t) * h, 0.0, 1.0)
    return h

def cd_1d(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    m0, m1 = n - 1 - p, 2 * n - 2 - p
    ms = np.arange(m0, m1 + 1)
    jf = p + ms - (n - 1); jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n); vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf): a[vf] += 1.0 - h[jf[vf]]
    if np.any(vb): a[vb] -= h[jb[vb]]
    cb = c[ms].copy()
    di = np.where(ms == n - 1)[0]
    hd = len(di) > 0
    if hd:
        cd_ = cb[di[0]]; a[di] = 0.0
    ua = 0.0
    if m0 > 0: ua = max(ua, float(c[:m0].max()))
    if m1 < 2 * n - 2: ua = max(ua, float(c[m1 + 1:].max()))
    cm = float(c.max())
    nm = a < -1e-15; pm = a > 1e-15; zm = np.abs(a) <= 1e-15
    lo, hi = ua, cm
    for _ in range(50):
        L = (lo + hi) / 2.0
        xl, xh = 0.0, 1.0
        ok = True
        if np.any(zm) and np.any(cb[zm] > L + 1e-15): ok = False
        if ok and np.any(nm): xl = max(xl, float(((L - cb[nm]) / a[nm]).max()))
        if ok and np.any(pm): xh = min(xh, float(((L - cb[pm]) / a[pm]).min()))
        if xl > xh + 1e-14: ok = False

                if not (max(xl, 0.0) < r1 and min(xh, 1.0) > r2): ok = False
        if ok: hi = L
        else: lo = L
    L = hi; xl, xh = 0.0, 1.0
    if np.any(nm): xl = max(xl, float(((L - cb[nm]) / a[nm]).max()))
    if np.any(pm): xh = min(xh, float(((L - cb[pm]) / a[pm]).min()))
    xl = max(xl, 0.0); xh = min(xh, 1.0)
    if hd:
        cq = L - cd_ + h0 - h0 * h0; ds = 1.0 - 4.0 * cq
        if ds >= 0:

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline: break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12: continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                m0, m1 = n - 1 - p, 2 * n - 2 - p
                ms = np.arange(m0, m1 + 1)
                jf = p + ms - (n - 1); jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n); vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf): a[vf] += 1.0 - h[jf[vf]]
                if np.any(vb): a[vb] -= h[jb[vb]]
                di = np.where(ms == n - 1)[0]
                c[ms] += delta * a
                if len(di) > 0: c[ms[di[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved: break
    return h, float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def lse_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    ex = np.exp(sh - sh.max())
    sm = ex / ex.sum()
    val = (m + np.log(ex.sum()) / alpha) * dx
    dk = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        sw = sm[k]
        if sw < 1e-16: continue
        d = dk[k]
        p_lo = max(0, -d); p_hi = min(n - 1, n - 1 - d)
        if p_lo <= p_hi:
            grad[p_lo:p_hi + 1] += sw * (1.0 - h[p_lo + d:p_hi + d + 1])
        p_lo2 = max(0, d); p_hi2 = min(n - 1, d + n - 1)
        if p_lo2 <= p_hi2:
            grad[p_lo2:p_hi2 + 1] -= sw * h[p_lo2 - d:p_hi2 - d + 1]
    grad *= dx
    return val, grad

def lse_opt(h, n, alphas, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline: break
        try:
            def _fun(x, n_=n, a_=alpha):
                return lse_obj_grad(x, n_, a_)[0]
            def _grad(x, n_=n, a_=alpha):
                return lse_obj_grad(x, n_, a_)[1]
            res = minimize(_fun, best_h, jac=_grad, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-12})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)
