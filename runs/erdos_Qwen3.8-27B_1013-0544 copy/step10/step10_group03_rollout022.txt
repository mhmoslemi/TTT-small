import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a_diag = a[diag_idx[0]]
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    L_lo, L_hi = unaff, float(np.max(c))
    for _ in range(40):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        d_lo, d_hi = 0.0, 0.0
        feasible = True
        if np.any(neg_mask):
            d_lo = max(d_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if np.any(pos_mask):
            d_hi = min(d_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if feasible and has_diag:
            disc = a_diag * a_diag - 4.0 * (L - c_diag)
            if disc >= 0:
                r1 = (a_diag - np.sqrt(disc)) / 2.0
                r2 = (a_diag + np.sqrt(disc)) / 2.0
                left_ok = d_lo <= r1 and d_hi >= d_lo
                right_ok = d_hi >= r2 and d_lo <= d_hi
                if left_ok:
                    d_hi = min(d_hi, r1)
                elif right_ok:
                    d_lo = max(d_lo, r2)
                else:
                    feasible = False
        if not feasible or d_lo > d_hi + 1e-12:
            L_lo = L
        else:
            L_hi = L
    L = L_hi
    d_lo, d_hi = 0.0, 0.0
    if np.any(neg_mask):
        d_lo = max(d_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        d_hi = min(d_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(h0 + d_lo, 0.0)
    x_hi = min(h0 + d_hi, 1.0)
    if has_diag:
        disc = a_diag * a_diag - 4.0 * (L - c_diag)
        if disc >= 0:
            r1 = (a_diag - np.sqrt(disc)) / 2.0
            r2 = (a_diag + np.sqrt(disc)) / 2.0
            if x_lo <= r1 <= x_hi:
                x_hi = min(x_hi, r1)
            elif r2 >= x_lo:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def fast_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                diag_i = np.where(ms == (n - 1))[0]
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def topk_obj(x, n, k, beta):
    c = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argpartition(c, -k)[-k:]
    top = c[idx]
    m = top.max()
    shifted = beta * (top - m)
    mx = shifted.max()
    lse = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / beta
    return lse * (2.0 / n)

def eq_fun(x, n):
    return x.sum() - n / 2.0

def eq_jac(x, n):
    return np.ones(n)

def slsqp_topk(h, n, k, beta, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]
    try:
        res = minimize(topk_obj, best_h, args=(n, k, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    except Exception:
        pass
    return best_h, best_c

def structured_perturb(h, n, rng, mode):
    if mode == 0:
        return project_feasible(h + rng.normal(0, 0.01, n), n)
    elif mode == 1:
        f = np.linspace(0, 2, n)
        freq = rng.uniform(1, 5)
        amp = rng.uniform(0.02, 0.08)
        return project_feasible(h + amp * np.sin(2 * np.pi * freq * f / 2.0), n)
    elif mode == 2:
        edges = sorted(rng.randint(0, n, size=rng.randint(3, 8)))
        pert = np.zeros(n)
        for e in edges:
            w = rng.randint(5, 20)
            s = rng.uniform(-0.05, 0.05)
            pert[e:e+w] += s
        return project_feasible(h + pert, n)
    else:
        block = rng.randint(0, n)
        w = rng.randint(20, min(100, n // 4))
        s = rng.uniform(-0.1, 0.1)
        pert = np.zeros(n)
        pert[block:block+w] = s
        return project_feasible(h + pert, n)

def multi_scale_opt(h_in, n_in, deadline):

    for n_try in [200, 300, 400, 500, 600, 700, 800, 900, 1000, 1200, 1500]:

            h_o, c_o = fast_cd(h_o, n_try, 80, deadline)
            if c_o < best_c - 1e-9:
