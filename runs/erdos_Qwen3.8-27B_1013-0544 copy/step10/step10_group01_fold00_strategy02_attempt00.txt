   ```python
       x_new = np.linspace(0, 2, target_n)
       h_new = np.interp(x_new, x_old, parent)
       
       # Repair function preservation
       return project_feasible(h_new, target_n)
   
   def project_feasible(h, n):
       target_sum = n/2
       total = h.sum()
       if abs(total - target_sum) < 1e-5: 
           return h.clip(0, 1)
       elif total < target_sum:
           # Distribute extra value
           h += (target_sum-total)/n * np.ones_like(h)
       else:
           # Deduct excess value
           h -= (total-target_sum)/n * np.ones_like(h)
       return h.clip(0, 1)
   ```
