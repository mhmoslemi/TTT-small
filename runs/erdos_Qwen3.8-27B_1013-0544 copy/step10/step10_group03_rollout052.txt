    c = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    shifted = alpha * (c - c.max())
    exp_vals = np.exp(shifted - shifted.max())  # numerical stability
    weights = exp_vals / exp_vals.sum()
    W_total = weights.sum()  # = 1.0

            t += 1
            grad, _ = smooth_grad(h, n, alpha)
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * grad * grad
            m_hat = m / (1.0 - 0.9 ** t)
            v_hat = v / (1.0 - 0.999 ** t)
            h = h - lr_c * m_hat / (np.sqrt(v_hat) + 1e-8)
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            if t % 200 == 0:
                lr_c *= 0.9
        h = best_h
    return best_h, best_c

def solve_1d(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-13 or h0 > 1.0 - 1e-13:
        return h0
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            L_lo = L
            continue
        if np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            L_lo = L
            continue
        if has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    L_lo = L
                    continue
        L_hi = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if h0 - r1 <= r2 - h0:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def pw_const_init(k, n, rng):
    vals = rng.uniform(0.1, 0.9, k)
    reps = np.repeat(vals, n // k)
    rem = n - len(reps)
    if rem > 0:
        reps = np.concatenate([reps, np.tile(vals[:rem], 1)])
    h = reps[-n:]
    return project_feasible(h, n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            sh = alpha * (c - m)
            mx = sh.max()
            lse = mx + np.log(np.sum(np.exp(sh - mx)))
            return (m + lse / alpha) * (2.0 / n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        n = 700
        h_best = np.full(n, 0.5)
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: ADAM from parent
    if time.time() < deadline - 250:
        h1, c1 = adam_opt(h_best, n, deadline - time.time() - 250)
        if c1 < best_c:
            h_best, best_c = h1, c1

    # Phase 2: SLSQP refinement
    if time.time() < deadline - 200:
        h2, c2 = slsqp_opt(h_best, n, [10000, 50000], 300, deadline - 200)
        if c2 < best_c:
            h_best, best_c = h2, c2

    # Phase 3: Coordinate descent
    if time.time() < deadline - 150:
        h3, c3 = coord_descent(h_best, n, 300, deadline - 150)
        if c3 < best_c:
            h_best, best_c = h3, c3

    # Phase 4: Multi-n exploration with diverse inits
    n_list = [500, 600, 800, 900, 1000]
    for n_try in n_list:
        if time.time() > deadline - 150:
            break
        if n_try == n:
            continue
        if time.time() > deadline - 80:
            break
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        for h_start in [h_int, pw_const_init(10, n_try, rng), pw_const_init(30, n_try, rng)]:
            if time.time() > deadline - 80:
                break
            h_o, c_o = adam_opt(h_start, n_try, 25)
            h_o, c_o = coord_descent(h_o, n_try, 100, deadline - 80)
            if c_o < best_c - 1e-9:
                h_best, best_c = h_o, c_o
                n = n_try

    # Phase 5: Perturbation + CD
    n = len(h_best)
    for r in range(25):
        if time.time() > deadline - 60:
            break
        s = 0.002 + 0.008 * r
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        h_o, c_o = coord_descent(hp, n, 150, deadline - 60)
        if c_o < best_c - 1e-10:
            h_best, best_c = h_o, c_o

    # Phase 6: Binary push + CD
    if time.time() < deadline - 50:
        for _ in range(5):
            if time.time() > deadline - 40:
                break
            hb = h_best.copy()
            mask_high = hb > 0.5
            hb[mask_high] = np.minimum(1.0, hb[mask_high] + 0.08)
            hb[~mask_high] = np.maximum(0.0, hb[~mask_high] - 0.08)
            hb = project_feasible(hb, n)
            h_o, c_o = coord_descent(hb, n, 200, deadline - 40)
            if c_o < best_c - 1e-10:
                h_best, best_c = h_o, c_o

    # Phase 7: Final intensive CD
    if time.time() < deadline - 2:
        n = len(h_best)
        h_f, c_f = coord_descent(h_best, n, 500, deadline - 2)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
