def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:

        h = np.clip(h, 0.0, 1.0)

def snap_binary(h, target):

    if abs(diff) > 1e-10:

f(h) = max_k [corr_k(h)] * dx

where corr_k(h) = Σ_i h[i] * (1-h[i+k]) (with appropriate bounds on i).

This is a convex function of h (max of linear functions). We want to minimize it subject to 0 ≤ h[i] ≤ 1 and Σh[i] = n/2.

For convex optimization over a polytope, the optimum is at a face of the polytope (not necessarily a vertex). The active constraints at the optimum are:
- Some h[i] = 0 or h[i] = 1 (bound constraints)
- Σh[i] = n/2 (equality constraint)
- The active correlations (those achieving the max) are all equal (KKT condition)

So the non-binary optimum has some coordinates at 0 or 1, and the rest possibly in (0,1). The number of non-binary coordinates is at most the number of active correlations + 1 (by the KKT conditions).

This suggests that the non-binary solution is "mostly binary" with a small number of intermediate values. I can exploit this by:

2. Identify which coordinates to "free up" (make non-binary)
3. Optimize only those coordinates

With a = h, v = 1-h, N = M = n:

- Term where n = i: h[i]*(1-h[i+k]), derivative w.r.t. h[i] is (1-h[i+k])
  Valid when max(0,-k) ≤ i ≤ min(n-1, n-1-k)
- Term where n+k = i, i.e., n = i-k: h[i-k]*(1-h[i]), derivative w.r.t. h[i] is -(1-h[i-k]) = h[i-k]-1
  Valid when max(0,-k) ≤ i-k ≤ min(n-1, n-1-k), i.e., k ≤ i ≤ n-1-k+k = n-1 and i-k ≥ 0 and i-k ≤ n-1

So: ∂C[k+n-1]/∂h[i] = (1-h[i+k])·[valid1] + (h[i-k]-1)·[valid2]

where valid1: max(0,-k) ≤ i ≤ min(n-1, n-1-k) and 0 ≤ i+k < n
      valid2: 0 ≤ i-k < n and max(0,-k) ≤ i-k ≤ min(n-1, n-1-k)

For k ≥ 0:
valid1: 0 ≤ i ≤ n-1-k (so i+k ≤ n-1)
valid2: k ≤ i ≤ n-1 (so i-k ≥ 0 and i-k ≤ n-1-k)

For k < 0, let k = -|k|:
valid1: |k| ≤ i ≤ n-1 (so i+k = i-|k| ≥ 0)
valid2: 0 ≤ i ≤ n-1-|k| (so i-k = i+|k| ≤ n-1)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_max_corr(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr))

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def snap_binary(h, target):
    z = (h > 0.5).astype(np.float64)
    diff = int(round(target - z.sum()))
    if diff > 0:
        idx = np.where(z == 0)[0]
        order = np.argsort(h[idx])[::-1][:diff]
        z[idx[order]] = 1.0
    elif diff < 0:
        idx = np.where(z == 1)[0]
        order = np.argsort(h[idx])[:(-diff)]
        z[idx[order]] = 0.0
    return z

def make_thue_morse(n, seed=42):
    target = n // 2
    tm = np.zeros(n)
    for i in range(n):
        x, c = i, 0
        while x:
            c ^= (x & 1)
            x >>= 1
        tm[i] = c
    actual = int(tm.sum())
    if actual != target:
        rng = np.random.RandomState(seed)
        if actual > target:
            idx = np.where(tm == 1)[0]
            tm[idx[rng.choice(len(idx), actual-target, replace=False)]] = 0.0
        else:
            idx = np.where(tm == 0)[0]
            tm[idx[rng.choice(len(idx), target-actual, replace=False)]] = 1.0
    return tm

def random_binary(n, seed=42):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h

def binary_ls(h, t0, budget, seed=42, max_rounds=300, K=1000):

    n = len(h)
    cur = h.copy()
    cur_c5 = compute_max_corr(cur)
    trial = cur.copy()
    for rnd in range(max_rounds):
        if time.time() - t0 > budget - 20:
            break
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        no, nz = len(ones), len(zeros)
        if no == 0 or nz == 0:
            break
        best_c = cur_c5
        bi, bj = -1, -1
        for _ in range(K):
            i = int(ones[rng.randint(no)])
            j = int(zeros[rng.randint(nz)])
            trial[i] = 0.0
            trial[j] = 1.0
            c = compute_max_corr(trial)
            if c < best_c - 1e-15:
                best_c = c
                bi, bj = i, j
            trial[i] = 1.0
            trial[j] = 0.0
        if bi >= 0:
            cur[bi] = 0.0
            cur[bj] = 1.0
            cur_c5 = best_c
        else:
            break
    return cur, cur_c5

def binary_ils(h_init, t0, budget, seed=42, n_cycles=100):

    n = len(h_init)
    best_h = h_init.copy()
    best_c = compute_max_corr(h_init)
    for cyc in range(n_cycles):
        if time.time() - t0 > budget - 30:
            break
        h_ls, c_ls = binary_ls(best_h, t0, budget, seed=seed + cyc, max_rounds=200, K=800)
        if c_ls < best_c:

        if c_ls2 < best_c:

    return best_h, best_c

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def sml_obj(h_in, beta, dx):
    return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0/n), method='SLSQP',
                     bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except:
        return h0

def coord_descent(cur, target, n, sweeps, t0, budget):
    best_c5 = compute_c5(cur)
    deltas = (0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002)
    for sw in range(sweeps):
        if time.time() - t0 > budget - 5:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget - 3:
                break
            for d in deltas:
                if cur[j] + d < 0 or cur[j] + d > 1:
                    continue
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved:
            break
    return cur, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_max_corr(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr))

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def snap_binary(h, target):
    z = (h > 0.5).astype(np.float64)
    diff = int(round(target - z.sum()))
    if diff > 0:
        idx = np.where(z == 0)[0]
        order = np.argsort(h[idx])[::-1][:diff]
        z[idx[order]] = 1.0
    elif diff < 0:
        idx = np.where(z == 1)[0]
        order = np.argsort(h[idx])[:(-diff)]
        z[idx[order]] = 0.0
    return z

def make_thue_morse(n, seed=42):
    target = n // 2
    tm = np.zeros(n)
    for i in range(n):
        x, c = i, 0
        while x:
            c ^= (x & 1)
            x >>= 1
        tm[i] = c
    actual = int(tm.sum())
    if actual != target:
        rng = np.random.RandomState(seed)
        if actual > target:
            idx = np.where(tm == 1)[0]
            tm[idx[rng.choice(len(idx), actual - target, replace=False)]] = 0.0
        else:
            idx = np.where(tm == 0)[0]
            tm[idx[rng.choice(len(idx), target - actual, replace=False)]] = 1.0
    return tm

def random_binary(n, seed=42):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def binary_ls(h, t0, t_end, seed=42, max_rounds=400, K=800):
    rng = np.random.RandomState(seed)
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_max_corr(cur)
    trial = cur.copy()
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        no, nz = len(ones), len(zeros)
        if no == 0 or nz == 0:
            break
        best_c = cur_c5
        bi, bj = -1, -1
        k_use = K if rnd < 100 else max(400, K // 2)
        for _ in range(k_use):
            i = int(ones[rng.randint(no)])
            j = int(zeros[rng.randint(nz)])
            trial[i] = 0.0
            trial[j] = 1.0
            c = compute_max_corr(trial)
            if c < best_c - 1e-15:
                best_c = c
                bi, bj = i, j
            trial[i] = 1.0
            trial[j] = 0.0
        if bi >= 0:
            cur[bi] = 0.0
            cur[bj] = 1.0
            cur_c5 = best_c
        else:
            break
    return cur, cur_c5

def binary_ils(h_init, t0, t_end, seed=42, n_cycles=300):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    best_h = h_init.copy()
    best_c = compute_max_corr(h_init)
    for cyc in range(n_cycles):
        if time.time() > t_end:
            break
        h_ls, c_ls = binary_ls(best_h, t0, t_end, seed=seed + cyc, max_rounds=200, K=700)
        if c_ls < best_c:
            best_c, best_h = c_ls, h_ls.copy()
        pert = best_h.copy()
        npert = rng.randint(3, 9)
        for _ in range(npert):
            ones = np.where(pert == 1)[0]
            zeros = np.where(pert == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            pert[i] = 0.0
            pert[j] = 1.0
        h_ls2, c_ls2 = binary_ls(pert, t0, t_end, seed=seed + cyc + 500, max_rounds=150, K=600)
        if c_ls2 < best_c:
            best_c, best_h = c_ls2, h_ls2.copy()
    return best_h, best_c

def smooth_max(x, beta):
    m = np.max(x)
    xc = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xc)) / len(x)) / beta

def sml_obj(h_in, beta, dx):
    return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target
    except Exception:
        return h0

def coord_descent(cur, target, n, sweeps, t0, t_end):
    best_c5 = compute_c5(cur)
    deltas = (0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002)
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        improved = False
        for j in range(n):
            if time.time() > t_end:
                break
            for d in deltas:
                if cur[j] + d < -0.01 or cur[j] + d > 1.01:
                    continue
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved:
            break
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    n = 384

    if 'initial_h_values' in globals() and initial_h_values is not None:

    inits = [snap_binary(h_parent, target),

        if time.time() > t_bin_end:

        h_res, c_res = binary_ils(inits[si], t0, t_bin_end, seed=seed + si * 100, n_cycles=200)
        if c_res < best_c_bin:
            best_c_bin = c_res
            best_h_bin = h_res.copy()

    best_c5 = best_c_bin * 2.0 / n
    cur = best_h_bin.copy()
    bounds = Bounds(np.zeros(n), np.ones(n))

    rem = budget_s - (time.time() - t0)

            rem = budget_s - (time.time() - t0)

            if c < best_c5:

                best_h_bin = cur.copy()

    rem = budget_s - (time.time() - t0)

        cur, c = coord_descent(cur, target, n, 40, t0, t0 + rem - 10)
        if c < best_c5:
            best_c5 = c
            best_h_bin = cur.copy()

        cur2 = do_slsqp(cur, n, target, 3000, bounds, 4000, 1e-15)
        c2 = compute_c5(cur2)
        if c2 < best_c5:
            best_c5 = c2
            best_h_bin = cur2.copy()

    best_h_bin = project_h(best_h_bin, target)
    best_c5 = compute_c5(best_h_bin)
    return best_h_bin, best_c5, n
