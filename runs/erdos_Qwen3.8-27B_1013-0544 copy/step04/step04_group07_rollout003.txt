        r = rng.random()
        if r < 0.55:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            scale = T * 15.0 + 0.008
            d = rng.uniform(-scale, scale)
            v1 = h[i] + d
            v2 = h[j] - d
            if v1 < 0.0 or v1 > 1.0 or v2 < 0.0 or v2 > 1.0:
                T *= cool
                continue
            h_new = h.copy()
            h_new[i] = v1
            h_new[j] = v2
            c_new = compute_c5(h_new)
        elif r < 0.80:
            blk = rng.randint(2, min(8, n))
            start = rng.randint(0, n - blk + 1)
            h_new = h.copy()
            h_new[start:start + blk] += rng.uniform(-0.015, 0.015, blk)
            h_new = project_h(h_new, target)
            c_new = compute_c5(h_new)
        elif r < 0.90:
            i, j = rng.randint(0, n, 2)
            if i == j:
                continue
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            c_new = compute_c5(h_new)
        else:
            idx = rng.randint(n)
            d = rng.uniform(-0.04, 0.04)
            h_new = h.copy()
            h_new[idx] += d
            h_new = project_h(h_new, target)
            c_new = compute_c5(h_new)

        steps += 1
        delta = c_new - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            cur_c5 = c_new
            if c_new < best_c5 - 1e-14:
                best_h = h.copy()
                best_c5 = c_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 5000:
            h = best_h.copy()
            cur_c5 = best_c5
            T = max(T, 3e-4)
            stall = 0
            kick_count += 1
            if kick_count > 25:
                break

        T *= cool

    return best_h, best_c5

def coord_descent(cur, target, n, t0, deadline):
    best_c5 = compute_c5(cur)
    deltas = (0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002)
    while time.time() < deadline:
        improved = False
        for j in range(n):
            if time.time() > deadline - 0.5:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved:
            break
    return cur, best_c5

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(320, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    n_pts = len(h_parent)
    target = n_pts / 2.0
    bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))

    # Phase 1: SLSQP to enter a good basin (short)
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        cur = do_slsqp(best_h, n_pts, target, 150, bounds, 1500, 1e-14)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5 = cur.copy(), cur_c5

    # Phase 2: Main SA search (bulk of time)
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        sa_deadline = t0 + min(remaining - 25, budget_s * 0.55)
        cur, cur_c5 = sa_search(best_h, n_pts, target, t0, sa_deadline, seed)
        if cur_c5 < best_c5:
            best_h, best_c5 = cur.copy(), cur_c5

    # Phase 3: SLSQP polish with high beta
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        cur = do_slsqp(best_h, n_pts, target, 800, bounds, 4000, 1e-15)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5 = cur.copy(), cur_c5

    # Phase 4: Second SA round from improved point
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        sa_deadline = t0 + min(remaining - 15, budget_s * 0.25)
        cur, cur_c5 = sa_search(best_h, n_pts, target, t0, sa_deadline, seed + 1)
        if cur_c5 < best_c5:
            best_h, best_c5 = cur.copy(), cur_c5

    # Phase 5: Coordinate descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        cd_deadline = t0 + min(remaining - 5, 80)
        cur, cur_c5 = coord_descent(best_h, target, n_pts, t0, cd_deadline)
        if cur_c5 < best_c5:
            best_h, best_c5 = cur.copy(), cur_c5

    # Phase 6: Final SLSQP polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        best_h = do_slsqp(best_h, n_pts, target, 3000, bounds, 5000, 1e-15)
        best_c5 = compute_c5(best_h)

    # Phase 7: Tiny SA kick for last improvements
    remaining = budget_s - (time.time() - t0)
    if remaining > 8:
        sa_deadline = t0 + min(remaining - 2, 30)
        cur, cur_c5 = sa_search(best_h, n_pts, target, t0, sa_deadline, seed + 2)
        if cur_c5 < best_c5:
            best_h, best_c5 = cur.copy(), cur_c5

    # Phase 8: Final coord descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 5:
        cd_deadline = t0 + min(remaining - 1, 20)
        cur, cur_c5 = coord_descent(best_h, target, n_pts, t0, cd_deadline)
        if cur_c5 < best_c5:
            best_h, best_c5 = cur.copy(), cur_c5

    best_h = project_h(best_h, n_pts / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_pts
