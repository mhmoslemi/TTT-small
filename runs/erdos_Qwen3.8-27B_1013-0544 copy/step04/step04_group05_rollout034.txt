def cut_plane_step(h, target, dx, active_lags):
    n = len(h)
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    # Linearize O_k around h
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx  # O_k values
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    
    for lag in active_lags:
        g = lag_grad(h, lag)
        # O_k(h) ≈ vals[idx] + g^T (h_var - h)
        # vals[idx] + g^T * h_var - g^T * h <= t

        constraints.append(vals[idx] + g @ h_var - g @ h <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve()
        if h_var.value is not None:
            return np.clip(h_var.value, 0, 1)
    except Exception:
        pass
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_grad(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:-lag])
        g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g

def sum_sq_obj(h, L, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    top = np.argsort(vals)[::-1][:L]
    return np.sum(vals[top] ** 2)

def sum_sq_grad(h, L, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    top = np.argsort(vals)[::-1][:L]
    grad = np.zeros(n)
    for idx in top:
        lag = int(idx) - (n - 1)
        grad += 2.0 * vals[idx] * lag_grad(h, lag)
    return grad

        return h_in.sum() - target

    def con_j(h_in):
        return ones_vec

    try:
        res = minimize(obj, h, jac=obj_g, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con, 'jac': con_j}],
                       options={'maxiter': max_iter, 'ftol': 1e-18})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h.copy(), compute_c5(h)

        xs = np.clip(beta * (c - m), -400, 400)
