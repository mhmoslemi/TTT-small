For coordinate descent, the key insight is:
- Given all h[j] for j≠i fixed, c[k] = A_k + B_k * h[i] for some constants A_k, B_k

Let me think about the 1D minimax problem:
minimize max_k (A_k + B_k * t) subject to 0 ≤ t ≤ 1

The function f(t) = max_k (A_k + B_k * t) is convex in t (max of linear functions).
We can find the minimum by:
- If all B_k ≥ 0: f is non-decreasing, minimum at t=0
- If all B_k ≤ 0: f is non-increasing, minimum at t=1
- Otherwise: the minimum is where two "active" lines cross

To find the minimum of a convex piecewise linear function, we can use binary search on the subgradient, or simply evaluate at the breakpoints.

The breakpoints are where A_i + B_i * t = A_j + B_j * t, i.e., t = (A_j - A_i) / (B_i - B_j) for B_i ≠ B_j.

There are O(n²) breakpoints, but we only need the ones in [0,1]. We can find the minimum by:
1. Evaluating f at t=0 and t=1
2. If f(0) < f(1), the minimum is in [0, some value]
3. Use ternary search or binary search on the subgradient

Actually, for a convex piecewise linear function, we can use a simple approach:
- Compute f(0) and f(1)
- The subgradient at t is the set of B_k for k achieving the max
- At the optimum, 0 is in the subgradient, meaning there exist active k with B_k > 0 and active k with B_k < 0 (or B_k = 0)

A practical approach: use scipy.optimize.minimize_scalar with bounds=(0,1) on the piecewise linear function. Since it's convex, this will converge quickly.

But wait - we also need to maintain the sum constraint. If we change h[i] by δ, we need to change some other h[j] by -δ. 

One approach: do coordinate descent in pairs. Pick (i, j) and optimize h[i] and h[j] jointly (with h[i] + h[j] = constant = old h[i] + h[j]). This is a 2D problem:
minimize max_k (A_k + B_k * h[i] + C_k * h[j]) subject to h[i] + h[j] = s, 0 ≤ h[i], h[j] ≤ 1

Since h[j] = s - h[i], this reduces to:
minimize max_k (A_k + (B_k - C_k) * h[i] + C_k * s) subject to max(0, s-1) ≤ h[i] ≤ min(1, s)

This is again a 1D convex problem!

So the algorithm is:
1. For each pair (i, j):
   a. Compute the 1D minimax problem
   b. Solve it (ternary search or exact)
   c. Update h[i] and h[j]

But O(n²) pairs is too many. Instead, we can cycle through pairs in a fixed order or use random pairs.

Actually, a simpler approach that still works well:
1. For each i (in order):
   a. Compute B_k for this i (how c[k] changes with h[i])
   b. Solve the 1D minimax to find optimal h[i]
   c. Update h[i] and redistribute the sum difference evenly (or to a specific j)

The redistribution step is what makes it approximate, but it should still converge.

def coord_descent(h, n, deadline, max_sweeps=100):
    dx = 2.0 / n
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                    break
            # Compute B_k: change in c[k] per unit change in h[i]
            # c[k] = sum_j h[j] * (1-h[j-(k-n+1)]) * dx
            # h[i] appears as:
            #   First factor: h[i] * (1-h[i-(k-n+1)]) for valid k
            #   Second factor (in 1-h): h[j] * (-1) * [j-(k-n+1)==i] for valid j, k
            # So dc[k]/dh[i] = dx * [(1-h[i-(k-n+1)]) - h[j] where j = i + k - n + 1]
            
            # Simpler: just compute c with h[i]=0 and h[i]=1
            h[i] = 0
            c0 = np.correlate(h, 1-h, mode='full') * dx
            h[i] = 1
            c1 = np.correlate(h, 1-h, mode='full') * dx
            h[i] = 0  # temporary
            B = c1 - c0  # B_k = change per unit h[i]
            A = c0  # c[k] when h[i]=0 (but h[i] is actually 0 here, so this is correct)
            
            # 1D minimax: min_t max_k (A_k + B_k * t) for t in [0,1]
            # Use ternary search (convex function)
            def f(t):
                return np.max(A + B * t)
            
            from scipy.optimize import minimize_scalar
            res = minimize_scalar(f, bounds=(0, 1), method='bounded',
                                  options={'xatol': 1e-10})
            t_opt = res.x
            
            h[i] = t_opt
            # Project to maintain sum
            diff = target - h.sum()
            if abs(diff) > 1e-14:
                # Redistribute to non-bound variables
                mask = (h > 1e-10) & (h < 1 - 1e-10)
                if mask.sum() > 0:
                    h[mask] += diff / mask.sum()
                    h = np.clip(h, 0, 1)
        
        c = compute_c5(h, n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            improved = True
        if not improved:
            break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def slsqp_step(h, n, alpha, maxiter, deadline):
    target = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        return smooth_max(c, alpha)
    
    def cons_fun(x):
        return x.sum() - target
    
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, deadline, max_sweeps=300, t_n=21):
    dx = 2.0 / n
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    t_vals = np.linspace(0.0, 1.0, t_n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            h_save = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1 - h, mode='full') * dx
            h[i] = 1.0
            c1 = np.correlate(h, 1 - h, mode='full') * dx
            B = c1 - c0
            A = c0
            h[i] = h_save
            
            # Find best t in [0,1] minimizing max_k(A_k + B_k*t)
            # Quick: evaluate on grid
            best_t = h_save
            best_val = np.max(A + B * h_save)
            for t in t_vals:
                val = np.max(A + B * t)
                if val < best_val:
                    best_val = val
                    best_t = t
            
            if abs(best_t - h_save) > 1e-8:
                h[i] = best_t
                diff = target - h.sum()
                if abs(diff) > 1e-12:
                    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
                    cnt = mask.sum()
                    if cnt > 0:

        c = compute_c5(h, n)
        if c < best_c - 1e-12:

            best_h = h.copy()
        # Early stopping if no improvement for several sweeps
        elif best_c - c < -1e-14:

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def slsqp_step(h, n, alpha, maxiter, deadline):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        return smooth_max(c, alpha)
    def cons_fun(x):
        return x.sum() - target
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, deadline, max_sweeps=300, t_n=21):
    dx = 2.0 / n
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    t_vals = np.linspace(0.0, 1.0, t_n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            h_save = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1 - h, mode='full') * dx
            h[i] = 1.0
            c1 = np.correlate(h, 1 - h, mode='full') * dx
            B = c1 - c0
            A = c0
            h[i] = h_save
            best_t = h_save
            best_val = np.max(A + B * h_save)
            for t in t_vals:
                val = np.max(A + B * t)
                if val < best_val:
                    best_val = val
                    best_t = t
            if abs(best_t - h_save) > 1e-8:
                h[i] = best_t
                diff = target - h.sum()
                if abs(diff) > 1e-12:
                    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
                    cnt = mask.sum()
                    if cnt > 0:
                        h[mask] = np.clip(h[mask] + diff / cnt, 0, 1)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break
    return best_h, best_c

def binary_local(h0, n, deadline, max_iter=5000):
    target = n // 2
    idx = np.argsort(-h0)
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    best_c = compute_c5(h, n)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    rng = np.random.RandomState(456)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        found = False

        for _ in range(min(n_attempts, 500)):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            h[i], h[j] = 1.0, 0.0
            c = compute_c5(h, n)
            if c < best_c - 1e-12:
                best_c = c
                ones.remove(j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(j)
                found = True
                stall = 0
                break
            else:
                h[i], h[j] = 0.0, 1.0
        if not found:
            stall += 1
            if stall > 10:
                break
    return h, best_c

    h_p = h + sigma * rng.randn(n)
