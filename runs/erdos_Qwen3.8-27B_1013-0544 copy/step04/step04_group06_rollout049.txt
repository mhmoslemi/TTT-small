import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    current_sum = h.sum()
    if abs(current_sum - target) < 1e-10:
        return h
    # Binary search for shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_lag(h, lag):
    """Gradient of correlation at given lag (actual lag, not index)."""
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n-lag]
    else:
        m = -lag
        g[:n-m] -= h[m:]
        g[m:] += (1.0 - h[:n-m])
    return g

def soft_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = full_corr(h)
    m = corr.max()
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    se = e.sum()
    val = (m + np.log(se / len(corr)) / beta) * dx
    # Gradient
    w = e / se
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        if w[j] < 1e-10:
            continue
        lag = j - (n - 1)
        g += w[j] * grad_lag(h, lag)
    return val, g * dx

def slsqp_fast(h_init, n, beta, maxiter=1000):
    target = n / 2.0
    bounds = [(0, 1)] * n
    
    def obj(x):
        return soft_max_obj(np.clip(x, 0, 1), beta)[0]
    
    def grad(x):
        return soft_max_obj(np.clip(x, 0, 1), beta)[1]
    
    def eq(x):
        return np.sum(x) - target
    
    cons = [{'type': 'eq', 'fun': eq, 'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_sum(res.x, target)

def subgrad_momentum(h, target, max_iter, t0, budget, n_active=8, momentum=0.5):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    v = np.zeros(n)  # momentum
    no_improve = 0
    
    for it in range(max_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = full_corr(best)
        m = corr.max()

        threshold = m - 0.02 / (2.0/n)  # 0.02 in C5 units
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(corr)[::-1][:n_active]
        if len(active) > 3 * n_active:
            active = np.argsort(corr)[::-1][:n_active]
        
        g = np.zeros(n)
        for idx in active:
            lag = int(idx) - (n - 1)
            g += grad_lag(best, lag)
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        # Try multiple step sizes
        improved = False
        for step in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n-lag]
    else:
        m = -lag
        g[:n-m] -= h[m:]
        g[m:] += (1.0 - h[:n-m])
    return g

def slsqp_opt(h, n, beta, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m0 = corr.max()
    
    def obj(x):
        xc = np.clip(x, 0, 1)
        c = np.correlate(xc, 1.0 - xc, mode="full")
        mm = c.max()
        sh = np.clip(beta * (c - mm), -500, 500)
        e = np.exp(sh)
        return (mm + np.log(e.sum() / len(c)) / beta) * (2.0 / n)
    
    def grad(x):
        xc = np.clip(x, 0, 1)
        c = np.correlate(xc, 1.0 - xc, mode="full")
        mm = c.max()
        sh = np.clip(beta * (c - mm), -500, 500)
        e = np.exp(sh)
        w = e / e.sum()
        g = np.zeros(n)
        for j in range(len(c)):
            if w[j] < 1e-12:
                continue
            lag = j - (n - 1)
            g += w[j] * grad_lag(xc, lag)
        return g * (2.0 / n)
    
    def eq(x):
        return x.sum() - target
    def eqj(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq, 'jac': eqj}]
    res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_sum(np.clip(res.x, 0, 1), target)
