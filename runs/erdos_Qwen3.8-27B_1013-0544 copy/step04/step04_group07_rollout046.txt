from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, beta, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # Use log-sum-exp for numerical stability
    shifted = np.clip(beta * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta) * (2.0 / n)

def sum_constraint(x, target):
    return np.sum(x) - target

def sum_jac(x, target):
    return np.ones(x.shape)

def slsqp_optimize(h0, n, target, beta, maxiter, ftol):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(smooth_max_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        if r.success or r.fun < smooth_max_obj(h0, beta, n):
            return project_h(r.x, target)
    except Exception:
        pass
    return h0

    corr = get_corr(h)
    n_corr = len(corr)
    mc = np.max(corr)

    thresh = mc - 1e-6 * (mc + 1e-10)

        active = active[np.argsort(corr[active])[::-1][:top_k]]

        else:

                g[i] += h[i + la] - (1 - h[i])
    return g / len(active)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, beta, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta) * (2.0 / n)

def sum_con(x, t):
    return np.sum(x) - t

def sum_jac(x, t):
    return np.ones(x.shape)

def slsqp_opt(h0, n, target, beta, maxiter, ftol):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(smooth_max_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        c_new = compute_c5(r.x)
        c_old = compute_c5(h0)
        if c_new < c_old:
            return project_h(r.x, target)
    except Exception:
        pass
    return h0

def vec_subgrad(h, n, top_k=25):
    corr = get_corr(h)
    mc = np.max(corr)
    thr = mc - 1e-3 * (mc + 1e-10)
    active = np.where(corr >= thr)[0]
    if len(active) > top_k:
        active = active[np.argsort(corr[active])[::-1][:top_k]]
    g = np.zeros(n)
    for idx in active:
        lag = int(idx) - (n - 1)
        if lag >= 0:
            m = n - lag
            g[:m] += (1.0 - h[lag:]) - h[:m]
        else:
            la = -lag
            m = n - la
            g[:m] += h[la:] - (1.0 - h[:m])
    return g / max(len(active), 1)

def nesterov_opt(h0, n, target, n_iter, step0, t0, budget):
    h = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    step = step0
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        g = vec_subgrad(y, n, top_k=30)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        h_new = project_h(y - step * g, target)
        mu = (1.0 - np.sqrt(1.0 + 4.0 / ((it + 1) ** 2))) / 2.0
        y = np.clip(h_new + mu * (h_new - h_prev), 0, 1)
        h_prev, h = h, h_new
        c = compute_c5(h)
        if c < best_c5 - 1e-16:
            best_h, best_c5, stall = h.copy(), c, 0
        else:
            stall += 1
            if stall % 200 == 0:
                step *= 0.8
    return best_h, best_c5

def coord_descent(h0, n, target, sweeps, t0, budget):
    h = h0.copy()
    best_c5 = compute_c5(h)
    deltas = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005]
    for sw in range(sweeps):
        if time.time() - t0 > budget - 2:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget - 1:
                break
            for d in deltas:
                trial = h.copy()
                trial[j] = np.clip(trial[j] + d, 0.0, 1.0)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-16:
                    h, best_c5, improved = trial, c, True
                    break
        if not improved:
            break
    return h, best_c5

def pair_descent(h0, n, target, sweeps, t0, budget, rng):
    h = h0.copy()
    best_c5 = compute_c5(h)
    for sw in range(sweeps):
        if time.time() - t0 > budget - 2:
            break
        improved = 0
        for _ in range(max(n // 2, 50)):
            if time.time() - t0 > budget - 1:
                break
            j, k = rng.choice(n, 2, replace=False)
            for d in (0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001):
                trial = h.copy()
                trial[j] = np.clip(trial[j] + d, 0.0, 1.0)
                trial[k] = np.clip(trial[k] - d, 0.0, 1.0)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-16:
                    h, best_c5 = trial, c
                    improved += 1
                    break
        if improved == 0:
            break
    return h, best_c5

def snap_binary(h, target):
    z = (h > 0.5).astype(float)
    diff = target - z.sum()
    if diff > 0.5:
        zeros = np.where(z == 0)[0]
        order = zeros[np.argsort(h[zeros])[::-1]]
        z[order[:int(diff)]] = 1.0
    elif diff < -0.5:
        ones = np.where(z == 1)[0]
        order = ones[np.argsort(h[ones])]
        z[order[:int(-diff)]] = 0.0
    return project_h(z, target)

def resample_to(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def multi_restart_optimize(h0, n, target, n_restarts, t0, budget, rng):
    best_h, best_c5 = h0.copy(), compute_c5(h0)
    for r in range(n_restarts):
        if time.time() - t0 > budget - 10:
            break
        noise = rng.randn(n) * 0.02
        h_try = project_h(h0 + noise, target)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            h_try = slsqp_opt(h_try, n, target, 5000, 2000, 1e-15)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_h, best_c5 = h_try.copy(), c_try
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
