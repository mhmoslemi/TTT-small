def cutting_plane(h0, n, target, t0, budget, max_rounds=20):
    cur = h0.copy()
    active_lags = set()
    # Initialize with top 5 lags
    c = compute_lags(cur)
    top = np.argsort(c)[::-1][:5]
    for idx in top:
        active_lags.add(int(idx) - (n - 1))
    
    for round in range(max_rounds):
        if budget - (time.time() - t0) < 10:
            break
        
        # Solve QP: maximize sum of autocorrelations at active lags
        def obj(h):
            total = 0.0
            for lag in active_lags:
                if 0 <= lag < n:
                    total += np.sum(h[:n-lag] * h[lag:])
                elif -n < lag < 0:
                    la = -lag
                    total += np.sum(h[la:] * h[:n-la])
            return -total / len(active_lags)  # minimize negative
        
        try:
            res = minimize(obj, cur, method='L-BFGS-B',
                          bounds=[(0,1)]*n,
                          options={'maxiter': 2000, 'gtol': 1e-10})
            cur = project_h(res.x, target)
        except:
            pass

        c = compute_lags(cur)
        top10 = np.argsort(c)[::-1][:10]

        active_lags = active_lags | new_lags
