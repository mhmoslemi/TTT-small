import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
        else:
            h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def obj_and_grad(h_in, beta, n):
    dx = 2.0 / n
    h = h_in
    c = 1.0 - h
    corr = np.correlate(h, c, mode="full")
    
    m = np.max(corr)
    w = np.exp(np.clip(beta * (corr - m), -500, 500))
    w /= w.sum()
    
    obj = (m + np.log(np.sum(np.exp(np.clip(beta * (corr - m), -500, 500)))) / beta) * dx
    # Simpler:
    obj = smooth_max(corr, beta) * dx
    
    # Gradient
    grad = np.zeros(n)
    for j in range(n):
        # Term1: sum_{s=j-(n-1)}^{j} w[n-1+s] * c[j-s]
        s_vals = np.arange(j - (n-1), j + 1)
        w_idx = n - 1 + s_vals
        c_idx = j - s_vals
        # All indices should be valid by construction
        grad[j] += np.dot(w[w_idx], c[c_idx])
        
        # Term2: sum_{s=-j}^{n-1-j} w[n-1+s] * h[j+s]
        s_vals2 = np.arange(-j, n - j)
        w_idx2 = n - 1 + s_vals2
        h_idx2 = j + s_vals2
        grad[j] -= np.dot(w[w_idx2], h[h_idx2])
    
    grad *= dx
    return obj, grad

def do_slsqp(h0, n, target, beta, maxiter=5000, ftol=1e-14):
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def fun_and_jac(x):
        h = project_h(x, target)
        return obj_and_grad(h, beta, n)
    
    # Actually, SLSQP doesn't handle projection well. Let me use equality constraint instead.
    def fun(x):
        return obj_and_grad(x, beta, n)[0]
    
    def jac(x):
        return obj_and_grad(x, beta, n)[1]
    
    def eq_fun(x):
        return np.sum(x) - target
    
    def eq_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        r = minimize(fun, h0, jac=jac, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.05, -0.05, 0.03, -0.03, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for sw in range(max_sweeps):
        if time.time() - t0 > budget - 3:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget - 2:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-15:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved:
            break
    return cur, best_c5
