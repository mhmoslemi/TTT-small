def grad_of_lag(h, lag):
    """Compute gradient of sum_i h[i]*(1-h[i+lag]) with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        # g = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
        # dg/dh[i] for i in [0, n-1-lag]: 1 - h[i+lag]
        # dg/dh[i+lag] for i in [0, n-1-lag]: -h[i]
        for i in range(n - lag):
            grad[i] += (1 - h[i + lag])
            grad[i + lag] += (-h[i])
    else:
        la = -lag
        for i in range(la, n):
            grad[i] += (1 - h[i - la])
            grad[i - la] += (-h[i])
    return grad

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=8000, ftol=1e-15):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def grad_of_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] += (-h[:n - lag])
    else:
        la = -lag
        if la < n:
            grad[la:] += (1.0 - h[:n - la])
            grad[:n - la] += (-h[la:])
    return grad

def balance_max(h, target, n, t0, budget, n_iter=500, top_k=6):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    step = 0.03
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[top_idx]
        temp = 0.01
        w = np.exp(top_vals / temp)
        w /= w.sum()
        grad = np.zeros(n)
        for kk in range(top_k):
            if w[kk] < 1e-8:
                continue
            lag = int(top_idx[kk]) - (n - 1)
            grad += w[kk] * grad_of_lag(cur, lag)
        step = 0.03 * (1.0 - 0.5 * it / max(n_iter, 1))
        trial = project_h(cur - step * grad, target)
        c = compute_c5(trial)
        if c < best_c5 - 1e-15:
            cur = trial
            best_c5 = c
        else:
            step *= 0.5
            trial = project_h(cur - step * grad, target)
            c = compute_c5(trial)
            if c < best_c5 - 1e-15:
                cur = trial
                best_c5 = c
            else:
                step *= 0.5
    return cur, best_c5

def targeted_coord_descent(h, target, n, t0, budget, max_sweeps=100):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.04, -0.04, 0.025, -0.025, 0.015, -0.015, 0.01, -0.01]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:8]
        critical_coords = set()
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if 0 <= lag < n:
                critical_coords.update(range(max(0, n - lag)))
            elif -n < lag < 0:
                la = -lag
                critical_coords.update(range(la, n))
        coord_list = list(critical_coords)
        rng_local = np.random.default_rng(sw + 42)
        rng_local.shuffle(coord_list)
        improved = False
        for j in coord_list:
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0.0, 1.0)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c
                    improved = True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

def block_exchange(h, target, n, t0, budget, n_tries=5000):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    rng = np.random.default_rng(12345)
    block_sizes = [4, 8, 12, 16, 24, 32, 48]
    for t in range(n_tries):
        if budget - (time.time() - t0) < 2:
            break
        bs = int(block_sizes[rng.integers(len(block_sizes))])
        if 2 * bs >= n:
            continue
        i = int(rng.integers(0, n - 2 * bs + 1))
        j = int(rng.integers(0, n - 2 * bs + 1))
        if abs(i - j) < bs + 1:
            continue
        trial = cur.copy()
        tmp = trial[i:i+bs].copy()
        trial[i:i+bs] = trial[j:j+bs]
        trial[j:j+bs] = tmp
        trial = project_h(trial, target)
        c = compute_c5(trial)
        if c < best_c5 - 1e-15:
            cur = trial
            best_c5 = c
    return cur, best_c5

def pair_swap(h, target, n, t0, budget, n_tries=10000):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    rng = np.random.default_rng(9999)
    for t in range(n_tries):
        if budget - (time.time() - t0) < 1:
            break
        a = int(rng.integers(n))
        b = int(rng.integers(n))
        if a == b:
            continue
        trial = cur.copy()
        trial[a], trial[b] = trial[b], trial[a]
        trial = project_h(trial, target)
        c = compute_c5(trial)
        if c < best_c5 - 1e-15:
            cur = trial
            best_c5 = c
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 384
        h_parent = np.zeros(n_init)
        h_parent[:n_init // 2] = 1.0

    n = len(h_parent)
    target = n / 2.0
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at moderate beta
    for beta in [300, 1000, 3000]:
        remaining = budget_s - (time.time() - t0)

            break
        best_h = do_slsqp(best_h, n, target, beta, maxiter=6000)

    # Phase 3: SLSQP high beta
    remaining = budget_s - (time.time() - t0)

        for beta in [8000, 20000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            best_h = do_slsqp(best_h, n, target, beta, maxiter=8000)
            best_c5 = compute_c5(best_h)

    # Phase 4: Targeted coordinate descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        best_h, best_c5 = targeted_coord_descent(best_h, target, n, t0, budget_s, max_sweeps=80)

    # Phase 5: Block exchanges
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        best_h, best_c5 = block_exchange(best_h, target, n, t0, budget_s, n_tries=8000)

    # Phase 6: Additional balance after structural moves
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        best_h, best_c5 = balance_max(best_h, target, n, t0, budget_s, n_iter=300, top_k=8)

    # Phase 7: SLSQP again
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        for beta in [10000, 30000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            best_h = do_slsqp(best_h, n, target, beta, maxiter=6000)
            best_c5 = compute_c5(best_h)

    # Phase 8: Pair swaps (fine-grained rearrangement)
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        best_h, best_c5 = pair_swap(best_h, target, n, t0, budget_s, n_tries=8000)
        best_h, best_c5 = balance_max(best_h, target, n, t0, budget_s, n_iter=200, top_k=5)

    # Phase 9: Try higher resolution
    remaining = budget_s - (time.time() - t0)
    for n_hr in [480, 640]:
        if remaining < 50:
            break
        target_hr = n_hr / 2.0
        cur_hr = project_h(resample_fn(best_h, n_hr), target_hr)
        for beta in [1000, 5000, 15000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            cur_hr = do_slsqp(cur_hr, n_hr, target_hr, beta, maxiter=5000)
            cur_hr, c_hr = balance_max(cur_hr, target_hr, n_hr, t0, budget_s, n_iter=150, top_k=5)
            if c_hr < best_c5:
                best_h = cur_hr
                best_c5 = c_hr
                n = n_hr
                target = target_hr

    # Phase 10: Multi-start perturbation
    remaining = budget_s - (time.time() - t0)
    rng = np.random.default_rng(seed + 200)
    for att in range(6):
        if remaining < 25:

        scale = 0.015 * (0.7 ** att)
        hp = best_h + rng.standard_normal(n) * scale
        hp = project_h(hp, target)
        hp = do_slsqp(hp, n, target, 5000, maxiter=4000)
        hp, c_try = balance_max(hp, target, n, t0, budget_s, n_iter=150, top_k=5)
        c_try = compute_c5(hp)
        if c_try < best_c5:
            best_h = hp
            best_c5 = c_try

    # Phase 11: Final polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        best_h, best_c5 = balance_max(best_h, target, n, t0, budget_s, n_iter=300, top_k=8)
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        best_h, best_c5 = targeted_coord_descent(best_h, target, n, t0, budget_s, max_sweeps=60)
    remaining = budget_s - (time.time() - t0)
    if remaining > 8:
        best_h = do_slsqp(best_h, n, target, 40000, maxiter=8000)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
