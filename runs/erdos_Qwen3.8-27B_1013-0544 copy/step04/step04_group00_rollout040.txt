```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def gradient_lse(h, n, alpha):
    """Gradient of (1/alpha)*log(sum exp(alpha*g)) where g = correlate(h,1-h,'full')*dx"""
    g = compute_all_g(h, n)
    # Softmax of alpha * g
    ag = alpha * g
    ag_max = ag.max()
    w = np.exp(ag - ag_max)
    w /= w.sum()
    
    # grad[j] = sum_k w[k] * dg_k/dh[j]
    # = sum_{t=0}^{n-1} w[n-1+j-t] * (1-2h[t])
    a = 1 - 2 * h
    js = np.arange(n)[:, None]
    ts = np.arange(n)[None, :]
    idx = (n - 1) + js - ts
    vals = w[idx]
    grad = vals @ a
    return grad, g.max()

def lse_optimize(h, n, alpha, step, max_iter, deadline):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        grad, gmax = gradient_lse(h, n, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        s = step
        improved = False
        for _ in range(30):

            c_new = compute_c5(h_new, n)
            if c_new < gmax - 1e-12:

                improved = True
                if c_new < best_c:

                best_h = h_new.copy()
                h = h_new
    
    return best_h, best_c

def momentum_optimize(h, n, alpha, deadline, max_iter=50000):
    """Momentum gradient descent on LSE-smoothed objective"""
    v = np.zeros(n)
    mu = 0.9
    lr = 2.0 / (alpha + 1)  # initial learning rate
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        grad, gmax = gradient_lse(h, n, alpha)
        v = mu * v - lr * grad
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

            v *= 0.5
            lr *= 0.9
            h_new = project_feasible(h + v, n)

                best_c = min(best_c, compute_c5(h_new, n))
                best_h = h_new.copy()
            h = h_new
        else:
            h = h_new
    
    return best_h, best_c

def binary_search(h, n, deadline, max_sweeps=100):
    """Greedy binary search for good binary solutions"""
    target = int(n // 2)
    # Start from rounded h
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    
    ones = set(np.where(h_bin == 1)[0])
    zeros = set(np.where(h_bin == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try each zero position
        for i in sorted(zeros):
            if time.time() > deadline:
                break
            best_j = None
            best_c_trial = best_c
            for j in sorted(ones):
                h_bin[i], h_bin[j] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                if c < best_c_trial:
                    best_c_trial = c
                    best_j = j
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0

                h_bin[i] = 1.0
                h_bin[best_j] = 0.0
                ones.discard(best_j)
                ones.add(i)
                zeros.discard(i)
                zeros.add(best_j)
                best_c = best_c_trial
                improved = True
                break  # restart sweep
        
        if not improved:
            break
    
    return h_bin, best_c

def local_refine(h, n, deadline, delta_list=None):

    if delta_list is None:
```

Second sum = sum_{u=0}^{n-1} w[n-1+j-u] * h[u]

For n=2, j=0: sum_{u=0}^{1} w[1-u] * h[u] = w[1]*h[0] + w[0]*h[1] ✓

So the correct formula is:
grad[j] = sum_{t=0}^{n-1} w[n-1-j+t] * (1-h[t]) - sum_{u=0}^{n-1} w[n-1+j-u] * h[u]

These are DIFFERENT index patterns! The first uses w[n-1-j+t] (increasing with t) and the second uses w[n-1+j-u] (decreasing with u).

Let me combine: rename t=u in both:
grad[j] = sum_{t=0}^{n-1} w[n-1-j+t] * (1-h[t]) - sum_{t=0}^{n-1} w[n-1+j-t] * h[t]

```python
    # Second term: for each j, we need w[j : j+n+1] (length n) dotted with h[::-1]...

    js = np.arange(n)[:, None]  # (n, 1)
    ts = np.arange(n)[None, :]  # (1, n)

    # Second term index: n-1+j-u  (where u is the dummy index, same as ts)
    idx2 = (n-1) + js - ts  # shape (n, n)
```

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def gradient_weighted(h, w, n):
    js = np.arange(n)[:, None]
    ts = np.arange(n)[None, :]
    idx1 = (n - 1) - js + ts
    idx2 = (n - 1) + js - ts
    return w[idx1] @ (1.0 - h) - w[idx2] @ h

def lse_iter(h, n, alpha, lr, max_iter, deadline, momentum=True):
    h = h.copy()
    v = np.zeros(n) if momentum else None
    mu = 0.85
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stagn = 0
    base_lr = lr
    js_arr = np.arange(n)[:, None]
    ts_arr = np.arange(n)[None, :]
    idx1 = (n - 1) - js_arr + ts_arr
    idx2 = (n - 1) + js_arr - ts_arr
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        gmax = g.max()
        ag = alpha * g
        am = ag.max()
        ex = np.exp(ag - am)
        ws = ex.sum()
        if ws < 1e-300:
            break
        w = ex / ws
        a = 1.0 - 2.0 * h
        grad = w[idx1] @ (1.0 - h) - w[idx2] @ h
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        if momentum:
            v = mu * v - lr * grad
            h_new = project_feasible(h + v, n)
        else:
            h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stagn = 0
        else:
            stagn += 1
            if stagn > 50:
                lr *= 0.85
                if momentum:
                    v *= 0.3
                stagn = 0
                if lr < base_lr * 1e-4:
                    break
        h = h_new
    return best_h, best_c

def binary_local(h, n, deadline, max_sweeps=50):
    target = int(n // 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = sorted(np.where(h_bin == 1)[0])
    zeros = sorted(np.where(h_bin == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            best_j = None
            bc = best_c
            for j in ones:
                h_bin[i], h_bin[j] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                if c < bc:
                    bc = c
                    best_j = j
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
            if best_j is not None:
                h_bin[i] = 1.0
                h_bin[best_j] = 0.0
                ones.remove(best_j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(best_j)
                ones.sort()
                zeros.sort()
                best_c = bc
                improved = True
                break
        if not improved:
            break
    return h_bin, best_c

def coord_refine(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005]
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(50):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                for s in (1, -1):
                    nv = h[i] + s * d
                    if 0.0 <= nv <= 1.0:
                        h[i] = nv
                        h2 = project_feasible(h, n)
                        c = compute_c5(h2, n)
                        if c < best_c:
                            best_c = c
                            h = h2
                            improved = True
                        else:
                            h[i] = nv - s * d
            if not improved:
                break
    return h, best_c

        xo = np.linspace(0, 2, n)

        if c < best_c:

            best_h = hh.copy()

            best_h = hb2.copy()
    # Phase 5: Coordinate refinement
    if time.time() < dl - 3:
        best_h, best_c = coord_refine(best_h, n, dl)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
