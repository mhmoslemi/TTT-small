```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') 
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_corr(h, n):
    """Full correlation array, length 2n-1."""
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_grad_max_c(h, n):

    Compute gradient of max_k c_k.
    Returns (grad, max_val, active_shifts)

def projected_gradient(h, n, deadline, max_iter=50000):
    """
    Nesterov accelerated projected gradient descent.
    Minimizes max_k c_k subject to sum(h)=n/2, 0<=h<=1.
    """
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    beta = 0.9  # momentum
    v = np.zeros(n)
    step = 0.1  # initial step size
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Nesterov lookahead
        h_nest = h + beta * v
        h_nest = np.clip(h_nest, 0, 1)
        h_nest = project_feasible(h_nest, n)
        
        # Compute gradient
        grad, c_val, _ = compute_grad_max_c(h_nest, n)
        
        # Project gradient onto sum=0 subspace
        grad -= np.mean(grad)
        
        # Update momentum
        v = beta * v - step * grad
        
        # Proposed new point
        h_new = h + v
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        
        # Evaluate

            no_improve = 0

        else:

            step *= 0.95  # decrease step
            if step < 1e-8:
                break
        
        h = h_new
        
        if no_improve > 500:
```

```python
def fast_pair_search_v2(h, n, deadline, step_sizes):
    """Fast pair search using incremental correlation updates."""
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx  # length 2n-1
    # corr[m] corresponds to shift k = m - (n-1)
    best_c = corr.max()
    
    for d in step_sizes:
        if time.time() > deadline:
            break
        for _pass in range(500):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d > 1.0 or h[j] - d < 0.0:
                        # Try reverse
                        if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                            # Evaluate h[i]-=d, h[j]+=d
                            new_max = _eval_move(h, n, i, j, -d, dx, corr)
                            if new_max is not None and new_max < best_c - 1e-15:
                                h[i] -= d
                                h[j] += d
                                _update_corr(h, n, i, j, -d, dx, corr)
                                best_c = new_max
                                improved = True
                        continue
                    
                    new_max = _eval_move(h, n, i, j, d, dx, corr)
                    if new_max is not None and new_max < best_c - 1e-15:
                        h[i] += d
                        h[j] -= d
                        _update_corr(h, n, i, j, d, dx, corr)
                        best_c = new_max
                        improved = True
                        # Continue with this improved h (don't undo)
                    
                    # Also try reverse: h[i] -= d, h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        new_max = _eval_move(h, n, i, j, -d, dx, corr)
                        if new_max is not None and new_max < best_c - 1e-15:
                            h[i] -= d
                            h[j] += d
                            _update_corr(h, n, i, j, -d, dx, corr)
                            best_c = new_max
                            improved = True
            
            if not improved:
                break
    
    return h, best_c
```

The `_eval_move` function: given current h, n, positions i, j, change d (h[i]+=d, h[j]-=d), and current corr array, compute the new max without modifying anything.

The `_update_corr` function: actually update the corr array after the move is accepted.

For `_eval_move`:
- The correlation value at shift k (index m = k + n - 1 in corr) changes only if i or j (or i±k or j±k) is in the valid range for shift k.

```python
def _eval_move_vec(h, n, i, j, d, dx, corr):
    # shifts: k = -(n-1), ..., n-1 -> indices 0 to 2n-2
    k = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # For each k, the valid range of m is [max(0,-k), min(n, n-k))
    # Affected m values: i, j, i-k, j-k (that are in valid range)
    
    # Instead of looping over k, let's compute the delta for all k at once.
    # delta_k = sum over affected m of (h'[m]*(1-h'[m+k]) - h[m]*(1-h[m+k]))
    
    # For position p (where p is i or j), the contribution to delta_k is:
    # If p is a valid "first index" m for shift k: i.e., max(0,-k) <= p < min(n, n-k)
    #   AND h[p] is changing (p == i or p == j):
    #   The term h[p]*(1-h[p+k]) changes by (Δh[p])*(1-h[p+k]) [if p+k is not also changing]
    #   or (Δh[p])*(1-h[p+k]+Δh[p+k]) + h[p]*(-Δh[p+k]) [if p+k is also changing]
    
    # This is getting complex. Let me just use a simple approach:
    # Compute the new corr for all affected shifts.
    
    # Position i changes by +d, position j changes by -d.
    # The correlation at shift k involves pairs (m, m+k).
    # Position p is involved in shift k if:
    #   p = m for some valid m: max(0,-k) <= p < min(n, n-k)  =>  -p <= k < n-p (and k > -(n-1), k < n-1)
    #   p = m+k for some valid m: max(0,-k) <= p-k < min(n, n-k) => p-n < k <= p (and k bounds)
    
    # Simpler: position p is in shift k iff:
    #   p is a first index: 0 <= p < n and 0 <= p+k < n => -p <= k <= n-1-p
    #   p is a second index: 0 <= p-k < n and 0 <= k... => 0 <= p-k < n => k <= p and k > p-n, i.e., p-n < k <= p
    # Combined: p is in shift k iff max(-p, p-n) < k <= min(n-1-p, p)
    # Since k is integer: max(-p, -(n-1)) <= k <= min(p, n-1-p)
    # I.e., -(n-1) <= k <= n-1 (always true for valid k) and:
    #   -p <= k <= p and k >= p-n+1 and k <= n-1-p
    #   => max(-p, p-n+1, -(n-1)) <= k <= min(p, n-1-p, n-1)
    #   => max(-(n-1-p), -(n-1)) <= k <= min(p, n-1-p)  [since -(n-1-p) = p-n+1... wait]
    #   
    # Let me just be explicit. For position p, the shifts k where p participates:
    # As first index (m=p): need 0 <= p+k < n, i.e., -p <= k < n-p
    # As second index (m+k=p, so m=p-k): need 0 <= p-k < n, i.e., p-n < k <= p
    # Union: max(-p, p-n+1) <= k <= min(n-1-p, p)
    # Since k is integer: k ∈ [max(-p, p-n+1), min(n-1-p, p)]
    # Note: p-n+1 = -(n-1-p) and -p. So lower = max(-p, -(n-1-p)) = -min(p, n-1-p)
    # Upper = min(n-1-p, p) = min(p, n-1-p)
    # So: k ∈ [-min(p, n-1-p), min(p, n-1-p)]
    
    # Great! So position p participates in shifts k from -r_p to r_p where r_p = min(p, n-1-p).
    
    # For the move (i, j, d):
    # r_i = min(i, n-1-i), r_j = min(j, n-1-j)
    # Affected shifts: k ∈ [-max(r_i,r_j), max(r_i,r_j)] (union of the two ranges)
    # But within this range, only shifts where i OR j participates are affected.
    
    ri = min(i, n - 1 - i)
    rj = min(j, n - 1 - j)
    k_lo = -max(ri, rj)
    k_hi = max(ri, rj)
    
    # For each k in [k_lo, k_hi]:
    # delta_c(k) = sum of changes over affected positions
    # Affected: i participates if |k| <= ri, j participates if |k| <= rj
    
    # Let me compute this vectorially:
    k_vals = np.arange(k_lo, k_hi + 1)
    new_corr_vals = np.zeros(len(k_vals))
    
    for idx_k, k in enumerate(k_vals):
        delta = 0.0
        # Position i:
        if abs(k) <= ri:
            # i is first index: m=i, need 0<=i+k<n (guaranteed by |k|<=ri)
            # i is second index: m=i-k, need 0<=i-k<n (guaranteed by |k|<=ri)
            # Change in h[i]*(1-h[i+k]) [i as first]:
            #   Δ = d * (1 - h[i+k]) [if i+k != j]
            #   or d*(1-h[i+k]+d) + h[i]*(-d) ... wait
            #   h'[i] = h[i]+d, h'[i+k] = h[i+k] (unless i+k == j, i.e., k = j-i)
            if i + k == j:
                # Both i and i+k=j are changing
                old_term = h[i] * (1 - h[j])
                new_term = (h[i]+d) * (1-(h[j]-d))
                delta += new_term - old_term
            else:
                old_term = h[i] * (1 - h[i+k])
                new_term = (h[i]+d) * (1 - h[i+k])
                delta += new_term - old_term
            # Change in h[i-k]*(1-h[i]) [i as second index, m=i-k]:
            if i - k == j:
                old_term = h[j] * (1 - h[i])
                new_term = (h[j]-d) * (1-(h[i]+d))
                delta += new_term - old_term
            else:
                old_term = h[i-k] * (1 - h[i])
                new_term = h[i-k] * (1-(h[i]+d))
                delta += new_term - old_term
        
        # Position j:
        if abs(k) <= rj:
            if j + k == i:
                old_term = h[j] * (1 - h[i])
                new_term = (h[j]-d) * (1-(h[i]+d))
                delta += new_term - old_term
            else:
                old_term = h[j] * (1 - h[j+k])
                new_term = (h[j]-d) * (1 - h[j+k])
                delta += new_term - old_term
            if j - k == i:
                old_term = h[i] * (1 - h[j])
                new_term = (h[i]+d) * (1-(h[j]-d))
                delta += new_term - old_term
            else:
                old_term = h[j-k] * (1 - h[j])
                new_term = h[j-k] * (1-(h[j]-d))
                delta += new_term - old_term
        
        new_c_k = corr[k + n - 1] + delta * dx
        new_corr_vals[idx_k] = new_c_k
    
    # The max is max of: unchanged corr values, and new_corr_vals
    # For speed, we can just check: is the new max less than the old max?
    # Old max = best_c (tracked separately)
    # New max = max(best_c among unaffected, max(new_corr_vals))
    # Unaffected values: corr[k] for k not in [k_lo, k_hi]
    # Their max is at most best_c (since best_c is the overall max)
    # So new_max = max(max_new_vals, max_unaffected)
    # max_unaffected <= best_c (could be equal if the old max was at an unaffected shift)
    
    max_new = new_corr_vals.max() if len(new_corr_vals) > 0 else 0
    # Check if old max was at an unaffected shift
    # Old max shifts: where corr == best_c
    # If any of those shifts are outside [k_lo, k_hi], then max_unaffected = best_c
    # Otherwise, max_unaffected < best_c (need to compute)
    
    # For simplicity, just compute:
    # new_max = max(max_new, max of unaffected)
    # max of unaffected = max(corr[0:k_lo+n-1], corr[k_hi+1+n-1:]) if those ranges exist
    idx_lo = k_lo + n - 1
    idx_hi = k_hi + n - 1
    if idx_lo > 0:
        max_left = corr[:idx_lo].max()
    else:
        max_left = 0
    if idx_hi < 2*n-2:
        max_right = corr[idx_hi+1:].max()
    else:
        max_right = 0
    max_unaff = max(max_left, max_right)
    
    return float(max(max_new, max_unaff))
```
