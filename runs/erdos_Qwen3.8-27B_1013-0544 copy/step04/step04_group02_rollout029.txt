```python
    #   dcorr/dh[i] includes (1-h[i+k]) if i+k < n, and -h[i-k] if i >= k
```

```python
def compute_smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    z = corr / beta
    z = z - z.max()
    p = np.exp(z)
    p /= p.sum()
    
    grad = np.zeros(n)
    # For each |k| = a (a = 0 to n-1):
    #   There are two lags with |k|=a: k=a (idx=n-1+a) and k=-a (idx=n-1-a)
    #   (except a=0 where they're the same)
    #   d/dh[i] = (1-h[i-a])*1[i>=a] - h[i+a]*1[i+a<n]
    
    for a in range(n):
        w_pos = p[n-1+a]  # weight for lag k=+a
        w_neg = p[n-1-a] if a > 0 else 0.0  # weight for lag k=-a
        w = w_pos + w_neg
        
        if w < 1e-15:
            continue
        
        # (1-h[i-a]) * 1[i>=a]
        if a == 0:
            grad[:] += w * (1.0 - h)
        else:
            grad[a:] += w * (1.0 - h[:n-a])
        
        # -h[i+a] * 1[i+a < n]
        if a == 0:
            grad[:] -= w * h
        else:
            grad[:n-a] -= w * h[a:]
    
    return dx * grad
```

Actually, for the coordinate descent, a simpler approach that works well:
- The minimum of a convex PWL function on [0,1] is at a "kink" (where two lines cross and form the upper envelope)
- I can find candidate kinks by: for each pair (i,j) where line i is above line j at t=0 and below at t=1, they cross at t* = (a_j - a_i)/(b_i - b_j)
- But this is O(n²) pairs, which is too many

Alternative: use the fact that at the optimum, the subgradient contains 0. The subgradient at t is [min B_i : i in argmax(a_i + B_i * t), max B_i : i in argmax(...)]. So I need a t where some active line has B_i < 0 and another has B_i > 0 (or one has B_i = 0).

Practical approach: 
1. Evaluate g at t=0 and t=1
2. Find the line max at t=0 (call it i0) and at t=1 (call it i1)
3. If B[i0] >= 0: minimum is at t=0
4. If B[i1] <= 0: minimum is at t=1
5. Otherwise: i0 has B[i0] < 0 and i1 has B[i1] > 0
   They cross at t* = (a[i0] - a[i1]) / (B[i1] - B[i0])
   If 0 < t* < 1, evaluate g(t*)
   The minimum is min(g(0), g(1), g(t*))

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, beta):
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(z))) / beta

def smooth_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -700.0, 700.0)
    p = np.exp(z)
    p /= p.sum()
    grad = np.zeros(n)
    for a in range(n):
        w_pos = p[n - 1 + a]
        w_neg = p[n - 1 - a] if a > 0 else 0.0
        w = w_pos + w_neg
        if w < 1e-15:
            continue
        if a == 0:
            grad[:] += w * (1.0 - h)
            grad[:] -= w * h
        else:
            grad[a:] += w * (1.0 - h[:n - a])
            grad[:n - a] -= w * h[a:]
    return dx * grad

def con_fun(x, ts):
    return x.sum() - ts

def con_jac(x, ts):
    return np.ones_like(x)

def slsqp_optimize(h, n, deadline, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    c5 = compute_c5(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
    betas = [30, 100, 300, 1000, 3000, 10000, 30000, 100000]
    for beta in betas:
        if time.time() > deadline:
            break
        for p in [0.0, 0.005, 0.015]:
            if time.time() > deadline:
                break
            x0 = h.copy()
            if p > 0:
                x0 = project_feasible(x0 + np.random.randn(n) * p, ts)
            try:
                res = minimize(smooth_obj, x0, args=(dx, beta), jac=smooth_grad,
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < c5:
                    h = h_new
                    c5 = c5_new
            except Exception:
                pass
    return h, c5

def subgradient_polish(h, n, deadline, seed):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    velocity = np.zeros(n)
    lr = 0.003
    stagnant = 0
    for it in range(20000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = corr.max()
        top_mask = corr >= idx_max - 0.01 * idx_max
        top_idx = np.where(top_mask)[0]
        if len(top_idx) == 0:
            top_idx = np.argsort(corr)[-3:]
        w = np.exp(20.0 * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        grad = np.zeros(n)
        for j, idx in enumerate(top_idx):
            k = int(idx - (n - 1))
            a = abs(k)
            if a == 0:
                grad[:] += w[j] * (1.0 - 2.0 * h)
            else:
                grad[a:] += w[j] * (1.0 - h[:n - a])
                grad[:n - a] -= w[j] * h[a:]
        velocity = 0.80 * velocity - lr * grad
        h_new = project_feasible(h + velocity, ts)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-12:
            h = h_new
            c5 = c_new
            stagnant = 0
            lr = min(lr * 1.003, 0.006)
        else:
            stagnant += 1
            velocity *= 0.5
            lr *= 0.90
        if stagnant > 200 and lr < 1e-6:
            break
    return h, c5

def coord_descent(h, n, deadline):
    ts = n / 2.0
    c5 = compute_c5(h)
    dx = 2.0 / n
    for sweep in range(500):
        if time.time() > deadline:
            break
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        for i in range(n):
            B = np.zeros(2 * n - 1)
            for a in range(n):
                w_pos = 1.0 if (n - 1 + a) == (n - 1) else 1.0
                # d/dh[i] for lag +a: (1-h[i-a])*[i>=a] - h[i+a]*[i+a<n]
                val = 0.0
                if i >= a:
                    val += (1.0 - h[i - a])
                if i + a < n:
                    val -= h[i + a]
                B[n - 1 + a] = val
                if a > 0:
                    B[n - 1 - a] = val
            a_arr = corr - B * h[i]
            candidates = [0.0, 1.0]
            i0 = int(np.argmax(a_arr))
            i1 = int(np.argmax(a_arr + B))
            if i0 != i1 and B[i1] != B[i0]:
                t_cross = (a_arr[i0] - a_arr[i1]) / (B[i0] - B[i1])
                if 0.0 < t_cross < 1.0:
                    candidates.append(t_cross)
            t_opt = min(candidates, key=lambda t: np.max(a_arr + B * t))
            if abs(t_opt - h[i]) > 1e-13:
                h[i] = t_opt
                improved = True
        h = project_feasible(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5:
            c5 = c5_new
        if not improved and time.time() - t0 > 5:
            break
    return h, c5

    best_h = None
```
