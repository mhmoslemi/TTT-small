    srx = np.sort(corr)[::-1][:k]

    vals = np.exp(np.clip(beta * (srx - m), -500, 500))
    return (m + np.log(np.sum(vals) / k) / beta) * dx

    return x.sum() - t

    return np.ones(x.shape)

def project_h(h, target_sum):
