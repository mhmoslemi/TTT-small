import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def thue_morse(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t.astype(float)

def get_top_lags(h, K=20):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:K]
    return idx

def subgrad_top(h, top_idx, n):
    g = np.zeros(n)
    for i in top_idx:
        lag = int(i) - (n - 1)
        if lag >= 0:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
        else:
            la = -lag
            g[la:] += (1.0 - h[:n - la])
            g[:n - la] -= h[la:]
    return g / len(top_idx)

def smooth_top_obj(h, beta, K, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    top = np.argsort(corr)[::-1][:K]
    vals = corr[top]
    m = np.max(vals)
    return (m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta) * (2.0 / n)

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def slsqp_opt(h0, n, target, beta, K, maxiter, ftol):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    def obj(x):
        return smooth_top_obj(x, beta, K, n)
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def nesterov_opt(h0, n, target, n_iter, t0, budget, K=20, step0=0.04):
    h = h0.copy()
    y = h0.copy()
    v = np.zeros(n)
    best_h, best_c5 = h.copy(), compute_c5(h)
    top_idx = get_top_lags(h, K)
    step = step0
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        g = subgrad_top(y, top_idx, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        v = 0.85 * v + 0.15 * g
        h_new = project_h(y - step * v, target)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-14:
            step = min(step * 1.04, 0.25)
            stall = 0
            best_h, best_c5 = h_new.copy(), c_new
        else:
            stall += 1
            if stall > 40:
                step *= 0.8
                stall = 0
        y = h_new
        if (it + 1) % 50 == 0:
            top_idx = get_top_lags(h_new, K)
    return best_h, best_c5

def coord_opt(h0, n, target, sweeps, t0, budget):
    h = h0.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(123)

                if c < best_c5 - 1e-14:
