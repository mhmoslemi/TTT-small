import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target}"""
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def get_top_lags(h, n_top):
    """Get indices of top n_top lags in the correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.argsort(corr)[::-1][:n_top], corr

def gradient_for_lags(h, top_idx):
    """Compute subgradient for the top lags."""
    n = len(h)
    g = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag == 0:
            g += (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += (1.0 - h[:-lag])
            g[:n - lag] -= h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += (1.0 - h[m2:])
            g[m2:] -= h[:n - m2]
    return g

def subgrad_step(h, target, n_active, step_sizes, t0, budget_s, min_budget=2.0):
    """One subgradient descent step with line search."""
    if budget_s - (time.time() - t0) < min_budget:
        return h, compute_c5(h)
    best_c5 = compute_c5(h)
    top_idx, _ = get_top_lags(h, n_active)
    g = gradient_for_lags(h, top_idx)
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, best_c5
    g /= gn
    best_h = h
    best_c5 = compute_c5(h)
    for step in step_sizes:
        ht = project_sum(h - step * g, target)
        c5t = compute_c5(ht)

            best_h = ht

def slsqp_opt(h_init, n, beta, maxiter=3000):
    """SLSQP optimization with soft-max objective."""
    target = n / 2.0
    bounds = [(0, 1)] * n
    
    def obj(x):
        n_local = len(x)
        dx = 2.0 / n_local
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        bc = np.clip(beta * (corr - m), -500, 500)
        e = np.exp(bc)
        se = e.sum()
        return (m + np.log(se / len(corr)) / beta) * dx

    def grad(x):
        n_local = len(x)
        dx = 2.0 / n_local
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        bc = np.clip(beta * (corr - m), -500, 500)
        e = np.exp(bc)
        se = e.sum()
        w = e / se
        g = np.zeros(n_local)
        nc = len(corr)
        for j in range(nc):
            if w[j] < 1e-11:
                continue
            lag = j - (n_local - 1)
            if lag == 0:
                g += w[j] * (1.0 - 2.0 * x)
            elif lag > 0:
                g[lag:] += w[j] * (1.0 - x[:-lag])
                g[:n_local - lag] -= w[j] * x[lag:]
            else:
                m2 = -lag
                g[:n_local - m2] += w[j] * (1.0 - x[m2:])
                g[m2:] -= w[j] * x[:n_local - m2]
        return g * dx

    def eq(x):
        return np.sum(x) - target

    def eqjac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eqjac}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
    return project_sum(res.x, target)

def coordinate_swap(h, target, n_swaps=100, t0=None, budget_s=None, min_budget=1.0):

    best = h.copy()
    best_c5 = compute_c5(best)
    n = len(best)
    targets = np.where(best > 0.5)[0]  # positions where h is "high"
    sources = np.where(best < 0.5)[0]  # positions where h is "low"
    
    for _ in range(n_swaps):
        if t0 is not None and budget_s - (time.time() - t0) < min_budget:
            break
        # Try swapping a "high" with a "low"
        i = np.random.choice(targets)
        j = np.random.choice(sources)
        ht = best.copy()
        ht[i], ht[j] = ht[j], ht[i]
        ht = project_sum(ht, target)
        c5t = compute_c5(ht)

            best_c5 = c5t
            # Update target/source lists
            if best[i] > 0.5:
                targets = np.where(best > 0.5)[0]
                sources = np.where(best < 0.5)[0]
    return best, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_lags(h, indices):
    n = len(h)
    g = np.zeros(n)
    for idx in indices:
        lag = int(idx) - (n - 1)
        if lag == 0:
            g += (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += (1.0 - h[:-lag])
            g[:n-lag] -= h[lag:]
        else:
            m = -lag
            g[:n-m] += (1.0 - h[m:])
            g[m:] -= h[:n-m]
    return g

def slsqp_solve(h0, n, beta, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    dx = 2.0 / n

    def f(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = c.max()
        z = np.clip(beta * (c - m), -400, 400)
        e = np.exp(z)
        return (m + np.log(e.sum() / len(c)) / beta) * dx

    def df(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = c.max()
        z = np.clip(beta * (c - m), -400, 400)
        e = np.exp(z)
        w = e / e.sum()
        g = np.zeros(n)
        nc = len(c)
        for j in range(nc):
            if w[j] < 1e-10:
                continue
            lag = j - (n - 1)
            if lag == 0:
                g += w[j] * (1.0 - 2.0 * x)
            elif lag > 0:
                g[lag:] += w[j] * (1.0 - x[:-lag])
                g[:n-lag] -= w[j] * x[lag:]
            else:
                m2 = -lag
                g[:n-m2] += w[j] * (1.0 - x[m2:])
                g[m2:] -= w[j] * x[:n-m2]
        return g * dx

    res = minimize(f, h0, jac=df, method='SLSQP', bounds=bounds,
                   constraints=[{'type':'eq','fun':lambda x: x.sum()-target,
                                'jac':lambda x: np.ones(n)}],
                   options={'maxiter':maxiter,'ftol':1e-18})
    return project_sum(res.x, target)

def subgrad_polish(h, target, n_active, max_it, t0, budget, min_rem=3):
    best = h.copy()
    bc = compute_c5(best)
    n = len(best)
    v = np.zeros(n)
    mu = 0.85
    stall = 0
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for it in range(max_it):
        if budget - (time.time()-t0) < min_rem:
            break
        corr = np.correlate(best, 1.0-best, mode="full")
        idx = np.argsort(corr)[::-1][:n_active]
        g = gradient_lags(best, idx)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        v = mu*v + (1-mu)*g
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            break
        v /= vn
        ok = False
        for s in steps:
            ht = project_sum(best - s*v, target)
            c5 = compute_c5(ht)
            if c5 < bc - 1e-16:
                best, bc = ht, c5
                ok = True
                break
        if not ok:
            stall += 1
            if stall > 100:
                scale = 0.006 * (1.0 + stall / 400.0)
                p = np.random.randn(n) * scale
                ht = project_sum(best + p, target)
                c5 = compute_c5(ht)
                if c5 < bc:
                    best, bc, stall, v = ht, c5, 0, np.zeros(n)
                elif stall > 1200:
                    break
        else:
            stall = 0
    return best, bc

def micro_polish(h, target, max_it, t0, budget, min_rem=1.0):
    best = h.copy()
    bc = compute_c5(best)
    n = len(best)
    for it in range(max_it):
        if budget - (time.time()-t0) < min_rem:
            break
        corr = np.correlate(best, 1.0-best, mode="full")
        top = np.argsort(corr)[::-1][:2]
        g = gradient_lags(best, top)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        ok = False
        for s in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]:
            ht = project_sum(best - s*g, target)
            c5 = compute_c5(ht)
            if c5 < bc - 1e-17:
                best, bc = ht, c5
                ok = True
                break
        if not ok and it > 2000:
            break
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        n0 = 240
        alpha = (1.0 + 5.0**0.5) / 2.0
        x = np.linspace(0, 2, n0)
        best_h = ((x * alpha) % 1.0 < 0.5).astype(float)
        best_h = project_sum(best_h, n0 / 2.0)

    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at current resolution
    n = len(best_h)
    target = n / 2.0
    for beta in [40, 150, 600, 2500, 10000]:
        if rem() < 80:
            break
        h_new = slsqp_solve(best_h, n, beta, 2000)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5
        best_h, best_c5 = subgrad_polish(best_h, target, 5, 400, t0, budget_s)

    # Phase 2: Aggressive momentum polish
    best_h, best_c5 = subgrad_polish(best_h, target, 6, 6000, t0, budget_s)

    # Phase 3: Try nearby resolutions
    for n_try in [400, 520, 560]:
        if rem() < 200:
            break
        if n_try == len(best_h):
            continue
        xo = np.linspace(0, 2, len(best_h))
        xn = np.linspace(0, 2, n_try)
        hi = project_sum(np.interp(xn, xo, best_h), n_try / 2.0)
        for beta in [100, 500, 2000]:
            if rem() < 70:
                break
            hn = slsqp_solve(hi, n_try, beta, 2000)
            c5 = compute_c5(hn)
            if c5 < best_c5:
                best_h, best_c5 = hn, c5
                n = n_try
                target = n_try / 2.0
            hi = hn
        best_h, best_c5 = subgrad_polish(best_h, target, 6, 1000, t0, budget_s)

    # Phase 4: Diverse restarts
    nb = len(best_h)
    tb = nb / 2.0
    starts = [best_h + np.random.randn(nb) * 0.03]
    starts.append((best_h > 0.5).astype(float))
    alpha2 = (1.0 + 5.0**0.5) / 2.0
    xg = np.linspace(0, 2, nb)
    starts.append(((xg * alpha2) % 1.0 < 0.5).astype(float))
    # quarter-interval pattern
    qp = np.zeros(nb)
    qp[:nb//4] = 1.0
    qp[3*nb//4:] = 1.0
    starts.append(qp)

    for st in starts:
        if rem() < 90:
            break
        ht = project_sum(st, tb)
        for beta in [400, 2000, 8000]:
            if rem() < 50:
                break
            hn = slsqp_solve(ht, nb, beta, 1500)
            c5 = compute_c5(hn)
            if c5 < best_c5:
                best_h, best_c5 = hn, c5
            ht = hn
        best_h, best_c5 = subgrad_polish(best_h, tb, 5, 500, t0, budget_s)

    # Phase 5: Final intensive polish with decreasing n_active
    nf = len(best_h)
    tf = nf / 2.0
    best_h, best_c5 = subgrad_polish(best_h, tf, 4, 5000, t0, budget_s)
    best_h, best_c5 = subgrad_polish(best_h, tf, 2, 8000, t0, budget_s)

    # Phase 6: Final SLSQP at very high beta
    if rem() > 50:
        hn = slsqp_solve(best_h, nf, 25000, 3000)
        c5 = compute_c5(hn)
        if c5 < best_c5:
            best_h, best_c5 = hn, c5
        best_h, best_c5 = subgrad_polish(best_h, tf, 2, 2000, t0, budget_s)

    # Phase 7: Micro-polish
    best_h, best_c5 = micro_polish(best_h, tf, 5000, t0, budget_s)

    return best_h, best_c5, len(best_h)
