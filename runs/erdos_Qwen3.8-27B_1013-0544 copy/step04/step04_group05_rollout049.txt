from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr * dx

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            idx = np.argsort(h)
            if diff > 0:
                idx = idx[-max(1, int(abs(diff) * 30)):]
            else:
                idx = idx[:max(1, int(abs(diff) * 30))]
            h[idx] += diff / len(idx)
        else:
            h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lags(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def slsqp_opt(h, target, dx, betas, top_m, max_iter, time_fn, min_time):

    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()

        return h_in.sum() - target
    def jac_fn(h_in):
        return np.ones(n)
    for beta in betas:
        if time_fn() < min_time:
            break
        def obj_fn(h_in, b=beta):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")

            vals = c[top]

                          options={'maxiter': max_iter, 'ftol': 1e-22})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)[0]
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def subgrad_opt(h, target, time_fn, max_iters, m_lags, steps, perturb_every, perturb_scale, momentum):

    best = h.copy()
    best_c5 = compute_c5(best)[0]
    vel = np.zeros(n)
    no_improve = 0
    for it in range(max_iters):
        if time_fn() < 3:
            break
        corr_raw = np.correlate(best, 1.0 - best, mode="full")
        m_act = min(m_lags, max(3, len(corr_raw) // 10))
        top = np.argsort(corr_raw)[::-1][:m_act]
        lags = [int(i) - (n - 1) for i in top]
        g = subgrad_for_lags(best, lags)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn
        found = False
        for step in steps:
            ht = project_h(best + step * d, target)
            c5t = compute_c5(ht)[0]
            if c5t < best_c5 - 1e-16:
                best = ht
                best_c5 = c5t
                no_improve = 0
                found = True
                break

                    best = ht
                    best_c5 = c5t
                    vel *= 0.5
                    no_improve = 0
            if no_improve > perturb_every * 12:
                break
    return best, best_c5

        improved = False

                    ht = project_h(ht, target)
                    c5t, _ = compute_c5(ht)
                    if c5t < best_c5 - 1e-15:

                        best_c5 = c5t
                        improved = True
                        break
                if best_c5 < compute_c5(best)[0] + 1e-15:
                    break
        if not improved:
            break
    return best, best_c5
