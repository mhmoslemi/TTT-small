    h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

# Precomputed gradient indices cache
_grad_cache = {}

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(h, target_sum):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    return np.clip(h - tau, 0.0, 1.0)

def _get_grad_indices(n):
    if n not in _grad_cache:
        s = np.arange(2 * n - 1) - (n - 1)
        m_idx = np.arange(n)[:, None]
        s_arr = s[None, :]
        mms = m_idx - s_arr
        mps = m_idx + s_arr
        vm = (mms >= 0) & (mms < n)
        vp = (mps >= 0) & (mps < n)
        _grad_cache[n] = (np.clip(mms, 0, n-1), np.clip(mps, 0, n-1),
                          vm.astype(np.float64), vp.astype(np.float64))
    return _grad_cache[n]

def compute_smooth_grad(h, n, alpha):
    """Compute gradient of log-sum-exp smoothed max correlation."""
    im, ip, mm, mp = _get_grad_indices(n)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    ac = alpha * c
    mx = ac.max()
    p = np.exp(ac - mx)
    p_sum = p.sum()
    p /= p_sum
    h_m = h[im]
    h_p = h[ip]
    t1 = (1.0 - h_m) * mm
    t2 = h_p * mp
    grad = dx * (t1 - t2).dot(p)
    return grad, c

def lse_value(c, alpha):
    ac = alpha * c
    mx = ac.max()
    return float((mx + np.log(np.sum(np.exp(ac - mx)))) / alpha)

def obj_lse(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    return lse_value(c, alpha)

def obj_pen(h, n, alpha, lam):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    f = lse_value(c, alpha)
    f += lam * (np.sum(h) - n / 2.0) ** 2
    return f

def _con_fun(h, n):
    return np.sum(h) - n / 2.0

def _con_jac(h, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_lse, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if np.all(np.isfinite(res.x)):
            return project_box_sum(res.x, n / 2.0)
    except Exception:
        pass
    return h.copy()

def lbfgs_opt(h, n, alpha, lam, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h, args=(n, alpha, lam), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        if np.all(np.isfinite(res.x)):
            return project_box_sum(res.x, n / 2.0)
    except Exception:
        pass
    return h.copy()

    x = h_init.copy()
    y = h_init.copy()
    best_h = x.copy()
    best_c = compute_c5(x, n)
    t = 1.0

        if time.time() > deadline:
            break
        
        # Compute gradient at y
        grad, _ = compute_smooth_grad(y, n, alpha)
        # Add penalty for constraint
        c_val = y.sum() - n / 2.0

        grad_total = grad + grad_penalty
        
        # Gradient step

        # Project onto feasible set
        x_new = project_box_sum(x_new, n / 2.0)
        
        # Check improvement

        if c_new < best_c:

            best_h = x_new.copy()

        t_new
