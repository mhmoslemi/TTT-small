import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def slsqp_con_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def slsqp_con_jac(x, n):
    return np.ones_like(x)

def slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh)))) / alpha

def slsqp_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
    else:
        w = np.exp(sh)
    w /= w.sum()
    idx = np.arange(n)
    ij = idx[:, None] - idx[None, :] + (n - 1)
    mask = (ij >= 0) & (ij < 2 * n - 1)
    W = np.where(mask, w[ij], 0.0)
    return dx * (W @ (1 - x) - W.T @ x)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h.copy(), args=(n, alpha),
                       jac=lambda x: slsqp_grad(x, n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

            break

                break
            improved = False
            for i in range(n):
                for j in range(i + 1, n):
                    if time.time() > deadline:
                        return h, best_c
                    hi = h[i] + d
                    hj = h[j] - d
                    if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
                        continue

                    # ΔC[r] = dx*d * [(1-h[i-r])*1_{0≤i-r<n} - (1-h[j-r])*1_{0≤j-r<n} - h[i+r]*1_{0≤i+r<n} + h[j+r]*1_{0≤j+r<n}]
                    # r ranges from -(n-1) to n-1, indexed as r + n - 1 in C
                    delta = np.zeros(2 * n - 1)
                    # Term 1: (1-h[i-r]) for r in [i-n+1, i]
                    r_lo1 = max(-(n-1), i - (n-1))
                    r_hi1 = min(n-1, i)
                    if r_lo1 <= r_hi1:
                        rs = np.arange(r_lo1, r_hi1 + 1)
                        delta[rs + n - 1] += (1 - h[i - rs])
                    # Term 2: -(1-h[j-r]) for r in [j-n+1, j]
                    r_lo2 = max(-(n-1), j - (n-1))
                    r_hi2 = min(n-1, j)
                    if r_lo2 <= r_hi2:
                        rs = np.arange(r_lo2, r_hi2 + 1)
                        delta[rs + n - 1] -= (1 - h[j - rs])
                    # Term 3: -h[i+r] for r in [-(i), n-1-i]
                    r_lo3 = max(-(n-1), -i)
                    r_hi3 = min(n-1, n - 1 - i)
                    if r_lo3 <= r_hi3:
                        rs = np.arange(r_lo3, r_hi3 + 1)
                        delta[rs + n - 1] -= h[i + rs]
                    # Term 4: h[j+r] for r in [-j, n-1-j]
                    r_lo4 = max(-(n-1), -j)
                    r_hi4 = min(n-1, n - 1 - j)
                    if r_lo4 <= r_hi4:
                        rs = np.arange(r_lo4, r_hi4 + 1)
                        delta[rs + n - 1] += h[j + rs]
                    
                    delta *= dx * d
                    new_max = np.max(C + delta)
                    if new_max < best_c - 1e-15:
                        h[i], h[j] = hi, hj
                        C = C + delta
                        best_c = new_max
                        improved = True
                    # Try other direction
                    hi2 = h[i] - d

                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:

                            h[i], h[j] = hi2, hj2
                            C = C - delta
                            best_c = new_max2
                            improved = True
            if not improved:
                break
    return h, best_c
