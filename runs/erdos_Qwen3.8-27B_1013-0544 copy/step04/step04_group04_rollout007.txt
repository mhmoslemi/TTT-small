def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    """Exact Euclidean projection onto {0<=x<=1, sum(x)=target_sum}"""
    h = h.copy()
    # Bisection on alpha: find alpha such that sum(clip(h - alpha, 0, 1)) = target_sum
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):

        s = np.clip(h - alpha, 0, 1).sum()
        if s > target_sum:
            lo = alpha
        else:
