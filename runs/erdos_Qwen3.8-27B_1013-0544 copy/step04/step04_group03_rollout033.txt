```python
def binary_sa(h, n, t0, budget, n_iter=500000, T_init=0.5, T_final=0.01):
    # h is binary with exactly n/2 ones
    corr = np.correlate(h, 1-h, mode='full')
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    rng = np.random.default_rng(12345)
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for it in range(n_iter):
        if time.time() - t0 > budget - 10:
            break
        T = T_init * (T_final / T_init) ** (it / n_iter)
        
        # Random swap
        ia = rng.integers(len(ones))
        ib = rng.integers(len(zeros))
        a = ones[ia]
        b = zeros[ib]
        
        # Perform swap
        h[a] = 0
        h[b] = 1
        ones[ia] = b
        zeros[ib] = a
        
        # Evaluate
        new_corr = np.correlate(h, 1-h, mode='full')
        new_max = float(np.max(new_corr))
        
        if new_max <= cur_max or rng.random() < np.exp(-(new_max - cur_max) / T):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            # Undo
            h[a] = 1
            h[b] = 0
            ones[ia] = a
            zeros[ib] = b
    
    return best_h, best_max
```

For a swap of a (1→0) and b (0→1):

```python
def swap_delta(h, a, b, n):
    """Compute change in correlation array when h[a] 1->0 and h[b] 0->1."""
    delta = np.zeros(2*n-1)
    # For k >= 0: c[k] = sum_{t=0}^{n-1-k} h[t+k]*(1-h[t])
    # a as h[t+k]: t = a-k, valid if k <= a <= n-1
    #   delta: -(1 - h[a-k]) if k <= a
    # a as 1-h[t]: t = a, valid if a <= n-1-k, i.e., k <= n-1-a
    #   delta: +h[a+k] if a+k <= n-1
    # b as h[t+k]: t = b-k, valid if k <= b <= n-1
    #   delta: +(1 - h[b-k]) if k <= b
    # b as 1-h[t]: t = b, valid if b <= n-1-k, i.e., k <= n-1-b
    #   delta: -h[b+k] if b+k <= n-1
    
    for k in range(n):
        d = 0.0
        if k <= a:
            d += (h[a-k] - 1.0)  # = -(1-h[a-k])
        if a + k <= n - 1:
            d += h[a+k]
        if k <= b:
            d += (1.0 - h[b-k])
        if b + k <= n - 1:
            d += (-h[b+k])
        delta[k + n - 1] = d
    
    for k in range(1, n):
        # c[-k] = sum_{s=0}^{n-1-k} h[s]*(1-h[s+k])
        d = 0.0
        if a <= n - 1 - k:
            d += (h[a+k] - 1.0)  # = -(1-h[a+k])
        if k <= a <= n - 1:
            d += h[a-k]
        if b <= n - 1 - k:
            d += (1.0 - h[b+k])
        if k <= b <= n - 1:
            d += (-h[b-k])
        delta[n - 1 - k] = d
    
    return delta
```

import time

def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:

            h = np.clip(h, 0.0, 1.0)
        else:

```python
def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def obj_fn(h_in, beta, n):
    corr = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)

def sum_eq(x, target):
    return np.sum(x) - target

def sum_jac(x, target):
    return np.ones(x.shape)

def slsqp_refine(h0, n, target, beta, maxiter=8000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(obj_fn, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def binary_sa(h, n, t0, t_end, n_iter, seed=42):
    rng = np.random.default_rng(seed)
    h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    T_init = 1.5
    T_final = 0.005
    stall = 0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        T = T_init * (T_final / T_init) ** (it / max(n_iter, 1))
        peak_idx = np.argmax(corr)
        use_targeted = (it < n_iter * 0.7 and rng.random() < 0.6)
        if use_targeted:
            lag = int(peak_idx) - (n - 1)
            found = False
            if lag > 0 and lag < n:
                valid_j = np.arange(n - lag)
                contrib = np.where((h[lag:] == 1) & (h[:n-lag] == 0))[0]
                if len(contrib) > 0:
                    jp = contrib[rng.integers(len(contrib))]
                    a = jp + lag
                    b_list = zeros[rng.integers(3 * len(zeros))]
                    b = b_list[rng.integers(len(b_list))]
                    if b != a:
                        found = True
            elif lag < 0 and -lag < n:
                m = -lag
                contrib = np.where((h[:n-m] == 1) & (h[m:] == 0))[0]
                if len(contrib) > 0:
                    jp = contrib[rng.integers(len(contrib))]
                    a = jp
                    b = jp + m
                    b_list = ones[rng.integers(3 * len(ones))]
                    b = b_list[rng.integers(len(b_list))]
                    if b != a:
                        found = True
            if not found:
                a = ones[rng.integers(len(ones))]
                b = zeros[rng.integers(len(zeros))]
        else:
            a = ones[rng.integers(len(ones))]
            b = zeros[rng.integers(len(zeros))]
        if a == b:
            continue
        h[a] = 0.0
        h[b] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        accept = (new_max <= cur_max) or (rng.random() < np.exp(-(new_max - cur_max) / max(T, 1e-10)))
        if accept:
            ia = np.where(ones == a)[0]
            if len(ia) > 0:
                ones[ia[0]] = b
            ib = np.where(zeros == b)[0]
            if len(ib) > 0:
                zeros[ib[0]] = a
            corr = new_corr
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[a] = 1.0
            h[b] = 0.0
            stall += 1
        if stall > 50000 and T < 0.01:
            break
    return best_h, best_max

    best_max = cur_max

    for _ in range(max_sweeps):
        if time.time() > t_end:

        improved = False
        n_ones = np.sum(h == 1)
        n_zeros = n - n_ones
        for _ in range(200):
            if time.time() > t_end:

            a = rng.integers(n)

            if no_improve > 50:
                break
    return best_h, best_max
```
