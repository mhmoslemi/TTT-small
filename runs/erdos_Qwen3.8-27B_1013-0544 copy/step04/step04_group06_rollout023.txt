import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    else:
        m = -lag
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    return g

def subgrad_max(h, k_top):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:k_top]
    g = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        g += grad_lag(h, lag)
    return g, corr

def slsqp_opt(h_init, beta, maxiter=3000, ftol=1e-15):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def objective(x):
        x = np.clip(x, 0.0, 1.0)
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = float(np.max(corr))
        shifted = np.clip(beta * (corr - m), -60.0, 0.0)
        e = np.exp(shifted)
        se = float(e.sum())
        val = (m + np.log(se / len(corr)) / beta) * dx
        w = e / se
        g = np.zeros(n)
        nc = len(corr)
        for j in range(nc):
            wj = w[j]
            if wj < 1e-13:
                continue
            lag = j - (n - 1)
            wj *= dx
            g += wj * grad_lag(x, lag)
        return val, g

    def f(x):
        return objective(x)[0]

    def gp(x):
        return objective(x)[1]

    def cons_fun(x):
        return [np.sum(x) - target]

    def cons_jac(x):
        return np.ones((1, n))

    res = minimize(f, h_init, jac=gp, method='SLSQP', bounds=bounds,
                   constraints={'type': 'eq', 'fun': cons_fun, 'jac': cons_jac},
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_sum(res.x, target)

    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    v = np.zeros(n)
    step = step0
    stall = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 5:
            break
        g, _ = subgrad_max(best, k_top)

                if stall > 4000:
                    break
    return best, best_c5
