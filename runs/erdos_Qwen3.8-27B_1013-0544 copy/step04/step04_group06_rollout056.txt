    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid

def subgrad_direction(h, top_k=10):
    """Compute subgradient of the max correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Top k correlations (normalized)

        if lag == 0:

        elif lag > 0:
            g[lag:] += (1.0 - h[:-lag])
            g[:n - lag] -= h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += (1.0 - h[m2:])
            g[m2:] -= h[:n - m2]
    return g

def coordinate_descent(h, target, top_k=5, max_iter=5000, t0=None, budget=None):

    n = len(h)

def obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    bc = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(bc)
    se = e.sum()
    val = (m + np.log(se / len(corr)) / beta) * dx
    w = e / se
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        if w[j] < 1e-11:
            continue
        lag = j - (n - 1)

        else:
            m2 = -lag
            g[:n - m2] += w[j] * (1.0 - h[m2:])
            g[m2:] -= w[j] * h[:n - m2]
    return val, g * dx

    bounds = [(0, 1)] * n
    
    def obj(x):
        return obj_grad(np.clip(x, 0, 1), beta)[0]
    
    def grad(x):
        return obj_grad(np.clip(x, 0, 1), beta)[1]
    
    def eq(x):
        return np.sum(x) - target
    
    def eqj(x):
        return np.ones(n)

    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_sum(res.x, target)

def nesterov_subgrad(h, target, top_k, max_iter, t0, budget):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()  # current point
    y = h.copy()  # extrapolated point
    best = h.copy()
    best_c5 = compute_c5(h)[0]
    
    for k in range(1, max_iter + 1):
        if budget - (time.time() - t0) < 3:
            break
        
        # Evaluate at y
        g, _ = subgrad_direction(y, top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        
        # Step: x_new = y - alpha_k * g, where alpha_k = c / sqrt(k)
        alpha = 0.1 / np.sqrt(k)
        x_new = project_sum(y - alpha * g, target)
        
        # Nesterov momentum
        t_k = (1 + np.sqrt(1 + 4 * (k-1)**2)) / 2
        beta_mom = (t_k - 1) / t_k
        y = x_new + beta_mom * (x_new - x)
        y = project_sum(y, target)
        x = x_new
        
        # Check improvement

    return best, best_c5

Let me think about convexity. The objective is:
f(h) = max_k ∫ h(x)(1-h(x+k)) dx = max_k (∫h(x+k)dx - ∫h(x)h(x+k)dx)

The term ∫h(x)h(x+k)dx is the autocorrelation, which is a quadratic form. For k=0, ∫h(1-h) = ∫h - ∫h² = 1 - ∫h². Since ∫h² is convex, ∫h(1-h) = 1 - ∫h² is concave.

For k≠0, ∫h(x)(1-h(x+k))dx = ∫h(x)dx - ∫h(x)h(x+k)dx = 1 - (autocorrelation at lag k). The autocorrelation at lag k is ∫h(x)h(x+k)dx, which is a positive semidefinite quadratic form (it's the inner product of h with its shifted version). So 1 - autocorrelation is concave.

Wait, that means each individual term ∫h(x)(1-h(x+k))dx is concave in h! And the max of concave functions is concave. So we're minimizing a concave function over a convex set, which is a non-convex problem (the minimum of a concave function is at the boundary).

This explains why the optimal solution is close to the boundary (values near 0 or 1). The problem is essentially: find a 0-1 valued function (approximately) that minimizes the maximum correlation.

    """Compute change in C(k) for all k when flipping p:1->0 and j:0->1."""
    # p goes from 1 to 0
    # j goes from 0 to 1
    delta = np.zeros(n - 1)  # for k=1,...,n-1
    
    # Contribution of p as first element (1->0 removes it)
    # k = 1 to n-1-p
    for k in range(1, n - p):

    # Contribution of p as second element (1->0 adds b[p-k])

    for k in range(1, min(j, n-1) + 1):

This is a quadratic function of h. Is it concave? The Hessian of g_k with respect to h would involve the cross terms h[i]*h[i+k]. The quadratic form h^T A h (where A has 1s at positions (i, i+k) and (i+k, i)) is not necessarily negative semidefinite. In fact, A has 1s on the k-th diagonal (both above and below the main diagonal), and the eigenvalues of such a matrix can be positive or negative.

So g_k is NOT necessarily concave! My earlier argument was wrong.

Let me reconsider. g_k(h) = sum_i h[i] - sum_i h[i]*h[i+k] (times dx). The first term is linear. The second term is -sum_i h[i]*h[i+k], which is a quadratic form with matrix -A_k (where A_k has 1s at (i, i+k) and (i+k, i)).

The Hessian of g_k is -2*A_k (symmetric part). The eigenvalues of A_k for a Toeplitz-like matrix with 1s on the k-th diagonal: for the matrix with 1s at positions (i, i+k) and (i+k, i) for valid i, the eigenvalues are 2*cos(2πjk/(n-k)) approximately.
