def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:

def gradient_for_lag(h, k, n):
    """Gradient of f_k(h) = sum h[i]*(1-h[i+k]) (without dx factor)"""

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-15:
        j = int(np.argmin(np.abs(h - 0.5)))
        h[j] = np.clip(h[j] + diff, 0.0, 1.0)
    return h

def gradient_for_lag(h, k, n):
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            grad[:m] += (1.0 - h[k:k+m])
            if n - 1 - k > 0:
                grad[k:n-1] += (-h[:n-1-k])
    else:
        la = -k
        m = n - la
        if m > 0:
            grad[la:] += (1.0 - h[:m])
            grad[:m] += (-h[la:])
    return grad

def solve_vfo(grad):
    n = len(grad)
    target = n / 2.0
    order = np.argsort(grad)
    h_vf = np.zeros(n)
    remaining = target
    for idx in order:
        if remaining >= 1.0 - 1e-12:
            h_vf[idx] = 1.0
            remaining -= 1.0
        elif remaining > 1e-12:
            h_vf[idx] = remaining
            remaining = 0.0
            break
    return h_vf

def fw_optimize(h, n, deadline, max_iter=80000, top_k=3):
    ts = n / 2.0
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    alpha = 1.0
    stagnant = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        tk = top_k if it > 200 else 1
        top_idx = np.argsort(corr)[-tk:]
        w = np.exp(40.0 * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        grad = np.zeros(n)
        for j in range(tk):
            k = int(top_idx[j]) - (n - 1)
            grad += w[j] * gradient_for_lag(h, k, n)
        h_vf = solve_vfo(grad)
        best_a = 0.0
        best_obj = c5
        for a in [alpha, 0.5 * alpha, 0.25 * alpha]:
            h_try = (1.0 - a) * h + a * h_vf
            h_try = project_feasible(h_try, ts)
            c_try = compute_c5(h_try)
            if c_try < best_obj:
                best_obj = c_try
                best_a = a
        if best_a > 0:
            h = (1.0 - best_a) * h + best_a * h_vf
            h = project_feasible(h, ts)
            c5 = compute_c5(h)
            alpha = min(best_a * 1.4, 1.0)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            alpha *= 0.5
            stagnant += 1
        if stagnant > 400 and alpha < 0.005:
            break
    return best_h, best_c5

def _con_fun(h, ts):
    return h.sum() - ts

def _con_jac(h, ts):
    return np.ones_like(h)

def _smooth_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -700.0, 700.0)
    return (m + np.log(np.sum(np.exp(sh))) / beta) * dx

def slsqp_polish(h, n, deadline, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (ts,)}]
    for attempt in range(6):
        if time.time() > deadline:
            break
        x0 = h.copy()
        if attempt > 0:
            noise = 0.003 * (attempt + 1)
            x0 = project_feasible(x0 + np.random.randn(n) * noise, ts)
        beta = 30000 + 20000 * attempt
        try:
            res = minimize(_smooth_obj, x0, args=(dx, beta), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 4000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        except Exception:
            pass
    return best_h, best_c5

def subgrad_polish(h, n, deadline, seed):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    velocity = np.zeros(n)
    lr = 0.003
    stagnant = 0
    for it in range(50000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-4:]
        w = np.exp(15.0 * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        grad = np.zeros(n)
        for j in range(4):
            k = int(top_idx[j]) - (n - 1)
            grad += w[j] * gradient_for_lag(h, k, n)
        velocity = 0.8 * velocity - lr * grad
        h_new = project_feasible(h + velocity, ts)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h_new.copy()
            stagnant = 0
            lr = min(lr * 1.01, 0.006)
        elif c_new < c5:
            h = h_new
            c5 = c_new
            stagnant += 1
        else:
            stagnant += 1
            velocity *= 0.3
            lr *= 0.9
        if stagnant > 400 and lr < 1e-5:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    if parent is None:
        parent = np.full(350, 0.5)
    pn = len(parent)
    parent = project_feasible(parent, pn / 2.0)

    best_c5 = compute_c5(parent)
    best_n = pn

    # Phase 1: FW on parent resolution
    d1 = t0 + budget_s * 0.45
    h1, c1 = fw_optimize(parent.copy(), pn, d1, max_iter=100000, top_k=3)
    if c1 < best_c5:

    # Try binary rounding then re-optimize
    order = np.argsort(best_h)[::-1]
    h_bin = np.zeros(best_n)
    h_bin[order[:best_n // 2]] = 1.0
    c_bin = compute_c5(h_bin)
    if c_bin < best_c5 + 0.01:
        h_b2, c_b2 = fw_optimize(h_bin, best_n, d1, max_iter=20000, top_k=5)
        if c_b2 < best_c5:
            best_h, best_c5 = h_b2.copy(), c_b2

    # Phase 2: SLSQP + subgradient polish
    d2 = t0 + budget_s * 0.65

    res_list = [best_n - 40, best_n - 20, best_n + 20, best_n + 40, 200, 280, 400, 450, 500, 550]
