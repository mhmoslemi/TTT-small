from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_fn(h_in, beta, n, target):
    corr = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(shifted)
    return (m + np.log(exps.sum()) / beta) * (2.0 / n)

def grad_fn(h_in, beta, n, target):
    corr = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    grad = np.zeros(n)
    lags = np.abs(np.arange(len(corr)) - (n - 1))
    sig = w > 1e-8
    for idx in np.where(sig)[0]:
        k = int(lags[idx])
        wk = w[idx]
        if k == 0:
            grad += wk * (1.0 - 2.0 * h_in)
        else:
            grad[k:] += wk * (1.0 - h_in[:n-k])
            grad[:n-k] -= wk * h_in[k:]
    return grad * (2.0 / n)

def cstr_fun(h_in, beta, n, target):
    return np.sum(h_in) - target

def cstr_jac(h_in, beta, n, target):
    return np.ones(n)

def do_slsqp(h0, n, target, beta, maxiter=5000, ftol=1e-16):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': cstr_fun, 'jac': cstr_jac, 'args': (beta, n, target)}]
    try:
        r = minimize(obj_fn, h0, args=(beta, n, target), jac=grad_fn,
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def subgrad_desc(h, target, n, n_iter, step_init, t0, budget, top_k=8):
    cur = h.copy()
    best_local = compute_c5(cur)
    stall = 0
    step = step_init
    mom = np.zeros(n)
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            k = abs(int(idx) - (n - 1))
            if k == 0:
                grad += 1.0 - 2.0 * cur
            else:
                grad[k:] += (1.0 - cur[:n-k])
                grad[:n-k] -= cur[k:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = 0.9 * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            stall = 0
            step = min(step * 1.03, 0.08)
        else:
            stall += 1
            step *= 0.75
            mom *= 0.3
            if stall > 300:
                break
    return cur, best_local

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.03, -0.03, 0.015, -0.015, 0.007, -0.007, 0.003, -0.003]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0.0, 1.0)
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:

    return cur, best_c5

def anneal_binary(h_bin, n, t0, budget, n_iter=50000):
    h = h_bin.copy()
    cur_max = np.max(np.correlate(h, 1.0 - h, mode="full"))
    best_h = h.copy()
    best_max = cur_max
    rng = np.random.default_rng(123)
    for it in range(n_iter):
        if budget - (time.time() - t0) < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        new_max = np.max(np.correlate(h, 1.0 - h, mode="full"))
        delta = new_max - cur_max
        if delta < 0:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            frac = it / max(n_iter, 1)
            temp = 8.0 * (0.01) ** frac
            if rng.random() < np.exp(-delta / max(temp, 1e-10)):
                cur_max = new_max
            else:
                h[a] = 1.0
                h[b] = 0.0
    return best_h, best_max

def block_perturb(h, n, rng):

    cur = h.copy()
    target = n / 2.0
    blen = rng.integers(4, max(5, n // 10))
    start = rng.integers(0, n - blen)
    # Randomly modify the block
    cur[start:start+blen] = np.clip(cur[start:start+blen] + rng.standard_normal(blen) * 0.3, 0, 1)
    return project_h(cur, target)

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()
    
    # Get parent

    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target = n / 2.0
    rng = np.random.default_rng(seed)

    if budget_s - (time.time() - t0) > 30:
        for beta in [200, 500, 1500, 5000, 15000]:

    # Phase 3: More SLSQP

            if budget_s - (time.time() - t0) < 10:

        # Make it exactly binary

        # Refine from binary

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_fn(h_in, beta, n, target):
    corr = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(shifted)
    return (m + np.log(exps.sum()) / beta) * (2.0 / n)

def grad_fn(h_in, beta, n, target):
    corr = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    grad = np.zeros(n)
    center = n - 1
    for idx in range(len(corr)):
        if w[idx] < 1e-10:
            continue
        wk = w[idx]
        lag = idx - center
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h_in)
        elif lag > 0:
            grad[lag:] += wk * (1.0 - h_in[:-lag])
            grad[:n - lag] -= wk * h_in[lag:]
        else:
            k = -lag
            grad[:n - k] += wk * (1.0 - h_in[k:])
            grad[k:] -= wk * h_in[:n - k]
    return grad * (2.0 / n)

def cstr_fun(h_in, beta, n, target):
    return np.sum(h_in) - target

def cstr_jac(h_in, beta, n, target):
    return np.ones(n)

def do_slsqp(h0, n, target, beta, maxiter=5000, ftol=1e-16):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': cstr_fun, 'jac': cstr_jac, 'args': (beta, n, target)}]
    try:
        r = minimize(obj_fn, h0, args=(beta, n, target), jac=grad_fn,
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def subgrad_desc(h, target, n, n_iter, step_init, t0, budget, top_k=8):
    cur = h.copy()
    best_local = compute_c5(cur)
    stall = 0
    step = step_init
    mom = np.zeros(n)
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        center = n - 1
        for idx in top_idx:
            lag = idx - center
            if lag == 0:
                grad += 1.0 - 2.0 * cur
            elif lag > 0:
                grad[lag:] += (1.0 - cur[:-lag])
                grad[:n - lag] -= cur[lag:]
            else:
                k = -lag
                grad[:n - k] += (1.0 - cur[k:])
                grad[k:] -= cur[:n - k]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = 0.9 * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            stall = 0
            step = min(step * 1.03, 0.08)
        else:
            stall += 1
            step *= 0.75
            mom *= 0.3
            if stall > 400:
                break
    return cur, best_local

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.03, -0.03, 0.015, -0.015, 0.007, -0.007, 0.003, -0.003]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0.0, 1.0)
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c5

def anneal_binary(h_bin, n, t0, budget, n_iter=80000):
    h = h_bin.copy()
    cur_max = np.max(np.correlate(h, 1.0 - h, mode="full"))
    best_h = h.copy()
    best_max = cur_max
    rng = np.random.default_rng(999)
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        new_max = np.max(np.correlate(h, 1.0 - h, mode="full"))
        delta = new_max - cur_max
        if delta < 0:
            cur_max = new_max
            stall = 0
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            stall += 1
            frac = min(it / max(n_iter, 1), 1.0)
            temp = 10.0 * (0.005) ** frac
            if stall > 5000 and temp < 0.01:
                temp = 0.5
                stall = 0
            if rng.random() < np.exp(-delta / max(temp, 1e-10)):
                cur_max = new_max
            else:
                h[a] = 1.0
                h[b] = 0.0
    return best_h, best_max

def block_perturb(h, n, rng):
    cur = h.copy()
    target = n / 2.0
    blen = int(rng.integers(3, max(4, n // 12)))
    start = int(rng.integers(0, n - blen))
    cur[start:start + blen] = np.clip(
        cur[start:start + blen] + rng.standard_normal(blen) * 0.4, 0, 1)
    return project_h(cur, target)

def shift_perturb(h, n, rng):
    cur = h.copy()
    target = n / 2.0
    s = int(rng.integers(1, max(2, n // 16)))
    direction = 1 if rng.random() > 0.5 else -1
    cur = np.roll(cur, direction * s)
    return project_h(cur, target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 384
        h_parent = np.zeros(n_init)
        h_parent[::2] = 1.0

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target = n / 2.0
    rng = np.random.default_rng(seed)

    # Phase 1: Gradient SLSQP with increasing beta
    if budget_s - (time.time() - t0) > 40:
        for beta in [150, 500, 2000, 8000, 25000]:
            if budget_s - (time.time() - t0) < 12:
                break
            best_h = do_slsqp(best_h, n, target, beta, maxiter=8000, ftol=1e-16)
            best_c5 = compute_c5(best_h)

    # Phase 2: Subgradient refinement
    if budget_s - (time.time() - t0) > 30:
        best_h, best_c5 = subgrad_desc(best_h, target, n, 12000, 0.02, t0, budget_s, top_k=8)

    # Phase 3: High-beta SLSQP
    if budget_s - (time.time() - t0) > 20:
        for beta in [8000, 30000, 80000]:
            if budget_s - (time.time() - t0) < 10:
                break
            best_h = do_slsqp(best_h, n, target, beta, maxiter=10000, ftol=1e-17)
            best_c5 = compute_c5(best_h)

    # Phase 4: Binary annealing
    if budget_s - (time.time() - t0) > 40:
        idx = np.argsort(best_h)[::-1]
        h_bin = np.zeros(n)
        h_bin[idx[:n // 2]] = 1.0
        h_bin, _ = anneal_binary(h_bin, n, t0, budget_s, n_iter=120000)
        cur = h_bin + 0.008 * rng.standard_normal(n)
        cur = project_h(cur, target)
        for beta in [500, 3000, 15000]:
            if budget_s - (time.time() - t0) < 10:
                break
            cur = do_slsqp(cur, n, target, beta, maxiter=8000)
            c_try = compute_c5(cur)
            if c_try < best_c5:
                best_h = cur.copy()
                best_c5 = c_try

    # Phase 5: Diverse restarts
    if budget_s - (time.time() - t0) > 80:
        for att in range(10):
            if budget_s - (time.time() - t0) < 25:
                break
            if att % 3 == 0:
                hp = block_perturb(best_h, n, rng)
            elif att % 3 == 1:
                scale = [0.04, 0.02, 0.01, 0.06][att % 4]
                hp = best_h + rng.standard_normal(n) * scale
                hp = project_h(hp, target)
            else:
                hp = shift_perturb(best_h, n, rng)
            hp = do_slsqp(hp, n, target, 800, maxiter=5000)
            hp = do_slsqp(hp, n, target, 5000, maxiter=6000)
            hp, _ = subgrad_desc(hp, target, n, 4000, 0.01, t0, budget_s, top_k=6)
            hp = do_slsqp(hp, n, target, 20000, maxiter=6000)
            c_try = compute_c5(hp)
            if c_try < best_c5:
                best_h = hp.copy()
                best_c5 = c_try

    # Phase 6: Final SLSQP + coordinate descent
    if budget_s - (time.time() - t0) > 8:
        best_h = do_slsqp(best_h, n, target, 50000, maxiter=12000, ftol=1e-17)
        best_c5 = compute_c5(best_h)
    if budget_s - (time.time() - t0) > 4:
        best_h, best_c5 = coord_descent(best_h, target, n, 200, t0, budget_s)

    return best_h, best_c5, n
