def smooth_max_grad(corr, beta):
    """Returns (smooth_max_value, gradient_weights) where gradient_weights[k] is the weight for lag k."""
    m = np.max(corr)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:

        h = np.clip(h, 0.0, 1.0)

    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def grad_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

        lag = idx - (n - 1)

        else:

            grad[la:] += w * (1.0 - h[:n-la])
            grad[:n-la] -= w * h[la:]

def binary_local_search(h0, target, n, t0, budget, max_iter=20000):

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def grad_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700, 700)
    exps = np.exp(shifted)
    total = np.sum(exps)
    sm_val = (m + np.log(total) / beta) * dx
    weights = exps / total
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-10:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n - lag]
        else:
            la = -lag
            grad[la:] += w * (1.0 - h[:n - la])
            grad[:n - la] -= w * h[la:]
    return sm_val, grad * dx

def adam_optimize(h0, n, target, n_iter, t0, budget, beta_start=200, lr_init=0.02):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    best_h, best_c5 = h.copy(), compute_c5(h)
    beta_cur = beta_start
    lr = lr_init
    stall = 0
    
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        if it > 0 and it % 400 == 0:
            beta_cur = min(beta_cur * 1.8, 200000)
            lr *= 0.8
            if stall > 200:
                lr *= 0.5
                stall = 0
        
        val, g = grad_smooth_max(h, beta_cur)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1 ** (it + 1))
        v_hat = v / (1 - b2 ** (it + 1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_h(h_new, target)
        
        c = compute_c5(h_new)
        if c < best_c5 - 1e-12:
            best_h, best_c5, stall = h_new.copy(), c, 0
        else:
            stall += 1
        h = h_new
    
    return best_h, best_c5

def binary_sa(h0, target, n, t0, budget, max_iter=50000):
    """Simulated annealing on binary configurations."""
    b = (h0 > 0.5).astype(float)
    diff = target - b.sum()
    if diff > 0:
        zeros = np.where(b == 0)[0]
        b[zeros[np.argsort(h0[zeros])[::-1]][:int(round(diff))]] = 1.0
    elif diff < 0:
        ones = np.where(b == 1)[0]
        b[ones[np.argsort(h0[ones])][:int(round(-diff))]] = 0.0
    
    best_b = b.copy()
    best_c5 = compute_c5(b)
    cur_c5 = best_c5
    rng = np.random.RandomState(77)
    T = 0.01
    
    for it in range(max_iter):
        if time.time() - t0 > budget - 5:
            break
        if it % 1000 == 0 and it > 0:
            T *= 0.95
        
        ones_idx = np.nonzero(b)[0]
        zeros_idx = np.nonzero(1 - b)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_b, best_c5 = b.copy(), new_c5
        else:
            b[i] = 1
            b[j] = 0
    
    return best_b, best_c5

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -700, 700)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)
    
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(384, 0.5)
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(h_parent)
    
    # Multi-resolution cascade
    resolutions = sorted(set([len(h_parent)] + [r for r in [384, 512, 640] if r >= len(h_parent)]))
    
    for n_pts in resolutions:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        
        target = n_pts / 2.0
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
        cur = project_h(resample(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts
        
        # Phase 1: ADAM (main optimization)
        rem = budget_s - (time.time() - t0)
        if rem > 80:
            cur, cur_c5 = adam_optimize(cur, n_pts, target, 8000, t0, budget_s, beta_start=200, lr_init=0.02)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts
        
        # Phase 2: Binary simulated annealing
        rem = budget_s - (time.time() - t0)
        if rem > 60:
            b, bc = binary_sa(cur, target, n_pts, t0, budget_s, max_iter=40000)
            if bc < best_c5:
                best_h, best_c5, best_n = b.copy(), bc, n_pts
                cur = b.copy()
                cur_c5 = bc
        
        # Phase 3: ADAM again (from binary or previous)
        rem = budget_s - (time.time() - t0)
        if rem > 50:
            cur = project_h(cur, target)
            cur, cur_c5 = adam_optimize(cur, n_pts, target, 5000, t0, budget_s, beta_start=500, lr_init=0.01)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts
        
        # Phase 4: SLSQP polish
        rem = budget_s - (time.time() - t0)
        if rem > 30:
            cur = do_slsqp(cur, n_pts, target, 3000, bounds, 3000, 1e-15)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts
    
    # Final polish at best resolution
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        nf = len(best_h)
        tf = nf / 2.0
        bf = Bounds(np.zeros(nf), np.ones(nf))
        best_h = do_slsqp(best_h, nf, tf, 5000, bf, 5000, 1e-15)
        best_c5 = compute_c5(best_h)
    
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)

    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:

                stall = 0

        m_hat = m / (1 - b1 ** (it + 1))
        v_hat = v / (1 - b2 ** (it + 1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_h(h_new, target)

        if c < best_c5 - 1e-12:
            best_h, best_c5, stall = h_new.copy(), c, 0
        else:
            stall += 1
        h = h_new
    return best_h, best_c5

def binary_sa(h0, target, n, t0, budget, max_iter=50000):
    b = (h0 > 0.5).astype(float)
    diff = target - b.sum()
    if diff > 0:
        zeros = np.where(b == 0)[0]
        b[zeros[np.argsort(h0[zeros])[::-1]][:int(round(diff))]] = 1.0
    elif diff < 0:
        ones = np.where(b == 1)[0]
        b[ones[np.argsort(h0[ones])][:int(round(-diff))]] = 0.0
    best_b = b.copy()
    best_c5 = compute_c5(b)
    cur_c5 = best_c5
    rng = np.random.RandomState(77)

    for it in range(max_iter):

            break
        if it % 2000 == 0 and it > 0:
            T *= 0.93
        ones_idx = np.nonzero(b)[0]
        zeros_idx = np.nonzero(1 - b)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        b[i] = 0; b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 < cur_c5 or (T > 1e-8 and rng.random() < np.exp((cur_c5 - new_c5) / T)):
            cur_c5 = new_c5

                best_b, best_c5 = b.copy(), new_c5
        else:
            b[i] = 1; b[j] = 0
    return best_b, best_c5

    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -700, 700)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break

        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)

        h = np.clip(h, 0.0, 1.0)

def resample(h, n_new):

def grad_smooth_max(h, beta):

    total = np.sum(exps)
