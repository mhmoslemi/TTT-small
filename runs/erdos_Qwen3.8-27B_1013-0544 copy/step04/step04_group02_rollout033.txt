import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def coord_grads(h, n):
    """Compute grad[i,k] = d(corr[k])/d(h[i]) for all i,k."""
    grad = np.zeros((n, 2 * n - 1))
    for i in range(n):
        # term1: shift s where 0 <= i-s < n, contribution = 1-h[i-s]
        # k = s + (n-1), s in [max(-(n-1), i-n+1), min(n-1, i)]
        s_lo = max(-(n - 1), i - n + 1)
        s_hi = min(n - 1, i)
        if s_lo <= s_hi:
            s_vals = np.arange(s_lo, s_hi + 1)
            k_idx = s_vals + (n - 1)
            idx = i - s_vals
            grad[i, k_idx] += 1.0 - h[idx]
        # term2: shift s where 0 <= i+s < n, contribution = -h[i+s]
        # k = s + (n-1), s in [max(-(n-1), -i), min(n-1, n-i-1)]
        t_lo = max(-(n - 1), -i)
        t_hi = min(n - 1, n - i - 1)
        if t_lo <= t_hi:
            t_vals = np.arange(t_lo, t_hi + 1)
            k_idx2 = t_vals + (n - 1)
            idx2 = i + t_vals
            grad[i, k_idx2] -= h[idx2]
    return grad

def solve_1d_max_linear(corr, b, dlo, dhi):
    """Minimize max_k(corr[k]+b[k]*d) s.t. dlo<=d<=dhi. Returns optimal d."""
    if dlo >= dhi - 1e-15:
        return 0.0
    lo, hi = dlo, dhi
    for _ in range(50):
        if hi - lo < 1e-15:
            break
        mid = 0.5 * (lo + hi)
        vals = corr + b * mid
        v_max = vals.max()
        active = vals >= v_max - 1e-13
        if not active.any():
            return mid
        sg_min = b[active].min()
        sg_max = b[active].max()
        if sg_min > 1e-14:
            hi = mid
        elif sg_max < -1e-14:
            lo = mid
        else:
            return mid
    return 0.5 * (lo + hi)

def coordinate_descent(h, n, max_sweeps, deadline, order_seed=0):
    """Cyclic coordinate descent with Lagrangian for sum constraint."""
    np.random.seed(order_seed)
    target_sum = n / 2.0
    lam = 0.0
    order = np.arange(n)
    best_c5 = compute_c5(h)
    stagnant = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if sweep % 20 == 0:
            order = np.random.permutation(n)
        grad = coord_grads(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full').copy()
        any_change = False
        for i in order:
            b = grad[i] + lam
            delta = solve_1d_max_linear(corr, b, -h[i], 1.0 - h[i])
            if abs(delta) > 1e-13:
                h[i] += delta
                corr += grad[i] * delta
                any_change = True
        s = h.sum()
        lam += 0.8 * (s - target_sum) / n
        lam = np.clip(lam, -3.0, 3.0)
        h = np.clip(h, 0.0, 1.0)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-9:
            best_c5 = c5
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 200 and sweep > 50:
            break
    return project_feasible(h, target_sum), compute_c5(project_feasible(h, target_sum))

def slsqp_phase(h, n, deadline, seed):
    """Run SLSQP with smooth max at various betas."""
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
    c5 = compute_c5(h)
    for beta in [50, 200, 1000, 5000]:
        if time.time() > deadline:
            break
        for pert in [0.0, 0.005, 0.015]:
            if time.time() > deadline:
                break
            x0 = h.copy()
            if pert > 0:
                x0 = project_feasible(x0 + np.random.randn(n) * pert, ts)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < c5:
                    h = h_new
                    c5 = c5_new
            except Exception:
                pass
    return h, c5

def subgradient_polish(h, n, deadline, seed):
    """Nesterov-accelerated subgradient on smooth max."""
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    velocity = np.zeros(n)
    lr = 0.003
    stagnant = 0
    for it in range(5000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top = np.argsort(corr)[-8:]
        w = np.exp(12.0 * (corr[top] - corr[top].max()))
        w /= w.sum()
        grad = np.zeros(n)
        for j, k in enumerate(top):
            lag = k - (n - 1)
            if lag >= 0:
                m = n - lag
                if m > 0:
                    idx = np.arange(m)
                    grad[idx] += (1.0 - h[idx + lag])
                    grad[idx + lag] -= h[idx]
            else:
                la = -lag
                m = n - la
                if m > 0:
                    idx = np.arange(m)
                    grad[idx + la] += (1.0 - h[idx])
                    grad[idx] -= h[idx + la]
            grad *= w[j]
        t = it / max(1, it + 100)
        lr_t = lr * (1 - t)
        y = h + (it / (it + 1)) * velocity / lr if it > 0 else h
        velocity_new = (it / (it + 1)) * 0.7 * velocity - lr_t * grad
        h_new = project_feasible(y - velocity_new / lr_t * 0.3, ts) if it > 0 else project_feasible(h, ts)
        h_new = project_feasible(h + velocity_new / max(lr_t, 1e-10) * 0.5, ts)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-11:
            h = h_new
            c5 = c_new
            velocity = velocity_new
            stagnant = 0
        else:
            stagnant += 1
            velocity *= 0.5
            lr *= 0.92
        if stagnant > 300 and lr < 3e-6:
            break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    if parent is None:
        parent = np.full(200, 0.5)
    pn = len(parent)

    # Phase 1: SLSQP warmstart (20% budget)
    d1 = t0 + budget_s * 0.20
    h1, c1 = slsqp_phase(parent.copy(), pn, d1, seed)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c1, pn

    # Phase 2: Coordinate descent (55% budget)
    d2 = t0 + budget_s * 0.75
    h2, c2 = coordinate_descent(h1.copy(), pn, max_sweeps=5000, deadline=d2, order_seed=seed)
    if c2 < best_c5:
        best_h, best_c5, best_n = h2.copy(), c2, pn

    # Coordinate descent from reversed
    if time.time() < d2 and pn > 2:
        h_rev = project_feasible(parent[::-1].copy(), pn / 2.0)
        c_rev = compute_c5(h_rev)
        if c_rev < best_c5 + 0.01:
            h3, c3 = coordinate_descent(h_rev.copy(), pn, max_sweeps=2000, deadline=d2, order_seed=seed+7)
            if c3 < best_c5:
                best_h, best_c5, best_n = h3.copy(), c3, pn

    # Phase 3: Subgradient polish (10%)
    d3 = t0 + budget_s * 0.85
    if best_h is not None:
        hp, cp = subgradient_polish(best_h.copy(), best_n, d3, seed + 50)
        if cp < best_c5:
            best_h, best_c5 = hp.copy(), cp

    # Phase 4: More coordinate descent on best (up to 92%)
    d4 = t0 + budget_s * 0.92
    if best_h is not None:
        h4, c4 = coordinate_descent(best_h.copy(), best_n, max_sweeps=3000, deadline=d4, order_seed=seed+111)
        if c4 < best_c5:
            best_h, best_c5 = h4.copy(), c4

    # Phase 5: Try nearby resolutions
    d5 = t0 + budget_s * 0.96
    if best_h is not None and time.time() < d5:
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        for n_try in [best_n - 20, best_n + 20]:
            if n_try < 100 or n_try > 500 or time.time() > d5:
                continue
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h0 = np.interp(x_new, x_old, best_h)
            h0 = project_feasible(h0, n_try / 2.0)
            h5, c5 = coordinate_descent(h0, n_try, max_sweeps=1000, deadline=d5, order_seed=seed + n_try)
            if c5 < best_c5:
                best_h, best_c5, best_n = h5.copy(), c5, n_try

    # Final projection and verification
    if best_h is None:
        best_h = project_feasible(parent.copy(), pn / 2.0)
        best_n = pn
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
