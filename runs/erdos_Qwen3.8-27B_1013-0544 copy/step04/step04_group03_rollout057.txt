                g[i + la] -= h_star[i]
        
        # C[k] at h_star
        if lag >= 0:
            c_val = np.sum(h_star[lag:] * (1.0 - h_star[:n - lag]))
        else:
            la = -lag
            c_val = np.sum(h_star[:n - la] * (1.0 - h_star[la:]))
        
        # Linearized: (c_val + g·(h - h_star)) * dx <= t
        # => dx * g·h - t <= dx * (c_val - g·h_star)
        rhs = dx * (c_val - np.dot(g, h_star))
        constraints.append(dx * cp.sum(cp.multiply(g, h_var)) - t_var <= rhs)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
        if h_var.value is not None:
            return project_h(h_var.value, n / 2.0)
    except:
        pass
    return h_star.copy()

def slsqp_opt(h0, n, target, beta, maxiter=5000):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(c)
        x_shift = np.clip(beta * (c - m), -400, 400)
        return (m + np.log(np.mean(np.exp(x_shift))) / beta) * (2.0 / n)
    
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(c)
        x_shift = np.clip(beta * (c - m), -400, 400)
        weights = np.exp(x_shift)
        weights /= np.sum(weights)
        # gradient of smooth max
        grad_sm = np.zeros(len(c))
        grad_sm += weights  # d/dc of smooth max = weights
        # Now d/smooth_max/dh[j]
        g = np.zeros(n)
        for k in range(len(c)):
            lag = k - (n - 1)
            if abs(lag) >= n:
                continue
            w = grad_sm[k]
            if lag >= 0:
                g[:n - lag] += w * (1.0 - x[lag:])
                g[lag:] += w * (-x[:n - lag])
            else:
                la = -lag
                g[la:] += w * (1.0 - x[:n - la])
                g[:n - la] += w * (-x[la:])
        return g * (2.0 / n)
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except:
        return h0

        h_lp = lp_step(cur, n, active_lags=active)
        if np.allclose(h_lp, cur, atol=1e-8):
            break
        best_alpha = 0.0
        best_trial = best_c5
        for alpha in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
            trial = project_h(cur + alpha * (h_lp - cur), target)
            c_try = compute_c5(trial)
            if c_try < best_trial:
                best_trial = c_try
                best_alpha = alpha
        if best_alpha > 0:
            cur = project_h(cur + best_alpha * (h_lp - cur), target)
            best_c5 = best_trial
        else:
            break
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(384, 0.5)
        h_parent[:192] = 0.7
        h_parent[192:] = 0.3
        h_parent = project_h(h_parent, 192.0)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)
    target = best_n / 2.0

    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        for beta in [300, 2000, 10000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 12:
                break
            h_new = slsqp_opt(best_h, best_n, target, beta, maxiter=6000)
            c_try = compute_c5(h_new)
            if c_try < best_c5:
                best_h = h_new
                best_c5 = c_try

    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        best_h, best_c5 = subgrad_thresh(best_h, target, best_n, 12000, 0.03, t0, budget_s, threshold=0.003)

    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        best_h, best_c5 = lp_polish(best_h, target, best_n, t0, budget_s, max_iters=40, gap=0.005)

    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        for beta in [20000, 80000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 12:
                break
            h_new = slsqp_opt(best_h, best_n, target, beta, maxiter=8000)
            c_try = compute_c5(h_new)
            if c_try < best_c5:
                best_h = h_new
                best_c5 = c_try

    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        best_h, best_c5 = subgrad_thresh(best_h, target, best_n, 8000, 0.015, t0, budget_s, threshold=0.001)

    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        best_h, best_c5 = lp_polish(best_h, target, best_n, t0, budget_s, max_iters=30, gap=0.003)

    remaining = budget_s - (time.time() - t0)
    for n_try in [320, 420, 448]:
        if remaining < 50:
            break
        target_try = n_try / 2.0
        h_try = project_h(resample_fn(best_h, n_try), target_try)
        for beta in [500, 5000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 12:
                break
            h_try = slsqp_opt(h_try, n_try, target_try, beta, maxiter=5000)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_h = h_try
                best_c5 = c_try
                best_n = n_try
                target = target_try
        remaining = budget_s - (time.time() - t0)
        if remaining > 25:
            h_try, c_try = lp_polish(h_try, n_try / 2.0, n_try, t0, budget_s, max_iters=15, gap=0.004)
            if c_try < best_c5:
                best_h = h_try
                best_c5 = c_try
                best_n = n_try
                target = target_try

    remaining = budget_s - (time.time() - t0)
    rng = np.random.default_rng(seed + 55)
    for att in range(5):
        if remaining < 25:
            break
        scales = [0.03, 0.015, 0.008, 0.025, 0.012]
        hp = best_h + rng.standard_normal(best_n) * scales[att]
        hp = project_h(hp, target)
        for beta in [1000, 5000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            hp = slsqp_opt(hp, best_n, target, beta, maxiter=4000)
            hp, _ = lp_polish(hp, target, best_n, t0, budget_s, max_iters=10, gap=0.004)
            c_try = compute_c5(hp)
            if c_try < best_c5:
                best_h = hp.copy()
                best_c5 = c_try

    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        best_h, best_c5 = subgrad_thresh(best_h, target, best_n, 5000, 0.01, t0, budget_s, threshold=0.0005)

    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        best_h, best_c5 = lp_polish(best_h, target, best_n, t0, budget_s, max_iters=20, gap=0.002)

    remaining = budget_s - (time.time() - t0)
    if remaining > 8:
        best_h, best_c5 = coord_polish(best_h, target, best_n, 150, t0, budget_s)

    return best_h, best_c5, best_n
