    dx = 2.0 / n
    g = np.correlate(x, 1 - x, 'full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    w = exps / exps.sum()

    # dg[k]/dh[j] = dx * [(1-x[j+s]) if 0<=j+s<n - x[j-s] if 0<=j-s<n]

Let me think about what "direct max targeting" means:
- Find k* = argmax_k g(k)
- Try to minimize g(k*) subject to g(k) <= g(k*) + eps for all k
- This is a convex optimization problem (well, g(k*) is not convex in h since it's a bilinear form, but...)

Actually, g(k) = sum_i h[i]*(1-h[i+s]) = sum_i h[i] - sum_i h[i]*h[i+s]
The first term is fixed (=n/2), so minimizing g(k) is equivalent to maximizing sum_i h[i]*h[i+s].
This is a quadratic form: h^T A_s h where A_s is a specific matrix.
Maximizing a quadratic form subject to box and sum constraints is generally non-convex.

However, if we fix which components are "active" (near 0 or 1), the problem becomes easier.

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, 'full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def softmax_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, 'full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        wk = w[k] * dx
        s = k - (n - 1)
        js = max(0, -s)
        je = min(n - 1, n - 1 - s)
        if js <= je:
            grad[js:je+1] += wk * (1 - x[js+s:je+1+s])
        js2 = max(0, s)
        je2 = min(n - 1, s + n - 1)
        if js2 <= je2:
            grad[js2:je2+1] -= wk * x[js2-s:je2+1-s]
    return grad
