import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad(h, indices):
    n = len(h)
    g = np.zeros(n)
    nh = n - 1
    for j in indices:
        lag = j - nh
        if lag == 0:
            g += (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += (1.0 - h[:-lag])
            g[:n - lag] -= h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += (1.0 - h[m2:])
            g[m2:] -= h[:n - m2]
    return g

def soft_max_obj(h, beta):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    s = np.clip(beta * (corr - m), -50, 50)
    return (m + np.log(np.exp(s).sum()) / beta) * dx

def soft_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    s = np.clip(beta * (corr - m), -50, 50)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    nh = n - 1
    for j in range(len(corr)):
        wj = w[j]
        if wj < 1e-12:
            continue
        lag = j - nh
        if lag == 0:
            g += wj * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wj * (1.0 - h[:-lag])
            g[:n - lag] -= wj * h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += wj * (1.0 - h[m2:])
            g[m2:] -= wj * h[:n - m2]
    return g * dx

def slsqp_step(h, n, beta, maxiter=1500, ftol=1e-16):
    from scipy.optimize import minimize
    target = n / 2.0
    bounds = [(0, 1)] * n
    def obj(x):
        return soft_max_obj(np.clip(x, 0, 1), beta)
    def grad(x):
        return soft_max_grad(np.clip(x, 0, 1), beta)
    def eq_fun(x):
        return np.sum(x) - target
    def eq_jac(x):
        return np.ones(n)
    res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_sum(res.x, target)

def multi_polish(h, target, t0, budget, max_iter=80000):
    n = len(h)
    best = h.copy()
    best_val = compute_c5(best)
    no_improve = 0
    steps = [0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002, 0.00008]
    active_strategies = [1, 2, 3, 5, 8, 13, 21]

    for it in range(max_iter):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        cmax = corr.max()
        improved = False

        # Cycle through active set sizes
        for n_act in active_strategies:
            if n_act >= len(corr):
                break
            top_idx = np.argsort(corr)[::-1][:n_act]
            g = compute_grad(best, top_idx)
            g -= g.mean()
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                continue
            g /= gnorm
            for sign in [-1, 1]:
                for st in steps:
                    ht = project_sum(best + sign * st * g, target)
                    c5t = compute_c5(ht)
                    if c5t < best_val - 1e-15:
                        best = ht
                        best_val = c5t
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 50 and no_improve % 80 == 0:
                scale = 0.004 * min(no_improve / 500.0, 1.0)
                pert = np.random.randn(n) * scale
                ht = project_sum(best + pert, target)
                c5t = compute_c5(ht)
                if c5t < best_val:
                    best = ht
                    best_val = c5t
                    no_improve = 0
            if no_improve > 4000:
                break
        else:
            no_improve = 0

    return best, best_val

    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

            best_c5 = c5_new

    # Phase 2: Intensive polishing at current resolution

    # Phase 3: Try higher resolution
    for n_try in [560, 640, 800]:

    best_h, best_c5 = multi_polish(best_h, n / 2.0, t0, budget_s, max_iter=200000)
