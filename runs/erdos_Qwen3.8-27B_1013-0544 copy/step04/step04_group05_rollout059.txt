    best = h.copy()
    best_c5 = compute_c5(best)
    for beta in betas:
        if time_budget_remaining() < 20:
            break

            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, beta) * dx

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_phase(h, target, dx, betas, max_iter_per_beta, rem_fn):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    for beta in betas:
        if rem_fn() < 25:
            break
        def obj(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, beta) * dx
        def con(h_in):
            return h_in.sum() - target
        def jac(h_in):
            return np.ones(n)
        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con, 'jac': jac}],
                          options={'maxiter': max_iter_per_beta, 'ftol': 1e-18})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def subgrad_phase(h, target, rem_fn, max_iters, m_start, m_end, step_init,
                  momentum, perturb_every, perturb_scale, weight_beta):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    step = step_init
    no_improve = 0
    for it in range(max_iters):
        if rem_fn() < 3:
            break
        frac = it / max(max_iters - 1, 1)
        m = max(5, int(m_start + (m_end - m_start) * frac))
        corr = np.correlate(best, 1.0 - best, mode="full")
        top_idx = np.argsort(corr)[::-1][:m]
        top_vals = corr[top_idx]
        w = np.exp(weight_beta * (top_vals - top_vals[0]))
        w /= np.sum(w)
        grad = np.zeros(n)
        for wi, ix in zip(w, top_idx):
            if wi < 1e-8:
                continue
            lag = int(ix) - (n - 1)
            if lag == 0:
                grad += wi * (1.0 - 2.0 * best)
            elif lag > 0:
                grad[lag:] += wi * (1.0 - best[:-lag])
                grad[:n - lag] -= wi * best[lag:]
            else:
                ml = -lag
                grad[:n - ml] += wi * (1.0 - best[ml:])
                grad[ml:] -= wi * best[:n - ml]
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1.0 - momentum) * (grad / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn
        improved = False
        s = step
        for _ in range(12):
            ht = project_h(best + s * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                no_improve = 0
                step = min(s * 1.4, step_init * 2.0)
                improved = True
                break
            s *= 0.6
        if not improved:
            step = max(s * 0.8, step_init * 0.01)
            no_improve += 1
            if no_improve % perturb_every == 0:
                p = np.random.randn(n) * perturb_scale
                ht = project_h(best + p, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    vel = np.zeros(n)
                    no_improve = 0
                    step = step_init
                elif no_improve > perturb_every * 15:
                    break
    return best, best_c5
