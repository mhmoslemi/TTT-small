```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def active_gradient(h, n, m_star):
    """O(n) analytical gradient of g(m_star) w.r.t. h."""
    shift = m_star - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    j1 = max(0, shift)
    j2 = min(n, shift + n)
    if j2 > j1:
        grad[j1:j2] += (1 - h[j1 - shift:j2 - shift])
    j3 = max(0, -shift)
    j4 = min(n, n - shift)
    if j4 > j3:
        grad[j3:j4] -= h[j3 + shift:j4 + shift]
    return grad * dx

def multi_active_gradient(h, n, weights, m_indices):
    """Gradient of weighted sum of active lags."""
    grad = np.zeros(n)
    dx = 2.0 / n
    for w, m in zip(weights, m_indices):
        shift = m - (n - 1)
        j1 = max(0, shift)
        j2 = min(n, shift + n)
        if j2 > j1:
            grad[j1:j2] += w * (1 - h[j1 - shift:j2 - shift])
        j3 = max(0, -shift)
        j4 = min(n, n - shift)
        if j4 > j3:
            grad[j3:j4] -= w * h[j3 + shift:j4 + shift]
    return grad * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgrad_descent(h, n, deadline, max_iter=20000):
    """Projected subgradient with backtracking and multi-lag smoothing."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    g = compute_all_g(h, n)
    m = g.max()
    lr = 0.5
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        g = compute_all_g(h, n)
        m = g.max()
        
        # Find top lags within 5% of max
        threshold = m - 0.05 * max(m, 1e-10)
        active_idx = np.where(g >= threshold)[0]
        
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(g)])
        
        # Use uniform weights for active lags
        weights = np.ones(len(active_idx)) / len(active_idx)
        grad = multi_active_gradient(h, n, weights, active_idx)
        
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        
        # Backtracking line search
        found = False
        step = lr
        for _ in range(30):
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-14:
                h = ht
                found = True
                break
            step *= 0.7
        
        if not found:
            lr *= 0.9
            if lr < 1e-14:
                break
        else:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            # Slowly increase lr if we're making progress
            lr = min(lr * 1.02, 2.0)
    
    return best_h, best_c
```

```python
def lse_gradient(h, n, alpha):
    """Gradient of log-sum-exp(α * g) / α w.r.t. h."""
    g = compute_all_g(h, n)
    # Softmax weights
    shifted = alpha * (g - g.max())
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    
    # Only compute gradient for lags with significant weight
    threshold = 1e-6
    active = np.where(weights > threshold)[0]
    
    grad = np.zeros(n)
    dx = 2.0 / n
    for m in active:
        w = weights[m]
        shift = m - (n - 1)
        j1 = max(0, shift)
        j2 = min(n, shift + n)
        if j2 > j1:
            grad[j1:j2] += w * (1 - h[j1 - shift:j2 - shift])
        j3 = max(0, -shift)
        j4 = min(n, n - shift)
        if j4 > j3:
            grad[j3:j4] -= w * h[j3 + shift:j4 + shift]
    return grad * dx, float(np.log(exp_shifted.sum()) / alpha + g.max())
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def lse_grad_and_val(h, n, alpha):
    """Compute gradient and value of (1/alpha)*log(sum(exp(alpha*g)))."""
    g = compute_all_g(h, n)
    amax = g.max()
    shifted = alpha * (g - amax)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    weights = exp_s / total
    lse_val = amax + np.log(total) / alpha
    
    # Find active lags
    thresh = 1e-8
    active = np.where(weights > thresh)[0]
    
    grad = np.zeros(n)
    dx = 2.0 / n
    for m in active:
        w = weights[m]
        shift = m - (n - 1)
        j1 = max(0, shift)
        j2 = min(n, shift + n)
        if j2 > j1:
            grad[j1:j2] += w * (1 - h[j1 - shift:j2 - shift])
        j3 = max(0, -shift)
        j4 = min(n, n - shift)
        if j4 > j3:
            grad[j3:j4] -= w * h[j3 + shift:j4 + shift]
    return grad * dx, lse_val

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def proj_grad_descent(h, n, deadline, alphas):
    """Projected gradient with LSE smooth approximation."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        lr = 1.0 / (alpha * 0.005 + 1.0)
        stall = 0
        prev_c = best_c
        for it in range(20000):
            if time.time() > deadline:
                break
            grad, _ = lse_grad_and_val(h, n, alpha)
            ng = np.linalg.norm(grad)
            if ng < 1e-15:
                break
            # Try step
            ht = project_feasible(h - lr * grad, n)
            ct = compute_c5(ht, n)
            if ct < prev_c - 1e-14:
                h = ht
                prev_c = ct
                if ct < best_c:
                    best_c = ct
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.05, 5.0 / (ng + 1e-10))
            else:
                stall += 1
                lr *= 0.85
                if stall > 40:
                    break
        h = best_h.copy()
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    """SLSQP on LSE smooth approximation."""
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        s = alpha * (g - m)
        sm = s.max()
        return m + (sm + np.log(np.exp(s - sm).sum())) / alpha
    
    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        s = alpha * (g - m)
        sm = s.max()
        exp_s = np.exp(s - sm)
        weights = exp_s / exp_s.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        active = np.where(weights > 1e-10)[0]
        for mi in active:
            w = weights[mi]
            shift = mi - (n - 1)
            j1 = max(0, shift)
            j2 = min(n, shift + n)
            if j2 > j1:
                grad[j1:j2] += w * (1 - x[j1 - shift:j2 - shift])
            j3 = max(0, -shift)
            j4 = min(n, n - shift)
            if j4 > j3:
                grad[j3:j4] -= w * x[j3 + shift:j4 + shift]
        return grad * dx
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h, n, deadline, max_sweeps=200):
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try moving each zero to a one position
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                h_bin[i], h_bin[j] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                if c < best_c:
                    best_c = c
                    improved = True
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c
```

       = (1-h[i+shift]) - (1-h[j+shift])
       = h[j+shift] - h[i+shift]

Wait, but h[i+shift] and h[j+shift] might also have changed if i+shift = j or j+shift = i. In general, for different i and j, the change in g(k) is:
Δg(k) = (1-h_new[i+shift_k]) - (1-h_new[j+shift_k]) + effects from h changing at i and j being used as the (1-h[n]) part.

Let me be more careful. Let shift_k = k - (N-1).

g(k) = ∑_n h[n+shift_k] * (1-h[n]) * dx

After swap (h[i]: 0→1, h[j]: 1→0):
g_new(k) = ∑_n h_new[n+shift_k] * (1-h_new[n]) * dx

The change comes from:
1. Terms where n+shift_k = i (i.e., n = i-shift_k): h changes from 0 to 1
   Contribution: (1-h_new[i-shift_k]... wait, n = i-shift_k, and (1-h[n]) = (1-h[i-shift_k])
   If i-shift_k = j, then h[j] changed from 1 to 0, so (1-h[j]) changed from 0 to 1.
   Otherwise, (1-h[i-shift_k]) is unchanged.
   
   New term: 1 * (1-h_new[i-shift_k])
   Old term: 0 * (1-h[i-shift_k]) = 0
   Change: (1-h_new[i-shift_k])

2. Terms where n+shift_k = j (i.e., n = j-shift_k): h changes from 1 to 0
   New term: 0 * (1-h_new[j-shift_k]) = 0
   Old term: 1 * (1-h[j-shift_k])
   Change: -(1-h_old[j-shift_k])

   But wait, if j-shift_k = i, then h[i] changed from 0 to 1, so (1-h[i]) changed from 1 to 0.
   Old: 1*(1-h_old[i]) = 1*1 = 1
   New: 0*(1-h_new[i]) = 0
   Change: -1

   If j-shift_k ≠ i:
   Old: 1*(1-h[j-shift_k])
   Change: -(1-h[j-shift_k]) [since h[j-shift_k] didn't change]

3. Terms where n = i: (1-h[n]) changes from 1 to 0
   The term is h[i+shift_k] * (1-h[i])
   Old: h[i+shift_k] * 1
   New: h_new[i+shift_k] * 0 = 0
   Change: -h[i+shift_k]
   
   But if i+shift_k = j, then h[j] changed from 1 to 0.
   Old: h[j] * 1 = 1
   New: h_new[j] * 0 = 0
   Change: -1
   If i+shift_k ≠ j:
   Change: -h[i+shift_k]

4. Terms where n = j: (1-h[n]) changes from 0 to 1
   The term is h[j+shift_k] * (1-h[j])
   Old: h[j+shift_k] * 0 = 0
   New: h_new[j+shift_k] * 1
   Change: h_new[j+shift_k]
   
   If j+shift_k = i: h_new[i] = 1, change = 1
   If j+shift_k ≠ i: change = h[j+shift_k]

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def lse_grad_val(h, n, alpha):
    g = compute_all_g(h, n)
    amax = g.max()
    shifted = alpha * (g - amax)
    expired = shifted < -50
    shifted[expired] = -50
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    weights = exp_s / total
    lse_val = amax + np.log(total) / alpha
    thresh = 1e-7
    active = np.where(weights > thresh)[0]
    grad = np.zeros(n)
    dx = 2.0 / n
    for mi in active:
        w = weights[mi]
        shift = mi - (n - 1)
        j1 = max(0, shift)
        j2 = min(n, shift + n)
        if j2 > j1:
            grad[j1:j2] += w * (1.0 - h[j1 - shift:j2 - shift])
        j3 = max(0, -shift)
        j4 = min(n, n - shift)
        if j4 > j3:
            grad[j3:j4] -= w * h[j3 + shift:j4 + shift]
    return grad * dx, lse_val

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def proj_grad(h, n, deadline, alphas):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        lr = 0.5
        stall = 0
        prev_c = compute_c5(h, n)
        for it in range(30000):
            if time.time() > deadline:
                break
            grad, _ = lse_grad_val(h, n, alpha)
            ng = np.linalg.norm(grad)
            if ng < 1e-15:
                break
            ht = project_feasible(h - lr * grad, n)
            ct = compute_c5(ht, n)
            if ct < prev_c - 1e-14:
                h = ht
                prev_c = ct
                if ct < best_c:
                    best_c = ct
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.03, 3.0 / (ng + 1e-10))
            else:
                stall += 1
                lr *= 0.82
                if stall > 50:
                    break
        h = best_h.copy()
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        s = alpha * (g - m)
        s[s < -50] = -50
        sm = s.max()
        return m + (sm + np.log(np.exp(s - sm).sum())) / alpha

    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        s = alpha * (g - m)
        s[s < -50] = -50
        sm = s.max()
        exp_s = np.exp(s - sm)
        weights = exp_s / exp_s.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        active = np.where(weights > 1e-10)[0]
        for mi in active:
            w = weights[mi]
            shift = mi - (n - 1)
            j1 = max(0, shift)
            j2 = min(n, shift + n)
            if j2 > j1:
                grad[j1:j2] += w * (1.0 - x[j1 - shift:j2 - shift])
            j3 = max(0, -shift)
            j4 = min(n, n - shift)
            if j4 > j3:
                grad[j3:j4] -= w * x[j3 + shift:j4 + shift]
        return grad * dx

    def con(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_search(h, n, deadline, max_sweeps=300):
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                if abs(i - j) == 1:
                    continue
                h_bin[i], h_bin[j] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                if c < best_c:
                    best_c = c
                    improved = True
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            # Try double swaps
            found = False
            for ii in range(len(zeros)):
                if time.time() > deadline:
                    break
                for jj in range(ii + 1, len(zeros)):
                    if time.time() > deadline:
                        break
                    for kk in range(len(ones)):
                        if time.time() > deadline:
                            break
                        for ll in range(kk + 1, len(ones)):
                            i1, i2 = zeros[ii], zeros[jj]
                            j1, j2 = ones[kk], ones[ll]
                            h_bin[i1], h_bin[i2] = 1.0, 1.0
                            h_bin[j1], h_bin[j2] = 0.0, 0.0
                            c = compute_c5(h_bin, n)
                            if c < best_c:
                                best_c = c
                                found = True
                                ones = [x for x in ones if x not in (j1, j2)]
                                ones.extend([i1, i2])
                                zeros = [x for x in zeros if x not in (i1, i2)]
                                zeros.extend([j1, j2])
                            else:
                                h_bin[i1], h_bin[i2] = 0.0, 0.0
                                h_bin[j1], h_bin[j2] = 1.0, 1.0
                            if found:
                                break
                        if found:
                            break
                    if found:
                        break
                if found:
                    break
            if not found and not improved:
                break
    return h_bin, best_c
```
