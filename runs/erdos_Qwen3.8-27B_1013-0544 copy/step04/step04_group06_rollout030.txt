```python
def exact_subgrad_step(h, alpha, n_top=1):
    """One step of subgradient method targeting the top n_top constraints."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_top]
    g = np.zeros(n)
    nc = len(corr)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag == 0:
            g += (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += (1.0 - h[:-lag])
            g[:n-lag] -= h[lag:]
        else:
            m2 = -lag
            g[:n-m2] += (1.0 - h[m2:])
            g[m2:] -= h[:n-m2]
    # Normalize gradient
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h
    h_new = project_sum(h - alpha * g / gn, n / 2.0)
    return h_new
```

```python
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_obj_grad(h, beta, penalty=1000.0):
    """Softmax approximation of max correlation, with integral penalty."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    bc = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(bc)
    se = e.sum()
    nc = len(corr)
    val = (m + np.log(se / nc) / beta) * dx + penalty * (np.sum(h) - n/2)**2 / n
    
    w = e / se
    g = np.zeros(n)
    for j in range(nc):
        if w[j] < 1e-12:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += w[j] * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w[j] * (1.0 - h[:-lag])
            g[:n-lag] -= w[j] * h[lag:]
        else:
            m2 = -lag
            g[:n-m2] += w[j] * (1.0 - h[m2:])
            g[m2:] -= w[j] * h[:n-m2]
    g *= dx
    # Add penalty gradient
    g += 2 * penalty * (np.sum(h) - n/2) / n * np.ones(n)
    return val, g

def slbfgs_optimize(h_init, n, beta, maxiter=2000, penalty=1000.0):
    bounds = [(0, 1)] * n
    res = minimize(softmax_obj_grad, h_init, jac=True, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-16})
    return project_sum(res.x, n / 2.0)

def subgrad_polish(h, max_iter, t0, budget, n_top=3):

    best = h.copy()
    best_c5 = compute_c5(best)
    alpha = 0.05
    no_improve = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        top_idx = np.argsort(corr)[::-1][:n_top]
        g = np.zeros(n)
        nc = len(corr)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                g += (1.0 - 2.0 * best)
            elif lag > 0:
                g[lag:] += (1.0 - best[:-lag])
                g[:n-lag] -= best[lag:]
            else:
                m2 = -lag
                g[:n-m2] += (1.0 - best[m2:])
                g[m2:] -= best[:n-m2]
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        h_try = project_sum(best - alpha * g, n / 2.0)
        c5_try = compute_c5(h_try)
```

Actually, I just realized something important. Let me re-examine the subgradient step.

The subgradient of max_k c_k at a point where c_{k*} = max is ∇c_{k*}. The negative subgradient direction is -∇c_{k*}. Moving in this direction decreases c_{k*} (for small enough steps). But it might increase other c_k values.

The key question is: what's the optimal step size? Too large and we overshoot (other constraints become active and the max increases). Too small and we make very slow progress.

The backtracking approach: start with some alpha, if it doesn't improve, reduce alpha. This is reasonable but might be slow.

A better approach: **line search**. For a given direction d, find the alpha that minimizes C₅(h - alpha*d). Since C₅ is the max of quadratic functions, this is the max of quadratic functions in alpha, which is piecewise quadratic. We can find the minimum by:
1. Computing all c_k(alpha) = c_k(h - alpha*d) for alpha ∈ [0, alpha_max]
2. Each c_k(alpha) is a quadratic in alpha
3. The max is a piecewise quadratic function
4. Find where it's minimized

But this is expensive because we need to evaluate all c_k for many alpha values.

A simpler approach: **golden section search** on alpha. Each evaluation costs O(n²) for the correlation. With maybe 20-30 evaluations, this is feasible.

Let me implement this:

```python
def line_search_alpha(h, d, t0, budget, n_eval=20):
    """Find optimal step size along direction d using golden section."""
    n = len(h)
    # First find a reasonable upper bound
    alpha_max = 0.1
    c5_0 = compute_c5(h)
    # Make sure alpha_max gives a valid point
    h_check = project_sum(h - alpha_max * d, n/2.0)
    if compute_c5(h_check) > c5_0 * 1.5:
        alpha_max = 0.01
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a, b = 0.0, alpha_max
    c5_a = c5_0
    c5_b = compute_c5(project_sum(h - b * d, n/2.0))
    for _ in range(n_eval):
        if budget - (time.time() - t0) < 1:
            break
        m1 = a + gr * (b - a)
        m2 = b - gr * (b - a)
        c5_m1 = compute_c5(project_sum(h - m1 * d, n/2.0))
        c5_m2 = compute_c5(project_sum(h - m2 * d, n/2.0))
        if c5_m1 < c5_m2:
            b = m2
            c5_b = c5_m2
        else:
            a = m1
            c5_a = c5_m1
    return (a + b) / 2
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_corr_single(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:-lag])
        g[:n - lag] -= h[lag:]
    else:
        m2 = -lag
        g[:n - m2] += (1.0 - h[m2:])
        g[m2:] -= h[:n - m2]
    return g

def softmax_obj_grad(h, beta, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    bc = np.clip(beta * (corr - m), -600, 600)
    e = np.exp(bc)
    se = e.sum()
    nc = len(corr)
    val = (m + np.log(se / nc) / beta) * dx + penalty * (np.sum(h) - n * 0.5) ** 2
    w = e / se
    g = np.zeros(n)
    for j in range(nc):
        if w[j] < 1e-12:
            continue
        lag = j - (n - 1)
        g += w[j] * grad_corr_single(h, lag)
    g *= dx
    g += 2.0 * penalty * (np.sum(h) - n * 0.5)
    return val, g

def lbfgs_opt(h_init, beta, penalty, maxiter):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj_grad, h_init, args=(beta, penalty), jac=True,
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50})
    return project_sum(res.x, n / 2.0)

def slsqp_opt(h_init, beta, maxiter):
    n = len(h_init)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return softmax_obj_grad(x, beta, 1e8)[0]
    def grad(x):
        return softmax_obj_grad(x, beta, 1e8)[1]
    eq = {'type': 'eq', 'fun': lambda x: np.sum(x) - target,
          'jac': lambda x: np.ones(n)}
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=[eq], options={'maxiter': maxiter, 'ftol': 1e-18})
    return project_sum(res.x, target)

def subgrad_polish(h_init, max_iter, t0, budget, n_top=1, alpha_init=0.01):
    n = len(h_init)
    best = h_init.copy()
    best_c5 = compute_c5(best)
    alpha = alpha_init
    no_imp = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 1.5:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        top = np.argsort(corr)[::-1][:n_top]
        g = np.zeros(n)
        for idx in top:
            lag = int(idx) - (n - 1)
            g += grad_corr_single(best, lag)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        h_try = project_sum(best - alpha * g, n / 2.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-16:
            best = h_try
            best_c5 = c5_try
            alpha = min(alpha * 1.015, 0.15)
            no_imp = 0
        else:
            alpha *= 0.82
            no_imp += 1
            if no_imp > 150:
                p = np.random.randn(n) * 0.004
                hp = project_sum(best + p, n / 2.0)
                cp = compute_c5(hp)
                if cp < best_c5:
                    best = hp
                    best_c5 = cp
                no_imp = 0
            if no_imp > 3000:
                break
    return best, best_c5

def multi_step_subgrad(h_init, t0, budget, n_top=1, total_iters=5000):
    n = len(h_init)
    best = h_init.copy()
    best_c5 = compute_c5(best)
    alpha = 0.03
    for it in range(total_iters):
        if budget - (time.time() - t0) < 1.0:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        top = np.argsort(corr)[::-1][:n_top]
        g = np.zeros(n)
        for idx in top:
            lag = int(idx) - (n - 1)
            g += grad_corr_single(best, lag)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        improved = False
        for step in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1,
                     alpha * 0.04, alpha * 0.01, 0.001, 0.0003]:
            h_try = project_sum(best - step * g, n / 2.0)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-17:
                best = h_try
                best_c5 = c5_try
                alpha = min(alpha * 1.02, 0.1)
                improved = True
                break
        if not improved:
            alpha *= 0.7
    return best, best_c5
```
