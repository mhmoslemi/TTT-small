        h_new = slsqp(best_h, target_final, 500000, maxiter=6000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Final micro-polish with very fine steps
    if tl() > 5:
        for it in range(5000):
            if tl() < 2:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            top_idx = np.argsort(corr)[::-1][:5]
            g = np.zeros(n_final)
            for idx in top_idx:
                lag = int(idx) - (n_final - 1)
                if lag == 0:
                    g += (1.0 - 2.0 * best_h)
                elif lag > 0:
                    g[lag:] += (1.0 - best_h[:-lag])
                    g[:n - lag] -= best_h[lag:]
                else:
                    m2 = -lag
                    g[:n_final - m2] += (1.0 - best_h[m2:])
                    g[m2:] -= best_h[:n_final - m2]
            gn = np.linalg.norm(g)
            if gn < 1e-15:
                break
            g /= gn
            for step in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0004, 0.0001]:
                ht = project_sum(best_h - step * g, target_final)
                c5t = compute_c5(ht)
                if c5t < best_c5 - 1e-16:
                    best_h = ht
                    best_c5 = c5t
                    break

    return best_h, best_c5, len(best_h)
