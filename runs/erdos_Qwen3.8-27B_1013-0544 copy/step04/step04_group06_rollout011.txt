            g[m2:] -= h[:n - m2]

    n = len(h)
    gn = np.linalg.norm(g)

        return h, best_c5

        c5_new = compute_c5(h_new)

        step *= 0.5
    return h, best_c5

            g[m2:] -= w[j] * h[:n - m2]

        return softmax_obj(np.clip(x, 0, 1), beta)[0]

                h_new = slsqp_opt(h_interp, n_try / 2.0, beta, maxiter=1500)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
                    n, target = n_try, n_try / 2.0
                h_interp = h_new
            if rem() > 80:
                best_h, best_c5 = subgrad_optimize(best_h, target, t0, time.time() + min(rem() - 30, 120),
                                                   k_start=1, k_max=5, patience=300)

    # Phase 4: Multi-restart exploration
    n_best = len(best_h)
    target_b = n_best / 2.0
    for restart in range(5):
        if rem() < 80:
            break
        if restart == 0:
            h_try = best_h.copy()
        elif restart == 1:
            h_try = project_sum(best_h + np.random.randn(n_best) * 0.03, target_b)
        elif restart == 2:
            h_try = best_h.copy()
            bs = max(5, n_best // 6)
            start = np.random.randint(0, n_best - bs)
            h_try[start:start+bs] = 1.0 - h_try[start:start+bs]
            h_try = project_sum(h_try, target_b)
        elif restart == 3:
            h_try = project_sum(best_h + np.random.randn(n_best) * 0.08, target_b)
        else:
            # Shift-based perturbation
            shift = np.random.randint(3, max(4, n_best // 20))
            h_try = np.roll(best_h, shift)
            h_try = project_sum(h_try, target_b)
        # Optimize the perturbed point
        h_new = slsqp_opt(h_try, target_b, 20000, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        else:
            # Try subgradient from the perturbed point
            h_sg, c5_sg = subgrad_optimize(h_new, target_b, t0, time.time() + 20,
                                           k_start=1, k_max=4, patience=100)
            if c5_sg < best_c5:
                best_h, best_c5 = h_sg, c5_sg

    # Phase 5: Final intensive polish
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h, best_c5 = subgrad_optimize(best_h, target_final, t0, time.time() + max(0, rem() - 5),
                                       k_start=1, k_max=5, patience=800)

    # Final micro-steps
    for _ in range(3000):
        if rem() < 2:
            break
        g, _ = topk_subgrad(best_h, 1)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g_dir = -g / gn
        for step in [0.003, 0.001, 0.0004, 0.0001]:
            h_try = project_sum(best_h + step * g_dir, target_final)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-17:
                best_h, best_c5 = h_try, c5_try
                break

    return best_h, best_c5, len(best_h)
