def project_h(h, target_sum):

        idx = np.argsort(vals)[::-1][:top_k_active]
        vals_active = vals[idx]

        vals_active = vals

    exp_vals = np.exp(shifted)

import numpy as np
from scipy.optimize import minimize
import time

_lse_beta = [100.0]
_lse_top_k = [50]
_lse_mu = [1000.0]
_lse_target = [0.0]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_vals_grad(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    nk = min(top_k, len(vals))
    idx = np.argsort(vals)[::-1][:nk]
    vals_active = vals[idx]
    m = vals_active[0]
    shifted = beta * (vals_active - m)
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    val = m + np.log(sum_exp) / beta
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for j in range(nk):
        w = weights[j] * dx
        k = int(idx[j])
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    return val, grad

def _obj(x):
    v, g = lse_vals_grad(x, _lse_beta[0], _lse_top_k[0])
    pen = _lse_mu[0] * (x.sum() - _lse_target[0]) ** 2
    return v + pen

def _grad(x):
    v, g = lse_vals_grad(x, _lse_beta[0], _lse_top_k[0])
    pen_grad = 2.0 * _lse_mu[0] * (x.sum() - _lse_target[0])
    return g + pen_grad

def do_lbfgsb(h0, betas, top_ks, mus, maxiter, deadline):
    n = len(h0)
    target = n / 2.0
    _lse_target[0] = target
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for i in range(len(betas)):
        if time.time() > deadline - 15:
            break
        _lse_beta[0] = betas[i]
        _lse_top_k[0] = top_ks[i]
        _lse_mu[0] = mus[i]
        try:
            res = minimize(_obj, h_cur, jac=_grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h_cur, c5_cur
