def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_at_lag(h, lag):
    """Compute gradient of correlation at given lag."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if lag == 0:
        g = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        g[lag:] += (1.0 - h[:-lag]) * dx
        g[:n - lag] -= h[lag:] * dx
    else:
        m = -lag
        g[:n - m] += (1.0 - h[m:]) * dx
        g[m:] -= h[:n - m] * dx
    return g

def projected_gradient_step(h, grad, target, max_step=0.5, min_step=1e-6):
    """Adaptive step size projected gradient descent."""
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(h)
    step = max_step
    for _ in range(30):
        if step < min_step:
            break
        h_new = project_sum(h - step * grad, target)
        val = compute_c5(h_new)
        if val < best_val:
            best_h = h_new
            best_val = val
            step = min(step * 1.5, max_step)
        else:
            step *= 0.5
    return best_h, best_val

    best_c5 = compute_c5(best)
    no_improve = 0

                    no_improve = 0

def spectral_flat_restart(h, target, rng):
    """Create a new starting point by flattening the spectrum of h."""
    n = len(h)
    H = np.fft.rfft(h - 0.5)  # Centered spectrum
    # Flatten: reduce peak-to-average ratio
    mag = np.abs(H)
    phase = np.angle(H)
    target_mag = np.full_like(mag, mag.mean()) * 0.5 + mag * 0.5
    H_new = target_mag * np.exp(1j * phase)
    h_new = np.fft.irfft(H_new, n=n) + 0.5
    h_new = project_sum(h_new + rng.randn(n) * 0.01, target)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_lag(h, lag):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if lag == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        g[lag:] += (1.0 - h[:-lag]) * dx
        g[:n - lag] -= h[lag:] * dx
    else:
        m = -lag
        g[:n - m] += (1.0 - h[m:]) * dx
        g[m:] -= h[:n - m] * dx
    return g

def softmax_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    bc = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(bc)
    se = e.sum()
    val = (m + np.log(se / len(corr)) / beta) * dx
    w = e / se
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        if w[j] < 1e-11:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += w[j] * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w[j] * (1.0 - h[:-lag])
            g[:n - lag] -= w[j] * h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += w[j] * (1.0 - h[m2:])
            g[m2:] -= w[j] * h[:n - m2]
    return val, g * dx

def slsqp_solve(h0, n, beta, maxiter, rem):
    target = n / 2.0
    bounds = [(0, 1)] * n
    def f(x):
        return softmax_obj(np.clip(x, 0, 1), beta)[0]
    def fgrad(x):
        return softmax_obj(np.clip(x, 0, 1), beta)[1]
    def feq(x):
        return np.sum(x) - target
    def feqj(x):
        return np.ones(n)
    res = minimize(f, h0, jac=fgrad, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': feq, 'jac': feqj}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_sum(res.x, target)

def grad_polish(h, target, k, maxiter, rem):
    n = len(h)
    h = h.copy()

    mom = np.zeros(n)
    stuck = 0
    for _ in range(maxiter):
        if rem() < 5:
            break

        improved = False
        for s in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            ht = project_sum(h - s * mom, target)
            v = compute_c5(ht)

                bc = compute_c5(h)
                mom[:] = 0.0
                stuck = 0
        else:
            stuck = 0
    return h, bc
