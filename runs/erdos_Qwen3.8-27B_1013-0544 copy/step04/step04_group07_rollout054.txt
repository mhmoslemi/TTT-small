    n = len(h0)
    target = n / 2.0
    h = snap_binary(h0, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    ones_idx = np.where(h >= 0.5)[0].tolist()
    zeros_idx = np.where(h < 0.5)[0].tolist()
    n_ones = len(ones_idx)
    T = 0.04
    T_min = 3e-5
    cool = 0.99997
    stall = 0
    while time.time() < deadline:
        if stall > 10000:
            T = max(T, 0.025)
            stall = 0
        ip = rng.randint(n_ones)
        jp = rng.randint(n_ones)
        i = ones_idx[ip]
        j = zeros_idx[jp]
        h[i] = 0.0
        h[j] = 1.0
        ones_idx[ip] = j
        zeros_idx[jp] = i
        c = compute_c5(h)
        d = c - cur_c5
        if d < 0 or rng.random() < np.exp(-d / T):
            cur_c5 = c
            if c < best_c5:
                best_h = h.copy()
                best_c5 = c
                stall = 0
            else:
                stall += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            ones_idx[ip] = i
            zeros_idx[jp] = j
            stall += 1
        T = max(T * cool, T_min)
    return best_h, best_c5

def polish_binary(h0, deadline, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h)
    improved = True
    iters = 0
    while improved and time.time() < deadline and iters < 50:
        improved = False
        iters += 1
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for _ in range(3000):
            if time.time() > deadline:
                return h, best_c5
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c = compute_c5(h)
            if c < best_c5 - 1e-15:
                best_c5 = c
                improved = True
                ones = np.where(h >= 0.5)[0]
                zeros = np.where(h < 0.5)[0]
            else:
                h[i], h[j] = 1.0, 0.0
    return h, best_c5

def continuous_refine(h0, deadline):
    n = len(h0)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    def smax(x, beta):
        m = np.max(x)
        return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500))) / len(x)) / beta
    def obj(h_in):
        corr = np.correlate(h_in, 1.0 - h_in, mode="full")
        return smax(corr, 800) * (2.0 / n)
    def con_fun(h_in):
        return np.sum(h_in) - target
    def con_jac(x):
        return np.ones(x.shape)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 5000, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def resample_to(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    np.random.seed(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(384, 0.5)
    n_parent = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    # Try resolutions: prioritize parent and nearby
    resolutions = sorted(set([256, 288, 320, n_parent, 448]))
    # Phase 1: Quick binary search at each resolution
    for n_pts in resolutions:
        if time.time() > deadline - 30:
            break
        target = n_pts / 2.0
        cur = resample_to(h_parent, n_pts)
        cur = project_h(cur, target)
        hb = snap_binary(cur, target)
        cb = compute_c5(hb)
        if cb < best_c5:
            best_h, best_c5 = hb.copy(), cb
    # Phase 2: Annealing - allocate time proportionally
    # Give more time to parent resolution and smaller ones
    weights = {}
    for r in resolutions:
        if r == n_parent:
            weights[r] = 3.0
        elif r <= 320:
            weights[r] = 2.0
        else:
            weights[r] = 1.0
    total_w = sum(weights.values())
    time_per_res = {}
    for r in resolutions:
        time_per_res[r] = (budget_s - 60) * weights[r] / total_w
    elapsed = time.time() - t0
    for idx, n_pts in enumerate(resolutions):
        if time.time() > deadline - 20:
            break
        alloc = time_per_res[n_pts] * 0.7
        res_deadline = min(time.time() + alloc, deadline - 20)
        target = n_pts / 2.0
        cur = resample_to(h_parent, n_pts)
        cur = project_h(cur, target)
        hb = snap_binary(cur, target)
        cb = compute_c5(hb)
        # Anneal
        if time.time() < res_deadline - 5:
            hb, cb = anneal(hb, res_deadline - 5, seed=seed + idx * 211)
            if cb < best_c5:
                best_h, best_c5 = hb.copy(), cb
        # Polish
        if time.time() < res_deadline:
            hb, cb = polish_binary(hb, res_deadline, seed=seed + 500 + idx)
            if cb < best_c5:
                best_h, best_c5 = hb.copy(), cb
    # Phase 3: Continued annealing at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 30:
        rem = deadline - 30 - time.time()
        hb, cb = anneal(best_h, deadline - 30, seed=seed + 9999)
        if cb < best_c5:
            best_h, best_c5 = hb.copy(), cb
        if time.time() < deadline - 25:
            hb, cb = polish_binary(hb, deadline - 25, seed=seed + 7777)
            if cb < best_c5:
                best_h, best_c5 = hb.copy(), cb
    # Phase 4: Continuous refinement
    if time.time() < deadline - 5:
        refined = continuous_refine(best_h, deadline - 3)
        c_ref = compute_c5(refined)
        if c_ref < best_c5:
            best_h, best_c5 = refined, c_ref
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
