def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

                best_h = h.copy()
        else:
            stagnant += 1
            velocity *= 0.4
            lr *= 0.87
        if stagnant > 400 and lr < 2e-6:
            break
    return best_h, best_c5

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def coarse_to_fine(h_parent, n_target, deadline, seed):
    np.random.seed(seed)
    n_coarse = max(25, n_target // 8)
    h_c = resample(h_parent, n_coarse)
    h_c = project_feasible(h_c, n_coarse / 2.0)
    h_c, c_c = slsqp_opt(h_c, n_coarse, deadline, seed, [50, 200, 1000, 5000])
    h_c, c_c = subgrad_polish(h_c, n_coarse, deadline, seed + 1)
    h_mid = resample(h_c, n_target // 2)
    h_mid = project_feasible(h_mid, n_target / 4.0)
    h_mid, c_mid = slsqp_opt(h_mid, len(h_mid), deadline, seed + 10, [100, 500, 2000])
    h_out = resample(h_mid, n_target)
    h_out = project_feasible(h_out, n_target / 2.0)
    return h_out, compute_c5(h_out)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    if parent is None:
        parent = np.full(200, 0.5)
    pn = len(parent)

    d1 = t0 + budget_s * 0.18
    if time.time() < d1:
        h_cf, c_cf = coarse_to_fine(parent.copy(), pn, d1, seed)
        if c_cf < best_c5:
            best_h, best_c5, best_n = h_cf.copy(), c_cf, pn

    d2 = t0 + budget_s * 0.45
    h2 = parent.copy()
    h2, c2 = slsqp_opt(h2, pn, d2, seed + 10, [80, 300, 1500, 8000, 40000])
    h2, c2 = lbfgsb_opt(h2, pn, d2, seed + 20)
    h2, c2 = subgrad_polish(h2, pn, d2, seed + 30)
    if c2 < best_c5:
        best_h, best_c5, best_n = h2.copy(), c2, pn

    d3 = t0 + budget_s * 0.62
    if time.time() < d3:
        variants = [parent[::-1].copy()]
        for s in [seed + 100, seed + 200, seed + 300]:
            np.random.seed(s)
            variants.append(np.clip(parent + np.random.randn(pn) * 0.04, 0, 1))
        for vi, v in enumerate(variants):
            if time.time() > d3:
                break
            v = project_feasible(v, pn / 2.0)
            v, cv = slsqp_opt(v, pn, d3, seed + 50 + vi * 10, [200, 1500, 8000, 40000])
            v, cv = subgrad_polish(v, pn, d3, seed + 90 + vi * 10)
            if cv < best_c5:
                best_h, best_c5, best_n = v.copy(), cv, pn

    d4 = t0 + budget_s * 0.78
    res_list = sorted(set([pn - 40, pn - 20, pn, pn + 20, pn + 40, pn + 80]))
    for n_try in res_list:
        if time.time() > d4:
            break
        if n_try < 60 or n_try > 800:
            continue
        h0 = resample(parent, n_try)
        h0 = project_feasible(h0, n_try / 2.0)
        h0, c0 = slsqp_opt(h0, n_try, d4, seed + n_try, [100, 1000, 10000, 50000])
        h0, c0 = subgrad_polish(h0, n_try, d4, seed + n_try + 1)
        if c0 < best_c5:
            best_h, best_c5, best_n = h0.copy(), c0, n_try

    d5 = t0 + budget_s * 0.95
    if best_h is not None:
        n = best_n
        best_h, best_c5 = slsqp_opt(best_h, n, d5, seed + 500, [50000, 200000, 800000])
        best_h, best_c5 = lbfgsb_opt(best_h, n, d5, seed + 600)
        best_h, best_c5 = subgrad_polish(best_h, n, d5, seed + 700)

    if best_h is None:
        best_h = parent.copy()
        best_n = pn
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
