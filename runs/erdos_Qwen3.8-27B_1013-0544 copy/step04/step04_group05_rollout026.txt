```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def grad_c5_approx(h, beta, top_m=50):
    """Approximate gradient of smooth max C5"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    top_idx = np.argsort(c)[::-1][:top_m]
    w = np.exp(beta * (c[top_idx] - m))
    w = w / np.sum(w)
    
    g = np.zeros(n)
    for i, idx in enumerate(top_idx):
        wi = w[i]
        if wi < 1e-14:
            continue
        lag = int(idx) - (n - 1)
        if lag == 0:
            g += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < n:
                g[lag:] += wi * (1.0 - h[:-lag])
            g[:n-lag] -= wi * h[lag:]
        else:
            mk = -lag
            g[:n-mk] += wi * (1.0 - h[mk:])
            g[mk:] -= wi * h[:n-mk]
    return g * dx

def lbfgsb_optimize(h, target, beta, maxiter, time_fn, top_m=50):
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        h_full = np.empty(n)
        h_full[:n-1] = x
        h_full[n-1] = target - x.sum()
        h_full = np.clip(h_full, 0, 1)
        c = np.correlate(h_full, 1.0 - h_full, mode="full")
        return smooth_max(c, beta) * dx
    
    def grad(x):
        h_full = np.empty(n)
        h_full[:n-1] = x
        h_full[n-1] = target - x.sum()
        h_full = np.clip(h_full, 0, 1)
        g = grad_c5_approx(h_full, beta, top_m)
        return g[:n-1] - g[n-1]  # chain rule
    
    x0 = h[:n-1].copy()
    bounds = [(0.0, 1.0)] * (n-1)
    
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-15})
        h_new = np.empty(n)
        h_new[:n-1] = res.x
        h_new[n-1] = target - res.x.sum()
        h_new = project_h(h_new, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def subgrad_refine(h, target, time_fn, max_iters, top_m, step_init, momentum):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    x = best.copy()
    v = np.zeros(n)
    step = step_init
    no_improve = 0
    
    for it in range(max_iters):
        if time_fn() < 2:
            break
        g = grad_c5_approx(x, 500, top_m)  # moderate beta for gradient
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        v = momentum * v + step * g / gn
        x = project_h(x - v, target)
        
        c5 = compute_c5(x)
        if c5 < best_c5 - 1e-13:
            best = x
            best_c5 = c5
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                step *= 0.9
        
        if no_improve > 2000:
            break
    
    return best, best_c5
