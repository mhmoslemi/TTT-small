    """Optimize h using Fourier basis expansion."""
    x_eval = np.linspace(0, 2, n_eval)
    
    # Build Fourier matrix
    X = np.zeros((n_eval, 2*n_basis + 1))
    X[:, 0] = 0.5  # constant term
    for k in range(1, n_basis + 1):
        X[:, 2*k-1] = np.cos(np.pi * k * x_eval)
        X[:, 2*k] = np.sin(np.pi * k * x_eval)
    
    # Initial coefficients from h_init
    coefs, _, _, _ = np.linalg.lstsq(X, h_init, rcond=None)
    
    def obj_params(params):
        h = X @ params
        # Penalty for box constraint violation
        viol = np.maximum(0, -h) ** 2 + np.maximum(0, h - 1) ** 2
        penalty = viol.sum()
        
        # Overlap computation

        g = np.correlate(h, 1 - h, mode='full') * dx
        m = g.max()
        alpha = 5000
        sh = alpha * (g - m)
        sm = sh.max()
        smooth = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        
        return smooth + 100 * penalty
    
    from scipy.optimize import minimize

                      options={'maxiter': 1000, 'ftol': 1e-15})
    
    h_opt = X @ result.x
    return project_feasible(h_opt, n_eval)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def fourier_optimize(n_eval, K, h_init, deadline, alpha=10000):
    """Optimize h using Fourier basis expansion."""
    x = np.linspace(0, 2, n_eval, endpoint=False)
    
    # Build Fourier design matrix
    cols = [np.ones(n_eval) * 0.5]
    for k in range(1, K + 1):
        cols.append(np.cos(np.pi * k * x))
        cols.append(np.sin(np.pi * k * x))
    X = np.column_stack(cols)
    n_params = X.shape[1]
    
    # Initial coefficients
    coefs, _, _, _ = np.linalg.lstsq(X, h_init, rcond=None)
    
    def obj_params(params):
        h = X @ params
        # Box constraint penalty
        penalty = (np.maximum(0, -h) ** 2 + np.maximum(0, h - 1) ** 2).sum()
        # Overlap

        sh = alpha * (g - m)
        sm = sh.max()
        lse = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        return lse + 1000.0 * penalty
    
    t_start = time.time()
    maxiter = max(50, int(30 / (time.time() - t_start + 1e-3)))  # Adaptive
    result = minimize(obj_params, coefs, method='L-BFGS-B',
                      options={'maxiter': 500, 'ftol': 1e-18})
    
    h_opt = X @ result.x
    return project_feasible(h_opt, n_eval)

def slsqp_refine(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n

        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

        m = g.max()

            h[i] -= eps

One more thought: instead of the expensive subgradient method (which computes the gradient by finite differences, O(n³)), I can compute the gradient analytically!

The gradient of g_k(h) = ∑_i h[i] * (1-h[i+k]) * dx with respect to h[j] is:
- For each i: ∂/∂h[j] [h[i] * (1-h[i+k])] = δ(i,j) * (1-h[i+k]) + h[i] * (-δ(i+k,j))
  = δ(i,j) * (1-h[i+k]) - h[i] * δ(i+k, j)

So ∂g_k/∂h[j] = (1-h[j+k]) - h[j-k] (with appropriate boundary handling)

The subgradient of max_k g_k is the gradient of g_{k*} where k* = argmax_k g_k.

So I can compute the subgradient in O(n) instead of O(n³)!
