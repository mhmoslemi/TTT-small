```python
from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def fast_de(h0, n, target, t0, time_limit, pop_size=50, F=0.8, CR=0.9, max_gen=2000):
    """Custom fast differential evolution"""
    rng = np.random.RandomState(12345)
    # Initialize population
    pop = np.zeros((pop_size, n))
    pop[0] = h0
    for i in range(1, pop_size):
        # Small perturbation around h0
        pop[i] = project_h(h0 + 0.1 * rng.randn(n), target)
    
    # Evaluate initial population
    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_c5 = scores[best_idx]
    
    for gen in range(max_gen):
        if time.time() - t0 > time_limit:
            break
        
        for i in range(pop_size):
            if time.time() - t0 > time_limit:
                break
            # DE/rand/1
            a, b, c = rng.choice(pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = project_h(mutant, target)
            
            # Crossover
            cross = rng.rand(n) < CR
            trial = np.where(cross, mutant, pop[i])
            trial = project_h(trial, target)
            
            score = compute_c5(trial)
            
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_c5:
                    best_c5 = score
                    best_h = trial.copy()
                    best_idx = i
    
    return best_h, best_c5

def subgrad_project(h, n, target, active_lags=None, num_lags=30):
    """Compute subgradient using active lags"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    if active_lags is None:
        thr = mc - 0.005 * (mc + 1e-10)
        idx = np.where(corr >= thr)[0]
    else:
        idx = active_lags
    if len(idx) > num_lags:
        order = np.argsort(corr[idx])[::-1]
        idx = idx[order[:num_lags]]
    g = np.zeros(n)
    for i in idx:
        lag = int(i) - (n - 1)
        if lag >= 0:
            if lag < n:
                g[:n - lag] += (1.0 - h[lag:])
                g[lag:] -= h[:n - lag]
        else:
            la = -lag
            if la < n:
                g[la:] += (1.0 - h[:n - la])
                g[:n - la] -= h[la:]
    return g / max(len(idx), 1), corr

def nesterov_subgrad(h0, n, target, n_iter, step0, t0, time_limit):
    h = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    step = step0
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        g, corr = subgrad_project(y, n, target)
        cnorm = np.linalg.norm(g)
        if cnorm < 1e-12:
            break
        g /= cnorm
        h_new = project_h(y - step * g, target)
        mu = (1.0 - np.sqrt(1.0 + 4.0 / ((it + 1) ** 2))) / 2.0
        y = np.clip(h_new + mu * (h_new - h_prev), 0, 1)
        y = project_h(y, target)
        h_prev, h = h, h_new
        c = compute_c5(h)
        if c < best_c5 - 1e-15:
            best_h, best_c5, stall = h.copy(), c, 0
        else:
            stall += 1
            if stall % 200 == 0:
                step *= 0.85
    return best_h, best_c5

def coord_descent(cur, target, n, sweeps, t0, time_limit):
    best_c5 = compute_c5(cur)
    deltas = (0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002)
    for sw in range(sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > time_limit - 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

def pair_descent(cur, target, n, sweeps, t0, time_limit, seed=42):
    best_c5 = compute_c5(cur)
    rng = np.random.RandomState(seed)
    for sw in range(sweeps):

        for _ in range(n):

                trial[k] -= d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 3:
            break
    return cur, best_c5

def snap_binary(h, target):
    z = (h > 0.5).astype(float)
    diff = target - z.sum()
    if diff > 1e-10:
        zeros = np.where(z == 0)[0]
        idx = zeros[np.argsort(h[zeros])[::-1]][:int(diff)]
        z[idx] = 1.0
    elif diff < -1e-10:
        ones = np.where(z == 1)[0]
        idx = ones[np.argsort(h[ones])][:int(-diff)]
        z[idx] = 0.0
    return project_h(z, target)

    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)
```

```python
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) < 1e-10:

    free = (h > 0) & (h < 1)
    if free.sum() == 0:
        return h  # Can't project, return as-is
    h[free] += diff / free.sum()
    h = np.clip(h, 0, 1)
    # One more iteration if needed
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        free = (h > 0) & (h < 1)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0, 1)
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def proj(h, t):
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) < 1e-14:
        return h
    for _ in range(50):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 0) & (h < 1.0)
        nf = free.sum()
        if nf == 0:
            h += d / len(h)
            h = np.clip(h, 0, 1)
            continue
        h[free] += d / nf
        h = np.clip(h, 0, 1)
    return h

def active_grad(h, n, nlags=50):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    thr = mc - 0.01 * (mc + 1e-10)
    idx = np.where(corr >= thr)[0]
    if len(idx) > nlags:
        idx = idx[np.argsort(corr[idx])[::-1][:nlags]]
    g = np.zeros(n)
    for i in idx:
        lag = int(i) - (n - 1)
        if lag >= 0 and lag < n:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
        elif lag < 0:
            la = -lag
            if la < n:
                g[la:] += (1.0 - h[:n - la])
                g[:n - la] -= h[la:]
    return g / max(len(idx), 1), mc

def nesterov_run(h0, n, target, n_it, step0, t0, t_end):
    h = h0.copy()
    hp = h0.copy()
    y = h0.copy()
    best = h.copy()
    best_c = compute_c5(h)
    step = step0
    stall = 0
    for it in range(n_it):
        if time.time() > t_end:
            break
        g, _ = active_grad(y, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        h_new = proj(y - step * g, target)
        mu = (1.0 - np.sqrt(1.0 + 4.0 / ((it+1)**2))) / 2.0
        y = np.clip(h_new + mu * (h_new - hp), 0, 1)
        y = proj(y, target)
        hp, h = h, h_new
        c = compute_c5(h)
        if c < best_c - 1e-16:
            best, best_c, stall = h.copy(), c, 0
        else:
            stall += 1
            if stall > 300:
                step *= 0.8
                stall = 0
    return best, best_c

def coord_dd(cur, target, n, sweeps, t0, t_end):
    bc = compute_c5(cur)
    deltas = (0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005)
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        for j in range(n):
            if time.time() > t_end - 1:
                break
            for d in deltas:
                tr = cur.copy()
                tr[j] += d
                tr = proj(tr, target)
                c = compute_c5(tr)
                if c < bc - 1e-16:
                    cur, bc, imp = tr, c, True
                    break
        if not imp and sw > 5:
            break
    return cur, bc

def pair_dd(cur, target, n, sweeps, t0, t_end, seed):
    bc = compute_c5(cur)
```

from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def proj(h, t):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += d / len(h)
            h = np.clip(h, 0, 1)
            continue
        h[free] += d / free.sum()
        h = np.clip(h, 0, 1)
    return h

def active_grad(h, n, nlags=40):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    thr = mc - 0.008 * (mc + 1e-10)
    idx = np.where(corr >= thr)[0]
    if len(idx) > nlags:
        idx = idx[np.argsort(corr[idx])[::-1][:nlags]]

        elif -n < lag < 0:

            g[la:] += (1.0 - h[:n-la])
            g[:n-la] -= h[la:]
