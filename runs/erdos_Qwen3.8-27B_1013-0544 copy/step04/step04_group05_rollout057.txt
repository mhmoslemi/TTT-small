def successive_lp(h0, target, active_tol, max_iter, time_fn):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time_fn() < 10:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        cmax = np.max(corr)
        
        # Active set: lags within tol of max
        active = corr >= cmax - active_tol
        active_idx = np.where(active)[0]
        
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        for idx in active_idx:
            k = idx - (n - 1)
            # Compute gradient of corr_k at h
            grad = np.zeros(n)
            if k == 0:
                grad = 1.0 - 2.0 * h
            elif k > 0:
                grad[k:] += (1.0 - h[:n-k])
                grad[:n-k] -= h[k:]
            else:
                m = -k
                grad[:n-m] += (1.0 - h[m:])
                grad[m:] -= h[:n-m]
            
            # Linearize: corr_k(h_new) ≈ corr_k(h) + grad @ (h_new - h)
            # Constraint: corr_k(h) + grad @ (h_new - h) <= t
            base = corr[idx]  # this is already * dx
            # grad @ (h_new - h) = grad @ h_new - grad @ h
            linear_part = grad @ h  # scalar
            # corr_k(h) + grad @ h_new - grad @ h <= t
            # corr_k(h) - grad @ h + grad @ h_new <= t
            rhs = base - linear_part
            constraints.append(grad @ h_var + rhs <= t_var)
        
        objective = cp.Minimize(t_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS)  # or another solver
        except:
            try:
                prob.solve(solver=cp.CBC)
            except:
                break
        
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        
        h_new = h_var.value
        h_new = project_h(h_new, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:

            else:

import numpy as np
from scipy.optimize import minimize, Bounds
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_optimize(h, target, betas, max_iter, time_fn):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    for beta in betas:
        if time_fn() < 10:
            break
        def obj(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, beta) * dx
        def con(h_in):
            return h_in.sum() - target
        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con, 'jac': lambda x: np.ones(n)}],
                          options={'maxiter': max_iter, 'ftol': 1e-18})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except:
            pass
    return best, best_c5

def successive_lp(h0, target, time_fn, tol_init=0.001, max_iter=500):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = compute_c5(h)
    tol = tol_init
    
    for it in range(max_iter):
        if time_fn() < 15:
            break
        
        corr_raw = np.correlate(h, 1.0 - h, mode="full")
        corr = corr_raw * dx
        cmax = np.max(corr)
        
        # Active set
        active = corr >= cmax - tol
        active_idx = np.where(active)[0]
        
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(corr)])
        
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        for idx in active_idx:
            k = idx - (n - 1)
            grad = np.zeros(n)
            if k == 0:
                grad = 1.0 - 2.0 * h
            elif k > 0:
                grad[k:] += (1.0 - h[:n-k])
                grad[:n-k] -= h[k:]
            else:
                m = -k
                grad[:n-m] += (1.0 - h[m:])
                grad[m:] -= h[:n-m]
            
            base = corr[idx]
            linear_const = base - grad @ h
            constraints.append(grad @ h_var + linear_const <= t_var)
        
        # Trust region (optional, for stability)
        # constraints.append(cp.norm(h_var - h, np.inf) <= 0.1)
        
        objective = cp.Minimize(t_var)
        prob = cp.Problem(objective, constraints)
        
        h_new = None
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate'] and h_var.value is not None:
                    h_new = h_var.value
                    break
            except:
                continue
        
        if h_new is None:
            # Try cvxpy default
            try:
                prob.solve()
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    h_new = h_var.value
            except:
                pass
        
        if h_new is None:
            tol *= 2
            if tol > 0.1:
                break
            continue
        
        h_new = project_h(h_new, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-12:
            h = h_new
            best_c5 = c5_new
            # Tighten tolerance as we improve
            tol = max(tol * 0.95, 0.0001)
        else:
            # Try with slightly larger active set
            tol *= 1.5
            if tol > 0.05:
                break
    
    return h, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_constraint(h_in, target):
    return h_in.sum() - target

def slsqp_jac(h_in):
    return np.ones(len(h_in))

def slsqp_optimize(h, target, betas, max_iter, time_fn):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    for beta in betas:
        if time_fn() < 10:
            break
        def obj_func(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, beta) * dx
        try:
            res = minimize(obj_func, best, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': slsqp_constraint, 'jac': slsqp_jac, 'args': (target,)}],
                          options={'maxiter': max_iter, 'ftol': 1e-18})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def compute_lag_grad(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad = 1.0 - 2.0 * h
    elif k > 0:
        grad[:n - k] += (1.0 - h[k:])
        grad[k:] -= h[:n - k]
    else:
        m = -k
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    return grad

def successive_lp(h0, target, time_fn, tol_init=0.002, max_iter=400, trust_init=0.15):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h)
    tol = tol_init
    trust = trust_init
    stall = 0
    for it in range(max_iter):
        if time_fn() < 15:
            break
        corr_raw = np.correlate(h, 1.0 - h, mode="full")
        corr = corr_raw * (2.0 / n)
        cmax = np.max(corr)
        active = corr >= cmax - tol
        active_idx = np.where(active)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(corr)])
        if len(active_idx) > 50:
            top = np.argsort(corr)[::-1][:50]
            active_idx = top
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        constraints.append(h_var - h <= trust)
        constraints.append(h - h_var <= trust)
        for idx in active_idx:
            k = idx - (n - 1)
            grad = compute_lag_grad(h, k)
            base = corr[idx]
            lc = base - (grad @ h)
            constraints.append(grad @ h_var + lc <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        h_new = None
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if prob.status in ('optimal', 'optimal_inaccurate') and h_var.value is not None:
                    h_new = h_var.value
                    break
            except Exception:
                continue
        if h_new is None:
            try:
                prob.solve()
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    h_new = h_var.value
            except Exception:
                pass
        if h_new is None:
            trust = max(trust * 0.5, 0.01)
            tol *= 1.5
            stall += 1
            if stall > 20 or tol > 0.1:
                break
            continue
        h_new = project_h(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            h = h_new
            best_c5 = c5_new
            tol = max(tol * 0.9, 0.0001)
            trust = min(trust * 1.2, 0.3)
            stall = 0
        else:
            stall += 1
            trust = max(trust * 0.7, 0.01)
            tol *= 1.3
            if stall > 30 or tol > 0.1:
                break
    return h, best_c5

def subgrad_refine(h, target, time_fn, max_iters=10000, m_max=15):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    momentum = 0.85
    no_improve = 0
    for it in range(max_iters):
        if time_fn() < 3:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        m_act = min(m_max, max(3, n // 30))
        top = np.argsort(corr)[::-1][:m_act]
        lags = [int(i) - (n - 1) for i in top]
        g = np.zeros(n)
        for lag in lags:
            g += compute_lag_grad(best, lag)
        g /= len(lags)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn
        found = False
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            ht = project_h(best + step * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                no_improve = 0
                found = True
                break
        if not found:
            no_improve += 1
            if no_improve >= 500 and it % 200 == 0:
                p = np.random.randn(n) * 0.005
                ht = project_h(best + p, target)
                if compute_c5(ht) < best_c5:
                    best = ht
                    best_c5 = compute_c5(best)
                    vel *= 0.5
                    no_improve = 0
            if no_improve > 3000:
                break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(400, 0.5)

    n0 = len(best_h)
    target0 = n0 / 2.0
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at base resolution
    best_h, best_c5 = slsqp_optimize(best_h, target0, [80, 300, 1000, 4000, 10000], 2500, rem)

    # Phase 2: Subgradient refinement
    best_h, best_c5 = subgrad_refine(best_h, target0, rem, max_iters=8000, m_max=12)

    # Phase 3: Successive LP at base resolution
    best_h, best_c5 = successive_lp(best_h, target0, rem, tol_init=0.002, max_iter=200, trust_init=0.12)

    # Phase 4: Upscale to 2x
    if rem() > 250:
        n2 = n0 * 2
        target2 = n2 / 2.0
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n2)
        h2 = project_h(np.interp(x_new, x_old, best_h), target2)
        c5_2 = compute_c5(h2)
        if rem() > 150:
            h2, c5_2 = slsqp_optimize(h2, target2, [200, 800, 3000, 8000], 1500, rem)
        if rem() > 80:
            h2, c5_2 = subgrad_refine(h2, target2, rem, max_iters=5000, m_max=15)
        if rem() > 40:
            h2, c5_2 = successive_lp(h2, target2, rem, tol_init=0.002, max_iter=250, trust_init=0.1)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2

    # Phase 5: Final successive LP polish
    nf = len(best_h)
    tf = nf / 2.0
    best_h, best_c5 = successive_lp(best_h, tf, rem, tol_init=0.001, max_iter=200, trust_init=0.08)

    # Phase 6: Tiny subgradient touch-up
    best_h, best_c5 = subgrad_refine(best_h, tf, rem, max_iters=3000, m_max=20)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
