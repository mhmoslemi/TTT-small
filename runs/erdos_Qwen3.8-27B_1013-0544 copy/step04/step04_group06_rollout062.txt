    if top_k < nc:
        idx = np.argsort(w)[::-1][:top_k]
    else:
        idx = np.where(w > 1e-12)[0]
    
    for j in idx:
        wj = w[j]
        if wj < 1e-12:
            continue
        lag = j - off
        if lag == 0:
            g += wj * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wj * (1.0 - h[:-lag])
            g[:n - lag] -= wj * h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += wj * (1.0 - h[m2:])
            g[m2:] -= wj * h[:n - m2]
    
    return val, g * dx

def slsqp(h_init, n, beta, maxiter=2000):
    target = n / 2.0
    bounds = [(0, 1)] * n
    res = minimize(
        lambda x: obj_grad(np.clip(x, 0, 1), beta)[0],
        h_init,
        jac=lambda x: obj_grad(np.clip(x, 0, 1), beta)[1],
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                      'jac': lambda x: np.ones(n)}],
        options={'maxiter': maxiter, 'ftol': 1e-16}
    )
    return project_sum(res.x, target)
