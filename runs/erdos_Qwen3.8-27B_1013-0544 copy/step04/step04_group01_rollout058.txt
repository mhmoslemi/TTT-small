def project_feasible_fast(h, n):

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Binary search for lambda
    lo, hi = -2.0, 2.0

        else:

    # (A ⋆ B)[k] = sum_i A[i] * B[i+k] for valid i
    # I want: sum_i A[i] * B[j-i+(n-1)] = sum_i A[i] * B[n-1+j-i]

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def hse_gradient_vec(h, n, alpha):
    dx = 2.0 / n

    c_max = c.max()
