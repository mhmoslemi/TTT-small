def compute_gradient_fast(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    shifted = np.clip(beta * (corr - corr.max()), -700.0, 700.0)
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    # For each lag, update the gradient
    for k in range(1, n):  # k is the magnitude of the lag
        # Positive lag k: weight w[n-1+k]
        wk = w[n-1+k]
        if wk > 1e-10:
            idx = np.arange(n-k)
            grad[idx] += wk * (1.0 - h[idx+k])
            grad[idx+k] -= wk * h[idx]
        
        # Negative lag k: weight w[n-1-k]
        wk = w[n-1-k]
        if wk > 1e-10:
            idx = np.arange(k, n)
            grad[idx] += wk * (1.0 - h[idx-k])
            grad[idx-k] -= wk * h[idx]
    
    return dx * grad

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_grad(h, beta, mu=0.0):
    """Objective and gradient for L-BFGS-B."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    sm_val = m + np.log(np.sum(np.exp(shifted))) / beta
    obj = sm_val * dx
    
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(1, n):
        wk = w[n-1+k]
        if wk > 1e-10:
            idx = np.arange(n-k)
            grad[idx] += wk * (1.0 - h[idx+k])
            grad[idx+k] -= wk * h[idx]
        wk = w[n-1-k]
        if wk > 1e-10:
            idx = np.arange(k, n)
            grad[idx] += wk * (1.0 - h[idx-k])
            grad[idx-k] -= wk * h[idx]
    grad *= dx
    
    if mu > 0:
        s = h.sum() - n/2.0
        obj += 0.5 * mu * s * s
        grad += mu * s
    
    return obj, grad

def lbfgs(h0, n, beta, mu, deadline, maxiter=300):
    if time.time() > deadline:
        return h0
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        return obj_grad(x, beta, mu)[0]
    
    def jac(x):
        return obj_grad(x, beta, mu)[1]
    
    try:
        res = minimize(fun, h0, jac=jac, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n/2.0)
    except Exception:
        return h0

def proj_grad(h0, n, beta, deadline, maxiter=10000):
    h = h0.copy()
    ts = n / 2.0
    c5 = compute_c5(h)
    vel = np.zeros(n)
    lr = 0.005
    momentum = 0.85
    stall = 0
    
    for it in range(maxiter):
        if time.time() > deadline:
            break
        _, grad = obj_grad(h, beta, 0.0)
        vel = momentum * vel - lr * grad
        h_new = project_feasible(h + vel, ts)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-12:
            h = h_new
            c5 = c_new
            stall = 0
            lr = min(lr * 1.01, 0.05)
        else:
            stall += 1
            vel *= 0.5
            lr *= 0.92
        if stall > 300 and lr < 1e-6:
            break
    return h, c5

def slsqp_refine(h0, n, deadline, beta):
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def cf(x):
        return x.sum() - ts
    def cj(x):
        return np.ones_like(x)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    
    def fun(x):
        return obj_grad(x, beta, 0.0)[0]
    def jac(x):
        return obj_grad(x, beta, 0.0)[1]
    
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 2000, 'ftol': 1e-14})
        return project_feasible(res.x, ts)
    except Exception:
        return h0
