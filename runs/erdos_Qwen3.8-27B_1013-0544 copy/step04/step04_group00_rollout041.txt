    return np.correlate(h, 1-h, mode='full') * dx

def compute_grad_smooth(h, n, alpha):
    c = compute_c_vec(h, n)
    dx = 2.0 / n
    # Softmax weights
    a = alpha * c
    a -= a.max()
    w = np.exp(a)
    w /= w.sum()
    # Gradient: dw_p/dh[j] * c_p contribution
    # For each lag p (0 to 2n-2), k_p = p - (n-1)
    # dc_p/dh[j] = dx * [(1-h[j+k_p]) * 1_{valid} - h[j-k_p] * 1_{valid}]

    # = dx * sum_p w_p * [(1-h[j+k_p]) * 1_{valid} - h[j-k_p] * 1_{valid}]
    
    grad = np.zeros(n)
    for p in range(2*n-1):
        if w[p] < 1e-15:
            continue
        k = p - (n-1)
        if k >= 0:
            # j+k < n => j < n-k
            # 1-h[j+k] for j = 0..n-k-1

            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(x, n, alpha):
    h = np.clip(x, 0, 1)
    c = compute_c_vec(h, n)
    a = alpha * c
    a -= a.max()
    e = np.exp(a)
    s = e.sum()
    obj = np.log(s) / alpha
    w = e / s
    # Gradient
    dx = 2.0 / n
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        wp = w[p]
        if wp < 1e-15:
            continue
        k = p - (n - 1)
        if k >= 0:
            grad[:n-k] += wp * dx * (1 - h[k:])
            grad[k:] += wp * dx * (-h[:n-k])
        else:
            kk = -k
            grad[kk:] += wp * dx * (1 - h[:n-kk])
            grad[:n-kk] += wp * dx * (-h[kk:])
    # Chain rule through clip (just pass through where 0<h<1)
    grad *= (h > 1e-10) * (h < 1 - 1e-10)
    return obj, grad

        return v

        _, g = smooth_obj_and_grad(x, n, alpha)
        return g

        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h

**Alternative: Reformulation as a min-max problem**

min_h max_k g_k(h)

This can be solved via the Lagrangian dual:
min_h max_{λ≥0, sum λ=1} sum_k λ_k g_k(h)

For fixed λ, the inner problem is:
min_h sum_k λ_k g_k(h) s.t. sum h = n/2, 0≤h≤1

= dx * sum_i h[i] * (S_i - R_i)

min_h dx * sum_i h[i] * (1 - R_i(h))

where R_i(h) = sum_k λ_k h[i+k] is a linear function of h.

This is: min_h dx * [sum_i h[i] - sum_i h[i] * R_i(h)]
= dx * [n/2 - sum_i h[i] * R_i(h)]

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

    return float(np.max(compute_c_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14: break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += d / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += d / s * h

    c = compute_c_vec(h, n)
    a = alpha * (c - c.max())
    e = np.exp(a)

    """Precompute f_i(k) and g_j(k) for binary swap analysis."""

    c = np.correlate(hb, 1-hb, mode='full') * dx  # shape (2n-1,)
    # k_p = p - (n-1) for p = 0..2n-2, so k ranges from -(n-1) to (n-1)
    
    # f_i(k) = -(1-h[i+k])*1_{0<=i+k<n} + h[i-k]*1_{0<=i-k<n}
    # For binary h: h[i+k] is 0 or 1

    G = np.zeros((n, 2*n-1))  # G[j, p] = g_j(k_p)

        for p in range(2*n-1):

            if 0 <= i+k < n:
                g += (1 - hb[i+k])
            if 0 <= i-k < n:
                g -= hb[i-k]
            G[i, p] = g
    return c, F, G, dx

                grad[:n - kk] += wp * dx * (-x[kk:])
    return grad

def lbfgsb_run(h, n, alpha, maxiter, deadline):
    def f(x):
        return smooth_max_val(x, n, alpha)
    def g(x):
        return smooth_max_grad(x, n, alpha)
    try:
        res = minimize(f, h.copy(), jac=g, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h

        m = c.max()

                    grad[:nk] += inv_k * dx * (1.0 - h[k:])
                    grad[k:] += inv_k * dx * (-h[:nk])
            else:

                if kk < n:

                    grad[:n - kk] += inv_k * dx * (-h[kk:])

            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-14:
                h = ht
                found = True
                break
        if not found:
            lr *= 0.8
            if lr < 1e-14:
                break
        else:
            cn = compute_c5(h, n)
            if cn < best_c:
                best_c = cn
                best_h = h.copy()
    return best_h, best_c

def binary_swap(h, n, deadline, max_sweeps=100):

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += d / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += d / s * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c_vec(x, n)
    a = alpha * (c - c.max())
