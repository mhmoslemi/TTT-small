from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_detail(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr, float(np.max(corr))

def project_onto_simplex_clip(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            return np.clip(h, 0, 1)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_at_max_lags(h, n_lags=20):
    """Compute subgradient focusing on the top lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    # Get top lags
    order = np.argsort(corr)[::-1][:n_lags]
    g = np.zeros(n)
    for i in order:
        lag = int(i) - (n - 1)
        if lag >= 0:
            g[:n-lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n-lag]
        else:
            la = -lag
            g[la:] += (1.0 - h[:n-la])
            g[:n-la] -= h[la:]
    return g / len(order), corr * (2.0/n), float(np.max(corr) * (2.0/n))

def smooth_max_c5(h, beta):
    """Smooth approximation of C5 using log-sum-exp."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(np.clip(beta*(corr - m), -500, 500))) / len(corr)) / beta

    return np.sum(x) - target

    return np.ones(x.shape)

def run_slsqp(h0, n, target, beta, maxiter=3000, ftol=1e-14):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    def obj(x):
        return smooth_max_c5(x, beta)
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_onto_simplex_clip(r.x, target)
    except:
        return h0

    """Coordinate descent with adaptive step size and binary search refinement."""
    best_c5 = compute_c5(h)
    n = len(h)
    step = step_init
    stall = 0
    
    for it in range(n_iter_per_coord):
        if time.time() - t0 > budget - 3:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget - 1:
                break
            # Try positive and negative moves

                trial = h.copy()
                trial[j] = np.clip(trial[j] + sign * step, 0, 1)
                trial = project_onto_simplex_clip(trial, target)

                        trial2[j] = np.clip(trial2[j] + sign * step * mult, 0, 1)

                    break
    return h, best_c5

def perturb_and_restart(h, target, magnitude, rng):
    """Add random perturbation and project."""
    noise = rng.uniform(-magnitude, magnitude, size=len(h))
    h_noisy = h + noise
    return project_onto_simplex_clip(h_noisy, target)

def block_descent(h, target, block_size, n_sweeps, t0, budget):
    """Optimize blocks of adjacent coordinates together."""
    n = len(h)
    best_c5 = compute_c5(h)
    for sw in range(n_sweeps):
        if time.time() - t0 > budget - 2:
            break
        improved = False
        for start in range(0, n, block_size):
            if time.time() - t0 > budget - 1:
                break

            # Try perturbing the block

                    c = compute_c5(trial_h)
                    if c < best_c5 - 1e-15:
                        h = trial_h
                        best_c5 = c
                        improved = True
                        break
                if improved:
                    break
    return h, best_c5

    x_old = np.linspace(0, 2, len(h))

    return np.interp(x_new, x_old, h)

Find a binary string of length n with n/2 ones that minimizes max_k |{i: s[i]=1, s[i+k]=0}| * dx.

This is equivalent to maximizing min_k |{i: s[i]=1, s[i+k]=1}| (the autocorrelation).

For a binary string, the autocorrelation at lag k is the number of positions where both s[i]=1 and s[i+k]=1. We want this to be as large as possible for all k simultaneously.

This is related to the concept of "perfect sequences" or "low autocorrelation sequences" in radar/communications, but inverted - we want HIGH autocorrelation at all lags.

A sequence with high autocorrelation at all lags is one that's "self-similar" or has a lot of structure. For example, a repeated pattern like 11001100... would have high autocorrelation at lags that are multiples of the period.

But we want high autocorrelation at ALL lags, which is a stronger condition. The only way to achieve this perfectly is if the sequence is all 1s (but we can't have that due to the constraint of exactly n/2 ones).

So the optimal binary sequence is one that "spreads out" the ones as evenly as possible while maintaining high overlap at all shifts. This is related to the concept of "Bose-Chowla sequences" or "difference sets."

A (v, k, λ) difference set is a subset D of a group of order v such that every non-identity element can be expressed as a difference d1-d2 (with d1, d2 ∈ D) in exactly λ ways. For cyclic difference sets, this gives sequences with constant autocorrelation.

For our problem, we want a binary sequence of length n with n/2 ones such that the autocorrelation is as uniform (and high) as possible. This is exactly the difference set problem!

If we can find a cyclic (n, n/2, λ) difference set, then the autocorrelation at all non-zero lags would be λ, and the overlap at all non-zero lags would be n/2 - λ. The overlap at lag 0 would be 0 (since s[i](1-s[i]) = 0 for binary s).

So if autocorr(k) = λ for all k ≠ 0, then overlap(k) = n/2 - λ for all k ≠ 0.

For a (v, k, λ) difference set in Z_v: k(k-1) = λ(v-1).
With v = n, k = n/2: (n/2)(n/2-1) = λ(n-1), so λ = (n/2)(n/2-1)/(n-1) = n(n-2)/(4(n-1)).

For this to be an integer, we need 4(n-1) | n(n-2). This is a strong condition.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def critical_subgrad(h, n_top=10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    order = np.argsort(corr)[::-1][:n_top]
    g = np.zeros(n)
    for i in order:
        lag = int(i) - (n - 1)
        if lag >= 0:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
        else:
            la = -lag
            g[la:] += (1.0 - h[:n - la])
            g[:n - la] -= h[la:]
    norm = np.linalg.norm(g)
    if norm > 1e-14:
        g /= norm
    return g, mc * (2.0 / n)

def smooth_max_c5(h, beta):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(np.clip(beta * (corr - m), -500, 500))) / len(corr)) / beta

def _sum_fun(x, target):
    return np.sum(x) - target

def _sum_jac(x, target):
    return np.ones(x.shape)

def run_slsqp(h0, n, target, beta, maxiter=3000, ftol=1e-14):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (target,)}]
    try:
        r = minimize(smooth_max_c5, h0, args=(beta,), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def nesterov_subgrad(h0, n, target, n_iter, step0, t0, deadline, n_top=10):
    h = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    step = step0
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        g, _ = critical_subgrad(y, n_top)
        h_new = project_h(y - step * g, target)
        mu = (1.0 - np.sqrt(1.0 + 4.0 / ((it + 1) ** 2))) / 2.0
        y = np.clip(h_new + mu * (h_new - h_prev), 0, 1)
        h_prev, h = h, h_new
        c = compute_c5(h)
        if c < best_c5 - 1e-15:
            best_h, best_c5, stall = h.copy(), c, 0
        else:
            stall += 1
            if stall % 50 == 0:
                step *= 0.8
                if step < 1e-6:
                    break
    return best_h, best_c5

def coord_descent_adapt(h, target, max_sweeps, t0, deadline, step_init=0.04, step_min=0.0005):
    best_c5 = compute_c5(h)
    n = len(h)
    step = step_init
    stall = 0
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for sign in (1, -1):
                for mult in (1.0, 0.5, 0.2):
                    trial = h.copy()
                    trial[j] = np.clip(trial[j] + sign * step * mult, 0, 1)
                    trial = project_h(trial, target)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-15:
                        h, best_c5, improved = trial, c, True
                        break
                if improved:
                    break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 2:
                step *= 0.6
                if step < step_min:
                    break
    return h, best_c5

def pair_search(h, target, n_iter, t0, deadline, base=0.03, rng_seed=7):
    best_c5 = compute_c5(h)
    n = len(h)
    rng = np.random.RandomState(rng_seed)
    step = base
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        j, k = rng.choice(n, 2, replace=False)
        improved = False
        for sign in (1, -1):
            for mult in (1.0, 0.5, 0.25):
                s = sign * step * mult
                trial = h.copy()
                trial[j] = np.clip(trial[j] + s, 0, 1)
                trial[k] = np.clip(trial[k] - s, 0, 1)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    h, best_c5, improved = trial, c, True
                    break
            if improved:
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 30:
                step *= 0.7
                if step < 1e-4:
                    break
    return h, best_c5

def lag_focused_opt(h, target, t0, deadline, n_top=8):
    """Iteratively focus on the most critical lags and reduce them."""
    best_c5 = compute_c5(h)
    n = len(h)
    dx = 2.0 / n
    for it in range(500):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        mc = np.max(corr)
        if mc < best_c5 - 1e-14:
            best_c5 = mc
        # Find top lags
        order = np.argsort(corr)[::-1][:n_top]
        # For each critical lag, find coordinated move that reduces it most
        improved = False
        for idx in order[:5]:
            if time.time() > deadline:
                break
            lag = int(idx) - (n - 1)
            # Compute which positions contribute most to this lag's overlap
            if lag >= 0:
                valid = slice(0, n - lag)
                contrib = h[valid] * (1.0 - h[lag:lag + len(h[valid])])
            else:
                la = -lag
                valid = slice(la, n)
                contrib = h[la:] * (1.0 - h[:n - la])
            # Find positions where h is high and h[+lag] is low (or vice versa)
            top_pos = np.argsort(contrib)[::-1][:3]
            for p in top_pos:
                if lag >= 0:
                    j = p
                    k = p + lag
                else:
                    j = p + (-lag)
                    k = p
                if j < 0 or j >= n or k < 0 or k >= n:
                    continue
                # Try to reduce overlap: decrease h[j] or increase h[k]
                for j_delta, k_delta in [(-0.03, 0.03), (0.03, -0.03), (-0.02, 0.02)]:
                    trial = h.copy()
                    trial[j] = np.clip(trial[j] + j_delta, 0, 1)
                    trial[k] = np.clip(trial[k] + k_delta, 0, 1)
                    trial = project_h(trial, target)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-15:
                        h, best_c5, improved = trial, c, True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved and it > 100:
            break
    return h, best_c5

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def perturb(h, target, mag, rng):
    return project_h(h + rng.uniform(-mag, mag, size=len(h)), target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(384, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    resolutions = sorted(set([len(h_parent), 384, 448, 512]))
    resolutions = [r for r in resolutions if 256 <= r <= 512]

    for n_pts in resolutions:
        rem = budget_s - (time.time() - t0)
        if rem < 120:
            break
        target = n_pts / 2.0
        cur = project_h(resample(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5 = cur.copy(), cur_c5

        # SLSQP with multiple beta
        for beta in [400, 1000, 2500]:
            rem = budget_s - (time.time() - t0)
            if rem < 25:
                break
            cur = run_slsqp(cur, n_pts, target, beta, maxiter=2500, ftol=1e-14)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5

        # Multi-round local search with perturbation
        for rnd in range(4):
            rem = budget_s - (time.time() - t0)
            if rem < 30:
                break
            mag = 0.04 * (0.6 ** rnd)
            cur = perturb(cur, target, mag, rng)

            # Nesterov subgradient
            rem = budget_s - (time.time() - t0)
            if rem > 20:
                cur, cur_c5 = nesterov_subgrad(cur, n_pts, target, 5000, 0.02, t0,
                                               time.time() + min(rem - 5, 60), n_top=8)

            # Lag-focused optimization
            rem = budget_s - (time.time() - t0)
            if rem > 15:
                cur, cur_c5 = lag_focused_opt(cur, target, t0, time.time() + min(rem - 5, 40))

            # SLSQP polish
            rem = budget_s - (time.time() - t0)
            if rem > 15:
                cur = run_slsqp(cur, n_pts, target, 1500, maxiter=1500, ftol=1e-13)
                cur_c5 = compute_c5(cur)

            # Adaptive coordinate descent
            rem = budget_s - (time.time() - t0)
            if rem > 15:
                cur, cur_c5 = coord_descent_adapt(cur, target, 8, t0, time.time() + min(rem - 3, 50),
                                                  step_init=0.02, step_min=0.001)

            # Pair search
            rem = budget_s - (time.time() - t0)
            if rem > 10:
                cur, cur_c5 = pair_search(cur, target, 3000, t0, time.time() + min(rem - 3, 40),
                                          base=0.025, rng_seed=seed + rnd)

            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5

        # Final fine-tuning on this resolution
        rem = budget_s - (time.time() - t0)
        if rem > 20:
            cur, cur_c5 = coord_descent_adapt(cur, target, 15, t0, time.time() + rem - 5,
                                              step_init=0.01, step_min=0.0003)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5
        rem = budget_s - (time.time() - t0)
        if rem > 10:
            cur, cur_c5 = pair_search(cur, target, 2000, t0, time.time() + rem - 3,
                                      base=0.015, rng_seed=seed + 99)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5

    # Final global polish
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        nf = len(best_h)
        tf = nf / 2.0
        best_h = run_slsqp(best_h, nf, tf, 3000, maxiter=6000, ftol=1e-15)
        best_c5 = compute_c5(best_h)
        rem = budget_s - (time.time() - t0)
        if rem > 25:
            best_h, best_c5 = nesterov_subgrad(best_h, nf, tf, 4000, 0.01, t0,
                                               time.time() + rem - 5, n_top=6)
        rem = budget_s - (time.time() - t0)
        if rem > 15:
            best_h, best_c5 = coord_descent_adapt(best_h, tf, 12, t0, time.time() + rem - 3,
                                                  step_init=0.008, step_min=0.0002)
        rem = budget_s - (time.time() - t0)
        if rem > 8:
            best_h, best_c5 = lag_focused_opt(best_h, tf, t0, time.time() + rem - 2, n_top=6)

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
