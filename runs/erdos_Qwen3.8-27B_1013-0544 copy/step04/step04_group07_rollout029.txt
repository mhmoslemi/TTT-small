def sa_binary_fast(h_int, n, n_iter, t_start, t_end, t0, budget, seed=42):
    h = h_int.copy()  # int array, 0 or 1
    dx = 2.0 / n
    
    # Initial correlation
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    # corr has length 2n-1, index n-1 corresponds to lag 0
    # corr[n-1+k] = R[k] for lag k
    
    max_corr = np.max(corr)
    best_h = h.copy()
    best_c5 = max_corr * dx
    
    rng = np.random.RandomState(seed)
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    # Precompute k array
    k_arr = np.arange(-(n-1), n)  # all lags
    
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        
        alpha = (it + 1) / n_iter
        T = t_start * (t_end / t_start) ** alpha
        
        # Random swap
        ii = rng.randint(len(ones_list))
        jj = rng.randint(len(zeros_list))
        i = ones_list[ii]
        j = zeros_list[jj]
        
        # Compute delta for all lags
        # h is 0/1 int array
        # For lag k: ΔR[k] = -(1-h[i+k]) - h[i-k] + (1-h[j+k]) - h[j-k]
        # with out-of-bounds treated as 0
        
        # Vectorized:
        # i+k_k valid when 0 <= i+k_k < n
        idx_ik = i + k_arr  # positions for h[i+k]
        idx_ik_k = i - k_arr  # positions for h[i-k]
        idx_jk = j + k_arr  # positions for h[j+k]
        idx_jk_k = j - k_arr  # positions for h[j-k]
        
        delta = np.zeros(2*n-1)
        # Term 1: -(1-h[i+k]) for valid i+k
        m1 = (idx_ik >= 0) & (idx_ik < n)
        delta[m1] -= (1 - h[idx_ik[m1]])
        # Term 2: -h[i-k] for valid i-k
        m2 = (idx_ik_k >= 0) & (idx_ik_k < n)
        delta[m2] -= h[idx_ik_k[m2]]
        # Term 3: +(1-h[j+k]) for valid j+k
        m3 = (idx_jk >= 0) & (idx_jk < n)
        delta[m3] += (1 - h[idx_jk[m3]])
        # Term 4: -h[j-k] for valid j-k
        m4 = (idx_jk_k >= 0) & (idx_jk_k < n)
        delta[m4] -= h[idx_jk_k[m4]]
        
        new_corr = corr + delta
        new_max = np.max(new_corr)
        
        # Accept/reject
        accept = False
        if new_max <= max_corr:
            accept = True
        else:
            delta_E = (new_max - max_corr) * dx
            if delta_E < T:

            corr = new_corr
            max_corr = new_max

            zeros_list[jj] = i
            if max_corr * dx < best_c5:
                best_h = h.copy()
                best_c5 = max_corr * dx
    
    return best_h.astype(float), best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def snap_to_binary(h, n):
    """Convert h to binary with exactly n/2 ones."""
    target = n // 2
    # Threshold
    z = (h >= 0.5).astype(int)
    # Adjust count
    ones = z.sum()
    if ones > target:
        # Need to flip some 1s to 0s: pick the ones with lowest h value
        one_idx = np.where(z == 1)[0]
        flip = one_idx[np.argsort(h[one_idx])[:ones - target]]
        z[flip] = 0
    elif ones < target:
        zero_idx = np.where(z == 0)[0]
        flip = zero_idx[np.argsort(-h[zero_idx])[:target - ones]]
        z[flip] = 1
    return z

def sa_binary(h_init, n, n_iter, t_init, t_final, t0, budget, seed=42, verbose=False):
    """Simulated annealing on binary h with exactly n/2 ones."""
    h = h_init.copy()
    dx = 2.0 / n
    
    # Initial correlation (vector of length 2n-1)
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    max_corr = int(np.max(corr))
    best_h = h.copy()
    best_c5 = max_corr * dx
    best_max_corr = max_corr
    
    rng = np.random.RandomState(seed)
    
    # Precompute k array for all lags
    k_arr = np.arange(-(n-1), n)  # length 2n-1
    
    # Track ones and zeros positions for random selection
    ones_pos = np.where(h == 1)[0]
    zeros_pos = np.where(h == 0)[0]
    
    # For faster max tracking, use a histogram
    # corr values are integers in [0, n]
    corr_hist = np.bincount(corr.astype(int), minlength=n+1)
    # max_corr is the largest index with non-zero count
    
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        
        alpha = (it + 1) / n_iter
        # Exponential cooling
        T = t_init * (t_final / t_init) ** alpha
        
        # Random swap: pick one from ones_pos and one from zeros_pos
        ii = rng.randint(len(ones_pos))
        jj = rng.randint(len(zeros_pos))
        i = ones_pos[ii]
        j = zeros_pos[jj]
        
        if i == j:
            continue
        
        # Compute delta correlation
        # For lag k (index = k + n - 1 in corr array):
        # delta = -(1-h[i+k]) - h[i-k] + (1-h[j+k]) - h[j-k]
        # where out-of-bounds h values are 0
        
        # Vectorized computation
        p1 = i + k_arr  # i+k for each lag
        p2 = i - k_arr  # i-k for each lag
        p3 = j + k_arr  # j+k for each lag
        p4 = j - k_arr  # j-k for each lag
        
        delta = np.zeros(2*n-1, dtype=np.int32)
        
        m = (p1 >= 0) & (p1 < n)
        delta[m] -= (1 - h[p1[m]])
        
        m = (p2 >= 0) & (p2 < n)
        delta[m] -= h[p2[m]]
        
        m = (p3 >= 0) & (p3 < n)
        delta[m] += (1 - h[p3[m]])
        
        m = (p4 >= 0) & (p4 < n)
        delta[m] -= h[p4[m]]
        
        new_corr = corr + delta
        new_max = int(np.max(new_corr))
        
        # Accept or reject
        accept = False
        if new_max <= max_corr:
            accept = True
        else:
            # Metropolis: accept with prob exp(-(new_max - max_corr)*dx / T)
            dE = (new_max - max_corr) * dx
            if dE < T:
                if rng.rand() < np.exp(-dE / max(T, 1e-15)):
                    accept = True
        
        if accept:
            # Update
            h[i], h[j] = 0, 1
            corr = new_corr
            max_corr = new_max
            ones_pos[ii] = j
            zeros_pos[jj] = i
            
            if max_corr < best_max_corr:
                best_h = h.copy()
                best_max_corr = max_corr
                best_c5 = max_corr * dx
                if verbose:
                    print(f"  it={it}, improved to max_corr={max_corr}, C5={best_c5:.6f}")
    
    return best_h.astype(float), best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def snap_to_binary(h, n):
    target = n // 2
    z = (h >= 0.5).astype(int)
    ones = int(z.sum())
    if ones > target:
        one_idx = np.where(z == 1)[0]
        flip = one_idx[np.argsort(h[one_idx])[:ones - target]]
        z[flip] = 0
    elif ones < target:
        zero_idx = np.where(z == 0)[0]
        flip = zero_idx[np.argsort(-h[zero_idx])[:target - ones]]
        z[flip] = 1
    return z

def sa_binary(h_init, n, max_iter, t_init, t_final, t0, deadline, seed=42):
    h = h_init.copy().astype(np.int8)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    max_corr = int(np.max(corr))
    best_h = h.copy()
    best_mc = max_corr
    rng = np.random.RandomState(seed)
    k_arr = np.arange(-(n-1), n)
    ones_pos = np.where(h == 1)[0].astype(np.int32).copy()
    zeros_pos = np.where(h == 0)[0].astype(np.int32).copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        alpha = (it + 1) / max_iter
        T = t_init * (t_final / t_init) ** alpha
        
        ii = rng.randint(len(ones_pos))
        jj = rng.randint(len(zeros_pos))
        i = int(ones_pos[ii])
        j = int(zeros_pos[jj])
        
        p1 = i + k_arr
        p2 = i - k_arr
        p3 = j + k_arr
        p4 = j - k_arr
        
        delta = np.zeros(2*n-1, dtype=np.int32)
        m = (p1 >= 0) & (p1 < n)
        delta[m] -= (1 - h[p1[m]])
        m = (p2 >= 0) & (p2 < n)
        delta[m] -= h[p2[m]]
        m = (p3 >= 0) & (p3 < n)
        delta[m] += (1 - h[p3[m]])
        m = (p4 >= 0) & (p4 < n)
        delta[m] -= h[p4[m]]
        
        new_corr = corr + delta
        new_max = int(np.max(new_corr))
        
        accept = False
        if new_max <= max_corr:
            accept = True
        else:
            dE = (new_max - max_corr) * dx
            if dE < T:
                if rng.rand() < np.exp(-dE / T):
                    accept = True
        
        if accept:
            h[i], h[j] = 0, 1
            corr = new_corr
            max_corr = new_max
            ones_pos[ii] = j
            zeros_pos[jj] = i
            if max_corr < best_mc:
                best_h = h.copy()
                best_mc = max_corr
    
    return best_h.astype(float), float(best_mc * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta*(x-m), -500, 500))) / len(x)) / beta

def sml_obj(h_in, beta, dx):
    return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0/n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

    for _ in range(200000):
        if time.time() > deadline:

        m = (p3>=0)&(p3<n); delta[m] += (1-h[p3[m]])
        m = (p4>=0)&(p4<n); delta[m] -= h[p4[m]]
        new_corr = corr + delta
        new_max = int(np.max(new_corr))
        if new_max < max_corr:

            corr = new_corr

            ones_pos[ii] = j
