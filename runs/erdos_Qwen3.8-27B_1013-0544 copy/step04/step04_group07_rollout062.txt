import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_eq(x, t):
    return float(np.sum(x) - t)

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def sml_objective(h_in, beta, n):
    corr = np.correlate(h_in, 1.0 - h_in, mode='full')
    m = np.max(corr)
    idx = np.where(corr >= m - 0.02 * (m + 1e-10))[0]
    if len(idx) == 0:
        idx = np.array([np.argmax(corr)])
    sub = corr[idx]
    sm = float(sub.max() + np.log(np.sum(np.exp(beta * (sub - sub.max())))) / beta)
    return sm * (2.0 / n)

def coord_sweep_fast(h, n, num_bisect, t0, budget):
    ncorr = 2 * n - 1
    for i in range(n):
        if time.time() - t0 > budget - 5:
            break
        h_old = h[i]
        base_corr = np.correlate(h, 1.0 - h, mode='full')
        slopes = np.zeros(ncorr)
        for k in range(ncorr):
            lag = k - (n - 1)
            s = 0.0
            j1 = i + lag
            if 0 <= j1 < n:
                s += 1.0 - h[j1]
            j2 = i - lag
            if 0 <= j2 < n:
                s -= h[j2]
            slopes[k] = s
        lo, hi = 0.0, 1.0
        for _ in range(num_bisect):
            mid = (lo + hi) / 2.0
            delta = mid - h_old
            corr = base_corr + delta * slopes
            corr[n - 1] = base_corr[n - 1] + delta * (1.0 - mid - h_old)
            mc = np.max(corr)
            active = np.where(corr >= mc - 1e-13)[0]
            min_s = float(np.min(slopes[active]))
            max_s = float(np.max(slopes[active]))
            if (n - 1) in active:
                s_n1 = 1.0 - 2.0 * mid
                if s_n1 < min_s:
                    min_s = s_n1
                if s_n1 > max_s:
                    max_s = s_n1
            if min_s > 1e-11:
                hi = mid
            elif max_s < -1e-11:
                lo = mid
            else:
                lo = hi = mid
                break
        h[i] = (lo + hi) / 2.0
    return h

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_eq(x, t):
    return float(np.sum(x) - t)

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def sml_objective(h_in, beta, n):
    corr = np.correlate(h_in, 1.0 - h_in, mode='full')
    m = np.max(corr)
    idx = np.where(corr >= m - 0.02 * (m + 1e-10))[0]
    if len(idx) == 0:
        idx = np.array([np.argmax(corr)])
    sub = corr[idx]
    sm = float(sub.max() + np.log(np.sum(np.exp(beta * (sub - sub.max())))) / beta)
    return sm * (2.0 / n)

def coord_sweep_fast(h, n, num_bisect, t0, budget):
    ncorr = 2 * n - 1
    for i in range(n):
        if time.time() - t0 > budget - 5:
            break
        h_old = h[i]
        base_corr = np.correlate(h, 1.0 - h, mode='full')
        slopes = np.zeros(ncorr)
        for k in range(ncorr):
            lag = k - (n - 1)
            s = 0.0
            j1 = i + lag
            if 0 <= j1 < n:
                s += 1.0 - h[j1]
            j2 = i - lag
            if 0 <= j2 < n:
                s -= h[j2]
            slopes[k] = s
        lo, hi = 0.0, 1.0
        for _ in range(num_bisect):
            mid = (lo + hi) / 2.0
            delta = mid - h_old
            corr = base_corr + delta * slopes
            corr[n - 1] = base_corr[n - 1] + delta * (1.0 - mid - h_old)
            mc = np.max(corr)
            active = np.where(corr >= mc - 1e-13)[0]
            min_s = float(np.min(slopes[active]))
            max_s = float(np.max(slopes[active]))
            if (n - 1) in active:
                s_n1 = 1.0 - 2.0 * mid
                if s_n1 < min_s:
                    min_s = s_n1
                if s_n1 > max_s:
                    max_s = s_n1
            if min_s > 1e-11:
                hi = mid
            elif max_s < -1e-11:
                lo = mid
            else:
                lo = hi = mid
                break
        h[i] = (lo + hi) / 2.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(320, 0.5)

    n_p = len(h_parent)
    h_parent = project_h(h_parent.copy(), n_p / 2.0)

    n_pts = 384
    target = n_pts / 2.0

    best_h = np.interp(np.linspace(0, 2, n_pts), np.linspace(0, 2, n_p), h_parent)
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)

    starts = [best_h.copy()]

    h_n = best_h + 0.15 * np.random.randn(n_pts)
    starts.append(project_h(h_n, target))

    idx = np.random.choice(n_pts, int(target), replace=False)
    h_rb = np.zeros(n_pts)
    h_rb[idx] = 1.0
    starts.append(h_rb)

    h_bl = np.zeros(n_pts)
    blen = n_pts // 8
    for b in range(0, n_pts, 2 * blen):
        h_bl[b:b + blen] = 1.0
    starts.append(project_h(h_bl, target))

    h_b2 = np.zeros(n_pts)
    q = n_pts // 4
    h_b2[:q] = 1.0
    h_b2[2 * q:3 * q] = 1.0
    starts.append(h_b2)

    for si, h_start in enumerate(starts):
        rem = budget_s - (time.time() - t0)
        if rem < 60:
            break
        cur = h_start.copy()
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h = cur.copy()
            best_c5 = cur_c5

        stall = 0
        pert_size = 0.1
        max_sweeps = 5000
        for sw in range(max_sweeps):
            rem = budget_s - (time.time() - t0)
            if rem < 40:
                break
            cur = coord_sweep_fast(cur, n_pts, 30, t0, budget_s)
            cur = project_h(cur, target)
            new_c5 = compute_c5(cur)
            if new_c5 < best_c5 - 1e-15:
                best_h = cur.copy()
                best_c5 = new_c5
                stall = 0
            elif new_c5 < cur_c5 - 1e-14:
                stall = 0
            else:
                stall += 1
            cur_c5 = new_c5
            if stall > 30:
                if rem > 30:
                    cur = cur + pert_size * np.random.randn(n_pts)
                    cur = project_h(cur, target)
                    pert_size *= 0.7
                    if pert_size < 0.005:
                        pert_size = 0.05
                    stall = 0
                else:
                    break

    rem = budget_s - (time.time() - t0)
    if rem > 80:
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
        cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
        for beta in [500, 2000]:
            rem = budget_s - (time.time() - t0)
            if rem < 30:
                break
            try:
                r = minimize(sml_objective, best_h.copy(), args=(beta, n_pts),
                             method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 5000, 'ftol': 1e-15})
                h_sq = project_h(r.x, target)
                c5_sq = compute_c5(h_sq)
                if c5_sq < best_c5:
                    best_h = h_sq
                    best_c5 = c5_sq
            except Exception:
                pass

    rem = budget_s - (time.time() - t0)
    if rem > 20:
        best_h = coord_sweep_fast(best_h, n_pts, 35, t0, budget_s)
        best_h = project_h(best_h, target)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
