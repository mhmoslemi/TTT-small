```python
    c = np.correlate(h, 1.0 - h, mode="full")

                if j >= la:
                    g -= h[j - la]
                grad[j] += w[m] * g
    
    return grad * (2.0 / n)
```

d(c_k)/d(h) has:
- Positions j >= k: add (1-h[j-k]) to grad[j]

grad[:n-k] -= h[k:]  [the second part, j from 0 to n-1-k, h[j+k] from h[k] to h[n-1]]

grad[:n-la] += (1 - h[la:])  [j from 0 to n-1-la, h[j+la] from h[la] to h[n-1]]
grad[la:] -= h[:n-la]  [j from la to n-1, h[j-la] from h[0] to h[n-1-la]]

```python
    return float(np.max(corr) * dx)

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0

            break
        g = smooth_max_grad(cur, n, beta)
        vel = momentum * vel - lr * g
        trial = project_h(cur + vel, target)

            best_h = trial.copy()
            cur = trial
            stall = 0

            vel *= 0.5

    c = np.correlate(h_in, 1.0 - h_in, mode="full")

    return np.sum(x_in) - target

def sum_eq_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=5000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]

                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(r.x, target)
    except Exception:
        return h0
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
        else:
            h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_snap(h, n):
    h_bin = np.zeros(n)
    h_bin[np.argsort(h)[::-1][:n // 2]] = 1.0
    return h_bin

def smooth_max_grad(h, n, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    cs = np.clip(beta * (c - c.max()), -500, 500)
    w = np.exp(cs)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(len(c)):
        k = m - (n - 1)
        wm = w[m]
        if wm < 1e-12:
            continue
        if k >= 0:
            grad[k:] += wm * (1.0 - h[:n - k])
            grad[:n - k] -= wm * h[k:]
        else:
            la = -k
            grad[:n - la] += wm * (1.0 - h[la:])
            grad[la:] -= wm * h[:n - la]
    return grad * (2.0 / n)

def grad_desc(h, target, n, deadline, beta=300, lr=0.002, mom=0.9):
    cur = h.copy()
    vel = np.zeros(n)
    best = compute_c5(cur)
    best_h = cur.copy()
    stall = 0
    for _ in range(300000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(cur, n, beta)
        vel = mom * vel - lr * g
        trial = project_h(cur + vel, target)
        ct = compute_c5(trial)
        if ct < best - 1e-15:
            best, best_h, cur = ct, trial.copy(), trial
            stall = 0
            lr = min(lr * 1.015, 0.015)
        else:
            stall += 1
            lr *= 0.85
            vel *= 0.5
            if lr < 1e-7 or stall > 400:
                break
    return best_h, best

    h = hb.copy()
    cur = compute_c5(h)
    bh, bc = h.copy(), cur
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    rng = np.random.default_rng(seed)
    while time.time() < deadline:
        bt, ba, bb = cur, -1, -1
        for _ in range(trials):
            a = ones[rng.integers(len(ones))]
            b = zeros[rng.integers(len(zeros))]
            if a == b:
                continue
            h[a], h[b] = 0.0, 1.0
            ct = compute_c5(h)
            h[a], h[b] = 1.0, 0.0
            if ct < bt - 1e-15:
                bt, ba, bb = ct, a, b
        if ba >= 0:
            h[ba], h[bb] = 0.0, 1.0
            cur = bt
            if cur < bc:
                bc, bh = cur, h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            stall += 1
            if stall >= stall_max:

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = c.max()
    cs = np.clip(beta * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(cs)) / len(cs)) / beta) * 2.0 / n

def sum_eq(x, t):
    return x.sum() - t

def sum_eq_j(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, maxiter=4000):
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=Bounds(np.zeros(n), np.ones(n)),
                     constraints=[{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_j, 'args': (target,)}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(r.x, target)
    except Exception:
        return h0
```
