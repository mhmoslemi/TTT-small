```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad_for_lag(h, lag):
    """Gradient of correlation at given lag with respect to h"""
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n-lag]
    else:
        m = -lag
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]
    return g

def lp_solve(h, n_active=50, t0=None, budget=None):
    """Solve LP to get descent direction"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Top n_active lags
    top_idx = np.argsort(corr)[::-1][:n_active]
    
    # Build constraint matrix A @ [δ, t] ≤ b
    # For each active lag: grad_lag · δ + (-1)*t ≤ -corr[lag]
    # i.e., grad_lag · δ - t ≤ -corr[lag]
    
    # Variables: [δ_0, ..., δ_{n-1}, t]
    # Objective: minimize t → c = [0,...,0, 1]
    
    m_cons = n_active
    A_ub = np.zeros((m_cons, n + 1))
    b_ub = np.zeros(m_cons)
    
    for i, idx in enumerate(top_idx):
        lag = int(idx) - (n - 1)
        g = compute_grad_for_lag(h, lag)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0  # -t
        b_ub[i] = -corr[idx] * dx  # We work with dx-scaled values
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    # Bounds: -h[i] ≤ δ[i] ≤ 1-h[i], t is free (or t ≥ some lower bound)
    # For linprog: bounds = [(lb, ub), ...]
    bounds = [(max(-h[i], -1.0), min(1.0 - h[i], 1.0)) for i in range(n)]
    bounds.append((None, None))  # t is free
    
    # Equality: sum(δ) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    delta = res.x[:n]
    t_new = res.x[n]
    
    return delta, t_new
```

```python
def softmax_obj_grad(h, beta):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    bc = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(bc)
    se = e.sum()
    val = (m + np.log(se / len(corr)) / beta) * dx
    w = e / se
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        if w[j] < 1e-11:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += w[j] * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w[j] * (1.0 - h[:-lag])
            g[:n - lag] -= w[j] * h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += w[j] * (1.0 - h[m2:])
            g[m2:] -= w[j] * h[:n - m2]
    return val, g * dx

                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
```
