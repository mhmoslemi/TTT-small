import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def get_active_lags(h, n, topm=20):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:topm]
    return idx, corr

def subgrad_topK(h, n, idx, corr):
    """Subgradient using specific active lags"""
    g = np.zeros(n)
    for i in idx:
        lag = int(i) - (n - 1)
        if lag >= 0:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
        else:
            la = -lag
            g[la:] += (1.0 - h[:n - la])
            g[:n - la] -= h[la:]
    return g / len(idx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500))) / len(x)) / beta

def sml_obj(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx

def sml_grad(h, beta, dx):
    # Gradient of smooth max of correlations
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    weights = np.exp(np.clip(beta * (corr - m), -500, 500))
    weights /= np.sum(weights)
    g = np.zeros(len(h))
    n = len(h)
    for i, w in enumerate(weights):
        if w < 1e-10:
            continue
        lag = i - (n - 1)
        if lag >= 0:
            g[:n - lag] += w * (1.0 - h[lag:])
            g[lag:] -= w * h[:n - lag]
        else:
            la = -lag
            g[la:] += w * (1.0 - h[:n - la])
            g[:n - la] -= w * h[la:]
    return g * dx, corr * weights  # return gradient and weighted corr

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0/n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def nesterov(h0, n, target, n_iter, step0, t0, budget, topm=30):
    h = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    step = step0
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        idx, corr = get_active_lags(y, n, topm)
        g = subgrad_topK(y, n, idx, corr)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        h_new = project_h(y - step * g, target)
        mu = (1.0 - np.sqrt(1.0 + 4.0/((it+1)**2))) / 2.0
        y = np.clip(h_new + mu * (h_new - h_prev), 0, 1)
        h_prev, h = h, h_new
        c = compute_c5(h)
        if c < best_c5 - 1e-15:
            best_h, best_c5, stall = h.copy(), c, 0
        else:
            stall += 1
            if stall % 100 == 0:
                step *= 0.7
    return best_h, best_c5

def frank_wolfe(h0, n, target, n_iter, t0, budget, topm=40):
    h = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        idx, corr = get_active_lags(h, n, topm)
        g = subgrad_topK(h, n, idx, corr)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        order = np.argsort(g)
        z = np.zeros(n)
        z[order[:int(target)]] = 1.0
        diff = target - z.sum()
        if diff > 1e-10:
            z[order[int(target)]] = diff
        elif diff < -1e-10:
            z[order[int(target)-1]] = 1.0 + diff
        alpha = 2.0 / (it + 2)
        h_new = np.clip((1-alpha)*h + alpha*z, 0, 1)
        h_new = project_h(h_new, target)
        c = compute_c5(h_new)
        if c < best_c5 - 1e-15:
            best_h, best_c5, stall = h_new.copy(), c, 0
        else:
            stall += 1
        h = h_new
        if stall > 500:
            break
    return best_h, best_c5

def coord_descent(cur, target, n, sweeps, t0, budget):
    best_c5 = compute_c5(cur)
    deltas = (0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002)
    for sw in range(sweeps):
        if time.time() - t0 > budget - 2:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget - 1:
                break
            orig = cur[j]
            for d in deltas:
                trial = cur.copy()
                trial[j] = np.clip(orig + d, 0, 1)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

def pair_descent(cur, target, n, sweeps, t0, budget, seed=42):
    best_c5 = compute_c5(cur)
    rng = np.random.RandomState(seed)
    for sw in range(sweeps):

            break
        improved = False
        for _ in range(n):

                trial[k] = np.clip(trial[k] - d, 0, 1)

                if c < best_c5 - 1e-14:
                    cur, best_c5, improved = trial, c, True

def fourier_init(n, M=8):
    """Create a Fourier-based initial guess"""
    x = np.linspace(0, 2, n)
    # Try a few Fourier modes
    h = np.full(n, 0.5)
    # Add a cosine mode that might help
    for m in range(1, M+1):
        h = h + 0.02 * np.cos(np.pi * m * x)  # small perturbation

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def simulated_anneal(cur, target, n, t0, budget, seed=123, T0=0.001, T1=0.0001):
    """Simulated annealing with structured moves"""
    rng = np.random.RandomState(seed)
    h = cur.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    n_iter = 0
    n_total = 20000
    for it in range(n_total):
        if time.time() - t0 > budget - 5:
            break
        n_iter = it
        T = T0 * (T1/T0) ** (it / n_total)
        # Random pair move
        j, k = rng.choice(n, 2, replace=False)
        d = rng.uniform(-0.05, 0.05)
        trial = h.copy()
        trial[j] = np.clip(trial[j] + d, 0, 1)
        trial[k] = np.clip(trial[k] - d, 0, 1)
        trial = project_h(trial, target)
        c = compute_c5(trial)
        if c < best_c5 - 1e-14:
            best_h = trial.copy()
            best_c5 = c
        delta = c - compute_c5(h)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = trial
    return best_h, best_c5

                break
            j, k = rng.choice(n, 2, replace=False)
            for d in (0.02, -0.02, 0.01, -0.01, 0.005, -0.005):
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0, 1)
                trial[k] = np.clip(trial[k] - d, 0, 1)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 3:
            break
    return cur, best_c5

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def perturbed_local(cur, target, n, t0, budget, seed=77):
    rng = np.random.RandomState(seed)
    best_h, best_c5 = cur.copy(), compute_c5(cur)
    bounds = Bounds(np.zeros(n), np.ones(n))
    for trial in range(8):
        if time.time() - t0 > budget - 10:
            break
        h_pert = cur + rng.normal(0, 0.008, n)
        h_pert = project_h(h_pert, target)
        h_pert = do_slsqp(h_pert, n, target, 3000, bounds, 800, 1e-13)
        h_pert, c = coord_descent(h_pert, target, n, 3, t0, budget)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = h_pert.copy(), c
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(384, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    n_parent = len(h_parent)
    resolutions = sorted(set([n_parent] + [r for r in [320, 448] if r != n_parent]))

    for n_pts in resolutions:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        target = n_pts / 2.0
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
        cur = project_h(resample(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5 = cur.copy(), cur_c5

        # SLSQP cascade with increasing beta
        for beta, maxit in [(300, 500), (2000, 3000), (8000, 5000)]:
            rem = budget_s - (time.time() - t0)
            if rem < 8:
                break
            cur = do_slsqp(cur, n_pts, target, beta, bounds, maxit, 1e-14)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5
            rem = budget_s - (time.time() - t0)
            if rem > 12:
                cur, cur_c5 = coord_descent(cur, target, n_pts, 5, t0, budget_s)
                if cur_c5 < best_c5:
                    best_h, best_c5 = cur.copy(), cur_c5

        # Nesterov subgradient
        rem = budget_s - (time.time() - t0)
        if rem > 25:
            cur, cur_c5 = nesterov(cur, n_pts, target, 4000, 0.015, t0, budget_s, topm=15)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5

        # Pair descent
        rem = budget_s - (time.time() - t0)
        if rem > 25:
            cur, cur_c5 = pair_descent(cur, target, n_pts, 40, t0, budget_s, seed=seed)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5

        # Coordinate descent
        rem = budget_s - (time.time() - t0)
        if rem > 20:
            cur, cur_c5 = coord_descent(cur, target, n_pts, 20, t0, budget_s)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5

        # High-beta SLSQP polish
        rem = budget_s - (time.time() - t0)
        if rem > 15:
            cur = do_slsqp(cur, n_pts, target, 20000, bounds, 10000, 1e-15)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5

    # Final phase at best resolution
    nf = len(best_h)
    tf = nf / 2.0
    bf = Bounds(np.zeros(nf), np.ones(nf))

    rem = budget_s - (time.time() - t0)
    if rem > 50:
        best_h, best_c5 = perturbed_local(best_h, tf, nf, t0, budget_s, seed=77)

    rem = budget_s - (time.time() - t0)
    if rem > 25:
        best_h = do_slsqp(best_h, nf, tf, 25000, bf, 8000, 1e-15)
        best_c5 = compute_c5(best_h)

    rem = budget_s - (time.time() - t0)
    if rem > 15:
        best_h, best_c5 = coord_descent(best_h, tf, nf, 40, t0, budget_s)

    rem = budget_s - (time.time() - t0)
    if rem > 12:
        best_h, best_c5 = pair_descent(best_h, tf, nf, 30, t0, budget_s, seed=123)

    rem = budget_s - (time.time() - t0)
    if rem > 12:
        best_h = do_slsqp(best_h, nf, tf, 40000, bf, 5000, 1e-15)
        best_c5 = compute_c5(best_h)

    rem = budget_s - (time.time() - t0)
    if rem > 8:
        best_h, best_c5 = coord_descent(best_h, tf, nf, 30, t0, budget_s)

    best_h = project_h(best_h, nf)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, nf
