```python
def project_feasible(h, target_sum):

        else:

def gradient_weighted_lags(h, lags, weights, n):
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag, n):
    """Correct gradient of corr[lag] wrt h."""
    grad = np.zeros(n)
    if lag == 0:
        # corr[0] = sum h[i]*(1-h[i]), d/dh[i] = 1-2*h[i]
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        m = n - lag
        # corr = sum_{i=0}^{m-1} h[i+lag]*(1-h[i])
        # d/dh[j] = (1-h[j-lag])*[j >= lag] - h[j+lag]*[j < m]
        if m > 0:
            grad[lag:] += (1.0 - h[:m])
            grad[:m] -= h[lag:]
    else:
        la = -lag
        m = n - la
        # corr = sum_{i=0}^{m-1} h[i]*(1-h[i+la])
        # d/dh[j] = (1-h[j+la])*[j < m] - h[j-la]*[j >= la]
        if m > 0:
            grad[:m] += (1.0 - h[la:])
            grad[la:] -= h[:m]
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

def gradient_descent(h, n, deadline, seed, lr_init=0.003, momentum_init=0.85,
                     alpha_init=5.0, alpha_max=50.0, top_k=10, max_iter=50000):
    """Main optimization loop with adaptive weights."""
    np.random.seed(seed)
    ts = n / 2.0
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    momentum = np.zeros(n)
    lr = lr_init
    alpha = alpha_init
    stagnant = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        max_corr = corr[top_idx].max()
        
        w = np.exp(alpha * (corr[top_idx] - max_corr))
        w /= w.sum()
        lags = top_idx - (n - 1)
        
        grad = np.zeros(n)
        for lag, wi in zip(lags, w):
            if wi > 1e-10:
                grad += wi * corr_gradient(h, int(lag), n)
        
        momentum = momentum_init * momentum - lr * grad
        h_new = project_feasible(h + momentum, ts)
        
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c_new
        
        if c_new < c5 - 1e-12:
            h = h_new
            c5 = c_new
            stagnant = 0
            lr = min(lr * 1.003, 0.008)
            if it % 100 == 0:
                alpha = min(alpha + 0.3, alpha_max)
        else:
            stagnant += 1
            momentum *= 0.5
            if stagnant > 50:
                lr *= 0.9
                alpha = max(alpha - 1.0, alpha_init)
                if stagnant > 200:
                    # Restart with perturbation
                    h = project_feasible(best_h + np.random.randn(n) * 0.01, ts)
                    c5 = compute_c5(h)
                    momentum = np.zeros(n)
                    lr = lr_init * 0.5
                    stagnant = 0
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Get parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(200, 0.5)
    
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_n = n

    h1, c1 = gradient_descent(h.copy(), n, d1, seed)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1, c1, n
    
    # Phase 2: SLSQP polish (20%)
    d2 = t0 + budget_s * 0.70
    n = best_n
    ts = n / 2.0
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
    
    for beta in [30, 100, 500, 2000, 10000]:
        if time.time() > d2:
            break
        try:
            res = minimize(obj_smooth, best_h.copy(), args=(dx, beta),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5, best_n = h_new, c5_new, n
        except Exception:
            pass
    
    # Phase 3: Try other resolutions (15%)
    d3 = t0 + budget_s * 0.85
    n_best = best_n
    for n_try in [n_best - 30, n_best + 30, n_best + 60, 250, 400, 500]:
        if time.time() > d3:
            break
        if n_try < 100 or n_try == n_best:
            continue
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)

    # Phase 4: Final fine-tuning (10%)
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag, n):
    """Correct gradient of the lag-th correlation term wrt h."""
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        m = n - lag
        if m > 0:
            grad[lag:] += (1.0 - h[:m])
            grad[:m] -= h[lag:]
    else:
        la = -lag
        m = n - la
        if m > 0:
            grad[:m] += (1.0 - h[la:])
            grad[la:] -= h[:m]
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

def gradient_descent(h, n, deadline, seed, lr_init=0.003, mom_init=0.85,
                     alpha_min=3.0, alpha_max=60.0, top_k=10, max_iter=50000):
    np.random.seed(seed)
    ts = n / 2.0
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    momentum = np.zeros(n)
    lr = lr_init
    alpha = alpha_min
    stagnant = 0
    cycle_len = 500

    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        max_corr = corr[top_idx].max()
        w = np.exp(alpha * (corr[top_idx] - max_corr))
        w /= w.sum()
        lags = (top_idx - (n - 1)).astype(int)
        grad = np.zeros(n)
        for lag, wi in zip(lags, w):
            if wi > 1e-10:
                grad += wi * corr_gradient(h, lag, n)
        # Clip gradient norm for stability
        gn = np.linalg.norm(grad)
        if gn > 5.0:
            grad *= 5.0 / gn
        momentum = mom_init * momentum - lr * grad
        h_new = project_feasible(h + momentum, ts)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-13:
            best_h = h_new.copy()
            best_c5 = c_new
        if c_new < c5 - 1e-13:
            h = h_new
            c5 = c_new
            stagnant = 0
            lr = min(lr * 1.002, 0.009)
            # Cyclic alpha: increase then reset
            phase = (it % cycle_len) / cycle_len
            alpha = alpha_min + (alpha_max - alpha_min) * phase
        else:
            stagnant += 1
            momentum *= 0.5
            if stagnant > 80:
                lr *= 0.92
                if stagnant > 250:
                    h = project_feasible(best_h + np.random.randn(n) * 0.008, ts)
                    c5 = compute_c5(h)
                    momentum = np.zeros(n)
                    lr = lr_init * 0.4
                    stagnant = 0
                    alpha = alpha_min
    return best_h, best_c5

def slsqp_polish(h, n, deadline, seed):
    ts = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
    betas = [20, 50, 150, 500, 2000, 8000, 30000]
    for beta in betas:
        if time.time() > deadline:
            break
        perts = [0.0, 0.003, 0.008] if beta < 1000 else [0.0]
        for p in perts:
            if time.time() > deadline:
                break
            x0 = best_h.copy()
            if p > 0:
                x0 = project_feasible(x0 + np.random.randn(n) * p, ts)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 4000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(200, 0.5)

    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_n = n

    # Phase 1: Gradient descent at parent resolution (45%)
    d1 = t0 + budget_s * 0.45
    h1, c1 = gradient_descent(h.copy(), n, d1, seed)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1, c1, n

    # Try reversed parent too
    if time.time() < d1 and n > 1:
        h_rev = project_feasible(h[::-1].copy(), n / 2.0)
        h2, c2 = gradient_descent(h_rev, n, d1, seed + 500)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, n

    # Phase 2: SLSQP polish (20%)
    d2 = t0 + budget_s * 0.65
    best_h, best_c5 = slsqp_polish(best_h.copy(), best_n, d2, seed + 100)
    best_n = best_h.shape[0]

    # Phase 3: Multi-resolution exploration (15%)
    d3 = t0 + budget_s * 0.80
    n_best = best_n
    for n_try in sorted(set([n_best-20, n_best+20, n_best+50, n_best+100, 200, 300, 400, 500])):
        if time.time() > d3:
            break
        if n_try < 80 or n_try == n_best:
            continue
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try / 2.0)
        h_new, c_new = gradient_descent(h_interp, n_try, d3, seed + n_try,
                                         lr_init=0.002, max_iter=20000)
        if c_new < best_c5:
            best_h, best_c5, best_n = h_new, c_new, n_try

    # Phase 4: Fine gradient descent (10%)
    d4 = t0 + budget_s * 0.92
    best_h, best_c5 = gradient_descent(best_h, best_n, d4, seed + 999,
                                        lr_init=0.0004, mom_init=0.92,
                                        alpha_min=10.0, alpha_max=100.0,
                                        top_k=15, max_iter=30000)
    best_n = best_h.shape[0]

    # Phase 5: Final SLSQP (5%)
    d5 = t0 + budget_s * 0.97
    best_h, best_c5 = slsqp_polish(best_h.copy(), best_n, d5, seed + 2000)

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
