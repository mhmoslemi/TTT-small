    """Compute weighted subgradient for given lags and weights."""
    grad = np.zeros(n)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def wsubgrad(h, n, top_idx, weights):
    grad = np.zeros(n)
    for j in range(len(top_idx)):
        k = int(top_idx[j])
        lag = k - (n - 1)
        w = weights[j]
        if lag >= 0:
            m = n - lag
            if m > 0:
                grad[:m] += w * (1.0 - h[lag:lag+m])
                grad[lag:lag+m] -= w * h[:m]
        else:
            la = -lag
            m = n - la
            if m > 0:
                grad[la:] += w * (1.0 - h[:m])
                grad[:m] -= w * h[la:]
    return grad

def subgrad_opt(h, n, deadline, seed, top_k=20, temp=20.0, lr0=0.02, mu=0.85, patience=4000):

        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(corr, -top_k)[-top_k:]
        vals = corr[idx]
        w = np.exp(temp * (vals - vals.max()))
        w /= w.sum()

                h = best_h.copy()
                vel = np.zeros(n)
                lr = lr0
                mu = 0.85
                stag = 0
        h = h_new

    return x.sum() - ts

    sh = np.clip(beta * (corr - m), -400.0, 400.0)
    return (m + np.log(np.sum(np.exp(sh))) / beta) * dx
