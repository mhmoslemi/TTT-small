import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def best_improvement(h, n, deadline, max_sweeps=200):
    """Best-improvement local search for binary h."""
    best = h.copy()
    best_c = compute_c5(best, n)
    ones = np.where(best == 1)[0].tolist()
    zeros = np.where(best == 0)[0].tolist()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                return best, best_c
            for j in ones:
                best[i], best[j] = 1.0, 0.0
                c = compute_c5(best, n)
                if c < best_c - 1e-15:
                    best_c = c
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
                    improved = True
                    break
                else:
                    best[i], best[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return best, best_c
