import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def compute_subgrad(h, top_k=12):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            g = np.zeros(n)
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
            grad += g
        else:
            M = -lag
            g = np.zeros(n)
            g[M:] += (1.0 - h[:n - M])
            g[:n - M] -= h[M:]
            grad += g
    grad /= top_k
    return grad, corr

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def slsqp_objective(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def do_slsqp(h0, n, target, beta, maxiter=8000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_objective, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def adam_subgrad(h0, target, n, n_iter, lr, t0, budget, top_k=16, beta1=0.85, beta2=0.999, eps=1e-8):
    cur = h0.copy().astype(np.float64)
    m = np.zeros(n)
    v = np.zeros(n)
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    t = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        t += 1
        grad, _ = compute_subgrad(cur, top_k=top_k)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        trial = project_h(cur + step, target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            cur = trial
            best_c5 = c_try
            best_h = cur.copy()
    return best_h, best_c5

def adaptive_adam(h0, target, n, n_iter, lr_init, t0, budget, top_k=16):
    cur = h0.copy().astype(np.float64)
    m = np.zeros(n)
    v = np.zeros(n)
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    lr = lr_init
    t = 0
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        t += 1
        grad, _ = compute_subgrad(cur, top_k=top_k)
        m = 0.85 * m + 0.15 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1.0 - 0.85**t)
        v_hat = v / (1.0 - 0.999**t)
        step = -lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        trial = project_h(cur + step, target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            cur = trial
            best_c5 = c_try
            best_h = cur.copy()
            stall = 0
            lr = min(lr * 1.01, 0.15)
        else:
            stall += 1
            lr *= 0.85
            if stall > 200:
                break
    return best_h, best_c5

def binary_anneal(h_bin, n, t0, budget, n_iter=300000, seed=123):
    h = h_bin.copy().astype(np.float64)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for it in range(n_iter):
        if budget - (time.time() - t0) < 5:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        new_c5 = compute_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            ones.remove(a)
            zeros.remove(b)
            ones.append(b)
            zeros.append(a)
        else:
            if it < n_iter:
                frac = it / max(n_iter, 1)
                temp = 0.15 * (0.005) ** frac
                if rng.random() < np.exp(-(new_c5 - cur_c5) / max(temp, 1e-15)):
                    cur_c5 = new_c5
                    ones.remove(a)
                    zeros.remove(b)
                    ones.append(b)
                    zeros.append(a)
                else:
                    h[a] = 1.0
                    h[b] = 0.0
    return best_h, best_c5

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.03, -0.03, 0.015, -0.015, 0.008, -0.008]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0.0, 1.0)
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
