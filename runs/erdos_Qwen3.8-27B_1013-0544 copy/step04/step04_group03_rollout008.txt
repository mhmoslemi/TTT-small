                    bad_zeros.append(i + wl)
        
        # Try targeted swaps
        found = False
        n_attempts = min(50, max(10, len(bad_ones) // 4)) if bad_ones else 30
        
        for _ in range(n_attempts):
            if bad_ones and bad_zeros and rng.random() < 0.7:
                i = bad_ones[rng.integers(len(bad_ones))]
                j = bad_zeros[rng.integers(len(bad_zeros))]
            else:
                ones = np.where(cur == 1)[0]
                zeros = np.where(cur == 0)[0]
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
            
            if cur[i] == cur[j]:
                continue
            
            cur[i], cur[j] = cur[j], cur[i]
            new_c5 = compute_c5(cur)
            
            if new_c5 < cur_c5 - 1e-15:
                cur_c5 = new_c5
                found = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = cur.copy()
                    no_improve = 0
            else:
                cur[i], cur[j] = cur[j], cur[i]
        
        if not found:
            no_improve += 1
            if no_improve > 200:
                # Random perturbation
                ones = np.where(cur == 1)[0]
                zeros = np.where(cur == 0)[0]
                for _ in range(5):
                    i = ones[rng.integers(len(ones))]
                    j = zeros[rng.integers(len(zeros))]
                    cur[i], cur[j] = cur[j], cur[i]
                cur_c5 = compute_c5(cur)
                no_improve = 0
                if cur_c5 > best_c5 + 0.01:

                    cur_c5 = best_c5
    
    return best_h, best_c5

One more idea: **Using the structure of the problem more directly.**

The overlap at lag $k$ is $\sum_i h[i](1-h[i+k])$. For a binary $h$, this equals the number of 1s in $h$ that have a 0 at distance $k$. If we think of the 1s as a set $A$, this is $|A| - |A \cap (A+k)| = |A| - R_A(k)$ where $R_A(k)$ is the autocorrelation of the indicator of $A$.

To minimize the maximum of $|A| - R_A(k)$, we want to maximize the minimum of $R_A(k)$ (for $k \neq 0$).

The autocorrelation of a binary sequence is related to its DFT. Specifically, by the Wiener-Khinchin theorem, the autocorrelation is the inverse DFT of the power spectrum $|H(k)|^2$ where $H(k)$ is the DFT of $h$.

So minimizing the maximum autocorrelation deficit is equivalent to (approximately) making the power spectrum as flat as possible. This suggests a spectral approach:
1. Compute the DFT of the current binary $h$
2. Try to modify $h$ to flatten the power spectrum
3. This is related to "spectral tiling" or "power spectrum flattening"

This is a well-studied problem in signal processing. The "zero autocorrelation bandwidth" (ZACB) problem is related.

However, implementing a full spectral optimization might be complex. Let me stick with the direct approach but add some spectral insights.

Actually, here's a simpler idea: **use the DFT to identify "bad frequencies" and design moves to fix them.**

If the power spectrum has peaks at certain frequencies, those correspond to periodic structures in $h$ that cause high autocorrelation at certain lags. We can try to "break" these periodic structures.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_snap(h):
    n = len(h)
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def directed_binary_opt(h_bin, t0, budget, seed=0, max_iters=300000):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = compute_c5(cur)
    best_c5 = cur_c5
    best_h = cur.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    patience = 500

    for it in range(max_iters):
        if budget - (time.time() - t0) < 15:
            break
        corr = compute_lags(cur)
        max_val = corr.max()
        # Find top critical lags
        n_crit = max(3, min(20, int(0.05 * (2 * n - 1))))
        top_lags = np.argsort(corr)[::-1][:n_crit]

        found = False
        n_attempts = 40
        for _ in range(n_attempts):
            # Choose a critical lag to target
            lag_idx = top_lags[rng.integers(n_crit)]
            lag = int(lag_idx) - (n - 1)
            if lag == 0:
                ones = np.where(cur == 1)[0]
                zeros = np.where(cur == 0)[0]
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
            elif lag > 0 and lag < n:
                # Find positions where cur[i+lag]=1 and cur[i]=0 (swapping helps)
                candidates_i = np.where((cur[:-lag] == 0) & (cur[lag:] == 1))[0]
                candidates_j = np.where((cur[:-lag] == 1) & (cur[lag:] == 0))[0]
                if len(candidates_i) > 0 and len(candidates_j) > 0 and rng.random() < 0.7:
                    i = candidates_j[rng.integers(len(candidates_j))] + lag
                    j = candidates_i[rng.integers(len(candidates_i))]
                else:
                    ones = np.where(cur == 1)[0]
                    zeros = np.where(cur == 0)[0]
                    i = ones[rng.integers(len(ones))]
                    j = zeros[rng.integers(len(zeros))]
            else:
                la = min(-lag, n - 1)
                if la > 0:
                    candidates_i = np.where((cur[la:] == 0) & (cur[:-la] == 1))[0]
                    candidates_j = np.where((cur[la:] == 1) & (cur[:-la] == 0))[0]
                    if len(candidates_i) > 0 and len(candidates_j) > 0 and rng.random() < 0.7:
                        i = candidates_j[rng.integers(len(candidates_j))]
                        j = candidates_i[rng.integers(len(candidates_i))] + la
                    else:
                        ones = np.where(cur == 1)[0]
                        zeros = np.where(cur == 0)[0]
                        i = ones[rng.integers(len(ones))]
                        j = zeros[rng.integers(len(zeros))]
                else:
                    ones = np.where(cur == 1)[0]
                    zeros = np.where(cur == 0)[0]
                    i = ones[rng.integers(len(ones))]
                    j = zeros[rng.integers(len(zeros))]

            if i == j or cur[i] == cur[j]:
                continue
            cur[i], cur[j] = cur[j], cur[i]
            new_c5 = compute_c5(cur)
            if new_c5 < cur_c5 - 1e-15:
                cur_c5 = new_c5
                found = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = cur.copy()
                    no_improve = 0
            else:
                cur[i], cur[j] = cur[j], cur[i]

        if not found:
            no_improve += 1
            if no_improve > patience:
                # Perturb: do 3-8 random swaps
                n_swaps = rng.integers(3, 9)
                for _ in range(n_swaps):
                    ones = np.where(cur == 1)[0]
                    zeros = np.where(cur == 0)[0]
                    i = ones[rng.integers(len(ones))]
                    j = zeros[rng.integers(len(zeros))]
                    cur[i], cur[j] = cur[j], cur[i]

                if cur_c5 > best_c5 + 0.02:
                    cur = best_h.copy()
                    cur_c5 = best_c5
                    patience = 500

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=8000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def subgrad_desc(h, target, n, n_iter, step_init, t0, budget, top_k=16):
    cur = h.copy()
    mom = np.zeros(n)
    best_local = compute_c5(cur)
    stall = 0
    step = step_init
    for it in range(n_iter):
        if budget - (time.time() - t0) < 5:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[:n - lag] += (1.0 - cur[lag:])
                    grad[lag:] -= cur[:n - lag]
            else:
                la = -lag
                grad[la:] += (1.0 - cur[:n - la])
                grad[:n - la] -= cur[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = 0.9 * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            stall = 0
            step = min(step * 1.03, 0.08)
        else:
            stall += 1
            step *= 0.75
            mom *= 0.5
            trial = project_h(cur + mom, target)
            c_try = compute_c5(trial)
            if c_try < best_local - 1e-15:
                cur = trial
                best_local = c_try
                stall = 0
            else:
                mom = np.zeros(n)
        if stall > 400:
            break
    return cur, best_local

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

    # Phase 2: Try shifted/reversed variants
    remaining = budget_s - (time.time() - t0)
    for variant_seed in [99, 200, 456]:

        rng_v = np.random.default_rng(variant_seed)

        ones = np.where(h_v == 1)[0]
        zeros = np.where(h_v == 0)[0]
        for _ in range(n_mut):
            i = ones[rng_v.integers(len(ones))]
            j = zeros[rng_v.integers(len(zeros))]
            h_v[i], h_v[j] = h_v[j], h_v[i]
        h_v_bin = binary_snap(h_v)
