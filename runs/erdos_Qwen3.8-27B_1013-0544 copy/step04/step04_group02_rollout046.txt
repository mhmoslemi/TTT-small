def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def soft_weighted_max(x, beta, k=10):
    # Weight only the top-k values for efficiency
    idx = np.argpartition(x, -k)[-k:]
    top = x[idx]
    m = np.max(top)
    shifted = np.clip(beta * (top - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)

        if abs(s - target_sum) < 1e-14:
            break

        else:

        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, lag, n):

            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]

        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

def obj_pen(h, n, dx, beta, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx + penalty * (h.sum() - n / 2.0) ** 2

def obj_pen_grad(h, n, dx, beta, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        grad += w[k] * subgrad_for_lag(h, lag, n)
    grad *= dx
    grad += 2.0 * penalty * (h.sum() - n / 2.0)
    return grad

def lbfgs_phase(h0, n, deadline, pen_start=50.0, pen_end=500.0, max_iter=2000):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    h = h0.copy()
    betas = [40, 100, 300, 1000, 3000, 10000]
    n_betas = len(betas)
    for bi, beta in enumerate(betas):
        if time.time() > deadline:
            break
        pen = pen_start * (pen_end / pen_start) ** (bi / max(n_betas - 1, 1))
        try:
            res = minimize(obj_pen, h, args=(n, dx, beta, pen),
                          jac=obj_pen_grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50})
            h_new = project_feasible(res.x, n / 2.0)
            if compute_c5(h_new) < compute_c5(h):
                h = h_new
        except Exception:
            pass
    return h

def fista_polish(h, n, deadline, seed, top_k=20, max_iter=8000):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    y = h.copy()
    t_mom = 1.0
    lr = 0.002
    stagnant = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        idx = np.argpartition(corr, -top_k)[-top_k:]
        top_vals = corr[idx]
        m = top_vals.max()
        w = np.exp(60.0 * (top_vals - m))
        w /= w.sum()
        grad = np.zeros(n)
        for j, k in enumerate(idx):
            if w[j] < 1e-12:
                continue
            lag = k - (n - 1)
            grad += w[j] * subgrad_for_lag(y, lag, n)
        h_new = project_feasible(y - lr * grad, ts)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-13:
            stagnant = 0
            lr = min(lr * 1.008, 0.005)
        else:
            stagnant += 1
            lr *= 0.93
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = h_new + (t_mom / t_next - 1.0) * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        c5 = c_new if c_new < c5 else c5
        y = project_feasible(y, ts)
        t_mom = t_next
        if stagnant > 300 and lr < 5e-6:
            break
    return h, c5

def swap_local_search(h, n, deadline, seed, n_trials=200):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        delta = 0.01 + 0.02 * np.random.rand()
        if h[i] >= delta and h[j] <= 1.0 - delta:
            h_try = h.copy()
            h_try[i] -= delta
            h_try[j] += delta
            h_try = project_feasible(h_try, ts)
            c_try = compute_c5(h_try)
            if c_try < c5 - 1e-12:
                h = h_try
                c5 = c_try
        elif h[j] >= delta and h[i] <= 1.0 - delta:
            h_try = h.copy()
            h_try[j] -= delta
            h_try[i] += delta
            h_try = project_feasible(h_try, ts)
            c_try = compute_c5(h_try)
            if c_try < c5 - 1e-12:
                h = h_try
                c5 = c_try
    return h, c5

def random_perturb(h, n, seed, scale=0.04):
    np.random.seed(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    h_new = h.copy()
    for freq in np.random.randint(1, max(2, n // 3), size=4):
        amp = scale * np.random.randn()
        phase = np.random.randn() * np.pi
        h_new += amp * np.cos(2.0 * np.pi * freq * x / 2.0 + phase)
        h_new += amp * 0.5 * np.sin(2.0 * np.pi * (freq + 1) * x / 2.0 + phase)
    return project_feasible(h_new, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
