                    hj = h[j] - d

                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

def optimize_one(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Subgradient phase
    if time.time() < deadline - 20:
        h, c = subgradient_deflate(best_h, n, n_steps=3000, deadline=time.time() + 80)
        if c < best_c:
            best_c = c
            best_h = h

    # SLSQP with increasing alpha
    for alpha in [2000, 10000, 50000, 200000]:
        if time.time() > deadline - 15:
            break
        ht = slsqp_run(best_h, n, alpha, 4000, 1e-16, deadline)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    # Pair search
    if time.time() < deadline - 10:
        if n <= 120:
            best_h, best_c = focused_pair_search(best_h, n, time.time() + 40,
                                                  [0.02, 0.01, 0.005, 0.002, 0.001])
        else:
            best_h, best_c = focused_pair_search(best_h, n, time.time() + 40,
                                                  [0.01, 0.005, 0.002])

    # Final SLSQP polish
    if time.time() < deadline - 5:
        ht = slsqp_run(best_h, n, 500000, 5000, 1e-17, deadline)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    # Final subgradient polish
    if time.time() < deadline - 3:
        ht, c = subgradient_deflate(best_h, n, n_steps=1000, deadline=deadline)
        if c < best_c:
            best_c = c
            best_h = ht

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 100

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize initial solution
    if h_init is not None:
        alloc = min(400, (deadline - t0) * 0.4)
        h_opt, c_opt = optimize_one(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt

    # Phase 2: Try different n values
    if best_h is not None:
        x_best = np.linspace(0, 2, len(best_h))
        for n_try in [100, 150, 80, 120]:
            if time.time() > deadline - 100:
                break
            if n_try == len(best_h):
                continue
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_best, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(200, (deadline - time.time()) * 0.25)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_one(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                x_best = np.linspace(0, 2, len(best_h))

    # Phase 3: Random restarts from best
    if best_h is not None and time.time() < deadline - 80:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 20
        n_restarts = max(3, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.03 + 0.10 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n_cur)
            hp = project_feasible(hp, n_cur)
            alloc = min(50, (deadline - time.time() - 30) * 0.3)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_one(hp, n_cur, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Final polish
    if best_h is not None and time.time() < deadline - 10:
        n_cur = len(best_h)
        # Subgradient polish
        if time.time() < deadline - 15:
            h_pol, c_pol = subgradient_deflate(best_h, n_cur, n_steps=2000,
                                                deadline=time.time() + 40)
            if c_pol < best_c:
                best_c = c_pol
                best_h = h_pol
        # SLSQP polish
        if time.time() < deadline - 5:
            ht = slsqp_run(best_h, n_cur, 1000000, 5000, 1e-17, deadline)
            c = compute_c5(ht, n_cur)
            if c < best_c:
                best_c = c
                best_h = ht
        # Tiny pair search
        if time.time() < deadline - 2:
            best_h, best_c = focused_pair_search(best_h, n_cur, deadline - 0.5,
                                                  [0.005, 0.002, 0.001])

    if best_h is None:
        n = 100
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
