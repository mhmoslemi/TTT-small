def smoothed_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -600, 600)
    exp_v = np.exp(shifted)
    se = np.sum(exp_v)
    return m + np.log(se) / beta

def smoothed_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -600, 600)
    exp_v = np.exp(shifted)
    w = exp_v / np.sum(exp_v)
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        wk = w[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:

        elif 0 < lag < n:

        elif -n < lag < 0:
