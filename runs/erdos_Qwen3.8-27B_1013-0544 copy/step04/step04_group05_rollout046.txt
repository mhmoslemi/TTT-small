import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad(h, lags):
    n = len(h)
    g = np.zeros(n)
    for lag in lags:
        if lag == 0:
            g += (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += (1.0 - h[:-lag])
            g[:n - lag] -= h[lag:]
        else:
            m = -lag
            g[:n - m] += (1.0 - h[m:])
            g[m:] -= h[:n - m]
    g /= len(lags)
    return g

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift))) / beta

def subgrad_adaptive(h, target, time_fn, max_iters, m_start, m_end,
                     step_start, step_end, mom_start, mom_end,
                     perturb_every, perturb_scale):

    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0

    for it in range(max_iters):
        if time_fn() < 3:
            break
        frac = float(it) / max(max_iters, 1)
        m_act = max(3, int(m_start + (m_end - m_start) * frac))
        m_act = min(m_act, n - 1)
        step_cur = step_start * (step_end / step_start) ** frac
        mom_cur = mom_start + (mom_end - mom_start) * frac

        top = np.argsort(corr)[::-1][:m_act]
        lags = [int(i) - (n - 1) for i in top]

        g = compute_subgrad(best, lags)
        vel = mom_cur * vel + (1.0 - mom_cur) * g
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn

        found = False
        s = step_cur
        for _ in range(25):
            if s < 1e-8:
                break
            ht = project_h(best + s * d, target)

                best_c5 = c5t
                no_improve = 0
                found = True

def slsqp_run(h, target, dx, beta, maxiter, rem_fn):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj_fn(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        return smooth_max(c, beta) * dx

    def con_fn(h_in):
        return h_in.sum() - target

    def jac_fn(h_in):
        return np.ones(n)

    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fn, 'jac': jac_fn}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)
