import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    return float(np.max(compute_corr(h)) * 2.0 / len(h))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += 1.0 - h[:-lag]
        g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g * (2.0 / n)

def subgrad_polish(h, target, n_top, t0, budget, max_iter=100000):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    s_prev, g_prev, bb_cnt = None, None, 0
    no_imp = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 2.0:
            break
        corr = compute_corr(best)
        top = np.argsort(corr)[::-1][:n_top]
        g = np.zeros(n)
        for idx in top:
            g += grad_lag(best, int(idx) - (n - 1))
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        step = 0.04
        if s_prev is not None and g_prev is not None:
            y = g - g_prev
            sy = np.dot(s_prev, y)
            yy = np.dot(y, y)
            ss = np.dot(s_prev, s_prev)
            if yy > 1e-16 and sy > 1e-16:
                step = ss / sy if bb_cnt % 2 else sy / yy
                step = float(np.clip(step, 1e-5, 0.3))
            bb_cnt += 1
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.04, 0.01]:
            ht = project_sum(best - step * mult * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                s_prev = ht - best
                best = ht
                best_c5 = c5t
                g_prev = g
                improved = True
                break
        if not improved:
            no_imp += 1
            if no_imp > 150 and it % 80 == 79:
                ht = project_sum(best + np.random.randn(n) * 0.01, target)
                if compute_c5(ht) < best_c5:

                    no_imp, s_prev, g_prev = 0, None, None
                elif no_imp > 4000:
                    break
        else:
            no_imp = 0
    return best, best_c5

def slsqp_opt(h_init, target, beta, maxiter=3000):
    n = len(h_init)
    dx = 2.0 / n
    nc = 2 * n - 1

    def obj(x):
        h = np.clip(x, 0, 1)
        corr = compute_corr(h) * dx
        m = np.max(corr)
        bc = np.clip(beta * (corr - m), -120, 120)
        return m + np.log(np.exp(bc).sum() / nc) / beta

    def grad(x):
        h = np.clip(x, 0, 1)
        corr = compute_corr(h) * dx
        m = np.max(corr)
        bc = np.clip(beta * (corr - m), -120, 120)
        e = np.exp(bc)
        se = e.sum()
        w = e / se
        g = np.zeros(n)
        for j in range(nc):
            if w[j] < 1e-12:
                continue
            g += w[j] * grad_lag(h, j - (n - 1))
        return g

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
    return project_sum(res.x, target)

def lkbfgs_opt(h_init, target, beta, maxiter=1000, pen=500.0):
    n = len(h_init)
    dx = 2.0 / n
    nc = 2 * n - 1

    def obj(x):
        h = np.clip(x, 0, 1)
        corr = compute_corr(h) * dx
        m = np.max(corr)
        bc = np.clip(beta * (corr - m), -120, 120)
        return m + np.log(np.exp(bc).sum() / nc) / beta + pen * (h.sum() - target) ** 2

    def grad(x):
        h = np.clip(x, 0, 1)
        corr = compute_corr(h) * dx
        m = np.max(corr)
        bc = np.clip(beta * (corr - m), -120, 120)
        e = np.exp(bc)
        se = e.sum()
        w = e / se
        g = np.zeros(n)
        for j in range(nc):
            if w[j] < 1e-12:
                continue
            g += w[j] * grad_lag(h, j - (n - 1))
        g += 2.0 * pen * (h.sum() - target)
        return g

    bounds = [(0, 1)] * n
    res = minimize(obj, h_init, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
    return project_sum(res.x, target)

def coord_descent(h, target, n_samp, t0, budget, max_rounds=10):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    for r in range(max_rounds):
        if budget - (time.time() - t0) < 3.0:
            break
        improved_any = False
        for i in range(n):
            if budget - (time.time() - t0) < 2.0:
                break
            best_val = best[i]
            cands = np.linspace(max(0, best_val - 0.3), min(1, best_val + 0.3), n_samp)
            cands = np.append(cands, [0.0, 1.0])
            c5s = []
            for cv in cands:
                ht = best.copy()
                ht[i] = cv
                ht = project_sum(ht, target)
                c5s.append(compute_c5(ht))
            c5s = np.array(c5s)
            bi = int(np.argmin(c5s))
            if c5s[bi] < best_c5 - 1e-14:
                best[i] = cands[bi]
                best = project_sum(best, target)
                best_c5 = compute_c5(best)
                improved_any = True
        if not improved_any:
            break
    return best, best_c5
