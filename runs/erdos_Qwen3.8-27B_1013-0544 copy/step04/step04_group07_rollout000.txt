    return compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(320, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(h_parent)

    n_par = len(h_parent)
    target_par = n_par / 2.0

    # Quick SLSQP on parent to establish baseline improvement
    cur = do_slsqp(best_h, n_par, target_par, 5000, 5000, 1e-15)
    cur_c5 = compute_c5(cur)
    if cur_c5 < best_c5:
        best_h, best_c5, best_n = cur.copy(), cur_c5, n_par

    # Differential Evolution at parent resolution for global exploration
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        pop_size = max(30, min(60, int(80000 / n_par)))
        max_iter = max(50, int((rem - 20) * 2))
        try:
            de_res = differential_evolution(
                de_objective, [(0.0, 1.0)] * n_par, args=(target_par,),
                maxiter=max_iter, popsize=pop_size, tol=1e-12,
                mutation=(0.4, 1.0), recombination=0.7,
                seed=seed, polish=False, updating='immediate',
                init='latinhypercube'
            )
            if de_res.fun < best_c5:
                cur = project_h(de_res.x, target_par)
                cur_c5 = compute_c5(cur)
                if cur_c5 < best_c5:
                    best_h, best_c5, best_n = cur.copy(), cur_c5, n_par
        except Exception:
            pass

    # Aggressive local refinement at parent resolution
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        cur = do_slsqp(best_h, n_par, target_par, 20000, 8000, 1e-15)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur.copy(), cur_c5, n_par

    rem = budget_s - (time.time() - t0)
    if rem > 40:
        best_h, best_c5 = nesterov_subgrad(best_h, n_par, target_par, 6000, 0.025, t0, budget_s)
        best_n = n_par

    rem = budget_s - (time.time() - t0)
    if rem > 30:
        best_h, best_c5 = pair_descent(best_h, target_par, n_par, 25, t0, budget_s, seed=seed + 1)
        best_n = n_par

    rem = budget_s - (time.time() - t0)
    if rem > 25:
        best_h, best_c5 = coord_descent(best_h, target_par, n_par, 25, t0, budget_s)
        best_n = n_par

    rem = budget_s - (time.time() - t0)
    if rem > 20:
        best_h = do_slsqp(best_h, n_par, target_par, 30000, 10000, 1e-15)
        best_c5 = compute_c5(best_h)

    # Try other resolutions
    for n_pts in [256, 320, 448]:
        rem = budget_s - (time.time() - t0)
        if rem < 120:
            break
        if n_pts == n_par:
            continue
        target = n_pts / 2.0
        cur = project_h(resample(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)

        # Quick DE at this resolution
        rem = budget_s - (time.time() - t0)
        if rem > 60:
            pop_size = max(20, min(50, int(60000 / n_pts)))
            max_iter = max(30, int((rem - 40) * 1.5))
            try:
                de_res = differential_evolution(
                    de_objective, [(0.0, 1.0)] * n_pts, args=(target,),
                    maxiter=max_iter, popsize=pop_size, tol=1e-12,
                    mutation=(0.4, 1.0), recombination=0.7,
                    seed=seed + n_pts, polish=False, updating='immediate',
                    init='latinhypercube'
                )
                if de_res.fun < cur_c5:
                    cur = project_h(de_res.x, target)
                    cur_c5 = compute_c5(cur)
            except Exception:
                pass

        # SLSQP refinement
        rem = budget_s - (time.time() - t0)
        if rem > 30:
            cur = do_slsqp(cur, n_pts, target, 15000, 5000, 1e-15)
            cur_c5 = compute_c5(cur)

        # Nesterov
        rem = budget_s - (time.time() - t0)
        if rem > 25:
            cur, cur_c5 = nesterov_subgrad(cur, n_pts, target, 4000, 0.02, t0, budget_s)

        # Pair + coord descent
        rem = budget_s - (time.time() - t0)
        if rem > 20:
            cur, cur_c5 = pair_descent(cur, target, n_pts, 15, t0, budget_s, seed=seed + n_pts + 100)

        rem = budget_s - (time.time() - t0)
        if rem > 15:
            cur, cur_c5 = coord_descent(cur, target, n_pts, 15, t0, budget_s)

        rem = budget_s - (time.time() - t0)
        if rem > 15:
            cur = do_slsqp(cur, n_pts, target, 20000, 15000, 1e-15)
            cur_c5 = compute_c5(cur)

        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

    # Final polish at best resolution
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        nf = len(best_h)
        tf = nf / 2.0
        best_h = do_slsqp(best_h, nf, tf, 50000, 20000, 1e-15)
        best_c5 = compute_c5(best_h)
        rem = budget_s - (time.time() - t0)
        if rem > 10:
            best_h, best_c5 = pair_descent(best_h, tf, nf, 15, t0, budget_s, seed=777)
        rem = budget_s - (time.time() - t0)
        if rem > 10:
            best_h, best_c5 = coord_descent(best_h, tf, nf, 15, t0, budget_s)

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
