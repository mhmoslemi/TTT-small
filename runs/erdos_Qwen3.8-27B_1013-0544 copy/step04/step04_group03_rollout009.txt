def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]

    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=8000, ftol=1e-16):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def lag_grad(h, lag, n):
    """Gradient of correlation at a specific lag w.r.t. h."""
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
    else:
        la = -lag
        grad[la:] += (1.0 - h[:n - la])
        grad[:n - la] -= h[la:]
    return grad

def subgrad_desc(h, target, n, n_iter, step_init, t0, budget, top_k=16, momentum=0.85):
    cur = h.copy()
    mom = np.zeros(n)
    best_local = compute_c5(cur)
    stall = 0
    step = step_init
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            grad += lag_grad(cur, lag, n)
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = momentum * mom - step * grad

            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=8000, ftol=1e-16):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def lag_grad(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
    else:
        la = -lag
        grad[la:] += (1.0 - h[:n - la])
        grad[:n - la] -= h[la:]
    return grad

def subgrad_desc(h, target, n, n_iter, step_init, t0, budget, top_k=12, momentum=0.85):
    cur = h.copy()
    mom = np.zeros(n)
    best_local = compute_c5(cur)
    stall = 0
    step = step_init
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            grad += lag_grad(cur, lag, n)
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = momentum * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            stall = 0
            step = min(step * 1.02, 0.06)
        else:
            stall += 1
            step *= 0.72
            mom *= 0.3
        if stall > 250:
            break
    return cur, best_local

def lag_focus_opt(h, target, n, max_iters, t0, budget, top_k=5):
    """Focus on reducing the top lags specifically."""
    cur = h.copy()
    best_c5 = compute_c5(cur)
    step = 0.01
    for it in range(max_iters):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        # Compute weighted gradient for top lags
        weights = np.zeros(len(corr))
        # Emphasize the top lags
        for i, idx in enumerate(top_idx):
            weights[idx] = (len(top_idx) - i + 1)
        total_w = weights.sum()
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            grad += weights[idx] * lag_grad(cur, lag, n)
        grad /= (total_w * np.linalg.norm(grad) + 1e-15)
        # Backtracking line search
        alpha = step
        improved = False
        for _ in range(10):
            trial = project_h(cur - alpha * grad, target)
            c_try = compute_c5(trial)
            if c_try < best_c5 - 1e-15:
                cur = trial
                best_c5 = c_try
                improved = True
                step = min(alpha * 1.5, 0.05)
                break
            alpha *= 0.5
        if not improved:
            step *= 0.7
            if step < 1e-5:
                break
    return cur, best_c5

def fine_coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    steps = [0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in steps:
                val = cur[j] + d
                if val < 0 or val > 1:
                    continue
                trial = cur.copy()
                trial[j] = val
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-15:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

def anneal_binary(h_bin, n, t0, budget, n_iter=60000):
    h = h_bin.copy()
    cur_max = float(np.max(compute_lags(h)))
    best_h = h.copy()
    best_max = cur_max
    rng = np.random.default_rng(888)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iter):
        if budget - (time.time() - t0) < 5:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        new_max = float(np.max(compute_lags(h)))
        if new_max < cur_max:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            frac = it / max(n_iter, 1)
            temp = 0.4 * (0.003) ** frac
            if rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-15)):
                cur_max = new_max
            else:
                h[a] = 1.0
                h[b] = 0.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
    return best_h, best_max

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 384
        h_parent = np.full(n_init, 0.5)
        h_parent[:n_init // 2] = 0.7
        h_parent[n_init // 2:] = 0.3
        h_parent = project_h(h_parent, n_init / 2.0)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)

    # Phase 1: Binary snap + anneal
    n_p = best_n
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        h_bin = binary_snap(best_h, n_p)
        h_bin, _ = anneal_binary(h_bin, n_p, t0, budget_s, n_iter=100000)
        cur = h_bin + 0.04 * np.random.randn(n_p)
        cur = project_h(cur, n_p / 2.0)
        for beta in [400, 1500]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            cur = do_slsqp(cur, n_p, n_p / 2.0, beta, maxiter=4000)
            c_try = compute_c5(cur)
            if c_try < best_c5:
                best_h = cur.copy()
                best_c5 = c_try

    # Phase 2: SLSQP with increasing sharpness
    n_cur = best_n
    target = n_cur / 2.0
    for beta in [800, 3000, 10000, 30000]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 12:
            break
        best_h = do_slsqp(best_h, n_cur, target, beta, maxiter=10000)
        best_c5 = compute_c5(best_h)

    # Phase 3: Subgradient descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        best_h, best_c5 = subgrad_desc(best_h, target, n_cur, 12000, 0.012, t0, budget_s, top_k=10)

    # Phase 4: Lag-focused optimization
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        best_h, best_c5 = lag_focus_opt(best_h, target, n_cur, 8000, t0, budget_s, top_k=6)

    # Phase 5: SLSQP refinement
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        for beta in [10000, 40000, 80000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            best_h = do_slsqp(best_h, n_cur, target, beta, maxiter=8000)
            best_c5 = compute_c5(best_h)

    # Phase 6: Higher resolution
    remaining = budget_s - (time.time() - t0)
    for n_hr in [480, 576]:
        if remaining < 80:
            break
        target_hr = n_hr / 2.0
        cur_hr = project_h(resample_fn(best_h, n_hr), target_hr)
        for beta in [500, 2000, 8000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            cur_hr = do_slsqp(cur_hr, n_hr, target_hr, beta, maxiter=5000)
            c_try = compute_c5(cur_hr)
            if c_try < best_c5:
                best_h = cur_hr.copy()
                best_c5 = c_try
                best_n = n_hr
                n_cur = n_hr
                target = target_hr
        # Quick subgradient at new resolution
        remaining = budget_s - (time.time() - t0)
        if remaining > 20:
            best_h, best_c5 = subgrad_desc(best_h, target, n_cur, 4000, 0.01, t0, budget_s, top_k=8)

    # Phase 7: Multi-start
    remaining = budget_s - (time.time() - t0)
    rng = np.random.default_rng(seed + 300)
    for att in range(6):
        if remaining < 20:
            break
        scales = [0.03, 0.015, 0.04, 0.01, 0.025, 0.02]
        hp = best_h + rng.standard_normal(n_cur) * scales[att]
        hp = project_h(hp, target)
        hp = do_slsqp(hp, n_cur, target, 3000, maxiter=5000)
        hp, _ = subgrad_desc(hp, target, n_cur, 3000, 0.008, t0, budget_s, top_k=8)
        hp, _ = lag_focus_opt(hp, target, n_cur, 2000, t0, budget_s, top_k=5)
        c_try = compute_c5(hp)
        if c_try < best_c5:
            best_h = hp.copy()
            best_c5 = c_try

    # Phase 8: Final fine coordinate descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        best_h, best_c5 = fine_coord_descent(best_h, target, n_cur, 80, t0, budget_s)

    # Phase 9: Final SLSQP
    remaining = budget_s - (time.time() - t0)
    if remaining > 8:
        for beta in [50000, 150000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 5:
                break
            best_h = do_slsqp(best_h, n_cur, target, beta, maxiter=12000, ftol=1e-18)
            best_c5 = compute_c5(best_h)

    # Final polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 3:
        best_h, best_c5 = fine_coord_descent(best_h, target, n_cur, 60, t0, budget_s)

    return best_h, best_c5, best_n
