def lag_directed_swap(h_bin, n, t0, budget, n_rounds=100, seed=42):
    """Targeted swap optimization focusing on worst lags."""
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    
    for rnd in range(n_rounds):
        if budget - (time.time() - t0) < 5:
            break
        
        lags = compute_lags(h)
        worst_lag = int(np.argmax(lags))
        lag = worst_lag - (n - 1)  # actual lag value
        
        # Find positions contributing to worst lag
        # For lag >= 0: contribution at position i is h[i] * (1-h[i+lag]) for i < n-lag
        # For lag < 0: contribution at position i+|lag| is h[i+|lag|] * (1-h[i]) for i < n-|lag|
        
        if lag >= 0:
            # h[i] * (1-h[i+lag]) for i in [0, n-lag)
            contrib_idx = np.arange(n - lag)
            contributors = contrib_idx[(h[contrib_idx] == 1) & (h[contrib_idx + lag] == 0)]
        else:
            la = -lag
            # h[i+la] * (1-h[i]) for i in [la, n)
            contrib_idx = np.arange(la, n)
            contributors = contrib_idx[(h[contrib_idx] == 1) & (h[contrib_idx - la] == 0)]
        
        if len(contributors) == 0:
            continue
        
        # Try flipping some contributors
        for _ in range(min(20, len(contributors))):
            if budget - (time.time() - t0) < 3:
                break
            a = contributors[rng.integers(len(contributors))]
            # Find a zero to swap with (prefer zeros that don't increase other lags)
            zeros = np.where(h == 0)[0]
            # Try a few random zeros
            for _ in range(5):
                b = zeros[rng.integers(len(zeros))]
                h[a] = 0.0
                h[b] = 1.0
                new_c5 = compute_c5(h)
                if new_c5 < cur_c5:
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    break
                h[a] = 1.0
                h[b] = 0.0
    
    return best_h, best_c5

So the problem is equivalent to: find a subset A of Z_n (or a subset of [0,2] discretized) of size n/2 that maximizes the minimum autocorrelation.

This is related to "optimal binary sequences" or "maximum autocorrelation" problems!

For binary sequences of length n with n/2 ones, the maximum possible minimum autocorrelation is bounded by various results. The connection to Welch bounds and related combinatorial designs might give us good initial constructions.

One good construction is based on **Barker sequences** or **m-sequences**, but those typically have different lengths.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def obj_fn(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_eq(x, target):
    return np.sum(x) - target

def sum_jac(x, target):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, maxiter=5000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(obj_fn, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(r.x, target)
    except:
        return h0

def binary_snap(h):
    n = len(h)
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def swap_anneal_binary(h_bin, n, t0, budget, n_iter=100000, seed=123, 
                        directed_frac=0.3):
    """Simulated annealing with mix of random and directed swaps."""
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        
        # Decide move type
        if rng.random() < directed_frac:
            # Directed: target worst lag
            lags = compute_lags(h)
            worst = int(np.argmax(lags))
            lag = worst - (n - 1)
            
            if lag >= 0 and lag < n:
                # c = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
                # Want to find i where h[i+lag]=1 and h[i]=0
                ones_at = np.where(h[lag:] == 1)[0] + lag
                zeros_at = np.where(h[:n-lag] == 0)[0]
                if len(ones_at) > 0 and len(zeros_at) > 0:
                    a = ones_at[rng.integers(len(ones_at))]
                    b = zeros_at[rng.integers(len(zeros_at))]
                else:
                    a = h[rnp.integers(n)]  # fallback random
                    b = (a + 1) % n
            else:
                a = rng.integers(n)
                b = rng.integers(n)
        else:
            # Random swap
            a = rng.integers(n)
            b = rng.integers(n)
            
        if h[a] == h[b]:
            continue
        if h[a] == 0:
            a, b = b, a  # ensure a is 1, b is 0
        
        h[a] = 0.0
        h[b] = 1.0
        new_c5 = compute_c5(h)
        
        if new_c5 <= cur_c5 + 1e-15:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            frac = it / max(n_iter, 1)
            temp = 0.003 * (0.01) ** frac
            if rng.random() < np.exp(-(new_c5 - cur_c5) / max(temp, 1e-18)):
                cur_c5 = new_c5
            else:
                h[a] = 1.0
                h[b] = 0.0
    
    return best_h, best_c5

**Coordinate descent for this problem**:
For each coordinate j, the optimal h[j] (given all other h[i] fixed) minimizes:
max_k c[k](h[j]) where c[k] is linear in h[j].

The minimum of a convex piecewise-linear function over [0,1] is at a kink point or at a boundary. The kink points are where two c[k] values are equal.

For efficiency, I can:
1. Compute the current c[k] for all k
2. Identify which k are "active" (near the maximum)
3. For the active k that depend on h[j], find the h[j] where they're equal
4. Evaluate the objective at those points plus boundaries

from scipy.optimize import minimize, Bounds
import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_eq(x, target):
    return np.sum(x) - target

def sum_jac(x, target):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, maxiter=5000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(obj_fn, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(r.x, target)
    except:
        return h0

    n = len(h)
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def swap_anneal(h_bin, n, t0, budget, n_iter, seed, l2_phase=0.2):
    """Two-phase annealing: first L2, then L∞."""
    h = h_bin.copy()
    rng = np.random.default_rng(seed)
    
    # Phase 1: L2 objective (minimize sum of squares of lags)
    n_l2 = int(n_iter * l2_phase)
    lags = compute_lags(h)
    cur_l2 = float(np.sum(lags**2))
    best_l2 = cur_l2
    best_h_l2 = h.copy()
    
    temp_start = 0.001
    for it in range(n_l2):
        if budget - (time.time() - t0) < 5:
            break
        a = rng.integers(n)
        b = rng.integers(n)
        if h[a] == h[b]:
            continue
        if h[a] == 0:
            a, b = b, a
        h[a] = 0.0
        h[b] = 1.0
        lags = compute_lags(h)
        new_l2 = float(np.sum(lags**2))
        if new_l2 <= cur_l2:
            cur_l2 = new_l2
            if new_l2 < best_l2:
                best_l2 = new_l2
                best_h_l2 = h.copy()
        else:
            frac = it / max(n_l2, 1)
            temp = temp_start * (0.1) ** frac
            if rng.random() < np.exp(-(new_l2 - cur_l2) / max(temp, 1e-18)):
                cur_l2 = new_l2
            else:
                h[a] = 1.0
                h[b] = 0.0
    
    h = best_h_l2.copy()
    
    # Phase 2: L∞ objective (minimize max lag)
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
