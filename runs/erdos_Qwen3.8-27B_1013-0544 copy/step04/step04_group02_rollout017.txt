import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def active_set_grad(h, n, top_k=10, temp=8.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    cmax = corr[top_idx[-1]]
    w = np.exp(temp * (corr[top_idx] - cmax))
    w /= w.sum()
    grad = np.zeros(n)
    for j, k in enumerate(top_idx):
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                idx = np.arange(m)
                grad[idx] += w[j] * (1.0 - h[idx + lag])
                grad[idx + lag] -= w[j] * h[idx]
        else:
            la = -lag
            m = n - la
            if m > 0:
                idx = np.arange(m)
                grad[idx + la] += w[j] * (1.0 - h[idx])
                grad[idx] -= w[j] * h[idx + la]
    return grad

    momentum = 0.45

        if time.time() > deadline:
            break
        y = h + momentum * velocity
        y = np.clip(y, 0.0, 1.0)

        alpha = lr

                velocity = momentum * velocity - alpha * grad
                h = candidate

            lr *= 0.65

    return best_h, best_c5

def frank_wolfe_optimize(h, n, deadline, seed, top_k=10):
    ts = n / 2.0
    h = project_feasible(h.copy(), ts)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta
