        g[lag:] -= h[:n-lag]
    else:

        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]

        g += w[j] * corr_gradient(h, j, n)

    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

            if budget_s - (time.time() - t0) < 25:
                break
            h_new = pg_solve(h_interp, target_try, beta, t0, budget_s, max_time=20)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                n = n_try
                target = target_try
            h_interp = h_new
        
        best_h, best_c5 = subgrad_polish(best_h, target, t0, budget_s, max_time=30)

    if budget_s - (time.time() - t0) > 20:
        best_h, best_c5 = subgrad_polish(best_h, target, t0, budget_s, max_time=0)
    
    return best_h, compute_c5(best_h), len(best_h)

Let me also think about a completely different idea: **using the fact that the optimal h might be close to a 0-1 function**. If we can find a good 0-1 pattern, we can then relax it slightly.

For the binary case, the problem is:
min max_k #{i : h[i]=1, h[i+k]=0}

subject to sum(h) = n/2, h[i] ∈ {0,1}.

This is equivalent to:
min max_k (n/2 - #{i : h[i]=1, h[i+k]=1})
= n/2 - max min_k #{i : h[i]=1, h[i+k]=1}

So we want to maximize the minimum "autocorrelation" of the binary sequence.

A greedy approach: start with a random binary sequence of weight n/2, then greedily flip bits to improve the minimum autocorrelation.

Let me implement this as an additional phase:

```python
def binary_optimize(h_init, target, t0, budget, max_time=30):
    """Optimize a binary (or near-binary) sequence to minimize max correlation."""
    n = len(h_init)
    # Threshold to binary
    h_bin = (h_init > 0.5).astype(float)
    # Fix sum
    while h_bin.sum() > target:
        # Flip a 1 to 0 that improves things most
        best_i = -1
        best_c5 = compute_c5(h_bin)
        for i in np.where(h_bin == 1)[0]:
            trial = h_bin.copy()
            trial[i] = 0
            c5 = compute_c5(trial)
            if c5 < best_c5:
                best_c5 = c5
                best_i = i
        if best_i < 0:
            break
        h_bin[best_i] = 0
    
    while h_bin.sum() < target:
        best_i = -1
        best_c5 = compute_c5(h_bin)
        for i in np.where(h_bin == 0)[0]:
            trial = h_bin.copy()
            trial[i] = 1
            c5 = compute_c5(trial)
            if c5 < best_c5:
                best_c5 = c5
                best_i = i
        if best_i < 0:
            break
        h_bin[best_i] = 1
    
    # Now do coordinate descent: try flipping each bit
    c5 = compute_c5(h_bin)
    no_improve = 0
    order = np.random.permutation(n)
    
    for epoch in range(100):
        if time.time() - t0 > budget - max_time:
            break
        improved = False
        for i in order:
            trial = h_bin.copy()
            trial[i] = 1.0 - trial[i]
            if trial.sum() != target:
                continue
            c5_trial = compute_c5(trial)
            if c5_trial < c5 - 1e-12:
                h_bin = trial
                c5 = c5_trial
                improved = True
        if not improved:
            break
    
    return h_bin, c5
```

This is O(n^2) per epoch (for the sum fix) and O(n * n) per epoch (for the coordinate descent, since each flip requires recomputing the correlation). For n=480, that's about 480 * 960 = 460K operations per flip, and 480 flips per epoch, so about 220M per epoch. That might be too slow.

Let me optimize: instead of recomputing the full correlation after each flip, I can compute the change in correlation analytically.

When we flip h[i] from 0 to 1 (or 1 to 0), how does each correlation c(k) change?

If h[i] changes by δ (δ = +1 for 0→1, δ = -1 for 1→0):
- c(k) changes by: δ * (1 - h[i+k]) + δ * (1 - h[i-k]) for appropriate k
  Wait, let me be more careful.
  
c(k) = Σ_j h[j] * (1 - h[j+k])

If h[i] changes by δ:
- In the term where j = i: contributes δ * (1 - h[i+k])
- In the term where j+k = i, i.e., j = i-k: contributes h[i-k] * (-δ) = -δ * h[i-k]

So Δc(k) = δ * (1 - h[i+k]) - δ * h[i-k] = δ * (1 - h[i+k] - h[i-k])

(with the understanding that h[j] = 0 for j outside [0, n-1].)

This means I can compute the effect of a flip in O(n) time (for all k), making the total per-epoch cost O(n^2).

Let me implement this:

```python
def flip_effect(h, i, n):
    """Compute the change in correlation vector if h[i] is flipped."""
    # δ = 1 - 2*h[i] (flipping changes h[i] by δ)
    # Actually, flipping means h[i] goes from h[i] to 1-h[i]
    # So δ = (1-h[i]) - h[i] = 1 - 2*h[i]
    delta = 1.0 - 2.0 * h[i]
    # Δc(k) = delta * (1 - h[i+k]) - delta * h[i-k] for each k
    # In terms of the full correlation array (length 2n-1):
    # The full correlation at index j corresponds to lag k = j - (n-1)
    # Δc at lag k = delta * (1 - h[i+k]) - delta * h[i-k]
    #             = delta * (1 - h[i+k] - h[i-k])
    
    effect = np.zeros(2*n-1)
    # For each k from -(n-1) to (n-1):
    # Δc(k) = delta * (1 - h[i+k] - h[i-k]) where h[j]=0 if j out of range
    for k in range(-(n-1), n):
        ik = i + k
        ik2 = i - k
        val = 0.0
        if 0 <= ik < n:
            val += (1.0 - h[ik])
        if 0 <= ik2 < n:
            val -= h[ik2]
        effect[k + (n-1)] = delta * val
    return effect
```

But this is O(n) per flip, and we do n flips per epoch, so O(n^2) per epoch. For n=480, that's about 230K per epoch. If we do 100 epochs, that's 23M total. That's fast!
