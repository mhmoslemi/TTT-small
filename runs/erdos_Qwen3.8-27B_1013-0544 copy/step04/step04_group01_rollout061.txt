        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
    # Phase 3: Pair search
    if time.time() < deadline - 5:
        best_h, best_c = fine_pair_search(best_h, n, deadline - 3,
                                          [0.01, 0.005, 0.002, 0.001, 0.0005])
    # Phase 4: Final SLSQP
    if time.time() < deadline - 4:
        ht = slsqp_optimize(best_h, n, 5000000, maxiter=5000, ftol=1e-18)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    # Phase 5: Ultra-fine pair search
    if time.time() < deadline - 1:
        best_h, best_c = fine_pair_search(best_h, n, deadline - 0.3,
                                          [0.0005, 0.0002, 0.0001])
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc_main = (deadline - t0) * 0.50
        h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc_main, rng)
        best_h, best_c = h_opt, c_opt
        # Try other resolutions
        for n_try in [150, 250]:
            if time.time() > deadline - 45:
                break
            remaining = deadline - time.time()
            alloc = min(90, remaining * 0.22)
            if alloc < 25:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            h_opt2, c_opt2 = optimize_n(h_interp, n_try, time.time() + alloc, rng)
            if c_opt2 < best_c:
                best_c = c_opt2
                best_h = h_opt2
    # Random restarts
    if best_h is not None and time.time() < deadline - 35:
        n = len(best_h)
        remaining = deadline - time.time() - 12
        n_restarts = max(2, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 18:
                break
            scale = 0.008 + 0.06 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 18) * 0.35)
            if alloc < 12:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Final polish
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        best_h, best_c = fine_pair_search(best_h, n, deadline - 1,
                                          [0.0003, 0.0002, 0.0001, 0.00005])
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c, len(best_h)
