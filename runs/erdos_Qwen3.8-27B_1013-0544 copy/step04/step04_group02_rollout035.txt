import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_sweep(h, n, c_full, dx, degree=200):
    """One full coordinate descent sweep. Returns new h, c_full, c5."""
    target_sum = n / 2.0
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    
    for j in range(n):
        v = h[j]
        
        # Compute A vector
        A = np.zeros(2*n - 1)
        idx1 = (j - lags >= 0) & (j - lags < n)
        idx2 = (j + lags >= 0) & (j + lags < n)
        A[lags[idx1] + (n-1)] += (1.0 - h[j - lags[idx1]])
        A[lags[idx2] + (n-1)] -= h[j + lags[idx2]]
        A[n-1] = 0.0  # handle lag=0 separately
        
        # Grid search for best t
        ts = np.linspace(0, 1, degree)
        # For m != n-1: c[m] + (t-v)*A[m]
        # For m = n-1: c[n-1] + t - t^2 - v + v^2
        linear_part = c_full[None, :] + (ts[:, None] - v) * A[None, :]
        # Fix the lag=0 entry
        linear_part[:, n-1] = c_full[n-1] + ts - ts**2 - v + v**2
        g = linear_part.max(axis=1)
        
        best_i = g.argmin()
        best_t = ts[best_i]
        best_g = g[best_i]
        
        # Refine
        if 0 < best_i < degree - 1:
            lo, hi = ts[best_i-1], ts[best_i+1]
            for _ in range(30):
                mid = (lo + hi) / 2
                tp = ts[best_i]  # current best
                # Evaluate at mid
                lin = c_full + (mid - v) * A
                lin[n-1] = c_full[n-1] + mid - mid**2 - v + v**2
                gm = lin.max()
                if gm < best_g:
                    best_g = gm
                    best_t = mid
                    # Determine which half
                    lin_lo = c_full + (lo - v) * A
                    lin_lo[n-1] = c_full[n-1] + lo - lo**2 - v + v**2
                    if lin_lo.max() < best_g:
                        hi = mid
                    else:
                        lo = mid
                else:
                    lin_hi = c_full + (hi - v) * A
                    lin_hi[n-1] = c_full[n-1] + hi - hi**2 - v + v**2
                    if lin_hi.max() < best_g:
                        lo = mid
                    else:
                        hi = mid
        elif best_i == 0:
            best_t = 0.0
        else:
            best_t = 1.0
        
        # Update
        if abs(best_t - v) > 1e-14:
            c_full = c_full + (best_t - v) * A
            c_full[n-1] = c_full[n-1] + best_t - best_t**2 - v + v**2

    return h, c_full

def smooth_max(x, beta):
