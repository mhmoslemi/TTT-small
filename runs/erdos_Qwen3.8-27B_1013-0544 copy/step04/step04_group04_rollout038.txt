def coordinate_descent(h, n_sweeps=100, target_sum=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    for sweep in range(n_sweeps):
        # Compute full correlation
        corr = np.correlate(h, 1.0 - h, mode="full")
        
        for j in range(n):
            # For each lag, compute the change in corr if h[j] changes
            # delta_c[lag] = delta * ((1-h[j-lag]) - h[j+lag])
            # We need to find optimal x in [0,1] minimizing max(corr + (x-h[j])*slopes)
            
            # Compute slopes for each k
            # k corresponds to lag = k - (n-1)
            slopes = np.zeros(2*n-1)
            for lag in range(-(n-1), n):
                i1 = j - lag  # where h[j] appears as h[i+lag]
                i2 = j + lag  # where h[j] appears as h[i]
                slope = 0.0
                if 0 <= i1 < n:
                    slope += (1.0 - h[i1])
                if 0 <= i2 < n:
                    slope -= h[i2]
                slopes[lag + n - 1] = slope
            
            # Now minimize max(corr[k] + (x - h[j]) * slopes[k]) over x in [0,1]
            # Let d = x - h[j], so we minimize max(corr[k] + d * slopes[k]) over d in [h[j]-1, 1-h[j]]
            # Interchangeably, minimize max(a[k] + b[k]*x) where a[k] = corr[k] - h[j]*slopes[k], b[k] = slopes[k]
            
            a = corr - h[j] * slopes
            b = slopes
            
            # Bisection for optimal x in [0,1]
            lo, hi = 0.0, 1.0
            for _ in range(40):
                x = (lo + hi) / 2
                vals = a + b * x
                mx = np.max(vals)
                active = vals >= mx - 1e-10
                if active.sum() == 0:
                    active = np.argmax(vals)
                    active = np.zeros(2*n-1, dtype=bool)
                    active[np.argmax(vals)] = True
                bg = np.mean(b[active])
                if bg > 0:
                    hi = x
                else:
                    lo = x
            
            h[j] = (lo + hi) / 2
        
        # Project to sum constraint
        if target_sum is not None:
            h = project_h(h, target_sum)
    
    return h

def compute_slopes(h, j):
    n = len(h)
    # slopes[k] for k in [0, 2n-2] corresponds to lag = k - (n-1)
    # slopes[k] = (1-h[j-lag]) * [valid] - h[j+lag] * [valid]
    
    lags = np.arange(2*n-1) - (n-1)  # lags from -(n-1) to n-1
    
    # (1-h[j-lag]) where 0 ≤ j-lag < n
    idx1 = j - lags  # indices into h
    valid1 = (idx1 >= 0) & (idx1 < n)
    term1 = np.where(valid1, 1.0 - h[idx1[valid1]], 0.0)
    
    # h[j+lag] where 0 ≤ j+lag < n
    idx2 = j + lags
    valid2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.where(valid2, h[idx2[valid2]], 0.0)
    
    return term1 - term2
