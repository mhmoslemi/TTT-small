def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break

            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def lse_value_and_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    peak = shifted.max()
    exps = np.exp(shifted - peak)
    logsum = peak + np.log(exps.sum())
    val = m + logsum / alpha
    w = exps / exps.sum()
    return val, w

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_val_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    pk = sh.max()
    ex = np.exp(sh - pk)
    val = m + (pk + np.log(ex.sum())) / alpha
    w = ex / ex.sum()
    return val, w

def lse_gradient(h, n, alpha, penalty=0.0):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    _, w = lse_val_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-18:
            continue
        lo = max(0, n - 1 - k)
        hi = min(n, 2 * n - 1 - k)
        idx = np.arange(lo, hi) + k - n + 1
        grad[lo:hi] += wk * (1.0 - h[idx])
        j_lo = max(0, lo + k - n + 1)
        j_hi = min(n, hi + k - n + 1)
        if j_lo < j_hi:
            i_idx = np.arange(j_lo, j_hi) - k + n - 1
            grad[j_lo:j_hi] -= wk * h[i_idx]
    if penalty > 0:
        grad += 2.0 * penalty * (h.sum() - n / 2.0)
    return grad

def lse_objective(h, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    val, _ = lse_val_weights(c, alpha)
    if penalty > 0:
        val += penalty * (h.sum() - n / 2.0) ** 2
    return val

def lbfgs_round(h, n, alpha, penalty, maxiter=400):
    def obj(v):
        return lse_objective(v, n, alpha, penalty)
    def grad(v):
        return lse_gradient(v, n, alpha, penalty)
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'gtol': 1e-12})
        return res.x
    except Exception:
        return h.copy()

def gradient_local_search(h, n, deadline, n_top=12):
    best_c = compute_c5(h, n)
    alphas = [20000, 50000, 200000, 500000]
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005]
    alpha_idx = 0
    step_idx = 0
    stagnant = 0
    for _ in range(5000):
        if time.time() > deadline:
            break
        if stagnant > 15:
            if alpha_idx < len(alphas) - 1:
                alpha_idx += 1
            elif step_idx < len(steps) - 1:
                step_idx += 1
            stagnant = 0
        g = lse_gradient(h, n, alphas[alpha_idx])
        idx_sorted = np.argsort(g)
        improved = False
        d = steps[step_idx]
        for i in idx_sorted[:n_top]:
            for j in idx_sorted[-n_top:]:
                if i == j:
                    continue
                hi = h[i] + d
                hj = h[j] - d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h[i], h[j] = hi, hj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                        stagnant = 0
                    else:
                        h[i], h[j] = hi - d, hj + d
        if not improved:
            stagnant += 1
    return h, best_c

def exhaustive_near_pair(h, n, deadline, radius=40, deltas=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(50):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def optimize_full(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    alphas = [200, 1000, 3000, 10000, 30000, 100000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        for pen in [1.0, 50.0]:
            if time.time() > deadline - 5:
                break
            ht = lbfgs_round(best_h, n, a, pen, maxiter=300)
            ht = project_feasible(ht, n)
            c = compute_c5(ht, n)
            if c < best_c - 1e-15:
                best_c = c
                best_h = ht.copy()

    if time.time() < deadline - 10:
        best_h, best_c = gradient_local_search(best_h, n, deadline - 5, n_top=12)

    if time.time() < deadline - 5:
        ht = lbfgs_round(best_h, n, 200000, 200.0, maxiter=300)
        ht = project_feasible(ht, n)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    if time.time() < deadline - 3:
        best_h, best_c = exhaustive_near_pair(best_h, n, deadline - 1, radius=30)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def smooth_init(n, rng, seed_offset=0):
    x = np.linspace(0, 2, n)
    n_four = rng.randint(2, 6)
    h = np.full(n, 0.5)
    for m in range(n_four):
        freq = rng.randint(1, 4) * (m + 1)
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(np.pi * freq * x + phase)
    h = project_feasible(h, n)
    return h
