import numpy as np
from scipy.optimize import minimize
import time

_global_beta = 100.0
_global_target = 0.0

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_at_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:-lag])
        g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g

def lse_value_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    bc = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(bc)
    se = e.sum()
    val = (m + np.log(se / len(corr)) / beta) * dx
    w = e / se
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        if w[j] < 1e-12:
            continue
        lag = j - (n - 1)
        g += w[j] * grad_at_lag(h, lag)
    return val, g * dx

def _slsqp_obj(x):
    h = np.clip(x, 0, 1)
    return lse_value_grad(h, _global_beta)[0]

def _slsqp_grad(x):
    h = np.clip(x, 0, 1)
    return lse_value_grad(h, _global_beta)[1]

def _slsqp_eq(x):
    return np.sum(x) - _global_target

def _slsqp_eqj(x):
    return np.ones(len(x))

def do_slsqp(h_init, beta, maxiter=3000):
    global _global_beta, _global_target
    _global_beta = beta
    _global_target = len(h_init) / 2.0
    n = len(h_init)
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eqj}]
    res = minimize(_slsqp_obj, h_init, jac=_slsqp_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_sum(res.x, n / 2.0)

def subgrad_polish(h, n_active, max_iter, t0, budget_s):
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(best)
    no_improve = 0
    steps_coarse = [0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0004]
    steps_fine = [0.002, 0.001, 0.0004, 0.0001, 5e-5]
    for it in range(max_iter):
        if budget_s - (time.time() - t0) < 3:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        top_idx = np.argsort(corr)[::-1][:n_active]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            g += grad_at_lag(best, lag)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        steps = steps_fine if it > 500 else steps_coarse
        improved = False
        for step in steps:
            ht = project_sum(best - step * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                improved = True
                break

                    best = ht
                    best_c5 = c5t
                    no_improve = 0
            if no_improve > 4000:
                break
        else:
            no_improve = 0
    return best, best_c5

            ht = project_sum(best + pert, target)
            if compute_c5(ht) < best_c5:
                best = ht
                best_c5 = compute_c5(ht)
    return best, best_c5
