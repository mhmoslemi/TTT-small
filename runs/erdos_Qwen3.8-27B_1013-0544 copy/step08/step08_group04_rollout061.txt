def project_feasible(h, n):

    target = 0.5 * n
    diff = target - float(h.sum())
    if diff > 0.0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0.0:
            h = h + (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 0.0:
            h = h + (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)

import numpy as np

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-12:
        return h
    if diff > 0.0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 1e-12:
            h = h + (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 1e-12:
            h = h + (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-10:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-10:
        idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def resample_h(old, n):
    n0 = len(old)

        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

    idx = np.argpartition(h, n - k)[n - k:]
    h2 = np.zeros(n)
    h2[idx] = 1.0
    return h2

    """One pass of coordinate descent. Returns updated h, c, cur."""
    cur = float(c.max())
    for i in range(n):
        hp = float(h[i])
        lo = min(i, center - i)
        hi = max(center + i, 2 * center - i)
        seg_len = hi - lo + 1

        j = i + s
        k_idx = i - s

        vj = (j >= 0) & (j < n)
        vi = (k_idx >= 0) & (k_idx < n)
        if np.any(vj):
            a[vj] += 1.0 - h[j[vj]]
        if np.any(vi):
            a[vi] -= h[k_idx[vi]]
        
        off = center - lo

        if seg_len > 1:

            asg = np.empty(seg_len - 1)
            if off > 0:

                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center+1:hi+1]
                asg[off:] = a[off+1:]
            qconst = c[center] - hp * (1.0 - hp)
        else:
            cseg = np.empty(0)
            asg = np.empty(0)
            qconst = c[center] - hp * (1.0 - hp)
        
        unaff = 0.0
        if lo > 0:
            unaff = max(unaff, float(c[:lo].max()))
        if hi + 1 < L:
            unaff = max(unaff, float(c[hi+1:].max()))

        n_grid = 20
        xg = np.linspace(0.0, 1.0, n_grid)
        d = xg - hp
        if cseg.size > 0:

        else:
            vm = np.zeros(n_grid)
        qv = qconst + xg * (1.0 - xg)
        fv = np.maximum(np.maximum(vm, qv), unaff)
        idx_b = int(np.argmin(fv))
        
        # Local refinement
        l = max(0.0, xg[idx_b] - 2.0 * (1.0 / (n_grid - 1)))
        r = min(1.0, xg[idx_b] + 2.0 * (1.0 / (n_grid - 1)))
        xg2 = np.linspace(l, r, 16)
        d2 = xg2 - hp
        if cseg.size > 0:

        else:
            vm2 = np.zeros(16)
        qv2 = qconst + xg2 * (1.0 - xg2)
        fv2 = np.maximum(np.maximum(vm2, qv2), unaff)
        idx2 = int(np.argmin(fv2))
        
        if float(fv2[idx2]) < float(fv[idx_b]):
            best_x = float(xg2[idx2])
            best_f = float(fv2[idx2])
        else:
            best_x = float(xg[idx_b])
            best_f = float(fv[idx_b])
        
        if best_f < cur - 1e-14:
            dxv = best_x - hp
            if cseg.size > 0:
                if off > 0:
                    c[lo:center] += asg[:off] * dxv
                if hi > center:
                    c[center+1:hi+1] += asg[off:] * dxv
            c[center] = qconst + best_x * (1.0 - best_x)
            h[i] = best_x
            cur = best_f
    
    return h, c, cur

import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-13:
        return h
    if diff > 0.0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 1e-12:
            h = h + (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 1e-12:
            h = h + (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-10:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-10:
        idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def binary_from_h(h):
    n = len(h)
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    h2 = np.zeros(n)
    h2[idx] = 1.0
    return h2

def cd_optimize(h, n, max_passes, deadline, fine=40):
    h = project_feasible(h, n).copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur = float(c.max())
    best_h = h.copy()
    best_c = (2.0 / n) * cur
    center = n - 1
    L = 2 * n - 1
    stall = 0
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            hp = h[i]
            lo = min(i, center - i)
            hi = max(center + i, 2 * center - i)
            seg_len = hi - lo + 1
            s_arr = center - np.arange(lo, hi + 1, dtype=np.int32)
            jj = i + s_arr
            kk = i - s_arr
            a = np.zeros(seg_len)
            vj = (jj >= 0) & (jj < n)
            vi = (kk >= 0) & (kk < n)
            if np.any(vj):
                a[vj] += 1.0 - h[jj[vj]]
            if np.any(vi):
                a[vi] -= h[kk[vi]]
            off = center - lo
            nseg = seg_len - 1
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(c[hi + 1:].max()))
            
            if nseg > 0:
                cseg = np.empty(nseg)
                asg = np.empty(nseg)
                if off > 0:
                    cseg[:off] = c[lo:center]
                    asg[:off] = a[:off]
                if hi > center:
                    cseg[off:] = c[center + 1:hi + 1]
                    asg[off:] = a[off + 1:]
                # Coarse grid
                xg = np.linspace(0.0, 1.0, fine)
                d = xg - hp
                vm = np.max(cseg[None, :] + d[:, None] * asg[None, :], axis=1)
                qv = qconst + xg * (1.0 - xg)
                fv = np.maximum(np.maximum(vm, qv), unaff)
                ib = int(np.argmin(fv))
                # Fine local
                w = 2.0 / (fine - 1)
                ll = max(0.0, xg[ib] - 2 * w)
                rr = min(1.0, xg[ib] + 2 * w)
                xg2 = np.linspace(ll, rr, 24)
                d2 = xg2 - hp
                vm2 = np.max(cseg[None, :] + d2[:, None] * asg[None, :], axis=1)
                qv2 = qconst + xg2 * (1.0 - xg2)
                fv2 = np.maximum(np.maximum(vm2, qv2), unaff)

                    bx = float(xg2[i2])
                    bf = float(fv2[i2])
                else:
                    bx = float(xg[ib])
                    bf = float(fv[ib])
            else:
                bx = hp
                bf = max(unaff, qconst + hp * (1.0 - hp))

                    if off > 0:
                        c[lo:center] += asg[:off] * dxv
                    if hi > center:
                        c[center + 1:hi + 1] += asg[off:] * dxv
                c[center] = qconst + bx * (1.0 - bx)
                h[i] = bx
                cur = bf
                improved = True
        
        h = project_feasible(h, n)

            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall >= 8:
            break
    
    return best_h, best_c

def thue_morse_binary(n):
    h = np.zeros(n)
    for i in range(n):
        b = bin(i).count('1')
        if b % 2 == 0:
            h[i] = 1.0
    ones = int(h.sum())
    if ones > n // 2:
        idx = np.where(h == 1.0)[0]
        extra = ones - n // 2
        pick = idx[:extra]
        h[pick] = 0.0
    elif ones < n // 2:
        idx = np.where(h == 0.0)[0]
        need = n // 2 - ones
        pick = idx[:need]
        h[pick] = 1.0
    return h

def binary_local_search(h_bin, n, deadline, max_iters=50000):

            break

            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0

        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        if stall > 5000 and it > 5000:
            break
    
    return best_h, best_c

def fourier_start(n, rng, K=15):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 * np.ones(n)
    for k in range(1, K + 1):
        amp = rng.uniform(0.06, 0.22) / np.sqrt(k)
        ph = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(k * np.pi * (x - 1.0) + ph)
    return project_feasible(h, n)

def triangle_start(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(1.0 - np.abs(x - 1.0), n)
