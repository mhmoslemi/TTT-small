```python
def compute_subgrad(h, n, top_k=5):
    c = np.correlate(h, 1-h, mode='full')
    max_c = c.max()
    # Get lags with c close to max
    threshold = max_c - 1e-8
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    grad = np.zeros(n)
    for m in active:
        # ∂c[m]/∂h[j] for all j
        # First term: (1-h[j+m-(n-1)]) where 0 <= j+m-(n-1) < n
        shift1 = m - (n-1)
        if shift1 >= 0:
            # h[j+shift1] for j in [0, n-1-shift1]
            size = n - shift1
            grad[:size] += (1 - h[shift1:])
        else:
            # h[j+shift1] = h[j+n+shift1] for j in [-shift1, n-1]
            size = n + shift1
            grad[-shift1:] += (1 - h[:size])
        
        # Second term: -h[j-m+(n-1)] where 0 <= j-m+(n-1) < n
        shift2 = (n-1) - m
        if shift2 >= 0:
            size = n - shift2
            grad[:size] -= h[shift2:]
        else:
            size = n + shift2
            grad[-shift2:] -= h[:size]
    
    # Average over active set
    grad /= len(active)
    return grad, c.max()
```

```python
def lp_step(h, n, n_active=10, step_size=0.05):

    c_max = c.max()
    
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:n_active]
    inactive = sorted_idx[n_active:]
    
    G = build_grad_matrix(h, n)
    
    # LP: min t
    # s.t. G[active] @ delta - t <= -c[active]
    #      G[inactive] @ delta <= c_max - c[inactive]
    #      max(-h, -step_size) <= delta <= min(1-h, step_size)
    #      sum(delta) = 0
    
    # Using scipy linprog (minimize c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds)
    # Variables: [delta (n), t (1)] total n+1
    
    # Objective: minimize t => c_obj = [0,...,0, 1]
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0
    
    A_ub = []
    b_ub = []
    
    # Active constraints: G[m] @ delta - t <= -c[m]
    for m in active:
        row = np.zeros(n+1)
        row[:n] = G[m]
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(-c[m])

    for m in inactive:

        row[:n] = G[m]
        A_ub.append(row)
        b_ub.append(c_max - c[m])

    b_ub = np.array(b_ub)

    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    lo = np.maximum(-h, -step_size)
    hi = np.minimum(1.0 - h, step_size)
    bounds = list(zip(lo, hi))
    bounds.append((−np.inf, np.inf))  # t is unbounded
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_grad_matrix(h, n):
    G = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            G[m, k:] = (1.0 - h[:n - k])
            G[m, :n - k] -= h[k:]
        else:
            G[m, :n + k] = (1.0 - h[-k:])
            G[m, -k:] -= h[:n + k]
    return G

def lp_polish(h, n, max_iters=800, deadline=None, step_size=0.04, n_active=20):
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(c.max())
    stagnation = 0
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        c_max = float(c.max())
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:n_active]
        inactive = sorted_idx[n_active:]
        G = build_grad_matrix(h, n)
        A_rows = []
        b_vals = []
        for m in active:
            row = np.zeros(n + 1)
            row[:n] = G[m]
            row[-1] = -1.0
            A_rows.append(row)
            b_vals.append(-c[m])
        for m in inactive:
            row = np.zeros(n + 1)
            row[:n] = G[m]
            A_rows.append(row)
            b_vals.append(c_max - c[m])
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        lo = np.maximum(-h, -step_size)
        hi = np.minimum(1.0 - h, step_size)
        bounds = list(zip(lo, hi))
        bounds.append((-1e10, 1e10))
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
        except Exception:
            break
        if res.status != 0 or res.x is None:
            break
        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-12:
            break
        h_new = project_feasible(h + delta, n)
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        new_max = float(c_new.max())
        if new_max < current_max - 1e-12:
            h = h_new
            c = c_new
            current_max = new_max
            stagnation = 0
            step_size = min(0.12, step_size * 1.05)
        else:
            h_try = project_feasible(h + 0.5 * delta, n)
            c_try = np.correlate(h_try, 1.0 - h_try, mode='full')

                step_size *= 0.7

def coord_cd(h, n, max_passes=5000, deadline=None, grid_size=150):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        idx_order = range(n) if p % 2 == 0 else range(n - 1, -1, -1)
        for idx in idx_order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 > 1.0 - 1e-10:
                continue
            lo_d = max(0.0, -h0) if h0 > 1e-10 else 0.0
            hi_d = 1.0 - h0
            if hi_d - lo_d < 1e-12:
                continue
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            unaff_max = 0.0
            if m_lo > 0:
                tmp = float(c[:m_lo].max())
                if tmp > unaff_max:
                    unaff_max = tmp
            if m_hi < 2 * n - 2:
                tmp = float(c[m_hi + 1:].max())
                if tmp > unaff_max:
                    unaff_max = tmp
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c[ms][None, :] + d_grid[:, None] * a[None, :]
            if has_diag:
                nv[:, diag_idx] -= d_grid[:, None] ** 2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            best_i = int(np.argmin(vals))
            best_d = float(d_grid[best_i])
            best_val = float(vals[best_i])
            if 0 < best_i < grid_size - 1:
                y = vals[best_i - 1:best_i + 2]
                x = d_grid[best_i - 1:best_i + 2]
                denom = y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1])
                if abs(denom) > 1e-15:
                    xv = 0.5*(y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1]))/denom
                    if x[0] < xv < x[2]:
                        nv1 = c[ms] + xv * a
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag_idx] -= xv ** 2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val, best_d = v1, float(xv)
            if abs(best_d) > 1e-10 and best_val < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c[ms] + best_d * a
                if has_diag:
                    c[ms[diag_idx]] -= best_d ** 2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True
        if not improved:
            break
        if (p + 1) % 400 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

            h_a, c_a = lp_polish(h_int, n_try, max_iters=200, deadline=deadline - 30,
                                  step_size=0.03, n_active=15)
            h_a, c_a = coord_cd(h_a, n_try, max_passes=1000, deadline=deadline - 30,
                                 grid_size=100)
            if c_a < best_c:
                best_h, best_c = h_a, c_a

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
