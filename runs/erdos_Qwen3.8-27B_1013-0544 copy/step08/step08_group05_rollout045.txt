import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx

def smooth_obj(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]

def sum_con(x, n):
    return x.sum() - 0.5 * n

def sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)

def slsqp_polish(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n)
    best, bc = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a), jac=smooth_grad,
                          method="SLSQP", bounds=bounds, constraints=cons,
                          options={"maxiter": maxiter, "ftol": 1e-12})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-13:
            best, bc = nh, nc
    return best, bc

def coord_cd(h, n, max_passes=3, deadline=None, grid=32, refine=12):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    diag = n - 1
    A = np.zeros(2 * n - 1)
    xs0 = np.linspace(0.0, 1.0, grid)
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = float(x[p])
            A.fill(0.0)
            A[p:p + n] += om
            st = diag - p
            A[st:st + n] -= x
            vals = c[None, :] + (xs0 - old)[:, None] * A[None, :]
            vals[:, n - 1] -= (xs0 - old) ** 2
            rowmax = vals.max(axis=1)
            i = int(np.argmin(rowmax))
            bx = float(xs0[i])
            if bx != old and rowmax[i] < c.max() - 1e-12:
                sp = 1.0 / (grid - 1)
                lo, hi = max(0.0, bx - sp), min(1.0, bx + sp)
                xx = np.linspace(lo, hi, refine + 1)
                vv = c[None, :] + (xx - old)[:, None] * A[None, :]
                vv[:, n - 1] -= (xx - old) ** 2
                j = int(np.argmin(vv.max(axis=1)))
                bx = float(xx[j])
                d = bx - old

                    c[p:p + n] += d * om
                    c[st:st + n] -= d * x
                    c[diag] -= d * d
                    x[p] = bx
                    om[diag - p] = 1.0 - bx
                    improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc:
            best, bc = xp, cp
        if not improved:
            break
    return best, bc
