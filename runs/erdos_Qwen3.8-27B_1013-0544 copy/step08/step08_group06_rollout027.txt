    n2 = 2 * n - 1
    nmid = n - 1

    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        if p % 2 == 0:
            idx_range = range(n)
        else:
            idx_range = range(n - 1, -1, -1)

        for idx in idx_range:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-12:
                lo_d, hi_d = 0.0, 1.0
            elif h0 > 1.0 - 1e-12:
                continue
            else:
                lo_d, hi_d = -h0, 1.0 - h0
            if hi_d - lo_d < 1e-14:
                continue

            m_lo = max(0, min(n-1-idx, idx))
            m_hi = min(n2, max(n2-idx, idx+n-1))
            ms = np.arange(m_lo, m_hi + 1)
            M = len(ms)

            a = np.zeros(M)
            jh = idx + ms - nmid
            vh = (jh >= 0) & (jh < n)
            if vh.any():
                a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + nmid
            vg = (jg >= 0) & (jg < n)
            if vg.any():
                a[vg] -= h[jg[vg]]

            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < n2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))

            diag_mask = (ms == nmid)
            has_diag = diag_mask.any()

            # Coarse grid
            d_coarse = np.linspace(lo_d, hi_d, coarse_size)
            nv = c_aff[None, :] + d_coarse[:, None] * a[None, :]
            if has_diag:
                nv[:, diag_mask] -= d_coarse[:, None] ** 2
            vals_c = np.maximum(unaff_max, nv.max(axis=1))
            best_i = int(np.argmin(vals_c))

            # Fine grid near coarse optimum
            w = max(1, coarse_size // 8)
            i_lo = max(0, best_i - w)
            i_hi = min(coarse_size - 1, best_i + w)
            d_fine = np.linspace(d_coarse[i_lo], d_coarse[i_hi], fine_size)
            nv2 = c_aff[None, :] + d_fine[:, None] * a[None, :]
            if has_diag:
                nv2[:, diag_mask] -= d_fine[:, None] ** 2
            vals_f = np.maximum(unaff_max, nv2.max(axis=1))
            best_j = int(np.argmin(vals_f))
            best_d = float(d_fine[best_j])
            best_val = float(vals_f[best_j])

            # Also check boundaries
            for db in (lo_d, hi_d):
                nv_b = c_aff + db * a
                if has_diag:
                    nv_b = nv_b.copy()
                    nv_b[diag_mask] -= db * db
                vb = max(unaff_max, float(nv_b.max()))
                if vb < best_val:
                    best_val = vb
                    best_d = db

            if abs(best_d) > 1e-10 and best_val < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] += best_d * a
                if has_diag:
                    c[ms[diag_mask]] -= best_d * best_d
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True

        if not improved:
            break
        if (p + 1) % 400 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())

    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_quick(h, n, deadline):
    if time.time() > deadline:
        return h, compute_c5(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in [3000, 30000]:
        if time.time() > deadline:
            break
        def obj(x):
            cc = np.correlate(x, 1.0 - x, mode='full')
            m = cc.max()
            sh = alpha * (cc - m)
            mx = sh.max()
            lse = mx + np.log(np.sum(np.exp(sh - mx)))
            return (m + lse / alpha) * (2.0 / n)
        def con_fun(x):
            return x.sum() - n / 2.0
        def con_jac(x):
            return np.ones(n)
        try:
            res = minimize(obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                          options={'maxiter': 200, 'ftol': 1e-16})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def binary_push(h, n, threshold=0.5):
    """Push h values toward 0 or 1, then re-project."""
    h2 = h.copy()
    h2[h2 > threshold] = 1.0
    h2[h2 <= threshold] = 0.0
    h2 = project_feasible(h2, n)
    return h2

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(400, 0.5)
        n = 400

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Quick SLSQP
    td1 = min(deadline - 500, t0 + 40)
    h1, c1 = slsqp_quick(h, n, td1)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Main coordinate descent
    td2 = deadline - 250
    h2, c2 = coord_cd(best_h, n, max_passes=8000, deadline=td2,
                      coarse_size=20, fine_size=80)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Perturbation + re-optimize
    td3 = deadline - 80
    for attempt in range(6):
        if time.time() > td3:
            break
        eps = 0.03 + 0.02 * attempt
        h_p = best_h.copy()
        mask = rng.random(n) < 0.4
        h_p[mask] += rng.uniform(-eps, eps, int(mask.sum()))
        h_p = project_feasible(h_p, n)
        h_opt, c_opt = coord_cd(h_p, n, max_passes=1500,
                                deadline=min(td3, t0 + budget_s - 60 - attempt * 5),
                                coarse_size=20, fine_size=80)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 4: Try binary push variant
    if time.time() < td3 - 10:
        h_bin = binary_push(best_h, n)
        h_bo, c_bo = coord_cd(h_bin, n, max_passes=1500,
                              deadline=td3, coarse_size=20, fine_size=80)
        if c_bo < best_c:
            best_h, best_c = h_bo, c_bo

    # Phase 5: Final fine polish
    h5, c5f = coord_cd(best_h, n, max_passes=5000, deadline=deadline,
                       coarse_size=40, fine_size=150)
    if c5f < best_c:
        best_h, best_c = h5, c5f

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
