def compute_c5(h, n):

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(55):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    if abs(res) > 1e-13:
        mask = y < 1.0 if res > 0 else y > 0.0
        if mask.any():
            room = (1.0 - y) if res > 0 else y
            tot = room.sum()
            if tot > 1e-12:
                y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def _smooth_val(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    return (m + float(np.log(e.sum())) / alpha) * (2.0 / n)

def _smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    p = e / e.sum()
    g = np.zeros(n)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-14:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n-d])
            g[:n-d] -= w * x[d:]
        else:
            e2 = -d
            g[:n-e2] += w * (1.0 - x[e2:])
            g[e2:] -= w * x[:n-e2]
    return (2.0 / n) * g

                       jac=_smooth_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        x = np.asarray(res.x, dtype=float)
        if np.all(np.isfinite(x)):
            return project_feasible(x, n)
    except Exception:
        pass
    return h

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(55):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    if abs(res) > 1e-13:
        mask = y < 1.0 if res > 0 else y > 0.0
        if mask.any():
            room = (1.0 - y) if res > 0 else y
            tot = room.sum()
            if tot > 1e-12:
                y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def _smooth_val(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    return (m + float(np.log(e.sum())) / alpha) * (2.0 / n)

def _smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    p = e / e.sum()
    g = np.zeros(n)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-14:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n-d])
            g[:n-d] -= w * x[d:]
        else:
            e2 = -d
            g[:n-e2] += w * (1.0 - x[e2:])
            g[e2:] -= w * x[:n-e2]
    return (2.0 / n) * g

def _slsqp(h, n, alpha, maxiter, deadline):
    if time.time() > deadline:
        return h
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_val, h, args=(n, alpha), method='SLSQP',
                       jac=_smooth_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        x = np.asarray(res.x, dtype=float)
        if np.all(np.isfinite(x)):
            return project_feasible(x, n)
    except Exception:
        pass
    return h

def _compute_A(h, n):
    A = np.zeros((n, 2*n - 1))
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[d:, k] += 1.0 - h[:n-d]
            A[:n-d, k] -= h[d:]
        else:
            e = -d
            A[:n-e, k] += 1.0 - h[e:]
            A[e:, k] -= h[:n-e]
    return A

def _pair_search(h, n, max_tries, grid_n, deadline, rng):

        return h, compute_c5(h, n)
    h = h.copy()

    A = _compute_A(h, n)
    off = n - 1
    accepts = 0
    no_imp = 0
    for trial in range(max_tries):
        if time.time() > deadline:

        cur_max = float(c.max())

            if d >= 0:

                if m <= 0: continue

            else:

                m = n - e
                if m <= 0: continue

        if hi - lo <= 1e-14:

        lines[:, off + j - i] += us * us
