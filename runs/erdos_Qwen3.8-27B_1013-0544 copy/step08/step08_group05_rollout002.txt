def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(55):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    y = np.clip(y, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():

            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return y


def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)


def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx


def smooth_obj(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]


def smooth_grad(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]


def penalized_obj(x, n, alpha, pen):
    v, g = smooth_val_grad(x, n, alpha)
    s = x.sum() - 0.5 * n
    v += 0.5 * pen * s * s
    g = g + pen * s
    return v, g

def coord_eval(c, A, xs, old, n):
    d = xs - old
    cand = c[None, :] + d[:, None] * A[None, :]
    cand[:, n - 1] -= d * d
    return cand.max(axis=1)


def coord_cd(h, n, max_passes=5, deadline=None, grid=40, refine=12, tol=1e-12):

def binarize(h, n):
    if n % 2:
        return None
    target = n // 2
    if target <= 0 or target >= n:
        b = np.zeros(n, dtype=float)
        b[:target] = 1.0
        return b
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b


def rand_one_zero(h, n, rng):
    i = int(np.argmax(h))
    for _ in range(8):
        i = int(rng.randint(n))
        if h[i] >= 0.5:
            break
    j = int(np.argmin(h))
    for _ in range(8):
        j = int(rng.randint(n))
        if h[j] < 0.5:
            break
    if i == j:
        i = int(np.argmax(h))
        j = int(np.argmin(h))
    return i, j


def binary_trial(c, h, om, i, j, n):
    d = c.copy()
    d[i:i + n] -= om
    st = n - 1 - i
    d[st:st + n] += h
    d[j:j + n] += om
    st = n - 1 - j
    d[st:st + n] -= h
    d[n - 1] -= 2.0
    d[n - 1 + i - j] += 1.0
    d[n - 1 + j - i] += 1.0
    return d, float(d.max())

def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)


def make_structured(n, pattern_type, rng):
    h = np.zeros(n)
    if pattern_type == 0:
        h[: n // 2] = 1.0
    elif pattern_type == 1:
        for i in range(0, n, 4):
            h[i:i + 2] = 1.0
    elif pattern_type == 2:
        for i in range(0, n, 3):
            h[i] = 1.0
    elif pattern_type == 3:
        h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    else:
        phase = rng.uniform(0, 2 * np.pi)
        for i in range(n):
            h[i] = 1.0 if np.sin(2 * np.pi * 3 * i / n + phase) > 0 else 0.0
    return project_simplex(h, n)

                best_h, best_c = h1, c1

    # Phase 3: SLSQP polish
    t3 = min(deadline - 100.0, time.time() + 80.0)
    if time.time() < t3:
        h1, c1 = slsqp_polish(best_h, n_best, [50000, 500000, 5000000], 200, t3)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 4: Try different n values
    cands = []
    base = n_best
    for dy in [-100, -50, 50, 100, 150, -150, 200, -200]:
        nt = base + dy
        if 100 <= nt <= 1000 and nt % 2 == 0 and nt != base:
            cands.append(nt)
    for nt in [200, 300, 400, 600, 700, 800]:
        if nt not in cands and 100 <= nt <= 1000:
            cands.append(nt)
    cands = cands[:7]

    tried_n = [n_best]
    for nt in cands:
        if time.time() > deadline - 80.0:
            break
        if nt in tried_n:
            continue
        tried_n.append(nt)
        ht = transfer_h(best_h, n_best, nt)
        cv = compute_c5(ht, nt)
        chunk = min(deadline - 60.0, time.time() + 50.0)
        if chunk > time.time() + 5.0:
            h1, c1 = lbgfbs_smooth(ht, nt, [5000, 50000], chunk)
            if c1 < best_c:
                ht = h1
                cv = c1
            h1, c1 = coord_cd(ht, nt, max_passes=3, deadline=chunk, grid=32, refine=10)
            if c1 < best_c:
                ht = h1
                cv = c1
        if cv < best_c:
            best_h, best_c, n_best = ht, cv, int(nt)

    # Phase 5: Final polish
    t5 = min(deadline - 20.0, time.time() + 60.0)
    if time.time() < t5:
        h1, c1 = lbgfbs_smooth(best_h, n_best, [100000, 1000000], t5)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h1, c1 = coord_cd(best_h, n_best, max_passes=4, deadline=t5, grid=50, refine=16)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 6: Final binary attempt if time remains
    if n_best % 2 == 0 and time.time() < deadline - 10.0:
        t6 = deadline - 5.0
        hb, cb = binary_optimize(best_h, n_best, rng, t6)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb

    final_h = project_simplex(best_h, n_best)
    final_c = float(compute_c5(final_h, n_best))
    return final_h, final_c, int(n_best)
