import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    # Same as before - project to [0,1] with sum = n/2
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(60):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def lag_gradient(h, n, k):
    """Compute gradient of c[k] w.r.t. h for a specific lag k."""
    g = np.zeros(n, dtype=float)
    d = k - (n - 1)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:d+m] += (1.0 - h[:m])
            g[:m] -= h[d:d+m]
    else:
        e = -d
        m = n - e
        if m > 0:
            g[:m] += (1.0 - h[e:e+m])
            g[e:e+m] -= h[:m]
    return (2.0 / n) * g

def active_set_subgrad(h, n, lr, n_iter, top_k, deadline):
    """Projected subgradient method on top-k active lags."""
    h = h.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cm = float(c.max())
        # Find top-k lags
        thresh = np.sort(c)[-top_k][0] if top_k < len(c) else c.min()
        active = np.flatnonzero(c >= thresh)
        k = len(active)
        # Average gradient
        g = np.zeros(n, dtype=float)
        for ak in active:
            g += lag_gradient(h, n, ak)
        g /= k
        # Step
        h_new = project_feasible(h - lr * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < cm:
            h = h_new
            # Decrease lr slightly
            lr *= 0.999
        else:
            lr *= 0.95
    return h, compute_c5(h, n)

    B = np.zeros(2*n - 1, dtype=float)
    off = n - 1
    i_part = np.zeros(2*n - 1, dtype=float)
    j_part = np.zeros(2*n - 1, dtype=float)

        d = k - off
        if d >= 0:
            # ∂c[k]/∂h[i] = (1-h[i-d]) if d<=i<=n-1, minus h[i+d] if 0<=i<=n-1-d

import numpy as np

from scipy.optimize import minimize

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low = z.min() - 2.0
    high = z.max() + 2.0
    for _ in range(60):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

# Smooth max for SLSQP
def smooth_val(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    z = alpha * (c - m)
    zm = z.max()
    e = np.exp(z - zm)
    s = e.sum()
    return (m + np.log(s) / alpha) * (2.0 / n)

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    z = alpha * (c - m)
    zm = z.max()
    e = np.exp(z - zm)
    p = e / e.sum()
    # gradient
    g = np.zeros(n, dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-12:
            continue
        d = k - off
        if d >= 0:
            m2 = n - d
            g[d:d+m2] += w * (1.0 - x[:m2])
            g[:m2] -= w * x[d:d+m2]
        else:
            e2 = -d
            m2 = n - e2
            g[:m2] += w * (1.0 - x[e2:e2+m2])
            g[e2:e2+m2] -= x[:m2]
    return (2.0 / n) * g

def sum_con(x, n):
    return x.sum() - 0.5 * n

def sum_jac(x, n):
    return np.ones_like(x)

def slsqp_opt(h, n, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    plans = [(30, 40), (200, 35), (1000, 30), (5000, 25), (20000, 20), (100000, 15)]
    for alpha, mx in plans:
        if time.time() > deadline - 2.0:
            break
        try:
            res = minimize(smooth_val, best_h, args=(n, alpha), method='SLSQP',
                          jac=smooth_grad, bounds=bounds, constraints=cons,
                          options={'maxiter': mx, 'ftol': 1e-14, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best_h, best_c = cur, cc
        except Exception:
            pass
    return best_h, best_c
