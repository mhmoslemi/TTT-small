        hi = max(m_hi_f, m_hi_b)

        jb = p - ms + (n - 1)

        vb = (jb >= 0) & (jb < n)

            a[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            a[vb] -= h[jb[vb]]
        c_base = c[ms]
        dm = (ms == (n - 1))
        has_diag = np.any(dm)
        if has_diag:
            c_diag = c_base[np.where(dm)[0][0]]
            a[dm] = 0.0
        unaff = 0.0
        if lo > 0:

        if hi < 2 * n - 2:
            unaff = max(unaff, c[hi + 1:].max())
        neg = a < -1e-15
        pos = a > 1e-15

        Lhi = c.max()

                break

                    r2 = (1.0 + np.sqrt(disc)) * 0.5
                    if xl < r1 and xh > r2:
                        pass
                    elif xl >= r1 and xh <= r2:
                        ok = False

            Llo = Llo if ok else L
        L = Lhi
        xl = 0.0
        xh = 1.0

        xl = max(xl, 0.0)
        xh = min(xh, 1.0)

def cd_run(h, n, max_passes, deadline, rng):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        h, improved = cd_one_pass(h, n, order)
        h = project_feasible(h, n)
        if not improved:
            break
    return h

def perturb_swap(h, n, rng, k=3):
    """Swap k random pairs of coordinates."""
    h = h.copy()
    for _ in range(k):
        i, j = rng.choice(n, 2, replace=False)
        h[i], h[j] = h[j], h[i]
    return h

def perturb_sine(h, n, rng):
    """Add a sinusoidal perturbation."""
    m = rng.randint(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.02, 0.12)
    x = np.linspace(0, 2, n)
    h = h + amp * np.sin(np.pi * m * x + phase)
    return project_feasible(h, n)

def perturb_block(h, n, rng):
    """Add a rectangular perturbation."""
    a = rng.randint(0, n)
    w = rng.randint(5, n // 4)
    b = min(n, a + w)
    amp = rng.choice([-1, 1]) * rng.uniform(0.03, 0.15)
    h = h.copy()
    h[a:b] += amp
    return project_feasible(h, n)

def perturb_gauss(h, n, rng):
    """Add a Gaussian bump."""
    x0 = rng.uniform(0, n)
    sigma = rng.uniform(3, n // 6)
    amp = rng.choice([-1, 1]) * rng.uniform(0.03, 0.1)
    indices = np.arange(n)
    h = h + amp * np.exp(-((indices - x0) / sigma) ** 2)
    return project_feasible(h, n)

        if abs(x_new - h0) > 1e-12:
            h[p] = x_new
            improved = True
    return h, improved

def cd_run(h, n, max_passes, deadline, rng):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        h, improved = cd_one_pass(h, n, order)
        h = project_feasible(h, n)
        if not improved:
            break
    return h

def perturb_swap(h, n, rng, k=5):
    h = h.copy()
    for _ in range(k):
        i, j = rng.choice(n, 2, replace=False)
        h[i], h[j] = h[j], h[i]
    return h

def perturb_sine(h, n, rng):
    m = rng.randint(1, 6)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.03, 0.15)
    x = np.arange(n, dtype=float)
    h = h + amp * np.sin(np.pi * m * x / n + phase)
    return project_feasible(h, n)

def perturb_block(h, n, rng):
    a = rng.randint(0, max(1, n - 10))
    w = rng.randint(5, max(6, n // 5))
    b = min(n, a + w)
    amp = rng.choice([-1, 1]) * rng.uniform(0.04, 0.18)
    h = h.copy()
    h[a:b] += amp
    return project_feasible(h, n)

def perturb_gauss(h, n, rng):
    x0 = rng.uniform(2, n - 2)
    sigma = rng.uniform(3, max(4, n // 8))
    amp = rng.choice([-1, 1]) * rng.uniform(0.04, 0.12)
    idx = np.arange(n, dtype=float)
    h = h + amp * np.exp(-0.5 * ((idx - x0) / sigma) ** 2)
    return project_feasible(h, n)

def perturb_noise(h, n, rng):
    s = rng.uniform(0.02, 0.08)
    return project_feasible(h + rng.normal(0, s, n), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(700, 0.5)
        n = 700

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP warmup
    if time.time() < deadline - 30:
        h1, c1 = slsqp_run(best_h, n, [3000, 20000, 100000], 250)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: CD convergence
    h2 = cd_run(best_h, n, 300, deadline, rng)
    c2 = compute_c5(h2, n)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Perturbation + refinement cycles
    perts = [perturb_swap, perturb_sine, perturb_block, perturb_gauss, perturb_noise]
    att = 0
    while time.time() < deadline - 120:
        att += 1
        pf = perts[att % len(perts)]
        hp = pf(best_h, n, rng)
        hp, cp = slsqp_run(hp, n, [5000, 30000], 120)
        hp = cd_run(hp, n, 80, deadline, rng)
        cp = compute_c5(hp, n)
        if cp < best_c:
            best_h, best_c = hp, cp

    # Phase 4: Different n values
    for n_try in [500, 600, 800, 1000, 900]:
        if time.time() > deadline - 50:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        h_opt, c_opt = slsqp_run(h_int, n_try, [5000, 50000], 200)
        h_opt = cd_run(h_opt, n_try, 100, deadline, rng)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            n = n_try

    # Phase 5: More perturbation cycles on best
    if time.time() < deadline - 30:
        sub_deadline = deadline - 10
        for _ in range(50):
            if time.time() > sub_deadline:
                break
            pf = perts[rng.randint(len(perts))]
            hp = pf(best_h, len(best_h), rng)
            hp = cd_run(hp, len(best_h), 60, sub_deadline, rng)
            cp = compute_c5(hp, len(best_h))
            if cp < best_c:
                best_h, best_c = hp, cp

    # Phase 6: Final polish
    best_h = cd_run(best_h, len(best_h), 100, deadline, rng)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
