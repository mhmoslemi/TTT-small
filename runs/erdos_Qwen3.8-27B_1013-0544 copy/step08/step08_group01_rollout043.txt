def smooth_jac(x,n,temp):
    c = np.correlate(x, 1.0-x, mode='full')
    m = float(c.max())
    exps = np.exp((c-m)/temp)
    tot = exps.sum()
    if tot <= 0: return np.zeros(n)
    p = exps / tot
    g = np.zeros(n)
    dx=2.0/n
    idxs = np.flatnonzero(p > 1e-12)
    k0 = n-1
    for idx in idxs:
        d = int(idx - k0)
        w = float(p[idx])
        if d >= 0:
            length = n - d
            g[:length] += w * (1.0 - x[d:])
            g[d:] -= w * x[:length]
        else:
            q = -d
            length = n - q
            g[q:] += w * (1.0 - x[:length])
            g[:length] -= w * x[q:]
    return g * dx

import numpy as np

from scipy.optimize import minimize, golden

def c_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_feasible(u, n, exclude=None):
    h = np.clip(np.array(u, dtype=float, copy=True), 0.0, 1.0)
    S = 0.5 * n
    diff = S - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0.0:
            room = 1.0 - h
            if exclude is not None:
                room[exclude] = 0.0
            tot = room.sum()
            if tot <= 1e-15 and exclude is not None:
                room = 1.0 - h
                tot = room.sum()
            if tot > 1e-15:
                h += (diff / tot) * room
        else:
            room = h.copy()
            if exclude is not None:
                room[exclude] = 0.0
            tot = room.sum()
            if tot <= 1e-15 and exclude is not None:
                room = h
                tot = room.sum()
            if tot > 1e-15:
                h -= ((-diff) / tot) * room
    diff = S - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0.0:
            room = 1.0 - h
            if exclude is not None:
                room[exclude] = 0.0
            i = int(np.argmax(room))
            h[i] = min(1.0, h[i] + diff)
        else:
            room = h.copy()
            if exclude is not None:
                room[exclude] = 0.0
            i = int(np.argmax(room))
            h[i] = max(0.0, h[i] + diff)
    return h

def grad_from_weights(x, n, w, scale=1.0):
    g = np.zeros(n)
    idxs = np.flatnonzero(w > 1e-14)
    k0 = n - 1
    for idx in idxs:
        d = int(idx - k0)
        ww = float(w[idx])
        if d >= 0:
            length = n - d
            if length > 0:
                g[d:] += ww * (1.0 - x[:length])
                g[:length] -= ww * x[d:]
        else:
            q = -d
            length = n - q
            if length > 0:
                g[:length] += ww * (1.0 - x[q:])
                g[q:] -= ww * x[:length]
    return g * scale

def smooth_obj(x, n, tau):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    e = np.exp((c - m) / tau)
    return float((m + tau * np.log(max(float(e.sum()), 1e-300))) * (2.0 / n))

def smooth_jac(x, n, tau):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    e = np.exp((c - m) / tau)
    tot = float(e.sum())
    if tot <= 1e-300:
        return np.zeros(n)
    p = e / tot
    return grad_from_weights(x, n, p, 2.0 / n)

def sum_con(x, n):
    return x.sum() - 0.5 * n

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_smooth(h, n, taus, maxiter_per, deadline):
    h = project_feasible(h, n).copy()
    best = eval_c5(h)
    if n > 1200:
        return h, best
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    for tau in taus:
        if time.time() > deadline:
            break
        rem = max(1.0, deadline - time.time())
        mi = max(20, int(min(maxiter_per, rem * 2)))
        try:
            res = minimize(smooth_obj, h, args=(n, tau), method='SLSQP',
                           bounds=bounds, constraints=cons, jac=smooth_jac,
                           options={'maxiter': mi, 'ftol': 1e-18, 'iprint': 0})
            hx = project_feasible(res.x, n)
            cx = eval_c5(hx)
            if cx < best - 1e-14:
                h, best = hx, cx
        except Exception:
            pass
    return h, best

def coord_obj(z, ca, sa, un, c0, s0, h0):
    dz = z - h0
    if ca.size:
        v = float(np.max(ca + sa * dz))
    else:
        v = -1e30
    if un > v:
        v = un
    q0 = c0 + s0 * dz - dz * dz
    return max(v, q0)

def try_transforms(h, n, deadline):
    best = h.copy()
    bc = eval_c5(best)
    bases = (best, best[::-1].copy(), 1.0 - best, 1.0 - best[::-1])
    step = max(1, n // 80)
    for base in bases:
        if time.time() > deadline:
            break
        b = project_feasible(base, n)
        c = eval_c5(b)
        if c < bc:
            best, bc = b, c
        for s in range(0, n, step):
            if time.time() > deadline:
                break
            hr = np.roll(base, s)
            c = eval_c5(hr)
            if c < bc:
                best, bc = hr.copy(), c
    best = project_feasible(best, n)
    return best, eval_c5(best)

def resample_h(h, m):
    n = len(h)
    if m == n:

    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    return project_feasible(np.interp(x1, x0, h, left=h[0], right=h[-1]), m)

            best_c = bc
        n = len(h)

    if time.time() < main_end - 20:
        for tau in (10.0, 1.0, 0.1, 0.01, 0.001, 1e-4):
            if time.time() > main_end - 10:
                break
            h, bc = slsqp_smooth(h, n, (tau,), 200, main_end - 10)
            if bc < best_c:
                best_c = bc
            n = len(h)

    if time.time() < main_end:
        h, bc = coordinate_refine(h, n, main_end, grid=90, radius=0.05,
                                  max_passes=50000, seed=seed + 1)
        if bc < best_c:
            best_c = bc
        n = len(h)

    start_end = dl - 100.0
    if time.time() < start_end - 30:
        for sc in (0.01, 0.03, 0.08, 0.2):
            if time.time() > start_end - 50:
                break
            sub = min(start_end, time.time() + 80.0)
            hp = project_feasible(h + rng.normal(0.0, sc, n), n)
            h2, c2 = slsqp_smooth(hp, n, (1.0, 0.1, 0.01), 80, min(sub, time.time() + 30.0))
            if time.time() < sub - 10.0:
                h2, c2 = coordinate_refine(h2, n, sub, grid=48, max_passes=500,
                                           seed=seed + int(1000 * sc))
            if c2 < best_c:
                h, best_c = h2, c2
                n = len(h)

        n0 = n
        for fac in (0.5, 0.75, 1.25, 1.5):
            if time.time() > start_end - 30:
                break
            nt = int(round(n0 * fac))
            if nt < 80 or nt > 1000:
                continue
            sub = min(start_end, time.time() + 80.0)
            st = resample_h(h, nt)
            st = project_feasible(st + rng.normal(0.0, 0.02, nt), nt)
            h2, c2 = coordinate_refine(st, nt, sub, grid=40, max_passes=1000, seed=seed + nt)
            if time.time() < sub - 15.0:
                h2, c2 = slsqp_smooth(h2, nt, (1.0, 0.1), 80, sub)
            if c2 < best_c:
                h, best_c, n = h2, c2, nt

    if time.time() < dl - 15:
        h, bc = coordinate_refine(h, n, dl - 5.0, grid=120, radius=0.01,
                                  max_passes=50000, seed=seed)
        if bc < best_c:
            best_c = bc
        n = len(h)

    if time.time() < dl - 5:
        h, bc = slsqp_smooth(h, n, (1e-3, 1e-4), 150, dl - 1.0)
        if bc < best_c:
            best_c = bc
        n = len(h)

    h = project_feasible(h, n)
    best_c = eval_c5(h)
    return h, best_c, n
