    bc = compute_c5(best, n)

    xs = np.linspace(0.0, 1.0, grid)

            break

            old = x[p]
            A.fill(0.0)
            A[p:p+n] += om
            st = diag - p
            A[st:st+n] -= x
            cur = c.max()
            # Evaluate max over all candidate values
            d = xs - old
            cand = c[None,:] + d[:,None] * A[None,:]
            cand[:,diag] -= d*d

    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:

        if mask.any():

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def coord_cd(h, n, max_passes, deadline, grid=128):
    best = project_simplex(h, n).copy()
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    A = np.zeros(L)
    xs = np.linspace(0.0, 1.0, grid)
    tol = 1e-12
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = x[p]
            A.fill(0.0)
            A[p:p+n] += om
            st = diag - p
            A[st:st+n] -= x
            cur = c.max()
            d = xs - old
            cand = c[None,:] + d[:,None] * A[None,:]
            cand[:,diag] -= d*d
            vals = cand.max(axis=1)
            i = vals.argmin()
            bx, bv = xs[i], vals[i]
            if bv < cur - tol:
                # Refine around bx
                sp = 1.0/(grid-1)
                lo_r = max(0.0, bx - sp)
                hi_r = min(1.0, bx + sp)
                xx = np.linspace(lo_r, hi_r, 21)
                d2 = xx - old
                cand2 = c[None,:] + d2[:,None]*A[None,:]
                cand2[:,diag] -= d2*d2
                vals2 = cand2.max(axis=1)
                j = vals2.argmin()
                bx, bv = xx[j], vals2[j]
                d = bx - old
                c[p:p+n] += d * om
                c[st:st+n] -= d * x
                c[diag] -= d*d
                x[p] = bx
                om[diag-p] = 1.0 - bx
                improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - tol:
            best = xp
            bc = cp
        if not improved:
            break
    return best, bc

def binary_sa(h_init, n, rng, deadline, flips=3, T0=0.05, cool=0.9997):
    target = n // 2
    idx = np.argpartition(h_init, target-1)[:target]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    best = h.copy()
    c = np.correlate(h, 1-h, mode='full')
    cm = c.max()
    best_cm = cm
    dx = 2.0/n
    T = T0
    while time.time() < deadline:
        f = rng.randint(n, size=flips)
        f = np.unique(f)
        if len(f) < 2:
            continue
        h_old = h.copy()
        for i in f:
            h[i] = 1.0 - h[i]
        c_new = np.correlate(h, 1-h, mode='full')
        cm_new = c_new.max()
        delta = cm_new - cm
        if delta < 0 or rng.random() < np.exp(-delta/(T+1e-10)):
            cm, c = cm_new, c_new
            if cm < best_cm:
                best, best_cm = h.copy(), cm
        else:
            h = h_old
        T *= cool
    return best, best_cm * dx
