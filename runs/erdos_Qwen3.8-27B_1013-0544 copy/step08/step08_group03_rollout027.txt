    # As m+d_k=pj (m=pj-d_k): the term was h[m]*(1-1)=0, now h[m]*(1-0)=h[m], so adds h[m]
    v3 = (pj + d_arr >= 0) & (pj + d_arr < n)
    v4 = (pj - d_arr >= 0) & (pj - d_arr < n)
    delta[v3] -= (1.0 - h[pj + d_arr[v3]])
    delta[v4] += h[pj - d_arr[v4]]
    return delta

def binary_sa(h, n, max_iters=50000, deadline=None, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h_b = np.where(h > 0.5, 1.0, 0.0).copy()
    n_ones = n // 2
    ones_mask = h_b == 1.0
    n_ones_current = ones_mask.sum()
    # Fix sum
    if n_ones_current > n_ones:
        excess = rng.choice(np.where(ones_mask)[0], size=n_ones_current - n_ones, replace=False)
        h_b[excess] = 0.0
    elif n_ones_current < n_ones:
        need = rng.choice(np.where(h_b == 0.0)[0], size=n_ones - n_ones_current, replace=False)
        h_b[need] = 1.0
    
    c = compute_corr(h_b, n)
    dx = 2.0 / n
    best_c = c.max() * dx
    best_h = h_b.copy()
    current_c_max = c.max()
    T = 0.005
    dx = 2.0 / n
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        ones = np.where(h_b == 1.0)[0]
        zeros = np.where(h_b == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        pi = zeros[rng.randint(len(zeros))]
        pj = ones[rng.randint(len(ones))]
        
        delta = swap_delta(c, h_b, n, pi, pj)
        new_c = c + delta
        new_max = new_c.max() * dx
        
        if new_max < current_c_max:
            # Accept
            h_b[pi] = 1.0
            h_b[pj] = 0.0
            c = new_c
            current_c_max = new_max
            if new_max < best_c:
                best_c = new_max
                best_h = h_b.copy()
        else:
            # Maybe accept (simulated annealing)
            if rng.random() < np.exp(-(new_max - current_c_max) / max(T, 1e-10)):
                h_b[pi] = 1.0
                h_b[pj] = 0.0
                c = new_c
                current_c_max = new_max
                if new_max < best_c:
                    best_c = new_max
                    best_h = h_b.copy()
        
        T *= 0.9997
    
    return best_h, best_c

    h_new[p] = best_t
    corr_new = corr + (best_t - h0) * A - (best_t*best_t - h0*h0) * C

def exact_cd_fast(h, n, max_passes=100, deadline=None):
    """Exact coordinate descent with precomputed correlation."""
    h = h.copy()
    c = compute_corr(h, n)
    current_max = c.max()
    d_arr = np.arange(2*n-1) - (n-1)
    
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            
            v1 = (p + d_arr >= 0) & (p + d_arr < n)
            v2 = (p - d_arr >= 0) & (p - d_arr < n)
            A = np.zeros(2*n-1)
            A[v1] += (1.0 - h[p + d_arr[v1]])
            A[v2] += (-h[p - d_arr[v2]])
            C = np.zeros(2*n-1)
            C[n-1] = 1.0
            
            # Golden section on t in [0,1]
            def f_obj(t):
                val = c + (t - h0) * A - (t*t - h0*h0) * C
                return float(np.max(val))
            
            from scipy.optimize import minimize_scalar
            res = minimize_scalar(f_obj, bounds=[0, 1], method='bounded', options={'xatol': 1e-10})
            t_star = res.x
            f_star = f_obj(t_star)
            f0 = f_obj(0.0)
            f1 = f_obj(1.0)
            
            if f0 <= f_star and f0 <= f1:
                t_new = 0.0
            elif f1 <= f_star and f1 <= f0:
                t_new = 1.0
            else:
                t_new = t_star
            
            if t_new != h0:
                c = c + (t_new - h0) * A - (t_new*t_new - h0*h0) * C

                if c.max() < current_max:
                    current_max = c.max()
                    improved = True

        c = compute_corr(h, n)
        current_max = c.max()

            break
