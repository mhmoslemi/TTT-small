def project_feasible(h,n):
    target=n/2.0
    h=np.clip(h.copy(),0.0,1.0)
    for _ in range(60):
        diff=target-h.sum()
        if abs(diff)<1e-13:
            break
        if diff>0.0:
            room=1.0-h; tot=room.sum()
            if tot<1e-15: break
            h += (diff/tot)*room
        else:
            tot=h.sum()
            if tot<1e-15: break
            h += (diff/tot)*h
    h=np.clip(h,0.0,1.0)

import numpy as np
import time


def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(60):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(4):
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            i = int(np.argmax(1.0 - h))
            room = 1.0 - h[i]
            if room <= 0.0:
                break
            h[i] += min(room, diff)
        else:
            i = int(np.argmax(h))
            room = h[i]
            if room <= 0.0:
                break
            h[i] -= min(room, -diff)
        diff = target - h.sum()
    return h


def binary_from_scores(scores, n):
    if n % 2:
        return None
    y = np.zeros(n, dtype=np.int32)
    order = np.argsort(-scores, kind='stable')
    y[order[:n // 2]] = 1
    return y


def binary_M_C(y, n):
    z = 1 - y
    D = np.correlate(y, z, mode='full')
    M = int(D.max())
    return M, 2.0 * float(M) / float(n), D


def eval_swap(y, z, i, j):
    yi = int(y[i])
    yj = int(y[j])
    y[i] = yj
    z[i] = 1 - yj
    y[j] = yi
    z[j] = 1 - yi
    D = np.correlate(y, z, mode='full')
    M = int(D.max())
    S = float(np.dot(D, D))
    y[i] = yi
    z[i] = 1 - yi
    y[j] = yj
    z[j] = 1 - yj
    return M, S


def apply_swap(y, z, i, j):
    yi = int(y[i])
    yj = int(y[j])
    y[i] = yj
    z[i] = 1 - yj
    y[j] = yi
    z[j] = 1 - yi


def generate_candidates(y, n, D, limit, tau):
    c = n - 1
    M = int(D.max())
    w1 = np.zeros(n, dtype=np.int32)
    w0 = np.zeros(n, dtype=np.int32)
    order = np.argsort(D, kind='stable')[::-1]
    direct = []

    for idx in order:
        val = int(D[idx])
        if val < M - tau:
            break
        k = idx - c
        if k == 0:
            continue
        if k > 0:
            i0 = 0
            i1 = n - k
        else:
            i0 = -k
            i1 = n
        a = y[i0:i1]
        b = y[i0 + k:i1 + k]
        pos = np.nonzero((a == 1) & (b == 0))[0]
        for off in pos:
            ii = int(i0 + off)
            jj = int(ii + k)
            w1[ii] += 1
            w0[jj] += 1
            if len(direct) < limit:
                direct.append((ii, jj))

    top1 = np.argsort(-w1, kind='stable')[:40]
    top0 = np.argsort(-w0, kind='stable')[:40]
    cart = []
    for ii in top1:
        if int(w1[ii]) <= 0:
            break
        for jj in top0:
            if int(w0[jj]) <= 0:
                break
            ii = int(ii)
            jj = int(jj)
            if ii == jj:
                continue
            if y[ii] == 1 and y[jj] == 0:
                cart.append((ii, jj))
                if len(cart) > limit:
                    cart = cart[:limit]
                    break
        if len(cart) > limit:
            break

    seen = set()
    out = []
    for pair in direct:
        if pair not in seen:
            seen.add(pair)
            out.append(pair)
            if len(out) >= limit:
                break
    if len(out) < limit:
        for pair in cart:
            if pair not in seen:
                seen.add(pair)
                out.append(pair)
                if len(out) >= limit:
                    break
    return out


def binary_refine(y, n, deadline, seed=1, max_eval=80, tau=3):
    y = np.array(y, dtype=np.int32, copy=True)
    z = 1 - y
    D = np.correlate(y, z, mode='full')
    M = int(D.max())
    S = float(np.dot(D, D))
    best_y = y.copy()
    best_M = M
    best_S = S
    rng = np.random.RandomState(seed)
    stall = 0
    rand_left = 180
    tau_cur = tau

    while time.time() < deadline:
        cand = generate_candidates(y, n, D, 2 * max_eval + 100, tau_cur)
        best_pair = None
        best_key = None
        for pair in cand[:max_eval]:
            m, s = eval_swap(y, z, pair[0], pair[1])
            if m < M or (m == M and s < S - 1.0):
                if best_key is None or m < best_key[0] or (m == best_key[0] and s < best_key[1]):
                    best_pair = pair
                    best_key = (m, s)

        if best_pair is not None:
            apply_swap(y, z, best_pair[0], best_pair[1])
            D = np.correlate(y, z, mode='full')
            M = int(D.max())
            S = float(np.dot(D, D))
            stall = 0
            if M < best_M or (M == best_M and S < best_S):
                best_M = M
                best_S = S
                best_y = y.copy()
            continue

        stall += 1
        moved = False
        if rand_left > 0 and stall <= 30:
            ones = np.nonzero(y == 1)[0]
            zeros = np.nonzero(y == 0)[0]
            if len(ones) and len(zeros):
                trials = min(5, rand_left)
                for _ in range(trials):
                    i = int(ones[rng.randint(len(ones))])
                    j = int(zeros[rng.randint(len(zeros))])
                    m, s = eval_swap(y, z, i, j)
                    if m < M or (m == M and s < S - 50.0):
                        apply_swap(y, z, i, j)
                        D = np.correlate(y, z, mode='full')
                        M = int(D.max())
                        S = float(np.dot(D, D))
                        if M < best_M or (M == best_M and S < best_S):
                            best_M = M
                            best_S = S
                            best_y = y.copy()
                        moved = True
                        break
                rand_left -= trials

        if moved:
            stall = 0
            continue

        tau_cur = min(6, tau_cur + 1)
        if stall > 30:
            break

    return best_y, 2.0 * float(best_M) / float(n), best_M


def coord_cd(h, n, max_passes=100, deadline=None, grid=80):
    h = np.asarray(h, dtype=float).copy()
    n = int(len(h))
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    cc = n - 1

    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        if p % 2 == 0:
            indices = range(n)
        else:
            indices = range(n - 1, -1, -1)

        for idx in indices:
            h0 = float(h[idx])
            lo = -h0
            hi = 1.0 - h0
            if hi - lo <= 1e-13:
                continue
            r = max(idx, n - 1 - idx)
            if r <= 0:
                continue
            s = cc - r
            e = cc + r + 1
            a = np.zeros(2 * r + 1)
            k = np.arange(-r, r + 1)
            first = (k >= -idx) & (k <= n - 1 - idx)
            if np.any(first):
                a[first] += 1.0 - h[idx + k[first]]
            second = (k >= idx - n + 1) & (k <= idx)
            if np.any(second):
                a[second] -= h[idx - k[second]]

            c_aff = c[s:e].copy()
            unaff = 0.0
            if s > 0:
                tm = float(c[:s].max())
                if tm > unaff:
                    unaff = tm
            if e < c.size:
                tm = float(c[e:].max())
                if tm > unaff:
                    unaff = tm

            dgrid = np.linspace(lo, hi, grid)
            mat = c_aff[None, :] + dgrid[:, None] * a[None, :]
            di = np.where(k == 0)[0]
            if di.size:
                mat[:, di] -= dgrid[:, None] ** 2
            vals = np.maximum(unaff, mat.max(axis=1))

            bi = int(np.argmin(vals))
            bd = float(dgrid[bi])
            bv = float(vals[bi])

            if 0 < bi < grid - 1:
                yv = vals[bi - 1:bi + 2]
                xd = dgrid[bi - 1:bi + 2]
                num = yv[0] * (xd[1] - xd[2]) + yv[1] * (xd[2] - xd[0]) + yv[2] * (xd[0] - xd[1])
                den = yv[0] * (xd[2] - xd[1]) + yv[1] * (xd[0] - xd[2]) + yv[2] * (xd[1] - xd[0])
                if abs(den) > 1e-15:
                    xv = num / den
                    if xd[0] < xv < xd[2]:
                        nv = c_aff + xv * a
                        if di.size:
                            nv = nv.copy()
                            nv[di] -= xv * xv
                        vv = max(unaff, float(nv.max()))
                        if vv < bv:
                            bv = vv
                            bd = xv

            if abs(bd) > 1e-11 and bv < cur - 1e-12:
                h[idx] = float(np.clip(h0 + bd, 0.0, 1.0))
                c[s:e] = c_aff + bd * a
                if di.size:
                    c[s + di] -= bd * bd
                new_cur = max(unaff, float(c[s:e].max()))
                if new_cur < cur:
                    cur = new_cur
                improved = True

        if not improved:
            break

        h = project_feasible(h, n)
        cv = compute_c5(h, n)
        if cv < best_c:
            best_c = cv
            best_h = h.copy()
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
