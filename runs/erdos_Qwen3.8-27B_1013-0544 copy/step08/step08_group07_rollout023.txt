def subgradient_step(h, n, lr, momentum_buf, active_tol=1e-8):
    """One step of projected subgradient descent with momentum."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    active = np.where(c >= m - active_tol)[0]

    sg = np.zeros(n)
    for k in active:
        shift = int(k)
        # term1: (1-h[i-shift]) for i where 0<=i-shift<n
        lo1 = max(0, shift)
        hi1 = min(n, shift + n)
        if lo1 < hi1:
            sg[lo1:hi1] += (1.0 - h[lo1-shift:hi1-shift])
        # term2: -h[i+shift] for i where 0<=i+shift<n
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if lo2 < hi2:
            sg[lo2:hi2] -= h[lo2+shift:hi2+shift]
    sg /= len(active)
    
    # Momentum update
    v = momentum_buf
    v = 0.9 * v - lr * sg
    h_new = h + v
    
    return project_feasible(h_new, n), v

def subgradient_opt(h, n, n_iter=5000,deadline=None, active_tol=1e-8):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    lr = 0.01

            break
        h, v = subgradient_step(h, n, lr, v, active_tol)
        c = compute_c5(h, n)
        if c < best_c:

            best_c = c

        lr = lr * 0.999

            k_hi = min(2 * n - 2, p)
            # Backward: h[p] in (1-h[j]) when j=p, affects c[k] where h[p+k] is valid, i.e., 0<=p+k<n => k in [-p, n-1-p]
            bk_lo = max(0, -p)  # = 0
            bk_hi = min(2 * n - 2, n - 1 - p)
            k_start = min(k_lo, bk_lo)
            k_end = max(k_hi, bk_hi)
            
            if k_start == k_end and k_start != 0:
                # Edge case, skip
                pass
            
            ms = np.arange(k_start, k_end + 1)
            n_aff = len(ms)
            
            # Compute linear coefficients
            a = np.empty(n_aff)
            for idx in range(n_aff):
                m = ms[idx]
                coeff = 0.0
                jm = p - m
                if 0 <= jm < n:
                    coeff += (1.0 - h[jm])
                jp = p + m
                if 0 <= jp < n:
                    coeff -= h[jp]
                a[idx] = coeff
            
            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(c[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[k_end + 1:].max()))
            
            # Check if m=0 is affected (quadratic term)
            has_quad = (k_start <= 0 <= k_end)
            quad_idx = (0 - k_start) if has_quad else -1
            
            # 1D optimization: find d in [-h0, 1-h0] minimizing max(unaff_max, max(c[ms] + d*a - d²*quad_indicator))
            c_aff_base = c[ms].copy()
            
            # Coarse grid
            best_d = 0.0
            best_val = current_max
            ng = 30
            for d in np.linspace(-h0, 1.0 - h0, ng):
                if abs(d) < 1e-15:
                    continue
                vals = c_aff_base + d * a
                if has_quad:
                    vals[quad_idx] -= d * d
                val = max(unaff_max, float(vals.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            
            # Fine grid around best
            if best_d != 0.0:
                span = (1.0 + h0) / ng
                lo = max(-h0, best_d - span)
                hi = min(1.0 - h0, best_d + span)
                for d in np.linspace(lo, hi, 50):
                    if abs(d) < 1e-15:
                        continue
                    vals = c_aff_base + d * a
                    if has_quad:
                        vals[quad_idx] -= d * d
                    val = max(unaff_max, float(vals.max()))
                    if val < best_val:
                        best_val = val
                        best_d = d
            
            if abs(best_d) > 1e-14 and best_val < current_max - 1e-15:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                # Update correlation incrementally
                c[ms] = c_aff_base + best_d * a
                if has_quad:
                    c[0] -= best_d * best_d
                current_max = max(unaff_max, float(c[ms].max()))
                any_improved = True
        
        if not any_improved:
            break
        # Re-project to maintain feasibility
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        # Recompute correlation (since projection may have changed things)
        c = np.correlate(h, 1.0 - h, mode='full')
    
    return h, best_c
