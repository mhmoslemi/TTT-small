        # Compute subgradient
        grad = np.zeros(n)
        for j in max_indices:
            lag = j - (n - 1)
            # c[j] = sum_k h[k+lag] * (1-h[k]) for valid k
            # d(c[j])/d(h[p]):
            #   If p+lag in [0,n): coefficient is (1-h[p]) [from h[p+lag] term, treating h[p] as the first factor... wait]
            #   Actually: c[j] = sum_k h[k+(j-(n-1))] * (1-h[k])
            #   Let lag = j-(n-1)
            #   c[j] = sum_k h[k+lag] * (1-h[k]) for k in valid range
            #   d/d(h[p]): 
            #     - when k+lag = p, i.e., k = p-lag: contributes (1-h[p-lag])
            #     - when k = p: contributes -h[p+lag]
            #   So d(c[j])/d(h[p]) = (1-h[p-lag]) - h[p+lag] (when both valid)
            #   or just one term if only one is valid
            
            k1 = p - lag  # k where k+lag = p → first factor is h[p]
            k2 = p        # k where k = p → second factor is (1-h[p])
            
            # Actually this is per-p, so let me vectorize differently.
            # For a given j (with lag = j-(n-1)):
            #   d(c[j])/d(h[p]) = [1 if 0<=p-lag<n else 0] * (1-h[p-lag if valid]) ... 
            #   This is getting complicated to vectorize per-j. Let me just do it per-j.
            
            # For this j, the gradient contribution to each p:
            #   If 0 <= p-lag < n: grad[p] += (1 - h[max(0, p-lag)]) ... no
            #   Let me think again.
            #   c[j] = sum_{k: 0<=k<n, 0<=k+lag<n} h[k+lag] * (1-h[k])
            #   d(c[j])/d(h[p]) = sum over terms involving h[p]:
            #     Term where k+lag = p → k = p-lag, valid if 0<=p-lag<n and 0<=k<n (same thing)
            #       Contribution: (1-h[p-lag])... no, (1-h[k]) = (1-h[p-lag])
            #     Term where k = p → k+lag = p+lag, valid if 0<=p+lag<n
            #       Contribution: -h[p+lag] (derivative of (1-h[p]) is -1, times h[p+lag])
            
            # So: d(c[j])/d(h[p]) = (1-h[p-lag]) * I(0<=p-lag<n) - h[p+lag] * I(0<=p+lag<n)
            
            # Vectorized over p:
            p_idx = np.arange(n)
            p_f = p_idx - lag  # where h[p] is the "first" factor
            p_b = p_idx + lag  # where h[p] is the "second" factor (in 1-h[p])
            valid_f = (p_f >= 0) & (p_f < n)
            valid_b = (p_b >= 0) & (p_b < n)
            g = np.zeros(n)
            if np.any(valid_f):
                g[valid_f] += (1.0 - h[p_f[valid_f]])
            if np.any(valid_b):
                g[valid_b] -= h[p_b[valid_b]]
            grad += g / len(max_indices)  # average over max indices
        
        # Learning rate
        lr = lr0 / (1.0 + 0.01 * it)
        # Gradient descent step
        h = h - lr * grad
        h = project_feasible(h, n)
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_passes=50, grid_size=200, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    for pidx in range(max_passes):
        if deadline is not None and time.time() > deadline:

            m_b_lo = max(0, p)

            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)

            jb = p - (ms - (n - 1))

def subgradient_step(h, n, lr, c, cmax):
    """One subgradient step using near-max terms."""
    thresh = cmax * 0.999
    max_idx = np.where(c >= thresh)[0]

    grad = np.zeros(n)
    p_idx = np.arange(n)
    for j in max_idx:
        lag = j - (n - 1)
        pf = p_idx - lag
        pb = p_idx + lag
        vf = (pf >= 0) & (pf < n)

        grad += g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_passes=50, grid_size=200, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    for pidx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        order = np.random.permutation(n)
        for p in order:
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + (ms - (n - 1))
            jb = p - (ms - (n - 1))
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c_aff = c[ms]
            diag = (ms == (n - 1))
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            c_vals = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if np.any(diag):
                c_vals[:, diag] -= d_grid[:, None] ** 2
            obj = np.maximum(unaff, c_vals.max(axis=1))
            bi = int(obj.argmin())
            if obj[bi] < current_max - 1e-15:
                h[p] = float(np.clip(h0 + d_grid[bi], 0.0, 1.0))
                improved = True
                c[ms] = c_aff + d_grid[bi] * a
                if np.any(diag):
                    c[ms[diag]] -= d_grid[bi] ** 2
                current_max = float(obj[bi])
        if not improved:
            break
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < best_c:
            best_c = nc
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(np.clip(x, 0, 1), 1.0 - np.clip(x, 0, 1), mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def lse_lbfgs(h, n, alphas, maxiter_each=300, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 20:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18, 'gtol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgrad_descent(h, n, max_it=2000, lr0=0.01, deadline=None):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    p_idx = np.arange(n)
    for it in range(max_it):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = float(c.max())
        cv = cmax * (2.0 / n)
        if cv < best_c:
            best_c = cv
            best_h = h.copy()
        thresh = cmax * 0.9995
        max_idx = np.where(c >= thresh)[0]
        if len(max_idx) == 0:
            max_idx = np.where(c >= cmax - 1e-10)[0]
        grad = np.zeros(n)
        for j in max_idx:
            lag = j - (n - 1)
            pf = p_idx - lag
            pb = p_idx + lag
            vf = (pf >= 0) & (pf < n)
            vb = (pb >= 0) & (pb < n)
            g = np.zeros(n)
            if np.any(vf):
                g[vf] += (1.0 - h[pf[vf]])
            if np.any(vb):
                g[vb] -= h[pb[vb]]
            grad += g
        grad /= len(max_idx)
        lr = lr0 / (1.0 + 0.003 * it)
        h = project_feasible(h - lr * grad, n)
    return best_h, best_c
