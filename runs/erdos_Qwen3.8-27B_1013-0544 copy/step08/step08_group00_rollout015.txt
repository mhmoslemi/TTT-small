    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) * 0.5
            r2 = (1.0 + np.sqrt(disc)) * 0.5
            if x_lo > r1 and x_hi < r2:
                return h0
            if x_lo < r1 and x_hi > r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo < r1:
                x_hi = min(x_hi, r1)
            elif x_hi > r2:
                x_lo = max(x_lo, r2)
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi:
        return h0
    return float((x_lo + x_hi) * 0.5)

def coord_cd(h, n, max_passes, deadline):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(400, 0.5)
        n = 400
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: SLSQP
    h, c = slsqp_optimize(h, n, [500, 5000, 50000], maxiter_each=250)
    if c < best_c:
        best_h, best_c = h.copy(), c

    # Phase 2: Strong coordinate descent
    h, c = coord_cd(h, n, 300, deadline)
    if c < best_c:
        best_h, best_c = h.copy(), c

    # Phase 3: Structured perturbation exploration
    x_ax = np.linspace(0, 2, n)
    perturbations = []
    for s in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25]:
        perturbations.append(('rand', s, None))
    for freq in [1, 2, 3, 4, 5, 7, 10, 14, 20]:
        for amp in [0.02, 0.05, 0.1]:
            perturbations.append(('sin', amp, freq))
    for freq in [2, 3, 5, 8]:
        for amp in [0.03, 0.06, 0.1]:
            perturbations.append(('cos', amp, freq))

    for ptype, param, freq in perturbations:
        if time.time() > deadline - 120:
            break
        if ptype == 'rand':
            h_pert = project_feasible(best_h + param * (rng.rand(n) - 0.5), n)
        elif ptype == 'sin':
            h_pert = project_feasible(best_h + param * np.sin(np.pi * freq * x_ax), n)
        else:
            h_pert = project_feasible(best_h + param * np.cos(np.pi * freq * x_ax), n)
        h_try, c_try = coord_cd(h_pert, n, 80, deadline)
        if c_try < best_c - 1e-7:
            best_h, best_c = h_try.copy(), c_try

    # Phase 4: Try different n
    x_best = np.linspace(0, 2, len(best_h))
    n_tries = [350, 500, 200, 1000, 150, 280]
    for n_try in n_tries:
        if time.time() > deadline - 180:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_best, best_h), n_try)
        h_t, c_t = slsqp_optimize(h_int, n_try, [1000, 10000], maxiter_each=200)
        h_t, c_t = coord_cd(h_t, n_try, 60, deadline)
        if c_t < best_c:
            best_h, best_c = h_t.copy(), c_t
            x_best = np.linspace(0, 2, len(best_h))
            n = len(best_h)
            x_ax = x_best

    # Phase 5: Random restarts at current n
    n_cur = len(best_h)
    remaining = deadline - time.time()
    if remaining > 150:
        n_restarts = max(3, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 80:
                break
            scale = 0.02 + 0.15 * (r / max(1, n_restarts - 1))
            h_pert = project_feasible(best_h + scale * rng.standard_normal(n_cur), n_cur)
            h_t, c_t = slsqp_optimize(h_pert, n_cur, [5000], maxiter_each=200)
            h_t, c_t = coord_cd(h_t, n_cur, 60, deadline)
            if c_t < best_c:
                best_h, best_c = h_t.copy(), c_t

    # Phase 6: Final refinement
    if time.time() < deadline - 10:
        h, c = slsqp_optimize(best_h, len(best_h), [200000, 1000000], maxiter_each=500)
        if c < best_c:
            best_h, best_c = h.copy(), c
        h, c = coord_cd(best_h, len(best_h), 80, deadline)
        if c < best_c:
            best_h, best_c = h.copy(), c

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
