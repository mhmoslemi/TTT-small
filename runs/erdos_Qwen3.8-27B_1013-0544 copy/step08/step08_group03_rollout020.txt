                            a, c1, fc1 = c1, d1, fd1
                            d1 = a + gr * (b - a)
                            fd1 = float(np.max(c0 + d1 * A - d1 * d1 * D))
                    ts = (a + b) * 0.5
                    fs = float(np.max(c0 + ts * A - ts * ts * D))
                    if fs < best_f:
                        best_t, best_f = ts, fs
            if best_f < cur_max - 1e-12:
                tn = float(np.clip(best_t, 0.0, 1.0))
                c += (tn - h0) * A - (tn * tn - h0 * h0) * D
                h[p] = tn
                cur_max = float(np.max(c))
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def _slsqp_sum_con(x, n):
    return x.sum() - n / 2.0

def _slsqp_sum_jac(x, n):
    return np.ones(len(x))

def _slsqp_obj_p(corr_vec, p, n):
    return (np.sum(corr_vec ** p)) ** (1.0 / p) * (2.0 / n)

def pnorm_slsqp(h, n, p_list, maxiter_each=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in p_list:
        if deadline is not None and time.time() > deadline:
            break
        def obj(x):
            cv = np.correlate(x, 1.0 - x, mode='full')
            return _slsqp_obj_p(cv, p, n)
        def gfun(x):
            return _slsqp_sum_con(x, n)
        def gjac(x):
            return _slsqp_sum_jac(x, n)
        cons = [{'type': 'eq', 'fun': gfun, 'jac': gjac}]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_each, 'ftol': 1e-16})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n = 400
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Geometric p-continuation SLSQP
    p_sched = [5, 15, 50, 150, 500, 1500, 5000]
    h1, c1 = pnorm_slsqp(best_h, n, p_sched, maxiter_each=350, deadline=deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: CD with boundary focus
    h2, c2 = cd_optimize(best_h, n, max_passes=500, deadline=deadline,
                         grid_pts=30, gs_iters=25, seed=seed)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Multi-scale exploration
    n_cands = [x for x in [300, 400, 500, 600, 750, 1000] if x != n]
    for nt in n_cands:
        if time.time() > deadline - 250:
            break
        hi = interpolate_h(best_h, n, nt)
        ho, co = pnorm_slsqp(hi, nt, [50, 200, 1000], 250, deadline)
        ho, co = cd_optimize(ho, nt, 250, deadline, grid_pts=25, gs_iters=20, seed=seed + nt)
        if co < best_c:
            best_h, best_c, n = ho, co, nt

    # Phase 4: Structured restarts targeting active k
    n = len(best_h)
    rem = deadline - time.time()
    if rem > 120:
        nr = max(2, int(rem / 130))
        for r in range(nr):
            if time.time() > deadline - 80:
                break
            if r % 4 == 3:
                hp = rng.uniform(0.2, 0.8, n)
            else:
                w = np.abs(best_h - 0.5) + 0.05
                sc = 0.015 + 0.035 * (r / max(1, nr - 1))
                hp = best_h + rng.normal(0, sc, n) * (w / w.mean())
            hp = project_feasible(hp, n)
            ho, co = pnorm_slsqp(hp, n, [200, 1000, 5000], 250, deadline)
            ho, co = cd_optimize(ho, n, 250, deadline, grid_pts=25, gs_iters=25, seed=seed + 500 + r)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 5: Final high-p polish + CD
    if time.time() < deadline - 30:
        hf, cf = pnorm_slsqp(best_h, n, [10000, 50000], 300, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 15:
        hf, cf = cd_optimize(best_h, n, 300, deadline, grid_pts=40, gs_iters=35, seed=seed + 999)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 5:
        hf, cf = cd_optimize(best_h, n, 200, deadline, grid_pts=50, gs_iters=40, seed=seed + 1999)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
