def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cd_adaptive(h, n, max_passes=5000, deadline=None, grid_size=64, bisect_iters=25):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        idx_order = range(n) if p % 2 == 0 else range(n-1, -1, -1)
        improved = False
        for idx in idx_order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            lo_d = -h0 if h0 > 0 else 0.0
            hi_d = 1.0 - h0 if h0 < 1 else 0.0
            if hi_d - lo_d < 1e-12:
                continue
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]

            diag_idx = np.where(ms == (n - 1))[0]

def smooth_grad_step(h, n, alpha, lr=0.01):
    """One step of gradient descent on log-sum-exp smoothed objective."""
    c = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * c
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return h, 0.0
    w /= w_sum
    
    n_corr = 2 * n - 1
    grad = np.zeros(n)
    for m in range(n_corr):
        if w[m] < 1e-12:
            continue
        shift = m - (n - 1)
        wm = w[m]
        if shift >= 0:
            s = shift
            if s == 0:
                # c = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
                # d/dh[p] = 1 - 2*h[p]
                grad += wm * (1.0 - 2.0 * h)
            else:
                # c = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
                # d/dh[p] += (1-h[p-s]) for s<=p<=n-1
                # d/dh[p] -= h[p+s] for 0<=p<=n-1-s
                grad[s:n] += wm * (1.0 - h[0:n-s])
                grad[0:n-s] -= wm * h[s:n]
        else:
            s = -shift
            # c = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
            # d/dh[p] += (1-h[p+s]) for 0<=p<=n-1-s
            # d/dh[p] -= h[p-s] for s<=p<=n-1
            grad[0:n-s] += wm * (1.0 - h[s:n])
            grad[s:n] -= wm * h[0:n-s]
    
    # Projected gradient step
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new, 0.0


def smooth_grad_descent(h, n, deadline=None, n_iter=3000, alpha_init=50, alpha_end=5000):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        progress = it / max(1, n_iter - 1)
        alpha = alpha_init * (alpha_end / alpha_init) ** progress
        lr = 0.1 / (1.0 + 0.05 * it)
        h_new, _ = smooth_grad_step(h, n, alpha, lr)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

    h_rev = h[::-1].copy()
