def pnorm_slsqp(h, n, p_list, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in p_list:
        if deadline is not None and time.time() > deadline:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return np.sum(c**p) ** (1.0/p) * (2.0/n)
        def sum_con(x):
            return x.sum() - n/2.0
        def sum_con_jac(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    """One CD sweep in the given order. Returns updated h, c, current_max."""
    current_max = float(c.max())
    n_improved = 0
    for p in order:
        if deadline is not None and time.time() > deadline:
            break
        h0 = h[p]
        if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
            continue
        pa = p + kvals - (n - 1)
        pb = p - kvals + (n - 1)
        B = np.zeros(2 * n - 1)
        va = (pa >= 0) & (pa < n)
        vb = (pb >= 0) & (pb < n)
        B[va] = 1.0 - h[pa[va]]
        B[vb] -= h[pb[vb]]
        
        # Vectorized grid
        grid = np.linspace(0.0, 1.0, grid_size)
        us = grid - h0
        vals = c[np.newaxis, :] + us[:, np.newaxis] * B[np.newaxis, :] - (us[:, np.newaxis]) ** 2 * D[np.newaxis, :]
        fvals = vals.max(axis=1)
        best_idx = int(np.argmin(fvals))
        
        lo = grid[max(0, best_idx - 1)]
        hi = grid[min(grid_size - 1, best_idx + 1)]
        if hi - lo > 1e-10:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = lo, hi
            c1 = b - gr * (b - a)
            d1 = a + gr * (b - a)
            fc1 = float(np.max(c + (c1 - h0) * B - (c1 - h0) ** 2 * D))
            fd1 = float(np.max(c + (d1 - h0) * B - (d1 - h0) ** 2 * D))
            for _ in range(gs_iters):
                if fc1 < fd1:
                    b = d1; d1 = c1; fd1 = fc1
                    c1 = b - gr * (b - a)
                    fc1 = float(np.max(c + (c1 - h0) * B - (c1 - h0) ** 2 * D))
                else:
                    a = c1; c1 = d1; fc1 = fd1
                    d1 = a + gr * (b - a)
                    fd1 = float(np.max(c + (d1 - h0) * B - (d1 - h0) ** 2 * D))
            t_star = (a + b) * 0.5
        else:
            t_star = grid[best_idx]
        
        f0 = float(np.max(c + (0.0 - h0) * B - (0.0 - h0) ** 2 * D))
        f1 = float(np.max(c + (1.0 - h0) * B - (1.0 - h0) ** 2 * D))
        fs = float(np.max(c + (t_star - h0) * B - (t_star - h0) ** 2 * D))
        
        if min(f0, f1, fs) < current_max - 1e-12:
            t_new = 0.0 if (f0 <= f1 and f0 <= fs) else (1.0 if (f1 <= f0 and f1 <= fs) else float(np.clip(t_star, 0.0, 1.0)))
            u = t_new - h0
            c = c + u * B - u * u * D
            h[p] = t_new
            current_max = float(c.max())
            n_improved += 1
    return h, c, current_max, n_improved

def exact_cd(h, n, max_passes=200, deadline=None, grid_size=40, gs_iters=30):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    kvals = np.arange(2 * n - 1)
    D = np.zeros(2 * n - 1)
    D[n - 1] = 1.0
    
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        h, c, _, n_imp = cd_sweep(h, c, n, kvals, D, list(range(n)), deadline, grid_size, gs_iters)
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full').astype(float)
        if n_imp == 0:
            break
    return h, compute_c5(h, n)

def slsqp_refine(h, n, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in [50, 200, 1000]:
        if deadline is not None and time.time() > deadline:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return np.sum(c ** p) ** (1.0 / p) * (2.0 / n)
        def cons_fun(x):
            return x.sum() - n / 2.0
        def cons_jac(x):
            return np.ones(n)
        try:
            res = minimize(obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                          options={'maxiter': 300, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
