            e = -d
            g[:n-e] += w*(1.0-h[e:])
            g[e:] -= w*h[:n-e]
    return (2.0/n)*g

def sv(h, n, a): return smooth_parts(h, n, a)[0]

def slsqp(h, n, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bnd = [(0.0, 1.0)]*n
    cons = [{'type':'eq','fun':sum_con,'jac':sum_jac,'args':(n,)}]
    ws = best_h.copy()
    for alpha, mx in [(20,40),(80,35),(300,30),(1200,25),(5000,20),(20000,18),(80000,15)]:
        if time.time() > deadline - 2: break
        try:
            r = minimize(sv, ws, args=(n,alpha), method='SLSQP', jac=smooth_grad,
                        bounds=bnd, constraints=cons,
                        options={'maxiter':mx,'ftol':1e-14,'disp':False})
            x = project_feasible(r.x, n)
            c = compute_c5(x, n)
            if c < best_c: best_h, best_c, ws = x, c, x
        except: pass
    return best_h, best_c

def compute_A(h, n):
    A = np.zeros((n, 2*n-1))
    off = n-1
    for k in range(2*n-1):
        d = k-off
        if d >= 0:
            A[d:,k] += 1.0 - h[:n-d]
            A[:n-d,k] -= h[d:]
        else:
            e = -d
            A[:n-e,k] += 1.0 - h[e:]
            A[e:,k] -= h[:n-e]
    return A

def pair_search(h, n, ntries, grid, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0-h, mode='full')
    A = compute_A(h, n)
    off = n-1
    accepts, stale = 0, 0
    for t in range(ntries):
        if time.time() > deadline: break
        cm = c.max()
        tol = max(0.015, 0.0008*cm)
        act = np.flatnonzero(c >= cm - tol)
        if t % 5 == 4 or act.size <= 2:
            i, j = rng.randint(n), rng.randint(n)
            if i == j: j = (j+1) % n
        else:
            k = act[rng.randint(act.size)]
            d = k - off
            if d >= 0:
                m = n - d
                if m <= 0: continue
                b = rng.randint(m); i, j = b+d, b
            else:
                e = -d; m = n - e
                if m <= 0: continue
                b = e + rng.randint(m); i, j = b-e, b
        S = h[i] + h[j]
        lo_u = max(0.0, S-1.0) - h[i]
        hi_u = min(1.0, S) - h[i]
        if hi_u - lo_u < 1e-14: stale += 1; continue
        B = A[i] - A[j]
        us = np.linspace(lo_u, hi_u, grid)
        lines = c[None,:] + us[:,None]*B[None,:]
        lines[:,off] -= 2.0*us**2
        if 0 <= off+i-j < 2*n-1: lines[:,off+i-j] += us**2
        if 0 <= off+j-i < 2*n-1: lines[:,off+j-i] += us**2
        vals = lines.max(axis=1)
        idx = vals.argmin()
        u = us[idx]
        bestv = vals[idx]
        if 0 < idx < grid-1 and bestv < cm - 1e-10:
            us2 = np.linspace(us[idx-1], us[idx+1], 12)
            lines2 = c[None,:] + us2[:,None]*B[None,:]
            lines2[:,off] -= 2.0*us2**2
            if 0 <= off+i-j < 2*n-1: lines2[:,off+i-j] += us2**2
            if 0 <= off+j-i < 2*n-1: lines2[:,off+j-i] += us2**2
            vals2 = lines2.max(axis=1)
            idx2 = vals2.argmin()
            if vals2[idx2] < bestv:
                u, bestv = us2[idx2], vals2[idx2]
        if bestv < cm - 1e-10:
            t2 = h[i] + u
            h[i], h[j] = t2, S - t2
            c = c + u*B
            c[off] -= 2.0*u*u
            if 0 <= off+i-j < 2*n-1: c[off+i-j] += u*u
            if 0 <= off+j-i < 2*n-1: c[off+j-i] += u*u
            A = compute_A(h, n)
            accepts += 1; stale = 0
            if accepts >= min(200, ntries//3): break
        else:
            stale += 1
        if stale > max(500, ntries//2): break
    return h, compute_c5(h, n)

def interp_h(h, m):
    n = len(h)
    if n == m: return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def apply_syms(h, n):
    c = compute_c5(h, n)
    hs = project_feasible(0.5*(h + h[::-1]), n)
    if compute_c5(hs, n) < c: h, c = hs, compute_c5(hs, n)
    hc = 1.0 - h
    if compute_c5(hc, n) < c: h, c = hc, compute_c5(hc, n)
    hsc = project_feasible(0.5*(h + 1.0 - h[::-1]), n)
    if compute_c5(hsc, n) < c: h, c = hsc, compute_c5(hsc, n)
    return h, c

def opt_instance(h, n, deadline, rng, quick=False):
    h = project_feasible(h, n)
    h, c = apply_syms(h, n)
    if not quick and time.time() < deadline - 25:
        h, c = slsqp(h, n, deadline)
    else:
        dl = time.time() + min(25.0, max(5.0, (deadline-time.time())*0.4))
        h, c = slsqp(h, n, dl)
    rem = deadline - time.time()
    if rem > 3 and n <= 1200:
        mt = int(min(10000 if not quick else 4000, max(400, rem*35)))
        h, c = pair_search(h, n, mt, 48, deadline, rng)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)): raise ValueError
    except:
        x = np.linspace(0, 2, 200, endpoint=False)
        h0 = project_feasible(0.5 + 0.3*np.sin(np.pi*x) + 0.15*np.sin(2*np.pi*x), 200)
    n0 = len(h0)
    best_h, best_c = apply_syms(project_feasible(h0, n0), n0)
    main_dl = hard - max(120.0, 0.22*float(budget_s))
    if main_dl < t0 + 20: main_dl = hard - 5
    best_h, best_c = opt_instance(best_h, n0, main_dl, rng, False)
    for m in [300, 400, 500, 600, 700, 800, 1000, 1200]:
        if time.time() > hard - 30: break
        if m == n0: continue
        rem = hard - time.time()
        if m > 600 and rem < 80: continue
        if m > 900 and rem < 140: continue
        if m > 1100 and rem < 170: continue
        dl = time.time() + min(55.0, 0.25*rem)
        if dl < time.time() + 10: continue
        ht, ct = opt_instance(interp_h(best_h, m), m, dl, rng, True)
        if ct < best_c: best_h, best_c = ht, ct
    if time.time() < hard - 40:
        n = len(best_h)
        for s in [0.04, 0.1, 0.2]:
            if time.time() > hard - 20: break
            hp = project_feasible(best_h + rng.normal(0, s, n), n)
            dl = min(hard-5, time.time()+35)
            ht, ct = opt_instance(hp, n, dl, rng, True)
            if ct < best_c: best_h, best_c = ht, ct
    if time.time() < hard - 5:
        n = len(best_h)
        rem = hard - 2 - time.time()
        if rem > 3 and n <= 1200:
            ht, ct = pair_search(best_h, n, int(min(5000, max(300, rem*30))), 64, hard-2, rng)
            if ct < best_c: best_h, best_c = ht, ct
    best_h = project_feasible(best_h, len(best_h))
    return best_h, compute_c5(best_h, len(best_h)), len(best_h)
