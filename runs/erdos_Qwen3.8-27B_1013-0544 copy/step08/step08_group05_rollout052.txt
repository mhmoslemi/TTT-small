def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(55):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx

def smooth_obj(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]

def sum_con(x, n):
    return x.sum() - 0.5 * n

def sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)

    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for a in alphas:

            res = minimize(
                smooth_obj, best, args=(n, a), jac=smooth_grad,

        nc = compute_c5(nh, n)

def binarize(h, n):

    target = n // 2

        return b
    idx = np.argpartition(h, target - 1)[:target]

                xx = np.linspace(lo, hi, refine + 1)
                d2 = (xx - old)[:, None]
                mm = c[None, :] + d2 * A[None, :]
                mm[:, diag] -= (xx - old) ** 2
                mm = mm.max(axis=1)
                j = int(np.argmin(mm))
                bx = float(xx[j])
                bv = float(mm[j])
                if lo < bx < hi:
                    sp2 = (hi - lo) / float(refine)
                    lo2 = max(0.0, bx - sp2)
                    hi2 = min(1.0, bx + sp2)
                    yy = np.linspace(lo2, hi2, 13)
                    d3 = (yy - old)[:, None]
                    ww = c[None, :] + d3 * A[None, :]
                    ww[:, diag] -= (yy - old) ** 2
                    ww = ww.max(axis=1)
                    k2 = int(np.argmin(ww))
                    bx = float(yy[k2])
                    bv = float(ww[k2])
                if bv < cur - tol:
                    d_val = bx - old
                    c[p:p + n] += d_val * om
                    c[st:st + n] -= d_val * x
                    c[diag] -= d_val * d_val
                    x[p] = bx
                    om[diag - p] = 1.0 - bx
                    improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - tol:
            best = xp
            bc = cp
        if not improved:
            break
    return best, bc

def binarize(h, n):
    if n % 2:
        return None
    target = n // 2
    if target <= 0 or target >= n:
        b = np.zeros(n, dtype=float)
        b[:target] = 1.0
        return b
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def binary_trial(c, h, om, i, j, n):
    d = c.copy()
    d[i:i + n] -= om
    st = n - 1 - i
    d[st:st + n] += h
    d[j:j + n] += om
    st = n - 1 - j
    d[st:st + n] -= h
    d[n - 1] -= 2.0
    d[n - 1 + i - j] += 1.0
    d[n - 1 + j - i] += 1.0
    return d, float(d.max())

def rand_one_zero(h, n, rng):
    i = int(np.argmax(h))
    for _ in range(8):
        i = int(rng.randint(n))
        if h[i] >= 0.5:
            break
    j = int(np.argmin(h))
    for _ in range(8):
        j = int(rng.randint(n))
        if h[j] < 0.5:
            break
    if i == j:
        i = int(np.argmax(h))
        j = int(np.argmin(h))
    return i, j

def binary_optimize(h_init, n, rng, deadline):
    b = binarize(h_init, n)
    if b is None:
        return None, np.inf
    dx = 2.0 / n
    best = project_simplex(b, n)
    bc = compute_c5(best, n)
    # SA-based binary optimization
    T0 = 0.5  # initial temperature (in units of correlation, not C5)
    T_min = 1e-8
    cooling = 0.95
    c = np.correlate(best, 1.0 - best, mode="full")
    cm = float(c.max())
    h = best.copy()
    om = 1.0 - h[::-1]
    stagnant = 0
    phase = 0
    while time.time() < deadline and stagnant < 5:
        T = T0 * (cooling ** phase)
        if T < T_min:
            T = T_min
        plateau = 0
        trials = 0
        phase_best = False
        while time.time() < deadline and plateau < 3000 and trials < 20000:
            i, j = rand_one_zero(h, n, rng)
            if i == j:
                continue
            c2, nm = binary_trial(c, h, om, i, j, n)
            trials += 1
            delta = nm - cm
            if delta < -1e-12:
                # Accept (improvement)
                accept = True
            elif T > T_min:
                # SA acceptance
                accept = rng.random() < np.exp(-delta / T)
            else:
                accept = False
            if accept:
                c = c2
                cm = nm
                h[i] = 0.0
                h[j] = 1.0
                om[n - 1 - i] = 1.0
                om[n - 1 - j] = 0.0
                plateau = 0
                if cm < bc:
                    best = h.copy()
                    bc = cm
                    phase_best = True
            else:
                plateau += 1
        phase += 1
        if phase_best:
            stagnant = 0
        else:
            stagnant += 1
        # Small random perturbation to escape
        if stagnant >= 2:
            for _ in range(4):
                i, j = rand_one_zero(h, n, rng)
                if i != j:
                    h[i] = 0.0
                    h[j] = 1.0
                    om[n - 1 - i] = 1.0
                    om[n - 1 - j] = 0.0
            c = np.correlate(h, 1.0 - h, mode="full")
            cm = float(c.max())
    return best, bc * dx

def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)
