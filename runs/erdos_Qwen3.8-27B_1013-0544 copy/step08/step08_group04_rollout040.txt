```
def make_grads(h, idx):
    n=len(h); T=idx.size; G=np.zeros((T,n)); center=n-1
    for a,p in enumerate(idx):
       s=int(p)-center
       g=G[a]
       if s==0:
          g += 1.0 - 2.0*h
       elif s>0:
          if s<n:
             g[:n-s] += 1.0 - h[s:]
             g[s:] -= h[:n-s]
       else:
          r=-s
          if r<n:
             g[r:] += 1.0 - h[:n-r]
             g[:n-r] -= h[r:]
    return G
```

```
def project_feasible(h,n):

          if tot>1e-15: h+=(diff/tot)*cap
          else:
```

```
def fourier_perturb(h,n,rng,amp=0.04,modes=8):
```

```python
import numpy as np
import time
from scipy.optimize import linprog

def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(3):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            cap = 1.0 - h
            tot = float(cap.sum())
            if tot > 1e-15:
                h += (diff / tot) * cap
            else:
                idx = np.where(h < 1.0)[0]
                if idx.size:
                    h[idx[0]] = 1.0
        else:
            tot = float(h.sum())
            if tot > 1e-15:
                h += (diff / tot) * h
            else:
                idx = np.where(h > 0.0)[0]
                if idx.size:
                    h[idx[0]] = 0.0
        h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if diff > 1e-14:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            h[idx[0]] = min(1.0, float(h[idx[0]] + diff))
    elif diff < -1e-14:
        idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = max(0.0, float(h[idx[0]] + diff))
    return h

def raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c(h, n):
    return float(np.max(raw_corr(h)) * (2.0 / n))

def as_even(n):
    n = int(n)
    if n % 2:
        n += 1
    return n

def resample_h(old, n):
    old = np.asarray(old, dtype=float)
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def default_h(n):
    x = np.linspace(0.0, 2.0, n)
    return project_feasible(0.5 + 0.22 * np.cos(np.pi * (x - 1.0)), n)

def fourier_perturb(h, n, rng, amp=0.04, modes=8):
    h2 = np.asarray(h, dtype=float).copy()
    x = (np.arange(n) + 0.5) * (2.0 / n)
    t = x - 1.0
    for k in range(1, modes + 1):
        h2 += (amp / k) * rng.uniform(-1.0, 1.0) * np.cos(k * np.pi * t)
        h2 += (0.5 * amp / k) * rng.uniform(-1.0, 1.0) * np.sin(k * np.pi * t)
    return project_feasible(h2, n)

def binary_pick(h, n, random=False, rng=None):
    h2 = np.zeros(n)
    if random:
        h2[rng.choice(n, n // 2, replace=False)] = 1.0
    else:
        k = n // 2
        idx = np.argpartition(h, n - k)[n - k:]
        h2[idx] = 1.0
    return h2

def coord_grid(hp, cseg, a, qconst, unaff, coarse=49, fine=33):
    xs = np.linspace(0.0, 1.0, coarse)
    d = xs - hp
    if a.size:
        vals = np.max(d[:, None] * a[None, :] + cseg[None, :], axis=1)
    else:
        vals = np.zeros(coarse)
    vals = np.maximum(vals, qconst + xs * (1.0 - xs))
    vals = np.maximum(vals, unaff)
    i = int(np.argmin(vals))
    x = float(xs[i])
    f = float(vals[i])
    fh = qconst + hp * (1.0 - hp)
    if cseg.size:
        m = float(np.max(cseg))
        if m > fh:
            fh = m
    if unaff > fh:
        fh = unaff
    if fh < f:
        x = float(hp)
        f = float(fh)
    span = 1.0 / (coarse - 1)
    for rad in (2.0, 0.6):
        lo = max(0.0, x - rad * span)
        hi = min(1.0, x + rad * span)
        if hi <= lo:
            continue
        xs2 = np.linspace(lo, hi, fine)
        d2 = xs2 - hp
        if a.size:
            v2 = np.max(d2[:, None] * a[None, :] + cseg[None, :], axis=1)
        else:
            v2 = np.zeros_like(xs2)
        v2 = np.maximum(v2, qconst + xs2 * (1.0 - xs2))
        v2 = np.maximum(v2, unaff)
        i2 = int(np.argmin(v2))
        f2 = float(v2[i2])
        x2 = float(xs2[i2])
        if f2 < f - 1e-13:
            x = x2
            f = f2
    return x, f

def cd_unconstrained(h, n, max_passes, deadline, project_every=10):
    h = project_feasible(h, n)
    c = raw_corr(h)
    cur = float(c.max())
    sc = 2.0 / n
    best_h = h.copy()
    best_c = cur * sc
    center = n - 1
    L = 2 * n - 1
    pe = max(3, int(project_every))
    for _ in range(int(max_passes)):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            hp = float(h[p])
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            seg = hi - lo + 1
            s = center - np.arange(lo, hi + 1, dtype=int)
            j = p + s
            i = p - s
            a = np.zeros(seg, dtype=float)
            vj = (j >= 0) & (j < n)
            if np.any(vj):
                a[vj] += 1.0 - h[j[vj]]
            vi = (i >= 0) & (i < n)
            if np.any(vi):
                a[vi] -= h[i[vi]]
            off = center - lo
            cseg = np.empty(seg - 1, dtype=float)
            asg = np.empty(seg - 1, dtype=float)
            if off:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center + 1:hi + 1]
                asg[off:] = a[off + 1:]
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = float(c[:lo].max())
            if hi + 1 < L:
                tmp = float(c[hi + 1:].max())
                if tmp > unaff:
                    unaff = tmp
            bx, bf = coord_grid(hp, cseg, asg, qconst, unaff)
            if bf < cur - 1e-10:
                dxv = bx - hp
                if off:
                    c[lo:center] += asg[:off] * dxv
                if hi > center:
                    c[center + 1:hi + 1] += asg[off:] * dxv
                c[center] = qconst + bx * (1.0 - bx)
                h[p] = bx
                cur = bf
                improved = True
            if (p + 1) % pe == 0:
                h = project_feasible(h, n)
                c = raw_corr(h)
                cur = float(c.max())
        h = project_feasible(h, n)
        c = raw_corr(h)
        cur = float(c.max())
        cval = cur * sc
        if cval < best_c - 1e-13:
            best_c = cval
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def make_grads(h, idx):
    n = len(h)
    T = len(idx)
    G = np.zeros((T, n), dtype=float)
    center = n - 1
    for a, p in enumerate(idx):
        s = int(p) - center
        g = G[a]
        if s == 0:
            g += 1.0 - 2.0 * h
        elif s > 0 and s < n:
            g[:n - s] += 1.0 - h[s:]
            g[s:] -= h[:n - s]
        elif s < 0:
            r = -s
            if r < n:
                g[r:] += 1.0 - h[:n - r]
                g[:n - r] -= h[r:]
    return G

def blend_search(h0, h1, n, points=13):
    h0 = project_feasible(h0, n)
    h1 = project_feasible(h1, n)
    best_h = h0.copy()
    best_c = compute_c(h0, n)
    best_a = 0.0
    for k in range(1, points + 1):
        a = float(k) / float(points)
        hp = project_feasible((1.0 - a) * h0 + a * h1, n)
        cp = compute_c(hp, n)
        if cp < best_c - 1e-14:
            best_c = cp
            best_h = hp.copy()
            best_a = a
    if 0.0 < best_a < 1.0:
        lo = max(0.0, best_a - 1.0 / points)
        hi = min(1.0, best_a + 1.0 / points)
        for k in range(1, 11):
            a = lo + (hi - lo) * float(k) / 10.0
            hp = project_feasible((1.0 - a) * h0 + a * h1, n)
            cp = compute_c(hp, n)
            if cp < best_c - 1e-14:
                best_c = cp
                best_h = hp.copy()
    return best_h, best_c

def lp_active_step(h0, n, K=150, dt=0.1, deadline=0.0):
    h0 = project_feasible(h0, n)
    c0 = raw_corr(h0)
    cmax = float(c0.max())
    best_h = h0.copy()
    best_c = cmax * (2.0 / n)
    lp_h = h0.copy()
    lp_c = best_c
    if time.time() > deadline:
        return best_h, best_c, lp_h, lp_c
    L = 2 * n - 1
    if int(K) <= 0:
        if n <= 700:
            idx = np.arange(L)
        else:
            idx = np.argsort(c0)[-int(min(300, L)):]
    else:
        kk = int(K)
        if kk >= L:
            idx = np.arange(L)
        else:
            idx = np.argsort(c0)[-kk:]
    G = make_grads(h0, idx)
    B = c0[idx] - G @ h0
    A = np.empty((idx.size, n + 1), dtype=float)
    A[:, :n] = G
    A[:, n] = -1.0
    b = -B
    obj = np.zeros(n + 1)
    obj[n] = 1.0
    Aeq = np.ones(n + 1)
    Aeq[n] = 0.0
    beq = np.array([0.5 * n])
    lb = np.maximum(0.0, h0 - dt)
    ub = np.minimum(1.0, h0 + dt)
    bounds = []
    for i in range(n):
        bounds.append((float(lb[i]), float(ub[i])))
    bounds.append((0.0, cmax + 10.0))
    try:
        if time.time() <= deadline:
            res = linprog(obj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq, bounds=bounds, method='highs')
            if res.x is not None and int(res.status) == 0 and np.all(np.isfinite(res.x[:n])):
                h1 = project_feasible(res.x[:n], n)
                c1 = compute_c(h1, n)
                if c1 < lp_c:
                    lp_h, lp_c = h1, c1
                bh, bc = blend_search(h0, h1, n)
                if bc < best_c:
                    best_h, best_c = bh, bc
    except Exception:
        pass
    return best_h, best_c, lp_h, lp_c

    pe = 7 if n <= 500 else (11 if n <= 800 else 17)

        if time.time() > deadline - 45 or c <= 0.38068:

        seed_h = bh if bc <= c1 else h1
        seed_c = bc if bc <= c1 else c1
        if seed_c < c:

        if seed_c < c + 0.015 and time.time() < deadline - 35:
            h2, c2 = cd_unconstrained(seed_h, n, 2, deadline - 35, pe)
            if c2 < c:

            elif seed_c < c:
                h, c = seed_h, seed_c

        h2, c2 = cd_unconstrained(h, n, int(final_passes), deadline - 25, 5)
        if c2 < c:
            h, c = h2, c2
    h = project_feasible(h, n)
    c = compute_c(h, n)
    return h, c
```
