def c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def proj(z, n):

    if z.size != n:

    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-8 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    lo = float(z.min()) - 2.0
    hi = float(z.max()) + 2.0
    for _ in range(60):
        tau = 0.5 * (lo + hi)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-14:
        room = (1.0 - y) if r > 0 else y
        m = room > 0
        if m.any():
            y[m] += r * room[m] / room.sum()
    y = np.clip(y, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-15:
        if r > 0:
            idx = int(np.argmax(1.0 - y))
            d = min(r, 1.0 - y[idx])

            r -= d

    return np.clip(y, 0.0, 1.0)


def sym(h, n):
    b = proj(h, n)
    bc = c5(b, n)
    for _ in range(3):
        cands = [0.5 * (b + b[::-1]), 1.0 - b, 0.5 * (b + 1.0 - b[::-1])]
        imp = False
        for y in cands:
            y = proj(y, n)
            v = c5(y, n)
            if v < bc:
                b, bc, imp = y, v, True
        if not imp:
            break
    return b, bc


def rstart(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.35 * np.cos(np.pi * (x - 1.0))
    for m in range(2, 9):

def soft(h, n, alpha):
    h = np.asarray(h, float)
    cc = np.correlate(h, 1.0 - h, mode='full')
    m = float(cc.max())
    z = alpha * (cc - m)
    e = np.exp(z - z.max())
    s = float(e.sum())
    p = e / s
    val = (m + math.log(s) / alpha) * (2.0 / n)
    return val, p, cc


def sval(h, n, alpha):
    return soft(h, n, alpha)[0]


def ggrad(cc, h, n, p):
    off = n - 1
    g = np.zeros(n, float)
    g += p[off] * (1.0 - 2.0 * h)
    for e in range(1, n):
        wp = float(p[off + e])
        wm = float(p[off - e])
        m = n - e
        if wp:
            g[e:] += wp * (1.0 - h[:m])
            g[:m] -= wp * h[e:]
        if wm:
            g[:m] += wm * (1.0 - h[e:])
            g[e:] -= wm * h[:m]
    return (2.0 / n) * g


def sgrad(h, n, alpha):
    _, p, cc = soft(h, n, alpha)
    return ggrad(cc, h, n, p)


def sfun(x, n):
    return float(np.asarray(x, float).sum() - 0.5 * n)


def sjac(x, n):
    return np.ones(np.asarray(x, float).shape, dtype=float)


def slsqp(h, n, deadline, plans):
    h = proj(h, n)
    best = h.copy()
    bc = c5(best, n)
    work = best
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac, 'args': (n,)}]
    for alpha, mi in plans:
        if time.time() > deadline - 1.0:
            break
        try:
            res = minimize(sval, work, args=(n, alpha), method='SLSQP',
                           jac=sgrad, bounds=bounds, constraints=cons,
                           options={'maxiter': mi, 'ftol': 1e-11})
            x = proj(res.x, n)
            v = c5(x, n)
        except Exception:
            continue
        if v < bc:
            best, bc, work = x, v, x
        elif v <= bc + 1e-8:
            work = x
    return best, bc


def add_der(D, h, n, p, s):
    off = n - 1
    D[off:off + p + 1] += s * (1.0 - h[p::-1])
    m = n - p
    D[off:off + m] += s * (-h[p:n])
    la = n - 1 - p
    if la > 0:
        D[off - la:off] += s * (1.0 - h[p + 1:n][::-1])
    if p > 0:
        D[off - p:off] += s * (-h[0:p])


def pB(h, n, i, j):
    B = np.zeros(2 * n - 1, float)
    add_der(B, h, n, i, 1.0)
    add_der(B, h, n, j, -1.0)
    return B


def pgrid(cc, B, n, e, us):
    v = cc[None, :] + us[:, None] * B[None, :]
    off = n - 1
    u2 = us * us
    v[:, off] -= 2.0 * u2
    if e > 0:
        v[:, off + e] += u2
        v[:, off - e] += u2
    return v.max(axis=1)


def peval(cc, B, n, e, u):
    v = cc + u * B
    off = n - 1
    u2 = u * u
    v[off] -= 2.0 * u2
    if e > 0:
        v[off + e] += u2
        v[off - e] += u2
    return float(v.max())


def pgen(cc, B, n, e, a, b, it):
    if b <= a + 1e-14:
        return a, peval(cc, B, n, e, a)
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    x1 = b - gr * (b - a)
    x2 = a + gr * (b - a)
    f1 = peval(cc, B, n, e, x1)
    f2 = peval(cc, B, n, e, x2)

            x2 = x1
            f2 = f1

        dl = min(hard - 60.0, time.time() + 320.0)
        best, bc = psearch(best, n0, dl, rng, 41)
    best, bc = sym(best, n0)
    cands = []
    if n0 < 800:
        cands.append(800)
    if n0 > 250:
        cands.append(250)
    for m in cands:
        if time.time() > hard - 110.0:
            break
        rem = hard - time.time()
        if m > 700 and rem < 140.0:
            continue
        if m < 300 and rem < 90.0:
            continue
        ht = inter(best, m)
        ht, ct = sym(ht, m)
        if ct < bc:
            best, bc, n0 = ht, ct, m
        cand = ht.copy()
        mc = len(cand)
        if time.time() < hard - 70.0:
            if mc <= 600 and time.time() < hard - 85.0:
                cand, ct = slsqp(cand, mc, min(hard - 85.0, time.time() + 25.0),
                                 [(20, 30), (80, 20), (300, 10)])
            pt = 60.0 if mc <= 300 else (110.0 if mc > 650 else 90.0)
            gp = 35 if mc <= 650 else 31
            cand, ct = psearch(cand, mc, min(hard - 55.0, time.time() + pt), rng, gp)
            cand, ct = sym(cand, mc)
            if ct < bc:
                best, bc, n0 = cand, ct, mc
    for s in (0.02, 0.05, 0.10):
        if time.time() > hard - 110.0:
            break
        n = len(best)
        hp = proj(best + rng.normal(0.0, s, n), n)
        hp, cp = sym(hp, n)
        if time.time() < hard - 60.0:
            hp, cp = psearch(hp, n, min(hard - 60.0, time.time() + 45.0), rng, 35)
            hp, cp = sym(hp, n)
            if cp < bc:
                best, bc, n0 = hp, cp, n
    if time.time() < hard - 3.0:
        n = len(best)
        gp = 51 if n <= 650 else 35
        best, bc = psearch(best, n, hard - 3.0, rng, gp)
        best, bc = sym(best, n)
    n = len(best)
    best = proj(best, n)
    bc = c5(best, n)
    return best, bc, n
