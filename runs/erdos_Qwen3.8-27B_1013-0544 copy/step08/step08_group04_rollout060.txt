import numpy as np

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            cap = 1.0 - h
            tot = cap.sum()
            if tot > 0:
                h += (diff / tot) * cap
        else:
            tot = h.sum()
            if tot > 0:
                h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - h.sum()
    if res > 1e-12:
        i = int(np.argmin(h))
        h[i] = min(1.0, h[i] + res)
    elif res < -1e-12:
        i = int(np.argmax(h))
        h[i] = max(0.0, h[i] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * (2.0 / n))

def fourier_adam(h_init, n, K, deadline, seed=42, lr=0.05):
