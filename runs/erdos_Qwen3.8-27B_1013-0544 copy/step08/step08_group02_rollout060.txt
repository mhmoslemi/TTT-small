def compute_c5(h, n):

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(50):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)

            m = n - d

    z = alpha * (c - m)
    zmax = float(z.max())

    # dc/dh for each k, weighted by p[k]

    g = np.zeros(n, dtype=float)

        else:

            m = n - e
            g[:m] += w * (1.0 - h[e:])
            g[e:] -= w * h[:m]
    g *= (2.0 / n)
    return val, g

def adam_optimize(h, n, alpha, steps, lr, deadline):
    """Adam optimization of smooth max."""
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(1, steps + 1):
        if time.time() > deadline:
            break
        val, g = smooth_max_val_grad(h, n, alpha)
        m_t = beta1 * m_t + (1 - beta1) * g
        v_t = beta2 * v_t + (1 - beta2) * g * g
        m_hat = m_t / (1 - beta1 ** t)
        v_hat = v_t / (1 - beta2 ** t)

        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h = new_h.copy()
            best_c = new_c
        h = new_h
    return best_h, best_c

                h[j] += best_u

import numpy as np
import time

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(50):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def col_for_index(h, n, idx):
    col = np.zeros(2 * n - 1, dtype=float)
    off = n - 1
    for d in range(-(n - 1), n):
        k = off + d
        if d >= 0:
            if idx <= n - 1 - d:
                col[k] += 1.0 - h[idx + d]
            if d > 0 and idx >= d:
                col[k] -= h[idx - d]
        else:
            e = -d
            if idx >= e:
                col[k] += 1.0 - h[idx - e]
            if idx <= n - 1 - e:
                col[k] -= h[idx + e]
    return col

def lag_gradient(h, n, d):
    gg = np.zeros(n, dtype=float)
    if d >= 0:
        m = n - d
        gg[:m] += 1.0 - h[d:]
        if d > 0:
            gg[d:] -= h[:m]
    else:
        e = -d
        m = n - e
        gg[:m] += 1.0 - h[e:]
        gg[e:] -= h[:m]
    return gg

def smooth_val_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zmax = float(z.max())
    e = np.exp(z - zmax)
    s = float(e.sum())
    p = e / s
    val = (m + np.log(s) / alpha) * (2.0 / n)
    off = n - 1
    g = np.zeros(n, dtype=float)
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-14:
            continue
        d = k - off
        gg = lag_gradient(h, n, d)
        g += w * gg
    g *= (2.0 / n)
    return val, g

def adam_optimize(h, n, alpha, steps, lr, deadline):
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(1, steps + 1):
        if time.time() > deadline:
            break
        _, g = smooth_val_grad(h, n, alpha)
        m_t = b1 * m_t + (1 - b1) * g
        v_t = b2 * v_t + (1 - b2) * g * g
        mh = m_t / (1 - b1 ** t)
        vh = v_t / (1 - b2 ** t)
        step = lr * mh / (np.sqrt(vh) + eps)
        new_h = project_feasible(h - step, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h = new_h.copy()
            best_c = new_c
        h = new_h
    return best_h, best_c

        gg = np.zeros(n, dtype=float)
        for kk in idxs:
            d = int(kk) - off
            gg += lag_gradient(h, n, d)
        pos_mask = gg > 0.005

        pos_idx = np.flatnonzero(pos_mask)

            col_i = col_for_index(h, n, i)
            col_j = col_for_index(h, n, j)
            B = col_j - col_i
            u_max = min(h[i], 1.0 - h[j])
            if u_max <= 1e-12:

            best_u = 0.0

                new_c[off] -= 2.0 * u * u
                ki = off + i - j
                kj = off + j - i
                u2 = u * u
                if 0 <= ki < 2 * n - 1:
                    new_c[ki] += u2
                if 0 <= kj < 2 * n - 1:
                    new_c[kj] += u2

                    best_nm = nm
                    best_u = u
            if best_u > 0 and best_nm < cur_max - 1e-11:

                no_improve = 0
                found = True
                break
        if not found:
            no_improve += 1
            if no_improve > max_no_imp:
                break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)
