def exact_coord_cd(h, n, max_passes=8, deadline=None):
    """Coordinate descent using subgradient bisection for the exact 1D subproblem."""
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        cur_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            v = float(x[p])
            # Compute A[m] for this p
            A = np.zeros(L)
            # Part 1: (1-h[p+m-(n-1)]) for valid p+m-(n-1)
            m1s = max(0, n - 1 - p)
            m1e = min(L - 1, 2 * n - 2 - p)
            if m1s <= m1e:
                idx = np.arange(m1s, m1e + 1)
                h_idx = p + idx - diag
                A[m1s:m1e+1] += (1.0 - x[h_idx])
            # Part 2: -h[p-m+(n-1)] for valid p-m+(n-1)
            m2s = max(0, p - (n - 1))
            m2e = min(L - 1, p + (n - 1))
            if m2s <= m2e:
                idx = np.arange(m2s, m2e + 1)
                h_idx = p - idx + diag
                A[m2s:m2e+1] += (-x[h_idx])
            
            # Find best d in [-v, 1-v] minimizing max_m(c[m] + d*A[m])
            lo_d = -v
            hi_d = 1.0 - v
            best_d = 0.0
            best_val = cur_max
            # Subgradient bisection
            for _ in range(40):
                mid_d = 0.5 * (lo_d + hi_d)
                vals = c + mid_d * A
                peak = float(vals.max())
                if peak < best_val:
                    best_val = peak
                    best_d = mid_d
                active = vals >= vals.max() - 1e-10
                slopes = A[active]
                if slopes.min() > 0:
                    hi_d = mid_d
                elif slopes.max() < 0:
                    lo_d = mid_d
                else:
                    break
            # Check endpoints too
            for d_check in [lo_d, hi_d, 0.0]:
                vals = c + d_check * A
                peak = float(vals.max())
                if peak < best_val:
                    best_val = peak
                    best_d = d_check
            # Verify with exact (including -d^2 at diag)
            d = best_d
            vals_exact = c + d * A
            vals_exact[diag] -= d * d
            exact_peak = float(vals_exact.max())
            if exact_peak < cur_max - 1e-10:
                # Apply
                x[p] = v + d
                c = vals_exact  # reuse
                cur_max = exact_peak
                improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - 1e-12:
            best, bc = xp, cp
        if not improved:
            break
    return best, bc

def binary_steepest(h_init, n, deadline, max_rounds=50):
    """Steepest descent in binary space: at each step, apply the single best swap."""
    if n % 2:
        return None, np.inf
    target = n // 2
    # Binarize: take top n/2 values
    idx = np.argpartition(h_init, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    best = b.copy()
    bc = np.correlate(best, 1.0 - best, mode="full")
    bcm = float(bc.max())
    dx = 2.0 / n
    L = 2 * n - 1
    diag = n - 1
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        h = best.copy()
        c = np.correlate(h, 1.0 - h, mode="full")
        cm = float(c.max())
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # For each pair (i in ones, j in zeros), compute new max
        # Vectorized: for each i, compute the effect of removing i and adding each j
        # Precompute B for all positions
        # B[p] = change in c when h[p] goes from 0 to 1 (i.e., d=1)
        # Actually, let me compute the effect directly.
        
        # When we swap i (1→0) and j (0→1):
        # Δc[m] = -A_i[m] + A_j[m] where A_p is the coefficient for position p
        # New c[m] = c[m] - A_i[m] + A_j[m]
        # New max = max_m(c[m] - A_i[m] + A_j[m])
        
        # For all i in ones: compute c_i = c - A_i (shape L)
        # For all j in zeros: compute A_j (shape L)
        # new_max[i,j] = max_m(c_i[i] + A_j[j])
        
        # This is: for each i, for each j: max(c_i + A_j)
        # = max of diagonal-like structure
        
        # Efficient approach: 
        # A_all = precomputed B matrix (n x L)
        # For i in ones: c_i = c - A_all[i]
        # For j in zeros: new_maxes = max(c_i[None,:] + A_all[zeros][:], axis=1)
        
        # Precompute A_all
        A_all = np.zeros((n, L), dtype=float)
        for p in range(n):
            m1s = max(0, n - 1 - p)
            m1e = min(L - 1, 2 * n - 2 - p)
            if m1s <= m1e:
                h_idx = p + np.arange(m1s, m1e + 1) - diag
                A_all[p, m1s:m1e+1] += (1.0 - h[h_idx])
            m2s = max(0, p - (n - 1))
            m2e = min(L - 1, p + (n - 1))
            if m2s <= m2e:
                h_idx = p - np.arange(m2s, m2e + 1) + diag
                A_all[p, m2s:m2e+1] += (-h[h_idx])
        
        # Now for each i in ones, find best j in zeros
        best_swap = None
        best_val = cm
        A_zeros = A_all[zeros]  # shape (num_zeros, L)
        for i in ones:
            c_i = c - A_all[i]  # shape (L,)
            # new_maxes[j] = max_m(c_i + A_zeros[j])
            new_maxes = np.max(c_i[None, :] + A_zeros, axis=1)
            j_idx = int(np.argmin(new_maxes))
            if new_maxes[j_idx] < best_val:
                best_val = new_maxes[j_idx]
                best_swap = (int(i), int(zeros[j_idx]))
        
        if best_swap is None:
            break
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        if best_val < bcm:
            best = h.copy()
            bcm = best_val
        # Check convergence
        if best_val >= cm - 1e-10:
            break
    
    return best, bcm * dx

def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)

import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.clip(h + mid, 0.0, 1.0).sum() < t:
            lo = mid
        else:
            hi = mid
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
    y = np.clip(y, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += d / mask.sum()
    return y

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx

def smooth_obj(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]

def sum_con(x, n):
    return x.sum() - 0.5 * n

def sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)

def slsqp_polish(h, n, alphas, maxiter, deadline):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a), jac=smooth_grad,
                           method="SLSQP", bounds=bounds, constraints=cons,
                           options={"maxiter": int(maxiter), "ftol": 1e-12})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-13:
            best, bc = nh, nc
    return best, bc

def exact_coord_cd(h, n, max_passes=10, deadline=None):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        cur_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            v = float(x[p])
            A = np.zeros(L)
            m1s = max(0, n - 1 - p)
            m1e = min(L - 1, 2 * n - 2 - p)
            if m1s <= m1e:
                h_idx = p + np.arange(m1s, m1e + 1) - diag
                A[m1s:m1e + 1] += (1.0 - x[h_idx])
            m2s = max(0, p - (n - 1))
            m2e = min(L - 1, p + (n - 1))
            if m2s <= m2e:
                h_idx = p - np.arange(m2s, m2e + 1) + diag
                A[m2s:m2e + 1] += (-x[h_idx])
            lo_d, hi_d = -v, 1.0 - v
            best_d, best_val = 0.0, cur_max
            for _ in range(45):
                mid_d = 0.5 * (lo_d + hi_d)
                vals = c + mid_d * A
                peak = float(vals.max())
                if peak < best_val:
                    best_val, best_d = peak, mid_d
                active = vals >= vals.max() - 1e-10
                slopes = A[active]
                if len(slopes) == 0:
                    break
                if slopes.min() > 1e-12:
                    hi_d = mid_d
                elif slopes.max() < -1e-12:
                    lo_d = mid_d
                else:
                    break
            for d_check in [max(0.0, lo_d), min(1.0, hi_d)]:
                vals = c + d_check * A
                peak = float(vals.max())
                if peak < best_val:
                    best_val, best_d = peak, d_check
            d = best_d
            vals_exact = c + d * A
            vals_exact[diag] -= d * d
            exact_peak = float(vals_exact.max())
            if exact_peak < cur_max - 1e-10:
                x[p] = v + d
                c = vals_exact
                cur_max = exact_peak
                improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - 1e-12:
            best, bc = xp, cp
        if not improved:
            break
    return best, bc

def compute_A_all(h, n):
    L = 2 * n - 1
    diag = n - 1
    A_all = np.zeros((n, L), dtype=float)
    for p in range(n):
        m1s = max(0, n - 1 - p)
        m1e = min(L - 1, 2 * n - 2 - p)
        if m1s <= m1e:
            h_idx = p + np.arange(m1s, m1e + 1) - diag
            A_all[p, m1s:m1e + 1] += (1.0 - h[h_idx])
        m2s = max(0, p - (n - 1))
        m2e = min(L - 1, p + (n - 1))
        if m2s <= m2e:
            h_idx = p - np.arange(m2s, m2e + 1) + diag
            A_all[p, m2s:m2e + 1] += (-h[h_idx])
    return A_all

def binary_steepest(h_init, n, deadline, max_rounds=60, rng=None):
    if n % 2:
        return None, np.inf
    target = n // 2
    idx = np.argpartition(h_init, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    best = b.copy()
    bc = np.correlate(best, 1.0 - best, mode="full")
    bcm = float(bc.max())
    dx = 2.0 / n
    stagnant = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        h = best.copy()
        c = np.correlate(h, 1.0 - h, mode="full")
        cm = float(c.max())
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        A_all = compute_A_all(h, n)
        A_zeros = A_all[zeros]
        best_swap, best_val = None, cm
        for i in ones:
            c_i = c - A_all[i]
            new_maxes = np.max(c_i[None, :] + A_zeros, axis=1)
            j_idx = int(np.argmin(new_maxes))
            if new_maxes[j_idx] < best_val:
                best_val = float(new_maxes[j_idx])
                best_swap = (int(i), int(zeros[j_idx]))
        if best_swap is None or best_val >= cm - 1e-10:
            stagnant += 1
            if stagnant >= 3:
                break
            if rng is not None:
                i, j = int(rng.choice(ones)), int(rng.choice(zeros))
                h[i], h[j] = 0.0, 1.0
            else:
                break
        else:
            i, j = best_swap
            h[i], h[j] = 0.0, 1.0
            stagnant = 0
            if best_val < bcm:
                best, bcm = h.copy(), best_val
    return best, bcm * dx

def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)
