def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = cap.sum()
        if tot > 0:
            h += (diff / tot) * cap
    else:
        tot = h.sum()
        if tot > 0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size:
                h[idx[0]] = min(1.0, h[idx[0]] + res)
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size:
                h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h
