def coord_sweep(h, n, c, order):

    h = h.copy()

        B = c - A * h[i]

            h[i] = opt
            total_imp += abs(delta)
    return h, c, total_imp

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(z - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - y.sum()
    if abs(res) > 1e-12:
        mask = (y < 1.0) if res > 0 else (y > 0.0)
        if mask.any():
            room = (1.0 - y) if res > 0 else y
            tot = room.sum()
            if tot > 1e-12:
                y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def get_coeff(h, n, i):
    d = np.arange(-(n - 1), n)
    ip = i + d
    im = i - d
    A = np.zeros(2 * n - 1)
    v1 = (ip >= 0) & (ip < n)
    v2 = (im >= 0) & (im < n)
    A[v1] += 1.0 - h[ip[v1]]
    A[v2] -= h[im[v2]]
    return A

def solve_1d(A, B, lo, hi):
    if lo >= hi:
        return 0.5 * (lo + hi)
    a, b = lo, hi
    for _ in range(45):
        mid = 0.5 * (a + b)
        if b - a < 1e-13:
            break
        vals = A * mid + B
        fv = float(np.max(vals))
        tol = 1e-10 * (1.0 + abs(fv))
        active = vals >= fv - tol
        if not active.any():
            active = (vals == fv)
        smin = float(np.min(A[active]))
        smax = float(np.max(A[active]))
        if smin > 0:
            b = mid
        elif smax < 0:
            a = mid
        else:
            break
    return 0.5 * (a + b)

def cd_sweep(h, n, c, order):
    h_new = h.copy()
    c_new = c.copy()
    imp = 0.0
    for i in order:
        A = get_coeff(h_new, n, i)
        B = c_new - A * h_new[i]
        opt = solve_1d(A, B, 0.0, 1.0)
        d = opt - h_new[i]
        if abs(d) > 1e-12:
            c_new += A * d
            h_new[i] = opt
            imp += abs(d)
    return h_new, c_new, imp

def cd_run(h, n, deadline, rng, max_sweeps=20000):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    best = compute_c5(h, n)
    order = np.arange(n)
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        if sw % 7 == 0:
            rng.shuffle(order)
        h, c, imp = cd_sweep(h, n, c, order)
        if sw % 30 == 29:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            cur = compute_c5(h, n)
            if cur < best:
                best = cur
        if imp < 1e-11:
            break
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    best = min(best, compute_c5(h, n))
    return h, best

def pair_cd(h, n, deadline, rng, n_pairs=5000):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best = compute_c5(h, n)
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        hi, hj = h[i], h[j]
        s = hi + hj
        lo = max(0.0, s - 1.0)
        hib = min(1.0, s)
        if hib - lo < 1e-14:
            continue
        Ai = get_coeff(h, n, i)
        Aj = get_coeff(h, n, j)
        diff = Ai - Aj
        D = Aj * s + c - Ai * hi - Aj * hj
        opt = solve_1d(diff, D, lo, hib)
        d = opt - hi
        if abs(d) > 1e-12:
            c += diff * d
            h[i] = opt
            h[j] = s - opt
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_fn(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = z.max()
    e = np.exp(z - zm)
    return (m + np.log(e.sum()) / alpha) * (2.0 / n)

def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = z.max()
    e = np.exp(z - zm)
    p = e / e.sum()
    g = np.zeros(n)
    ir = np.arange(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-14:
            continue
        d = k - (n - 1)
        gp = np.zeros(n)
        ip = ir + d
        im = ir - d
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        gp[vp] += 1.0 - h[ip[vp]]
        gp[vm] -= h[im[vm]]
        g += w * gp
    return (2.0 / n) * g

def slsqp_polish(h, n, deadline):
    h = project_feasible(h, n)
    best = compute_c5(h, n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - 0.5 * n, 'jac': lambda x: np.ones(n)}]
    for alpha in [30, 150, 800, 3000]:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_fn, h, args=(n, alpha), method='SLSQP',
                          jac=smooth_grad, bounds=bnds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-12, 'disp': False})
            x = project_feasible(np.asarray(res.x, dtype=float), n)
            cc = compute_c5(x, n)
            if cc < best:
                h, best = x, cc
        except Exception:
            pass
    return h, best

def try_symmetries(h, n):
    best, bc = h.copy(), compute_c5(h, n)
    ht = project_feasible(h[::-1].copy(), n)
    if compute_c5(ht, n) < bc:
        best, bc = ht, compute_c5(ht, n)
    ht = project_feasible(1.0 - h, n)
    ct = compute_c5(ht, n)
    if ct < bc:
        best, bc = ht, ct
    ht = project_feasible(0.5 * (h + h[::-1]), n)
    ct = compute_c5(ht, n)
    if ct < bc:
        best, bc = ht, ct
    return best, bc

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)
