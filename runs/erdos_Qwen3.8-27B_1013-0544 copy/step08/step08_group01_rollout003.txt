Actually, for the Frank-Wolfe / cutting plane approach, the standard approach is:
1. Solve the LP to get h_new
2. Do a line search: find alpha in [0,1] that minimizes f((1-alpha)*h_old + alpha*h_new)
3. Update h = (1-alpha)*h_old + alpha*h_new

This guarantees monotone decrease of the objective (at least for the linearized part).

Now, for the coordinate descent, let me think about how to make it faster. The key bottleneck is computing the effect of changing h[p] on all lags. For each p, the relevant lags are those m where both i=p and j=p±offset are valid. The offset is m-(n-1), so the valid m range is [p - (n-1), p + (n-1)] ∩ [0, 2n-2].

For a given p, the number of affected lags is at most 2*min(p, n-1-p) + 2 or so. On average it's about n.

A vectorized approach: for each p, compute the effect of changing h[p] by d on all lags simultaneously.

c[m] as a function of h[p] = h0[p] + d:
- For lags where p appears as the "first index": c[m] += d * (1-h[p+m-(n-1)])
- For lags where p appears in the "(1-h)" part: c[m] += -d * h[p-m+(n-1)]

So the change in c[m] is: d * (1-h[p+m-(n-1)]) - d * h[p-m+(n-1)] for valid indices.

The new max is: max(max of unaffected lags, max of (c[m] + change[m]) for affected m).

This is what the previous code was doing. The question is whether we can speed it up.

import numpy as np
from scipy.optimize import minimize, golden
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n, max_iter=2000):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    shifted = alpha * c
    mx = shifted.max()
    lse = (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return lse * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-19})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def lp_cutting_plane(h, n, top_count=20, max_iters=25):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(max_iters):
        c = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
        top_idx = np.argsort(c)[-top_count:]
        h_var = cp.Variable(n)
        t = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2]
        for m in top_idx:
            offset = m - (n - 1)
            grad = np.zeros(n)
            for i in range(n):
                j = i + offset
                if 0 <= j < n:
                    grad[i] += (1.0 - best_h[j])
                j2 = i - offset
                if 0 <= j2 < n:
                    grad[i] -= best_h[j2]
            constraints.append(c[m] + dx * cp.sum(cp.multiply(grad, h_var - best_h)) <= t)
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            break
        if h_var.value is None:
            break
        h_lp = project_feasible(np.clip(h_var.value, 0, 1), n)
        best_alpha = 0.0
        best_line_c = best_c
        for alpha in np.linspace(0, 1, 21):
            h_try = project_feasible((1 - alpha) * best_h + alpha * h_lp, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_line_c:
                best_line_c = c_try
                best_alpha = alpha
        if best_alpha > 0:
            best_h = project_feasible((1 - best_alpha) * best_h + best_alpha * h_lp, n)
            best_c = best_line_c
        else:
            break
    return best_h, best_c

def coord_cd(h, n, max_passes=200, deadline=None, grid_size=80):
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo, hi = -h0, 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals = np.array([obj_1d(d, c_aff, a_coeff, diag_local, unaff_max) for d in d_grid])
            best_idx = int(np.argmin(vals))
            best_d = d_grid[best_idx]
            best_val = float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                gl, gh = d_grid[best_idx - 1], d_grid[best_idx + 1]
                try:
                    res = golden(obj_1d, gl, gh,
                                 args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=60)
                    if res.fun < best_val:
                        best_d, best_val = float(res.x), float(res.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def subgradient_polish(h, n, max_iter=200, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.02
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_norm = c * dx
        top_k = 3
        top_idx = np.argsort(c_norm)[-top_k:]
        weights = np.exp(100 * (c_norm[top_idx] - c_norm[top_idx].max()))
        weights /= weights.sum()
        sg = np.zeros(n)
        for m in top_idx:
            offset = m - (n - 1)
            for i in range(n):
                j = i + offset
                if 0 <= j < n:
                    sg[i] += (1.0 - h[j])
                j2 = i - offset
                if 0 <= j2 < n:
                    sg[i] -= h[j2]
        h_new = project_feasible(h - lr * sg * dx, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            h, best_c, stall = h_new, c_new, 0
        else:
            lr *= 0.8
            stall += 1
            if stall > 40 or lr < 1e-12:
                break
    return h, best_c
