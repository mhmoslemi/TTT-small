import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_correlation(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def multi_lag_gradient(h, n, num_lags=3):
    """Compute gradient of sum of top num_lags overlaps."""
    c = compute_correlation(h, n)
    # Get top lags
    top_indices = np.argsort(c)[-num_lags:]
    m_star = top_indices
    sg = np.zeros(n)
    for k in m_star:
        d = k - (n - 1)
        # c[k] = sum_i h[i] * (1-h[i+d]) for valid i
        # d c[k]/d h[i] = (1-h[i+d]) - h[i-d] (for valid indices)
        for i in range(n):
            term1 = 0.0
            if 0 <= i + d < n:
                term1 = 1.0 - h[i + d]
            term2 = 0.0
            if 0 <= i - d < n:
                term2 = h[i - d]
            sg[i] += term1 - term2
    return sg, c

def focused_descent(h, n, max_iter=500, deadline=None, num_lags=3, lr_init=0.01):
    """Descent focusing on the worst lags."""
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    stall = 0
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        c = compute_correlation(h, n)
        # Get top lags with some softness
        sorted_c = np.sort(c)[::-1]
        # Use top num_lags
        threshold = sorted_c[min(num_lags, len(c)) - 1]
        active = c >= threshold
        
        # Compute gradient for active lags
        d_weight = np.zeros(n)  # gradient contribution
        for k in np.where(active)[0]:
            d = k - (n - 1)
            for i in range(n):

                if 0 <= i + d < n:
                    term1 = 1.0 - h[i + d]

                if 0 <= i - d < n:

        c_new = compute_c5(h_new, n)

    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, k):
    """Gradient of c[k] w.r.t. h."""
    d = k - (n - 1)
    sg = np.zeros(n)
    if d == 0:
        sg = 1.0 - 2.0 * h
    elif d > 0:
        sg[:n - d] += (1.0 - h[d:])
        sg[d:] -= h[:n - d]
    else:
        dd = -d
        sg[dd:] += (1.0 - h[:n - dd])
        sg[:n - dd] -= h[dd:]
    return sg

def multilag_descent(h, n, max_iter=2000, deadline=None, num_lags=5):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.005
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        # Select top lags
        top_idx = np.argpartition(c, -num_lags)[-num_lags:]
        sg = np.zeros(n)
        for k in top_idx:
            sg += lag_gradient(h, n, k)
        sg /= num_lags
        h_new = project_feasible(h - lr * sg, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            h = h_new
            best_c = c_new
            stall = 0
        else:
            lr *= 0.85
            stall += 1
            if stall > 100 or lr < 1e-14:
                break
    return h, best_c

def coord_cd(h, n, max_passes=100, deadline=None, grid_size=60):
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            # Find affected correlation indices
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            jf_arr = p + ms - (n - 1)
            jb_arr = p - ms + (n - 1)
            valid_f = (jf_arr >= 0) & (jf_arr < n)
            valid_b = (jb_arr >= 0) & (jb_arr < n)
            if np.any(valid_f):
                a_coeff[valid_f] += 1.0 - h[jf_arr[valid_f]]
            if np.any(valid_b):
                a_coeff[valid_b] -= h[jb_arr[valid_b]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            # Local 1D optimization
            # c_aff + d*a_coeff (with diagonal correction for the h[p]*(1-h[p]) term)
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo = -h0
            hi = 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals = np.empty(len(d_grid))
            for gi, d in enumerate(d_grid):
                nv = c_aff + d * a_coeff
                if diag_local is not None:
                    nv[diag_local] -= d * d
                vals[gi] = max(float(nv.max()), unaff_max)
            best_idx = int(np.argmin(vals))
            best_d = d_grid[best_idx]
            best_val = float(vals[best_idx])
            # Refine with golden section if interior
            if 0 < best_idx < len(d_grid) - 1:
                gl = d_grid[best_idx - 1]
                gh = d_grid[best_idx + 1]
                try:
                    def obj_1d(d):
                        nv = c_aff + d * a_coeff
                        if diag_local is not None:
                            nv = nv.copy()
                            nv[diag_local] -= d * d
                        return max(float(nv.max()), unaff_max)
                    res = golden(obj_1d, gl, gh, maxiter=40)
                    if res.fun < best_val:
                        best_d = float(res.x)
                        best_val = float(res.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-15:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Multi-lag gradient descent
    if time.time() < deadline - 300:
        h1, c1 = multilag_descent(best_h, n, 3000, deadline, num_lags=3)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h1, c1 = multilag_descent(best_h, n, 3000, deadline, num_lags=5)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h1, c1 = multilag_descent(best_h, n, 3000, deadline, num_lags=10)
        if c1 < best_c:
            best_h, best_c = h1, c1
    
    # Phase 2: Coordinate descent
    if time.time() < deadline - 200:
        h2, c2 = coord_cd(best_h, n, 200, deadline, grid_size=80)
        if c2 < best_c:
            best_h, best_c = h2, c2
    
    # Phase 3: More coordinate descent passes
    if time.time() < deadline - 150:
        h2, c2 = coord_cd(best_h, n, 200, deadline, grid_size=100)
