    # Part B: h[k] appears in second arg at position k (as 1-h[k])
    # (1-h[k]) appears in c[p] when i=k
    # c[p] has term h[k+m]*(1-h[k]) where m=p-(n-1)
    # For validity: 0 ≤ k+m < n → 0 ≤ k+(p-(n-1)) < n → n-1-k ≤ p ≤ 2n-2-k
    p_lo_b = max(0, n-1-k)
    p_hi_b = min(2*n-2, 2*n-2-k)
    for p in range(p_lo_b, p_hi_b + 1):
        j = k + (p - (n-1))
        # Before: h[j]*(1-h[k]) = h[j]*0 = 0 (since h[k]=1)
        # After: h[j]*(1-h'[k]) = h[j]*1 = h[j]. Change: +h[j]
        effect[p] += h[j]
    
    return effect

def flip_add_effect(k, h, n):
    """Effect on c of changing h[k] from 0 to 1."""
    # This is the negative of flip_remove in "reverse"
    # Before: h[k]=0, so 1-h[k]=1
    # After: h[k]=1, so 1-h[k]=0
    effect = np.zeros(2*n-1)
    
    # Part A: h[k] in first arg (was 0, now 1)
    # c[p] term h[i+m]*(1-h[i]): h[k] appears when i+m=k
    # Before: 0*(1-h[i]) = 0. After: 1*(1-h[i]) = 1-h[i]. Change: +(1-h[i])
    p_lo = max(0, k)
    p_hi = min(2*n-2, k + n - 1)
    for p in range(p_lo, p_hi + 1):
        i = k - (p - (n-1))
        effect[p] += (1 - h[i])
    
    # Part B: h[k] in second arg (was 1-h[k]=1, now 1-h[k]=0)
    # c[p] term h[j]*(1-h[k]): Before: h[j]*1 = h[j]. After: h[j]*0 = 0. Change: -h[j]
    p_lo_b = max(0, n-1-k)
    p_hi_b = min(2*n-2, 2*n-2-k)
    for p in range(p_lo_b, p_hi_b + 1):
        j = k + (p - (n-1))
        effect[p] -= h[j]
    
    return effect
```

    c = compute_corr(h)
    dx = 2.0 / n
    current_c5 = float(c.max()) * dx
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for round_num in range(max_rounds):
        if time.time() > deadline:
            break
        
        # Find the bottleneck lag
        p_star = int(np.argmax(c))
        m_star = p_star - (n - 1)
        
        # For each zero position j, compute score_j = h[j-m*] + h[j+m*] (with boundary handling)
        # For each one position k, compute score_k = h[k-m*] + h[k+m*]
        # We want to swap: choose j with high score_j and k with low score_k
        
        def boundary_sum(pos, m):
            """Compute h[pos-m] + h[pos+m] with h=0 outside [0,n-1]."""
            s = 0.0
            if 0 <= pos - m < n:
                s += h[pos - m]
            if 0 <= pos + m < n:
                s += h[pos + m]
            return s
        
        # Compute scores for all zeros and ones
        # This is O(n) each
        zero_scores = np.array([boundary_sum(j, m_star) for j in zeros])
        one_scores = np.array([boundary_sum(k, m_star) for k in ones])
        
        # Try the best few pairs
        best_improvement = 0
        best_pair = None
        
        # Sort: best zeros (highest score) and best ones (lowest score)
        top_zeros = np.argsort(-zero_scores)[:min(20, len(zeros))]
        bottom_ones = np.argsort(one_scores)[:min(20, len(ones))]
        
        for zi in top_zeros:
            for oi in bottom_ones:
                j = zeros[zi]
                k = ones[oi]
                # Compute full effect
                effect = flip_remove_effect(k, h, n) + flip_add_effect(j, h, n)
                new_c = c + effect
                new_max = float(new_c.max())
                improvement = current_c5 - new_max * dx
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_pair = (k, j)
                    best_new_c = new_c
        
        if best_pair is None or best_improvement < 1e-10:
            break
        
        k, j = best_pair
        h[k] = 0
        h[j] = 1
        c = best_new_c
        ones.remove(k)
        ones.append(j)  # wait, j was already removed from zeros
        zeros.remove(j)
        current_c5 = float(c.max()) * dx
    
    return h, current_c5

        new_c5 = float(new_c.max()) * dx
        delta = new_c5 - current_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(tau, 1e-10)):
            h[k] = 0
            h[j] = 1
            c = new_c
            current_c5 = new_c5
            no_improve = 0
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            tau *= 1.001  # slight increase after move (optional)
        else:
            no_improve += 1
        
        tau *= cooldown
    
    return best_h, best_c5
```

```python
def guided_local_search(h, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy().astype(float)
    c = compute_corr(h)
    dx = 2.0 / n
    current_c5 = float(c.max()) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    stagnation = 0
    
    while time.time() < deadline:
        # Find bottleneck
        p_star = int(np.argmax(c))
        m_star = p_star - (n - 1)
        
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        # Compute scores for guided search
        # For remove k: score_r(k) = h[k+m*] + h[k-m*] (want small)
        # For add j: score_a(j) = h[j+m*] + h[j-m*] (want large)
        
        km_p = ones_idx + m_star
        km_m = ones_idx - m_star
        score_r = np.zeros(len(ones_idx))
        mask_p = (km_p >= 0) & (km_p < n)
        mask_m = (km_m >= 0) & (km_m < n)
        score_r[mask_p] += h[km_p[mask_p]]
        score_r[mask_m] += h[km_m[mask_m]]
        
        kj_p = zeros_idx + m_star
        kj_m = zeros_idx - m_star
        score_a = np.zeros(len(zeros_idx))
        mask_p2 = (kj_p >= 0) & (kj_p < n)
        mask_m2 = (kj_m >= 0) & (kj_m < n)
        score_a[mask_p2] += h[kj_p[mask_p2]]
        score_a[mask_m2] += h[kj_m[mask_m2]]
        
        # Try top candidates
        k_order = np.argsort(score_r)  # smallest first (best to remove)
        j_order = np.argsort(-score_a)  # largest first (best to add)
        
        top_k_count = min(15, len(k_order))
        top_j_count = min(15, len(j_order))
        
        best_new_c5 = current_c5
        best_pair = None
        best_new_c = None
        
        for ki in range(top_k_count):
            k = ones_idx[k_order[ki]]
            for ji in range(top_j_count):
                j = zeros_idx[j_order[ji]]
                if abs(int(k) - int(j)) < 1:  # skip same position (shouldn't happen)
                    continue
                eff = flip_remove_effect(int(k), h, n) + flip_add_effect(int(j), h, n)
                new_c = c + eff
                new_c5_val = float(new_c.max()) * dx
                if new_c5_val < best_new_c5 - 1e-12:
                    best_new_c5 = new_c5_val
                    best_pair = (int(k), int(j))
                    best_new_c = new_c
        
        if best_pair is not None:
            k, j = best_pair
            h[k] = 0
            h[j] = 1
            c = best_new_c
            current_c5 = best_new_c5
            stagnation = 0
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            stagnation += 1
            if stagnation >= 30:
                # Random kick: do 5 random pair flips
                for _ in range(5):
                    ones_ij = np.where(h == 1)[0]
                    zeros_ij = np.where(h == 0)[0]
                    k = int(ones_ij[rng.randint(len(ones_ij))])
                    j = int(zeros_ij[rng.randint(len(zeros_ij))])
                    eff = flip_remove_effect(k, h, n) + flip_add_effect(j, h, n)
                    c = c + eff
                    h[k] = 0
                    h[j] = 1
                current_c5 = float(c.max()) * dx
                stagnation = 0
    
    return best_h, best_c5
```

- Computing scores: O(n)
- Trying 15×15 = 225 candidates, each O(n): O(225n)
- Total: O(225n) per iteration
