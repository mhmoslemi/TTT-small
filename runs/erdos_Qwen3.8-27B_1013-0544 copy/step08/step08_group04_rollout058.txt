                sq = np.sqrt(disc)
                for x1 in [(-B + sq) / 2.0, (-B - sq) / 2.0]:
                    if 0.0 <= x1 <= 1.0:
                        cands.append(x1)
    best_x, best_f = hp, f_val(hp, hp, cseg, a, qconst, unaff)
    for x in cands:
        if x < 0 or x > 1:
            continue
        f = f_val(x, hp, cseg, a, qconst, unaff)
        if f < best_f - 1e-14:
            best_f, best_x = f, x
    return best_x, best_f

def coord_descent(h, n, max_passes, deadline, project_every=10):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h, best_c = h.copy(), (2.0 / n) * cur
    center = n - 1
    L = 2 * n - 1
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            hp = float(h[p])
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            seg = hi - lo + 1
            s = center - np.arange(lo, hi + 1, dtype=int)
            j, i = p + s, p - s
            a = np.zeros(seg, dtype=float)
            vj = (j >= 0) & (j < n)
            vi = (i >= 0) & (i < n)
            if vj.any():
                a[vj] += 1.0 - h[j[vj]]
            if vi.any():
                a[vi] -= h[i[vi]]
            off = center - lo
            cseg = np.empty(max(seg - 1, 0), dtype=float)
            asg = np.empty(max(seg - 1, 0), dtype=float)
            if off > 0:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center + 1:hi + 1]
                asg[off:] = a[off + 1:]
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(c[hi + 1:].max()))
            bx, bf = opt_coord(hp, cseg, asg, qconst, unaff)
            if bf < cur - 1e-12:
                dxv = bx - hp
                if off > 0:
                    c[lo:center] += asg[:off] * dxv
                if hi > center:
                    c[center + 1:hi + 1] += asg[off:] * dxv
                c[center] = qconst + bx * (1.0 - bx)
                h[p] = bx
                cur = bf
                improved = True
            if (p + 1) % project_every == 0:
                h = project_feasible(h, n)
                c = np.correlate(h, 1.0 - h, mode='full')
                cur = float(c.max())
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        cval = (2.0 / n) * cur
        if cval < best_c - 1e-14:
            best_c, best_h = cval, h.copy()
        if not improved:
            break
    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    z = alpha * c
    zm = float(z.max())
    return ((zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    center = n - 1
    z = alpha * c
    z -= float(z.max())
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * x)
        elif s > 0:
            g[:n - s] += wm * (1.0 - x[s:])
            g[s:] -= wm * x[:n - s]
        else:
            r = -s
            g[r:] += wm * (1.0 - x[:n - r])
            g[:n - r] -= wm * x[r:]
    return (2.0 / n) * g

def _sum_con(x, n):
    return float(x.sum()) - 0.5 * n

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h = project_feasible(h, n)
    best_c = compute_c(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_jac, bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
            x = project_feasible(res.x, n)
            c = compute_c(x, n)
            if c < best_c:
                best_h, best_c = x, c
        except Exception:
            pass
    return best_h, best_c

def fourier_h(n, rng, K=12):
    h = 0.5 * np.ones(n)
    x = np.arange(n, dtype=float)
    for k in range(1, K + 1):
        amp = rng.uniform(0.05, 0.30) / (0.5 + 0.5 * k)
        h += amp * np.cos(2 * np.pi * k * x / n + rng.uniform(0, 2 * np.pi))
    return project_feasible(h, n)

def golden_binary(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    half = n // 2
    for k in range(half):
        pos = int(np.floor((k + 1) * phi)) % n
        h[pos] = 1.0
    return project_feasible(h, n)

def optimize_combo(h, n, deadline, slsqp_alphas, slsqp_max, cd_passes, pe):
    h = project_feasible(h, n)
    c = compute_c(h, n)
    if n <= 800 and time.time() < deadline - 60:
        h2, c2 = slsqp_opt(h, n, slsqp_alphas, slsqp_max, deadline)
        if c2 < c:
            h, c = h2, c2
    if time.time() < deadline - 15:
        h2, c2 = coord_descent(h, n, cd_passes, deadline, pe)
        if c2 < c:
            h, c = h2, c2
    if n <= 600 and time.time() < deadline - 50:
        h3, c3 = slsqp_opt(h, n, slsqp_alphas[-2:], max(20, slsqp_max // 2), deadline)
        if c3 < c:
            h, c = h3, c3
    if time.time() < deadline - 10:
        h4, c4 = coord_descent(h, n, 3, deadline, pe)
        if c4 < c:
            h, c = h4, c4
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = project_feasible(0.5 + 0.3 * np.sin(np.pi * np.arange(500, dtype=float) / 250), 500)
    n_init = len(h_init)
    if n_init % 2:
        n_init += 1
        h_init = resample_h(h_init, n_init)
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c(best_h, n_init)

    # Phase 1: optimize initial at its n
    h1, c1 = optimize_combo(best_h, n_init, deadline,
                            (30, 300, 3000, 30000), 100, 10, 8)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: multi-scale exploration
    scale_ns = [n_init, 400, 550, 700, 300, 800]
    for nt in scale_ns:
        if time.time() > deadline - 180:
            break
        if best_c <= 0.38075:
            break
        h0 = resample_h(best_h, nt)
        pe = 6 if nt <= 400 else (10 if nt <= 700 else 15)
        passes = 8 if nt <= 500 else (5 if nt <= 700 else 3)
        alphas = (50, 500, 5000) if nt <= 500 else (30, 300)
        smax = 80 if nt <= 500 else 50
        h1, c1 = optimize_combo(h0, nt, deadline, alphas, smax, passes, pe)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 3: diverse restarts
    for r in range(20):
        if time.time() > deadline - 120:
            break
        if best_c <= 0.38072:
            break
        nt = n_init if r % 2 == 0 else 400
        base = resample_h(best_h, nt)
        typ = r % 5
        if typ == 0:
            h0 = project_feasible(base + rng.normal(0, 0.03, nt), nt)
        elif typ == 1:
            h0 = binary_from_h(base)
        elif typ == 2:
            h0 = fourier_h(nt, rng, K=10 + r % 5)
        elif typ == 3:
            h0 = golden_binary(nt)
        else:
            h0 = project_feasible(base * rng.uniform(0.8, 1.2, nt), nt)
        h1, c1 = optimize_combo(h0, nt, deadline,
                                (100, 1000, 10000), 60, 4, 8)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 4: final aggressive refinement at best n
    if time.time() < deadline - 40:
        fn = len(best_h)
        best_h, best_c = optimize_combo(best_h, fn, deadline,
                                        (200, 2000, 20000), 120, 15, 5)

    # Phase 5: try binary refinement
    if time.time() < deadline - 20:
        hb = binary_from_h(best_h)
        cbi = compute_c(hb, len(hb))
        if cbi < best_c:
            hb, cbi = coord_descent(hb, len(hb), 6, deadline, 5)
            if cbi < best_c:
                best_h, best_c = hb, cbi

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c(best_h, len(best_h))
    return best_h, best_c, len(best_h)
