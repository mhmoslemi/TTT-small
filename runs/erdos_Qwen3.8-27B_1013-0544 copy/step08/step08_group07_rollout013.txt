                if f1 < f2:

            best_d = (lo3 + hi3) / 2.0
            best_val = float(np.max(np.where(np.arange(2*n-1) == n-1,

            if best_val < cur_max - 1e-12:

                improved = True
        if not improved:

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    cs = alpha * (c - c.max())
    w = np.exp(cs)
    s = w.sum()
    if s < 1e-300:
        return np.zeros(n)
    w /= s
    n_full = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_full):
        wi = w[k]
        if wi < 1e-16:
            continue
        m = k - (n - 1)
        if m >= 0:
            nm = n - m
            if nm > 0:
                grad[:nm] += wi * (1.0 - h[m:])
                grad[m:] -= wi * h[:nm]
        else:
            ma = -m
            if ma < n:
                grad[ma:] += wi * (1.0 - h[:ma])
            if n - ma > 0:
                grad[:n-ma] -= wi * h[ma:]
    return grad

def smooth_grad_descent(h, n, deadline, max_iter=30000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 30.0
    lr = 0.03
    stall_count = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it > 0 and it % 1000 == 0:
            alpha *= 1.8
            lr *= 0.88
        g = smooth_grad(h, n, alpha)
        g_norm = np.sqrt(np.mean(g**2))
        if g_norm < 1e-10:
            break
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-9:
            best_c = c_new
            best_h = h_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 2000:
                lr *= 0.7
                stall_count = 0
        h = h_new
    return best_h, best_c

def coord_descent(h, n, max_passes=150, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    idx = np.arange(2 * n - 1)
    diag_mask = (idx == n - 1).astype(float)
    for p in range(max_passes):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        improved = False
        for j in range(n):
            if deadline and time.time() > deadline:
                break
            h0 = h[j]
            if h0 < 5e-4 or h0 > 1.0 - 5e-4:
                continue
            m_range = np.arange(-(n-1), n)
            a = np.zeros(2*n-1)
            m_lo = max(-j, -(n-1))
            m_hi = min(n-1-j, n-1)
            if m_lo <= m_hi:
                ii = (m_range >= m_lo) & (m_range <= m_hi)
                a[ii] += (1.0 - h[j + m_range[ii]])
            m_lo2 = max(j-n+1, -(n-1))
            m_hi2 = min(j, n-1)
            if m_lo2 <= m_hi2:
                ii2 = (m_range >= m_lo2) & (m_range <= m_hi2)
                a[ii2] -= h[j - m_range[ii2]]
            lo = -h0
            hi = 1.0 - h0
            lo3, hi3 = lo, hi
            for _ in range(45):
                m1 = lo3 + (hi3 - lo3) / 3.0
                m2 = hi3 - (hi3 - lo3) / 3.0
                f1 = float(np.max(c + m1 * a - m1*m1 * diag_mask))
                f2 = float(np.max(c + m2 * a - m2*m2 * diag_mask))
                if f1 < f2:
                    hi3 = m2
                else:
                    lo3 = m1
            bd = (lo3 + hi3) / 2.0
            bv = float(np.max(c + bd * a - bd*bd * diag_mask))
            if bv < cur_max - 1e-11:
                h[j] = np.clip(h0 + bd, 0.0, 1.0)
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < best_c:
            best_c = nc
    return h, compute_c5(h, n)

def simulated_annealing(h, n, deadline, n_steps=8000, T0=0.002, T1=0.00005):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_now = best_c
    for step in range(n_steps):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (step / max(1, n_steps - 1))
        r1 = np.random.randint(n)
        r2 = np.random.randint(n)
        if r1 == r2:
            continue
        delta = 0.01 + 0.04 * np.random.rand()
        if np.random.rand() < 0.5:
            if h[r1] < delta or h[r2] > 1.0 - delta:
                continue
            h[r1] -= delta
            h[r2] += delta
        else:
            if h[r2] < delta or h[r1] > 1.0 - delta:
                continue
            h[r2] -= delta
            h[r1] += delta
        c_new = compute_c5(h, n)
        if c_new < c_now or np.random.rand() < np.exp(-(c_new - c_now) / max(T, 1e-8)):
            c_now = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[r1] += delta if r1 != r2 else 0
            # revert
            if np.random.rand() < 0.5:
                h[r1] += delta
                h[r2] -= delta
            else:
                h[r2] += delta
                h[r1] -= delta
    return best_h, best_c

def structured_start(n, kind, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    if kind == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x)
    elif kind == 1:
        h = 0.5 + 0.3 * np.sin(3 * np.pi * x) + 0.15 * np.sin(5 * np.pi * x)
    elif kind == 2:
        h = 0.5 + 0.25 * np.sin(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif kind == 3:
        h = (0.4 + 0.3 * np.sin(2 * np.pi * 1.5 * x)).clip(0, 1)
    else:
        h = rng.uniform(0, 1, n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    try:

    best_c = compute_c5(best_h, n)
    # Phase 1: smooth gradient descent

    if c1 < best_c:

    h2, c2 = coord_descent(best_h, n, 200, d2)
    if c2 < best_c:

    # Phase 3: multi-scale
    x_orig = np.linspace(0, 2, n, endpoint=False)

        if time.time() > deadline - 120 or ns >= n:
            break
        x_s = np.linspace(0, 2, ns, endpoint=False)
        h_s = project_feasible(np.interp(x_s, x_orig, best_h, period=2) + 0.03 * rng.randn(ns), ns)
        d_s = min(deadline - 100, time.time() + 80)
        h_s, c_s = smooth_grad_descent(h_s, ns, d_s, max_iter=8000)
        h_s, c_s = coord_descent(h_s, ns, 80, min(deadline - 80, time.time() + 60))
        x_big = np.linspace(0, 2, n, endpoint=False)
        h_back = project_feasible(np.interp(x_big, x_s, h_s, period=2), n)
        h_back, c_back = coord_descent(h_back, n, 60, min(deadline - 60, time.time() + 50))
        if c_back < best_c:
            best_h, best_c = h_back, c_back
    # Phase 4: simulated annealing

        if c4 < best_c:
            best_h, best_c = h4, c4
    # Phase 5: restarts with different structures
    if time.time() < deadline - 40:

            if time.time() > deadline - 20:
                break
            hp = structured_start(n, r, rng)
            if r >= 3:

            d_r = min(deadline - 15, time.time() + 40)
            h_r, c_r = smooth_grad_descent(hp, n, d_r, max_iter=3000)
            h_r, c_r = coord_descent(h_r, n, 50, min(deadline - 10, time.time() + 30))
            if c_r < best_c:

    # Phase 6: final polish
    if time.time() < deadline - 5:
        h_f, c_f = coord_descent(best_h, n, 100, deadline)
        if c_f < best_c:

            h_g, c_g = smooth_grad_descent(best_h, n, deadline - 1, max_iter=5000)
            if c_g < best_c:
                best_h, best_c = h_g, c_g
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
