import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_onto_box_simplex(x, c):
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(x - mid, 0.0, 1.0).sum() > c:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def ls_grad(h, n, beta):
    """Compute gradient of logsumexp(r*beta)/beta where r = correlate(h,1-h,'full')*dx."""
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full') * dx
    r_max = r.max()
    exp_r = np.exp(beta * (r - r_max))
    w = exp_r / exp_r.sum()
    g = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        l = k - (n - 1)
        if 0 <= l < n:
            g[l:] += wk * omh[:n - l]
            g[:n - l] -= wk * h[l:]
        elif -n < l < 0:
            m = -l
            g[:n - m] += wk * omh[m:]
            g[m:] -= wk * h[:n - m]
    return g * dx, r_max

def ls_slsqp(h, n, beta, maxiter=500, deadline=None):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        r = np.correlate(x, 1.0 - x, mode='full') * dx
        rm = r.max()
        return rm + np.log(np.exp(beta * (r - rm)).sum()) / beta
    
    def obj_grad(x):
        g, _ = ls_grad(x, n, beta)
        return g
    
    res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    new_h = project_onto_box_simplex(res.x, target)
    return new_h, compute_c5(new_h, n)

def exact_cd(h, n, max_passes=200, deadline=None, grid_pts=25):
    h = h.copy()
    target = n / 2.0
    n2 = 2 * n - 1
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full').astype(float)
        curr = c.max()
        improved = False
        for p in range(n):
            if deadline and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            # Compute A: effect on c[k] of changing h[p] from h0 to h0 + delta
            # c[k] = sum_i h[i+k'l]*(1-h[i'l]) where k'l is lag
            # h[p] appears in c[k] when:
            #   As h[i+k'l] with i+k'l = p, i.e., i = p - l
            #   As h[i'l] (in 1-h[i'l]) with i'l = p, i.e., i = p
            # For lag l = k - (n-1):
            #   h[p] = h[i+l] when i = p-l, valid if 0<=p-l and 0<=i<=n-1-l (for l>=0)
            #     or m<=p-l<=n-1 (for l<0, m=-l)
            #   h[p] = h[i] when i = p, valid if 0<=p<=n-1-l (for l>=0)
            #     or m<=p<=n-1 (for l<0)
            A = np.zeros(n2)
            D = (np.arange(n2) == (n-1)).astype(float)
            for l in range(-(n-1), n):
                k = l + (n-1)
                # h[p] as h[i+l]: i = p-l
                i = p - l
                if l >= 0:
                    if 0 <= i <= n-1-l:
                        A[k] += (1.0 - h[i])
                else:
                    m = -l
                    if m <= i <= n-1:
                        A[k] += (1.0 - h[i])
                # h[p] as h[i] (in 1-h[i]): contribution is -h[i+l] = -h[p+l]
                if l >= 0:
                    if 0 <= p <= n-1-l:
                        A[k] += -h[p+l]
                else:
                    m = -l
                    if m <= p <= n-1:
                        A[k] += -h[p+l]
            # c_new[k] = c[k] - h0*A[k] + t*A[k] - (h0^2 - t^2)*D[k]
            #           = c[k] + (t-h0)*A[k] - (t^2-h0^2)*D[k]
            # Minimize max_k c_new[k] over t in [0,1]
            c0 = c - h0 * A + h0 * h0 * D
            # f(t) = max(c0 + t*A - t^2*D)
            grid = np.linspace(0.0, 1.0, grid_pts)
            fv = np.array([np.max(c0 + t*A - t*t*D) for t in grid])
            bi = int(np.argmin(fv))
            # Golden section refinement
            lo = grid[max(0, bi-1)]
            hi = grid[min(grid_pts-1, bi+1)]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = lo, hi
            c1 = b - gr*(b-a)
            d1 = a + gr*(b-a)
            fc1 = np.max(c0 + c1*A - c1*c1*D)
            fd1 = np.max(c0 + d1*A - d1*d1*D)
            for _ in range(25):
                if fc1 < fd1:
                    b = d1; d1 = c1; fd1 = fc1
                    c1 = b - gr*(b-a)
                    fc1 = np.max(c0 + c1*A - c1*c1*D)
                else:
                    a = c1; c1 = d1; fc1 = fd1
                    d1 = a + gr*(b-a)
                    fd1 = np.max(c0 + d1*A - d1*d1*D)
            t_star = (a+b) * 0.5
            f_star = np.max(c0 + t_star*A - t_star*t_star*D)
            f0 = np.max(c0)
            f1 = np.max(c0 + A - D)
            if min(f_star, f0, f1) < curr - 1e-12:

                curr = c.max()
                improved = True
        h = project_onto_box_simplex(h, target)
        if not improved:
            break
    return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_onto_box_simplex(x, c):
    lo, hi = -1.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(x - mid, 0.0, 1.0).sum() > c:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def ls_grad(h, n, beta):
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full') * dx
    r_max = r.max()
    exp_r = np.exp(beta * (r - r_max))
    w = exp_r / exp_r.sum()
    g = np.zeros(n)
    omh = 1.0 - h
    n2 = 2 * n - 1
    for k in range(n2):
        wk = w[k]
        if wk < 1e-10:
            continue
        l = k - (n - 1)
        if 0 <= l < n:
            g[l:] += wk * omh[:n - l]
            g[:n - l] -= wk * h[l:]
        elif -n < l < 0:
            m = -l
            g[:n - m] += wk * omh[m:]
            g[m:] -= wk * h[:n - m]
    return g * dx

def ls_slsqp(h, n, beta, maxiter=400, deadline=None):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        r = np.correlate(x, 1.0 - x, mode='full') * dx
        rm = r.max()
        return rm + np.log(np.sum(np.exp(beta * (r - rm)))) / beta

    def obj_grad(x):
        return ls_grad(x, n, beta)

    def cons_f(x):
        return x.sum() - target

    def cons_j(x):
        return np.ones(n)

    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_f, 'jac': cons_j}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        new_h = project_onto_box_simplex(res.x, target)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def ls_gradient_descent(h, n, n_iters, beta, deadline):
    h = h.copy()
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 1.0 / beta
    for it in range(n_iters):
        if time.time() > deadline:
            break
        g = ls_grad(h, n, beta)
        h_new = project_onto_box_simplex(h - alpha * g, target)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new
        if c_new > compute_c5(h, n) + 1e-10:
            alpha *= 0.5
        h = h_new
    return best_h, best_c

def exact_cd(h, n, max_passes=300, deadline=None, grid_pts=30):
    h = h.copy()
    target = n / 2.0
    n2 = 2 * n - 1
    mid_idx = n - 1
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full').astype(float)
        curr = c.max()
        improved = False
        for p in range(n):
            if deadline and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            A = np.zeros(n2)
            for l in range(-(n - 1), n):
                k = l + (n - 1)
                i = p - l
                if l >= 0:
                    if 0 <= i <= n - 1 - l:
                        A[k] += (1.0 - h[i])
                    if 0 <= p <= n - 1 - l and p + l < n:
                        A[k] += -h[p + l]
                else:
                    m = -l
                    if m <= i <= n - 1:
                        A[k] += (1.0 - h[i])
                    if m <= p <= n - 1 and p + l >= 0:
                        A[k] += -h[p + l]
            D = (np.arange(n2) == mid_idx).astype(float)
            c0 = c - h0 * A + h0 * h0 * D
            grid = np.linspace(0.0, 1.0, grid_pts)
            fv = np.array([np.max(c0 + t * A - t * t * D) for t in grid])
            bi = int(np.argmin(fv))
            lo = grid[max(0, bi - 1)]
            hi = grid[min(grid_pts - 1, bi + 1)]
            if hi - lo < 1e-14:
                t_star = grid[bi]
            else:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = lo, hi
                c1 = b - gr * (b - a)
                d1 = a + gr * (b - a)
                fc1 = np.max(c0 + c1 * A - c1 * c1 * D)
                fd1 = np.max(c0 + d1 * A - d1 * d1 * D)
                for _ in range(25):
                    if fc1 < fd1:
                        b = d1; d1 = c1; fd1 = fc1
                        c1 = b - gr * (b - a)
                        fc1 = np.max(c0 + c1 * A - c1 * c1 * D)
                    else:
                        a = c1; c1 = d1; fc1 = fd1
                        d1 = a + gr * (b - a)
                        fd1 = np.max(c0 + d1 * A - d1 * d1 * D)
                t_star = (a + b) * 0.5
            f_star = np.max(c0 + t_star * A - t_star * t_star * D)
            f0 = np.max(c0)
            f1 = np.max(c0 + A - D)
            if min(f_star, f0, f1) < curr - 1e-12:
                if f0 <= f_star and f0 <= f1:
                    t_new = 0.0
                elif f1 <= f_star and f1 <= f0:
                    t_new = 1.0
                else:
                    t_new = float(np.clip(t_star, 0.0, 1.0))
                c += (t_new - h0) * A - (t_new * t_new - h0 * h0) * D
                h[p] = t_new
                curr = c.max()
                improved = True
        h = project_onto_box_simplex(h, target)
        if not improved:
            break
    return h, compute_c5(h, n)

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_onto_box_simplex(h_new, n_new / 2.0)
