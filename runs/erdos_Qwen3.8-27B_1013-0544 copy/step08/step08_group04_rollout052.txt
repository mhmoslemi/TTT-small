import numpy as np

from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-12:
        return h
    if diff > 0.0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0.0:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 0.0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-12:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            h[idx[np.argmax(1.0 - h[idx])]] = min(1.0, h[idx[np.argmax(1.0 - h[idx])]] + res)
    elif res < -1e-12:
        idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[np.argmin(h[idx])]] = max(0.0, h[idx[np.argmin(h[idx])]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    z = alpha * c
    zm = float(z.max())
    return ((zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    center = n - 1
    z = alpha * c
    z = z - z.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm <= 1e-14:
            continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * x)
        elif s > 0:
            g[:n - s] += wm * (1.0 - x[s:])
            g[s:] -= wm * x[:n - s]
        else:
            r = -s
            g[r:] += wm * (1.0 - x[:n - r])
            g[:n - r] -= wm * x[r:]
    return (2.0 / n) * g

def sum_con(x, n):
    return float(x.sum()) - 0.5 * n

def sum_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter, deadline):
    best_h = project_feasible(h, n)
    best_c = compute_c(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_jac, bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
            x = project_feasible(res.x, n)
            c = compute_c(x, n)

            if np.any(vj):

            off = center - lo
            cseg = np.empty(seg - 1, dtype=float)

            if off:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center + 1:hi + 1]

            qconst = c[center] - hp * (1.0 - hp)

                    c[center + 1:hi + 1] += asg[off:] * dxv

        if cval < best_c - 1e-14:

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-13:
        return h
    if diff > 0.0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 1e-14:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 1e-14:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size:
                best = idx[np.argmax(1.0 - h[idx])]
                h[best] = min(1.0, h[best] + res)
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size:
                best = idx[np.argmin(h[idx])]
                h[best] = max(0.0, h[best] + res)

    z = alpha * c
