def project_simplex(h,n):
    h=np.asarray(h,dtype=float).clip(0,1).copy()
    t=0.5*n
    s=h.sum()
    if abs(s-t)<1e-13: return h
    lo,hi=-1.0,1.0
    for _ in range(50):
       m=0.5*(lo+hi)
       if np.clip(h+m,0,1).sum() < t: lo=m
       else: hi=m
    y=np.clip(h+hi,0,1)

import numpy as np, time
from scipy.optimize import minimize
try:
    from scipy.optimize import differential_engine
except Exception:
    differential_engine = None

BT = {}

def project_simplex(h, n):
    h = np.clip(np.asarray(h, float), 0.0, 1.0).copy()
    t = 0.5 * n
    lo, hi = -1.0, 1.0
    for _ in range(50):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    for eps in (1e-12, 1e-14):
        if abs(d) > eps:
            msk = (y > 1e-10) & (y < 1.0 - 1e-10)
            if msk.any():
                y[msk] += d / msk.sum()
            else:
                idx = int(np.argmax(y if d < 0 else 1.0 - y))
                y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
            y = np.clip(y, 0.0, 1.0)
            d = t - y.sum()
    return y

def c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def smooth_val_grad(x, n, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    dx = 2.0 / n
    r = a * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    w = ee / z
    A = np.convolve(1.0 - x, w, mode='full')
    B = np.convolve(x, w[::-1], mode='full')
    s = n - 1
    return (m + np.log(z) / a) * dx, (A - B)[s:s + n] * dx

def so(x, n, a):
    return smooth_val_grad(x, n, a)[0]

def sg(x, n, a):
    return smooth_val_grad(x, n, a)[1]

def sum_con(x, n):
    return x.sum() - 0.5 * n

def sum_jac(x, n):
    return np.ones_like(x)

def slsqp_polish(h, n, alphas, maxit, deadline):
    best = project_simplex(h, n)
    bc = c5(best, n)
    bd = [(0.0, 1.0)] * n
    co = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            r = minimize(so, best, args=(n, float(a)), method='SLSQP', jac=sg,
                         bounds=bd, constraints=co,
                         options={'maxiter': int(maxit), 'ftol': 1e-12})
            nh = project_simplex(r.x, n)
            nc = c5(nh, n)
            if nc < bc - 1e-13:
                best, bc = nh, nc
        except Exception:
            pass
    return best, bc

def cand_vals(c, A, xs, old, n):
    d = xs - old
    o = c[None, :] + d[:, None] * A[None, :]
    o[:, n - 1] -= d * d
    return o.max(axis=1)

def coord_cd(h, n, deadline, grid=36, refine=12, passes=50, order=None):
    best = project_simplex(h, n)
    bc = c5(best, n)
    L = 2 * n - 1
    dg = n - 1
    xg = np.linspace(0.0, 1.0, grid)
    if order is None:
        order = np.arange(n)
    for _ in range(passes):
        if time.time() >= deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode='full')
        om = 1.0 - x[::-1]
        A = np.empty(L)
        imp = False
        cnt = 0
        for p in order:
            if time.time() >= deadline:
                break
            old = float(x[p])
            A.fill(0.0)
            A[p:p + n] += om
            st = dg - p
            A[st:st + n] -= x
            cur = float(c.max())
            vals = cand_vals(c, A, xg, old, n)
            i = int(np.argmin(vals))
            bx = float(xg[i])
            bv = float(vals[i])
            if bv < cur - 1e-11:
                sp = 1.0 / (grid - 1)
                lo = max(0.0, bx - sp)
                hi = min(1.0, bx + sp)
                xx = np.linspace(lo, hi, refine + 1)
                mm = cand_vals(c, A, xx, old, n)
                j = int(np.argmin(mm))
                if mm[j] < bv - 1e-13:
                    bx = float(xx[j])
                    bv = float(mm[j])
                if bv < cur - 1e-11:
                    d = bx - old
                    c[p:p + n] += d * om
                    c[st:st + n] -= d * x
                    c[dg] -= d * d
                    x[p] = bx
                    om[dg - p] = 1.0 - bx
                    imp = True
            cnt += 1
            if cnt % 64 == 0 and abs(x.sum() - 0.5 * n) > 1e-7:
                x = project_simplex(x, n)
                c = np.correlate(x, 1.0 - x, mode='full')
                om = 1.0 - x[::-1]
        xp = project_simplex(x, n)
        cp = c5(xp, n)
        if cp < bc - 1e-12:
            best, bc = xp, cp
        if not imp:
            break
    return best, bc

def wgrad(h, n, w):
    A = np.convolve(1.0 - h, w, mode='full')
    B = np.convolve(h, w[::-1], mode='full')
    s = n - 1
    return (A - B)[s:s + n] * (2.0 / n)

def active_grad(h, n, iters, deadline, qs=(1, 2, 3, 5, 10, 20)):
    best = project_simplex(h, n)
    cur = best.copy()
    bc = c5(cur, n)
    L = 2 * n - 1
    stag = 0
    for _ in range(iters):
        if time.time() >= deadline:
            break
        imp = False
        for q in qs:
            if time.time() >= deadline:
                break
            c = np.correlate(cur, 1.0 - cur, mode='full')
            if q >= L:
                idx = np.arange(L)
            else:
                idx = np.argpartition(c, L - q)[-q:]
            w = np.zeros(L)
            w[idx] = 1.0 / q
            g = wgrad(cur, n, w)
            g -= g.mean()
            gn = np.linalg.norm(g)
            if gn < 1e-10:
                continue
            dirn = -g / gn
            nh_best = None
            cn_best = bc
            for tau in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
                nh = project_simplex(cur + tau * dirn, n)
                nc = c5(nh, n)
                if nc < cn_best:
                    cn_best = nc
                    nh_best = nh
            if nh_best is not None and cn_best < bc - 1e-13:
                cur = nh_best
                bc = cn_best
                best = nh_best
                imp = True
                break
        if not imp:
            stag += 1
            if stag >= 3:
                break
        else:
            stag = 0
    return best, bc

def transfer_h(h, n, nt):
    if nt == n:
        return project_simplex(h, nt).copy()
    return project_simplex(np.interp(np.linspace(0.0, 2.0, nt),
                                     np.linspace(0.0, 2.0, n), h), nt)

def gen_smooth(n, rng):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 * np.ones(n)
    for m in range(1, 9):
        amp = 0.40 / (0.6 * m + 0.4)
        a = rng.uniform(-1.0, 1.0) * amp
        b = rng.uniform(-1.0, 1.0) * amp
        h = h + a * np.cos(np.pi * m * x) + b * np.sin(np.pi * m * x)
    return project_simplex(np.clip(h, 0.0, 1.0), n)

def gen_blocks(n, rng):
    mm = min(n, int(rng.randint(12, min(80, n // 2) + 1)))
    if mm <= 1:
        return project_simplex(0.5 * np.ones(n), n)
    cuts = np.sort(rng.choice(np.arange(1, n), mm - 1, replace=False))
    lengths = np.diff(np.concatenate(([0], cuts, [n])))
    vals = rng.uniform(0.0, 1.0, mm)
    h = np.empty(n)
    p = 0
    for l, v in zip(lengths, vals):
        l = int(l)
        h[p:p + l] = float(v)
        p += l
    return project_simplex(h, n)

def perturb(h, n, rng, sigma):
    return project_simplex(h + sigma * rng.randn(n), n)

def get_basis(n, K):
    key = (n, K)
    if key not in BT:
        x = np.linspace(0.0, 2.0, n)
        cols = [np.ones(n)]
        for k in range(1, K + 1):
            c = np.cos(np.pi * k * x)
            s = np.sin(np.pi * k * x)
            cols.append(c)
            cols.append(s)
        BT[key] = np.column_stack(cols)
    return BT[key]

def fourier_fun(theta, n, K):
    B = get_basis(n, K)
    h = np.asarray(theta, float) @ B.T
    return c5(project_simplex(np.clip(h, 0.0, 1.0), n), n)

def fourier_h(theta, n, K):
    B = get_basis(n, K)
    h = np.asarray(theta, float) @ B.T
    return project_simplex(np.clip(h, 0.0, 1.0), n)

def improve(h, n, deadline, mode=1):
    h = project_simplex(h, n)
    c = c5(h, n)
    if mode >= 3:
        as_, cs, gr, rf, pa = 35, 50, 52, 18, 30
        qs = (1, 2, 3, 5, 10, 20, 40)
    elif mode == 2:
        as_, cs, gr, rf, pa = 20, 30, 44, 14, 20
        qs = (1, 2, 3, 5, 10, 20)
    else:
        as_, cs, gr, rf, pa = 10, 15, 36, 10, 12
        qs = (1, 2, 3, 5, 10)
    t1 = min(deadline, time.time() + as_)
    if time.time() < t1:
        h, c = active_grad(h, n, 40 if mode == 3 else 15, t1, qs)
    t2 = min(deadline, time.time() + cs)
    h, c = coord_cd(h, n, t2, grid=gr, refine=rf, passes=pa)
    if mode == 3 and time.time() < deadline - 50:
        t3 = min(deadline, time.time() + 55)
        h, c = slsqp_polish(h, n, (2000, 20000, 200000), 50, t3)
        t3b = min(deadline, time.time() + 20)
        h, c = coord_cd(h, n, t3b, grid=52, refine=18, passes=8)
    if time.time() < deadline:
        h, c = coord_cd(h, n, deadline, grid=min(64, gr + 10),
                        refine=min(24, rf + 6), passes=pa)
    return h, c
