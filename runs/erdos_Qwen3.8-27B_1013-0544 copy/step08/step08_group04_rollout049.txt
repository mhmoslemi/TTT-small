import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 1e-12:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 1e-12:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-10:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-10:
        idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    z = alpha * c
    zm = float(z.max())
    val = ((zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)
    # Add small penalty for constraint violation
    pen = 100.0 * (float(x.sum()) - 0.5 * n) ** 2
    return val + pen

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    center = n - 1
    z = alpha * c - alpha * c.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * x)
        elif s > 0:
            g[:n - s] += wm * (1.0 - x[s:])
            g[s:] -= wm * x[:n - s]
        else:
            r = -s
            g[r:] += wm * (1.0 - x[:n - r])
            g[:n - r] -= wm * x[r:]
    g *= (2.0 / n)
    # Penalty gradient
    g += 200.0 * (float(x.sum()) - 0.5 * n)
    return g

def slsqp_optimize(h, n, alphas, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_jac, bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
            x = project_feasible(res.x, n)
            c = compute_c5(x, n)
            if c < best_c:
                best_h, best_c = x, c
        except Exception:
            pass
    return best_h, best_c

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)

        return h

        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 1e-12:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 1e-12:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-10:
        idx = np.where(h < 1.0)[0]

            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-10:
        idx = np.where(h > 0.0)[0]

            h[idx[0]] = max(0.0, h[idx[0]] + res)

    zm = float(z.max())
    val = ((zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)

def shift_gradient_step(h, n, alpha, deadline):
    """Use bottleneck-shift gradient to propose a new h."""
    c = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    # Find bottleneck shifts (top few by c value)
    top_k = 10
    top_idx = np.argsort(c)[-top_k:]
    # Compute ascent direction for min autocorrelation = -max overlap
    # Gradient of c_k w.r.t h_p: (1-h_{p+k}) - h_{p-k} for the k corresponding to index
    d = np.zeros(n)
    for m in top_idx:
        s = center - m  # this is the shift k
        if s == 0:
            d += (1.0 - 2.0 * h)
        elif s > 0 and s < n:
            d[:n - s] += (1.0 - h[s:])
            d[s:] -= h[:n - s]
        elif s < 0 and -s < n:
            r = -s
            d[r:] += (1.0 - h[:n - r])
            d[:n - r] -= h[r:]
    # Center to preserve sum
    d -= d.mean()
    h_new = project_feasible(h + alpha * d, n)
    return h_new
