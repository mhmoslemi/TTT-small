import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x):
    return x.sum() - x.size / 2.0

def sum_con_jac(x):
    return np.ones(x.size)

def slsqp_optimize(h, n, alphas, maxiter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
        try:

                          bounds=bounds, constraints=cons,

            if new_c < best_c:

            pass

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x):
    return x.sum() - x.size / 2.0

def sum_con_jac(x):
    return np.ones(x.size)

def slsqp_optimize(h, n, alphas, maxiter=400):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

                break

            elif h0 > 1.0 - 1e-12:

            ms = np.arange(m_lo, m_hi+1)
            a = np.zeros(len(ms))
            jh = idx + ms - (n-1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n-1)

            a[vg] -= h[jg[vg]]
            c_aff = c[ms].copy()
            unaff_max = 0.0
            if m_lo > 0:

            if m_hi < 2*n-2:

            diag_mask = (ms == (n-1))

            d_grid = np.linspace(lo_d, hi_d, grid_size)

                        v1 = max(unaff_max, float(nv1.max()))

                if new_max < current_c5:
                    current_c5 = new_max
                improved = True
        if not improved:
            break

    return h, compute_c5(h, n)

            pass
    return best_h, best_c

def coord_cd(h, n, max_passes=2000, deadline=None, grid_size=100):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        idx_order = range(n) if p % 2 == 0 else range(n-1, -1, -1)
        improved = False
        for idx in idx_order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-12:
                lo_d, hi_d = 0.0, 1.0
            elif h0 > 1.0 - 1e-12:
                continue
            else:
                lo_d, hi_d = -h0, 1.0 - h0
            if hi_d - lo_d < 1e-12:
                continue
            m_lo = max(0, min(n-1-idx, idx))
            m_hi = min(2*n-2, max(2*n-2-idx, idx+n-1))
            ms = np.arange(m_lo, m_hi+1)
            a = np.zeros(len(ms))
            jh = idx + ms - (n-1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n-1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms].copy()
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            diag_mask = (ms == (n-1))
            has_diag = diag_mask.any()
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if has_diag:
                di = np.where(diag_mask)[0]
                nv[:, di] -= d_grid[:, None] ** 2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            best_i = int(np.argmin(vals))
            best_d = float(d_grid[best_i])
            best_val = float(vals[best_i])
            if 0 < best_i < grid_size - 1:
                y = vals[best_i-1:best_i+2]
                x = d_grid[best_i-1:best_i+2]
                denom = y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1])
                if abs(denom) > 1e-15:
                    xv = 0.5*(y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1]))/denom
                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv * a
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[di] -= xv**2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val, best_d = v1, float(xv)
            if abs(best_d) > 1e-10 and best_val < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[ms[di]] -= best_d**2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def shake_and_optimize(h, n, rng, n_shakes=50, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    temperature = 0.18
    no_improve = 0
    for s in range(n_shakes):
        if deadline is not None and time.time() > deadline:
            break
        if no_improve > 12:
            temperature = min(0.35, temperature * 1.15)
            no_improve = 0
        else:
            temperature = max(0.04, temperature * 0.96)
        pert_type = rng.randint(4)
        h_pert = best_h.copy()
        if pert_type == 0:
            np_ = rng.randint(max(10, n//10), max(25, n//4))
            idxs = rng.choice(n, size=np_, replace=False)
            h_pert[idxs] += rng.randn(np_) * temperature
        elif pert_type == 1:
            bl = rng.randint(max(5, n//20), max(20, n//5))
            st = rng.randint(0, max(1, n - bl))
            h_pert[st:st+bl] = 1.0 - h_pert[st:st+bl]
        elif pert_type == 2:
            bl = rng.randint(max(15, n//6), max(50, n//2))
            st = rng.randint(0, max(1, n - bl))
            t = np.linspace(0, 2*np.pi, bl)
            fr = rng.uniform(1.0, 4.0)
            h_pert[st:st+bl] += temperature * np.sin(fr*t + rng.uniform(0, 2*np.pi))
        else:
            n1 = rng.randint(10, 40)
            idxs1 = rng.choice(n, size=n1, replace=False)
            h_pert[idxs1] += rng.randn(n1) * temperature * 1.3
        h_pert = project_feasible(h_pert, n)
        rem = (deadline - time.time()) / (n_shakes - s) if deadline else 8.0
        dl = time.time() + min(rem, 12.0)
        h_opt, c_opt = coord_cd(h_pert, n, max_passes=250,
                                deadline=min(dl, deadline) if deadline else dl, grid_size=50)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 20
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(500, 0.5)
        n = 500
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    # Phase 1: SLSQP
    if time.time() < deadline - 350:
        h1, c1 = slsqp_optimize(h, n, [100, 500, 2000], maxiter=300)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Coordinate descent
    h2, c2 = coord_cd(best_h, n, max_passes=1500, deadline=deadline - 280, grid_size=100)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Shake and re-optimize
    if time.time() < deadline - 180:
        h3, c3 = shake_and_optimize(best_h, n, rng, n_shakes=55, deadline=deadline - 140)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Different resolutions
    for n_try in [400, 300]:
        if time.time() > deadline - 80:
            break
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = slsqp_optimize(h_int, n_try, [2000, 10000], maxiter=200)
        h_opt, c_opt = coord_cd(h_opt, n_try, max_passes=800, deadline=deadline - 60, grid_size=80)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    # Phase 5: Final high-precision coordinate descent
    h5, c5 = coord_cd(best_h, len(best_h), max_passes=5000, deadline=deadline - 30, grid_size=250)
    if c5 < best_c:
        best_h, best_c = h5, c5
    # Phase 6: Final shake at precision
    if time.time() < deadline - 10:
        h6, c6 = shake_and_optimize(best_h, len(best_h), rng, n_shakes=15, deadline=deadline - 5)
        if c6 < best_c:
            best_h, best_c = h6, c6
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
