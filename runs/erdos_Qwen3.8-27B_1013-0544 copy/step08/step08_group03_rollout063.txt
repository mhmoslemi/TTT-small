```python
def project_feasible(h, n):

            proj = np.clip(proj, 0.0, 1.0)
```

```python
def lse_gradient(h, n, tau):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    exp_vals = np.exp(tau * (c - c_max))
    w = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for p in range(n):
        s1 = np.dot(one_minus_h, w[p:p+n][::-1])
        s2 = np.dot(h, w[n-1-p:2*n-1-p])
        grad[p] = s1 - s2 + 2.0 * h[p] * w[n-1]
    return grad
```

```python
def fista_lse(h, n, tau, steps=1000, step_size=None, deadline=None):
    h_cur = h.copy()
    h_prev = h.copy()
    t_k = 1.0
    if step_size is None:
        # Estimate Lipschitz constant
        # L ~ 4*tau*n (rough bound)
        step_size = 1.0 / (4.0 * tau * n)
    for _ in range(steps):
        if deadline is not None and time.time() > deadline:
            break
        # FISTA momentum
        t_prev = t_k
        y = h_cur + (t_prev - 1.0) / (t_k + 1.0) * (h_cur - h_prev)
        # Gradient at y
        g = lse_gradient(project_feasible(y, n), n, tau)
        # Gradient step
        z = y - step_size * g
        # Projection
        h_next = project_feasible(z, n)
        # Update t
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        h_prev = h_cur
        h_cur = h_next
    return h_cur
```
