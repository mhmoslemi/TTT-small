def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-14:
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        if mask.any():
            y[mask] += d / mask.sum()
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_obj(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    r = alpha * (c - m)
    r -= r.max()
    return float((m + np.log(r.exp().sum()) / alpha) * (2.0 / n))

def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    r = alpha * (c - m)
    r -= r.max()
    w = np.exp(r)
    w /= w.sum()
    dx = 2.0 / n
    A = np.convolve(1.0 - h, w, mode="full")
    B = np.convolve(h, w[::-1], mode="full")
    s = n - 1
    return (A - B)[s:s+n] * dx

def sum_fun(h, n):
    return h.sum() - 0.5 * n

    return np.ones(n)

def slsqp_cont(h, n, alphas, maxiter, deadline):
