import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def fast_coord_cd(h, n, max_passes=2000, deadline=None, grid_size=64):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    deff = 2 * n - 2
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        order = range(n) if p % 2 == 0 else range(n - 1, -1, -1)
        for idx in order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 <= 1e-10 or h0 >= 1.0 - 1e-10:
                continue
            lo_d, hi_d = -h0, 1.0 - h0
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(deff, max(deff - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            M = len(ms)
            a = np.zeros(M)
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms]
            parts = []
            if m_lo > 0:
                parts.append(c[:m_lo].max())
            if m_hi < deff:
                parts.append(c[m_hi + 1:].max())
            unaff_max = max(parts) if parts else 0.0
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            diag_pos = n - 1 - m_lo
            has_diag = (n - 1 >= m_lo) and (n - 1 <= m_hi)
            if has_diag:
                nv[:, diag_pos] -= d_grid[:, None] ** 2
            row_max = nv.max(axis=1)
            vals = np.maximum(unaff_max, row_max)
            best_i = int(np.argmin(vals))
            best_val = float(vals[best_i])
            if best_val >= current_c5 - 1e-14:
                continue
            best_d = d_grid[best_i]
            if 0 < best_i < grid_size - 1:
                y = vals[best_i - 1:best_i + 2]
                x = d_grid[best_i - 1:best_i + 2]
                denom = (y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2] * (x[0] - x[1]))
                if abs(denom) > 1e-12:
                    xv = 0.5 * (y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2] * (x[0] - x[1])) / denom
                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv * a
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag_pos] -= xv ** 2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val:
                            best_val = v1
                            best_d = xv
            h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
            c[ms] = c_aff + best_d * a
            if has_diag:
                c[n - 1] -= best_d ** 2
            new_max = max(unaff_max, float(c[ms].max()))
            if new_max < current_c5:
                current_c5 = new_max
                improved = True
        if not improved:
            break
        if (p + 1) % 500 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def pair_cd(h, n, max_passes=200, deadline=None, gs=24):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    deff = 2 * n - 2
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for idx in range(0, n - 1, 2):
            if deadline is not None and time.time() > deadline:
                break
            h0a, h0b = h[idx], h[idx + 1]
            if (h0a <= 1e-10 and h0b <= 1e-10) or (h0a >= 1-1e-10 and h0b >= 1-1e-10):
                continue
            ga = np.linspace(max(-h0a, 0), min(1-h0a, 1), gs) if h0a > 1e-10 and h0a < 1-1e-10 else np.array([0.0])
            gb = np.linspace(max(-h0b, 0), min(1-h0b, 1), gs) if h0b > 1e-10 and h0b < 1-1e-10 else np.array([0.0])
            if len(ga) == 1 and len(gb) == 1:
                continue
            best_val = current_c5
            best_da, best_db = 0.0, 0.0
            for da in ga:
                for db in gb:
                    if abs(da) < 1e-12 and abs(db) < 1e-12:
                        continue
                    na, nb = h0a + da, h0b + db
                    if na < -1e-12 or na > 1+1e-12 or nb < -1e-12 or nb > 1+1e-12:
                        continue
                    cn = c.copy()
                    for ii, dd in ((idx, da), (idx+1, db)):
                        m1 = max(0, min(n-1-ii, ii))
                        m2 = min(deff, max(deff-ii, ii+n-1))
                        ms = np.arange(m1, m2+1)
                        a = np.zeros(len(ms))
                        jh = ii + ms - (n-1)
                        vh = (jh >= 0) & (jh < n)
                        a[vh] += (1.0 - h[jh[vh]])
                        jg = ii - ms + (n-1)
                        vg = (jg >= 0) & (jg < n)
                        a[vg] -= h[jg[vg]]
                        cn[ms] += dd * a
                        if n-1 >= m1 and n-1 <= m2:
                            cn[n-1] -= dd * dd
                    vm = float(cn.max())
                    if vm < best_val - 1e-13:
                        best_val = vm
                        best_da, best_db = da, db
            if abs(best_da) > 1e-10 or abs(best_db) > 1e-10:
                h[idx] = float(np.clip(h0a + best_da, 0.0, 1.0))
                h[idx+1] = float(np.clip(h0b + best_db, 0.0, 1.0))
                c = np.correlate(h, 1.0 - h, mode='full')
                current_c5 = float(c.max())
                improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)
