def coord_cd_exact(h, n, max_passes=5000, deadline=None, use_quadratic_check=True):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        direction = range(n) if p % 2 == 0 else range(n-1, -1, -1)
        improved = False
        
        for idx in direction:
            if deadline is not None and time.time() > deadline:
                break
            
            h0 = h[idx]
            if h0 < 1e-12:
                lo_d, hi_d = 0.0, 1.0
            elif h0 > 1.0 - 1e-12:
                continue
            else:
                lo_d, hi_d = -h0, 1.0 - h0
            
            if hi_d - lo_d < 1e-12:
                continue
            
            # Determine affected c indices
            m_lo = max(0, min(n-1-idx, idx))
            m_hi = min(2*n-2, max(2*n-2-idx, idx+n-1))
            ms = np.arange(m_lo, m_hi+1)
            M = len(ms)
            
            # Compute coefficient a for each affected c[m]
            a = np.zeros(M)
            jh = idx + ms - (n-1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n-1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            
            # b = current c values for affected indices
            b = c[ms].copy()
            
            # Unaffected max
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            
            # Find exact 1D minimum (ignoring quadratic)
            best_d, best_val = exact_1d_min(a, b, unaff_max, lo_d, hi_d)
            
            # Check quadratic (diagonal)
            if use_quadratic_check:
                diag_pos = np.where(ms == (n-1))[0]
                if len(diag_pos) > 0:
                    a_diag = a[diag_pos[0]]
                    b_diag = b[diag_pos[0]]
                    c_diag_at_d = b_diag + a_diag * best_d - best_d**2
                    if c_diag_at_d > best_val + 1e-10:
                        # Quadratic is active; do fine grid search
                        grid = np.linspace(lo_d, hi_d, 200)
                        vals = np.maximum(unaff_max, (b[None,:] + grid[:,None]*a[None,:]).max(axis=1))
                        # Subtract d^2 at diagonal
                        vals -= (grid**2)[np.newaxis, :] * 0  # placeholder
                        # Actually need to handle the diagonal separately
                        nv = b[None,:] + grid[:,None]*a[None,:]
                        nv[:, diag_pos[0]] -= grid**2
                        vals = np.maximum(unaff_max, nv.max(axis=1))
                        best_i = int(np.argmin(vals))
                        best_d = grid[best_i]
                        best_val = float(vals[best_i])
            
            if abs(best_d) > 1e-12 and best_val < current_c5 - 1e-13:
                h[idx] = h0 + best_d
                # Update c
                c[ms] = b + best_d * a
                diag_pos = np.where(ms == (n-1))[0]
                if len(diag_pos) > 0:
                    c[ms[diag_pos[0]]] -= best_d**2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True
        
        if not improved:
            break
        if (p+1) % 200 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_1d_min(a, b, unaff_max, lo, hi):
    if hi - lo < 1e-14:
        d = lo
        val = max(unaff_max, float(np.max(a * d + b))) if len(b) > 0 else unaff_max
        return d, val
    val_lo = max(unaff_max, float(np.max(a * lo + b)))
    val_hi = max(unaff_max, float(np.max(a * hi + b)))
    t_lo = unaff_max
    t_hi = max(val_lo, val_hi)
    best_d = 0.5 * (lo + hi)
    pos_mask = a > 1e-14
    neg_mask = a < -1e-14
    zero_mask = ~(pos_mask | neg_mask)
    for _ in range(50):
        t_mid = 0.5 * (t_lo + t_hi)
        lb = lo
        ub = hi
        feasible = True
        if np.any(neg_mask):
            lb = max(lb, float(np.max((t_mid - b[neg_mask]) / a[neg_mask])))
        if np.any(pos_mask):
            ub = min(ub, float(np.min((t_mid - b[pos_mask]) / a[pos_mask])))
        if np.any(zero_mask) and float(np.max(b[zero_mask])) > t_mid + 1e-12:
            feasible = False
        elif lb > ub + 1e-12:
            feasible = False
        if feasible:
            t_hi = t_mid
            best_d = 0.5 * (lb + ub)
        else:
            t_lo = t_mid
    best_d = float(np.clip(best_d, lo, hi))
    return best_d, t_hi

def coord_cd_exact(h, n, max_passes=5000, deadline=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    prev_c5 = current_c5
    stagnant = 0
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        direction = range(n) if p % 2 == 0 else range(n - 1, -1, -1)
        improved = False
        for idx in direction:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-12:
                lo_d, hi_d = 0.0, 1.0
            elif h0 > 1.0 - 1e-12:
                continue
            else:
                lo_d, hi_d = -h0, 1.0 - h0
            if hi_d - lo_d < 1e-12:
                continue
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            b = c[ms].copy()
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            best_d, best_val = exact_1d_min(a, b, unaff_max, lo_d, hi_d)
            diag_pos = np.where(ms == (n - 1))[0]
            if len(diag_pos) > 0:
                cdiag = b[diag_pos[0]] + a[diag_pos[0]] * best_d - best_d ** 2
                if cdiag > best_val + 1e-10:
                    grid = np.linspace(lo_d, hi_d, 300)
                    nv = b[None, :] + grid[:, None] * a[None, :]
                    nv[:, diag_pos[0]] -= grid ** 2
                    vals = np.maximum(unaff_max, nv.max(axis=1))
                    bi = int(np.argmin(vals))
                    best_d = float(grid[bi])
                    best_val = float(vals[bi])
            if abs(best_d) > 1e-12 and best_val < current_c5 - 1e-13:
                h[idx] = h0 + best_d
                c[ms] = b + best_d * a
                if len(diag_pos) > 0:
                    c[ms[diag_pos[0]]] -= best_d ** 2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True
        if not improved:
            break
        if (p + 1) % 150 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
        if (p + 1) % 50 == 0:
            if abs(prev_c5 - current_c5) < 1e-12:
                stagnant += 1
                if stagnant > 4:
                    break
            else:
                stagnant = 0
            prev_c5 = current_c5
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    if deadline - time.time() > 300:
        for alpha in [500, 5000, 50000]:
            if time.time() > deadline - 300:
                break
            try:

                         'jac': lambda x: np.ones(n)}]
                res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                               bounds=[(0.0, 1.0)] * n, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-16})

                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
            except Exception:
                pass
    # Phase 2: Main exact coordinate descent
    h2, c2 = coord_cd_exact(best_h, n, max_passes=8000, deadline=deadline - 200)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Random restarts with perturbation
    for r in range(8):
        if deadline - time.time() < 80:
            break
        h_r = best_h.copy()
        n_pert = max(5, n // 20)
        idx_pert = rng.choice(n, n_pert, replace=False)
        h_r[idx_pert] = rng.random(n_pert)
        h_r = project_feasible(h_r, n)
        h_opt, c_opt = coord_cd_exact(h_r, n, max_passes=3000, deadline=deadline - 50)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    # Phase 4: Try different n by interpolation
    for n_try in [300, 400, 600]:
        if deadline - time.time() < 120:
            break
        if n_try == n:
            continue
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_cd_exact(h_int, n_try, max_passes=3000, deadline=deadline - 60)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    # Phase 5: Final polish
    if deadline - time.time() > 30:
        h_fin, c_fin = coord_cd_exact(best_h, len(best_h), max_passes=5000, deadline=deadline)
        if c_fin < best_c:
            best_h, best_c = h_fin, c_fin
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
