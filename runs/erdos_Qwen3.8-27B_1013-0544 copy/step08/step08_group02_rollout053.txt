def sm_val(h, n, alpha):
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    z = alpha*(c-m)
    em = z.max()
    e = np.exp(z-em)
    s = e.sum()
    return (m + np.log(s)/alpha)*(2.0/n)

def sm_grad(h, n, alpha):
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    z = alpha*(c-m)
    em = z.max()
    e = np.exp(z-em)
    p = e/e.sum()
    off = n-1
    g = np.zeros(n)
    for k in range(2*n-1):
        w = p[k]
        if w < 1e-12: continue
        d = k-off
        if d >= 0:
            m_n = n-d
            g[:m_n] += w*(1.0-h[d:d+m_n])
            g[d:] -= w*h[:m_n]
        else:

            m_n = n-e_i
            g[e_i:] += w*(1.0-h[:m_n])
            g[:m_n] -= w*h[e_i:e_i+m_n]
    return (2.0/n)*g

def sum_jac(x, n): return np.ones_like(x)

def opt_slsqp(h, n, deadline, alpha_start=20, steps=6):

    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work = best_h.copy()
    bounds = [(0.0, 1.0)]*n
    cons = [{'type':'eq','fun':sum_con,'jac':sum_jac,'args':(n,)}]
    alphas = [alpha_start * (5**i) for i in range(steps)]
    max_iters = [60, 50, 45, 40, 30, 25]
    for i, (alpha, mx) in enumerate(zip(alphas, max_iters)):
        if time.time() > deadline - 3.0: break
        try:
            res = minimize(sm_val, work, args=(n, alpha), method='SLSQP',
                          jac=sm_grad, bounds=bounds, constraints=cons,
                          options={'maxiter': mx, 'ftol': 1e-15, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)): continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best_h, best_c = cur, cc
            work = cur if cc <= best_c else best_h
        except: pass
    return best_h, best_c

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.clip(np.nan_to_num(np.asarray(z, dtype=float), nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > target: lo = mid
        else: hi = mid
    y = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    res = target - y.sum()
    if abs(res) > 1e-12:
        mask = (y < 1.0-1e-12) if res > 0 else (y > 1e-12)
        if mask.any():
            room = (1.0-y) if res > 0 else y
            tot = room[mask].sum()
            if tot > 1e-14: y[mask] += (res/tot)*room[mask]
    return np.clip(y, 0.0, 1.0)

def sm_val(h, n, alpha):
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    z = alpha*(c - m)
    em = z.max()
    e = np.exp(z - em)
    s = e.sum()
    return (m + np.log(s)/alpha) * (2.0/n)

def sm_grad(h, n, alpha):
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    z = alpha*(c-m)
    em = z.max()
    e = np.exp(z-em)
    p = e / e.sum()
    off = n - 1
    g = np.zeros(n)
    for k in range(2*n-1):
        w = p[k]
        if w < 1e-12: continue
        d = k - off
        if d >= 0:
            m_n = n - d
            g[:m_n] += w*(1.0-h[d:d+m_n])
            g[d:] -= w*h[:m_n]
        else:
            e_i = -d
            m_n = n - e_i
            g[e_i:] += w*(1.0-h[:m_n])
            g[:m_n] -= w*h[e_i:e_i+m_n]
    return (2.0/n)*g

def _sum_con(x, n):
    return x.sum() - 0.5*n

def _sum_jac(x, n):
    return np.ones_like(x)

def opt_slsqp(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [(20,80),(100,60),(500,50),(2000,40),(10000,30),(50000,25),(200000,20)]
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work = best_h.copy()
    bounds = [(0.0, 1.0)]*n
    cons = [{'type':'eq','fun':_sum_con,'jac':_sum_jac,'args':(n,)}]
    for alpha, mx in alphas:
        if time.time() > deadline - 2.0: break
        try:
            res = minimize(sm_val, work, args=(n, alpha), method='SLSQP',
                         jac=sm_grad, bounds=bounds, constraints=cons,
                         options={'maxiter':mx, 'ftol':1e-15, 'disp':False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)): continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c: best_h, best_c = cur, cc
            work = cur if cc <= best_c else best_h
        except: pass
    return best_h, best_c

def pair_refine(h, n, deadline, rng, max_tries=5000):
    h = h.copy()
    c = np.correlate(h, 1.0-h, mode='full')
    off = n - 1
    accepts = 0
    no_imp = 0
    for _ in range(max_tries):
        if time.time() > deadline: break
        cur_max = float(c.max())
        # Pick which shifts to consider
        tol = max(0.01, 0.001*cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        # Pick a random pair from active shifts
        kk = active[rng.randint(active.size)]
        d = int(kk) - off
        if d >= 0:
            m = n - d
            if m <= 0: continue
            j = rng.randint(m)
            i = j + d
        else:
            e = -d
            m = n - e
            if m <= 0: continue
            j = e + rng.randint(m)
            i = j - e
        # Compute the gradient directions for this pair
        # A[i,k] = d f_k / d h[i], computed implicitly
        Bi = _pair_grad_i(h, n, i)
        Bj = _pair_grad_j(h, n, j)
        B = Bi - Bj
        # Optimal u: minimize max_k (c[k] + u*B[k] + quad[k]*u^2)
        # For now, use the quadratic approximation
        quad = np.zeros(2*n-1)
        quad[off] = -2.0
        quad[off + i - j] = 1.0
        quad[off + j - i] = 1.0
        # Find best u in [max(0,-h[i]), min(1-h[i], h[j])]
        lo_u = -h[i]
        hi_u = min(1.0-h[i], h[j])
        if hi_u - lo_u < 1e-14: continue
        # Parabolic approximation for each k, then take max
        # For speed, try a few candidate u values
        best_u = 0.0
        best_val = cur_max

            if val < best_val:

                best_u = trial_u

            h[j] -= best_u
            c = c + best_u*B + best_u**2*quad
            accepts += 1
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 2000:
            # Reset tolerance
            no_imp = 0
            tol = max(0.005, 0.0005*cur_max)
    return h, compute_c5(h, n)
