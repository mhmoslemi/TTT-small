def coord_cd(h, n, max_passes=4, deadline=None, grid=48, tol=1e-12):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    A = np.zeros(L)
    xs0 = np.linspace(0.0, 1.0, grid)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = float(x[p])
            A.fill(0.0)
            A[p:p + n] += om
            st = diag - p
            A[st:st + n] -= x
            cur = float(c.max())
            d = xs0[None, :] - old
            cand = c[None, :] + d[:, None] * A[None, :]
            cand[:, diag] -= d * d
            vals = cand.max(axis=1)
            i = int(np.argmin(vals))
            bx = float(xs0[i])
            bv = float(vals[i])
            if bv < cur - tol:
                # Refine with ternary-search-like approach
                sp = 1.0 / (grid - 1)
                lo = max(0.0, bx - 2 * sp)
                hi = min(1.0, bx + 2 * sp)
                xx = np.linspace(lo, hi, 32)
                d2 = xx - old
                cand2 = c[None, :] + d2[:, None] * A[None, :]
                cand2[:, diag] -= d2 * d2
                mm = cand2.max(axis=1)
                j = int(np.argmin(mm))
                if mm[j] < bv:
                    bx = float(xx[j])
                    bv = float(mm[j])
                if bv < cur - tol:
                    dd = bx - old
                    c[p:p + n] += dd * om
                    c[st:st + n] -= dd * x
                    c[diag] -= dd * dd
                    x[p] = bx

                    improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - tol:

    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)

    else:

def binary_trial(c, h, om, i, j, n):
    d = c.copy()
    d[i:i + n] -= om
    st = n - 1 - i
    d[st:st + n] += h
    d[j:j + n] += om
    st = n - 1 - j
    d[st:st + n] -= h
    d[n - 1] -= 2.0
    d[n - 1 + i - j] += 1.0
    d[n - 1 + j - i] += 1.0
    return d, float(d.max())

def binary_sa(h_init, n, rng, deadline, t_start=0.01, t_end=0.001, cooling=0.9999):
    """Simulated annealing on binary space."""
    b = binarize(h_init, n)
    if b is None:
        return None, np.inf
    dx = 2.0 / n
    best = b.copy()
    c = np.correlate(best, 1.0 - best, mode="full")
    bcm = float(c.max())
    cur = b.copy()
    cc = c.copy()
    cm = bcm
    T = t_start
    stagnant = 0
    while time.time() < deadline and T > t_end * 0.5:
        i, j = rand_one_zero(cur, n, rng)
        if i == j:
            continue
        om = 1.0 - cur[::-1]
        c2, nm = binary_trial(cc, cur, om, i, j, n)
        if nm < cm - 1e-10:
            cc = c2
            cm = nm
            cur[i] = 0.0
            cur[j] = 1.0
            if cm < bcm:
                best = cur.copy()
                bcm = cm
                stagnant = 0
            stagnant = max(0, stagnant - 1)
        else:
            # Accept with probability exp(-(nm-cm)/T)
            if nm - cm < 10 * T:
                p = np.exp(-(nm - cm) / T)
                if rng.random() < p:
                    cc = c2
                    cm = nm
                    cur[i] = 0.0
                    cur[j] = 1.0
        stagnant += 1
        if stagnant > 500:
            T *= 0.9
            stagnant = 0
        T *= cooling
    return best, bcm * dx

def binary_optimize(h_init, n, rng, deadline):

    b = binarize(h_init, n)
    if b is None:
        return None, np.inf
    dx = 2.0 / n
    best = b.copy()
    bc = np.correlate(best, 1.0 - best, mode="full")
    bcm = float(bc.max())
    
    # Phase 1: Greedy local search
    cur = best.copy()
    c = bc.copy()
    cm = bcm
    no_improve = 0
    while time.time() < deadline and no_improve < 20000:
        i, j = rand_one_zero(cur, n, rng)
        if i == j:
            continue
        om = 1.0 - cur[::-1]
        c2, nm = binary_trial(c, cur, om, i, j, n)
        if nm < cm - 1e-10:
            c = c2
            cm = nm
            cur[i] = 0.0
            cur[j] = 1.0
            om[n - 1 - i] = 1.0
            om[n - 1 - j] = 0.0
            no_improve = 0
        else:
            no_improve += 1
        if cm < bcm:
            best = cur.copy()
            bcm = cm
    
    # Phase 2: SA from current state
    if time.time() < deadline:
        sa_best, sa_c = binary_sa(cur, n, rng, deadline, t_start=0.005, t_end=0.0005, cooling=0.99995)
        if sa_best is not None and sa_c < bcm * dx:
            best = sa_best
            bcm = sa_c / dx
    
    # Phase 3: Another greedy pass
    if time.time() < deadline - 5:
        cur = best.copy()
        c = np.correlate(cur, 1.0 - cur, mode="full")
        cm = float(c.max())
        no_improve = 0
        while time.time() < deadline - 5 and no_improve < 10000:
            i, j = rand_one_zero(cur, n, rng)
            if i == j:
                continue

                cm = nm

                cur[j] = 1.0
                om[n - 1 - i] = 1.0
                om[n - 1 - j] = 0.0
                no_improve = 0
            else:
                no_improve += 1
            if cm < bcm:
                best = cur.copy()
                bcm = cm

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx

    return smooth_val_grad(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]

    return x.sum() - 0.5 * n

    return np.ones_like(x, dtype=float)

        c2, nm = binary_trial(c, cur, om, i, j, n)
        if nm < cm - 1e-10:
            c, cm = c2, nm
            cur[i], cur[j] = 0.0, 1.0
            if cm < bcm:
                best, bcm = cur.copy(), cm
                stall = 0
        else:
            diff = nm - cm
            if diff < 8.0 * T and rng.random() < np.exp(-diff / T):
                c, cm = c2, nm
                cur[i], cur[j] = 0.0, 1.0
        stall += 1
        if stall > 800:
            T *= 0.85
            stall = 0
        T *= cool
    return best, bcm

def binary_optimize(h_init, n, rng, deadline):
    b = binarize(h_init, n)
    if b is None:
        return None, np.inf
    dx = 2.0 / n
    best, bcm = binary_greedy(b, n, rng, min(deadline, time.time() + 40.0))
    # SA phase
    if time.time() < deadline:
        sa_b, sa_c = binary_sa(best, n, rng, min(deadline, time.time() + 80.0))
        if sa_c < bcm:
            best, bcm = sa_b, sa_c
    # Re-greedy after SA
    if time.time() < deadline:
        best2, bcm2 = binary_greedy(best, n, rng, deadline, max_no_improve=8000)
        if bcm2 < bcm:
            best, bcm = best2, bcm2
    # Kick-start perturbations
    while time.time() < deadline - 5.0:
        pert = best.copy()
        for _ in range(6):
            i, j = rand_one_zero(pert, n, rng)
            if i != j:
                pert[i], pert[j] = 0.0, 1.0
        p2, c2 = binary_greedy(pert, n, rng, min(deadline - 5.0, time.time() + 20.0))
        if c2 < bcm:
            best, bcm = p2, c2
        else:
            break
    return best, bcm * dx

def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(5.0, budget_s - 5.0)
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 500
        h0 = np.full(n0, 0.5)
    h0 = project_simplex(h0, n0)
    best_h, best_c, n_best = h0.copy(), float(compute_c5(h0, n0)), int(n0)
    rev = h0[::-1]
    for v in (0.5 * (h0 + rev), 0.5 * (h0 + (1.0 - rev))):
        vv = project_simplex(v, n_best)
        cv = compute_c5(vv, n_best)
        if cv < best_c:
            best_h, best_c = vv, cv
    # Phase 1: SLSQP + coord descent on initial
    t1 = min(deadline - 250.0, t0 + 100.0)
    if time.time() < t1:
        h1, c1 = slsqp_polish(best_h, n_best, [500, 3000, 15000, 80000], 80, t1)
        if c1 < best_c:
            best_h, best_c = h1, c1
    t1b = min(deadline - 200.0, time.time() + 60.0)
    if time.time() < t1b:
        h1, c1 = coord_cd(best_h, n_best, max_passes=10, deadline=t1b, grid=56)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Binary optimization with SA
    t2 = min(deadline - 120.0, time.time() + 250.0)
    if n_best % 2 == 0 and time.time() < t2:
        hb, cb = binary_optimize(best_h, n_best, rng, t2)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb
            t2b = min(deadline - 90.0, time.time() + 60.0)
            if time.time() < t2b:
                h1, c1 = slsqp_polish(best_h, n_best, [5000, 30000], 60, t2b)
                if c1 < best_c:
                    best_h, best_c = h1, c1
                h1, c1 = coord_cd(best_h, n_best, max_passes=5, deadline=t2b, grid=56)
                if c1 < best_c:
                    best_h, best_c = h1, c1
    # Phase 3: Multi-resolution
    cands = [n_best, 400, 600, 350, 550, 450, 300, 650, 700, 380]
    tried = set()
    for nt in cands:
        if len(tried) >= 5:
            break
        if time.time() > deadline - 70.0:
            break
        if nt in tried or nt < 200 or nt > 900 or nt % 2:
            continue
        tried.add(nt)
        ht = transfer_h(best_h, n_best, nt)
        cv = compute_c5(ht, nt)
        if cv < best_c:
            best_h, best_c, n_best = ht, cv, int(nt)
        chunk = min(deadline - 50.0, time.time() + 70.0)
        if time.time() < chunk:
            h1, c1 = slsqp_polish(ht, nt, [3000, 15000], 50, chunk)
            h1, c1 = coord_cd(h1, nt, max_passes=4, deadline=chunk, grid=48)
            if c1 < best_c:
                best_h, best_c, n_best = h1, c1, int(nt)
        if nt % 2 == 0:
            chunk2 = min(deadline - 40.0, time.time() + 50.0)
            if time.time() < chunk2:
                hb, cb = binary_optimize(best_h, nt, rng, chunk2)
                if hb is not None and cb < best_c:
                    best_h, best_c, n_best = hb, cb, int(nt)
    # Phase 4: Final polish
    tf = min(deadline - 10.0, time.time() + 50.0)
    if time.time() < tf:
        h1, c1 = slsqp_polish(best_h, n_best, [10000, 60000], 80, tf)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h1, c1 = coord_cd(best_h, n_best, max_passes=4, deadline=tf, grid=56)
        if c1 < best_c:
            best_h, best_c = h1, c1
    if n_best % 2 == 0 and time.time() < deadline - 3.0:
        hb, cb = binary_optimize(best_h, n_best, rng, deadline - 2.0)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb
    final_h = project_simplex(best_h, n_best)
    final_c = float(compute_c5(final_h, n_best))
    return final_h, final_c, int(n_best)
