import numpy as np
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lp_step(h, n, solver=None):
    """Solve LP: fix g=1-h, find h* minimizing max_k sum h*[n+k]*g[n]"""
    g = 1.0 - h
    dx = 2.0 / n
    var = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [var >= 0, var <= 1, cp.sum(var) == n / 2.0]
    
    # Build constraint matrix efficiently
    # c[m] = sum_n var[n+m] * g[n] for valid n
    # We'll build this as a sparse matrix
    rows = []
    cols = []
    vals = []
    
    for m in range(-(n-1), n):
        # valid n: 0 <= n < n_points and 0 <= n+m < n_points
        n_lo = max(0, -m)
        n_hi = min(n - 1, n - 1 - m)
        for pos in range(n_lo, n_hi + 1):
            rows.append(m + n - 1)  # row index in the constraint matrix
            cols.append(pos + m)     # column index (index into var)
            vals.append(g[pos])
    
    # Build sparse matrix
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints.append(A @ var <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    if solver is None:
        try:
            prob.solve(solver=cp.GLOP)
        except:
            prob.solve(solver=cp.ECOS)
    else:
        prob.solve(solver=solver)
    
    if var.value is not None:
        return np.clip(var.value, 0.0, 1.0)
    return h.copy()

    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        pos_mask = h < 1.0
        neg_mask = h > 0.0
        if diff > 0:
            tot = (1.0 - h[pos_mask]).sum()
            if tot > 0:
                h[pos_mask] += diff * (1.0 - h[pos_mask]) / tot
        else:
            tot = h[neg_mask].sum()
            if tot > 0:
                h[neg_mask] += diff * h[neg_mask] / tot
    return np.clip(h, 0.0, 1.0)

def build_A(g, n):
    A = lil_matrix((2*n-1, n))

        i_hi = min(n - 1, k)
        idx = np.arange(i_lo, i_hi + 1)
