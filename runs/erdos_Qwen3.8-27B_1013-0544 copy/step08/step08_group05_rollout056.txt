    """

    # d(h[i]/di) contribution:
    #   As first position: (1-h[i+m-n+1]) for m where 0<=i+m-n+1<n, i.e., n-1-i <= m <= 2n-2-i
    #   As second position: -h[i-(m-n+1)] = -h[i-m+n-1] for m where 0<=i-m+n-1<n, i.e., i <= m <= i+n-1

def binary_optimize_v2(h_bin, n, rng, deadline, max_evals=500000):

    dx = 2.0 / n
    h = h_bin.copy()

    cm = float(c.max())
    best_h = h.copy()

    best_c5 = cm * dx

    T_start = 0.5 * dx  # in units of c (before multiplying by dx)

    T_min = 0.1

            bad_indices = np.where(mask)[0] + dd

            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = int(rng.choice(ones))

                i, j = idx, idx - dd  # h[i]=1, h[j]=0
        
        # Try the swap

        new_h[i] = 0.0

def fast_swap(c, h, i, j, n):
    """Compute new correlation after swap h[i]:1->0, h[j]:0->1. O(n) time."""
    new_c = c.copy()
    
    # Effect of h[i]: 1 -> 0
    # First position: for m where 0 <= i+m-n+1 < n, i.e., n-1-i <= m <= 2n-2-i
    lo = max(0, n - 1 - i)
    hi = min(2*n - 2, 2*n - 2 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        # h[i + m - n + 1]
        new_c[lo:hi+1] -= (1.0 - h[i + m - n + 1])
    
    # Second position: for m where 0 <= i-m+n-1 < n, i.e., i <= m <= i+n-1 (and 0<=m<=2n-2)
    lo = max(0, i)
    hi = min(2*n - 2, i + n - 1)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        # h[i - m + n - 1] = h[(i + n - 1) - m]
        new_c[lo:hi+1] += h[(i + n - 1) - m]
    
    # Effect of h[j]: 0 -> 1
    lo = max(0, n - 1 - j)
    hi = min(2*n - 2, 2*n - 2 - j)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        new_c[lo:hi+1] += (1.0 - h[j + m - n + 1])
    
    lo = max(0, j)
    hi = min(2*n - 2, j + n - 1)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        new_c[lo:hi+1] -= h[(j + n - 1) - m]
    
    return new_c

import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def binarize(h, n):
    if n % 2:
        return None
    target = n // 2
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def fast_swap_update(c, h, i, j, n):
    """Update correlation after h[i]:1->0, h[j]:0->1. Returns new c."""
    nc = c.copy()
    # h[i] 1->0: first pos
    lo = max(0, n - 1 - i)
    hi = min(2 * n - 2, 2 * n - 2 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        nc[lo:hi+1] -= (1.0 - h[i + m - n + 1])
    # h[i] 1->0: second pos
    lo = max(0, i)
    hi = min(2 * n - 2, i + n - 1)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        nc[lo:hi+1] += h[(i + n - 1) - m]
    # h[j] 0->1: first pos
    lo = max(0, n - 1 - j)
    hi = min(2 * n - 2, 2 * n - 2 - j)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        nc[lo:hi+1] += (1.0 - h[j + m - n + 1])
    # h[j] 0->1: second pos
    lo = max(0, j)
    hi = min(2 * n - 2, j + n - 1)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        nc[lo:hi+1] -= h[(j + n - 1) - m]

def binary_sa(h_init, n, rng, deadline, max_evals=2000000):
    """Simulated annealing on binary solutions with fast swap evaluation."""
    dx = 2.0 / n
    h = h_init.copy()
    c = np.correlate(h, 1.0 - h, mode="full")
    cm = float(c.max())
    best_h = h.copy()
    best_cm = cm
    
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    T = 3.0
    T_min = 0.3
    total_evals = max_evals
    cool = (T_min / T) ** (1.0 / total_evals)
    
    evals = 0
    stagnation = 0
    start_time = time.time()
    
    while time.time() < deadline and evals < total_evals:
        # Find worst lag
        k_star = int(np.argmax(c))
        d_star = k_star - n + 1
        
        # Get bad pairs for worst lag
        if d_star == 0:
            # c[center] = sum h[i](1-h[i]) = 0 for binary, skip
            i = int(rng.choice(list(ones)))
            j = int(rng.choice(list(zeros)))
        elif d_star > 0:
            mask = (h[:n-d_star] == 1) & (h[d_star:] == 0)
            bads = np.where(mask)[0]
            if len(bads) > 0:
                idx = int(bads[rng.randint(len(bads))])
                i = idx
                j = idx + d_star
            else:
                i = int(rng.choice(list(ones)))
                j = int(rng.choice(list(zeros)))
        else:
            dd = -d_star
            mask = (h[dd:] == 1) & (h[:-dd] == 0)
            bads = np.where(mask)[0]
            if len(bads) > 0:
                idx = int(bads[rng.randint(len(bads))])
                i = idx + dd  # this is where h=1
                j = idx      # this is where h=0
            else:
                i = int(rng.choice(list(ones)))
                j = int(rng.choice(list(zeros)))
        
        if i not in ones or j not in zeros:
            continue
        
        # Perform swap
        new_c = fast_swap_update(c, h, i, j, n)
        new_cm = float(new_c.max())
        evals += 1
        
        delta = new_cm - cm
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            h[i] = 0.0
            h[j] = 1.0
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            c = new_c
            cm = new_cm
            if cm < best_cm - 1e-12:
                best_cm = cm
                best_h = h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1
        
        T *= cool
        
        if stagnation > 50000 and T > T_min:
            # Restart temperature
            T = max(T * 0.5, T_min)
            stagnation = 0
    
    return best_h, best_cm * dx

def smooth_polish(h, n, alphas, maxiter, deadline):
    """Polish using SLSQP with smooth max."""
    h = project_simplex(h, n)
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": lambda x, n: x.sum() - 0.5*n, "jac": lambda x, n: np.ones(n), "args": (n,)}]
    
    for a in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            return smooth_obj_and_grad(x, n, a)[0]
        def grad(x):
            return smooth_obj_and_grad(x, n, a)[1]
        try:
            res = minimize(obj, best, jac=grad, method="SLSQP",
                          bounds=bounds, constraints=cons,
                          options={"maxiter": maxiter, "ftol": 1e-10})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-12:
            best = nh
            bc = nc
    return best, bc
