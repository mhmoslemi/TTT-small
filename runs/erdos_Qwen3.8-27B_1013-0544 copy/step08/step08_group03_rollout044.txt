import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_cd(h, n, max_passes=200, deadline=None, grid_pts=25, gs_iters=30):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    D = np.zeros(2 * n - 1)
    D[n - 1] = 1.0
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            A = np.zeros(2 * n - 1)
            k1_lo = max(0, n - 1 - p)
            k1_hi = min(2 * n - 2, 2 * n - 2 - p)
            if k1_lo <= k1_hi:
                idx1 = np.arange(k1_lo, k1_hi + 1)
                A[idx1] += (1.0 - h[p + idx1 - (n - 1)])
            k2_lo = max(0, p - n + 1)
            k2_hi = min(2 * n - 2, p)
            if k2_lo <= k2_hi:
                idx2 = np.arange(k2_lo, k2_hi + 1)
                A[idx2] += -h[p - idx2 + (n - 1)]
            c0 = corr
            u_lo, u_hi = -h0, 1.0 - h0
            grid = np.linspace(u_lo, u_hi, grid_pts)
            vals = np.max(c0[None, :] + grid[:, None] * A[None, :] - (grid ** 2)[:, None] * D[None, :], axis=1)
            bi = int(np.argmin(vals))
            lo = grid[max(0, bi - 1)]
            hi = grid[min(grid_pts - 1, bi + 1)]
            if hi - lo > 1e-14:
                g = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - g * (hi - lo)
                d1 = lo + g * (hi - lo)
                fc1 = float(np.max(c0 + c1 * A - c1 * c1 * D))
                fd1 = float(np.max(c0 + d1 * A - d1 * d1 * D))
                for _ in range(gs_iters):
                    if fc1 < fd1:
                        hi, d1, fd1 = d1, c1, fc1
                        c1 = hi - g * (hi - lo)
                        fc1 = float(np.max(c0 + c1 * A - c1 * c1 * D))
                    else:
                        lo, c1, fc1 = c1, d1, fd1
                        d1 = lo + g * (hi - lo)
                        fd1 = float(np.max(c0 + d1 * A - d1 * d1 * D))
                t_star = (lo + hi) * 0.5
            else:
                t_star = grid[bi]
            t_star = float(np.clip(t_star, 0.0, 1.0))
            if abs(t_star - h0) > 1e-12:
                u = t_star - h0
                f_new = float(np.max(c0 + u * A - u * u * D))
                if f_new < current_max - 1e-14:
                    corr = c0 + u * A - u * u * D
                    h[p] = t_star
                    current_max = f_new
                    improved = True
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        current_max = float(np.max(corr))
        if not improved:
            break
    return h, current_max

def pnorm_slsqp(h, n, p_list, maxiter_each=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for p in p_list:
        if deadline is not None and time.time() > deadline:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full') * dx
            return float(np.sum(c ** p)) ** (1.0 / p)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
