3. Identify the top-k lags (e.g., those within 5% of max)

6. Repeat

def project_simplex(h, n):
    target = 0.5 * n
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + lo, 0.0, 1.0)
    # Fix any residual
    d = target - result.sum()
    if abs(d) > 1e-12:
        mask = (result > 1e-9) & (result < 1.0 - 1e-9)
        if mask.any():
            result[mask] += d / mask.sum()
    return np.clip(result, 0.0, 1.0)

def compute_c5(h, n):
    return float(np.correlate(h, 1.0 - h, mode="full").max()) * (2.0 / n)

def binarize(h, n):

        return None

    b = np.zeros(n, dtype=float)
    b[idx] = 1.0

def smooth_val_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    w = ee / z
    dx = 2.0 / n
    s = n - 1
    A = np.convolve(1.0 - h, w, mode="full")
    B = np.convolve(h, w[::-1], mode="full")
    grad = (A - B)[s:s + n] * dx
    val = (m + np.log(z) / alpha) * dx
    return val, grad

    s = h.sum() - 0.5 * n
    v += penalty * s * s
    g = g + 2.0 * penalty * s * np.ones(n)
    return v, g

        nm = float(c2.max())
        delta = nm - cm
        if delta < -1e-9 or (T > T_min and rng.rand() < np.exp(-max(delta, 0.0) / T)):
            h, c, cm = h2, c2, nm
            stall = 0
            if nm < bcm:
                best, bcm = h.copy(), nm
        else:
            stall += 1
        T = max(T_min, T * 0.9997)
        if stall > 5000:
            h, c = best.copy(), np.correlate(best, 1.0 - best, mode="full")
            cm = float(c.max())
            T = max(T_min * 3, 1.0)
            stall = 0
            if T < T_min + 0.05:
                break
    return best, bcm * (2.0 / n)

def transfer_h(h, n, nt):
    if nt == n:
        return project_simplex(h, n)
    return project_simplex(np.interp(np.linspace(0, 2, nt), np.linspace(0, 2, n), h), nt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + budget_s - 8.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 500
        h0 = np.full(n, 0.5)

    h0 = project_simplex(h0, n)
    best = h0.copy()
    best_c = compute_c5(h0, n)

    rev = h0[::-1]
    for v in (0.5 * (h0 + rev), 0.5 * (h0 + (1.0 - rev))):
        vv = project_simplex(v, n)
        cv = compute_c5(vv, n)
        if cv < best_c:
            best, best_c = vv, cv

    if n % 2 == 0:
        hb = binarize(best, n)
        if hb is not None:
            cb = compute_c5(hb, n)
            if cb < best_c:
                best, best_c = hb, cb

    t_bin = min(deadline - 500.0, t0 + 180.0)
    if n % 2 == 0 and time.time() < t_bin:
        bb, bc = binary_sa(binarize(best, n), n, rng, t_bin)
        if bc < best_c:
            best, best_c = bb, bc

    t_lbfgs = min(deadline - 300.0, t_bin + 300.0)
    if time.time() < t_lbfgs:
        h1, c1 = lbfgs_phase(best, n, t_lbfgs)
        if c1 < best_c:
            best, best_c = h1, c1

    t_cd1 = min(deadline - 150.0, t_lbfgs + 120.0)
    if time.time() < t_cd1:
        best, best_c = coord_descent(best, n, 5, t_cd1, grid=56, refine=14)

    t_slq = min(deadline - 80.0, t_cd1 + 100.0)
    if time.time() < t_slq:
        h1, c1 = slsqp_phase(best, n, t_slq)
        if c1 < best_c:
            best, best_c = h1, c1
        best, best_c = coord_descent(best, n, 3, t_slq, grid=64, refine=16)

    t_multi = deadline - 30.0
    cands = [400, 600, 450, 550, 350, 480, 520]
    for nt in cands:
        if time.time() > t_multi:
            break
        if nt % 2 or nt == n or nt < 100 or nt > 900:
            continue
        ht = transfer_h(best, n, nt)
        cv = compute_c5(ht, nt)
        if cv < best_c:
            best, best_c, n = ht, cv, nt
        if time.time() < t_multi - 5:
            h1, c1 = lbfgs_phase(ht, nt, min(t_multi, time.time() + 30.0))
            if c1 < best_c:
                best, best_c, n = h1, c1, nt
            best, best_c = coord_descent(best, n, 2, min(t_multi, time.time() + 20.0), grid=40, refine=10)

    if time.time() < deadline - 3.0:
        best, best_c = coord_descent(best, n, 3, deadline - 3.0, grid=64, refine=16)

    final_h = project_simplex(best, n)
    final_c = compute_c5(final_h
, n)
    return final_h, final_c, n
