def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            cap = 1.0 - h
            tot = cap.sum()
            if tot > 0:
                h += (diff / tot) * cap
        else:
            tot = h.sum()
            if tot > 0:
                h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-14:
        idx = np.where((h < 1.0) if res > 0 else (h > 0.0))[0]
        if idx.size:
            h[idx[0]] += res
            h[idx[0]] = np.clip(h[idx[0]], 0.0, 1.0)
    return h

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            cap = 1.0 - h
            tot = cap.sum()
            if tot > 0:
                h += (diff / tot) * cap
        else:
            tot = h.sum()
            if tot > 0:
                h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-14:
        idx = np.where((h < 1.0) if res > 0 else (h > 0.0))[0]
        if idx.size:
            h[idx[0]] += res
            h[idx[0]] = np.clip(h[idx[0]], 0.0, 1.0)
    return h

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(np.array(old, dtype=float), n)

def binary_search(h_init, n, iters, deadline, seed=0):

def coord_descent(h, n, max_passes, deadline, proj_every=4):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h = h.copy()
    best_val = cur
    center = n - 1
    L = 2 * n - 1
    pass_i = 0
    for _ in range(max_passes):
        pass_i += 1
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            hp = h[p]
            s_lo = p - (n - 1)
            s_hi = (n - 1) - p
            s_range = np.arange(s_lo, s_hi + 1)
            A = np.zeros(len(s_range))
            for idx_s in range(len(s_range)):
                sv = s_range[idx_s]
                co = 0.0
                if 0 <= p + sv < n:
                    co += 1.0 - h[p + sv]
                if 0 <= p - sv < n:
                    co -= h[p - sv]
                A[idx_s] = co
            s_idx = center + s_range
            B = c[s_idx] - hp * A
            zmask = (s_range == 0)
            B0 = B[zmask][0] if zmask.any() else 0.0
            nz = ~zmask
            Bnz = B[nz]
            Anz = A[nz]
            aff = np.zeros(L, dtype=bool)
            aff[s_idx] = True
            unaff = float(c[~aff].max()) if (~aff).any() else 0.0
            xb, vb = solve_1d(B0, Bnz, Anz, unaff)
            if vb < cur - 1e-11:
                h[p] = xb
                c[s_idx] = B + A * xb
                if zmask.any():
                    c[center] = B0 + xb - xb * xb
                cur = max(float(c[s_idx].max()), unaff)
                improved = True
        if pass_i % proj_every == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            cur = float(c.max())
        if cur < best_val:
            best_val = cur
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_val * (2.0 / n)
