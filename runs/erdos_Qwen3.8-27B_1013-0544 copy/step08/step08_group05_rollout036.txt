def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(55):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    y = np.clip(y, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += d / mask.sum()
    return y

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * dx

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = c.max()
        r = alpha * (c - m)
        r -= r.max()
        ee = np.exp(r)
        return (m + np.log(ee.sum()) / alpha) * (2.0 / n)
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = c.max()
        r = alpha * (c - m)
        r -= r.max()
        ee = np.exp(r)
        w = ee / ee.sum()
        A = np.convolve(1.0 - x, w, mode="full")
        B = np.convolve(x, w[::-1], mode="full")
        s = n - 1
        return (A - B)[s:s+n] * (2.0 / n)
    
    def sum_fun(x):
        return x.sum() - 0.5 * n
    def sum_jac(x):
        return np.ones_like(x)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_fun, "jac": sum_jac}]
    try:
        res = minimize(obj, h, jac=grad, method="SLSQP", bounds=bounds,
                       constraints=cons, options={"maxiter": maxiter, "ftol": 1e-12})

    except:
