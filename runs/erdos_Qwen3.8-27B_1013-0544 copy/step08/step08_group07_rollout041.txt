def eval_c(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * n
    s = float(h.sum())
    if abs(s - target) < 1e-13:
        return h
    if s < target:
        lo, hi = 0.0, 1.0
        for _ in range(60):
            mid = 0.5 * (lo + hi)
            vals = h + mid
            np.clip(vals, 0.0, 1.0, out=vals)
            if float(vals.sum()) < target:
                lo = mid
            else:
                hi = mid
        mid = 0.5*(lo+hi)
        np.clip(h + mid, 0.0, 1.0, out=h)
    else:
        lo, hi = 0.0, 1.0
        for _ in range(60):
            mid = 0.5*(lo+hi)
            vals = h - mid
            np.clip(vals, 0.0, 1.0, out=vals)
            if float(vals.sum()) > target:
                lo = mid
            else:
                hi = mid
        mid=0.5*(lo+hi)
        np.clip(h - mid, 0.0, 1.0, out=h)
    diff = target - float(h.sum())
    if abs(diff) > 1e-8:
       if diff > 0:
          room = 1.0 - h
          tot = float(room.sum())
          if tot > 1e-15: h += (diff / tot) * room
       else:
          tot = float(h.sum())
          if tot > 1e-15: h += (diff / tot) * h
       np.clip(h,0,1,out=h)
    return h

def linear_interval(const, pos_m, pos_b, neg_m, neg_b, dlo, dhi, T):
    if T < const - 1e-12:
       return None
    ilo = dlo; ihi = dhi
    if len(pos_m):
       v = float(np.min((T - pos_b) / pos_m))
       if v < ihi: ihi = v
    if ihi < ilo - 1e-12: return None
    if len(neg_m):
       v = float(np.max((T - neg_b) / neg_m))
       if v > ilo: ilo = v
    if ilo > ihi + 1e-12: return None
    return (ilo, ihi)

    T1 = hi + 1e-10
    if T1 >= curr_max:
        T1 = curr_max - 1e-11
    if T1 < hi:
        T1 = hi

    d = choose_coord(const, pos_m, pos_b, neg_m, neg_b, cd, ad, dlo, dhi, T1)
    if d is None or abs(d) < 1e-12:
        T2 = min(curr_max - 1e-12, hi + 1e-7)
        if T2 > T1 and T2 < curr_max:
            d = choose_coord(const, pos_m, pos_b, neg_m, neg_b, cd, ad, dlo, dhi, T2)
    if d is None or abs(d) < 1e-12:
        return None

    new_sub = c_raw + a_raw * d
    if 0 <= diag_rel < len(new_sub):
        new_sub[diag_rel] -= d * d

    new_max = const
    mx = float(np.max(new_sub))
    if mx > new_max:
        new_max = mx

    if new_max >= curr_max - 1e-10:
        return None

    h[p] = float(np.clip(h0 + d, 0.0, 1.0))
    c[mlo:mhi + 1] = new_sub
    return new_max, d

def interp_to(h, n):
    if len(h) == n:
        return project_feasible(h.copy(), n)
    old = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new = np.linspace(0.0, 2.0, n, endpoint=False)
    return project_feasible(np.interp(new, old, h), n)

def structured_start(n, kind, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    if kind == 0:
        h = 0.5 + 0.4 * np.sin(2.0 * np.pi * x)
    elif kind == 1:
        h = 0.5 + 0.4 * np.sin(4.0 * np.pi * x)
    elif kind == 2:
        h = (np.sin(4.0 * np.pi * x) > 0.0).astype(float)
    elif kind == 3:
        h = 0.5 + 0.3 * np.sin(2.0 * np.pi * x) + 0.15 * np.sin(6.0 * np.pi * x)
    elif kind == 4:
        h = rng.uniform(0.0, 1.0, n)
    else:
        h = (rng.rand(n) > 0.5).astype(float)
    return project_feasible(h, n)

def cd_polish(h_in, end_time, steps):
    n = len(h_in)
    h = project_feasible(h_in, n)
    c = eval_c(h)
    target = 0.5 * n
    cur_sum = float(h.sum())

    best_h = h.copy()
    best_pix = float(np.max(c))
    best_bound = best_pix * (2.0 / n)

    drift_limit = max(0.5, 0.01 * n)

    for step in steps:
        stalling = 0
        sweeps = 0
        while time.time() < end_time - 5:
            if sweeps >= 80:
                break

            moved = 0
            cur = float(np.max(c))
            pref, suff = make_pref_suff(c)

            start = 0 if (sweeps % 2 == 0) else n - 1
            stop = n if start == 0 else -1
            stride = 1 if start == 0 else -1

            for p in range(start, stop, stride):
                if time.time() >= end_time - 5:
                    break

                cur = float(np.max(c))
                res = exact_one_move(h, c, p, step, pref, suff, cur)
                if res is not None:
                    new_max, d = res
                    cur = new_max
                    cur_sum += d
                    moved += 1

                    if abs(cur_sum - target) > drift_limit:
                        h = project_feasible(h, n)
                        c = eval_c(h)
                        cur_sum = float(h.sum())
                        cur = float(np.max(c))
                        pref, suff = make_pref_suff(c)
                    else:
                        pref, suff = make_pref_suff(c)

            if moved:
                h = project_feasible(h, n)
                c = eval_c(h)
                cur_sum = float(h.sum())
                cur = float(np.max(c))
                b = cur * (2.0 / n)
                if b < best_bound - 1e-14:
                    best_bound = b
                    best_pix = cur
                    best_h = h.copy()

            sweeps += 1
            if moved == 0:
                stalling += 1
                if stalling >= 2:
                    break
            else:
                stalling = 0

            if cur > best_pix + max(0.5, 0.02 * best_pix):
                h = best_h.copy()
                c = eval_c(h)
                cur = float(np.max(c))
                cur_sum = float(h.sum())

    return best_h, best_bound

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(1.0, budget_s - 10.0)
    rng = np.random.RandomState(seed)

    try:
        base = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        base = structured_start(400, 3, rng)

    n0 = len(base)
    base = project_feasible(base, n0)
    best_h = base.copy()
    best_c = bound_from_h(best_h)

    end1 = min(deadline, t0 + max(60.0, 0.38 * budget_s))
    h1, c1 = cd_polish(best_h, end1, [0.16, 0.08, 0.04, 0.02, 0.01])
    if c1 < best_c:
        best_h, best_c = h1, c1

    reserve = max(60.0, 0.15 * budget_s)
    final_guard = deadline - reserve

    interp_ns = [32, 48, 64, 96, 144, 200, 320]
    start_cfgs = [
        (32, 0), (32, 2), (48, 1), (64, 3),
        (96, 4), (96, 5), (144, 2), (192, 3)
    ]

    for ns in interp_ns:
        if time.time() > final_guard:
            break
        if ns == len(best_h):
            continue
        remaining = final_guard - time.time()
        if remaining < 25:
            break
        alloc = min(85.0, max(22.0, 0.14 * remaining))
        hc = interp_to(best_h, ns)
        hc = project_feasible(hc + 0.004 * rng.randn(ns), ns)
        hc, cc = cd_polish(hc, time.time() + alloc, [0.22, 0.10, 0.045, 0.02])
        if cc < best_c:
            best_h, best_c = hc, cc

    for ns, kind in start_cfgs:
        if time.time() > final_guard:
            break
        remaining = final_guard - time.time()
        if remaining < 25:
            break
        alloc = min(70.0, max(20.0, 0.14 * remaining))
        hr = structured_start(ns, kind, rng)
        hr, cr = cd_polish(hr, time.time() + alloc, [0.25, 0.12, 0.05, 0.02])
        if cr < best_c:
            best_h, best_c = hr, cr

        if len(best_h) != ns and time.time() < final_guard - 30:
            hb = interp_to(hr, ns)
            alloc2 = min(35.0, max(12.0, 0.05 * remaining))
            hb, cb = cd_polish(hb, time.time() + alloc2, [0.08, 0.035, 0.015])
            if cb < best_c:
                best_h, best_c = hb, cb

    use_left = deadline - time.time()
    if use_left > max(320.0, 0.3 * budget_s) and len(best_h) < 800:
        n_up = min(900, max(400, 2 * len(best_h)))
        alloc = min(use_left - max(120.0, 0.12 * budget_s), max(160.0, 0.2 * budget_s))
        if alloc > 120:
            hu = interp_to(best_h, n_up)
            hu, cu = cd_polish(hu, time.time() + alloc, [0.06, 0.03, 0.015, 0.008])
            if cu < best_c:
                best_h, best_c = hu, cu

    if time.time() < deadline - 8:
        hf, cf = cd_polish(best_h, deadline - 4, [0.025, 0.012, 0.006, 0.003])
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    final_c = bound_from_h(best_h)
    return best_h, final_c, len(best_h)
