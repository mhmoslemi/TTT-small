        return h0
    m_lo_f = max(0, n - 1 - p)
    m_hi_f = min(2 * n - 2, n - 1 + p)
    m_lo_b = max(0, p)
    m_hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(m_lo_f, m_lo_b)
    m_hi = max(m_hi_f, m_hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p + (n - 1) - ms
    valid_f = (jf >= 0) & (jf < n)
    valid_b = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(valid_f):
        a[valid_f] += (1.0 - h[jf[valid_f]])
    if np.any(valid_b):
        a[valid_b] -= h[jb[valid_b]]
    c_aff = c[ms]
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(c[m_hi + 1:])))
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_aff[diag_idx[0]]
        a[diag_idx] = 0.0
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo = max(unaff_max, 0.0)
    L_hi = float(np.max(c))
    for _ in range(60):
        if L_hi - L_lo < 1e-14:
            break
        L_mid = (L_lo + L_hi) // 2 + (L_lo + L_hi) % 2 * 0 + (L_lo + L_hi) / 2.0
        L_mid = (L_lo + L_hi) / 2.0
        feasible = True
        if np.any(zero_mask) and np.any(c_aff[zero_mask] > L_mid + 1e-15):
            feasible = False
        x_low = 0.0
        x_high = 1.0
        if feasible and np.any(neg_mask):
            bounds = (L_mid - c_aff[neg_mask]) / a[neg_mask] + h0
            x_low = max(x_low, float(np.max(bounds)))
        if feasible and np.any(pos_mask):
            bounds = (L_mid - c_aff[pos_mask]) / a[pos_mask] + h0
            x_high = min(x_high, float(np.min(bounds)))
        if feasible and x_low > x_high + 1e-14:
            feasible = False
        if feasible and has_diag:
            Cq = L_mid - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if x_low >= r1 - 1e-14 and x_high <= r2 + 1e-14:
                    feasible = False
                elif x_low < r1 and x_high > r2:
                    if h0 - r1 <= r2 - h0:
                        x_high = min(x_high, r1)
                    else:
                        x_low = max(x_low, r2)
                elif x_low < r1:
                    x_high = min(x_high, r1)
                else:
                    x_low = max(x_low, r2)
        if feasible:
            L_hi = L_mid
        else:
            L_lo = L_mid
    L = L_hi
    x_low = 0.0
    x_high = 1.0
    if np.any(neg_mask):
        bounds = (L - c_aff[neg_mask]) / a[neg_mask] + h0
        x_low = max(x_low, float(np.max(bounds)))
    if np.any(pos_mask):
        bounds = (L - c_aff[pos_mask]) / a[pos_mask] + h0
        x_high = min(x_high, float(np.min(bounds)))
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_low < r1 and x_high > r2:
                if h0 - r1 <= r2 - h0:
                    x_high = min(x_high, r1)
                else:
                    x_low = max(x_low, r2)
            elif x_low >= r1 and x_high <= r2:
                if h0 - r1 <= r2 - h0:
                    x_high = min(x_high, r1)
                else:
                    x_low = max(x_low, r2)
            elif x_low < r1:
                x_high = min(x_high, r1)
            else:
                x_low = max(x_low, r2)
    x_low = max(x_low, 0.0)
    x_high = min(x_high, 1.0)
    if x_low > x_high:
        return h0
    return float(np.clip((x_low + x_high) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            if h[p] < 1e-13 or h[p] > 1.0 - 1e-13:
                continue
            x_new = solve_1d(c, h, p, n)
            if abs(x_new - h[p]) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
    return h, best_c

def subgradient_perturb(h, n, magnitude, rng):
    """Perturb h in the direction that reduces the max correlate."""
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    active = np.where(c > cm * 0.99)[0]
    g = np.zeros(n)
    for m in active:
        shift = m - (n - 1)
        for i in range(n):
            j1 = i + shift
            j2 = i - shift
            if 0 <= j1 < n:
                g[i] += (1.0 - h[j1])
            if 0 <= j2 < n:
                g[i] += h[j2]
    g = g / (np.linalg.norm(g) + 1e-15)
    h_pert = h - magnitude * g
    return project_feasible(h_pert, n)

def random_perturb(h, n, sigma, rng):
    h_pert = h + rng.normal(0, sigma, n)
    return project_feasible(h_pert, n)

def slsqp_refine(h, n, deadline):
    """Quick SLSQP refinement using soft-max."""
    from scipy.optimize import minimize
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        alpha = 5000
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        lse = mx + np.log(np.sum(np.exp(s - mx)))
        return (m + lse / alpha) * dx
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        alpha = 5000
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        weights = np.exp(s - mx)
        weights /= weights.sum()
        g = np.zeros(n)
        for mi in active_indices(weights, c, n, top_k=20):
            w = weights[mi]
            shift = mi - (n - 1)
            for i in range(n):
                j1 = i + shift
                j2 = i - shift
                if 0 <= j1 < n:
                    g[i] += w * (1.0 - x[j1])
                if 0 <= j2 < n:
                    g[i] -= w * x[j2]
        return g * dx
    def active_indices(weights, c, n, top_k=20):
        idx = np.argsort(weights)[-top_k:]
        return idx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < compute_c5(h, n):
            return h_new
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(700, 0.5)
        n_init = 700
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: Exact CD from parent
    remaining = deadline - time.time()
    max_passes = max(10, int(remaining * 0.3 / 1.5))
    h1, c1 = exact_cd(best_h, n_init, max_passes, t0 + (deadline - t0) * 0.3)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: SLSQP refinement
    if time.time() < deadline - 200:
        h2 = slsqp_refine(best_h, n_init, deadline)
        c2 = compute_c5(h2, n_init)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h3, c3 = exact_cd(h2, n_init, 30, deadline)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 3: Multi-restart with perturbation + CD
    n = len(best_h)
    strategies = []
    for i in range(30):
        if i % 3 == 0:
            sigma = 0.01 + 0.03 * (i / 30.0)
            strategies.append(('rand', sigma))
        elif i % 3 == 1:
            mag = 0.02 + 0.05 * (i / 30.0)
            strategies.append(('subgrad', mag))
        else:
            sigma = 0.005 + 0.02 * (i / 30.0)
            strategies.append(('rand', sigma))
    for si, (stype, param) in enumerate(strategies):
        if time.time() > deadline - 60:
            break
        pass_budget = deadline - time.time() - 30
        if pass_budget < 10:
            break
        if stype == 'rand':
            h_p = random_perturb(best_h, n, param, rng)
        else:
            h_p = subgradient_perturb(best_h, n, param, rng)
        max_p = max(5, int(pass_budget / 2.5))
        h_r, c_r = exact_cd(h_p, n, max_p, deadline)
        if c_r < best_c:
            best_h, best_c = h_r, c_r
    # Phase 4: Try different n values
    for n_try in [500, 800, 1000, 900]:
        if time.time() > deadline - 80:
            break
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        pass_budget = deadline - time.time() - 40
        if pass_budget < 15:
            break
        max_p = max(5, int(pass_budget / 3.0))
        h_t, c_t = exact_cd(h_int, n_try, max_p, deadline)
        if c_t < best_c:
            best_h, best_c = h_t, c_t
    # Phase 5: Final polish with many CD passes
    if time.time() < deadline - 15:
        n = len(best_h)
        max_p = max(20, int((deadline - time.time() - 10) / 1.5))
        hf, cf = exact_cd(best_h, n, max_p, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
