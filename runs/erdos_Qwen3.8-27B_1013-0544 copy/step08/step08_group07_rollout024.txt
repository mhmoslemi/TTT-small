def solve_qp(h_t, n, alpha, c_vals, grads):
    h = cp.Variable(n)

    prob = cp.Problem(obj, cons)
    prob.solve()
    return h.value

        grad_penalty_t = -2000 * np.sum(active)
        return np.concatenate([grad_prox + grad_penalty_u, grad_penalty_t])
    
    x0 = np.zeros(n + 1)
    res = minimize(objective, x0, jac=objective_grad, method='L-BFGS-B',
                   options={'maxiter': 200, 'ftol': 1e-15})
    if not res.success:
        return project_feasible(h, n)
    u = res.x[:n]
    h_new = project_feasible(h + u, n)
    return h_new

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grads(h, n):
    n_shifts = 2 * n - 1
    grads = np.zeros((n_shifts, n))
    for idx in range(n_shifts):
        s = idx - (n - 1)
        if s >= 0:
            m = n - s
            grads[idx, :m] += (1.0 - h[s:])
            grads[idx, s:] -= h[:m]
        else:
            ss = -s
            m = n - ss
            grads[idx, ss:] += (1.0 - h[:m])
            grads[idx, :m] -= h[ss:]
    return grads

def lp_prox_step(h, n, deadline):
    c_vals = np.correlate(h, 1.0 - h, mode='full')
    grads = compute_grads(h, n)
    n_shifts = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([grads, -np.ones((n_shifts, 1))])
    b_ub = -c_vals.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lb = -h.copy()
    ub = 1.0 - h.copy()
    bounds = list(zip(lb, ub)) + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return h.copy()
    if not res.success:
        return h.copy()
    u = res.x[:n]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for frac in [1.0, 0.6, 0.4, 0.25, 0.1]:
        if time.time() > deadline: break
        h_try = project_feasible(h + frac * u, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_h = h_try
            best_c = c_try
    return best_h

def active_subgrad_step(h, n, step_size, deadline):
    c_vals = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(c_vals.max())
    # Active set: constraints within 5% of max
    tol = 0.05 * cmax
    active = c_vals > (cmax - tol)
    if not np.any(active):
        return h.copy()
    grads = compute_grads(h, n)
    g = grads[active].mean(axis=0)
    # Projected gradient descent
    h_new = h - step_size * g
    h_new = project_feasible(h_new, n)
    return h_new

def coord_cd(h, n, max_passes, deadline, grid_size=60):
    h = h.copy()
    best_c = compute_c5(h, n)
    for pidx in range(max_passes):
        if time.time() > deadline: break
        c_prev = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c_prev.max())
        improved = False
        for p in range(n):
            if time.time() > deadline: break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            # Compute affected correlation values
            m_lo = max(0, p - (n - 1))
            m_hi = min(2 * n - 2, p + (n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb): a[vb] -= h[jb[vb]]
            c_aff = c_prev[ms]
            unaff_max = 0.0
            if m_lo > 0: unaff_max = max(unaff_max, float(c_prev[:m_lo].max()))
            if m_hi < 2 * n - 2: unaff_max = max(unaff_max, float(c_prev[m_hi+1:].max()))
            diag_mask = (ms == (n - 1))
            best_d = 0.0
            best_val = current_max
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            for d in d_grid:
                if abs(d) < 1e-14: continue
                nv = c_aff + d * a
                if np.any(diag_mask): nv = nv.copy(); nv[diag_mask] -= d * d
                val = max(unaff_max, float(nv.max()))

                    best_d = d

                if np.any(diag_mask): c_prev[ms[diag_mask]] -= best_d * best_d
        if not improved: break
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c: best_c = new_c
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alphas, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 20: break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

- From the "forward" part: (1-h[p+0]) = 1-h[p] (since jf = p + (n-1) - (n-1) = p, so we add 1-h[p])
- From the "backward" part: -h[p-0] = -h[p] (since jb = p - (n-1) + (n-1) = p, so we subtract h[p])

So a at the diagonal position = (1-h[p]) - h[p] = 1 - 2h[p].

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grads(h, n):
    n_shifts = 2 * n - 1
    grads = np.zeros((n_shifts, n))
    for idx in range(n_shifts):
        s = idx - (n - 1)
        if s >= 0:
            m = n - s
            grads[idx, :m] += (1.0 - h[s:])
            grads[idx, s:] -= h[:m]
        else:
            ss = -s
            m = n - ss
            grads[idx, ss:] += (1.0 - h[:m])
            grads[idx, :m] -= h[ss:]
    return grads

def lp_prox_step(h, n, deadline):
    c_vals = np.correlate(h, 1.0 - h, mode='full')
    grads = compute_grads(h, n)
    n_shifts = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([grads, -np.ones((n_shifts, 1))])
    b_ub = -c_vals.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = list(zip(-h, 1.0 - h)) + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return h.copy()
    if not res.success:
        return h.copy()
    u = res.x[:n]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for frac in [1.0, 0.6, 0.4, 0.25, 0.15]:
        if time.time() > deadline: break
        h_try = project_feasible(h + frac * u, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_h = h_try
            best_c = c_try
    return best_h

def coord_cd(h, n, max_passes, deadline, grid_size=60):
    h = h.copy()
    best_c = compute_c5(h, n)
    for pidx in range(max_passes):
        if time.time() > deadline: break
        c_prev = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c_prev.max())
        improved = False
        for p in range(n):
            if time.time() > deadline: break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_lo = max(0, p - (n - 1))
            m_hi = min(2 * n - 2, p + (n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb): a[vb] -= h[jb[vb]]
            c_aff = c_prev[ms]
            unaff_max = 0.0
            if m_lo > 0: unaff_max = max(unaff_max, float(c_prev[:m_lo].max()))
            if m_hi < 2 * n - 2: unaff_max = max(unaff_max, float(c_prev[m_hi+1:].max()))
            diag_mask = (ms == (n - 1))
            best_d = 0.0
            best_val = current_max
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            for d in d_grid:
                if abs(d) < 1e-14: continue
                nv = c_aff + d * a
                if np.any(diag_mask): nv[diag_mask] -= d * d
                val = max(unaff_max, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-14 and best_val < current_max:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                c_prev[ms] = c_aff + best_d * a
                if np.any(diag_mask): c_prev[ms[diag_mask]] -= best_d * best_d
        if not improved: break
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c: best_c = new_c
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alphas, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 20: break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgrad_step(h, n, step_size, tol_frac=0.03):
    c_vals = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(c_vals.max())
    active = c_vals > (cmax - tol_frac * cmax)
    if not np.any(active): return h.copy()
    grads = compute_grads(h, n)
    g = grads[active].mean(axis=0)
    h_new = project_feasible(h - step_size * g, n)
    return h_new
