import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if diff > 0.0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0.0:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 0.0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = np.clip(h[idx[0]] + res, 0.0, 1.0)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def subgrad_step(h, n, k_star, alpha):
    """Take a subgradient step for the k_star-th correlation term."""
    d = k_star - (n - 1)
    # grad[j] = (1-h[j+d]) if j+d valid, minus h[j-d] if j-d valid
    grad = np.zeros(n)
    # First part: (1-h[j+d]) for valid j+d
    j_range = slice(0, n)
    jd = np.arange(n) + d
    valid_jd = (jd >= 0) & (jd < n)
    grad[valid_jd] += (1.0 - h[jd[valid_jd]])
    # Second part: -h[j-d] for valid j-d
    jd2 = np.arange(n) - d
    valid_jd2 = (jd2 >= 0) & (jd2 < n)
    grad[valid_jd2] -= h[jd2[valid_jd2]]
    return grad

def softmax_active(c, n, top_m=5):
    """Get weights for top-m active terms using softmax."""
    center = n - 1
    # Focus on shifts near the center (small lags)
    L = len(c)
    # Get top m values
    idx = np.argsort(c)[::-1][:top_m]
    vals = c[idx]
    alpha = 50.0
    z = alpha * (vals - vals[0])
    w = np.exp(z)
    w /= w.sum()
    return idx, w

def subgradient_descent(h, n, max_iters, deadline, initial_alpha=0.01):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    L = 2 * n - 1
    scale = 2.0 / n
    alpha = initial_alpha
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Find active terms
        idx, w = softmax_active(c, n, top_m=5)
        # Compute weighted subgradient
        grad = np.zeros(n)
        for m in range(len(idx)):
            k_star = idx[m]
            g = subgrad_step(h, n, k_star, w[m])
            grad += w[m] * g
        
        # Step
        h_new = h - alpha * grad * scale  # scale to account for the dx factor
        h_new = project_feasible(h_new, n)
        
        # Evaluate
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c_val = float(c_new.max()) * scale
        
        if c_val < best_c:
            best_c = c_val
            best_h = h_new.copy()
        
        h = h_new
        c = c_new
        
        # Update step size (diminishing)
        alpha = initial_alpha / (1.0 + 0.01 * it)
    
    return best_h, best_c

            n_off = np.where(h == 0)[0]

                best_c = cur * (2.0/n)
                best_h = h.copy()
        else:

        T = max(T_end, T * 0.9999)

import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0:
            h += (diff / tot) * cap
    elif diff < 0:
        tot = float(h.sum())
        if tot > 0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        idx = np.where(h < 1.0)[0] if res > 0 else np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = np.clip(h[idx[0]] + res, 0.0, 1.0)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * (2.0 / n))

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def triangle_h(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(1.0 - np.abs(x - 1.0), n)

def binary_from_h(h):
    n = len(h)
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    h2 = np.zeros(n)
    h2[idx] = 1.0
    return h2

def subgradient_descent(h, n, max_iters, deadline, alpha0=0.05):
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c = compute_c(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    scale = 2.0 / n
    center = n - 1
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Find top-3 active terms
        top_idx = np.argsort(c)[::-1][:3]
        top_vals = c[top_idx]
        w = np.exp(20.0 * (top_vals - top_vals[0]))
        w /= w.sum()
        
        grad = np.zeros(n)
        for m in range(3):
            k_star = top_idx[m]
            d = k_star - center
            jd = np.arange(n) + d
            v1 = (jd >= 0) & (jd < n)
            grad[v1] += w[m] * (1.0 - h[jd[v1]])
            jd2 = np.arange(n) - d
            v2 = (jd2 >= 0) & (jd2 < n)
            grad[v2] -= w[m] * h[jd2[v2]]
        
        alpha = alpha0 / (1.0 + 0.05 * it)
        h_new = project_feasible(h - alpha * grad, n)
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c_val = float(c_new.max()) * scale
        
        if c_val < best_c:
            best_c = c_val
            best_h = h_new.copy()
        h = h_new
        c = c_new
    
    return best_h, best_c

def coord_descent(h, n, max_passes, deadline, project_every=10):
    """Simplified coordinate descent for continuous h."""
    h = project_feasible(h, n).copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h = h.copy()
    best_c = cur * (2.0 / n)
    center = n - 1
    L = 2 * n - 1
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            hp = float(h[p])
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            seg = hi - lo + 1
            s = center - np.arange(lo, hi + 1, dtype=int)
            j = p + s
            i = p - s
            a = np.zeros(seg)
            vj = (j >= 0) & (j < n)
            vi = (i >= 0) & (i < n)
            if np.any(vj):
                a[vj] += 1.0 - h[j[vj]]
            if np.any(vi):
                a[vi] -= h[i[vi]]
            
            off = center - lo
            # Build the affected c values excluding center
            cseg_parts = []
            a_parts = []
            if off > 0:
                cseg_parts.append(c[lo:center])
                a_parts.append(a[:off])
            if hi > center:
                cseg_parts.append(c[center+1:hi+1])
                a_parts.append(a[off+1:])
            
            cseg = np.concatenate(cseg_parts) if cseg_parts else np.array([])
            a_seg = np.concatenate(a_parts) if a_parts else np.array([])
            
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(c[hi+1:].max()))
            
            # Find optimal x for this coordinate
            # f(x) = max(qconst + x(1-x), max(cseg + a_seg*(x-hp)), unaff)
            # Try golden section + check critical points
            best_x = hp
            best_f = max(qconst + hp*(1-hp), float(cseg.max() + a_seg*(hp-hp)) if a_seg.size else 0, unaff)
            
            # Sample candidates
            xg = np.linspace(0, 1, 16)
            for x in xg:
                if a_seg.size:
                    vmax = float(np.max(cseg + a_seg * (x - hp)))
                else:
                    vmax = 0.0
                f = max(unaff, qconst + x*(1-x), vmax)
                if f < best_f:
                    best_f = f
                    best_x = x
            
            if best_f < cur - 1e-12:
                dx = best_x - hp
                if off > 0:
                    c[lo:center] += a[:off] * dx
                if hi > center:
                    c[center+1:hi+1] += a[off+1:] * dx
                c[center] = qconst + best_x * (1-best_x)
                h[p] = best_x
                cur = best_f
                improved = True
            
            if (p+1) % project_every == 0:
                h = project_feasible(h, n)
                c = np.correlate(h, 1.0 - h, mode='full')
                cur = float(c.max())
        
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        cval = cur * (2.0/n)
        if cval < best_c - 1e-13:
            best_c = cval
            best_h = h.copy()
        if not improved:
            break
    
    return best_h, best_c

def binary_search(h0, n, max_iters, deadline, rng):
    """Local search on binary solutions with smart moves."""
    k = n // 2
    idx = np.argpartition(h0, n-k)[n-k:]
    h = np.zeros(n)
    h[idx] = 1.0
    
    c = np.correlate(h, 1-h, mode='full')
    cur_max = float(c.max())
    best_h = h.copy()
    best_c = cur_max * (2.0/n)
    scale = 2.0/n
    center = n - 1
    
    T = 0.5 * cur_max
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Choose move type
        r = rng.random()
        if r < 0.4:
            # Swap: pick random on/off pair
            on = np.where(h == 1)[0]
            off = np.where(h == 0)[0]
            i = on[rng.randint(len(on))]
            j = off[rng.randint(len(off))]
        elif r < 0.7:
            # Targeted: pick pair at distance = worst shift
            k_star = int(np.argmax(c))
            d = abs(k_star - center)
            if d == 0 or d >= n:
                d = 1
            candidates = np.arange(d, n - d)
            if len(candidates) == 0:
                continue
            i = candidates[rng.randint(len(candidates))]
            j = i + d
            if h[i] == h[j]:
                continue  # need different values for a swap
            # Make sure one is on, other is off
            if h[i] == 0 and h[j] == 1:
                i, j = j, i
            # Now h[i]=1, h[j]=0
        else:
            # Swap-2: two pairs
            on = np.where(h == 1)[0]
            off = np.where(h == 0)[0]
            if len(on) < 2 or len(off) < 2:
                continue
            i1, i2 = on[rng.randint(len(on), 2)]
            j1, j2 = off[rng.randint(len(off), 2)]
            # Apply double swap
            h[i1], h[i2], h[j1], h[j2] = 0, 0, 1, 1
            c_new = np.correlate(h, 1-h, mode='full')
            new_max = float(c_new.max())
            delta = new_max - cur_max
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_max = new_max
                c = c_new
                if new_max * scale < best_c:
                    best_c = new_max * scale
                    best_h = h.copy()
            else:
                h[i1], h[i2], h[j1], h[j2] = 1, 1, 0, 0
            T = max(0.001 * cur_max, T * 0.999)
            continue
        
        # Apply single swap
        h[i], h[j] = 1 - h[i], 1 - h[j]
        c_new = np.correlate(h, 1-h, mode='full')
        new_max = float(c_new.max())
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_max = new_max
            c = c_new
            if new_max * scale < best_c:
                best_c = new_max * scale
                best_h = h.copy()
        else:
            h[i], h[j] = 1 - h[i], 1 - h[j]
        
        # Cool
        T = max(0.001 * cur_max, T * 0.9995)
    
    return best_h, best_c

import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if diff > 0.0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0.0:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 0.0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = np.clip(h[idx[0]] + res, 0.0, 1.0)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * (2.0 / n))

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def triangle_h(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(1.0 - np.abs(x - 1.0), n)

def binary_from_h(h):
    n = len(h)
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    h2 = np.zeros(n)
    h2[idx] = 1.0
    return h2

def subgradient_descent(h, n, max_iters, deadline, alpha0=0.08):
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c = compute_c(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    scale = 2.0 / n
    center = n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        top_idx = np.argsort(c)[::-1][:4]
        top_vals = c[top_idx]
        w = np.exp(30.0 * (top_vals - top_vals[0]))
        w /= w.sum()
        grad = np.zeros(n)
        for m in range(len(top_idx)):
            d = int(top_idx[m]) - center
            if d == 0:
                continue
            jd = np.arange(n) + d
            v1 = (jd >= 0) & (jd < n)
            if np.any(v1):
                grad[v1] += w[m] * (1.0 - h[jd[v1]])
            jd2 = np.arange(n) - d
            v2 = (jd2 >= 0) & (jd2 < n)
            if np.any(v2):
                grad[v2] -= w[m] * h[jd2[v2]]
        alpha = alpha0 / (1.0 + 0.02 * it)
        h_new = project_feasible(h - alpha * grad, n)
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c_val = float(c_new.max()) * scale
        if c_val < best_c:
            best_c = c_val
            best_h = h_new.copy()
        h = h_new
        c = c_new
    return best_h, best_c

def coord_1d(hp, cseg, a_seg, qconst, unaff):
    if a_seg.size:
        vmax0 = float(np.max(cseg + a_seg * (0.0 - hp)))
        vmax1 = float(np.max(cseg + a_seg * (1.0 - hp)))
    else:
        vmax0 = vmax1 = 0.0
    lo = max(0.0, unaff, qconst)
    hi = min(1.0, 1.0)
    xg = np.linspace(0.0, 1.0, 24)
    best_x = hp
    best_f = 1e100
    for x in xg:
        d = x - hp
        if a_seg.size:
            vm = float(np.max(cseg + a_seg * d))
        else:
            vm = 0.0
        f = max(unaff, qconst + x * (1.0 - x), vm)
        if f < best_f:
            best_f = f
            best_x = x
    lo2 = max(0.0, best_x - 0.08)
    hi2 = min(1.0, best_x + 0.08)
    xg2 = np.linspace(lo2, hi2, 24)
    for x in xg2:
        d = x - hp
        if a_seg.size:
            vm = float(np.max(cseg + a_seg * d))
        else:
            vm = 0.0
        f = max(unaff, qconst + x * (1.0 - x), vm)
        if f < best_f:
            best_f = f
            best_x = x
    return best_x, best_f

def coord_descent(h, n, max_passes, deadline, project_every=8):
    h = project_feasible(h, n).copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h = h.copy()
    best_c = cur * (2.0 / n)
    center = n - 1
    L = 2 * n - 1
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            hp = float(h[p])
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            seg = hi - lo + 1
            s = center - np.arange(lo, hi + 1, dtype=int)
            j_idx = p + s
            i_idx = p - s
            a = np.zeros(seg)
            vj = (j_idx >= 0) & (j_idx < n)
            vi = (i_idx >= 0) & (i_idx < n)
            if np.any(vj):
                a[vj] += 1.0 - h[j_idx[vj]]
            if np.any(vi):
                a[vi] -= h[i_idx[vi]]
            off = center - lo
            parts_c, parts_a = [], []
            if off > 0:
                parts_c.append(c[lo:center])
                parts_a.append(a[:off])
            if hi > center:
                parts_c.append(c[center+1:hi+1])
                parts_a.append(a[off+1:])
            cseg = np.concatenate(parts_c) if parts_c else np.array([0.0])
            a_seg = np.concatenate(parts_a) if parts_a else np.array([0.0])
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(c[hi+1:].max()))
            bx, bf = coord_1d(hp, cseg, a_seg, qconst, unaff)
            if bf < cur - 1e-12:
                dxv = bx - hp
                if off > 0:
                    c[lo:center] += a[:off] * dxv
                if hi > center:
                    c[center+1:hi+1] += a[off+1:] * dxv
                c[center] = qconst + bx * (1.0 - bx)
                h[p] = bx
                cur = bf
                improved = True
            if (p + 1) % project_every == 0:
                h = project_feasible(h, n)
                c = np.correlate(h, 1.0 - h, mode='full')
                cur = float(c.max())
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        cval = cur * (2.0 / n)
        if cval < best_c - 1e-14:
            best_c = cval
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def binary_local_search(h0, n, max_iters, deadline, rng):
    k = n // 2
    idx = np.argpartition(h0, n - k)[n - k:]
    h = np.zeros(n)
    h[idx] = 1.0
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(c.max())
    best_h = h.copy()
    best_c = cur_max * (2.0 / n)
    scale = 2.0 / n
    center = n - 1
    T = 0.3 * cur_max
    T_floor = 0.005 * cur_max
    on_mask = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        r = rng.random()
        if r < 0.35:
            on = np.where(h == 1)[0]
            off = np.where(h == 0)[0]
            i = int(on[rng.randint(len(on))])
            j = int(off[rng.randint(len(off))])
        elif r < 0.65:
            k_star = int(np.argmax(c))
            d = abs(k_star - center)
            if d < 1 or d >= n:
                d = rng.randint(1, n // 2)
            valid = np.arange(0, n - d)
            if len(valid) == 0:
                continue
            i = int(valid[rng.randint(len(valid))])
            j = i + d
            if h[i] == h[j]:
                i2 = int(rng.randint(n))
                if h[i2] != h[i]:
                    j = i2
                else:
                    continue
            if h[i] == 0:
                i, j = j, i
        else:
            on = np.where(h == 1)[0]
            off = np.where(h == 0)[0]
            if len(on) < 2 or len(off) < 2:
                continue
            i1, i2 = int(on[rng.randint(len(on))]), int(on[rng.randint(len(on))])
            j1, j2 = int(off[rng.randint(len(off))]), int(off[rng.randint(len(off))])
            if i1 == i2 or j1 == j2:
                continue
            h[i1], h[i2], h[j1], h[j2] = 0, 0, 1, 1
            c_new = np.correlate(h, 1.0 - h, mode='full')
            nm = float(c_new.max())
            delta = nm - cur_max
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                cur_max = nm
                c = c_new
                if nm * scale < best_c:
                    best_c = nm * scale
                    best_h = h.copy()
            else:
                h[i1], h[i2], h[j1], h[j2] = 1, 1, 0, 0
            T = max(T_floor, T * 0.999)
            continue
        h[i], h[j] = 0.0, 1.0
        c_new = np.correlate(h, 1.0 - h, mode='full')
        nm = float(c_new.max())
        delta = nm - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_max = nm
            c = c_new
            if nm * scale < best_c:
                best_c = nm * scale
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_floor, T * 0.9995)
    return best_h, best_c
