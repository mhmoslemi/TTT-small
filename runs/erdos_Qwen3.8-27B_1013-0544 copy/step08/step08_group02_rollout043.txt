def c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def proj(z, n):
    z = np.asarray(z, dtype=float).ravel()
    target = 0.5 * n
    if z.size == 0:
        z = np.full(n, 0.5)
    elif z.size != n:
        a = np.linspace(0.0, 2.0, z.size, endpoint=False)
        b = np.linspace(0.0, 2.0, n, endpoint=False)
        z = np.interp(b, a, z)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(float(z.sum()) - target) <= 1e-8:
        return np.clip(z, 0.0, 1.0)
    lo, hi = float(z.min()) - 2.0, float(z.max()) + 2.0
    for _ in range(50):
        t = 0.5 * (lo + hi)
        s = float(np.clip(z - t, 0.0, 1.0).sum())
        if s > target:
            lo = t
        else:
            hi = t
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - float(y.sum())
    if abs(r) > 1e-14:
        m = (y < 1.0) if r > 0 else (y > 0.0)
        room = (1.0 - y) if r > 0 else y
        if m.any() and float(room.sum()) > 1e-13:
            y[m] += (r / float(room.sum())) * room[m]
    y = np.clip(y, 0.0, 1.0)
    d = target - float(y.sum())
    if abs(d) > 1e-12:
        i = int(np.argmax(1.0 - y) if d > 0 else np.argmax(y))
        y[i] += d
    return np.clip(y, 0.0, 1.0)

def interp(h, m):
    n = len(h)
    a = np.linspace(0.0, 2.0, n, endpoint=False)
    b = np.linspace(0.0, 2.0, m, endpoint=False)
    return proj(np.interp(b, a, h), m)

def rand_start(n, rg):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, 8):
        a = rg.uniform(-0.4, 0.4)
        p = rg.uniform(0.0, 2.0 * np.pi)
        h += a * np.sin(0.5 * np.pi * m * x + p) + 0.5 * a * np.cos(np.pi * m * x + p)

    h = proj(h, n)
    best, bc = h, c5(h, n)
    for z in (1.0 - h, h[::-1], 1.0 - h[::-1], 0.5 * (h + h[::-1]), 0.5 * (h + 1.0 - h[::-1])):

        v = c5(z, n)
        if v < bc:

def sgrad_p(x, n, p):
    g = np.zeros(n)
    off = n - 1
    for k in range(2 * n - 1):
        w = float(p[k])
        if w > 1e-13:
            d = k - off
            if d >= 0:

                m = n - e
                g[:m] += w * (1.0 - x[e:])
                g[e:] -= w * x[:m]

    e = np.exp(z)
    s = float(e.sum())
    return e / s, m, np.log(s)

def sval(x, n, al):
    _, m, ls = sinfo(x, n, al)
    return (m + ls / al) * (2.0 / n)

def sgrad(x, n, al):
    p, _, _ = sinfo(x, n, al)
    return sgrad_p(x, n, p)

def pval(x, n, al, mu):
    t = 0.5 * n
    d = float(x.sum()) - t
    return sval(x, n, al) + 0.5 * mu * d * d / (t * t)

def pgrad(x, n, al, mu):
    t = 0.5 * n
    d = float(x.sum()) - t
    return sgrad(x, n, al) + (mu * d / (t * t)) * np.ones_like(x)

def lbfgs(h, n, bd, plans, mu, dl):
    b = proj(h, n)
    bc = c5(b, n)
    w = b.copy()
    for al, mx in plans:
        if time.time() > dl - 2.0:
            break
        try:
            r = minimize(pval, w, args=(n, al, mu), method="L-BFGS-B", jac=pgrad, bounds=bd,
                         options={"maxiter": int(mx), "maxfun": int(mx * 4), "ftol": 1e-14,
                                  "gtol": 1e-7, "maxls": 20, "disp": False})
        except Exception:
            continue
        x = proj(np.asarray(r.x, dtype=float), n)
        v = c5(x, n)
        if v < bc:
            b, bc, w = x, v, x
        elif v <= bc + 1e-8:
            w = x
    return b, bc

def buildG(h, n, idx, off):
    d = (idx - off).astype(int)
    m = d.size
    G = np.zeros((n, m))
    for c in range(m):
        dd = int(d[c])
        if dd >= 0:
            m2 = n - dd
            if m2 > 0:
                G[:m2, c] += 1.0 - h[dd:]
                G[dd:, c] -= h[:m2]
        else:
            e = -dd
            m2 = n - e
            if m2 > 0:
                G[e:, c] += 1.0 - h[:m2]
                G[:m2, c] -= h[e:]
    return G

def rebuild(h, c, n, K):
    L = 2 * n - 1
    off = n - 1
    cm = float(c.max())
    near = np.flatnonzero(c >= cm - 1e-7)
    cap = min(L, max(K, 400))
    if near.size > cap:
        idx = np.argpartition(c, -cap)[-cap:]
    else:
        idx = near
    idx = np.unique(idx)
    idx = idx[np.argsort(c[idx])[::-1]]
    pos = np.full(L, -1, dtype=int)
    pos[idx] = np.arange(idx.size)
    return idx, pos, idx - off, buildG(h, n, idx, off)

def updG(G, r, delta, ad, n):
    if delta == 0.0:
        return
    rows = r - ad
    m = (rows >= 0) & (rows < n)
    if m.any():
        i = np.nonzero(m)[0]
        G[rows[i], i] -= delta
    rows = r + ad
    m = (rows >= 0) & (rows < n)
    if m.any():
        i = np.nonzero(m)[0]
        G[rows[i], i] -= delta

def actvals(cs, B, us, pos, off, i, j):
    v = cs[None, :] + us[:, None] * B[None, :]
    q = us * us
    p = int(pos[off])
    if p >= 0:
        v[:, p] -= 2.0 * q
    p = int(pos[off + i - j])
    if p >= 0:
        v[:, p] += q
    p = int(pos[off + j - i])
    if p >= 0:
        v[:, p] += q
    return v.max(axis=1)

def line(cs, G, i, j, pos, off, lo, hi):
    if hi - lo < 1e-11:
        return 0.0, 1e100
    B = G[i] - G[j]
    us = lo + (hi - lo) * GA
    v = actvals(cs, B, us, pos, off, i, j)
    bi = int(np.argmin(v))
    bu = float(us[bi])
    bv = float(v[bi])
    wd = (hi - lo) / 32.0
    for _ in range(2):
        uu = np.clip(bu + wd * 0.7 * RA, lo, hi)
        if float(uu.max()) - float(uu.min()) < 1e-12:
            break
        v2 = actvals(cs, B, uu, pos, off, i, j)
        b2 = int(np.argmin(v2))
        bu2 = float(uu[b2])
        if abs(bu2 - bu) < 1e-10:
            bu = bu2
            bv = float(v2[b2])
            break
        bu = bu2
        bv = float(v2[b2])
        wd *= 0.5
    return bu, bv

def ver(c, h, n, i, j, u, off):
    B = np.zeros(2 * n - 1)
    om = 1.0 - h
    rv = h[::-1]
    B[off - i:2 * n - 1 - i] += om
    B[i:i + n] -= rv
    B[off - j:2 * n - 1 - j] -= om
    B[j:j + n] += rv
    cn = c + u * B
    q = u * u
    cn[off] -= 2.0 * q
    cn[off + i - j] += q
    cn[off + j - i] += q
    return cn

def add(i, j, n, out, seen):
    if i > j:
        i, j = j, i
    if i == j:
        return
    k = i * n + j
    if k not in seen:
        seen.add(k)
        out.append((i, j))

def select(h, G, ad, n, m, R, rg):
    out = []
    seen = set()
    for p in range(m):
        d = int(ad[p])
        if d == 0:
            add(int(np.argmax(h)), int(np.argmin(h)), n, out, seen)
            continue
        e = abs(d)
        cnt = n - e
        if cnt < 2:
            continue
        B = G[:cnt, p] - G[e:, p]
        if cnt <= R:
            sel = np.arange(cnt)
        else:
            sel = np.argpartition(-np.abs(B), R)[:R]
            sel = sel[np.argsort(-np.abs(B[sel]))]
        for bb in sel:
            add(int(bb), int(bb) + e, n, out, seen)
        rb = int(rg.randint(cnt))
        add(rb, rb + e, n, out, seen)
    for _ in range(int(min(300, max(40, m)))):
        add(int(rg.randint(n)), int(rg.randint(n)), n, out, seen)
    return out

def fullpass(h, c, cs, G, idx, pos, ad, n, step, dl, ma):
    off = n - 1
    cm = float(c.max())
    acc = 0
    for i in range(n - 1):
        hi = h[i]
        for j in range(i + 1, n):
            if time.time() > dl or acc >= ma:
                return h, c, cm, acc
            hj = h[j]
            lo = -hi + 1e-12
            up = min(1.0 - hi, hj) - 1e-12
            lo = max(lo, -step)
            up = min(up, step)
            if up - lo < 1e-11:
                continue
            u, av = line(cs, G, i, j, pos, off, lo, up)
            if u == 0.0 or av >= cm - 1e-7:
                continue
            cn = ver(c, h, n, i, j, u, off)
            nm = float(cn.max())
            if nm < cm - 1e-8:
                h[i] += u
                h[j] -= u
                updG(G, i, u, ad, n)
                updG(G, j, -u, ad, n)
                c = cn
                cs = c[idx]
                cm = nm
                acc += 1
                if pos[int(np.argmax(c))] < 0:
                    return h, c, cm, acc
    return h, c, cm, acc

def local(h, n, dl, K=200, rg=None):
    if rg is None:
        rg = np.random.RandomState(999)
    h = proj(h, n)
    c = np.correlate(h, 1.0 - h, mode="full")
    off = n - 1
    stalls = 0
    fs = 0
    while time.time() < dl:
        c = np.correlate(h, 1.0 - h, mode="full")
        cm = float(c.max())
        idx, pos, ad, G = rebuild(h, c, n, K)
        m = idx.size
        cs = c[idx]
        step = 0.12 if stalls == 0 else (0.30 if stalls == 1 else 0.05)
        if m <= 60 and fs < 8 and n <= 600:
            h, c, cm, acc = fullpass(h, c, cs, G, idx, pos, ad, n, step, dl, 600)
            if acc:
                fs += 1
                stalls = 0
            else:
                stalls += 1
                if stalls > 2:
                    break
            continue
        R = max(2, min(12, 2500 // max(1, m)))
        cands = select(h, G, ad, n, m, R, rg)
        pa = 0
        for i, j in cands:
            if time.time() > dl or pa >= 150:
                break
            hi = h[i]
            hj = h[j]
            lo = -hi + 1e-12
            up = min(1.0 - hi, hj) - 1e-12
            lo = max(lo, -step)
            up = min(up, step)
            if up - lo < 1e-11:
                continue
            u, av = line(cs, G, i, j, pos, off, lo, up)
            if u == 0.0 or av >= cm - 1e-7:
                continue
            cn = ver(c, h, n, i, j, u, off)
            nm = float(cn.max())
            if nm < cm - 1e-8:
                h[i] += u
                h[j] -= u
                updG(G, i, u, ad, n)
                updG(G, j, -u, ad, n)
                c = cn
                cs = c[idx]
                cm = nm
                pa += 1
                if pos[int(np.argmax(c))] < 0:
                    break
        if pa == 0:
            if stalls == 2 and n <= 550 and m <= 200 and time.time() < dl - 30.0:
                h, c, cm, acc = fullpass(h, c, cs, G, idx, pos, ad, n, 0.08, dl, 150)
                fs += 1
                if acc:
                    stalls = 0
                    continue
            stalls += 1
            if stalls > 2:
                break
        else:
            stalls = 0
    c = np.correlate(h, 1.0 - h, mode="full")
    return h, float(c.max()) * 2.0 / n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = float(budget_s)
    dl = t0 + B - 6.0
    rg = np.random.RandomState(seed)
    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel()
    except Exception:
        h0 = None
    if h0 is None or h0.size < 2:
        h = rand_start(500, rg)
    else:
        if h0.size < 200 or h0.size > 1000 or h0.size % 2:
            mm = 500 if (h0.size < 200 or h0.size > 1000) else h0.size + 1
            h = interp(h0, mm)
        else:
            h = h0.copy()
    h, cb = sym(h, len(h))
    n = len(h)
    bd = [(0.0, 1.0)] * n
    lbdl = min(dl - 100.0, time.time() + 80.0)
    if dl - time.time() > 90.0 and time.time() < lbdl:
        hb, vb = lbfgs(h, n, bd, [(30, 40), (150, 30), (800, 20)], 1e5, lbdl)
        if vb < cb:
            h, cb = hb, vb
    md = dl - max(120.0, 0.18 * B)
    if md < time.time() + 20.0:
        md = dl - 20.0
    h, cb = local(h, n, md, 200, rg)
    if n < 640 and time.time() < dl - 160.0:
        res = min(70.0, 0.35 * (dl - time.time()))
        if res > 30.0:
            ht = interp(h, 640)
            bm = [(0.0, 1.0)] * 640
            h2, v2 = lbfgs(ht, 640, bm, [(150, 25), (600, 20)], 1e5, time.time() + 0.4 * res)
            h3, v3 = local(h2, 640, time.time() + 0.55 * res, 180, rg)
            if v3 < cb:
                h, cb, n, bd = h3, v3, 640, bm
    if time.time() < dl - 70.0:
        hp = proj(h + rg.normal(0.0, 0.04, n), n)
        hb, vb = lbfgs(hp, n, bd, [(100, 20), (400, 15)], 1e5, time.time() + 15.0)
        hc, vc = local(hb, n, time.time() + 25.0, 160, rg)
        if vc < cb:
            h, cb = hc, vc
    if time.time() < dl - 10.0:
        h, cb = local(h, n, dl - 3.0, 160, rg)
    h = proj(h, len(h))
    cb = c5(h, len(h))
    return h, cb, len(h)
