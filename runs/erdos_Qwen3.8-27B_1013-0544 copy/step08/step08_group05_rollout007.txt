def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)

    w = ee / z

    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx

    """Compute new max after flipping h[i]:1->0, h[j]:0->1 without full copy."""
    L = 2 * n - 1
    diag = n - 1
    st_i = diag - i
    st_j = diag - j

    changed = np.zeros(L, dtype=bool)

                    bx = float(xx[j])
                    bv = float(mm[j].max())
                if lo < bx < hi:
                    sp2 = (hi - lo) / float(refine)
                    lo2 = max(0.0, bx - sp2)
                    hi2 = min(1.0, bx + sp2)
                    yy = np.linspace(lo2, hi2, 9)
                    ww = c[None, :] + (yy - old)[:, None] * A[None, :]
                    ww[:, n - 1] -= (yy - old) ** 2
                    k = int(np.argmin(ww.max(axis=1)))
                    if ww[k].max() < bv - tol:
                        bx = float(yy[k])
                        bv = float(ww[k].max())
                if bv < cur - tol:
                    d = bx - old
                    c[p:p + n] += d * om
                    c[st:st + n] -= d * x
                    c[diag] -= d * d
                    x[p] = bx
                    om[diag - p] = 1.0 - bx
                    improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - tol:
            best = xp
            bc = cp
        if not improved:
            break
    return best, bc

def binarize(h, n):
    if n % 2:
        return None
    target = n // 2
    if target <= 0 or target >= n:
        return None
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def binary_sa(h_init, n, rng, deadline, T0=0.5, T1=0.005, max_rounds=60, tpr=20000):
    if n % 2:
        return None, np.inf
    dx = 2.0 / n
    h = binarize(h_init, n)
    if h is None:
        return None, np.inf
    c = np.correlate(h, 1.0 - h, mode="full")
    cm = float(c.max())
    best = h.copy()
    best_cm = cm
    diag = n - 1
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (rnd / max(1, max_rounds))
        for t in range(tpr):
            if time.time() > deadline:
                break
            for _ in range(8):
                i = int(rng.randint(n))
                if h[i] > 0.5:
                    break
            else:
                continue
            for _ in range(8):
                j = int(rng.randint(n))
                if h[j] < 0.5:
                    break
            else:
                continue
            if i == j:
                continue
            om = 1.0 - h[::-1]
            st_i = diag - i
            st_j = diag - j
            d = c.copy()
            d[i:i + n] -= om
            d[st_i:st_i + n] += h
            d[j:j + n] += om
            d[st_j:st_j + n] -= h
            d[diag] -= 2.0
            d[diag + i - j] += 1.0
            d[diag + j - i] += 1.0
            nm = float(d.max())
            delta = nm - cm
            if delta < -1e-12 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
                c = d
                cm = nm
                h[i] = 0.0
                h[j] = 1.0
                if cm < best_cm:
                    best = h.copy()
                    best_cm = cm
    return best, float(best_cm) * dx

def iterated_binary(h_init, n, rng, deadline, n_restarts=5):
    best_h, best_c = binary_sa(h_init, n, rng, min(deadline, time.time() + 60),
                               T0=1.0, T1=0.01, max_rounds=30, tpr=25000)
    if best_h is None:
        return None, np.inf
    perm = np.zeros(n, dtype=float)
    perm[rng.permutation(n)] = 1.0
    starts = [best_h]
    for r in range(n_restarts - 1):
        pert = best_h.copy()
        nflip = rng.randint(4, 12)
        ones_idx = np.where(pert > 0.5)[0]
        zeros_idx = np.where(pert < 0.5)[0]
        for _ in range(nflip):
            a = ones_idx[int(rng.randint(len(ones_idx)))]
            b = zeros_idx[int(rng.randint(len(zeros_idx)))]
            pert[a] = 0.0
            pert[b] = 1.0
        starts.append(pert)
    for s in starts:
        if time.time() > deadline:
            break
        chunk = min(deadline, time.time() + max(20, (deadline - time.time()) // 3))
        bh, bc = binary_sa(s, n, rng, chunk, T0=0.3, T1=0.003, max_rounds=25, tpr=20000)
        if bh is not None and bc < best_c:
            best_h, best_c = bh, bc
    return best_h, best_c

def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h, period=n if False else None), n_tgt)

def transfer_h2(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    y = np.linspace(0.0, 2.0, n_tgt, endpoint=False) + 1.0 / n_tgt
    hi = np.interp(y, x, h)
    return project_simplex(hi, n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(5.0, budget_s - 5.0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    h0 = project_simplex(h0, n0)
    best_h = h0.copy()
    best_c = float(compute_c5(h0, n0))
    n_best = int(n0)

    rev = h0[::-1]
    for v in (0.5 * (h0 + rev), 0.5 * (h0 + (1.0 - rev))):
        vv = project_simplex(v, n_best)
        cv = compute_c5(vv, n_best)
        if cv < best_c:
            best_h, best_c = vv, cv

    if n_best % 2 == 0:
        hb = binarize(h0, n_best)
        if hb is not None:
            cb = compute_c5(hb, n_best)
            if cb < best_c:
                best_h, best_c = hb, cb

    # Phase 1: Intensive binary search on best n
    t1 = min(deadline - 250.0, t0 + 350.0)
    if n_best % 2 == 0 and time.time() < t1:
        hb, cb = iterated_binary(best_h, n_best, rng, t1, n_restarts=4)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb

    # Phase 2: Continuous polish
    t2 = min(deadline - 200.0, time.time() + 150.0)
    if time.time() < t2:
        h1, c1 = slsqp_polish(best_h, n_best, [5000, 50000, 500000], 100, t2)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h1, c1 = coord_cd(best_h, n_best, max_passes=6, deadline=t2, grid=32, refine=12)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 3: Resolution sweep
    n_cands = [300, 200, 400, 600, 500, 700, 800, 350, 450, 550]
    tried = {n_best}
    for nt in n_cands:
        if time.time() > deadline - 120.0:
            break
        if nt in tried or nt % 2 or nt < 100 or nt > 1000:
            continue
        tried.add(nt)
        ht = transfer_h2(best_h, n_best, nt)
        cv = compute_c5(ht, nt)
        chunk = min(deadline - 100.0, time.time() + 80.0)
        if nt % 2 == 0 and time.time() < chunk:
            hb2, cb2 = iterated_binary(ht, nt, rng, chunk, n_restarts=2)
            if hb2 is not None and cb2 < cv:
                ht, cv = hb2, cb2
        if time.time() < chunk:
            h1, c1 = slsqp_polish(ht, nt, [20000], 60, chunk)
            if c1 < cv:
                ht, cv = h1, c1
            h1, c1 = coord_cd(ht, nt, max_passes=3, deadline=chunk, grid=24, refine=8)
            if c1 < cv:
                ht, cv = h1, c1
        if cv < best_c:
            best_h, best_c, n_best = ht, cv, int(nt)

    # Phase 4: Final polish
    tf1 = min(deadline - 30.0, time.time() + 100.0)
    if time.time() < tf1:
        h1, c1 = slsqp_polish(best_h, n_best, [50000, 500000, 5000000], 150, tf1)
        if c1 < best_c:
            best_h, best_c = h1, c1
    tf2 = min(deadline - 10.0, time.time() + 80.0)
    if time.time() < tf2:
        h1, c1 = coord_cd(best_h, n_best, max_passes=4, deadline=tf2, grid=40, refine=16)
        if c1 < best_c:
            best_h, best_c = h1, c1
    if n_best % 2 == 0 and time.time() < deadline - 5.0:
        hb, cb = binary_sa(best_h, n_best, rng, deadline, T0=0.2, T1=0.002, max_rounds=30, tpr=30000)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb

    final_h = project_simplex(best_h, n_best)
    final_c = float(compute_c5(final_h, n_best))
    return final_h, final_c, int(n_best)
