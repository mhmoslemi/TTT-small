Actually, you know what, let me think about this differently. The problem is essentially:

min_h max_k sum_i h[i]*(1-h[i+k]) * dx

This is a min-max problem. The optimal solution likely has the property that the max is achieved at several k values simultaneously (equioscillation).

A different approach: **Use a cutting-plane / Kelley's cutting plane method**.

1. Start with an initial h
2. Compute all c_k values
3. Identify the current max k*
4. Solve a subproblem: minimize c_{k*} subject to constraints
5. This gives a new h
6. Repeat

But the subproblem is non-convex (h[i]*(1-h[j]) is bilinear).

Alternative: **Linearize the non-linear terms**.

At the current h, linearize h[i]*(1-h[j]) around h:
h[i]*(1-h[j]) ≈ h[i]*(1-h[j]) + (h[i]-h[i])*(1-h[j]) + h[i]*(-(h[j]-h[j]))
= h[i]*(1-h[j]) + delta_i*(1-h[j]) - h[i]*delta_j

This is a DCG (directional derivative based) or linearization approach.

Actually, for the max term, we can use a different approach:

**Eposcillation-based approach**:
1. Compute c_k for all k
2. Let k_max be the set of k where c_k is close to the max (within epsilon)
3. Try to equalize c_k for k in k_max
4. This is like a "water filling" approach

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    ncorr = 2 * n - 1
    mid = n - 1
    if order is None:
        order = range(n)
    current_max = float(corr.max())
    improved = False
    
    for p in order:
        h0 = h[p]
        if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
            continue
        
        # Build A and D vectors
        A = np.zeros(ncorr)
        D = np.zeros(ncorr)
        D[mid] = 1.0

        # As h[i] with i=p: need 0<=p+k<n → k in [-p, n-1-p]
        # As h[i+k] with i+k=p: need 0<=p-k<n → k in [p-n+1, p]

                vals[mid] = qval

                q = corr[mid] + u * A[mid] - (t*t - h0*h0)

            c1 = b - gr*(b-a)

                    a, c1, fc1 = c1, d1, fd1

            if f_star < best_f:

        best_f = min(best_f, f0, f1, fh0)

            best_t = 1.0

            h[p] = best_t
            current_max = float(corr.max())
            improved = True
    
    return h, current_max, improved

    # The p-th term changes from h0*(1-h0) to t*(1-t)
    # Delta = t*(1-t) - h0*(1-h0) = (t-h0) - (t^2-h0^2)
    # So: corr[mid](t) = corr[mid] + (t-h0)*1 - (t^2-h0^2)*1
    # Hence A[mid] = 1, D[mid] = 1.
    
    # The naive computation (treating h[p+k] and h[p-k] as fixed) gives:
    # A[mid] = (1-h[p]) - h[p] = 1-2*h[p]  (using the OLD value h[p]=h0)
    # But the TRUE coefficient is 1 (for the linear part after extracting D).
    
    # So we need to correct: A[mid] should be 1, not 1-2*h0.
    A[mid] = 1.0
    
    return A

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def cd_pass(h, n, order, grid_pts=32, gs_iters=25):
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mid = n - 1
    current_max = float(corr.max())
    improved = False
    for p in order:
        h0 = h[p]
        if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
            continue
        A = np.zeros(2 * n - 1)
        ks1 = np.arange(-p, n - p)
        A[mid + ks1] += (1.0 - h[:n - p])
        ks2 = np.arange(p - n + 1, p + 1)
        A[mid + ks2] -= h[p::-1]
        A[mid] = 1.0
        grid = np.linspace(0.0, 1.0, grid_pts)
        us = grid - h0
        mat = corr[None, :] + us[:, None] * A[None, :]
        mat[:, mid] -= grid * grid - h0 * h0
        fvals = mat.max(axis=1)
        best_i = int(np.argmin(fvals))
        lo = grid[max(0, best_i - 1)]
        hi = grid[min(grid_pts - 1, best_i + 1)]
        best_f = fvals[best_i]
        if hi - lo > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = lo, hi
            def feval(t):
                u = t - h0
                v = float(np.max(corr + u * A))
                q = corr[mid] + u - (t * t - h0 * h0)
                return max(v, q)
            c1 = b - gr * (b - a)
            d1 = a + gr * (b - a)
            fc1 = feval(c1)
            fd1 = feval(d1)
            for _ in range(gs_iters):
                if fc1 < fd1:
                    b, d1, fd1 = d1, c1, fc1
                    c1 = b - gr * (b - a); fc1 = feval(c1)
                else:
                    a, c1, fc1 = c1, d1, fd1
                    d1 = a + gr * (b - a); fd1 = feval(d1)
            ts = (a + b) * 0.5
            fs = feval(ts)
            if fs < best_f:
                best_f = fs
                grid[best_i] = ts
        f0 = feval(0.0)
        f1 = feval(1.0)
        best_f = min(best_f, f0, f1)
        if f0 <= best_f + 1e-15:
            best_t = 0.0
        elif f1 <= best_f + 1e-15:
            best_t = 1.0
        else:
            best_t = float(grid[best_i])
        if best_f < current_max - 1e-12:
            delta = best_t - h0
            corr += delta * A
            corr[mid] -= (best_t * best_t - h0 * h0)
            h[p] = best_t
            current_max = float(corr.max())
            improved = True
    return h, current_max, improved

def pnorm_slsqp(h, n, p_list, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in p_list:
        if deadline is not None and time.time() > deadline:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return (np.sum(c ** p) * (2.0 / n)) ** (1.0 / p)
        def cons_fun(x):
            return x.sum() - n / 2.0
        def cons_jac(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
