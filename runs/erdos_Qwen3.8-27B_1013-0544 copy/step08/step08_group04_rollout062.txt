```python
    b_p, b_n = b[pos], b[neg]

        ok = False

                if float(np.max(v)) > ll: ll = float(np.max(v))
            if a_p.size:

                if float(np.min(v)) < lh: lh = float(np.min(v))
            if ll <= 1.0+tol and lh >= -tol:

                elif g >= -1e-14:

                    r1 = 0.5-0.5*np.sqrt(disc)
                    r2 = 0.5+0.5*np.sqrt(disc)

                    else: ok = max(r2,ll) <= min(1.0,lh)+tol

        else: low = mid

            f = max(f, unaff)
            if f < best_f - 1e-14:
```

```python
import numpy as np

from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = cap.sum()
        if tot > 1e-14:
            h += (diff / tot) * cap
    else:
        tot = h.sum()
        if tot > 1e-14:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - h.sum()
    if res > 1e-12:
        idx = np.where(h < 1.0)[0]
        if idx.size: h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-12:
        idx = np.where(h > 0.0)[0]
        if idx.size: h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    z = alpha * (c - c.max())
    return (c.max() + np.log(np.sum(np.exp(z))) / alpha) * (2.0 / n)

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    z = alpha * (c - c.max())
    w = np.exp(z); w /= w.sum()
    center = n - 1
    g = np.zeros(n)
    for m in range(2*n-1):
        wm = w[m]
        if wm < 1e-15: continue
        s = m - center
        if s == 0: g += wm * (1.0 - 2.0*x)
        elif s > 0:
            g[:n-s] += wm*(1.0-x[s:]); g[s:] -= wm*x[:n-s]
        else:
            r = -s; g[r:] += wm*(1.0-x[:n-r]); g[:n-r] -= wm*x[r:]
    return (2.0/n) * g

def _sc(x, n): return float(x.sum()) - 0.5*n

def _sj(x, n): return np.ones(n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h = project_feasible(h, n)
    best_c = compute_c(best_h, n)
    bounds = [(0.0, 1.0)]*n
    cons = [{'type':'eq', 'fun':_sc, 'jac':_sj, 'args':(n,)}]
    for alpha in alphas:
        if time.time() > deadline - 5: break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          jac=smooth_jac, bounds=bounds, constraints=cons,
                          options={'maxiter':maxiter, 'ftol':1e-14})
            x = project_feasible(res.x, n)
            c = compute_c(x, n)
            if c < best_c: best_h, best_c = x, c
        except Exception: pass
    return best_h, best_c
```

Let me also think about the Fourier/Adam approach more carefully.

For the Fourier approach:
- Parameterize h(x) = sigmoid(c + sum_{k=1}^K [a_k cos(k*π*x') + b_k sin(k*π*x')])
- where x' = x - 1 (shifted to center)
- Total params: 2K + 1 (including c)
- After each update, project to satisfy integral constraint (adjust c)
- Use Adam with lr = 0.01, decay over time

The advantage is that this is a low-dimensional smooth optimization, and Adam can navigate it well.

For the gradient, I'll use the analytic form (as derived above). The key computation is:
1. Compute h from params
2. Compute correlation
3. Compute smooth max weights
4. Compute gradient w.r.t. h (the "backprop through correlation")
5. Compute gradient w.r.t. params (chain rule through sigmoid and trig)

Steps 1-3 are straightforward. Step 4 is the O(n^2) part (the loop over active weights). Step 5 is O(n*K).

For n=300, K=10: step 4 is ~300*300 = 90000 (but only over active weights, which might be ~50-100). So ~5000-10000 ops. Very fast.

```python
def fourier_adam(h, n, K, steps, lr, deadline):
    x = (np.arange(n)+0.5)*(2.0/n)
    t = np.pi*(x-1.0)
    # Initialize params from current h via least squares
    # h ~ sigmoid(f) => f ~ log(h/(1-h)) = logit(h)
    h_clipped = np.clip(h, 1e-6, 1-1e-6)
    logit_h = np.log(h_clipped / (1.0-h_clipped))
    # Fit Fourier: logit_h ≈ c + sum a_k cos + b_k sin
    # Build design matrix
    A = np.ones((n, 2*K+1))
    for k in range(1, K+1):
        A[:, 2*k-1] = np.cos(k*t)
        A[:, 2*k] = np.sin(k*t)
    # Least squares
    params, _, _, _ = np.linalg.lstsq(A, logit_h, rcond=None)
    # Adam state
    m = np.zeros(2*K+1)
    v = np.zeros(2*K+1)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_params = params.copy()
    best_h = project_feasible(1.0/(1.0+np.exp(-A @ params)), n)
    best_c = compute_c(best_h, n)
    alpha_fm = 100.0  # smooth max parameter for fourier
    
    for step in range(steps):
        if time.time() > deadline: break
        # Compute h and gradient
        f = A @ params
        h_f = 1.0/(1.0+np.exp(-f))
        sz = h_f*(1.0-h_f)
        corr = np.correlate(h_f, 1.0-h_f, mode='full')
        z = alpha_fm*(corr - corr.max())
        w = np.exp(z); w /= w.sum()
        center = n-1
        gh = np.zeros(n)
        for mm in range(2*n-1):
            wm = w[mm]
            if wm < 1e-13: continue
            s = mm - center
            if s == 0: gh += wm*(1.0-2.0*h_f)
            elif s > 0:
                gh[:n-s] += wm*(1.0-h_f[s:]); gh[s:] -= wm*h_f[:n-s]
            else:
                r = -s; gh[r:] += wm*(1.0-h_f[:n-r]); gh[:n-r] -= wm*h_f[r:]
        gh *= 2.0/n
        # Gradient w.r.t. params
        grad = A.T @ (gh * sz)
        # Adam update
        t_step = step + 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*grad**2
        m_hat = m/(1-beta1**t_step)
        v_hat = v/(1-beta2**t_step)
        params -= lr*m_hat/(np.sqrt(v_hat)+eps)
        # Project: adjust c to satisfy integral
        f2 = A @ params
        h2 = 1.0/(1.0+np.exp(-f2))
        # Adjust params[0] (c) to get sum(h2) = n/2
        # This requires a 1D solve
        for _ in range(30):
            h2 = 1.0/(1.0+np.exp(-(params[0] + A[:,1:] @ params[1:])))
            diff = n/2 - h2.sum()
            if abs(diff) < 1e-10: break
            # Newton step: d(sum)/dc = sum(h(1-h))
            sh = float(np.sum(h2*(1.0-h2)))
            if sh < 1e-10: break
            params[0] += diff/sh
        h_final = project_feasible(1.0/(1.0+np.exp(-A @ params)), n)
        c_val = compute_c(h_final, n)
        if c_val < best_c:
            best_c = c_val
            best_h = h_final.copy()
            best_params = params.copy()
    return best_h, best_c
```
