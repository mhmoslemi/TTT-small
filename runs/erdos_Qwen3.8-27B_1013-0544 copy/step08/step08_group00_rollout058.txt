import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx), corr

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c, _ = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c, _ = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def coordinate_update(h, n, p, corr):
    """Optimal single-coordinate update for index p."""
    h0 = h[p]
    if h0 < 1e-15 or h0 > 1.0 - 1e-15:
        return h0
    
    m_lo = max(0, n - 1 - p)
    m_hi_f = min(2*n - 2, n - 1 + p)
    m_lo_b = max(0, p)
    m_hi_b = min(2*n - 2, p + n - 1)
    m_lo = min(m_lo, m_lo_b)
    m_hi = max(m_hi_f, m_hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    valid_f = (jf >= 0) & (jf < n)
    valid_b = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(valid_f):
        a[valid_f] += (1.0 - h[jf[valid_f]])
    if np.any(valid_b):
        a[valid_b] -= h[jb[valid_b]]
    c_base = corr[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
    
    unaff = 0.0
    if m_lo > 0:
        unaff = max(unaff, float(corr[:m_lo].max()))
    if m_hi < 2*n - 2:
        unaff = max(unaff, float(corr[m_hi+1:].max()))
    
    cur_max = float(corr.max())
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    
    # Bisection on L
    L_lo = max(unaff, 0.0)
    L_hi = cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo = 0.0
        x_hi = 1.0
        ok = True
        if ok and np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            ok = False
        if ok and np.any(neg_mask):
            vals = (L - c_base[neg_mask]) / a[neg_mask]
            x_lo = max(x_lo, float(np.max(vals)))
        if ok and np.any(pos_mask):
            vals = (L - c_base[pos_mask]) / a[pos_mask]
            x_hi = min(x_hi, float(np.min(vals)))
        if x_lo > x_hi + 1e-14:
            ok = False
        if ok and has_diag:
            # c_diag + d*(1-2h0) - d^2 <= L where d = x-h0
            # -d^2 + (1-2h0)*d + (c_diag - L) <= 0
            # d^2 - (1-2h0)*d + (L - c_diag) >= 0
            # roots of d^2 - (1-2h0)*d + (L-c_diag) = 0
            b = -(1.0 - 2.0*h0)
            c_coef = L - c_diag
            disc = b*b - 4.0*c_coef
            if disc >= 0:
                r1 = (-b - np.sqrt(disc)) / 2.0
                r2 = (-b + np.sqrt(disc)) / 2.0
                # Need d^2 - (1-2h0)d + (L-c_diag) >= 0
                # i.e., d <= r1 or d >= r2
                d_lo = x_lo - h0
                d_hi = x_hi - h0
                # Check if [d_lo, d_hi] is entirely in (r1, r2) → infeasible
                if d_lo >= r1 and d_hi <= r2:
                    ok = False
                # Otherwise, we can still find feasible d
            else:
                # No real roots, quadratic is always positive (coeff of d^2 is 1 > 0)
                # So d^2 - (1-2h0)d + (L-c_diag) > 0 always → feasible
                pass
        if ok:
            L_hi = L
        else:
            L_lo = L
    
    L = L_hi
    # Find feasible range for x
    x_lo = 0.0
    x_hi = 1.0
    if np.any(neg_mask):
        vals = (L - c_base[neg_mask]) / a[neg_mask]
        x_lo = max(x_lo, float(np.max(vals)))
    if np.any(pos_mask):
        vals = (L - c_base[pos_mask]) / a[pos_mask]
        x_hi = min(x_hi, float(np.min(vals)))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    
    if has_diag:
        b = -(1.0 - 2.0*h0)
        c_coef = L - c_diag
        disc = b*b - 4.0*c_coef
        if disc >= 0:
            r1 = (-b - np.sqrt(disc)) / 2.0
            r2 = (-b + np.sqrt(disc)) / 2.0
            # d must satisfy d <= r1 or d >= r2, where d = x - h0
            # x must satisfy x <= h0+r1 or x >= h0+r2
            xc_lo = max(x_lo, 0.0)
            xc_hi = min(x_hi, 1.0)
            if xc_lo >= r1 + h0 and xc_hi <= r2 + h0:
                return h0  # infeasible
            # We need to pick x outside [h0+r1, h0+r2]
            # Candidate from left: min(xc_lo, h0+r1) if feasible
            # Candidate from right: max(xc_hi, h0+r2) if feasible
            candidates = set()
            candidates.add(float(np.clip(x_lo, 0.0, 1.0)))
            candidates.add(float(np.clip(x_hi, 0.0, 1.0)))
            candidates.add(float(np.clip(h0 + r1, 0.0, 1.0)))
            candidates.add(float(np.clip(h0 + r2, 0.0, 1.0)))
        else:
            candidates = {float(np.clip(x_lo, 0.0, 1.0)),
                          float(np.clip(x_hi, 0.0, 1.0)),
                          float(np.clip((x_lo+x_hi)/2.0, 0.0, 1.0))}
    else:
        if x_lo > x_hi:
            return h0
        candidates = {float(np.clip(x_lo, 0.0, 1.0)),
                      float(np.clip(x_hi, 0.0, 1.0)),
                      float(np.clip((x_lo+x_hi)/2.0, 0.0, 1.0))}
    
    best_x = h0
    best_val = cur_max
    for x in candidates:
        x = float(np.clip(x, 0.0, 1.0))
        d = x - h0
        nv = c_base + d * a
        if has_diag:
            nv[diag_idx] = c_diag + d * (1.0 - 2.0*h0) - d * d
        val = max(unaff, float(nv.max()))
        if val < best_val:
            best_val = val
            best_x = x
    return best_x

def exact_cd(h, n, max_passes, deadline, focus_top=10):
    """Exact coordinate descent with peak-focused ordering."""
    h = h.copy()
    for pass_num in range(max_passes):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(corr.max())
        
        # Get ordering: prioritize coordinates in top correlations
        top_k = np.argsort(corr)[-focus_top:]
        # Find which h indices are involved in top-k correlations
        involved = set()
        for m in top_k:
            m = int(m)
            lo = max(0, m - (n-1))
            hi = min(n-1, n-1 + m - (n-1))
            for j in range(max(0, n-1-m), min(n, n+m-(n-1)+1)):
                i = j
                partner = i + m - (n-1)
                if 0 <= i < n and 0 <= partner < n:
                    involved.add(i)
                    involved.add(partner)
        
        # Create ordering: involved first (reverse order for diversity), then others
        involved_list = list(involved)
        remaining = [i for i in range(n) if i not in involved]
        order = involved_list + remaining
        
        improved = False
        for p in order:
            if time.time() > deadline:
                break
            x_new = coordinate_update(h, n, p, corr)
            if abs(x_new - h[p]) > 1e-13:
                h[p] = x_new
                improved = True
        
        if not improved:
            break
        h = project_feasible(h, n)
    
    c, _ = compute_c5(h, n)
    return h, c

def grid_cd(h, n, max_passes, deadline, grid_size=80):
    """Grid-based coordinate descent (faster per step, less precise)."""
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(corr.max())
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            m_lo = max(0, n-1-p)
            m_hi_f = min(2*n-2, n-1+p)
            m_lo_b = max(0, p)
            m_hi_b = min(2*n-2, p+n-1)
            m_lo = min(m_lo, m_lo_b)
            m_hi = max(m_hi_f, m_hi_b)
            ms = np.arange(m_lo, m_hi+1)
            jf = p + ms - (n-1)
            jb = p - ms + (n-1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c_aff = corr[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(corr[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff = max(unaff, float(corr[m_hi+1:].max()))
            diag_m = (ms == (n-1))
            d_grid = np.linspace(-h0, 1.0-h0, grid_size)
            best_d = 0.0
            best_val = cur_max
            for d in d_grid:
                if abs(d) < 1e-14:
                    continue
                nv = c_aff + d * a
                if np.any(diag_m):
                    nv[diag_m] -= d * (2.0*h0 - 1.0)  # Wait, this is wrong
                    # Correct: nv[diag] = c_diag + d*(1-2h0) - d^2
                    # So the correction from c_diag is: d*(1-2h0) - d^2
                    # But we already added d*a, and a[diag] = 1-2h0
                    # So we need to subtract d^2
                    nv[diag_m] -= d * d
                val = max(unaff, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    c, _ = compute_c5(h, n)
    return h, c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

            hp = project_feasible(best_h + rng.normal(0, s, n), n)
            h_o, c_o = slsqp_run(hp, n, [5000, 20000], 250, deadline)
            h_o, c_o = grid_cd(h_o, n, 50, deadline, grid_size=40)
            if c_o < best_c:

    # Phase 5: Exact CD for final polishing
    if time.time() < deadline - 15:
        n = len(best_h)
        max_passes = max(5, int((deadline - time.time() - 5) / 3.0))
        hf, cf = exact_cd(best_h, n, max_passes, deadline, focus_top=15)
        if cf < best_c:
            best_h, best_c = hf, cf
    
    # Phase 6: Final SLSQP + exact CD
    if time.time() < deadline - 5:
        n = len(best_h)
        hf, cf = slsqp_run(best_h, n, [100000, 500000], 500, deadline)
        if cf < best_c:

        hf, cf = exact_cd(best_h, n, 10, deadline, focus_top=20)
        if cf < best_c:
            best_h, best_c = hf, cf
    
    best_h = project_feasible(best_h, len(best_h))
    best_c, _ = compute_c5(best_h, len(best_h))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx), corr

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=400, deadline=None):
    best_h = h.copy()
    best_c, _ = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c, _ = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def coordinate_update(h, n, p, corr):
    h0 = h[p]
    if h0 < 1e-15 or h0 > 1.0 - 1e-15:
        return h0
    m_lo = max(0, n - 1 - p)
    m_hi_f = min(2 * n - 2, n - 1 + p)
    m_lo_b = max(0, p)
    m_hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(m_lo, m_lo_b)
    m_hi = max(m_hi_f, m_hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = corr[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
    unaff = 0.0
    if m_lo > 0:
        unaff = max(unaff, float(corr[:m_lo].max()))
    if m_hi < 2 * n - 2:
        unaff = max(unaff, float(corr[m_hi + 1:].max()))
    cur_max = float(corr.max())
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo = max(unaff, 0.0)
    L_hi = cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo = 0.0
        x_hi = 1.0
        ok = True
        if ok and np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            ok = False
        if ok and np.any(neg_mask):
            vals = (L - c_base[neg_mask]) / a[neg_mask]
            x_lo = max(x_lo, float(np.max(vals)))
        if ok and np.any(pos_mask):
            vals = (L - c_base[pos_mask]) / a[pos_mask]
            x_hi = min(x_hi, float(np.min(vals)))
        if x_lo > x_hi + 1e-14:
            ok = False
        if ok and has_diag:
            b = -(1.0 - 2.0 * h0)
            cc = L - c_diag
            disc = b * b - 4.0 * cc
            if disc >= 0:
                r1 = (-b - np.sqrt(disc)) / 2.0
                r2 = (-b + np.sqrt(disc)) / 2.0
                d_lo = x_lo - h0
                d_hi = x_hi - h0
                if d_lo >= r1 - 1e-14 and d_hi <= r2 + 1e-14:
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    if np.any(neg_mask):
        vals = (L - c_base[neg_mask]) / a[neg_mask]
        x_lo = max(x_lo, float(np.max(vals)))
    if np.any(pos_mask):
        vals = (L - c_base[pos_mask]) / a[pos_mask]
        x_hi = min(x_hi, float(np.min(vals)))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    candidates = set()
    if has_diag:
        b = -(1.0 - 2.0 * h0)
        cc = L - c_diag
        disc = b * b - 4.0 * cc
        if disc >= 0:
            r1 = (-b - np.sqrt(disc)) / 2.0
            r2 = (-b + np.sqrt(disc)) / 2.0
            x_lr = h0 + r1
            x_rr = h0 + r2
            candidates.add(float(np.clip(min(x_lo, x_lr), 0.0, 1.0)))
            candidates.add(float(np.clip(max(x_hi, x_rr), 0.0, 1.0)))
            candidates.add(float(np.clip(x_lo, 0.0, 1.0)))
            candidates.add(float(np.clip(x_hi, 0.0, 1.0)))
        else:
            candidates.add(float(np.clip(x_lo, 0.0, 1.0)))
            candidates.add(float(np.clip(x_hi, 0.0, 1.0)))
            candidates.add(float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0)))
    else:
        if x_lo > x_hi:
            return h0
        candidates.add(float(np.clip(x_lo, 0.0, 1.0)))
        candidates.add(float(np.clip(x_hi, 0.0, 1.0)))
        candidates.add(float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0)))
    best_x = h0
    best_val = cur_max
    for x in candidates:
        x = float(np.clip(x, 0.0, 1.0))
        d = x - h0
        nv = c_base.copy() + d * a
        if has_diag:
            nv[diag_idx] = c_diag + d * (1.0 - 2.0 * h0) - d * d
        val = max(unaff, float(nv.max()))
        if val < best_val:
            best_val = val
            best_x = x
    return best_x

def exact_cd(h, n, max_passes, deadline, reverse_every=2):
    h = h.copy()
    for pass_num in range(max_passes):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(corr.max())
        direction = 1 if (pass_num // reverse_every) % 2 == 0 else -1
        order = range(n) if direction == 1 else range(n - 1, -1, -1)
        improved = False
        for p in order:
            if time.time() > deadline:
                break
            x_new = coordinate_update(h, n, p, corr)
            if abs(x_new - h[p]) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    c, _ = compute_c5(h, n)
    return h, c

def grid_cd(h, n, max_passes, deadline, grid_size=80):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(corr.max())
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi_f = min(2 * n - 2, n - 1 + p)
            m_lo_b = max(0, p)
            m_hi_b = min(2 * n - 2, p + n - 1)
            m_lo = min(m_lo, m_lo_b)
            m_hi = max(m_hi_f, m_hi_b)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c_aff = corr[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(corr[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(corr[m_hi + 1:].max()))
            diag_m = (ms == (n - 1))
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            best_d = 0.0
            best_val = cur_max
            for d in d_grid:
                if abs(d) < 1e-14:
                    continue
                nv = c_aff + d * a
                if np.any(diag_m):
                    nv[diag_m] -= d * d
                val = max(unaff, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    c, _ = compute_c5(h, n)
    return h, c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def subgradient_step(h, n, step_size, deadline):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    top_mask = corr >= c_max - 1e-6
    g = np.zeros(n)
    for p in range(n):
        m_lo = max(0, n - 1 - p)
        m_hi_f = min(2 * n - 2, n - 1 + p)
        m_lo_b = max(0, p)
        m_hi_b = min(2 * n - 2, p + n - 1)
        m_lo = min(m_lo, m_lo_b)
        m_hi = max(m_hi_f, m_hi_b)
        ms = np.arange(m_lo, m_hi + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        a = np.zeros(len(ms))
        if np.any(vf):
            a[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            a[vb] -= h[jb[vb]]
        active = top_mask[ms]
        if np.any(active):
            g[p] = a[active].mean()
    h_new = h - step_size * g
    h_new = project_feasible(h_new, n)
    return h_new
