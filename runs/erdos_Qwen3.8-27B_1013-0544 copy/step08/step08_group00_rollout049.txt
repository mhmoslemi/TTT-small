        wi = w[k_idx]
        if m == 0:
            g += wi * (1.0 - 2.0 * h)
        elif m > 0:
            g[:n-m] += wi * (1.0 - h[m:])
            g[n-m:] += -wi * h[:m]

            m2 = -m
            g[m2:] += wi * (1.0 - h[:n-m2])
            g[:n-m2] += -wi * h[m2:]
    return g

    #     So h[p-m] * (1-h[p]) with m = k-(n-1), need p-m >= 0, so m <= p
    #     k = (n-1)+m, m in [-p, -1], so k in [n-1-p, n-2]

    m_lo = n - 1 - p  # k = n-1-p (from backward, m=-p)
    m_hi = 2 * n - 2 - p  # k = 2n-2-p (from forward, m=n-1-p)

    m_hi = min(2*n-2, m_hi)

    #   Forward (m > 0, i.e., k > n-1): h[p] * (1-h[p+m]) -> x*(1-h[p+m])
    #     Change: (x - h0) * (1-h[p+m])
    #   Backward (m < 0, i.e., k < n-1): h[p+m] * (1-h[p]) -> (1-x)*h[p+m]
    #     Change: -(x - h0) * h[p+m]
    #   Self (m = 0, k = n-1): h[p]*(1-h[p]) -> x*(1-x)
    #     Change: x - x*x - h0 + h0*h0 = (x-h0) - (x²-h0²) = (x-h0)(1-x-h0)

        cands.append((1.0 - 2.0*h0) * 0.5)
    
    best_x = h0
    best_val = cur_max
    for d in cands:
        dd = float(np.clip(d, -h0, 1.0-h0))
        c_test = c_base + dd * a
        if has_diag:
            c_test[diag_i] -= dd*dd
        val = max(unaff_max, float(c_test.max()))
        if val < best_val:
            best_val = val
            best_x = h0 + dd
    
    return float(np.clip(best_x, 0.0, 1.0))
