    improved = 0
    for p in range(n):

        if h0 < 1e-12 or h0 > 1.0 - 1e-12:
            continue

        m_b_hi = min(2*n-2, p+n-1)
        m_lo = min(m_f_lo, m_b_lo)
        m_hi = max(m_f_hi, m_b_hi)
        ms = np.arange(m_lo, m_hi+1)
        a = np.zeros(len(ms))
        jf = p + ms - (n-1)
        vf = (jf >= 0) & (jf < n)
        if np.any(vf):
            a[vf] += 1.0 - h[jf[vf]]
        jb = p - ms + (n-1)
        vb = (jb >= 0) & (jb < n)
        if np.any(vb):
            a[vb] -= h[jb[vb]]
        c_aff = c[ms].copy()
        unaff = 0.0
        if m_lo > 0:
            unaff = max(unaff, c[:m_lo].max())
        if m_hi < 2*n-2:
            unaff = max(unaff, c[m_hi+1:].max())
        cur_max = max(unaff, c_aff.max())
        diag = (ms == (n-1))
        best_d = 0.0
        best_val = cur_max
        # Coarse
        dc = np.linspace(-h0, 1.0-h0, 15)
        for d in dc:
            if abs(d) < 1e-14: continue
            nv = c_aff + d*a
            if np.any(diag): nv[diag] -= d*d
            v = max(unaff, nv.max())
            if v < best_val:
                best_val = v; best_d = d
        # Fine
        if best_d != 0.0:
            span = (1.0-h0)/15.0
            lo = max(-h0, best_d-span)
            hi = min(1.0-h0, best_d+span)
            df = np.linspace(lo, hi, grid)
        else:
            df = np.linspace(-h0*0.2, (1.0-h0)*0.2, grid)
        for d in df:
            if abs(d) < 1e-14: continue
            nv = c_aff + d*a
            if np.any(diag): nv[diag] -= d*d
            v = max(unaff, nv.max())
            if v < best_val:
                best_val = v; best_d = d
        if abs(best_d) > 1e-14 and best_val < cur_max - 1e-14:
            h[p] = np.clip(h0 + best_d, 0.0, 1.0)
            c[ms] = c_aff + best_d * a
            if np.any(diag): c[ms[diag]] -= best_d*best_d
            improved += 1
    return h, c, improved

def cd_with_vibration(h, n, max_passes=500, deadline=None, rng=None, grid=80):

                stuck_count = 0
    return best_h if 'best_h' in dir() else h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def grad_at_lag(h, n, m):
    g = np.zeros(n)
    offset = m - (n - 1)
    ps = np.arange(max(0, -offset), min(n, n - offset))
    if len(ps) > 0:
        g[ps] += 1.0 - h[ps + offset]
    ps2 = np.arange(max(0, offset), min(n, n + offset))
    if len(ps2) > 0:
        g[ps2] -= h[ps2 - offset]
    return g

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:

            if tot < 1e-15:

def lbfgs_smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

                           bounds=bounds, constraints=cons,
