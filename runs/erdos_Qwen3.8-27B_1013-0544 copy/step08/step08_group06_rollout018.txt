import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def cd_step(h, n, c, current_c5, idx, grid_size):
    """Try to improve coordinate idx. Returns (new_h_val, new_c5, changed)."""
    h0 = h[idx]
    if h0 < 1e-12:
        lo_d, hi_d = 0.0, 1.0
    elif h0 > 1.0 - 1e-12:
        return h0, current_c5, False
    else:
        lo_d, hi_d = -h0, 1.0 - h0
    if hi_d - lo_d < 1e-12:
        return h0, current_c5, False

    m_lo = max(0, min(n - 1 - idx, idx))
    m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
    ms = np.arange(m_lo, m_hi + 1)
    M = len(ms)

    a = np.zeros(M)
    jh = idx + ms - (n - 1)
    vh = (jh >= 0) & (jh < n)
    a[vh] += (1.0 - h[jh[vh]])
    jg = idx - ms + (n - 1)
    vg = (jg >= 0) & (jg < n)
    a[vg] -= h[jg[vg]]

    c_aff = c[ms]
    unaff_max = 0.0
    if m_lo > 0:
        t = float(c[:m_lo].max())
        if t > unaff_max:
            unaff_max = t
    if m_hi < 2 * n - 2:
        t = float(c[m_hi + 1:].max())
        if t > unaff_max:
            unaff_max = t

    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0

    d_grid = np.linspace(lo_d, hi_d, grid_size)
    nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
    if has_diag:
        nv[:, diag_idx] -= d_grid[:, None] ** 2
    vals = np.maximum(unaff_max, nv.max(axis=1))

    best_i = int(np.argmin(vals))
    best_d = float(d_grid[best_i])
    best_val = float(vals[best_i])

    # Parabolic refinement
    if 0 < best_i < grid_size - 1:
        y = vals[best_i - 1:best_i + 2]
        x = d_grid[best_i - 1:best_i + 2]
        denom = (y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2] * (x[0] - x[1]))
        if abs(denom) > 1e-15:
            xv = 0.5 * (y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2] * (x[0] - x[1])) / denom
            if x[0] < xv < x[2]:
                nv1 = c_aff + xv * a
                if has_diag:
                    nv1 = nv1.copy()
                    nv1[diag_idx] -= xv ** 2
                v1 = max(unaff_max, float(nv1.max()))
                if v1 < best_val - 1e-14:
                    best_val = v1
                    best_d = xv

    if abs(best_d) > 1e-10 and best_val < current_c5 - 1e-14:
        new_val = float(np.clip(h0 + best_d, 0.0, 1.0))
        c[ms] = c_aff + best_d * a
        if has_diag:
            c[ms[diag_idx]] -= best_d ** 2
        new_max = max(unaff_max, float(c[ms].max()))
        return new_val, new_max, True
    return h0, current_c5, False

            if changed:

        if not improved:

            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subspace_slsqp(h, n, subspace_size, deadline):
    """Optimize a random subset of coordinates jointly."""
    if time.time() > deadline:
        return h, compute_c5(h, n)
    rng = np.random.RandomState(int(time.time() * 1000) % 100000)
    idxs = rng.choice(n, size=subspace_size, replace=False)
    idxs.sort()
    x0 = h[idxs].copy()
    fixed = h.copy()
    
    def obj(x):
        fixed_sub = fixed.copy()
        fixed_sub[idxs] = x
        return compute_c5(fixed_sub, n)
    
    def constraint(x):
        return x.sum() + (h.sum() - x0.sum()) - n / 2.0
    
    def jac(x):
        return np.ones(subspace_size)
    
    bounds = [(0.0, 1.0)] * subspace_size
    cons = [{'type': 'eq', 'fun': constraint, 'jac': jac}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 200, 'ftol': 1e-12})
        new_h = h.copy()
        new_h[idxs] = np.clip(res.x, 0, 1)
        new_h = project_feasible(new_h, n)
        new_c = compute_c5(new_h, n)
        old_c = compute_c5(h, n)
        if new_c < old_c:
            return new_h, new_c
    except Exception:
        pass
    return h, compute_c5(h, n)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

            continue

        a = np.zeros(len(ms))
        jh = idx + ms - (n - 1)

        a[vh] += (1.0 - h[jh[vh]])
        jg = idx - ms + (n - 1)
        vg = (jg >= 0) & (jg < n)
        a[vg] -= h[jg[vg]]
        c_aff = c[ms].copy()
        unaff_max = 0.0
        if m_lo > 0:
            t = c[:m_lo].max()
            if t > unaff_max:

        if m_hi < 2 * n - 2:

            if abs(denom) > 1e-15:

def binary_round(h, n):
    k = n // 2
    idx = np.argsort(h)[::-1][:k]

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - n)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    return x.sum() - n / 2.0
