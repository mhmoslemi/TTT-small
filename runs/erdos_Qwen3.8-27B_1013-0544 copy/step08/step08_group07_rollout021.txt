def coord_cd(h, n, max_passes=200, deadline=None, grid_size=100):
    """Coordinate descent with grid search + refinement."""
    h = h.copy()
    best_c = compute_c5(h, n)
    off = n - 1
    for pidx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c_prev = compute_corr(h, n)
        current_max = float(c_prev.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            # Sensitivity vector
            a = np.zeros(2*n - 1)
            jf = p + np.arange(2*n-1) - off
            valid_f = (jf >= 0) & (jf < n)
            a[valid_f] += (1.0 - h[jf[valid_f]])
            jb = p - np.arange(2*n-1) + off
            valid_b = (jb >= 0) & (jb < n)
            a[valid_b] -= h[jb[valid_b]]
            
            diag = (np.arange(2*n-1) == off)

            best_val = current_max

                if abs(d) < 1e-14:
                    continue

                if np.any(diag):
                    nv = nv.copy()
                    nv[diag] -= d * d

                    best_val = val
                    best_d = d

                span = (1.0 - h0) / 30.0

        new_c = compute_c5(h, n)

def subgradient_polish(h, n, n_iter=200, deadline=None):
    """Subgradient method with BB step for the (non-smooth) max objective."""
    h = h.copy()
    dx = 2.0 / n
    c = compute_corr(h, n)
    best_c = float(c.max()) * dx
    best_h = h.copy()
    off = n - 1
    
    step = 0.5
    prev_h = h.copy()
    prev_grad = None
    
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        c = compute_corr(h, n)
        cur_max = c.max()
        
        # Find active set (within tolerance of max)
        tol = 0.01 * cur_max + 1e-10
        active = c >= (cur_max - tol)
        
        # Compute subgradient: average gradient over active set
        grad = np.zeros(n)
        n_active = active.sum()
        for m in np.where(active)[0]:
            k = m - off
            lo1 = max(0, -k)
            hi1 = min(n, n - k)
            if hi1 > lo1:
                grad[lo1:hi1] += (1.0 - h[lo1+k:hi1+k])
            lo2 = max(0, k)
            hi2 = min(n, n + k)
            if hi2 > lo2:
                grad[lo2:hi2] -= h[lo2-k:hi2-k]
        grad /= n_active
        
        # BB step size
        if prev_grad is not None:
            s = h - prev_h
            y = grad - prev_grad
            sy = s @ y
            yy = y @ y
            ss = s @ s
            if yy > 1e-15 and sy > 1e-15:
                step = ss / sy  # BB1
            elif yy > 1e-15:
                step = sy / yy  # BB2
            step = max(0.01, min(step, 2.0))
        
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_corr(h_new, n)
        obj_new = float(c_new.max()) * dx
        if obj_new < best_c:
            best_h = h_new.copy()
            best_c = obj_new
        
        prev_h = h.copy()
        prev_grad = grad.copy()
        h = h_new
        step *= 0.95
    
    return best_h, best_c

        best_h, best_c = h1, c1

    # Phase 2: Subgradient polish
    if time.time() < deadline - 80:
        h2, c2 = subgrad_polish(best_h, n, 200, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Coordinate descent
    if time.time() < deadline - 120:
        h3, c3 = coord_cd(best_h, n, 250, deadline, grid_size=90)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Another L-BFGS-B round with very high alpha
    if time.time() < deadline - 100:
        h4, c4 = lbfgs_opt(best_h, n, [200000, 1000000, 5000000], 600, deadline)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: More coordinate descent
    if time.time() < deadline - 60:
        h5, c5 = coord_cd(best_h, n, 300, deadline, grid_size=110)
        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 6: Try different n values
    n_curr = len(best_h)
    x_curr = np.linspace(0, 2, n_curr, endpoint=False)
    for n_try in [n_curr - 100, n_curr + 100, n_curr + 200]:
        if n_try < 200 or time.time() > deadline - 100:
            continue
        x_s = np.linspace(0, 2, n_try, endpoint=False)
        h_try = project_feasible(np.interp(x_s, x_curr, best_h), n_try)
        h_try, c_try = lbfgs_opt(h_try, n_try, [5000, 50000, 200000], 300, deadline)
        if time.time() < deadline - 50:
            h_try, c_try = coord_cd(h_try, n_try, 80, deadline, grid_size=70)
        if c_try < best_c:
            best_h = h_try
            best_c = c_try

    # Phase 7: Final subgradient + coord polish
    n_fin = len(best_h)
    if time.time() < deadline - 30:
        hf, cf = subgrad_polish(best_h, n_fin, 150, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 5:
        hf, cf = coord_cd(best_h, n_fin, 100, deadline, grid_size=130)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
