def slsqp_step(h, n, alpha, maxiter, ftol, bounds, cons):
    def f(x):
        return smooth_max_val(x, n, alpha)
    def g(x):
        return grad_smooth(x, n, alpha)
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return np.asarray(res.x, dtype=float)
    except Exception:
        return h

def pair_search(h, n, max_tries, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    off = n - 1
    no_improve = 0
    for trial in range(max_tries):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        tol = max(0.025, 0.0015 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        if trial % 7 == 6 or active.size <= 2:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
                if i == j:
                    continue
        else:
            kk = int(active[int(rng.randint(active.size))])
            d = kk - off
            if d >= 0:
                m = n - d
                if m <= 0:
                    continue
                b = int(rng.randint(m))
                i, j = b + d, b
            else:
                e = -d
                m = n - e
                if m <= 0:
                    continue
                b = e + int(rng.randint(m))
                i, j = b - e, b
        if i == j:
            continue
        h_i, h_j = h[i], h[j]
        lo = max(-h_i, -(1.0 - h_j))
        hi = min(1.0 - h_i, h_j)
        if hi - lo < 1e-14:
            no_improve += 1
            continue
        # Golden section
        invphi = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = lo, hi
        c1 = a + (1.0 - invphi) * (b - a)
        c2 = a + invphi * (b - a)

        def ev(u):
            if abs(u) < 1e-15:
                return cur_max
            hn = h.copy()
            hn[i] = h_i + u
            hn[j] = h_j - u
            return float(np.max(np.correlate(hn, 1.0 - hn, mode='full')))

        f1, f2 = ev(c1), ev(c2)
        for _ in range(10):
            if f1 < f2:
                b, c2, f2 = c2, c1, f1
                c1 = a + (1.0 - invphi) * (b - a)
                f1 = ev(c1)
            else:
                a, c1, f1 = c1, c2, f2
                c2 = a + invphi * (b - a)
                f2 = ev(c2)
        best_u = 0.5 * (a + b)
        best_val = ev(best_u)
        if best_val < cur_max - 1e-10:
            h[i] = h_i + best_u
            h[j] = h_j - best_u
            c = np.correlate(h, 1.0 - h, mode='full')
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break
    return h, compute_c5(h, n)

def coord_search(h, n, deadline, rng, ngrid=40):
    h = h.copy()
    cur_c = compute_c5(h, n)
    improve = True
    sweeps = 0
    while improve and time.time() < deadline:
        improve = False
        sweeps += 1
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                return h, compute_c5(h, n)
            hi = h[i]
            lo, hiT = 0.0, 1.0
            if lo >= hiT:
                continue
            us = np.linspace(lo, hiT, ngrid)
            best_t = hi
            best_v = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            h[i] = 0.0
            for u in us:
                h[i] = u
                v = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                if v < best_v:
                    best_v = v
                    best_t = u
            h[i] = best_t
            if best_t != hi:
                improve = True
        # Project to maintain sum constraint
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c >= cur_c - 1e-12:
            improve = False
        else:
            cur_c = new_c
    return h, compute_c5(h, n)

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def random_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, 10):
        a = rng.uniform(-0.4, 0.4)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += a * np.sin(0.5 * np.pi * m * x + ph)
        h += 0.5 * a * np.cos(np.pi * m * x + ph)
    return project_feasible(np.clip(h, 0.0, 1.0), n)

def optimize_full(h, n, deadline, rng, quick=False):
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    # Symmetries
    for sym_h in [0.5*(h + h[::-1]), 1.0 - h, 0.5*(h + 1.0 - h[::-1])]:
        sh = project_feasible(sym_h, n)
        sc = compute_c5(sh, n)
        if sc < c:
            h, c = sh, sc
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    # SLSQP phases
    if not quick:
        slsqp_plans = [(20, 60), (100, 50), (500, 40), (2000, 30), (10000, 20), (50000, 15)]
    else:
        slsqp_plans = [(50, 30), (500, 25), (5000, 20)]
    if n <= 200 and not quick:
        slsqp_plans.append((200000, 10))
    for alpha, mx in slsqp_plans:
        if time.time() > deadline - 5.0:
            break
        x = slsqp_step(h, n, alpha, mx, 1e-14, bounds, cons)
        x = project_feasible(x, n)
        xc = compute_c5(x, n)
        if xc < c:
            h, c = x, xc
    # Pair search (main workhorse)
    if time.time() < deadline - 3.0:
        rem = deadline - time.time()
        mt = int(min(8000 if not quick else 3000, max(500, rem * 25)))
        h, c = pair_search(h, n, mt, deadline - 2.0, rng)
    # Coordinate search
    if time.time() < deadline - 3.0:
        h, c = coord_search(h, n, deadline - 2.0, rng, ngrid=30)
    # Final pair polish
    if time.time() < deadline - 1.0:
        h, c = pair_search(h, n, 500, deadline - 0.5, rng)
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)
    # Get initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        h0 = random_start(200, rng)
    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    # Symmetries on initial
    for sym_h in [0.5*(best_h + best_h[::-1]), 1.0 - best_h, 0.5*(best_h + 1.0 - best_h[::-1])]:
        sh = project_feasible(sym_h, n0)
        sc = compute_c5(sh, n0)
        if sc < best_c:
            best_h, best_c = sh, sc
    if hard - t0 < 40.0:
        return best_h, best_c, n0
    # Main optimization at native resolution
    main_dl = hard - max(100.0, 0.2 * float(budget_s))
    best_h, best_c = optimize_full(best_h, n0, main_dl, rng, quick=False)
    # Try multiple resolutions
    cands = sorted(set([int(n0 * 0.4), int(n0 * 0.6), int(n0 * 0.8), 100, 150, 200, 256, 320, 400]),
                   key=lambda x: abs(x - n0))
    for m in cands:
        if time.time() > hard - 30.0:
            break
        if m < 50 or m == n0:
            continue
        rem = hard - time.time()
        if rem < 40.0:
            break
        if m > 400 and rem < 80.0:
            continue
        dl = time.time() + min(55.0, 0.25 * rem)
        if dl < time.time() + 10.0:
            continue
        ht = interp_h(best_h, m)
        ht, ct = optimize_full(ht, m, dl, rng, quick=(m > 300))
        if ct < best_c:
            best_h, best_c = ht, ct
    # Perturbation restarts
    for s in [0.04, 0.1, 0.2, 0.4]:
        if time.time() > hard - 35.0:
            break
        rem = hard - time.time()
        if rem < 30.0:
            break
        n = len(best_h)
        hp = project_feasible(best_h + rng.normal(0.0, s, n), n)
        dl = time.time() + min(45.0, 0.3 * rem)
        ht, ct = optimize_full(hp, n, dl, rng, quick=True)
        if ct < best_c:
            best_h, best_c = ht, ct
    # Final polish
    if time.time() < hard - 5.0:
        n = len(best_h)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
        x = slsqp_step(best_h, n, 100000, 15, 1e-14, bounds, cons)
        x = project_feasible(x, n)
        xc = compute_c5(x, n)
        if xc < best_c:
            best_h, best_c = x, xc
        if time.time() < hard - 2.0:
            best_h, best_c = pair_search(best_h, n, 300, hard - 1.0, rng)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
