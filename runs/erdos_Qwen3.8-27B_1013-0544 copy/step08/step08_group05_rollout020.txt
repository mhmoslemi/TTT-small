    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(55):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:

def compute_c5(h, n):

    return float(c.max()) * (2.0 / n)

def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    grad = (A - B)[s:s + n] * dx
    val = (m + np.log(z) / alpha) * dx
    return val, grad

Let me try a different angle: **What if we use the "duality" of the problem?**

The max correlation is achieved at some lag k*. If we can characterize what makes a particular lag "bad", we can design h to avoid it.

For lag k, the overlap is $\sum_i h[i](1-h[i+k])$ (with boundary effects). If h is binary, this is the number of "1 followed by 0 at distance k" patterns.

To minimize ALL lags simultaneously, we want a sequence where, for every k, the number of 1→0 transitions at distance k is small.

This is related to the concept of "balanced sequences" or "low-discrepancy sequences" in the discrete setting.

**Key insight**: A sequence that is "as random as possible" (in the sense of having uniform autocorrelation) might be good. But we also need the integral constraint (exactly half 1s).

**New idea: Use a greedy construction**

Start with a random binary sequence with exactly n/2 ones. Then, at each step, find the lag with maximum correlation, and modify the sequence to reduce that specific correlation.

This is like a "targeted" local search that focuses on the worst lag.

def targeted_binary_search(h_init, n, rng, deadline, max_iter=100000):
    """Binary local search that targets the worst lag."""
    h = h_init.copy()
    c = np.correlate(h, 1.0 - h, mode="full")
    best_c = float(c.max())
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Find the worst lag
        k_worst = int(np.argmax(c))
        cm = float(c.max())
        
        # Try to reduce correlation at k_worst
        # The correlation at lag k_worst - (n-1) is sum_i h[i]*(1-h[i+k])
        # To reduce it, find a position i where h[i]=1 and h[i+k]=0
        # and either set h[i]=0 (find j where h[j]=0 and set to 1)
        # This is a swap move targeting the worst lag
        
        k = k_worst - (n - 1)  # actual lag
        # Find candidate swaps
        if k > 0:
            # c[k_worst] = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
            # Find i where h[i]=1, h[i-k]=0
            # Swap: set h[i]=0, h[j]=1 where h[j]=0
            # This reduces c[k_worst] by 1 if h[j-k] is anything and h[j] was 0
            # But it might increase other lags
            cand_i = [i for i in range(k, n) if h[i] == 1 and h[i-k] == 0]
            cand_j = [j for j in range(n) if h[j] == 0]
        elif k < 0:
            kk = -k
            # c[k_worst] = sum_{i=0}^{n-1-kk} h[i]*(1-h[i+kk])
            cand_i = [i for i in range(n - kk) if h[i] == 1 and h[i+kk] == 0]
            cand_j = [j for j in range(n) if h[j] == 0]
        else:
            cand_i = []
        
        if not cand_i or not cand_j:
            # No improvement possible at this lag, try random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
        else:
            i = int(rng.choice(cand_i))
            j = int(rng.choice(cand_j))
        
        if i == j:
            continue
        
        # Evaluate the swap using the fast method
        c2, nm = binary_trial(c, h, 1.0 - h[::-1], i, j, n)
        if nm < cm - 1e-9:
            c = c2
            h[i] = 0.0
            h[j] = 1.0
            if nm < best_c:
                best_c = nm
                best_h = h.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(55):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
    y = np.clip(y, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += d / mask.sum()
    return y

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx

def pen_obj(x, n, alpha, lam):
    val, g = smooth_val_grad(x, n, alpha)
    return val + lam * (x.sum() - 0.5 * n) ** 2

def pen_grad(x, n, alpha, lam):
    _, g = smooth_val_grad(x, n, alpha)
    g = g + 2.0 * lam * (x.sum() - 0.5 * n)
    return g

def lbfgs_polish(h, n, deadline, alphas=None, lams=None):
    if alphas is None:
        alphas = [2000, 10000, 50000]
    if lams is None:
        lams = [0.0, 1.0, 10.0]
    best = project_simplex(h.copy(), n)
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        for lam in lams:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(pen_obj, best.copy(), args=(n, a, lam),
                               jac=pen_grad, method='L-BFGS-B',
                               bounds=bounds, options={'maxiter': 150, 'gtol': 1e-5})
                nh = project_simplex(res.x, n)
                nc = compute_c5(nh, n)
                if nc < bc - 1e-13:
                    best, bc = nh, nc
            except Exception:
                pass
    return best, bc

def coord_cd(h, n, max_passes=4, deadline=None, grid=32, refine=10, tol=1e-12):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    xs0 = np.linspace(0.0, 1.0, grid)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = float(x[p])
            A = np.zeros(L)
            A[p:p + n] += om
            st = diag - p
            A[st:st + n] -= x
            cur = float(c.max())
            d = xs0 - old
            cand = c[None, :] + d[:, None] * A[None, :]
            cand[:, diag] -= d * d
            i = int(np.argmin(cand.max(axis=1)))
            bx = float(xs0[i])
            if bx != old and cand[i, :].max() < cur - tol:
                sp = 1.0 / (grid - 1)
                lo = max(0.0, bx - sp)
                hi = min(1.0, bx + sp)
                xx = np.linspace(lo, hi, refine + 1)
                d2 = xx - old
                cand2 = c[None, :] + d2[:, None] * A[None, :]
                cand2[:, diag] -= d2 * d2
                j = int(np.argmin(cand2.max(axis=1)))
                bx = float(xx[j])
                if bx != old and cand2[j, :].max() < cur - tol:
                    d = bx - old
                    c[p:p + n] += d * om
                    c[st:st + n] -= d * x
                    c[diag] -= d * d
                    x[p] = bx
                    om[diag - p] = 1.0 - bx
                    improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - tol:
            best, bc = xp, cp
        if not improved:
            break
    return best, bc

def binary_trial(c, h, om, i, j, n):
    d = c.copy()
    d[i:i + n] -= om
    st = n - 1 - i
    d[st:st + n] += h
    d[j:j + n] += om
    st = n - 1 - j
    d[st:st + n] -= h
    d[n - 1] -= 2.0
    d[n - 1 + i - j] += 1.0
    d[n - 1 + j - i] += 1.0
    return d, float(d.max())

def targeted_binary(h_init, n, rng, deadline):
    h = h_init.copy()
    om = 1.0 - h[::-1]
    c = np.correlate(h, 1.0 - h, mode="full")
    best = h.copy()
    bc = float(c.max())
    stale = 0
    while time.time() < deadline:
        cm = float(c.max())
        kw = int(np.argmax(c))
        k = kw - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        found = False
        tries = 0
        while tries < 50 and not found:
            tries += 1
            if k > 0 and len(ones) > 0:
                ci = ones[ones >= k]
                ci = ci[np.array([h[int(i) - k] == 0 for i in ci])]
                if len(ci) > 0:
                    i = int(ci[rng.randint(len(ci))])
                else:
                    i = int(rng.choice(ones))
            elif k < 0:
                kk = -k
                ci = ones[ones < n - kk]
                ci = ci[np.array([h[int(i) + kk] == 0 for i in ci])]
                if len(ci) > 0:
                    i = int(ci[rng.randint(len(ci))])
                else:
                    i = int(rng.choice(ones))
            else:
                i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            if i == j:
                continue
            c2, nm = binary_trial(c, h, om, i, j, n)
            if nm < cm - 1e-9:
                c = c2
                h[i] = 0.0
                h[j] = 1.0
                om[n - 1 - i] = 1.0
                om[n - 1 - j] = 0.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if nm < bc:
                    best = h.copy()
                    bc = nm
                    stale = 0
                else:
                    stale += 1
                found = True
        if not found:
            stale += 1
            if stale > 200:
                break
    return best, bc * (2.0 / n)

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)
