def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def project_feasible(z, n):
    # Same as before (binary search for threshold)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5*n
    low, high = z.min()-1, z.max()+1
    for _ in range(60):
        tau = 0.5*(low+high)
        if np.clip(z-tau, 0, 1).sum() > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5*(low+high), 0, 1)
    res = target - y.sum()
    if abs(res) > 1e-12:
        mask = (y < 1 - 1e-12) if res > 0 else (y > 1e-12)
        room = (1-y) if res > 0 else y
        tot = room[mask].sum()
        if tot > 1e-12:
            y[mask] += (res/tot)*room[mask]
    return np.clip(y, 0, 1)

def get_grad_for_j(h, n, j):
    """Compute dc[k]/dh[j] for all k"""
    n2 = 2*n - 1
    g = np.zeros(n2)
    off = n - 1
    for k in range(n2):
        d = k - off
        e = abs(d)
        if e == 0:
            g[k] = 1.0 - 2.0*h[j]
        else:
            val = 0.0
            if e <= j <= n-1:
                val += (1.0 - h[j-e])
            if 0 <= j <= n-1-e:
                val -= h[j+e]
            g[k] = val
    return g

def coord_min_1d(c, g_j, h0, n, n_grid=200):
    """Find t in [0,1] that minimizes f(t) = (2/n)*max_k c_k(t)"""
    off = n - 1
    # c_k(t) = c[k] + (t-h0)*g_j[k] for k != off
    # c_off(t) = c[off] + t*(1-t) - h0*(1-h0)
    t_grid = np.linspace(0, 1, n_grid)
    best_t = h0
    best_val = np.inf
    for t in t_grid:
        vals = c + (t - h0) * g_j
        vals[off] = c[off] + t*(1-t) - h0*(1-h0)
        m = vals.max()
        if m < best_val:
            best_val = m
            best_t = t
    # Refine around best_t
    lo = max(0, best_t - 1.0/(n_grid-1))
    hi = min(1, best_t + 1.0/(n_grid-1))
    t_refine = np.linspace(lo, hi, 100)
    for t in t_refine:
        vals = c + (t - h0) * g_j
        vals[off] = c[off] + t*(1-t) - h0*(1-h0)
        m = vals.max()
        if m < best_val:
            best_val = m
            best_t = t
    return best_t

    h = h.copy()
    c5 = compute_c5(h, n)
    if sweep_order is None:
        sweep_order = list(range(n))
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0-h, mode='full')
        improved = 0
        for j in sweep_order:
            if time.time() > deadline:

            h0 = h[j]
            g_j = get_grad_for_j(h, n, j)
            t_new = coord_min_1d(c, g_j, h0, n, 200)
            if abs(t_new - h0) > 1e-10:
                h[j] = t_new
                # Update c (recompute is faster than incremental for small changes)
                # Actually, let's recompute c after each sweep
                improved += 1
        c = np.correlate(h, 1.0-h, mode='full')
        # Renormalize (coordinate descent might change the sum slightly)
        h = project_feasible(h, n)
        c5_new = compute_c5(h, n)
        if c5_new < c5 - 1e-10:
            c5 = c5_new
        elif improved == 0:

    return h, compute_c5(h, n)
