    if time.time() < hard - 60.0:
        dl3 = min(hard - 40.0, time.time() + 60.0)
        best_h, best_c = slsqp_refine(best_h, n0, [(1000, 30), (3000, 25), (10000, 20)], dl3)
    
    # Phase 4: More pair search
    if time.time() < hard - 30.0:
        dl4 = min(hard - 15.0, time.time() + 80.0)
        best_h, best_c = pair_search(best_h, n0, 5000, dl4, rng)
    
    # Phase 5: Try other resolutions
    n_cur = len(best_h)
    for m in [400, 600, 800, 1000]:
        if time.time() > hard - 30.0:
            break
        if m == n_cur:
            continue
        rem = hard - time.time()
        if rem < 40.0:
            break
        dl = time.time() + min(35.0, 0.3 * rem)
        ht = interp_h(best_h, m)
        ht, ct = slsqp_refine(ht, m, [(100, 30), (500, 25)], dl)
        ht, ct = pair_search(ht, m, 2000, dl, rng)
        if ct < best_c:
            best_h, best_c = ht, ct
    
    # Phase 6: Final pair search polish
    if time.time() < hard - 3.0:
        n_f = len(best_h)
        best_h, best_c = pair_search(best_h, n_f, 4000, hard - 1.0, rng)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
