    # Apply 1-swap (i1, j1)
    d = c.copy()
    d[i1:i1+n] -= om
    st = n - 1 - i1
    d[st:st+n] += h
    d[j1:j1+n] += om
    st = n - 1 - j1
    d[st:st+n] -= h
    d[n-1] -= 2.0
    d[n-1+i1-j1] += 1.0
    d[n-1+j1-i1] += 1.0

    # om[n-1-j1]: was 1-h[j1] = 1, now 1-h[j1] = 0 → changes from 1 to 0

    h2 = h.copy()

    d[n-1] -= 2.0

    d[n-1+j2-i2] += 1.0

import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        m = 0.5*(lo+hi)
        if np.clip(h+m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h+hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-10) & (y < 1.0-1e-10)
        if mask.any():
            y[mask] += d/mask.sum()
        else:
            idx = int(np.argmax(1.0-y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx]+d, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0-h, mode="full")
    return float(c.max()) * (2.0/n)

def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0-x, mode="full")
    m = float(c.max())
    sh = alpha*(c-m)
    sh -= sh.max()
    ee = np.exp(sh)
    z = float(ee.sum())
    w = ee/z
    dx = 2.0/n
    A = np.convolve(1.0-x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n-1
    return (m+np.log(z)/alpha)*dx, (A-B)[s:s+n]*dx

def smooth_obj(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]

def sum_con(x, n):
    return x.sum() - 0.5*n

def sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)

def true_subgrad_and_val(x, n):
    c = np.correlate(x, 1.0-x, mode="full")
    idx = int(np.argmax(c))
    k = idx - (n-1)
    g = np.zeros(n)
    om = 1.0-x
    if k >= 0:
        g[:n-k] += om[k:]
        g[k:] -= x[:n-k]
    else:
        g[-k:] += om[:n+k]
        g[:n+k] -= x[-k:]
    return g, float(c.max())*(2.0/n)

def subgrad_phase(x, n, deadline):
    x = project_simplex(x, n).copy()
    best = x.copy()
    bc = compute_c5(best, n)
    v = np.zeros(n)
    eta = 0.003
    gamma = 0.92
    stall = 0
    while time.time() < deadline:
        g, cv = true_subgrad_and_val(x, n)
        v = gamma*v - eta*g
        x_new = project_simplex(x+v, n)
        cn = compute_c5(x_new, n)
        if cn < bc - 1e-11:
            best = x_new.copy()
            bc = cn
            stall = 0
        else:
            stall += 1
        x = x_new
        if stall > 0 and stall % 60 == 0:
            eta *= 0.88
            gamma = max(0.88, gamma*0.998)
            if eta < 1e-7:
                break
    return best, bc

def slsqp_polish(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n)
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)]*n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a), jac=smooth_grad,
                          method="SLSQP", bounds=bounds, constraints=cons,
                          options={"maxiter": int(maxiter), "ftol": 1e-14})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-14:
            best = nh
            bc = nc
    return best, bc

def coord_eval(c, A, xs, old, n):
    d = xs - old
    cand = c[None,:] + d[:,None]*A[None,:]
    cand[:, n-1] -= d*d
    return cand.max(axis=1)

    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2*n-1
    diag = n-1

    xs0 = np.linspace(0.0, 1.0, grid)

            vals = coord_eval(c, A, xs0, old, n)
            i = int(np.argmin(vals))

            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 4:
                break
            for _ in range(4):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if h[i] > 0.5 and h[j] < 0.5:
                    h[i] = 0.0
                    h[j] = 1.0
                elif h[j] > 0.5 and h[i] < 0.5:
                    h[j] = 0.0
                    h[i] = 1.0
    return best, bcm*dx

def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(5.0, budget_s - 5.0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 500
        h0 = np.full(n0, 0.5)

    h0 = project_simplex(h0, n0)
    best_h = h0.copy()
    best_c = float(compute_c5(h0, n0))
    n_best = int(n0)

    for v in (0.5*(h0+h0[::-1]), 0.5*(h0+(1.0-h0[::-1]))):
        vv = project_simplex(v, n_best)
        cv = compute_c5(vv, n_best)
        if cv < best_c:
            best_h = vv.copy()
            best_c = cv

    if n_best % 2 == 0:
        hb = binarize(h0, n_best)
        if hb is not None:
            cb = compute_c5(hb, n_best)
            if cb < best_c:
                best_h = hb.copy()
                best_c = cb

    # Phase 1: Subgradient exploration
    t_sg = min(deadline - 400.0, t0 + 40.0)
    if time.time() < t_sg:
        h1, c1 = subgrad_phase(best_h, n_best, t_sg)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: SLSQP with increasing alpha
    t_sls = min(deadline - 250.0, time.time() + 250.0)
    h1, c1 = slsqp_polish(best_h, n_best, [5000, 50000, 500000, 5000000], 200, t_sls)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 3: Coordinate descent
    t_cd = min(deadline - 150.0, time.time() + 150.0)
    h1, c1 = coord_cd(best_h, n_best, max_passes=8, deadline=t_cd, grid=48, refine=20)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 4: Binary optimization with 1-swap and 2-swap
    if n_best % 2 == 0:
        t_bin = min(deadline - 80.0, time.time() + 200.0)
        hb, cb = binary_optimize(best_h, n_best, rng, t_bin)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb
            # Polish after binary
            t_pol = min(deadline - 30.0, time.time() + 60.0)
            h1, c1 = slsqp_polish(best_h, n_best, [50000, 500000], 150, t_pol)
            if c1 < best_c:
                best_h, best_c = h1, c1
            h1, c1 = coord_cd(best_h, n_best, max_passes=4, deadline=min(deadline-10.0, time.time()+40.0), grid=48, refine=20)
            if c1 < best_c:
                best_h, best_c = h1, c1

    # Phase 5: Try different resolutions
    cands = [400, 600, 500, 350, 700, 450]
    tried = [n_best]
    for nt in cands:
        if len(tried) >= 5:
            break
        if time.time() > deadline - 60.0:
            break
        if nt in tried or nt < 100 or nt > 1000 or nt % 2:
            continue
        tried.append(nt)
        ht = transfer_h(best_h, n_best, nt)
        cv = compute_c5(ht, nt)
        if cv < best_c:
            best_h, best_c, n_best = ht, cv, int(nt)
        chunk = min(deadline - 40.0, time.time() + 50.0)
        h1, c1 = slsqp_polish(best_h, n_best, [50000, 500000], 80, chunk)
        if c1 < best_c:
            best_h, best_c, n_best = h1, c1, int(n_best)
        h1, c1 = coord_cd(best_h, n_best, max_passes=3, deadline=chunk, grid=36, refine=16)
        if c1 < best_c:
            best_h, best_c, n_best = h1, c1, int(n_best)

    # Phase 6: Final polish
    tf1 = min(deadline - 15.0, time.time() + 60.0)
    if time.time() < tf1:
        h1, c1 = slsqp_polish(best_h, n_best, [100000, 1000000, 5000000], 200, tf1)
        if c1 < best_c:
            best_h, best_c = h1, c1
    tf2 = min(deadline - 3.0, time.time() + 30.0)
    if time.time() < tf2:
        h1, c1 = coord_cd(best_h, n_best, max_passes=5, deadline=tf2, grid=52, refine=24)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Final binary attempt
    if n_best % 2 == 0 and time.time() < deadline - 5.0:
        hb, cb = binary_optimize(best_h, n_best, rng, deadline)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb

    final_h = project_simplex(best_h, n_best)
    final_c = float(compute_c5(final_h, n_best))
    return final_h, final_c, int(n_best)
