from scipy.optimize import golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj1(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def compute_gradient(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    sw = weights.sum()
    if sw < 1e-30:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights /= sw
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w_val = weights[m_idx]
        if w_val < 1e-12:
            continue
        shift = m_idx - (n - 1)
        if shift >= 0:
            grad[:n - shift] += w_val * (1.0 - h[shift:])
            grad[shift:] -= w_val * h[:n - shift]
        else:
            s = -shift
            grad[s:] += w_val * (1.0 - h[:n - s])
            grad[:n - s] -= w_val * h[s:]
    return grad, m, c

            if time.time() > deadline:
