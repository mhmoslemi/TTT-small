Actually, I realize there's an even simpler approach. Since c_k(u) is convex in u (it's the max of convex quadratics), and we want to minimize it on an interval, the minimum is where the subgradient contains 0. For a max of functions, the subgradient at u is the convex hull of {f_k'(u) : f_k(u) = max}. 

A simple approach: use bisection on u, checking the sign of the "slope" of the max function. The slope of the max function at u is the max of {f_k'(u) : f_k is active} if the max is achieved uniquely, or the interval [min, max] of {f_k'(u) : f_k is active} otherwise.

For bisection: 
- If the minimum slope > 0, the function is increasing, so the minimum is to the left
- If the maximum slope < 0, the function is decreasing, so the minimum is to the right
- Otherwise, we're at the minimum (or very close)

Let me implement this:

```python
def solve_1d_pair(c, B, Q, off, k1, k2, lo_u, hi_u, iters=50):
    """Minimize max_k(c[k] + u*B[k] + u²*Q[k]) over u in [lo_u, hi_u].
    k1, k2 are the indices where Q is nonzero (or -1 if no Q term)."""
    a, b = lo_u, hi_u
    for _ in range(iters):
        if b - a < 1e-14:
            break
        m = 0.5 * (a + b)
        # Compute the max and its subgradient at m
        vals = c + m * B + m * m * Q
        mx = float(vals.max())
        # Find active set
        active = (vals >= mx - 1e-12)
        # Subgradient: convex hull of {B[k] + 2m*Q[k] for active k}
        sg_lo = float((B[active] + 2 * m * Q[active]).min())
        sg_hi = float((B[active] + 2 * m * Q[active]).max())
        if sg_lo > 0:
            # Function is increasing at m, go left
            b = m
        elif sg_hi < 0:
            # Function is decreasing at m, go right
            a = m
        else:
            # 0 is in subgradient, we're at the minimum
            break
    return 0.5 * (a + b)
```

This is clean and efficient. Each iteration is O(n) for the max and O(active) for the subgradient. With n=500, each pair evaluation is ~500 * 50 = 25K operations.

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-14 and z.max() <= 1.0 + 1e-14 and abs(z.sum() - target) <= 1e-10:
        return np.clip(z, 0.0, 1.0)
    lo, hi = float(z.min()) - 1.0, float(z.max()) + 1.0
    for _ in range(60):
        tau = 0.5 * (lo + hi)
        if np.clip(z - tau, 0.0, 1.0).sum() > target:
            lo = tau
        else:
            hi = tau
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - y.sum()
    if abs(res) > 1e-14:
        mask = (y < 1.0) if res > 0 else (y > 0.0)
        room = (1.0 - y) if res > 0 else y
        tot = room[mask].sum()
        if tot > 1e-15:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def smooth_val(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    return (m + float(np.log(e.sum())) / alpha) * (2.0 / n)

def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    p = e / e.sum()
    off = n - 1
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-13:
            continue
        d = k - off
        if d >= 0:
            g[d:] += p[k] * (1.0 - h[:n-d])
            g[:n-d] -= p[k] * h[d:]
        else:
            r = -d
            g[:n-r] += p[k] * (1.0 - h[r:])
            g[r:] -= p[k] * h[:n-r]
    return (2.0 / n) * g

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones(len(x))

def slsqp_opt(h, n, deadline, alphas=(30, 150, 600, 2500, 10000, 40000)):
    best_h, best_c = h.copy(), compute_c5(h, n)
    wh = best_h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    for a in alphas:
        if time.time() > deadline - 2.0:
            break
        try:
            mx = max(15, min(400, int(20000.0 / a)))
            res = minimize(smooth_val, wh, args=(n, a), method='SLSQP',
                          jac=smooth_grad, bounds=bounds, constraints=cons,
                          options={'maxiter': mx, 'ftol': 1e-15})
            x = project_feasible(res.x, n)
            c = compute_c5(x, n)
            if c < best_c:
                best_h, best_c = x, c
            wh = best_h
        except Exception:
            pass
    return best_h, best_c

def compute_B(h, i, j, n):
    off = n - 1
    B = np.zeros(2 * n - 1)
    for d in range(-(n-1), n):
        k = off + d
        gi = 0.0
        if 0 <= i + d < n:
            gi = 1.0 - h[i + d]
        if 0 <= i - d < n:
            gi -= h[i - d]
        gj = 0.0
        if 0 <= j + d < n:
            gj = 1.0 - h[j + d]
        if 0 <= j - d < n:
            gj -= h[j - d]
        B[k] = gi - gj
    return B

def solve_1d(c, B, Q, lo_u, hi_u):
    a, b = lo_u, hi_u
    if b - a < 1e-14:
        return 0.5 * (a + b)
    for _ in range(50):
        if b - a < 1e-14:
            break
        mid = 0.5 * (a + b)
        vals = c + mid * B + mid * mid * Q
        mx = float(vals.max())
        am = (vals >= mx - 1e-12)
        sg_lo = float((B[am] + 2.0 * mid * Q[am]).min())
        sg_hi = float((B[am] + 2.0 * mid * Q[am]).max())
        if sg_lo > 1e-13:
            b = mid
        elif sg_hi < -1e-13:
            a = mid
        else:
            break
    return 0.5 * (a + b)

def pair_search(h, n, max_iters, deadline, rng, tol_init=0.04):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(c.max())
    off = n - 1
    acc = 0
    stag = 0
    max_acc = min(800, max(200, max_iters // 4))
    for it in range(max_iters):
        if time.time() > deadline or acc >= max_acc:
            break
        tol = max(0.008, tol_init * cmax * (1.0 - 0.5 * it / max_iters))
        active = np.flatnonzero(c >= cmax - tol)
        if it % 9 == 8 or active.size <= 2:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
        else:
            k = int(active[int(rng.randint(active.size))])
            d = k - off
            if d >= 0:
                m = n - d
                b = int(rng.randint(m))
                i, j = b + d, b
            else:
                r = -d
                m = n - r
                b = r + int(rng.randint(m))
                i, j = b - r, b
        hi, hj = h[i], h[j]
        S = hi + hj
        lo_u = max(-hi, hj - 1.0)
        hi_u = min(1.0 - hi, hj)
        if hi_u - lo_u < 1e-13:
            stag += 1
            continue
        B = compute_B(h, i, j, n)
        Q = np.zeros(2 * n - 1)
        k1 = off + (j - i)
        k2 = off + (i - j)
        if 0 <= k1 < 2 * n - 1:
            Q[k1] = 1.0
        if 0 <= k2 < 2 * n - 1 and k2 != k1:
            Q[k2] = 1.0
        u = solve_1d(c, B, Q, lo_u, hi_u)
        if abs(u) < 1e-14:
            stag += 1
            continue
        nm = float((c + u * B + u * u * Q).max())
        if nm < cmax - 1e-12:
            h[i] = hi + u
            h[j] = hj - u
            c = c + u * B + u * u * Q
            cmax = nm
            acc += 1
            stag = 0
        else:
            stag += 1
        if acc > 0 and acc % 150 == 0:
            c = np.correlate(h, 1.0 - h, mode='full')
            cmax = float(c.max())
        if stag > max(300, max_iters // 3):
            break
    return project_feasible(h, n), compute_c5(h, n)

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def random_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, 10):
        a = rng.uniform(-0.5, 0.5)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += a * np.sin(0.5 * np.pi * m * x + ph)
        h += 0.5 * a * np.cos(np.pi * m * x + ph)
    h += 0.1 * (x / 2.0 - 0.5)
    return project_feasible(np.clip(h, 0.0, 1.0), n)

def try_symmetries(h, n):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for sym in [0.5*(h + h[::-1]), 1.0 - h, 0.5*(h + 1.0 - h[::-1]), h[::-1]]:
        sh = project_feasible(sym, n)
        sc = compute_c5(sh, n)
        if sc < best_c:
            best_h, best_c = sh, sc
    return best_h, best_c

def optimize_instance(h, n, deadline, rng, quick=False):
    h, c = try_symmetries(project_feasible(h, n), n)
    if not quick and time.time() < deadline - 20.0:

        h, c = slsqp_opt(h, n, min(deadline - 5.0, time.time() + 15.0), (50, 300, 2000))
    if time.time() < deadline - 3.0:
        rem = deadline - time.time()
        mi = int(min(60000 if not quick else 15000, max(3000, rem * 40)))
        h, c = pair_search(h, n, mi, deadline, rng)
    return project_feasible(h, n), compute_c5(h, n)
