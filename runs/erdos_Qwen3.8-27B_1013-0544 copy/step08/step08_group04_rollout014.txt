            lh = 1.0

            c[center] = qconst + best_x * (1.0 - best_x)

import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size:
                h[idx[0]] = min(1.0, h[idx[0]] + res)
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size:
                h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):

def resample_h(old, n):

    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

            i_idx = p - s_vals

def perturb_h(h, n, rng, mode):
    """Apply a perturbation to h."""
    h2 = h.copy()
    if mode == 0:
        h2 += rng.normal(0, 0.04, n)
    elif mode == 1:
        # Block flip
        width = rng.randint(n // 6, n // 3)
        start = rng.randint(0, n - width)
        h2[start:start+width] = 1.0 - h2[start:start+width]
    elif mode == 2:
        # Smooth bump
        c = rng.uniform(0.2, 1.8)
        w = rng.uniform(0.1, 0.5)
        amp = rng.uniform(-0.3, 0.3)
        x = (np.arange(n) + 0.5) * (2.0 / n)
        h2 += amp * np.exp(-0.5 * ((x - c) / w) ** 2)
    elif mode == 3:
        # Shift a block
        width = rng.randint(n // 5, n // 2)
        start = rng.randint(0, n - width)
        shift = rng.randint(1, max(2, n // 10))
        h2[start+shift:start+width+shift] = h2[start:start+width]
    return project_feasible(h2, n)
