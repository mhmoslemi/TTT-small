    """Compute gradient of (1/beta)*log(sum(exp(beta*c_k*dx))) with respect to h."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx

    weights = np.exp(beta * (c - max_c))

```python
def lse_gradient(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_scaled = c * dx
    max_c = c_scaled.max()
    w = np.exp(beta * (c_scaled - max_c))
    w /= w.sum()
    
    grad = np.zeros(n)
    for p in range(n):
        k_lo = n - 1 - p
        k_hi = 2 * n - 2 - p
        ks = np.arange(k_lo, k_hi + 1)
        j1 = p + ks - n + 1  # always in [0, n-1]
        j2 = p - ks + n - 1  # may be out of range
        valid2 = (j2 >= 0) & (j2 < n)
        term = (1.0 - h[j1])  # always valid
        term[valid2] += -h[j2[valid2]]
        grad[p] = np.sum(w[ks] * term) * dx
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_cd_fast(h, n, max_passes=200, deadline=None, grid_pts=40, gs_iters=40):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    D = (np.arange(2*n-1) == (n-1)).astype(float)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            # Build A vector
            A = np.zeros(2*n-1)
            k_lo = n - 1 - p
            k_hi = 2*n - 2 - p
            ks = np.arange(k_lo, k_hi + 1)
            j1 = p + ks - n + 1
            j2 = p - ks + n - 1
            valid2 = (j2 >= 0) & (j2 < n)
            A[ks] += (1.0 - h[j1])
            if np.any(valid2):
                A[ks[valid2]] += (-h[j2[valid2]])
            
            c0 = c - h0 * A + h0*h0 * D
            # 1D minimization: min_t max(c0 + t*A - t^2*D)
            grid = np.linspace(0.0, 1.0, grid_pts)
            fvals = np.empty(grid_pts)
            for i, t in enumerate(grid):
                fvals[i] = float(np.max(c0 + t*A - t*t*D))
            best_idx = int(np.argmin(fvals))
            lo = grid[max(0, best_idx-1)]
            hi = grid[min(grid_pts-1, best_idx+1)]
            if hi - lo < 1e-14:
                t_star = grid[best_idx]
            else:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = lo, hi
                c1 = b - gr*(b-a)
                d1 = a + gr*(b-a)
                fc1 = float(np.max(c0 + c1*A - c1*c1*D))
                fd1 = float(np.max(c0 + d1*A - d1*d1*D))
                for _ in range(gs_iters):
                    if fc1 < fd1:
                        b = d1; d1 = c1; fd1 = fc1
                        c1 = b - gr*(b-a)
                        fc1 = float(np.max(c0 + c1*A - c1*c1*D))
                    else:
                        a = c1; c1 = d1; fc1 = fd1
                        d1 = a + gr*(b-a)
                        fd1 = float(np.max(c0 + d1*A - d1*d1*D))
                t_star = (a+b)*0.5
            # Check boundaries
            f_star = float(np.max(c0 + t_star*A - t_star*t_star*D))
            f0 = float(np.max(c0))
            f1 = float(np.max(c0 + A - D))
            if min(f_star, f0, f1) < current_max - 1e-12:
                if f0 <= f_star and f0 <= f1:
                    t_new = 0.0
                elif f1 <= f_star and f1 <= f0:
                    t_new = 1.0
                else:
                    t_new = float(np.clip(t_star, 0.0, 1.0))
                c += (t_new - h0) * A - (t_new*t_new - h0*h0) * D
                h[p] = t_new
                current_max = float(np.max(c))
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def lse_pg(h, n, n_steps=100, deadline=None):
    """Log-sum-exp projected gradient descent."""
    h = h.copy()
    dx = 2.0 / n
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        beta = 20.0 + 50.0 * step / max(1, n_steps)
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        max_c = c.max()
        w = np.exp(beta * (c - max_c))
        w /= w.sum()
        # Compute gradient
        grad = np.zeros(n)
        for p in range(n):
            k_lo = n - 1 - p
            k_hi = 2*n - 2 - p
            ks = np.arange(k_lo, k_hi + 1)
            j1 = p + ks - n + 1
            j2 = p - ks + n - 1
            valid2 = (j2 >= 0) & (j2 < n)
            term = (1.0 - h[j1]).copy()
            if np.any(valid2):
                term[valid2] += -h[j2[valid2]]
            grad[p] = np.dot(w[ks], term)
        # Step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        lr = 0.5 / (beta * grad_norm)
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        c_old = compute_c5(h, n)
        if c_new < c_old:
            h = h_new
        else:
            # Try smaller step
            lr *= 0.3
            h_new = project_feasible(h - lr * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < c_old:
                h = h_new
    return h, compute_c5(h, n)

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_cd(h, n, max_passes=200, deadline=None, gp=40, gi=40):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    D = (np.arange(2 * n - 1) == (n - 1)).astype(float)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            A = np.zeros(2 * n - 1)
            k_lo = n - 1 - p
            k_hi = 2 * n - 2 - p
            ks = np.arange(k_lo, k_hi + 1)
            j1 = p + ks - n + 1
            j2 = p - ks + n - 1
            v2 = (j2 >= 0) & (j2 < n)
            A[ks] = (1.0 - h[j1])
            if np.any(v2):
                A[ks[v2]] += -h[j2[v2]]
            c0 = c - h0 * A + h0 * h0 * D
            grid = np.linspace(0.0, 1.0, gp)
            fvals = np.empty(gp)
            for i in range(gp):
                fvals[i] = np.max(c0 + grid[i] * A - grid[i] * grid[i] * D)
            bi = int(np.argmin(fvals))
            lo = grid[max(0, bi - 1)]
            hi = grid[min(gp - 1, bi + 1)]
            if hi - lo < 1e-14:
                ts = grid[bi]
            else:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = lo, hi
                c1 = b - gr * (b - a)
                d1 = a + gr * (b - a)
                fc1 = np.max(c0 + c1 * A - c1 * c1 * D)
                fd1 = np.max(c0 + d1 * A - d1 * d1 * D)
                for _ in range(gi):
                    if fc1 < fd1:
                        b, d1, fd1 = d1, c1, fc1
                        c1 = b - gr * (b - a)
                        fc1 = np.max(c0 + c1 * A - c1 * c1 * D)
                    else:
                        a, c1, fc1 = c1, d1, fd1
                        d1 = a + gr * (b - a)
                        fd1 = np.max(c0 + d1 * A - d1 * d1 * D)
                ts = (a + b) * 0.5
            fs = np.max(c0 + ts * A - ts * ts * D)
            f0 = np.max(c0)
            f1 = np.max(c0 + A - D)
            if min(fs, f0, f1) < current_max - 1e-12:
                if f0 <= fs and f0 <= f1:
                    tn = 0.0
                elif f1 <= fs and f1 <= f0:
                    tn = 1.0
                else:
                    tn = float(np.clip(ts, 0.0, 1.0))
                c += (tn - h0) * A - (tn * tn - h0 * h0) * D
                h[p] = tn
                current_max = float(np.max(c))
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def lse_pg(h, n, n_steps=60, deadline=None):
    h = h.copy()
    dx = 2.0 / n
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        beta = 30.0 + 80.0 * step / max(1, n_steps)
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        mc = c.max()
        w = np.exp(beta * (c - mc))
        w /= w.sum()
        grad = np.zeros(n)
        for p in range(n):
            k_lo = n - 1 - p
            k_hi = 2 * n - 2 - p
            ks = np.arange(k_lo, k_hi + 1)
            j1 = p + ks - n + 1
            j2 = p - ks + n - 1
            v2 = (j2 >= 0) & (j2 < n)
            term = (1.0 - h[j1]).copy()
            if np.any(v2):
                term[v2] += -h[j2[v2]]
            grad[p] = np.dot(w[ks], term)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        lr = 0.4 / (beta * gn)
        for _ in range(8):
            hn = project_feasible(h - lr * grad, n)
            cn = compute_c5(hn, n)
            co = compute_c5(h, n)
            if cn < co:
                h = hn
                break
            lr *= 0.4
    return h, compute_c5(h, n)

    xn = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(xn, xo, h_old), n_new)
