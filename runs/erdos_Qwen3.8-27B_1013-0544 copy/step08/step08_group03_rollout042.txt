def grad_weights(w, h, n):
    h = np.asarray(h, dtype=float)
    m = 2*n - 1
    c = n - 1
    g = np.zeros(n, dtype=float)
    wc = w[c]
    if wc:
        g += wc * (1.0 - 2.0 * h)
    if n > 1:
        wP = np.zeros(n)
        wN = np.zeros(n)
        wP[1:] = w[c+1:m]
        wN[1:] = w[:c][::-1]
        x1 = 1.0 - h
        L1 = np.convolve(x1, wP, mode='full')[:n]
        L2 = np.convolve(h, wN, mode='full')[:n]
        R1 = np.convolve(x1[::-1], wN, mode='full')[:n][::-1]
        R2 = np.convolve(h[::-1], wP, mode='full')[:n][::-1]
        g = g + L1 - L2 + R1 - R2
    return (2.0 / n) * g

import numpy as np
from scipy.optimize import minimize
import time
import math

def compute_c(h, n):
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def grad_weights(w, h, n):
    h = np.asarray(h, dtype=float)
    m = 2 * n - 1
    c = n - 1
    g = np.zeros(n, dtype=float)
    wc = w[c]
    if wc:
        g += wc * (1.0 - 2.0 * h)
    if n > 1:
        wP = np.zeros(n)
        wN = np.zeros(n)
        wP[1:] = w[c + 1:m]
        wN[1:] = w[:c][::-1]
        x1 = 1.0 - h
        L1 = np.convolve(x1, wP, mode='full')[:n]
        L2 = np.convolve(h, wN, mode='full')[:n]
        R1 = np.convolve(x1[::-1], wN, mode='full')[:n][::-1]
        R2 = np.convolve(h[::-1], wP, mode='full')[:n][::-1]
        g = g + L1 - L2 + R1 - R2
    return (2.0 / n) * g

def _sum_fun_jac(x, beta, M, n):
    h = np.asarray(x, dtype=float)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    cm = float(c.max())
    if beta < 1e-3:
        act = (c >= cm - 1e-9).astype(float)
        tot = act.sum()
        if tot <= 0:
            act = np.ones_like(c)
            tot = act.sum()
        w = act / tot
        val = cm
    else:
        z = beta * (c - cm)
        e = np.exp(z)
        s = float(e.sum())
        if s <= 1e-300 or not math.isfinite(s):
            w = np.ones_like(c) / c.size
            val = cm
        else:
            w = e / s
            val = cm + math.log(s) / beta
    g = grad_weights(w, h, n)
    dev = float(h.sum()) - 0.5 * n
    val = val + M * dev * dev
    g = g + 2.0 * M * dev
    return float(val), g

def project_box_sum(h, n):
    target = 0.5 * n
    hp = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(80):
        diff = target - hp.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            m = hp < 1.0
            if not m.any():
                break
            add = diff / m.sum()
            hp[m] = np.minimum(1.0, hp[m] + add)
        else:
            m = hp > 0.0
            if not m.any():
                break
            add = -diff / m.sum()
            hp[m] = np.maximum(0.0, hp[m] - add)
    return hp

def make_bounds(n):
    return [(0.0, 1.0)] * n

def resample_h(h, n_old, n_new):
    if n_old == n_new:
        return project_box_sum(h, n_new)
    xo = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(xn, xo, np.asarray(h, dtype=float))
    return project_box_sum(y, n_new)

def global_start(n, rng, kind):
    if kind == 0:
        t = np.zeros(n, dtype=np.uint8)
        for i in range(1, n):
            t[i] = t[i // 2] ^ (i & 1)
        hp = 1.0 - t.astype(float)
        return project_box_sum(hp, n)
    if kind == 1:
        x = (np.arange(n) + 0.5) * (2.0 / n)
        alpha = math.sqrt(2.0) - 1.0
        hp = ((x * alpha) % 1.0 < 0.5).astype(float)
        hp += 0.02 * (rng.rand(n) - 0.5)
        return project_box_sum(hp, n)
    if kind == 2:
        return project_box_sum(rng.rand(n), n)
    if kind == 3:
        x = (np.arange(n) + 0.5) * (2.0 / n)
        hp = 0.5 + 0.42 * np.sin(3.0 * np.pi * x) + 0.22 * np.sin(7.0 * np.pi * x + 0.8)
        hp += 0.05 * rng.normal(0.0, 1.0, n)
        return project_box_sum(hp, n)
    hp = rng.rand(n)
    q = np.partition(hp, n // 2)[n // 2]
    hp = (hp >= q).astype(float)
    return project_box_sum(hp, n)

def make_initial(n, seed):
    return global_start(n, np.random.RandomState(seed), 3)

def perturb_h(h, n, rng, kind):
    if kind == 0:
        return project_box_sum(h + 0.03 * rng.normal(0.0, 1.0, n), n)
    if kind == 1:
        return project_box_sum(h + 0.10 * rng.normal(0.0, 1.0, n), n)
    if kind == 2:
        return global_start(n, rng, 2)
    if kind == 3:
        return global_start(n, rng, 3)
    if kind == 4:
        return global_start(n, rng, 0)
    if kind == 5:
        return global_start(n, rng, 4)
    return project_box_sum(h + 0.05 * rng.normal(0.0, 1.0, n), n)

def coord_cd(h, n, max_passes, deadline=None, sum_penalty=0.01, grid1=13, grid2=25):
    h = project_box_sum(h, n).copy()
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cur = float(c.max())
    M = 2 * n - 1
    center = n - 1
    target = 0.5 * n
    gridA = np.linspace(0.0, 1.0, grid1)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        h_pass = h.copy()
        c_pass = c.copy()
        cur_pass = cur
        cur_sum = float(h.sum())
        any_imp = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = float(h[p])
            A = np.zeros(M)
            left = p
            if left:
                hL = h[p - 1::-1]
                A[n:n + left] += dx * (1.0 - hL)
                A[n - 2:n - left - 2:-1] -= dx * hL
            right = n - 1 - p
            if right:
                hR = h[p + 1:n]
                A[n:n + right] -= dx * hR
                A[n - 2:n - right - 2:-1] += dx * (1.0 - hR)
            c_work = c.copy()
            c_work[center] = -1e100
            shifts = gridA - h0
            m = (c_work[None, :] + shifts[:, None] * A[None, :]).max(axis=1)
            cent = c[center] + dx * (shifts - (gridA * gridA - h0 * h0))
            hard = np.maximum(m, cent)
            f = hard
            if sum_penalty:
                dev = cur_sum - target + shifts
                f = np.maximum(f, sum_penalty * dev * dev)
            i = int(np.argmin(f))
            best_t = float(gridA[i])
            if 0 < i < gridA.size - 1:
                lo = gridA[i - 1]
                hi = gridA[i + 1]
                gridB = np.linspace(lo, hi, grid2)
                shifts = gridB - h0
                m = (c_work[None, :] + shifts[:, None] * A[None, :]).max(axis=1)
                cent = c[center] + dx * (shifts - (gridB * gridB - h0 * h0))
                hard = np.maximum(m, cent)
                if sum_penalty:
                    dev = cur_sum - target + shifts
                    f = np.maximum(hard, sum_penalty * dev * dev)
                j = int(np.argmin(f))
                if f[j] < f[i]:
                    best_t = float(gridB[j])
            if abs(best_t - h0) > 1e-14:
                shift = best_t - h0
                c += shift * A
                c[center] += dx * (shift - (best_t * best_t - h0 * h0))
                new_max = float(c.max())
                if new_max < cur - 1e-13:
                    h[p] = best_t
                    cur = new_max
                    cur_sum += shift
                    any_imp = True
                else:
                    c[center] -= dx * (shift - (best_t * best_t - h0 * h0))
                    c -= shift * A
        if not any_imp:
            break
        h_proj = project_box_sum(h, n)
        c_proj = np.correlate(h_proj, 1.0 - h_proj, mode='full') * dx
        new_cur = float(c_proj.max())
        if new_cur < cur_pass - 1e-12:
            h, c, cur = h_proj, c_proj, new_cur
        else:
            h, c, cur = h_pass, c_pass, cur_pass
    h = project_box_sum(h, n)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return h, float(c.max())

def get_base_iter(n):
    if n <= 192:
        return 700
    if n <= 384:
        return 450
    if n <= 640:
        return 320
    return 220

def get_schedule(mode, n):
    b = get_base_iter(n)
    if mode == 0:
        return [(8, 1e4, b), (30, 1e4, b), (120, 1e5, b), (500, 1e5, int(0.9 * b)), (2500, 1e6, int(0.7 * b))]
    if mode == 1:
        return [(20, 1e4, int(0.9 * b)), (150, 1e5, int(0.9 * b)), (800, 1e6, int(0.7 * b))]
    return [(50, 1e4, int(0.6 * b)), (400, 1e5, int(0.5 * b))]

def optimize_n(h, n, deadline, mode):
    h = project_box_sum(h, n)
    c0 = compute_c(h, n)
    if mode <= 1 and n <= 800 and time.time() < deadline - 30:
        h, c0 = coord_cd(h, n, 5, min(deadline, time.time() + 12), 0.01, 9, 15)
    sched = get_schedule(mode, n)
    bounds = make_bounds(n)
    for beta, M, mi in sched:
        now = time.time()
        if now > deadline - 12:
            break
        mi = int(min(mi, max(12, int(deadline - now))))
        if mi < 12:
            break
        try:
            res = minimize(_sum_fun_jac, h, args=(beta, M, n), method='L-BFGS-B', jac=True,
                           bounds=bounds, options={'maxiter': mi, 'maxfun': 2 * mi,
                                                   'ftol': 1e-12, 'gtol': 1e-7,
                                                   'maxls': 20, 'disp': False})
            hn = np.asarray(res.x, dtype=float)
        except Exception:
            continue
        hn = project_box_sum(hn, n)
        cn = compute_c(hn, n)
        if cn < c0 - 1e-12:
            h, c0 = hn, cn
    if mode == 0:
        cd_passes = 60 if n <= 500 else 32
    elif mode == 1:
        cd_passes = 26 if n <= 600 else 14
    else:
        cd_passes = 8
    if time.time() < deadline - 5:
        h, c0 = coord_cd(h, n, cd_passes, deadline - 2, 0.01, 13, 25)
    h = project_box_sum(h, n)
    return h, compute_c(h, n)
