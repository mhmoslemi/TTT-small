import numpy as np

from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 1e-15:
            h += (diff / tot) * cap
    elif diff < 0:
        tot = float(h.sum())
        if tot > 1e-15:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-12:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-12:
        idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def c_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c_max(h, n):
    return float(np.max(c_full(h)) * (2.0 / n))

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if diff > 1e-15:
        cap = 1.0 - h
        t = float(cap.sum())
        if t > 1e-15:
            h += (diff / t) * cap
    elif diff < -1e-15:
        t = float(h.sum())
        if t > 1e-15:
            h += (diff / t) * h
    h = np.clip(h, 0.0, 1.0)

            T = high

            cands = [0.0, 1.0, hp, 0.5]
            if T >= low - tol:
                ll = 0.0
                lh = 1.0
                for jj in range(seg):
                    if quad[jj] > 0:
                        disc = acoef[jj]**2 - 4.0 * (T - qconst[jj])
                        if disc >= 0:
                            r1 = (acoef[jj] - np.sqrt(disc)) / 2.0
                            r2 = (acaoef[jj] + np.sqrt(disc)) / 2.0
                            if 0 <= r1 <= 1:
                                cands.extend([max(0,r1), min(1,r2)])
                            elif r2 <= 1 and r2 >= 0:
                                ll = max(ll, r2)
                            if r1 <= 1 and r1 >= 0:
                                lh = min(lh, r1)
                    else:
                        if acoef[jj] > 1e-14:
                            lh = min(lh, (T - qconst[jj]) / acoef[jj])
                        elif acoef[jj] < -1e-14:
                            ll = max(ll, (T - qconst[jj]) / acoef[jj])
                ll = max(0.0, ll)
                lh = min(1.0, lh)
                if ll <= lh + tol:
                    cands.extend([ll, lh, 0.5*(ll+lh)])
            best_x = hp
            best_f = max(unaff, float(np.max(qconst + acoef*hp - quad*hp*hp)))
            for x in cands:
                x = float(np.clip(x, 0.0, 1.0))
                f = max(unaff, float(np.max(qconst + acoef*x - quad*x*x)))
                if f < best_f - 1e-12:
                    best_f = f
                    best_x = x
            if best_f < cur - 1e-11:
                dx = best_x - hp
                # Update c for affected lags
                for jj in range(seg):
                    k = lag_lo + jj
                    c[k] = qconst[jj] + acoef[jj]*best_x - quad[jj]*best_x*best_x
                h[p] = best_x
                cur = best_f
                improved = True
            if (p + 1) % project_every == 0:
                h = project_feasible(h, n)
                c = c_full(h)
                cur = float(c.max())
        h = project_feasible(h, n)
        c = c_full(h)
        cur = float(c.max())
        cv = (2.0/n) * cur
        if cv < best_c - 1e-14:
            best_c = cv
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def dual_refine(h, n, deadline, n_outer=30, n_inner=8):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = c_max(h, n)
    L = 2*n - 1
    c = c_full(h)
    beta = 30.0
    for outer in range(n_outer):
        if time.time() > deadline - 30:
            break
        z = beta * c / (2.0/n)
        z -= z.max()
        p = np.exp(z)
        p /= p.sum()
        for inner in range(n_inner):
            if time.time() > deadline - 30:
                break
            g = wgrad(h, n, p)
            lr = 0.02 / (1.0 + inner * 0.3)
            h_new = h - lr * g
            h_new = project_feasible(h_new, n)
            c_new = c_full(h_new)
            obj_new = float(p @ c_new)
            obj_old = float(p @ c)
            if obj_new < obj_old:
                h = h_new
                c = c_new
            else:
                lr *= 0.5
                h_new = h - lr * g
                h_new = project_feasible(h_new, n)
                c_new = c_full(h_new)
                if float(p @ c_new) < obj_old:
                    h = h_new
                    c = c_new
        cv = (2.0/n) * float(c.max())
        if cv < best_c:
            best_c = cv
            best_h = h.copy()
        beta *= 1.15
    return best_h, best_c

def symmetrize(h):
    n = len(h)
    return project_feasible(0.5*(h + h[::-1]), n)

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0)+0.5)*(2.0/n0)
    x1 = (np.arange(n)+0.5)*(2.0/n)
    return project_feasible(np.interp(x1, x0, old), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 12.0
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n0 = 500
        x = (np.arange(n0)+0.5)*(2.0/n0)
        h_init = project_feasible(1.0 - np.abs(x - 1.0), n0)
    n_init = len(h_init)
    if n_init % 2:
        h_init = resample_h(h_init, n_init+1)
        n_init += 1
    best_h = project_feasible(h_init, n_init)
    best_c = c_max(best_h, n_init)

    # Phase 1: SLSQP on initial
    if time.time() < deadline - 200:
        h1, c1 = slsqp_opt(best_h, n_init, (30, 300, 3000), 60, deadline)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Coordinate descent
    if time.time() < deadline - 100:
        h2, c2 = cd_max(best_h, n_init, 6, deadline, project_every=8)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Symmetric optimization
    if time.time() < deadline - 150:
        hs = symmetrize(best_h)
        cs = c_max(hs, n_init)
        hs, cs = cd_max(hs, n_init, 5, deadline, project_every=8)
        if cs < best_c:
            best_h, best_c = hs, cs

    # Phase 4: Dual refinement
    if time.time() < deadline - 100:
        h4, c4 = dual_refine(best_h, n_init, deadline, n_outer=20, n_inner=6)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Multi-scale
    for nt in [n_init, 400, 600, 300, 450, 800]:
        if time.time() > deadline - 120:
            break
        if best_c <= 0.38075:
            break
        if nt == n_init:
            continue
        h0 = resample_h(best_h, nt)
        h1, c1 = slsqp_opt(h0, nt, (50, 500), 40, deadline)
        h2, c2 = cd_max(h1, nt, 4, deadline, project_every=10)
        if c2 < best_c:
            best_h, best_c = h2, nt if len(h2)==nt else best_c
            best_h = h2
        h3, c3 = dual_refine(best_h, len(best_h), deadline, n_outer=10, n_inner=5)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 6: Random restarts
    for r in range(20):
        if time.time() > deadline - 80:
            break
        if best_c <= 0.38070:
            break
        nt = len(best_h)
        base = resample_h(best_h, nt)
        typ = r % 4
        if typ == 0:
            h0 = project_feasible(base + rng.normal(0, 0.015, nt), nt)
        elif typ == 1:
            h0 = project_feasible(base + rng.normal(0, 0.05, nt), nt)
        elif typ == 2:
            h0 = symmetrize(base)
            h0 = project_feasible(h0 + rng.normal(0, 0.02, nt), nt)
        else:
            x = (np.arange(nt)+0.5)*(2.0/nt)
            t = x - 1.0
            h0 = 0.5 + 0.3*np.cos(2*np.pi*t) + 0.1*np.cos(4*np.pi*t) + rng.normal(0,0.05,nt)
            h0 = project_feasible(h0, nt)
        h1, c1 = cd_max(h0, nt, 3, deadline, project_every=12)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h2, c2 = dual_refine(h1, nt, deadline, n_outer=8, n_inner=4)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 7: Final polish
    if time.time() < deadline - 20:
        h7, c7 = cd_max(best_h, len(best_h), 4, deadline, project_every=6)
        if c7 < best_c:
            best_h, best_c = h7, c7
    if time.time() < deadline - 10:
        h8, c8 = slsqp_opt(best_h, len(best_h), (200, 2000), 50, deadline)
        if c8 < best_c:
            best_h, best_c = h8, c8

    best_h = project_feasible(best_h, len(best_h))
    best_c = c_max(best_h, len(best_h))
    return best_h, best_c, len(best_h)
