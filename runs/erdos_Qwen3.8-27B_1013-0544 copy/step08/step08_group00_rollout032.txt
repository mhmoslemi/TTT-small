import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        active = (h > 0) & (h < 1)
        if not np.any(active):
            break
        if diff > 0:
            room = 1.0 - h
            tot = room[active].sum()
            if tot < 1e-15:
                break
            h[active] = np.clip(h[active] + diff / tot * room[active], 0.0, 1.0)
        else:
            tot = h[active].sum()
            if tot < 1e-15:
                break
            h[active] = np.clip(h[active] + diff / tot * h[active], 0.0, 1.0)
    return h

def solve_1d(h, c, p, n):
    """Exact 1D subproblem: find x minimizing max_m c'[m] subject to 0<=x<=1."""
    h0 = h[p]
    # Compute coefficients A[m] for m in relevant range
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)  # first index position
    jb = p - ms + (n - 1)  # second index position
    valid_f = (jf >= 0) & (jf < n)
    valid_b = (jb >= 0) & (jb < n)
    
    A = np.zeros(len(ms))
    A[valid_f] += (1.0 - h[jf[valid_f]])
    A[valid_b] -= h[jb[valid_b]]
    
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    
    # Unaffected max
    unaff = 0.0
    if m_lo > 0:
        unaff = max(unaff, float(c[:m_lo].max()))
    if m_hi < 2 * n - 2:
        unaff = max(unaff, float(c[m_hi + 1:].max()))
    
    cur_max = float(c.max())
    
    # Bisection on L
    L_lo = unaff
    L_hi = cur_max
    
    for _ in range(60):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        
        # Check feasibility: exists x in [0,1] s.t. all constraints satisfied
        x_lo = 0.0
        x_hi = 1.0
        feasible = True
        
        # Linear constraints (m ≠ n-1)
        for idx in range(len(ms)):
            m = ms[idx]
            if has_diag and m == (n - 1):
                continue
            a = A[idx]
            cb = c_base[idx]
            if a > 1e-14:
                # cb + (x-h0)*a <= L => x <= (L-cb)/a + h0
                x_hi = min(x_hi, (L - cb) / a + h0)
            elif a < -1e-14:
                # cb + (x-h0)*a <= L => x >= (L-cb)/a + h0
                x_lo = max(x_lo, (L - cb) / a + h0)

            L_lo = L

            # x^2 - x + (L - cb_d + h0 - h0^2) >= 0

                    # Both are non-empty since x_lo < r1 and x_hi > r2
                    # So feasible (we'll pick the better one later)
                    pass
                elif x_lo >= r1 and x_hi <= r2:
                    feasible = False
                # Other cases: the constraint is automatically satisfied
        
        if feasible:
            L_hi = L
        else:
            L_lo = L
    
    L = L_hi
    
    # Now find the best x in [0,1] satisfying all constraints
    x_lo = max(0.0, 0.0)
    x_hi = min(1.0, 1.0)
    
    for idx in range(len(ms)):

        if has_diag and m == (n - 1):

        cb = c_base[idx]
        if a > 1e-14:

        elif a < -1e-14:
            x_lo = max(x_lo, (L - cb) / a + h0)

    x_hi = min(x_hi, 1.0)

        cb_d = c_base[di]

                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:

                # Pick closest to h0
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    
    if x_lo > x_hi:
        return h0
    # Pick midpoint for stability (or the value closest to h0)
    if abs(h0 - x_lo) <= abs(x_hi - h0):
        return float(np.clip(h0 if h0 >= x_lo and h0 <= x_hi else x_lo, 0.0, 1.0))
    else:
        return float(np.clip(h0 if h0 >= x_lo and h0 <= x_hi else x_hi, 0.0, 1.0))

def coord_cd_exact(h, n, max_passes, deadline, order='seq'):
    h = h.copy()
    n_pass = 0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        if order == 'random':
            perm = np.random.permutation(n)
        else:
            perm = range(n)
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in perm:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            x_new = solve_1d(h, c, p, n)
            if abs(x_new - h0) > 1e-10:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        n_pass += 1
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_run(h, n, alphas, maxiter_each=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        def sum_con(x_, n_):
            return x_.sum() - n_ / 2.0
        def sum_con_jac(x_, n_):
            return np.ones(n_)
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            if res.success or res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def solve_1d_vectorized(h, c, p, n):
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    valid_f = (jf >= 0) & (jf < n)
    valid_b = (jb >= 0) & (jb < n)
    
    A = np.zeros(len(ms))
    A[valid_f] += (1.0 - h[jf[valid_f]])
    A[valid_b] -= h[jb[valid_b]]
    
    c_base = c[ms].copy()
    diag_mask = (ms == (n - 1))
    has_diag = np.any(diag_mask)
    
    unaff = 0.0
    if m_lo > 0:
        unaff = max(unaff, float(c[:m_lo].max()))
    if m_hi < 2 * n - 2:
        unaff = max(unaff, float(c[m_hi + 1:].max()))
    
    cur_max = float(c.max())
    
    # Separate linear and quadratic constraints
    lin_mask = ~diag_mask
    
    # For linear constraints: c_base[i] + (x-h0)*A[i] <= L
    # If A[i] > 0: x <= (L - c_base[i])/A[i] + h0
    # If A[i] < 0: x >= (L - c_base[i])/A[i] + h0

    neg_mask = lin_mask & (A < -1e-14)
    zero_mask = lin_mask & (np.abs(A) <= 1e-14)

    L_hi = cur_max

        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        
        x_lo = 0.0
        x_hi = 1.0
        feasible = True
        
        if np.any(zero_mask):
            if np.any(c_base[zero_mask] > L):
                feasible = False
        
        if feasible and np.any(neg_mask):
            bounds_lo = (L - c_base[neg_mask]) / A[neg_mask] + h0
            x_lo = max(x_lo, float(np.max(bounds_lo)))

            cb_d = c_base[di]

            disc = 1.0 - 4.0 * C_val
            if disc >= 0:

                # Infeasible if [x_lo, x_hi] ⊆ (r1, r2)
                if x_lo >= r1 and x_hi <= r2:

        else:
            L_lo = L

    x_hi = 1.0

        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo < r1 and x_hi > r2:
                if h0 - max(x_lo, 0.0) <= min(x_hi, 1.0) - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    
    if x_lo > x_hi:
        return h0
    # Return the value in [x_lo, x_hi] closest to h0
    if x_lo <= h0 <= x_hi:
        return h0
    elif h0 < x_lo:
        return x_lo
    else:
        return x_hi

    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
