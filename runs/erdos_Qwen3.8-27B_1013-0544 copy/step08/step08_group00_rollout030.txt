```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=600, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

def power_obj(x, n, p):
    c = np.correlate(x, 1.0 - x, mode='full')
    return (np.sum(c.astype(float)**p) ** (1.0/p)) * (2.0/n)

def smooth_obj(x, n, alpha):

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    grad = np.zeros(n)

    if n_active == 0:
        return h
    for m in np.where(active)[0]:
        offset = m - (n - 1)
        # Forward: h[p]*(1-h[p+offset])

        if np.any(valid_b):

    grad /= n_active
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def _solve_1d(c, h, p, n):
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi_f = min(2 * n - 2, n - 1 + p)
    m_lo_b = max(0, p)
    m_hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(m_lo, m_lo_b)
    m_hi = max(m_hi_f, m_hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    dpos = np.where(ms == (n - 1))[0]
    has_d = len(dpos) > 0
    if has_d:
        c_diag = c_base[dpos[0]]
        a[dpos] = 0.0
    unaff = 0.0
    if m_lo > 0:
        unaff = max(unaff, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff = max(unaff, float(c[m_hi + 1:].max()))
    cur_max = float(np.max(c))
    L_lo, L_hi = unaff, cur_max
    nm = a < -1e-15
    pm = a > 1e-15
    zm = np.abs(a) <= 1e-15
    for _ in range(55):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        xlo, xhi = 0.0, 1.0
        ok = True
        if np.any(zm) and np.any(c_base[zm] > L):
            ok = False
        if ok and np.any(nm):
            xlo = max(xlo, float(np.max((L - c_base[nm]) / a[nm])))
        if ok and np.any(pm):
            xhi = min(xhi, float(np.min((L - c_base[pm]) / a[pm])))
        if xlo > xhi + 1e-15:
            ok = False
        if ok and has_d:
            A = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * A
            if disc < 0:
                pass
            else:
                sq = np.sqrt(disc)
                r1 = (1.0 - sq) / 2.0
                r2 = (1.0 + sq) / 2.0
                xc_lo, xc_hi = max(xlo, 0.0), min(xhi, 1.0)
                if xc_lo >= r1 and xc_hi <= r2:
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    xlo, xhi = 0.0, 1.0
    if np.any(nm):
        xlo = max(xlo, float(np.max((L - c_base[nm]) / a[nm])))
    if np.any(pm):
        xhi = min(xhi, float(np.min((L - c_base[pm]) / a[pm])))
    xlo, xhi = max(xlo, 0.0), min(xhi, 1.0)
    if has_d:
        A = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * A

            sq = np.sqrt(disc)
            r1 = (1.0 - sq) / 2.0
            r2 = (1.0 + sq) / 2.0
            if xlo < r1 and xhi > r2:

                    xlo = max(xlo, r2)

    xlo, xhi = max(xlo, 0.0), min(xhi, 1.0)
    if xlo > xhi:
        return h0
    return float(np.clip((xlo + xhi) / 2.0, 0.0, 1.0))

def cd_exact(h, n, max_passes, deadline, order=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if order is None:
            idx = range(n)
        elif order == 'rev':
            idx = range(n - 1, -1, -1)
        else:
            idx = order
        for p in idx:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            x_new = _solve_1d(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < best_c:
            best_c = nc
    return h, best_c

def subgrad_step(h, n, step):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    cx = c.max()
    active = np.where(c >= cx * 0.98)[0]
    if len(active) == 0:
        return h.copy()
    grad = np.zeros(n)
    for m in active:
        off = m - (n - 1)
        jf = np.arange(n) + off
        vf = (jf >= 0) & (jf < n)
        if np.any(vf):
            grad[vf] += (1.0 - h[jf[vf]])
        jb = np.arange(n) - off
        vb = (jb >= 0) & (jb < n)
        if np.any(vb):
            grad[np.arange(n)[vb]] -= h[np.arange(n)[vb]]
    grad /= len(active)
    return project_feasible(h - step * grad, n)

def try_n_transfer(best_h, best_c, n_try, deadline, rng):
    n_src = len(best_h)
    if n_try == n_src or n_try < 100 or n_try > 2000:
        return best_h, best_c
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xn = np.linspace(0, 2, n_try, endpoint=False)
    h_int = np.interp(xn, xs, best_h)
    h_int = project_feasible(h_int, n_try)
    h_opt, c_opt = cd_exact(h_int, n_try, 40, deadline)
    if c_opt < best_c:
        return h_opt, c_opt
    return best_h, best_c
```
