import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_idx(n):
    cache = {}
    for p in range(n):
        ks = np.arange(max(0, p), min(2*n-2, p+n-1)+1)
        iv = p + (n-1) - ks
        valid = (iv >= 0) & (iv < n)
        diag = valid & (iv == p)
        nd = valid & ~diag
        ks2 = np.arange(max(0, n-1-p), min(2*n-2, 2*n-2-p)+1)
        jv = ks2 - (n-1) + p
        valid2 = (jv >= 0) & (jv < n)
        nd2 = valid2 & (jv != (n-1-p))
        cache[p] = (ks, iv, nd, ks2, jv, nd2)
    return cache

def solve_1d(c0, A, D, n_grid=80, n_gs=55):
    grid = np.linspace(0.0, 1.0, n_grid)
    fvals = np.array([np.max(c0 + t*A - t*t*D) for t in grid])
    bi = int(np.argmin(fvals))
    lo = grid[max(0, bi-1)]
    hi = grid[min(n_grid-1, bi+1)]
    if hi - lo < 1e-15:
        return grid[bi]
    gr = (np.sqrt(5.0)-1.0)/2.0
    a, b = lo, hi
    x1 = b - gr*(b-a); x2 = a + gr*(b-a)
    f1 = np.max(c0 + x1*A - x1*x1*D)
    f2 = np.max(c0 + x2*A - x2*x2*D)
    for _ in range(n_gs):
        if f1 < f2:
            b, x2, f2 = x2, x1, f1
            x1 = b - gr*(b-a)
            f1 = np.max(c0 + x1*A - x1*x1*D)
        else:
            a, x1, f1 = x1, x2, f2
            x2 = a + gr*(b-a)
            f2 = np.max(c0 + x2*A - x2*x2*D)
    return (a+b)*0.5

def cd_pass(h, c, idx, n, order, grid_pts, gs_iters, deadline):
    D = (np.arange(2*n-1) == n-1).astype(float)
    improved = False
    for p in order:
        if deadline is not None and time.time() > deadline:
            break
        h0 = h[p]
        if h0 <= 1e-14 or h0 >= 1.0-1e-14:
            continue
        ks, iv, nd, ks2, jv, nd2 = idx[p]
        A = np.zeros(2*n-1)
        if nd.any():
            A[ks[nd]] += (1.0 - h[iv[nd]])
        A[n-1] += 1.0
        if nd2.any():
            A[ks2[nd2]] += (-h[jv[nd2]])
        c0 = c - h0*A + h0*h0*D
        t = solve_1d(c0, A, D, grid_pts, gs_iters)
        fs = np.max(c0 + t*A - t*t*D)
        f0 = np.max(c0)
        f1 = np.max(c0 + A - D)
        cur_max = np.max(c)
        if min(fs, f0, f1) < cur_max - 1e-13:
            if f0 <= fs and f0 <= f1:
                tn = 0.0
            elif f1 <= fs and f1 <= f0:
                tn = 1.0
            else:
                tn = max(0.0, min(1.0, t))
            c += (tn-h0)*A - (tn*tn-h0*h0)*D
            h[p] = tn
            improved = True
    return h, c, improved

def cd_multi(h, n, max_passes, deadline, grid_pts=80, gs_iters=55, seed=42):
    h = h.copy()
    idx = build_idx(n)
    c = np.correlate(h, 1.0-h, mode='full').astype(float)
    rng = np.random.RandomState(seed)
    orders = [list(range(n)), list(range(n-1, -1, -1))]
    for _ in range(8):
        o = list(range(n))
        rng.shuffle(o)
        orders.append(o)
    stall_count = 0
    for pi in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = orders[pi % len(orders)]
        h, c, imp = cd_pass(h, c, idx, n, order, grid_pts, gs_iters, deadline)
        h = project_feasible(h, n)
        if not imp:
            stall_count += 1
            if stall_count >= 3:
                break
        else:
            stall_count = 0
    return h, compute_c5(h, n)

def pnorm_opt(h, n, p, maxiter=400, deadline=None):
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full')
        return float(np.sum(c**p) ** (1.0/p)) * (2.0/n)
    def sfun(x):
        return x.sum() - n/2.0
    def sjac(x):
        return np.ones(n)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0,1.0)]*n,
                   constraints=[{'type':'eq','fun':sfun,'jac':sjac}],
                   options={'maxiter':maxiter,'ftol':1e-16})
    return project_feasible(res.x, n)

def interp_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old))
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def active_set_cd(h, n, deadline, grid_pts=100, gs_iters=60, n_active=20):
    """Focus on coordinates that most affect the top lags."""
    h = h.copy()
    idx = build_idx(n)
    c = np.correlate(h, 1.0-h, mode='full').astype(float)
    D = (np.arange(2*n-1) == n-1).astype(float)
    for _ in range(200):
        if deadline is not None and time.time() > deadline:
            break
        cmax = np.max(c)
        active_lags = np.where(c > cmax - 1e-6*cmax)[0]
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0-1e-14:
                continue
            ks, iv, nd, ks2, jv, nd2 = idx[p]
            A = np.zeros(2*n-1)
            if nd.any():
                A[ks[nd]] += (1.0 - h[iv[nd]])
            A[n-1] += 1.0
            if nd2.any():
                A[ks2[nd2]] += (-h[jv[nd2]])
            c0 = c - h0*A + h0*h0*D
            # Only consider active lags for the max
            c0_a = c0[active_lags]
            A_a = A[active_lags]
            D_a = D[active_lags]
            t = solve_1d(c0_a, A_a, D_a, grid_pts, gs_iters)
            # Verify against full
            fs_full = np.max(c0 + t*A - t*t*D)
            f0 = np.max(c0)
            f1 = np.max(c0 + A - D)
            if min(fs_full, f0, f1) < cmax - 1e-13:

                    tn = 0.0
                elif f1 <= fs_full and f1 <= f0:
                    tn = 1.0
                else:
                    tn = max(0.0, min(1.0, t))
                c += (tn-h0)*A - (tn*tn-h0*h0)*D
                h[p] = tn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)
