import numpy as np
import time

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    lo, hi = -1.0, 1.0
    for _ in range(50):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def binarize(h, n):
    if n % 2:
        return None
    target = n // 2
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def binary_trial(c, h, om, i, j, n):
    """Compute new correlation after flipping h[i]:1->0, h[j]:0->1."""
    d = c.copy()
    d[i:i+n] -= om
    st = n - 1 - i
    d[st:st+n] += h
    d[j:j+n] += om
    st = n - 1 - j
    d[st:st+n] -= h
    d[n-1] -= 2.0
    if i != j:
        d[n-1+i-j] += 1.0
        d[n-1+j-i] += 1.0
    return d, float(d.max())

def targeted_binary(h, n, deadline, rng, max_iters=200000):
    """Binary local search with targeted moves and random restarts."""
    best = h.copy()
    bc = float(np.correlate(best, 1.0 - best, mode="full").max())
    dx = 2.0 / n
    L = 2 * n - 1
    diag = n - 1
    
    h_cur = best.copy()
    c_cur = np.correlate(h_cur, 1.0 - h_cur, mode="full").copy()
    om = 1.0 - h_cur[::-1]
    cm = bc
    
    # Temperature for simulated annealing
    temp = 0.5
    cool = 0.9999
    
    stagnant = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Choose move type
        if rng.random() < 0.6:
            # Targeted: find max lag, try to reduce it
            k_max = int(np.argmax(c_cur))
            d = k_max - diag
            if d > 0:
                # c[k_max] = sum of h[i](1-h[i+d]) for i=0..n-1-d
                # Try flipping (i, i+d) where h[i]=1, h[i+d]=0
                candidates = []
                for rep in range(5):
                    i = int(rng.randint(n - d))
                    if h_cur[i] > 0.5 and h_cur[i+d] < 0.5:
                        candidates.append((i, i+d))
                if not candidates:
                    # Try (i+d, i) direction: flip h[i+d]=0->1, h[i]=1->0 is same
                    # Try random pair
                    i = int(np.argmin(h_cur))
                    j = int(np.argmax(h_cur))
                    candidates = [(j, i)]
                i, j = candidates[0]
            elif d < 0:
                dd = -d
                candidates = []
                for rep in range(5):
                    i = int(rng.randint(n - dd))
                    if h_cur[i+dd] > 0.5 and h_cur[i] < 0.5:
                        candidates.append((i+dd, i))  # flip 1->0 at i+dd, 0->1 at i
                if not candidates:
                    i = int(np.argmin(h_cur))
                    j = int(np.argmax(h_cur))
                    candidates = [(j, i)]
                i, j = candidates[0]
            else:
                i = int(np.argmax(h_cur))
                j = int(np.argmin(h_cur))
        else:
            # Random move
            i = int(np.argmax(h_cur))
            j = int(np.argmin(h_cur))
            # Perturb
            for _ in range(3):
                ri = int(rng.randint(n))
                if h_cur[ri] > 0.5:
                    i = ri
                    break
            for _ in range(3):
                rj = int(rng.randint(n))
                if h_cur[rj] < 0.5:
                    j = rj
                    break
            if i == j:
                i = int(np.argmax(h_cur))
                j = int(np.argmin(h_cur))
        
        if h_cur[i] < 0.5 or h_cur[j] > 0.5:
            continue  # Not a valid flip
        
        # Try the flip
        c_new, nm = binary_trial(c_cur, h_cur, om, i, j, n)
        
        # Simulated annealing acceptance
        accept = False
        if nm < cm:
            accept = True
        else:
            delta = nm - cm
            if delta < temp:
                if rng.random() < np.exp(-delta / max(temp, 1e-10)):
                    accept = True
        
        if accept:
            c_cur = c_new
            cm = nm
            h_cur[i] = 0.0
            h_cur[j] = 1.0
            om[diag - i] = 1.0
            om[diag - j] = 0.0
            if cm < bc:
                best = h_cur.copy()
                bc = cm
                stagnant = 0
            temp *= cool
        else:
            stagnant += 1
            temp *= cool
        
        if stagnant > 500:
            # Restart with perturbation
            for _ in range(10):
                ri = int(rng.randint(n))
                rj = int(rng.randint(n))
                if h_cur[ri] > 0.5 and h_cur[rj] < 0.5:
                    h_cur[ri] = 0.0
                    h_cur[rj] = 1.0
                    om[diag - ri] = 1.0
                    om[diag - rj] = 0.0
            c_cur = np.correlate(h_cur, 1.0 - h_cur, mode="full")
            cm = float(c_cur.max())
            stagnant = 0
            temp = 0.5
    
    return best, bc * dx

def fista_optimize(h, n, deadline, alpha=5000, max_iter=5000):
    """FISTA for the smooth max approximation."""
    dx = 2.0 / n
    best = h.copy()
    best_c = compute_c5(best, n)
    x = h.copy()
    z = h.copy()
    t = 1.0
    L = 2 * n - 1
    diag = n - 1
    
    # Estimate Lipschitz constant
    # The gradient of smooth_max has Lipschitz constant bounded by alpha * (n * dx) roughly
    # Let's use a conservative estimate
    step_size = 1.0 / (2.0 * n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c = np.correlate(z, 1.0 - z, mode="full")
        cm = float(c.max())
        scaled = alpha * (c - cm)
        w = np.exp(scaled - scaled.max())
        w /= w.sum()
        
        # Gradient via convolution
        # grad[i] = sum_k w[k] * d c[k] / d z[i]
        # c[k] = sum_i z[i] * (1-z[i + k - diag])
        # d c[k] / d z[i] = (1-z[i+k-diag]) - z[i-k+diag] (for valid indices)
        # 
        # Let j = k - diag
        # If j >= 0: (1-z)[i+j] valid for i < n-j, z[i-j] valid for i >= j
        # If j < 0: (1-z)[i+j] = (1-z)[i-|j|] valid for i >= |j|, z[i+|j|] valid for i < n-|j|
        
        grad = np.zeros(n)
        for k in range(L):
            wk = w[k]
            if wk < 1e-12:
                continue
            j = k - diag
            if j >= 0:
                grad[:n-j] += wk * (1.0 - z[j:])
                grad[j:] -= wk * z[:n-j]
            else:
                jj = -j
                grad[jj:] += wk * (1.0 - z[:n-jj])
                grad[:n-jj] -= wk * z[jj:]
        
        grad *= dx
        
        # FISTA update
        x_new = project_simplex(z - step_size * grad, n)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = x_new + ((t - 1.0) / t_new) * (x_new - x)
        z = np.clip(z, 0.0, 1.0)  # Keep in bounds (not full projection for speed)
        x = x_new
        t = t_new
        
        cx = compute_c5(x, n)
        if cx < best_c:
            best = x.copy()
            best_c = cx
    
    return best, best_c

def transfer_h(h, n, n_tgt):

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    lo, hi = -1.0, 1.0
    for _ in range(50):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:

            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)

    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def binary_trial(c, h, om, i, j, n):
    d = c.copy()
    d[i:i+n] -= om
    st = n - 1 - i
    d[st:st+n] += h
    d[j:j+n] += om
    st = n - 1 - j
    d[st:st+n] -= h
    d[n-1] -= 2.0
    d[n-1+i-j] += 1.0
    d[n-1+j-i] += 1.0
    return d, float(d.max())

def binary_sa(h, n, deadline, rng, max_trials=500000):
    best = h.copy()
    bc = float(np.correlate(best, 1.0 - best, mode="full").max())
    h_cur = best.copy()
    c_cur = np.correlate(h_cur, 1.0 - h_cur, mode="full").copy()
    om = 1.0 - h_cur[::-1]
    cm = bc
    diag = n - 1
    temp = 1.0
    cool = 0.99995
    stagnant = 0
    trials = 0
    consecutive_fail = 0
    while time.time() < deadline and trials < max_trials:
        k_max = int(np.argmax(c_cur))
        d_lag = k_max - diag
        moved = False
        for attempt in range(3):
            if d_lag > 0:
                cands = []
                for _ in range(8):
                    i = int(rng.randint(n - d_lag))
                    if h_cur[i] > 0.5 and h_cur[i + d_lag] < 0.5:
                        cands.append(i)
                if cands:
                    i = cands[0]
                else:
                    i = int(np.argmax(h_cur))
            elif d_lag < 0:
                dd = -d_lag
                cands = []
                for _ in range(8):
                    i = int(rng.randint(n - dd))
                    if h_cur[i + dd] > 0.5 and h_cur[i] < 0.5:
                        cands.append(i + dd)
                if cands:
                    i = cands[0]
                else:
                    i = int(np.argmax(h_cur))
            else:
                i = int(np.argmax(h_cur))
            for _ in range(5):
                j = int(rng.randint(n))
                if h_cur[j] < 0.5:
                    break
            if h_cur[i] < 0.5 or h_cur[j] > 0.5 or i == j:
                continue
            c_new, nm = binary_trial(c_cur, h_cur, om, i, j, n)
            trials += 1
            accept = nm < cm - 1e-12
            if not accept and nm - cm < temp:
                if rng.random() < np.exp(-(nm - cm) / max(temp, 1e-12)):
                    accept = True
            if accept:
                c_cur = c_new
                cm = nm
                h_cur[i] = 0.0
                h_cur[j] = 1.0
                om[diag - i] = 1.0
                om[diag - j] = 0.0
                temp *= cool
                moved = True
                if cm < bc - 1e-14:
                    best = h_cur.copy()
                    bc = cm
                    stagnant = 0
                    consecutive_fail = 0
                else:
                    consecutive_fail += 1
                break
        if not moved:
            consecutive_fail += 1
            if consecutive_fail > 200:
                for _ in range(15):
                    ri = int(rng.randint(n))
                    rj = int(rng.randint(n))
                    if h_cur[ri] > 0.5 and h_cur[rj] < 0.5:
                        h_cur[ri] = 0.0
                        h_cur[rj] = 1.0
                        om[diag - ri] = 1.0
                        om[diag - rj] = 0.0
                c_cur = np.correlate(h_cur, 1.0 - h_cur, mode="full")
                cm = float(c_cur.max())
                consecutive_fail = 0
                temp = max(temp * 2.0, 0.5)
        temp *= cool
    return best, bc * (2.0 / n)

def smooth_val_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    w = ee / z
    dx = 2.0 / n
    diag = n - 1
    grad = np.zeros(n)
    g = 1.0 - h
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-10:
            continue
        j = k - diag
        lo1 = max(0, j)
        hi1 = min(n, j + n)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * g[0:hi1 - lo1]
        lo2 = max(0, -j)
        hi2 = min(n, n - j)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[0:hi2 - lo2]
    return (m + np.log(z) / alpha) * dx, grad * dx

def smooth_obj(h, n, alpha):
    return smooth_val_grad(h, n, alpha)[0]

def smooth_grad(h, n, alpha):
    return smooth_val_grad(h, n, alpha)[1]

def sum_con(h, n):
    return h.sum() - 0.5 * n

def sum_con_jac(h, n):
    return np.ones_like(h, dtype=float)

def slsqp_polish(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n)
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a), jac=smooth_grad,
                           method="SLSQP", bounds=bounds, constraints=cons,
                           options={"maxiter": maxiter, "ftol": 1e-12})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-14:
            best, bc = nh, nc
    return best, bc

def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h.copy(), n_tgt)
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(10.0, budget_s - 8.0)
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h0))
    except Exception:
        n = 500
        h0 = np.full(n, 0.5)
    h0 = project_simplex(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    n_best = n
    hb = binarize(h0, n)
    if hb is not None:
        hc = compute_c5(hb, n)
        if hc < best_c:
            best_h, best_c = hb, hc
    if time.time() > deadline - 60.0:
        fh = project_simplex(best_h, n_best)
        return fh, compute_c5(fh, n_best), int(n_best)
    t1 = min(deadline - 400.0, t0 + 0.42 * budget_s)
    if n % 2 == 0 and time.time() < t1:
        hb = binarize(best_h, n)
        if hb is not None:
            hb, hc = binary_sa(hb, n, t1, rng)
            if hc < best_c:
                best_h, best_c = hb, hc
    t2 = min(deadline - 250.0, t0 + 0.62 * budget_s)
    if time.time() < t2:
        hf, cf = slsqp_polish(best_h, n, [5000, 20000], 100, t2)
        if cf < best_c:
            best_h, best_c = hf, cf
    t3 = min(deadline - 120.0, t0 + 0.82 * budget_s)
    if n % 2 == 0 and time.time() < t3:
        hb = binarize(best_h, n)
        if hb is not None:
            hb, hc = binary_sa(hb, n, t3, rng, max_trials=300000)
            if hc < best_c:
                best_h, best_c = hb, hc
    if time.time() < deadline - 60.0:
        for nt in [450, 550, 400, 600]:
            if time.time() > deadline - 40.0:
                break
            if nt == n_best or nt < 200 or nt > 800 or nt % 2:
                continue
            ht = transfer_h(best_h, n_best, nt)
            ht_c = compute_c5(ht, nt)
            if nt % 2 == 0:
                htb = binarize(ht, nt)
                if htb is not None:
                    htb_c = compute_c5(htb, nt)
                    if htb_c < ht_c:
                        ht, ht_c = htb, htb_c
                    chunk = min(deadline - 40.0, time.time() + 50.0)
                    ht2, ht2c = binary_sa(ht, nt, chunk, rng, max_trials=100000)
                    if ht2c < ht_c:
                        ht, ht_c = ht2, ht2c
                    chunk2 = min(deadline - 30.0, time.time() + 25.0)
                    if time.time() < chunk2:
                        ht3, ht3c = slsqp_polish(ht, nt, [20000], 60, chunk2)
                        if ht3c < ht_c:
                            ht, ht_c = ht3, ht3c
            if ht_c < best_c:
                best_h, best_c, n_best = ht, ht_c, int(nt)
    if time.time() < deadline - 15.0:
        hf, cf = slsqp_polish(best_h, n_best, [50000], 80, deadline - 10.0)
        if cf < best_c:
            best_h, best_c = hf, cf
    final_h = project_simplex(best_h, n_best)
    final_c = float(compute_c5(final_h, n_best))
    return final_h, final_c, int(n_best)
