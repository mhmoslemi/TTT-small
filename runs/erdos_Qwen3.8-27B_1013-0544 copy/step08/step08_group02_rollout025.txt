def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    z = np.clip(z, 0.0, 1.0)
    diff = target - z.sum()
    if abs(diff) < 1e-10:
        return z
    if diff > 0:

    else:

        return z

    tot = room.sum()

    z = alpha * (c - m)

    e = np.exp(z - zm)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    target = 0.5 * n
    diff = target - z.sum()
    if abs(diff) < 1e-10:
        return z
    if diff > 0:
        mask = z < 1.0 - 1e-10
        room = 1.0 - z
    else:
        mask = z > 1e-10
        room = z.copy()
    room[~mask] = 0.0
    tot = room.sum()
    if tot < 1e-12:
        return z
    z[mask] += (diff / tot) * room[mask]
    return np.clip(z, 0.0, 1.0)

def compute_A(h, n):
    """Compute Jacobian matrix A where A[i,k] = dc[k]/dh[i]"""
    A = np.zeros((n, 2*n-1))
    off = n - 1
    for k in range(2*n-1):
        d = k - off
        if d >= 0:
            # c[k] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            A[:n-d, k] = 1.0 - h[d:n]  # derivative w.r.t. first factor
            A[d:, k] += -h[:n-d]        # derivative w.r.t. second factor
        else:
            e = -d
            # c[k] = sum_{i=e}^{n-1} h[i]*(1-h[i-e])
            A[e:, k] = 1.0 - h[:n-e]    # derivative w.r.t. h[i] (first factor)
            A[:n-e, k] += -h[e:]         # derivative w.r.t. h[i-e] (second factor)
    return A

def smooth_max_val(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    return (m + float(np.log(e.sum())) / alpha) * 2.0 / n

def smooth_max_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    p = e / e.sum()
    A = compute_A(h, n)
    return (2.0 / n) * (A @ p)

def pair_descent(h, n, max_iters, deadline):
    """Pair coordinate descent with exact line search"""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    A = compute_A(h, n)
    off = n - 1
    cur_max = float(c.max())
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Find active lags
        tol = 0.005 * cur_max
        active = np.flatnonzero(c >= cur_max - tol)
        
        # Pick a pair that involves an active lag
        if active.size > 0:
            k_star = active[np.random.randint(active.size)]
        else:
            k_star = int(np.argmax(c))
        
        d = k_star - off
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            b = np.random.randint(m)
            i, j = b + d, b
        else:
            e = -d
            m = n - e
            if m <= 0:
                continue
            b = e + np.random.randint(m)
            i, j = b - e, b
        
        # Line search: minimize max_k (c[k] + u*(A[i,k]-A[j,k]) + u^2*Q[k])
        B = A[i] - A[j]
        # Quadratic terms
        Q = np.zeros(2*n-1)
        if j > i:
            Q[off + (j-i)] = -1.0
        if i > j:
            Q[off + (i-j)] = 1.0
        
        # Valid range for u
        u_lo = max(-h[i], h[j] - 1.0 + 1e-10)
        u_hi = min(1.0 - h[i], h[j] - 1e-10)
        if u_hi - u_lo < 1e-12:
            continue
        
        # The function f(u) = max_k (c[k] + u*B[k] + u^2*Q[k]) is convex in u
        # Use golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b_ = u_lo, u_hi
        c1 = b_ - gr * (b_ - a)
        c2 = a + gr * (b_ - a)
        
        def f_of_u(u):
            return float(np.max(c + u * B + u*u * Q))
        
        f1, f2 = f_of_u(c1), f_of_u(c2)
        for _ in range(30):
            if f1 < f2:
                b_, c2, f2 = c2, c1, f1
                c1 = b_ - gr * (b_ - a)
                f1 = f_of_u(c1)
            else:
                a, c1, f1 = c1, c2, f2
                c2 = a + gr * (b_ - a)
                f2 = f_of_u(c2)
        
        u_best = (a + b_) / 2.0
        f_best = f_of_u(u_best)
        
        if f_best < cur_max - 1e-10:
            h[i] += u_best
            h[j] -= u_best
            c = c + u_best * B + u_best*u_best * Q
            # Recompute A
            A = compute_A(h, n)
            cur_max = f_best
    
    return h, compute_c5(h, n)

            q_k, q_sign = off + (i - j), 1.0
        u_lo = max(-h[i], h[j] - 1.0)
        u_hi = min(1.0 - h[i], h[j])
        if u_hi - u_lo < 1e-12:
            no_improve += 1
            continue
        a2, b3 = u_lo + 1e-12, u_hi - 1e-12
        c1v = b3 - gr * (b3 - a2)
        c2v = a2 + gr * (b3 - a2)
        tmp = c + c1v * B
        tmp[q_k] += c1v * c1v * q_sign
        f1v = float(tmp.max())
        tmp = c + c2v * B
        tmp[q_k] += c2v * c2v * q_sign
        f2v = float(tmp.max())
        for _ in range(18):
            if f1v < f2v:
                b3, c2v, f2v = c2v, c1v, f1v
                c1v = b3 - gr * (b3 - a2)
                tmp = c + c1v * B
                tmp[q_k] += c1v * c1v * q_sign
                f1v = float(tmp.max())
            else:
                a2, c1v, f1v = c1v, c2v, f2v
                c2v = a2 + gr * (b3 - a2)
                tmp = c + c2v * B
                tmp[q_k] += c2v * c2v * q_sign
                f2v = float(tmp.max())
        u_best = (a2 + b3) * 0.5
        tmp = c + u_best * B
        tmp[q_k] += u_best * u_best * q_sign
        f_best = float(tmp.max())
        if f_best < cur_max - 1e-12:
            h[i] += u_best
            h[j] -= u_best
            c = c + u_best * B
            c[q_k] += u_best * u_best * q_sign
            cur_max = f_best
            no_improve = 0
            if (it + 1) % 20 == 0:
                A = compute_A(h, n)
        else:
            no_improve += 1
            if no_improve > 400:
                break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def interp_to(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def try_transforms(h, n):
    c = compute_c5(h, n)
    best, best_c = h.copy(), c
    ht = project_feasible(0.5 * (h + h[::-1]), n)
    ct = compute_c5(ht, n)
    if ct < best_c:
        best, best_c = ht, ct
    ht = 1.0 - h
    ct = compute_c5(ht, n)
    if ct < best_c:
        best, best_c = ht, ct
    ht = project_feasible(0.5 * (h + 1.0 - h[::-1]), n)
    ct = compute_c5(ht, n)
    if ct < best_c:
        best, best_c = ht, ct
    return best, best_c

def optimize_one(h, n, deadline, rng):
    h, c = try_transforms(h, n)
    if time.time() < deadline - 80:
        for alpha in [50, 300]:
            if time.time() > deadline - 100:
                break
            h = slsqp(h, n, alpha, 80, deadline)
            c = compute_c5(h, n)
    if time.time() < deadline - 30:
        h, c = pair_descent(h, n, 5000, deadline - 30, rng)
    if time.time() < deadline - 20:
        for alpha in [1000, 5000]:
            if time.time() > deadline - 40:
                break
            h = slsqp(h, n, alpha, 60, deadline)
            c = compute_c5(h, n)
    if time.time() < deadline - 5:
        h, c = pair_descent(h, n, 5000, deadline, rng)
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 10.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)):
            raise ValueError("bad init")
    except Exception:
        n0 = 500
        x = np.linspace(0, 2, n0, endpoint=False)
        h0 = 0.5 + 0.25 * np.sin(np.pi * x) + 0.1 * np.sin(2.0 * np.pi * x)
        h0 = project_feasible(h0, n0)
    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    main_dl = hard - 150.0
    if main_dl < t0 + 40:
        main_dl = hard
    best_h, best_c = optimize_one(best_h, n0, main_dl, rng)
    for m in [200, 300, 400, 250, 350]:
        if time.time() > hard - 50:
            break
        if m == n0:
            continue
        rem = hard - time.time()
        if rem < 40:
            break
        dl = time.time() + min(30.0, 0.25 * rem)
        ht = interp_to(best_h, m)
        ht, ct = optimize_one(ht, m, dl, rng)
        if ct < best_c:
            best_h, best_c = ht, ct
    n = len(best_h)
    if time.time() < hard - 30:
        for s in [0.02, 0.05, 0.1]:
            if time.time() > hard - 20:
                break
            hp = project_feasible(best_h + rng.normal(0, s, n), n)
            dl = min(hard - 10, time.time() + 20)
            ht, ct = optimize_one(hp, n, dl, rng)
            if ct < best_c:
                best_h, best_c = ht, ct
    if time.time() < hard - 8:
        n2 = len(best_h)
        best_h = slsqp(best_h, n2, 20000, 40, hard)
        if time.time() < hard - 3:
            best_h, best_c = pair_descent(best_h, n2, 3000, hard, rng)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
