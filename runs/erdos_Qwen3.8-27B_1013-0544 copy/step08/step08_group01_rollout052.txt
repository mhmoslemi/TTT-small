            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            jf_arr = p + ms - (n - 1)
            jb_arr = p - ms + (n - 1)
            valid_f = (jf_arr >= 0) & (jf_arr < n)
            valid_b = (jb_arr >= 0) & (jb_arr < n)
            if np.any(valid_f):
                a_coeff[valid_f] += 1.0 - h[jf_arr[valid_f]]
            if np.any(valid_b):
                a_coeff[valid_b] -= h[jb_arr[valid_b]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = float(c[:m_lo].max())
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo = -h0
            hi = 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals_mat = c_aff[None, :] + d_grid[:, None] * a_coeff[None, :]
            if diag_local is not None:
                vals_mat[:, diag_local] -= d_grid[:, None] * d_grid[:, None]
            vals = np.maximum(vals_mat.max(axis=1), unaff_max)
            best_idx = int(np.argmin(vals))
            best_d = d_grid[best_idx]
            best_val = float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                gl = d_grid[best_idx - 1]
                gh = d_grid[best_idx + 1]
                try:
                    res = golden(obj_1d, gl, gh,
                                 args=(c_aff, a_coeff, diag_local, unaff_max),
                                 maxiter=golden_iter)
                    if res.fun < best_val - 1e-15:
                        best_d = float(res.x)
                        best_val = float(res.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
    return h, compute_c5(h, n)

def gradient_step(h, n, alpha, lr, deadline):
    c = np.correlate(h, 1.0 - h, mode='full')
    exp_c = np.exp(alpha * (c - c.max()))
    weights = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for p in range(n):
        jf = p - (n - 1)
        for m in range(max(0, p), min(2 * n - 1, p + n)):
            s = m - (n - 1)
            j_fwd = p + s
            j_bwd = p - s
            g = 0.0
            if 0 <= j_fwd < n:
                g += weights[m] * (1.0 - h[j_fwd])
            if 0 <= j_bwd < n:
                g -= weights[m] * h[j_bwd]
            grad[p] += g
    h_new = project_feasible(h - lr * grad, n)
    return h_new

def smooth_lsqp(h, n, alpha, maxiter=500):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * (2.0 / n)
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        exp_c = np.exp(alpha * (c - m))
        w = exp_c / exp_c.sum()
        grad = np.zeros(n)
        for p in range(n):
            for m_idx in range(max(0, p), min(2 * n - 1, p + n)):
                s = m_idx - (n - 1)
                jf = p + s
                jb = p - s
                g = 0.0
                if 0 <= jf < n:
                    g += w[m_idx] * (1.0 - x[jf])
                if 0 <= jb < n:
                    g -= w[m_idx] * x[jb]
                grad[p] += g
        return grad * (2.0 / n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP with moderate alpha
    if time.time() < deadline - 300:
        h1, c1 = smooth_lsqp(best_h, n, 2000, 400)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Extended coordinate descent
    if time.time() < deadline - 200:
        h2, c2 = coord_cd(best_h, n, 3000, deadline - 100,
                          grid_size=50, golden_iter=50)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Try nearby n values
    x_orig = np.linspace(0, 2, len(best_h))
    n_list = [n - 20, n - 10, n, n + 10, n + 20, n - 40, n + 40, n + 80]
    n_list = sorted(set(max(100, x) for x in n_list if x < 900))
    for n_try in n_list:
        if time.time() > deadline - 100:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_cd(h_int, n_try, 500, deadline - 50,
                                grid_size=50, golden_iter=40)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new
    # Phase 4: Structured restarts on best n
    n_cur = len(best_h)
    for r in range(8):
        if time.time() > deadline - 60:
            break
        s = 0.003 + 0.008 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        h_o, c_o = coord_cd(hp, n_cur, 2000, deadline - 40,
                            grid_size=50, golden_iter=40)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
    # Phase 5: Final fine-grained coordinate descent
    if time.time() < deadline - 20:
        hf, cf = coord_cd(best_h, len(best_h), 3000, deadline - 5,
                          grid_size=80, golden_iter=60)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Check 1-h symmetry
    n_cur = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
