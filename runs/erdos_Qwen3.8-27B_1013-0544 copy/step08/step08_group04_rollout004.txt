import numpy as np
import time

def project_feasible(h, n):
    target = 0.5 * n
    h = h.astype(float).copy()
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * (2.0 / n))

def get_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _obj_at(x, hp, cseg, aneg, qconst, unaff):
    val = qconst + x * (1.0 - x)
    if unaff > val:
        val = unaff
    if aneg.size:
        v = float(np.max(cseg + aneg * x))
        if v > val:
            val = v
    return val

def cd_optimize(h, n, max_rounds=10, deadline=None, project_every=8):
    h = h.copy()
    c = get_correlations(h)
    cur = float(c.max())
    best_h = h.copy()
    best_c = cur
    center = n - 1
    L = 2 * n - 1
    golden = 0.618033988749895

    for rnd in range(max_rounds):
        if deadline and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if deadline and time.time() > deadline:
                break
            hp = float(h[p])
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            seg = hi - lo + 1
            ss = center - np.arange(lo, hi + 1, dtype=np.int64)
            jj = p + ss
            ii = p - ss
            a = np.zeros(seg)
            vj = (jj >= 0) & (jj < n)
            vi = (ii >= 0) & (ii < n)
            if vj.any():
                a[vj] += 1.0 - h[jj[vj]]
            if vi.any():
                a[vi] -= h[ii[vi]]
            off = center - lo
            csg, asg = [], []
            if off > 0:
                csg.append(c[lo:center])
                asg.append(a[:off])
            if hi > center:
                csg.append(c[center + 1:hi + 1])
                asg.append(a[off + 1:])
            if csg:
                cseg = np.concatenate(csg)
                aneg = np.concatenate(asg)
            else:
                cseg = np.empty(0)
                aneg = np.empty(0)
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(c[hi + 1:].max()))
            # Golden section on [0,1]
            a_g, b_g = 0.0, 1.0
            c_g = b_g - golden * (b_g - a_g)
            d_g = a_g + golden * (b_g - a_g)
            fc = _obj_at(c_g, hp, cseg, aneg, qconst, unaff)
            fd = _obj_at(d_g, hp, cseg, aneg, qconst, unaff)
            for _ in range(35):
                if fc < fd:
                    b_g = d_g; d_g = c_g; fd = fc
                    c_g = b_g - golden * (b_g - a_g)
                    fc = _obj_at(c_g, hp, cseg, aneg, qconst, unaff)
                else:
                    a_g = c_g; c_g = d_g; fc = fd
                    d_g = a_g + golden * (b_g - a_g)
                    fd = _obj_at(d_g, hp, cseg, aneg, qconst, unaff)
            x_best = 0.5 * (a_g + b_g)
            f_best = _obj_at(x_best, hp, cseg, aneg, qconst, unaff)

                fc2 = _obj_at(x_chk, hp, cseg, aneg, qconst, unaff)
                if fc2 < f_best:
                    f_best = fc2
                    x_best = x_chk
            if f_best < cur - 1e-11:
                h[p] = x_best
                c = get_correlations(h)
                cur = float(c.max())
                improved = True

                c = get_correlations(h)
                cur = float(c.max())

        c = get_correlations(h)
        cur = float(c.max())
        if cur < best_c:
            best_c = cur
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c * (2.0 / n)

def fourier_perturb(h, n, rng, modes=12, scale=0.08):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    t = x - 1.0
    pert = np.zeros(n)
    for k in range(1, modes + 1):
        amp = rng.uniform(-scale / k, scale / k)
        perm = rng.choice([0, 1])
        if perm:
            pert += amp * np.cos(k * np.pi * t)
        else:
            pert += amp * np.sin(k * np.pi * t)
    return project_feasible(h + pert, n)

def shift_target_perturb(h, n, rng, strength=0.05):
    c = get_correlations(h)
    idx = int(np.argmax(c))
    center = n - 1
    k_shift = idx - center
    h2 = h.copy()
    if k_shift > 0 and k_shift < n:
        mask = np.zeros(n)
        mask[:n - k_shift] = h[:n - k_shift] * h[k_shift:]
        h2 = h2 - strength * mask
    elif k_shift < 0 and -k_shift < n:
        ks = -k_shift
        mask = np.zeros(n)
        mask[ks:] = h[ks:] * h[:n - ks]
        h2 = h2 - strength * mask
    else:
        h2 = h2 - strength * h * (1.0 - h)
    return project_feasible(h2, n)

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old.copy(), n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def binary_from_h(h):
    n = len(h)
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    h2 = np.zeros(n)
    h2[idx] = 1.0
    return h2
