   def rnd_sine(n, rng):
       x = np.arange(n, dtype=float) * (2.0 / n)
       h = 0.5 + 0.40 * np.sin(np.pi * x + rng.uniform(0, 2 * np.pi)) + 0.14 * np.sin(np.pi * 3 * x + rng.uniform(0, 2 * np.pi))
       return proj(h, n)

   def rnd_beatty(n, rng):
       i = np.arange(n)
       h = ((i * 0.6180339887498949 + rng.uniform(0, 1)) % 1.0 > 0.5).astype(float)
       return proj(h, n)

   def rnd_square(n, rng):
       # 2 blocks crossing mix
       a = sorted([int(rng.integers(1, n - 1)), int(rng.integers(1, n - 1)), n // 2])
       h = np.zeros(n)
       h[a[0]:a[1]] = 1; h[a[1]:a[2]] = 0; h[a[2]:] = 1
       return proj(h, n)

   def smooth_obj(x, n, alpha):
       c = np.correlate(x, 1.0 - x, mode='full')
       m = float(c.max())
       sh = alpha * (c - m)
       mx = float(sh.max())
       lse = mx + float(np.sum(np.exp(sh - mx)))
       return (m + np.log(lse) / alpha) * (2.0 / n)

   def sumeq(x, n):
       return x.sum() - n / 2.0

   def sumeqj(x, n):
       return np.ones(n)

   def lbfgs_polish(h, n, alphas, maxit, dl):
       bh = h.copy()
       bc = c5_of(bh, n)
       for al in alphas:
           if time.time() > dl - 70:
               break
           try:
               cons = ({'type': 'eq', 'fun': sumeq, 'jac': sumeqj, 'args': (n,)})
               res = minimize(smooth_obj, bh, args=(n, float(al)), method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n, constraints=cons,
                              options={'maxiter': int(maxit), 'ftol': 1e-20, 'gtol': 1e-16})
               nh = proj(res.x, n)
               nc = c5_of(nh, n)
               if nc < bc:
                   bh, bc = nh, nc
           except Exception:
               pass
       return bh, bc

    p = 4
    d = 0.07
    ht = hh.copy()
    ht[p] += d
    ct = np.correlate(ht, 1.0 - ht, mode='full')
    c0 = np.correlate(hh, 1.0 - hh, mode='full')
    m = arange_kp(p, n)
    da = (m == n - 1).astype(float)
    best = 0
    best_e = 1e30
    for sw in (0, 1):
        old = _SWP
        _SWP = sw
        A = _line_bld(p, m, hh, n)
        _SWP = old
        nu = c0[m] + d * (A + da * (1.0 - 2.0 * hh[p])) - d * d * da
        e = float(np.max(np.abs(nu - ct[m])))
        if e < best_e:
            best_e = e
            best = sw
    return best


def fa_lin(A, Bv, K, t):
    return max(K, float(np.max(Bv + A * t)))


def fc_con(Aq, Cq, Qc, t):
    return Cq + Aq * t - Qc * t * t


def conic_line_min(A, Bv, K, Aq, Cq, Qc, lo, hi):
    a, b = lo, hi
    for _ in range(26):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if fc_con(Aq, Cq, Qc, m1) - fa_lin(A, Bv, K, m1) <= fc_con(Aq, Cq, Qc, m2) - fa_lin(A, Bv, K, m2):
            a = m1
        else:
            b = m2
    ts = 0.5 * (a + b)
    cand = [lo, hi]
    a, b = lo, hi
    for _ in range(34):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if fa_lin(A, Bv, K, m1) <= fa_lin(A, Bv, K, m2):
            b = m2
        else:
            a = m1
    cand.append(0.5 * (a + b))
    if fc_con(Aq, Cq, Qc, ts) - fa_lin(A, Bv, K, ts) > -1e-12:
        L = lo
        if fc_con(Aq, Cq, Qc, lo) - fa_lin(A, Bv, K, lo) < 0.0:
            la, lb = lo, ts
            for _ in range(46):
                md = 0.5 * (la + lb)
                if fc_con(Aq, Cq, Qc, md) > fa_lin(A, Bv, K, md):
                    la = md
                else:
                    lb = md
            L = 0.5 * (la + lb)
        cand.append(L)
        R = hi
        if fc_con(Aq, Cq, Qc, hi) - fa_lin(A, Bv, K, hi) < 0.0:
            la, lb = ts, hi
            for _ in range(46):
                md = 0.5 * (la + lb)
                if fc_con(Aq, Cq, Qc, md) > fa_lin(A, Bv, K, md):
                    la = md
                else:
                    lb = md
            R = 0.5 * (la + lb)
        cand.append(R)
    if lo < 0.0 < hi:
        cand.append(0.0)
    bt = cand[0]
    bf = None
    for t in cand:
        f = max(fa_lin(A, Bv, K, t), fc_con(Aq, Cq, Qc, t))
        if bf is None or f < bf:
            bf = f
            bt = t
    return float(bt), float(bf)


def single_pass(h, c, n, deadline=None):
    cur = float(c.max())
    L2 = 2 * n - 1
    imp = 0
    for p in range(n):
        if deadline is not None and time.time() > deadline:
            break
        hp = h[p]
        lo = hp - 1.0
        hi = 1.0 - hp
        if hi - lo < 1e-13:
            continue
        matep = max(1e-12, 1e-11 * cur)
        m = arange_kp(p, n)
        Kk = 0.0
        if m[0] > 0:
            Kk = max(Kk, float(c[:m[0]].max()))
        if m[-1] < L2 - 1:
            Kk = max(Kk, float(c[m[-1] + 1:].max()))
        A = _line_bld(p, m, h, n)
        d0 = (m == n - 1)
        C0 = c[m] - hp * A - d0 * (hp - hp * hp)
        Ava = A + d0 * (1.0 - 2.0 * hp)
        qq = int(np.nonzero(d0)[0][0])
        t, fv = conic_line_min(A, C0, Kk, float(Ava[qq]), float(C0[qq]), 1.0, lo, hi)
        if fv < cur - matep and abs(t) > 1e-13:
            h[p] = hp + t
            c[m] = C0 + t * Ava - t * t * d0
            if fv < cur:
                cur = fv
            imp += 1
    return h, c, imp, cur


def greedy_pair_pass(h, c, n, rng, rounds, deadline=None):
    cur = float(c.max())
    L2 = 2 * n - 1
    imp = 0
    bad = 0
    for _r in range(rounds):
        tt = time.time()
        if deadline is not None and tt > deadline:
            break
        act = np.nonzero(c > cur - max(1.0, cur * 0.01))[0]
        best = None
        for _b in range(40):
            used = False
            if act.size and rng.random() < 0.30:
                ms = int(act[int(rng.integers(act.size))])
                k = ms - (n - 1)
                if k != 0:
                    ilo = max(0, -k)
                    ihi = min(n, n - k)
                    if ihi > ilo:
                        i = int(rng.integers(ilo, ihi))
                        j = i + k
                        used = True
            if not used:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
            bi = h[i]
            bj = h[j]
            lo = max(-bi, bj - 1.0)
            hi = min(bj, 1.0 - bi)
            if hi - lo < 1e-13:
                continue
            mi = arange_kp(i, n)
            mj = arange_kp(j, n)
            mask = np.zeros(L2, bool)
            mask[mi] = True
            mask[mj] = True
            aff = np.nonzero(mask)[0]
            Kk = float(c[~mask].max())
            A1 = _line_bld(i, aff, h, n)
            A2 = _line_bld(j, aff, h, n)
            Ag = A1 - A2
            d0 = (aff == n - 1)
            own = (bi - bi * bi) + (bj - bj * bj)
            Cx = c[aff] - bi * A1 - bj * A2 - d0 * own
            lnu = (1.0 - 2.0 * bi) - (1.0 - 2.0 * bj)
            qq = int(np.nonzero(d0)[0][0])
            Av = Ag + d0 * lnu
            t, _fv = conic_line_min(Ag, Cx, Kk, float(Av[qq]), float(Cx[qq]), 2.0, lo, hi)
            newaff = Cx + t * Ag + d0 * lnu * t - 2.0 * d0 * t * t
            ftr = max(Kk, float(newaff.max()))
            matep = max(1e-12, 1e-11 * cur)
            if abs(t) > 1e-13 and ftr < cur - matep and (best is None or ftr < best[3]):
                best = (i, j, t, ftr, aff, Cx, Ag, d0, lnu, bi, bj)
        if best is None:
            bad += 1
            if bad >= 4:
                break
            continue
        bad = 0
        i, j, t, ftr, aff, Cx, Ag, d0, lnu, bi, bj = best
        h[i] = bi + t
        h[j] = bj - t
        c[aff] = Cx + t * Ag + d0 * lnu * t - 2.0 * d0 * t * t
        cur = ftr
        imp += 1
    return h, c, imp, cur


def cand_sine(n, rng):
    x = np.arange(n, dtype=float) * (2.0 / n)
    h = (0.5
         + 0.40 * np.sin(np.pi * x + float(rng.uniform(0.0, 2.0 * np.pi)))
         + 0.16 * np.sin(np.pi * 3 * x + float(rng.uniform(0.0, 2.0 * np.pi))))
    return proj(h, n)


def cand_beatty(n, rng):
    i = np.arange(n, dtype=float)
    h = ((i * 0.6180339887498949 + float(rng.uniform(0.0, 1.0))) % 1.0 > 0.5).astype(float)
    return proj(h, n)


def cand_square(n, rng):
    mid = n // 2
    a1 = int(rng.integers(0, n))
    a2 = int(rng.integers(0, n))
    cut = [a1, a2, mid]
    cut.sort()
    h = np.zeros(n)
    h[cut[0]:cut[1]] = 1.0
    h[cut[2]:] = 1.0
    return proj(h, n)


def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    lse = mx + float(np.sum(np.exp(sh - mx)))
    return (m + np.log(lse) / alpha) * (2.0 / n)


def sumeq(x, n):
    return x.sum() - n / 2.0


def sumeqj(x, n):
    return np.ones(n)


def lbfgs_polish(h, n, alphas, maxit, dl):
    bh = h.copy()
    bc = c5_of(bh, n)
    for al in alphas:
        if time.time() > dl - 50:
            break
        try:
            cons = ({'type': 'eq', 'fun': sumeq, 'jac': sumeqj, 'args': (n,)})
            res = minimize(smooth_obj, bh, args=(n, float(al)), method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n, constraints=cons,
                           options={'maxiter': int(maxit), 'ftol': 1e-20, 'gtol': 1e-15})
            nh = proj(res.x, n)
            nc = c5_of(nh, n)
            if nc < bc:
                bh, bc = nh, nc
        except Exception:
            pass
    return bh, bc
