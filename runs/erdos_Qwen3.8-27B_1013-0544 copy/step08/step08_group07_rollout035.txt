3. Check the objective at each such delta (and at boundaries)
4. Pick the best

    h = h.copy()
    c = get_correlation(h, n)
    best_c = float(c.max()) * (2.0/n)
    nm1 = n - 1
    total_lags = 2*n - 1

            break
        current_max = float(c.max())
        improved = 0
        
        for p in range(n):
            h0 = h[p]

                continue
            
            m_lo = max(0, p - n + 1, n - 1 - p)
            m_hi = min(total_lags - 1, p + n - 1, total_lags - 1 - p)
            if m_lo > m_hi:
                continue
            ms = np.arange(m_lo, m_hi + 1)
            i1 = p - ms + nm1
            i2 = p + ms - nm1
            v1 = (i1 >= 0) & (i1 < n)
            v2 = (i2 >= 0) & (i2 < n)
            a = np.zeros(len(ms))
            a[v1] += (1.0 - h[i1[v1]])

            if m_hi < total_lags - 1:

            # Get the top K affected correlations
            K = min(15, len(ms))
            top_local = np.argsort(c_aff)[-K:]

            d_lo = -h0
            d_hi = 1.0 - h0
            candidates = [d_lo, d_hi, 0.0]
            
            for ii in range(K):
                for jj in range(ii+1, K):
                    if abs(a[top_local[ii]] - a[top_local[jj]]) > 1e-12:
                        d = (c_aff[top_local[jj]] - c_aff[top_local[ii]]) / (a[top_local[ii]] - a[top_local[jj]])

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-15: break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-15: break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def get_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj(x, n, alpha):
    c = get_corr(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_jac(x, n):
    return np.ones(len(x))

def lbfgs_opt(h, n, alphas, maxiter=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline and time.time() > deadline - 30:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def cd_analytical(h, n, max_passes=2000, deadline=None, topK=20):
    """Fast coordinate descent using analytical breakpoints."""
    h = h.copy()
    c = get_corr(h, n)
    best_c = float(c.max()) * (2.0 / n)
    nm1 = n - 1
    tot = 2 * n - 1
    
    for pidx in range(max_passes):
        if deadline and time.time() > deadline:
            break
        current_max = float(c.max())
        improved = 0
        
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            m_lo = max(0, p - n + 1, nm1 - p)
            m_hi = min(tot - 1, p + n - 1, tot - 1 - p)
            if m_lo > m_hi:
                continue
            ms = np.arange(m_lo, m_hi + 1)
            i1 = p - ms + nm1
            i2 = p + ms - nm1
            v1 = (i1 >= 0) & (i1 < n)
            v2 = (i2 >= 0) & (i2 < n)
            a = np.zeros(len(ms))
            if np.any(v1): a[v1] += 1.0 - h[i1[v1]]
            if np.any(v2): a[v2] -= h[i2[v2]]
            c_aff = c[ms]
            
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < tot - 1:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            
            diag_mask = (ms == nm1)
            
            # Focus on top K affected + unaff_max level
            K = min(topK, len(ms))
            top_idx = np.argsort(c_aff)[-K:]
            
            d_lo, d_hi = -h0, 1.0 - h0
            candidates = np.array([d_lo, d_hi, 0.0])
            
            # Crossings among top K
            for ii in range(K):
                ti = top_idx[ii]
                for jj in range(ii + 1, K):
                    tj = top_idx[jj]
                    da = a[ti] - a[tj]
                    if abs(da) > 1e-12:
                        d = (c_aff[tj] - c_aff[ti]) / da
                        if d_lo < d < d_hi:
                            candidates = np.append(candidates, d)
                # Crossing with unaff_max
                if a[ti] > 1e-12:
                    d = (unaff_max - c_aff[ti]) / a[ti]
                    if d_lo < d < d_hi:
                        candidates = np.append(candidates, d)
            
            # Evaluate
            best_d = 0.0
            best_val = current_max
            for d in candidates:
                nv = c_aff + d * a
                if np.any(diag_mask):
                    nv[diag_mask] -= d * d
                val = max(unaff_max, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            
            if abs(best_d) > 1e-12 and best_val < current_max - 1e-14:
                h[p] = max(0.0, min(1.0, h0 + best_d))
                c[ms] = c_aff + best_d * a
                if np.any(diag_mask):
                    c[ms[diag_mask]] -= best_d * best_d
                improved += 1
        
        if improved == 0:
            break
        h = project_feasible(h, n)
        c = get_corr(h, n)
        cv = float(c.max()) * (2.0 / n)
        if cv < best_c:
            best_c = cv
    
    return h, best_c

def binary_round(h, n):
    """Round to binary with exactly n/2 ones."""
    h_sorted = np.sort(h)[::-1]
    threshold = h_sorted[n // 2 - 1] if n % 2 == 0 else h_sorted[n // 2]
    # For even n: exactly n/2 ones
    if n % 2 == 0:
        binary = (h >= threshold).astype(float)
        if binary.sum() > n / 2:
            # Need to reduce: set the smallest "1"s to 0
            ones = np.where(binary == 1)[0]
            if len(ones) > n / 2:
                excess = len(ones) - n / 2
                ones_sorted = np.sort(ones, key=lambda i: h[ones]) # this won't work without lambda
                # Use argsort
                h_ones = h[ones]
                order = np.argsort(h_ones)[:int(excess)]
                binary[ones[order]] = 0
        binary = (h >= h_sorted[int(n/2) - 1]).astype(float)
        # Ensure exactly n/2
        if binary.sum() != n / 2:
            binary = (h >= h_sorted[int(n/2) - 1]).astype(float)
            # Still might not work due to ties; use rank-based
            ranks = np.argsort(np.argsort(h))
            binary = (ranks >= n - n//2).astype(float)
    else:
        ranks = np.argsort(np.argsort(h))
        binary = (ranks >= n - n//2).astype(float)
    return binary

def cd_binary(h, n, max_passes=500, deadline=None):
    """Coordinate descent on binary h (flip moves)."""
    h = binary_round(h, n).copy()
    c = get_corr(h, n)
    best_c = float(c.max()) * (2.0 / n)
    nm1 = n - 1
    tot = 2 * n - 1
    
    for pidx in range(max_passes):
        if deadline and time.time() > deadline:
            break
        current_max = float(c.max())
        improved = 0
        for p in range(n):
            h0 = h[p]
            # Try flipping
            new_h = 1.0 - h0
            # Compute effect
            m_lo = max(0, p - n + 1, nm1 - p)
            m_hi = min(tot - 1, p + n - 1, tot - 1 - p)
            ms = np.arange(m_lo, m_hi + 1)
            i1 = p - ms + nm1
            i2 = p + ms - nm1
            v1 = (i1 >= 0) & (i1 < n)
            v2 = (i2 >= 0) & (i2 < n)
            a = np.zeros(len(ms))
            if np.any(v1): a[v1] += 1.0 - h[i1[v1]]
            if np.any(v2): a[v2] -= h[i2[v2]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < tot - 1:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            diag_mask = (ms == nm1)
            delta = new_h - h0  # +1 or -1
            nv = c_aff + delta * a
            if np.any(diag_mask):
                nv[diag_mask] -= delta * delta  # = -1
            val = max(unaff_max, float(nv.max()))
            if val < current_max - 1e-14:
                h[p] = new_h
                c[ms] = nv
                current_max = val
                improved += 1
        if improved == 0:
            break
        cv = float(c.max()) * (2.0 / n)
        if cv < best_c:
            best_c = cv
    
    return h, best_c

            best_val = float(vals[bi])

                h[p] = max(0.0, min(1.0, h0 + best_d))
                c[ms] = c_aff + best_d * a
                if np.any(diag_mask):
                    c[ms[diag_mask]] -= best_d * best_d
                improved += 1
        if improved == 0:
            break
        h = project_feasible(h, n)
        c = get_corr(h, n)
        cv = float(c.max()) * (2.0 / n)
        if cv < best_c:
            best_c = cv
    return h, best_c

def binary_round(h, n):
    ranks = np.argsort(np.argsort(h))
    binary = (ranks >= n - n // 2).astype(float)
    return binary

def cd_binary(h, n, max_passes=500, deadline=None):
    h = binary_round(h, n).copy()
    c = get_corr(h, n)
    best_c = float(c.max()) * (2.0 / n)
    nm1 = n - 1
    tot = 2 * n - 1
    for pidx in range(max_passes):
        if deadline and time.time() > deadline:
            break
        current_max = float(c.max())
        improved = 0
        for p in range(n):
            h0 = h[p]
            m_lo = max(0, min(p, nm1 - p))
            m_hi = min(tot - 1, max(p + nm1, tot - 1 - p))
            if m_lo > m_hi:
                continue
            ms = np.arange(m_lo, m_hi + 1)
            i1 = p - ms + nm1
            i2 = p + ms - nm1
            v1 = (i1 >= 0) & (i1 < n)
            v2 = (i2 >= 0) & (i2 < n)
            a = np.zeros(len(ms))
            if np.any(v1):
                a[v1] += 1.0 - h[i1[v1]]
            if np.any(v2):
                a[v2] -= h[i2[v2]]
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < tot - 1:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            diag_mask = (ms == nm1)
            delta = 1.0 - 2.0 * h0
            nv = c_aff + delta * a
            if np.any(diag_mask):
                nv[diag_mask] -= delta * delta
            val = max(unaff, float(nv.max()))
            if val < current_max - 1e-14:
                h[p] = 1.0 - h0
                c[ms] = nv
                current_max = val
                improved += 1
        if improved == 0:
            break
        cv = float(c.max()) * (2.0 / n)
        if cv < best_c:
            best_c = cv
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 300
        x = np.linspace(0, 2, n, endpoint=False)
        h0 = 0.5 + 0.35 * np.sin(2 * np.pi * x) + 0.15 * np.sin(4 * np.pi * x)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: L-BFGS
    if time.time() < deadline - 100:
        h, c = lbfgs_opt(best_h, n, [500, 5000, 50000], 600, deadline)
        if c < best_c:
            best_h, best_c = h, c
    # Phase 2: Fast CD
    if time.time() < deadline - 60:
        h, c = cd_fast(best_h, n, 3000, deadline, topK=15)
        if c < best_c:
            best_h, best_c = h, c
    # Phase 3: Binary exploration
    if time.time() < deadline - 80:
        hb, cb = cd_binary(best_h, n, 800, deadline)
        if cb < best_c:
            best_h, best_c = hb, cb
        hc = project_feasible(hb + 0.04 * rng.randn(n), n)
        hc, cc = cd_fast(hc, n, 2000, deadline, topK=15)
        if cc < best_c:
            best_h, best_c = hc, cc
    # Phase 4: Multi-n
    for n_try in [200, 256, 350, 400, 500, 600]:
        if time.time() > deadline - 60:
            break
        if n_try == n:
            continue
        x_s = np.linspace(0, 2, len(best_h), endpoint=False)
        x_d = np.linspace(0, 2, n_try, endpoint=False)
        ht = project_feasible(np.interp(x_d, x_s, best_h), n_try)
        ht, ct = cd_fast(ht, n_try, 2000, deadline, topK=12)
        if ct < best_c:
            best_h, best_c = ht, ct
    # Phase 5: Perturbed restart
    if time.time() < deadline - 30:
        n2 = len(best_h)
        hp = project_feasible(best_h + 0.02 * rng.randn(n2), n2)
        hp, cp = cd_fast(hp, n2, 2000, deadline, topK=18)
        if cp < best_c:
            best_h, best_c = hp, cp
    # Final polish
    if time.time() < deadline - 2:
        n3 = len(best_h)
        hf, cf = cd_fast(best_h, n3, 2000, deadline, topK=25)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
