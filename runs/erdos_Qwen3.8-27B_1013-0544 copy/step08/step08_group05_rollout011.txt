def coord_eval_fn(c, A, xs, old, n):
    d = xs - old
    cand = c[None, :] + d[:, None] * A[None, :]
    cand[:, n - 1] -= d * d
    return cand.max(axis=1)

def coord_cd(h, n, max_passes, deadline, grid, refine):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    A = np.zeros(L)
    xs0 = np.linspace(0.0, 1.0, grid)
    tol = 1e-12
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = float(x[p])
            A.fill(0.0)
            A[p:p + n] += om
            st = diag - p
            A[st:st + n] -= x
            cur = float(c.max())
            vals = coord_eval_fn(c, A, xs0, old, n)
            i = int(np.argmin(vals))
            bx = float(xs0[i])
            bv = float(vals[i])
            if bv < cur - tol:
                sp = 1.0 / (grid - 1)
                lo = max(0.0, bx - sp)
                hi = min(1.0, bx + sp)
                xx = np.linspace(lo, hi, refine + 1)

import numpy as np
from scipy.optimize import minimize
import time


def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(55):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
    return np.clip(y, 0.0, 1.0)


def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)


def swap_update(c, h, om, i, j, n):
    d = c.copy()
    d[i:i + n] -= om
    st = n - 1 - i
    d[st:st + n] += h
    d[j:j + n] += om
    st = n - 1 - j
    d[st:st + n] -= h
    d[n - 1] -= 2.0
    d[n - 1 + i - j] += 1.0
    d[n - 1 + j - i] += 1.0
    return d


def binarize(h, n):
    if n % 2:
        return None
    target = n // 2
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b


def sa_optimize(b, n, rng, time_limit, T0=0.4):
    c = np.correlate(b, 1.0 - b, mode="full")
    best_b = b.copy()
    best_c = float(c.max())
    om = 1.0 - b[::-1]
    T = T0
    t_end = time.time() + time_limit
    stuck = 0
    iters = 0
    while time.time() < t_end:
        cm = float(c.max())
        ones_pos = np.where(b == 1.0)[0]
        zeros_pos = np.where(b == 0.0)[0]
        i = int(rng.choice(ones_pos))
        j = int(rng.choice(zeros_pos))
        new_c = swap_update(c, b, om, i, j, n)
        new_max = float(new_c.max())
        delta = new_max - cm
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / (T + 1e-14))):
            c = new_c
            b[i] = 0.0
            b[j] = 1.0
            om[n - 1 - i] = 1.0
            om[n - 1 - j] = 0.0
            if new_max < best_c - 1e-15:
                best_c = new_max
                best_b = b.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            stuck += 1
        iters += 1
        if iters % 800 == 0:
            T *= 0.996
            if stuck > 25000 and time.time() < t_end - 5:
                ndiv = min(20, n // 15)
                for _ in range(ndiv):
                    ones_pos = np.where(b == 1.0)[0]
                    zeros_pos = np.where(b == 0.0)[0]
                    ii = int(rng.choice(ones_pos))
                    jj = int(rng.choice(zeros_pos))
                    c = swap_update(c, b, om, ii, jj, n)
                    b[ii] = 0.0
                    b[jj] = 1.0
                    om[n - 1 - ii] = 1.0
                    om[n - 1 - jj] = 0.0
                T = max(T, T0 * 0.25)
                stuck = 0
    return best_b, best_c * (2.0 / n)


def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx


def smooth_obj(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]


def smooth_grad(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]


def sum_con(x, n):
    return x.sum() - 0.5 * n


def sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)


def slsqp_polish(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n)
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a), jac=smooth_grad,
                           method="SLSQP", bounds=bounds, constraints=cons,
                           options={"maxiter": int(maxiter), "ftol": 1e-14})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-14:
            best = nh
            bc = nc
    return best, bc


def coord_eval_fn(c, A, xs, old, n):
    d = xs - old
    cand = c[None, :] + d[:, None] * A[None, :]
    cand[:, n - 1] -= d * d
    return cand.max(axis=1)


def coord_cd(h, n, max_passes, deadline, grid, refine):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    A = np.zeros(L)
    xs0 = np.linspace(0.0, 1.0, grid)
    tol = 1e-13
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = float(x[p])
            A.fill(0.0)
            A[p:p + n] += om
            st = diag - p
            A[st:st + n] -= x
            cur = float(c.max())
            vals = coord_eval_fn(c, A, xs0, old, n)
            idx = int(np.argmin(vals))
            bx = float(xs0[idx])
            bv = float(vals[idx])
            if bv < cur - tol:
                sp = 1.0 / (grid - 1)
                lo = max(0.0, bx - sp)
                hi = min(1.0, bx + sp)
                xx = np.linspace(lo, hi, refine + 1)
                mm = coord_eval_fn(c, A, xx, old, n)
                j = int(np.argmin(mm))
                if mm[j] < bv - tol:
                    bx = float(xx[j])
                    bv = float(mm[j])
                if lo < bx < hi:
                    sp2 = (hi - lo) / float(refine)
                    lo2 = max(0.0, bx - sp2)
                    hi2 = min(1.0, bx + sp2)
                    yy = np.linspace(lo2, hi2, 9)
                    ww = coord_eval_fn(c, A, yy, old, n)
                    k = int(np.argmin(ww))
                    if ww[k] < bv - tol:
                        bx = float(yy[k])
                        bv = float(ww[k])
                if bv < cur - tol:
                    d = bx - old
                    c[p:p + n] += d * om
                    c[st:st + n] -= d * x
                    c[diag] -= d * d
                    x[p] = bx
                    om[diag - p] = 1.0 - bx
                    improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - tol:
            best = xp
            bc = cp
        if not improved:
            break
    return best, bc


def transfer_h(h, n_src, n_tgt):
    if n_tgt == n_src:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n_src)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)
