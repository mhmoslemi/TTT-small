    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff/tot)*room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff/tot)*h, 0.0, 1.0)
    return h

def gentle_project(h, n, strength=0.1):
    """Project with damping to avoid large jumps."""
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    # Distribute proportionally to available room
    residual = diff * strength
    room = 1.0 - h
    tot = room.sum()
    if tot < 1e-15: return h
    h = np.clip(h + (residual/tot)*room, 0.0, 1.0)
    return h

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def cd_pass(h, n, c, deadline):
    for p in range(n):
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
            continue
        m_lo = max(0, n - 1 - p)
        m_hi = min(2 * n - 2, n - 1 + p)
        ms = np.arange(m_lo, m_hi + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        a = np.zeros(len(ms))
        if np.any(vf):
            a[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            a[vb] -= h[jb[vb]]
        c_aff = c[ms]
        unaff = 0.0
        if m_lo > 0:
            unaff = max(unaff, float(c[:m_lo].max()))
        if m_hi < 2 * n - 2:
            unaff = max(unaff, float(c[m_hi + 1:].max()))
        diag_pos = np.where(ms == n - 1)[0]
        has_diag = len(diag_pos) > 0
        di = diag_pos[0] if has_diag else -1
        Llo, Lhi = unaff, float(c.max())
        for _ in range(48):
            if Lhi - Llo < 3e-14:
                break
            L = (Llo + Lhi) / 2.0
            xlo, xhi = 0.0, 1.0
            ok = True
            mnd = np.ones(len(ms), dtype=bool)
            if has_diag:
                mnd[di] = False
            if np.any(mnd):

                neg = am < -1e-14
                zro = np.abs(am) <= 1e-14
                if np.any(zro) and np.any(cm[zro] > L + 1e-13):
                    ok = False
                elif np.any(pos):
                    xhi = min(xhi, float(np.min((L - cm[pos]) / am[pos])) + h0)
                if ok and np.any(neg):
                    xlo = max(xlo, float(np.max((L - cm[neg]) / am[neg])) + h0)
            if ok and has_diag:
                dlo, dhi = xlo - h0, xhi - h0
                if dlo > dhi:
                    ok = False
                else:
                    ds = (1.0 - 2.0 * h0) / 2.0
                    if dlo <= ds <= dhi:

                            ok = False
                    else:

                                ok = False
                                break
            if ok and xlo <= xhi + 1e-12:
                Lhi = L
            else:
                Llo = L
        Lopt = Lhi
        xlo, xhi = 0.0, 1.0
        mnd = np.ones(len(ms), dtype=bool)
        if has_diag:
            mnd[di] = False
        if np.any(mnd):
            am, cm = a[mnd], c_aff[mnd]
            pos = am > 1e-14

            if np.any(pos):
                xhi = min(xhi, float(np.min((Lopt - cm[pos]) / am[pos])) + h0)
            if np.any(neg):
                xlo = max(xlo, float(np.max((Lopt - cm[neg]) / am[neg])) + h0)
        if has_diag:

                w1 = max(0.0, hi1 - lo1) + 1e-12

def multi_cd(h, n, deadline, proj_every=1):
    c = np.correlate(h, 1.0 - h, mode='full')
    passes = 0
    while time.time() < deadline:
        h, c = cd_pass(h, n, c, deadline)
        c = np.correlate(h, 1.0 - h, mode='full')
        passes += 1
        if passes % max(1, proj_every) == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, c
