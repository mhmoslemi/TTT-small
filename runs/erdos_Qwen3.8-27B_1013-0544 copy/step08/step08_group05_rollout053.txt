def nesterov_optimize(h0, n, alpha, max_iter, step_lr, momentum, deadline):
    h = h0.copy()
    v = np.zeros(n)
    f_prev = compute_c5(h, n)
    best_h = h.copy()
    best_f = f_prev
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Nesterov lookahead
        h_nest = project_simplex(h - momentum * v, n)
        f, g = smooth_val_grad(h_nest, n, alpha)
        # Center gradient (for sum constraint)
        g = g - g.mean()
        # Update
        v = momentum * v + g
        h_new = project_simplex(h - step_lr * v, n)
        f_new = compute_c5(h_new, n)
        if f_new < f_prev:
            h = h_new
            f_prev = f_new
            if f_new < best_f:
                best_h = h.copy()
                best_f = f_new
        else:
            # Backtrack
            step_lr *= 0.5
            h_new = project_simplex(h - step_lr * v, n)
            f_new = compute_c5(h_new, n)
            if f_new < f_prev:
                h = h_new
                f_prev = f_new
            else:
                v *= 0  # reset momentum on failure
    
    return best_h, best_f
