import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            room = (1.0 - h) * mask
            tot = room.sum()
            h += (diff / tot) * room if tot > 1e-15 else 0
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            tot = h[mask].sum()
            h[mask] += (diff / tot) * h[mask] if tot > 1e-15 else 0
    return np.clip(h, 0.0, 1.0)

def exact_cd(h, n, max_passes=500, deadline=None, gs_iters=35):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        cmax = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            # Compute A vector: A[k] = (1-h[p+shift])*1_{valid} - h[p-shift]*1_{valid}
            shift = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
            p_plus = p + shift
            p_minus = p - shift
            A = np.zeros(2*n-1)
            mask1 = (p_plus >= 0) & (p_plus < n)
            mask2 = (p_minus >= 0) & (p_minus < n)
            A[mask1] += (1.0 - h[p_plus[mask1]])
            A[mask2] -= h[p_minus[mask2]]
            
            # F(t) = max(c + (t-h0)*A - (t^2-h0^2)*1_{k=n-1})
            B = np.zeros(2*n-1)
            B[n-1] = 1.0  # subtract (t^2-h0^2) at k=n-1
            
            # Golden section
            a, b = 0.0, 1.0
            c1 = b - gr*(b-a)
            d1 = a + gr*(b-a)
            fc1 = float(np.max(c + (c1-h0)*A - (c1*c1-h0*h0)*B))
            fd1 = float(np.max(c + (d1-h0)*A - (d1*d1-h0*h0)*B))
            for _ in range(gs_iters):
                if fc1 < fd1:
                    b = d1; d1 = c1; fd1 = fc1
                    c1 = b - gr*(b-a)
                    fc1 = float(np.max(c + (c1-h0)*A - (c1*c1-h0*h0)*B))
                else:
                    a = c1; c1 = d1; fc1 = fd1
                    d1 = a + gr*(b-a)
                    fd1 = float(np.max(c + (d1-h0)*A - (d1*d1-h0*h0)*B))
            t_star = (a+b)*0.5
            f0 = float(np.max(c - h0*A + h0*h0*B))
            f1 = float(np.max(c + (1-h0)*A - (1-h0*h0)*B))
            fstar = float(np.max(c + (t_star-h0)*A - (t_star*t_star-h0*h0)*B))
            if min(f0, f1, fstar) < cmax - 1e-14:
                if f0 <= f1 and f0 <= fstar:
                    t_new = 0.0
                elif f1 <= f0 and f1 <= fstar:
                    t_new = 1.0
                else:
                    t_new = t_star
                c += (t_new-h0)*A - (t_new*t_new-h0*h0)*B
                h[p] = t_new
                cmax = float(c.max())
                improved = True
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0-h, mode='full').astype(float)
        if not improved:
            break
    return h, compute_c5(h, n)

```python
def pnorm_grad(h, n, p):
    c = np.correlate(h, 1.0-h, mode='full')
    if p <= 1:
        w = np.ones_like(c)
    else:
        w = p * np.maximum(c, 0) ** (p-1)  # c should be non-negative
    # grad[j] = sum_k w[k] * [(1-h[j+shift_k])*1_{valid} - h[j-shift_k]*1_{valid}]
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] == 0:
            continue
        shift_k = k - (n-1)
        # First term: (1-h[j+shift_k]) for j where 0<=j+shift_k<n
        js = max(0, -shift_k)
        je = min(n, n - shift_k)
        if js < je:
            grad[js:je] += w[k] * (1.0 - h[js+shift_k:je+shift_k])
        # Second term: -h[j-shift_k] for j where 0<=j-shift_k<n
        js2 = max(0, shift_k)
        je2 = min(n, n + shift_k)
        if js2 < je2:
            grad[js2:je2] -= w[k] * h[js2-shift_k:je2-shift_k]
    return grad
```

This is O(n²) per call (loop over k, each O(1) numpy ops). For n=500: 999 iterations, each with ~4 numpy slice ops of average length 250. Total: ~999 * 4 * 250 = 1M operations. In numpy: ~1-2ms.

SLSQP with this gradient: maybe 200 iterations * (1 gradient + 1 function eval) = 200 * 3ms = 600ms. Very fast!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            room = (1.0 - h) * mask
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            tot = h[mask].sum()
            if tot < 1e-15:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def pnorm_obj(h, n, p):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.sum(c ** p))

def pnorm_grad(h, n, p):
    c = np.correlate(h, 1.0 - h, mode='full')
    w = p * c ** (p - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-30:
            continue
        shift_k = k - (n - 1)
        js = max(0, -shift_k)
        je = min(n, n - shift_k)
        if js < je:
            grad[js:je] += wk * (1.0 - h[js + shift_k:je + shift_k])
        js2 = max(0, shift_k)
        je2 = min(n, n + shift_k)
        if js2 < je2:
            grad[js2:je2] -= wk * h[js2 - shift_k:je2 - shift_k]
    return grad

def slsqp_opt(h, n, p, maxiter=500, deadline=None):
    def obj(x):
        return pnorm_obj(x, n, p)
    def grad(x):
        return pnorm_grad(x, n, p)
    bounds = [(0.0, 1.0)] * n
    def sum_con(x):
        return x.sum() - n / 2.0
    def sum_con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        return new_h
    except Exception:
        return h.copy()
