            if tot < 1e-15:
                break

def corr_gradient(h, n, lag):

    g = np.zeros(n)
    if lag >= 0:
        if lag > 0:
            g[lag:] += 1.0 - h[:n-lag]
            g[:n-lag] -= h[lag:]

            g[:] = 1.0 - 2.0 * h
    else:
        l = -lag
        g[:n-l] += 1.0 - h[l:]
        g[l:] -= h[:n-l]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def corr_grad(h, n, lag):
    g = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += 1.0 - h[:n - lag]
        g[:n - lag] -= h[lag:]
    else:
        l = -lag
        g[:n - l] += 1.0 - h[l:]
        g[l:] -= h[:n - l]
    return g

def subgrad_opt(h, n, deadline, top_m=3, momentum=0.9):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.06
    v = np.zeros(n)
    stall = 0
    for it in range(30000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        if top_m <= 1:
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            g = corr_grad(h, n, lag)
        else:
            top_idx = np.argpartition(c, -top_m)[-top_m:]
            g = np.zeros(n)
            for idx in top_idx:
                lag = idx - (n - 1)
                g += corr_grad(h, n, lag)
            g /= top_m
        v = momentum * v - eta * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        h = h_new
        if c_new < best_c - 1e-10:
            best_h = h.copy()
            best_c = c_new
            stall = 0
            eta = min(eta * 1.03, 0.25)
        else:
            stall += 1
            if stall % 40 == 0:
                eta *= 0.92
                momentum = min(momentum + 0.01, 0.95)
    return best_h, best_c

    for pidx in range(max_passes):

        current_max = float(c_prev.max())
        improved = False

                break

            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue

            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                a[vb] -= h[jb[vb]]

        return project_feasible(h, n)
    rng = np.random.RandomState(0)

    for trial in range(8):

            h = build_h(c)
