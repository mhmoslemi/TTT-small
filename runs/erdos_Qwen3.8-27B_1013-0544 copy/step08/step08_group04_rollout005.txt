def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 1e-14:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 1e-14:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-10:
        idx = np.where(h < 1.0)[0]
        for i in idx:
            add = min(res, 1.0 - h[i])
            h[i] += add
            res -= add
            if res < 1e-12:
                break
    elif res < -1e-10:
        idx = np.where(h > 0.0)[0]
        for i in idx:
            sub = min(-res, h[i])
            h[i] -= sub
            res += sub
            if res > -1e-12:
                break
    return h

def cd_pass(h, n, order, deadline, bisection_iters=22):
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    center = n - 1
    L = 2 * n - 1
    for p in order:
        if time.time() > deadline:
            break
        hp = float(h[p])
        lo = min(p, center - p)
        hi = max(center + p, 2 * center - p)
        seg = hi - lo + 1
        s = center - np.arange(lo, hi + 1, dtype=int)
        j = p + s
        i = p - s
        a = np.zeros(seg, dtype=float)
        vj = (j >= 0) & (j < n)
        vi = (i >= 0) & (i < n)
        if np.any(vj):
            a[vj] += (1.0 - h[j[vj]])
        if np.any(vi):
            a[vi] -= h[i[vi]]
        off = center - lo
        cseg = np.empty(seg - 1, dtype=float)
        asg = np.empty(seg - 1, dtype=float)
        if off:
            cseg[:off] = c[lo:center]
            asg[:off] = a[:off]
        if hi > center:
            cseg[off:] = c[center + 1:hi + 1]
            asg[off:] = a[off + 1:]
        qconst = c[center] - hp * (1.0 - hp)
        unaff = 0.0
        if lo > 0:
            unaff = max(unaff, float(c[:lo].max()))
        if hi + 1 < L:
            unaff = max(unaff, float(c[hi + 1:].max()))
        bx, bf = optimize_coordinate(hp, cseg, asg, qconst, unaff, bisection_iters)
        if bf < cur - 1e-12:
            dxv = bx - hp
            if off:
                c[lo:center] += asg[:off] * dxv
            if hi > center:
                c[center + 1:hi + 1] += asg[off:] * dxv
            c[center] = qconst + bx * (1.0 - bx)
            h[p] = bx
            cur = bf
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    return h, cur * (2.0 / n), cur

def cd_multi(h, n, max_passes, deadline, bisection_iters=22):
    h = project_feasible(h, n)
    c = compute_c(h, n)
    best_h = h.copy()
    best_c = c
    fwd = np.arange(n)
    bwd = np.arange(n - 1, -1, -1)
    for p in range(max_passes):
        if time.time() > deadline:
            break
        if p % 3 == 0:
            order = fwd
        elif p % 3 == 1:
            order = bwd
        else:
            order = np.random.permutation(n)
        h, c_raw, _ = cd_pass(h, n, order, deadline, bisection_iters)
        c_val = c_raw * (2.0 / n)
        if c_val < best_c - 1e-14:
            best_c = c_val
            best_h = h.copy()
    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    z = alpha * c
    zm = float(z.max())
    return ((zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)

def sum_con(x, n):
    return float(x.sum()) - 0.5 * n

def slsqp_step(h, n, alphas, maxiter, deadline):
    best_h = project_feasible(h, n)
    best_c = compute_c(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline - 30:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-11, 'iprint': 0})
            x = project_feasible(res.x, n)
            c = compute_c(x, n)
            if c < best_c:
                best_h, best_c = x, c
        except Exception:
            pass
    return best_h, best_c
