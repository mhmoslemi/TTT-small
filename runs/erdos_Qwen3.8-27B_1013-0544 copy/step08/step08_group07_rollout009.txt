import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    # Top-k smooth max for sharper focus
    k = min(30, len(c))
    topk = np.partition(c, len(c)-k)[len(c)-k:]
    shifted = alpha * (topk - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 20:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-20, 'gtol': 1e-14, 'maxfun': maxiter_each * 3})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    k = min(40, len(c))
    topk = np.partition(c, len(c) - k)[len(c) - k:]
    shifted = alpha * (topk - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alpha, maxiter=1000, deadline=None):

    if deadline is not None and time.time() > deadline - 10:
        return best_h, best_c
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15,
                                'maxfun': maxiter * 5})
        new_h = project_feasible(res.x, n)
