import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(55):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def sv_val(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max() * (2.0 / n)
    z = alpha * (c * (2.0 / n) - m)
    zm = z.max()
    return m + np.log(np.exp(z - zm).sum()) / alpha

def sv_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    z = alpha * (c - m)
    zm = z.max()
    e = np.exp(z - zm)
    p = e / e.sum()
    g = np.zeros(n)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-14:
            continue
        d = k - off
        if d >= 0:
            g[:n-d] += w * (1.0 - x[d:])
            g[d:] -= w * x[:n-d]
        else:
            e2 = -d
            g[e2:] += w * (1.0 - x[:n-e2])
            g[:n-e2] -= w * x[e2:]
    return (2.0 / n) * g

def slsqp(h, n, deadline, alphas):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    for alpha, mx in alphas:
        if time.time() > deadline - 2.0:
            break
        try:
            res = minimize(sv_val, best_h, args=(n, alpha), method='SLSQP',
                          jac=sv_grad, bounds=bounds, constraints=cons,
                          options={'maxiter': mx, 'ftol': 1e-15, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best_h, best_c = cur, cc
        except Exception:
            pass
    return best_h, best_c

def compute_A(h, n):
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[:n-d, k] = 1.0 - h[d:]
            A[d:, k] -= h[:n-d]
        else:
            e = -d
            A[e:, k] = 1.0 - h[:n-e]
            A[:n-e, k] -= h[e:]
    return A

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(55):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def sv_val(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    z = alpha * (c - m)
    zm = z.max()
    return m + np.log(np.exp(z - zm).sum()) / alpha

def sv_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    z = alpha * (c - m)
    zm = z.max()
    e = np.exp(z - zm)
    p = e / e.sum()
    g = np.zeros(n)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-14:
            continue
        d = k - off
        if d >= 0:
            g[:n-d] += w * (1.0 - x[d:])
            g[d:] -= w * x[:n-d]
        else:
            e2 = -d
            g[e2:] += w * (1.0 - x[:n-e2])
            g[:n-e2] -= w * x[e2:]
    return (2.0 / n) * g

def slsqp(h, n, deadline, alphas):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    for alpha, mx in alphas:
        if time.time() > deadline - 2.0:
            break
        try:
            res = minimize(sv_val, best_h, args=(n, alpha), method='SLSQP',
                          jac=sv_grad, bounds=bounds, constraints=cons,
                          options={'maxiter': mx, 'ftol': 1e-15, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best_h, best_c = cur, cc
        except Exception:
            pass
    return best_h, best_c

def compute_A(h, n):
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[:n-d, k] = 1.0 - h[d:]
            A[d:, k] -= h[:n-d]
        else:
            e = -d
            A[e:, k] = 1.0 - h[:n-e]
            A[:n-e, k] -= h[e:]
    return A

def pair_search(h, n, grid, deadline, rng, max_stale=600):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    A = compute_A(h, n)
    off = n - 1
    stale = 0
    total_tries = 0
    while time.time() < deadline:
        total_tries += 1
        cur_max = c.max()
        tol = max(0.015, 0.0008 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(c.argmax())])
        if total_tries % 7 == 6 or active.size <= 2:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
        else:
            kk = active[int(rng.randint(active.size))]
            d = kk - off
            if d >= 0:
                m = n - d
                if m <= 1:
                    continue
                b = int(rng.randint(m))
                i, j = b + d, b
            else:
                e = -d
                m = n - e
                if m <= 1:
                    continue
                b = e + int(rng.randint(m))
                i, j = b - e, b
        hi, hj = h[i], h[j]
        S = hi + hj
        lo = max(0.0, S - 1.0)
        hiT = min(1.0, S)
        if hiT - lo < 1e-14:
            continue
        u_grid = np.linspace(lo - hi, hiT - hi, grid)
        B = A[i] - A[j]
        lines = c[None, :] + u_grid[:, None] * B[None, :]
        u2 = u_grid * u_grid
        lines[:, off] -= 2.0 * u2
        lines[:, off + i - j] += u2
        lines[:, off + j - i] += u2
        vals = lines.max(axis=1)
        idx = int(np.argmin(vals))
        if 1 <= idx <= grid - 2:
            v0, v1, v2 = vals[idx-1], vals[idx], vals[idx+1]
            den = v0 - 2.0*v1 + v2
            if den > 1e-14:
                hs = u_grid[1] - u_grid[0]
                up = u_grid[idx] - 0.5 * hs * (v0 - v2) / den
                if u_grid[idx-1] < up < u_grid[idx+1]:
                    lp = c + up * B
                    lp = lp.copy()
                    lp[off] -= 2.0 * up * up
                    lp[off + i - j] += up * up
                    lp[off + j - i] += up * up
                    lv = lp.max()
                    if lv < vals[idx]:
                        idx_p = idx
                        vals[idx] = lv
                        u_grid[idx] = up
        if vals[idx] < cur_max - 1e-10:
            u = u_grid[idx]
            old = hi + hj
            t = np.clip(hi + u, 0.0, 1.0)
            u = t - hi
            h[i] = t
            h[j] = old - t
            B = A[i] - A[j]
            u2 = u * u
            c = c + u * B
            c[off] -= 2.0 * u2
            c[off + i - j] += u2
            c[off + j - i] += u2
            stale = 0
            A = compute_A(h, n)
            if stale < 0:
                pass
            if total_tries % 100 == 0:
                c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        else:
            stale += 1
            if stale >= max_stale:
                h = project_feasible(h + rng.normal(0, 0.008, n), n)
                c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
                A = compute_A(h, n)
                stale = 0
    return h, compute_c5(h, n)
