def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(55):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def smooth_components(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    s = float(e.sum())
    p = e / s
    return (m + np.log(s) / alpha) * (2.0 / n), p, c

def grad_weights(x, n, p):
    g = np.zeros(n, dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-13:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n - d])
            g[:n - d] -= w * x[d:]
        else:
            e = -d
            g[:n - e] += w * (1.0 - x[e:])
            g[e:] -= w * x[:n - e]
    return (2.0 / n) * g

def smooth_val(x, n, alpha):
    return smooth_components(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    _, p, _ = smooth_components(x, n, alpha)
    return grad_weights(x, n, p)

def slsqp_stage(h, n, bounds, cons, plans, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work_h = best_h.copy()
    for alpha, mx in plans:
        if time.time() > deadline - 1.0:
            break
        try:
            res = minimize(smooth_val, work_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_grad, bounds=bounds, constraints=cons,
                           options={'maxiter': mx, 'ftol': 1e-15, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best_h, best_c = cur, cc
            work_h = cur if cc <= best_c + 1e-10 else best_h
        except Exception:
            pass
    return best_h, best_c

def compute_A(h, n):
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            m = n - d
            if m > 0:
                A[:m, k] += (1.0 - h[d:d + m])
                A[d:d + m, k] -= h[:m]
        else:
            e = -d
            m = n - e
            if m > 0:
                A[:m, k] += (1.0 - h[e:e + m])
                A[e:e + m, k] -= h[:m]
    return A

def pair_search(h, n, max_iters, deadline, rng, grid=40, accept_tol=1e-9):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    A = compute_A(h, n)
    off = n - 1
    accepts = 0
    no_improve = 0
    max_no_improve = max(500, max_iters // 3)
    for trial in range(max_iters):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        tol = max(0.03, 0.002 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        if trial % 7 == 6 or active.size <= 2:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
                if i == j:
                    continue
        else:
            kk = int(active[int(rng.randint(active.size))])
            d = kk - off
            if d >= 0:
                m = n - d
                if m <= 0:
                    continue
                b = int(rng.randint(m))
                i = b + d
                j = b
            else:
                e = -d
                m = n - e
                if m <= 0:
                    continue
                b = e + int(rng.randint(m))
                i = b - e
                j = b
        old_sum = h[i] + h[j]
        lo = max(0.0, old_sum - 1.0)
        hi = min(1.0, old_sum)
        if hi - lo <= 1e-14:
            no_improve += 1
            continue
        B = A[i] - A[j]
        us = np.linspace(lo - h[i], hi - h[i], grid)
        u2 = us * us
        lines = c[None, :] + us[:, None] * B[None, :]
        lines[:, off] -= 2.0 * u2
        qi = off + i - j
        qj = off + j - i
        if 0 <= qi < len(c):
            lines[:, qi] += u2
        if 0 <= qj < len(c):
            lines[:, qj] += u2
        vals = lines.max(axis=1)
        idx = int(np.argmin(vals))
        worst = float(vals[idx])
        if worst < cur_max - accept_tol:
            u = float(us[idx])
            t = float(np.clip(h[i] + u, 0.0, 1.0))
            if abs(t - h[i]) < 1e-15:
                no_improve += 1
                continue
            h[i] = t
            h[j] = old_sum - t
            cn = c + u * B
            cn[off] -= 2.0 * u * u
            if 0 <= qi < len(c):
                cn[qi] += u * u
            if 0 <= qj < len(c):
                cn[qj] += u * u
            c = cn
            accepts += 1
            no_improve = 0
            A[i] = _a_col(i, h, n)
            A[j] = _a_col(j, h, n)
            if accepts % 25 == 0:
                c = np.correlate(h, 1.0 - h, mode='full')
                A = compute_A(h, n)
        else:
            no_improve += 1
        if no_improve > max_no_improve:
            break
    return h, compute_c5(h, n)

def _a_col(i, h, n):
    col = np.zeros(2 * n - 1, dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            if 0 <= i < n - d:
                col[k] += (1.0 - h[i + d])
            if d <= i < n:
                col[k] -= h[i - d]
        else:
            e = -d
            if e <= i < n:
                col[k] += (1.0 - h[i - e])
            if 0 <= i < n - e:
                col[k] -= h[i + e]
    return col

def apply_symmetries(h, n):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    hs = project_feasible(0.5 * (h + h[::-1]), n)
    cs = compute_c5(hs, n)
    if cs < best_c:
        best_h, best_c = hs, cs
    hc = 1.0 - best_h
    cs = compute_c5(hc, n)
    if cs < best_c:
        best_h, best_c = hc, cs
    hsc = project_feasible(0.5 * (best_h + 1.0 - best_h[::-1]), n)
    cs = compute_c5(hsc, n)
    if cs < best_c:
        best_h, best_c = hsc, cs
    return best_h, best_c

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def make_random_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, 10):
        a = rng.uniform(-0.4, 0.4)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += a * np.sin(0.5 * np.pi * m * x + ph)
        h += 0.5 * a * np.cos(np.pi * m * x + ph)
    return project_feasible(np.clip(h, 0.0, 1.0), n)

def optimize_one(h, n, deadline, rng, quick=False):
    h = project_feasible(h, n)
    h, c = apply_symmetries(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    if quick:
        plans = [(50, 30), (200, 30), (1000, 25), (5000, 20)]
    else:
        plans = [(20, 60), (100, 50), (500, 40), (2000, 30), (10000, 25), (50000, 20)]
    if not quick and time.time() < deadline - 30.0:
        plans = [(10, 80), (50, 60), (200, 50), (1000, 40), (5000, 30), (20000, 25), (80000, 20), (200000, 15)]
    h, c = slsqp_stage(h, n, bounds, cons, plans, deadline)
    if n <= 800 and time.time() < deadline - 3.0:
        rem = deadline - time.time()
        mt = int(min(8000 if quick else 15000, max(300 if quick else 1000, rem * 25)))
        h, c = pair_search(h, n, mt, deadline, rng)
    if not quick and time.time() < deadline - 5.0:
        h, c = slsqp_stage(h, n, bounds, cons, [(50000, 25), (200000, 20), (500000, 15)], deadline)
        if n <= 800 and time.time() < deadline - 2.0:
            rem = deadline - time.time()
            mt = int(min(10000, max(500, rem * 20)))
            h, c = pair_search(h, n, mt, deadline, rng)
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
        if n0 < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        n0 = 500
        h0 = make_random_start(n0, rng)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    best_h, best_c = apply_symmetries(best_h, n0)
    if hard - t0 < 40.0:
        return best_h, compute_c5(best_h, n0), n0
    main_dl = hard - max(80.0, 0.22 * float(budget_s))
    if main_dl < t0 + 25.0:
        main_dl = hard
    best_h, best_c = optimize_one(best_h, n0, main_dl, rng, False)
    cands = [320, 400, 480, 550, 640, 800]
    for m in cands:
        if time.time() > hard - 30.0:
            break
        if m == len(best_h):
            continue
        rem = hard - time.time()
        if m > 500 and rem < 80.0:
            continue
        if m > 700 and rem < 120.0:
            continue
        dl = time.time() + min(50.0, 0.18 * rem)
        if dl < time.time() + 10.0:
            continue
        ht, ct = optimize_one(interp_h(best_h, m), m, dl, rng, True)
        if ct < best_c:
            best_h, best_c = ht, ct
    if time.time() < hard - 60.0:
        n = len(best_h)
        for s in [0.02, 0.05, 0.1, 0.2]:
            if time.time() > hard - 30.0:
                break
            hp = project_feasible(best_h + rng.normal(0.0, s, n), n)
            dl = min(hard - 10.0, time.time() + 45.0)
            ht, ct = optimize_one(hp, n, dl, rng, True)
            if ct < best_c:
                best_h, best_c = ht, ct
    if time.time() < hard - 8.0:
        n = len(best_h)
        h2, c2 = pair_search(best_h, n, 20000, hard - 4.0, rng, grid=45)
        if c2 < best_c:
            best_h, best_c = h2, c2
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
