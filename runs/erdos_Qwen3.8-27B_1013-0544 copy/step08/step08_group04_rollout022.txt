Let me also think about a different angle: **what's the theoretical lower bound?**

For any h with ∫h = 1 on [0,2]:
At shift 0: ∫h(1-h)dx = 1 - ∫h²dx ≤ 1 - (∫h)²/2 = 1 - 1/2 = 1/2 (by Cauchy-Schwarz, ∫h² ≥ (∫h)²/2 = 1/2)

Actually, ∫h(1-h)dx = 1 - ∫h²dx. To minimize this, maximize ∫h²dx. The max of ∫h² given ∫h=1 on [0,2] is 1 (achieved by h=1 on a set of measure 1, 0 elsewhere). So the min of shift-0 is 0.

For binary: shift-0 = 0 exactly.

For other shifts, the lower bound is harder. Let's think about it differently.

∫h(x)(1-h(x+k))dx for a specific k. If h is binary (indicator of set A with |A|=1):
= |A ∩ (A+k)^c ∩ [0,2-k]| (for k>0, approximately)
= |A ∩ [0,2-k]| - |A ∩ (A+k)|

For the optimal arrangement, we want this to be small for ALL k simultaneously.

This is related to the concept of "cross-intersecting families" or "difference sets" in combinatorics.

            s = center - np.arange(lo, hi+1)

            i = p - s  # p-m
            
            a = np.zeros(hi - lo + 1)
            vj = (j >= 0) & (j < n)
            vi = (i >= 0) & (i < n)
            a[vj] += 1.0 - h[j[vj]]
            a[vi] -= h[i[vi]]
            
            # c_seg = c[lo:hi+1] - a*hp (the "base" without the x*1 contribution at center)
            # Actually: c[m] for m in [lo,hi] is: base[m] + a[m]*h[p]
            # where at m=center, the contribution is h[p]*(1-h[p]) = h[p] - h[p]^2
            # For m != center, the contribution is linear: a[m]*h[p]
            
            # So: c[m](x) = c[m] - a[m]*hp + a[m]*x for m != center
            #       c[center](x) = c[center] - hp*(1-hp) + x*(1-x)
            
            c_seg = c[lo:hi+1].copy()
            a_seg = a.copy()
            qconst = c_seg[center - lo] - hp*(1-hp)  # constant part of center
            c_seg[center - lo] = qconst  # will handle separately
            
            # Unaffected shifts
            if lo > 0:
                unaff = float(c[:lo].max())
            else:
                unaff = 0.0
            if hi+1 < L:
                um = float(c[hi+1:].max())
                if um > unaff:
                    unaff = um
            
            # Binary search for optimal x
            # f(x) = max(unaff, qconst + x*(1-x), max(m!=center: c_seg[m] - a_seg[m]*hp + a_seg[m]*x))
            # Note: c_seg[m] - a_seg[m]*hp is the base value (without x contribution)
            base = c_seg - a_seg * hp
            base[center - lo] = qconst  # center is special
            
            lo_x = 0.0
            hi_x = 1.0
            for _ in range(30):
                mid_x = 0.5*(lo_x + hi_x)
                # f(mid_x)
                if a_seg[center-lo] == 0:  # shouldn't happen
                    val_center = qconst + mid_x*(1-mid_x)
                else:
                    val_center = qconst + mid_x*(1-mid_x)
                # Linear parts
                lin = base + a_seg * mid_x
                lin[center-lo] = val_center
                f_val = max(unaff, float(lin.max()))
                
                # Check if we can do better
                # This is a piecewise linear + quadratic function
                # For now, use bisection on the subgradient
                # Actually, let's just use ternary search or golden section
                
            # Simpler: just try a few candidate values
            candidates = [0.0, 1.0, 0.5, hp]
            # Also try values that make the worst shift equal to unaff or to the quadratic
            # For the quadratic: x*(1-x) = target → x = (1±sqrt(1-4*target))/2
            for target_val in [unaff, float(base.max()), 0.25]:
                disc = 1.0 - 4.0*(target_val - qconst)
                if disc >= 0:
                    candidates.append(0.5 + 0.5*np.sqrt(disc))
                    candidates.append(0.5 - 0.5*np.sqrt(disc))
            # For linear: a[m]*x + base[m] = target → x = (target-base[m])/a[m]
            for m_idx in range(len(a_seg)):
                if m_idx == center - lo:
                    continue
                if abs(a_seg[m_idx]) > 1e-12:
                    for tv in [unaff, float(base.max())]:
                        xv = (tv - base[m_idx]) / a_seg[m_idx]
                        if 0 <= xv <= 1:
                            candidates.append(xv)
            
            best_x = hp
            best_f = max(unaff, float((base + a_seg*hp).max()))
            # Fix center
            center_idx = center - lo
            lin_at_hp = base + a_seg*hp
            lin_at_hp[center_idx] = qconst + hp*(1-hp)
            best_f = max(unaff, float(lin_at_hp.max()))
            
            for x in candidates:
                if x < 0 or x > 1:
                    continue
                lin = base + a_seg*x
                lin[center_idx] = qconst + x*(1-x)
                f_val = max(unaff, float(lin.max()))
                if f_val < best_f - 1e-12:
                    best_f = f_val
                    best_x = x
            
            if best_x != hp:
                dx = best_x - hp
                c[lo:hi+1] += a_seg * dx
                c[center+lo] = qconst + best_x*(1-best_x)  # wait, center_idx in c is 'center'
                # Actually c[center] is at index center in the full array
                c[center] = qconst + best_x*(1-best_x)
                h[p] = best_x
                cur = max(float(c.max()), unaff)  # unaff is already in c
                cur = float(c.max())
                if cur * (2.0/n) < best_val - 1e-12:
                    best_val = cur * (2.0/n)
                    best_h = h.copy()
                improved = True
        
        if not improved:
            break
    
    return best_h, best_val

import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-12:
        idx = np.where((h < 1) if res > 0 else (h > 0))[0]
        if idx.size:
            h[idx[0]] += res
    return h

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def make_binary(h, rng=None):
    n = len(h)
    k = n // 2
    if rng is not None:
        h = h + rng.uniform(0, 1e-7, n)
    idx = np.argpartition(h, n - k)[n - k:]
    h_b = np.zeros(n)
    h_b[idx] = 1.0
    return h_b

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

    best_h = h.copy()

    center = n - 1

        q = zeros[rng.randint(len(zeros))]

        qm = q - m_arr

        m1 = (pm >= 0) & (pm < n)

        delta = np.zeros(2*n-1)

        pos1 = center + (q - p)
