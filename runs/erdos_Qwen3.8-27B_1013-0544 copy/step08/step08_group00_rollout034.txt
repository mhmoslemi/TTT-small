def lbfgs_optimize(h, n, alpha, maxiter=500, deadline=None):
    """L-BFGS-B optimization with smooth max objective. Uses numerical gradient."""
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        if deadline is not None and time.time() > deadline:
            return float('inf')
        return smooth_max_obj_scalar(x, n, alpha)
    
    cons_sum = h.sum() - n / 2.0
    
    # Use penalty for sum constraint since L-BFGS-B doesn't support equality constraints
    def penalized_obj(x):
        if deadline is not None and time.time() > deadline:
            return float('inf')
        base = smooth_max_obj_scalar(x, n, alpha)
        penalty = 1e4 * (x.sum() - n / 2.0) ** 2
        return base + penalty

    shifted = alpha * c
    mx = shifted.max()
    lse = mx / alpha + np.log(np.sum(np.exp(shifted - mx))) / alpha
