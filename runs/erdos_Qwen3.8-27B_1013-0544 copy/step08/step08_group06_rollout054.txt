        if abs(diff) < 1e-14:
            break
        if diff > 0:

            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def subgradient_method(h, n, max_iter=5000, deadline=None):
    """Subgradient method focusing on peak correlations."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = corr * dx
        peak = c.max()
        
        if peak < best_c - 1e-15:
            best_c = peak
            best_h = h.copy()
        
        # Find active k values (within small fraction of peak)
        threshold = peak - 0.05 * peak
        active = np.where(c >= threshold)[0]
        
        # Compute subgradient for active k values
        grad = np.zeros(n)
        for m in active:
            s = m - (n - 1)
            j = np.arange(n)
            # term1: (1 - h[j+s]) for valid j+s
            v1 = (j + s >= 0) & (j + s < n)
            term1 = np.zeros(n)
            term1[v1] = 1.0 - h[j[v1] + s]
            # term2: h[j-s] for valid j-s
            v2 = (j - s >= 0) & (j - s < n)
            term2 = np.zeros(n)
            term2[v2] = h[j[v2] - s]
            grad += (term1 - term2)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        step = 2.0 * best_c / (gnorm * np.sqrt(it + 1))
        step = min(step, 0.5)
        
        new_h = h - step * (dx) * grad / gnorm
        new_h = project_feasible(new_h, n)
        new_c = compute_c5(new_h, n)
        
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        h = new_h
    
    return best_h, best_c

def make_interval_h(a, b, c, n):

    h = np.zeros(n)
    x = np.arange(n) * (2.0 / n) + 1.0 / n
    h[(x < a)] = 1.0
    h[(x >= b) & (x < b + c)] = 1.0

def make_periodic_h(period, duty, n):

    h = np.zeros(n)
    x = np.arange(n) * (2.0 / n) + 1.0 / n
    phase = x % period
    h[phase < period * duty] = 1.0
    return project_feasible(h, n)

def smooth_objective(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha)

    return x.sum() - n / 2.0

    return np.ones(n)

where x_i = i·dx.

Now, h(x_i + k) = h(x_i + m·dx) where m = k/dx.

In the array, h(x_i + m·dx) = h[i+m] (if in bounds).

So the sum is: Σ_i h[i] · (1-h[i+m]) · dx = dx · Σ_i h[i] · (1-h[i+m])

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    lse = mx + np.log(np.sum(np.exp(sh - mx)))
    return m + lse / alpha

    return x.sum() - n / 2.0
