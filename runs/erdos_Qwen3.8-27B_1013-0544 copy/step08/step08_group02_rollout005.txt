def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(60):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def smooth_components(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    s = float(e.sum())
    p = e / s
    return (m + float(np.log(s)) / alpha) * (2.0 / n), p, c

def grad_weights(x, n, p):
    g = np.zeros(n, dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-13:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n - d])
            g[:n - d] -= w * x[d:]
        else:
            e = -d
            g[:n - e] += w * (1.0 - x[e:])
            g[e:] -= w * x[:n - e]
    return (2.0 / n) * g

def smooth_val(x, n, alpha):
    return smooth_components(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    _, p, _ = smooth_components(x, n, alpha)
    return grad_weights(x, n, p)

def slsqp_opt(h, n, alpha, maxiter, deadline):
    from scipy.optimize import minimize
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    best = h.copy()
    best_c = compute_c5(h, n)
    try:
        res = minimize(smooth_val, h, args=(n, alpha), method='SLSQP',
                       jac=smooth_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        x = np.asarray(res.x, dtype=float)
        if np.all(np.isfinite(x)):
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best, best_c = cur, cc
    except Exception:
        pass
    return best, best_c

    off = n - 1

        # d = 0

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low, high = float(z.min()) - 2.0, float(z.max()) + 2.0
    for _ in range(60):
        tau = 0.5 * (low + high)
        if np.clip(z - tau, 0.0, 1.0).sum() > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def smooth_components(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    s = float(e.sum())
    p = e / s
    return (m + float(np.log(s)) / alpha) * (2.0 / n), p, c

def grad_weights(x, n, p):
    g = np.zeros(n, dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-13:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n - d])
            g[:n - d] -= w * x[d:]
        else:
            e = -d
            g[:n - e] += w * (1.0 - x[e:])
            g[e:] -= w * x[:n - e]
    return (2.0 / n) * g

def smooth_val(x, n, alpha):
    return smooth_components(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    _, p, _ = smooth_components(x, n, alpha)
    return grad_weights(x, n, p)

def compute_A(h, n):
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    A[:, off] = 1.0 - 2.0 * h
    for d in range(1, n):
        A[:n-d, off+d] = 1.0 - h[d:]
        A[d:, off+d] += -h[:n-d]
        A[d:, off-d] = 1.0 - h[:n-d]
        A[:n-d, off-d] += -h[d:]
    return A

def slsqp_opt(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    best, best_c = h.copy(), compute_c5(h, n)
    try:
        res = minimize(smooth_val, h, args=(n, alpha), method='SLSQP',
                       jac=smooth_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        x = np.asarray(res.x, dtype=float)
        if np.all(np.isfinite(x)):
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best, best_c = cur, cc
    except Exception:
        pass
    return best, best_c

def pair_search_v2(h, n, max_iter, deadline, rng, grid=50):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    off = n - 1
    A = compute_A(h, n)
    accepts, no_imp = 0, 0
    max_no_imp = max(400, max_iter // 3)
    g = np.zeros(n)
    
    def recompute_A():
        nonlocal A
        A = compute_A(h, n)
    
    for trial in range(max_iter):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        if trial % 80 == 0:
            alpha = 300.0
            z = alpha * (c - cur_max)
            e = np.exp(z - z.max())
            p = e / e.sum()
            g = A @ p
        if trial % 11 == 0:
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
        else:
            best_v, best_i, best_j = -1, 0, 1
            for _ in range(6):
                ii, jj = int(rng.randint(n)), int(rng.randint(n))
                if ii == jj:
                    jj = (jj + 1) % n
                v = abs(g[ii] - g[jj])
                if v > best_v:
                    best_v, best_i, best_j = v, ii, jj
            i, j = best_i, best_j
        hi, hj = h[i], h[j]
        S = hi + hj
        lo, hiT = max(0.0, S - 1.0), min(1.0, S)
        if hiT - lo <= 1e-14:
            no_imp += 1
            continue
        us = np.linspace(lo - hi, hiT - hi, grid)
        B = A[i] - A[j]
        lines = c + us[:, None] * B[None, :]
        u2 = us * us
        lines[:, off] -= 2.0 * u2
        lines[:, off + i - j] += u2
        lines[:, off + j - i] += u2
        vals = lines.max(axis=1)
        idx = int(np.argmin(vals))
        if vals[idx] < cur_max - 1e-10:
            u = float(us[idx])
            t = float(h[i] + u)
            h[i], h[j] = t, S - t
            c = c + u * B
            uv = u * u
            c[off] -= 2.0 * uv
            c[off + i - j] += uv
            c[off + j - i] += uv
            accepts += 1
            no_imp = 0
            if accepts % 15 == 0:
                recompute_A()
            if accepts % 60 == 0:
                c = np.correlate(h, 1.0 - h, mode='full')
                if accepts > 200:
                    max_no_imp = max(200, max_iter // 4)
        else:
            no_imp += 1
        if no_imp > max_no_imp:
            break
    return h, compute_c5(h, n)

def apply_symmetries(h, n):
    c = compute_c5(h, n)
    hs = project_feasible(0.5 * (h + h[::-1]), n)
    cs = compute_c5(hs, n)
    if cs < c:
        h, c = hs, cs
    hc = 1.0 - h
    cs = compute_c5(hc, n)
    if cs < c:
        h, c = hc, cs
    hsc = project_feasible(0.5 * (h + 1.0 - h[::-1]), n)
    cs = compute_c5(hsc, n)
    if cs < c:
        h, c = hsc, cs
    return h, c

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def optimize_at_n(h, n, deadline, rng, quick=False):
    h, c = apply_symmetries(h, n)
    if not quick and time.time() < deadline - 40.0:
        for alpha, mx in [(30, 40), (150, 35), (800, 30), (4000, 25), (20000, 20)]:
            if time.time() > deadline - 20.0:
                break
            h, c = slsqp_opt(h, n, alpha, mx)
    rem = deadline - time.time()
    if rem > 5.0 and n <= 800:
        mt = int(min(8000 if not quick else 3000, max(500, rem * 25)))
        h, c = pair_search_v2(h, n, mt, deadline, rng, grid=55)
    h = project_feasible(h, n)
    return h, compute_c5(h, n)
