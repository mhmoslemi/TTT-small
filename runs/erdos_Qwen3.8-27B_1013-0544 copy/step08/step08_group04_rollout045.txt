import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 1e-15:
            h = h + (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 1e-15:
            h = h + (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-12:
        idx = np.where(h < 1.0 - 1e-14)[0]
        if idx.size:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-12:
        idx = np.where(h > 1e-14)[0]
        if idx.size:
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def corr_array(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def do_coord_pass(h, n, order, deadline):
    c = corr_array(h, n)
    center = n - 1
    L = 2 * n - 1
    for p in order:
        if time.time() > deadline:
            break
        hp = float(h[p])
        lo = min(p, center - p)
        hi = max(center + p, 2 * center - p)
        seg = hi - lo + 1
        s_arr = center - np.arange(lo, hi + 1, dtype=int)
        j = p + s_arr
        i = p - s_arr
        a = np.zeros(seg, dtype=float)
        vj = (j >= 0) & (j < n)
        vi = (i >= 0) & (i < n)
        if np.any(vj):
            a[vj] += 1.0 - h[j[vj]]
        if np.any(vi):
            a[vi] -= h[i[vi]]
        off = center - lo
        ns = seg - 1
        cseg = np.empty(ns, dtype=float)
        asg = np.empty(ns, dtype=float)
        if off > 0:
            cseg[:off] = c[lo:center]
            asg[:off] = a[:off]
        if hi > center:
            cseg[off:] = c[center + 1:hi + 1]
            asg[off:] = a[off + 1:]
        qconst = c[center] - hp * (1.0 - hp)
        unaff = 0.0
        if lo > 0:
            unaff = max(unaff, float(c[:lo].max()))
        if hi + 1 < L:
            unaff = max(unaff, float(c[hi + 1:].max()))
        pos = asg > 1e-12
        neg = asg < -1e-12
        b = cseg - asg * hp
        def check_t(t):
            if t < unaff - 1e-13:
                return False, (0.0, 1.0)
            ll = 0.0
            lh = 1.0
            if np.any(pos):
                v = float(np.min((t - b[pos]) / asg[pos]))
                if v < lh:
                    lh = v
            if np.any(neg):

                if v > ll:
                    ll = v
            if ll > 1.0 + 1e-12 or lh < -1e-12:
                return False, (0.0, 1.0)
            g = t - qconst
            if g < -1e-12:
                return False, (0.0, 1.0)
            if g <= 0.25:

                sq = np.sqrt(disc)
                r1 = 0.5 - 0.5 * sq
                r2 = 0.5 + 0.5 * sq

                L2 = max(r2, ll)
                U2 = min(1.0, lh)

            else:

        t_opt = t_hi
        feasible, interval = check_t(t_opt)
        best_x = hp

                c[center + 1:hi + 1] += asg[off:] * dx
            c[center] = qconst + best_x * (1.0 - best_x)
            h[p] = best_x
    h = project_feasible(h, n)
    return h, float(corr_array(h, n).max())

def optimize_cd(h, n, max_passes, deadline, rng):
    h = project_feasible(h, n)
    best_c = compute_c(h, n)
    best_h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        h, cur_raw = do_coord_pass(h, n, order, deadline)
        c_val = compute_c(h, n)
        if c_val < best_c - 1e-14:
            best_c = c_val
            best_h = h.copy()
    return best_h, best_c

def fourier_perturb(base, n, rng, n_modes=8, amplitude=0.15):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    t = x - 1.0
    h = base.copy()
    for k in range(1, n_modes + 1):
        amp = rng.uniform(-amplitude / k, amplitude / k)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(k * np.pi * t + phase)
    return project_feasible(h, n)

def binary_from_h(h, n):
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    h2 = np.zeros(n)
    h2[idx] = 1.0
    return h2

def block_flip(base, n, rng):
    h0 = base.copy()
    blen = rng.randint(15, max(16, n // 4))
    bstart = rng.randint(0, max(1, n - blen))
    h0[bstart:bstart + blen] = 1.0 - h0[bstart:bstart + blen]
    return project_feasible(h0, n)
