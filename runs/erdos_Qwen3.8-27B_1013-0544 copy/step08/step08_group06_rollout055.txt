```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c_at_lag(h, n, delta):
    """Gradient of c[k] where k = delta + (n-1), i.e., shift = delta."""
    # c[k] = sum_{i: i, i-delta in [0,n-1]} h[i](1-h[i-delta])
    # = sum_{i valid} h[i] - sum_{i valid} h[i]h[i-delta]
    grad = np.zeros(n)
    
    if delta == 0:
        # c = sum h[i](1-h[i]) = sum h[i] - sum h[i]^2
        # d/dh[j] = 1 - 2*h[j]
        valid = np.ones(n, dtype=bool)
        grad[valid] = 1.0 - 2.0 * h[valid]
    elif delta > 0:
        # i and i-delta in [0,n-1] means i in [delta, n-1]
        valid = np.zeros(n, dtype=bool)
        valid[delta:] = True
        # First term: sum_{i=delta}^{n-1} h[i], grad = 1 for i in [delta, n-1]
        grad[delta:] += 1.0
        # Second term: sum_{i=delta}^{n-1} h[i]*h[i-delta]
        # d/dh[j]: if j is in [delta, n-1] (as i): contribution h[j-delta]
        #           if j is in [0, n-1-delta] (as i-delta): contribution h[j+delta]
        grad[delta:] -= h[:n-delta]  # h[j-delta] for j in [delta, n-1]
        grad[:n-delta] -= h[delta:]  # h[j+delta] for j in [0, n-1-delta]
    else:
        # delta < 0, let d = -delta > 0
        # i and i-delta = i+d in [0,n-1] means i in [0, n-1-d]
        d = -delta
        valid = np.zeros(n, dtype=bool)
        valid[:n-d] = True
        grad[:n-d] += 1.0
        # Second term: sum_{i=0}^{n-1-d} h[i]*h[i+d]
        # d/dh[j]: if j in [0, n-1-d] (as i): h[j+d]
        #           if j in [d, n-1] (as i+d): h[j-d]
        grad[:n-d] -= h[d:]  # h[j+d] for j in [0, n-1-d]
        grad[d:] -= h[:n-d]  # h[j-d] for j in [d, n-1]
    
    return grad

    h = h.copy()
    best_h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(c.max())
    
    # Adaptive step size
    alpha = 1.0 / np.sqrt(n)
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        
        # Find top lags
        top_indices = np.argsort(c)[-top_k:]
        # Weight: give more weight to larger c values
        c_top = c[top_indices]
        w = (c_top - c_top.min() + 0.1) ** 2
        w /= w.sum()

        for idx, wi in zip(top_indices, w):

            grad += wi * grad_c_at_lag(h, n, delta)
        
        # Step

        h_new = project_feasible(h_new, n)
        
        # Evaluate
        c = np.correlate(h_new, 1.0 - h, mode='full')
        cur_c = float(c.max())
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_new.copy()
        
        # Adaptive step size: decrease if no improvement
        # Barzilai-Borwein style or simple decay
        alpha *= 0.999
        
        h = h_new
    
    return best_h, best_c
```

```python
def weighted_autocorr_grad(h, n, p_weights):
    """Compute gradient of -sum_m p_m * c[m] w.r.t. h.
    p_weights is array of length 2n-1, indexed by m=0,...,2n-2.
    Returns the direction to ADD to h (i.e., -gradient of the objective).
    """
    # Direction d[j] = sum_m p[m] * (contribution of lag delta_m to d/dh[j])
    # For lag delta (shift), contribution to d/dh[j] is:
    #   h[j-delta] (if valid) + h[j+delta] (if valid)  [for delta != 0]
    #   2*h[j] [for delta = 0]
    # But we're minimizing sum p[m]*c[m], and c[m] involves (1-h), so:
    # c[m] = sum_valid h[i] - sum_valid h[i]*h[i-delta_m]
    # d(c[m])/d(h[j]) = I[j valid for m] - h[j-delta_m]*I - h[j+delta_m]*I
    # 
    # For the gradient descent direction on sum p[m]*c[m]:
    # The "const" part (sum h[i] terms) gives a uniform gradient, which will be
    # handled by the projection (since we fix the sum).
    # The quadratic part gives: -sum_m p[m] * (h[j-delta_m] + h[j+delta_m])
    # So the direction to DECREASE the objective is: h + alpha * sum_m p[m] * (h[j-delta_m] + h[j+delta_m])
    
    direction = np.zeros(n)
    # For each lag index m (0 to 2n-2), delta = m - (n-1)
    for m in range(2*n-1):
        if p_weights[m] < 1e-15:
            continue
        delta = m - (n-1)
        if delta == 0:
            direction += p_weights[m] * 2.0 * h
        elif delta > 0:
            direction[delta:] += p_weights[m] * h[:n-delta]
            direction[:n-delta] += p_weights[m] * h[delta:]
        else:
            d = -delta
            direction[:n-d] += p_weights[m] * h[d:]
            direction[d:] += p_weights[m] * h[:n-d]
    
    return direction
```

```python
            direction = weighted_autocorr_grad(h, n, p)
            h_new = h + alpha * direction
            h_new = project_feasible(h_new, n)
            
            # Check improvement
            c = np.correlate(h_new, 1-h_new, mode='full')
            cur_c = float(c.max())
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_new.copy()
            
            h = h_new
        
        # Update p (multiplicative weight)
        c = np.correlate(h, 1-h, mode='full')
        # p[m] proportional to exp(beta * c[m])
        # Normalize to sum to 1
        c_shift = c - c.max()
        p = np.exp(beta * c_shift)
        p /= p.sum()
        
        # Decay step size
        alpha *= 0.95
    
    return best_h, best_c
```

Now, for the gradient computation in the MW method:
c[m] = ∑_{valid i} h[i] * (1-h[i-δ]) where δ = m-(n-1)

For the "direction to decrease c[m]":
∂c[m]/∂h[j] = I[j valid for m] - h[j-δ]·I[j-δ valid] - h[j+δ]·I[j+δ valid]

The "const" part (I[j valid]) is the same for all j in the valid range, and since we have the constraint ∑h = n/2, this just shifts everything uniformly and gets absorbed by the projection. So the meaningful gradient direction is:
-g_m[j] = h[j-δ]·I + h[j+δ]·I (the part that varies with j)

And to DECREASE c[m], we INCREASE h in the direction -∂c[m]/∂h, which means:
h ← h + α * (h[j-δ] + h[j+δ]) (the autocorrelation direction)

This makes sense: to decrease the overlap at lag δ, we want to increase the autocorrelation at lag δ, which means making h more "correlated" with its shifted version.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def autocorr_direction(h, n, p_w):
    """Compute direction to increase weighted autocorrelation (decrease weighted overlap)."""
    direction = np.zeros(n)
    for m in range(2 * n - 1):
        w = p_w[m]
        if w < 1e-15:
            continue
        delta = m - (n - 1)
        if delta == 0:
            direction += w * 2.0 * h
        elif delta > 0:
            direction[delta:] += w * h[:n - delta]
            direction[:n - delta] += w * h[delta:]
        else:
            d = -delta
            direction[:n - d] += w * h[d:]
            direction[d:] += w * h[:n - d]
    return direction

def mw_optimize(h, n, max_outer=300, inner_steps=5, beta=50.0, deadline=None):
    h = h.copy()
    best_h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(c.max())
    p = np.ones(2 * n - 1) / (2 * n - 1)
    alpha = 3.0 / n
    stall_count = 0
    for outer in range(max_outer):
        if deadline is not None and time.time() > deadline:
            break
        for _ in range(inner_steps):
            if deadline is not None and time.time() > deadline:
                break
            direction = autocorr_direction(h, n, p)
            d_norm = np.linalg.norm(direction)
            if d_norm < 1e-12:
                break
            h_new = h + alpha * direction / d_norm
            h_new = project_feasible(h_new, n)
            c = np.correlate(h_new, 1.0 - h_new, mode='full')
            cur_c = float(c.max())
            if cur_c < best_c - 1e-10:
                best_c = cur_c
                best_h = h_new.copy()
                stall_count = 0
            else:
                stall_count += 1
            h = h_new
        if stall_count > inner_steps * 3:
            alpha *= 0.7
            stall_count = 0
            if alpha < 1e-6:
                break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_shift = c - c.max()
        p = np.exp(beta * c_shift)
        p /= p.sum()
        alpha *= 0.995
    return best_h, best_c

def coord_cd(h, n, max_passes=5000, deadline=None, grid_size=100):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        idx_order = range(n) if p % 2 == 0 else range(n - 1, -1, -1)
        improved = False
        for idx in idx_order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 > 1.0 - 1e-10:
                continue
            lo_d = max(-h0, 0.0) if h0 < 1e-10 else -h0
            hi_d = min(1.0 - h0, 1.0) if h0 > 1.0 - 1e-10 else 1.0 - h0
            if hi_d - lo_d < 1e-12:
                continue
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            M = len(ms)
            a = np.zeros(M)
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                tmp = float(c[:m_lo].max())
                if tmp > unaff_max:
                    unaff_max = tmp
            if m_hi < 2 * n - 2:
                tmp = float(c[m_hi + 1:].max())
                if tmp > unaff_max:
                    unaff_max = tmp
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if has_diag:
                nv[:, diag_idx] -= d_grid[:, None] ** 2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            best_i = int(np.argmin(vals))
            best_d = d_grid[best_i]
            best_val = float(vals[best_i])
            if 0 < best_i < grid_size - 1:
                y = vals[best_i - 1:best_i + 2]
                x = d_grid[best_i - 1:best_i + 2]
                denom = (y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2] * (x[0] - x[1]))
                if abs(denom) > 1e-15:
                    xv = 0.5 * (y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2] * (x[0] - x[1])) / denom
                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv * a
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag_idx] -= xv ** 2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val = v1
                            best_d = xv
            if abs(best_d) > 1e-10 and best_val < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[ms[diag_idx]] -= best_d ** 2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True
        if not improved:
            break
        if (p + 1) % 500 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def perturb_and_optimize(h, n, rng, deadline, strength=0.05):
    h_pert = h + rng.randn(n) * strength
    h_pert = project_feasible(h_pert, n)
    h_opt, c_opt = mw_optimize(h_pert, n, max_outer=50, inner_steps=3, beta=100, deadline=deadline)
    return h_opt, c_opt
```
