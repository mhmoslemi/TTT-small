def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_deep(h, n, alphas, maxiter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:

                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18})

            new_c = compute_c5(new_h, n)

    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))

                y0, y1, y2 = vals[best_i-1], vals[best_i], vals[best_i+1]
                x0, x1, x2 = d_grid[best_i-1], d_grid[best_i], d_grid[best_i+1]
                denom = (y0*(x1-x2) + y1*(x2-x0) + y2*(x0-x1))
                if abs(denom) > 1e-18:
                    xv = 0.5*(y0*(x1-x2) + y1*(x2-x0) + y2*(x0-x1))/denom
                    if x0 < xv < x2:
                        nv1 = c_aff + xv * a
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag_pos] -= xv**2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-15:
                            best_val = v1
                            best_d = xv
            if abs(best_d) > 1e-12 and best_val < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[ms[diag_pos]] -= best_d**2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def try_interpolate_optimize(best_h, best_c, n_try, deadline, grid_size=150):
    x_orig = np.linspace(0, 2, len(best_h))
    x_new = np.linspace(0, 2, n_try)
    h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
    h_opt, c_opt = coord_cd(h_int, n_try, max_passes=50000, deadline=deadline, grid_size=grid_size)
    if c_opt < best_c:
        return h_opt, c_opt
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(400, 0.5)
        n = 400
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    # Phase 1: Brief SLSQP
    t1 = t0 + min(60, budget_s * 0.05)
    if time.time() < t1:
        h1, c1 = slsqp_deep(h, n, [500, 5000], maxiter=300)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Coordinate descent, medium grid
    t2 = t1 + min(250, budget_s * 0.2)
    h2, c2 = coord_cd(best_h, n, max_passes=100000, deadline=t2, grid_size=150)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Coordinate descent, fine grid
    t3 = t2 + min(250, budget_s * 0.2)
    h3, c3 = coord_cd(best_h, len(best_h), max_passes=100000, deadline=t3, grid_size=300)
    if c3 < best_c:
        best_h, best_c = h3, c3
    # Phase 4: Try higher resolution
    t4 = t3 + min(200, budget_s * 0.18)
    if time.time() < t4 - 30:
        n2 = min(len(best_h) * 2, 1000)
        if n2 > len(best_h):
            best_h, best_c = try_interpolate_optimize(best_h, best_c, n2, t4, grid_size=100)
    # Phase 5: Try lower resolution (sometimes better)
    t5 = t4 + min(100, budget_s * 0.08)
    if time.time() < t5 - 20:
        n3 = max(len(best_h) // 2, 250)
        if n3 < len(best_h):
            bh2, bc2 = try_interpolate_optimize(best_h, best_c, n3, t5, grid_size=200)
            if bc2 < best_c:
                best_h, best_c = bh2, bc2
    # Phase 6: Final high-precision coordinate descent
    h6, c6 = coord_cd(best_h, len(best_h), max_passes=100000, deadline=deadline, grid_size=500)
    if c6 < best_c:
        best_h, best_c = h6, c6
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
