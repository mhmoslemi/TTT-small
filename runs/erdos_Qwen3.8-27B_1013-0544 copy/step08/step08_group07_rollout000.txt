import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(x.shape)

def coord_descent(h, n, max_passes, deadline, grid_size=64, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    dac = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(corr.max()) * dac
    best_h = h.copy()
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_a_lo = max(0, p)
            m_a_hi = min(2 * n - 2, p + n - 1)
            m_b_lo = max(0, n - 1 - p)
            m_b_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_lo = min(m_a_lo, m_b_lo)
            m_hi = max(m_a_hi, m_b_hi)
            if m_lo >= m_hi:
                continue
            ms = np.arange(m_lo, m_hi + 1)
            coeffs = np.zeros(len(ms))
            idx_a = p - ms + n - 1
            va = (idx_a >= 0) & (idx_a < n)
            coeffs[va] += (1.0 - h[idx_a[va]])
            idx_b = ms - p + n - 1
            vb = (idx_b >= 0) & (idx_b < n)
            coeffs[vb] -= h[idx_b[vb]]
            diag = (ms == (n - 1))
            c_aff = corr[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = float(corr[:m_lo].max())
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(corr[m_hi + 1:].max()))
            cur_max = max(unaff, float(c_aff.max()))
            d_vals = np.linspace(-h0 * 0.999, (1.0 - h0) * 0.999, grid_size)
            nv = c_aff[np.newaxis, :] + d_vals[:, np.newaxis] * coeffs[np.newaxis, :]
            if np.any(diag):
                nv[:, diag] -= d_vals[:, np.newaxis] ** 2
            new_maxes = np.maximum(unaff, nv.max(axis=1))
            idx = int(np.argmin(new_maxes))
            best_d = d_vals[idx]
            best_val = new_maxes[idx]
            if best_val < cur_max - 1e-14:
                h[p] = h0 + best_d
                improved = True
                corr[ms] = c_aff + best_d * coeffs
                if np.any(diag):
                    corr[ms[diag]] -= best_d * best_d
        if not improved:
            break
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_c = float(corr.max()) * dac
        if new_c < best_c:
            best_c = new_c
            best_h = h.copy()
    return best_h, best_c
