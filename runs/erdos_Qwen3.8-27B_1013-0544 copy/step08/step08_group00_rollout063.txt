def exact_cd_pass(h, n, deadline, order=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    n_c = 2*n - 1  # length of c
    
    if order is None:
        order = np.arange(n)
    
    for p in order:
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-15 or h0 > 1.0 - 1e-15:
            continue
        
        # Affected range
        lo1 = max(0, p)
        hi1 = min(n_c-1, p + n - 1)
        lo2 = max(0, n - 1 - p)
        hi2 = min(n_c-1, 2*n - 2 - p)
        lo = min(lo1, lo2)
        hi = max(hi1, hi2)
        
        ms = np.arange(lo, hi + 1)
        idx = ms - lo  # local indices
        
        # Compute a coefficients
        j_first = p - ms + (n - 1)
        valid_first = (j_first >= 0) & (j_first < n)
        j_second = p + ms - (n - 1)
        valid_second = (j_second >= 0) & (j_second < n)
        
        a = np.zeros(len(ms))
        a[valid_first] += (1.0 - h[j_first[valid_first]])
        a[valid_second] -= h[j_second[valid_second]]
        
        has_diag = (lo <= n-1 <= hi)
        diag_pos = (n-1) - lo if has_diag else -1
        
        # Unaffected max
        unaff_max = 0.0
        if lo > 0:
            unaff_max = float(c[:lo].max())
        if hi < n_c - 1:
            unaff_max = max(unaff_max, float(c[hi+1:].max()))
        
        c_base = c[ms].copy()
        cur_max = max(unaff_max, float(c_base.max()))
        
        # Bisection for optimal delta
        d_lo = -h0
        d_hi = 1.0 - h0
        
        for _ in range(45):
            d_mid = (d_lo + d_hi) / 2.0
            new_vals = c_base + a * d_mid
            if has_diag:
                new_vals[diag_pos] -= d_mid * d_mid
            
            f_val = max(unaff_max, float(new_vals.max()))
            
            # Find active set
            active = new_vals >= f_val - 1e-13
            if not np.any(active):
                # f_val determined by unaff_max
                # In this case, we want to minimize f_val which is fixed
                # So any delta that keeps max(new_vals) <= unaff_max is fine
                # Just pick the midpoint that minimizes the max
                # Actually if f_val = unaff_max and unaff_max > max(new_vals),
                # then we're already at the minimum for this coordinate
                break
            
            if has_diag:
                # For diagonal: derivative is a[diag] - 2*d_mid
                active_grads = a[active].copy()
                if diag_pos >= 0 and active[diag_pos]:
                    active_grads[diag_pos] = a[diag_pos] - 2.0 * d_mid
            else:
                active_grads = a[active]
            
            g_lo = float(active_grads.min())
            g_hi = float(active_grads.max())
            
            if g_lo > 1e-13:
                d_hi = d_mid
            elif g_hi < -1e-13:
                d_lo = d_mid
            else:
                break
        
        best_d = (d_lo + d_hi) / 2.0
        best_d = float(np.clip(best_d, -h0, 1.0 - h0))
        
        if abs(best_d) > 1e-14:
            # Update
            h[p] = h0 + best_d
            c[ms] += best_d * a
            if has_diag:
                c[lo + diag_pos] -= best_d * best_d
    
    return h, c

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def exact_cd_pass(h, n, deadline, order=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    nc = 2 * n - 1
    if order is None:
        order = np.arange(n)
    total_change = 0.0
    for p in order:
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-15 or h0 > 1.0 - 1e-15:
            continue
        lo1 = max(0, p)
        hi1 = min(nc - 1, p + n - 1)
        lo2 = max(0, n - 1 - p)
        hi2 = min(nc - 1, 2 * n - 2 - p)
        lo = min(lo1, lo2)
        hi = max(hi1, hi2)
        ms = np.arange(lo, hi + 1)
        jf = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        jb = p + ms - (n - 1)
        vb = (jb >= 0) & (jb < n)
        a = np.zeros(len(ms))
        a[vf] += (1.0 - h[jf[vf]])
        a[vb] -= h[jb[vb]]
        has_diag = (lo <= n - 1 <= hi)
        dpos = (n - 1) - lo if has_diag else -1
        unaff_max = 0.0
        if lo > 0:
            unaff_max = float(c[:lo].max())
        if hi < nc - 1:
            unaff_max = max(unaff_max, float(c[hi + 1:].max()))
        c_base = c[ms]
        d_lo_b = -h0
        d_hi_b = 1.0 - h0
        for _ in range(50):
            dm = (d_lo_b + d_hi_b) / 2.0
            nv = c_base + a * dm
            if has_diag:
                nv[dpos] -= dm * dm
            fv = max(unaff_max, float(nv.max()))
            active = nv >= fv - 1e-12
            if not np.any(active):
                break
            if has_diag:
                ag = a[active].copy()
                if dpos >= 0 and active[dpos]:
                    ag[dpos] = a[dpos] - 2.0 * dm
            else:
                ag = a[active]
            gmin = float(ag.min())
            gmax = float(ag.max())
            if gmin > 1e-12:
                d_hi_b = dm
            elif gmax < -1e-12:
                d_lo_b = dm
            else:
                break
        bd = float(np.clip((d_lo_b + d_hi_b) / 2.0, -h0, 1.0 - h0))
        if abs(bd) > 1e-14:
            h[p] = h0 + bd
            c[ms] += bd * a
            if has_diag:
                c[lo + dpos] -= bd * bd
            total_change += abs(bd)
    return h, c, total_change

**Subgradient method:**
1. Compute c = correlate(h, 1-h)
2. Find active set: S = {k : c[k] = max(c)} (or close to max)
3. Compute subgradient: g = (1/|S|) * ∑_{k∈S} ∂c[k]/∂h
4. Take a step: h_new = project(h - step_size * g)
5. Repeat

The subgradient of c[k] with respect to h[i] is:
∂c[k]/∂h[i] = ∂/∂h[i] [∑_j h[j]*(1-h[j-(k-n+1)])]

This is computed similarly to the CD coefficients. For a given k, the dependence on h[i] comes from:
- h[i] as first factor: (1-h[i-(k-n+1)]) if valid
- h[i] in (1-h[j]): -h[j] where j = i+(k-n+1), i.e., the term is -h[i+(k-n+1)] if valid

So ∂c[k]/∂h[i] = (1-h[i-k+n-1])*I[valid] - h[i+k-n+1]*I[valid]

The subgradient of max_k c[k] is the average of ∂c[k]/∂h over active k.

This gives a vector of length n. Computing it takes O(n * |S|) where |S| is the number of active k's. If many k's are active (which happens near the optimum), this is O(n²).

The projected subgradient step is:
h_new = project(h - α * g)

where α is the step size. We can use a backtracking line search to find the right α.

This is O(n²) per iteration, same as one CD pass. But it updates all coordinates simultaneously, which might converge faster.

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def exact_cd_pass(h, n, deadline, order=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    nc = 2 * n - 1
    if order is None:
        order = np.arange(n)
    total_change = 0.0
    for p in order:
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-15 or h0 > 1.0 - 1e-15:
            continue
        lo1 = max(0, p)
        hi1 = min(nc - 1, p + n - 1)
        lo2 = max(0, n - 1 - p)
        hi2 = min(nc - 1, 2 * n - 2 - p)
        lo = min(lo1, lo2)
        hi = max(hi1, hi2)

        jf = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        jb = p + ms - (n - 1)
        vb = (jb >= 0) & (jb < n)
        a = np.zeros(len(ms))
        a[vf] += (1.0 - h[jf[vf]])
        a[vb] -= h[jb[vb]]
        has_diag = (lo <= n - 1 <= hi)
        dpos = (n - 1) - lo if has_diag else -1
        unaff_max = 0.0

def subgrad_step(h, n, alpha, deadline):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(c.max())
    tol = 1e-4 * max(1.0, cmax)
    active_k = np.where(c >= cmax - tol)[0]
    if len(active_k) == 0:
        return h, cmax
    g = np.zeros(n)
    n_act = len(active_k)
    for k in active_k:
        s = int(k - (n - 1))
        if s >= 0:
            idx_f = np.arange(s, n)
            g[idx_f] += (1.0 - h[idx_f - s])
            idx_s = np.arange(0, max(1, n - s))

            g[idx_f] += (1.0 - h[idx_f - s])

    g /= n_act
    h_new = project_feasible(h - alpha * g, n)

def smooth_slsqp_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    shifted = alpha * (c - m)
    mx = float(shifted.max())
    lse = mx + float(np.log(np.sum(np.exp(shifted - mx))))
    return (m + lse / alpha) * (2.0 / n)

def smooth_slsqp_sum_con(x, n):
    return x.sum() - n / 2.0

def smooth_slsqp_sum_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': smooth_slsqp_sum_con, 'jac': smooth_slsqp_sum_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_slsqp_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
