            if m_lo > 0: unaff_max = max(unaff_max, float(c_prev[:m_lo].max()))
            if m_hi < 2*n-2: unaff_max = max(unaff_max, float(c_prev[m_hi+1:].max()))
            diag = (ms == (n-1))
            best_d = 0.0
            best_val = current_max
            # Two-level grid search
            nc = 25
            d_coarse = np.linspace(-h0, 1.0-h0, nc)
            for d in d_coarse:
                if abs(d) < 1e-14: continue
                nv = c_aff + d*a
                if np.any(diag): nv[diag] -= d*d
                val = max(unaff_max, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            if best_d != 0.0:
                span = (1.0-h0)/nc
                lo = max(-h0, best_d - span)
                hi = min(1.0-h0, best_d + span)
                d_fine = np.linspace(lo, hi, grid_size)
                for d in d_fine:
                    if abs(d) < 1e-14: continue
                    nv = c_aff + d*a
                    if np.any(diag): nv[diag] -= d*d
                    val = max(unaff_max, float(nv.max()))
                    if val < best_val:
                        best_val = val
                        best_d = d
            if abs(best_d) > 1e-14 and best_val < current_max:
                h[p] = float(np.clip(h0+best_d, 0.0, 1.0))
                improved = True
                c_prev[ms] = c_aff + best_d*a
                if np.any(diag): c_prev[ms[diag]] -= best_d*best_d
                if best_val < current_max:
                    current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
    return h, compute_c5(h, n)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
