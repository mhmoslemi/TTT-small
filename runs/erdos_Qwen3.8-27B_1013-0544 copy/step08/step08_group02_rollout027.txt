    best_i = 0
    prev_v = 1e18
    prev2_v = 1e18
    prev_u = us[0]
    prev2_u = us[0]
    for idx in range(n_g):
        v = f_at(us[idx])
        if v < best_v:
            best_v = v
            best_u = us[idx]
            best_i = idx
        prev2_v, prev2_u = prev_v, prev_u
        prev_v, prev_u = v, us[idx]
    if best_v >= cmax - 1e-10:
        return None, None
    # Parabolic refinement
    if 0 < best_i < n_g - 1:
        u0, u1, u2 = us[best_i-1], us[best_i], us[best_i+1]
        f0, f1, f2 = f_at(u0), best_v, f_at(u2)
        den = (u1 - u0) * (f1 - f2) - (u1 - u2) * (f1 - f0)
        if abs(den) > 1e-15:
            num = 0.5 * ((f1 - f2) * (u1 - u0)**2 - (f1 - f0) * (u1 - u2)**2)
            u_par = u1 - num / den
            if lo_u - 1e-12 < u_par < hi_u + 1e-12:
                u_par = float(np.clip(u_par, lo_u, hi_u))
                fp = f_at(u_par)
                if fp < best_v:
                    best_v = fp
                    best_u = u_par
    return best_v, best_u

def intensive_pair_search(h, n, max_tries, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    A = compute_A(h, n)
    off = n - 1
    cmax = float(c.max())
    accepts = 0
    no_improve = 0
    for trial in range(max_tries):
        if time.time() > deadline:
            break
        cmax = float(c.max())
        tol = max(0.015, 0.001 * cmax)
        active = np.flatnonzero(c >= cmax - tol)
        if trial % 7 == 6 or active.size <= 2:
            i = int(rng.randint(n)); j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
                if i == j:
                    continue
        else:
            kk = int(active[int(rng.randint(active.size))])
            d = kk - off
            if d >= 0:
                m = n - d
                if m <= 0:
                    continue
                b = int(rng.randint(m)); i = b + d; j = b
            else:
                e_ = -d
                m = n - e_
                if m <= 0:
                    continue
                b = e_ + int(rng.randint(m)); i = b - e_; j = b
        S = h[i] + h[j]
        lo_u = max(0.0, S - 1.0) - h[i]
        hi_u = min(1.0, S) - h[i]
        if hi_u - lo_u <= 1e-14:
            continue
        B = A[i] - A[j]
        nv, u = pair_line_search(c, B, off, i, j, lo_u, hi_u, cmax)
        if nv is None:
            no_improve += 1
            continue
        old = h[i] + h[j]
        t = float(np.clip(h[i] + u, 0.0, 1.0))
        h[i] = t
        h[j] = old - t
        # Update c
        c = c + u * B
        c[off] -= 2.0 * u * u
        di = off + i - j
        dj = off + j - i
        if 0 <= di < len(c):
            c[di] += u * u
        if 0 <= dj < len(c):
            c[dj] += u * u
        cmax = float(c.max())
        accepts += 1
        no_improve = 0
        A = compute_A(h, n)
        if accepts % 15 == 0:
            c = np.correlate(h, 1.0 - h, mode='full')
        if no_improve > max(400, max_tries // 3):
            break
    return h, compute_c5(h, len(h))

def slsqp_stage(h, n, bounds, cons, plans, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work_h = best_h.copy()
    for alpha, mx in plans:
        if time.time() > deadline - 1.0:
            break
        try:
            res = minimize(smooth_val, work_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_grad, bounds=bounds, constraints=cons,
                           options={'maxiter': mx, 'ftol': 1e-14, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best_h, best_c = cur, cc
            work_h = cur if cc <= best_c + 1e-8 else best_h
        except Exception:
            pass
    return best_h, best_c

def lbfgs_stage(h, n, bounds, plans, mu, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work_h = best_h.copy()
    for alpha, mx in plans:
        if time.time() > deadline - 1.0:
            break
        try:
            res = minimize(pen_val, work_h, args=(n, alpha, mu), method='L-BFGS-B',
                           jac=pen_jac, bounds=bounds,
                           options={'maxiter': mx, 'ftol': 1e-14, 'maxls': 20, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best_h, best_c = cur, cc
            work_h = cur if cc <= best_c + 1e-7 else best_h
        except Exception:
            pass
    return best_h, best_c

def random_smooth_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, 8):
        a = rng.uniform(-0.45, 0.45)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += a * np.sin(0.5 * np.pi * m * x + ph)
        h += 0.5 * a * np.cos(np.pi * m * x + ph)
    h += 0.08 * (x / 2.0 - 0.5)
    return project_feasible(np.clip(h, 0.0, 1.0), n)

def optimize_instance(h, n, deadline, rng, quick=False):
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    hs = project_feasible(0.5 * (h + h[::-1]), n)
    cs = compute_c5(hs, n)
    if cs < c:
        h, c = hs, cs
    hc = 1.0 - h
    cs = compute_c5(hc, n)
    if cs < c:
        h, c = hc, cs
    hsc = project_feasible(0.5 * (h + 1.0 - h[::-1]), n)
    cs = compute_c5(hsc, n)
    if cs < c:
        h, c = hsc, cs
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    if (not quick) and time.time() < deadline - 30.0:
        h, c = lbfgs_stage(h, n, bounds, [(50, 40), (300, 30), (1500, 25)], 1e5, deadline)
    if quick:
        plans = [(50, 25), (500, 25), (5000, 20)]
        if n <= 200:
            plans.append((20000, 15))
    else:
        plans = [(20, 50), (100, 50), (500, 40), (2500, 30), (10000, 25), (50000, 20)]
        if time.time() < deadline - 25.0:
            plans.append((200000, 15))
    h, c = slsqp_stage(h, n, bounds, cons, plans, deadline)
    if n <= 800 and time.time() < deadline - 5.0:
        rem = deadline - time.time()
        mt = int(min(8000 if not quick else 2000, max(500, rem * 25)))
        h, c = intensive_pair_search(h, n, mt, deadline, rng)
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        h0 = random_smooth_start(200, rng)
    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    # Symmetries
    for transform in [lambda x: 0.5*(x + x[::-1]), lambda x: 1.0 - x,
                      lambda x: 0.5*(x + 1.0 - x[::-1])]:
        ht = project_feasible(transform(best_h), n0)
        ct = compute_c5(ht, n0)
        if ct < best_c:
            best_h, best_c = ht, ct
    if hard - t0 < 45.0:
        return best_h, compute_c5(best_h, n0), n0
    # Main optimization at original resolution
    main_dl = hard - max(100.0, 0.30 * float(budget_s))
    if main_dl < t0 + 30.0:
        main_dl = t0 + 60.0
    best_h, best_c = optimize_instance(best_h, n0, main_dl, rng, False)
    # Try other resolutions
    cands = [320, 400, 500, 640, 800]
    for m in cands:
        if time.time() > hard - 60.0:
            break
        if m == len(best_h):
            continue
        rem = hard - time.time()
        if m > 600 and rem < 80.0:
            continue
        dl = time.time() + min(40.0, 0.18 * rem)
        if dl < time.time() + 8.0:
            continue
        ht, ct = optimize_instance(interp_h(best_h, m), m, dl, rng, True)
        if ct < best_c:
            best_h, best_c = ht, ct
    # Random perturbations
    if time.time() < hard - 80.0:
        n = len(best_h)
        for s in [0.02, 0.05, 0.10, 0.20]:
            if time.time() > hard - 40.0:
                break
            hp = project_feasible(best_h + rng.normal(0.0, s, n), n)
            dl = min(hard - 10.0, time.time() + 35.0)
            ht, ct = optimize_instance(hp, n, dl, rng, True)
            if ct < best_c:
                best_h, best_c = ht, ct
    # Final intensive pair search
    if time.time() < hard - 10.0:
        n = len(best_h)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
        hd = hard - 3.0
        ht, ct = slsqp_stage(best_h, n, bounds, cons, [(50000, 20), (200000, 15)], hd)
        if ct < best_c:
            best_h, best_c = ht, ct
        if n <= 1000 and time.time() < hd - 3.0:
            rem = hd - time.time()
            mt = int(min(15000, max(1000, rem * 30)))
            ht, ct = intensive_pair_search(best_h, n, mt, hd, rng)
            if ct < best_c:
                best_h, best_c = ht, ct
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
