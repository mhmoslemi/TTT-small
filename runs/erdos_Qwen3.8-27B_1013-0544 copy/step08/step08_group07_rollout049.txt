        lo_b = max(0, d)
        hi_b = min(n, n + d)

def coord_cd(h, n, max_passes, deadline, grid_size=80):

    best_c = compute_c5(h, n)
    for pidx in range(max_passes):
        if time.time() > deadline:
            break

        current_max = float(c_prev.max())
        improved = False

            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += 1.0 - h[jf[valid_f]]
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            c_aff = c_prev[ms]
            unaff_max = 0.0
            if m_lo > 0:
                um = float(c_prev[:m_lo].max())
                unaff_max = max(unaff_max, um)
            if m_hi < 2 * n - 2:
                um = float(c_prev[m_hi + 1:].max())
                unaff_max = max(unaff_max, um)
            diag_mask = (ms == (n - 1))
            # Coarse then fine
            d_coarse = np.linspace(-h0, 1.0 - h0, 16)
            best_d = 0.0
            best_val = current_max
            for d in d_coarse:
                if abs(d) < 1e-14:
                    continue

                if np.any(diag_mask):
                    nv[diag_mask] -= d * d

                    best_d = d

                    val = max(unaff_max, float(nv.max()))
                    if val < best_val:

                        best_d = d
            if abs(best_d) > 1e-14 and best_val < current_max:

                improved = True
                c_prev[ms] = c_aff + best_d * a
                if np.any(diag_mask):
                    c_prev[ms[diag_mask]] -= best_d * best_d
        if not improved:
            break
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_obj_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    dx = 2.0 / n
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-12:
            continue
        d = k - (n - 1)
        lo_f = max(0, -d)
        hi_f = min(n, n - d)
        if lo_f < hi_f:
            g[lo_f:hi_f] += w[k] * (1.0 - x[lo_f + d:hi_f + d])
        lo_b = max(0, d)
        hi_b = min(n, n + d)
        if lo_b < hi_b:
            g[lo_b:hi_b] -= w[k] * x[lo_b - d:hi_b - d]
    return g * dx

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 20:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           jac=smooth_obj_grad, bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16, 'gtol': 1e-10})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def leveling(h, n, top_m, lr, deadline):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    dx = 2.0 / n
    idx = np.argsort(c)[::-1][:top_m]
    g = np.zeros(n)
    for k in idx:
        d = k - (n - 1)
        lo_f = max(0, -d)
        hi_f = min(n, n - d)
        if lo_f < hi_f:
            g[lo_f:hi_f] += (1.0 - h[lo_f + d:hi_f + d])
        lo_b = max(0, d)
        hi_b = min(n, n + d)
        if lo_b < hi_b:
            g[lo_b:hi_b] -= h[lo_b - d:hi_b - d]
    g *= dx / top_m
    best_h = h.copy()
    best_val = m * dx
    for i in range(8):
        step = lr * (0.5 ** i)
        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_val:
            best_h = h_new
            best_val = c_new
    return best_h, best_val

def coord_cd(h, n, max_passes, deadline, grid_size=80):
    h = h.copy()
    best_c = compute_c5(h, n)
    for pidx in range(max_passes):
        if time.time() > deadline:
            break
        c_prev = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c_prev.max())
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_lo = min(max(0, n - 1 - p), max(0, p))
            m_hi = max(min(2 * n - 2, 2 * n - 2 - p), min(2 * n - 2, p + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += 1.0 - h[jf[valid_f]]
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            c_aff = c_prev[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c_prev[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c_prev[m_hi + 1:].max()))
            diag_mask = (ms == (n - 1))
            best_d = 0.0
            best_val = current_max
            d_coarse = np.linspace(-h0, 1.0 - h0, 16)
            for d in d_coarse:
                if abs(d) < 1e-14:
                    continue
                nv = c_aff + d * a
                if np.any(diag_mask):
                    nv[diag_mask] -= d * d
                val = max(unaff_max, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            if best_d != 0.0:
                span = (1.0 - h0) / 16.0
                d_fine = np.linspace(max(-h0, best_d - span), min(1.0 - h0, best_d + span), grid_size)
                for d in d_fine:
                    if abs(d) < 1e-14:
                        continue
                    nv = c_aff + d * a
                    if np.any(diag_mask):
                        nv[diag_mask] -= d * d
                    val = max(unaff_max, float(nv.max()))
                    if val < best_val:
                        best_val = val
                        best_d = d
            if abs(best_d) > 1e-14 and best_val < current_max:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                c_prev[ms] = c_aff + best_d * a
                if np.any(diag_mask):
                    c_prev[ms[diag_mask]] -= best_d * best_d
        if not improved:
            break
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
    return h, compute_c5(h, n)

def smooth_pass(h, n, sigma):
    kernel_size = int(6 * sigma * n) | 1
    if kernel_size < 3:
        kernel_size = 3
    x = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    h_s = np.convolve(h, kernel, mode='same')
    return project_feasible(h_s, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 200
        x = np.linspace(0, 2, n_init, endpoint=False)
        h_init = project_feasible(0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.15 * np.sin(4 * np.pi * x), n_init)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS with analytic gradient, wider alpha range
    h1, c1 = lbfgs_opt(best_h, n_init, [100, 500, 2000, 10000, 50000, 200000, 1000000], 800, deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Leveling - equalize top correlations
    lr = 0.1
    for it in range(300):
        if time.time() > deadline - 300:
            break
        h2, c2 = leveling(best_h, n_init, 8, lr, deadline)
        if c2 < best_c - 1e-10:
            best_h, best_c = h2, c2
        else:
            lr *= 0.8
            if lr < 1e-7:
                break

    # Phase 3: Coordinate descent
    h3, c3 = coord_cd(best_h, n_init, 250, deadline, grid_size=60)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: Smoothing + re-optimize (escape local minima)
    if time.time() < deadline - 200:
        for sigma_frac in [0.005, 0.01, 0.02]:
            if time.time() > deadline - 150:
                break
            n = len(best_h)
            h_sm = smooth_pass(best_h, n, sigma_frac)
            h_sm, c_sm = lvl = lbfgs_opt(h_sm, n, [5000, 50000, 200000], 400, deadline)
            h_sm, c_sm = coord_cd(h_sm, n, 100, deadline, grid_size=50)
            if c_sm < best_c:
                best_h, best_c = h_sm, c_sm

    # Phase 5: Try different resolutions
    for n_try in [350, 500, 650]:
        if time.time() > deadline - 150:
            break
        if n_try == len(best_h):
            continue
        x_orig = np.linspace(0, 2, len(best_h), endpoint=False)
        x_try = np.linspace(0, 2, n_try, endpoint=False)
        h_try = project_feasible(np.interp(x_try, x_orig, best_h) + 0.005 * rng.randn(n_try), n_try)
        h_t, c_t = lbfgs_opt(h_try, n_try, [2000, 20000, 100000], 400, deadline)
        h_t, c_t = coord_cd(h_t, n_try, 100, deadline, grid_size=50)
        if c_t < best_c:
            best_h, best_c = h_t, c_t

    # Phase 6: Final leveling with more top indices
    lr = 0.05
    n = len(best_h)
    for it in range(200):
        if time.time() > deadline - 80:
            break
        h2, c2 = leveling(best_h, n, 15, lr, deadline)
        if c2 < best_c - 1e-10:
            best_h, best_c = h2, c2
        else:
            lr *= 0.85
            if lr < 1e-7:
                break

    # Phase 7: Final coordinate descent polish
    h3, c3 = coord_cd(best_h, n, 300, deadline, grid_size=100)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Final micro-leveling
    lr = 0.02
    for it in range(50):
        if time.time() > deadline - 2:
            break
        h2, c2 = leveling(best_h, len(best_h), 20, lr, deadline)
        if c2 < best_c - 1e-12:
            best_h, best_c = h2, c2
        else:
            lr *= 0.7

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
