And the second derivative d²c[m]/dh[p]² is:
- From the term (1-h[p-m]): if p-m = p, i.e., m = 0, then we have (1-h[p]) and d/dh[p] of that is -1
  So d²c[m]/dh[p]² = -[m=0 and 0 <= p-0 <= n-1] = -1 if m=0
- Actually wait: (1-h[p-m]) has derivative -1 with respect to h[p-m], not h[p]. 
  If m ≠ 0, then p-m ≠ p, so (1-h[p-m]) doesn't depend on h[p], and its derivative w.r.t. h[p] is 0.
  If m = 0, then (1-h[p]) has derivative -1 w.r.t. h[p].

So d²c[0]/dh[p]² = -1 (for all p), and d²c[m]/dh[p]² = 0 for m ≠ 0.

This confirms: only c[0] has the quadratic term in h[p]. All other correlations are linear in h[p].

So the 1D objective for coordinate p is:
f_p(d) = max_m [c[m] + d * a_m(p) - d² * [m=0]] * dx

where a_m(p) = [0 <= p-m <= n-1] * (1-h[p-m]) - [0 <= p+m <= n-1] * h[p+m]

This is the same as what the previous code computes! The only difference is that the previous code evaluates this on a grid and uses golden section. We can do better:

**Exact 1D minimization**: f_p(d) is the max of:
- For m ≠ 0: linear functions c[m] + d * a_m(p) (max of lines = convex piecewise linear)
- For m = 0: c[0] + d * a_0(p) - d² (a concave parabola)

The max of a concave parabola and convex piecewise linear function... this is not necessarily convex or concave. But we can find its minimum by:
1. Finding all breakpoints of the piecewise linear part
2. Checking the minimum of f_p at each breakpoint and at the critical point of the parabola

This is exact but might be O(n) per coordinate, which is the same as the grid approach. The grid approach with 80-120 points should be close to exact.

def nesterov_gradient(h, n, alpha, max_iter=2000, deadline=None):
    """Nesterov accelerated gradient on log-sum-exp smoothed objective."""
    dx = 2.0 / n
    h = h.copy()
    y = h.copy()  # Nesterov momentum variable
    beta = 0.9  # momentum
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for k in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        beta = (1 + np.sqrt(1 + 4 * beta * beta)) / 2  # Nesterov's beta
        # Actually, the standard Nesterov beta is (1 - t_{k-1}) / t_k where t_k = (1+sqrt(1+4t_{k-1}^2))/2
        # Let me use a fixed beta = 0.9 for simplicity
        
        # Compute gradient at y
        c = np.correlate(y, 1.0 - y, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        probs = np.exp(shifted - mx)
        probs /= probs.sum()
        # Gradient of log-sum-exp
        # d/dh[p] of lse(alpha * c) = alpha * sum_m probs[m] * dc[m]/dh[p]
        grad = np.zeros(n)
        for p in range(n):
            for m_idx in range(len(c)):
                if probs[m_idx] < 1e-10:
                    continue
                lag = m_idx - (n - 1)
                val = 0.0
                if 0 <= p - lag <= n - 1:
                    val += (1.0 - y[p - lag])
                if 0 <= p + lag <= n - 1:
                    val -= y[p + lag]
                grad[p] += probs[m_idx] * val
        grad *= alpha * dx
        
        # Step
        step = 1.0 / (alpha * dx)  # Lipschitz-based step size
        h_new = project_feasible(y - step * grad, n)
        y = (1 + beta) * h_new - beta * y
        y = np.clip(y, 0, 1)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
    
    return best_h, best_c

        improved = False
        order = list(range(n))
        for p in order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            lag_lo = max(p - (n - 1), -p)
            lag_hi = min(p, (n - 1) - p)
            idx_lo = lag_lo + (n - 1)
            idx_hi = lag_hi + (n - 1)
            c_sub = c[idx_lo:idx_hi + 1]
            lags = np.arange(lag_lo, lag_hi + 1)
            a = np.zeros(len(lags))
            for i in range(len(lags)):
                lag = lags[i]
                v = 0.0
                if 0 <= p - lag <= n - 1:
                    v += 1.0 - h[p - lag]
                if 0 <= p + lag <= n - 1:
                    v -= h[p + lag]
                a[i] = v
            diag_idx = (0 - lag_lo) if lag_lo <= 0 <= lag_hi else None
            unaff = 0.0
            if idx_lo > 0:
                unaff = max(unaff, float(c[:idx_lo].max()))
            if idx_hi < 2 * n - 2:
                unaff = max(unaff, float(c[idx_hi + 1:].max()))
            if unaff >= cur_max - 1e-13:
                continue
            lo_d = -h0
            hi_d = 1.0 - h0
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            vals = np.empty(grid_size)
            for gi in range(grid_size):
                v = c_sub + d_grid[gi] * a
                if diag_idx is not None:
                    v[diag_idx] -= d_grid[gi] * d_grid[gi]
                vals[gi] = max(float(np.max(v)), unaff)
            bi = int(np.argmin(vals))
            bd = float(d_grid[bi])
            bv = float(vals[bi])
            if 0 < bi < grid_size - 1:
                gl = float(d_grid[bi - 1])
                gh = float(d_grid[bi + 1])
                try:
                    res = golden(f1d, gl, gh, args=(c_sub, a, diag_idx, unaff), maxiter=45)
                    if res.fun < bv:
                        bd, bv = float(res.x), float(res.fun)
                except Exception:
                    pass
            if bv < cur_max - 1e-13:
                h[p] = float(np.clip(h0 + bd, 0.0, 1.0))
                improved = True
                cur_max = min(cur_max, bv)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_grad_step(h, n, alpha, lr, deadline=None, max_iter=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w /= w.sum()
        grad = np.zeros(n)
        active = np.where(w > 1e-6)[0]
        for mi in active:
            lag = int(mi) - (n - 1)
            pf_lo = max(0, lag)
            pf_hi = min(n - 1, n - 1 + lag)
            if pf_lo <= pf_hi:
                grad[pf_lo:pf_hi + 1] += w[mi] * (1.0 - h[pf_lo - lag:pf_hi - lag + 1])
            pb_lo = max(0, -lag)
            pb_hi = min(n - 1, n - 1 - lag)
            if pb_lo <= pb_hi:
                grad[pb_lo:pb_hi + 1] -= w[mi] * h[pb_lo + lag:pb_hi + lag + 1]
        h_new = project_feasible(h - lr * grad * (2.0 / n) * alpha, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            h = h_new
            best_c = c_new
        else:
            lr *= 0.8
            if lr < 1e-8:
                break
    return h, best_c

def block_perturb(h, n, rng, block_frac=0.15, mag=0.1):
    h2 = h.copy()
    blen = max(5, int(n * block_frac))
    start = rng.randint(0, n - blen)
    offset = rng.choice([-1, 1]) * mag
    h2[start:start + blen] += offset
    return project_feasible(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Initial coordinate descent
    if time.time() < deadline - 300:
        h1, c1 = coord_cd(best_h, n, 150, deadline, grid_size=60)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Smooth gradient to escape
    if time.time() < deadline - 250:
        h2, c2 = smooth_grad_step(best_h, n, alpha=30.0, lr=0.3, deadline=deadline, max_iter=300)
        h2, c2 = coord_cd(h2, n, 80, deadline, grid_size=60)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Multiple perturbations
    n_cur = len(best_h)
    perturbations = []
    for sig in [0.015, 0.04, 0.08, 0.12]:
        perturbations.append(project_feasible(best_h + rng.normal(0, sig, n_cur), n_cur))
    for _ in range(3):
        perturbations.append(block_perturb(best_h, n_cur, rng, 0.12, 0.08))
        perturbations.append(block_perturb(best_h, n_cur, rng, 0.25, 0.15))
    perturbations.append(project_feasible(1.0 - best_h, n_cur))

    for pi, hp in enumerate(perturbations):
        if time.time() > deadline - 200:
            break
        passes = 100 if pi < 3 else 60
        h3, c3 = coord_cd(hp, n_cur, passes, deadline, grid_size=50)
        if c3 < best_c:
            best_h, best_c = h3, c3
            n_cur = len(best_h)

    # Phase 4: Try different n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 500, 700, 800]:
        if time.time() > deadline - 150:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h, kind='cubic'), n_try)
        h4, c4 = coord_cd(h_int, n_try, 100, deadline, grid_size=50)
        if c4 < best_c:
            best_h, best_c = h4, c4
            x_orig = np.linspace(0, 2, n_try)

    # Phase 5: Final high-precision refinement
    n_cur = len(best_h)
    if time.time() < deadline - 30:
        hf, cf = coord_cd(best_h, n_cur, 300, deadline, grid_size=100)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Phase 6: One more smooth+coord cycle
    if time.time() < deadline - 20:
        h6, c6 = smooth_grad_step(best_h, len(best_h), alpha=50.0, lr=0.2, deadline=deadline, max_iter=100)
        h6, c6 = coord_cd(h6, len(best_h), 100, deadline, grid_size=80)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Check 1-h symmetry
    h_flip = project_feasible(1.0 - best_h, len(best_h))
    c_flip = compute_c5(h_flip, len(best_h))
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
        # Quick polish on the flip
        if time.time() < deadline - 5:
            hf, cf = coord_cd(best_h, len(best_h), 50, deadline, grid_size=80)
            if cf < best_c:
                best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
