def coord_descent(h, n, max_passes=300, deadline=None, gs_iters=35):
    h = h.copy()
    n_c = 2*n - 1
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_max = float(c.max())
    
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            # A[k] for k = 0..n_c-1, representing shift k-(n-1)
            ks = np.arange(-(n-1), n)
            i1 = p + ks
            i2 = p - ks
            v1 = (i1 >= 0) & (i1 < n)
            v2 = (i2 >= 0) & (i2 < n)
            A = np.zeros(n_c)
            A[v1] = 1.0 - h[i1[v1]]
            A[v2] += h[i2[v2]]
            # Diagonal correction
            D = np.zeros(n_c); D[n-1] = 1.0
            
            c_base = c - h0*A + h0*h0*D
            gr = (np.sqrt(5.0)-1.0)/2.0
            a_, b_ = 0.0, 1.0
            c1_ = b_ - gr*(b_-a_)
            d1_ = a_ + gr*(b_-a_)
            fc1_ = np.max(c_base + c1_*A - c1_*c1_*D)
            fd1_ = np.max(c_base + d1_*A - d1_*d1_*D)
            for _ in range(gs_iters):
                if fc1_ < fd1_:
                    b_ = d1_; d1_ = c1_; fd1_ = fc1_
                    c1_ = b_ - gr*(b_-a_)
                    fc1_ = np.max(c_base + c1_*A - c1_*c1_*D)
                else:
                    a_ = c1_; c1_ = d1_; fc1_ = fd1_
                    d1_ = a_ + gr*(b_-a_)
                    fd1_ = np.max(c_base + d1_*A - d1_*d1_*D)
            t_star = (a_+b_)*0.5
            f_mid = np.max(c_base + t_star*A - t_star*t_star*D)
            f_lo = np.max(c_base)
            f_hi = np.max(c_base + A - D)
            
            if f_lo <= f_mid and f_lo <= f_hi:
                t_new = 0.0
            elif f_hi <= f_mid and f_hi <= f_lo:
                t_new = 1.0
            else:
                t_new = float(t_star)
            
            f_best = np.max(c_base + t_new*A - t_new*t_new*D)
            if f_best < cur_max - 1e-13:
                c += (t_new-h0)*A - (t_new*t_new-h0*h0)*D
                h[p] = t_new
                cur_max = float(c.max())
                improved = True
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0-h, mode='full').astype(float)
        cur_max = float(c.max())
        if not improved and _ > 5:
            break
    return h, compute_c5(h, n)

def pnorm_slp(h, n, p, maxiter=300, deadline=None):
    def obj(x):
        cv = np.correlate(x, 1.0 - x, mode='full')

            grad[i] = gval * 2.0
        return grad * (np.sum(ol ** 2) ** (0.5 - 1.0/p)) if np.sum(ol**2) > 0 else np.zeros(n)
    def sfun(x):
