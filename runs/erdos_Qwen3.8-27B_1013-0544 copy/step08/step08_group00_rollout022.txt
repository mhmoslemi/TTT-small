import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:

def coord_cd_grid(h, n, max_passes, deadline, grid_size=80):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):

            ms = np.arange(m_lo, m_hi + 1)

            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):

                c_diag = c_aff[diag_idx[0]]
                a[diag_idx] = 0.0
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

            tot = room.sum()

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            if has_diag:
                c_diag = c_aff[diag_idx[0]]
                a[diag_idx] = 0.0
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            best_d = 0.0
            best_val = current_max
            for d in d_grid:
                if abs(d) < 1e-15:
                    continue
                nv = c_aff + d * a
                if has_diag:
                    nv[diag_idx] = c_diag + (h0 + d) * (1.0 - h0 - d)
                val = max(unaff_max, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-13:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def coord_cd_exact(h, n, max_passes, deadline):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, p + n - 1)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            c_base = c[ms]
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            if has_diag:
                a[diag_idx] = 0.0
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            T_diag = None
            if has_diag:
                T_diag = None  # computed per L in bisection
            L_lo = unaff_max
            L_hi = current_max
            for _ in range(50):
                if L_hi - L_lo < 1e-14:
                    break
                L = (L_lo + L_hi) / 2.0
                x_lo = 0.0
                x_hi = 1.0
                feasible = True
                pm = a > 1e-14
                nm = a < -1e-14
                zm = np.abs(a) <= 1e-14
                if np.any(zm) and np.any(c_base[zm] > L + 1e-14):
                    feasible = False
                if feasible and np.any(nm):
                    x_lo = max(x_lo, float(np.max((L - c_base[nm]) / a[nm])) + h0)
                if feasible and np.any(pm):
                    x_hi = min(x_hi, float(np.min((L - c_base[pm]) / a[pm])) + h0)
                x_lo = max(x_lo, 0.0)
                x_hi = min(x_hi, 1.0)
                if x_lo > x_hi + 1e-13:
                    feasible = False
                if feasible and has_diag:
                    T = L - float(c[diag_idx[0]]) + h0 * (1.0 - h0)
                    if T < -1e-14:
                        feasible = False
                    elif T <= 0.25:
                        disc = 1.0 - 4.0 * T
                        r1 = (1.0 - np.sqrt(max(0.0, disc))) / 2.0
                        r2 = (1.0 + np.sqrt(max(0.0, disc))) / 2.0
                        if x_lo > r1 and x_hi < r2:
                            feasible = False
                if feasible:
                    L_hi = L
                else:
                    L_lo = L
            L = L_hi
            x_lo = 0.0
            x_hi = 1.0
            if np.any(pm):
                x_hi = min(x_hi, float(np.min((L - c_base[pm]) / a[pm])) + h0)
            if np.any(nm):
                x_lo = max(x_lo, float(np.max((L - c_base[nm]) / a[nm])) + h0)
            x_lo = max(x_lo, 0.0)
            x_hi = min(x_hi, 1.0)
            best_x = None
            if has_diag:
                T = L - float(c[diag_idx[0]]) + h0 * (1.0 - h0)
                if T <= 0.25 and T >= -1e-14:
                    disc = 1.0 - 4.0 * max(0.0, T)
                    r1 = (1.0 - np.sqrt(disc)) / 2.0
                    r2 = (1.0 + np.sqrt(disc)) / 2.0
                    cands=[]
                    if x_lo <= r1 + 1e-13:
                        cands.append(max(x_lo, 0.0))
                        cands.append(min(x_hi, r1))
                    if x_hi >= r2 - 1e-13:
                        cands.append(max(x_lo, r2))
                        cands.append(x_hi)
                    if cands:
                        best_val = float('inf')
                        for xc in cands:
                            xc = float(np.clip(xc, 0.0, 1.0))
                            nv = c_base + (xc - h0) * a
                            nv[diag_idx] = float(c[diag_idx[0]]) + xc*(1.0-xc) - h0*(1.0-h0)
                            val = max(unaff_max, float(nv.max()))
                            if val < best_val:
                                best_val = val
                                best_x = xc
            if best_x is None:
                best_x = float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))
            if abs(best_x - h0) > 1e-12:
                h[p] = best_x
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgradient_shake(h, n, rng, n_steps, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    step = 0.5
    for s in range(n_steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = c.max()
        active = np.where(c > cmax - 0.02 * cmax)[0]
        if len(active) == 0:
            active = np.array([c.argmax()])
        g = np.zeros(n)
        for m in active:
            k = m - (n - 1)
            gf = np.zeros(n)
            idx_f = np.arange(max(0, -k), min(n, n - k))
            idx_s = idx_f + k
            gf[idx_f] += (1.0 - h[idx_s])
            gf[idx_s] -= h[idx_f]
            g += gf / len(active)
        g *= dx
        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
            h = h_new
            step = min(1.0, step * 1.05)
        else:
            step *= 0.7
            h = project_feasible(h_new + rng.normal(0, 0.01, n), n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: Grid CD
    h1, c1 = coord_cd_grid(best_h, n_init, 300, deadline, 100)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: SLSQP
    if time.time() < deadline - 120:
        h2, c2 = slsqp_run(best_h, n_init, [3000, 30000, 300000], 500)
        if c2 < best_c:
            best_h, best_c = h2, c2
            h3, c3 = coord_cd_grid(best_h, n_init, 100, deadline, 100)
            if c3 < best_c:
                best_h, best_c = h3, c3
    # Phase 3: Exact CD
    if time.time() < deadline - 40:
        h4, c4 = coord_cd_exact(best_h, n_init, 40, deadline)
        if c4 < best_c:
            best_h, best_c = h4, c4
    # Phase 4: Subgradient shake
    if time.time() < deadline - 80:
        n = len(best_h)
        h5, c5 = subgradient_shake(best_h, n, rng, 200, deadline)
        if c5 < best_c:
            best_h, best_c = h5, c5
            h6, c6 = coord_cd_grid(best_h, n, 60, deadline, 80)
            if c6 < best_c:
                best_h, best_c = h6, c6
    # Phase 5: Try other n
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [500, 800, 600, 900]:
        if time.time() > deadline - 120:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_cd_grid(h_int, n_try, 60, deadline, 80)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = np.linspace(0, 2, n_try)
            if time.time() < deadline - 30:
                he, ce = coord_cd_exact(best_h, n_try, 15, deadline)
                if ce < best_c:
                    best_h, best_c = he, ce
    # Phase 6: Multi-restart
    if time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(2, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            s = 0.01 + 0.04 * (r / max(1, n_restarts - 1))
            if r % 3 == 0:
                hp = project_feasible(best_h + rng.normal(0, s, n), n)
            elif r % 3 == 1:
                hp = best_h.copy()
                bs = max(10, n // 8)
                idx = rng.randint(0, max(1, n - bs))
                hp[idx:idx+bs] = rng.uniform(0, 1, bs)
                hp = project_feasible(hp, n)
            else:
                hp = project_feasible(np.full(n, 0.5) + rng.normal(0, 0.3, n), n)
            h_o, c_o = coord_cd_grid(hp, n, 40, deadline, 60)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 7: Final polish
    if time.time() < deadline - 8:
        n = len(best_h)
        hf, cf = coord_cd_grid(best_h, n, 30, deadline, 200)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = coord_cd_exact(best_h, n, 10, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
