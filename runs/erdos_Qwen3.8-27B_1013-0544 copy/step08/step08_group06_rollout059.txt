import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_slp_lp(h, n):
    """Build the LP for one SLP iteration. Returns (c_obj, A_ub, b_ub, A_eq, b_eq, bounds)."""
    c = np.correlate(h, 1.0 - h, mode='full')
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    b_ub = np.zeros(n_constraints)
    
    j = np.arange(n)
    for k in range(n_constraints):
        l = k - (n - 1)
        # coeff[j] = (1-h[j+l]) * (0<=j+l<n) - h[j-l] * (0<=j-l<n)
        jl = j + l
        valid1 = (jl >= 0) & (jl < n)
        coeff = np.zeros(n)
        coeff[valid1] += (1.0 - h[jl[valid1]])
        jn = j - l
        valid2 = (jn >= 0) & (jn < n)
        coeff[valid2] += -h[jn[valid2]]
        
        A_ub[k, :n] = coeff
        A_ub[k, n] = -1.0
        b_ub[k] = c[k] - np.dot(coeff, h)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    
    return c_obj, A_ub, b_ub, A_eq, b_eq, bounds, c

def slp_solve(h, n, max_iter=200, deadline=None):
    """Run SLP iterations."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c_obj, A_ub, b_ub, A_eq, b_eq, bounds, c = build_slp_lp(h, n)
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            break
        h_new = project_feasible(res.x[:n], n)
        
        # Line search
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(alpha * h_new + (1.0 - alpha) * h, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-13:
                best_h = h_try
                best_c = c_try
                h = h_try
                improved = True
                break
        
        if not improved:
            # Try accepting even small improvements
            c_try = compute_c5(h_new, n)
            if c_try < best_c - 1e-13:
                best_h = h_new.copy()
                best_c = c_try
                h = h_new
            else:
                break
    
    return best_h, best_c

def coord_cd(h, n, max_passes=500, deadline=None, grid_size=80):

    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')

        if deadline is not None and time.time() > deadline:

                continue
            else:
                lo_d, hi_d = -h0, 1.0 - h0
            if hi_d - lo_d < 1e-14:
                continue

            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            M = len(ms)
            
            a = np.zeros(M)
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]

            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if has_diag:
                nv[:, diag_pos] -= d_grid[:, None] ** 2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            
            best_i = int(np.argmin(vals))

            best_val = float(vals[best_i])

                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag_pos] -= xv ** 2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val = v1
                            best_d = xv
            
            if abs(best_d) > 1e-12 and best_val < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[ms[diag_pos]] -= best_d ** 2
                new_max = max(unaff_max, float(c[ms].max()))

            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    
    h = project_feasible(h, n)

                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[ms[diag_pos]] -= best_d ** 2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True
        if not improved:
            break
        if (p + 1) % 150 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: SLP to get close to optimum
    h1, c1 = slp_solve(h, n, max_iter=300, deadline=deadline - 500)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Coordinate descent fine-tuning
    h2, c2 = coord_cd(best_h, len(best_h), max_passes=1500, deadline=deadline - 300, grid_size=80)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: SLP again to escape CD local min
    h3, c3 = slp_solve(best_h, len(best_h), max_iter=200, deadline=deadline - 150)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: Final CD polish with finer grid
    h4, c4 = coord_cd(best_h, len(best_h), max_passes=3000, deadline=deadline - 20, grid_size=120)
    if c4 < best_c:
        best_h, best_c = h4, c4

    # Phase 5: Multi-resolution exploration
    n_list = [300, 400, 600]
    for n_try in n_list:
        if time.time() > deadline - 30:
            break
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_t, c_t = slp_solve(h_int, n_try, max_iter=80, deadline=deadline - 15)
        h_t, c_t = coord_cd(h_t, n_try, max_passes=500, deadline=deadline - 10, grid_size=60)
        if c_t < best_c:
            best_h, best_c = h_t, c_t

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
