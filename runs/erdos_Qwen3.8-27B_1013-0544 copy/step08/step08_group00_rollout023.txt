            feasible = False
        if feasible and np.any(neg_mask):
            needed = (L - c_base[neg_mask]) / a[neg_mask]
            x_lo = max(x_lo, float(needed.max()))
        if feasible and np.any(pos_mask):
            needed = (L - c_base[pos_mask]) / a[pos_mask]
            x_hi = min(x_hi, float(needed.min()))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc < -1e-14:
                feasible = False
            elif disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (x_lo <= r1 and x_hi >= r2):
                    if x_lo < r1 and x_hi > r2:
                        pass
                    else:
                        feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(((L - c_base[neg_mask]) / a[neg_mask]).max()))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(((L - c_base[pos_mask]) / a[pos_mask]).min()))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo <= r1 and x_hi >= r2:
                pass
            elif x_lo < r2 and x_hi > r1:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            else:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi:
        return h0
    return float(np.clip(0.5 * (x_lo + x_hi), 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, skip_bounds=True):
    h = h.copy()
    for it in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = 0
        perm = np.arange(n)
        for p in perm:
            if time.time() > deadline:
                break
            h0 = h[p]
            if skip_bounds and (h0 < 1e-12 or h0 > 1.0 - 1e-12):
                continue
            xn = solve_1d_exact(c, h, p, n)
            if abs(xn - h0) > 1e-11:
                h[p] = xn
                improved += 1
        if improved == 0:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgradient_step(h, n, step_size, deadline):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_vals = c * dx
    t = c_vals.max()
    active = c_vals > t - 0.01
    grad = np.zeros(n)
    for m in np.where(active)[0]:
        for p in range(n):
            jf = p + m - (n - 1)
            jb = p - m + (n - 1)
            if 0 <= jf < n:
                grad[p] += (1.0 - h[jf]) * (1.0 / len(np.where(active)[0]))
            if 0 <= jb < n:
                grad[p] -= h[jb] * (1.0 / len(np.where(active)[0]))
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new, compute_c5(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    best_h = None
    best_c = 1e9

    def try_update(h, n):
        nonlocal best_h, best_c
        c = compute_c5(h, n)
        if c < best_c:
            best_h, best_c = h.copy(), c
            return True
        return False

    # Phase 1: Multiple structured initializations
    n = 700
    inits = [
        linear_ramp(n),
        s_shaped(n, 0.08),
        s_shaped(n, 0.12),
        s_shaped2(n, 0.06, 0.04),
    ]
    for init in inits:
        h_try = coord_descent(init, n, 40, deadline)
        if h_try[1] < best_c:
            best_h, best_c = h_try[0], h_try[1]

    # Phase 2: Detailed coordinate descent on best
    if best_h is not None:
        h2, c2 = coord_descent(best_h, n, 200, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Try different n values by interpolation
    for n_try in [500, 900, 1200]:
        if time.time() > deadline - 150:
            break
        if best_h is None or n_try == len(best_h):
            continue
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, 60, deadline)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 4: Random restarts with perturbation
    if best_h is not None and time.time() < deadline - 100:
        n_best = len(best_h)
        remaining = deadline - time.time() - 50
        n_restarts = max(3, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            s = 0.01 + 0.03 * (r / max(1, n_restarts - 1))
            hp = project_feasible(best_h + rng.normal(0, s, n_best), n_best)
            h_o, c_o = coord_descent(hp, n_best, 50, deadline)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
        # Also try S-shaped restarts
        for a in [0.05, 0.1, 0.15, 0.2]:
            if time.time() > deadline - 50:
                break
            hp = s_shaped(n_best, a)
            h_o, c_o = coord_descent(hp, n_best, 40, deadline)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 5: Final high-precision passes
    if best_h is not None:
        n_best = len(best_h)
        hf, cf = coord_descent(best_h, n_best, 300, deadline, skip_bounds=False)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
