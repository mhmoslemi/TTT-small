            if tot < 1e-15:

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def minimax_slsqp(h, n, deadline, max_rounds=10):
    """Optimize using weighted sum of critical constraints."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for round_i in range(max_rounds):
        if time.time() > deadline:
            break
        c = np.correlate(best_h, 1.0 - best_h, mode='full')
        m = c.max()
        # Find critical k values (within 1% of max)
        threshold = m * 0.99
        critical = np.where(c >= threshold)[0]
        # Weight: more weight to more critical k's
        weights = np.exp(100 * (c[critical] - m))
        weights = weights / weights.sum()
        # Build objective: weighted sum of critical correlations
        def obj(x):
            cc = np.correlate(x, 1.0 - x, mode='full')
            return sum(w * cc[k] for w, k in zip(weights, critical)) * (2.0/n)
        def obj_jac(x):
            # Gradient of the objective
            grad = np.zeros(n)
            for w, k in zip(weights, critical):
                for i in range(n):
                    j = i + k - (n - 1)
                    if 0 <= j < n:
                        grad[i] += w * (1.0 - x[j]) * (2.0/n)
                    ip = i - k + (n - 1)
                    if 0 <= ip < n:
                        grad[i] -= w * x[ip] * (2.0/n)
            return grad
        # Optimize
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(obj, best_h, jac=obj_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if m < 0:
        m = 0
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alphas, maxiter=500):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except:
            pass
    return best_h, best_c

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def coord_cd(h, n, max_passes=200, deadline=None, grid_size=100):
    h = h.copy()
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            jf_arr = p + ms - (n - 1)
            jb_arr = p - ms + (n - 1)
            valid_f = (jf_arr >= 0) & (jf_arr < n)
            valid_b = (jb_arr >= 0) & (jb_arr < n)
            if np.any(valid_f):
                a_coeff[valid_f] += 1.0 - h[jf_arr[valid_f]]
            if np.any(valid_b):
                a_coeff[valid_b] -= h[jb_arr[valid_b]]
            c_aff = c[ms].copy()
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo = -h0
            hi = 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals = np.array([obj_1d(d, c_aff, a_coeff, diag_local, unaff_max) for d in d_grid])
            best_idx = int(np.argmin(vals))
            best_d, best_val = d_grid[best_idx], float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                gl, gh = d_grid[best_idx - 1], d_grid[best_idx + 1]
                try:
                    res = golden(obj_1d, gl, gh,
                                args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=100)
                    if res.fun < best_val:
                        best_d, best_val = float(res.x), float(res.fun)
                except:
                    pass
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def critical_slsqp(h, n, deadline, rounds=5):
    best_h, best_c = h.copy(), compute_c5(h, n)

    for r in range(rounds):
        if time.time() > deadline:

        c = np.correlate(best_h, 1.0 - best_h, mode='full')
        m = c.max()
        critical = np.where(c >= m * 0.98)[0]

            cc = np.correlate(x, 1.0 - x, mode='full')
            return np.max(cc[critical]) * dx
        
        def crit_jac(x):
            cc = np.correlate(x, 1.0 - x, mode='full')
            max_k = critical[np.argmax(cc[critical])]

            for i in range(n):
                j = i + max_k - (n - 1)
                if 0 <= j < n:
                    grad[i] += (1.0 - x[j]) * dx
                ip = i - max_k + (n - 1)
                if 0 <= ip < n:
                    grad[i] -= x[ip] * dx
            return grad

        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]

            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo = -h0
            hi = 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals = np.array([obj_1d(d, c_aff, a_coeff, diag_local, unaff_max) for d in d_grid])
            best_idx = int(np.argmin(vals))
            best_d, best_val = d_grid[best_idx], float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                gl, gh = d_grid[best_idx - 1], d_grid[best_idx + 1]
                try:
                    res = golden(obj_1d, gl, gh,
                                args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=100)
                    if res.fun < best_val:
                        best_d, best_val = float(res.x), float(res.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def build_crit_obj(x, n, crit_k):
    dx = 2.0 / n
    cc = np.correlate(x, 1.0 - x, mode='full')
    return float(cc[crit_k].max()) * dx

def build_crit_jac(x, n, crit_k):
    dx = 2.0 / n
    cc = np.correlate(x, 1.0 - x, mode='full')
    k_star = int(crit_k[np.argmax(cc[crit_k])])
    grad = np.zeros(n)
    for i in range(n):
        j = i + k_star - (n - 1)
        if 0 <= j < n:
            grad[i] += (1.0 - x[j]) * dx
        ip = i - k_star + (n - 1)
        if 0 <= ip < n:
            grad[i] -= x[ip] * dx
    return grad

def critical_slsqp(h, n, deadline, rounds=8):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for r in range(rounds):
        if time.time() > deadline:
            break
        c = np.correlate(best_h, 1.0 - best_h, mode='full')
        m = c.max()
        thr = m * 0.97
        crit_k = np.where(c >= thr)[0]
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        for _ in range(3):
            if time.time() > deadline:
                break
            c = np.correlate(best_h, 1.0 - best_h, mode='full')
            m = c.max()
            crit_k = np.where(c >= m * 0.97)[0]
            try:
                def _obj(x):
                    return build_crit_obj(x, n, crit_k)
                def _jac(x):
                    return build_crit_jac(x, n, crit_k)
                res = minimize(_obj, best_h, jac=_jac, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-15})
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c - 1e-15:
                    best_h, best_c = new_h, new_c
            except Exception:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP at base n
    if time.time() < deadline - 400:
        h1, c1 = slsqp_opt(best_h, n, [100, 1000, 5000, 20000], 600)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Critical constraint SLSQP
    if time.time() < deadline - 300:
        h2, c2 = critical_slsqp(best_h, n, deadline - 300, rounds=10)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Thorough coordinate descent
    if time.time() < deadline - 200:
        h3, c3 = coord_cd(best_h, n, 500, deadline - 200, grid_size=120)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Multi-n exploration
    x_orig = np.linspace(0, 2.0, len(best_h))
    for n_try in [200, 300, 400, 500, 700, 800]:
        if time.time() > deadline - 80:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2.0, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = slsqp_opt(h_int, n_try, [500, 5000, 20000], 400)
        h_o, c_o = critical_slsqp(h_o, n_try, deadline - 80, rounds=5)
        h_o, c_o = coord_cd(h_o, n_try, 150, deadline - 80, grid_size=80)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = x_new
    # Phase 5: Random perturbations at best n
    n_cur = len(best_h)
    for r in range(8):
        if time.time() > deadline - 40:
            break
        s = 0.01 + 0.03 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        h_o, c_o = critical_slsqp(hp, n_cur, deadline - 40, rounds=4)
        h_o, c_o = coord_cd(h_o, n_cur, 100, deadline - 40, grid_size=80)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
    # Phase 6: Final high-precision polish
    n_cur = len(best_h)
    if time.time() < deadline - 20:
        hf, cf = slsqp_opt(best_h, n_cur, [50000, 200000], 600)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 5:
        hf, cf = coord_cd(best_h, n_cur, 800, deadline - 2, grid_size=200)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
