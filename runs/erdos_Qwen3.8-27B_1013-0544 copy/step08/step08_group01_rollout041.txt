def slp_direction(h,n,scale,margin,topk,q=None):
    if q is None: q=q_counts(h,n)
    cur=float(np.max(q))
    if cur <= 0: return None
    idx=np.where(q > cur - margin)[0]
    if idx.size == 0: idx=np.array([int(np.argmax(q))])
    if idx.size > topk:
       # keep argmax and topk by q among idx
       part=np.argpartition(q[idx], -topk)[-topk:]
       idx=idx[part]
    # ensure argmax included
    amax=int(np.argmax(q))
    if amax not in idx: idx=np.append(idx,amax)  # but could exceed topk; okay
    S=idx.size
    A=slp_gradient_rows(h,n,idx)

    A_ub=np.empty((S,n+1)); A_ub[:,:n]=A; A_ub[:,n]=-1.0
    b_ub=-base
    A_eq=np.zeros((1,n+1)); A_eq[0,:n]=1.0
    b_eq=np.array([0.0])

    if res.status not in (0,3) or res.x is None: return None
    d=res.x[:n]
    if not np.all(np.isfinite(d)): return None
    d -= d.sum()/n
    lo=-scale*h; up=scale*(1-h)

import numpy as np
import time
from scipy.optimize import golden, linprog

TARGET_C5 = 0.3808

def q_counts(h, n):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, n):
    return float(np.max(q_counts(h, n)) * (2.0 / float(n)))

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(120):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        m = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(m.sum())
        if cnt == 0:
            break
        h[m] += diff / float(cnt)
    return np.clip(h, 0.0, 1.0)

def fast_project(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    diff = target - h.sum()
    if abs(diff) > 1e-11:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-13:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 1e-13:
                h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    m = (h > 1e-8) & (h < 1.0 - 1e-8)
    if m.any() and abs(diff) > 1e-13:
        h[m] += diff / float(m.sum())
    return np.clip(h, 0.0, 1.0)

def resample_h(h, n2):
    n1 = len(h)
    if n1 == n2:
        return project_feasible(h, n2)
    x1 = np.arange(n1, dtype=float) * (2.0 / float(n1))
    x2 = np.arange(n2, dtype=float) * (2.0 / float(n2))
    f = np.interp(x2, x1, h)
    return project_feasible(f, n2)

def anti_project(h):
    n = len(h)
    if n % 2:
        return np.asarray(h, dtype=float).copy()
    m = n // 2
    g = 0.5 * (h[:m] + 1.0 - h[m:])
    out = np.empty(n, dtype=float)
    out[:m] = g
    out[m:] = 1.0 - g
    return project_feasible(out, n)

def make_anti_2step(a, w, n):
    m = n // 2
    t = np.arange(m, dtype=float) / float(m)
    g = np.where(t < a, w, 1.0 - w)
    out = np.empty(n, dtype=float)
    out[:m] = g
    out[m:] = 1.0 - g
    return project_feasible(out, n)

def anti_family_candidate(h, n, deadline):
    best = np.asarray(h, dtype=float).copy()
    bestc = compute_c5(best, n)
    if n % 2:
        return best, bestc
    for a in np.linspace(0.08, 0.92, 9):
        for w in np.linspace(0.0, 1.0, 7):
            if time.time() > deadline:
                return best, bestc
            hc = make_anti_2step(float(a), float(w), n)
            cc = compute_c5(hc, n)
            if cc < bestc:
                best = hc
                bestc = cc
    return best, bestc

def slp_gradient_rows(h, n, ms):
    out = np.empty((len(ms), n), dtype=float)
    nm1 = n - 1
    for r, m in enumerate(ms):
        lag = int(m - nm1)
        a = abs(lag)
        row = out[r]
        row.fill(0.0)
        if a == 0:
            row[:] = 1.0 - 2.0 * h
        elif lag > 0:
            row[:n - a] = 1.0 - h[a:]
            row[a:] += -h[:n - a]
        else:
            row[:n - a] = -h[a:]
            row[a:] += 1.0 - h[:n - a]
    return out

def slp_direction(h, n, scale, margin, topk, q):
    cur = float(np.max(q))
    L = len(q)
    topk = max(8, min(int(topk), L))
    if topk >= L:
        top_idx = np.arange(L)
    else:
        top_idx = np.argpartition(q, L - topk)[-topk:]
    act_idx = np.where(q > cur - margin)[0]
    idx = np.unique(np.concatenate((top_idx, act_idx)))
    if idx.size > topk:
        idx = idx[np.argpartition(q[idx], -topk)[-topk:]]
    base = q[idx]
    A = slp_gradient_rows(h, n, idx)
    S = A.shape[0]
    c = np.zeros(n + 1, dtype=float)
    c[-1] = 1.0
    A_ub = np.empty((S, n + 1), dtype=float)
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -base
    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1, dtype=float)
    bounds = []
    for i in range(n):
        bounds.append((-float(scale) * float(h[i]), float(scale) * float(1.0 - h[i])))
    bounds.append((0.0, cur + 1.0))
    try:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method="highs", options={"presolve": True})
        except TypeError:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method="simplex")
    except Exception:
        return None
    if res is None or getattr(res, "status", 1) != 0 or getattr(res, "x", None) is None:
        return None
    d = np.asarray(res.x[:n], dtype=float)
    if not np.all(np.isfinite(d)):
        return None
    d -= d.sum() / float(n)
    lo = -scale * h
    up = scale * (1.0 - h)
    d = np.clip(d, lo, up)
    s = d.sum()
    if abs(s) > 1e-10:
        mask = (h > 1e-7) & (h < 1.0 - 1e-7)
        if mask.sum() > 0:
            d[mask] -= s / float(mask.sum())
    d = np.clip(d, lo, up)
    if np.max(np.abs(d)) < 1e-11:
        return None
    return d

def slp_alpha_h(a, h, d, n):
    return fast_project(h + a * d, n)

def slp_alpha_obj(a, h, d, n):
    return compute_c5(slp_alpha_h(a, h, d, n), n)

def slp_polish(h, n, deadline, stop_c=TARGET_C5):
    best = np.asarray(h, dtype=float).copy()
    bestc = compute_c5(best, n)
    scales = (0.25, 0.12, 0.06, 0.03, 0.015, 0.0075)
    margins = (6.0, 3.0, 1.5, 0.8, 0.4, 0.2)
    topks = (80, 140, 200, 280, 340, 400)
    alphas = (0.10, 0.25, 0.50, 0.75, 1.00)
    fails = 0
    for _ in range(90):
        if time.time() > deadline or bestc <= stop_c:
            break
        q = q_counts(best, n)
        improved = False
        for si in range(len(scales)):
            if time.time() > deadline:
                break
            d = slp_direction(best, n, scales[si], margins[si], topks[si], q)
            if d is None:
                continue
            ba = 0.0
            bc = bestc
            for a in alphas:
                v = slp_alpha_obj(a, best, d, n)
                if v < bc:
                    bc = v
                    ba = a
            if ba <= 0.0:
                continue
            if ba >= 1.0:
                lb, ub = 0.75, 1.0
            elif ba <= 0.10:
                lb, ub = 0.0, 0.25
            else:
                lb = 0.0
                ub = 1.0
                for a in alphas:
                    if abs(a - ba) < 1e-12:
                        continue
                    if a < ba and a > lb:
                        lb = a
                    if a > ba and a < ub:
                        ub = a
                if ub <= lb + 1e-8:
                    lb = max(0.0, ba - 0.25)
                    ub = min(1.0, ba + 0.25)
            if ub > lb + 1e-8:
                try:
                    rg = golden(slp_alpha_obj, lb, ub, args=(best, d, n), maxiter=26)
                    if rg.fun < bc:
                        bc = float(rg.fun)
                        ba = float(rg.x)
                except Exception:
                    pass
            if bc < bestc - 1e-11:
                best = slp_alpha_h(ba, best, d, n)
                bestc = bc
                improved = True
                break
        if improved:
            fails = 0
        else:
            fails += 1
            if fails > 8:
                break
    return best, bestc

def obj_1d_scalar(d, base, a, diag, U):
    v = float(np.max(base + a * d))
    if diag is not None:
        vd = float(base[diag] + a[diag] * d - d * d)
        if vd > v:
            v = vd
    if U > v:
        v = U
    return v

def coord_cd(h, n, max_passes, deadline, grid_size=36, max_step=0.05, use_golden=False):
    grid_size = int(max(8, grid_size))
    best = np.asarray(h, dtype=float).copy()
    bestc = compute_c5(best, n)
    h = best.copy()
    L = 2 * n - 1
    stall = 0
    for _ in range(int(max_passes)):
        if time.time() > deadline:
            break
        c = q_counts(h, n)
        cur = float(c.max())
        pass_up = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = float(h[p])
            lo_d = max(-h0, -max_step)
            hi_d = min(1.0 - h0, max_step)
            if hi_d - lo_d < 1e-13:
                continue
            left = min(p, n - 1 - p)
            right = max(2 * n - 2 - p, n - 1 + p)
            base = c[left:right + 1].copy()
            mlen = right - left + 1
            a = np.zeros(mlen, dtype=float)
            ms = np.arange(left, right + 1, dtype=int)
            jf = p + ms - (n - 1)
            vf = (jf >= 0) & (jf < n)
            if np.any(vf):
                a[vf] += 1.0 - h[jf[vf]]
            jb = p - ms + (n - 1)
            vb = (jb >= 0) & (jb < n)
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            U = 0.0
            if left > 0:
                U = float(c[:left].max())
            if right < L - 1:
                t = float(c[right + 1:].max())
                if t > U:
                    U = t
            diag = n - 1 - left
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            vals = base + a[:, None] * d_grid[None, :]
            vals[diag, :] -= d_grid * d_grid
            maxv = vals.max(axis=0)
            if U > 0.0:
                np.maximum(maxv, U, out=maxv)
            bi = int(np.argmin(maxv))
            bd = float(d_grid[bi])
            bv = float(maxv[bi])
            if 0 < bi < grid_size - 1:
                y0 = float(maxv[bi - 1])
                y1 = float(maxv[bi])
                y2 = float(maxv[bi + 1])
                den = y0 - 2.0 * y1 + y2
                if den > 1e-13:
                    tau = 0.5 * (y0 - y2) / den
                    dp = d_grid[bi] + tau * (d_grid[bi + 1] - d_grid[bi])
                    dp = float(np.clip(dp, lo_d, hi_d))
                    vp = obj_1d_scalar(dp, base, a, diag, U)
                    if vp < bv:
                        bv = vp
                        bd = dp
                if use_golden:
                    gl = float(d_grid[bi - 1])
                    gh = float(d_grid[bi + 1])
                    if gh > gl:
                        try:
                            rg = golden(obj_1d_scalar, gl, gh, args=(base, a, diag, U), maxiter=22)
                            if rg.fun < bv:
                                bv = float(rg.fun)
                                bd = float(rg.x)
                        except Exception:
                            pass
            if bv < cur - 1e-12:
                h[p] = float(np.clip(h0 + bd, 0.0, 1.0))
                newv = base + a * bd
                newv[diag] -= bd * bd
                c[left:right + 1] = newv
                cur = bv
                pass_up = True
        hp = fast_project(h, n)
        cp = compute_c5(hp, n)
        if cp < bestc:
            bestc = cp
            best = hp.copy()
            h = hp
            if pass_up:
                stall = 0
            else:
                stall += 1
        else:
            if cp <= bestc + 1e-9:
                h = hp
            else:
                h = best.copy()
            stall += 1
        if not pass_up and stall > 3:
            break
    return best, bestc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s) - 25.0)
    rng = np.random.RandomState(int(seed))
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        if h0.size == 0:
            h0 = None
    except Exception:
        h0 = None
    if h0 is None:
        n = 600
        h0 = np.full(n, 0.5, dtype=float)
    n = len(h0)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    flip = project_feasible(1.0 - best_h, n)
    cf = compute_c5(flip, n)
    if cf < best_c:
        best_h, best_c = flip, cf

    ap = anti_project(best_h)
    cap = compute_c5(ap, n)
    if cap < best_c:
        best_h, best_c = ap, cap

    if time.time() < deadline - 320.0:
        fam_h, fam_c = anti_family_candidate(best_h, n, deadline - 300.0)
        if fam_c < best_c:
            best_h, best_c = fam_h, fam_c

    if time.time() < deadline - 250.0 and best_c > TARGET_C5:
        best_h, best_c = slp_polish(best_h, n, deadline - 250.0, stop_c=TARGET_C5)

    if time.time() < deadline - 170.0:
        best_h, best_c = coord_cd(best_h, n, 22, deadline - 170.0,
                                  grid_size=34, max_step=0.07, use_golden=False)

    if time.time() < deadline - 90.0:
        best_h, best_c = coord_cd(best_h, n, 38, deadline - 90.0,
                                  grid_size=56, max_step=0.022, use_golden=False)

    if time.time() < deadline - 45.0:
        best_h, best_c = coord_cd(best_h, n, 22, deadline - 45.0,
                                  grid_size=88, max_step=0.009, use_golden=True)

    if time.time() < deadline - 160.0 and best_c > TARGET_C5:
        for n_try in (720, 840):
            if time.time() > deadline - 80.0:
                break
            h_try = resample_h(best_h, n_try)
            c_try = compute_c5(h_try, n_try)
            if c_try >= best_c + 1e-4:
                continue
            if time.time() < deadline - 120.0:
                h_try, c_try = slp_polish(h_try, n_try, deadline - 120.0, stop_c=best_c)
            if time.time() < deadline - 55.0:
                h_try, c_try = coord_cd(h_try, n_try, 18, deadline - 55.0,
                                        grid_size=52, max_step=0.018, use_golden=True)
            if c_try < best_c:
                best_h, best_c = h_try, c_try

    if time.time() < deadline - 25.0:
        ap = anti_project(best_h)
        cap = compute_c5(ap, len(ap))
        if cap < best_c:
            best_h, best_c = ap, cap
        flip = project_feasible(1.0 - best_h, len(best_h))
        cf = compute_c5(flip, len(flip))
        if cf < best_c:
            best_h, best_c = flip, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, float(best_c), int(len(best_h))
