    z = np.asarray(z, dtype=float)

    y = np.clip(z - 0.5*(low+high), 0.0, 1.0)

    if abs(res) > 1e-12:

    return np.clip(y, 0.0, 1.0)

2. The function is convex in u

If the max slope in the active set is < 0, f is decreasing, so go right.
If the min slope in the active set is > 0, f is increasing, so go left.
Otherwise, we're at the minimum.

        mask = (y < 1) if res > 0 else (y > 0)
        room = (1-y) if res > 0 else y
        tot = room[mask].sum()
        if tot > 1e-15: y[mask] += (res/tot)*room[mask]
    return np.clip(y, 0, 1)

def compute_A(h, n):
    nk = 2*n-1
    A = np.zeros((n, nk))
    off = n-1
    for k in range(nk):
        d = k - off
        if d >= 0:
            m = n - d
            A[:m, k] = 1.0 - h[d:]
            A[d:, k] -= h[:m]
        else:
            e = -d
            m = n - e
            A[e:, k] = 1.0 - h[:m]
            A[:m, k] -= h[e:]
    return A

def exact_u(c, direction, lo, hi):
    """Minimize max(c + u*direction) for u in [lo,hi] via bisection on subgradient."""
    def fval(u): return (c + u*direction).max()
    if hi - lo < 1e-14: return 0.5*(lo+hi)
    a, b = lo, hi
    for _ in range(45):
        mid = 0.5*(a+b)
        v = c + mid*direction
        m = v.max()
        act = v >= m - 1e-14
        if not act.any(): break
        sl = direction[act].min()
        su = direction[act].max()
        if su < -1e-15: a = mid
        elif sl > 1e-15: b = mid
        else: break
    u_mid = 0.5*(a+b)
    cands = [lo, hi, u_mid, 0.0]
    return min(cands, key=fval)

def exact_coord_descent(h, n, deadline, rng, max_rounds=5000):
    h = h.copy()
    c = compute_corr(h)
    A = compute_A(h, n)
    cur_max = float(c.max())
    no_improve = 0
    for r in range(max_rounds):
        if time.time() > deadline: break
        if no_improve > 2000: break
        cur_max = float(c.max())
        n_tries = min(300, max(100, n))
        improved = False
        for t in range(n_tries):
            if time.time() > deadline: break
            if rng.random() < 0.75:
                active = np.flatnonzero(c >= cur_max - 0.005*cur_max)
                if active.size == 0: active = np.array([int(c.argmax())])
                kk = active[rng.randint(active.size)]
                gi = np.abs(A[:, active]).max(axis=1) if active.size > 1 else np.abs(A[:, kk])
                probs = gi + 1e-8
                probs /= probs.sum()
                i = int(rng.choice(n, p=probs))
                j = int(rng.randint(n))
                if i == j: j = (j+1) % n
            else:
                i = int(rng.randint(n)); j = int(rng.randint(n))
                if i == j: j = (j+1) % n
            dirc = A[i] - A[j]
            lo = max(-h[i], h[j]-1.0)
            hi = min(1.0-h[i], h[j])
            if hi - lo < 1e-13: continue
            u_opt = exact_u(c, dirc, lo, hi)
            new_max = float((c + u_opt*dirc).max())
            if new_max < cur_max - 1e-14:
                h[i] += u_opt; h[j] -= u_opt
                c = compute_corr(h)
                A = compute_A(h, n)
                cur_max = float(c.max())
                improved = True
                no_improve = 0
        if not improved:
            no_improve += 1
            if no_improve % 300 == 0:
                idx = rng.randint(n)
                h[idx] = min(1.0, max(0.0, h[idx]+rng.uniform(-0.15, 0.15)))
                h = project_feasible(h, n)
                c = compute_corr(h)
                A = compute_A(h, n)
    return h, compute_c5(h, n)

def smooth_components(h, n, alpha):
    c = compute_corr(h)
    m = float(c.max())
    z = alpha*(c - m)
    zm = float(z.max())
    e = np.exp(z-zm)
    s = float(e.sum())
    p = e/s
    return (m + np.log(s)/alpha)*(2.0/n), p, c

def smooth_val(x, n, alpha):
    return smooth_components(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    _, p, _ = smooth_components(x, n, alpha)
    g = np.zeros(n)
    nk = 2*n-1
    off = n-1
    for k in range(nk):
        w = p[k]
        if w <= 1e-14: continue
        d = k - off
        if d >= 0:
            m = n-d
            g[:m] += w*(1.0-x[d:])
            g[d:] -= w*x[:m]
        else:
            e = -d; m = n-e
            g[e:] += w*(1.0-x[:m])
            g[:m] -= w*x[e:]
    return (2.0/n)*g

def sum_con(x, n): return x.sum() - 0.5*n
def sum_jac(x, n): return np.ones_like(x)

def slsqp_stage(h, n, deadline, plans):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work = best_h.copy()
    bounds = [(0.0,1.0)]*n
    cons = [{'type':'eq','fun':sum_con,'jac':sum_jac,'args':(n,)}]
    for alpha, mx in plans:
        if time.time() > deadline - 1: break
        try:
            res = minimize(smooth_val, work, args=(n,alpha), method='SLSQP',
                          jac=smooth_grad, bounds=bounds, constraints=cons,
                          options={'maxiter':mx, 'ftol':1e-14, 'disp':False})
            x = np.asarray(res.x, dtype=float)
            if np.all(np.isfinite(x)):
                x = project_feasible(x, n)
                cc = compute_c5(x, n)
                if cc < best_c: best_h, best_c = x, cc
                work = x if cc <= best_c+1e-8 else best_h
        except: pass
    return best_h, best_c

def pen_val(x, n, alpha, mu):
    v,_,_ = smooth_components(x, n, alpha)
    dev = x.sum() - 0.5*n
    return v + 0.5*mu*dev*dev

def pen_jac(x, n, alpha, mu):
    _,p,_ = smooth_components(x, n, alpha)
    g = np.zeros(n)
    nk = 2*n-1; off = n-1
    for k in range(nk):
        w = p[k]
        if w <= 1e-14: continue
        d = k - off
        if d >= 0:
            m = n-d
            g[:m] += w*(1.0-x[d:]); g[d:] -= w*x[:m]
        else:
            e=-d; m=n-e
            g[e:] += w*(1.0-x[:m]); g[:m] -= w*x[e:]
    g = (2.0/n)*g
    g += mu*(x.sum()-0.5*n)
    return g

def lbfgs_stage(h, n, deadline, mu):
    best_h = h.copy(); best_c = compute_c5(best_h, n)
    work = best_h.copy()
    bounds = [(0.0,1.0)]*n
    for alpha, mx in [(100,50),(500,40),(2000,30),(10000,25)]:
        if time.time() > deadline - 1: break
        try:
            res = minimize(pen_val, work, args=(n,alpha,mu), method='L-BFGS-B',
                          jac=pen_jac, bounds=bounds,
                          options={'maxiter':mx,'ftol':1e-14,'maxls':20,'disp':False})
            x = np.asarray(res.x, dtype=float)
            if np.all(np.isfinite(x)):
                x = project_feasible(x, n)
                cc = compute_c5(x, n)
                if cc < best_c: best_h, best_c = x, cc
                work = x if cc <= best_c+1e-7 else best_h
        except: pass
    return best_h, best_c

def interp_h(h, m):
    n = len(h)
    if n == m: return project_feasible(h.copy(), m)
    xo = np.linspace(0, 2, n, endpoint=False)
    xn = np.linspace(0, 2, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def symmetrize(h, n):
    h = h.copy()
    hs = project_feasible(0.5*(h + h[::-1]), n)
    if compute_c5(hs, n) < compute_c5(h, n): h = hs
    hc = 1.0 - h
    if compute_c5(hc, n) < compute_c5(h, n): h = 1.0 - h
    hsc = project_feasible(0.5*(h + 1.0 - h[::-1]), n)
    if compute_c5(hsc, n) < compute_c5(h, n): h = hsc
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 4.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 4: raise ValueError
    except:
        h0 = project_feasible(np.full(200, 0.5) + 0.1*rng.randn(200), 200)
    n = len(h0)
    best_h = project_feasible(symmetrize(h0, n), n)
    best_c = compute_c5(best_h, n)

    # Phase 1: LBFGS warm start
    if time.time() < hard - 200:
        dl = time.time() + min(60, 0.12*(hard-t0))
        best_h, best_c = lbfgs_stage(best_h, n, dl, 1e4)

    # Phase 2: SLSQP with smooth max
    if time.time() < hard - 150:
        dl = time.time() + min(120, 0.25*(hard-t0))
        plans = [(50,40),(200,35),(1000,30),(5000,25),(20000,20),(100000,15)]
        best_h, best_c = slsqp_stage(best_h, n, dl, plans)

    # Phase 3: Exact coordinate descent (main workhorse)
    if time.time() < hard - 60:
        dl = hard - 60
        best_h, best_c = exact_coord_descent(best_h, n, dl, rng, max_rounds=8000)

    # Phase 4: Try different resolutions
    for m in [100, 200, 300, 400, 600, 800]:
        if time.time() > hard - 40: break
        if m == n: continue
        rem = hard - time.time()
        if rem < 30: break
        dl = time.time() + min(35, 0.18*rem)
        if dl < time.time() + 10: continue
        ht = interp_h(best_h, m)
        ht = symmetrize(ht, m)
        ht, ct = exact_coord_descent(ht, m, dl, rng, max_rounds=3000)
        if ct < best_c:
            best_h, best_c = ht, ct

    # Phase 5: Noise restarts at current resolution
    n_cur = len(best_h)
    for s in [0.02, 0.05, 0.1, 0.2]:
        if time.time() > hard - 20: break
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        dl = min(hard - 8, time.time() + 25)
        ht, ct = exact_coord_descent(hp, n_cur, dl, rng, max_rounds=2000)
        if ct < best_c: best_h, best_c = ht, ct

    # Phase 6: Final SLSQP polish
    if time.time() < hard - 5:
        tl = compute_c5(best_h, len(best_h))
        dl2 = hard - 2
        best_h, best_c = slsqp_stage(best_h, len(best_h), dl2, [(5000,20),(50000,15)])

    # Phase 7: Final coord descent
    if time.time() < hard - 3:
        best_h, best_c = exact_coord_descent(best_h, len(best_h), hard - 1, rng, max_rounds=3000)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
