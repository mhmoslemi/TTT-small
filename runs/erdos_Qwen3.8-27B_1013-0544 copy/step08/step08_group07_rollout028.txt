            # This is the 1D "minimax" of affine functions

```python
def compute_coeffs(p, h, n):
    """Return (affected_lags, coefficients) for coordinate p."""
    # Forward: m from max(0, n-1-p) to min(2n-2, 2n-2-p)
    f_lo = max(0, n - 1 - p)
    f_hi = min(2*n - 2, 2*n - 2 - p)
    # Backward: m from max(0, p-n+1) to min(2n-2, p)
    b_lo = max(0, p - n + 1)
    b_hi = min(2*n - 2, p)
    
    # Build coefficient array for all lags in [min(f_lo,b_lo), max(f_hi,b_hi)]
    total_lo = min(f_lo, b_lo)
    total_hi = max(f_hi, b_hi)
    
    coeffs = np.zeros(total_hi - total_lo + 1)
    
    # Forward contributions
    if f_hi >= f_lo:
        ms = np.arange(f_lo, f_hi + 1)
        js = p + ms - n + 1  # j indices, all valid by construction
        coeffs[ms - total_lo] += (1.0 - h[js])
    
    # Backward contributions
    if b_hi >= b_lo:
        ms = np.arange(b_lo, b_hi + 1)
        is_ = p - ms + n - 1  # i indices, all valid by construction
        coeffs[ms - total_lo] -= h[is_]
    
    lags = np.arange(total_lo, total_hi + 1)
    return lags, coeffs
```
