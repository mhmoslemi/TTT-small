    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):

        return z

    momentum = np.zeros(n)

import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(55):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def smooth_components(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    s = float(e.sum())
    p = e / s
    return (m + float(np.log(s)) / alpha) * (2.0 / n), p, c

def grad_weights(x, n, p):
    g = np.zeros(n, dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-13:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n - d])
            g[:n - d] -= w * x[d:]
        else:
            e = -d
            g[:n - e] += w * (1.0 - x[e:])
            g[e:] -= w * x[:n - e]
    return (2.0 / n) * g

def smooth_val(x, n, alpha):
    return smooth_components(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    _, p, _ = smooth_components(x, n, alpha)
    return grad_weights(x, n, p)

def slsqp_stage(h, n, bounds, cons, plans, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work_h = best_h.copy()
    for alpha, mx in plans:
        if time.time() > deadline - 1.0:
            break
        try:
            res = minimize(smooth_val, work_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_grad, bounds=bounds, constraints=cons,
                           options={'maxiter': mx, 'ftol': 1e-14, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best_h, best_c = cur, cc
            work_h = cur if cc <= best_c + 1e-8 else best_h
        except Exception:
            pass
    return best_h, best_c

def compute_A(h, n):
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[d:, k] += 1.0 - h[:n - d]
            A[:n - d, k] -= h[d:]
        else:
            e = -d
            A[:n - e, k] += 1.0 - h[e:]
            A[e:] -= h[:n - e]
    return A

def eval_pair_brent(h, c, A, i, j, n):
    hi, hj = h[i], h[j]
    S = hi + hj
    lo = max(0.0, S - 1.0)
    hiT = min(1.0, S)
    if hiT - lo <= 1e-14:
        return None
    cur_max = float(c.max())
    off = n - 1
    B = A[i] - A[j]
    ki = off + i - j
    kj = off + j - i

    def obj(delta):
        cn = c + delta * B
        u2 = delta * delta
        peak = float(cn.max())
        v0 = cn[off] - 2.0 * u2
        if v0 > peak:
            peak = v0
        if 0 <= ki < len(cn):
            vk = cn[ki] + u2
            if vk > peak:
                peak = vk
        if 0 <= kj < len(cn):
            vk = cn[kj] + u2
            if vk > peak:
                peak = vk
        return peak

    res = minimize_scalar(obj, bounds=(lo, hiT), method='brent',
                          options={'xatol': 1e-12, 'maxiter': 40})
    if res.success and res.fun < cur_max - 1e-10:
        return float(res.x)
    return None

def pair_search(h, n, max_tries, deadline, rng):
    if n < 2 or max_tries <= 0:
        return h, compute_c5(h, n)
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    A = compute_A(h, n)
    accepts = 0
    no_improve = 0
    max_accept = min(300, max(80, max_tries // 3))
    for trial in range(max_tries):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        tol = max(0.012, 0.0008 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        if trial % 7 == 6 or active.size <= 2:
            i = int(rng.randint(n)); j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
        else:
            kk = int(active[int(rng.randint(active.size))])
            d = kk - (n - 1)
            if d >= 0:
                m = n - d
                if m <= 0:
                    continue
                b = int(rng.randint(m)); i = b + d; j = b
            else:
                e = -d
                m = n - e
                if m <= 0:
                    continue
                b = e + int(rng.randint(m)); i = b - e; j = b
        u = eval_pair_brent(h, c, A, i, j, n)
        if u is None:
            no_improve += 1
            continue
        old = float(h[i] + h[j])
        t = float(np.clip(h[i] + u, 0.0, 1.0))
        u = t - float(h[i])
        if abs(u) < 1e-15:
            no_improve += 1
            continue
        B = A[i] - A[j]
        u2 = u * u
        cn = c + u * B
        off = n - 1
        cn[off] -= 2.0 * u2
        ki = off + i - j
        kj = off + j - i
        if 0 <= ki < len(cn):
            cn[ki] += u2
        if 0 <= kj < len(cn):
            cn[kj] += u2
        nm = float(cn.max())
        if nm < cur_max - 1e-9:
            h[i] = t
            h[j] = old - t
            c = cn
            accepts += 1
            no_improve = 0
            if accepts >= max_accept:
                break
            A = compute_A(h, n)
            if accepts % 40 == 0:
                c = np.correlate(h, 1.0 - h, mode='full')
        else:
            no_improve += 1
        if no_improve > max(500, max_tries // 3):
            break
    return h, compute_c5(h, n)

def subgrad_descent(h, n, deadline, rng):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.03
    mu = 0.85
    mom = np.zeros(n)
    for it in range(800):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        if cur_max < best_c - 1e-7:
            best_c = cur_max
            best_h = h.copy()
        tol = max(0.001, 0.0005 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        z = (c[active] - cur_max) / max(tol, 1e-10)
        w = np.exp(z)
        w /= w.sum()
        g = np.zeros(n)
        off = n - 1
        for idx, ki in enumerate(active):
            wi = w[idx]
            d = ki - off
            if d >= 0:
                g[d:] += wi * (1.0 - h[:n - d])
                g[:n - d] -= wi * h[d:]
            else:
                e = -d
                g[:n - e] += wi * (1.0 - h[e:])
                g[e:] -= wi * h[:n - e]
        g *= 2.0 / n
        mom = mu * mom - lr * g
        h = project_feasible(h + mom, n)
        lr *= 0.997
    return best_h, best_c

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def optimize_instance(h, n, deadline, rng, quick=False):
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    hs = project_feasible(0.5 * (h + h[::-1]), n)
    cs = compute_c5(hs, n)
    if cs < c:
        h, c = hs, cs
    hc = 1.0 - h
    cs = compute_c5(hc, n)
    if cs < c:
        h, c = hc, cs
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    if not quick and time.time() < deadline - 15.0:
        plans = [(20, 60), (100, 50), (500, 40), (2000, 30), (8000, 25), (30000, 20)]
        if time.time() < deadline - 10.0:
            plans.append((100000, 15))
    else:
        plans = [(50, 40), (300, 35), (2000, 30)]
        if n <= 400:
            plans.append((8000, 20))
    h, c = slsqp_stage(h, n, bounds, cons, plans, deadline)
    if n <= 1500 and time.time() < deadline - 4.0:
        dl = min(deadline - 1.0, time.time() + max(8.0, 0.12 * (deadline - time.time())))
        h, c = subgrad_descent(h, n, dl, rng)
    if n <= 1500 and time.time() < deadline - 2.0:
        rem = deadline - time.time()
        mt = int(min(3000 if quick else 10000, max(200, rem * 35)))
        h, c = pair_search(h, n, mt, deadline, rng)
    h = project_feasible(h, n)
    return h, compute_c5(h, n)
