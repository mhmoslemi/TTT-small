from scipy.optimize import minimize
import time

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    lo, hi = -1.0, 1.0
    for _ in range(55):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def binarize(h, n):
    target = n // 2
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx

def smooth_obj(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]

def sum_con(x, n):
    return x.sum() - 0.5 * n

def sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)

    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a), jac=smooth_grad,
                          method="SLSQP", bounds=bounds, constraints=cons,
                          options={"maxiter": maxiter, "ftol": 1e-12})
        except Exception:

        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-13:
            best = nh
            bc = nc
    return best, bc

    d = xs - old
    cand = c[None, :] + d[:, None] * A[None, :]
    cand[:, n - 1] -= d * d
    return cand.max(axis=1)

    bc = compute_c5(best, n)

            vals = coord_eval(c, A, xs0, old, n)

            bx = float(xs0[i])

    c = np.correlate(h, 1.0 - h, mode="full")

def binarize(h, n):
    target = n // 2
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    lo, hi = -1.0, 1.0
    for _ in range(55):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
    return np.clip(y, 0.0, 1.0)

def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx

def smooth_obj(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]

    return x.sum() - 0.5 * n

    return np.ones_like(x, dtype=float)

    h = project_simplex(h, n)
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a), jac=smooth_grad,
                          method="SLSQP", bounds=bounds, constraints=cons,
                          options={"maxiter": maxiter, "ftol": 1e-12})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)

            best = nh
            bc = nc

    """Simulated annealing for binary overlap minimization."""
    b = b0.copy()
    c = np.correlate(b, 1.0 - b, mode="full")
    cm = float(c.max())
    best = b.copy()
    best_c = cm
    
    T = T_init
    tmp = np.empty(2 * n - 1)
    
    while time.time() < deadline and T > min_T:
        ones_pos = np.where(b > 0.5)[0]
        zeros_pos = np.where(b < 0.5)[0]
        i = int(rng.choice(ones_pos))
        j = int(rng.choice(zeros_pos))

        np.copyto(tmp, c)
        tmp[i:i+n] -= om
        st = n-1-i
        tmp[st:st+n] += b
        tmp[j:j+n] += om
        st = n-1-j
        tmp[st:st+n] -= b
        tmp[n-1] -= 2.0

        d2 = j - i
        tmp[n-1+d1] += 1.0
        tmp[n-1+d2] += 1.0

            b[i] = 0.0
            b[j] = 1.0

                best = b.copy()

            b[i] = 0.0
            b[j] = 1.0
            k -= 1
    return b

    bc = compute_c5(best, n)

        cp = compute_c5(xp, n)
        if cp < bc - tol:

def sa_binary(b0, n, rng, deadline, T_init=5.0, cool=0.99999, min_T=0.3):
    b = b0.copy()
    c = np.correlate(b, 1.0 - b, mode="full")
    cm = float(c.max())
    best = b.copy()
    best_c = cm
    T = T_init
    tmp = np.empty(2 * n - 1)
    while time.time() < deadline and T > min_T:
        i = int(rng.choice(np.where(b > 0.5)[0]))
        j = int(rng.choice(np.where(b < 0.5)[0]))
        om = 1.0 - b[::-1]
        np.copyto(tmp, c)
        tmp[i:i+n] -= om
        st = n - 1 - i
        tmp[st:st+n] += b
        tmp[j:j+n] += om
        st = n - 1 - j
        tmp[st:st+n] -= b
        tmp[n-1] -= 2.0
        tmp[n-1 + i - j] += 1.0
        tmp[n-1 + j - i] += 1.0
        new_m = float(tmp.max())
        delta = new_m - cm
        if delta < 0 or rng.random() < np.exp(-delta / T):

            cm = new_m
            b[i], b[j] = 0.0, 1.0
            if cm < best_c:
                best_c, best = cm, b.copy()
        T *= cool
    return best, best_c * (2.0 / n)

def sa_restarts(b0, n, rng, deadline, max_rounds=30):
    best, best_c = b0.copy(), compute_c5(b0, n)
    for r in range(max_rounds):
        if time.time() > deadline or deadline - time.time() < 5:
            break
        chunk = min(deadline - time.time(), max(10.0, (deadline - time.time()) / max(1, max_rounds - r)))
        rd = time.time() + chunk
        T_init = 5.0 if r < 3 else 3.0
        cool = 0.99999
        b, c = sa_binary(best, n, rng, rd, T_init=T_init, cool=cool, min_T=0.3)
        if c < best_c:
            best, best_c = b.copy(), c
        else:
            if r >= 2:
                best = perturb_binary(best, n, rng, min(0.3, 0.05 + 0.03 * (r - 2)))
    return best, best_c

def compute_delta(b, n):
    """Compute delta[i, m] = change in c[m] when b[i] goes from 0 to 1."""
    delta = np.zeros((n, 2*n-1))
    one_minus_b = 1.0 - b
    # For each lag L = m - (n-1):
    # delta[i, m] += one_minus_b[i-L] if 0 <= i-L < n
    # delta[i, m] -= b[i+L] if 0 <= i+L < n
    
    # First term: for L from 0 to n-1 (m = n-1+L, i from L to n-1):
    #   delta[i, n-1+L] += one_minus_b[i-L]
    # For L from 1 to n-1 (m = n-1-L, i from 0 to n-1-L):
    #   delta[i, n-1-L] += one_minus_b[i+L]
    
    # Second term: for L from 0 to n-1 (m = n-1+L, i from 0 to n-1-L):
    #   delta[i, n-1+L] -= b[i+L]
    # For L from 1 to n-1 (m = n-1-L, i from L to n-1):
    #   delta[i, n-1-L] -= b[i-L]
    
    # Let me use a different approach: build it up
    # L = 0: m = n-1, no change (as we showed)
    # For L = 1 to n-1:
    #   m_pos = n-1+L: delta[i, m_pos] += one_minus_b[i-L] for i>=L, and -= b[i+L] for i+L<n
    #   m_neg = n-1-L: delta[i, m_neg] += one_minus_b[i+L] for i+L<n, and -= b[i-L] for i>=L
    
    for L in range(1, n):
        mp = n - 1 + L
        mn = n - 1 - L
        # m_pos = n-1+L:
        #   one_minus_b[i-L] valid for i in [L, n-1]
        delta[L:, mp] += one_minus_b[:n-L]
        #   -b[i+L] valid for i in [0, n-1-L]
        delta[:n-L, mp] -= b[L:]
        # m_neg = n-1-L:
        #   one_minus_b[i+L] valid for i in [0, n-1-L]
        delta[:n-L, mn] += one_minus_b[L:]
        #   -b[i-L] valid for i in [L, n-1]
        delta[L:, mn] -= b[:n-L]
    
    return delta
