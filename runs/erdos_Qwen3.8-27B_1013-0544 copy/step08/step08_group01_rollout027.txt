def binary_local_search(h_bin, n, deadline, max_swaps=1000):
    h = h_bin.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    c = np.correlate(h, 1-h, mode='full')
    current_max = float(c.max())
    
    for swap_count in range(max_swaps):
        if deadline is not None and time.time() > deadline:
            break
        best_improvement = 1e-10
        best_pq = None
        best_new_c = None
        
        for p in ones:
            for q in zeros:
                # Compute delta
                k_range = np.arange(2*n-1)
                # delta[k] = h[p+k-(n-1)] + h[p-k+(n-1)] - h[q+k-(n-1)] - h[q-k+(n-1)]
                # where h[idx] = 0 if idx out of range
                delta = np.zeros(2*n-1)
                # Forward from p
                idx_f_p = p + k_range - (n-1)
                vf = (idx_f_p >= 0) & (idx_f_p < n)
                delta[vf] += h[idx_f_p[vf]]
                # Backward from p
                idx_b_p = p - k_range + (n-1)
                vb = (idx_b_p >= 0) & (idx_b_p < n)
                delta[vb] += h[idx_b_p[vb]]
                # Forward from q
                idx_f_q = q + k_range - (n-1)
                vf_q = (idx_f_q >= 0) & (idx_f_q < n)
                delta[vf_q] -= h[idx_f_q[vf_q]]
                # Backward from q
                idx_b_q = q - k_range + (n-1)
                vb_q = (idx_b_q >= 0) & (idx_b_q < n)
                delta[vb_q] -= h[idx_b_q[vb_q]]
                
                new_c = c + delta
                new_max = float(new_c.max())
                improvement = current_max - new_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_pq = (p, q)
                    best_new_c = new_c
        
        if best_pq is None:
            break
        
        p, q = best_pq
        h[p] = 0
        h[q] = 1
        c = best_new_c
        current_max = float(c.max())
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
    
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def c5_1d_val(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def exact_1d_vec(d_lo, d_hi, c_aff, a_coeff, diag_local, unaff_max):
    n_grid = 40
    d_grid = np.linspace(d_lo, d_hi, n_grid)
    vals = c_aff[None, :] + d_grid[:, None] * a_coeff[None, :]
    if diag_local is not None:
        vals[:, diag_local] -= d_grid[:, None] ** 2
    fv = vals.max(axis=1)
    fv = np.maximum(fv, unaff_max)
    bi = int(fv.argmin())
    if bi <= 0:
        return d_grid[0]
    if bi >= n_grid - 1:
        return d_grid[-1]
    a, b = d_grid[bi-1], d_grid[bi+1]
    for _ in range(50):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if c5_1d_val(m1, c_aff, a_coeff, diag_local, unaff_max) <= c5_1d_val(m2, c_aff, a_coeff, diag_local, unaff_max):
            b = m2
        else:
            a = m1
