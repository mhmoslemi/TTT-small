    cm = float(np.max(cseg)) if m else 0.0
    qcur = qconst + hp * (1.0 - hp)
    low = max(0.0, unaff, qconst)
    if zmax > low:
        low = zmax
    high = max(low + 1e-11, unaff, qconst, qcur, cm)
    if zmax > high:
        high = zmax
    high += 1e-8
    a_pos = a[pos]
    a_neg = a[neg]
    b_pos = b[pos]
    b_neg = b[neg]
    pc = a_pos.size
    nc = a_neg.size
    for _ in range(iters):
        mid = 0.5 * (low + high)
        feas = False
        if mid >= unaff - tol and mid >= qconst - tol and mid >= zmax - tol:
            ll = 0.0
            lh = 1.0
            if nc:
                v = float(np.max((mid - b_neg) / a_neg))
                if v > ll:
                    ll = v
            if pc:
                v = float(np.min((mid - b_pos) / a_pos))
                if v < lh:
                    lh = v
            if ll <= 1.0 + tol and lh >= -tol:
                g = mid - qconst
                if g > 0.25:
                    feas = max(0.0, ll) <= min(1.0, lh) + tol
                elif g > -1e-12:
                    g = max(0.0, g)
                    disc = max(0.0, 1.0 - 4.0 * g)
                    r1 = 0.5 - 0.5 * np.sqrt(disc)
                    r2 = 0.5 + 0.5 * np.sqrt(disc)
                    if max(0.0, ll) <= min(r1, lh) + tol:
                        feas = True
                    elif max(r2, ll) <= min(1.0, lh) + tol:
                        feas = True
        if feas:
            high = mid
        else:
            low = mid
    t = high
    cands = [hp, t, 0.0, 1.0, 0.5]
    best_x = hp
    best_f = coord_eval(hp, hp, cseg, a, qconst, unaff)
    for x in cands:
        if x < -1e-12 or x > 1.0 + 1e-12:
            continue
        xc = min(1.0, max(0.0, x))
        f = coord_eval(xc, hp, cseg, a, qconst, unaff)
        if f < best_f - 1e-13:
            best_f = f
            best_x = xc
    return best_x, best_f

def cd_sweep(h, n, deadline, project_every=12):
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    center = n - 1
    L = 2 * n - 1
    improved = False
    n_improve = 0
    for p in range(n):
        if time.time() > deadline:
            return h, (2.0 / n) * cur, True
        hp = float(h[p])
        lo = min(p, center - p)
        hi = max(center + p, 2 * center - p)
        seg = hi - lo + 1
        s_arr = center - np.arange(lo, hi + 1, dtype=np.int64)
        j = p + s_arr
        i = p - s_arr
        a = np.zeros(seg, dtype=float)
        vj = (j >= 0) & (j < n)
        vi = (i >= 0) & (i < n)
        if np.any(vj):
            a[vj] += (1.0 - h[j[vj]])
        if np.any(vi):
            a[vi] -= h[i[vi]]
        off = center - lo
        cseg = np.empty(seg - 1, dtype=float)
        asg = np.empty(seg - 1, dtype=float)
        if off:
            cseg[:off] = c[lo:center]
            asg[:off] = a[:off]
        if hi > center:
            cseg[off:] = c[center + 1:hi + 1]
            asg[off:] = a[off + 1:]
        qconst = c[center] - hp * (1.0 - hp)
        unaff = 0.0
        if lo > 0:
            unaff = max(unaff, float(c[:lo].max()))
        if hi + 1 < L:
            unaff = max(unaff, float(c[hi + 1:].max()))
        bx, bf = optimize_coordinate(hp, cseg, asg, qconst, unaff, 22)
        if bf < cur - 1e-13:
            dxv = bx - hp
            if off:
                c[lo:center] += asg[:off] * dxv
            if hi > center:
                c[center + 1:hi + 1] += asg[off:] * dxv
            c[center] = qconst + bx * (1.0 - bx)
            h[p] = bx
            cur = bf
            improved = True
            n_improve += 1
        if (p + 1) % project_every == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            cur = float(c.max())
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    return h, (2.0 / n) * cur, improved

def vdc_binary(n, rng):
    """Van der Corput inspired binary construction."""
    k = n // 2
    idx = np.arange(n)
    # Use bit-reversal as a low-discrepancy sequence
    bits = np.zeros(n, dtype=np.int64)
    for i in range(n):
        v = i
        b = 0
        m = max(1, (1 << int(np.ceil(np.log2(n)))))
        for _ in range(int(np.ceil(np.log2(n))) + 1):
            b = (b << 1) | (v & 1)
            v >>= 1
        bits[i] = b
    order = np.argsort(bits)
    h = np.zeros(n)
    h[order[:k]] = 1.0
    return project_feasible(h, n)

def fourier_perturb(base, n, rng):
    h = base.copy()
    x = (np.arange(n) + 0.5) * (2.0 / n) - 1.0
    pert = np.zeros(n)
    for k in range(1, 8):
        amp = rng.uniform(0.02, 0.12) / (k * 0.7)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(k * np.pi * x + phase)
    return project_feasible(h + pert, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = None

    if h_init is not None:
        n_base = len(h_init)
        if n_base % 2:
            n_base += 1
            h_init = resample_h(h_init, n_base)
    else:
        n_base = 500
        x = (np.arange(n_base) + 0.5) * (2.0 / n_base)
        h_init = project_feasible(1.0 - np.abs(x - 1.0), n_base)

    best_h = project_feasible(h_init, n_base)
    best_c = compute_c(best_h, n_base)

    # Phase 1: Intensive coordinate descent
    for sweep in range(80):
        if time.time() > deadline - 400:
            break
        h, c, improved = cd_sweep(best_h, n_base, deadline, project_every=10)
        if c < best_c - 1e-13:
            best_h, best_c = h, c
        if not improved:
            break

    # Phase 2: Multi-resolution
    for nt in [n_base, 700, 900, 400, 320, 256, 200]:
        if time.time() > deadline - 200:
            break
        h0 = resample_h(best_h, nt)
        for sweep in range(25):
            if time.time() > deadline - 200:
                break
            h0, c0, improved = cd_sweep(h0, nt, deadline, project_every=8)
            if c0 < best_c - 1e-13:
                if nt == n_base:
                    best_h, best_c = h0, c0
                else:
                    hr = resample_h(h0, n_base)
                    cr = compute_c(hr, n_base)
                    if cr < best_c:
                        best_h, best_c = hr, cr
            if not improved:
                break

    # Phase 3: Structured perturbations
    for it in range(50):
        if time.time() > deadline - 60:
            break
        typ = it % 6
        if typ == 0:
            h0 = fourier_perturb(best_h, n_base, rng)
        elif typ == 1:
            h0 = best_h.copy()
            t = rng.uniform(0.35, 0.65)
            h0[h0 > t] = 1.0
            h0[h0 <= t] = 0.0
            h0 = project_feasible(h0, n_base)
        elif typ == 2:
            h0 = 1.0 - best_h[::-1].copy()
            h0 = project_feasible(h0, n_base)
        elif typ == 3:
            h0 = vdc_binary(n_base, rng)
        elif typ == 4:
            h0 = best_h.copy()
            mid = (h0 > 0.2) & (h0 < 0.8)
            if mid.any():
                flip = rng.random(n_base) > 0.5
                h0[mid & flip] = 1.0
                h0[mid & ~flip] = 0.0
            h0 = project_feasible(h0, n_base)
        else:
            h0 = project_feasible(best_h + rng.normal(0, 0.04, n_base), n_base)

        for sweep in range(12):
            if time.time() > deadline - 60:
                break
            h0, c0, improved = cd_sweep(h0, n_base, deadline, project_every=10)
            if c0 < best_c - 1e-13:
                best_h, best_c = h0, c0
            if not improved:
                break

    # Phase 4: Final intensive polish
    for sweep in range(100):
        if time.time() > deadline - 3:
            break
        best_h, best_c, improved = cd_sweep(best_h, n_base, deadline, project_every=6)
        if not improved:
            break

    best_h = project_feasible(best_h, n_base)
    best_c = compute_c(best_h, n_base)
    return best_h, best_c, n_base
