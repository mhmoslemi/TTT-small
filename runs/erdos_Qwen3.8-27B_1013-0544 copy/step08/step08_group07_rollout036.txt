```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def compute_correlation(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def fw_subgradient(h, n):
    """Compute subgradient of f(h) = max_m c_m(h)."""
    c = compute_correlation(h, n)
    max_val = c.max()
    # Find active set (within tolerance)
    tol = 1e-10 * max_val + 1e-15
    active = np.where(c >= max_val - tol)[0]
    # Average gradient over active set
    g = np.zeros(n)
    for m in active:
        shift = m - (n - 1)
        # c_m = sum_j h[j] * (1 - h[j + shift])
        # d c_m / d h[i] = (1 - h[i+shift])_valid - h[i-shift]_valid
        # First term: h[i] appears as h[j] where j = i, contributing (1 - h[i+shift]) if i+shift valid
        # Second term: h[i] appears as h[j+shift] where j+shift = i, i.e., j = i-shift, contributing -h[i-shift] if i-shift valid
        # (1 - h[i+shift]) for valid i+shift
        idx1 = np.arange(n) + shift
        valid1 = (idx1 >= 0) & (idx1 < n)
        g[valid1] += (1.0 - h[idx1[valid1]])
        # -h[i-shift] for valid i-shift
        idx2 = np.arange(n) - shift
        valid2 = (idx2 >= 0) & (idx2 < n)
        g[valid2] -= h[idx2[valid2]]
    g /= len(active)
    return g, max_val

def fw_linear_min(g, n):
    """Find s in S that minimizes <g, s>. S = {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2
    idx = np.argsort(g)
    s = np.zeros(n)
    s[idx[:int(target)]] = 1.0
    return s

def frank_wolfe(h_init, n, max_iter=500, deadline=None):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for k in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        g, max_val = fw_subgradient(h, n)
        s = fw_linear_min(g, n)
        # Standard FW step size
        t = 2.0 / (k + 2)
        h_new = (1 - t) * h + t * s
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if np.linalg.norm(h_new - h) < 1e-10:
            break
        h = h_new
    return best_h, best_c
```

    """One step of linearized cutting plane."""
    c = np.correlate(h, 1.0-h, mode='full')
    max_val = c.max()
    # Active set
    tol = 0.01 * max_val
    active = np.where(c >= max_val - tol)[0]
    
    # Compute gradients of active c[m]
    # For active m: gradient g[m][i] = (1-h[i+m-(n-1)])[valid] - h[i-m+(n-1)][valid]
    # We want to find h' that minimizes the max linearized c
    # LP: min t s.t. c[m] + g_m·(h'-h) ≤ t for m in active, h' ∈ S
    #     Minimize sum of linearized c for active m (simpler)
    
    # Simpler: minimize sum_{m in active} (c[m] + g_m·(h'-h))
    # = sum(c[active]) + (sum_{m in active} g_m)·(h'-h)
    # So we want to minimize (sum_g)·h' where sum_g = sum of active gradients
    
    g_total = np.zeros(n)
    for m in active:
        shift = m - (n-1)
        for i in range(n):
            j1 = i + shift
            j2 = i - shift
            if 0 <= j1 < n:
                g_total[i] += (1.0 - h[j1])
            if 0 <= j2 < n:
                g_total[i] -= h[j2]
    
    # Minimize g_total · h' subject to 0≤h'≤1, sum(h')=n/2
    # Solution: h'[i] = 1 where g_total[i] is smallest (n/2 of them), 0 elsewhere
    idx = np.argsort(g_total)
    h_new = np.zeros(n)
    h_new[idx[:n//2]] = 1.0
    
    # Mix with current: h = (1-α)h + α*h_new
    return h_new, g_total

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

            if tot < 1e-15: break

            tot = h.sum()
            if tot < 1e-15: break

def compute_grad_c(h, n, m):
    """Compute gradient of c[m] w.r.t. h."""
    shift = m - (n - 1)
    g = np.zeros(n)
    idx = np.arange(n)
    j1 = idx + shift
    j2 = idx - shift
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    g[v1] += (1.0 - h[j1[v1]])
    g[v2] -= h[j2[v2]]
    return g

        if c_fw < best_c:

        h_fw, c_fw = frank_wolfe(best_h, n, max_iter=400, deadline=deadline - 40, tol=0.003)
        if c_fw < best_c:
            best_h, best_c = h_fw, c_fw

    # Phase 2: CD polishing
    if time.time() < deadline - 40:
        h_cd, c_cd = cd_fast(best_h, n, passes=200, deadline=deadline - 40, grid=50, rng=rng)
        if c_cd < best_c:
            best_h, best_c = h_cd, c_cd

    # Phase 3: FW with tighter tolerance + CD
    if time.time() < deadline - 50:
        h_fw, c_fw = frank_wolfe(best_h, n, max_iter=500, deadline=deadline - 50, tol=0.001)
        if c_fw < best_c:
            best_h, best_c = h_fw, c_fw
        h_cd, c_cd = cd_fast(best_h, n, passes=200, deadline=deadline - 40, grid=50, rng=rng)
        if c_cd < best_c:
            best_h, best_c = h_cd, c_cd

    # Phase 4: Kicks + CD
    n_kicks = 6
    for k in range(n_kicks):
        if time.time() > deadline - 120:
            break
        nk = rng.randint(n // 15, n // 4)
        idx = rng.choice(n, nk, replace=False)
        s = 0.03 + 0.08 * (k / max(1, n_kicks - 1))
        kick_h = best_h.copy()
        kick_h[idx] += rng.normal(0, s, nk)
        kick_h = project_feasible(kick_h, n)
        h_k, c_k = cd_fast(kick_h, n, passes=150, deadline=deadline - 100, grid=45, rng=rng)
        h_k, c_k = frank_wolfe(h_k, n, max_iter=200, deadline=deadline - 100, tol=0.005)
        if c_k < best_c:
            best_h, best_c = h_k, c_k

    # Phase 5: Multi-scale
    if time.time() < deadline - 80:
        for n_s in [300, 200]:
            if time.time() > deadline - 40:
                break
            x_s = np.linspace(0, 2, n_s, endpoint=False)
            x_orig = np.linspace(0, 2, n, endpoint=False)
            h_s = np.interp(x_s, x_orig, best_h)
            h_s = project_feasible(h_s, n_s)
            h_s, c_s = frank_wolfe(h_s, n_s, max_iter=300, deadline=deadline - 60, tol=0.005)
            h_s, c_s = cd_fast(h_s, n_s, passes=200, deadline=deadline - 40, grid=40, rng=rng)
            h_back = np.interp(x_orig, x_s, h_s)
            h_back = project_feasible(h_back, n)
            h_back, c_back = cd_fast(h_back, n, passes=100, deadline=deadline - 20, grid=50, rng=rng)
            if c_back < best_c:
                best_h, best_c = h_back, c_back

    # Phase 6: Fourier-based starts
    if time.time() < deadline - 60:
        for fm in [4, 7, 12]:
            if time.time() > deadline - 30:
                break
            h_f = fourier_start(n, fm, rng)
            h_f, c_f = frank_wolfe(h_f, n, max_iter=300, deadline=deadline - 40, tol=0.01)
            h_f, c_f = cd_fast(h_f, n, passes=150, deadline=deadline - 30, grid=45, rng=rng)
            if c_f < best_c:
                best_h, best_c = h_f, c_f

    # Phase 7: L-BFGS-B final polish
    if time.time() < deadline - 20:
        h_l, c_l = lbfgs_refine(best_h, n, [5000, 20000, 100000], maxiter=400, deadline=deadline - 10)
        if c_l < best_c:
            best_h, best_c = h_l, c_l

    # Phase 8: Final CD
    if time.time() < deadline - 3:
        h_f, c_f = cd_fast(best_h, n, passes=150, deadline=deadline - 2, grid=80, rng=rng)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
