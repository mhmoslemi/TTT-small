```python
import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def coord_cd(h, n, max_passes=200, deadline=None, grid_size=80):
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            q_arr = p - ms + (n - 1)
            p_arr = p + ms - (n - 1)
            valid_q = (q_arr >= 0) & (q_arr < n)
            valid_p = (p_arr >= 0) & (p_arr < n)
            a_coeff[valid_q] += (1.0 - h[q_arr[valid_q]])
            a_coeff[valid_p] -= h[p_arr[valid_p]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo = -h0
            hi = 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals = np.array([obj_1d(d, c_aff, a_coeff, diag_local, unaff_max) for d in d_grid])
            best_idx = int(np.argmin(vals))
            best_d = d_grid[best_idx]
            best_val = float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                gl, gh = d_grid[best_idx - 1], d_grid[best_idx + 1]
                try:
                    res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=50)
                    if res.fun < best_val:
                        best_d, best_val = float(res.x), float(res.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def pair_cd(h, n, max_passes=30, deadline=None, pairs_per_pass=200):
    h = h.copy()
    rng = np.random.RandomState(123)
    current_max = compute_c5(h, n)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for _ in range(pairs_per_pass):
            if deadline is not None and time.time() > deadline:
                break
            i, j = rng.randint(0, n, 2)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            if hi < 1e-10 or hi > 1 - 1e-10 or hj < 1e-10 or hj > 1 - 1e-10:
                continue
            m_lo = max(0, n - 1 - max(i, j))
            m_hi = min(2 * n - 2, n - 1 + max(i, j))
            ms = np.arange(m_lo, m_hi + 1)
            # grad for i
            gi = np.zeros(len(ms))
            qi = i - ms + (n - 1)
            pi = i + ms - (n - 1)
            vq = (qi >= 0) & (qi < n)
            vp = (pi >= 0) & (pi < n)
            gi[vq] += (1.0 - h[qi[vq]])
            gi[vp] -= h[pi[vp]]
            # grad for j
            gj = np.zeros(len(ms))
            qj = j - ms + (n - 1)
            pj = j + ms + (n - 1) - (n-1)  # same as j + ms - n + 1
            qj = j - ms + (n - 1)
            pj = j + ms - (n - 1)
            vq = (qj >= 0) & (qj < n)
            vp = (pj >= 0) & (pj < n)
            gj[vq] += (1.0 - h[qj[vq]])
            gj[vp] -= h[pj[vp]]
            # overlap
            ov = np.zeros(len(ms))
            mij = i - j + (n - 1)
            mj = j - i + (n - 1)
            if m_lo <= mij <= m_hi:
                ov[mij - m_lo] = 1.0
            if m_lo <= mj <= m_hi:
                ov[mj - m_lo] = 1.0
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = float(c[:m_lo].max())
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            d_lo = max(-hi, hj - 1.0)
            d_hi = min(1.0 - hi, hj)
            if d_hi - d_lo < 1e-10:
                continue
            grid = np.linspace(d_lo, d_hi, 60)
            best_d, best_v = 0.0, current_max
            for d in grid:
                v = max(float((c_aff + d * (gi - gj) + d * d * ov).max()), unaff)
                if v < best_v:
                    best_v, best_d = v, d
            if best_v < current_max - 1e-13:
                h[i] = hi + best_d
                h[j] = hj - best_d
                np.clip(h, 0.0, 1.0, out=h)
                current_max = best_v
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def topk_polish(h, n, k=3, max_iter=200, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.002
    stall = 0
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c)[-k:]
        G = np.zeros(n)
        for mi in top_idx:
            for i in range(n):
                q = i + mi - (n - 1)
                if 0 <= q < n:
                    G[i] += (1.0 - h[q])
                p = i - mi + (n - 1)
                if 0 <= p < n:
                    G[i] -= h[p]
        G /= k
        h_new = project_feasible(h - step * G, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            h = h_new
            best_c = c_new
            stall = 0
        else:
            step *= 0.6
            stall += 1
            if stall > 20 or step < 1e-8:
                break
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=400):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def coord_cd(h, n, max_passes=200, deadline=None, grid_size=80):
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            q_arr = p - ms + (n - 1)
            p_arr = p + ms - (n - 1)
            vq = (q_arr >= 0) & (q_arr < n)
            vp = (p_arr >= 0) & (p_arr < n)
            a_coeff[vq] += (1.0 - h[q_arr[vq]])
            a_coeff[vp] -= h[p_arr[vp]]
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = float(c[:m_lo].max())
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo, hi = -h0, 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals = np.array([obj_1d(d, c_aff, a_coeff, diag_local, unaff) for d in d_grid])
            best_idx = int(np.argmin(vals))
            best_d, best_val = float(d_grid[best_idx]), float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                try:
                    res = golden(obj_1d, d_grid[best_idx-1], d_grid[best_idx+1],
                                 args=(c_aff, a_coeff, diag_local, unaff), maxiter=60)
                    if res.fun < best_val:
                        best_d, best_val = float(res.x), float(res.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def pair_cd(h, n, max_passes=30, deadline=None, pairs_per_pass=200):

        current_max = float(c.max())
        improved = False

            i, j = rng.randint(0, n, 2)

            hi, hj = h[i], h[j]
            if hi < 1e-10 or hi > 1 - 1e-10 or hj < 1e-10 or hj > 1 - 1e-10:
                continue
            m_lo = max(0, n - 1 - max(i, j))
            m_hi = min(2 * n - 2, n - 1 + max(i, j))
            ms = np.arange(m_lo, m_hi + 1)

            qi = i - ms + (n - 1)
            pi = i + ms - (n - 1)
            vq = (qi >= 0) & (qi < n)
            vp = (pi >= 0) & (pi < n)
            gi[vq] += (1.0 - h[qi[vq]])
            gi[vp] -= h[pi[vp]]
            gj = np.zeros(len(ms))
            qj = j - ms + (n - 1)
            pj = j + ms - (n - 1)
            vq2 = (qj >= 0) & (qj < n)
            vp2 = (pj >= 0) & (pj < n)
            gj[vq2] += (1.0 - h[qj[vq2]])
            gj[vp2] -= h[pj[vp2]]
            ov = np.zeros(len(ms))
            mij = i - j + (n - 1)
            mj = j - i + (n - 1)
            if m_lo <= mij <= m_hi:
                ov[mij - m_lo] = 1.0
            if m_lo <= mj <= m_hi:
                ov[mj - m_lo] = 1.0
            c_aff = c[ms]
            unaff = 0.0

                h[j] = np.clip(hj - best_d, 0.0, 1.0)
                current_max = best_v
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

            v1 = (idx1 >= 0) & (idx1 < n)

    for n_try in [500, 700, 450]:
        if time.time() > deadline - 40:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = slsqp_run(h_int, n_try, [500, 5000], 300)
        h_opt, c_opt = coord_cd(h_opt, n_try, 60, deadline, grid_size=80)
        h_opt, c_opt = pair_cd(h_opt, n_try, 15, deadline, pairs_per_pass=100)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new
    n_cur = len(best_h)
    if time.time() < deadline - 20:
        hf, cf = coord_cd(best_h, n_cur, 200, deadline, grid_size=200)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 5:
        hf, cf = pair_cd(best_h, n_cur, 15, deadline, pairs_per_pass=150)
        if cf < best_c:
            best_h, best_c = hf, cf
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
