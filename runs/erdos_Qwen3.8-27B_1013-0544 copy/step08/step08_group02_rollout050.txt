def compute_A(h, n):
    """A[i, k] = d(c_k)/d(h[i]) for all i, k. Shape (n, 2n-1)."""
    A = np.zeros((n, 2 * n - 1))

    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[d:, k] = 1.0 - h[:n - d]
            A[:n - d, k] -= h[d:]
        else:
            e2 = -d
            A[:n - e2, k] = 1.0 - h[e2:]
            A[e2:, k] -= h[:n - e2]
    return A

def pair_line_search(h, c, A, i, j, n, grid_coarse, grid_fine):

    B = A[i] - A[j]
    off = n - 1
    hi, hj = h[i], h[j]
    t_lo = max(-hi, -(1.0 - hj))
    t_hi = min(1.0 - hi, hj)
    if t_hi - t_lo < 1e-14:
        return None, None
    
    # Coarse search
    ts = np.linspace(t_lo, t_hi, grid_coarse)
    best_t, best_val = 0.0, float(c.max())
    for t in ts:
        cn = c + t * B
        cn[off] -= 2.0 * t * t
        cn[off + i - j] += t * t
        cn[off + j - i] += t * t
        v = cn.max()
        if v < best_val:
            best_val = v
            best_t = t
    
    # Fine search around best_t
    if best_t > t_lo + 1e-14 and best_t < t_hi - 1e-14:
        span = (t_hi - t_lo) / grid_coarse
        ts2 = np.linspace(best_t - span, best_t + span, grid_fine)
        ts2 = ts2[(ts2 >= t_lo) & (ts2 <= t_hi)]
        for t in ts2:
            cn = c + t * B
            cn[off] -= 2.0 * t * t
            cn[off + i - j] += t * t
            cn[off + j - i] += t * t

                best_val = v
                best_t = t

                A = compute_A(h, n)
        if stagnation > max_stag:
            break
    return h, compute_c5(h, n)

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        n0 = 400
        x = np.linspace(0, 2, n0, endpoint=False)
        h0 = 0.5 + 0.25 * np.sin(2 * np.pi * x) + 0.12 * np.sin(4 * np.pi * x + 0.3)
        h0 = project_feasible(h0, n0)

    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)

    # Symmetries
    hs = project_feasible(0.5 * (best_h + best_h[::-1]), n0)
    cs = compute_c5(hs, n0)
    if cs < best_c:
        best_h, best_c = hs, cs

    # Phase 1: SLSQP moderate (120s)
    dl1 = min(hard - 400.0, t0 + 150.0)
    if time.time() < dl1:
        best_h, best_c = slsqp_opt(best_h, n0, [50, 200, 800, 3000], 150, dl1)

    # Phase 2: Main pair search (350s)
    dl2 = min(hard - 300.0, t0 + 500.0)
    if time.time() < dl2:
        best_h, best_c = enhanced_pair_search(best_h, n0, 12000, dl2, rng)

    # Phase 3: Resolution sweep
    for m in [320, 384, 448, 480, 520, 560, 640, 700]:
        if time.time() > hard - 200.0:
            break
        if m == n0:
            continue
        rem = hard - time.time()
        if rem < 50:
            break
        if m > 550 and rem < 80:
            continue
        dl = time.time() + min(40.0, 0.12 * rem)
        if dl < time.time() + 12:
            continue
        h_m = interp_h(best_h, m)
        h_m, c_m = slsqp_opt(h_m, m, [200, 1000], 100, dl)
        rem2 = hard - time.time() - 50.0
        if rem2 > 10:
            h_m, c_m = enhanced_pair_search(h_m, m, min(4000, int(rem2 * 20)), time.time() + min(30, rem2), rng)
        if c_m < best_c:
            best_h, best_c = h_m, c_m

    # Phase 4: Fine SLSQP (100s)
    if time.time() < hard - 50.0:
        n = len(best_h)
        dl4 = min(hard - 30.0, time.time() + 80.0)
        best_h, best_c = slsqp_opt(best_h, n, [5000, 20000, 80000], 300, dl4)

    # Phase 5: Final pair search
    if time.time() < hard - 10.0:
        n = len(best_h)
        rem = hard - time.time() - 8.0
        mt = int(min(8000, max(1000, rem * 30)))
        best_h, best_c = enhanced_pair_search(best_h, n, mt, hard - 3.0, rng)

    # Phase 6: Nano SLSQP
    if time.time() < hard - 3.0:
        n = len(best_h)
        best_h, best_c = slsqp_opt(best_h, n, [200000], 200, hard - 2.0)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
