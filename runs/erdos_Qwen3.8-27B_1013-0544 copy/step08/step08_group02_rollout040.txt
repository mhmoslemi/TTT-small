import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)


def raw_c(h):
    return np.correlate(h, 1.0 - h, mode="full")


def c_cq(h, n):
    cv = raw_c(h)
    cq = float(cv.max())
    return cv, cq, cq * 2.0 / n


def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(60):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = (y < 1.0) if res > 0 else (y > 0.0)
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)


def sum_con(x, n):
    return float(x.sum() - 0.5 * n)


def sum_jac(x, n):
    return np.ones_like(x, dtype=float)


def smooth_components(h, n, alpha):
    c = raw_c(h)
    m = float(c.max())
    if alpha <= 1e-8:
        p = np.zeros(c.shape, dtype=float)
        p[int(np.argmax(c))] = 1.0
        return c, m, p
    z = alpha * (c - m)
    z = z - float(z.max())
    e = np.exp(z)
    s = float(e.sum())
    p = e / s
    val = m + float(np.log(s)) / alpha
    return c, val, p


def smooth_val(x, n, alpha):
    return smooth_components(x, n, alpha)[1]


def grad_weights(h, p, n):
    g = np.zeros(n, dtype=float)
    off = n - 1
    for k in np.nonzero(p > 1e-14)[0]:
        w = float(p[k])
        d = int(k) - off
        if d >= 0:
            m = n - d
            g[d:] += w * (1.0 - h[:m])
            g[:m] -= w * h[d:]
        else:
            e = -d
            m = n - e
            g[:m] += w * (1.0 - h[e:])
            g[e:] -= w * h[:m]
    return g


def smooth_grad(x, n, alpha):
    _, _, p = smooth_components(x, n, alpha)
    return grad_weights(x, p, n)


def lbf_val(x, n, alpha, Tq, gamma, active, mu):
    c, val, _ = smooth_components(x, n, alpha)
    res = val
    if gamma > 0 and active is not None and len(active):
        pos = np.maximum(c[active] - Tq, 0.0)
        res += 0.5 * gamma * float(np.dot(pos, pos))
    dev = float(x.sum() - 0.5 * n)
    if mu > 0:
        res += 0.5 * mu * dev * dev
    return res


def lbf_grad(x, n, alpha, Tq, gamma, active, mu):
    c, _, p = smooth_components(x, n, alpha)
    g = grad_weights(x, p, n)
    if gamma > 0 and active is not None and len(active):
        pos = np.maximum(c[active] - Tq, 0.0)
        w = np.zeros(c.shape, dtype=float)
        w[active] = gamma * pos
        g += grad_weights(x, w, n)
    dev = float(x.sum() - 0.5 * n)
    if mu > 0:
        g += mu * dev
    return g


def lbf_stage(h, n, bounds, plans, Tq, gamma, active, mu, deadline):
    best = h.copy()
    bestc = compute_c5(best, n)
    work = best.copy()
    for alpha, mx in plans:
        rem = deadline - time.time()
        if rem < 5.0:
            break
        mx = int(min(mx, max(10, rem * 25)))
        try:
            res = minimize(
                lbf_val, work, args=(n, alpha, Tq, gamma, active, mu),
                jac=lbf_grad, method="L-BFGS-B", bounds=bounds,
                options={"maxiter": mx, "ftol": 1e-14, "maxls": 20, "disp": False}
            )
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < bestc:
                best, bestc = cur, cc
            work = cur if cc <= bestc + 1e-10 else best
        except Exception:
            pass
    return best, bestc


def slsqp_stage(h, n, bounds, cons, plans, deadline):
    best = h.copy()
    bestc = compute_c5(best, n)
    work = best.copy()
    for alpha, mx in plans:
        rem = deadline - time.time()
        if rem < 5.0:
            break
        mx = int(min(mx, max(10, rem * 2)))
        try:
            res = minimize(
                smooth_val, work, args=(n, alpha),
                jac=smooth_grad, method="SLSQP", bounds=bounds, constraints=cons,
                options={"maxiter": mx, "ftol": 1e-13, "disp": False}
            )
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < bestc:
                best, bestc = cur, cc
            work = cur if cc <= bestc + 1e-10 else best
        except Exception:
            pass
    return best, bestc


def compute_A(h, n):
    L = 2 * n - 1
    off = n - 1
    A = np.zeros((n, L), dtype=float)
    for d in range(1, n):
        m = n - d
        A[d:, off + d] += 1.0 - h[:m]
        A[:m, off + d] -= h[d:d + m]
        A[:m, off - d] += 1.0 - h[d:d + m]
        A[d:d + m, off - d] -= h[:m]
    A[:, off] = 1.0 - 2.0 * h
    return A


def top_indices(c, k):
    k = int(min(k, c.size))
    if k <= 0:
        return np.array([], dtype=int)
    return np.argpartition(c, -k)[-k:]


def select_active(c, Tq, cap):
    cap = int(min(cap, c.size))
    if cap <= 0:
        return np.array([], dtype=int)
    idx = np.where(c > Tq - 0.05)[0]
    if idx.size == 0:
        idx = top_indices(c, cap)
    elif idx.size > cap:
        take = np.argpartition(c[idx], -cap)[-cap:]
        idx = idx[take]
    return np.sort(idx)


def add_cand(lst, u, lo, hi):
    if lo - 1e-12 <= u <= hi + 1e-12:
        if u < lo:
            u = lo
        elif u > hi:
            u = hi
        lst.append(float(u))


def pair_choose_u(h, c, A, i, j, n, active):
    if active is None or len(active) == 0 or i == j:
        return None
    hii = float(h[i])
    hjj = float(h[j])
    lo = max(-hii, hjj - 1.0)
    upper = min(hjj, 1.0 - hii)
    if upper <= lo + 1e-15:
        return None

    B = A[i] - A[j]
    MA = len(active)
    CA = c[active]
    BA = B[active]
    coef = np.zeros(MA, dtype=float)
    corr = []
    off = n - 1
    for kk, cv in ((off, -2.0), (off + i - j, 1.0), (off + j - i, 1.0)):
        if kk < active[0] or kk > active[-1]:
            continue
        idx = int(np.searchsorted(active, kk))
        if idx < MA and active[idx] == kk:
            coef[idx] += cv
            corr.append(idx)

    cands = [lo, upper, 0.25 * (lo + upper), 0.5 * (lo + upper), 0.75 * (lo + upper)]
    for a in range(MA):
        ca = coef[a]
        ba = BA[a]
        if abs(ca) > 1e-14:
            add_cand(cands, -ba / (2.0 * ca), lo, upper)
        for b in range(a + 1, MA):
            c2 = ca - coef[b]
            c1 = ba - BA[b]
            c0 = CA[a] - CA[b]
            if abs(c2) > 1e-14:
                disc = c1 * c1 - 4.0 * c2 * c0
                if disc >= -1e-12:
                    disc = max(0.0, disc)
                    sq = float(np.sqrt(disc))
                    add_cand(cands, (-c1 + sq) / (2.0 * c2), lo, upper)
                    add_cand(cands, (-c1 - sq) / (2.0 * c2), lo, upper)
            elif abs(c1) > 1e-14:
                add_cand(cands, -c0 / c1, lo, upper)

    arr = np.asarray(cands, dtype=float)
    arr = arr[(arr >= lo - 1e-12) & (arr <= upper + 1e-12)]
    if arr.size == 0:
        return None
    arr = np.clip(arr, lo, upper)
    arr.sort()
    uniq = []
    last = None
    for uu in arr:
        if last is None or abs(uu - last) > 1e-9:
            uniq.append(float(uu))
            last = float(uu)
    if not uniq:
        return None

    listv = []
    for u in uniq:
        vals = CA + u * BA
        for p in corr:
            vals[p] += coef[p] * u * u
        vm = float(vals.max())
        listv.append((vm, u))
        listv.sort()
        if len(listv) > 5:
            listv.pop()

    cur_max = float(c.max())
    best_full = cur_max
    best_u = None
    for _, u in listv:
        u2 = u * u
        cn = c + u * B
        cn[off] -= 2.0 * u2
        cn[off + i - j] += u2
        cn[off + j - i] += u2
        fm = float(cn.max())
        if fm < best_full:
            best_full = fm
            best_u = u
    if best_u is not None and best_full < cur_max - 1e-10:
        return float(best_u)
    return None


def targeted_pair_search(h, c, A, n, max_tries, deadline, rng, topK=16):
    if n < 2 or max_tries <= 0:
        return h, raw_c(h)
    h = h.copy()
    c = c.copy()
    A = A.copy()
    L = 2 * n - 1
    off = n - 1

    active = np.sort(top_indices(c, topK))
    accepts = 0
    no_improve = 0
    max_accept = min(250, max(40, max_tries // 12))

    for trial in range(max_tries):
        if time.time() > deadline:
            break
        if trial % 97 == 0:

            d = kk - off

            else:

                a = t - e

            ha = float(h[a])
            hb = float(h[b2])
            if abs(ha - hb) < 1e-15:
                if trial % 5 == 3:
                    i, j = a, b2
                else:
                    continue
            elif hb > ha:
                if trial % 5 == 3:
                    i, j = b2, a
                else:
                    i, j = a, b2
            else:
                if trial % 5 == 3:
                    i, j = a, b2
                else:
                    i, j = b2, a
        else:
            i = int(rng.randint(n))

                i, j = j, i

        u = pair_choose_u(h, c, A, i, j, n, active)

        else:
            new_i = float(h[i]) + u

                no_improve += 1
                continue
            h[i] = new_i
            h[j] = new_j
            u2 = u * u
            c += u * (A[i] - A[j])

            c[off + i - j] += u2
            c[off + j - i] += u2
            accepts += 1
            no_improve = 0
            A = compute_A(h, n)
            if accepts % 20 == 0:
                c = raw_c(h)
            active = np.sort(top_indices(c, topK))
            if accepts >= max_accept:
                break
        if no_improve > max(600, max_tries // 2):
            break

    h = project_feasible(h, n)
    return h, raw_c(h)


def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)


def random_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, 7):
        a = rng.uniform(-0.35, 0.35)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += a * np.sin(0.5 * np.pi * m * x + ph)
        h += 0.5 * a * np.cos(np.pi * m * x + ph)
    return project_feasible(np.clip(h, 0.0, 1.0), n)


def optimize_instance(h0, n, deadline, rng, quick=False):
    h = project_feasible(h0, n)
    cvec, cq, bestc = c_cq(h, n)

    hx = project_feasible(0.5 * (h + h[::-1]), n)
    cx = compute_c5(hx, n)
    if cx < bestc:
        h, bestc = hx, cx
        cvec, cq, bestc = c_cq(h, n)

    hx = 1.0 - h
    cx = compute_c5(hx, n)
    if cx < bestc:
        h, bestc = hx, cx
        cvec, cq, bestc = c_cq(h, n)

    hx = project_feasible(0.5 * (h + 1.0 - h[::-1]), n)
    cx = compute_c5(hx, n)
    if cx < bestc:
        h, bestc = hx, cx
        cvec, cq, bestc = c_cq(h, n)

    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_jac, "args": (n,)}]
    empty = np.array([], dtype=int)

    if quick:
        if time.time() < deadline - 12.0:
            cvec, cq, _ = c_cq(h, n)
            Tq = min(cq - 0.02, 0.38065 * 0.5 * n)
            active = select_active(cvec, Tq, 40)
            h, bestc = lbf_stage(h, n, bounds, [(200, 50), (1000, 50)], Tq, 50.0, active, 1e4, deadline - 5.0)

        if time.time() < deadline - 6.0:
            h, bestc = slsqp_stage(h, n, bounds, cons, [(100, 50), (500, 40)], deadline - 3.0)
    else:
        if time.time() < deadline - 50.0:
            h, bestc = lbf_stage(
                h, n, bounds,
                [(100, 50), (500, 80), (2000, 70), (6000, 50)],
                0.0, 0.0, empty, 1e4, deadline - 30.0
            )

        if time.time() < deadline - 50.0:
            targetQ = 0.38065 * 0.5 * n
            steps = (0.10, 0.04, 0.018, 0.009, 0.0045)
            gammas = (20.0, 60.0, 120.0, 250.0, 500.0)
            for st, gm in zip(steps, gammas):
                if time.time() > deadline - 25.0:
                    break
                cvec, cq, _ = c_cq(h, n)
                Tq = min(cq - st, targetQ)
                if Tq < 0.0:
                    Tq = 0.0
                active = select_active(cvec, Tq, 60)
                if active.size == 0:
                    break
                plans = [(200, 80), (1000, 70)]
                if time.time() < deadline - 80.0:
                    plans.append((4000, 50))
                h, bestc = lbf_stage(h, n, bounds, plans, Tq, gm, active, 1e4, deadline - 5.0)
                cvec, cq, _ = c_cq(h, n)
                if cq <= targetQ + 1e-7:
                    break

        if time.time() < deadline - 15.0:
            plans = [(50, 80), (300, 60)]
            if time.time() < deadline - 80.0:
                plans.append((1500, 50))
            h, bestc = slsqp_stage(h, n, bounds, cons, plans, deadline - 3.0)

    if n <= 800 and time.time() < deadline - 4.0:
        cvec, cq, _ = c_cq(h, n)
        A = compute_A(h, n)
        rem = deadline - time.time()
        mt = int(min(12000 if not quick else 3000, max(300, rem * 20)))
        h, cvec = targeted_pair_search(h, cvec, A, n, mt, deadline, rng, 16)
        h = project_feasible(h, n)
        bestc = compute_c5(h, n)

    return h, bestc


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 3.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        h0 = random_start(500, rng)

    if h0.size > 800:
        h0 = interp_h(h0, 500)

    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)

    if hard - t0 < 25.0:
        return best_h, best_c, n0

    if hard - t0 < 220.0:
        best_h, best_c = optimize_instance(best_h, n0, hard, rng, True)
        best_h = project_feasible(best_h, n0)
        return best_h, compute_c5(best_h, n0), n0

    main_deadline = hard - max(150.0, 0.28 * float(budget_s))
    if main_deadline < t0 + 60.0:
        main_deadline = hard
    best_h, best_c = optimize_instance(best_h, n0, main_deadline, rng, False)

    for m in (400, 320, 250, 200):
        if time.time() > hard - 40.0:
            break
        if m == len(best_h):
            continue
        rem = hard - time.time()
        if rem < 75.0:
            break
        dl = time.time() + min(75.0, 0.20 * rem)
        if dl < time.time() + 10.0:
            continue
        ht = interp_h(best_h, m)
        ht, ct = optimize_instance(ht, m, dl, rng, True)
        if ct < best_c - 1e-12:
            best_h, best_c = ht, ct

    if time.time() < hard - 90.0:
        n = len(best_h)
        for s in (0.02, 0.05, 0.10):
            if time.time() > hard - 35.0:
                break
            hp = project_feasible(best_h + rng.normal(0.0, s, n), n)
            dl = min(hard - 10.0, time.time() + 45.0)
            ht, ct = optimize_instance(hp, n, dl, rng, True)
            if ct < best_c - 1e-12:
                best_h, best_c = ht, ct

    if time.time() < hard - 12.0:
        best_h, best_c = optimize_instance(best_h, len(best_h), hard, rng, True)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
