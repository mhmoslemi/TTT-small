import numpy as np
from scipy.optimize import minimize
import time

def project_h(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = h.sum() - target
    if abs(diff) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = result.sum() - target
    if abs(diff) > 1e-12:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
    return np.clip(result, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_c5_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = float(c.max())
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    ee = np.exp(shifted)
    z = float(ee.sum())
    w = ee / z
    dx = 2.0 / n
    A = np.convolve(1.0 - h, w, mode="full")
    B = np.convolve(h, w[::-1], mode="full")
    s = n - 1
    val = (m + np.log(z) / alpha) * dx
    grad = (A - B)[s:s + n] * dx
    return val, grad

def sum_penalty(h, n, mu):
    return mu * (h.sum() - 0.5 * n) ** 2

def obj_pen(h, n, alpha, mu):
    v, g = smooth_c5_grad(h, n, alpha)
    d = h.sum() - 0.5 * n
    g = g + 2.0 * mu * d * np.ones_like(h)
    return v + sum_penalty(h, n, mu), g

def coord_descent(h, n, max_passes=8, deadline=None, grid=24, refine=8):
    best = project_h(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    A = np.zeros(L)
    xs0 = np.linspace(0.0, 1.0, grid)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = float(x[p])
            A.fill(0.0)
            A[p:p + n] += om
            st = diag - p
            if st >= 0:
                A[st:st + n] -= x
            cur = float(c.max())
            vals = c[None, :] + (xs0 - old)[:, None] * A[None, :]
            vals[:, diag] -= (xs0 - old) ** 2
            i = int(np.argmin(vals.max(axis=1)))
            bd = float(xs0[i] - old)
            if vals[i].max() < cur - 1e-12:
                # Refine
                sp = 1.0 / (grid - 1)
                lo = max(0.0, xs0[i] - sp)
                hi = min(1.0, xs0[i] + sp)
                xx = np.linspace(lo, hi, refine + 1)
                vv = c[None, :] + (xx - old)[:, None] * A[None, :]
                vv[:, diag] -= (xx - old) ** 2
                k = int(np.argmin(vv.max(axis=1)))
                bd = float(xx[k] - old)
                if bd != 0.0:
                    c[p:p + n] += bd * om
                    if st >= 0:
                        c[st:st + n] -= bd * x
                    c[diag] -= bd * bd
                    x[p] = old + bd
                    om[diag - p] = 1.0 - x[p]
                    improved = True
        xp = project_h(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - 1e-13:
            best = xp
            bc = cp
        if not improved:
            break
    return best, bc

def lfgs_optimize(h0, n, alpha, mu, deadline, maxiter=200):
    """L-BFGS-B with penalty for sum constraint."""
    bounds = [(0.0, 1.0)] * n
    
    def obj_grad(x):
        v, g = smooth_c5_grad(x, n, alpha)
        d = x.sum() - 0.5 * n
        g = g + 2.0 * mu * d
        return v + 0.5 * mu * d * d, g
    
    try:
        res = minimize(obj_grad, h0, jac=True, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        result = project_h(res.x, n)
        return result, compute_c5(result, n)
    except:
        return project_h(h0, n), compute_c5(h0, n)

def transfer(h, n, n_tgt):

import numpy as np
from scipy.optimize import minimize
import time

def project_h(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    if abs(h.sum() - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = result.sum() - target
    if abs(diff) > 1e-12:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
    return np.clip(result, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_obj_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = float(c.max())
    sh = alpha * (c - m)
    sh -= sh.max()
    ee = np.exp(sh)
    z = float(ee.sum())
    w = ee / z
    dx = 2.0 / n
    A = np.convolve(1.0 - h, w, mode="full")
    B = np.convolve(h, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx

def coord_descent(h, n, max_passes, deadline, grid=32, refine=12):
    best = project_h(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    A = np.zeros(L)
    xs0 = np.linspace(0.0, 1.0, grid)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = float(x[p])
            A.fill(0.0)
            A[p:p + n] += om
            st = diag - p
            if st >= 0:
                A[st:st + n] -= x
            cur = float(c.max())
            cand = c[None, :] + (xs0 - old)[:, None] * A[None, :]
            cand[:, diag] -= (xs0 - old) ** 2
            mx = cand.max(axis=1)
            i = int(np.argmin(mx))
            if mx[i] < cur - 1e-12:
                sp = 1.0 / (grid - 1)
                lo = max(0.0, xs0[i] - sp)
                hi = min(1.0, xs0[i] + sp)
                xx = np.linspace(lo, hi, refine + 1)
                vv = c[None, :] + (xx - old)[:, None] * A[None, :]
                vv[:, diag] -= (xx - old) ** 2
                j = int(np.argmin(vv.max(axis=1)))
                bd = float(xx[j] - old)
                if abs(bd) > 1e-14:
                    c[p:p + n] += bd * om
                    if st >= 0:
                        c[st:st + n] -= bd * x
                    c[diag] -= bd * bd
                    x[p] = old + bd
                    om[diag - p] = 1.0 - x[p]
                    improved = True
        xp = project_h(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - 1e-14:
            best = xp
            bc = cp
        if not improved:
            break
    return best, bc

def lbfgs_opt(h0, n, alpha, mu, t_end, maxiter=150):
    bounds = [(0.0, 1.0)] * n
    def og(x):
        v, g = smooth_obj_grad(x, n, alpha)
        d = x.sum() - 0.5 * n
        g = g + 2.0 * mu * d
        return v + 0.5 * mu * d * d, g
    try:
        res = minimize(og, h0, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-11})
        rh = project_h(res.x, n)
        return rh, compute_c5(rh, n)
    except Exception:
        return project_h(h0, n), compute_c5(h0, n)

def transfer(h, n, n_tgt):
    if n == n_tgt:
        return project_h(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_h(np.interp(y, x, h), n_tgt)
