from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            cap = 1.0 - h
            tot = cap.sum()
            if tot > 0:
                h += (diff / tot) * cap
        else:
            tot = h.sum()
            if tot > 0:
                h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
        # Final fix
        res = target - h.sum()
        if abs(res) > 1e-12:
            if res > 0:
                idx = np.where(h < 1.0)[0]
            else:
                idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] = np.clip(h[idx[0]] + res, 0.0, 1.0)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, 'full')
    return float(c.max() * (2.0 / n))

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, 'full')
    z = alpha * c
    zm = z.max()
    return ((zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, 'full')
    center = n - 1
    z = alpha * c - (alpha * c).max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * x)
        elif s > 0:
            g[:n-s] += wm * (1.0 - x[s:])
            g[s:] -= wm * x[:n-s]
        else:
            r = -s
            g[r:] += wm * (1.0 - x[:n-r])
            g[:n-r] -= wm * x[r:]
    return (2.0 / n) * g

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = 0.5 * n - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            cap = 1.0 - h
            tot = cap.sum()
            if tot > 0:
                h += (diff / tot) * cap

            if h.sum() > 0:

            if len(idx) > 0:

    return np.correlate(h, 1.0 - h, 'full')

    return float(get_corr(h).max() * (2.0 / n))

    return x.sum() - 0.5 * n

    c = get_corr(x)
    z = alpha * c
    zm = z.max()
    return ((zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)

            r = -s
            g[r:] += wm * (1.0 - x[:n-r])
            g[:n-r] -= wm * x[r:]

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-10:

            tot = cap.sum()
            if tot > 0:

    return ((zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)
