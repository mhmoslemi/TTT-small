from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(60):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def smooth_components(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    s = float(e.sum())
    p = e / s
    return (m + float(np.log(s)) / alpha) * (2.0 / n), p, c

def grad_weights(x, n, p):
    g = np.zeros(n, dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-13:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n - d])
            g[:n - d] -= w * x[d:]
        else:
            e = -d
            g[:n - e] += w * (1.0 - x[e:])
            g[e:] -= w * x[:n - e]
    return (2.0 / n) * g

def smooth_val(x, n, alpha):
    return smooth_components(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    _, p, _ = smooth_components(x, n, alpha)
    return grad_weights(x, n, p)

def slsqp_optimize(h, n, bounds, cons, plans, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work_h = best_h.copy()
    for alpha, mx in plans:
        if time.time() > deadline - 1.0:
            break
        try:
            res = minimize(smooth_val, work_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_grad, bounds=bounds, constraints=cons,

            cc = compute_c5(cur, n)

            work_h = cur if cc <= best_c + 1e-8 else best_h
        except Exception:
            pass

def compute_A(h, n):
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[d:, k] += 1.0 - h[:n - d]
            A[:n - d, k] -= h[d:]
        else:
            e = -d
            A[:n - e, k] += 1.0 - h[e:]
            A[e:, k] -= h[:n - e]
    return A

def pair_search(h, n, max_tries, grid, deadline, rng):

    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    A = compute_A(h, n)
    accepts = 0
    no_improve = 0
    max_accept = min(200, max(60, max_tries // 4))
    off = n - 1
    
    for trial in range(max_tries):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        tol = max(0.01, 0.001 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        
        # Choose pair
        if trial % 7 == 6 or active.size <= 2:
            i = int(rng.randint(n)); j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
        else:
            kk = int(active[int(rng.randint(active.size))])
            d = kk - off
            if d >= 0:
                m = n - d
                if m <= 0:

                b = int(rng.randint(m)); i = b + d; j = b
            else:
                e = -d
                m = n - e
                if m <= 0:
                    continue
                b = e + int(rng.randint(m)); i = b - e; j = b
        
        # Find best u for this pair
        hi, hj = h[i], h[j]
        S = hi + hj
        lo = max(0.0, S - 1.0)
        hiT = min(1.0, S)
        if hiT - lo <= 1e-14:

            continue
        
        # Use finer grid for the search
        us = np.linspace(lo - hi, hiT - hi, grid)
        B = A[i] - A[j]

        u2 = us * us
        lines[:, off] -= 2.0 * u2
        lines[:, off + i - j] += u2
        lines[:, off + j - i] += u2
        vals = lines.max(axis=1)

            h[i] = t
            h[j] = S - t
            c = cn
            accepts += 1
            no_improve = 0

            if accepts % 15 == 0:

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n

    s = float(e.sum())

                           jac=smooth_grad, bounds=bounds, constraints=cons,
