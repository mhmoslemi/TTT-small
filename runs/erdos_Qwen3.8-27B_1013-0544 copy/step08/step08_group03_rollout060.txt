    # Linear constraints: for k != quad_k, c0[k] + t*A[k] <= v
    # => t <= (v - c0[k])/A[k] if A[k] > 0
    # => t >= (v - c0[k])/A[k] if A[k] < 0

    mask_neg = (A < -1e-14) & (np.arange(len(A)) != quad_k)
    mask_zero = (np.abs(A) <= 1e-14) & (np.arange(len(A)) != quad_k)
    
    def feasible(v):
        L = 0.0
        U = 1.0
        if np.any(mask_zero) and np.any(c0[mask_zero] > v + 1e-15):
            return None
        if np.any(mask_pos):
            U = min(U, float(np.min((v - c0[mask_pos]) / A[mask_pos])))
        if np.any(mask_neg):
            L = max(L, float(np.max((v - c0[mask_neg]) / A[mask_neg])))
        if L > U:

        # Quadratic constraint at quad_k
        a_q = A[quad_k]

            r2 = (a_q + sd) / 2.0

            if max(L, 0.0) <= min(U, r1, 1.0):

            if max(L, r2, 0.0) <= min(U, 1.0):

Actually, one more important thought: the `cd_AD_fast` function computes A and D from the CURRENT h. But in the CD loop, after we update h[idx], the A and D for OTHER coordinates change! So we need to recompute the correlation after each update (or update it incrementally).

In the previous code, the correlation is updated incrementally:
c += (t_new - h0) * A - (t_new² - h0²) * D

This is correct because the change in the correlation when h[idx] changes is exactly (t_new - h0) * A - (t_new² - h0²) * D, where A and D were computed with the current h.

But wait: A depends on h! Specifically, A[n-1+m] includes terms like (1-h[idx+m]). When we update h[idx], the A vectors for OTHER coordinates that involve h[idx] will change.

However, in the incremental update of c, we're using the A that was computed BEFORE the update. After the update, c is correct (because the formula for the change is exact given the pre-update h). But if we then compute A for the next coordinate, we should use the UPDATED h.

So the algorithm should be:
1. Compute c = correlate(h, 1-h)
2. For each idx:
   a. Compute A, D using CURRENT h (which includes all previous updates in this pass)
   b. Compute c0 = c - h0*A + h0²*D (the "base" correlation without h[idx]'s contribution)
   c. Find t* that minimizes max(c0 + t*A - t²*D)
   d. Update: c = c0 + t*A - t²*D, h[idx] = t

Wait, but step (b) assumes that c = c0 + h0*A - h0²*D, which means c0 = c - h0*A + h0²*D. This is correct because the contribution of h[idx] to the correlation is h0*A - h0²*D (by construction of A and D).

But A depends on h! Specifically, A[n-1+m] = (1-h[idx+m])*I(valid) - h[idx-m]*I(valid). If h[idx+m] has been updated in a previous step of this pass, then A reflects that update. This is correct! We want A to reflect the current state of h (with all previous updates applied).

So the algorithm is self-consistent: we use the current h to compute A and D, find the optimal t, update h and c, and move to the next coordinate.

The only issue is that after updating h[idx], the A vectors for subsequent coordinates that involve idx will be different. But since we recompute A for each coordinate using the current h, this is handled correctly.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cd_AD(h, idx, n):
    total = 2 * n - 1
    A = np.zeros(total)
    D = np.zeros(total)
    A[n-1] = 1.0
    D[n-1] = 1.0
    m = np.arange(1, n)
    ip = idx + m
    im = idx - m
    vp = ip < n
    vm = im >= 0
    if np.any(vp):
        A[n - 1 + m[vp]] += (1.0 - h[ip[vp]])
        A[n - 1 - m[vp]] += (-h[ip[vp]])
    if np.any(vm):
        A[n - 1 + m[vm]] += (-h[im[vm]])
        A[n - 1 - m[vm]] += (1.0 - h[im[vm]])
    return A, D

def exact_1d(c0, A, D, n):
    quad_k = n - 1
    mask = np.arange(len(c0)) != quad_k
    ap = A[mask]; ap_pos = ap[ap > 1e-14]; ap_neg = ap[ap < -1e-14]
    cp = c0[mask]
    # Precompute for speed
    pos_idx = np.where((A > 1e-14) & mask)[0]
    neg_idx = np.where((A < -1e-14) & mask)[0]
    zero_bad = np.where((np.abs(A) <= 1e-14) & mask & (c0 > 0))[0]  # will check at runtime
    
    a_q = A[quad_k]
    c_q = c0[quad_k]
    
    def feasible(v):
        L, U = 0.0, 1.0
        if len(pos_idx) > 0:
            U = min(U, float(np.min((v - c0[pos_idx]) / A[pos_idx])))
        if len(neg_idx) > 0:
            L = max(L, float(np.max((v - c0[neg_idx]) / A[neg_idx])))
        # Check zero-A constraints
        zero_mask = (np.abs(A) <= 1e-14) & mask
        if np.any(zero_mask) and np.any(c0[zero_mask] > v + 1e-14):
            return None
        if L > U:
            return None
        # Quadratic
        disc = a_q * a_q - 4.0 * (v - c_q)
        if disc >= 0:
            sd = np.sqrt(disc)
            r1 = (a_q - sd) * 0.5
            r2 = (a_q + sd) * 0.5
            if L <= min(U, r1):
                return (L, min(U, r1))
            if max(L, r2) <= U:
                return (max(L, r2), U)
            return None
        return (L, U)
    
    f0 = float(np.max(c0))
    f1 = float(np.max(c0 + A - D))
    fm = float(np.max(c0 + 0.5*A - 0.25*D))
    v_hi = min(f0, f1, fm)
    v_lo = 0.0
    
    best_iv = feasible(v_hi)
    for _ in range(45):
        v_mid = (v_lo + v_hi) * 0.5
        iv = feasible(v_mid)
        if iv is not None:
            v_hi = v_mid
            best_iv = iv
        else:
            v_lo = v_mid
    
    if best_iv is None:
        return 0.5
    return (best_iv[0] + best_iv[1]) * 0.5

def exact_cd(h, n, max_passes=100, deadline=None, seed=42):
    h = h.copy()
    rng = np.random.RandomState(seed)
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        current_max = float(c.max())
        order = rng.permutation(n)
        for idx in order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            A, D = cd_AD(h, idx, n)
            c0 = c - h0 * A + h0 * h0 * D
            t_star = exact_1d(c0, A, D, n)
            f_new = float(np.max(c0 + t_star * A - t_star * t_star * D))
            if f_new < current_max - 1e-12:
                c = c0 + t_star * A - t_star * t_star * D
                h[idx] = t_star
                current_max = f_new
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def pnorm_obj(h, n, p):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.sum((c * dx)**p)) ** (1.0 / p)

def pnorm_slsqp(h, n, p, maxiter=300, deadline=None):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return float(np.sum((c * dx)**p))
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def interp_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old))
    x_new = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)

def cd_AD(h, idx, n):
