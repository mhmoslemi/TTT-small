def pair_line_search(h, c, i, j, n, B, Q):

    a1 = hi - gr * (hi - lo)
    b1 = lo + gr * (hi - lo)
    fa = float((c + a1 * B + a1 * a1 * Q).max())
    fb = float((c + b1 * B + b1 * b1 * Q).max())

            b1 = lo + gr * (hi - lo)

    v_star = float((c + u_star * B + u_star * u_star * Q).max())

        best_u = u_star

        if cs < c: h, c = hs, cs
    if not quick and time.time() < deadline - 20:
        h, c = lbfgs_stage(h, n, [(50, 40), (300, 30)], 1e4, deadline)
    if quick:
        plans = [(50, 30), (500, 30)]
    else:
        plans = [(20, 50), (100, 50), (500, 40), (2500, 30), (10000, 25)]
        if time.time() < deadline - 20: plans.append((50000, 20))
    h, c = slsqp_stage(h, n, plans, deadline)
    if n <= 1000 and time.time() < deadline - 3:
        rem = deadline - time.time()
        mt = int(min(5000, max(200, rem * 25)))
        h, c = pair_search(h, n, mt, deadline, rng)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 4.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)): raise ValueError
    except Exception:
        n0 = 200
        x = np.linspace(0, 2, n0, endpoint=False)
        h0 = project_feasible(np.clip(0.5 + 0.3*np.sin(np.pi*x), 0, 1), n0)
    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    if hard - t0 < 30: return best_h, best_c, n0
    main_dl = hard - max(100, 0.22 * budget_s)
    if main_dl < t0 + 20: main_dl = hard
    best_h, best_c = optimize_at_res(best_h, n0, main_dl, rng, False)
    for m in [128, 160, 200, 256, 320, 400, 480, 640, 800]:
        if time.time() > hard - 18: break
        if m == n0: continue
        rem = hard - time.time()
        if m > 400 and rem < 55: continue
        dl = time.time() + min(30, 0.12 * rem)
        if dl < time.time() + 5: continue
        ht, ct = optimize_at_res(interp_h(best_h, m), m, dl, rng, True)
        if ct < best_c: best_h, best_c = ht, ct
    if time.time() < hard - 45:
        n = len(best_h)
        for s in [0.04, 0.08, 0.15]:
            if time.time() > hard - 18: break
            hp = project_feasible(best_h + s * rng.randn(n), n)
            dl = min(hard - 5, time.time() + 28)
            ht, ct = optimize_at_res(hp, n, dl, rng, True)
            if ct < best_c: best_h, best_c = ht, ct
    if time.time() < hard - 3:
        n = len(best_h)
        hd = hard - 1
        h2, c2 = pair_search(best_h, n, int(max(300, (hd - time.time()) * 35)), hd, rng)
        if c2 < best_c: best_h, best_c = h2, c2
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
