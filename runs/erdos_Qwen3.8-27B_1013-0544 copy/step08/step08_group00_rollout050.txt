def subgradient_step(h, n, step_size, deadline):
    """Take a subgradient step for the max of correlations."""
    c = np.correlate(h, 1.0 - h, mode='full')
    threshold = c.max() - 0.01 * c.max()
    active = c >= threshold
    # Subgradient of c_m with respect to h[i]:
    # dc_m/dh[i] = (1 - h[i+m]) for the forward part + (-h[i-m]) for the backward part
    # (plus quadratic term for the diagonal)
    grad = np.zeros(n)
    for m_idx in np.where(active)[0]:
        m = m_idx - (n - 1)
        # forward: i + m in [0, n-1], i in [max(0,-m), min(n-1, n-1-m)]
        i_lo = max(0, -m)
        i_hi = min(n - 1, n - 1 - m)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += (1.0 - h[i_lo + m:i_hi + m + 1])
        # backward: i - m in [0, n-1], i in [max(0,m), min(n-1, n-1+m)]
        i_lo2 = max(0, m)
        i_hi2 = min(n - 1, n - 1 + m)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2+1] -= h[i_lo2 - m:i_hi2 - m + 1]
    # Normalize and take step
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-10:
        return h
    h_new = h - step_size * grad / grad_norm
    h_new = project_feasible(h_new, n)
    return h_new

        best_h, best_c = h1, c1

    # Phase 2: Coarse grid CD
    h2, c2 = coord_descent(best_h, n_init, 150, deadline - 500, grid_size=30)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Finer grid CD
    h3, c3 = coord_descent(best_h, n_init, 100, deadline - 350, grid_size=60)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: SLSQP polish
    h4, c4 = slsqp_optimize(best_h, n_init, [50000, 200000, 500000], 300)
    if c4 < best_c:
        best_h, best_c = h4, c4

    # Phase 5: Bisection CD for high precision
    h5, c5 = bisection_cd(best_h, n_init, 20, deadline - 200)
    if c5 < best_c:
        best_h, best_c = h5, c5

    # Phase 6: Iterated local search
    n = len(best_h)
    scales = [0.08, 0.05, 0.03, 0.02, 0.01, 0.005]
    for scale in scales:
        if time.time() > deadline - 120:
            break
        for attempt in range(2):
            if time.time() > deadline - 100:
                break
            hp = perturb_h(best_h, n, rng, scale)
            ho, co = slsqp_optimize(hp, n, [5000, 50000], 200)
            ho, co = coord_descent(ho, n, 50, deadline - 60, grid_size=50)
            ho, co = bisection_cd(ho, n, 8, deadline - 30)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 7: Try alternative resolution
    if time.time() < deadline - 100:
        x_orig = np.linspace(0, 2, n)
        for n_try in [n // 2, n * 2]:
            if time.time() > deadline - 60:
                break
            if n_try < 100 or n_try > 2000:
                continue
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
            ho, co = slsqp_optimize(h_int, n_try, [5000, 50000], 200)
            ho, co = coord_descent(ho, n_try, 50, deadline - 30, grid_size=50)
            hof, cof = bisection_cd(ho, n_try, 5, deadline - 10)
            if cof < co:
                ho, co = hof, cof
            if co < best_c:
                best_h, best_c = ho, co
                n = n_try
                x_orig = x_new

    # Phase 8: Final polish
    if time.time() < deadline - 5:
        n = len(best_h)
        hf, cf = slsqp_optimize(best_h, n, [2000000, 5000000], 500)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = bisection_cd(best_h, n, 10, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
