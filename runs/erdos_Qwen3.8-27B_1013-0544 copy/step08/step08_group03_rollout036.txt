                      random_order=True, rng=rng)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Multi-n exploration
    n_candidates = [400, 450, 500, 550, 600, 700]
    if n_init in n_candidates:
        n_candidates.remove(n_init)
    for nc in n_candidates:
        if time.time() > deadline - 200:
            break
        h_int = interp_to_n(best_h, len(best_h), nc)
        h_o, c_o = exact_cd(h_int, nc, 100, deadline, grid_pts=18, gs_iters=20,
                           random_order=True, rng=rng)
        if c_o < best_c:
            best_h, best_c = h_o, c_o

    # Phase 4: Perturbation + re-CD loop
    n = len(best_h)
    for i in range(40):
        if time.time() > deadline - 50:
            break
        scale = 0.02 + 0.003 * (i % 8)
        strat = i % 4
        if strat == 0:
            h_try = project_feasible(best_h + rng.normal(0, scale, n), n)
        elif strat == 1:
            h_try = best_h.copy()
            bmask = (h_try > 0.15) & (h_try < 0.85)
            h_try[bmask] += rng.normal(0, scale * 2.5, int(bmask.sum()))
            h_try = project_feasible(h_try, n)
        elif strat == 2:
            h_try = best_h.copy()
            for _ in range(4):
                idx = rng.choice(n, 2, replace=False)
                h_try[idx[0]], h_try[idx[1]] = h_try[idx[1]], h_try[idx[0]]
            h_try = project_feasible(h_try + rng.normal(0, scale * 0.4, n), n)
        else:
            # Antisymmetric perturbation: push up where low, down where high
            h_try = best_h.copy()
            low_mask = h_try < 0.4
            high_mask = h_try > 0.6
            h_try[low_mask] += rng.uniform(0, scale * 2, int(low_mask.sum()))
            h_try[high_mask] -= rng.uniform(0, scale * 2, int(high_mask.sum()))
            h_try = project_feasible(h_try, n)

        h_o, c_o = exact_cd(h_try, n, 80, deadline, grid_pts=16, gs_iters=18,
                           random_order=True, rng=rng)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            # Polishing CD after improvement
            h_p, c_p = exact_cd(best_h, n, 120, deadline, grid_pts=20, gs_iters=22,
                               random_order=True, rng=rng)
            if c_p < best_c:
                best_h, best_c = h_p, c_p

    # Phase 5: Final high-p SLSQP + CD
    if time.time() < deadline - 50:
        hf, cf = pnorm_slsqp(best_h, n, [500, 2000, 10000], 300, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 8:
        hf, cf = exact_cd(best_h, n, 300, deadline, grid_pts=24, gs_iters=28)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 3:
        hf, cf = exact_cd(best_h, n, 100, deadline, grid_pts=30, gs_iters=35)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
