def coord_descend(h, n, max_passes, deadline, project_every=8):

    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h, best_c = h.copy(), (2.0/n) * cur
    center = n - 1
    L = 2*n - 1
    
    for _ in range(max_passes):
        if time.time() > deadline: break
        improved = False
        for p in range(n):
            if time.time() > deadline: break
            hp = h[p]
            lo = min(p, center - p)
            hi = max(center + p, 2*center - p)
            seg = hi - lo + 1
            s = center - np.arange(lo, hi+1)
            j = p + s
            i = p - s
            a = np.zeros(seg)
            vj = (j >= 0) & (j < n)
            vi = (i >= 0) & (i < n)
            a[vj] += 1.0 - h[j[vj]]
            a[vi] -= h[i[vi]]
            off = center - lo
            cseg = np.empty(seg - 1)
            asg = np.empty(seg - 1)
            if off:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center+1:hi+1]
                asg[off:] = a[off+1:]
            qconst = c[center] - hp*(1.0-hp)
            unaff = 0.0
            if lo > 0: unaff = max(unaff, float(c[:lo].max()))
            if hi+1 < L: unaff = max(unaff, float(c[hi+1:].max()))
            
            bx, bf = optimize_coordinate(hp, cseg, asg, qconst, unaff)
            if bf < cur - 1e-11:
                dxv = bx - hp
                if off: c[lo:center] += asg[:off]*dxv
                if hi > center: c[center+1:hi+1] += asg[off:]*dxv
                c[center] = qconst + bx*(1.0-bx)
                h[p] = bx
                cur = bf
                improved = True
            if (p+1) % project_every == 0:
                h = project_feasible(h, n)
                c = np.correlate(h, 1.0-h, mode='full')
                cur = float(c.max())
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0-h, mode='full')
        cur = float(c.max())
        cval = (2.0/n)*cur
        if cval < best_c - 1e-14:
            best_c, best_h = cval, h.copy()
        if not improved: break
    return best_h, best_c

def fourier_refine(h, n, K, deadline):
    x = (np.arange(n) + 0.5) * (2.0/n)
    cols = [np.cos(k*np.pi*x) for k in range(1, K+1)]
    cols += [np.sin(k*np.pi*x) for k in range(1, K+1)]
    B = np.column_stack(cols)
    coeffs, _, _, _ = np.linalg.lstsq(B, h - 0.5, rcond=None)
    
    def obj(w):
        hv = 0.5 + B @ w
        pen = 0.0
        neg = hv < 0
        pos = hv > 1
        if neg.any(): pen += 100.0 * (hv[neg].sum())**2
        if pos.any(): pen += 100.0 * ((hv[pos]-1).sum())**2
        hc = np.clip(hv, 0, 1)
        c = np.correlate(hc, 1.0-hc, mode='full')
        return float(np.max(c) * (2.0/n)) + pen
    
    try:
        res = minimize(obj, coeffs, method='L-BFGS-B', maxiter=200,
                       options={'ftol': 1e-15, 'maxfun': 500})
        h2 = project_feasible(0.5 + B @ res.x, n)
        c2 = compute_c5(h2, n)
        return h2, c2
    except:
        return h, compute_c5(h, n)

        if tot > 1e-14:

            hp = h[p]
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            seg = hi - lo + 1
            s = center - np.arange(lo, hi + 1)
            j, i = p + s, p - s
            a = np.zeros(seg)
            vj = (j >= 0) & (j < n)
            vi = (i >= 0) & (i < n)
            a[vj] += 1.0 - h[j[vj]]
            a[vi] -= h[i[vi]]
            off = center - lo
            cseg = np.empty(seg - 1)
            asg = np.empty(seg - 1)
            if off:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center+1:hi+1]
                asg[off:] = a[off+1:]
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0: unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L: unaff = max(unaff, float(c[hi+1:].max()))
            bx, bf = optimize_coordinate(hp, cseg, asg, qconst, unaff)
            if bf < cur - 1e-11:
                dxv = bx - hp
                if off: c[lo:center] += asg[:off] * dxv
                if hi > center: c[center+1:hi+1] += asg[off:] * dxv
                c[center] = qconst + bx * (1.0 - bx)
                h[p] = bx
                cur = bf
                improved = True
            if (p + 1) % project_every == 0:
                h = project_feasible(h, n)
                c = np.correlate(h, 1.0 - h, mode='full')
                cur = float(c.max())
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        cval = (2.0 / n) * cur
        if cval < best_c - 1e-14:
            best_c, best_h = cval, h.copy()
        if not improved: break
    return best_h, best_c

def fourier_refine(h, n, K, deadline):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    cols = [np.cos(k * np.pi * x) for k in range(1, K + 1)]
    cols += [np.sin(k * np.pi * x) for k in range(1, K + 1)]
    B = np.column_stack(cols)
    coeffs, _, _, _ = np.linalg.lstsq(B, h - 0.5, rcond=None)
    def obj(w):
        hv = 0.5 + B @ w
        pen = 0.0
        neg, pos = hv < 0, hv > 1
        if neg.any(): pen += 80.0 * (hv[neg].sum())**2
        if pos.any(): pen += 80.0 * ((hv[pos] - 1).sum())**2
        hc = np.clip(hv, 0, 1)
        c = np.correlate(hc, 1.0 - hc, mode='full')
        return float(np.max(c) * (2.0 / n)) + pen
    try:
        res = minimize(obj, coeffs, method='L-BFGS-B', maxiter=150,
                       options={'ftol': 1e-15, 'maxfun': 400})
        h2 = project_feasible(0.5 + B @ res.x, n)
        return h2, compute_c5(h2, n)
    except Exception:
        return h, compute_c5(h, n)

def make_start(n, typ, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    if typ == 0:
        return project_feasible(1.0 - np.abs(x - 1.0), n)
    elif typ == 1:
        return project_feasible(0.5 * np.ones(n) + 0.3 * np.cos(np.pi * x), n)
    elif typ == 2:
        idx = rng.choice(n, n // 2, replace=False)
        h = np.zeros(n)
        h[idx] = 1.0
        return h
    elif typ == 3:
        return project_feasible(0.5 + 0.4 * np.cos(2 * np.pi * x) + 0.1 * np.sin(3 * np.pi * x), n)
    else:
        s = rng.uniform(0.5, 1.5)
        return project_feasible(np.exp(-s**2 * (x - 1.0)**2), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 12.0
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = make_start(500, 0, rng)
    n = len(h_init)
    if n % 2:
        n += 1
        h_init = resample_h(h_init, n)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: CD with random order
    best_h, best_c = coord_descend(best_h, n, 8, deadline, project_every=8, random_order=True, rng=rng)
    # Phase 2: Fourier refinement then CD
    if time.time() < deadline - 250:
        for K in [6, 10, 14]:
            if time.time() > deadline - 220: break
            h2, c2 = fourier_refine(best_h, n, K, deadline)
            if c2 < best_c:
                best_h, best_c = h2, c2
                best_h, best_c = coord_descend(best_h, n, 5, deadline, project_every=6, random_order=True, rng=rng)
    # Phase 3: Multi-resolution
    for nt in [400, 350, 300, 256, 500, 600, 200, 450]:
        if time.time() > deadline - 80: break
        if best_c <= 0.38070: break
        h0 = resample_h(best_h, nt)
        pe = 5 if nt <= 350 else 8
        passes = 5 if nt <= 350 else 3
        h1, c1 = coord_descend(h0, nt, passes, deadline, project_every=pe, random_order=True, rng=rng)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 4: Diverse restarts
    for r in range(35):
        if time.time() > deadline - 40: break
        if best_c <= 0.38065: break
        nt = [len(best_h), 300, 400, 256, 500][r % 5]
        typ = r % 6
        if typ < 5:
            h0 = make_start(nt, typ, rng)
        else:
            base = resample_h(best_h, nt)
            sig = [0.002, 0.005, 0.01, 0.03, 0.06, 0.12][r % 6]
            h0 = project_feasible(base + rng.normal(0, sig, nt), nt)
        passes = 3 if nt > 400 else 5
        pe = 10 if nt > 500 else 6
        h1, c1 = coord_descend(h0, nt, passes, deadline, project_every=pe, random_order=True, rng=rng)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Final polish
    if time.time() < deadline - 15:
        best_h, best_c = coord_descend(best_h, len(best_h), 8, deadline, project_every=4, random_order=True, rng=rng)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
