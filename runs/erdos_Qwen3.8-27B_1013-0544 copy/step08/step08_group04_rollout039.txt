import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(np.asarray(h, float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if diff > 0:
        cap = 1.0 - h
        tot = cap.sum()
        if tot > 1e-15:
            h += (diff / tot) * cap
    elif diff < 0:
        tot = h.sum()
        if tot > 1e-15:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - h.sum()
    if res > 1e-12:
        idx = np.where(h < 1.0)[0]
        if len(idx):
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-12:
        idx = np.where(h > 0.0)[0]
        if len(idx):
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def smooth_fg(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    z = alpha * c
    zmax = z.max()
    e = np.exp(z - zmax)
    es = e.sum()
    w = e / es
    f = (zmax + np.log(es)) / alpha * (2.0 / n)
    center = n - 1
    g = np.zeros(n)
    for sh in range(-(n - 1), n):
        wm = w[center + sh]
        if wm < 1e-15:
            continue
        if sh == 0:
            g += wm * (1.0 - 2.0 * h)
        elif sh > 0:
            g[:n - sh] += wm * (1.0 - h[sh:])
            g[sh:] -= wm * h[:n - sh]
        else:
            r = -sh
            g[r:] += wm * (1.0 - h[:n - r])
            g[:n - r] -= wm * h[r:]
    g *= (2.0 / n)
    return f, g

def lbgfs_opt(h, n, alphas, deadline):
    best = h.copy()
    best_c = compute_c(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_fg, best, args=(n, alpha), method='L-BFGS-B',
                           jac=True, bounds=bounds,
                           options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10,
                                    'maxls': 50, 'iprint': 0})
            x = project_feasible(res.x, n)
            c = compute_c(x, n)
            if c < best_c:
                best, best_c = x, c
        except Exception:
            pass
    return best, best_c

def sharpen(h, n, t):
    h2 = h.copy()
    h2[h2 < t] = 0.0
    h2[h2 > 1.0 - t] = 1.0
    return project_feasible(h2, n)

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old.copy(), n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def coord_eval(x, hp, cseg, a, qconst, unaff):
    if a.size:
        vmax = float(np.max(cseg + a * (x - hp)))
    else:
        vmax = 0.0
    return max(max(unaff, qconst + x * (1.0 - x)), vmax)

def optimize_coord(hp, cseg, a, qconst, unaff, iters=20):
    tol = 1e-10
    b = cseg - a * hp
    pos = a > 1e-12
    neg = a < -1e-12
    zero = ~(pos | neg)
    zmax = float(np.max(b[zero])) if zero.any() else -1e100
    cm = float(np.max(cseg)) if a.size else 0.0
    low = max(0.0, unaff, qconst, zmax)
    high = max(low + 1e-11, qconst, qconst + hp * (1.0 - hp), cm, unaff) + 1e-8
    a_pos, a_neg = a[pos], a[neg]
    b_pos, b_neg = b[pos], b[neg]
    for _ in range(iters):
        mid = 0.5 * (low + high)
        feas = False
        if mid >= unaff - tol and mid >= qconst - tol and mid >= zmax - tol:
            ll, lh = 0.0, 1.0
            if a_neg.size:
                v = float(np.max((mid - b_neg) / a_neg))
                if v > ll:
                    ll = v
            if a_pos.size:

                    if max(0.0, ll) <= min(r1, lh) + tol:

                    else:
                        feas = max(r2, ll) <= min(1.0, lh) + tol
        if feas:

        else:

    best_x, best_f = hp, coord_eval(hp, hp, cseg, a, qconst, unaff)

        idx = int(np.argmin(vals))
        if vals[idx] < best_f:

    return best_x, best_f

def coord_descent(h, n, max_passes, deadline):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h, best_c = h.copy(), (2.0 / n) * cur
    center = n - 1
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            hp = float(h[p])
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            seg = hi - lo + 1
            s_arr = center - np.arange(lo, hi + 1, dtype=int)
            j = p + s_arr
            i = p - s_arr
            a = np.zeros(seg, dtype=float)
            vj = (j >= 0) & (j < n)
            vi = (i >= 0) & (i < n)
            if np.any(vj):
                a[vj] += 1.0 - h[j[vj]]
            if np.any(vi):
                a[vi] -= h[i[vi]]
            off = center - lo
            cseg = np.empty(seg - 1, dtype=float)
            asg = np.empty(seg - 1, dtype=float)
            if off > 0:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center + 1:hi + 1]
                asg[off:] = a[off + 1:]
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < 2 * n - 1:
                unaff = max(unaff, float(c[hi + 1:].max()))
            bx, bf = optimize_coord(hp, cseg, asg, qconst, unaff, 20)
            if bf < cur - 1e-12:
                dxv = bx - hp
                if off > 0:
                    c[lo:center] += asg[:off] * dxv
                if hi > center:
                    c[center + 1:hi + 1] += asg[off:] * dxv
                c[center] = qconst + bx * (1.0 - bx)
                h[p] = bx
                cur = bf
                improved = True
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        if (2.0 / n) * cur < best_c:
            best_c = (2.0 / n) * cur
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c
