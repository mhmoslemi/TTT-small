        h += theta[2*k] * np.sin(k * np.pi * x)

    cm = float(c.max())
    return cm + np.log(float(np.sum(np.exp(alpha * (c - cm))))) / alpha

    v_hat = v / (1 - b2**t)
    params -= lr * m_hat / (np.sqrt(v_hat) + eps)
    state['m'] = m
    state['v'] = v

    val = max(unaff, qconst + x * (1.0 - x))
    return max(val, vmax)

def coord_vector_eval(xg, hp, cseg, a, qconst, unaff):
    d = xg - hp
    if a.size:
        vmax = np.max(cseg[None, :] + d[:, None] * a[None, :], axis=1)
    else:
        vmax = np.zeros_like(xg)
    vals = np.maximum(vmax, qconst + xg * (1.0 - xg))
    return np.maximum(vals, unaff)

def optimize_coordinate(hp, cseg, a, qconst, unaff, iters=16):
    tol = 1e-10
    m = a.size
    b = cseg - a * hp
    pos = a > 1e-12
    neg = a < -1e-12
    zero = ~(pos | neg)
    zmax = float(np.max(b[zero])) if zero.any() else -1e100
    cm = float(np.max(cseg)) if m else 0.0
    qcur = qconst + hp * (1.0 - hp)
    low = max(0.0, unaff, qconst)
    if zmax > low:
        low = zmax
    high = max(low + 1e-11, unaff, qconst, qcur, cm)
    if zmax > high:
        high = zmax
    high += 1e-8
    a_pos = a[pos]
    a_neg = a[neg]
    b_pos = b[pos]
    b_neg = b[neg]
    pc = a_pos.size
    nc = a_neg.size
    for _ in range(iters):
        mid = 0.5 * (low + high)
        feas = False
        if mid >= unaff - tol and mid >= qconst - tol and mid >= zmax - tol:
            ll = 0.0
            lh = 1.0
            if nc:
                v = float(np.max((mid - b_neg) / a_neg))
                if v > ll:
                    ll = v
            if pc:
                v = float(np.min((mid - b_pos) / a_pos))
                if v < lh:
                    lh = v
            if ll <= 1.0 + tol and lh >= -tol:
                g = mid - qconst
                if g > 0.25:
                    Lq = max(0.0, ll)
                    Uq = min(1.0, lh)
                    feas = Lq <= Uq + tol
                elif g > -1e-12:
                    if g < 0.0:
                        g = 0.0
                    r1 = 0.5 - 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                    r2 = 0.5 + 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                    Lq = max(0.0, ll)
                    Uq = min(r1, lh)
                    if Lq <= Uq + tol:
                        feas = True
                    else:
                        Lq = max(r2, ll)
                        Uq = min(1.0, lh)
                        feas = Lq <= Uq + tol
        if feas:
            high = mid
        else:
            low = mid
    t = high
    cands = [0.0, 1.0, hp, 0.5]
    if t >= unaff - tol and t >= qconst - tol and t >= zmax - tol:
        ll = 0.0
        lh = 1.0
        if nc:
            v = float(np.max((t - b_neg) / a_neg))
            if v > ll:
                ll = v
        if pc:
            v = float(np.min((t - b_pos) / a_pos))
            if v < lh:
                lh = v
        if ll <= 1.0 + tol and lh >= -tol:
            g = t - qconst
            if g > 0.25:
                branches = [(max(0.0, ll), min(1.0, lh))]
            elif g >= -1e-12:
                if g < 0.0:
                    g = 0.0
                r1 = 0.5 - 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                r2 = 0.5 + 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                branches = [(0.0, r1), (r2, 1.0)]
            else:
                branches = []
            for Lq, Uq in branches:
                L = max(Lq, ll, 0.0)
                U = min(Uq, lh, 1.0)
                if L <= U + tol:
                    cands.extend([L, U, 0.5 * (L + U)])
    best_x = hp
    best_f = coord_eval(hp, hp, cseg, a, qconst, unaff)
    for x in cands:
        if x < 0.0 or x > 1.0:
            continue
        f = coord_eval(x, hp, cseg, a, qconst, unaff)
        if f < best_f - 1e-12:
            best_f = f
            best_x = x
    if best_f > high + 1e-4 and m:
        xg = np.linspace(0.0, 1.0, 24)
        vals = coord_vector_eval(xg, hp, cseg, a, qconst, unaff)
        idx = int(np.argmin(vals))
        x = float(xg[idx])
        f = float(vals[idx])
        if f < best_f:
            best_f = f
            best_x = x
        l = max(0.0, x - 0.04)
        r = min(1.0, x + 0.04)
        if r <= l:
            l, r = 0.0, 1.0
        xg2 = np.linspace(l, r, 24)
        vals2 = coord_vector_eval(xg2, hp, cseg, a, qconst, unaff)
        idx2 = int(np.argmin(vals2))
        x2 = float(xg2[idx2])
        f2 = float(vals2[idx2])
        if f2 < best_f:
            best_f = f2
            best_x = x2
    return best_x, best_f

def cd_pass(h, n, order, c, cur, project_every=8):
    center = n - 1
    L = 2 * n - 1
    count = 0
    for p in order:
        hp = float(h[p])
        lo = min(p, center - p)
        hi = max(center + p, 2 * center - p)
        seg = hi - lo + 1
        s = center - np.arange(lo, hi + 1, dtype=int)
        j = p + s
        i = p - s
        a = np.zeros(seg, dtype=float)
        vj = (j >= 0) & (j < n)
        vi = (i >= 0) & (i < n)
        if np.any(vj):
            a[vj] += 1.0 - h[j[vj]]
        if np.any(vi):
            a[vi] -= h[i[vi]]
        off = center - lo
        cseg = np.empty(seg - 1, dtype=float)
        asg = np.empty(seg - 1, dtype=float)
        if off:
            cseg[:off] = c[lo:center]
            asg[:off] = a[:off]
        if hi > center:
            cseg[off:] = c[center + 1:hi + 1]
            asg[off:] = a[off + 1:]
        qconst = c[center] - hp * (1.0 - hp)
        unaff = 0.0
        if lo > 0:
            unaff = max(unaff, float(c[:lo].max()))
        if hi + 1 < L:
            unaff = max(unaff, float(c[hi + 1:].max()))
        bx, bf = optimize_coordinate(hp, cseg, asg, qconst, unaff, 16)
        if bf < cur - 1e-10:
            dxv = bx - hp
            if off:
                c[lo:center] += asg[:off] * dxv
            if hi > center:
                c[center + 1:hi + 1] += asg[off:] * dxv
            c[center] = qconst + bx * (1.0 - bx)
            h[p] = bx
            cur = bf
            count += 1
        count2 = count
        if count2 % project_every == 0 and count2 > 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            cur = float(c.max())
    return h, c, cur, count

def subgrad_step(h, n, lr, alpha, deadline):
    c = np.correlate(h, 1.0 - h, mode='full')
    z = alpha * (c - c.max())
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    center = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n - s] += wm * (1.0 - h[s:])
            g[s:] -= wm * h[:n - s]
        else:
            r = -s
            g[r:] += wm * (1.0 - h[:n - r])
            g[:n - r] -= wm * h[r:]
    g *= (2.0 / n)
    h_new = project_feasible(h - lr * g, n)
    return h_new

def optimize_cd(h, n, max_passes, deadline, rng):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h = h.copy()
    best_c = (2.0 / n) * cur
    for p in range(max_passes):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        h, c, cur, improved = cd_pass(h, n, order, c, cur)
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        cval = (2.0 / n) * cur
        if cval < best_c - 1e-14:
            best_c = cval
            best_h = h.copy()
        if improved == 0:
            break
    return best_h, best_c

def optimize_subgrad(h, n, steps, deadline, rng):
    h = project_feasible(h, n)
    best_c = compute_c(h, n)
    best_h = h.copy()
    lr = 0.005
    alpha = 200.0
    for s in range(steps):
        if time.time() > deadline:
            break
        h = subgrad_step(h, n, lr, alpha, deadline)
        c = compute_c(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if s % 50 == 49:
            lr *= 0.9
            alpha *= 1.3
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n = 400
        x = (np.arange(n) + 0.5) * 2.0 / n
        h_init = 1.0 - np.abs(x - 1.0)
    best_h = project_feasible(h_init, len(h_init))
    best_c = compute_c(best_h, len(best_h))

    # Phase 1: CD on initial size
    n0 = len(best_h)
    if n0 % 2:
        n0 += 1
        best_h = resample_h(best_h, n0)
    best_h, best_c = optimize_cd(best_h, n0, 20, deadline, rng)

    # Phase 2: Try multiple sizes with CD + subgradient
    sizes = [350, 400, 450, 300, 500, 256]
    for nt in sizes:
        if time.time() > deadline - 120:
            break
        if best_c <= 0.38075:
            break
        h0 = resample_h(best_h, nt)
        h1, c1 = optimize_cd(h0, nt, 15, deadline, rng)
        if c1 < best_c:
            best_h, best_c = h1, c1
            # Add subgradient polish
            if time.time() < deadline - 60:
                h2, c2 = optimize_subgrad(best_h, nt, 80, deadline, rng)
                if c2 < best_c:
                    best_h, best_c = h2, c2
                    h3, c3 = optimize_cd(best_h, nt, 8, deadline, rng)
                    if c3 < best_c:
                        best_h, best_c = h3, c3

    # Phase 3: Restart with perturbations
    for r in range(12):
        if time.time() > deadline - 80:
            break
        if best_c <= 0.38070:
            break
        nt = len(best_h) if r % 2 else 400
        base = resample_h(best_h, nt)
        typ = r % 3
        if typ == 0:
            h0 = project_feasible(base + rng.normal(0, 0.015, nt), nt)
        elif typ == 1:
            # Binary snap with random shift
            k = nt // 2
            idx = np.argpartition(base, nt - k)[nt - k:]
            h0 = np.zeros(nt)
            h0[idx] = 1.0
            h0 = project_feasible(h0, nt)
        else:
            # Fourier-based perturbation
            x = (np.arange(nt) + 0.5) * 2.0 / nt
            pert = np.zeros(nt)
            for kk in range(1, 6):
                amp = rng.uniform(0.01, 0.04)
                ph = rng.uniform(0, 2 * np.pi)
                pert += amp * np.cos(kk * np.pi * x + ph)
            h0 = project_feasible(base + pert, nt)
        h1, c1 = optimize_cd(h0, nt, 10, deadline, rng)
        if time.time() < deadline - 40:
            h1b, c1b = optimize_subgrad(h1, nt, 40, deadline, rng)
            if c1b < c1:
                h1, c1 = h1b, c1b
            h1c, c1c = optimize_cd(h1, nt, 5, deadline, rng)
            if c1c < c1:
                h1, c1 = h1c, c1c
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 4: Final polish on best
    if time.time() < deadline - 20:
        best_h, best_c = optimize_cd(best_h, len(best_h), 15, deadline, rng)
    if time.time() < deadline - 10:
        best_h, best_c = optimize_subgrad(best_h, len(best_h), 30, deadline, rng)
        if time.time() < deadline - 5:
            best_h, best_c = optimize_cd(best_h, len(best_h), 5, deadline, rng)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c(best_h, len(best_h))
    return best_h, best_c, len(best_h)
