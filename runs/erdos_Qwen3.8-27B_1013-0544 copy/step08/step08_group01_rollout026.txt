def subgradient_balance(h, n, max_iter=200, deadline=None, k_lags=5):
    """Subgradient method that balances the top k_lags."""
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.03
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        # Get top k lags
        top_idx = np.argsort(c)[-k_lags:]
        sg = np.zeros(n)
        for mk in top_idx:
            m = mk - (n - 1)
            for i in range(n):
                j = i + m
                if 0 <= j < n:
                    sg[i] += (1.0 - h[j])
                ip = i - m
                if 0 <= ip < n:
                    sg[i] -= h[ip]
        sg /= k_lags
        h_new = project_feasible(h - lr * sg, n)
        c_new = compute_c5(h_new, n)

            h = h_new
            best_c = c_new
            stall = 0
        else:
            lr *= 0.8
            stall += 1
            if stall > 40 or lr < 1e-12:
                break
    return h, best_c

            if tot < 1e-14:

import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mse = shifted.max()
    lse = mse + np.log(np.sum(np.exp(shifted - mse)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def obj_1d_vec(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def coord_cd(h, n, max_passes=200, deadline=None, grid_size=60):
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            jf_arr = p + ms - (n - 1)
            jb_arr = p - ms + (n - 1)
            valid_f = (jf_arr >= 0) & (jf_arr < n)
            valid_b = (jb_arr >= 0) & (jb_arr < n)
            if np.any(valid_f):
                a_coeff[valid_f] += (1.0 - h[jf_arr[valid_f]])
            if np.any(valid_b):
                a_coeff[valid_b] -= h[jb_arr[valid_b]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo = -h0
            hi = 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals = c_aff[None, :] + d_grid[:, None] * a_coeff[None, :]
            if diag_local is not None:
                vals[:, diag_local] -= (d_grid[:, None]) ** 2
            vals = vals.max(axis=1)
            np.maximum(vals, unaff_max, out=vals)
            best_idx = int(np.argmin(vals))
            best_d = d_grid[best_idx]
            best_val = float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                gl = d_grid[best_idx - 1]
                gh = d_grid[best_idx + 1]
                try:
                    res = golden(obj_1d_vec, gl, gh,
                                 args=(c_aff, a_coeff, diag_local, unaff_max),
                                 maxiter=50)
                    if res.fun < best_val:
                        best_d = float(res.x)
                        best_val = float(res.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_run(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
