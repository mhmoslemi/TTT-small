def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    iters = 0
    while abs(diff) > 1e-10 and iters < 100:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff * room / tot, 0, 1)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h * (target / tot), 0, 1)
        diff = target - h.sum()
        iters += 1
    return h

            jf = p + ms - (n - 1)

            vf = (jf >= 0) & (jf < n)

            for d in dc:

                nv[d_idx] -= d * d
                v = max(um, float(nv.max()))

                    best_d = d

        c = np.correlate(h, 1.0 - h, mode='full')
        cc = float(c.max()) * dx

import numpy as np

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff * room / tot, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h * (target / tot), 0.0, 1.0)
    return h
