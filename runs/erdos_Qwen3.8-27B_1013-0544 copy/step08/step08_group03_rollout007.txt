def slsqp_pnorm(h, n, p, maxiter=500, deadline=None):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return (np.sum(c ** p) ** (1.0 / p)) * dx
    def sum_con(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        return new_h, new_c
    except Exception:
        return h, compute_c5(h, n)
