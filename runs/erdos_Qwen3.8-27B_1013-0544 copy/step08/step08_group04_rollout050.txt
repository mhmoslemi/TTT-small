def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 1e-14:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 1e-14:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size:
                h[idx[0]] = min(1.0, h[idx[0]] + res)
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size:
                h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def resample_h(old, n):

    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def fourier_to_h(coeffs, n, K):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.full(n, 0.5)
    for k in range(1, K+1):
        h += coeffs[2*(k-1)] * np.cos(k * np.pi * x)
        h += coeffs[2*(k-1)+1] * np.sin(k * np.pi * x)
    return h

def fourier_obj(coeffs, n, K, penalty=1000.0):
    h = fourier_to_h(coeffs, n, K)
    h_clipped = np.clip(h, 0.0, 1.0)
    # Penalty for box violations
    viol = np.sum(np.maximum(0, -h)**2 + np.maximum(0, h-1)**2)
    c = compute_c(h_clipped, n)
    return c + penalty * viol

def binary_local_search(h, n, max_iters=2000, deadline=None):
    # Convert to binary
    hb = np.zeros(n)
    k = n // 2
    idx = np.argpartition(h, n-k)[n-k:]
    hb[idx] = 1.0
    c = np.correlate(hb, 1.0-hb, mode='full')
    cur = float(c.max())
    best_c = (2.0/n) * cur
    best_hb = hb.copy()
    
    ones = np.where(hb == 1)[0]
    zeros = np.where(hb == 0)[0]
    
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        # Try a random swap
        i = int(ones[np.random.randint(len(ones))])
        j = int(zeros[np.random.randint(len(zeros))])
        if i == j:
            continue
        # Swap
        hb[i], hb[j] = hb[j], hb[i]
        c = np.correlate(hb, 1.0-hb, mode='full')
        new_c = float(c.max())
        if new_c < cur - 1e-10:
            cur = new_c
            if cur * (2.0/n) < best_c:
                best_c = cur * (2.0/n)
                best_hb = hb.copy()
            # Update lists
            ones = np.where(hb == 1)[0]
            zeros = np.where(hb == 0)[0]
        else:
            # Revert
            hb[i], hb[j] = hb[j], hb[i]
    
    return best_hb, best_c

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if diff > 0.0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 1e-14:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 1e-14:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-12:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-12:
        idx = np.where(h > 0.0)[0]

            h[idx[0]] = max(0.0, h[idx[0]] + res)

def sum_con(x, n):
    return float(x.sum()) - 0.5 * n

def sum_jac(x, n):
    return np.ones(n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    z = alpha * c
    zm = float(z.max())
    return ((zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    center = n - 1
    z = alpha * c - c.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm <= 1e-15:
            continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * x)
        elif s > 0:
            g[:n - s] += wm * (1.0 - x[s:])
            g[s:] -= wm * x[:n - s]
        else:
            r = -s
            g[r:] += wm * (1.0 - x[:n - r])
            g[:n - r] -= wm * x[r:]
    return (2.0 / n) * g

    best_c = compute_c(best_h, n)

    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_jac, bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-13, 'iprint': 0})
            x = project_feasible(res.x, n)
            c = compute_c(x, n)
            if c < best_c:

        except Exception:
