def gradp(h,n,p):
    g=np.zeros(n); off=n-1
    if p[off]: g+=p[off]*(1-2*h)
    for d in range(1,n):
      wp=p[off+d]
      if wp>1e-14: g[d:]+=wp*(1-h[:n-d]); g[:n-d]-=wp*h[d:]
      wn=p[off-d]
      if wn>1e-14: g[:n-d]+=wn*(1-h[d:]); g[d:]-=wn*h[:n-d]
    return (2./n)*g

import numpy as np
import time
from scipy.optimize import minimize

def eval_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def project_feasible(z, n):
    z = np.nan_to_num(np.asarray(z, dtype=float), nan=0.5, posinf=1.0, neginf=0.0)
    if z.size != n:
        if z.size == 0:
            return np.full(n, 0.5)
        z = np.interp(np.linspace(0.0, 1.0, n), np.linspace(0.0, 1.0, z.size), z)
    target = 0.5 * n
    lo = float(z.min()) - 2.0
    hi = float(z.max()) + 2.0
    for _ in range(48):
        tau = 0.5 * (lo + hi)
        if np.clip(z - tau, 0.0, 1.0).sum() > target:
            lo = tau
        else:
            hi = tau
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - y.sum()
    if r:
        mask = (y < 1.0) if r > 0 else (y > 0.0)
        room = (1.0 - y) if r > 0 else y
        s = float(room[mask].sum())
        if s > 1e-15:
            y[mask] += (r / s) * room[mask]
    return np.clip(y, 0.0, 1.0)

def random_smooth_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, 8):
        a = rng.uniform(-0.45, 0.45)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += a * np.sin(0.5 * np.pi * m * x + ph)
        h += 0.5 * a * np.cos(np.pi * m * x + ph)
    h += 0.08 * (x / 2.0 - 0.5)
    h += rng.normal(0.0, 0.02, n)
    return project_feasible(np.clip(h, 0.0, 1.0), n)

def smooth_components(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    e = np.exp(alpha * (c - m))
    s = float(e.sum())
    return (m + float(np.log(s)) / alpha) * (2.0 / n), e / s, c

def grad_weights(h, n, p):
    g = np.zeros(n, dtype=float)
    off = n - 1
    w0 = float(p[off])
    if w0 > 1e-14:
        g += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = float(p[off + d])
        if wp > 1e-14:
            g[d:] += wp * (1.0 - h[:n - d])
            g[:n - d] -= wp * h[d:]
        wn = float(p[off - d])
        if wn > 1e-14:
            g[:n - d] += wn * (1.0 - h[d:])
            g[d:] -= wn * h[:n - d]
    return (2.0 / n) * g

def smooth_val(x, n, alpha):
    return smooth_components(x, n, alpha)[0]

def smooth_jac(x, n, alpha):
    return grad_weights(x, n, smooth_components(x, n, alpha)[1])

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones(n, dtype=float)

def pen_val(x, n, alpha, mu):
    val, p, _ = smooth_components(x, n, alpha)
    dev = float(x.sum() - 0.5 * n)
    return val + 0.5 * mu * dev * dev

def pen_jac(x, n, alpha, mu):
    _, p, _ = smooth_components(x, n, alpha)
    dev = float(x.sum() - 0.5 * n)
    return grad_weights(x, n, p) + mu * dev

def lbfgs_stage(h, n, bounds, plans, mu, deadline):
    best = h.copy()
    bc = eval_c5(best, n)
    work = best.copy()
    for alpha, mx in plans:
        if time.time() > deadline - 2.0:
            break
        try:
            r = minimize(pen_val, work, args=(n, alpha, mu), method='L-BFGS-B',
                         jac=pen_jac, bounds=bounds,
                         options={'maxiter': mx, 'ftol': 1e-12, 'maxls': 20, 'disp': False})
            x = project_feasible(np.asarray(r.x, dtype=float), n)
            cc = eval_c5(x, n)
            if cc < bc:
                best, bc, work = x, cc, x.copy()
            elif cc < bc + 1e-7:
                work = x
        except Exception:
            pass
    return best, bc

def slsqp_stage(h, n, bounds, cons, plans, deadline):
    best = h.copy()
    bc = eval_c5(best, n)
    work = best.copy()
    for alpha, mx in plans:
        if time.time() > deadline - 2.0:
            break
        try:
            r = minimize(smooth_val, work, args=(n, alpha), method='SLSQP',
                         jac=smooth_jac, bounds=bounds, constraints=cons,
                         options={'maxiter': mx, 'ftol': 1e-12, 'disp': False})
            x = project_feasible(np.asarray(r.x, dtype=float), n)
            cc = eval_c5(x, n)
            if cc < bc:
                best, bc, work = x, cc, x.copy()
        except Exception:
            pass
    return best, bc

def transform_best(h, n):
    best = project_feasible(h, n)
    bc = eval_c5(best, n)
    base = best
    cands = [base, base[::-1], 1.0 - base, 1.0 - base[::-1]]
    for s in (n // 6, n // 4, n // 3, n // 2):
        if s == 0:
            continue
        r = np.roll(base, s)
        cands.extend([r, r[::-1], 1.0 - r, 1.0 - r[::-1]])
    if n % 2 == 0:
        th = np.zeros(n, dtype=float)
        th[np.argsort(h)[::-1][:n // 2]] = 1.0
        cands.extend([th, th[::-1], 1.0 - th])
    for v in cands:
        v = project_feasible(v, n)
        cc = eval_c5(v, n)
        if cc < bc:
            best, bc = v, cc
    return best, bc

def frank_wolfe_stage(h, n, deadline):
    if n % 2 != 0:
        return h, eval_c5(h, n)
    best = h.copy()
    bc = eval_c5(best, n)
    work = best.copy()
    for alpha in (20, 80, 300, 1000):
        if time.time() > deadline - 5.0:
            break
        for _ in range(20):
            if time.time() > deadline - 2.0:
                break
            _, p, _ = smooth_components(work, n, alpha)
            g = grad_weights(work, n, p)
            v = np.zeros(n, dtype=float)
            v[np.argsort(g)[:n // 2]] = 1.0
            if np.allclose(v, work):
                break
            bt = 0.0
            bw = eval_c5(work, n)
            for t in (0.1, 0.25, 0.4, 0.55, 0.7, 0.85, 1.0):
                cc = eval_c5((1.0 - t) * work + t * v, n)
                if cc < bw:
                    bw, bt = cc, t
            if bt > 0.0:
                work = project_feasible((1.0 - bt) * work + bt * v, n)
                cc = eval_c5(work, n)
                if cc < bc:
                    best, bc = work, cc
            else:
                break
    return best, bc

def compute_A(h, n):
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    A[:, off] = 1.0 - 2.0 * h
    for d in range(1, n):
        A[d:, off + d] += 1.0 - h[:n - d]
        A[:n - d, off + d] -= h[d:]
        A[:n - d, off - d] += 1.0 - h[d:]
        A[d:, off - d] -= h[:n - d]
    return A

def max_for_us(us, c, B, i, j, n):
    L = us[:, None] * B[None, :]
    L += c[None, :]
    off = n - 1
    d = i - j
    u2 = us * us
    L[:, off] -= 2.0 * u2
    L[:, off + d] += u2
    L[:, off - d] += u2
    return np.max(L, axis=1)

def eval_pair_u(h, c, A, i, j, n, grid, cur_max=None, refine=True, ref_grid=61):
    if i == j:
        return None
    ai = float(h[i])
    bj = float(h[j])
    s = ai + bj
    lo = s - 1.0
    lo = 0.0 if lo < 0.0 else lo
    hi = 1.0 if s > 1.0 else s
    ulo = lo - ai
    uhi = hi - ai
    if uhi - ulo < 1e-13:
        return None
    if cur_max is None:
        cur_max = float(c.max())
    B = A[i] - A[j]
    us = np.linspace(ulo, uhi, int(max(5, grid)))
    vals = max_for_us(us, c, B, i, j, n)
    im = int(np.argmin(vals))
    if vals[im] >= cur_max - 1e-9:
        return None
    if refine:
        ctr = float(us[im])
        half = (uhi - ulo) * 0.12
        a = ctr - half
        b = ctr + half
        if a < ulo:
            a = ulo
        if b > uhi:
            b = uhi
        if b - a > 1e-12:
            us2 = np.linspace(a, b, int(ref_grid))
            vals2 = max_for_us(us2, c, B, i, j, n)
            im2 = int(np.argmin(vals2))
            if vals2[im2] < vals[im] - 1e-12:
                return float(us2[im2])
    return float(us[im])

def apply_pair(h, i, j, u):
    old = float(h[i] + h[j])
    t = float(h[i] + u)
    if t < 0.0:
        t = 0.0
    elif t > 1.0:
        t = 1.0
    h[i] = t
    h[j] = old - t

def pair_search(h, n, deadline, rng, grid=17, ref_grid=65, top_lags=10,
                random_tries=150, max_cycles=300, max_accept=6000):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    A = compute_A(h, n)
    acc = 0
    tl = int(top_lags * (500.0 / max(n, 1)))
    if tl < 3:
        tl = 3
    if tl > 14:
        tl = 14
    for _ in range(max_cycles):
        if time.time() > deadline:
            break
        improved = 0
        cur_max = float(c.max())
        order = np.argsort(c)[-tl:]
        for kk in order:
            if time.time() > deadline:
                break
            d = int(kk) - (n - 1)
            if d == 0:
                continue
            m = n - abs(d)
            if m <= 1:
                continue
            inds = np.arange(m)
            if m > 256:
                rng.shuffle(inds)
            for b in inds:
                if time.time() > deadline:
                    break
                if d > 0:
                    i = b + d
                    j = b
                else:
                    i = b
                    j = b - d
                u = eval_pair_u(h, c, A, i, j, n, grid, cur_max, True, ref_grid)
                if u is not None:
                    apply_pair(h, i, j, u)
                    c = np.correlate(h, 1.0 - h, mode='full')
                    A = compute_A(h, n)
                    cur_max = float(c.max())
                    improved += 1
                    acc += 1
                    if acc >= max_accept:
                        return h, eval_c5(h, n)
        for _ in range(random_tries):
            if time.time() > deadline:
                break
            cur_max = float(c.max())
            tol = max(0.05, 0.002 * cur_max)
            act = np.flatnonzero(c >= cur_max - tol)
            if act.size:
                kk = int(act[int(rng.randint(act.size))])
            else:
                kk = int(np.argmax(c))
            d = int(kk) - (n - 1)
            if d == 0:
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    j = (j + 1) % n
            else:
                e = abs(d)
                m = n - e
                b = int(rng.randint(m))
                if d > 0:
                    i = b + d
                    j = b
                else:
                    i = b
                    j = b - d
            u = eval_pair_u(h, c, A, i, j, n, grid, cur_max, True, ref_grid)
            if u is not None:
                apply_pair(h, i, j, u)
                c = np.correlate(h, 1.0 - h, mode='full')
                A = compute_A(h, n)
                improved += 1
                acc += 1
                if acc >= max_accept:
                    return h, eval_c5(h, n)
        if improved == 0:
            break
    return h, eval_c5(h, n)

def optimize_instance(h, n, deadline, rng, quick=False):
    h = project_feasible(h, n)
    h, c = transform_best(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    if (not quick) and time.time() < deadline - 35.0:
        h, c = lbfgs_stage(h, n, bounds, [(20, 120), (80, 90), (300, 60)], 1e5, deadline)
    if time.time() < deadline - 15.0:
        if quick:
            plans = [(20, 40), (80, 40), (300, 30)]
        else:
            plans = [(20, 80), (80, 80), (300, 60), (1000, 45)]
        h, c = slsqp_stage(h, n, bounds, cons, plans, deadline)
    if (not quick) and n % 2 == 0 and time.time() < deadline - 10.0:
        h, c = frank_wolfe_stage(h, n, deadline)
    if time.time() < deadline - 1.0:
        h, c = pair_search(
            h, n, deadline, rng,
            grid=13 if quick else 17,
            ref_grid=45 if quick else 65,
            top_lags=6 if quick else 10,
            random_tries=60 if quick else 150
        )
    return h, eval_c5(h, n)
