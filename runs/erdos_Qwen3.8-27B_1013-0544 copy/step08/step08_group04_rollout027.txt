def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx):
                h[idx[0]] = min(1.0, h[idx[0]] + diff)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx):
                h[idx[0]] = max(0.0, h[idx[0]] + diff)
    return h

def compute_correlations(h):
    """Compute full correlation of h with 1-h"""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c(corr, n):
    return float(corr.max()) * (2.0 / n)

def gradient_active(corr, h, n, top_k=5):
    """Compute gradient of top_k worst lags, averaged"""
    m = 2 * n - 1
    center = n - 1
    # Find top_k lags
    idx = np.argsort(corr)[::-1][:top_k]
    g = np.zeros(n)
    for mi in idx:
        s = mi - center
        if s == 0:
            g += (1.0 - 2.0 * h)
        elif s > 0:
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
        else:
            r = -s
            g[r:] += (1.0 - h[:n-r])
            g[:n-r] -= h[r:]
    return g / len(idx)

def gradient_softmax(corr, h, n, beta=100.0):
    """Compute softmax-weighted gradient"""
    m = 2 * n - 1
    center = n - 1
    z = beta * corr
    z -= z.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    for mi in range(m):
        if w[mi] < 1e-14:

        s = mi - center
        wm = w[mi]
        if s == 0:

        elif s > 0:
            g[:n-s] += wm * (1.0 - h[s:])
            g[s:] -= wm * h[:n-s]
        else:
            r = -s
            g[r:] += wm * (1.0 - h[:n-r])
            g[:n-r] -= wm * h[r:]
    return g

    h = project_feasible(h.copy(), n)
    corr = compute_correlations(h)
    best_c = compute_c(corr, n)
    best_h = h.copy()
    
    y = h.copy()
    t = 1.0
    alpha = 2.0 / n  # initial step size
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = compute_correlations(y)
        c = compute_c(corr, n)
        
        # Gradient

        g = gradient_softmax(corr, y, n, beta)
        
        # Step

        h_new = project_feasible(z, n)
        
        # FISTA update

        y = h_new + ((t - 1.0) / t_new) * (h_new - h)
        y = np.clip(y, 0.0, 1.0)  # keep y in bounds (soft)
        h = h_new
        t = t_new

        c_new = compute_c(corr_new, n)
        if c_new < c - 1e-15:
            alpha *= 1.05
        else:
            alpha *= 0.9
        alpha = min(alpha, 0.1)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
    
    return best_h, best_c

import numpy as np

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = 0.5 * n
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if diff > 1e-12:
        idx = np.where(h < 1.0)[0]
        for i in idx:
            add = min(1.0 - h[i], diff)
            h[i] += add
            diff -= add
            if diff <= 1e-14: break
    elif diff < -1e-12:
        idx = np.where(h > 0.0)[0]
        for i in idx:
            rem = min(h[i], -diff)
            h[i] -= rem
            diff += rem
            if diff >= -1e-14: break
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c_val(corr, n):
    return float(corr.max()) * (2.0 / n)

def grad_topk(corr, h, n, k=3):
    center = n - 1
    top = np.argpartition(corr, -k)[-k:]
    g = np.zeros(n)
    for mi in top:
        s = mi - center
        if s == 0:
            g += (1.0 - 2.0 * h)
        elif s > 0:
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
        else:
            r = -s
            g[r:] += (1.0 - h[:n-r])
            g[:n-r] -= h[r:]
    return g / k

def grad_softmax(corr, h, n, beta):
    center = n - 1
    z = beta * (corr - corr.max())
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12: continue
        s = m - center
        wm = w[m]
        if s == 0:
            g += wm * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n-s] += wm * (1.0 - h[s:])
            g[s:] -= wm * h[:n-s]
        else:
            r = -s
            g[r:] += wm * (1.0 - h[:n-r])
            g[:n-r] -= wm * h[r:]
    return g

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old.copy(), n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def fourier_h(n, seed):
    rng = np.random.RandomState(seed)
    K = min(15, n//8)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 * np.ones(n)
    for k in range(K):
        amp = rng.uniform(0.03, 0.2) / (1.0 + 0.3*k)
        ph = rng.uniform(0, 2*np.pi)
        h += amp * np.cos((k+1)*np.pi*x + ph)
    return project_feasible(h, n)

def binary_structured(n, seed):
