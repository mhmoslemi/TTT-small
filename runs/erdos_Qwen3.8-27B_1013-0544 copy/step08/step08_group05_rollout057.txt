def project_simplex(h, n):

    target = 0.5 * n
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h + mid, 0, 1).sum() < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(h + hi, 0, 1)

    if abs(diff) > 1e-14:
        free = (y > 1e-10) & (y < 1 - 1e-10)

        else:

import numpy as np
from scipy.optimize import minimize
import time


def project_simplex(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * n
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h + mid, 0, 1).sum() < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(h + hi, 0, 1)
    diff = target - y.sum()
    if abs(diff) > 1e-14:
        free = (y > 1e-10) & (y < 1 - 1e-10)
        if free.any():
            y[free] += diff / free.sum()
        else:
            idx = np.argmax(1 - y) if diff > 0 else np.argmax(y)
            y[idx] = np.clip(y[idx] + diff, 0, 1)
    return np.clip(y, 0, 1)


def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)


def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    r = alpha * (c - m)
    r = r - r.max()
    ee = np.exp(r)
    z = ee.sum()
    dx = 2.0 / n
    val = (m + np.log(z) / alpha) * dx
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    grad = (A - B)[s:s + n] * dx
    return val, grad


def slsqp_fn(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]


def slsqp_jac(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]


def slsqp_cfun(x, n):
    return x.sum() - 0.5 * n


def slsqp_cjac(x, n):
    return np.ones(n)


def slsqp_polish(x0, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_cfun, 'jac': slsqp_cjac, 'args': (n,)}]
    try:
        res = minimize(slsqp_fn, x0, args=(n, alpha), jac=slsqp_jac,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        x_new = project_simplex(res.x, n)
        f_new = compute_c5(x_new, n)
        if f_new < compute_c5(x0, n):
            return x_new, f_new
    except Exception:
        pass
    return x0, compute_c5(x0, n)


def proj_grad_descent(x0, n, alphas, deadline, max_it=50000):
    x = project_simplex(x0, n)
    best_x = x.copy()
    best_f = compute_c5(x, n)
    dx = 2.0 / n
    for alpha in alphas:
        if time.time() > deadline:
            break
        step = dx * 0.3
        stag = 0
        for it in range(max_it):
            if time.time() > deadline + 3:
                break
            f, g = smooth_val_grad(x, n, alpha)
            for _ in range(25):
                y = project_simplex(x - step * g, n)
                f2, _ = smooth_val_grad(y, n, alpha)
                if f2 <= f + 1e-4 * np.dot(g, y - x) + 1e-15:
                    break
                step *= 0.5
            x = y
            f_new = compute_c5(x, n)
            if f_new < best_f - 1e-15:
                best_f = f_new
                best_x = x.copy()
                stag = 0
                step = min(step * 1.08, dx)
            else:
                stag += 1
                step *= 0.85
            if stag > 800:
                break
    return best_x, best_f


def coord_cd(x0, n, max_passes=6, deadline=None, tol=1e-14):
    x = project_simplex(x0, n)
    best_f = compute_c5(x, n)
    L = 2 * n - 1
    diag = n - 1
    dx = 2.0 / n
    gr_ratio = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for j in range(n):
            old = x[j]
            st = diag - j
            a_lo, b_hi = 0.0, 1.0
            c1 = b_hi - gr_ratio * (b_hi - a_lo)
            c2 = a_lo + gr_ratio * (b_hi - a_lo)

            def eval_t(t):
                d = t - old
                cn = c.copy()
                cn[j:j + n] += d * om
                cn[st:st + n] -= d * x
                cn[diag] -= d * d
                return cn.max()

            f1, f2 = eval_t(c1), eval_t(c2)

                else:
