import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size: h[idx[0]] = min(1.0, h[idx[0]] + res)
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size: h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def optimize_coord(hp, cseg, a, qconst, unaff, iters=20):

    m = a.size
    b = cseg - a * hp
    pos = a > 1e-12
    neg = a < -1e-12

    zmax = float(np.max(b[zero])) if zero.any() else -1e100
    cm = float(np.max(cseg)) if m else 0.0
    qcur = qconst + hp * (1.0 - hp)
    low = max(0.0, unaff, qconst, zmax)
    high = max(low + 1e-11, unaff, qconst, qcur, cm) + 1e-8
    a_pos = a[pos]; a_neg = a[neg]
    b_pos = b[pos]; b_neg = b[neg]

        feas = False

                    r2 = 0.5 + 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                    if max(0.0, ll) <= min(r1, lh) + tol: feas = True
                    else: feas = max(r2, ll) <= min(1.0, lh) + tol
        if feas: high = mid

    t = high

def cd_full_pass(h, n, order, deadline, project_every=10):
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    center = n - 1
    L = 2 * n - 1
    improved = 0
    for li, p in enumerate(order):
        if time.time() > deadline: break
        hp = float(h[p])
        lo = min(p, center - p)
        hi = max(center + p, 2 * center - p)
        seg = hi - lo + 1
        s = center - np.arange(lo, hi + 1, dtype=int)
        j = p + s
        i = p - s
        a = np.zeros(seg, dtype=float)
        vj = (j >= 0) & (j < n)
        vi = (i >= 0) & (i < n)
        if np.any(vj): a[vj] += 1.0 - h[j[vj]]
        if np.any(vi): a[vi] -= h[i[vi]]
        off = center - lo
        cseg = np.empty(seg - 1, dtype=float)
        asg = np.empty(seg - 1, dtype=float)
        if off:
            cseg[:off] = c[lo:center]
            asg[:off] = a[:off]
        if hi > center:
            cseg[off:] = c[center+1:hi+1]
            asg[off:] = a[off+1:]
        qconst = c[center] - hp * (1.0 - hp)
        unaff = 0.0
        if lo > 0: unaff = max(unaff, float(c[:lo].max()))
        if hi + 1 < L: unaff = max(unaff, float(c[hi+1:].max()))
        bx, bf = optimize_coord(hp, cseg, asg, qconst, unaff)
        if bf < cur - 1e-10:
            dxv = bx - hp
            if off: c[lo:center] += asg[:off] * dxv
            if hi > center: c[center+1:hi+1] += asg[off:] * dxv
            c[center] = qconst + bx*(1.0-bx)
            h[p] = bx
            cur = bf
            improved += 1
        if (li + 1) % project_every == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            cur = float(c.max())
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    return h, (2.0/n)*cur, improved

def run_cd(h, n, max_passes, deadline, rng):
    h = project_feasible(h, n)
    c = compute_c(h, n)
    best_h = h.copy()
    best_c = c
    for p in range(max_passes):

        h, c, imp = cd_full_pass(h, n, order, deadline)
        if c < best_c - 1e-13:
            best_c = c; best_h = h.copy()
        if imp == 0: break
    return best_h, best_c

def perturb_block(h, n, rng):
    """Swap two random blocks of h"""
    h2 = h.copy()
    blen = rng.randint(n // 10, n // 3)
    i = rng.randint(0, n - blen)
    j = rng.randint(0, n - blen)
    if abs(i - j) < blen:
        j = (i + blen + rng.randint(0, n - 2*blen)) % (n - blen)
    h2[i:i+blen], h2[j:j+blen] = h2[j:j+blen].copy(), h2[i:i+blen].copy()
    return project_feasible(h2, n)

def perturb_fourier(h, n, rng, n_modes=6):
    """Add random low-frequency Fourier perturbation"""
    t = (np.arange(n) + 0.5) * (2.0 / n)
    h2 = h.copy()
    for k in range(1, n_modes + 1):
        amp = rng.uniform(-0.15, 0.15) / k
        h2 += amp * np.cos(k * np.pi * (t - 1.0))
    return project_feasible(h2, n)

    return project_feasible(h + rng.normal(0, sigma, n), n)

import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size:
                h[idx[0]] = min(1.0, h[idx[0]] + res)
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size:
                h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def optimize_coord(hp, cseg, a, qconst, unaff, iters=22):
    tol = 1e-10
    m = a.size
    b = cseg - a * hp
    pos = a > 1e-12
    neg = a < -1e-12
    zero = ~(pos | neg)
    zmax = float(np.max(b[zero])) if zero.any() else -1e100
    cm = float(np.max(cseg)) if m else 0.0
    qcur = qconst + hp * (1.0 - hp)
    low = max(0.0, unaff, qconst, zmax)
    high = max(low + 1e-11, unaff, qconst, qcur, cm) + 1e-8
    a_pos = a[pos]; a_neg = a[neg]
    b_pos = b[pos]; b_neg = b[neg]
    pc = a_pos.size; nc = a_neg.size
    for _ in range(iters):
        mid = 0.5 * (low + high)
        feas = False
        if mid >= unaff - tol and mid >= qconst - tol and mid >= zmax - tol:
            ll = 0.0; lh = 1.0
            if nc:
                v = float(np.max((mid - b_neg) / a_neg))
                if v > ll: ll = v
            if pc:
                v = float(np.min((mid - b_pos) / a_pos))
                if v < lh: lh = v
            if ll <= 1.0 + tol and lh >= -tol:
                g = mid - qconst
                if g > 0.25:
                    feas = max(0.0, ll) <= min(1.0, lh) + tol
                elif g > -1e-12:
                    if g < 0.0: g = 0.0
                    r1 = 0.5 - 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                    r2 = 0.5 + 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                    if max(0.0, ll) <= min(r1, lh) + tol:
                        feas = True
                    else:
                        feas = max(r2, ll) <= min(1.0, lh) + tol
        if feas:
            high = mid
        else:
            low = mid
    t = high
    cands = [0.0, 1.0, hp, 0.5]
    if t >= unaff - tol and t >= qconst - tol and t >= zmax - tol:
        ll = 0.0; lh = 1.0
        if nc:
            v = float(np.max((t - b_neg) / a_neg))
            if v > ll: ll = v
        if pc:
            v = float(np.min((t - b_pos) / a_pos))
            if v < lh: lh = v
        if ll <= 1.0 + tol and lh >= -tol:
            g = t - qconst
            if g > 0.25:
                cands.extend([max(0.0, ll), min(1.0, lh), 0.5*(max(0.0,ll)+min(1.0,lh))])
            elif g >= -1e-12:
                if g < 0.0: g = 0.0
                r1 = 0.5 - 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                r2 = 0.5 + 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))

    best_f = max(unaff, qconst + hp*(1.0-hp))

        val = max(val, unaff)
        if val < best_f - 1e-13:

            best_x = x
    return best_x, best_f

def cd_pass(h, n, order, deadline, pe):
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    center = n - 1
    L = 2 * n - 1
    improved = 0
    for li, p in enumerate(order):
        if time.time() > deadline:
            break
        hp = float(h[p])
        lo = min(p, center - p)
        hi = max(center + p, 2 * center - p)
        seg = hi - lo + 1
        s = center - np.arange(lo, hi + 1, dtype=int)
        j = p + s
        i = p - s
        a = np.zeros(seg, dtype=float)
        vj = (j >= 0) & (j < n)
        vi = (i >= 0) & (i < n)
        if np.any(vj):
            a[vj] += 1.0 - h[j[vj]]
        if np.any(vi):
            a[vi] -= h[i[vi]]
        off = center - lo
        cseg = np.empty(seg - 1, dtype=float)
        asg = np.empty(seg - 1, dtype=float)
        if off:
            cseg[:off] = c[lo:center]
            asg[:off] = a[:off]
        if hi > center:
            cseg[off:] = c[center+1:hi+1]
            asg[off:] = a[off+1:]
        qconst = c[center] - hp * (1.0 - hp)
        unaff = 0.0
        if lo > 0:
            unaff = max(unaff, float(c[:lo].max()))
        if hi + 1 < L:
            unaff = max(unaff, float(c[hi+1:].max()))
        bx, bf = optimize_coord(hp, cseg, asg, qconst, unaff)
        if bf < cur - 1e-10:
            dxv = bx - hp
            if off:
                c[lo:center] += asg[:off] * dxv
            if hi > center:
                c[center+1:hi+1] += asg[off:] * dxv
            c[center] = qconst + bx*(1.0-bx)
            h[p] = bx
            cur = bf
            improved += 1
        if (li + 1) % pe == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            cur = float(c.max())
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    return h, (2.0/n)*cur, improved

def run_cd(h, n, max_passes, deadline, rng):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c(h, n)
    pe = max(5, n // 50)
    for p in range(max_passes):
        if time.time() > deadline:
            break
        order = rng.permutation(np.arange(n))
        h, c, imp = cd_pass(h, n, order, deadline, pe)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
        if imp == 0:
            break
    return best_h, best_c

def perturb_gauss(h, n, rng, sigma):
    return project_feasible(h + rng.normal(0, sigma, n), n)

def perturb_block(h, n, rng):
    h2 = h.copy()
    blen = rng.randint(max(2, n//15), n//3)
    i = rng.randint(0, n - blen)
    j = rng.randint(0, n - blen)
    if abs(i - j) <= blen:
        j = (i + blen + rng.randint(0, max(1, n - 2*blen))) % max(1, n - blen)
    h2[i:i+blen], h2[j:j+blen] = h2[j:j+blen].copy(), h2[i:i+blen].copy()
    return project_feasible(h2, n)

def perturb_fourier(h, n, rng, kmodes):
    t = (np.arange(n) + 0.5) * (2.0/n) - 1.0
    h2 = h.copy()
    for k in range(1, kmodes+1):
        amp = rng.uniform(-0.12, 0.12) / (k**0.7)
        h2 += amp * np.cos(k * np.pi * t + rng.uniform(0, 2*np.pi))
    return project_feasible(h2, n)

def perturb_binary(h, n, rng):
    thresh = np.quantile(h, 0.35 + 0.3*rng.random())
    h2 = np.where(h > thresh, 1.0, 0.0)
    return project_feasible(h2, n)
