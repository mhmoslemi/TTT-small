def project_simplex(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m

    d = t - y.sum()

            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)

def binary_search_improve(h, n, rng, deadline):
    """Binary optimization with smarter moves"""
    # Binarize
    target = n // 2
    if target <= 0 or target >= n:
        return None, np.inf
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    dx = 2.0 / n
    
    best = b.copy()
    c = compute_corr(best, n)
    bc = float(c.max())
    
    stagnant = 0
    while time.time() < deadline:
        # Try individual flips that improve
        ones = np.where(best == 1.0)[0]
        zeros = np.where(best == 0.0)[0]
        found = False
        # Adaptive: try more when stuck less
        n_trials = 2000 + 5000 * min(stagnant // 1000, 5)
        for _ in range(n_trials):
            if time.time() > deadline:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            if i == j:
                continue
            # Quick evaluate using precomputed structure
            best[i] = 0.0
            best[j] = 1.0
            c2 = compute_corr(best, n)
            cm = float(c2.max())
            if cm < bc - 1e-12:
                c = c2
                bc = cm
                ones = np.where(best == 1.0)[0]
                zeros = np.where(best == 0.0)[0]
                stagnant = 0
                found = True
            else:
                best[i] = 1.0
                best[j] = 0.0
                stagnant += 1
            if found and stagnant > 2000:
                break
        
        if stagnant > 15000:
            # Random restart: flip several pairs
            for _ in range(5):
                i = int(rng.choice(ones))
                j = int(rng.choice(zeros))
                if i != j:
                    best[i] = 0.0
                    best[j] = 1.0
            c = compute_corr(best, n)
            bc_new = float(c.max())
            if bc_new < bc + 0.3 * bc:
                bc = bc_new  # accept even if slightly worse for exploration
            else:
                # revert
                best = project_simplex(best, n)
                c = compute_corr(best, n)
                bc = float(c.max())
            stagnant = 0
    
    return best, bc * dx

def slsqp_smooth(h, n, alphas, maxiter, deadline):
    """SLSQP on smooth approximation"""
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        a = alphas[0] if isinstance(alphas, list) else alphas
        cm = float(c.max())
        r = a * (c - cm)
        r -= r.max()
        return (cm + np.log(np.exp(r).sum()) / a) * (2.0 / n)
    
    def grads(x, a):
        c = np.correlate(x, 1.0 - x, mode="full")
        cm = float(c.max())
        r = a * (c - cm)
        r -= r.max()
        ee = np.exp(r)
        w = ee / float(ee.sum())
        A = np.convolve(1.0 - x, w, mode="full")
        B = np.convolve(x, w[::-1], mode="full")
        s = n - 1
        return (A - B)[s:s+n] * (2.0 / n)
    
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": lambda x: x.sum() - 0.5*n, "jac": lambda x: np.ones(n)}]
    
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(obj, best, args=(a,), jac=lambda x, aa: grads(x, aa),
                          method="SLSQP", bounds=bounds, constraints=cons,
                          options={"maxiter": maxiter, "ftol": 1e-12})
            nh = project_simplex(res.x, n)
            nc = compute_c5(nh, n)
            if nc < bc - 1e-13:
                best = nh
                bc = nc
        except Exception:
            continue
    return best, bc
