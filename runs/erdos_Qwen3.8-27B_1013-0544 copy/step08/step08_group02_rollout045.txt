import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.nan_to_num(np.asarray(z, dtype=float), nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    lo = float(z.min()) - 2.0
    hi = float(z.max()) + 2.0
    for _ in range(60):
        t = 0.5 * (lo + hi)
        if np.clip(z - t, 0.0, 1.0).sum() > target:
            lo = t
        else:
            hi = t
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-13:
        m = y < 1.0 if r > 0 else y > 0.0
        if m.any():
            rm = (1.0 - y) if r > 0 else y
            y[m] += (r / rm.sum()) * rm[m]
    return np.clip(y, 0.0, 1.0)

def compute_A(h, n):
    off = n - 1
    d = np.arange(-(n-1), n, dtype=int)
    A = np.zeros((n, 2*n-1))
    for i in range(n):
        m1 = (i + d >= 0) & (i + d < n)
        m2 = (i - d >= 0) & (i - d < n)
        A[i, m1] = 1.0 - h[i + d[m1]]
        A[i, m2] -= h[i - d[m2]]
    return A

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x)

def smooth_val(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    z -= float(z.max())
    e = np.exp(z)
    return (m + float(np.log(e.sum())) / alpha) * (2.0 / n)

def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    z -= float(z.max())
    e = np.exp(z)
    p = e / float(e.sum())
    off = n - 1
    d = np.arange(-(n-1), n, dtype=int)
    g = np.zeros(n)
    for i in range(n):
        m1 = (i + d >= 0) & (i + d < n)
        m2 = (i - d >= 0) & (i - d < n)
        bg = np.zeros(2*n-1)
        bg[m1] = 1.0 - h[i + d[m1]]
        bg[m2] -= h[i - d[m2]]
        g[i] = float(np.dot(p, bg))
    return (2.0 / n) * g

def slsqp_opt(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_val, h, args=(n, alpha), method='SLSQP',
                       jac=smooth_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False})
        x = np.asarray(res.x, dtype=float)
        if np.all(np.isfinite(x)):
            return project_feasible(x, n)
    except Exception:
        pass
    return h

def pair_search(h, n, max_tries, deadline, rng, grid=80):
    h = h.copy()
    A = compute_A(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    off = n - 1
    d_arr = np.arange(-(n-1), n, dtype=int)
    best_c = float(c.max())
    no_imp = 0
    for trial in range(max_tries):
        if time.time() > deadline:
            break
        if no_imp > 3000:
            break
        cur_max = float(c.max())
        if cur_max < best_c - 1e-10:
            best_c = cur_max
        # Select pair
        tol = max(0.02, 0.01 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if trial % 5 == 4 or active.size <= 1:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
        else:
            kk = int(active[int(rng.randint(active.size))])
            dd = kk - off
            if dd >= 0:
                m = n - dd
                if m <= 0:
                    continue
                b = int(rng.randint(m))
                i, j = b, b + dd
            else:
                e = -dd
                m = n - e
                if m <= 0:
                    continue
                b = int(rng.randint(m))
                i, j = b + e, b
        # Compute B = A[i] - A[j]
        B = A[i] - A[j]
        # Line search: minimize max_k(c[k] + u*B[k]) for u in valid range
        S = h[i] + h[j]
        u_lo = max(0.0, S - 1.0) - h[i]
        u_hi = min(1.0, S) - h[i]
        if u_hi - u_lo < 1e-14:
            no_imp += 1
            continue
        us = np.linspace(u_lo, u_hi, grid)
        vals = (c + us[:, None] * B[None, :]).max(axis=1)
        idx = int(np.argmin(vals))
        # Refine
        lo_r = us[max(0, idx - 2)]
        hi_r = us[min(grid - 1, idx + 2)]
        if hi_r > lo_r + 1e-14:
            us2 = np.linspace(lo_r, hi_r, 30)
            vals2 = (c + us2[:, None] * B[None, :]).max(axis=1)
            idx2 = int(np.argmin(vals2))
            u = float(us2[idx2])
        else:
            u = float(us[idx])
        if abs(u) < 1e-15:
            no_imp += 1
            continue
        new_i = h[i] + u
        new_j = h[j] - u
        if new_i < -1e-12 or new_i > 1 + 1e-12 or new_j < -1e-12 or new_j > 1 + 1e-12:
            no_imp += 1
            continue

        nm = float(c_new.max())
        if nm < cur_max - 1e-10:
            h[i] = new_i
            h[j] = new_j
            c = c_new
            # Update A rows for i and j
            for idx2 in (i, j):
                m1 = (idx2 + d_arr >= 0) & (idx2 + d_arr < n)
                m2 = (idx2 - d_arr >= 0) & (idx2 - d_arr < n)
                A[idx2, :] = 0.0
                A[idx2, m1] = 1.0 - h[idx2 + d_arr[m1]]
                A[idx2, m2] -= h[idx2 - d_arr[m2]]
            no_imp = 0
        else:
            no_imp += 1
    return h, compute_c5(h, n)

def optimize(h, n, deadline, rng):
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    # Try symmetries
    hs = project_feasible(0.5 * (h + h[::-1]), n)
    cs = compute_c5(hs, n)
    if cs < c:
        h, c = hs, cs
    hc = 1.0 - h
    cs = compute_c5(hc, n)
    if cs < c:
        h, c = hc, cs
    # SLSQP with increasing alpha
    if time.time() < deadline - 30:
        for alpha, mx in [(50, 40), (200, 30), (1000, 25), (5000, 20)]:
            if time.time() > deadline - 30:
                break
            h2 = slsqp_opt(h, n, alpha, mx, deadline)
            c2 = compute_c5(h2, n)
            if c2 < c:
                h, c = h2, c2
    # Pair search (main workhorse)
    rem = deadline - time.time()
    if rem > 2:
        max_tries = int(min(500000, max(5000, rem * 5000)))
        h, c = pair_search(h, n, max_tries, deadline, rng, grid=80)
    return h, c
