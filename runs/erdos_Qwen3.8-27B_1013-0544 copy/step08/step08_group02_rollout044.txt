**Key new idea: Direct minimax via active constraint analysis**

Instead of smooth max + SLSQP, use the following approach:
1. Identify the top M active shifts (those within ε of the max)
2. Solve the KKT conditions: at the optimum, the gradient of the weighted sum of active constraints should be zero (weighted by the Lagrange multipliers of the active constraints)
3. This gives a system of linear equations for the Lagrange multipliers

Specifically:
- Let k1, ..., kM be the active shifts
- At the optimum: ∑_{m=1}^{M} λ_m * ∇c[k_m] = μ * 1 (for free variables)
- And λ_m ≥ 0, ∑ λ_m = 1
- This is a quadratic program: min ∑ λ_m^2 subject to A^T λ = μ*1, λ ≥ 0, 1^T λ = 1

Where A is the matrix of gradients of the active constraints.

This is a nice convex QP that can be solved quickly. The resulting gradient direction is the "steepest descent" for the min-max problem.

        if active.size > n_active:
            order = c[active]
            top = np.argsort(order)[-n_active:]
            active = active[top]
        
        # Average gradient direction from active constraints
        A = grad_corr_k_vec(h, n)
        g = A[:, active].mean(axis=1)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        
        # Backtracking line search
        alpha = step
        for _ in range(10):
            h_new = h - alpha * g
            h_new = project_sum(h_new, n)
            c_new = compute_corr(h_new)
            if c_new.max() < cmax * (1.0 - 0.01 * alpha * gn):
                h = h_new
                break
            alpha *= 0.7
        else:
            h = h_new  # accept anyway if it's not too bad
            cc = compute_c5(h, n)
            if cc > best_c * 1.05:
                h = best_h.copy()
                break
        
        step = max(0.001, step * 0.995)
    return best_h, best_c

def apply_symmetries(h, n):
    candidates = [h.copy(),
                  0.5 * (h + h[::-1]),
                  1.0 - h,
                  0.5 * (h + 1.0 - h[::-1])]
    best_h, best_c = h.copy(), compute_c5(h, n)
    for c in candidates:
        c = project_sum(c, n)
        cc = compute_c5(c, n)
        if cc < best_c:
            best_h, best_c = c, cc
    return best_h, best_c

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_sum(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_sum(np.interp(xn, xo, h), m)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 4.0
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        n0 = 200
        x = np.linspace(0, 2, n0, endpoint=False)
        h0 = np.clip(0.5 + 0.3*np.sin(np.pi*x) + 0.2*np.cos(2*np.pi*x) + 0.05*np.sin(3*np.pi*x), 0, 1)
        h0 = project_sum(h0, n0)
    
    n0 = len(h0)
    best_h, best_c = apply_symmetries(project_sum(h0, n0), n0)
    
    # Phase 1: SLSQP optimization on original resolution
    phase1_deadline = hard - max(200.0, 0.35 * budget_s)
    if phase1_deadline > t0 + 20:
        best_h, best_c = slsqp_opt(best_h, n0, phase1_deadline, alpha_init=30, n_plans=4)
    
    # Phase 2: Minimax subgradient
    if time.time() < hard - 150:
        dl = min(hard - 120, time.time() + 80)
        h_sub, c_sub = minimax_subgrad(best_h, n0, dl, n_steps=2000, n_active=10)
        if c_sub < best_c:
            best_h, best_c = h_sub, c_sub
    
    # Phase 3: Pair search on best resolution
    if time.time() < hard - 60:
        dl = min(hard - 40, time.time() + 100)
        best_h, best_c = pair_search(best_h, n0, 3000, 40, dl, rng)
    
    # Phase 4: Try different resolutions
    for m in [300, 400, 600]:
        if time.time() > hard - 50:
            break
        if m == n0:
            continue
        rem = hard - time.time()
        if rem < 40:
            break
        dl = time.time() + min(35.0, 0.2 * rem)
        ht = interp_h(best_h, m)
        ht, ct = apply_symmetries(ht, m)
        ht, ct = slsqp_opt(ht, m, dl - 10, alpha_init=20, n_plans=2)
        if time.time() < dl:
            ht, ct = pair_search(ht, m, 1000, 30, dl, rng)
        if ct < best_c:
            best_h, best_c = ht, ct
    
    # Phase 5: Final pair search with more budget
    if time.time() < hard - 10:
        dl = hard - 2
        n = len(best_h)
        best_h, best_c = pair_search(best_h, n, 5000, 50, dl, rng)
    
    # Phase 6: One more SLSQP polish
    if time.time() < hard - 5:
        n = len(best_h)
        best_h, best_c = slsqp_opt(best_h, n, hard - 1, alpha_init=200, n_plans=2)
    
    # Final pair search in remaining time
    if time.time() < hard - 2:
        n = len(best_h)
        best_h, best_c = pair_search(best_h, n, 1000, 40, hard - 1, rng)
    
    best_h = project_sum(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
