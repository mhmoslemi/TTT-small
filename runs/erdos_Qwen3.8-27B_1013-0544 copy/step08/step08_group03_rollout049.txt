    A = np.zeros(2 * n - 1)
    q = np.arange(n)
    mask = q != p
    qv = q[mask]
    A[n - 1 + p - qv] += 1.0 - h[qv]
    A[n - 1 - p + qv] += -h[qv]
    A[n - 1] += 1.0 - 2.0 * h[p]
    return A

def find_best_t(c0, A, D, grid_pts=60, gs_iters=45):
    def f(t):
        return float(np.max(c0 + t * A - t * t * D))
    grid = np.linspace(0.0, 1.0, grid_pts)
    fvals = np.array([f(t) for t in grid])
    best_idx = int(np.argmin(fvals))
    a = grid[max(0, best_idx - 1)]
    b = grid[min(grid_pts - 1, best_idx + 1)]
    if b - a < 1e-14:
        return float(grid[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    d1 = a + gr * (b - a)
    fc1, fd1 = f(c1), f(d1)
    for _ in range(gs_iters):
        if fc1 < fd1:
            b, d1, fd1 = d1, c1, fc1
            c1 = b - gr * (b - a); fc1 = f(c1)
        else:
            a, c1, fc1 = c1, d1, fd1
            d1 = a + gr * (b - a); fd1 = f(d1)
    return (a + b) * 0.5

def exact_cd(h, n, max_passes=5000, deadline=None, grid_pts=50, gs_iters=35, random_order=False, seed=0):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    D = (np.arange(2 * n - 1) == (n - 1)).astype(float)
    current_max = float(np.max(c))
    rng = np.random.RandomState(seed)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n) if random_order else range(n)
        for p in order:
            if deadline is not None and time.time() > deadline:
                return h, compute_c5(h, n)
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            A = compute_A(p, h, n)
            c0 = c - h0 * A + h0 * h0 * D
            t_star = find_best_t(c0, A, D, grid_pts, gs_iters)
            f_star = float(np.max(c0 + t_star * A - t_star * t_star * D))
            f0 = float(np.max(c0))
            f1 = float(np.max(c0 + A - D))
            best_f = min(f_star, f0, f1)
            if best_f < current_max - 1e-13:
                if f0 <= f_star and f0 <= f1:
                    t_new = 0.0
                elif f1 <= f_star and f1 <= f0:
                    t_new = 1.0
                else:
                    t_new = float(np.clip(t_star, 0.0, 1.0))
                delta = t_new - h0
                c += delta * A - (t_new * t_new - h0 * h0) * D
                h[p] = t_new
                current_max = float(np.max(c))
                improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_max_slsqp(h, n, beta, maxiter=600, deadline=None):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return float((2.0 / n) * (1.0 / beta) * np.log(np.sum(np.exp(beta * c))))
    def sum_con(x):
        return x.sum() - n / 2.0
    def sum_con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def pnorm_slsqp(h, n, p_list, maxiter_each=600, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in p_list:
        if deadline is not None and time.time() > deadline:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return float(np.sum(c ** p) ** (1.0 / p) * (2.0 / n))
        def sc(x):
            return x.sum() - n / 2.0
        def sj(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sc, 'jac': sj}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Long exact CD on initial grid
    h1, c1 = exact_cd(best_h, n_init, 3000, deadline, grid_pts=50, gs_iters=40, random_order=True, seed=seed)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Smooth max SLSQP (log-sum-exp smoothing)
    if time.time() < deadline - 60:
        for beta in [20, 50, 100, 200]:
            if time.time() > deadline - 30:
                break
            hs, cs = smooth_max_slsqp(best_h, len(best_h), beta, 400, deadline)
            if cs < best_c:
                best_h, best_c = hs, cs
                hf, cf = exact_cd(best_h, len(best_h), 500, deadline, grid_pts=50, gs_iters=40)
                if cf < best_c:
                    best_h, best_c = hf, cf

    # Phase 3: Multi-resolution exploration
    x_list = [350, 400, 450, 550, 600, 700, 800]
    for n_try in x_list:
        if time.time() > deadline - 120:
            break
        if abs(n_try - len(best_h)) < 50:
            continue
        h_int = interpolate_h(best_h, len(best_h), n_try)
        hOpt, cOpt = exact_cd(h_int, n_try, 1500, deadline, grid_pts=50, gs_iters=35, random_order=True, seed=seed + n_try)
        if cOpt < best_c:
            best_h, best_c = hOpt, cOpt

    # Phase 4: p-norm SLSQP on best grid
    n_best = len(best_h)
    if time.time() < deadline - 80:
        hp, cp = pnorm_slsqp(best_h, n_best, [100, 500, 2000], 400, deadline)
        if cp < best_c:
            best_h, best_c = hp, cp
            hf, cf = exact_cd(best_h, n_best, 1000, deadline, grid_pts=55, gs_iters=45)
            if cf < best_c:
                best_h, best_c = hf, cf

    # Phase 5: Random restarts with structured perturbations
    if time.time() < deadline - 60:
        n_restarts = max(2, int((deadline - time.time() - 40) / 100))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            if r % 3 == 0:
                hp = project_feasible(best_h + rng.normal(0, 0.05, n_best), n_best)
            elif r % 3 == 1:
                hp = project_feasible(best_h + rng.normal(0, 0.15, n_best), n_best)
            else:
                hp = project_feasible(rng.uniform(0, 1, n_best), n_best)
            ho, co = exact_cd(hp, n_best, 800, deadline, grid_pts=50, gs_iters=35, random_order=True, seed=seed * 7 + r)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 6: Final refinement CD
    if time.time() < deadline - 5:
        hf, cf = exact_cd(best_h, len(best_h), 5000, deadline, grid_pts=70, gs_iters=50)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
