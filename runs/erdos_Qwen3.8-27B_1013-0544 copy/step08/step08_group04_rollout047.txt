import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase: add to positions where h < 1
        mask = h < 1.0
        tot = float(mask.sum())
        if tot > 0:
            step = min(1.0, diff / tot)
            h[mask] += step
            h = np.clip(h, 0, 1)
            # Fine adjust
            resid = target - float(h.sum())
            if abs(resid) > 1e-12:
                idx = np.where(h < 1.0 if resid > 0 else h > 0.0)[0]
                if idx.size:
                    h[idx[0]] += np.clip(resid, -h[idx[0]], 1-h[idx[0]])
    else:
        mask = h > 0.0
        tot = float(mask.sum())
        if tot > 0:
            step = min(1.0, -diff / tot)
            h[mask] -= step
            h = np.clip(h, 0, 1)
            resid = target - float(h.sum())
            if abs(resid) > 1e-12:
                idx = np.where(h > 0.0 if resid < 0 else h < 1.0)[0]
                if idx.size:
                    h[idx[0]] += np.clip(resid, -h[idx[0]], 1-h[idx[0]])
    return h

def compute_corr(h, n):
    """Compute shifted correlations c[m] for m = 0..2n-2 (full mode)"""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c(h, n):
    c = compute_corr(h, n)
    return float(np.max(c) * (2.0 / n))

def triangle_h(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(1.0 - np.abs(x - 1.0), n)

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def binary_from_h(h):
    n = len(h)
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    h2 = np.zeros(n)
    h2[idx] = 1.0
    return h2

def basis_h(n, rng, modes=10):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 * np.ones(n)
    t = x - 1.0
    for k in range(1, modes + 1):
        amp = rng.uniform(0.0, 0.3 / k)
        sign = 1 if rng.rand() < 0.5 else -1
        h += sign * amp * np.cos(k * np.pi * t)
    return project_feasible(h, n)

import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
        tot = float(mask.sum())
        if tot > 0:
            h[mask] += min(1.0, diff / tot)
            h = np.clip(h, 0.1, 1.0)
            resid = target - h.sum()
            if abs(resid) > 1e-10:
                idx = np.where(h < 1.0)[0]
                if idx.size:
                    h[idx[0]] = np.clip(h[idx[0]] + resid, 0, 1)
    else:
        mask = h > 0.0
        tot = float(mask.sum())
        if tot > 0:
            h[mask] -= min(1.0, -diff / tot)
            h = np.clip(h, 0.0, 0.9)
            resid = target - h.sum()
            if abs(resid) > 1e-10:
                idx = np.where(h > 0.0)[0]
                if idx.size:
                    h[idx[0]] = np.clip(h[idx[0]] + resid, 0, 1)
    return np.clip(h, 0.0, 1.0)

def compute_c(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def triangle_h(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(1.0 - np.abs(x - 1.0), n)

def basis_h(n, rng, modes=10):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 * np.ones(n)
    t = x - 1.0
    for k in range(1, modes + 1):
        amp = rng.uniform(0.0, 0.25 / k)
        sgn = 1.0 if rng.rand() < 0.5 else -1.0
        h += sgn * amp * np.cos(k * np.pi * t + rng.uniform(0, np.pi))
    return project_feasible(h, n)

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def binary_from_h(h):
    n = len(h)
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    h2 = np.zeros(n)
    h2[idx] = 1.0
    return h2

def coord_optimize(h, n, p, c, center):
    """Find optimal value for coordinate p given current h and correlation c."""
    hp = h[p]
    lo_m = max(0, center - p)
    hi_m = min(2 * n - 2, center + p)
    seg = hi_m - lo_m + 1
    a = np.zeros(seg)
    for offset in range(seg):
        m = lo_m + offset
        q1 = p + m - center
        q2 = p - m + center
        if 0 <= q1 < n:
            a[offset] += (1.0 - h[q1])
        if 0 <= q2 < n:
            a[offset] -= h[q2]
    b = c[lo_m:hi_m + 1].copy() - hp * a
    # Binary search on the optimal max value
    zmax_zero = float(np.max(b[np.abs(a) < 1e-14])) if np.any(np.abs(a) < 1e-14) else -1e100
    cur_max = float(np.max(b + hp * a))
    lo_v = max(0.0, zmax_zero)
    hi_v = cur_max + 1e-10
    for _ in range(40):
        mid_v = 0.5 * (lo_v + hi_v)
        ll = 0.0
        lh = 1.0
        pos = a > 1e-14
        neg = a < -1e-14
        if np.any(neg):
            v = float(np.max((mid_v - b[neg]) / a[neg]))
            if v > ll:
                ll = v
        if np.any(pos):
            v = float(np.min((mid_v - b[pos]) / a[pos]))
            if v < lh:
                lh = v
        if ll <= 1.0 + 1e-12 and lh >= -1e-12:
            hi_v = mid_v
        else:
            lo_v = mid_v
    t_new = 0.0
    t_best = hp
    f_best = cur_max
    # Try the boundary and the found value
    ci = hi_v
    ll = 0.0
    lh = 1.0
    pos = a > 1e-14
    neg = a < -1e-14
    if np.any(neg):
        v = float(np.max((ci - b[neg]) / a[neg]))
        if v > ll:
            ll = v
    if np.any(pos):
        v = float(np.min((ci - b[pos]) / a[pos]))
        if v < lh:
            lh = v
    if ll <= 1.0 + 1e-12 and lh >= -1e-12:
        t_new = ll
    for t_try in [t_new, 0.0, 1.0, hp]:
        if t_try < 0 or t_try > 1:
            continue
        new_c_seg = b + t_try * a
        new_max = max(float(np.max(new_c_seg)), -1e100)
        # Also check unaffected entries
        if lo_m > 0:
            new_max = max(new_max, float(c[:lo_m].max()))
        if hi_m + 1 < 2 * n - 1:
            new_max = max(new_max, float(c[hi_m + 1:].max()))
        new_max = max(new_max, zmax_zero)
        if new_max < f_best:
            f_best = new_max
            t_best = t_try
    return t_best, f_best

def fast_cd(h, n, deadline, max_passes=300, priority=True):
    h = project_feasible(h, n)
    center = n - 1
    L = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h = h.copy()
    best_c = cur

        if time.time() > deadline:
            break
        improved = False
        if priority:
            # Compute priority for each coordinate (approximate improvement)
            priorities = []

                if time.time() > deadline:
                    break
                hp = h[p]
                lo_m = max(0, center - p)
                hi_m = min(L - 1, center + p)
                # Estimate improvement: how much does the max decrease?
                # Quick estimate: current max minus what it would be if we set h[p] optimally
                # Use a simpler proxy: the contribution of position p to the current max
                m_max = int(np.argmax(c))
                d = m_max - center
                q1 = p + d
                q2 = p - d
                contrib = 0.0
                if 0 <= q1 < n:
                    contrib += h[p] * (1.0 - h[q1])
                if 0 <= q2 < n:
                    contrib += h[q2] * (1.0 - h[p])
                priorities.append((-contrib, p))
            priorities.sort()
            order = [p for _, p in priorities]
        else:
            order = list(range(n))
        for p in order:
            if time.time() > deadline:
                break
            t_new, f_new = coord_optimize(h, n, p, c, center)
            if abs(t_new - h[p]) < 1e-10:
                continue
            if f_new < cur - 1e-12:
                hp = h[p]
                dx = t_new - hp
                lo_m = max(0, center - p)
                hi_m = min(L - 1, center + p)
                for m in range(lo_m, hi_m + 1):
                    q1 = p + m - center
                    q2 = p - m + center
                    delta = 0.0
                    if 0 <= q1 < n:
                        delta += (1.0 - h[q1])
                    if 0 <= q2 < n:
                        delta -= h[q2]
                    c[m] += dx * delta

                h[p] = t_new
                cur = f_new
                improved = True

            break

        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())

            best_c = cur
            best_h = h.copy()
    return best_h, best_c

def subgradient_step(h, n, alpha, deadline):
    """One subgradient step on the worst shift."""
    center = n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    m_max = int(np.argmax(c))
    d = m_max - center
    grad = np.zeros(n)
    for p in range(n):
        q1 = p + d
        q2 = p - d
        g = 0.0
        if 0 <= q1 < n:
            g += (1.0 - h[q1])
        if 0 <= q2 < n:
            g -= h[q2]
        grad[p] = g
    h = project_feasible(h - alpha * grad, n)
    return h
