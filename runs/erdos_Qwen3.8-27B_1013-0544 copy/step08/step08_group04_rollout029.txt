    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def cd_optimize(h, n, max_passes, deadline, proj_every=8):
    h = project_feasible(h, n)
    L = 2 * n - 1
    center = n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(c.max())
    best_h = h.copy()
    best_c = (2.0 / n) * cur_max
    s_arr = np.arange(L, dtype=np.int32) - center
    for pss in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = np.arange(n)
        for p in order:
            if time.time() > deadline:
                break
            hp = float(h[p])
            ps = p - s_arr.astype(np.int64)
            ph = p + s_arr.astype(np.int64)
            a = np.ones(L)
            vp = (ps >= 0) & (ps < n)
            vh = (ph >= 0) & (ph < n)
            a[vp] -= h[ps[vp]]
            a[vh] -= h[ph[vh]]
            qconst = c[center] - hp * (1.0 - hp)
            non_c = np.ones(L, dtype=bool)
            non_c[center] = False
            unaff_mask = (a == 0) & non_c
            unaff = float(np.max(c[unaff_mask])) if unaff_mask.any() else 0.0
            aff_mask = (np.abs(a) > 1e-14) & non_c
            t_lo = max(unaff, 0.0)
            t_hi = max(cur_max, qconst + hp * (1.0 - hp)) + 1e-10
            for _ in range(24):
                t_mid = 0.5 * (t_lo + t_hi)
                if t_mid < unaff:
                    t_lo = t_mid
                    continue
                pos = aff_mask & (a > 1e-14)
                neg = aff_mask & (a < -1e-14)
                x_hi = 1.0
                x_lo = 0.0
                if pos.any():
                    x_hi = min(x_hi, float(np.min(hp + (t_mid - c[pos]) / a[pos])))
                if neg.any():
                    x_lo = max(x_lo, float(np.max(hp + (t_mid - c[neg]) / a[neg])))
                if x_lo > x_hi + 1e-13:
                    t_lo = t_mid
                    continue
                g = t_mid - qconst
                if g < -1e-13:
                    t_lo = t_mid
                    continue
                if g < 0.25:
                    disc = max(0.0, 1.0 - 4.0 * g)
                    sd = np.sqrt(disc)
                    r1 = 0.5 * (1.0 - sd)
                    r2 = 0.5 * (1.0 + sd)
                    L1 = max(0.0, x_lo)
                    U1 = min(r1, x_hi)
                    L2 = max(r2, x_lo)
                    U2 = min(1.0, x_hi)
                    if not (L1 <= U1 + 1e-13 or L2 <= U2 + 1e-13):
                        t_lo = t_mid
                        continue
                t_hi = t_mid
            if t_hi >= cur_max - 1e-13:
                continue
            g = t_hi - qconst
            cands = [0.0, 1.0, hp, 0.5]
            if g >= 0 and g < 0.25:
                sd = np.sqrt(max(0.0, 1.0 - 4.0 * g))
                cands.extend([0.5 - 0.5 * sd, 0.5 + 0.5 * sd])
            if aff_mask.any():
                pos2 = aff_mask & (a > 1e-14)
                neg2 = aff_mask & (a < -1e-14)
                if pos2.any():
                    cands.append(float(np.min(hp + (t_hi - c[pos2]) / a[pos2])))
                if neg2.any():
                    cands.append(float(np.max(hp + (t_hi - c[neg2]) / a[neg2])))
            best_x = hp
            best_new = cur_max
            for xc in cands:
                xc = float(np.clip(xc, 0.0, 1.0))
                new_center = qconst + xc * (1.0 - xc)
                aff_new = float(np.max(c[aff_mask] + a[aff_mask] * (xc - hp))) if aff_mask.any() else 0.0
                nm = max(aff_new, new_center, unaff)
                if nm < best_new - 1e-14:
                    best_new = nm
                    best_x = xc
            if best_new < cur_max - 1e-13:
                dxv = best_x - hp
                c[aff_mask] += a[aff_mask] * dxv
                c[center] = qconst + best_x * (1.0 - best_x)
                h[p] = best_x
                cur_max = best_new
                improved = True
        if (pss + 1) % proj_every == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            cur_max = float(c.max())
        cval = (2.0 / n) * cur_max
        if cval < best_c - 1e-14:
            best_c = cval
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def fourier_h(n, K, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    t = x - 1.0
    h = 0.5 * np.ones(n)
    for k in range(1, K + 1):
        amp = rng.uniform(-0.25, 0.25) / np.sqrt(k)
        h += amp * np.cos(k * np.pi * t)
    return project_feasible(h, n)

def triangle_h(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.clip(1.0 - np.abs(x - 1.0) * 2.0, 0.0, 1.0), n)

def optimize_start(h, n, deadline, cd_passes=10, proj_every=8):
    h = project_feasible(h, n)
    c0 = compute_c(h, n)
    h, c1 = cd_optimize(h, n, cd_passes, deadline, proj_every)
    if c1 < c0:
        h, c0 = h, c1
    rem = deadline - time.time()
    if rem > 15:
        h2, c2 = cd_optimize(h, n, max(2, int(rem / max(n, 1) * 5)), deadline, 20)
        if c2 < c0:
            h, c0 = h2, c2
    return h, c0

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 10.0
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = triangle_h(500)
    n_init = len(h_init)
    if n_init % 2:
        n_init += 1
        h_init = resample_h(h_init, n_init)
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c(best_h, n_init)
    # Phase 1: optimize from initial
    best_h, best_c = optimize_start(best_h, n_init, deadline,
                                    cd_passes=12, proj_every=6)
    # Phase 2: multi-resolution
    for nt in [200, 300, 400, 500, 600]:
        if time.time() > deadline - 150:
            break
        h0 = resample_h(best_h, nt)
        h1, c1 = optimize_start(h0, nt, deadline,
                                cd_passes=max(4, int(300 / nt * 3)), proj_every=10)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 3: diverse restarts
    for r in range(40):
        if time.time() > deadline - 80:
            break
        if best_c <= 0.3806:
            break
        typ = r % 5
        nt = int(np.random.choice([250, 300, 400, 500, 600]))
        if typ == 0:
            h0 = resample_h(best_h, nt)
            h0 = project_feasible(h0 + rng.normal(0, 0.03, nt), nt)
        elif typ == 1:
            h0 = fourier_h(nt, rng.randint(4, 12), rng)
        elif typ == 2:
            h0 = np.zeros(nt)
            idx = rng.choice(nt, nt // 2, replace=False)
            h0[idx] = 1.0
        elif typ == 3:
            h0 = triangle_h(nt)
        else:
            base = resample_h(best_h, nt)
            h0 = project_feasible(base * rng.uniform(0.7, 1.3, nt), nt)
        h1, c1 = optimize_start(h0, nt, deadline,
                                cd_passes=max(3, int(200 / nt * 3)), proj_every=12)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 4: final polish on best
    if time.time() < deadline - 20:
        best_h, best_c = cd_optimize(best_h, len(best_h),
                                     max(3, int((deadline - time.time()) / max(len(best_h), 1) * 3)),
                                     deadline, proj_every=15)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c(best_h, len(best_h))
    return best_h, best_c, len(best_h)
