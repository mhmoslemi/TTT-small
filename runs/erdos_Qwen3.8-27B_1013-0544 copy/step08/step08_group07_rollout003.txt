import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    if alpha < 10:
        return m * (2.0 / n)
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alphas, maxiter_each=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 20:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18, 'gtol': 1e-13})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgrad_descent(h, n, n_iters=5000, deadline=None, lr0=0.05):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        kmax = int(np.argmax(c))
        L = abs(kmax - (n - 1))
        if L == 0:
            break
        grad = np.zeros(n)
        idx_ge = np.arange(L, n)
        grad[idx_ge] += (1.0 - h[idx_ge - L])
        idx_lt = np.arange(0, n - L)
        grad[idx_lt] -= h[idx_lt + L]
        lr = lr0 / (1.0 + it * 0.002)
        h = project_feasible(h - lr * grad, n)
        c_val = compute_c5(h, n)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
    return best_h, best_c

def coord_cd(h, n, max_passes=500, deadline=None, n_grid=24):
    h = h.copy()
    best_c = compute_c5(h, n)
    full_len = 2 * n - 1
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c_prev = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c_prev.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m1_lo = max(0, n - 1 - p)
            m1_hi = min(full_len - 1, 2 * n - 2 - p)
            m2_lo = max(0, p)
            m2_hi = min(full_len - 1, p + n - 1)
            m_lo = min(m1_lo, m2_lo)
            m_hi = max(m1_hi, m2_hi)
            ms = np.arange(m_lo, m_hi + 1)
            j1 = p + ms - (n - 1)
            valid1 = (j1 >= 0) & (j1 < n)
            j2 = p - ms + (n - 1)
            valid2 = (j2 >= 0) & (j2 < n)
            a = np.zeros(len(ms))
            if np.any(valid1):
                a[valid1] += (1.0 - h[j1[valid1]])
            if np.any(valid2):
                a[valid2] -= h[j2[valid2]]
            diag_mask = (ms == (n - 1))
            unaff_max = 0.0

                    if val < best_val:
                        best_val = val
                        best_d = d
            if abs(best_d) > 1e-14 and best_val < current_max - 1e-15:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                c_aff_new = c_aff + best_d * a
                if np.any(diag_mask):
                    c_aff_new = c_aff_new.copy()
                    c_aff_new[diag_mask] -= best_d * best_d
                c_prev[ms] = c_aff_new
        if not improved:
            break
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
    return h, compute_c5(h, n)

def block_subgrad(h, n, n_iters=3000, deadline=None):
    """Subgradient with random block averaging for more stable steps."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(0)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        grad = np.zeros(n)
        # Weight by top few peaks
        top_k = min(5, len(c))
        top_indices = np.argsort(c)[-top_k:]
        weights = c[top_indices] - c[top_indices[0]] + 1.0
        for ki in top_indices:
            L = abs(ki - (n - 1))
            if L == 0 or L >= n:
                continue
            g = np.zeros(n)
            g[L:] += (1.0 - h[:n-L])
            g[:n-L] -= h[L:]
            grad += weights[top_indices.tolist().index(ki)] * g
        lr = 0.03 / (1.0 + it * 0.001)
        h = project_feasible(h - lr * grad, n)
        c_val = compute_c5(h, n)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
    return best_h, best_c
