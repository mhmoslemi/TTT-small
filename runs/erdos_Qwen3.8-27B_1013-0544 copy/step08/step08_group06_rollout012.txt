import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_cd(h, n, max_passes=5000, deadline=None, grid_size=100):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        idx_order = range(n) if p % 2 == 0 else range(n - 1, -1, -1)
        improved = False
        for idx in idx_order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-10:
                lo_d, hi_d = 0.0, 1.0
            elif h0 > 1.0 - 1e-10:
                continue
            else:
                lo_d, hi_d = -h0, 1.0 - h0
            if hi_d - lo_d < 1e-12:
                continue
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            M = len(ms)
            a = np.zeros(M)
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            has_diag = (n - 1) in ms
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if has_diag:
                diag_pos = np.where(ms == (n - 1))[0]
                nv[:, diag_pos] -= d_grid[:, None] ** 2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            best_i = int(np.argmin(vals))
            best_d = d_grid[best_i]
            best_val = float(vals[best_i])
            if 0 < best_i < grid_size - 1:
                y = vals[best_i - 1:best_i + 2]
                x = d_grid[best_i - 1:best_i + 2]
                denom = (y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2] * (x[0] - x[1]))
                if abs(denom) > 1e-15:
                    xv = 0.5 * (y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2] * (x[0] - x[1])) / denom
                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv * a
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag_pos] -= xv ** 2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val = v1
                            best_d = xv
            if abs(best_d) > 1e-10 and best_val < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[ms[diag_pos]] -= best_d ** 2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True
        if not improved:
            break
        if (p + 1) % 200 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def pair_cd(h, n, max_passes=200, deadline=None, grid_size=15):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for idx in range(0, n - 1, 2):
            if deadline is not None and time.time() > deadline:
                break
            idx2 = idx + 1
            h0a, h0b = h[idx], h[idx2]
            if h0a < 1e-10 and h0b < 1e-10:
                continue
            if h0a > 1 - 1e-10 and h0b > 1 - 1e-10:
                continue
            ga = np.linspace(-h0a if h0a > 0 else 0.0, 1.0 - h0a if h0a < 1 else 1.0, grid_size)
            gb = np.linspace(-h0b if h0b > 0 else 0.0, 1.0 - h0b if h0b < 1 else 1.0, grid_size)
            best_val = current_c5
            best_da, best_db = 0.0, 0.0
            for da in ga:
                for db in gb:
                    if abs(da) < 1e-12 and abs(db) < 1e-12:
                        continue
                    na, nb = h0a + da, h0b + db
                    if na < -1e-12 or na > 1 + 1e-12 or nb < -1e-12 or nb > 1 + 1e-12:
                        continue
                    c_new = c.copy()
                    for ii, dd in ((idx, da), (idx2, db)):
                        m1 = max(0, min(n - 1 - ii, ii))
                        m2 = min(2 * n - 2, max(2 * n - 2 - ii, ii + n - 1))
                        ms = np.arange(m1, m2 + 1)
                        aa = np.zeros(len(ms))
                        jh = ii + ms - (n - 1)
                        vh = (jh >= 0) & (jh < n)
                        aa[vh] += (1.0 - h[jh[vh]])
                        jg = ii - ms + (n - 1)
                        vg = (jg >= 0) & (jg < n)
                        aa[vg] -= h[jg[vg]]
                        c_new[ms] += dd * aa
                        dm = (ms == (n - 1))
                        if np.any(dm):
                            c_new[ms[dm]] -= dd * dd
                    vm = float(c_new.max())
                    if vm < best_val - 1e-13:
                        best_val = vm
                        best_da, best_db = da, db
            if abs(best_da) > 1e-10 or abs(best_db) > 1e-10:
                h[idx] = float(np.clip(h0a + best_da, 0.0, 1.0))
                h[idx2] = float(np.clip(h0b + best_db, 0.0, 1.0))
                c = np.correlate(h, 1.0 - h, mode='full')
                current_c5 = float(c.max())
                improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
