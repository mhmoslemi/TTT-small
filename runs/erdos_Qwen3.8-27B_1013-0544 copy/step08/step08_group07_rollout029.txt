def smooth_max_with_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')

    mx = shifted.max()
    exps = np.exp(shifted - mx)
    wsum = exps.sum()
    weights = exps / wsum
    lse = mx + np.log(wsum)
    val = (m + lse / alpha) * dx
    grad = np.zeros(n)
    mo = n - 1
    for idx in range(2 * n - 1):

            grad[:n - k] -= w * x[k:]
        else:
            kk = -k
            grad[:n - kk] += w * (1.0 - x[kk:])
            grad[kk:] -= w * x[:n - kk]

        h = 0.5 + 0.4 * np.sin(2 * np.pi * x)

        h = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.2 * np.sin(4 * np.pi * x)
    elif kind == 2:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * x * 1.5)
    elif kind == 3:
        h = (np.sin(2 * np.pi * x) > 0).astype(float)
    else:
        h = rng.uniform(0, 1, n)
    return proj(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = structured_start(n, 0, rng)
    best_h = proj(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: L-BFGS-B with annealing
    h1, c1 = lbfgs_opt(best_h, n, [200, 1000, 5000, 20000, 100000, 500000, 2000000], 600, deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Coordinate descent
    h2, c2 = coord_cd(best_h, n, 250, deadline, grid_size=100)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Multi-scale
    x_orig = np.linspace(0, 2, n, endpoint=False)
    for n_s in [150, 200, 250, 350]:
        if time.time() > deadline - 80:
            break
        x_s = np.linspace(0, 2, n_s, endpoint=False)
        h_s = proj(np.interp(x_s, x_orig, best_h) + 0.01 * rng.randn(n_s), n_s)
        h_s, c_s = lbfgs_opt(h_s, n_s, [500, 5000, 50000, 500000], 400, deadline)
        h_s, c_s = coord_cd(h_s, n_s, 150, deadline, grid_size=80)
        x_b = np.linspace(0, 2, n, endpoint=False)
        h_b = proj(np.interp(x_b, x_s, h_s), n)
        h_b, c_b = coord_cd(h_b, n, 100, deadline, grid_size=80)
        if c_b < best_c:
            best_h, best_c = h_b, c_b
    # Phase 4: High-alpha polish
    if time.time() < deadline - 40:
        hf, cf = lbfgs_opt(best_h, n, [500000, 2000000, 10000000], 800, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = coord_cd(best_h, n, 300, deadline, grid_size=150)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Phase 5: Final coordinate descent
    if time.time() < deadline - 3:
        best_h, best_c = coord_cd(best_h, n, 150, deadline, grid_size=200)
    best_h = proj(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
