def coord_cd(h, n, max_passes, deadline, grid, refine):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    A = np.zeros(L)
    xs0 = np.linspace(0.0, 1.0, grid)
    tol = 1e-12
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = float(x[p])
            A.fill(0.0)
            A[p:p + n] += om
            st = diag - p
            A[st:st + n] -= x
            cur = float(c.max())
            vals = c[None, :] + (xs0 - old)[:, None] * A[None, :]
            vals[:, n - 1] -= (xs0 - old) * (xs0 - old)
            i = int(np.argmin(vals.max(axis=1)))
            bx = float(xs0[i])
            bv = float(vals[i].max())
            if bv < cur - tol:
                sp = 1.0 / (grid - 1)
                lo = max(0.0, bx - sp)
                hi = min(1.0, bx + sp)
                xx = np.linspace(lo, hi, refine + 1)
                mm = c[None, :] + (xx - old)[:, None] * A[None, :]
                mm[:, n - 1] -= (xx - old) * (xx - old)
                j = int(np.argmin(mm.max(axis=1)))
                if mm[j].max() < bv - tol:
                    bx = float(xx[j])
                    bv = float(mm[j].max())
                if lo < bx < hi:
                    sp2 = (hi - lo) / float(refine)
                    lo2 = max(0.0, bx - sp2)
                    hi2 = min(1.0, bx + sp2)
                    yy = np.linspace(lo2, hi2, 9)
                    ww = c[None, :] + (yy - old)[:, None] * A[None, :]
                    ww[:, n - 1] -= (yy - old) * (yy - old)
                    k = int(np.argmin(ww.max(axis=1)))
                    if ww[k].max() < bv - tol:
                        bx = float(yy[k])
                        bv = float(ww[k].max())
                if bv < cur - tol:
                    d = bx - old
                    c[p:p + n] += d * om
                    c[st:st + n] -= d * x
                    c[diag] -= d * d
                    x[p] = bx
                    om[diag - p] = 1.0 - bx
                    improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - tol:
            best = xp
            bc = cp
        if not improved:
            break
    return best, bc

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(55):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    y = np.clip(y, 0.0, 1.0)
    return y

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def binarize(h, n):
    if n % 2:
        return None
    target = n // 2
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx

def smooth_obj(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]

def sum_con(x, n):
    return x.sum() - 0.5 * n

def sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)

def slsqp_polish(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n)
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a), jac=smooth_grad,
                          method="SLSQP", bounds=bounds, constraints=cons,
                          options={"maxiter": int(maxiter), "ftol": 1e-12})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-13:
            best = nh
            bc = nc
    return best, bc

def coord_cd(h, n, max_passes, deadline, grid=32):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    A = np.zeros(L)
    xs0 = np.linspace(0.0, 1.0, grid)
    tol = 1e-12
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = float(x[p])
            A.fill(0.0)
            A[p:p + n] += om
            st = diag - p
            A[st:st + n] -= x
            cur = float(c.max())
            d = xs0 - old
            vals = c[None, :] + d[:, None] * A[None, :]
            vals[:, diag] -= d * d
            rowmax = vals.max(axis=1)
            i = int(np.argmin(rowmax))
            bx = float(xs0[i])
            if rowmax[i] < cur - tol:
                d = bx - old
                c[p:p + n] += d * om
                c[st:st + n] -= d * x
                c[diag] -= d * d
                x[p] = bx
                om[diag - p] = 1.0 - bx
                improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - tol:
            best = xp
            bc = cp
        if not improved:
            break
    return best, bc

def block_swap_move(x, n, rng, L):
    """Generate a block swap move: swap two non-overlapping blocks of length L."""
    if 2 * L > n:
        return None, None
    a = rng.randint(0, n - L + 1)
    remaining = list(range(0, a)) + list(range(a + L, n - L + 1))
    if not remaining:
        return None, None
    b = remaining[rng.randint(len(remaining))]
    if a > b:
        a, b = b, a
    if b < a + L:
        return None, None
    return a, b
