def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            cap = 1.0 - h
            tot = float(cap.sum())
            if tot > 0:
                h += (diff / tot) * cap
        else:
            tot = float(h.sum())

    c = np.correlate(h, 1.0 - h, mode='full')

        h_new = (1.0 - step) * h + step * v
        h_new = project_feasible(h_new, n)
        c_new = compute_c(h_new, n)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
        h = h_new
    return best_h, best_c

        if neg.any():

                if -1e-6 <= v <= 1.0 + 1e-6:
                    cands.append(float(np.clip(v, 0, 1)))
    xg = np.linspace(0, 1, 80)
    if m:
        vals = np.maximum(np.maximum(qconst + xg * (1 - xg), unaff),
                          np.max(cseg[None, :] + a[None, :] * (xg[:, None] - hp), axis=1))
    else:
        vals = np.maximum(qconst + xg * (1 - xg), unaff)
    vals = np.maximum(vals, unaff)
    idx = int(np.argmin(vals))
    cands.append(float(xg[idx]))
    best_x, best_f = hp, f_at(hp)
    for x in cands:
        x = float(np.clip(x, 0, 1))
        f = f_at(x)
        if f < best_f - 1e-13:
            best_f = f
            best_x = x
    if idx > 0 and idx < 79:
        xa, xb = float(xg[idx-1]), float(xg[idx+1])
        xg2 = np.linspace(max(0, xa - 0.01), min(1, xb + 0.01), 40)
        if m:
            v2 = np.maximum(np.maximum(qconst + xg2*(1-xg2), unaff),
                            np.max(cseg[None,:] + a[None,:]*(xg2[:,None]-hp), axis=1))
        else:
            v2 = np.maximum(qconst + xg2*(1-xg2), unaff)
        idx2 = int(np.argmin(v2))
        x2 = float(xg2[idx2])
        f2 = f_at(x2)
        if f2 < best_f - 1e-13:
            best_f = f2
            best_x = x2
    return best_x, best_f

def cd_optimize(h, n, max_passes, deadline, rng):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(c.max())
    best_h = h.copy()
    best_c = (2.0 / n) * cur_max
    for p in range(max_passes):
        if time.time() > deadline:
            break
        h, c, cur_max, improved = cd_pass(h, n, rng, c, cur_max)
        if p % 6 == 5:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            cur_max = float(c.max())
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        cval = (2.0 / n) * cur_max
        if cval < best_c - 1e-14:
            best_c = cval
            best_h = h.copy()
        if improved == 0:
            break
    return best_h, best_c

def full_optimize(h, n, deadline, rng, use_slsqp=True, cd_passes=10):
    h = project_feasible(h, n)
    c_val = compute_c(h, n)
    if use_slsqp and n <= 600 and time.time() < deadline - 120:
        for alpha in [30, 300, 3000]:
            if time.time() > deadline - 60:
                break
            h2, c2 = slsqp_opt(h, n, alpha, 60, deadline)
            if c2 < c_val:
                h, c_val = h2, c2
    if time.time() < deadline - 30:
        h2, c2 = cd_optimize(h, n, cd_passes, deadline, rng)
        if c2 < c_val:
            h, c_val = h2, c2
    if use_slsqp and n <= 600 and time.time() < deadline - 80:
        h3, c3 = slsqp_opt(h, n, 5000, 40, deadline)
        if c3 < c_val:
            h, c_val = h3, c3
    if time.time() < deadline - 20:
        h4, c4 = cd_optimize(h, n, 15, deadline, rng)
        if c4 < c_val:
            h, c_val = h4, c2 if False else c4
            h, c_val = h4, c4
    return h, c_val

def basis_h(n, rng, n_modes=12):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    t = x - 1.0
    h = 0.5 * np.ones(n)
    for k in range(1, n_modes + 1):
        amp = rng.uniform(-0.35, 0.35) / (0.5 + 0.3 * k)
        h += amp * np.cos(k * np.pi * t)
        amp2 = rng.uniform(-0.2, 0.2) / (0.5 + 0.3 * k)
        h += amp2 * np.sin(k * np.pi * t)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = basis_h(500, rng)
    n = len(h_init)
    if n % 2:
        n += 1
        x0 = (np.arange(len(h_init)) + 0.5) * (2.0 / len(h_init))
        x1 = (np.arange(n) + 0.5) * (2.0 / n)
        h_init = project_feasible(np.interp(x1, x0, h_init), n)
    best_h = project_feasible(h_init, n)
    best_c = compute_c(best_h, n)
    # Phase 1: SLSQP + CD on initial
    best_h, best_c = full_optimize(best_h, n, deadline, rng, use_slsqp=True, cd_passes=12)
    # Phase 2: Symmetry enforcement + re-optimize
    if time.time() < deadline - 200:
        hs = symmetrize(best_h, n)
        cs = compute_c(hs, n)
        if cs < best_c:
            best_c = cs
        h2, c2 = full_optimize(hs, n, deadline, rng, use_slsqp=True, cd_passes=15)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Multi-restart
    for r in range(30):
        if time.time() > deadline - 100:
            break
        if best_c <= 0.38075:
            break
        typ = r % 5
        if typ == 0:
            h0 = project_feasible(best_h + rng.normal(0, 0.015, n), n)
        elif typ == 1:
            h0 = basis_h(n, rng, n_modes=rng.randint(6, 16))
        elif typ == 2:
            h0 = symmetrize(best_h + rng.normal(0, 0.03, n), n)
        elif typ == 3:
            h0 = project_feasible(best_h + rng.normal(0, 0.05, n), n)
        else:
            k = n // 2
            base = np.zeros(n)
            idx = rng.choice(n, k, replace=False)
            base[idx] = 1.0
            h0 = project_feasible(base * 0.7 + best_h * 0.3, n)
        use_s = (r % 2 == 0) and time.time() < deadline - 150
        passes = 8 if use_s else 5
        h1, c1 = full_optimize(h0, n, deadline, rng, use_slsqp=use_s, cd_passes=passes)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 4: Final heavy polish
    if time.time() < deadline - 25:
        best_h, best_c = cd_optimize(best_h, n, 40, deadline, rng)
    # Phase 5: Symmetry polish
    if time.time() < deadline - 20:
        hs = symmetrize(best_h, n)
        cs = compute_c(hs, n)
        if cs < best_c:
            best_h = hs
            best_c = cs
            best_h, best_c = cd_optimize(best_h, n, 20, deadline, rng)
    best_h = project_feasible(best_h, n)
    best_c = compute_c(best_h, n)
    return best_h, best_c, n
