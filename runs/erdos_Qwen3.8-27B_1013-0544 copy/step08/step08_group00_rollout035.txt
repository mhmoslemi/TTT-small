def coord_descent(h, n, max_passes, deadline, worst_first=True):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = 0
        order = list(range(n))
        if worst_first:
            max_lags = np.where(c >= c.max() - 0.5 * (2.0 / n))[0]
            involved = set()
            for m in max_lags:
                lo = max(0, m - (n - 1))
                hi = min(n - 1, m)
                for p in range(lo, hi + 1):
                    involved.add(p)
            involved_list = sorted(involved, key=lambda p: -c[max(0, min(2*n-2, n-1+abs(p - (n-1))))])
            rest = [p for p in order if p not in involved]
            order = involved_list + rest
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord_exact(h, c, p, n)
            if abs(x_new - h0) > 1e-10:
                h[p] = x_new
                improved += 1
        if improved == 0:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_optimize(h, n, alphas, maxiter_each, deadline):
    def obj(x, alpha):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * (2.0 / n)
    def sc(x):
        return x.sum() - n / 2.0
    def scj(x):
        return np.ones(n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
        try:
            res = minimize(obj, best_h, args=(alpha,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
