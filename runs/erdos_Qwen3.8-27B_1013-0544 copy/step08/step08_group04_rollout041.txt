def project(h, n=0):
    h = np.clip(np.asarray(h, float), 0.0, 1.0)
    n = len(h)
    d = 0.5 * n - h.sum()
    if d > 0.0:
        cap = 1.0 - h
        s = cap.sum()
        if s > 0.0:
            h += (d / s) * cap
    elif d < 0.0:
        s = h.sum()
        if s > 0.0:
            h += (d / s) * h
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - h.sum()
    if d > 1e-12:
        i = np.flatnonzero(h < 1.0)
        if i.size:
            h[i[0]] = min(1.0, h[i[0]] + d)
    elif d < -1e-12:
        i = np.flatnonzero(h > 0.0)
        if i.size:
            h[i[0]] = max(0.0, h[i[0]] + d)
    return h

def calc(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def resample(h, n):
    n0 = len(h)
    if n0 == n:

    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project(np.interp(x1, x0, h), n)

    return st

import numpy as np
import time

def project(h, n=0):
    h = np.clip(np.asarray(h, float), 0.0, 1.0)
    n = len(h)
    d = 0.5 * n - h.sum()
    if d > 0.0:
        s = (1.0 - h).sum()
        if s > 0.0:
            h += (d / s) * (1.0 - h)
    elif d < 0.0:
        s = h.sum()
        if s > 0.0:
            h += (d / s) * h
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - h.sum()
    if d > 1e-12:
        i = np.flatnonzero(h < 1.0)
        if i.size:
            h[i[0]] = min(1.0, h[i[0]] + d)
    elif d < -1e-12:
        i = np.flatnonzero(h > 0.0)
        if i.size:
            h[i[0]] = max(0.0, h[i[0]] + d)
    return h

def calc(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def resample(h, n):
    n0 = len(h)
    if n0 == n:
        return project(h, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project(np.interp(x1, x0, h), n)

def default_h(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    t = np.abs(x - 1.0)
    return project(0.5 + 0.3 * (t - t.mean()), n)

def sym(h):
    return project(0.5 * (h + h[::-1]), len(h))

def binfrom(h):
    n = len(h)
    k = n // 2
    z = np.zeros(n)
    z[np.argpartition(h, n - k)[n - k:]] = 1.0
    return z

def basis_h(n, rng, K=7, amp=0.32, even=False):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    t = x - 1.0
    h = 0.5 * np.ones(n)
    for k in range(1, K + 1):
        h += rng.uniform(-amp / k, amp / k) * np.cos(k * np.pi * t)
        if not even:
            h += rng.uniform(-amp / k, amp / k) * np.sin(k * np.pi * t)
    return project(h, n)

def evalx(x, hp, c, a, q, u):
    v = float(np.max(c + a * (x - hp))) if a.size else u
    return max(u, q + x * (1.0 - x), v)

def opt_coord(hp, c, a, q, u, it=16):
    b = c - a * hp
    p = a > 1e-12
    ng = a < -1e-12
    z = b[~(p | ng)]
    zmax = float(np.max(z)) if z.size else -1e100
    cm = float(np.max(c)) if c.size else 0.0
    low = max(0.0, u, q, zmax)
    high = max(low + 1e-11, u, q, q + hp * (1.0 - hp), cm)
    high = max(high, zmax) + 1e-8
    ap = a[p]
    an = a[ng]
    bp = b[p]
    bn = b[ng]
    for _ in range(it):
        if high - low < 1e-9:
            break
        mid = 0.5 * (low + high)
        ok = mid >= u - 1e-10 and mid >= q - 1e-10 and mid >= zmax - 1e-10
        ll = 0.0
        lh = 1.0
        if ok:
            if an.size:
                ll = max(ll, float(np.max((mid - bn) / an)))
            if ap.size:
                lh = min(lh, float(np.min((mid - bp) / ap)))
            if not (ll <= 1.0 + 1e-10 and lh >= -1e-10):
                ok = False
            else:
                g = mid - q
                if g > 0.25:
                    ok = max(0.0, ll) <= min(1.0, lh) + 1e-10
                elif g > -1e-12:
                    g = max(0.0, g)
                    r = 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                    r1 = 0.5 - r
                    r2 = 0.5 + r
                    ok = (max(0.0, ll) <= min(r1, lh) + 1e-10) or (max(r2, ll) <= min(1.0, lh) + 1e-10)
                else:
                    ok = False
        high, low = (mid, low) if ok else (high, mid)
    T = high
    cands = [0.0, 1.0, hp, 0.5]
    if T >= u - 1e-10 and T >= q - 1e-10 and T >= zmax - 1e-10:
        ll = 0.0
        lh = 1.0
        if an.size:
            ll = max(ll, float(np.max((T - bn) / an)))
        if ap.size:
            lh = min(lh, float(np.min((T - bp) / ap)))
        if ll <= 1.0 + 1e-10 and lh >= -1e-10:
            g = T - q
            br = []
            if g > 0.25:
                br = [(max(0.0, ll), min(1.0, lh))]
            elif g >= -1e-12:
                g = max(0.0, g)
                r = 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                br = [(0.0, 0.5 - r), (0.5 + r, 1.0)]
            for A, B in br:
                C = max(A, ll, 0.0)
                D = min(B, lh, 1.0)
                if C <= D + 1e-10:
                    cands.extend((C, D, 0.5 * (C + D)))
    bx = hp
    bf = evalx(bx, hp, c, a, q, u)
    for x in cands:
        if 0.0 <= x <= 1.0:
            f = evalx(float(x), hp, c, a, q, u)
            if f < bf - 1e-13:
                bf = f
                bx = float(x)
    return bx, bf

def cdc(h, n, deadline, passes=8, pe=8, temp=0.0, rng=None):
    h = project(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best = h.copy()
    bcur = cur
    ct = n - 1
    L = 2 * n - 1
    if rng is None:
        rng = np.random.RandomState(12345)
    stall = 0
    for it in range(passes):
        if time.time() > deadline:
            break
        if it % 2 == 0:
            order = np.arange(n)
        elif it % 3 == 0:
            order = np.arange(n - 1, -1, -1)
        else:
            order = rng.permutation(n)
        imp = False
        for k, p in enumerate(order):
            if time.time() > deadline:
                break
            hp = float(h[p])
            lo = min(p, ct - p)
            hi = max(ct + p, 2 * ct - p)
            s = ct - np.arange(lo, hi + 1, 1)
            j = p + s
            i = p - s
            a = np.zeros(hi - lo + 1)
            vj = (j >= 0) & (j < n)
            if np.any(vj):
                a[vj] += 1.0 - h[j[vj]]
            vi = (i >= 0) & (i < n)
            if np.any(vi):
                a[vi] -= h[i[vi]]
            off = ct - lo
            cs = np.empty(hi - lo)
            as_ = np.empty(hi - lo)
            if off:
                cs[:off] = c[lo:ct]
                as_[:off] = a[:off]
            if hi > ct:
                cs[off:] = c[ct + 1:hi + 1]
                as_[off:] = a[off + 1:]
            q = c[ct] - hp * (1.0 - hp)
            u = 0.0
            if lo:
                u = max(u, float(c[:lo].max()))
            if hi + 1 < L:
                u = max(u, float(c[hi + 1:].max()))
            bx, bf = opt_coord(hp, cs, as_, q, u, 16)
            if bf < cur - 1e-12:
                dx = bx - hp
                if off:
                    c[lo:ct] += as_[:off] * dx
                if hi > ct:
                    c[ct + 1:hi + 1] += as_[off:] * dx
                c[ct] = q + bx * (1.0 - bx)
                h[p] = bx
                cur = bf
                imp = True
            elif temp > 0.0:
                candx = hp
                candf = cur + temp
                for xx in (0.0, 1.0, 1.0 - hp, hp + rng.uniform(-0.25, 0.25), rng.uniform(0.0, 1.0)):
                    xx = float(np.clip(xx, 0.0, 1.0))
                    if abs(xx - hp) > 1e-8:
                        ff = evalx(xx, hp, cs, as_, q, u)
                        if ff < candf:
                            candf = ff
                            candx = xx
                if candf < cur + temp and abs(candx - hp) > 1e-8:
                    dx = candx - hp
                    if off:
                        c[lo:ct] += as_[:off] * dx
                    if hi > ct:
                        c[ct + 1:hi + 1] += as_[off:] * dx
                    c[ct] = q + candx * (1.0 - candx)
                    h[p] = candx
                    cur = candf
                    imp = True
            if (k + 1) % pe == 0:
                h = project(h, n)
                c = np.correlate(h, 1.0 - h, mode='full')
                cur = float(c.max())
                if cur < bcur:
                    bcur = cur
                    best = h.copy()
        h = project(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        if cur < bcur:
            bcur = cur
            best = h.copy()
        if not imp:
            stall += 1
            if temp == 0.0 or stall >= 3:
                break
        else:
            stall = 0
        if temp > 0.0:
            temp *= 0.8
    h = project(best, n)
    return h, calc(h, n)

def pair(h, n, deadline, tries=80, rng=None):
    h = project(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best = h.copy()
    bcur = cur
    ct = n - 1
    if rng is None:
        rng = np.random.RandomState(987)
    for _ in range(tries):
        if time.time() > deadline:
            break
        mx = float(c.max())
        idx = np.flatnonzero(c >= mx - max(1e-6, 5e-5 * mx))
        if idx.size > 32:
            idx = idx[np.argsort(c[idx])[-32:]]
        m = idx[rng.randint(idx.size)]
        s = ct - int(m)
        if s >= 0:
            if n - s <= 0:
                continue
            i = int(rng.randint(n - s))
            j = i + s
        else:
            r = -s
            if n - r <= 0:
                continue
            i = r + int(rng.randint(n - r))
            j = i - r
        hi, hj = float(h[i]), float(h[j])
        d = hj - hi
        cands = []
        if abs(d) > 1e-5:
            cands = [d, 0.5 * d, -0.5 * d]
        cands += [0.02, -0.02, 0.05, -0.05]
        bh = None
        cc = None
        bc = cur
        for dd in cands:
            x = hi + dd
            y = hj - dd
            if x < 0.0 or x > 1.0 or y < 0.0 or y > 1.0 or abs(dd) < 1e-8:
                continue
            h2 = h.copy()
            h2[i] = x
            h2[j] = y
            c2 = np.correlate(h2, 1.0 - h2, mode='full')
            cm = float(c2.max())
            if cm < bc - 1e-12:
                bc = cm
                bh = h2
                cc = c2
        if bh is not None:
            h = bh
            c = cc
            cur = bc
            if cur < bcur:
                bcur = cur
                best = h.copy()
    return best, bcur * (2.0 / float(n))

def starts(n, best, rng):
    b0 = resample(best, n) if best is not None else default_h(n)
    st = [b0, binfrom(b0)]
    x = (np.arange(n) + 0.5) * (2.0 / n)
    t = np.abs(x - 1.0)
    for p in (1.0, 2.0, 3.0):
        q = t ** p
        st.append(project(0.5 + 0.45 * (q - q.mean()), n))
        st.append(project(0.5 - 0.30 * (q - q.mean()), n))
    st.append(basis_h(n, rng, 7, 0.32, False))
    st.append(basis_h(n, rng, 7, 0.32, True))
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    st.append(h)
    half = rng.uniform(0.0, 1.0, n // 2)
    st.append(project(np.concatenate((half, half[::-1])), n))
    return st

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 12.0
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, float).copy()
    except Exception:
        h = default_h(500)
    n = len(h)
    if n % 2:
        n += 1
        h = resample(h, n)
    if n > 900:
        h = resample(h, 800)
        n = 800
    h = project(h, n)
    bh = h
    bc = calc(h, n)
    bn = n

    lim = min(deadline, t0 + 240.0)
    if time.time() < lim:
        h1, c1 = cdc(bh, bn, lim, 24, 8, 0.0, rng)
        if c1 < bc:
            bh, bc, bn = h1, c1, bn
        h1 = sym(bh)
        c1 = calc(h1, bn)
        if c1 < bc:
            bh, bc = h1, c1
        if time.time() < deadline - 300.0:
            h1, c1 = pair(bh, bn, deadline - 300.0, 100, rng)
            if c1 < bc:
                bh, bc, bn = h1, c1, bn

    ml = deadline - 240.0
    ulist = []
    for nn in [bn, 128, 160, 256, 320, 400, 512, 640, 768]:
        nn = int(nn)
        if nn % 2:
            nn += 1
        if 80 <= nn <= 900 and nn not in ulist:
            ulist.append(nn)

    for n in ulist:
        if time.time() > ml:
            break
        for hs in starts(n, bh, rng):
            if time.time() > ml:
                break
            h = project(hs, n)
            c0 = calc(h, n)
            h2 = sym(h)
            c2 = calc(h2, n)
            if c2 < c0:
                h, c0 = h2, c2
            tlim = min(ml, time.time() + (22.0 if n <= 450 else 35.0))
            h1, c1 = cdc(h, n, tlim, 6 if n <= 450 else 4, 10 if n > 500 else 8, 0.0, rng)
            if c1 < c0:
                h, c0 = h1, c1
            h2 = sym(h)
            c2 = calc(h2, n)
            if c2 < c0:
                h, c0 = h2, c2
            if c0 < bc + 0.02 and time.time() < ml - 5.0:
                h1, c1 = pair(h, n, max(time.time() + 2.0, ml - 5.0), 45, rng)
                if c1 < c0:
                    h, c0 = h1, c1
            if c0 < bc:
                bh, bc, bn = h, c0, n

    pl = deadline - 120.0
    r = 0
    while time.time() < pl:
        r += 1
        n = bn
        h = bh.copy()
        typ = r % 6
        if typ == 0:
            h = project(h + rng.normal(0.0, 0.015, n), n)
        elif typ == 1:
            h = binfrom(h)
        elif typ == 2:
            for _ in range(8):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i != j:
                    h[i], h[j] = h[j], h[i]
        elif typ == 3:
            x = (np.arange(n) + 0.5) * (2.0 / n)
            t = x - 1.0
            h = project(h + rng.uniform(-0.025, 0.025) * np.cos(3.0 * np.pi * t)
                        + rng.uniform(-0.025, 0.025) * np.sin(2.0 * np.pi * t), n)
        elif typ == 4:
            h = sym(project(h + rng.normal(0.0, 0.01, n), n))
        else:
            n2 = 500 if bn < 500 else 320
            n = n2
            h = resample(bh, n)
        tlim = min(pl, time.time() + (25.0 if n <= 500 else 40.0))
        h1, c1 = cdc(h, n, tlim, 4, 8, 0.06 * (n / 500.0), rng)
        if c1 < bc:
            bh, bc, bn = h1, c1, n
        if time.time() < pl - 5.0:
            h1, c1 = pair(h1, n, max(time.time() + 1.0, pl - 5.0), 35, rng)
            if c1 < bc:
                bh, bc, bn = h1, c1, n

    if time.time() < deadline - 2.0:
        h1, c1 = cdc(bh, bn, deadline, 60, 5, 0.0, rng)
        if c1 < bc:
            bh, bc, bn = h1, c1, bn
    if time.time() < deadline - 1.0:
        h1, c1 = pair(bh, bn, deadline, 40, rng)
        if c1 < bc:
            bh, bc, bn = h1, c1, bn
    h1 = sym(bh)
    c1 = calc(h1, bn)
    if c1 < bc:
        bh, bc = h1, c1
    bh = project(bh, bn)
    bc = calc(bh, bn)
    return bh, bc, bn
