def project_simplex(h, n):

    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(55):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)

    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)


def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)


def grad_for_lag(h, shift, n):
    g = np.zeros(n)
    if shift >= 0:
        g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g


def active_subgrad(h, n, frac=0.15):
    c = np.correlate(h, 1.0 - h, mode="full")
    cm = float(c.max())
    diag = n - 1
    spread = cm - float(c.min())
    thr = cm - spread * frac
    active = np.where(c >= thr)[0]
    q = len(active)
    if q == 0:
        return h, cm
    g = np.zeros(n)
    for k in active:
        g += grad_for_lag(h, k - diag, n)
    g /= q
    g *= (2.0 / n)
    return g, cm

def slsqp_smooth(h, n, alphas, maxiter, deadline):
    dx = 2.0 / n
    L = 2 * n - 1
    diag = n - 1

    def smooth_val(x, alpha):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = float(c.max())
        r = alpha * (c - m)
        r -= r.max()
        return (m + np.log(float(np.exp(r).sum())) / alpha) * dx

    def smooth_jac(x, alpha):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = float(c.max())
        r = alpha * (c - m)
        r -= r.max()
        ee = np.exp(r)
        w = ee / float(ee.sum())
        A = np.convolve(1.0 - x, w, mode="full")
        B = np.convolve(x, w[::-1], mode="full")
        return (A - B)[diag:diag + n] * dx

    def sum_fun(x):
        return x.sum() - 0.5 * n

    def sum_jac(x):
        return np.ones(n)

    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_fun, "jac": sum_jac}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_val, best, args=(a,), jac=smooth_jac,
                           method="SLSQP", bounds=bounds, constraints=cons,

            nh = project_simplex(res.x, n)
            nc = compute_c5(nh, n)

    t3 = min(deadline - 200.0, t0 + 0.60 * budget_s)
    if time.time() < t3:
        h3, c3 = paired_cd(best_h, n_best, max_passes=8, deadline=t3, top_pairs=300, rng=rng)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Simple coordinate descent
    t4 = min(deadline - 150.0, t0 + 0.70 * budget_s)
    if time.time() < t4:
        h4, c4 = coord_simple(best_h, n_best, max_passes=4, deadline=t4)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Binary optimization
    t5 = min(deadline - 100.0, t0 + 0.85 * budget_s)
    if n_best % 2 == 0 and time.time() < t5:
        hb, cb = binary_swap_search(best_h, n_best, rng, max_rounds=5, deadline=t5)
        if cb < best_c:
            best_h, best_c = hb, cb
            # Polish after binary
            t5b = min(deadline - 50.0, time.time() + 60.0)
            h5, c5 = slsqp_smooth(best_h, n_best, [50000, 500000], 80, t5b)
            if c5 < best_c:
                best_h, best_c = h5, c5
            h5, c5 = paired_cd(best_h, n_best, max_passes=4, deadline=t5b, top_pairs=200, rng=rng)
            if c5 < best_c:
                best_h, best_c = h5, c5

    # Phase 6: Try transfer to other resolutions
    cands = [400, 600, 350, 550, 450]
    tried = [n_best]
    for nt in cands:
        if time.time() > deadline - 60.0:
            break
        if nt in tried or nt < 100 or nt > 1000 or nt % 2:
            continue
        tried.append(nt)
        ht = transfer_h(best_h, n_best, nt)
        cv = compute_c5(ht, nt)
        chunk = min(deadline - 40.0, time.time() + 50.0)
        ht, cv = paired_cd(ht, nt, max_passes=3, deadline=chunk, top_pairs=150, rng=rng)
        if cv < best_c:
            best_h, best_c, n_best = ht, cv, nt

    # Final polish
    tf = max(deadline - 2.0, time.time())
    if time.time() < tf:
        hf, cf = slsqp_smooth(best_h, n_best, [200000], 60, tf)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = paired_cd(best_h, n_best, max_passes=3, deadline=tf, top_pairs=100, rng=rng)
        if cf < best_c:
            best_h, best_c = hf, cf

    final_h = project_simplex(best_h, n_best)
    final_c = float(compute_c5(final_h, n_best))
    return final_h, final_c, int(n_best)
