import numpy as np
from scipy.optimize import minimize
import time


def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    d = t - h.sum()
    if abs(d) < 1e-13:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)


def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)


def lse_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    s = n - 1
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx


def _lf(x, n, alpha):
    return lse_val_grad(x, n, alpha)[0]


def _lj(x, n, alpha):
    return lse_val_grad(x, n, alpha)[1]


def _cf(x, n):
    return x.sum() - 0.5 * n


def _cj(x, n):
    return np.ones_like(x)


def smooth_optim(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n)
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _cf, "jac": _cj, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        for method in ("L-BFGS-B", "SLSQP"):
            if time.time() > deadline:
                break
            try:
                if method == "L-BFGS-B":

                nh = project_simplex(res.x, n)
                nc = compute_c5(nh, n)
                if nc < bc - 1e-13:
                    best, bc = nh, nc
            except Exception:

    return best, bc


def binarize_h(h, n):
    if n % 2:
        return None
    target = n // 2
    if target <= 0 or target >= n:
        return None
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b


def binary_trial(c, h, om, i, j, n):
    d = c.copy()
    d[i:i + n] -= om
    st = n - 1 - i
    d[st:st + n] += h
    d[j:j + n] += om
    st2 = n - 1 - j
    d[st2:st2 + n] -= h
    d[n - 1] -= 2.0
    d[n - 1 + i - j] += 1.0
    d[n - 1 + j - i] += 1.0
    return d, float(d.max())


def bkand(h, n, rng):

    j = int(np.argmin(h))
    for _ in range(6):
        i2 = int(rng.randint(n))

            i = int(rng.randint(n))
        j = int(rng.randint(n))
        while h[j] >= 0.5:
            j = int(rng.randint(n))
    return i, j


def binary_search(h_init, n, rng, deadline, max_restarts=8):

        return None, np.inf
    dx = 2.0 / n

                h[i] = 0.0

def coord_cd(h, n, max_passes=3, deadline=None, grid=24, refine=8):

    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1

            if deadline is not None and time.time() > deadline:
                break

            A[p:p + n] += om
            st = diag - p
            if st >= 0:
                A[st:st + n] -= x
            else:
                A[0:n + st] -= x[-st:]
                A[n + st:n + st + n] -= x[:-(n + st)] if n + st < n else x
            cur = float(c.max())
            d = xs0 - old

                cand2 = c[None, :] + d2[:, None] * A[None, :]

def transfer_h(h, n, n_tgt):
