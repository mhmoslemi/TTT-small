            if d_hi > r1 and d_hi < r2:
                d_lo = max(d_lo, r2)
            if d_lo > r1 and d_hi < r2:
                if d_lo - d_hi < 1e-12:
                    pass
                elif r1 - d_lo < d_hi - r2:
                    d_hi = r1
                else:
                    d_lo = r2
    if d_lo > d_hi + 1e-12:
        return h0
    return float(np.clip(h0 + (d_lo + d_hi) / 2.0, 0.0, 1.0))

def bc_pass(h, n, deadline):
    c = np.correlate(h, 1.0 - h, mode='full')
    improved = False
    for p in range(n):
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-15 or h0 > 1.0 - 1e-15:
            continue
        x_new = solve_1d(c, h, p, n)
        if abs(x_new - h0) > 1e-10:
            h[p] = x_new
            improved = True
    return h, improved

def cd_loop(h, n, max_passes, deadline):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        h, imp = bc_pass(h, n, deadline)
        if not imp:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def bc_pass_reversed(h, n, deadline):
    c = np.correlate(h, 1.0 - h, mode='full')
    improved = False
    for p in range(n - 1, -1, -1):
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-15 or h0 > 1.0 - 1e-15:
            continue
        x_new = solve_1d(c, h, p, n)
        if abs(x_new - h0) > 1e-10:
            h[p] = x_new
            improved = True
    return h, improved

def cd_loop_rev(h, n, max_passes, deadline):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        h, imp = bc_pass_reversed(h, n, deadline)
        if not imp:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def random_start(n, rng):
    h = rng.uniform(0, 1, n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_init = 400
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: Forward CD on initial
    h1, c1 = cd_loop(best_h, n_init, 300, deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1
    
    # Phase 2: Reverse CD
    h2, c2 = cd_loop_rev(best_h, n_init, 200, deadline)
    if c2 < best_c:
        best_h, best_c = h2, c1 if c1 < best_c else c2
        best_c = min(best_c, c2)
    
    # Phase 3: More forward CD (polishing)
    h3, c3 = cd_loop(best_h, n_init, 300, deadline)
    if c3 < best_c:
        best_h, best_c = h3, c3
    
    # Phase 4: Shake and re-optimize
    n = len(best_h)
    n_shakes = 6
    for s in range(n_shakes):
        if time.time() > deadline - 100:
            break
        sigma = 0.02 + 0.03 * (s / n_shakes)
        perturb = rng.normal(0, sigma, n)
        hp = project_feasible(best_h + perturb, n)
        hs, cs = cd_loop(hp, n, 150, deadline)
        if cs < best_c:
            best_h, best_c = hs, cs
            n = len(best_h)
    
    # Phase 5: Try different n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 500, 600]:
        if time.time() > deadline - 150:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        ho, co = cd_loop(h_int, n_try, 250, deadline)
        if co < best_c:
            best_h, best_c = ho, co
            x_orig = np.linspace(0, 2, n_try)
            n = n_try
    
    # Phase 6: Final alternating CD (forward + reverse) for high precision
    n = len(best_h)
    for _ in range(20):
        if time.time() > deadline - 10:
            break
        h, imp = bc_pass(best_h, n, deadline)
        h = project_feasible(h, n)
        h, imp2 = bc_pass_reversed(h, n, deadline)
        h = project_feasible(h, n)
        c_val = compute_c5(h, n)
        if c_val < best_c:
            best_h, best_c = h, c_val
        if not imp and not imp2:
            break
    
    # Phase 7: Boundary push heuristic
    if time.time() < deadline - 30:
        n = len(best_h)
        for trial in range(5):
            if time.time() > deadline - 15:
                break
            h_try = best_h.copy()
            # Pick a random subset to push to boundary
            n_push = max(5, n // 10)
            idx = rng.choice(n, size=n_push, replace=False)
            for i in idx:
                if h_try[i] < 0.5:
                    h_try[i] = 0.0
                else:
                    h_try[i] = 1.0
            h_try = project_feasible(h_try, n)
            hs, cs = cd_loop(h_try, n, 100, deadline)
            if cs < best_c:
                best_h, best_c = hs, cs
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
