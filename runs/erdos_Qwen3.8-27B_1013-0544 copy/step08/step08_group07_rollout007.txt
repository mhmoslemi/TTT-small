    lo, hi = -1.5, 1.5

def subgrad_single(h, n):
    """Subgradient using only the maximizing lag."""
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    s = k_star - (n - 1)  # shift value
    g = np.zeros(n)
    # g[p] = (1 - h[p-s]) - h[p+s] with boundary handling
    # p-s in [0,n) means p in [s, s+n), valid for p in [0,n)
    lo = max(0, s)

    if s >= 0:
        # p-s valid when p >= s
        g[s:] += (1.0 - h[:n-s])
        # p+s valid when p+s < n, i.e., p < n-s
        g[:n-s] -= h[s:]
    else:
        # p-s valid when p > s, i.e., p >= 1 (since s<0, p-s < p < n always, and p-s >= 0 when p >= -s)
        p_lo = -s
        g[p_lo:] += (1.0 - h[:n+s])
        # p+s valid when p+s in [0,n), i.e., p >= -s and p < n-s
        g[p_lo:n-s if n-s < n else n] -= h[s+p_lo:]
    return g

def subgrad_smooth(h, n, alpha):

    c = np.correlate(h, 1.0 - h, mode='full')
    shifted = c - c.max()
    w = np.exp(shifted / alpha)

def subgrad_at_lag(y, n, s):
    g = np.zeros(n)
    lo1 = max(0, s)
    hi1 = min(n, s + n)
    g[lo1:hi1] += (1.0 - y[lo1 - s:hi1 - s])
    lo2 = max(0, -s)
    hi2 = min(n, n - s)
    g[lo2:hi2] -= y[lo2 + s:hi2 + s]
    return g

def nesterov_opt(h, n, deadline, max_iter=50000, eta0=0.05):

def smooth_lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    lse = mx + np.log(np.sum(np.exp(sh - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_jac(x, n):
    return np.ones(n)
