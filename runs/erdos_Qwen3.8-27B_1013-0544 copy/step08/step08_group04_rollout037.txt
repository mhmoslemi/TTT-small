```python
import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            cap = 1.0 - h
            tot = float(cap.sum())
            if tot > 0:
                h += (diff / tot) * cap
        else:
            tot = float(h.sum())
            if tot > 0:
                h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-12:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-12:
        idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old.copy(), n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)
```

```python
import numpy as np

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            cap = 1.0 - h
            tot = float(cap.sum())
            if tot > 0:
                h += (diff / tot) * cap
        else:
            tot = float(h.sum())
            if tot > 0:
                h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-10:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-10:
        idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h
```

```python
import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            cap = 1.0 - h
            tot = float(cap.sum())
            if tot > 0:
                h += (diff / tot) * cap
        else:
            tot = float(h.sum())
            if tot > 0:
                h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-10:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-10:
        idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def resample_h(old, n):
    n0 = len(old)

        return project_feasible(old.copy(), n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def to_binary(h, n):

def optimize_coord(hp, cseg, a, qconst, unaff):

def subgrad_step(h, n, steps, deadline):

    best_h, best_c = h.copy(), compute_c(h, n)
    step = 0.05

        k_max = int(np.argmax(c))
        s = k_max - (n - 1)
        g = np.zeros(n)
        if s > 0:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
        elif s < 0:

        h = project_feasible(h - step * g, n)
        cv = compute_c(h, n)
```
