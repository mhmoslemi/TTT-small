def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_refine(h, n, alphas, maxiter=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 20:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgrad_update(h, n, top_k, step, lam):
    c = np.correlate(h, 1.0 - h, mode='full')
    if top_k >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argsort(c)[::-1][:top_k]
    grad = np.zeros(n)
    for k_idx in idx:
        s = int(k_idx) - (n - 1)
        if s >= 0:
            f_lo, f_hi = 0, n - 1 - s
        else:
            f_lo, f_hi = -s, n - 1
        if f_hi >= f_lo:
            grad[f_lo:f_hi + 1] += (1.0 - h[f_lo + s:f_hi + 1 + s])
        if s >= 0:
            b_lo, b_hi = s, n - 1
        else:
            b_lo, b_hi = 0, n - 1 + s
        if b_hi >= b_lo:
            grad[b_lo:b_hi + 1] -= h[b_lo - s:b_hi + 1 - s]
    grad /= len(idx)
    grad += lam
    return h - step * grad, float(c.max())

def run_subgrad(h, n, deadline, step0=0.04, top_k=30, max_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = step0
    lam = 0.0
    h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        h, _ = subgrad_update(h, n, top_k, step, lam)
        h = project_feasible(h, n)
        if it % 80 == 0:
            nc = compute_c5(h, n)
            if nc < best_c:
                best_h = h.copy()
                best_c = nc
            if nc < best_c * 1.002:
                step = min(step * 1.05, 0.25)
            else:
                step *= 0.85
            lam = 0.005 * (n / 2.0 - h.sum())
    return best_h, best_c

def coord_cd(h, n, max_passes=200, deadline=None, n_grid=60, rng=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    for pidx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c_prev = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c_prev.max())
        improved = False
        if rng is not None and pidx > 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for p in order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, p + n - 1)
            ms = np.arange(m_lo, m_hi + 1)
            jp_f = p + ms - (n - 1)
            jp_b = p - ms + (n - 1)
            a = np.zeros(len(ms))
            vf = (jp_f >= 0) & (jp_f < n)
            vb = (jp_b >= 0) & (jp_b < n)
            if np.any(vf):
                a[vf] += 1.0 - h[jp_f[vf]]
            if np.any(vb):
                a[vb] -= h[jp_b[vb]]
            c_aff = c_prev[ms].copy()
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c_prev[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c_prev[m_hi + 1:].max()))
            diag_mask = (ms == (n - 1))
            d_coarse = np.linspace(-h0, 1.0 - h0, 25)
            best_d = 0.0
            best_val = current_max
            for d in d_coarse:
                if abs(d) < 1e-14:
                    continue
                nv = c_aff + d * a
                if np.any(diag_mask):
                    nv[diag_mask] -= d * d
                val = max(unaff_max, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            if best_d != 0.0:
                span = (1.0 - h0) / 25.0
                lo = max(-h0, best_d - span)
                hi = min(1.0 - h0, best_d + span)
                d_fine = np.linspace(lo, hi, n_grid)
            else:
                d_fine = np.linspace(-h0 * 0.2, (1.0 - h0) * 0.2, n_grid)
            for d in d_fine:
                if abs(d) < 1e-14:
                    continue
                nv = c_aff + d * a
                if np.any(diag_mask):
                    nv[diag_mask] -= d * d
                val = max(unaff_max, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-14 and best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                c_prev[ms] = c_aff + best_d * a
                if np.any(diag_mask):
                    c_prev[ms[diag_mask]] -= best_d * best_d
        if not improved:
            break
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < best_c:
            best_c = nc
    return h, best_c

def smooth_lse(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    lse = mx + np.log(np.sum(np.exp(sh - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sfun(x, n):
    return x.sum() - n / 2.0

def _sjac(x, n):
    return np.ones(n)

def lbfgs_refine(h, n, alphas, maxiter=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_lse, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=[(0, 1)] * n,
                           constraints=[{'type': 'eq', 'fun': _sfun, 'jac': _sjac, 'args': (n,)}],
                           options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def binary_push(h, n, rng):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for trial in range(3):
        t = 0.3 + 0.4 * np.random.uniform()
        hb = (h > t).astype(float)
        hb = project_feasible(hb + 0.01 * rng.randn(n), n)
        nc = compute_c5(hb, n)
        if nc < best_c:
            best_c = nc
            best_h = hb.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 200
        x = np.linspace(0, 2, n, endpoint=False)
        h_init = 0.5 + 0.4 * np.sin(2 * np.pi * x)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: Subgradient for global optimization
    if time.time() < deadline - 200:
        h1, c1 = run_subgrad(best_h, n, deadline - 200, 0.04, 30, 20000)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: L-BFGS with high alpha for polishing the landscape
    if time.time() < deadline - 150:
        h2, c2 = lbfgs_refine(best_h, n, [5000, 50000, 200000], 500, deadline - 150)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Coordinate descent
    if time.time() < deadline - 60:
        h3, c3 = coord_cd(best_h, n, 300, deadline - 60, 70, rng)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Multi-resolution
    if time.time() < deadline - 80:
        x_big = np.linspace(0, 2, n, endpoint=False)
        for ns in [100, 160, 250]:
            if time.time() > deadline - 80 or ns >= n:
                break
            xs = np.linspace(0, 2, ns, endpoint=False)
            hs = project_feasible(np.interp(xs, x_big, best_h) + 0.02 * rng.randn(ns), ns)
            hs, cs = run_subgrad(hs, ns, deadline - 80, 0.04, 20, 5000)
            hs, cs = coord_cd(hs, ns, 100, deadline - 80, 50, rng)
            hb = project_feasible(np.interp(x_big, xs, hs), n)
            hb, cb = coord_cd(hb, n, 80, deadline - 80, 50, rng)
            if cb < best_c:
                best_h, best_c = hb, cb
    # Phase 5: Diverse restarts
    if time.time() < deadline - 60:
        rem = max(1, deadline - 60 - time.time())
        nr = max(3, int(rem / 50))
        for r in range(nr):
            if time.time() > deadline - 30:
                break
            s = 0.015 + 0.04 * (r / max(1, nr - 1))
            if r % 3 == 2:
                hp = project_feasible(rng.uniform(0, 1, n), n)
            else:
                hp = project_feasible(best_h + rng.normal(0, s, n), n)
            ho, co = run_subgrad(hp, n, deadline - 30, 0.03, 20, 3000)
            ho, co = coord_cd(ho, n, 80, deadline - 30, 50, rng)
            if co < best_c:
                best_h, best_c = ho, co
    # Phase 6: Binary push heuristic
    if time.time() < deadline - 20:
        hb, cb = binary_push(best_h, n, rng)
        if cb < best_c:
            best_h, best_c = hb, cb
    # Final polish
    if time.time() < deadline - 2:
        h5, c5 = coord_cd(best_h, n, 200, deadline - 2, 100, rng)
        if c5 < best_c:
            best_h, best_c = h5, c5
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
