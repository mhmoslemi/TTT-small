```python
import numpy as np
from scipy.optimize import golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, n, alpha):
    c = np.correlate(h, 1.0-h, mode='full')
    c_max = c.max()
    m_star = int(np.argmax(c))
    
    # Weights for smooth max
    shifted = alpha * (c - c_max)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    total = exp_s.sum()
    weights = exp_s / total
    
    # Gradient of smooth max part
    # first_term[p] = sum_i weights[i] * (1-h[p-i+(n-1)]) [valid]
    # second_term[p] = sum_i weights[i] * h[p+i-(n-1)] [valid]
    
    p_idx = np.arange(n)[:, None]
    i_idx = np.arange(2*n-1)[None, :]
    
    j_first = p_idx - i_idx + (n-1)
    valid_first = (j_first >= 0) & (j_first < n)
    omh = np.where(valid_first, (1.0-h)[np.clip(j_first, 0, n-1)], 0.0)
    first_term = (weights[None, :] * omh).sum(axis=1)
    
    j_second = p_idx + i_idx - (n-1)
    valid_second = (j_second >= 0) & (j_second < n)
    h_vals = np.where(valid_second, h[np.clip(j_second, 0, n-1)], 0.0)
    second_term = (weights[None, :] * h_vals).sum(axis=1)
    
    grad_smooth = first_term - second_term
    
    # Gradient of hard max
    j1 = np.arange(n) - m_star + (n-1)
    v1 = (j1 >= 0) & (j1 < n)
    grad_max = np.zeros(n)
    grad_max[v1] += (1.0 - h[j1[v1]])
    
    j2 = np.arange(n) + m_star - (n-1)
    v2 = (j2 >= 0) & (j2 < n)
    grad_max[v2] -= h[j2[v2]]
    
    dx = 2.0 / n
    grad = (grad_max + grad_smooth) * dx
    return grad

def adam_optimize(h, n, alpha, n_steps, deadline, lr=0.01):
    h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    t = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
```

```python
def compute_grad_sparse(h, n, alpha, top_k=5):
    c = np.correlate(h, 1.0-h, mode='full')
    c_max = c.max()
    m_star = int(np.argmax(c))
    
    shifted = alpha * (c - c_max)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    total = exp_s.sum()

    # Find top k weights
    top_idx = np.argsort(weights)[::-1][:top_k]
    top_w = weights[top_idx]
    
    dx = 2.0 / n
```
