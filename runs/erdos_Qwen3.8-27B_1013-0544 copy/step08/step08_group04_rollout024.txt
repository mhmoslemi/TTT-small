import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if diff > 0.0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0.0:
            h = h + (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 0.0:
            h = h + (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-12:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-12:
        idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def soft_max_val(c, alpha):
    z = alpha * c
    zm = float(z.max())
    return float((zm + np.log(np.sum(np.exp(z - zm)))) / alpha)

def soft_max_h(h, n, alpha):
    """Objective: soft max of correlate(h, 1-h), scaled by 2/n."""
    c = np.correlate(h, 1.0 - h, mode='full')
    return soft_max_val(c, alpha) * (2.0 / n)

def soft_max_h_grad(h, n, alpha):
    """Gradient of soft max objective w.r.t. h."""
    c = np.correlate(h, 1.0 - h, mode='full')
    L = 2 * n - 1
    center = n - 1
    z = alpha * c
    z = z - z.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(L):
        wm = w[m]
        if wm < 1e-14:
            continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n - s] += wm * (1.0 - h[s:])
            g[s:] -= wm * h[:n - s]
        else:
            r = -s
            g[r:] += wm * (1.0 - h[:n - r])
            g[:n - r] -= wm * h[r:]
    return (2.0 / n) * g

def optimize_lbfgs(h, n, alpha, deadline, mu_start=1000.0, maxiter=300):
    """L-BFGS-B with penalty for the equality constraint."""
    def obj_with_penalty(x):
        val = soft_max_h(x, n, alpha)
        penalty = mu_start * (float(x.sum()) - 0.5 * n) ** 2
        return val + penalty
    
    def grad_with_penalty(x):
        g = soft_max_h_grad(x, n, alpha)
        g = g + 2.0 * mu_start * (float(x.sum()) - 0.5 * n)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_with_penalty, h, jac=grad_with_penalty,
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        h_out = project_feasible(res.x, n)
        c_out = compute_c(h_out, n)
        return h_out, c_out
    except Exception:
        return h, compute_c(h, n)
