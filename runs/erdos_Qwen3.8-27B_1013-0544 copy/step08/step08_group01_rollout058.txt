def cross_corr_centered(a, b):
    """a has length 2n-1 (centered), b has length n
    Returns result[j] = sum_k a[k+(n-1)] * b[j+k] for valid indices"""
    n = len(b)
    result = np.zeros(n)
    for d in range(-(n-1), n):
        # a[d+(n-1)] * b[j+d] for valid j
        a_val = a[d + (n-1)]
        if a_val == 0:
            continue
        if d >= 0:

            result[:n-d] += a_val * b[d:]
        else:
            # j+d >= 0 => j >= -d
            result[-d:] += a_val * b[:n+d]
    return result

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=float)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    cx = c * dx
    m = cx.max()
    w = np.exp(alpha * (cx - m))
    w /= w.sum()
    A = w  # length 2n-1, centered at n-1
    A_rev = A[::-1]
    
    def cc(a, b):
        nb = len(b)
        res = np.zeros(nb)
        for d in range(-(nb-1), nb):
            v = a[d + nb - 1]
            if v > 1e-15:
                if d >= 0:
                    res[:nb-d] += v * b[d:nb]
                else:
                    res[-d:nb] += v * b[0:nb+d]
        return res
    
    return dx * (1.0 - cc(A, h) - cc(A_rev, h)), w, c

            lo = -h0_val

import numpy as np
from scipy.optimize import golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=float)
    lo = min(y.min(), 0.0) - 1.0
    hi = max(y.max(), 1.0) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(c, alpha, dx):
    cx = c * dx
    m = cx.max()
    return m + np.log(np.sum(np.exp(alpha * (cx - m)))) / alpha

def grad_smooth(c, h, n, alpha, dx):
    cx = c * dx
    m = cx.max()
    w = np.exp(alpha * (cx - m))
    w /= w.sum()
    # grad[j] = dx * sum_k w[k] * ((1-h[j-k+n-1]) - h[j+k-n+1])
    # Using the cross-correlation formulation
    A = w  # length 2n-1
    A_rev = A[::-1]
    
    def cc(a, b):
        nb = len(b)
        res = np.zeros(nb)
        for d in range(-(nb-1), nb):
            v = a[d + nb - 1]
            if v > 1e-18:
                if d >= 0:
                    res[:nb-d] += v * b[d:]
                else:
                    res[-d:] += v * b[:nb+d]
        return res
    
    return dx * (1.0 - cc(A, h) - cc(A_rev, h))

def fista(h0, n, alpha, max_iter, deadline, L_mult=1.0):
    dx = 2.0 / n
    h = h0.copy()
    h_prev = h0.copy()
    t_k = 1.0
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    # Estimate Lipschitz constant
    c = np.correlate(h, 1-h, mode='full')
    L = 4.0 * dx * alpha * L_mult
    step = 1.0 / L if L > 0 else 1.0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        y = h + ((t_k - 1.0) / (t_k + 1.0)) * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        c_y = np.correlate(y, 1.0 - y, mode='full')
        g = grad_smooth(c_y, y, n, alpha, dx)
        h_new = project_feasible(y - step * g, n)
        h_prev = h
        h = h_new
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) * 0.5
        if it % 200 == 0:
            cc_val = compute_c5(h, n)
            if cc_val < best_c:
                best_c = cc_val
                best_h = h.copy()
    
    cc_val = compute_c5(h, n)
    if cc_val < best_c:
        best_c = cc_val
        best_h = h.copy()
    return best_h, best_c

def coord_cd(h0, n, max_passes, deadline, grid_size=40):
    h = h0.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = c.max()
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0v = h[p]
            if h0v < 1e-12 or h0v > 1.0 - 1e-12:
                continue
            d_vals = np.arange(2*n-1) - (n-1)
            i1 = p - d_vals
            i2 = p + d_vals
            a = np.ones(2*n-1)
            v1 = (i1 >= 0) & (i1 < n)
            v2 = (i2 >= 0) & (i2 < n)
            a[v1] -= h[i1[v1]]
            a[v2] -= h[i2[v2]]
            lo = -h0v
            hi = 1.0 - h0v
            # Find min of max(c + d*a) over d in [lo, hi]
            # Use golden section
            try:
                def objective(d):
                    return float(np.max(c + d * a))
                res = golden(objective, lo, hi, maxiter=80)
                if res.fun < cur_max - 1e-14:
                    h[p] = np.clip(h0v + res.x, 0.0, 1.0)
                    improved = True
                    cur_max = res.fun
            except Exception:
                pass
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)
