import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -h.max() - 1.0, 1.0 - h.min() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / max(1, mask.sum())
            h = np.clip(h, 0.0, 1.0)
    return h

def exact_cd_active(h, n, max_passes=200, active_M=150, deadline=None):
    h = h.copy()
    c = get_correlation(h)
    current_max = float(c.max())
    
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        active = np.argsort(c)[-active_M:]
        improved = False
        c_vals_active = c[active]
        
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            
            # Compute B_p[k] for active k's
            B = np.empty(active_M)
            Cv = np.zeros(active_M)
            for idx, k in enumerate(active):
                k = int(k)
                if k == n - 1:
                    B[idx] = 1.0
                    Cv[idx] = 1.0
                else:
                    j1 = p + (n - 1 - k)
                    j2 = p - (n - 1 - k)
                    val = 0.0
                    if 0 <= j1 < n:
                        val += (1.0 - h[j1])
                    if 0 <= j2 < n:
                        val -= h[j2]
                    B[idx] = val
            
            # Golden section search for t in [0,1]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_f, b_f = 0.0, 1.0
            c1 = b_f - gr * (b_f - a_f)
            d1 = a_f + gr * (b_f - a_f)

                    fc1 = float(_f(c1))
                else:
                    a_f = c1; c1 = d1; fc1 = fd1
                    d1 = a_f + gr * (b_f - a_f)
                    fd1 = float(_f(d1))
            t_star = (a_f + b_f) * 0.5
            
            f0 = float(_f(0.0))
            f1 = float(_f(1.0))
            fs = float(_f(t_star))
            
            if min(f0, f1, fs) < current_max - 1e-10:
                if f0 <= f1 and f0 <= fs:

                elif f1 <= f0 and f1 <= fs:
                    t_new = 1.0

                c_full = get_correlation(h)

                    c_vals_active = c[active]

def pnorm_optimize(h, n, p, maxiter=500, deadline=None):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return float(np.sum(c**p)) ** (1.0/p) * dx
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        # d/dx of sum(c^p)^(1/p) is complex, use numerical or skip
        return None
    def sum_con(x):
        return x.sum() - n / 2.0
    def sum_con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except:
        return h, compute_c5(h, n)

    c = get_correlation(h)
    current_max = float(c.max())

                    c = c_new
                    current_max = new_max
                    improved = True
                    break
                else:
                    h[p] = h0
        if not improved:
            break
    
    return h, compute_c5(h, n)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -h.max() - 1.0, 1.0 - h.min() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h
