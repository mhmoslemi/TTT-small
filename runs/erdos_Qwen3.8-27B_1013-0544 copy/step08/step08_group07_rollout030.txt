import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, topk):
    c = np.correlate(x, 1.0 - x, mode='full')
    k = min(topk, len(c))
    idx = np.argpartition(-c, k)[:k]
    topk_vals = c[idx]
    mx = topk_vals.max()
    lse = mx + np.log(np.sum(np.exp(topk_vals - mx)))
    return (c.max() + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alphas, maxiter_each=500, deadline=None, topk=10):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 30:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha, topk), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16, 'gtol': 1e-10,
                                    'maxls': 50})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
        if deadline is not None and time.time() > deadline - 30:
            break
    return best_h, best_c

def coord_cd_fast(h, n, max_passes=300, deadline=None, n_coarse=80, n_fine=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    for pidx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c_prev = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c_prev.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_mask = (ms == (n - 1))
            c_aff = c_prev[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c_prev[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c_prev[m_hi + 1:].max()))
            d_vals = np.linspace(-h0, 1.0 - h0, n_coarse)
            d_col = d_vals[:, np.newaxis]
            new_cs = c_aff[np.newaxis, :] + d_col * a[np.newaxis, :]
            if np.any(diag_mask):
                new_cs[:, diag_mask] -= (d_vals ** 2)[:, np.newaxis]
            max_per_d = new_cs.max(axis=1)
            obj_per_d = np.maximum(max_per_d, unaff_max)
            best_idx = obj_per_d.argmin()
            best_val = obj_per_d[best_idx]
            best_d = d_vals[best_idx]
            if best_d != 0.0:
                span = 2.0 / n_coarse
                lo = max(-h0, best_d - span)
                hi = min(1.0 - h0, best_d + span)
                d_fine = np.linspace(lo, hi, n_fine)
                dc = d_fine[:, np.newaxis]
                new_cs_f = c_aff[np.newaxis, :] + dc * a[np.newaxis, :]
                if np.any(diag_mask):
                    new_cs_f[:, diag_mask] -= (d_fine ** 2)[:, np.newaxis]
                mpf = new_cs_f.max(axis=1)
                obj_f = np.maximum(mpf, unaff_max)
                fi = obj_f.argmin()
                if obj_f[fi] < best_val:
                    best_val = obj_f[fi]
                    best_d = d_fine[fi]
            if abs(best_d) > 1e-13 and best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                c_prev[ms] = c_aff + best_d * a
                if np.any(diag_mask):
                    c_prev[ms[diag_mask]] -= best_d * best_d
        if not improved:
            break
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
    return h, compute_c5(h, n)

def coord_cd_bipass(h, n, max_passes=200, deadline=None):

    h = h.copy()

        if deadline is not None and time.time() > deadline:

        forward = (pidx % 2 == 0)
        order = range(n) if forward else range(n - 1, -1, -1)
        c_prev = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c_prev.max())
        improved = False
        for p in order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_mask = (ms == (n - 1))
            c_aff = c_prev[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c_prev[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c_prev[m_hi + 1:].max()))
            d_vals = np.linspace(-h0, 1.0 - h0, 100)
            d_col = d_vals[:, np.newaxis]
            new_cs = c_aff[np.newaxis, :] + d_col * a[np.newaxis, :]
            if np.any(diag_mask):
                new_cs[:, diag_mask] -= (d_vals ** 2)[:, np.newaxis]
            obj = np.maximum(new_cs.max(axis=1), unaff_max)
            bi = obj.argmin()
            if obj[bi] < current_max - 1e-14 and abs(d_vals[bi]) > 1e-13:
                h[p] = float(np.clip(h0 + d_vals[bi], 0.0, 1.0))
                improved = True
                c_prev[ms] = c_aff + d_vals[bi] * a
                if np.any(diag_mask):
                    c_prev[ms[diag_mask]] -= d_vals[bi] ** 2
                current_max = max(unaff_max, float(c_prev.max()))
        if not improved:
            break
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
    return h, compute_c5(h, n)

def struct_start(n, kind, rng):

        h = 0.5 + 0.3 * np.sin(2 * np.pi * t) * (1 - t)
    else:

    return project_feasible(h, n)
