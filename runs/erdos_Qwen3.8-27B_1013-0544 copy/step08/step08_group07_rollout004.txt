def project_feasible(h, n):

            if tot < 1e-15:

            ho, co = fast_cd(ho, n, 150, deadline, n_grid=45, n_golden=35, rng=rng)
            if co < best_c:
                best_h, best_c = ho, co
    
    # Phase 5: Final intensive CD
    if time.time() < deadline - 15:
        hf, cf = fast_cd(best_h, n, 500, deadline, n_grid=60, n_golden=50, rng=rng)
        if cf < best_c:
            best_h, best_c = hf, cf
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
