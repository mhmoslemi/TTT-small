                pk2 = p + k
                if 0 <= pk < n:
                    a[i] += 1.0 - h[pk]
                if 0 <= pk2 < n:
                    a[i] -= h[pk2]
                    if k == 0:
                        b[i] = 1.0
            c_aff = c[m_lo:m_hi + 1]
            umax = 0.0
            if m_lo > 0:
                umax = max(umax, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                umax = max(umax, float(c[m_hi + 1:].max()))
            lo_d, hi_d = -h0, 1.0 - h0
            d_grid = np.linspace(lo_d, hi_d, 14)
            best_d, best_v = 0.0, current_max
            for d in d_grid:
                if abs(d) < 1e-15:
                    continue
                v = _obj_at_d(d, c_aff, a, b, umax)
                if v < best_v:
                    best_v, best_d = v, d
            if best_d != 0.0 and best_v < current_max:
                span = (hi_d - lo_d) / 14.0
                lo2 = max(lo_d, best_d - span)
                hi2 = min(hi_d, best_d + span)
                c1 = hi2 - gr * (hi2 - lo2)
                d1 = lo2 + gr * (hi2 - lo2)
                f1 = _obj_at_d(c1, c_aff, a, b, umax)
                f2 = _obj_at_d(d1, c_aff, a, b, umax)
                for _ in range(28):
                    if f1 < f2:
                        hi2, d1, f2 = d1, c1, f1
                        c1 = hi2 - gr * (hi2 - lo2)
                        f1 = _obj_at_d(c1, c_aff, a, b, umax)
                    else:
                        lo2, c1, f1 = c1, d1, f2
                        d1 = lo2 + gr * (hi2 - lo2)
                        f2 = _obj_at_d(d1, c_aff, a, b, umax)
                best_d = (lo2 + hi2) / 2.0
                best_v = _obj_at_d(best_d, c_aff, a, b, umax)
            if abs(best_d) > 1e-13 and best_v < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[m_lo:m_hi + 1] = c_aff + best_d * a - best_d * best_d * b
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < best_c:
            best_c = nc
    return h, compute_c5(h, n)

def gradient_step_smoothed(h, n, alpha, lr, deadline, n_steps=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m = float(c.max())
        shifted = (c - m) / alpha
        max_shifted = float(shifted.max())
        weights = np.exp(shifted - max_shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            break
        weights /= w_sum
        grad = np.zeros(n)
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-6 or h0 > 1.0 - 1e-6:
                continue
            k_lo = max(-(n - 1), p - n + 1, -p)
            k_hi = min(n - 1, p, n - 1 - p)
            if k_lo > k_hi:
                grad[p] = 2.0 / n
                continue
            ks = np.arange(k_lo, k_hi + 1)
            a = np.zeros(len(ks))
            for i in range(len(ks)):
                k = ks[i]
                pk = p - k
                pk2 = p + k
                if 0 <= pk < n:
                    a[i] += 1.0 - h[pk]
                if 0 <= pk2 < n:
                    a[i] -= h[pk2]
                    if k == 0:
                        a[i] -= 2.0 * h0
            idx = n - 1 + ks
            grad[p] = float(np.dot(weights[idx], a)) * (2.0 / n)
        h -= lr * grad
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < best_c:
            best_c = nc
        else:
            if step > 20:
                lr *= 0.95
    return h, best_c

def interpolate_back(h_small, n_small, n_big):
    x_s = np.linspace(0, 2, n_small, endpoint=False)
    x_b = np.linspace(0, 2, n_big, endpoint=False)
    return np.interp(x_b, x_s, h_small)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 400
        x = np.linspace(0, 2, n, endpoint=False)
        h_init = 0.5 + 0.35 * np.sin(2 * np.pi * x) + 0.12 * np.sin(4 * np.pi * x)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: Thorough coordinate descent
    h1, c1 = coord_descent(best_h, n, 300, deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Gradient smoothing step
    if time.time() < deadline - 120:
        h2, c2 = gradient_step_smoothed(best_h, n, 500.0, 0.002, deadline, 100)
        h2, c2 = coord_descent(h2, n, 150, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Diverse perturbations
    perturbations = []
    for s in [0.015, 0.03, 0.05, 0.08, 0.02, 0.04]:
        perturbations.append(('gauss', s))
    for bs in [8, 15, 30, 50]:
        perturbations.append(('block', bs))
    for f in [1, 2, 3, 5, 7]:
        perturbations.append(('spec', f))
    for ns in [150, 200, 250, 300]:
        if ns < n:
            perturbations.append(('interp', ns))
    idx = np.arange(len(perturbations))
    rng.shuffle(idx)
    perturbations = [perturbations[i] for i in idx]
    for name, param in perturbations:
        if time.time() > deadline - 80:
            break
        hp = best_h.copy()
        if name == 'gauss':
            hp += rng.normal(0, param, n)
        elif name == 'block':
            start = rng.randint(0, n - param)
            hp[start:start + param] = rng.uniform(0, 1, param)
        elif name == 'spec':
            x = np.linspace(0, 2, n, endpoint=False)
            amp = rng.uniform(0.04, 0.12)
            hp += amp * np.sin(2 * np.pi * param * x + rng.uniform(0, 2 * np.pi))
        elif name == 'interp':
            x_orig = np.linspace(0, 2, n, endpoint=False)
            x_s = np.linspace(0, 2, param, endpoint=False)
            h_s = np.interp(x_s, x_orig, best_h) + 0.02 * rng.randn(param)
            h_s = project_feasible(h_s, param)
            h_s, c_s = coord_descent(h_s, param, 200, deadline)
            hp = interpolate_back(h_s, param, n)
        hp = project_feasible(hp, n)
        h_new, c_new = coord_descent(hp, n, 200, deadline)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
    # Phase 4: Try different n values from scratch
    if time.time() < deadline - 100:
        for n_try in [200, 300, 350]:
            if time.time() > deadline - 60:
                break
            x_t = np.linspace(0, 2, n_try, endpoint=False)
            x_orig = np.linspace(0, 2, n, endpoint=False)
            ht = project_feasible(np.interp(x_t, x_orig, best_h), n_try)
            ht, ct = coord_descent(ht, n_try, 300, deadline)
            if ct < best_c:
                ht_back = project_feasible(interpolate_back(ht, n_try, n), n)
                ht_back, ct2 = coord_descent(ht_back, n, 100, deadline)
                if ct2 < best_c:
                    best_h, best_c = ht_back, ct2
                else:
                    # use smaller n solution directly
                    if ct < best_c:
                        best_h, best_c = ht, ct
    # Phase 5: Final polish with many coordinate descent passes
    if time.time() < deadline - 10:
        hf, cf = coord_descent(best_h, len(best_h), 400, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
