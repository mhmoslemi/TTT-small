            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return y

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_val_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - h, w, mode="full")
    B = np.convolve(h, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A[s:s+n] - B[s:s+n]) * dx

def _smooth_obj(x, n, a):
    return smooth_val_grad(x, n, a)[0]

def _smooth_grad(x, n, a):
    return smooth_val_grad(x, n, a)[1]

def _sum_con(x, n):
    return x.sum() - 0.5 * n

def _sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def slsqp_polish(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n)
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _sum_con, "jac": _sum_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj, best, args=(n, a), jac=_smooth_grad,
                           method="SLSQP", bounds=bounds, constraints=cons,
                           options={"maxiter": int(maxiter), "ftol": 1e-12})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-13:
            best = nh
            bc = nc
    return best, bc

def binary_trial(c, om, h, i, j, n):
    d = c.copy()
    d[i:i+n] -= om
    st = n - 1 - i
    d[st:st+n] += h
    d[j:j+n] += om
    st = n - 1 - j
    d[st:st+n] -= h
    d[n-1] -= 2.0
    d[n-1+i-j] += 1.0
    d[n-1+j-i] += 1.0
    return d, float(d.max())

def binarize(h, n):
    if n % 2:
        return None
    target = n // 2
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def binary_sa(h_init, n, rng, deadline):
    b = binarize(h_init, n)
    if b is None:
        return None, np.inf
    dx = 2.0 / n
    best = b.copy()
    c = np.correlate(b, 1.0 - b, mode="full")
    bc = float(c.max())
    cm = bc
    om = 1.0 - b[::-1].copy()
    
    T = 0.001
    T_min = 1e-8
    cooling = 0.99995
    plateau = 0
    
    while time.time() < deadline:
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if b[i] == 0.0:
            i = int(np.argmax(b))
        if b[j] == 1.0:
            j = int(np.argmin(b))
        if i == j:
            continue
        
        c2, nm = binary_trial(c, om, b, i, j, n)
        delta = nm - cm
        
        if delta < 0 or (T > T_min and rng.rand() < np.exp(-delta / T)):
            c = c2
            cm = nm
            b[i] = 0.0
            b[j] = 1.0
            om[n-1-i] = 1.0
            om[n-1-j] = 0.0
            if cm < bc:
                best = b.copy()
                bc = cm
            plateau = 0
        else:
            plateau += 1
        
        T *= cooling
        if plateau > 3000:
            T = max(T * 5.0, 0.001)
            plateau = 0
        if T < T_min and plateau > 5000:
            break
    
    return best, bc * dx

def momentum_pg(h, n, deadline, alpha0=5000, lr=0.05, mu=0.85, max_iter=20000):
    best = h.copy()
    bc = compute_c5(best, n)
    x = h.copy()
    v = np.zeros(n)
    alpha = alpha0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it > 0 and it % 500 == 0:
            alpha = min(alpha * 3.0, 1e7)
        _, grad = smooth_val_grad(x, n, alpha)
        v = mu * v - lr * grad
        x_new = project_simplex(x + v, n)
        cn = compute_c5(x_new, n)
        if cn < bc:
            best = x_new.copy()
            bc = cn
        x = x_new
    
    return best, bc

def coord_cd(h, n, max_passes=4, deadline=None, grid=32, refine=10, tol=1e-12):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    A = np.zeros(L)
    xs0 = np.linspace(0.0, 1.0, grid)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = float(x[p])
            A.fill(0.0)
            A[p:p+n] += om
            st = diag - p
            A[st:st+n] -= x
            cur = float(c.max())
            d = xs0 - old
            cand = c[None,:] + d[:,None] * A[None,:]
            cand[:, n-1] -= d*d
            vals = cand.max(axis=1)
            ii = int(np.argmin(vals))
            bx = float(xs0[ii])
            bv = float(vals[ii])
            if bv < cur - tol:
                sp = 1.0/(grid-1)
                lo = max(0.0, bx-sp)
                hi = min(1.0, bx+sp)
                xx = np.linspace(lo, hi, refine+1)
                d2 = xx - old
                cand2 = c[None,:] + d2[:,None]*A[None,:]
                cand2[:,n-1] -= d2*d2
                mm = cand2.max(axis=1)
                jj = int(np.argmin(mm))
                if mm[jj] < bv - tol:
                    bx = float(xx[jj])
                    bv = float(mm[jj])
                if lo < bx < hi:
                    sp2 = (hi-lo)/float(refine)
                    lo2 = max(0.0, bx-sp2)
                    hi2 = min(1.0, bx+sp2)
                    yy = np.linspace(lo2, hi2, 9)
                    d3 = yy - old
                    cand3 = c[None,:] + d3[:,None]*A[None,:]
                    cand3[:,n-1] -= d3*d3
                    ww = cand3.max(axis=1)
                    kk = int(np.argmin(ww))
                    if ww[kk] < bv - tol:
                        bx = float(yy[kk])
                        bv = float(ww[kk])
                if bv < cur - tol:
                    d_d = bx - old
                    c[p:p+n] += d_d * om
                    c[st:st+n] -= d_d * x
                    c[diag] -= d_d*d_d
                    x[p] = bx
                    om[diag-p] = 1.0 - bx
                    improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - tol:
            best = xp
            bc = cp
        if not improved:
            break
    return best, bc

def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)

                b[i] = 0.0
                b[j] = 1.0
                om[n-1-i] = 1.0
                om[n-1-j] = 0.0
                ones.remove(i)
                zeros.remove(j)
                ones.append(j)
                zeros.append(i)
                if cm < best_corr:
                    best = b.copy()
                    best_corr = cm
                plateau = 0
            else:
                plateau += 1
            T *= cooling
            if plateau > 2500:
                T = max(T * 4.0, 0.0008)
                plateau = 0
            if T < T_min and plateau > 4000:
                break
    return best, best_corr * dx

def coord_cd(h, n, max_passes=4, deadline=None, grid=32, refine=10, tol=1e-12):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    A = np.zeros(L)
    xs0 = np.linspace(0.0, 1.0, grid)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = float(x[p])
            A.fill(0.0)
            A[p:p+n] += om
            st = diag - p
            A[st:st+n] -= x
            cur = float(c.max())
            d = xs0 - old
            cand = c[None, :] + d[:, None] * A[None, :]
            cand[:, n-1] -= d * d
            vals = cand.max(axis=1)
            ii = int(np.argmin(vals))
            bx, bv = float(xs0[ii]), float(vals[ii])
            if bv < cur - tol:
                sp = 1.0 / (grid - 1)
                lo, hi = max(0.0, bx-sp), min(1.0, bx+sp)
                xx = np.linspace(lo, hi, refine+1)
                d2, cand2 = xx - old, c[None,:] + (xx-old)[:,None]*A[None,:]
                cand2[:, n-1] -= d2*d2
                mm = cand2.max(axis=1)
                jj = int(np.argmin(mm))
                if mm[jj] < bv - tol:
                    bx, bv = float(xx[jj]), float(mm[jj])
                if lo < bx < hi:
                    sp2 = (hi-lo)/float(refine)
                    yy = np.linspace(max(0.0,bx-sp2), min(1.0,bx+sp2), 9)
                    d3 = yy - old
                    cand3 = c[None,:] + d3[:,None]*A[None,:]
                    cand3[:,n-1] -= d3*d3
                    ww = cand3.max(axis=1)
                    kk = int(np.argmin(ww))
                    if ww[kk] < bv - tol:
                        bx, bv = float(yy[kk]), float(ww[kk])
                if bv < cur - tol:
                    dd = bx - old
                    c[p:p+n] += dd*om
                    c[st:st+n] -= dd*x
                    c[diag] -= dd*dd
                    x[p] = bx
                    om[diag-p] = 1.0 - bx
                    improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - tol:
            best, bc = xp, cp
        if not improved:
            break
    return best, bc

def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)

def momentum_pg(h, n, deadline, alpha0=5000, lr=0.04, mu=0.85):
    best, bc = h.copy(), compute_c5(h, n)
    x, v, alpha = h.copy(), np.zeros(n), alpha0
    while time.time() < deadline:
        _, grad = smooth_val_grad(x, n, alpha)
        v = mu * v - lr * grad
        x = project_simplex(x + v, n)
        cn = compute_c5(x, n)
        if cn < bc:
            best, bc = x.copy(), cn
        alpha = min(alpha * 1.02, 5e6)
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(5.0, budget_s - 5.0)
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0, h0 = 500, np.full(500, 0.5)
    h0 = project_simplex(h0, n0)
    best_h, best_c, n_best = h0.copy(), compute_c5(h0, n0), n0
    rev = h0[::-1]
    for v in (0.5*(h0+rev), 0.5*(h0+(1.0-rev))):
        vv = project_simplex(v, n_best)
        cv = compute_c5(vv, n_best)
        if cv < best_c:
            best_h, best_c = vv, cv
    t1 = min(deadline - 500.0, t0 + 50.0)
    if time.time() < t1:
        h1, c1 = slsqp_polish(best_h, n_best, [5000, 50000, 500000], 80, t1)
        if c1 < best_c:
            best_h, best_c = h1, c1
    t2 = min(deadline - 400.0, t0 + 120.0)
    if time.time() < t2:
        h1, c1 = momentum_pg(best_h, n_best, t2)
        if c1 < best_c:
            best_h, best_c = h1, c1
    t3 = min(deadline - 120.0, t0 + 550.0)
    if n_best % 2 == 0 and time.time() < t3:
        hb, cb = binary_sa(best_h, n_best, rng, t3)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb
    t4 = min(deadline - 80.0, time.time() + 50.0)
    if time.time() < t4:
        h1, c1 = coord_cd(best_h, n_best, max_passes=5, deadline=t4, grid=40, refine=14)
        if c1 < best_c:
            best_h, best_c = h1, c1
    for nt in [400, 600, 450, 550]:
        if time.time() > deadline - 60.0:
            break
        if nt == n_best or nt % 2 or nt < 100 or nt > 800:
            continue
        ht = transfer_h(best_h, n_best, nt)
        chunk = min(deadline - 50.0, time.time() + 40.0)
        if time.time() >= chunk:
            continue
        h1, c1 = binary_sa(ht, nt, rng, chunk)
        if h1 is not None:
            cv1 = compute_c5(h1, nt)
            if cv1 < best_c:
                best_h, best_c, n_best = h1, cv1, nt
    t5 = min(deadline - 5.0, time.time() + 40.0)
    if time.time() < t5:
        h1, c1 = slsqp_polish(best_h, n_best, [50000, 500000], 80, t5)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h1, c1 = coord_cd(best_h, n_best, max_passes=3, deadline=t5, grid=40, refine=14)
        if c1 < best_c:
            best_h, best_c = h1, c1
    if n_best % 2 == 0 and time.time() < deadline - 5.0:
        hb, cb = binary_sa(best_h, n_best, rng, deadline - 2.0)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb
    final_h = project_simplex(best_h, n_best)
    final_c = float(compute_c5(final_h, n_best))
    return final_h, final_c, int(n_best)
