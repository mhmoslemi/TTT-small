            cands = [hp, 0.0, 1.0, 0.5, max(0.0, min(1.0, Lx)), max(0.0, min(1.0, Ux))]
            disc = 1.0 - 4.0 * (t - q_c)
            if disc >= 0:
                sq = np.sqrt(disc)
                cands.append(max(0.0, min(1.0, 0.5 * (1 - sq))))
                cands.append(max(0.0, min(1.0, 0.5 * (1 + sq))))
            best_x, best_v = hp, cur_val
            for xc in cands:
                if xc < 0 or xc > 1:
                    continue
                if len(alpha) and np.any(alpha * xc + beta > t + 1e-10):
                    continue
                if -xc * xc + xc + q_c > t + 1e-10:
                    continue
                val = max(unaff, float(np.max(alpha * xc + beta)) if len(alpha) else 0.0,
                          float(-xc * xc + xc + q_c))
                if val < best_v - 1e-13:
                    best_v = val
                    best_x = xc
            if best_x != hp and abs(best_x - hp) > 1e-11:
                dx = best_x - hp
                if len(m_nz):
                    c[center + m_nz] += alpha * dx
                c[center] += best_x * (1 - best_x) - hp * (1 - hp)
                h[p] = best_x
                nm = float(c.max())
                if nm < cur:
                    cur = nm
                    improved = True
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        cv = (2.0 / n) * cur
        if cv < best_c - 1e-14:
            best_c = cv
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def fourier_perturb(h, n, rng, scale=0.04):
    x = (np.arange(n) + 0.5) * 2.0 / n
    t = x - 1.0
    pert = np.zeros(n)
    for k in range(1, 7):
        amp = rng.uniform(-scale, scale) / k
        pert += amp * np.cos(k * np.pi * t)
    return project_feasible(h + pert, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 12.0
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n = 500
        x = (np.arange(n) + 0.5) * 2.0 / n
        h = project_feasible(1.0 - np.abs(x - 1.0), n)
    n = len(h)
    if n % 2:
        n += 1
        h = resample_h(h, n)
    best_h = project_feasible(h, n)
    best_c = c5_val(best_h)
    # Phase 1: aggressive CD
    best_h, best_c = cd_solve(best_h, n, 20, deadline)
    # Phase 2: targeted restarts
    for r in range(60):
        if time.time() > deadline - 60:
            break
        if best_c <= 0.38070:
            break
        typ = r % 6
        if typ == 0:
            h0 = fourier_perturb(best_h, n, rng, 0.02)
        elif typ == 1:
            h0 = project_feasible(best_h + rng.normal(0, 0.01, n), n)
        elif typ == 2:
            k = n // 2
            idx = np.argpartition(best_h, n - k)[n - k:]
            h0 = np.zeros(n)
            h0[idx] = 1.0
        elif typ == 3:
            h0 = project_feasible((best_h + best_h[::-1]) * 0.5, n)
        elif typ == 4:
            h0 = fourier_perturb(best_h, n, rng, 0.06)
        else:
            h0 = project_feasible(best_h + rng.uniform(-0.02, 0.02, n), n)
        passes = 10 if time.time() < deadline - 200 else 6
        h1, c1 = cd_solve(h0, n, passes, deadline)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    # Phase 3: try smaller resolutions
    for nt in [400, 320, 256, 200]:
        if time.time() > deadline - 50:
            break
        h0 = resample_h(best_h, nt)
        h1, c1 = cd_solve(h0, nt, 12, deadline)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    # Final polish
    if time.time() < deadline - 5:
        best_h, best_c = cd_solve(best_h, len(best_h), 15, deadline)
    best_h = project_feasible(best_h, len(best_h))
    best_c = c5_val(best_h)
    return best_h, best_c, len(best_h)
