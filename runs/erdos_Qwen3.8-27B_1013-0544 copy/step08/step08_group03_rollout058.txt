    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cd_update(c, h, p, n):

def exact_cd_fast(h, n, max_passes, deadline, n_grid=15, n_gs=25, order='random', rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(c.max())
    
    for sweep in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        indices = list(range(n))
        if order == 'random' and rng is not None:
            rng.shuffle(indices)
        
        for p in indices:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            
            A, D = cd_update(c, h, p, n)
            
            # Minimize max(c + u*A - u²*D) for u in [-h0, 1-h0]
            u_lo = -h0
            u_hi = 1.0 - h0
            
            # Grid search
            grid = np.linspace(u_lo, u_hi, n_grid)
            best_u = 0.0
            best_val = current_max  # max(c)
            for u in grid:
                val = float(np.max(c + u * A - u * u * D))
                if val < best_val:
                    best_val = val
                    best_u = u
            
            # Golden section refinement
            if best_val < current_max - 1e-14:
                # Find bracket
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a = u_lo
                b = u_hi
                # Use grid result to narrow
                gi = int(np.argmin([float(np.max(c + u * A - u * u * D)) for u in grid]))
                a = grid[max(0, gi-1)]
                b = grid[min(len(grid)-1, gi+1)]
                
                c1 = b - gr * (b - a)
                d1 = a + gr * (b - a)
                fc1 = float(np.max(c + c1 * A - c1 * c1 * D))
                fd1 = float(np.max(c + d1 * A - d1 * d1 * D))
                for _ in range(n_gs):
                    if fc1 < fd1:
                        b = d1
                        d1 = c1
                        fd1 = fc1
                        c1 = b - gr * (b - a)
                        fc1 = float(np.max(c + c1 * A - c1 * c1 * D))
                    else:
                        a = c1
                        c1 = d1
                        fc1 = fd1
                        d1 = a + gr * (b - a)
                        fd1 = float(np.max(c + d1 * A - d1 * d1 * D))
                best_u = (a + b) * 0.5
                best_val = float(np.max(c + best_u * A - best_u * best_u * D))
            
            if best_val < current_max - 1e-13:
                h[p] = h0 + best_u
                c = c + best_u * A - best_u * best_u * D
                current_max = best_val
                improved = True

            break

        c = np.correlate(h, 1.0 - h, mode='full').astype(float)
        current_max = float(c.max())
    
    return h, current_max
