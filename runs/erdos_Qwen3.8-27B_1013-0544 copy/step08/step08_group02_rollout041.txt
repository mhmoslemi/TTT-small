import numpy as np

from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)))


def project_feasible(z, target=None):
    z = np.asarray(z, dtype=float)
    n = z.size
    if target is None:
        target = 0.5 * n
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    if float(z.min()) >= -1e-12 and float(z.max()) <= 1.0 + 1e-12 and abs(float(z.sum()) - float(target)) <= 1e-8 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(60):
        tau = 0.5 * (low + high)
        s = float(np.clip(z - tau, 0.0, 1.0).sum())
        if s > float(target):
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = float(target) - float(y.sum())
    if res > 1e-13:
        room = 1.0 - y
        tot = float(room.sum())
        if tot > 1e-14:
            y = y + (res / tot) * room
    elif res < -1e-13:
        room = y
        tot = float(room.sum())
        if tot > 1e-14:
            y = y + (res / tot) * room
    return np.clip(y, 0.0, 1.0)


def make_feasible(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    target = 0.5 * n
    if float(np.min(h)) >= -1e-12 and float(np.max(h)) <= 1.0 + 1e-12 and abs(float(h.sum()) - target) <= 1e-7:
        return np.clip(h, 0.0, 1.0)
    return project_feasible(h, target)


def smooth_components(x, n, beta):
    beta = max(float(beta), 1e-6)
    c = (2.0 / n) * np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    z = beta * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    s = float(e.sum())
    val = m + (zm + np.log(s)) / beta
    p = e / s
    return val, p, c


def smooth_val(x, n, beta):
    return smooth_components(x, n, beta)[0]


def grad_raw(x, p):
    n = len(x)
    r = n - 1
    p0 = p[r]
    P = p[r + 1:]
    N = p[:r][::-1]
    W = P + N
    w = np.zeros(n, dtype=float)
    if n > 1:
        w[1:] = W
    left = np.convolve(x, w, mode="full")[:n]
    right = np.convolve(x, w[::-1], mode="full")[r:2 * n - 1]
    linP = np.zeros(n, dtype=float)
    if n > 1:
        linP[1:] = np.cumsum(P)
    prefixN = np.empty(n, dtype=float)
    prefixN[0] = 0.0
    if n > 1:
        prefixN[1:] = np.cumsum(N)
    linN = prefixN[n - 1 - np.arange(n)]
    return p0 * (1.0 - 2.0 * x) + linP + linN - left - right


def smooth_grad(x, n, beta):
    _, p, _ = smooth_components(x, n, beta)
    return (2.0 / n) * grad_raw(x, p)


def pen_val(x, n, beta, mu):
    val, _, _ = smooth_components(x, n, beta)
    dev = float(x.sum() - 0.5 * n)
    return val + 0.5 * mu * dev * dev


def pen_jac(x, n, beta, mu):
    _, p, _ = smooth_components(x, n, beta)
    g = (2.0 / n) * grad_raw(x, p)
    dev = float(x.sum() - 0.5 * n)
    return g + mu * dev


def lbfgs_stage(h, n, plans, deadline):
    best_h = np.asarray(h, dtype=float).copy()
    best_c = compute_c5(best_h)
    work = best_h.copy()
    bounds = [(0.0, 1.0)] * n
    for beta, mu, mx in plans:
        if time.time() > deadline - 2.0:
            break
        try:
            res = minimize(
                pen_val,
                work,
                args=(n, beta, mu),
                method="L-BFGS-B",
                jac=pen_jac,
                bounds=bounds,
                options={"maxiter": int(mx), "ftol": 1e-15, "maxls": 40, "disp": False},
            )
            x = np.asarray(res.x, dtype=float)
            if np.all(np.isfinite(x)):
                x = project_feasible(x, 0.5 * n)
                cc = compute_c5(x)
                if cc < best_c:
                    best_h = x.copy()
                    best_c = cc
                if cc <= best_c + 1e-5:
                    work = x
        except Exception:
            pass
    return best_h, best_c


def sum_con(x, n):
    return float(x.sum() - 0.5 * n)


def sum_jac(x, n):
    return np.ones_like(x, dtype=float)


def slsqp_stage(h, n, beta, mx, deadline):
    best_h = make_feasible(h).copy()
    best_c = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_jac, "args": (n,)}]
    work = best_h.copy()
    try:
        res = minimize(
            smooth_val,
            work,
            args=(n, beta),
            method="SLSQP",
            jac=smooth_grad,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(mx), "ftol": 1e-12, "disp": False},
        )
        x = project_feasible(np.asarray(res.x, dtype=float), 0.5 * n)
        cc = compute_c5(x)
        if cc < best_c:
            best_h = x.copy()
            best_c = cc
    except Exception:
        pass
    return best_h, best_c


def add_row(B, h, a, n, sign):
    off = n - 1
    if a > 0:
        B[off + 1:off + 1 + a] += sign * (1.0 - h[a - 1::-1])
    m = n - 1 - a
    if m > 0:
        B[off + 1:off + 1 + m] += sign * (-h[a + 1:a + 1 + m])
        B[off - m:off] += sign * (1.0 - h[a + 1:a + 1 + m][::-1])
    B[off] += sign * (1.0 - 2.0 * h[a])
    if a > 0:
        B[off - a:off] += sign * (-h[:a])


def pair_B(h, i, j, n):
    B = np.zeros(2 * n - 1, dtype=float)
    add_row(B, h, i, n, 1.0)
    add_row(B, h, j, n, -1.0)
    B *= 2.0 / n
    return B


def eval_update(c, B, us, i, j, n, dx, off):
    uu = us * us
    mat = c + us[:, None] * B
    q1 = off + i - j
    q2 = off + j - i
    mat[:, q1] += dx * uu
    if q2 != q1:
        mat[:, q2] += dx * uu
    mat[:, off] -= 2.0 * dx * uu
    return mat.max(axis=1)


def pair_line_search(c, B, h, i, j, n, dx, off, cur_m):
    hi = h[i]
    hj = h[j]
    lo = -hi
    hi_u = 1.0 - hj
    if hi_u - lo <= 1e-14:
        return None
    us = np.linspace(lo, hi_u, 21)
    vals = eval_update(c, B, us, i, j, n, dx, off)
    idx = int(np.argmin(vals))
    val = float(vals[idx])
    if val >= cur_m - 1e-11:
        return None
    step = (hi_u - lo) / 20.0
    a = us[idx] - 2.0 * step
    b = us[idx] + 2.0 * step
    if a < lo:
        a = lo
    if b > hi_u:
        b = hi_u
    if b - a > 1e-13:
        us2 = np.linspace(a, b, 73)
        vals2 = eval_update(c, B, us2, i, j, n, dx, off)
        idx2 = int(np.argmin(vals2))
        val2 = float(vals2[idx2])
        if val2 < val:
            us = us2
            idx = idx2
            val = val2
    return float(us[idx]), float(val)


def pair_search(h, n, max_tries, deadline, rng):
    h = make_feasible(h).copy()
    n = len(h)
    dx = 2.0 / n
    off = n - 1
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    best_h = h.copy()
    best_c = compute_c5(h)
    max_tries = max(1, int(max_tries))
    max_accept = min(50000, max(1000, max_tries // 3))
    noimp_break = max(800, max_tries // 20)
    trial = 0
    accepts = 0
    noimp = 0
    while trial < max_tries and accepts < max_accept and time.time() < deadline:
        trial += 1
        if accepts and accepts % 24 == 0:
            c = np.correlate(h, 1.0 - h, mode="full") * dx
        M = float(c.max())
        if M < best_c - 1e-12:
            best_c = M
            best_h = h.copy()
        tol = max(0.0008, 0.005 * M)
        active = np.flatnonzero(c >= M - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        if trial % 9 == 8:
            i = int(rng.randint(n))
            j = (i + 1 + int(rng.randint(max(1, n - 1)))) % n
            if i == j:
                continue
        else:
            k = int(active[int(rng.randint(active.size))])
            d = k - off
            if d == 0:
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    j = (j + 1) % n
                if i == j:
                    continue
            elif d > 0:
                m = n - d
                if m <= 0:
                    continue
                if trial & 1:
                    b = int(np.argmax(h[d:d + m] * (1.0 - h[:m])))
                else:
                    b = int(rng.randint(m))
                i = b + d
                j = b
            else:
                e = -d
                m = n - e
                if m <= 0:
                    continue
                if trial & 1:
                    b = int(np.argmax(h[:m] * (1.0 - h[e:e + m])))
                else:
                    b = int(rng.randint(m))
                i = b
                j = b + e
            if i == j:
                j = (j + 1) % n
                if i == j:
                    continue
        B = pair_B(h, i, j, n)
        move = pair_line_search(c, B, h, i, j, n, dx, off, M)
        if move is None:
            noimp += 1
            if noimp > noimp_break:
                break
            continue
        u, nv = move
        hi_old = h[i]
        hj_old = h[j]
        uu = u * u
        c = c + u * B
        c[off] -= 2.0 * dx * uu
        c[off + i - j] += dx * uu
        c[off + j - i] += dx * uu
        t = hi_old + u
        if t < 0.0:
            t = 0.0
        elif t > 1.0:
            t = 1.0
        h[i] = t
        h[j] = hi_old + hj_old - t
        accepts += 1
        noimp = 0
    h2 = make_feasible(h)
    c2 = compute_c5(h2)
    if c2 <= best_c + 1e-9:
        return h2, c2
    bh = make_feasible(best_h)
    bc = compute_c5(bh)
    return bh, bc


def improve_sym(h):
    h = make_feasible(h)
    c = compute_c5(h)
    x = make_feasible(0.5 * (h + h[::-1]))
    c2 = compute_c5(x)
    if c2 < c:
        h, c = x, c2
    x = make_feasible(1.0 - h)
    c2 = compute_c5(x)
    if c2 < c:
        h, c = x, c2
    x = make_feasible(0.5 * (h + 1.0 - h[::-1]))
    c2 = compute_c5(x)
    if c2 < c:
        h, c = x, c2
    return h, c


def interp_h(h, m):
    n = len(h)
    if n == m:
        return make_feasible(h)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), 0.5 * m)


def random_smooth_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5, dtype=float)
    for m in range(1, 7):
        a = rng.uniform(-0.25, 0.25)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += a * np.sin(0.5 * np.pi * m * x + ph)
        h += 0.5 * a * np.cos(np.pi * m * x + ph)
    h += 0.05 * (x / 2.0 - 0.5)
    return make_feasible(np.clip(h, 0.0, 1.0))


def make_sinusoid(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    f = 1 + int(rng.randint(3))
    amp = rng.uniform(0.75, 0.97)
    ph = rng.uniform(0.0, 2.0 * np.pi)
    h = 0.5 + 0.5 * amp * np.cos(np.pi * f * x + ph)
    return make_feasible(np.clip(h, 0.0, 1.0))


def binary_start(n, rng):
    h = np.zeros(n, dtype=float)
    if n > 0:
        idx = rng.permutation(n)
        cnt = n // 2
        h[idx[:cnt]] = 1.0
        if n % 2:
            h[idx[cnt]] = 0.5
    return h


def optimize_instance(h, n, deadline, rng, quick=False):
    h, c = improve_sym(h)
    best_h = h.copy()
    best_c = c
    if time.time() < deadline - 10.0:
        if quick:
            plans = [(8, 30, 40), (40, 80, 30)]
        else:
            plans = [(5, 20, 100), (15, 60, 80), (60, 150, 50), (200, 400, 30), (600, 1000, 20)]
        best_h, best_c = lbfgs_stage(best_h, n, plans, deadline)
    if (not quick) and n <= 600 and time.time() < deadline - 30.0:
        h2, c2 = slsqp_stage(best_h, n, 150, 8, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2
    if time.time() < deadline - 2.0:
        rem = deadline - time.time()
        if quick:
            tries = int(min(20000, max(500, rem * 80.0)))
        else:
            tries = int(min(80000, max(2000, rem * 120.0)))
        best_h, best_c = pair_search(best_h, n, tries, deadline, rng)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
        h0 = make_feasible(h0)
    except Exception:
        h0 = random_smooth_start(250, rng)
    best_h, best_c = improve_sym(h0)
    n0 = len(best_h)
    reserve = min(260.0, 0.3 * float(budget_s))
    main_dl = hard - reserve
    if main_dl < t0 + 60.0:
        main_dl = hard
    if hard - t0 > 50.0:
        try:
            best_h, best_c = optimize_instance(best_h, n0, main_dl, rng, False)
        except Exception:
            pass
    tried = {len(best_h)}
    sizes = []
    for fac in (0.25, 0.5, 0.64, 0.8, 1.0, 1.25, 1.5):
        m = int(round(n0 * fac))
        if m < 64:
            m = 64
        if m > 800:
            m = 800
        if m % 2:
            m += 1
        sizes.append(m)
    for m in (128, 250, 500, 640):
        if m % 2:
            m += 1
        if 64 <= m <= 800:
            sizes.append(m)
    uniq_sizes = []
    for m in sizes:
        if m not in tried and m not in uniq_sizes:
            uniq_sizes.append(m)
    uniq_sizes = uniq_sizes[:5]
    for idx, m in enumerate(uniq_sizes):
        if time.time() > hard - 35.0:
            break
        rem = hard - time.time()
        per = min(80.0, rem / max(1, len(uniq_sizes) - idx) * 1.1)
        if per < 30.0:
            continue
        try:
            h = interp_h(best_h, m)
            h, c = optimize_instance(h, m, time.time() + per, rng, True)
            tried.add(m)
            if c < best_c:
                best_h, best_c = h, c
        except Exception:
            tried.add(m)
    if time.time() < hard - 60.0:
        for sig in (0.04, 0.08, 0.15):
            if time.time() > hard - 35.0:
                break
            n = len(best_h)
            hp = project_feasible(best_h + rng.normal(0.0, sig, n), 0.5 * n)
            per = min(55.0, max(25.0, (hard - time.time()) * 0.35))
            try:
                h, c = optimize_instance(hp, n, time.time() + per, rng, True)
                if c < best_c:
                    best_h, best_c = h, c
            except Exception:
                pass
    if time.time() < hard - 50.0:
        for starter in (make_sinusoid, binary_start):
            if time.time() > hard - 35.0:
                break
            n = len(best_h)
            h = starter(n, rng)
            per = min(45.0, max(20.0, (hard - time.time()) * 0.4))
            try:
                h, c = optimize_instance(h, n, time.time() + per, rng, True)
                if c < best_c:
                    best_h, best_c = h, c
            except Exception:
                pass
    if time.time() < hard - 8.0:
        n = len(best_h)
        rem = hard - time.time()
        tries = int(min(100000, max(2000, rem * 300.0)))
        try:
            h, c = pair_search(best_h, n, tries, hard - 2.0, rng)
            if c < best_c:
                best_h, best_c = h, c
        except Exception:
            pass
    best_h = make_feasible(best_h)
    best_c = compute_c5(best_h)
    return best_h, float(best_c), int(len(best_h))
