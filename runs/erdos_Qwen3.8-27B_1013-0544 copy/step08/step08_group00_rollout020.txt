            if np.any(vb):
                a[vb] += h[jb[vb]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            if has_diag:
                a[diag_pos] = 2.0 * h0
            q_base = q[ms]
            if m_lo == 0 and m_hi == 2 * n - 2:
                unaff_min = np.inf
            else:
                cands = []
                if m_lo > 0:
                    cands.append(float(q[:m_lo].min()))
                if m_hi < 2 * n - 2:
                    cands.append(float(q[m_hi + 1:].min()))
                unaff_min = min(cands) if cands else np.inf
            T_lo = min_q
            T_hi = min(unaff_min, float(q_base.max()) + float(np.abs(a).max()) + 1.0)
            if T_hi <= T_lo:
                continue
            for _ in range(n_bisect):
                if T_hi - T_lo < 1e-13:
                    break
                T = (T_lo + T_hi) / 2.0
                t_lo_b = -h0
                t_hi_b = 1.0 - h0
                feasible = True
                if has_diag:
                    lin_mask = np.ones(len(ms), dtype=bool)
                    lin_mask[diag_pos] = False
                else:
                    lin_mask = np.ones(len(ms), dtype=bool)
                a_lin = a[lin_mask]
                q_lin = q_base[lin_mask]
                pos_m = a_lin > 1e-14
                neg_m = a_lin < -1e-14
                zero_m = np.abs(a_lin) <= 1e-14
                if np.any(zero_m) and float(q_lin[zero_m].min()) < T - 1e-14:
                    feasible = False
                if feasible:
                    if np.any(pos_m):
                        t_lo_b = max(t_lo_b, float(((T - q_lin[pos_m]) / a_lin[pos_m]).max()))
                    if np.any(neg_m):
                        t_hi_b = min(t_hi_b, float(((T - q_lin[neg_m]) / a_lin[neg_m]).min()))
                if feasible and has_diag:
                    qd = q_base[diag_pos[0]]
                    rhs = T - qd + h0 * h0
                    if rhs > 0:
                        t_lo_b = max(t_lo_b, float(np.sqrt(rhs) - h0))
                if t_lo_b > t_hi_b + 1e-14:
                    feasible = False
                if feasible:
                    T_lo = T
                else:
                    T_hi = T
            T_star = T_lo
            t_opt_lo = -h0
            t_opt_hi = 1.0 - h0
            if has_diag:
                lin_mask2 = np.ones(len(ms), dtype=bool)
                lin_mask2[diag_pos] = False
            else:
                lin_mask2 = np.ones(len(ms), dtype=bool)
            a_lin2 = a[lin_mask2]
            q_lin2 = q_base[lin_mask2]
            pos2 = a_lin2 > 1e-14
            neg2 = a_lin2 < -1e-14
            if np.any(pos2):
                t_opt_lo = max(t_opt_lo, float(((T_star - q_lin2[pos2]) / a_lin2[pos2]).max()))
            if np.any(neg2):
                t_opt_hi = min(t_opt_hi, float(((T_star - q_lin2[neg2]) / a_lin2[neg2]).min()))
            if has_diag:
                qd = q_base[diag_pos[0]]
                rhs = T_star - qd + h0 * h0
                if rhs > 0:
                    t_opt_lo = max(t_opt_lo, float(np.sqrt(rhs) - h0))
            t_opt_lo = max(t_opt_lo, -h0)
            t_opt_hi = min(t_opt_hi, 1.0 - h0)
            if t_opt_lo <= t_opt_hi:
                t_new = (t_opt_lo + t_opt_hi) / 2.0
                if abs(t_new) > 1e-11:
                    h[p] = float(np.clip(h0 + t_new, 0.0, 1.0))
                    any_improved = True
        if not any_improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def grid_cd(h, n, max_passes, deadline, grid_size=40):
    h = h.copy()
    for sweep in range(max_passes):
        if time.time() > deadline:
            break
        q = np.correlate(h, h, mode='full')
        min_q_prev = float(q.min())
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            m_lo = max(0, p - (n - 1))
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += h[jf[vf]]
            if np.any(vb):
                a[vb] += h[jb[vb]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            q_base = q[ms]
            if m_lo == 0 and m_hi == 2 * n - 2:
                unaff_min = np.inf
            else:
                cands = []
                if m_lo > 0:
                    cands.append(float(q[:m_lo].min()))
                if m_hi < 2 * n - 2:
                    cands.append(float(q[m_hi + 1:].min()))
                unaff_min = min(cands) if cands else np.inf
            ts = np.linspace(-h0, 1.0 - h0, grid_size)
            best_t = 0.0
            best_obj = float(-np.inf)
            for t in ts:
                new_q = q_base + t * a
                if has_diag:
                    new_q[diag_pos] = q_base[diag_pos] + t * (2.0 * h0) + t * t
                obj = min(float(new_q.min()), unaff_min)
                if obj > best_obj:
                    best_obj = obj
                    best_t = t
            if abs(best_t) > 1e-12:
                h[p] = float(np.clip(h0 + best_t, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: SLSQP
    if time.time() < deadline - 300:
        h1, c1 = slsqp_opt(best_h, n_init, [500, 5000, 50000], 250)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Grid CD for fast convergence
    if time.time() < deadline - 200:
        h2, c2 = grid_cd(best_h, n_init, 30, deadline, grid_size=35)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Try different n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [500, 600, 800, 400]:
        if time.time() > deadline - 250:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        hp, cp = slsqp_opt(h_int, n_try, [2000, 20000], 150)
        hp, cp = grid_cd(hp, n_try, 20, deadline, grid_size=35)
        if cp < best_c:
            best_h, best_c = hp, cp
            x_orig = np.linspace(0, 2, n_try)
    # Phase 4: Exact CD on best
    if time.time() < deadline - 100:
        n = len(best_h)
        remaining = deadline - time.time() - 60
        max_passes = max(5, int(remaining / 3.0))
        hf, cf = exact_cd(best_h, n, max_passes, deadline, n_bisect=35)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Phase 5: Multi-restart
    if time.time() < deadline - 120:
        n = len(best_h)
        remaining = deadline - time.time() - 80
        n_restarts = max(2, int(remaining / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 80:
                break
            if r % 3 == 0:
                s = 0.01 + 0.03 * (r / max(1, n_restarts))
                hp = project_feasible(best_h + rng.normal(0, s, n), n)
            elif r % 3 == 1:
                hp = project_feasible(rng.uniform(0.2, 0.8, n), n)
            else:
                hp = project_feasible(best_h + rng.randn(n) * 0.02, n)
            ho, co = grid_cd(hp, n, 8, deadline, grid_size=30)
            ho, co = exact_cd(ho, n, 5, deadline, n_bisect=30)
            if co < best_c:
                best_h, best_c = ho, co
    # Phase 6: Final refinement
    if time.time() < deadline - 15:
        n = len(best_h)
        hf, cf = exact_cd(best_h, n, 10, deadline, n_bisect=45)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
