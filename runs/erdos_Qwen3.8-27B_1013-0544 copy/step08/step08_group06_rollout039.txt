# SLSQP helper
def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = c.max()
    shifted = alpha * (c - mx)
    mx2 = shifted.max()
    lse = mx2 + np.log(np.sum(np.exp(shifted - mx2)))
    return (mx + lse/alpha) * (2.0/n)

def sum_con(x, n):
    return x.sum() - n/2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter=500):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=[(0,1)]*n,
                          constraints=[{'type':'eq','fun':sum_con,'jac':sum_con_jac,'args':(n,)}],
                          options={'maxiter':maxiter,'ftol':1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except:
            pass
    return best_h, best_c

            if tot < 1e-15:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_cd(h, n, max_passes=5000, deadline=None, grid_size=100):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        if p % 2 == 0:
            idx_order = range(n)
        else:
            idx_order = range(n - 1, -1, -1)
        improved = False
        for idx in idx_order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-12:
                lo_d, hi_d = 0.0, 1.0
            elif h0 > 1.0 - 1e-12:
                continue
            else:
                lo_d, hi_d = -h0, 1.0 - h0
            if hi_d - lo_d < 1e-12:
                continue
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                t = float(c[:m_lo].max())
                if t > unaff_max:
                    unaff_max = t
            if m_hi < 2 * n - 2:
                t = float(c[m_hi + 1:].max())
                if t > unaff_max:
                    unaff_max = t
            dm = (ms == (n - 1))
            dpos = int(np.argmax(dm)) if dm.any() else -1
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if dpos >= 0:
                nv[:, dpos] -= d_grid[:, None] ** 2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            bi = int(np.argmin(vals))
            bd = float(d_grid[bi])
            bv = float(vals[bi])
            if 0 < bi < grid_size - 1:
                y0, y1, y2 = vals[bi - 1], vals[bi], vals[bi + 1]
                x0, x1, x2 = d_grid[bi - 1], d_grid[bi], d_grid[bi + 1]
                denom = y0 * (x1 - x2) + y1 * (x2 - x0) + y2 * (x0 - x1)
                if abs(denom) > 1e-15:
                    xv = 0.5 * (y0 * (x1 - x2) + y1 * (x2 - x0) + y2 * (x0 - x1)) / denom
                    if x0 < xv < x2:
                        nv1 = c_aff + xv * a
                        if dpos >= 0:
                            nv1 = nv1.copy()
                            nv1[dpos] -= xv * xv
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < bv - 1e-14:
                            bv, bd = v1, xv
            if abs(bd) > 1e-10 and bv < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + bd, 0.0, 1.0))
                c[ms] = c_aff + bd * a
                if dpos >= 0:
                    c[ms[dpos]] -= bd * bd
                nm = max(unaff_max, float(c[ms].max()))
                if nm < current_c5:
                    current_c5 = nm
                improved = True
        if not improved:
            break
        if (p + 1) % 300 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = c.max()
    shifted = alpha * (c - mx)
    mx2 = shifted.max()
    lse = mx2 + np.log(np.sum(np.exp(shifted - mx2)))
    return (mx + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter=300):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}],
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 500
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h, best_c = h.copy(), compute_c5(h, n)

    # Phase 1: SLSQP warm-up
    if time.time() < deadline - 700:
        h1, c1 = slsqp_optimize(h, n, [1000, 10000, 100000], maxiter=300)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Coarse coordinate descent for fast progress
    h2, c2 = coord_cd(best_h, n, max_passes=2000, deadline=t0 + 150, grid_size=60)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Medium grid refinement
    h3, c3 = coord_cd(best_h, n, max_passes=5000, deadline=t0 + 450, grid_size=120)
    if c3 < best_c:
        best_h, best_c = h3, c2 if c3 >= best_c else c3

    # Phase 4: Random restarts - try perturbations
    for restart in range(6):
        if time.time() > deadline - 300:
            break
        sigma = 0.02 * (1.0 - 0.1 * restart)
        perturbed = best_h + rng.randn(n) * sigma
        perturbed = project_feasible(perturbed, n)
        hr, cr = coord_cd(perturbed, n, max_passes=1500, deadline=None, grid_size=100)
        if cr < best_c - 1e-10:
            best_h, best_c = hr, cr

    # Phase 5: Try different n values via interpolation
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 600, 500]:
        if time.time() > deadline - 150:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        hn, cn = coord_cd(h_int, n_try, max_passes=3000, deadline=None, grid_size=110)
        if cn < best_c - 1e-10:
            best_h, best_c = hn, cn
            x_orig = x_new

    # Phase 6: Fine grid final polish
    h6, c6 = coord_cd(best_h, len(best_h), max_passes=10000, deadline=deadline - 5, grid_size=200)
    if c6 < best_c:
        best_h, best_c = h6, c6

    # Phase 7: Ultra-fine on best
    if time.time() < deadline - 3:
        h7, c7 = coord_cd(best_h, len(best_h), max_passes=5000, deadline=deadline, grid_size=250)
        if c7 < best_c:
            best_h, best_c = h7, c7

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
