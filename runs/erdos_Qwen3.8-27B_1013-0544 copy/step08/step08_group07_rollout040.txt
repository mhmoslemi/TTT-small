```python
def minimize_1d_convex(func, a, b, tol=1e-10):
    """Minimize a convex function on [a, b] using ternary search."""
    for _ in range(100):
        m1 = a + (b-a)/3
        m2 = b - (b-a)/3
        if func(m1) < func(m2):
            b = m2
        else:
            a = m1
        if b - a < tol:
            break
    return (a+b)/2
```

```python
def fourier_to_h(params, n):
    """Convert Fourier params to h values."""
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, len(params)//2 + 1):
        a = params[2*k-2]
        b = params[2*k-1]
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    return h

def fourier_obj(params, n, alpha=100):
    h = np.clip(fourier_to_h(params, n), 0, 1)
    h = project_feasible(h, n)
    c = np.correlate(h, 1-h, mode='full')
    m = c.max()
    lse = m + np.log(np.sum(np.exp(alpha*(c - m)))) / alpha
    return lse * (2.0/n)
```

```python
def compute_grad_for_p(h, p, n):
    """Compute J[m, p] = dc[m]/dh[p] for all m."""
    J = np.zeros(2*n-1)
    # First term: (1-h[p-k]) for valid k
    # k >= 0: k in [0, p], m = n-1+k
    for k in range(0, p+1):
        m = n - 1 + k
        J[m] += (1.0 - h[p-k])
    # k < 0: k in [p-n+1, -1], m = n-1+k
    k_lo = max(0 - (n-1), p - n + 1)  # = p-n+1 (since p < n)
    for k in range(k_lo, 0):
        m = n - 1 + k
        J[m] += (1.0 - h[p-k])
    # Second term: -h[p+k] for valid k
    # k >= 0: k in [0, n-1-p], m = n-1+k
    for k in range(0, n - p):
        m = n - 1 + k
        J[m] -= h[p+k]
    # k < 0: k in [-p, -1], m = n-1+k
    for k in range(-p, 0):
        m = n - 1 + k
        J[m] -= h[p+k]
    return J
```

```python
            # Use grid search with refinement
            d_lo = -h0
            d_hi = 1.0 - h0

                c_new[n-1] -= d*d

                    best_val = val

                    val = c_new.max()

                        best_d = d

                    h[p] = h0 + best_d
                    c += best_d * J
                    c[n-1] -= best_d * best_d
                    c_max = c.max()
                    improved = True
        
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        if not improved:
            break
    
    return h, best_c
```

```python
def compute_J(h, n):
    """Compute gradient matrix J[m,p] = dc[m]/dh[p]. Shape (2n-1, n)."""
    J = np.zeros((2*n-1, n))
    # k=0: J[n-1, :] = 1 - 2*h
    J[n-1, :] = 1.0 - 2.0*h
    # k>0
    for k in range(1, n):
        # m_pos = n-1+k
        # First term: J[n-1+k, k:] += (1-h[:n-k])
        # Second term: J[n-1+k, :n-k] += -h[k:]
        J[n-1+k, k:] += (1.0 - h[:n-k])
        J[n-1+k, :n-k] += -h[k:]
        # m_neg = n-1-k
        # First term: J[n-1-k, :n-k] += (1-h[k:])
        # Second term: J[n-1-k, k:] += -h[:n-k]
        J[n-1-k, :n-k] += (1.0 - h[k:])
        J[n-1-k, k:] += -h[:n-k]
    return J
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_J(h, n):
    J = np.zeros((2 * n - 1, n))
    J[n - 1, :] = 1.0 - 2.0 * h
    for k in range(1, n):
        J[n - 1 + k, k:] += (1.0 - h[:n - k])
        J[n - 1 + k, :n - k] += -h[k:]
        J[n - 1 - k, :n - k] += (1.0 - h[k:])
        J[n - 1 - k, k:] += -h[:n - k]
    return J

def coord_descent(h, n, max_passes=300, deadline=None, grid_size=80):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_passes):

            break

        c_max = float(c.max())
        J = compute_J(h, n)
        improved = False
        c_nd = c.copy()
        c_nd[n - 1] = -1e18
        for p in range(n):
            if deadline and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            Jp = J[:, p]
            d_lo, d_hi = -h0, 1.0 - h0
            d_grid = np.linspace(d_lo, d_hi, grid_size)
            vals_lin = (c_nd[None, :] + d_grid[:, None] * Jp[None, :]).max(axis=1)
            vals_diag = c[n - 1] + d_grid * Jp[n - 1] - d_grid ** 2
            vals = np.maximum(vals_lin, vals_diag)
            idx = int(vals.argmin())
            if vals[idx] >= c_max - 1e-13:
                continue
            lo = max(d_lo, d_grid[max(0, idx - 2)])
            hi = min(d_hi, d_grid[min(grid_size - 1, idx + 2)])
            d_fine = np.linspace(lo, hi, grid_size)
            vals_lin_f = (c_nd[None, :] + d_fine[:, None] * Jp[None, :]).max(axis=1)
            vals_diag_f = c[n - 1] + d_fine * Jp[n - 1] - d_fine ** 2
            vals_f = np.maximum(vals_lin_f, vals_diag_f)
            idx_f = int(vals_f.argmin())
            best_d = float(d_fine[idx_f])
            best_val = float(vals_f[idx_f])
            if best_val < c_max - 1e-13:
                h[p] = h0 + best_d
                c += best_d * Jp
                c[n - 1] -= best_d * best_d
                c_nd[n - 1] = -1e18
                c_max = best_val
                improved = True
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif not improved:
            break
    return h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n) + 1e5 * (x.sum() - n / 2.0) ** 2

def lbfgs_opt(h, n, alphas, maxiter=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline and time.time() > deadline - 20:
            break
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def fourier_h(params, n, x):
    h_f = np.full(n, params[0])
    for k in range(1, len(params) // 2 + 1):
        if 2 * k < len(params):
            h_f += params[2 * k - 1] * np.cos(np.pi * k * x)
        if 2 * k < len(params):
            h_f += params[2 * k] * np.sin(np.pi * k * x)
    return project_feasible(h_f, n)

def fourier_initial(h, n, M):
    x = np.linspace(0, 2, n, endpoint=False)
    params = np.zeros(2 * M + 1)
    params[0] = h.mean()
    for k in range(1, M + 1):
        params[2 * k - 1] = 2.0 / n * np.sum(h * np.cos(np.pi * k * x))
        params[2 * k] = 2.0 / n * np.sum(h * np.sin(np.pi * k * x))
    return params
```
