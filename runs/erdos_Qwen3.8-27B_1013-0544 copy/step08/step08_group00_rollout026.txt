def nesterov_smoothed(h, n, alphas, max_iter, deadline):
    """Nesterov-accelerated projected gradient on smoothed max."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m = 2*n - 1  # number of correlation lags
    
    for alpha in alphas:
        t_k = 1.0  # Nesterov momentum parameter
        x = h.copy()  # current point
        y = h.copy()  # extrapolated point
        step = 0.1  # initial step size
        
        for it in range(max_iter):
            if time.time() > deadline:
                break
            
            # Compute weights
            c = np.correlate(y, 1-y, mode='full') * dx
            w = np.exp(alpha * (c - c.max()))
            w /= w.sum()
            
            # Compute gradient at y
            T1 = np.correlate(w, 1-y, mode='full')[:n]
            T2 = np.correlate(w[::-1], y, mode='full')[n-1:n-1+n]
            grad = dx * (T1 - T2)
            
            # Nesterov update

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def compute_subgradient(h, n, top_frac=0.05):
    """Compute averaged subgradient of max_k c_k using top correlations."""
    dx = 2.0 / n
    m = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_k = max(1, int(top_frac * m))
    top_idx = np.argpartition(c, -top_k)[-top_k:]
    w = np.zeros(m)
    w[top_idx] = 1.0 / top_k
    # Gradient via convolution
    T1 = np.correlate(w, 1.0 - h, mode='full')[:n]
    T2 = np.correlate(w[::-1], h, mode='full')[n-1:n-1+n]
    return dx * (T1 - T2), c

def nesterov_subgrad(h, n, deadline, top_frac=0.05):

    m = 2 * n - 1

    best_h = h.copy()
    x = h.copy()
    v = np.zeros(n)
    beta = 0.7
    step = 0.05
    t = 1.0
    y = h.copy()
    
    for it in range(50000):
        if time.time() > deadline:
            break
        
        c = np.correlate(y, 1.0 - y, mode='full')
        c_scaled = c * (2.0 / n)
        c_max = float(c_scaled.max())
        if c_max < best_c:

        # Subgradient
        top_k = max(1, int(top_frac * m))
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        w = np.zeros(m)
        w[top_idx] = 1.0 / top_k
        T1 = np.correlate(w, 1.0 - y, mode='full')[:n]
        T2 = np.correlate(w[::-1], y, mode='full')[n-1:n-1+n]
        grad = (2.0 / n) * (T1 - T2)
        
        # Nesterov
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0

        x = y_new
        y = y_new

def slsqp_smooth(h, n, alphas, maxiter_each, deadline):
    """SLSQP with softmax-smoothed objective."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            shifted = alpha * (c - m)
            mx = shifted.max()
            lse = mx + np.log(np.sum(np.exp(shifted - mx)))
            return (m + lse / alpha) * dx
        
        def obj_grad(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            shifted = alpha * (c - m)
            mx = shifted.max()
            w = np.exp(shifted - mx)
            w /= w.sum()
            T1 = np.correlate(w, 1.0 - x, mode='full')[:n]
            T2 = np.correlate(w[::-1], x, mode='full')[n-1:n-1+n]
            return dx * (T1 - T2)
        
        cons = [
            {'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}
        ]
        try:
            res = minimize(obj, best_h, args=(), method='SLSQP',
                           bounds=bounds, constraints=cons, jac=obj_grad,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def compute_gradient(h, w, n):
    """Grad of sum(w[m]*c[m]) w.r.t. h using FFT-based correlate/convolve."""
    dx = 2.0 / n
    T1 = np.correlate(w, 1.0 - h, mode='full')[:n]
    T2 = np.convolve(w, h, mode='full')[n - 1:n - 1 + n]
    return dx * (T1 - T2)

def slsqp_with_grad(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for alpha in alphas:
        if time.time() > deadline:
            break

        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            shifted = alpha * (c - m)
            mx = shifted.max()
            lse = mx + np.log(np.sum(np.exp(shifted - mx)))
            return (m + lse / alpha) * dx

        def obj_grad(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            shifted = alpha * (c - m)
            mx = shifted.max()
            w = np.exp(shifted - mx)
            w /= w.sum()
            return compute_gradient(x, w, n)

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, jac=obj_grad,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def coord_cd_exact(h, n, max_passes, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    m_total = 2 * n - 1
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        improved = False
        order = np.argsort(h * (1.0 - h))[::-1]
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p, p)
            m_hi = min(m_total - 1, n - 1 + p, p + n - 1)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c_base = c[ms]
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < m_total - 1:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            L_lo, L_hi = unaff, cur_max
            for _ in range(50):
                if L_hi - L_lo < 1e-15:
                    break
                L = (L_lo + L_hi) / 2.0
                x_lo, x_hi = 0.0, 1.0
                feasible = True
                neg_m = a < -1e-15
                pos_m = a > 1e-15
                zero_m = np.abs(a) <= 1e-15
                if np.any(zero_m) and np.any(c_base[zero_m] > L):
                    feasible = False
                elif feasible and np.any(neg_m):
                    x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
                if feasible and np.any(pos_m):
                    x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
                if x_lo > x_hi + 1e-14:
                    feasible = False
                if feasible and has_diag:
                    di = diag_idx[0]
                    cd = c_base[di]
                    ad = a[di]
                    disc = ad * ad - 4.0 * (L - cd)
                    if disc >= 0:
                        r1 = (ad - np.sqrt(disc)) / 2.0
                        r2 = (ad + np.sqrt(disc)) / 2.0
                        if r1 < r2:
                            if x_lo < r1 and x_hi > r2:
                                x_hi_new = min(x_hi, r1)
                                x_lo_new = max(x_lo, r2)
                                if x_lo_new > x_hi_new:
                                    feasible = False
                                else:
                                    x_lo, x_hi = x_lo_new, x_hi_new
                            elif x_lo >= r1 and x_hi <= r2:
                                feasible = False
                if feasible:
                    L_hi = L
                else:
                    L_lo = L
            L = L_hi
            x_lo, x_hi = 0.0, 1.0
            neg_m = a < -1e-15
            pos_m = a > 1e-15
            zero_m = np.abs(a) <= 1e-15
            if np.any(zero_m) and np.any(c_base[zero_m] > L):
                continue
            if np.any(neg_m):
                x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
            if np.any(pos_m):
                x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
            if x_lo > x_hi:
                continue
            if has_diag:
                di = diag_idx[0]
                cd = c_base[di]
                ad = a[di]
                disc = ad * ad - 4.0 * (L - cd)
                if disc >= 0:
                    r1 = (ad - np.sqrt(disc)) / 2.0
                    r2 = (ad + np.sqrt(disc)) / 2.0
                    if x_lo < r1:
                        x_hi = min(x_hi, r1)
                    elif x_hi > r2:
                        x_lo = max(x_lo, r2)
                    elif x_lo >= r1 and x_hi <= r2:
                        continue
            x_new = float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))
            if abs(x_new - h0) > 1e-12:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < best_c:
            best_c = nc
    return h, best_c

def subgrad_nesterov(h, n, deadline, top_frac=0.03, max_iter=30000):
    dx = 2.0 / n
    m = 2 * n - 1
    best_c = compute_c5(h, n)
    best_h = h.copy()
    x = h.copy()
    y = h.copy()
    v = np.zeros(n)
    beta = 0.6
    step = 0.03
    t = 1.0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(y, 1.0 - y, mode='full')
        cm = float(c.max() * dx)
        if cm < best_c:
            best_c = cm
            best_h = y.copy()
        top_k = max(1, int(top_frac * m))
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        w = np.zeros(m)
        w[top_idx] = 1.0 / top_k
        grad = compute_gradient(y, w, n)
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mom = (t - 1.0) / t_next
        y_new = x + mom * (x - y) - step * grad
        y_new = project_feasible(y_new, n)
        v = beta * v + (1.0 - beta) * grad
        x = y_new
        y = y_new
        t = t_next
        if it % 2000 == 1999:
            step *= 0.85
    return best_h, best_c

def try_rounding(h, n):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(n):
        candidates = [0.0, 1.0]
        for cv in candidates:
            if abs(h[p] - cv) > 0.3:
                continue
            test = h.copy()
            test[p] = cv
            test = project_feasible(test, n)
            tc = compute_c5(test, n)
            if tc < best_c - 1e-12:
                best_c = tc
                best_h = test.copy()
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:

        n = 500

    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with analytic gradient
    h1, c1 = slsqp_with_grad(best_h, n, [300, 1000, 5000, 30000], 600, deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Nesterov subgradient
    if time.time() < deadline - 80:
        h2, c2 = subgrad_nesterov(best_h, n, deadline - 50, top_frac=0.02)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Exact coordinate descent
    if time.time() < deadline - 40:
        h3, c3 = coord_cd_exact(best_h, n, 100, deadline - 10)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Rounding
    if time.time() < deadline - 20:
        h4, c4 = try_rounding(best_h, n)
        if c4 < best_c:
            best_h, best_c = h4, c4
        h3b, c3b = coord_cd_exact(best_h, n, 30, deadline - 5)
        if c3b < best_c:
            best_h, best_c = h3b, c3b

    # Phase 5: Try other n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [350, 800, 550]:
        if time.time() > deadline - 120:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = slsqp_with_grad(h_int, n_try, [3000, 20000], 300, deadline)
        if time.time() < deadline - 30:
            h_opt, c_opt = coord_cd_exact(h_opt, n_try, 60, deadline - 5)
        h_opt, c_opt = try_rounding(h_opt, n_try)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = np.linspace(0, 2, n_try)

    # Phase 6: Multi-restart perturbation
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        n_restarts = max(2, int((deadline - time.time() - 60) / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 60:
                break
            s = 0.01 + 0.05 * (r / max(1, n_restarts - 1))
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            h_o, c_o = slsqp_with_grad(hp, n_cur, [5000, 50000], 200, deadline)
            if time.time() < deadline - 10:

            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 7: Final exact CD
    if time.time() < deadline - 5:
        h7, c7 = coord_cd_exact(best_h, len(best_h), 50, deadline)
        if c7 < best_c:
            best_h, best_c = h7, c7

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
