import numpy as np
from scipy.optimize import minimize
import time


def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-14:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-13:
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)


def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)


def soft_max_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    val = (m + np.log(z) / alpha) * dx
    grad = (A - B)[s:s + n] * dx
    return val, grad


def proj_grad_optimize(h, n, alphas, max_iters, deadline):
    h = project_simplex(h, n).copy()
    best = h.copy()
    best_c = compute_c5(best, n)
    best_v = soft_max_val_grad(best, n, alphas[0])[0]
    v = np.zeros(n)
    for ai, alpha in enumerate(alphas):
        if time.time() > deadline:
            break
        lr = 0.5 / (alpha * (2.0 / n))
        mom = 0.9
        for it in range(max_iters):
            if time.time() > deadline:
                break
            val, grad = soft_max_val_grad(h, n, alpha)
            v = mom * v - lr * grad
            h_new = project_simplex(h + v, n)
            val_new = soft_max_val_grad(h_new, n, alpha)[0]
            if val_new < val:
                h = h_new
                lr = min(lr * 1.1, 5.0 / (alpha * (2.0 / n)))
            else:
                v = 0.9 * v
                lr *= 0.5
                h_new = project_simplex(h + v, n)

                    h = h_new
            c_true = compute_c5(h, n)
            if c_true < best_c - 1e-10:
                best_c = c_true
                best = h.copy()
    return best, best_c

        return None, np.inf

    b = np.zeros(n, dtype=float)
    idx = np.argpartition(h, target)[:target]
    b[idx] = 1.0
    best = b.copy()

    bcm = float(bc.max())
    cur = b.copy()
    c = np.correlate(cur, 1.0 - cur, mode="full")
    cm = float(c.max())
    dx = 2.0 / n
    ones = np.where(cur == 1.0)[0]
    zeros = np.where(cur == 0.0)[0]
    T = t0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if rnd % 100 == 0:
            T *= cool
            if T < 1e-6:
                T = 1e-6
        i = ones[int(rng.randint(len(ones)))]
        j = zeros[int(rng.randint(len(zeros)))]
        if i == j:
            continue
        om = 1.0 - cur[::-1]
        d = c.copy()
        d[i:i + n] -= om
        st_i = diag_off(i, n)
        d[st_i:st_i + n] += cur
        d[j:j + n] += om
        st_j = diag_off(j, n)
        d[st_j:st_j + n] -= cur
        d[n - 1] -= 2.0
        d[n - 1 + i - j] += 1.0
        d[n - 1 + j - i] += 1.0
        nm = float(d.max())
        delta = nm - cm
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-9)):
            c = d
            cm = nm
            cur = cur.copy()
            cur[i], cur[j] = cur[j], cur[i]
            oi = np.where(cur == 1.0)[0]
            oz = np.where(cur == 0.0)[0]
            ones, zeros = oi, oz
            if cm < bcm:
                bcm = cm
                best = cur.copy()
    return best, bcm * dx


def diag_off(p, n):

        h1, c1 = proj_grad_optimize(best_h, n_best, [10000, 100000], 300, t_phase2)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h1, c1 = coord_cd(best_h, n_best, max_passes=8, deadline=t_phase2, grid=48, refine=16)
        if c1 < best_c:
            best_h, best_c = h1, c1

    t_phase3 = min(deadline - 60.0, t0 + 0.85 * budget_s)
    if n_best % 2 == 0 and time.time() < t_phase3:
        hb, cb = binary_sa(best_h, n_best, rng, t_phase3, t0=2.0, cool=0.985)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb
            t_ext = min(deadline - 40.0, time.time() + 60.0)
            h1, c1 = proj_grad_optimize(best_h, n_best, [50000, 200000], 200, t_ext)
            if c1 < best_c:
                best_h, best_c = h1, c1
            h1, c1 = coord_cd(best_h, n_best, max_passes=4, deadline=t_ext, grid=48, refine=16)
            if c1 < best_c:
                best_h, best_c = h1, c1

    cands = [600, 400, 500, 450, 550, 700]
    tried = set([n_best])
    for nt in cands:
        if time.time() > deadline - 50.0:
            break
        if nt in tried or nt < 200 or nt > 1000 or nt % 2:
            continue
        tried.add(nt)
        ht = transfer_h(best_h, n_best, nt)
        cv0 = compute_c5(ht, nt)
        chunk = min(deadline - 30.0, time.time() + 45.0)
        h1, c1 = proj_grad_optimize(ht, nt, [5000, 50000], 100, chunk)
        h1, c1 = coord_cd(h1, nt, max_passes=3, deadline=chunk, grid=32, refine=10)
        if c1 < best_c:
            best_h, best_c, n_best = h1, c1, nt

    tf = min(deadline - 5.0, time.time() + 100.0)
    if time.time() < tf:
        h1, c1 = proj_grad_optimize(best_h, n_best, [50000, 500000], 200, tf)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h1, c1 = coord_cd(best_h, n_best, max_passes=4, deadline=tf, grid=48, refine=16)
        if c1 < best_c:
            best_h, best_c = h1, c1

    if n_best % 2 == 0 and time.time() < deadline - 10.0:
        hb, cb = binary_sa(best_h, n_best, rng, deadline, t0=0.5, cool=0.99)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb

    final_h = project_simplex(best_h, n_best)
    final_c = float(compute_c5(final_h, n_best))
    return final_h, final_c, int(n_best)
