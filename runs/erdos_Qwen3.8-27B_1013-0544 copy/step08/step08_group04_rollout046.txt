import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 1e-14:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 1e-14:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-12:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-12:
        idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def triangle_h(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(1.0 - np.abs(x - 1.0), n)

def basis_h(n, rng, modes=10):
    x = (np.arange(n) + 0.5) * (2.0 / n) - 1.0
    h = 0.5 * np.ones(n)
    scale = rng.uniform(0.15, 0.35)
    for k in range(1, modes + 1):
        h += rng.uniform(-scale / k, scale / k) * np.cos(k * np.pi * x)
    return project_feasible(h, n)

def binary_from_h(h):
    n = len(h)
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    h2 = np.zeros(n)
    h2[idx] = 1.0
    return h2

def _eval_coord(x, cseg, a, qconst, unaff):
    if a.size:
        vmax = float(np.max(cseg + a * x))
    else:
        vmax = 0.0
    val = max(unaff, qconst + x * (1.0 - x))
    return max(val, vmax)

def _check_feas(t, unaff, qconst, zmax, a_pos, a_neg, b_pos, b_neg):
    if t < unaff - 1e-12 or t < qconst - 1e-12 or t < zmax - 1e-12:
        return False
    ll, lh = 0.0, 1.0
    if a_neg.size:
        v = float(np.max((t - b_neg) / a_neg))
        if v > ll:
            ll = v
    if a_pos.size:
        v = float(np.min((t - b_pos) / a_pos))
        if v < lh:
            lh = v
    if ll > 1.0 + 1e-10 or lh < -1e-10:
        return False
    g = t - qconst
    if g > 0.25:
        return max(0.0, ll) <= min(1.0, lh) + 1e-10
    elif g >= -1e-12:
        if g < 0:
            g = 0.0
        disc = max(0.0, 1.0 - 4.0 * g)
        r1 = 0.5 - 0.5 * np.sqrt(disc)
        r2 = 0.5 + 0.5 * np.sqrt(disc)
        if max(0.0, ll) <= min(r1, lh) + 1e-10:
            return True
        return max(r2, ll) <= min(1.0, lh) + 1e-10
    return False

def _opt_coord(hp, cseg, a, qconst, unaff):
    m = a.size
    b = cseg - a * hp
    pos = a > 1e-12
    neg = a < -1e-12
    zero = ~(pos | neg)
    zmax = float(np.max(b[zero])) if zero.any() else -1e100
    cm = float(np.max(cseg)) if m else 0.0
    low = max(0.0, unaff, qconst, zmax)
    high = max(low + 1e-11, unaff, qconst, cm, qconst + hp * (1.0 - hp)) + 1e-8
    a_pos, a_neg = a[pos], a[neg]
    b_pos, b_neg = b[pos], b[neg]
    for _ in range(20):
        mid = 0.5 * (low + high)
        if _check_feas(mid, unaff, qconst, zmax, a_pos, a_neg, b_pos, b_neg):
            high = mid
        else:
            low = mid
    t = high
    cands = [0.0, 1.0, hp, 0.5]
    if t >= unaff - 1e-10 and t >= qconst - 1e-10 and t >= zmax - 1e-10:
        ll, lh = 0.0, 1.0
        if a_neg.size:
            v = float(np.max((t - b_neg) / a_neg))
            if v > ll:
                ll = v
        if a_pos.size:
            v = float(np.min((t - b_pos) / a_pos))
            if v < lh:
                lh = v
        if ll <= 1.0 + 1e-10 and lh >= -1e-10:
            g = t - qconst
            if g > 0.25:
                branches = [(max(0.0, ll), min(1.0, lh))]
            elif g >= -1e-12:
                if g < 0:
                    g = 0.0
                disc = max(0.0, 1.0 - 4.0 * g)
                r1 = 0.5 - 0.5 * np.sqrt(disc)
                r2 = 0.5 + 0.5 * np.sqrt(disc)
                branches = [(0.0, r1), (r2, 1.0)]
            else:
                branches = []
            for Lq, Uq in branches:
                L = max(Lq, ll, 0.0)
                U = min(Uq, lh, 1.0)
                if L <= U + 1e-10:
                    cands.extend([L, U, 0.5 * (L + U)])
    best_x, best_f = hp, _eval_coord(hp, b, a, qconst, unaff)
    for x in cands:
        if x < 0.0 or x > 1.0:
            continue
        f = _eval_coord(x, b, a, qconst, unaff)
        if f < best_f - 1e-13:
            best_f, best_x = f, x
    if best_f > t + 1e-4 and m:
        xg = np.linspace(0.0, 1.0, 24)
        vals = np.maximum(np.maximum(np.max(b[None, :] + xg[:, None] * a[None, :], axis=1),
                                      qconst + xg * (1.0 - xg)), unaff)
        idx = int(np.argmin(vals))
        x, f = float(xg[idx]), float(vals[idx])
        if f < best_f:
            best_f, best_x = f, x
    return best_x, best_f

def cd_optimize(h, n, max_passes, deadline, proj_every=8):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h, best_c = h.copy(), (2.0 / n) * cur
    center, L = n - 1, 2 * n - 1
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = list(range(n))
        for p in order:
            if time.time() > deadline:
                break
            hp = float(h[p])
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            s_arr = center - np.arange(lo, hi + 1, dtype=int)
            j_idx = p + s_arr
            i_idx = p - s_arr
            a = np.zeros(hi - lo + 1, dtype=float)
            vj = (j_idx >= 0) & (j_idx < n)
            vi = (i_idx >= 0) & (i_idx < n)
            if np.any(vj):
                a[vj] += 1.0 - h[j_idx[vj]]
            if np.any(vi):
                a[vi] -= h[i_idx[vi]]
            off = center - lo
            cseg = np.empty(hi - lo, dtype=float)
            asg = np.empty(hi - lo, dtype=float)
            if off:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center + 1:hi + 1]
                asg[off:] = a[off + 1:]
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(c[hi + 1:].max()))
            bx, bf = _opt_coord(hp, cseg, asg, qconst, unaff)
            if bf < cur - 1e-12:
                dxv = bx - hp
                if off:
                    c[lo:center] += asg[:off] * dxv
                if hi > center:
                    c[center + 1:hi + 1] += asg[off:] * dxv
                c[center] = qconst + bx * (1.0 - bx)
                h[p] = bx
                cur = bf
                improved = True
            if (p + 1) % proj_every == 0:
                h = project_feasible(h, n)
                c = np.correlate(h, 1.0 - h, mode='full')
                cur = float(c.max())
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        cval = (2.0 / n) * cur
        if cval < best_c - 1e-14:
            best_c, best_h = cval, h.copy()
        if not improved:
            break
    return best_h, best_c

def subgrad_polish(h, n, steps, deadline):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    cur = float(c.max())
    best_h, best_c = h.copy(), (2.0 / n) * cur
    lr = 0.05
    for it in range(steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        tol = cur * 0.05
        active = np.where(c >= cur - tol)[0]
        g = np.zeros(n)
        for m in active:
            s = m - center
            w = (c[m] - (cur - tol)) / max(tol, 1e-15)
            for p in range(n):
                jp, ip = p + s, p - s
                gv = 0.0
                if 0 <= jp < n:
                    gv += (1.0 - h[jp])
                if 0 <= ip < n:
                    gv -= h[ip]
                if s == 0 and p == ip == jp:
                    gv = 1.0 - 2.0 * h[p]
                g[p] += w * gv
        g /= max(1.0, float(np.linalg.norm(g)))
        h = project_feasible(h - lr * g * n / 500.0, n)
        lr *= 0.97
        cval = (2.0 / n) * float(np.max(np.correlate(h, 1.0 - h, mode='full')))
        if cval < best_c - 1e-14:
            best_c, best_h = cval, h.copy()
    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    z = alpha * c
    z -= z.max()
    return (z.max() + np.log(np.sum(np.exp(z)))) / alpha * (2.0 / n)

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    center = n - 1
    z = alpha * c
    z -= z.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * x)
        elif s > 0:
            g[:n - s] += wm * (1.0 - x[s:])
            g[s:] -= wm * x[:n - s]
        else:
            r = -s
            g[r:] += wm * (1.0 - x[:n - r])
            g[:n - r] -= wm * x[r:]
    return (2.0 / n) * g

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h = project_feasible(h, n)
    best_c = compute_c(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - 0.5 * n,
             'jac': lambda x, n=n: np.ones(n)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_jac, bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
            x = project_feasible(res.x, n)
            c = compute_c(x, n)
            if c < best_c:
                best_h, best_c = x, c
        except Exception:
            pass
    return best_h, best_c
