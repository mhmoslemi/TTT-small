    """Compute A[k] = d(r[k])/d(h[p]) for given p and current h."""
    A = np.ones(2*n-1)  # base: derivative of sum_i h[i] w.r.t. h[p] is 1

**Alternative: Use CVXPY with a quadratic constraint**

The problem is:
minimize z
subject to:

  0 ≤ h ≤ 1

The constraint r[k] ≤ z is:
S - autocorr[k] ≤ z
autocorr[k] ≥ S - z

The autocorrelation is a quadratic form: autocorr[k] = h^T A_k h where A_k is a symmetric matrix.

So the constraint is h^T A_k h ≥ S - z, which is a non-convex quadratic constraint (we need h^T A_k h to be above a threshold, and A_k may be indefinite).

This makes it a non-convex QCQP, which CVXPY can't handle directly.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_influence(p, h, n):
    A = np.ones(2 * n - 1)
    k_lo = max(0, n - 1 - p)
    k_hi = min(2 * n - 2, 2 * n - 2 - p)
    if k_lo <= k_hi:
        idx = p + np.arange(k_lo, k_hi + 1) - n + 1
        A[k_lo:k_hi + 1] -= h[idx]
    k_lo2 = max(0, p)
    k_hi2 = min(2 * n - 2, p + n - 1)
    if k_lo2 <= k_hi2:
        idx2 = p - np.arange(k_lo2, k_hi2 + 1) + n - 1
        A[k_lo2:k_hi2 + 1] -= h[idx2]
    return A

def coordinate_descent(h, n, max_passes=500, deadline=None, gs_iters=12):
    h = h.copy()
    nd = n - 1
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    gr = (np.sqrt(5.0) + 5.0) ** -1 * (np.sqrt(5.0) - 1.0)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    tmp = np.empty(2 * n - 1)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        cmax = float(c.max())
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            A = get_influence(p, h, n)
            c0 = c - h0 * A
            c0[nd] += h0 * h0
            a, b = 0.0, 1.0
            c1 = b - gr * (b - a)
            d1 = a + gr * (b - a)
            tmp[:] = c0 + c1 * A
            tmp[nd] -= c1 * c1
            fc1 = float(tmp.max())
            tmp[:] = c0 + d1 * A
            tmp[nd] -= d1 * d1
            fd1 = float(tmp.max())
            for _ in range(gs_iters):
                if fc1 < fd1:
                    b = d1; d1 = c1; fd1 = fc1
                    c1 = b - gr * (b - a)
                    tmp[:] = c0 + c1 * A
                    tmp[nd] -= c1 * c1
                    fc1 = float(tmp.max())
                else:
                    a = c1; c1 = d1; fc1 = fd1
                    d1 = a + gr * (b - a)
                    tmp[:] = c0 + d1 * A
                    tmp[nd] -= d1 * d1
                    fd1 = float(tmp.max())
            t_star = (a + b) * 0.5
            tmp[:] = c0 + t_star * A
            tmp[nd] -= t_star * t_star
            f_star = float(tmp.max())
            f0 = float(c0.max())
            tmp[:] = c0 + A
            tmp[nd] -= 1.0
            f1 = float(tmp.max())
            if f0 <= f_star and f0 <= f1:
                t_new = 0.0
            elif f1 <= f_star and f1 <= f0:
                t_new = 1.0
            else:
                t_new = t_star
            f_new = min(f0 if t_new == 0 else (f1 if t_new == 1 else f_star), f_star)
            if t_new == 0:
                f_new = f0
            elif t_new == 1:
                f_new = f1
            if f_new < cmax - 1e-13:
                delta = t_new - h0
                c += delta * A
                c[nd] -= (t_new * t_new - h0 * h0)
                h[p] = t_new
                cmax = f_new
                improved = True
        h = project_feasible(h, n)
        c_new = np.correlate(h, 1.0 - h, mode='full').astype(float)
        if not np.allclose(c, c_new, atol=1e-12):
            c = c_new
        if not improved:
            break
    return h, compute_c5(h, n)

def pnorm_slsqp(h, n, p_list, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for pv in p_list:
        if deadline is not None and time.time() > deadline:
            break
        def obj(x):
            cc = np.correlate(x, 1.0 - x, mode='full')
            return float(np.sum(cc ** pv)) ** (1.0 / pv) * (2.0 / n)
        def sfun(x):
            return x.sum() - n / 2.0
        def sjac(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_each, 'ftol': 1e-16, 'maxls': 40})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
