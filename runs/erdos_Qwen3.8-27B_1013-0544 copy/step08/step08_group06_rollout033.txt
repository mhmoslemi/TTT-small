import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_objective(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if alpha > 50000:
        obj = m * (2.0/n)
    else:
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        obj = (m + lse / alpha) * (2.0/n)
    obj += 100.0 * (x.sum() - n/2.0)**2
    return obj

def lse_gradient(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    cmax = c.max()
    shifted = alpha * (c - cmax)
    mx = shifted.max()
    W = np.exp(shifted - mx)
    W /= W.sum()
    
    m = 2*n - 1
    # Build W_mat: W_mat[j, i] = W[i - j + n - 1]
    offsets = np.arange(n)
    W_mat = np.array([W[n-1-j+offsets] for j in range(n)])
    
    P = W_mat @ x
    Q = W_mat.T @ x
    R = W_mat @ np.ones(n)
    
    grad = (2.0/n) * (R - P - Q) + 200.0 * (x.sum() - n/2.0)
    return grad

def lbfgs_optimize(h, n, alphas, maxiter=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(lse_objective, best_h, jac=lse_gradient, args=(n, alpha),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
