                    ndm[di] = False
                xl = 0.0
                xh = 1.0
                ok = True
                if np.any(ndm):
                    cb = c_base[ndm]
                    av = a[ndm]
                    if np.any(cb > L + 1e-16):
                        ok = False
                    else:
                        neg = av < -1e-15
                        pos = av > 1e-15
                        if np.any(neg):
                            xl = max(xl, float(np.max((L - cb[neg]) / av[neg])))
                        if np.any(pos):
                            xh = min(xh, float(np.min((L - cb[pos]) / av[pos])))
                if ok and has_d:
                    cd = float(c[nd])
                    bc = 1.0 - 2.0 * h0
                    disc = bc*bc + 4.0*(cd - L)
                    if disc < 0:
                        ok = False
                    else:
                        sq = np.sqrt(disc)
                        r1 = (bc - sq) / 2.0
                        r2 = (bc + sq) / 2.0
                        if not (xl <= r1 or xh >= r2):
                            ok = False
                if ok and xl > xh + 1e-14:
                    ok = False
                if ok:
                    L_hi = L
                else:
                    L_lo = L
            L = L_hi
            if L >= cm - 1e-14:
                continue
            ndm = np.ones(len(ms), dtype=bool)
            if has_d:
                ndm[di] = False
            xl = 0.0
            xh = 1.0
            if np.any(ndm):
                cb = c_base[ndm]
                av = a[ndm]
                neg = av < -1e-15
                pos = av > 1e-15
                if np.any(neg):
                    xl = max(xl, float(np.max((L - cb[neg]) / av[neg])))
                if np.any(pos):
                    xh = min(xh, float(np.min((L - cb[pos]) / av[pos])))
            if has_d:
                cd = float(c[nd])
                bc = 1.0 - 2.0 * h0
                disc = bc*bc + 4.0*(cd - L)
                if disc >= 0:
                    sq = np.sqrt(disc)
                    r1 = (bc - sq) / 2.0
                    r2 = (bc + sq) / 2.0
                    cands = []
                    if xl <= r1:
                        cands.append(min(max(xl, 0.0), r1))
                    if xh >= r2:
                        cands.append(max(xl, r2))
                    if cands:
                        best_d = min(cands, key=lambda x: abs(x))
                        nv = float(np.clip(h0 + best_d, 0.0, 1.0))
                        if abs(nv - h0) > 1e-13:
                            h[p] = nv
                            improved = True
            else:
                xl = max(xl, 0.0)
                xh = min(xh, 1.0)
                if xl <= xh:
                    if xl > 0:
                        d = xl
                    elif xh < 0:
                        d = xh
                    else:
                        d = 0.0
                    nv = float(np.clip(h0 + d, 0.0, 1.0))
                    if abs(nv - h0) > 1e-13:
                        h[p] = nv
                        improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def freeze_and_optimize(h, n, threshold, deadline):
    h = h.copy()
    frozen = (h < threshold) | (h > 1.0 - threshold)
    free = ~frozen
    if free.sum() < 5 or frozen.sum() < 5:
        return h, compute_c5(h, n)
    frozen_vals = h[frozen].copy()
    h[frozen] = np.where(h[frozen] < 0.5, 0.0, 1.0)
    h = project_feasible(h, n)
    h[frozen] = np.where(frozen_vals < 0.5, 0.0, 1.0)
    target = n / 2.0
    frozen_sum = h[frozen].sum()
    free_target = target - frozen_sum
    if free_target < 0 or free_target > free.sum():
        return h, compute_c5(h, n)
    h[free] = project_feasible(h[free], int(free_target * 2)) if False else h[free]
    scale = free_target / max(h[free].sum(), 1e-15)
    h[free] = np.clip(h[free] * scale, 0.0, 1.0)
    for _ in range(50):
        diff = free_target - h[free].sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h[free]
            tot = room.sum()
            if tot < 1e-15:
                break
            h[free] = np.clip(h[free] + (diff/tot)*room, 0.0, 1.0)
        else:
            tot = h[free].sum()
            if tot < 1e-15:
                break
            h[free] = np.clip(h[free] + (diff/tot)*h[free], 0.0, 1.0)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_init = 400
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: SLSQP warm start
    h1, c1 = run_slsqp(best_h, n_init, [500, 2000, 10000, 50000], 350)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Grid CD
    h2, c2 = cd_grid(best_h, n_init, 250, deadline, gs=60)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Exact CD
    if time.time() < deadline - 60:
        n = len(best_h)
        rem = deadline - time.time() - 40
        mp = max(10, int(rem / 3.5))
        hf, cf = cd_exact(best_h, n, mp, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Phase 4: Try other n
    for n_try in [500, 400, 600, 800]:
        if time.time() > deadline - 150:
            break
        if n_try == len(best_h):
            continue
        xo = np.linspace(0, 2, len(best_h))
        xn = np.linspace(0, 2, n_try)
        hi = project_feasible(np.interp(xn, xo, best_h), n_try)
        ho, co = run_slsqp(hi, n_try, [2000, 10000, 50000], 250)
        ho, co = cd_grid(ho, n_try, 100, deadline, gs=50)
        if co < best_c:
            best_h, best_c = ho, co
        if time.time() < deadline - 50:
            hf, cf = cd_exact(best_h, len(best_h), 20, deadline)
            if cf < best_c:
                best_h, best_c = hf, cf
    # Phase 5: Multi-restart with perturbation
    if time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 60
        n_restarts = max(2, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 60:
                break
            s = 0.01 + 0.06 * (r / max(1, n_restarts - 1))
            hp = project_feasible(best_h + rng.normal(0, s, n), n)
            ho, co = run_slsqp(hp, n, [5000, 50000], 200)
            ho, co = cd_grid(ho, n, 50, deadline, gs=40)
            if co < best_c:
                best_h, best_c = ho, co
    # Phase 6: Final exact CD polish
    if time.time() < deadline - 10:
        n = len(best_h)
        rem = deadline - time.time() - 5
        mp = max(5, int(rem / 3.0))
        hf, cf = cd_exact(best_h, n, mp, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
