def paired_swap_local_search(h, max_sweeps=100, tol=1e-10):
    n = len(h)
    h = h.copy()
    S = np.correlate(h, 1.0-h, mode="full")
    
    for sweep in range(max_sweeps):
        improved = False
        for j1 in range(n):
            # Compute delta_j1 once
            delta_j1 = compute_delta(h, j1)
            best_gain = 0
            best_j2 = -1
            best_delta = 0
            for j2 in range(n):
                if j2 == j1: continue
                delta_j2 = compute_delta(h, j2)
                g = delta_j1 - delta_j2
                # Find optimal delta
                d_min = max(-h[j1], h[j2]-1)
                d_max = min(1-h[j1], h[j2])
                if d_min >= d_max: continue
                opt_delta, new_max = ternary_search(S, g, d_min, d_max)
                old_max = S.max()
                gain = old_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_j2 = j2
                    best_delta = opt_delta
            if best_j2 >= 0 and best_gain > tol:
                h[j1] += best_delta
                h[best_j2] -= best_delta

def compute_delta_matrix(h):
    """Compute delta[j, m] for all j and all lags m."""
    n = len(h)
    delta = np.zeros((n, 2*n-1))
    for j in range(n):

        # lag valid when 0 ≤ j+lag ≤ n-1, i.e., -j ≤ lag ≤ n-1-j
        lag_lo = max(-(n-1), -j)
        lag_hi = min(n-1, n-1-j)
        idx = np.arange(lag_lo + (n-1), lag_hi + (n-1) + 1)
        h_idx = np.arange(j + lag_lo, j + lag_hi + 1)
        delta[j, idx] += (1.0 - h[h_idx])
        
        # Second term: -h[j-lag] for valid lag
        # lag valid when 0 ≤ j-lag ≤ n-1, i.e., j-(n-1) ≤ lag ≤ j
        lag_lo2 = max(-(n-1), j-(n-1))

        delta[j, idx2] -= h[h_idx2]

    """Find delta in [d_min, d_max] minimizing max(S + delta*g)."""
    for _ in range(50):

        d2 = d_max - (d_max - d_min) / 3.0

        if f1 < f2:

        else:
            d_min = d1
    d_opt = (d_min + d_max) / 2.0
    f_opt = np.max(S + d_opt * g)
    return d_opt, f_opt

def compute_delta_matrix(h):
    n = len(h)
    nlag = 2 * n - 1

    if cnt > 0:

```python
def find_best_swap(h, j1, delta_mat, S):
    """Find the best (j2, delta) for swapping with j1."""
    n = len(h)
    # G[j2, m] = delta_mat[j1, m] - delta_mat[j2, m]
    G = delta_mat[j1][np.newaxis, :] - delta_mat  # shape (n, 2n-1)
    
    # Bounds
    dlo = np.maximum(-h[j1], h - 1.0)  # shape (n,)
    dhi = np.minimum(1.0 - h[j1], h)   # shape (n,)
    
    # Mark infeasible
    feasible = dlo < dhi
    feasible[j1] = False  # exclude self
    
    if not np.any(feasible):
        return -1, 0.0, np.inf
    
    # Parallel ternary search
    dlo_v = dlo.copy()
    dhi_v = dhi.copy()
    
    for _ in range(35):
        m1 = dlo_v + (dhi_v - dlo_v) / 3.0
        m2 = dhi_v - (dhi_v - dlo_v) / 3.0
        f1 = np.max(S[np.newaxis, :] + m1[:, None] * G, axis=1)
        f2 = np.max(S[np.newaxis, :] + m2[:, None] * G, axis=1)
        # Where f1 < f2, shrink from the right
        shrink_right = f1 < f2
        dhi_v[shrink_right] = m2[shrink_right]
        dlo_v[~shrink_right] = m1[~shrink_right]
    
    d_opt = 0.5 * (dlo_v + dhi_v)
    f_opt = np.max(S[np.newaxis, :] + d_opt[:, None] * G, axis=1)
    
    # Mask infeasible
    f_opt[~feasible] = np.inf
    
    best_idx = int(np.argmin(f_opt))
    best_gain = float(np.max(S) - f_opt[best_idx])
    return best_idx, float(d_opt[best_idx]), float(f_opt[best_idx])
```

   = (a+δ)*(1-h'[j1-lag]) - a*(1-h[j1-lag])
   = (a+δ)*(1-h[j1-lag]) - a*(1-h[j1-lag]) + (a+δ)*(h[j1-lag]-h'[j1-lag])
   = δ*(1-h[j1-lag]) + (a+δ)*(h[j1-lag]-h'[j1-lag])
   
   The second term is nonzero only if j1-lag = j2, in which case h'[j2] = b-δ, so h[j1-lag]-h'[j1-lag] = δ.
   Then this term = (a+δ)*δ = aδ + δ²

3. Terms where n+lag = j2, i.e., n = j2-lag: h'[j2] * (1-h'[j2-lag]) - h[j2] * (1-h[j2-lag])
   = (b-δ)*(1-h'[j2-lag]) - b*(1-h[j2-lag])
   = (b-δ)*(1-h[j2-lag]) - b*(1-h[j2-lag]) + (b-δ)*(h[j2-lag]-h'[j2-lag])
   = -δ*(1-h[j2-lag]) + (b-δ)*(h[j2-lag]-h'[j2-lag])
   
   The second term is nonzero only if j2-lag = j1, i.e., lag = j2-j1.
   Then h'[j1] = a+δ, so h[j2-lag]-h'[j2-lag] = a-(a+δ) = -δ.
   This term = (b-δ)*(-δ) = -bδ + δ²

4. Terms where n = j2: h'[j2+lag] * (1-(b-δ)) - h[j2+lag] * (1-b) = h'[j2+lag]*δ
   h'[j2+lag] ≠ h[j2+lag] only if j2+lag = j1, i.e., lag = j1-j2.

So the total change has:
- Linear terms in δ (from items 1, 2, 3, 4)
- Quadratic terms in δ (from the cross terms in items 2 and 3, which only activate when j1-lag=j2 or j2-lag=j1)

The quadratic terms are at most O(δ²), which for small δ is negligible. But for the exact computation, they matter.

However, for the ternary search, I'm using the linear approximation. This means the computed δ is slightly off from the true optimum, but for small δ (which is typical near the optimum), the error is O(δ²) which is very small.

In practice, this should be fine. The algorithm will converge, just not to the exact optimum of the true (slightly nonlinear) function. But since we re-evaluate S after each swap, the next iteration will correct for the error.

This can be computed exactly by just recomputing the correlation after the swap. The ternary search tells us what δ to use, and then we recompute S exactly. So the linear approximation is fine for finding a good δ, and the exact recomputation keeps us on track.
