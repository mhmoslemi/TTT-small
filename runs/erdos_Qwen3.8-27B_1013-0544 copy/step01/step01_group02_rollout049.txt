import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _sum_con(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def _slsqp(h_init, n, alpha, iters, tl, bounds, cons):
    if tl() < 5:
        return h_init, compute_c5(h_init, n)
    try:
        r = minimize(obj_smooth, h_init, args=(n, alpha), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': iters, 'ftol': 1e-18})
        h_t = project_feasible(r.x, n)
        return h_t, compute_c5(h_t, n)
    except Exception:
        return h_init, compute_c5(h_init, n)

def _coord_descent(h, n, tl, dsets):
    c_best = compute_c5(h, n)
    for dset in dsets:
        improved = True
        while improved and tl() > 10:
            improved = False
            for i in range(n):

                            hj = h[j] - dd

                            if c < c_best - 1e-14:
                                c_best = c
                                improved = True
                            else:
                                h[i], h[j] = hi - dd, hj + dd
    return h, c_best
