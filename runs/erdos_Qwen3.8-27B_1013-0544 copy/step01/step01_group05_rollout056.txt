                        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                                     'jac': lambda x: np.ones(n_points)}],
                        options={'maxiter': 3000, 'ftol': 1e-14})
                    if res.x is not None:
                        h_ref = project_feasible(res.x, target_sum)
                        c5_ref = compute_c5(h_ref, dx)
                        if c5_ref < best_c5_loc:
                            best_c5_loc = c5_ref
                            best_h_loc = h_ref
                except Exception:
                    pass

            if best_c5_loc < best_c5:
                best_c5 = best_c5_loc
                best_h = best_h_loc.copy()
                best_n = n_points

    # Fine-tuning pass on best solution
    if best_h is not None and tleft() > 12:
        n_points = best_n
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        h = best_h.copy()
        step_size = 0.0008
        momentum = np.zeros(n_points)
        tau = dx * 1.5
        for it in range(8000):
            if tleft() < 3:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            g = compute_smooth_grad(corr, h, tau, 8)
            momentum = 0.5 * momentum - step_size * g
            h_new = project_feasible(h + momentum, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            else:
                step_size *= 0.92
                if step_size < 1e-10:
                    break
            h = h_new

    # Random perturbation + quick polish
    if best_h is not None and tleft() > 20:
        n_points = best_n
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        for _ in range(30):
            if tleft() < 5:
                break
            hp = best_h + rng.normal(0, 0.008, n_points)
            hp = project_feasible(hp, target_sum)
            cp = compute_c5(hp, dx)
            h = hp.copy()
            ss = 0.0005
            mom = np.zeros(n_points)
            for _ in range(300):
                if tleft() < 3:
                    break
                corr = np.correlate(h, 1.0 - h, mode='full')
                g = compute_smooth_grad(corr, h, dx, 6)
                mom = 0.5 * mom - ss * g
                h = project_feasible(h + mom, target_sum)
                ct = compute_c5(h, dx)
                if ct < cp:
                    cp = ct
                    hp = h.copy()
                else:
                    ss *= 0.88
                    if ss < 1e-9:
                        break
            if cp < best_c5:
                best_c5 = cp
                best_h = hp.copy()

    if best_h is None:
        best_h = project_feasible(h_init_70, 35.0)
        best_n = 70
        best_c5 = compute_c5(best_h, 2.0 / 70)

    return (best_h, best_c5, best_n)
