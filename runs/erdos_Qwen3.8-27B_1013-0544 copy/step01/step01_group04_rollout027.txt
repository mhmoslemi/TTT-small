def project_feasible(h, n):

        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:

        free = (h > 1e-10) & (h < 1.0 - 1e-10)

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)

    """Compute max_k corr_k when h[j] = t, using precomputed coefficients."""

    c0 = t - t*t + corr_0_base
    # corr_k = a_coeffs[k] * t + b_consts[k] for k >= 1
    cks = a_coeffs * t + b_consts
    return max(c0, np.max(cks))

def _minimize_1d(c0_base, a_coeffs, b_consts):

        c0 = t - t * t + c0_base
        cks = a_coeffs * t + b_consts

def lbfgs_softmax(h, n, max_iter=500, tol=1e-10):
    h = project_feasible(h.copy(), n)
    target = n / 2.0
    dx = 2.0 / n
    tau = 80.0
    penalty = 1e4
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        soft = (1.0 / tau) * np.log(np.sum(np.exp(tau * corr)))
        pen = penalty * (np.sum(x) - target) ** 2
        return soft + pen
    def obj_grad(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        exp_c = np.exp(tau * corr)
        w = exp_c / np.sum(exp_c)
        # d(corr_k)/dx[j] via chain rule
        grad = np.zeros(n)
        one_minus_x = 1.0 - x
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            for i in range(n):
                ik = i + k
                if ik < n:
                    grad[i] += wk * one_minus_x[ik]
                jk = i - k
                if jk >= 0:
                    grad[jk] -= wk * x[i]
        grad = grad * dx
        pen_grad = 2.0 * penalty * (np.sum(x) - target)
        return grad + pen_grad
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8})
    return project_feasible(res.x, n)
