For h of length n, np.correlate(h, 1-h, mode='full') gives an array of length 2n-1.
The k-th element (0-indexed) is:
corr[k] = Σ_i h[i+k-(n-1)] * (1-h[i]) for valid i

The value corr[k] = Σ_i h[i+j] * (1-h[i]) where j = k-(n-1).

For j=0: corr[n-1] = Σ_i h[i] * (1-h[i]) = Σ_i h[i] - Σ_i h[i]^2

from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-11:
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def make_smooth_obj(h0, n, dx, target, beta):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        return float((m + np.log(np.sum(np.exp(c - m))) / beta) * dx)
    return obj, bounds, cons

def slsqp_smooth(h0, n, dx, target, beta, max_iter=100):
    obj, bounds, cons = make_smooth_obj(h0, n, dx, target, beta)
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, n)
    return project_feasible(h0.copy(), n)
