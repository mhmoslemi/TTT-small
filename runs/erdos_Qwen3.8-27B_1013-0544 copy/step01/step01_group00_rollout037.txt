import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_onto_sum(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = corr - m
    weights = np.exp(shifted / temp)
    wsum = weights.sum()
    weights /= wsum
    obj = m + temp * np.log(wsum)
    
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-14:
            continue
        s = m_idx - (n - 1)
        if s >= 0:
            grad[:n-s] += w * dx * (1.0 - h[s:])
            grad[s:] -= w * dx * h[:n-s]
        else:
            grad[-s:] += w * dx * (1.0 - h[:n+s])
            grad[:n+s] -= w * dx * h[-s:]

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_onto_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_obj_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = corr - m
    weights = np.exp(shifted / temp)
    wsum = weights.sum()
    weights /= wsum
    obj = m + temp * np.log(wsum)
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-12:
            continue
        s = m_idx - (n - 1)
        if s >= 0:
            grad[:n - s] += w * dx * (1.0 - h[s:])
            grad[s:] -= w * dx * h[:n - s]
        else:
            grad[-s:] += w * dx * (1.0 - h[:n + s])
            grad[:n + s] -= w * dx * h[-s:]
    return obj, grad


def t_objective(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def run_pgd(h, temps, n_steps, step_size, momentum, time_limit):
    n = len(h)
    h_cur = h.copy()
    h_mom = np.zeros(n)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    t_start = time.time()
    
    for temp in temps:
        for it in range(n_steps):
            if time.time() - t_start > time_limit:
                return best_h, best_c5
            _, grad = smooth_obj_grad(h_cur, temp)
            h_mom = momentum * h_mom - step_size * grad
            h_new = h_cur + h_mom
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_onto_sum(h_new, n * 0.5)
            if it % 100 == 0 or it == n_steps - 1:
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            h_cur = h_new
    return best_h, best_c5


def slsqp_optimize(h0, max_iter, ftol):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), min(c5_cur + 0.02, 1.0))
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_onto_sum(h_new, n * 0.5)
    return h_new, compute_c5(h_new)
