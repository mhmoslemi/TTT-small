import numpy as np
from scipy.optimize import minimize, LBFGS_B
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective_and_grad(h, temp, lam=1.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) / temp
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    w /= w_sum
    obj = m + temp * np.log(w_sum)
    
    # Gradient of log-sum-exp part
    # dPhi/dh[s] = dx * sum_j w[j] * [(1-h[s+n-1-j])*1_{valid} - h[s-(n-1-j)]*1_{valid}]
    grad_lse = np.zeros(n)
    for s in range(n):
        # First term: sum_j w[j] * (1-h[s+n-1-j]) where 0 <= s+n-1-j < n
        # => j ranges from max(0, s) to min(2n-2, s+n-1)
        j_lo = max(0, s)
        j_hi = min(2*n-2, s+n-1)
        if j_lo <= j_hi:
            idx = s + n - 1 - np.arange(j_lo, j_hi+1)
            grad_lse[s] += np.dot(w[j_lo:j_hi+1], 1.0 - h[idx])
        # Second term: sum_j w[j] * h[s-(n-1-j)] where 0 <= s-n+1+j < n
        # => j ranges from max(0, n-1-s) to min(2n-2, n-s)
        j_lo2 = max(0, n-1-s)

            grad_lse[s] -= np.dot(w[j_lo2:j_hi2+1], h[idx])
    grad_lse *= dx
    
    # Penalty gradient for equality constraint
    pen = lam * (np.sum(h) - n*0.5)
    grad_pen = 2.0 * lam * (np.sum(h) - n*0.5) * np.ones(n)
    
    total_obj = obj + lam * (np.sum(h) - n*0.5)**2
    return total_obj, grad_lse + grad_pen

def lse_optimize(h0, temp, lam=1.0, max_iter=200):
    n = len(h0)
    h = h0.copy()
    def fun_and_grad(h):
        return lse_objective_and_grad(h, temp, lam)
    res = minimize(fun_and_grad, h, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)]*n,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n*0.5)
    return h_new, compute_c5(h_new)

def thue_morse(n):
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)

def t_objective(v):
    return v[-1]

def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j

def optimize_t_var(h0, max_iter=400, ftol=1e-10):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.02)
    bounds = [(0.0, 1.0)]*n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n*0.5)
    return h_new, compute_c5(h_new)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def lse_fun_and_grad(h, temp, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(corr))
    shifted = (corr - m) / temp
    shifted -= np.max(shifted)
    exp_s = np.exp(shifted)
    w_sum = float(np.sum(exp_s))
    w = exp_s / w_sum
    obj = m + temp * np.log(w_sum)

    grad = np.zeros(n)
    for s in range(n):
        j_lo = s
        j_hi = min(2 * n - 2, s + n - 1)
        if j_lo <= j_hi:
            idx = s + n - 1 - np.arange(j_lo, j_hi + 1)
            grad[s] += float(np.dot(w[j_lo:j_hi + 1], 1.0 - h[idx]))
        j_lo2 = max(0, n - 1 - s)
        j_hi2 = min(2 * n - 2, 2 * n - 2 - s)
        if j_lo2 <= j_hi2:
            idx2 = s - (n - 1) + np.arange(j_lo2, j_hi2 + 1)
            grad[s] -= float(np.dot(w[j_lo2:j_hi2 + 1], h[idx2]))
    grad *= dx

    pen_val = lam * (np.sum(h) - n * 0.5) ** 2
    pen_grad = 2.0 * lam * (np.sum(h) - n * 0.5)
    return obj + pen_val, grad + pen_grad


def lse_optimize(h0, temp, lam, max_iter=200):
    n = len(h0)

    def wrapper(h):
        return lse_fun_and_grad(h, temp, lam)

    res = minimize(wrapper, h0.copy(), method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-7})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def thue_morse_init(n):
    tm = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    h = (1 - tm).astype(float)
    return project_h(h, n * 0.5)


def interp_h(h_old, n_new):
    x_old = np.linspace(0.0, 1.0, len(h_old))
    x_new = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)


def t_objective(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return v[-1] - corr


def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def optimize_t_var(h0, max_iter=400, ftol=1e-10):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def refine_from(h0, temps, base_lam, max_iter, time_limit_fn):
    h_cur = h0.copy()
    best_local = compute_c5(h_cur)
    for temp in temps:
        if time_limit_fn():
            break
        lam = base_lam
        h_new, c5_new = lse_optimize(h_cur, temp, lam, max_iter)
        if c5_new < best_local:
            h_cur = h_new
            best_local = c5_new
        h_new, c5_new = lse_optimize(h_cur, temp * 0.5, lam * 3, max_iter)
        if c5_new < best_local:
            h_cur = h_new
            best_local = c5_new
    return h_cur, best_local

        return budget_s - (time.time() - t0)

    # Phase 1: Optimize each start
