            if no_improve > 15000:
                cur_h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0
                T = max(T, 0.003)

    # Phase 4: Multi-start local optimization
    if remaining() > 100:
        for restart in range(8):
            if remaining() < 30:
                break
            scale = rng.choice([0.05, 0.1, 0.15, 0.2, 0.3])
            h_pert = best_h + rng.normal(0, scale, size=n)
            h_pert = project_feasible(h_pert, n)
            h_try = slsqp_polish(h_pert, n, 3000, maxiter=2000, ftol=1e-15)
            c = compute_c5(h_try, n)
            if c < best_c5:
                best_h, best_c5 = h_try, c

    # Phase 5: Fine 2-point swaps
    if remaining() > 30:
        count = 0
        half_life = 0
        while remaining() > 10 and count < 150000:
            count += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if half_life == 0:
                d = 0.001 * rng.choice([1, 2, 3, 5, 8])
            else:
                d = 0.0005 * rng.choice([1, 2, 4])
            for s in [1, -1]:
                di = s * d
                hi = best_h[i] + di
                hj = best_h[j] - di
                if 0 <= hi <= 1 and 0 <= hj <= 1:
                    h_try = best_h.copy()
                    h_try[i] = hi
                    h_try[j] = hj
                    c = compute_c5(h_try, n)
                    if c < best_c5:
                        best_h = h_try
                        best_c5 = c
                        half_life = 500
                        break
            half_life = max(0, half_life - 1)

    # Phase 6: Final SLSQP polish
    if remaining() > 5:
        for alpha in [5000, 15000, 50000]:
            if remaining() < 5:
                break
            h_try = slsqp_polish(best_h, n, alpha, maxiter=5000, ftol=1e-18)
            c = compute_c5(h_try, n)
            if c < best_c5:
                best_h, best_c5 = h_try, c

    # Final feasibility check
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)

    return best_h, best_c5, n
