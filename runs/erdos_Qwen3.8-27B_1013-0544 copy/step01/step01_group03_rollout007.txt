    return float(np.max(corr) * dx)

def smooth_max_val(obj, tau):
    x = obj / tau
    x_max = np.max(x)
    return float(tau * (x_max + np.log(np.sum(np.exp(x - x_max)))))

def fix_sum(h, target):
    h = np.array(h, float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_smooth_grad(h, tau):
    """Compute gradient of smooth max of correlation"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = corr / tau
    x -= x.max()
    exp_x = np.exp(x)
    p = exp_x / exp_x.sum()
    # p[k] for k=0..2n-2
    
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for j in range(n):
        # First term: sum over m where both j-l and l are valid
        # grad[j] += p[l+(n-1)] * (1-h[j-l]) for l where 0<=j-l<=n-1
        # Let m = j-l, so l = j-m, valid when 0<=m<=n-1 and 0<=j-m+(n-1)<=2n-2
        # j-m+(n-1) >= 0 → m <= j+n-1 (always true)
        # j-m+(n-1) <= 2n-2 → m >= j-n+1
        # So m ranges from max(0, j-n+1) to min(n-1, j+n-1) = n-1 (since j<=n-1)
        m_lo = max(0, j - n + 1)
        m_hi = n - 1
        if m_lo <= m_hi:
            m = np.arange(m_lo, m_hi + 1)
            l = j - m  # l = j - m
            grad[j] += np.dot(one_minus_h[m], p[l + n - 1])
        # Second term: grad[j] -= p[l+(n-1)] * h[j+l] for l where 0<=j+l<=n-1
        # Let m = j+l, so l = m-j, valid when 0<=m<=n-1 and 0<=m-j+(n-1)<=2n-2
        # m-j+(n-1) >= 0 → m >= j-n+1 (may or may not be active)
        # m-j+(n-1) <= 2n-2 → m <= j+n-1 (always true since m<=n-1)
        m_lo2 = max(0, j - n + 1)
        m_hi2 = n - 1
        if m_lo2 <= m_hi2:
            m = np.arange(m_lo2, m_hi2 + 1)
            l = m - j
            grad[j] -= np.dot(h[m], p[l + n - 1])
    return grad, p, corr.copy()

def projected_gradient_step(h, grad, target, step_size):
    """One projected gradient step with sum constraint"""
    h_new = h - step_size * grad
    return fix_sum(h_new, target)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(obj, tau):
    x = obj / tau
    x_max = np.max(x)
    return float(tau * (x_max + np.log(np.sum(np.exp(x - x_max)))))

def fix_sum(h, target):
    h = np.array(h, float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    else:
        mask = h > 1e-12
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, tau):
    """Gradient of smooth max of correlation. O(n^2)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    y = corr / tau
    y -= y.max()
    e = np.exp(y)
    p = e / e.sum()  # length 2n-1
    
    grad = np.zeros(n)
    for j in range(n):
        m_lo = max(0, j - n + 1)
        for m in range(m_lo, n):
            l1 = j - m
            idx1 = l1 + n - 1
            if 0 <= idx1 <= 2*n - 2:
                grad[j] += p[idx1] * (1.0 - h[m])
            l2 = m - j
            idx2 = l2 + n - 1
            if 0 <= idx2 <= 2*n - 2:
                grad[j] -= p[idx2] * h[m]
    return grad

import numpy as np
from scipy.optimize import minimize
from scipy.linalg import toeplitz
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(obj, tau):
    x = obj / tau
    x_max = np.max(x)
    return float(tau * (x_max + np.log(np.sum(np.exp(x - x_max)))))

def fix_sum(h, target):
    h = np.array(h, float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    else:
        mask = h > 1e-12
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    y = corr / tau
    y -= y.max()
    e = np.exp(y)
    p = e / e.sum()
    first_col = p[0:n][::-1]
    first_row = p[n-1:2*n-1]
    M1 = toeplitz(first_col, first_row)
    grad = M1 @ (1.0 - h) - M1.T @ h
    return grad

def obj_grad_lbfgsb(x, tau, n, penalty):
    target = n / 2.0
    corr = np.correlate(x, 1.0 - x, mode="full")
    f = smooth_max_val(corr, tau)
    g = smooth_grad(x, tau)
    s_diff = x.sum() - target
    f += 0.5 * penalty * s_diff * s_diff
    g = g + penalty * s_diff * np.ones(n)
    return f, g

        h_init = np.full(120, 0.5)
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    
    def rem():
        return budget_s - (time.time() - t0)
    
    def do_optimize(h_start, tau, maxiter, penalty=500.0):
        n = len(h_start)
        res = minimize(obj_grad_lbfgsb, h_start, args=(tau, n, penalty),
                      method='L-BFGS-B', bounds=[(0.0,1.0)]*n,
                      options={'maxiter': maxiter, 'ftol': 1e-15})

import numpy as np
from scipy.optimize import minimize
from scipy.linalg import toeplitz
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(obj, tau):
    x = obj / tau
    x_max = np.max(x)
    return float(tau * (x_max + np.log(np.sum(np.exp(x - x_max)))))

def fix_sum(h, target):
    h = np.array(h, float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    else:
        mask = h > 1e-12
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    y = corr / tau
    y -= y.max()
    e = np.exp(y)
    p = e / e.sum()
    first_col = p[0:n][::-1]
    first_row = p[n-1:2*n-1]
    M1 = toeplitz(first_col, first_row)
    return M1 @ (1.0 - h) - M1.T @ h

def obj_grad_lbfgsb(x, tau, n, penalty):
    target = n / 2.0
    corr = np.correlate(x, 1.0 - x, mode="full")
    f = smooth_max_val(corr, tau)
    g = smooth_grad(x, tau)
    s = x.sum() - target
    f += 0.5 * penalty * s * s
    g = g + penalty * s
    return f, g

def anneal_lbfgsb(h_start, h_best, c5_best, tau_list, iters, penalty, budget_s, t0):
    h_cur = h_start.copy()
    n = len(h_cur)
    for tau in tau_list:
        if budget_s - (time.time() - t0) < 8:
            break
        try:
            res = minimize(obj_grad_lbfgsb, h_cur, args=(tau, n, penalty),
                          method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': iters, 'ftol': 1e-14, 'gtol': 1e-8})
            h_new = fix_sum(res.x, n / 2.0)
            c5 = evaluate_c5(h_new)
            if c5 < c5_best:
                h_best = h_new.copy()
                c5_best = c5
            h_cur = h_new
        except Exception:
            pass
    return h_cur, h_best, c5_best

def slsqp_polish(h_start, h_best, c5_best, tau, maxiter, budget_s, t0):
    n = len(h_start)
    target = n / 2.0
    if budget_s - (time.time() - t0) < 10:
        return h_start, h_best, c5_best
    try:
        res = minimize(obj_grad_lbfgsb, h_start, args=(tau, n, 0.0),
                      method='SLSQP', bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                   'jac': lambda x: np.ones(n)}],
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        h_new = fix_sum(res.x, target)
        c5 = evaluate_c5(h_new)
        if c5 < c5_best:
            h_best = h_new.copy()
            c5_best = c5
        return h_new, h_best, c5_best
    except Exception:
        return h_start, h_best, c5_best
