    sum_exp = np.sum(exp_x)

            h_t = h_new

    # Phase 5: Final fine-tuning at best resolution
    hn_final = len(best_h)
    for tau in [5, 3, 2]:
        if _rem() < 5:
            break
        h_new, c5_new = _do_optimize(best_h, hn_final, tau, 1000.0, 500)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
