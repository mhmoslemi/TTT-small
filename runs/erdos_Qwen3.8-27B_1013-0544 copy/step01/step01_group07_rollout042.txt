import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_simplex_box(x, S):
    """Project x onto {h: sum(h)=S, 0<=h<=1} using bisection on lambda."""
    n = len(x)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def subgradient_max(h):
    """Compute subgradient of max correlation."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    
    grad = np.ones(n)
    # Subtract h[j-lag] where defined
    if lag >= 0:
        grad[lag:] -= h[:n-lag]
    else:
        grad[:n+lag] -= h[-lag:]
    # Subtract h[j+lag] where defined  
    if lag >= 0:
        grad[:n-lag] -= h[lag:]
    else:
        grad[-lag:] -= h[:n+lag]
    
    return grad, corr, k_star

def adam_step(params, grad, beta1, beta2, lr, t, eps=1e-8):
    m = params['m']
    v = params['v']
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    update = lr * m_hat / (np.sqrt(v_hat) + eps)
    return update, m, v

    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_simplex_box(x, S):
    n = len(x)
    if S <= 0:
        return np.zeros(n)
    if S >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def corr_lag_grad(x, lag, n):
    grad = np.ones(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] -= x[:n-lag]
            grad[:n-lag] -= x[lag:]
    else:
        if -lag < n:
            grad[:n+lag] -= x[-lag:]
            grad[-lag:] -= x[:n+lag]
    return grad

def lbfgsb_optimize(h, dx, S, t, penalty, maxiter=300):
    n = len(h)
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        max_c = np.max(corr)
        shifted = t * (corr - max_c)
        # Log-sum-exp for numerical stability
        obj_val = (max_c + (1.0/t) * np.log(np.sum(np.exp(shifted)))) * dx
        obj_val += penalty * (np.sum(x) - S)**2
        return float(obj_val)
    
    def grad(x):
        corr = np.correlate(x, 1 - x, mode="full")
        max_c = np.max(corr)
        shifted = t * (corr - max_c)
        exp_s = np.exp(shifted)
        total = np.sum(exp_s)
        weights = exp_s / total
        
        g = np.zeros(n)
        for i in range(len(corr)):
            if weights[i] < 1e-10:
                continue
            lag = i - (n - 1)
            g += weights[i] * corr_lag_grad(x, lag, n)
        g *= dx
        g += 2.0 * penalty * (np.sum(x) - S)
        return g
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        return res.x
    except Exception:
        return h

    w = np.exp(t_soft * (top_vals - top_vals.max()))

        grad += ww * corr_lag_grad(h, lag, n)
    
    h_new = h - lr * grad
    h_new = project_simplex_box(h_new, S)

import numpy as np
import time
from scipy.optimize import minimize

_opt = {'t': 1.0, 'dx': 0.01, 'S': 1.0, 'penalty': 1.0}

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_simplex_box(x, S):
    n = len(x)
    if S <= 0:
        return np.zeros(n)
    if S >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def corr_lag_grad(x, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += 1.0 - x[:n - lag]
            if n - lag > 0:
                grad[:n - lag] -= x[lag:]
    else:
        if -lag < n:
            if n + lag > 0:
                grad[:n + lag] += 1.0 - x[-lag:]
            grad[-lag:] -= x[:n + lag]
    return grad

def _obj(x):
    t = _opt['t']
    dx = _opt['dx']
    S = _opt['S']
    penalty = _opt['penalty']
    corr = np.correlate(x, 1 - x, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    obj_val = (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx
    obj_val += penalty * (np.sum(x) - S) ** 2
    return float(obj_val)

def _grad(x):
    t = _opt['t']
    dx = _opt['dx']
    S = _opt['S']
    penalty = _opt['penalty']
    n = len(x)
    corr = np.correlate(x, 1 - x, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    g = np.zeros(n)
    for i in range(len(corr)):
        if weights[i] < 1e-10:
            continue
        lag = i - (n - 1)
        g += weights[i] * corr_lag_grad(x, lag, n)
    g *= dx
    g += 2.0 * penalty * (np.sum(x) - S)
    return g

def lbfgsb_optimize(h, dx, S, t, penalty, maxiter=300):
    _opt['t'] = t
    _opt['dx'] = dx
    _opt['S'] = S
    _opt['penalty'] = penalty
    n = len(h)
    try:
        res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        return res.x
    except Exception:
        return h

def subgradient_step(h, S, lr, top_k=3):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    top_idx = np.argsort(corr)[-top_k:]
    top_vals = corr[top_idx]
    t_soft = 20.0
    w = np.exp(t_soft * (top_vals - top_vals.max()))
    w /= w.sum()
    grad = np.zeros(n)
    for ww, i in zip(w, top_idx):
        lag = i - (n - 1)
        grad += ww * corr_lag_grad(h, lag, n)
    h_new = h - lr * grad
    return project_simplex_box(h_new, S)

def make_fourier_h(n, rng, n_harm=5):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.0, 0.15)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(k * np.pi * x + phase)
    return h
