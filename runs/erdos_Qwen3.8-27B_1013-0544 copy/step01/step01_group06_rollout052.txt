import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    ws = np.sum(w)
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            end = n - k
            grad[:end] += wi * (1.0 - h[k:])
            grad[k:] += wi * (-h[:end])
        else:
            p = -k
            end = n - p
            grad[p:] += wi * (1.0 - h[:end])
            grad[:end] += wi * (-h[p:])
    return grad * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def optimize_with_line_search(h0, target_sum, deadline, rng, alpha_init=80):
