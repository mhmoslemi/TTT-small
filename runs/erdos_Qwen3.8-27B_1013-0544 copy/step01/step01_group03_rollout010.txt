def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        # Add to elements most in interior
        order = np.argsort(h)  # smallest first, add to those
        for i in order:
            room = 1.0 - h[i]
            add = min(room, diff)
            h[i] += add
            diff -= add
            if diff <= 1e-15:
                break
    else:
        diff = -diff
        order = np.argsort(-h)  # largest first, subtract from those
        for i in order:
            room = h[i]
            sub = min(room, diff)
            h[i] -= sub
            diff -= sub
            if diff <= 1e-15:
                break

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    # Water-filling projection
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h + mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def compute_subgrad(h, lag_idx, n):
    shift = lag_idx - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        s = shift
        if s < n:
            grad[0:n-s] += -h[s:n]
            grad[s:n] += 1.0 - h[0:n-s]
    else:
        s = -shift
        if s < n:
            grad[0:n-s] += 1.0 - h[s:n]
            grad[s:n] += -h[0:n-s]
    return grad

def project_simplex_box(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h + mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def slsqp_optimize(h_start, n_pts, tau, maxiter=200):
    target = n_pts / 2.0
    bnds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, tau)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n_pts)}]
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        x = fix_sum(res.x, target)
        c5 = evaluate_c5(x)
        return x, c5
    except Exception:
        return h_start, None

def subgradient_phase(h, n, target, t0, budget_s, max_iters=20000):
    h = h.copy()
    momentum = np.zeros(n)
    beta = 0.85
    best = evaluate_c5(h)
    for it in range(max_iters):
        if budget_s - (time.time() - t0) < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        t_star = np.argmax(corr)
        # Weighted subgradient from top lags
        top_k = min(10, len(corr))
        thresh = corr[t_star] * 0.85
        top_lags = np.where(corr >= thresh)[0]
        top_lags = top_lags[:top_k]
        weights = (corr[top_lags] - thresh) / (corr[t_star] - thresh + 1e-15)
        grad = np.zeros(n)
        for tl in top_lags:
            grad += weights[list(top_lags).index(tl)] * compute_subgrad(h, tl, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        alpha = 0.5 / np.sqrt(it + 1.0)
        momentum = beta * momentum + (1 - beta) * (grad / gnorm)
        h_new = h - alpha * momentum * 5.0
        h_new = project_simplex_box(h_new, target)
        c5_new = evaluate_c5(h_new)
        if c5_new < best:
            h = h_new
            best = c5_new
        else:
            h = project_simplex_box(h + 0.1 * alpha * momentum * 5.0, target)
    return h, best

def binary_local_search(h, n, target, t0, budget_s, max_passes=50):
    hb = (h >= 0.5).astype(float)
    hb = fix_sum(hb, target)
    best_c5 = evaluate_c5(hb)
    best_h = hb.copy()
    ones = np.where(hb == 1.0)[0]
    zeros = np.where(hb == 0.0)[0]
    for p in range(max_passes):
        if budget_s - (time.time() - t0) < 5:
            break
        improved = False
        for i in ones:
            if improved and len(ones) > 10:
                ones = np.where(best_h == 1.0)[0]
                zeros = np.where(best_h == 0.0)[0]
            for j in zeros:
                if i == j:
                    continue
                best_h[i] = 0.0
                best_h[j] = 1.0
                c5_new = evaluate_c5(best_h)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    improved = True
                    ones = np.where(best_h == 1.0)[0]
                    zeros = np.where(best_h == 0.0)[0]

                else:
                    best_h[i] = 1.0
                    best_h[j] = 0.0
                if budget_s - (time.time() - t0) < 3:
                    return best_h, best_c5
        if not improved:
            break
    return best_h, best_c5
