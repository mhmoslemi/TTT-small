        # corr[m] = sum_{i in V} h[i] - sum_{i in V} h[i]*h[i-lag]
        #
        # d/d(h[j]):
        # First term: 1 if j in V, else 0
        # Second term: 
        #   d/d(h[j]) sum_{i in V} h[i]*h[i-lag]
        #   = (product rule) sum_{i in V} [delta(i,j)*h[i-lag] + h[i]*delta(i-lag,j)]
        #   = h[j-lag] (if j in V) + h[j+lag] (if j+lag in V)
        #
        # So: d(corr[m])/d(h[j]) = 1[j in V] - h[j-lag]*1[j in V] - h[j+lag]*1[j+lag in V]
        #   = 1[j in V] * (1 - h[j-lag]) - h[j+lag]*1[j+lag in V]
        
        if lag >= 0:
            V_lo, V_hi = lag, n - 1
        else:
            V_lo, V_hi = 0, n - 1 + lag
        
        # For j in V: contribution is (1 - h[j-lag])
        # For j+lag in V (i.e., j in V+lag): contribution is -h[j+lag] = -h[i] where i=j+lag, so -h[i]
        # Wait, let me just compute it directly.
        
        j_in_V = np.zeros(n, dtype=bool)
        j_in_V[V_lo:V_hi+1] = True
        
        # First part: 1[j in V] * (1 - h[j-lag])
        part1 = np.zeros(n)
        for j in range(V_lo, V_hi + 1):
            part1[j] = 1.0 - h[j - lag]
        
        # Second part: -h[j+lag] * 1[j+lag in V]
        # j+lag in V means V_lo <= j+lag <= V_hi, i.e., V_lo-lag <= j <= V_hi-lag
        part2 = np.zeros(n)
        j2_lo = max(0, V_lo - lag)
        j2_hi = min(n - 1, V_hi - lag)
        for j in range(j2_lo, j2_hi + 1):
            part2[j] = -h[j + lag]
        
        grad += w[m] * (part1 + part2)
    
    return grad * dx
