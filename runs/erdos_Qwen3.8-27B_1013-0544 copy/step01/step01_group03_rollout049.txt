    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def slsqp_run(h0, n, tau, maxiter):
    target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, tau)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                  constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
    x = fix_sum(res.x, target)
    return x, evaluate_c5(x)

def subgradient_polish(h, n, steps=1000, alpha_init=0.01):
    """Polish using subgradient of the true max."""
    dx = 2.0 / n
    h = h.copy()
    alpha = alpha_init
    for step in range(steps):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        k_star = np.argmax(corr)
        
        # Compute gradient of corr[k_star] w.r.t. h
        # corr[k] = sum_i h[i] * (1-h[i']) where the pairing depends on k
        # For numpy correlate: out[k] = sum_i a[i] * v[flip_idx]
        # Let's just do it numerically with a small perturbation in the direction
        # that reduces the max
        
        # Actually, let's compute the gradient analytically
        # From the empirical pattern: out[k] involves h[i]*h[j] pairs
        # where i and j are related by the lag k
        
        # Simpler: just use the "effective gradient" approach
        # The subgradient of max_k g_k at the point is (1/|S|) * Σ_{k in S} ∇g_k
        # where S = {k : g_k = max}
        
        # For simplicity, use just k_star
        # ∇g_{k*}[j] = ∂corr[k*]/∂h[j] * dx
        
        # Numerical gradient (central difference with step 1e-5)
        eps = 1e-5
        grad = np.zeros(n)
        for j in range(n):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[j] = min(1, h[j] + eps)
            h_minus[j] = max(0, h[j] - eps)
            c_plus = np.correlate(h_plus, 1-h_plus, mode="full")[k_star] * dx
            c_minus = np.correlate(h_minus, 1-h_minus, mode="full")[k_star] * dx
            grad[j] = (c_plus - c_minus) / (h_plus[j] - h_minus[j]) if h_plus[j] != h_minus[j] else 0
        
        # Projected gradient step
        h_new = h - alpha * grad
        # Project to {x : 0 ≤ x ≤ 1, sum(x) = n/2}
        h_new = np.clip(h_new, 0, 1)
        h_new = fix_sum(h_new, n/2)
        
        if evaluate_c5(h_new) < evaluate_c5(h):
            h = h_new
            alpha = min(alpha * 1.05, 0.1)
        else:
            alpha *= 0.9
            if alpha < 1e-6:
                break
