        x = x_lo + frac * (x_hi - x_lo)
        h[i], h[j] = x, s - x
        v = np.max(np.correlate(h, 1-h, mode="full")) * dx
        if v < best_val:
            best_val, best_x = v, x
    for _ in range(10):
        step = (x_hi - x_lo) * 0.04
        improved = False
        for d in [step, -step, step*0.4, -step*0.4]:
            xn = best_x + d
            if x_lo <= xn <= x_hi:
                h[i], h[j] = xn, s - xn
                v = np.max(np.correlate(h, 1-h, mode="full")) * dx
                if v < best_val:
                    best_val, best_x = v, xn
                    improved = True
        if not improved:
            break
    h[i], h[j] = best_x, s - best_x
    return best_val

def pair_sweep(h, dx, n_sweeps=3, offsets=None):
    n = len(h)
    if offsets is None:
        offsets = [1, 2, 3, 4, 5, 8, 16, n//3, n//2]
    for _ in range(n_sweeps):
        for i in range(n):
            for off in offsets:
                j = (i + off) % n
                if j != i:
                    pair_optimize(h, i, j, dx)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = fix_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    best_n = n

    # Phase 1: Quick SLSQP warmup
    for t in [5, 15, 40, 100, 250]:
        if elapsed() > budget_s - 200:
            break
        h_new = slsqp_optimize(best_h, dx, target_sum, t=t, maxiter=400)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h, best_c5 = h_new.copy(), c5_new

    # Phase 2: Aggressive pairwise coordinate descent (main workhorse)
    for sweep_batch in range(50):
        if elapsed() > budget_s - 80:
            break
        c5_before = compute_c5(best_h, dx)
        pair_sweep(best_h, dx, n_sweeps=1)
        c5_mid = compute_c5(best_h, dx)
        if c5_mid < best_c5:
            best_c5 = c5_mid
        # If stalled, try random pairs
        if c5_mid >= c5_before - 1e-8:
            for _ in range(n):
                i, j = np.random.randint(0, n, 2)
                if i != j:
                    pair_optimize(best_h, i, j, dx)
            c5_rand = compute_c5(best_h, dx)
            if c5_rand < best_c5:
                best_c5 = c5_rand

    # Phase 3: SLSQP refinement on improved point
    if elapsed() < budget_s - 60:
        for t in [50, 150, 400]:
            if elapsed() > budget_s - 50:
                break
            h_new = slsqp_optimize(best_h, dx, target_sum, t=t, maxiter=600)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new

    # Phase 4: Random perturbations + pair descent
    n_trials = 0
    while elapsed() < budget_s - 50 and n_trials < 300:
        n_trials += 1
        h_trial = best_h.copy()
        n_pert = np.random.randint(2, 8)
        idx = np.random.choice(n, n_pert, replace=False)
        h_trial[idx] += np.random.uniform(-0.35, 0.35, n_pert)
        h_trial = fix_sum(h_trial, target_sum)
        c5_t = compute_c5(h_trial, dx)
        if c5_t < best_c5 * 1.05:
            pair_sweep(h_trial, dx, n_sweeps=2)
            c5_t = compute_c5(h_trial, dx)
            if c5_t < best_c5:
                best_h, best_c5 = h_trial.copy(), c5_t
            elif elapsed() < budget_s - 40:
                h_trial = slsqp_optimize(h_trial, dx, target_sum, t=60, maxiter=300)
                c5_t = compute_c5(h_trial, dx)
                if c5_t < best_c5:
                    pair_sweep(h_trial, dx, n_sweeps=1)
                    c5_t = compute_c5(h_trial, dx)
                    if c5_t < best_c5:
                        best_h, best_c5 = h_trial.copy(), c5_t

    # Phase 5: Try n=80 (downsample)
    if elapsed() < budget_s - 30:
        n2 = 80
        dx2 = 2.0 / n2
        ts2 = n2 / 2.0
        x_old = np.linspace(0, 2, n + 1)
        x_new = np.linspace(0, 2, n2 + 1)
        h2 = np.interp(x_new[:-1], x_old, best_h)
        h2 = fix_sum(h2, ts2)
        h2 = slsqp_optimize(h2, dx2, ts2, t=30, maxiter=500)
        pair_sweep(h2, dx2, n_sweeps=5)
        c5_2 = compute_c5(h2, dx2)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2, c5_2, n2

    # Phase 6: Final polish
    if elapsed() < budget_s - 5:
        dx_f = 2.0 / best_n
        ts_f = best_n / 2.0
        h_final = slsqp_optimize(best_h, dx_f, ts_f, t=1000, maxiter=2000)
        pair_sweep(h_final, dx_f, n_sweeps=3)
        c5_final = compute_c5(h_final, dx_f)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
