                    h_bin[i1] = 1.0; h_bin[i2] = 1.0
                    h_bin[j1] = 0.0; h_bin[j2] = 0.0

        T = max(T_min, T * cooling)

    if bin_best_c5 < best_c5:
        best_c5 = bin_best_c5
        best_h = bin_best_h.copy()

    # Phase 2: SLSQP refinement (30% time)
    phase2_end = t0 + budget_s * 0.65

    # From best binary with perturbation
    h_work = bin_best_h.copy()
    h_work += np.random.uniform(-0.08, 0.08, n)
    d = target_sum - np.sum(h_work)
    h_work = np.clip(h_work + d / n, 0.0, 1.0)

    for t_val in [10, 30, 80, 200, 500]:
        if time.time() > phase2_end - 30:
            break
        h_work = slsqp_full(h_work, dx, target_sum, t_val, maxiter=500)
        cw = compute_c5(h_work, dx)
        if cw < best_c5:
            best_c5 = cw
            best_h = h_work.copy()

    # From initial values
    h_init = h0.copy()
    for t_val in [50, 150, 400]:
        if time.time() > phase2_end - 20:
            break
        h_init = slsqp_full(h_init, dx, target_sum, t_val, maxiter=400)
        ci = compute_c5(h_init, dx)
        if ci < best_c5:
            best_c5 = ci
            best_h = h_init.copy()

    # Phase 3: Random group SLSQP (20% time)
    phase3_end = t0 + budget_s * 0.88
    gs = 8
    for _ in range(2000):
        if time.time() > phase3_end:
            break
        start = np.random.randint(0, n - gs + 1)
        idx = list(range(start, start + gs))
        ht = slsqp_sub(best_h, dx, idx, t_val=100, maxiter=120)
        ct = compute_c5(ht, dx)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 4: Final polish (7% time)
    if time.time() < t0 + budget_s * 0.97:
        for t_val in [400, 1000]:
            if time.time() > t0 + budget_s * 0.96:
                break
            h_f = slsqp_full(best_h, dx, target_sum, t_val, maxiter=1000)
            cf = compute_c5(h_f, dx)
            if cf < best_c5:
                best_c5 = cf
                best_h = h_f

    # Validate and return
    best_h = np.clip(best_h, 0.0, 1.0)
    d = target_sum - np.sum(best_h)
    if abs(d) > 1e-10:
        best_h = np.clip(best_h + d / n, 0.0, 1.0)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, best_n
