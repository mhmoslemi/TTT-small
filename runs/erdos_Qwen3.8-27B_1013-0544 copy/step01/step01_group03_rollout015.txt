def project_simplex(h, c):
    """Project h onto {x: 0 <= x <= 1, sum(x) = c}."""
    # Bisection on lambda
    lo, hi = -1.0, 2.0
    for _ in range(60):

        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > c:

        else:

    x -= np.max(x)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(obj, tau):
    x = tau * obj
    xm = np.max(x)
    return float(xm / tau + np.log(np.sum(np.exp(x - xm))) / tau)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def project_box_sum(h, c):
    """Project h onto {x: 0<=x<=1, sum(x)=c} via bisection on shift."""
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > c:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad_h(h, tau):
    """Compute gradient of smooth_max(correlate(h, 1-h), tau) w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x -= np.max(x)
    w = np.exp(x)
    w /= w.sum()
    # Truncate small weights
    w[w < 1e-12] = 0.0
    active = np.where(w > 0)[0]
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in active:
        m = k - (n - 1)
        wk = w[k]
        # Term 1: (1-h[j+m]) for j in [max(0,-m), min(n, n-m))
        j1s = max(0, -m)
        j1e = min(n, n - m)
        if j1e > j1s:
            grad[j1s:j1e] += wk * one_minus_h[j1s + m:j1e + m]
        # Term 2: -h[j-m] for j in [max(0,m), min(n, n+m))
        j2s = max(0, m)
        j2e = min(n, n + m)
        if j2e > j2s:
            grad[j2s:j2e] -= wk * h[j2s - m:j2e - m]
    return grad

    """Projected gradient descent on smooth max with backtracking."""
    h = h0.copy()

    alpha = alpha0
    for _ in range(n_steps):
        grad = smooth_max_grad_h(h, tau)
        # Backtracking line search
        for _bt in range(20):
            h_try = project_box_sum(h - alpha * grad, target)
            f_try = smooth_max_val(np.correlate(h_try, 1.0 - h_try, mode="full"), tau)
            if f_try < f_prev:
                break
            alpha *= 0.5
        else:
            break  # No improvement
        h = h_try
        f_prev = f_try
        alpha = min(alpha * 1.1, alpha0)  # Slowly increase step size
    return h

            best_c5 = c5_new

    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
