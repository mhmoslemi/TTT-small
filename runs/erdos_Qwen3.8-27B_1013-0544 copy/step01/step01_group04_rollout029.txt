def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_smoothed_grad(h, dx, sigma):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_vals = np.exp((corr - max_c) / sigma)
    Z = np.sum(exp_vals)
    weights = exp_vals / Z
    obj = (max_c + sigma * np.log(Z)) * dx
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-10:
            continue
        k = m - (n - 1)
        if k >= 0:
            for i in range(n - k):
                grad[i] += w * (1.0 - h[i + k])
                grad[i + k] -= w * h[i]
        else:
            kk = -k
            for i in range(kk, n):
                grad[i] += w * (1.0 - h[i - kk])
                grad[i - kk] -= w * h[i]
    grad *= dx
    return obj, grad

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = np.sum(proj) - target
    if abs(diff) > 1e-14:
        free = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if np.any(free):
            proj[free] -= diff / np.sum(free)
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def slsqp_optimize(h_start, n, dx, max_iter=200):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraint = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(
        lambda x: compute_c5(x, dx),
        h_start, method='SLSQP', bounds=bounds,
        constraints=constraint,
        options={'maxiter': max_iter, 'ftol': 1e-14}
    )
    return project_feasible(res.x, n)

def smoothed_grad_descent(h, n, dx, sigma_start, sigma_end, max_iter, lr_init):
    h = h.copy()
    best = compute_c5(h, dx)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        frac = it / max_iter
        sigma = sigma_start * (sigma_end / sigma_start) ** frac
        _, grad = compute_smoothed_grad(h, dx, sigma)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new, dx)
        if c5 < best:
            best = c5
            best_h = h_new.copy()
        h = h_new
        lr *= 0.998
    return best_h, best

def block_perturb(h, rng, n):
    h = h.copy()
    n_blocks = rng.randint(2, 5)
    block_size = max(1, n // (2 * n_blocks))
    for _ in range(rng.randint(1, n_blocks + 1)):
        start = rng.randint(0, n - block_size)
        h[start:start + block_size] = 1.0 - h[start:start + block_size]
    return project_feasible(h, n)

def gaussian_bump(h, rng, n):
    h = h.copy()
    center = rng.randint(0, n)
    width = rng.uniform(3, 10)
    amp = rng.uniform(-0.3, 0.3)
    x = np.arange(n)
    h += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    return project_feasible(h, n)

def coord_descent_sweep(h, n, dx, max_sweeps=5):
    h = h.copy()
    best = compute_c5(h, dx)
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            h_old = h[j]
            def obj_1d(v):
                h[j] = v
                return compute_c5(h, dx)
            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-8})
            h[j] = res.x
            h = project_feasible(h, n)
            c5_now = compute_c5(h, dx)
            if c5_now < best - 1e-14:
                best = c5_now
                improved = True
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n

    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)

    def elapsed():
        return time.time() - start_time

    # Phase 1: Smoothed gradient descent with multiple sigma schedules (250s)
    phase1_end = 250
    sigma_scheds = [(0.5, 0.02, 0.005), (1.0, 0.05, 0.01), (0.3, 0.01, 0.003)]
    sched_idx = 0
    while elapsed() < phase1_end:
        ss, se, lr = sigma_scheds[sched_idx % len(sigma_scheds)]
        if sched_idx > 0:
            h_try = best_h + rng.randn(n) * 0.04
            h_try = project_feasible(h_try, n)
        else:
            h_try = best_h.copy()
        h_new, c5_new = smoothed_grad_descent(h_try, n, dx, ss, se, 300, lr)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        sched_idx += 1

    # Phase 2: SLSQP with diverse restarts (400s)
    phase2_end = 650
    attempt = 0
    no_improve_count = 0
    while elapsed() < phase2_end:
        if attempt == 0:
            h_try = best_h.copy()
        elif attempt % 4 == 1:
            h_try = block_perturb(best_h, rng, n)
        elif attempt % 4 == 2:
            h_try = gaussian_bump(best_h, rng, n)
        elif attempt % 4 == 3:
            h_try = best_h + rng.randn(n) * (0.06 + 0.04 * rng.rand())
            h_try = project_feasible(h_try, n)
        else:
            h_try = project_feasible(1.0 - best_h[::-1], n)
            h_try = 0.5 * h_try + 0.5 * project_feasible(best_h, n)

        h_new = slsqp_optimize(h_try, n, dx, max_iter=150)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
            no_improve_count = 0
        else:
            no_improve_count += 1
        attempt += 1
        if no_improve_count > 30:
            no_improve_count = 0

    # Phase 3: Coordinate descent sweeps (200s)
    phase3_end = budget_s - 80
    while elapsed() < phase3_end:
        h_cd, c5_cd = coord_descent_sweep(best_h, n, dx, max_sweeps=3)
        if c5_cd < best_c5 - 1e-12:
            best_h = h_cd
            best_c5 = c5_cd
        else:
            h_pert = gaussian_bump(best_h, rng, n)
            h_cd, c5_cd = coord_descent_sweep(h_pert, n, dx, max_sweeps=3)
            if c5_cd < best_c5 - 1e-12:
                best_h = h_cd
                best_c5 = c5_cd

    # Phase 4: Final SLSQP polish
    if elapsed() < budget_s - 15:
        h_final = slsqp_optimize(best_h, n, dx, max_iter=400)
        c5_final = compute_c5(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    # One more coord descent pass on the final result
    if elapsed() < budget_s - 5:
        h_cd, c5_cd = coord_descent_sweep(best_h, n, dx, max_sweeps=2)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    return (best_h, best_c5, len(best_h))
