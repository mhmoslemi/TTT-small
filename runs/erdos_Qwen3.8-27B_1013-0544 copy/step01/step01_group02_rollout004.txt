import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def slsqp_refine(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def systematic_swaps(h, n, deltas, max_rounds=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    interior = np.where((best_h > 1e-6) & (best_h < 1.0 - 1e-6))[0]
    for rnd in range(max_rounds):
        improved = False
        for i in range(len(interior)):
            for j in range(i + 1, len(interior)):
                ii, jj = interior[i], interior[j]
                for d in deltas:
                    for s in [1, -1]:
                        di = s * d
                        hi = best_h[ii] + di
                        hj = best_h[jj] - di
                        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
                            continue
                        best_h[ii] = hi
                        best_h[jj] = hj
                        c = compute_c5(best_h, n)
                        if c < best_c5:
                            best_c5 = c
                            improved = True
                        else:
                            best_h[ii] -= di
                            best_h[jj] += di
        if not improved:
            break
    return best_h, best_c5

def active_subgradient(h, n, top_k=5):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    top_idx = np.argsort(c)[-top_k:]
    grad = np.zeros(n)
    for active in top_idx:
        lag = int(active) - (n - 1)
        vs1 = max(0, lag)
        ve1 = min(n, n + lag)
        if vs1 < ve1:
            idx = np.arange(vs1, ve1)
            grad[idx] += (1.0 - h[idx - lag])
        vs2 = max(0, -lag)
        ve2 = min(n, n - lag)
        if vs2 < ve2:
            idx = np.arange(vs2, ve2)
            grad[idx] -= h[idx + lag]
    return grad / top_k

def subgradient_descent(h, n, max_iters=3000, rate0=0.02):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    cur_h = h.copy()
    for it in range(max_iters):
        rate = rate0 / (1.0 + 0.005 * it)
        g = active_subgradient(cur_h, n, top_k=5)
        cur_h = project_feasible(cur_h - rate * g, n)
        c = compute_c5(cur_h, n)
        if c < best_c5:
            best_h = cur_h.copy()
            best_c5 = c
    return best_h, best_c5
