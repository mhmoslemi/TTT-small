def corr_gradient(h, d):
    """Compute gradient of correlation at lag d w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    # full[n-1+d] = sum_i h[i+d] * (1-h[i])
    # d/dh[j] = (1-h[j-d]) [when j-d is valid, i.e., j >= d] - h[j+d] [when j+d < n]
    if d >= 0:
        # h[i+d] terms: j = i+d, so i = j-d, valid when j >= d
        # contribution: (1-h[i]) = (1-h[j-d])
        valid_j = range(d, n)
        for j in valid_j:
            grad[j] += (1 - h[j-d])
        # -h[i+d] terms: j = i, so i+d = j+d, valid when j+d < n
        valid_j2 = range(0, n-d)
        for j in valid_j2:
            grad[j] -= h[j+d]
    else:
        dd = -d
        # h[i-dd] terms: j = i-dd, so i = j+dd, valid when j+dd < n
        # contribution: (1-h[i]) = (1-h[j+dd])

        valid_j2 = range(dd, n)
        for j in valid_j2:
            grad[j] -= h[j-dd]
    return grad * (2.0/n)  # dx factor

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Binary search for the right threshold
    # We want to add 'diff' to h while keeping it in [0,1]
    # This is like finding a water level
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_trial = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_trial) - target
        if s > 0:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo+hi)/2, 0.0, 1.0)
    # Final correction
    diff2 = target - np.sum(h)
    if diff2 != 0:
        mask = (h < 1.0) if diff2 > 0 else (h > 0.0)
        h[mask] += diff2 / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_at_lag(h, d):
    """Correlation value at lag d (where full index = n-1+d)"""
    n = len(h)
    dx = 2.0 / n
    if d >= 0:
        m = min(n - d, n)
        return np.sum(h[d:d+m] * (1 - h[:m])) * dx
    else:
        dd = -d
        m = min(n - dd, n)
        return np.sum(h[:m] * (1 - h[dd:dd+m])) * dx

def corr_gradient(h, d):
    """Gradient of correlation at lag d w.r.t. h (including dx)"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if d >= 0:
        m = min(n - d, n)
        if m > 0:
            grad[d:d+m] += (1 - h[:m])
            grad[:m] -= h[d:d+m]
    else:

        m = min(n - dd, n)
        if m > 0:
            grad[:m] += (1 - h[dd:dd+m])
            grad[dd:dd+m] -= h[:m]

def pnorm_grad(h, p):
    corr = full_corr(h)
    val = pnorm_obj(h, p)
    # d/dh of (sum corr^p)^(1/p) = (1/p) * (sum corr^p)^(1/p - 1) * p * corr^(p-1) * d(corr)/dh
    # = (sum corr^p)^(1/p - 1) * corr^(p-1) * d(corr)/dh

    h1, c51 = subgrad_opt(h, n_iter=8000, K=5)
    elapsed = time.time() - t0
    if elapsed > budget - 40:
        return h1, c51
    h2, c52 = slsqp_tvar(h1, max_iter=300)
    if c52 < c51:
        h1, c51 = h2, c52
    return h1, c51

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_h(h, n * 0.5)
    c5 = compute_c5(h)
    h_best = h.copy()
    c5_best = c5

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Subgradient with momentum (fast, many iterations)
    if remaining() > 200:
        h_new, c5_new = subgrad_opt(h, n_iter=50000, K=5)
        if c5_new < c5_best:
            h, c5 = h_new, c5_new
            h_best, c5_best = h.copy(), c5

    # Phase 2: SLSQP t-variable refinement
    if remaining() > 150:
        h_new, c5_new = slsqp_tvar(h, max_iter=600, ftol=1e-12)
        if c5_new < c5_best:
            h, c5 = h_new, c5_new
            h_best, c5_best = h.copy(), c5

    # Phase 3: Another subgradient pass (different K)
    if remaining() > 150:
        h_new, c5_new = subgrad_opt(h, n_iter=40000, K=3)
        if c5_new < c5_best:
            h, c5 = h_new, c5_new
            h_best, c5_best = h.copy(), c5

    # Phase 4: p-norm SLSQP (smooth surrogate)
    if remaining() > 120:
        for p in [40, 100]:
            if remaining() < 60:
                break
            h_new, c5_new = slsqp_pnorm(h, p, max_iter=250)
            if c5_new < c5_best:
                h, c5 = h_new, c5_new
                h_best, c5_best = h.copy(), c5

    # Phase 5: Final SLSQP polish
    if remaining() > 80:
        h_new, c5_new = slsqp_tvar(h_best, max_iter=800, ftol=1e-13)
        if c5_new < c5_best:
            h_best, c5_best = h_new.copy(), c5_new

    # Phase 6: Basin hopping with subgradient recovery
    if remaining() > 60:
        scales = [0.02, 0.05, 0.01, 0.04, 0.08, 0.03, 0.06]
        for scale in scales:
            if remaining() < 50:
                break
            h_p = h_best + rng.randn(n) * scale
            h_p = project_h(h_p, n * 0.5)
            h_c, c5_c = try_candidate(h_p, t0, budget_s)
            if c5_c < c5_best:
                h_best, c5_best = h_c.copy(), c5_c

    # Phase 7: Try coarser resolutions
    for n_coarse in [90, 120]:
        if remaining() < 120:
            break
        h_c0 = interp_h(h_best, n_coarse)
        h_c, c5_c = try_candidate(h_c0, t0, budget_s)
        if c5_c < c5_best:
            h_best, c5_best = h_c.copy(), c5_c

    # Phase 8: Final polish on best
    if remaining() > 50:
        h_new, c5_new = slsqp_tvar(h_best, max_iter=500, ftol=1e-13)
        if c5_new < c5_best:
            h_best, c5_best = h_new.copy(), c5_new

    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
