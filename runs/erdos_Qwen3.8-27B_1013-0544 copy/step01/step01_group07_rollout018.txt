import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    return (m + (1.0/t) * np.log(np.sum(np.exp(t*(corr - m))))) * dx

def project_h(vec, target_sum):
    """Project onto {x: 0<=x<=1, sum(x)=target_sum}"""
    n = len(vec)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(vec - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(vec - (lo+hi)/2.0, 0.0, 1.0)

def grad_corr_at_lag(h, lag):
    n = len(h)
    i = np.arange(n)
    t1 = np.zeros(n)
    t2 = np.zeros(n)
    idx1 = i - lag
    m1 = (idx1 >= 0) & (idx1 < n)
    t1[m1] = 1.0 - h[idx1[m1]]
    idx2 = i + lag
    m2 = (idx2 >= 0) & (idx2 < n)
    t2[m2] = -h[idx2[m2]]
    return t1 + t2

    best = h.copy()
    best_c5 = compute_c5(h, dx)
    corr_full = np.correlate(h, 1-h, mode="full")
    
    for it in range(n_iters):
        # Compute correlations
        corr = np.correlate(h, 1-h, mode="full")
        # Find top m
        top_idx = np.argsort(corr)[-top_m:]
        top_corr = corr[top_idx]
        w = top_corr / top_corr.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for j, k_idx in enumerate(top_idx):
            lag = k_idx - (n-1)
            grad += w[j] * grad_corr_at_lag(h, lag)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:

        alpha = min(alpha, 0.1)  # cap step size

            best = h_new.copy()
            best_c5 = c5_new

def reweighting_optimization(h, dx, target_sum, n_iters=100):
    n = len(h)
    n_corr = 2*n - 1
    w = np.ones(n_corr) / n_corr
    
    for it in range(n_iters):
        # Solve: minimize sum_k w[k] * corr_k(h) subject to constraints
        # This is a quadratic program (since corr_k is bilinear in h)
        # Actually it's: minimize sum_k w[k] * sum_i h[i] * (1-h[m(i,k)])
        # = sum_i h[i] - sum_k w[k] * sum_i h[i]*h[m(i,k)]
        # The first term is linear, the second is quadratic (but indefinite)
        # So it's a non-convex QP in general
        
        # For now, use SLSQP
        def obj(h_vec):
            corr = np.correlate(h_vec, 1-h_vec, mode="full")
            return float(np.dot(w, corr)) * dx
        
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0,1)]*n,
                      constraints=[{'type':'eq','fun':lambda x: np.sum(x)-target_sum}],
                      options={'maxiter':200})
        h_new = project_h(res.x, target_sum)
        
        # Update weights
        corr_new = np.correlate(h_new, 1-h_new, mode="full")
        # Exponential update: w[k] proportional to exp(eta * corr_new[k])
        eta = 5.0
        w = np.exp(eta * (corr_new - np.max(corr_new)))
        w = w / w.sum()
        
        h = h_new
    
    return h, compute_c5(h, dx)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    return (m + (1.0/t) * np.log(np.sum(np.exp(t*(corr - m))))) * dx

def project_h(vec, target_sum):
    n = len(vec)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(vec - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(vec - (lo + hi) / 2.0, 0.0, 1.0)

def grad_at_lag(h, lag):
    n = len(h)
    i = np.arange(n)
    g = np.zeros(n)
    idx1 = i - lag
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += 1.0 - h[idx1[m1]]
    idx2 = i + lag
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def nesterov_subgrad(h, dx, target_sum, time_limit, top_m_start=1, top_m_end=6):
    t0 = time.time()
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    h_prev = h.copy()
    step = 0.04
    it = 0
    stagnation = 0
    
    while time.time() - t0 < time_limit:
        it += 1
        frac = min(1.0, it / 2000.0)
        top_m = max(1, int(round(top_m_start + frac * (top_m_end - top_m_start))))
        
        corr = np.correlate(h, 1 - h, mode="full")
        top_idx = np.argsort(corr)[-top_m:]
        top_vals = corr[top_idx]
        w = top_vals / top_vals.sum()
        
        grad = np.zeros(n)
        for j in range(top_m):
            lag = int(top_idx[j]) - (n - 1)
            grad += w[j] * grad_at_lag(h, lag)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Nesterov momentum
        momentum = (it + 0.5 - 1.0) / (it + 0.5 + 2.0)
        y = h + momentum * (h - h_prev)
        y = project_h(y, target_sum)
        
        h_prev = h.copy()
        h_new = project_h(y - step * grad / gnorm, target_sum)
        
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best = h_new.copy()
            best_c5 = c5_new
            stagnation = 0
        else:
            stagnation += 1
            if stagnation % 50 == 0:
                step *= 0.9
                if step < 0.005:
                    step = 0.005
                    h = project_h(best + 0.05 * np.random.randn(n), target_sum)
                    h_prev = h.copy()
                    stagnation = 0
        
        h = h_new
        step = min(0.06, step * 1.001)
    
    return best, best_c5

def block_slsqp(h, dx, target_sum, block_size=4, n_passes=4, t=30):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for p in range(n_passes):
        rng = np.random.RandomState(p * 7919 + 13)
        order = list(range(0, n, block_size))
        rng.shuffle(order)
        
        for start in order:
            end = min(start + block_size, n)
            if end - start < 2:
                continue
            idx = np.arange(start, end)
            bs = len(idx)
            tgt = target_sum * bs / n
            
            def obj(x, idx=idx):
                ht = best.copy()
                ht[idx] = x
                return smooth_max_obj(ht, t, dx)
            
            try:
                res = minimize(obj, best[idx].copy(), method='SLSQP',
                              bounds=[(0.0, 1.0)] * bs,
                              constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt}],
                              options={'maxiter': 60, 'ftol': 1e-9})
                best[idx] = np.clip(res.x, 0.0, 1.0)
                c5_new = compute_c5(best, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
            except Exception:
                pass
    
    return best, best_c5
