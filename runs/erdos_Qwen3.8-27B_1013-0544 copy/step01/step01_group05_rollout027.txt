import numpy as np
from scipy.optimize import minimize
import time

_G_DX = None
_G_TGT = None
_QP_HCUR = None
_QP_LAGS = None
_QP_MU = None

def project_feasible(x, target_sum):
    lo = -1.0
    hi = 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def active_gradient(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for L in lags:
        L = int(L)
        if L == 0:
            grad[:] += 1.0 - 2.0 * h
        elif L > 0:
            grad[:n-L] += 1.0 - h[L:]
            grad[L:] -= h[:n-L]
        else:
            L2 = -L
            grad[L2:] += 1.0 - h[:n-L2]
            grad[:n-L2] -= h[L2:]
    grad /= float(len(lags))
    return grad

def _slsqp_f(x):
    return compute_c5(x, _G_DX)

def _slsqp_ieq(x):
    return np.sum(x) - _G_TGT

def _qp_obj(x):
    n = len(x)
    quad_sum = 0.0
    for L in _QP_LAGS:
        L = int(L)
        if L == 0:
            quad_sum += np.sum(x) - np.sum(x * x)
        elif L > 0:
            quad_sum += np.sum(x[:n-L]) - np.sum(x[:n-L] * x[L:])
        else:
            L2 = -L
            quad_sum += np.sum(x[L2:]) - np.sum(x[L2:] * x[:n-L2])
    return 0.5 * np.sum((x - _QP_HCUR)**2) + (_QP_MU / len(_QP_LAGS)) * quad_sum

def _qp_grad(x):
    n = len(x)
    g = (x - _QP_HCUR).copy()
    sg = np.zeros(n)
    for L in _QP_LAGS:
        L = int(L)
        if L == 0:
            sg[:] += 1.0 - 2.0 * x
        elif L > 0:
            sg[:n-L] += 1.0 - x[L:]
            sg[L:] -= x[:n-L]
        else:
            L2 = -L
            sg[L2:] += 1.0 - x[:n-L2]
            sg[:n-L2] -= x[L2:]
    g += (_QP_MU / len(_QP_LAGS)) * sg
    return g

def fista(h0, dx, tgt, n_iter):
    n = len(h0)
    h = h0.copy()
    hp = h0.copy()
    t = 1.0
    step = 0.008
    bh = h0.copy()
    bc = compute_c5(h0, dx)
    for _ in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        cm = float(np.max(corr))
        tol = max(0.05, 0.008 * cm)
        idx = np.where(corr >= cm - tol)[0]
        if len(idx) < 3:
            idx = np.argsort(corr)[-5:]
        elif len(idx) > 20:
            idx = np.argsort(corr)[-20:]
        lags = idx - (n - 1)
        g = active_gradient(h, lags)
        y = h + ((t - 1.0) / (t + 1.0)) * (h - hp)
        h_new = project_feasible(y - step * g, tgt)
        c5n = compute_c5(h_new, dx)
        if c5n < bc:
            bc = c5n
            bh = h_new.copy()
        if c5n > bc + 1e-8:
            step *= 0.8
        else:
            step = min(step * 1.015, 0.04)
        hp = h
        h = h_new
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
    return bh, bc

def qp_refine(h0, dx, tgt, n_rounds, time_limit):
    global _QP_HCUR, _QP_LAGS, _QP_MU, _G_DX, _G_TGT
    start = time.time()
    n = len(h0)
    h = h0.copy()
    bh = h0.copy()
    bc = compute_c5(h0, dx)
    mu = 1.0
    _G_DX = dx
    _G_TGT = tgt
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_ieq}]
    for rnd in range(n_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cm = float(np.max(corr))
        tol = max(0.03, 0.005 * cm)
        idx = np.where(corr >= cm - tol)[0]
        if len(idx) < 3:
            idx = np.argsort(corr)[-5:]
        elif len(idx) > 15:
            idx = np.argsort(corr)[-15:]
        lags = list(idx - (n - 1))
        _QP_HCUR = h.copy()
        _QP_LAGS = lags
        _QP_MU = mu
        try:
            res = minimize(_qp_obj, h, jac=_qp_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-12})
            if res.success and res.x is not None:
                h_new = project_feasible(res.x, tgt)
                c5n = compute_c5(h_new, dx)
                if c5n < bc:
                    bc = c5n
                    bh = h_new.copy()
                h = h_new
        except Exception:
            pass
        mu *= 0.7
    return bh, bc

    start_time = time.time()
    rng = np.random.RandomState(seed)

    bounds = [(0.0, 1.0)] * n_points

    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)

    # Phase 1: FISTA (main global-ish search) - 200s
    if tleft() > 250:
        h_opt, c5_opt = fista(h0, dx, tgt, 4000)
        if c5_opt < best_c5:

            best_h = h_opt

    # Phase 2: QP refinement - 250s

        if c5_opt < best_c5:

        if c5_opt < best_c5:

            best_h = h_opt

    # Phase 5: SLSQP direct on original objective - 100s
    if tleft() > 80:
        try:

            if res.x is not None:

    if tleft() > 30:

            tol = max(0.01, 0.002 * cm)
            idx = np.where(corr >= cm - tol)[0]
            if len(idx) < 1:
                idx = np.array([np.argmax(corr)])

            lags = idx - (n_points - 1)

            c5n = compute_c5(h_new, dx)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()
            else:
                step *= 0.95
                if step < 1e-10:

            h = h_new

    # Ensure feasibility
    best_h = project_feasible(best_h, tgt)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h, dx)

    return (best_h, best_c5, n_points)
