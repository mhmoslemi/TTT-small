def lsq_step(h, dx, tau, step_size, momentum, mu, target_sum):
    """One log-sum-exp gradient step with momentum and projection."""
    n = len(h)
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = tau * corr * dx
    scaled_max = np.max(scaled)
    weights = np.exp(scaled - scaled_max)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= wsum
    
    # Only use top lags for efficiency
    top_k = min(10, len(corr))
    top_idx = np.argsort(weights)[-top_k:]
    top_weights = weights[top_idx]
    tw_sum = top_weights.sum()
    if tw_sum > 0:
        top_weights /= tw_sum
    else:
        top_weights = np.zeros_like(top_weights)
        top_weights[-1] = 1.0
    
    grad = np.zeros(n)
    for i, idx in enumerate(top_idx):
        L = int(idx) - n_half
        grad += top_weights[i] * compute_grad_lag(h, L)
    
    momentum = mu * momentum - step_size * grad
    h_new = project_feasible(h + momentum, target_sum)
    return h_new, momentum, float(np.max(corr)) * dx
