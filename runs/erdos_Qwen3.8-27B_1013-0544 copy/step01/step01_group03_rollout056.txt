    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)

    h = np.asarray(h, dtype=float).copy()

        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()

            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:

from scipy.optimize import minimize, differential_evolution, Bounds
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_c5(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    log_sum = np.log(np.sum(np.exp(corr - cmax)))
    return float((cmax + log_sum / tau) * dx)


def fix_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def _slsqp_obj(x, tau, target):
    return smooth_max_c5(x, tau)


def _slsqp_cons(x, tau, target):
    return np.sum(x) - target


def _slsqp_jac(x, tau, target):
    return np.ones(len(x))


def do_slsqp(h0, target, tau, maxiter=300):

        res = minimize(_slsqp_obj, h0, args=(tau, target), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        x = fix_sum(res.x, target)
        return x, evaluate_c5(x)

        return h0, evaluate_c5(h0)
