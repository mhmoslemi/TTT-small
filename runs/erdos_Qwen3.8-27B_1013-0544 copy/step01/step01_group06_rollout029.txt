import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / n)

def grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n-k]
    else:
        p = -k
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0.0, n_old, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def optimize_h(h0, target_sum, deadline, rng, k_init=3, k_max=15, lr0=0.02, beta=0.85):
    n = len(h0)
    h = project_box_sum(h0.copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    k_top = k_init
    mom = np.zeros(n)
    no_imp = 0
    stalls = 0
    
    while time.time() < deadline:
        corr = compute_corr(h)
        cur_max = float(np.max(corr))
        cur_c5 = cur_max * (2.0 / n)
        
        # Top-k gradient
        top_idx = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(n)
        for idx in top_idx:
            grad += grad_lag(h, int(idx) - (n - 1))
        grad *= (2.0 / (n * k_top))
        
        # Momentum
        mom = beta * mom + (1.0 - beta) * grad
        direction = -mom
        
        # Backtracking
        step = lr
        accepted = False
        for _ in range(10):
            h_try = project_box_sum(h + step * direction, target_sum)
            c_try = float(np.max(compute_corr(h_try))) * (2.0 / n)
            if c_try < cur_c5 - 1e-14:
                h = h_try
                accepted = True
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h.copy()
                no_imp = 0
                break
            step *= 0.5

                h = project_box_sum(h + rng.randn(n) * sigma, target_sum)

                mom[:] = 0

            elif best_c5 > 0.39:

            if no_imp == 0:
                lr = min(lr * 1.03, 0.08)
    
    return best_h, best_c5
