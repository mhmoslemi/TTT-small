            pop[i] = best_h + rng.randn(n) * rng.uniform(0.05, 0.15)
        elif r < 0.65:
            pop[i] = rng.random(n)
        else:
            k = rng.randint(2, 7)
            base = rng.random(k)
            rng_range = base.max() - base.min()
            if rng_range > 1e-10:
                base = (base - base.min()) / rng_range
            else:
                base = 0.5 * np.ones(k)
            pop[i] = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, k), base)
        pop[i] = project_feasible(pop[i], n)
        fit[i] = softmax_c5(pop[i], dx, 30)

    F = 0.7
    Cr = 0.6
    de_gen = 0

    while elapsed() < phase2_end:
        de_generation(pop, fit, rng, F, Cr, n, dx, target_sum, 30)
        bi = np.argmin(fit)
        c5_check = compute_c5(pop[bi], dx)
        if c5_check < best_c5:
            best_h = pop[bi].copy()
            best_c5 = c5_check

        if de_gen % 20 == 10:
            h_pol = slsqp_soft(pop[bi], n, dx, target_sum, 30, max_iter=150)
            c5_pol = compute_c5(h_pol, dx)
            if c5_pol < best_c5:
                best_h = h_pol
                best_c5 = c5_pol
                pop[0] = best_h.copy()
                fit[0] = softmax_c5(best_h, dx, 30)

        if de_gen % 60 == 59:
            F = max(0.3, F * 0.9)
            Cr = min(0.9, Cr + 0.02)
        de_gen += 1

    # Phase 3: Upsample to 2n, optimize there
    phase3_end = budget_s * 0.78
    n2 = 2 * n
    dx2 = 2.0 / n2
    target2 = n2 / 2.0
    h_up = project_feasible(np.repeat(best_h, 2), n2)
    c5_up = compute_c5(h_up, dx2)

    h_up_curr = h_up.copy()
    for lam in [20, 50, 100]:
        if elapsed() > phase3_end:
            break
        h_new = slsqp_soft(h_up_curr, n2, dx2, target2, lam, max_iter=200)
        c5_new = compute_c5(h_new, dx2)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            h_up_curr = h_new.copy()
        elif c5_new < c5_up:
            h_up_curr = h_new.copy()
            c5_up = c5_new

    # Short DE at n2
    N_pop2 = 20
    pop2 = np.zeros((N_pop2, n2))
    fit2 = np.zeros(N_pop2)
    pop2[0] = h_up_curr.copy()
    fit2[0] = softmax_c5(h_up_curr, dx2, 30)
    for i in range(1, N_pop2):
        pop2[i] = h_up_curr + rng.randn(n2) * 0.05
        pop2[i] = project_feasible(pop2[i], n2)
        fit2[i] = softmax_c5(pop2[i], dx2, 30)

    de2_gen = 0
    while elapsed() < phase3_end:
        de_generation(pop2, fit2, rng, 0.5, 0.7, n2, dx2, target2, 30)
        bi2 = np.argmin(fit2)
        c5_check2 = compute_c5(pop2[bi2], dx2)
        if c5_check2 < best_c5:
            best_h = pop2[bi2].copy()
            best_c5 = c5_check2
        if de2_gen % 15 == 7:
            h_pol2 = slsqp_soft(pop2[bi2], n2, dx2, target2, 30, max_iter=100)
            c5_pol2 = compute_c5(h_pol2, dx2)
            if c5_pol2 < best_c5:
                best_h = h_pol2
                best_c5 = c5_pol2
        de2_gen += 1

    # Phase 4: Final high-precision polishing at best resolution
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    target_final = n_final / 2.0
    phase4_end = budget_s - 15

    for lam in [100, 200, 400]:
        if elapsed() > phase4_end:
            break
        h_new = slsqp_soft(best_h, n_final, dx_final, target_final, lam, max_iter=400)
        c5_new = compute_c5(h_new, dx_final)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Fine multi-restart
    while elapsed() < phase4_end:
        scale = rng.uniform(0.003, 0.01)
        h_try = best_h + rng.randn(n_final) * scale
        h_try = project_feasible(h_try, n_final)
        h_try = slsqp_soft(h_try, n_final, dx_final, target_final, 200, max_iter=200)
        c5_try = compute_c5(h_try, dx_final)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try

    return (best_h, best_c5, n_final)
