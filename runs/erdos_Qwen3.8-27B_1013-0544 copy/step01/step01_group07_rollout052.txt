        h2, c2 = descent(h2, dx, s, 0.025, 4000, kk, 200, 'topk')

            best_h, best_c = h2, c2

    # Phase 6: Final polish with small steps
    if time.time() - t0 < budget_s - 5:
        h, c = descent(best_h, dx, s, 0.008, 10000, 3, 600, 'topk')
        if c < best_c:
            best_h, best_c = h, c

    best_h = np.clip(best_h, 0, 1)
    c5 = compute_c5(best_h, dx)
    return best_h, c5, n
