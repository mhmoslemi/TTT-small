import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-14:
        mask = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj


def lse_value_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exp_vals = np.exp(tau * (corr - m))
    p = exp_vals / np.sum(exp_vals)
    val = m + (1.0 / tau) * np.log(np.sum(exp_vals))
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        if p[i] < 1e-16:
            continue
        s = i - (n - 1)
        if s >= 0:
            grad[0:n - s] += p[i] * (1.0 - h[s:n])
            if n - s > 0:
                grad[s:] -= p[i] * h[0:n - s]
        else:
            ns = -s
            grad[ns:] += p[i] * (1.0 - h[0:n - ns])
            if n - ns > 0:
                grad[0:n - ns] -= p[i] * h[ns:]
    grad *= dx
    return val, grad


def lse_pen_obj(h, tau, mu, n):
    val, grad = lse_value_grad(h, tau)
    penalty = mu * (np.sum(h) - 0.5 * n) ** 2
    pgrad = 2.0 * mu * (np.sum(h) - 0.5 * n) * np.ones(n)
    return val + penalty, grad + pgrad


def opt_lse_lbfgs(h0, tau, mu=500.0, max_iter=600):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def fn(h):
        v, g = lse_pen_obj(h, tau, mu, n)
        return v, g

    res = minimize(fn, h0.copy(), jac=True, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def t_objective(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def opt_t_slsqp(h0, max_iter=800, ftol=1e-13):
    n = len(h0)
    c5 = compute_c5(h0)
    v0 = np.append(h0.copy(), c5)
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0 / n)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def active_set_opt(h0, top_k=50, max_iter=500):
    n = len(h0)
    dx = 2.0 / n
    corr = np.correlate(h0, 1.0 - h0, mode='full') * dx
    active_idx = np.argsort(-corr)[:top_k]

    def obj(v):
        return v[-1]

    def ineq(v):
        hh = v[:n]
        t = v[-1]
        c = np.correlate(hh, 1.0 - hh, mode='full') * dx
        return t - c[active_idx]

    def eqf(v):
        return np.sum(v[:n]) - n * 0.5

    def eqj(v):
        j = np.zeros(len(v))
        j[:n] = 1.0
        return j

    c5 = compute_c5(h0)
    v0 = np.append(h0.copy(), c5)
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0 / n)]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf, 'jac': eqj}]
    res = minimize(obj, v0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-13})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def make_perturbation(base, rng, scale):
    h = base + rng.randn(len(base)) * scale
    return project_h(h, len(base) * 0.5)
