```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr * dx  # scaled for numerical stability
    
    # Softmax weights
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    w = exp_x / exp_x.sum()  # weights sum to 1
    
    # Gradient of smooth max = sum_i w_i * grad f_i
    # where f_i = corr[i] * dx
    # grad f_i w.r.t. h[j] = dx * d(corr[i])/dh[j]
    
    N = n
    grad = np.zeros(N)
    
    for i in range(len(corr)):
        if w[i] < 1e-15:
            continue
        lag = i - (N - 1)
        n_min = max(0, -lag)
        n_max = min(N - 1, N - 1 - lag)
        for j in range(n_min, n_max + 1):
            j1 = j + lag
            j2 = j
            grad[j1] += w[i] * (1 - h[j2])
            grad[j2] -= w[i] * h[j1]
    
    grad *= dx
    return grad
```

```python
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(corr, h, tau, dx):
    n = len(h)
    x = tau * corr * dx
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    w = exp_x / exp_x.sum()
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for i in active:
        wi = w[i]
        lag = i - (n - 1)
        lo = max(0, -lag)
        hi = min(n - 1, n - 1 - lag)
        for j in range(lo, hi + 1):
            j1 = j + lag
            j2 = j
            grad[j1] += wi * (1.0 - h[j2])
            grad[j2] -= wi * h[j1]
    grad *= dx
    return grad

def slsqp_objective(x, tau, dx):
    c = np.correlate(x, 1.0 - x, mode="full")
    xv = tau * c * dx
    xm = xv.max()
    return float(xm / tau + np.log(np.sum(np.exp(xv - xm))) / tau)

def optimize_pg(h_init, target, tau_list, max_iter, t0, budget):
    n = len(h_init)
    dx = 2.0 / n
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    for tau in tau_list:
        elapsed = time.time() - t0
        if budget - elapsed < 10:
            break
        h_cur = best_h.copy()
        step = 0.5
        no_improve = 0
        for it in range(max_iter):
            elapsed = time.time() - t0
            if budget - elapsed < 5:
                break
            corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
            cur_c5 = float(np.max(corr) * dx)
            grad = smooth_max_grad(corr, h_cur, tau, dx)
            h_try = project_feasible(h_cur - step * grad, target)
            corr_try = np.correlate(h_try, 1.0 - h_try, mode="full")
            c5_try = float(np.max(corr_try) * dx)
            if c5_try < cur_c5 - 1e-15:
                h_cur = h_try
                no_improve = 0
                step = min(step * 1.03, 8.0)
                if c5_try < best_c5:
                    best_h = h_try.copy()
                    best_c5 = c5_try
            else:
                no_improve += 1
                step *= 0.75
                if no_improve > 60:
                    break
    return best_h, best_c5

def do_slsqp_polish(h_init, t0, budget):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    _sl_target = target
    bnds = Bounds(np.zeros(n), np.ones(n))
    h_best = h_init.copy()
    c5_best = evaluate_c5(h_best)
    for tau in [500, 2000, 5000]:
        if budget - (time.time() - t0) < 15:
            break
        try:
            res = minimize(slsqp_objective, h_best, args=(tau, dx),
                          method='SLSQP', bounds=bnds,
                          constraints=[{'type': 'eq', 'fun': _sl_eq_fun, 'jac': _sl_eq_jac}],
                          options={'maxiter': 400, 'ftol': 1e-15})
            x = project_feasible(res.x, target)
            c5 = evaluate_c5(x)
            if c5 < c5_best:
                h_best = x
                c5_best = c5
        except Exception:
            pass
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(120, 0.5)

    n = len(best_h)
    target = n / 2.0
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Main projected gradient optimization
    tau_list = [20, 50, 100, 200, 500, 1000, 3000]
    best_h, best_c5 = optimize_pg(best_h, target, tau_list, 500, t0, budget_s)

    # Phase 2: SLSQP polish
    if budget_s - (time.time() - t0) > 30:
        best_h, best_c5 = do_slsqp_polish(best_h, t0, budget_s)

    # Phase 3: Multi-start exploration
    n = len(best_h)
    target = n / 2.0
    n_starts = 20
    for s in range(n_starts):
        if budget_s - (time.time() - t0) < 20:
            break
        scale = 0.02 + 0.15 * (s / float(n_starts))
        h_pert = best_h + np.random.normal(0, scale, n)
        h_pert = project_feasible(h_pert, target)
        h_new, c5_new = optimize_pg(h_pert, target, [100, 500, 2000], 250, t0, budget_s)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: Try different resolutions
    n_cur = len(best_h)
    for tn in [70, 90, 100, 130, 140, 160]:
        if budget_s - (time.time() - t0) < 35:
            break
        if tn == n_cur:
            continue
        h_interp = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, n_cur), best_h)
        h_interp = project_feasible(h_interp, tn / 2.0)
        h_new, c5_new = optimize_pg(h_interp, tn / 2.0, [50, 200, 1000, 3000], 350, t0, budget_s)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            n_cur = tn

    # Phase 5: Final SLSQP polish on best
    if budget_s - (time.time() - t0) > 25:
        best_h, best_c5 = do_slsqp_polish(best_h, t0, budget_s)

    # Phase 6: Additional projected gradient with very high tau
    if budget_s - (time.time() - t0) > 20:
        n = len(best_h)
        target = n / 2.0
        best_h, best_c5 = optimize_pg(best_h, target, [5000, 10000, 20000], 400, t0, budget_s)

    # Ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n_final
```
