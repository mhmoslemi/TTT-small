def make_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def obj_ls(x, n, alpha):
    """Log-sum-exp smooth surrogate."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    if alpha * (c.max() - m) > 700:
        return m + c.max() / alpha + np.log(n * 2 - 1) / alpha
    shifted = alpha * (c - m)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-14: break
            h += (diff / t) * room
        else:
            t = h.sum()
            if t < 1e-14: break
            h += (diff / t) * h
    return np.clip(h, 0.0, 1.0)

def binary_local_search(h, n, rng, max_time):
    """Local search on binary sequences using swaps."""
    t_start = time.time()
    best = h.copy()
    best_c5 = compute_c5(best, n)
    
    # Precompute: for each pair (i,j) with different values, the effect of swapping
    # We'll use a simpler approach: random swaps with annealing
    
    T = 0.1  # Temperature for simulated annealing
    T_min = 0.001
    cooling = 0.995
    
    for it in range(100000):
        if time.time() - t_start > max_time:
            break
        if T < T_min:
            T = T_min
        
        i = rng.randint(n)
        j = rng.randint(n)
        if best[i] == best[j]:
            continue
        
        # Swap
        old_i, old_j = best[i], best[j]
        best[i], best[j] = old_j, old_i
        c_try = compute_c5(best, n)
        
        if c_try < best_c5:
            best_c5 = c_try
        else:
            # Accept with probability exp(-(c_try - best_c5) / T)
            delta = c_try - best_c5
            if rng.rand() < np.exp(-delta / T):
                best_c5 = c_try
            else:
                best[i], best[j] = old_i, old_j
        
        T *= cooling
    
    return best, best_c5

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-14: break
            h = h + (diff / t) * room
        else:
            t = h.sum()
            if t < 1e-14: break
            h = h + (diff / t) * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def grad_c5(x, n):
    """Compute subgradient of C5(x) = max_k correlate(x, 1-x)[k] * dx."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full')
    max_val = c.max()
    # Find all lags that achieve (approximately) the max
    tol = 1e-10
    max_lags = np.where(c >= max_val - tol)[0]
    # Compute gradient for each max lag and average
    grad = np.zeros(n)
    for k_idx in max_lags:
        # k is the lag: k_idx - (n-1)
        k = k_idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            # c_k = sum_{i=0}^{n-1-k} x[i+k]*(1-x[i])
            for i in range(n - k):
                j = i + k  # index of x in first term
                g[j] += (1 - x[i])
                g[i] += (-x[i + k])
        else:
            # c_k = sum_{i=-k}^{n-1} x[i+k]*(1-x[i])
            # Let m = -k
            m = -k
            for i in range(m, n):
                j = i - m  # index of x in first term (i + k = i - m)
                g[j] += (1 - x[i])
                g[i] += (-x[i - m])
        grad += g
    grad /= len(max_lags)
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-14:
                break
            h = h + (diff / t) * room
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h = h + (diff / t) * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def obj_lse(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def sum_jacobian(x, n):
    return np.ones(n)

def gradient_max_corr(h, n, tol=1e-8):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_val = c.max()
    max_idx = np.where(c >= max_val - tol)[0]
    grad = np.zeros(n)
    nm1 = n - 1
    for k_idx in max_idx:
        k = k_idx - nm1
        g = np.zeros(n)
        if k >= 0:
            lo = 0
            hi = n - 1 - k
            for ii in range(lo, hi + 1):
                j1 = ii + k
                g[j1] += (1 - h[ii])
                g[ii] += (-h[ii + k])
        else:
            m = -k
            lo = m
            hi = n - 1
            for ii in range(lo, hi + 1):
                j1 = ii - m
                g[j1] += (1 - h[ii])
                g[ii] += (-h[ii - m])
        grad += g
    grad /= max(1, len(max_idx))
    return grad * dx

    t_start = time.time()
    h = h.copy()
    c5 = compute_c5(h, n)
    T = 0.04
    T_min = 0.00005
    best_h = h.copy()
    best_c5 = c5
    while time.time() - t_start < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c_new = compute_c5(h, n)
        if c_new < c5:
            c5 = c_new
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
        else:
            delta = c_new - c5
            if delta < 10 * T and rng.rand() < np.exp(-delta / T):
                c5 = c_new
            else:
                h[i], h[j] = h[j], h[i]
        T = max(T_min, T * 0.9985)
    return best_h, best_c5

def slsqp_refine(h, n, alpha_list, max_iter, bounds, cons, time_limit):
    t_start = time.time()
    h = h.copy()
    c5 = compute_c5(h, n)
    for alpha in alpha_list:
        if time.time() - t_start > time_limit:
            break
        try:
            res = minimize(obj_lse, h, args=(n, alpha), method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': max_iter, 'ftol': 1e-18})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < c5 - 1e-15:
                h = h_try
                c5 = c
        except Exception:
            pass
    return h, c5
