        if new_c5 < best_c5 - 1e-15:
            h = h_new
            best_h = h.copy()
            best_c5 = new_c5
            stall = 0
            if lr < 0.025 and step_count % 10 == 0:
                lr *= 1.04
        else:
            h = best_h.copy()
            v *= 0.5
            lr *= 0.93
            stall += 1
            if stall > 40:
                stall_count += 1
                size = 0.03 + 0.02 * min(stall_count, 5) * 0.5
                h_pert = best_h + rng.randn(n) * size
                h_pert = project_box_sum(h_pert, target_sum)
                pc5 = compute_c5(h_pert)
                if pc5 < best_c5:
                    best_h = h_pert.copy()
                    best_c5 = pc5
                h = best_h.copy()
                v = np.zeros(n)
                lr = 0.006
                alpha = min(alpha + 20, 350.0)
                beta = 0.5 + 0.3 * (stall_count % 3)
                stall = 0
                if stall_count > 12:
                    break

    return best_h, best_c5

def interp_to(h_old, n_old, n_new):
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old) + 1.0 / n_old
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new) + 1.0 / n_new
    return np.interp(x_new, x_old, h_old)

def make_binary_start(n, rng):
    pos = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[pos] = 1.0
    return h

def make_smoothed_binary(n, rng, iters=200):
    h = make_binary_start(n, rng)
    for _ in range(iters):
        i = rng.randint(n)
        j = (i + rng.randint(1, 5)) % n
        if h[i] + h[j] == 1.0:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        elif h[i] == h[j]:
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = project_box_sum(0.5 + 0.3 * rng.randn(200), 100.0)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    def finalize(h):
        h = project_box_sum(h, len(h) / 2.0)
        c5 = compute_c5(h)
        return h, c5

    # Phase 1: Primary optimization at initial resolution
    n1 = len(h_init)
    d1 = t0 + min(budget_s * 0.3, 250.0)
    h1, c5_1 = optimize(h_init.copy(), n1 / 2.0, d1, rng, alpha_init=60)
    h1, c5_1 = finalize(h1)
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1, c5_1, n1

    # Phase 2: Alternative start - smoothed binary
    if remaining() > 150:
        h_bin = make_smoothed_binary(n1, rng, iters=500)
        h_bin = project_box_sum(h_bin, n1 / 2.0)
        d2 = t0 + budget_s * 0.5
        h2, c5_2 = optimize(h_bin, n1 / 2.0, d2, rng, alpha_init=40)
        h2, c5_2 = finalize(h2)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2, c5_2, n1

    # Phase 3: Try lower resolution (easier to optimize)
    if remaining() > 120:
        n3 = 100
        h3_init = interp_to(best_h, best_n, n3)
        h3_init = project_box_sum(h3_init, n3 / 2.0)
        d3 = t0 + budget_s * 0.68
        h3, c5_3 = optimize(h3_init, n3 / 2.0, d3, rng, alpha_init=80)
        h3, c5_3 = finalize(h3)
        if c5_3 < best_c5:
            best_h, best_c5, best_n = h3, c5_3, n3

    # Phase 4: Try higher resolution
    if remaining() > 120:
        n4 = min(best_n + 50, 300)
        h4_init = interp_to(best_h, best_n, n4)
        h4_init = project_box_sum(h4_init, n4 / 2.0)
        d4 = t0 + budget_s * 0.85
        h4, c5_4 = optimize(h4_init, n4 / 2.0, d4, rng, alpha_init=100)
        h4, c5_4 = finalize(h4)
        if c5_4 < best_c5:
            best_h, best_c5, best_n = h4, c5_4, n4

    # Phase 5: Final polish at best resolution
    if remaining() > 20:
        d5 = t0 + budget_s - 5
        h5, c5_5 = optimize(best_h.copy(), best_n / 2.0, d5, rng, alpha_init=200)
        h5, c5_5 = finalize(h5)
        if c5_5 < best_c5:
            best_h, best_c5, best_n = h5, c5_5, best_n

    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
