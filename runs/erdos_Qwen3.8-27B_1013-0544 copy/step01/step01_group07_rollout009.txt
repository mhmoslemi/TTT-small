```python
import time
from scipy.optimize import minimize, lbfgsb

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_values(h):
    return np.correlate(h, 1 - h, mode="full")

def grad_corr_max(h, top_m=1):
    """Compute subgradient of max_k corr[k]."""
    corr = np.correlate(h, 1 - h, mode="full")
    n = len(h)
    # Get top m indices
    idx = np.argsort(corr)[-top_m:]
    weights = np.ones(len(idx)) / len(idx)
    
    # For each selected lag j = k - (n-1), compute gradient
    # corr[(n-1)+j] = sum_{i in V_j} h[i] * (1-h[i+j])
    # d/dh[m] = (1-h[m+j])*[m in V_j] - h[m-j]*[m-j in V_j]
    grad = np.zeros(n)
    for w, k in zip(weights, idx):
        j = k - (n - 1)
        # V_j = {i : max(0,-j) <= i <= min(n-1, n-1-j)}
        i_lo = max(0, -j)
        i_hi = min(n - 1, n - 1 - j)
        # First term: (1-h[m+j]) for m in V_j where m+j in [0,n-1]
        # m ranges from i_lo to i_hi, m+j ranges from i_lo+j to i_hi+j
        m_range = np.arange(i_lo, i_hi + 1)
        # m+j must be in [0, n-1]
        valid_j = (m_range + j >= 0) & (m_range + j <= n - 1)
        if np.any(valid_j):
            grad[m_range[valid_j]] += w * (1 - h[m_range[valid_j] + j])
        # Second term: -h[m-j] for m-j in V_j, i.e., m in V_j + j
        m2_lo = max(0, -j) + j  # = max(j, 0)
        m2_hi = min(n - 1, n - 1 - j) + j  # = min(n-1+j, n-1)
        m2_lo = max(0, m2_lo)
        m2_hi = min(n - 1, m2_hi)
        if m2_hi >= m2_lo:
            m2_range = np.arange(m2_lo, m2_hi + 1)
            grad[m2_range] -= w * h[m2_range - j]
    
    return grad

def project_constraints(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    # Use bisection on lambda
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Exact fix

        free = (h_proj > 0.01) & (h_proj < 0.99)

            h_proj[free] += diff / np.sum(free)
            h_proj = np.clip(h_proj, 0.0, 1.0)
        else:
            h_proj += diff / len(h_proj)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = lr_init
    
    for it in range(n_iter):
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute subgradient
        g = grad_corr_max(h, top_m=top_m)
        
        # Backtracking line search
        alpha = lr
        cancelled = False
        for _ in range(30):
            h_trial = project_constraints(h - alpha * g, target_sum)
            c5_trial = compute_c5(h_trial, dx)
            if c5_trial < c5 - 1e-10:
                cancelled = True
                break
            alpha *= 0.5
        
        if cancelled:
            h = h_trial
        else:
            # Try a small random perturbation to escape

                pert = np.random.randn(n) * 0.01
                h_trial = project_constraints(h + pert, target_sum)
                if compute_c5(h_trial, dx) < best_c5:
                    h = h_trial
            else:
                h = project_constraints(h - alpha * g, target_sum)
        
        # Adaptive learning rate
        if cancelled and alpha > lr * 0.1:
            lr = max(alpha * 1.1, lr * 0.5)
        elif not cancelled:
            lr = lr * 0.9
    
    return best_h, best_c5
```

```python
#   Clipped to [0, n-1]: m ranges from max(0,-j) to min(n-1, n-1-j)
m2_start = max(0, -j)
m2_end = min(n-1, n-1-j)
if m2_end >= m2_start:
    g[m2_start:m2_end+1] -= w_j * h[m2_start:m2_end+1]
```
