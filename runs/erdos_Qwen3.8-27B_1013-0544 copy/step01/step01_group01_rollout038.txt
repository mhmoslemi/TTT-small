    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    active = corr >= max_val - tol * max_val
    n_active = np.sum(active)
    
    grad = np.zeros(n)
    for m in np.where(active)[0]:
        k = int(m - (n - 1))
        if k >= 0:
            # f_k = dx * sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
            # grad[i] = dx * (1-h[i+k]) for i in 0..n-1-k
            # grad[i+k] -= dx * h[i] for i in 0..n-1-k
            for i in range(n - k):
                grad[i] += dx * (1 - h[i+k])
                grad[i+k] -= dx * h[i]
        else:
            j = -k
            for i in range(j, n):
                grad[i] += dx * (1 - h[i-j])
                grad[i-j] -= dx * h[i]
    
    grad /= n_active
    return grad, max_val * dx, active

This gives us intuition: we want h to be "spread out" so that R_0 isn't too large, while also making the off-diagonal R_k's not too small.

This is essentially a "flat spectrum" problem: we want the autocorrelation to be as flat as possible.

def compute_c5(h):
    n = len(h)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / beta
