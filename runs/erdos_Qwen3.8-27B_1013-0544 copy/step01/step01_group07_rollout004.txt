    fixed_sum = target_sum - (np.sum(h) - np.sum(h[idx]))
    sub_h = h[idx].copy()
    
    def sub_obj(x):
        h_sub = h.copy()
        h_sub[idx] = x
        return smooth_max_obj(h_sub, t, dx)
    
    try:
        res = minimize(
            sub_obj, sub_h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * group_size,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - fixed_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
        h_new = h.copy()
        h_new[idx] = np.clip(res.x, 0, 1)
        return h_new
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def elapsed():
        return time.time() - t0
    
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = fix_sum(h0, target_sum)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    best_n = n
    
    # Phase 1: SLSQP with increasing t
    for t in [5, 10, 20, 50, 100, 200, 400, 800]:
        if elapsed() > budget_s - 250:
            break
        h_new = slsqp_step(best_h, dx, target_sum, t, 600)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 2: Fourier-space SLSQP
    x = np.arange(n) * dx
    n_modes = 20
    coefs = real_to_fourier(best_h, x, n_modes)
    
    if elapsed() < budget_s - 200:
        try:
            res = minimize(
                lambda c: fourier_obj_wrapper(c, x, n_modes, dx, n),
                coefs.copy(),
                method='SLSQP',
                bounds=[(-0.5, 0.5)] * n_modes,
                options={'maxiter': 400, 'ftol': 1e-12}
            )
            h_new = np.clip(fourier_eval(res.x, x), 0, 1)
            h_new = fix_sum(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                coefs = real_to_fourier(best_h, x, n_modes)
        except Exception:
            pass
    
    # Phase 3: Simulated annealing in Fourier space
    if elapsed() < budget_s - 150:
        coefs_sa = real_to_fourier(best_h, x, n_modes)
        current_c5 = best_c5
        T = 100.0
        T_floor = 0.05
        cooling = 0.997
        n_iter = 2000
        
        for i in range(n_iter):
            if elapsed() > budget_s - 120:
                break
            T = max(T_floor, T * cooling)
            
            n_pert = np.random.randint(1, 7)
            idx_p = np.random.choice(n_modes, n_pert, replace=False)
            coefs_new = coefs_sa.copy()
            scale = 0.01 * (1.0 + 50.0 / T)
            coefs_new[idx_p] += np.random.normal(0, scale, n_pert)
            
            h_new = np.clip(fourier_eval(coefs_new, x), 0, 1)
            h_new = fix_sum(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            
            if c5_new < current_c5:
                coefs_sa = coefs_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            else:
                diff = (c5_new - current_c5) / T
                if diff < 50 and np.random.random() < np.exp(-diff):
                    coefs_sa = coefs_new
                    current_c5 = c5_new
        
        # Refine after SA
        h_ref = np.clip(fourier_eval(coefs_sa, x), 0, 1)
        h_ref = fix_sum(h_ref, target_sum)
        if compute_c5(h_ref, dx) < best_c5:
            best_h = h_ref.copy()
            best_c5 = compute_c5(best_h, dx)
        best_h = slsqp_step(best_h, dx, target_sum, 100, 500)
        best_c5 = compute_c5(best_h, dx)
    
    # Phase 4: Targeted worst-shift fixing with group descent
    if elapsed() < budget_s - 100:
        for _ in range(80):
            if elapsed() > budget_s - 80:
                break
            h_new = group_desc_step(best_h, dx, target_sum, 8, 30, 60)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # Phase 5: Multi-scale perturbation + refinement
    if elapsed() < budget_s - 60:
        for trial in range(30):
            if elapsed() > budget_s - 40:
                break
            # Different perturbation strategies
            strategy = trial % 3
            h_trial = best_h.copy()
            if strategy == 0:
                # Sparse perturbation
                n_p = np.random.randint(3, 10)
                idx = np.random.choice(n, n_p, replace=False)
                h_trial[idx] += np.random.uniform(-0.3, 0.3, n_p)
            elif strategy == 1:
                # Fourier perturbation
                pp = np.arange(n) * dx
                mode = np.random.randint(2, 15)
                phase = np.random.uniform(0, 2 * np.pi)
                amp = np.random.uniform(0.05, 0.15)
                h_trial += amp * np.cos(mode * np.pi * pp + phase)
            else:
                # Block perturbation
                block = np.random.randint(10, 40)
                start = np.random.randint(0, n - block)
                h_trial[start:start+block] += np.random.uniform(-0.1, 0.1)
            
            h_trial = fix_sum(h_trial, target_sum)
            h_new = slsqp_step(h_trial, dx, target_sum, 50, 300)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # Phase 6: Try n=80 resolution
    if elapsed() < budget_s - 50:
        n2 = 80
        dx2 = 2.0 / n2
        ts2 = n2 / 2.0
        x2 = np.arange(n2) * dx2
        h2 = np.interp(x2, np.arange(n) * dx, best_h)
        h2 = fix_sum(h2, ts2)
        
        for t in [20, 50, 100, 300]:
            if elapsed() > budget_s - 30:
                break
            h2 = slsqp_step(h2, dx2, ts2, t, 500)
        c5_2 = compute_c5(h2, dx2)
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2
            best_n = n2
    
    # Phase 7: Final high-precision polish
    if elapsed() < budget_s - 10:
        dx_final = 2.0 / len(best_h)
        ts_final = len(best_h) / 2.0
        for t in [200, 500, 1000]:
            if elapsed() > budget_s - 5:
                break
            best_h = slsqp_step(best_h, dx_final, ts_final, t, 1500)
    
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    dx_final = 2.0 / len(best_h)
    best_c5 = compute_c5(best_h, dx_final)
    return best_h, best_c5, len(best_h)
