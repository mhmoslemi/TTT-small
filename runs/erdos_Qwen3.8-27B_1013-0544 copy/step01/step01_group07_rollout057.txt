    n = len(h)
    # G[k,j] = I(0 <= j-k+n-1 < n) * (1-h[j-k+n-1]) - I(0 <= j+k-n+1 < n) * h[j+k-n+1]

    corr = np.correlate(h, 1 - h, mode="full")

def compute_grad_matrix(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min1 = max(0, k - n + 1)
        j_max1 = min(n, k + 1)
        for j in range(j_min1, j_max1):
            G[k, j] += (1 - h[j - k + n - 1])
        j_min2 = max(0, n - k - 1)
        j_max2 = min(n, n - k + 1)
        for j in range(j_min2, j_max2):
            G[k, j] -= h[j + k - n + 1]
    return G

def project(h, target_sum):
    lo, hi = -1.5, 1.5

        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:

            hi = mid

```python
def compute_grad_matrix(h):
    n = len(h)
    # G[k,j] = I(0 <= j-k+n-1 < n) * (1-h[j-k+n-1]) - I(0 <= j+k-n+1 < n) * h[j+k-n+1]
    j = np.arange(n)
    k = np.arange(2*n-1)
    # First term index: j - k + n - 1
    idx1 = j[None, :] - k[:, None] + n - 1  # shape (2n-1, n)
    valid1 = (idx1 >= 0) & (idx1 < n)
    # Second term index: j + k - n + 1
    idx2 = j[None, :] + k[:, None] - n + 1
    valid2 = (idx2 >= 0) & (idx2 < n)
    # Gather values (use 0 where invalid)
    vals1 = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)
    vals2 = np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
    return vals1 - vals2
```

Actually, let me try a different approach altogether: **Frank-Wolfe (Conditional Gradient) method**.

The Frank-Wolfe method for minimizing f(h) over a convex set C:
1. Compute gradient g at current point h
2. Find s = argmin_{x in C} g^T x (the "linear oracle")
3. Update: h ← h + γ * (s - h) for some step size γ

For our problem, C = {h : 0 ≤ h ≤ 1, sum(h) = n/2}. The linear oracle (minimizing g^T h over C) can be solved in O(n log n): sort the components of g, and the solution is to set h_i = 1 for the smallest |target_sum| components (with one fractional value).

This is a well-known approach and often works well for problems with simple constraints.

Let me implement this:

```python
def fw_step(h, g, target_sum, gamma):
    """Frank-Wolfe update step."""
    n = len(h)
    # Linear oracle: minimize g^T x subject to 0 <= x <= 1, sum(x) = target_sum
    # Solution: sort g, set x_i = 1 for smallest g_i, 0 for largest, 
    # with one fractional value
    order = np.argsort(g)
    s = np.zeros(n)
    s[order[:int(target_sum)]] = 1.0
    frac = target_sum - int(target_sum)
    if frac > 0:
        s[order[int(target_sum)]] = frac
    # Update
    h_new = (1 - gamma) * h + gamma * s
    return h_new
```

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_grad_matrix(h):
    n = len(h)
    j = np.arange(n)
    k = np.arange(2 * n - 1)
    idx1 = j[None, :] - k[:, None] + n - 1
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j[None, :] + k[:, None] - n + 1
    valid2 = (idx2 >= 0) & (idx2 < n)
    vals1 = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n - 1)], 0.0)
    vals2 = np.where(valid2, h[np.clip(idx2, 0, n - 1)], 0.0)
    return vals1 - vals2

def project(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_direction(h, topk):
    corr = np.correlate(h, 1 - h, mode='full')
    sorted_vals = np.sort(corr)[::-1]
    threshold = sorted_vals[min(topk, len(corr)) - 1]
    active = np.where(corr >= threshold)[0]
    G = compute_grad_matrix(h)
    g = G[active].sum(axis=0) / len(active)
    return g

def fw_line_search(h_curr, s, dx, best_c5, n_gammas=9):
    best_h = h_curr.copy()
    best_val = best_c5
    gammas = np.linspace(0.05, 0.95, n_gammas)
    for gamma in gammas:
        h_try = (1.0 - gamma) * h_curr + gamma * s
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_val:
            best_val = c5_try
            best_h = h_try.copy()
    return best_h, best_val

def slsqp_polish(h, dx, target_sum, t=100, maxiter=2000):
    n = len(h)
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        max_c = np.max(corr)
        shifted = t * (corr - max_c)
        return (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project(res.x, target_sum)
    except Exception:
        return h

def binary_cd(h, target_sum, dx, n_rounds=5):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    for _ in range(n_rounds):
        improved = False

            for val in [0.0, 1.0]:
                if abs(h[j] - val) < 1e-6:
                    continue
                h_try = h.copy()
                h_try[j] = val
                h_try = project(h_try, target_sum)
                c5_try = compute_c5(h_try, dx)
                if c5_try < best_c5 - 1e-9:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def fw_loop(h_start, target_sum, dx, n, max_iters, topk_start, topk_end, n_gammas=9):

            best_h = h_try.copy()
        h_fw = h_try
    return best_h, best_c5
