```python
def get_active_lags(corr, n_active=10, epsilon=0.01):

    # Lags within epsilon of max, plus top n_active
```

```python
def active_set_descent(h, dx, target_sum, n_active=10, max_iter=10000, 
                       momentum=0.9, step_init=0.01, time_limit=None):
    n = len(h)
    t0 = time.time()
    v = np.zeros(n)  # momentum
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr, dx)
    
    for it in range(max_iter):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        
        corr = np.correlate(h_curr, 1 - h_curr, mode="full")
        c5_val = np.max(corr) * dx
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_curr.copy()
        
        # Get active lags
        active = get_active_lags(corr, n_active=n_active)
        
        # Compute gradient
        grad = compute_grad_active(h_curr, active, n)
        
        # Momentum update
        v = momentum * v - step_init * grad
        
        # Projected step
        h_trial = h_curr + v
        
        # Project to feasible set
        h_trial = project_feasible(h_trial, target_sum)
        
        # Check if improvement
        c5_trial = compute_c5(h_trial, dx)
        if c5_trial < c5_val:
            h_curr = h_trial
            # Adaptive step size: increase if good
            step_init = min(step_init * 1.05, 0.1)
        else:
            # Backtrack
            step_init *= 0.7
            if step_init < 1e-8:
                step_init = 0.01
                v = np.zeros(n)
    
    return best_h, best_c5
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_single_lag(h, k, n):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        m = -k
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]
    return g

def grad_active(h, active_lags, n):
    g = np.zeros(n)
    for li in active_lags:
        k = li - (n - 1)
        g += grad_single_lag(h, k, n)
    return g

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - np.sum(hp)
    if abs(diff) > 1e-12:
        mask = (hp > 1e-6) & (hp < 1.0 - 1e-6)
        cnt = np.sum(mask)
        if cnt > 0:
            hp[mask] += diff / cnt
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def smooth_max(vals, t):
    m = np.max(vals)
    return m + (1.0 / t) * np.log(np.sum(np.exp(t * (vals - m))))

def active_gradient_descent(h, dx, target_sum, time_limit, n_active=20,
                            step_init=0.04, momentum=0.82):
    n = len(h)
    t_start = time.time()
    v = np.zeros(n)
    h_c = h.copy()
    best_h = h_c.copy()
    best_c5 = compute_c5(h_c, dx)
    step = step_init
    stall = 0

    while time.time() - t_start < time_limit:
        corr = np.correlate(h_c, 1.0 - h_c, mode="full")
        c5_raw = np.max(corr)
        c5_val = c5_raw * dx
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h_c.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                break
        frac = 1.0 - 0.02 * (1.0 + 0.005 * stall)
        active = np.where(corr >= c5_raw * frac)[0]
        if len(active) < 10:
            active = np.argsort(corr)[-n_active:]
        elif len(active) > 40:
            active = np.argsort(corr)[-n_active:]
        g = grad_active(h_c, active, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        v = momentum * v - step * g / gn
        h_t = project_feasible(h_c + v, target_sum)
        c5_t = compute_c5(h_t, dx)
        if c5_t < c5_val - 1e-13:
            h_c = h_t
            step = min(step * 1.06, 0.4)
        else:
            step *= 0.65
            if step < 1e-7:
                step = step_init
                v *= 0.4

    return best_h, best_c5

def lbfgs_refine(h, dx, target_sum, t_smooth=50, frac=0.98, maxiter=400):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_raw = np.max(corr)
    active = np.where(corr >= c5_raw * frac)[0]
    if len(active) < 5:
        active = np.argsort(corr)[-10:]
    pen = 50.0

    def obj(x):
        hx = np.clip(x, 0.0, 1.0)
        cx = np.correlate(hx, 1.0 - hx, mode="full")
        return smooth_max(cx[active], t_smooth) * dx + pen * (np.sum(hx) - target_sum) ** 2

    def jac(x):
        hx = np.clip(x, 0.0, 1.0)
        cx = np.correlate(hx, 1.0 - hx, mode="full")
        m = np.max(cx[active])
        w = np.exp(t_smooth * (cx[active] - m))
        w /= np.sum(w)
        g = np.zeros(n)
        for j, ai in enumerate(active):
            g += w[j] * grad_single_lag(hx, ai - (n - 1), n)
        g *= dx
        g += 2.0 * pen * (np.sum(hx) - target_sum)
        return g

    try:
        res = minimize(obj, h.copy(), jac=jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'gtol': 1e-11})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h

def coordinate_polish(h, dx, target_sum, n_rounds=20):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best, dx)
    deltas = [0.001, 0.003, 0.008, 0.02, -0.001, -0.003, -0.008, -0.02]
    for _ in range(n_rounds):
        improved = False
        for i in range(n):
            for d in deltas:
                h_t = best.copy()
                h_t[i] = np.clip(h_t[i] + d, 0.0, 1.0)
                h_t = project_feasible(h_t, target_sum)
                c5_t = compute_c5(h_t, dx)
                if c5_t < best_c5 - 1e-13:
                    best = h_t
                    best_c5 = c5_t
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5
```
