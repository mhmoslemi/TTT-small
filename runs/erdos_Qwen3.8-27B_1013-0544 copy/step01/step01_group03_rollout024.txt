    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

    x = tau * values
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

    x = tau * corr
    x_max = np.max(x)
    return np.exp(x - x_max) / np.sum(np.exp(x - x_max))

def gradient_corr_smooth_max(h, tau):

    corr = np.correlate(h, 1.0 - h, mode="full")
    weights = compute_weights(corr, tau)
    n_lags = 2 * n - 1
    grad = np.zeros(n)

        w = weights[k]
        if w < 1e-10:

        m = k - (n - 1)
        if m == 0:
            grad += w * (1.0 - 2.0 * h)
        elif m > 0:
            grad[m:] += w * (1.0 - h[:n-m])
            grad[:n-m] -= w * h[m:]
        else:
            grad[:n+m] += w * (1.0 - h[-m:])
            grad[-m:] -= w * h[:n+m]
    
    return grad

def projected_gradient_descent(h_init, n, tau, steps=5000, lr=None):
    """PGD with projection onto {0<=h<=1, sum(h)=n/2}."""
    h = h_init.copy()
    target = n / 2.0
    h = fix_sum(h, target)
    
    if lr is None:
        lr = 1.0 / (tau * n)  # Lipschitz constant estimate
    
    best_h = h.copy()
    best_val = evaluate_c5(h)
    
    for step in range(steps):
        grad = gradient_corr_smooth_max(h, tau)
        # Project gradient onto the sum constraint (remove mean)
        grad = grad - np.mean(grad)
        h_new = h - lr * grad
        h_new = fix_sum(h_new, target)
        
        val = evaluate_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_val

def binary_local_search(h_bin, n, max_iter=10000):
    """Local search on binary solutions: flip pairs of bits."""
    h = h_bin.copy()
    target = n / 2
    assert abs(h.sum() - target) < 1e-10
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    best_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):

        improved = False
        # Sample random swaps
        for _ in range(200):

            j = np.random.randint(len(zeros))
            a, b = ones[i], zeros[j]

            h[b] = 1.0

                improved = True

                h[b] = 0.0

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(values, tau):
    x = tau * values
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def compute_weights(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    w = np.exp(x - x_max)
    s = w.sum()
    if s < 1e-300:
        return np.full(len(corr), 1.0 / len(corr))
    return w / s

def gradient_smooth_max(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    weights = compute_weights(corr, tau)
    n_lags = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_lags):
        w = weights[k]
        if w < 1e-8:
            continue
        m = k - (n - 1)
        if m == 0:
            grad += w * (1.0 - 2.0 * h)
        elif m > 0:
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
        else:
            mp = -m
            grad[:n - mp] += w * (1.0 - h[mp:])
            grad[mp:] -= w * h[:n - mp]
    return grad

def pgd_nesterov(h_init, n, tau, steps=5000, lr_base=0.5):
    h = h_init.copy()
    target = n / 2.0
    h = fix_sum(h, target)
    v = np.zeros(n)
    mu = 0.9
    lr = lr_base / (tau ** 0.5)
    best_h = h.copy()
    best_val = evaluate_c5(h)
    patience = 300
    no_improve = 0
    
    for step in range(steps):
        h_y = h + mu * v
        h_y = fix_sum(h_y, target)
        grad = gradient_smooth_max(h_y, tau)
        grad = grad - np.mean(grad)
        v = mu * v - lr * grad
        h_new = h + v
        h_new = fix_sum(h_new, target)
        h = h_new
        val = evaluate_c5(h)
        if val < best_val - 1e-10:
            best_val = val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > patience:
            break
    return best_h, best_val

def binary_local_search(h_bin, n, max_rounds=2000, n_trials=100):
    h = h_bin.copy()
    best_c5 = evaluate_c5(h)
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    rng = np.random.RandomState(42)
    
    for r in range(max_rounds):
        improved = False
        for _ in range(n_trials):
            i = int(rng.randint(len(ones)))
            j = int(rng.randint(len(zeros)))
            a, b = ones[i], zeros[j]
            h[a] = 0.0
            h[b] = 1.0
            c5 = evaluate_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                improved = True
                ones[i] = b
                zeros[j] = a
            else:
                h[a] = 1.0
                h[b] = 0.0
        if not improved:
            break
    return h, best_c5

def to_binary(h, n):
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def slsqp_refine(h, n, tau, maxiter=400):
    target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    
    def obj_func(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max_val(corr, tau)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_func, h, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = fix_sum(res.x, target)
        c5_new = evaluate_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h, evaluate_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def rem():
        return budget_s - (time.time() - t0)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(120, 0.5)
    
    n0 = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: PGD from parent with adaptive tau
    for tau in [30, 100, 500, 2000, 8000]:
        if rem() < 40:
            break
        h_new, c5_new = pgd_nesterov(best_h, n0, tau, steps=4000, lr_base=1.0)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 2: SLSQP polish
    if rem() > 50:
        h_new, c5_new = slsqp_refine(best_h, n0, tau=3000, maxiter=500)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 3: Binary local search from rounded solution
    if rem() > 80:
        h_bin = to_binary(best_h, n0)
        h_b, c5_b = binary_local_search(h_bin, n0, max_rounds=3000, n_trials=200)
        if c5_b < best_c5:
            best_h = h_b
            best_c5 = c5_b
        # Try binary at different n
        for tn in [80, 100, 140, 200]:
            if rem() < 30:
                break
            if tn % 2 != 0:
                continue
            h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, n0), best_h)
            h_bin_t = to_binary(h_t, tn)
            h_bt, c5_bt = binary_local_search(h_bin_t, tn, max_rounds=2000, n_trials=150)
            if c5_bt < best_c5:
                best_h = h_bt
                best_c5 = c5_bt
                n0 = tn
    
    # Phase 4: Multi-resolution continuous optimization
    for tn in [40, 60, 80, 100, n0, n0 + 40]:
        if rem() < 50:
            break
        if tn % 2 != 0 or tn < 20:
            continue
        x_old = np.linspace(0, 1, len(best_h))
        x_new = np.linspace(0, 1, tn)
        h_t = np.interp(x_new, x_old, best_h)
        h_t = fix_sum(h_t, tn / 2.0)
        for tau in [50, 300, 1500]:

            h_new, c5_new = pgd_nesterov(h_t, tn, tau, steps=3000, lr_base=1.0)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                n0 = tn
                x_old = np.linspace(0, 1, len(best_h))
            h_t = h_new
            if rem() < 15:
                break
    
    # Phase 5: Diverse restarts
    n_cur = len(best_h)
    for _ in range(30):
        if rem() < 40:
            break
        # Generate diverse initial points
        strategy = _ % 3
        if strategy == 0:
            h_r = np.random.uniform(0, 1, n_cur)
        elif strategy == 1:

        else:
            # Reversed + perturbed
            h_r = best_h[::-1].copy()
            h_r += np.random.normal(0, 0.1, n_cur)
        h_r = fix_sum(h_r, n_cur / 2.0)
        h_new, c5_new = pgd_nesterov(h_r, n_cur, 500, steps=2000, lr_base=1.0)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 6: Final polish

        h_new, c5_new = slsqp_refine(best_h, n_f, tau=10000, maxiter=300)

    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
