import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def smooth_obj_and_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    obj = m + T * np.log(np.sum(np.exp((corr - m) / T)))
    
    exp_corr = np.exp((corr - m) / T)
    w = exp_corr / np.sum(exp_corr)
    
    grad = np.zeros(n)
    lags = np.arange(2 * n - 1) - (n - 1)
    
    for k in range(2 * n - 1):
        lag = lags[k]
        wk = w[k] * dx
        # First term: (1-h[i-lag])
        i_start = max(0, lag)
        i_end = min(n, n + lag)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += wk * (1.0 - h[idx - lag])
        # Second term: -h[i+lag]
        i_start2 = max(0, -lag)
        i_end2 = min(n, n - lag)
        if i_start2 < i_end2:
            idx2 = np.arange(i_start2, i_end2)
            grad[idx2] -= wk * h[idx2 + lag]
    
    # Penalty gradient
    cv = np.sum(h) - n * 0.5
    obj += penalty * cv * cv
    grad += 2.0 * penalty * cv * np.ones(n)
    
    return obj, grad

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def opt_lbgfsb(h0, T, penalty, max_iter=500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_obj_and_grad(x, T, penalty)[0]
    
    def grad(x):
        return smooth_obj_and_grad(x, T, penalty)[1]
    
    res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-14})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new), res.nit

def opt_tvar_slsqp(h0, max_iter=300):
    n = len(h0)
    dx = 2.0 / n
    
    def t_obj(v):
        return v[-1]
    
    def t_ineq(v):
        h = v[:n]
        t = v[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corr
    
    def t_eq(v):
        return np.sum(v[:n]) - n * 0.5
    
    def t_eq_jac(v):
        j = np.zeros(len(v))
        j[:n] = 1.0
        return j
    
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}
    ]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-10})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

def interp_h(h_old, n_new):
    n_old = len(h_old)

    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)

    n = len(h0)
    dx = 2.0 / n
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': tvar_ineq, 'args': (n, dx)},
        {'type': 'eq', 'fun': tvar_eq, 'jac': tvar_eq_jac, 'args': (n,)}
    ]
    res = minimize(tvar_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-10})
    h_new = project_h(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)


def interp_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)


def subgradient_step(h, lr, n_active=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    active = np.where(corr >= m - 1e-10)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    elif len(active) > n_active:
        idx = np.argsort(corr[active])[::-1][:n_active]
        active = active[idx]
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in active:
        lag = lags[k]
        i_min = max(0, lag)
        i_max = min(n - 1, n - 1 + lag)
        if i_min <= i_max:
            idx = np.arange(i_min, i_max + 1)
            grad[idx] += dx * (1.0 - h[idx - lag])
        i_min2 = max(0, -lag)
        i_max2 = min(n - 1, n - 1 - lag)
        if i_min2 <= i_max2:
            idx2 = np.arange(i_min2, i_max2 + 1)
            grad[idx2] -= dx * h[idx2 + lag]
    grad /= len(active)
    h_new = h - lr * grad
    h_new = project_h(h_new, n * 0.5)
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_h(h, n * 0.5)
    c5 = compute_c5(h)
    h_best = h.copy()
    c5_best = c5

    penalty = 1e5

    # Phase 1: L-BFGS-B at current resolution, decreasing temperature
    temps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for T in temps:
        if tl() < 60:
            break
        h_new, c5_new = opt_lbgfsb(h, T, penalty, max_iter=500)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            h = h_new

    # Phase 2: Subgradient refinement at low temperature scale
    if tl() > 120:
        h_sg = h_best.copy()
        for i in range(200):
            if tl() < 80:
                break
            lr = 0.01 / (1.0 + 0.01 * i)
            h_sg = subgradient_step(h_sg, lr)
            c5_sg = compute_c5(h_sg)
            if c5_sg < c5_best:
                h_best = h_sg.copy()
                c5_best = c5_sg
        # Polish the subgradient result with L-BFGS-B
        if tl() > 60:
            h_new, c5_new = opt_lbgfsb(h_sg, 0.001, penalty, max_iter=300)
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new

    # Phase 3: T-variable SLSQP polish
    if tl() > 100:
        h_new, c5_new = opt_tvar(h_best, max_iter=500)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 4: Higher resolution
    for n_new in [360, 540]:
        if tl() < 200:
            break
        h_cur = interp_h(h_best, n_new)
        c5_cur = compute_c5(h_cur)
        temps2 = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        for T in temps2:
            if tl() < 80:
                break
            h_new, c5_new = opt_lbgfsb(h_cur, T, penalty, max_iter=500)
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new
                h_cur = h_new
        if tl() > 80:
            h_new, c5_new = opt_tvar(h_cur, max_iter=500)
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new

    # Phase 5: Random restarts with diverse perturbations
    n_best = len(h_best)
    n_restarts = 25
    for i in range(n_restarts):
        if tl() < 50:
            break
        scale = 0.015 + 0.008 * (i % 8)
        h_p = h_best + rng.randn(n_best) * scale
        h_p = project_h(h_p, n_best * 0.5)
        # Quick L-BFGS-B at moderate temperature
        h_new, c5_new = opt_lbgfsb(h_p, 0.005, penalty, max_iter=300)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            # Refine with lower temperature
            h_new2, c5_new2 = opt_lbgfsb(h_best, 0.001, penalty, max_iter=300)
            if c5_new2 < c5_best:
                h_best = h_new2.copy()
                c5_best = c5_new2

    # Phase 6: Final SLSQP polish
    if tl() > 50:
        h_new, c5_new = opt_tvar(h_best, max_iter=500)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Final cleanup
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
