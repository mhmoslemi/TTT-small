def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference to variables not at bounds
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_objective(h, temp, penalty_weight):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    # Log-sum-exp
    sms = m + temp * np.log(np.sum(np.exp((corr - m) / temp)))
    # Penalty for sum constraint
    penalty = penalty_weight * (np.sum(h) - n * 0.5) ** 2
    return sms + penalty

def project_h(h, target_sum):
    h = np.asarray(h, dtype=float)
    n = len(h)
    # Find lambda such that sum(clip(h + lambda, 0, 1)) = target_sum
    lo = -1.0  # h[i] + lo <= 0 for all i (if h[i] >= 0)
    hi = 1.0   # h[i] + hi >= 1 for all i (if h[i] <= 1)
    
    def sum_at_lambda(lam):
        return np.sum(np.clip(h + lam, 0.0, 1.0))
    
    # Bisection
    for _ in range(60):
        mid = (lo + hi) / 2
        s = sum_at_lambda(mid)
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2
    return np.clip(h + lam, 0.0, 1.0)

        h_new = opt_lbfgs(h, temp, 2000.0, 500)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            h_best = h.copy()

    # Phase 4: Basin hopping with random perturbations
    for i in range(30):
        if tl() < 70:
            break
        scale = 0.003 + 0.004 * (i % 12)
        h_p = h_best + rng.randn(n) * scale
        h_p = project_h(h_p, n * 0.5)
        h_new = opt_lbfgs(h_p, 0.003, 500.0, 200)
        c5_new = compute_c5(h_new)
        if c5_new < c5 or (c5_new < c5 + 0.0005 and i < 20):
            h_pol = opt_slsqp(h_new, max_iter=400)
            c5_pol = compute_c5(h_pol)
            if c5_pol < c5:
                h = h_pol
                c5 = c5_pol
                h_best = h.copy()

    # Phase 5: Block perturbation restarts
    for i in range(15):
        if tl() < 50:
            break
        h_p = h_best.copy()
        bl = rng.randint(8, min(35, n // 4))
        st = rng.randint(0, n - bl)
        h_p[st:st + bl] = rng.uniform(0, 1, bl)
        h_p = project_h(h_p, n * 0.5)
        h_new = opt_lbfgs(h_p, 0.002, 500.0, 200)
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 0.00005:
            h_pol = opt_slsqp(h_new, max_iter=400)
            c5_pol = compute_c5(h_pol)
            if c5_pol < c5:
                h = h_pol
                c5 = c5_pol
                h_best = h.copy()

    # Phase 6: Try circular shifts of best solution
    for shift in [1, 2, 3, -1, -2]:
        if tl() < 40:
            break
        h_shift = np.roll(h_best, shift)
        h_shift = project_h(h_shift, n * 0.5)
        h_new = opt_lbfgs(h_shift, 0.002, 500.0, 200)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h_pol = opt_slsqp(h_new, max_iter=400)
            c5_pol = compute_c5(h_pol)
            if c5_pol < c5:
                h = h_pol
                c5 = c5_pol
                h_best = h.copy()

    # Phase 7: Final SLSQP polish
    if tl() > 25:
        h_new = opt_slsqp(h_best, max_iter=900)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h_best = h_new
            c5 = c5_new

    # Ensure valid output
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, n * 0.5)
    c5 = compute_c5(h_best)
    return h_best, c5, n
