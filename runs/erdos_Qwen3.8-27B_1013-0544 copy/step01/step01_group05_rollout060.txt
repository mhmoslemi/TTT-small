import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    lo = -2.0
    hi = 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    logsum = cmax + np.log(np.exp(corr - cmax).sum())
    w = np.exp(corr - logsum)
    grad = np.zeros(n)
    nh = n - 1
    for i in range(len(corr)):
        wi = w[i]
        if wi < 1e-7:
            continue
        L = i - nh
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[:n - L] += wi * (1.0 - h[L:])
            grad[L:] -= wi * h[:n - L]
        else:
            L2 = -L
            grad[:n - L2] -= wi * h[L2:]
            grad[L2:] += wi * (1.0 - h[:n - L2])
    return grad * (2.0 / n)

def active_set_grad(h, nmax=12, tol_frac=0.001, tol_abs=0.002):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    thr = cmax - max(tol_abs, cmax * tol_frac)
    act = np.where(corr >= thr)[0]
    if len(act) < 3:
        act = np.argsort(corr)[-nmax:]
    elif len(act) > 25:
        act = np.argsort(corr)[-nmax:]
    grad = np.zeros(n)
    nh = n - 1
    for i in act:
        L = i - nh
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            grad[:n - L] += (1.0 - h[L:])
            grad[L:] -= h[:n - L]
        else:
            L2 = -L
            grad[:n - L2] -= h[L2:]
            grad[L2:] += (1.0 - h[:n - L2])
    return grad / len(act)

def slsqp_refine(h, dx, tsum, maxiter=2000):
    n = len(h)
    def obj(x):
        return compute_c5(x, dx)
    def cfun(x):
        return np.sum(x) - tsum
    def cjac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and res.x is not h:
            hf = project_feasible(res.x, tsum)
            return hf, compute_c5(hf, dx)
    except Exception:
        pass
    return h.copy(), compute_c5(h, dx)
