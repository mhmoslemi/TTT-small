def compute_c5_binary(h):
    """Compute C5 for a binary (or continuous) h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_lags(h):
    """Compute all lag overlaps."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx  # shape (2n-1,)

    """Convert h to a binary vector with exactly n/2 ones,
    using the fractional parts as a guide."""
    n = len(h)
    target = n // 2
    # Sort by value, take top target as 1s
    idx = np.argsort(-h)[:target]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def local_search(h_bin, n, dx, n_iterations=100000, temperature=0.01, cool_rate=0.999):
    """Simulated annealing local search on binary vectors."""
    target = n // 2
    current = h_bin.copy()
    current_lags = compute_all_lags(current)
    current_c5 = np.max(current_lags)
    
    best = current.copy()
    best_c5 = current_c5
    
    temp = temperature
    on_indices = np.where(current == 1)[0]
    off_indices = np.where(current == 0)[0]
    
    for it in range(n_iterations):
        # Random swap
        i = off_indices[np.random.randint(len(off_indices))]
        j = on_indices[np.random.randint(len(on_indices))]
        
        # Try the swap
        current[i] = 1
        current[j] = 0
        new_lags = compute_all_lags(current)
        new_c5 = np.max(new_lags)
        
        delta = new_c5 - current_c5
        if delta < 0 or np.random.random() < np.exp(-delta / temp):
            # Accept
            current_c5 = new_c5
            current_lags = new_lags
            # Update index sets
            off_indices = np.where(current == 0)[0]
            on_indices = np.where(current == 1)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = current.copy()
        else:
            # Reject
            current[i] = 0
            current[j] = 1
        
        temp *= cool_rate
    
    return best, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def smooth_max_and_grad(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    max_shifted = np.max(shifted)
    weights = np.exp(shifted - max_shifted)
    weights /= np.sum(weights)
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    val = m + log_sum / beta
    return val, weights


def project_feasible(z, S):
    n = len(z)
    lo = z.min() - S - 1.0
    hi = z.max() + S + 1.0

def gradient_of_smooth_max(h, weights, beta, dx, n):

    return grad

    f_current = compute_c5(h)
    step_scale = 0.05
    c_armijo = 1e-4
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        progress = it / max(float(max_iter), 1)
        beta = beta_start + (beta_end - beta_start) * progress
        lags = compute_all_lags(h)
        f_smooth, weights = smooth_max_and_grad(lags, beta)
        grad = gradient_of_smooth_max(h, weights, beta, dx, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        lr = step_scale
        for _ in range(40):
            h_trial = project_feasible(h - lr * grad, S)
            f_trial = compute_c5(h_trial)
            if f_trial <= f_current - c_armijo * lr * grad_norm ** 2:
                break
            lr *= 0.5
        else:
            h_trial = h
            f_trial = f_current
        if f_trial < f_current - 1e-15:
            h = h_trial
            f_current = f_trial
            step_scale = min(step_scale * 1.02, 0.1)

            step_scale *= 0.9
    return h, f_current


def slsqp_refine(h0, S, n, dx, max_iter, beta):
    bounds = Bounds(np.zeros(n), np.ones(n))

    def con_fun(x):
        return np.sum(x) - S

    def con_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

    def obj_func(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max_and_grad(corr * dx, beta)[0]

    try:
        res = minimize(obj_func, h0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, S)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return h0, compute_c5(h0)
