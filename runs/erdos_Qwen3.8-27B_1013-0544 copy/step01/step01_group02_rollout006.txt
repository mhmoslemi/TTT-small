```python
def subgradient_direction(h, n, tol=1e-6):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    m_max = c.max()
    active = np.where(c >= m_max - tol)[0]
    grad = np.zeros(n)
    for m in active:
        tau = m - (n-1)
        for j in range(n):
            g = 0.0
            if 0 <= j - tau <= n-1:
                g += (1 - h[j-tau])
            if 0 <= j + tau <= n-1:
                g -= h[j+tau]
            grad[j] += g * dx
    grad /= len(active)
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(c.max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(len(h))

def subgrad_dir(h, n, tol=1e-5):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m_max = c.max()
    active = np.where(c >= m_max - tol)[0]
    grad = np.zeros(n)
    for m in active:
        tau = int(m) - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j - tau <= n - 1:
                g += (1.0 - h[j - tau])
            if 0 <= j + tau <= n - 1:
                g -= h[j + tau]
            grad[j] += g * dx
    grad /= max(len(active), 1)
    return grad

def subgrad_polish(h, n, max_iter=5000, base_lr=0.002):
    h = h.copy()
    best = compute_c5(h, n)
    for it in range(max_iter):
        lr = base_lr / (1.0 + 0.01 * it)
        g = subgrad_dir(h, n)
        h_new = project_feasible(h - lr * g, n)
        c = compute_c5(h_new, n)
        if c < best - 1e-12:
            best = c
            h = h_new
        else:
            base_lr *= 0.95
    return h, best

def smooth_random_start(n, seed):
    rng = np.random.RandomState(seed)
    k = min(10, n // 3)
    coeffs = rng.randn(k + 1)
    phases = rng.uniform(0, 2 * np.pi, k + 1)
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    for i in range(k + 1):
        h += coeffs[i] * np.cos(2 * np.pi * (i + 1) * x / 2 + phases[i])
    h = 0.5 + 0.3 * np.tanh(h)
    return project_feasible(h, n)

    # Phase 1: Aggressive continuation SLSQP

            try:

    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
```
