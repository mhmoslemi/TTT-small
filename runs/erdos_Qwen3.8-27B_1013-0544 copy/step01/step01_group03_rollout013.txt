    target = n / 2.0

    idx = np.argsort(h0)[::-1]

    zeros = np.where(h == 0.0)[0]

def project_feasible_exact(h, target):
    """Exact projection onto {0<=x<=1, sum(x)=target}."""
    h = h.astype(float)
    # Bisection on lambda
    lo, hi = -2.0, 2.0  # lambda range
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Tiny fixup
    diff = target - h_proj.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h_proj < 1.0
        else:
            mask = h_proj > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def project_feasible_exact(h, target):
    h = h.astype(float)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h_proj.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h_proj < 1.0
        else:
            mask = h_proj > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def project_feasible_simple(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h = h.copy()
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    j_max = int(np.argmax(corr))
    k = j_max - (n - 1)
    max_val = corr[j_max] * dx
    grad = np.zeros(n)
    if k >= 0:
        if k == 0:
            grad[:] = 1.0 - h
            grad[:] -= h
        else:
            grad[k:] = 1.0 - h[:n - k]
            grad[:n - k] -= h[k:]
    else:
        kp = -k
        grad[:n - kp] = 1.0 - h[kp:]
        grad[kp:] -= h[:n - kp]
    return max_val, grad

def subgradient_descent(h0, n, max_iters, alpha0, momentum=0.5):
    target = n / 2.0
    h = h0.copy().astype(float)
    v = np.zeros(n)
    h_avg = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    for t in range(1, max_iters + 1):
        alpha = alpha0 / (1.0 + 0.05 * t)
        _, grad = compute_subgradient(h)
        v = momentum * v - alpha * grad
        h = project_feasible_exact(h + v, target)
        h_avg += (h - h_avg) / t
        if t % 100 == 0:
            c5_a = evaluate_c5(h_avg)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_avg.copy()
            c5_c = evaluate_c5(h)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h.copy()
    c5_a = evaluate_c5(h_avg)
    if c5_a < best_c5:
        best_c5 = c5_a
        best_h = h_avg.copy()
    return best_h, best_c5

def slsqp_refine(h0, n, tau=2000, maxiter=300):
    target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max_val(corr, tau)
    def con_fun(x):
        return np.sum(x) - target
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        x = project_feasible_exact(res.x, target)
        c5 = evaluate_c5(x)
        return x, c5
    except Exception:
        return h0, evaluate_c5(h0)

def binary_sa(h0, n, n_iter, T_start=0.02, T_end=0.001):
    target = int(n / 2.0)
    idx = np.argsort(h0)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    curr_c5 = best_c5
    ones = set(np.where(h == 1.0)[0].tolist())
    zeros = set(np.where(h == 0.0)[0].tolist())
    for t in range(n_iter):
        T = T_start * (T_end / T_start) ** (t / max(n_iter, 1))
        i = list(ones)[np.random.randint(len(ones))]
        j = list(zeros)[np.random.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = evaluate_c5(h)
        delta = new_c5 - curr_c5
        if delta < 0 or (T > 1e-8 and np.random.random() < np.exp(-delta / T)):
            curr_c5 = new_c5
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[j] = 0.0
            h[i] = 1.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    n0 = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP from parent with increasing tau
    for tau in [100, 500, 2000, 5000]:
        if rem() < 20:
            break
        h_new, c5_new = slsqp_refine(best_h, n0, tau=tau, maxiter=400)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Binary SA at n0
    if rem() > 60:

        if c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin
        # SLSQP from best binary

                best_c5 = c5_new

            hp = best_h + 0.03 * np.random.randn(n0)

            if c5_new < best_c5:

        h_t = project_feasible_exact(h_t, tn / 2.0)
