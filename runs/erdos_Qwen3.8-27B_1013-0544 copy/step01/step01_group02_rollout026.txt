import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(c.max())

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def smooth_obj(h, n, alpha):
    c = get_corr(h, n)
    m = c.max()
    s = alpha * (c - m)
    if s.max() > 700:
        return m + (s.max() + np.log(s.max() + np.sum(np.exp(s - s.max())))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    s = alpha * (c - c.max())
    p = np.exp(s)
    p /= p.sum()
    w = p * dx
    grad = np.zeros(n)
    for k in range(2*n-1):
        L = k - (n-1)
        wk = w[k]
        if wk < 1e-16:
            continue
        if L == 0:
            grad += wk * (1.0 - 2.0*h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:n-L])
            grad[:n-L] -= wk * h[L:]
        else:
            M = -L
            grad[:n-M] += wk * (1.0 - h[M:])
            grad[M:] -= wk * h[:n-M]
    return grad

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14: break
            h += (d/t) * r
        else:
            t = h.sum()
            if t < 1e-14: break
            h += (d/t) * h
    return np.clip(h, 0.0, 1.0)

def grad_desc(h, n, alpha, max_it, lr):
    h = h.copy()
    f_old = smooth_obj(h, n, alpha)
    for _ in range(max_it):
        g = smooth_grad(h, n, alpha)
        g -= g.mean()  # maintain sum constraint
        gn = np.linalg.norm(g)
        if gn < 1e-15: break
        step = lr

        for _ in range(40):

            f_new = smooth_obj(h_new, n, alpha)
            if f_new < f_old - 1e-4*step*gn*gn:
                ok = True
                break
            step *= 0.5
        if ok:
            h = h_new
            f_old = f_new
        else:
            lr *= 0.5
            if lr < 1e-14: break
    return h

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(c.max())

def get_corr(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def smooth_obj(h, n, alpha):
    c = get_corr(h, n)
    m = c.max()
    s = alpha * (c - m)
    if s.max() > 700:
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def smooth_grad(h, n, alpha):
    c = get_corr(h, n)
    s = alpha * (c - c.max())
    p = np.exp(s)
    p /= p.sum()
    w = p * (2.0 / n)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        wk = w[k]
        if wk < 1e-16:
            continue
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:n - L])
            grad[:n - L] -= wk * h[L:]
        else:
            M = -L
            grad[:n - M] += wk * (1.0 - h[M:])
            grad[M:] -= wk * h[:n - M]
    return grad

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()

        if abs(d) < 1e-14:

            t = r.sum()

                break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def grad_desc(h, n, alpha, max_it, lr):
    h = h.copy()

        if ok:
            h = h_new
            f_old = f_new
        else:
            lr *= 0.4
            if lr < 1e-14:
                break
    return h
