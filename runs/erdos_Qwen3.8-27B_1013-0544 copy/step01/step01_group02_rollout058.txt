    dx = 2.0 / n

    c_raw = np.correlate(h, 1 - h, mode='full')  # shape (2n-1,)

    # c_raw[m] = sum_i h[i] * (1-h[i + lags[m]])
    # Terms with h[j]: 
    #   i=j: h[j]*(1-h[j+lags[m]]) -> deriv = (1-h[j+lags[m]]) if valid
    #   i+lags[m]=j -> i=j-lags[m]: h[j-lags[m]]*(1-h[j]) -> deriv = -h[j-lags[m]] if valid

    # Vectorized
    dg = g[:, None] - g[None, :]  # dg[m,l] = g[m] - g[l]
    dc = c_raw[None, :] - c_raw[:, None]  # dc[m,l] = c_raw[l] - c_raw[m]

    d_bp = np.full_like(dg, np.nan)
    d_bp[mask] = dc[mask] / dg[mask]
    
    d_flat = d_bp[mask]
    d_flat = d_flat[np.isfinite(d_flat)]
    d_flat = d_flat[(d_flat >= lo - 1e-12) & (d_flat <= hi + 1e-12)]
    d_flat = np.clip(d_flat, lo, hi)
    
    candidates = np.unique(np.concatenate([d_flat, [lo, hi]]))

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + mx / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def coord_descent(h, n, max_sweeps=20):
    """Exact coordinate descent using breakpoint analysis."""
    h = h.copy()
    dx = 2.0 / n
    c_raw_init = np.correlate(h, 1 - h, mode='full')
    c5_init = c_raw_init.max() * dx
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            lags = np.arange(-(n-1), n)
            valid_plus = (j + lags >= 0) & (j + lags < n)
            valid_minus = (j - lags >= 0) & (j - lags < n)
            
            g = np.zeros(2*n-1)
            g[valid_plus] += (1.0 - h[j + lags[valid_plus]])
            g[valid_minus] -= h[j - lags[valid_minus]]
            
            lo = -h[j]
            hi = 1.0 - h[j]
            
            if hi - lo < 1e-15:
                continue
            
            c_raw = np.correlate(h, 1 - h, mode='full')
            
            dg = g[:, None] - g[None, :]
            dc = c_raw[None, :] - c_raw[:, None]
            mask = np.abs(dg) > 1e-12
            d_bp = np.full(dg.shape, np.nan)
            d_bp[mask] = dc[mask] / dg[mask]
            d_flat = d_bp[mask]
            d_flat = d_flat[np.isfinite(d_flat)]
            d_flat = d_flat[(d_flat >= lo - 1e-12) & (d_flat <= hi + 1e-12)]
            d_flat = np.clip(d_flat, lo, hi)
            
            candidates = np.unique(np.concatenate([d_flat, [lo, hi]]))
            vals = c_raw[None, :] + candidates[:, None] * g[None, :]
            obj = vals.max(axis=1)
            best_idx = np.argmin(obj)
            best_d = candidates[best_idx]
            
            new_val = h[j] + best_d
            new_c5 = obj[best_idx] * dx
            
            if new_c5 < c5_init - 1e-14:
                h[j] = new_val
                c5_init = new_c5
                improved = True
        
        if not improved:

c_m(d) = c_m(0) + d * (g_m[i] - g_m[j])

where g_m[i] is the derivative of c_m w.r.t. h[i] (computed as before).

So it's still piecewise linear in d, and we can use the same breakpoint approach. The "g" vector for the pair is g_pair[m] = g_m[i] - g_m[j].

This is O((2n-1)²) per pair, same as single coordinate. And there are O(n²) pairs = 6400 pairs for n=80. So a full pair sweep is 6400 * 2M = 12.8B operations. That's too much.

Let me limit to "important" pairs: pairs where both i and j contribute to the max correlation. Or just do a subset of pairs.

Actually, let me skip the pair optimization for now and focus on making the single-coordinate descent really good, combined with SLSQP and random restarts.

Let me also think about the "cycling" problem in coordinate descent. In general, coordinate descent can cycle (revisit the same points). For our problem, since the objective is linear in each coordinate (piecewise linear), and we're taking the exact minimum in each coordinate, cycling is unlikely but possible. I'll add a detection mechanism: if we visit the same point twice, stop.

Actually, for piecewise linear objectives, coordinate descent converges in finite steps (it can't cycle because each step either makes a unique improvement or stops). The number of steps is bounded by the number of "vertices" of the feasible region intersected with the constraint, which is finite. So it will terminate.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + mx / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return np.sum(x) - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def optimize_coord(h, j, n):
    c_raw = np.correlate(h, 1 - h, mode='full')
    lags = np.arange(-(n - 1), n)
    vp = (j + lags >= 0) & (j + lags < n)
    vm = (j - lags >= 0) & (j - lags < n)
    g = np.zeros(2 * n - 1)
    g[vp] += (1.0 - h[j + lags[vp]])
    g[vm] -= h[j - lags[vm]]
    lo = -h[j]
    hi = 1.0 - h[j]
    if hi - lo < 1e-15:
        return h[j]
    dg = g[:, None] - g[None, :]
    dc = c_raw[None, :] - c_raw[:, None]
    mask = np.abs(dg) > 1e-14
    with np.errstate(divide='ignore', invalid='ignore'):
        d_bp = np.where(mask, dc / dg, np.nan)
    d_flat = d_bp[mask]
    d_flat = d_flat[np.isfinite(d_flat)]
    d_flat = d_flat[(d_flat >= lo) & (d_flat <= hi)]
    candidates = np.unique(np.concatenate([d_flat, [lo, hi, 0.0]]))
    candidates = candidates[(candidates >= lo) & (candidates <= hi)]
    vals = c_raw[None, :] + candidates[:, None] * g[None, :]
    obj = vals.max(axis=1)
    best_idx = np.argmin(obj)
    return h[j] + candidates[best_idx]

def optimize_pair(h, i, j, n):
    c_raw = np.correlate(h, 1 - h, mode='full')
    lags = np.arange(-(n - 1), n)
    vp_i = (i + lags >= 0) & (i + lags < n)
    vm_i = (i - lags >= 0) & (i - lags < n)
    vp_j = (j + lags >= 0) & (j + lags < n)
    vm_j = (j - lags >= 0) & (j - lags < n)
    g = np.zeros(2 * n - 1)
    g[vp_i] += (1.0 - h[i + lags[vp_i]])
    g[vm_i] -= h[i - lags[vm_i]]
    g[vp_j] -= (1.0 - h[j + lags[vp_j]])
    g[vm_j] += h[j - lags[vm_j]]
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-15:
        return h[i], h[j]
    dg = g[:, None] - g[None, :]
    dc = c_raw[None, :] - c_raw[:, None]
    mask = np.abs(dg) > 1e-14
    with np.errstate(divide='ignore', invalid='ignore'):
        d_bp = np.where(mask, dc / dg, np.nan)
    d_flat = d_bp[mask]
    d_flat = d_flat[np.isfinite(d_flat)]
    d_flat = d_flat[(d_flat >= lo) & (d_flat <= hi)]
    candidates = np.unique(np.concatenate([d_flat, [lo, hi, 0.0]]))
    candidates = candidates[(candidates >= lo) & (candidates <= hi)]
    vals = c_raw[None, :] + candidates[:, None] * g[None, :]
    obj = vals.max(axis=1)
    best_idx = np.argmin(obj)
    d = candidates[best_idx]
    return h[i] + d, h[j] - d

def coord_descent(h, n, max_sweeps=30, use_pairs=False):
    h = h.copy()
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            new_val = optimize_coord(h, j, n)
            if abs(new_val - h[j]) > 1e-14:
                h[j] = new_val
                improved = True
        if use_pairs and improved:
            for i in range(n):
                for j in range(i + 1, n):
                    ni, nj = optimize_pair(h, i, j, n)
                    if abs(ni - h[i]) > 1e-14 or abs(nj - h[j]) > 1e-14:
                        h[i], h[j] = ni, nj
                        improved = True
        if not improved:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)

        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]

    # Phase 1: SLSQP continuation
    for alpha in [50, 200, 1000, 5000, 20000, 100000]:
        if budget_s - (time.time() - t0) < 15:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-18})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5 - 1e-15:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    # Phase 2: Coordinate descent
    if budget_s - (time.time() - t0) > 20:
        h_try = coord_descent(best_h, n, max_sweeps=50, use_pairs=False)
        c = compute_c5(h_try, n)
        if c < best_c5:

    # Phase 3: Pair coordinate descent

        h_try = coord_descent(best_h, n, max_sweeps=10, use_pairs=True)
        c = compute_c5(h_try, n)
        if c < best_c5:
            best_h, best_c5 = h_try, c

    # Phase 4: SLSQP polish
    for alpha in [50000, 200000]:
        if budget_s - (time.time() - t0) < 15:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-18})
            h_try = project_feasible(res.x, n)

        except Exception:
            pass

    # Phase 5: Random restarts with local search
    no_improve_count = 0
    while budget_s - (time.time() - t0) > 80:
        if no_improve_count < 20:

        else:
            scale = rng.lognormal(mean=-2.0, sigma=1.2)
        scale = min(scale, 0.5)
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = project_feasible(h_pert, n)

        c = compute_c5(h_try, n)

        if c < best_c5 - 1e-14:
            best_h, best_c5 = h_try, c
            no_improve_count = 0
            if budget_s - (time.time() - t0) > 20:
                try:
                    res = minimize(obj_smooth, best_h, args=(n, 50000), method='SLSQP',
                                  bounds=bounds, constraints=cons,
                                  options={'maxiter': 1000, 'ftol': 1e-18})
                    h_try2 = project_feasible(res.x, n)
                    c2 = compute_c5(h_try2, n)
                    if c2 < best_c5 - 1e-15:
                        best_h, best_c5 = h_try2, c2
                except Exception:
                    pass
        else:
            no_improve_count += 1

    # Phase 6: Final intensive coordinate descent with pairs
    if budget_s - (time.time() - t0) > 15:
        h_try = coord_descent(best_h, n, max_sweeps=30, use_pairs=True)
        c = compute_c5(h_try, n)
        if c < best_c5:
            best_h, best_c5 = h_try, c

    if budget_s - (time.time() - t0) > 5:

                          bounds=bounds, constraints=cons,

            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)

    return best_h, best_c5, n
