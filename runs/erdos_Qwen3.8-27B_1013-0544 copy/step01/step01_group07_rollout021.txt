                direction[m:] += w * h[:n-m]

                direction[:n+m] += w * h[-m:]

def slsqp_optimize(h, dx, target_sum, t, maxiter=1000):
    n = len(h)
    def obj(x):
        x_clipped = np.clip(x, 0, 1)
        return smooth_obj(x_clipped, t, dx)
    def obj_grad(x):
        x_clipped = np.clip(x, 0, 1)
        corr = np.correlate(x_clipped, 1 - x_clipped, mode="full")
        shifted = t * (corr - corr.max())
        exp_s = np.exp(shifted)
        weights = exp_s / exp_s.sum()
        # Gradient of smooth max wrt h
        grad = np.zeros(n)
        for k, w in enumerate(weights):
            if w < 1e-10:
                continue
            m = k - (n - 1)
            # d(corr[k])/d(h[j]) = (1-h[j+m]) - h[j-m]
            if m >= 0:
                grad[:n-m] += w * (1 - x_clipped[m:])
                grad[m:] += -w * x_clipped[:n-m]
            else:
                grad[-m:] += w * (1 - x_clipped[:n+m])
                grad[:n+m] += -w * x_clipped[-m:]
        return grad * dx
    # Use equality constraint for sum
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h.copy(), jac=obj_grad, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_box_sum(res.x, target_sum)
    except:
        return fix_sum(h, target_sum)

def block_descent(h, dx, target_sum, block_size=20, n_rounds=100, t=50):

                         options={'maxiter': 200, 'ftol': 1e-12})
            h_new = h.copy()
            h_new[idx] = np.clip(res.x, 0.0, 1.0)
            h_new = fix_sum(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
        except:
            pass
    return h, best_c5

    n = len(h)
