```python
def project_feasible(x, target_sum, n):
    """Project x onto {0<=x<=1, sum(x)=target_sum}."""
    x = x.copy()
    x = np.clip(x, 0, 1)
    
    # Now fix the sum
    diff = target_sum - np.sum(x)
    if abs(diff) < 1e-12:
        return x
    
    if diff > 0:
        # Need to increase sum: add to elements not at upper bound
        mask = x < 1.0
        if not np.any(mask):
            return x
        # Water-filling
        headroom = 1.0 - x[mask]
        # Find threshold such that sum(min(headroom, t)) = diff
        # Sort headroom
        h_sorted = np.sort(headroom)
        cumsum = np.cumsum(h_sorted)
        # Find the point where cumsum >= diff
        idx = np.searchsorted(cumsum, diff)
        if idx >= len(h_sorted):
            # All headroom is used, distribute evenly
            x[mask] += diff / np.sum(headroom)
        else:
            threshold = (diff - (cumsum[idx-1] if idx > 0 else 0)) / (len(h_sorted) - idx)
            x[mask] += np.minimum(headroom, threshold)
    else:
        # Need to decrease sum: subtract from elements not at lower bound
        mask = x > 0.0
        if not np.any(mask):
            return x
        capacity = x[mask]
        cap_sorted = np.sort(capacity)
        cumsum = np.cumsum(cap_sorted)
        deficit = -diff
        idx = np.searchsorted(cumsum, deficit)
        if idx >= len(cap_sorted):
            x[mask] -= deficit / np.sum(capacity)
        else:
            threshold = (deficit - (cumsum[idx-1] if idx > 0 else 0)) / (len(cap_sorted) - idx)
            x[mask] -= np.minimum(capacity, threshold)
    
    return np.clip(x, 0, 1)
```

```python
    """Solve the 1D subproblem for pair (i, j).
    Returns optimal h_i value."""

    lo = max(0.0, c - 1.0)
    hi = min(1.0, c)
    
    if hi - lo < 1e-10:
        return h[i]
    
    # Ternary search for min of max of convex functions
    for _ in range(50):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        
        # Evaluate objective at m1 and m2
        h[i], h[j] = m1, c - m1
        obj1 = np.max(compute_corr(h))
        
        h[i], h[j] = m2, c - m2
        obj2 = np.max(compute_corr(h))
        
        if obj1 < obj2:
            hi = m2
        else:
            lo = m1

    return opt
```

```python
def project_feasible(x, target_sum):
    x = np.clip(x, 0.0, 1.0).copy()
    diff = target_sum - x.sum()
    if abs(diff) < 1e-14:
        return x
    if diff > 0:
        mask = x < 1.0 - 1e-15
        if not np.any(mask):
            return x
        headroom = (1.0 - x[mask])
        total_head = headroom.sum()
        if total_head <= 0:
            return x
        # Water filling
        sorted_idx = np.argsort(headroom)
        sorted_h = headroom[sorted_idx]
        cum = np.cumsum(sorted_h)
        idx = np.searchsorted(cum, diff)
        if idx >= len(sorted_h):
            x[mask] += diff / total_head
        else:
            prev = cum[idx-1] if idx > 0 else 0.0
            t = (diff - prev) / (len(sorted_h) - idx)
            x[mask] += np.minimum(headroom, t)
    else:
        mask = x > 1e-15
        if not np.any(mask):
            return x
        cap = x[mask]
        total_cap = cap.sum()
        if total_cap <= 0:
            return x
        deficit = -diff
        sorted_idx = np.argsort(cap)
        sorted_c = cap[sorted_idx]
        cum = np.cumsum(sorted_c)
        idx = np.searchsorted(cum, deficit)
        if idx >= len(sorted_c):
            x[mask] -= deficit / total_cap
        else:
            prev = cum[idx-1] if idx > 0 else 0.0
            t = (deficit - prev) / (len(sorted_c) - idx)
            x[mask] -= np.minimum(cap, t)
    return np.clip(x, 0.0, 1.0)
```
