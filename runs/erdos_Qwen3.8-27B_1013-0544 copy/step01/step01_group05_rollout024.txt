Wait, I need to be careful. L-BFGS-B minimizes a scalar function. The objective is max_k corr_k * dx. The log-sum-exp approximation is (1/T) * log(sum exp(T * corr_k)) * dx. For large T, this approaches max(corr_k) * dx.

The gradient of (1/T) * log(sum exp(T * corr_k)) with respect to h is:
= (1/T) * (1/sum exp(T*corr_k)) * sum exp(T*corr_k) * T * d(corr_k)/dh
= sum_k w_k * d(corr_k)/dh

where w_k = exp(T*corr_k) / sum_j exp(T*corr_j).

So the gradient of the smoothed objective (times dx) is:
dx * sum_k w_k * grad_lag(h, k-nh)

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    lo = float(np.min(x)) - 1.5
    hi = float(np.max(x)) + 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if float(np.sum(np.clip(x - mid, 0.0, 1.0))) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad_lag(h, L):
    n = len(h)
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h)[:n - L]
        g[:n - L] -= h[L:]
    else:
        L2 = -L
        g[:n - L2] += (1.0 - h)[L2:]
        g[L2:] -= h[:n - L2]
    return g

def compute_

_weighted_grad(h, K, T):
    nh = len(h) - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-K:]
    if T > 0:
        tv = corr[top_idx]
        w = np.exp(T * (tv - tv.max()))
        w /= w.sum()
    else:
        w = np.ones(K) / K
    g = np.zeros(len(h))
    for i, wi in zip(top_idx, w):
        g += wi * grad_lag(h, int(i) - nh)
    return g

def local_opt(h, dx, ts, rng, t_end, K, ss_start, mu=0.5, restart_sig=0.015):
    n = len(h)
    best = h.copy()
    bc = compute_c5(h, dx)
    ss = ss_start
    mom = np.zeros(n)
    stag = 0
    while time.time() < t_end:
        g = compute_weighted_grad(h, K, 3.0)
        mom = mu * mom - ss * g
        h2 = project_feasible(h + mom, ts)
        c5 = compute_c5(h2, dx)
        if c5 < bc:
            bc = c5
            best = h2.copy()
            h = h2
            stag = 0
        else:
            h = h2
            stag += 1
            if stag >= 8:
                h = best + rng.normal(0, restart_sig, n)
                h = project_feasible(h, ts)
                mom = np.zeros(n)
                ss *= 0.75
                stag = 0
                if ss < 1e-10:
                    break
    return best, bc

def smooth_obj_grad(x, dx, ts, T, pen):
    n = len(x)
    nh = n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = corr.max()
    exps = np.exp(T * (corr - cmax))
    log_sum = cmax + np.log(np.sum(exps))
    obj = log_sum / T * dx + pen * (np.sum(x) - ts) ** 2
    w = exps / exps.sum()
    g = np.zeros(n)
    active = np.where(w > 1e-5)[0]
    for i in active:
        g += w[i] * grad_lag(x, int(i) - nh)
    g *= dx
    g += 2.0 * pen * (np.sum(x) - ts)
    return obj, g

def lbfgsb_opt(h, dx, ts, t_end, T=5.0, pen=500.0):
    n = len(h)
    def fun(x):
        return smooth_obj_grad(x, dx, ts, T, pen)[0]
    def gfun(x):
        return smooth_obj_grad(x, dx, ts, T, pen)[1]
    res = minimize(fun, h, jac=gfun, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': 3000, 'maxfun': 10000,
                            'ftol': 1e-16, 'gtol': 1e-12,
                            'iprint': -1, 'maxls': 50})
    if time.time() < t_end and res.x is not None:
        h2 = project_feasible(res.x, ts)
        return h2, compute_c5(h2, dx)
    return h, compute_c5(h, dx)

def make_periodic_init(n, rng):
    h = np.zeros(n)
    n_comp = rng.randint(1, 4)
    for _ in range(n_comp):
        freq = rng.randint(1, max(2, n // 2))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.15, 0.5)
        h += amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
    return np.clip(0.45 + 0.35 * h, 0.0, 1.0)
