    h = h.copy()
    h = np.clip(h, 0.0, 1.0)

            if total < 1e-15:

            h = np.clip(h, 0.0, 1.0)

            if total < 1e-15:
                break
            scale = diff / total
            h += scale * h
            h = np.clip(h, 0.0, 1.0)

def build_periodic(m, values, n):
    """Build h by repeating m values, each repeated n//m times."""
    block_size = n // m
    h = np.tile(values, block_size)
    if len(h) < n:
        h = np.concatenate([h, values[:n - len(h)]])
    return h[:n]

def opt_periodic(m, n, budget_s, t0):
    """Optimize m periodic values."""
    # Parameters: m values, constraint sum = m/2
    # Use m-1 free parameters (last determined by constraint)
    
    def obj(params):
        if budget_s - (time.time() - t0) < 5:
            return 1e10
        v = np.zeros(m)
        v[:m-1] = params
        v[m-1] = m/2.0 - params.sum()
        if np.any(v < -0.01) or np.any(v > 1.01):
            return 1e10
        v = np.clip(v, 0, 1)

        h = build_periodic(m, v, n)
        h = project_feasible(h, n)
        return compute_c5(h, n)
    
    # Multiple random starts

    best_v = None

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-15:
                break
            h = np.clip(h + diff / t * room, 0.0, 1.0)
        else:
            t = h.sum()
            if t < 1e-15:
                break
            h = np.clip(h + diff / t * h, 0.0, 1.0)
    return h

def smooth_obj(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def con_fun(h, n):
    return np.sum(h) - n / 2.0

def con_jac(h, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000, ftol=1e-18):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (n,)}]
    res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                  bounds=bounds, constraints=cons,
                  options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, n)

def thue_morse(n):
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t
