from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def soft_max_c5(h, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr * tau
    m = np.max(shifted)
    val = (m + np.log(np.sum(np.exp(shifted - m)))) / tau
    return val * dx

def project_box_sum(x, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    return result

def active_set_gradient(h, n_active=10):
    """Compute averaged gradient of top n_active correlations"""
    n = len(h)
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    tol = max(0.05, c_max * 0.005)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < 2:
        active = np.argsort(corr)[-2:]
    elif len(active) > n_active:
        active = np.argsort(corr)[-n_active:]
    
    grad = np.zeros(n)
    for idx in active:
        k = int(idx) - n_half
        # d corr[k] / d h[j] = (1-h[j-k]) * [0<=j-k<n] - h[j+k] * [0<=j+k<n]
        j = np.arange(n)
        # First term: valid when 0 <= j-k < n, i.e., k <= j < k+n
        mask1 = (j >= max(0, k)) & (j < min(n, k + n))
        # Wait, 0 <= j-k < n means k <= j < k+n
        mask1 = (j - k >= 0) & (j - k < n)
        # Second term: valid when 0 <= j+k < n, i.e., -k <= j < n-k
        mask2 = (j + k >= 0) & (j + k < n)
        
        contrib = np.zeros(n)
        if np.any(mask1):
            contrib[mask1] += (1.0 - h[j[mask1] - k])
        if np.any(mask2):
            contrib[mask2] -= h[j[mask2] + k]
        grad += contrib
    grad /= float(len(active))
    return grad, c_max * (2.0/n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def soft_max_value(h, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr * tau
    m = np.max(shifted)
    return (m + np.log(np.sum(np.exp(shifted - m)))) / tau

def project_box_sum(x, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def active_set_gradient(h, dx, n_active=10, tol_frac=0.003):
    n = len(h)
    k_offset = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    tol = max(0.02, c_max * tol_frac)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < 2:
        active = np.argsort(corr)[-2:]
    elif len(active) > n_active:
        active = np.argsort(corr)[-n_active:]
    grad = np.zeros(n)
    for idx in active:
        k = int(idx) - k_offset
        i_start = max(0, -k)
        i_end = min(n, n - k)
        for i in range(i_start, i_end):
            grad[i] += (1.0 - h[i + k])
            grad[i + k] -= h[i]
    grad /= float(len(active))
    return grad * dx

def softmax_gradient(h, tau, dx, n_active=20):
    n = len(h)
    k_offset = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr * tau
    m = np.max(shifted)
    weights = np.exp(shifted - m)
    weights /= np.sum(weights)
    if n_active < len(weights):
        top_idx = np.argsort(weights)[-n_active:]
        mask = np.zeros(len(weights), dtype=bool)
        mask[top_idx] = True
        weights[~mask] = 0.0
        ws = weights.sum()
        if ws > 0:
            weights /= ws
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = int(idx) - k_offset
        i_start = max(0, -k)
        i_end = min(n, n - k)
        for i in range(i_start, i_end):
            grad[i] += w * (1.0 - h[i + k])
            grad[i + k] -= w * h[i]
    return grad * dx

    """Optimize a binary sequence to minimize max correlation"""
    dx = 2.0 / n_points
    target = n_points // 2

    ones = np.arange(n_points)
    rng.shuffle(ones)
    h[ones[:target]] = 1.0
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for it in range(max_iters):
        if it % 10 == 0:

            if rng.random() < 0.1:

                i2 = rng.randint(n_points)

                    pass  # just a swap, sum unchanged
