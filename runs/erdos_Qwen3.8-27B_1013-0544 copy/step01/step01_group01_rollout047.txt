        mm = (idx - lag >= 0) & (idx - lag < n)
        grad[mp] += w * (1.0 - x[idx[mp] + lag])
        grad[mm] += w * (-x[idx[mm] - lag])
    return grad * _dx_current


def _eq_fun(x):
    return x.sum() - _target_sum


def _eq_jac(x):
    return np.ones(len(x))


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = h.copy()
    for _ in range(30):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def get_active_lags(h, tol=0.01):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    return np.where(corr >= mx - tol)[0], mx


def run(seed=42, budget_s=1100, **kwargs):
    global _target_sum, _beta_current, _dx_current
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    _target_sum = target_sum
    _dx_current = dx

    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

    def time_left():
        return budget_s - (time.time() - start_time)

    # Phase 1: Multi-scale gradient optimization
    beta_schedule = [30, 60, 120, 250, 500, 1000]
    for beta in beta_schedule:
        if time_left() < 40:
            break
        _beta_current = beta
        max_iter = 800 if beta <= 120 else 500
        try:
            res = minimize(_obj_func, best_h, method='SLSQP', jac=_jac_func,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-15})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5 - 1e-10:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 2: Diverse restarts with exact gradient
    no_improve = 0
    attempt = 0
    while time_left() > 25 and no_improve < 80:
        attempt += 1
        h0 = best_h.copy()
        strategy = attempt % 7

        if strategy == 0:
            scale = 0.02 + 0.08 * np.exp(-attempt / 50.0)
            h0 += np.random.randn(n_points) * scale
        elif strategy == 1:
            # Active-set targeted: reduce correlation at active lags
            active, mx = get_active_lags(best_h, 0.015)
            for idx in active[:8]:
                lag = idx - (n_points - 1)
                if lag == 0:
                    continue
                pairs = []
                for j in range(max(1, -lag), min(n_points, n_points - lag)):
                    pairs.append((j, j + lag if lag > 0 else j - lag))
                if pairs:
                    for (j, k) in pairs[:5]:
                        if best_h[j] * (1 - best_h[k]) > 0.2:
                            h0[j] -= 0.015
                            h0[k] += 0.015
                        elif best_h[k] * (1 - best_h[j]) > 0.2:
                            h0[k] -= 0.015
                            h0[j] += 0.015
        elif strategy == 2:
            nb = np.random.randint(3, max(4, n_points // 5))
            st = np.random.randint(0, n_points - nb)
            h0[st:st + nb] = np.roll(h0[st:st + nb], np.random.randint(1, nb))
        elif strategy == 3:
            seg = np.random.randint(4, max(5, n_points // 3))
            s1 = np.random.randint(0, n_points - seg)
            s2 = np.random.randint(0, n_points - seg)
            h0[s1:s1 + seg] = best_h[s2:s2 + seg].copy()
        elif strategy == 4:
            freq = np.random.uniform(2, 8)
            phase = np.random.uniform(0, 2 * np.pi)
            xg = np.linspace(0, 2, n_points)
            h0 += 0.04 * np.sin(freq * xg + phase)
            h0 += np.random.randn(n_points) * 0.01
        elif strategy == 5:
            # Block exchange
            bl = np.random.randint(4, max(5, n_points // 4))
            s1 = np.random.randint(0, n_points - bl)
            s2 = np.random.randint(0, n_points - bl)
            tmp = h0[s1:s1 + bl].copy()
            h0[s1:s1 + bl] = h0[s2:s2 + bl]
            h0[s2:s2 + bl] = tmp
        else:
            # Gentle annealing
            h0 += np.random.randn(n_points) * 0.01
            h0 += 0.02 * np.random.choice([-1, 1], n_points) * (h0 < 0.5).astype(float) * (h0 > 0.5).astype(float)

        h0 = project_h(h0, target_sum)

        _beta_current = 150 if attempt % 4 != 0 else 300
        try:
            res = minimize(_obj_func, h0, method='SLSQP', jac=_jac_func,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-14})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5 - 1e-10:
                best_h, best_c5 = h_new.copy(), c5
                no_improve = 0
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

    # Phase 3: Final high-precision polish
    if time_left() > 15:
        for beta in [800, 1500, 3000]:
            if time_left() < 10:
                break
            _beta_current = beta
            try:
                res = minimize(_obj_func, best_h, method='SLSQP', jac=_jac_func,
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-16})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Final verification
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
