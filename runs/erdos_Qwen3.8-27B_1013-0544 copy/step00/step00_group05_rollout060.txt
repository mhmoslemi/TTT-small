from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = np.clip((corr - m) / beta, -100, 100)
    return float(m + beta * np.log(np.sum(np.exp(shifted))))

def make_obj(h_init, beta):
    n = len(h_init)
    def obj(h):
        return smooth_c5(h, beta)
    def grad(h):
        # Finite difference gradient
        g = np.zeros(n)
        eps = 1e-7
        base = smooth_c5(h, beta)
        for i in range(n):
            h[i] += eps
            g[i] = (smooth_c5(h, beta) - base) / eps
            h[i] -= eps
        return g
    return obj, grad

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        h += diff / n
    return h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    corr_scaled = corr * dx
    m = np.max(corr_scaled)
    shifted = np.clip((corr_scaled - m) / beta, -100, 100)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    fval = m + beta * np.log(np.sum(exp_vals))
    
    # Compute gradient
    # grad = (dx/beta) * sum_k weights[k] * grad_c(k)
    # where grad_c(k) = d(c(k))/dh for the k-th element of full correlation
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-15:
            continue
        lag = k - (n - 1)
        d = abs(lag)
        if d == 0:
            # grad_c(p) = 1 - 2*h[p]
            grad += (dx / beta) * weights[k] * (1.0 - 2.0 * h)
        else:
            # For p from d to n-1: (1-h[p-d])
            # For p from 0 to n-1-d: -h[p+d]
            # Combined: for p in [d, n-1-d]: (1-h[p-d]) - h[p+d]
            #           for p in [n-1-d+1, n-1]: (1-h[p-d]) (if d < n-1-d)
            #           for p in [0, d-1]: -h[p+d] (if d > 0)

    return fval, grad
