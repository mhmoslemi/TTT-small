    n = len(h)

    return float(corr.max() * dx)

def normalize_h(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.argsort(1.0 - h)
    else:
        idx = np.argsort(h)
    remaining = abs(diff)
    for i in idx:
        if remaining < 1e-14:
            break
        if diff > 0:
            add = min(1.0 - h[i], remaining)
            h[i] += add
            remaining -= add
        else:
            sub = min(h[i], remaining)
            h[i] -= sub
            remaining -= sub
    return h
