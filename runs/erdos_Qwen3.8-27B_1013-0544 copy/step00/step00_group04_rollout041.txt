    if n_target is None:
        n = len(h)
    else:
        n = n_target
    # Upsample if needed
    if n_target is not None and n_target != len(h):
        old_indices = np.linspace(0, 2, len(h))
        new_indices = np.linspace(0, 2, n_target)
        h_interp = np.interp(new_indices, old_indices, h)

    sorted_vals = np.sort(h_interp)

    h_bin = (h_interp >= threshold).astype(float)

    n_ones = int(np.sum(h_bin == 1))

            n_ones -= 1

def simulated_annealing(h, n, max_iters, rng, time_limit=None, start_time=None):
    dx = 2.0 / n
    h = h.copy().astype(np.float64)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    current_c5 = float(np.max(np.correlate(h, 1-h, mode='full')) * dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    T_start = 0.1
    T_end = 0.001
    n_total = max_iters
    
    for it in range(max_iters):
        if start_time is not None and time.time() - start_time > time_limit:
            break
        
        T = T_start * (T_end/T_start) ** (it / max(1, n_total-1))
        
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        ones.remove(i)
        zeros.remove(j)
        ones.append(j)
        zeros.append(i)
        
        corr = np.correlate(h, 1-h, mode='full')
        new_c5 = float(np.max(corr) * dx)
        
        if new_c5 < current_c5 or rng.random() < np.exp((current_c5 - new_c5) / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            ones.remove(j)
            zeros.remove(i)
            ones.append(i)
            zeros.append(j)
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_binary(h, n):
    if n != len(h):
        old_x = np.linspace(0, 2, len(h), endpoint=False)
        new_x = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(new_x, old_x, h)
    n_half = n // 2
    sorted_idx = np.argsort(h)
    h_bin = np.zeros(n, dtype=np.float64)
    h_bin[sorted_idx[n - n_half:]] = 1.0
    return h_bin

def sa_binary(h, n, rng, t_end):
    dx = 2.0 / n
    h = h.copy()
    curr = compute_c5(h)
    best = curr
    best_h = h.copy()
    T = 0.015
    while time.time() < t_end:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        h[i], h[j] = 0.0, 1.0
        new = compute_c5(h)
        d = new - curr
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            curr = new
            if new < best:
                best = new
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T * 0.99998, 0.002)
    return best_h, best

        new_x = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(new_x, old_x, h)
    n_half = n // 2
    sorted_idx = np.argsort(h)
    h_bin = np.zeros(n, dtype=np.float64)
    h_bin[sorted_idx[n - n_half:]] = 1.0
    return h_bin

def perturb(h, rng, n_flips=None):
    h = h.copy()
    n = len(h)
    if n_flips is None:
        n_flips = max(2, n // 14)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    for _ in range(n_flips):
        a = int(ones[rng.integers(len(ones))])
        b = int(zeros[rng.integers(len(zeros))])
        h[a], h[b] = 1.0 - h[a], h[b]
    return h

def sa_binary(h, n, rng, t_end):
    dx = 2.0 / n
    h = h.copy()
    curr = compute_c5(h)
    best = curr
    best_h = h.copy()
    T = 0.015
    while time.time() < t_end:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        h[i] = 0.0
        h[j] = 1.0
        new = compute_c5(h)
        d = new - curr
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            curr = new
            if new < best:
                best = new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T * 0.99998, 0.002)
    return best_h, best

def subgrad_refine(h, n, t_end):
    dx = 2.0 / n
    h = h.copy().astype(np.float64)
    best = compute_c5(h)
    best_h = h.copy()
    lr = 0.05
    while time.time() < t_end:
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = int(np.argmax(corr))
        k = m - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            grad[:n-k] += dx * (1.0 - h[k:])
            grad[k:] -= dx * h[:n-k]
        else:
            kp = -k
            grad[kp:] += dx * (1.0 - h[:n-kp])
            grad[:n-kp] -= dx * h[kp:]
        h_new = h - lr * grad
        for _ in range(50):
            offset = (n / 2.0 - h_new.sum()) / n
            h_new = np.clip(h_new + offset, 0.0, 1.0)
            if abs(h_new.sum() - n / 2.0) < 1e-12:
                break
        c5_new = compute_c5(h_new)
        if c5_new < best:
            best = c5_new
            best_h = h_new.copy()
        h = h_new
        lr = max(lr * 0.9995, 0.001)
    return best_h, best
