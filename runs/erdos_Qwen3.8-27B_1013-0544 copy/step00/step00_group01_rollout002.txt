    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Softmax weights
    scaled = corr / (dx * temperature)
    scaled -= np.max(scaled)
    weights = np.exp(scaled)
    weights /= weights.sum()
    
    # Compute gradient as weighted sum
    grad = np.zeros(n)

            g[0:n-s] -= h[s:n]

    return grad * dx

    return float(np.max(corr) * dx)

    """Gradient of the correlation at shift s (|lag| = s) with respect to h."""
    n = len(h)
    g = np.zeros(n)

        # Also try a slightly perturbed version
        h_pert = proj_feasible(h_interp + rng.randn(n) * 0.03, n)
        for h_start in [h_interp, h_pert]:
            if time.time() > t_limit2:
                break
            hh, cc = optimize_loop(h_start, n, t0, t_limit2, rng, lr_init=0.004, temp_init=0.005)
            if cc < c5_best:
                h_best, c5_best = hh, cc

    # Phase 3: random restarts and binary perturbations at best n
    n_best = len(h_best)
    t_limit3 = t0 + budget_s * 0.95
    restart_count = 0
    while time.time() < t_limit3:
        restart_count += 1
        # Mix of continuous and binary perturbations
        if restart_count % 3 == 0:
            # Binary perturbation: flip some entries
            h_try = h_best.copy()
            n_flip = max(1, n_best // 8)
            idx = rng.choice(n_best, size=n_flip, replace=False)
            h_try[idx] = 1.0 - h_try[idx]
            h_try = proj_feasible(h_try, n_best)
        else:
            scale = 0.05 + 0.1 * (restart_count % 5) / 5.0
            h_try = proj_feasible(h_best + rng.randn(n_best) * scale, n_best)

        hh, cc = optimize_loop(h_try, n_best, t0, t_limit3, rng, lr_init=0.003, temp_init=0.003)
        if cc < c5_best:
            h_best, c5_best = hh, cc

    # Final polish: very small steps on the best solution
    if time.time() < t0 + budget_s * 0.99:
        hh, cc = optimize_loop(h_best, len(h_best), t0, t0 + budget_s * 0.99,
                               rng, lr_init=0.001, temp_init=0.001, temp_floor=0.0005, beta=0.9)
        if cc < c5_best:
            h_best, c5_best = hh, cc

    return h_best, c5_best, len(h_best)
