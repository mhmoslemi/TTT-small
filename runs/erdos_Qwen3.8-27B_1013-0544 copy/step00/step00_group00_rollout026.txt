import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(v, S):
    lo = -5.0
    hi = 5.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def c5_exact(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    cm = np.max(c)
    return float(cm + np.log(np.sum(np.exp(alpha * (c - cm)))) / alpha)

class _Obj:
    def __init__(self, n, alpha):
        self.n = n
        self.alpha = alpha
    def __call__(self, x):
        return smooth_obj(x, self.n, self.alpha)

class _Con:
    def __init__(self, S):
        self.S = S
    def __call__(self, x):
        return float(np.sum(x) - self.S)
    def jac(self, x):
        return np.ones_like(x)

def do_slsqp(h_start, n, alpha, S, bounds, maxiter, ftol):
    obj = _Obj(n, alpha)
    con = _Con(S)
    cons = [{'type': 'eq', 'fun': con, 'jac': con.jac}]
    try:
        res = minimize(obj, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_start

def make_sinusoidal_start(n, S, freq):
    x = np.arange(n, dtype=float) / n
    h = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * x)
    return project_feasible(h, S)

def make_random_start(n, S, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    return project_feasible(h, S)
