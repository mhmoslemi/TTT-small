def smooth_c5(h, dx, n, beta=100):
    """Compute smooth max of correlation and its gradient."""
    one_minus_h = 1 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    # Log-sum-exp for numerical stability
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    log_sum = max_corr + (1.0/beta) * np.log(np.exp(shifted).sum())
    
    # Softmax weights
    weights = np.exp(shifted)
    weights /= weights.sum()

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-10:
        h[:] = 0.5
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # Fine adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute the difference among non-saturated entries
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_smooth_c5(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    weights = np.exp(shifted)
    ws = weights.sum()
    if ws < 1e-300:
        weights = np.zeros(2*n-1)
        weights[int(np.argmax(corr))] = 1.0

            if m > 0:

                grad[:m] -= w * h[la:]

    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, n, beta):
    """Gradient of smooth_max(corr) * dx."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    sh = beta * (corr - mc)
    w = np.exp(sh)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(2*n-1)
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n-1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                g[:m] += wk * (1.0 - h[lag:])
                g[lag:] -= wk * h[:m]
        else:
            la = -lag
            m = n - la
            if m > 0:
                g[la:] += wk * (1.0 - h[:m])
                g[:m] -= wk * h[la:]
    return g * dx

        mom = mu * mom - step * g
        hn = project(h + mom, n)
        c = compute_c5(hn)

                mom[:] = 0
                step = 0.008
                ni = 0
    return bh, bc
