def project_onto_constraint(y, target_sum):
    """Project y onto {h: sum(h) = target_sum, 0 <= h <= 1}"""
    n = len(y)
    # Binary search on t such that sum(clip(y - t, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0  # t range where solution must be
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid

def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = np.max(corr)
    logits = T * (corr - m)
    logits -= np.max(logits)
    exp_l = np.exp(logits)
    w = exp_l / exp_l.sum()
    
    grad = np.zeros(n)

            g[-s:] += (1 - h[:n+s])
            g[:-s] += -h[-s:]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    logits = T * (corr - m)
    logits -= np.max(logits)
    exp_l = np.exp(logits)
    w = exp_l / exp_l.sum()
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-13:
            continue
        s = k - (n - 1)
        if s >= 0:
            g = np.zeros(n)
            if n - s > 0:
                g[:n-s] += (1.0 - h[s:])
            if s < n:

        time_left = budget_s * 0.98 - (time.time() - t_start)
        if time_left > 5:
            h_bs, c5_bs = binary_local_search(best_h, time_limit=time_left)
            if c5_bs < best_c5:
                best_c5, best_h, best_n = c5_bs, h_bs, best_n

    return best_h, best_c5, best_n
