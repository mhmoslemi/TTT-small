import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h)) * dx)

def smooth_c5_and_grad(h, dx, alpha):
    n = len(h)
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(overlaps)
    shifted = overlaps - m
    exp_vals = np.exp(alpha * shifted)
    wsum = float(np.sum(exp_vals))
    obj = m + np.log(wsum) / alpha
    weights = exp_vals / wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n - s]) * dx
            grad[:n - s] -= w * h[s:] * dx
        else:
            s2 = -s
            grad[:n - s2] += w * (1.0 - h[s2:]) * dx
            grad[s2:] -= w * h[:n - s2] * dx
    return float(obj), grad

def fix_sum(h, n):
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        room = 1.0 - h
        mask = room > 1e-8
        if mask.sum() == 0:
            return h
        h[mask] += diff * room[mask] / room[mask].sum()
    else:
        mask = h > 1e-8
        if mask.sum() == 0:
            return h
        h[mask] += diff * h[mask] / h[mask].sum()
    return np.clip(h, 0.0, 1.0)

def optimize_slsqp(h, n, dx, alpha, max_iter=300):
    constraints = [{'type': 'eq',
                    'fun': lambda x: np.sum(x) - n / 2.0,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def fun_and_grad(x):
        return smooth_c5_and_grad(x, dx, alpha)

    try:
        result = minimize(fun_and_grad, h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': max_iter, 'ftol': 1e-10})
        return np.clip(result.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def optimize_continue(h, n, dx, alphas=None, max_iter_per=200, time_check=None):
    if alphas is None:
        alphas = [20, 50, 100, 300]
    h_curr = h.copy()
    best_local = h_curr.copy()
    best_c5_local = compute_c5(h_curr)
    for alpha in alphas:
        if time_check is not None and time_check() < 30:
            break
        h_opt = optimize_slsqp(h_curr, n, dx, alpha, max_iter=max_iter_per)
        h_opt = fix_sum(h_opt, n)
        c5_val = compute_c5(h_opt)
        if c5_val < best_c5_local:
            best_c5_local = c5_val
            best_local = h_opt.copy()
        h_curr = h_opt.copy()
    return best_local, best_c5_local

def try_exchange_moves(h, n, dx, rng, max_iters=300, time_check=None):
    h_try = h.copy()
    best_c5 = compute_c5(h_try)
    for _ in range(max_iters):
        if time_check is not None and time_check() < 5:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Swap: set h[i]=1-h[j], h[j]=1-h[i] preserves sum
        new_i = 1.0 - h_try[j]
        new_j = 1.0 - h_try[i]
        if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
            continue
        orig_i, orig_j = h_try[i], h_try[j]
        h_try[i], h_try[j] = new_i, new_j
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
        else:
            h_try[i], h_try[j] = orig_i, orig_j
    return h_try, best_c5
