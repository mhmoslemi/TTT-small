import numpy as np
import time

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-10:
        idx = np.argmin(np.abs(hp - 0.5))
        hp[idx] = np.clip(hp[idx] + diff, 0.0, 1.0)
    return hp

def objective_and_grad(h, n, beta):
    ov = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    m = np.max(ov)
    shifted = ov - m
    exps = np.exp(beta * shifted)
    logsum = m + np.log(np.sum(exps)) / beta
    weights = exps / np.sum(exps)
    
    # Compute gradient
    grad = np.zeros(n)
    # Correlation index j corresponds to lag k = j - (n-1)
    # For lag k: |k| = |j - (n-1)|
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-12:
            continue
        lag = abs(j - (n - 1))
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[:n-lag] += w * (1.0 - h[lag:])
            grad[lag:] += w * (-h[:n-lag])
    return logsum, grad

    # Optimize function for a given h

        for beta in [5.0, 10.0, 20.0, 50.0, 100.0]:
            if not time_ok(time_frac):
                break
            lr = 0.05
            best_obj = objective_and_grad(h, n, beta)[0]
            for it in range(max_iter):

                obj, grad = objective_and_grad(h, n, beta)
                if obj < best_obj:

                h_new = h - lr * grad
                h_new = project_feasible(h_new, n)
                obj_new, _ = objective_and_grad(h_new, n, beta)

                    lr = min(lr * 1.08, 0.2)
                else:

import numpy as np
import time

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-12:
        idx = np.argmax(np.minimum(hp, 1.0 - hp))
        hp[idx] = np.clip(hp[idx] + diff, 0.0, 1.0)
    return hp

def obj_and_grad(h, n, beta):
    ov = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    m = np.max(ov)
    ex = np.exp(beta * (ov - m))
    Z = np.sum(ex)
    obj = m + np.log(Z) / beta
    p = ex / Z
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = p[j]
        if w < 1e-14:
            continue
        lag = abs(j - (n - 1))
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] += w * (-h[:n - lag])
    return obj, grad

def gradient_descent(h, n, t0, budget_s, frac=0.6, max_iter=4000):
    for beta in [5.0, 15.0, 40.0, 100.0]:
        if (time.time() - t0) > budget_s * frac:
            break
        lr = 0.03
        for it in range(max_iter):
            if (time.time() - t0) > budget_s * frac:
                break
            obj, grad = obj_and_grad(h, n, beta)
            h_new = project_feasible(h - lr * grad, n)
            obj_new, _ = obj_and_grad(h_new, n, beta)
            if obj_new < obj:
                h = h_new
                lr = min(lr * 1.06, 0.15)
            else:
                lr *= 0.75
            if lr < 1e-10:
                break
    return h

def swap_search(h, n, t0, budget_s, frac=0.95, max_rounds=200):
    c = evaluate_c5(h, n)
    for _ in range(max_rounds):
        if (time.time() - t0) > budget_s * frac:
            break
        improved = False
        idx1 = np.argsort(h)[:min(10, n // 2)]
        idx2 = np.argsort(h)[-min(10, n // 2):]
        for i in idx1:
            for j in idx2:
                if i == j:
                    continue
                h[i], h[j] = h[j], h[i]
                h = project_feasible(h, n)
                c_new = evaluate_c5(h, n)
                if c_new < c - 1e-12:
                    c = c_new
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]
                    h = project_feasible(h, n)
        if not improved:
            break
    return h, c

def binarize(h, n):
    ones = np.sum(h >= 0.5)
    h_bin = (h >= 0.5).astype(float)
    if ones != n // 2:
        h_bin = np.sort(h_bin)[::-1]
        order = np.argsort(-h)
        h_bin = np.zeros(n)
        h_bin[order[:n // 2]] = 1.0
    return project_feasible(h_bin, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals():
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 48
        h_best = np.zeros(n)
        h_best[:n // 2] = 1.0
        h_best = project_feasible(h_best, n)

    n_best = len(h_best)
    c_best = evaluate_c5(h_best, n_best)

    # Phase 1: Direct optimization at best n
    h_opt = gradient_descent(h_best.copy(), n_best, t0, budget_s, frac=0.35)
    c = evaluate_c5(h_opt, n_best)
    if c < c_best:
        h_best, c_best = h_opt, c

    # Phase 2: Try other n values
    for n_try in [48, 50, 54, 64, 80, 96, 120]:
        if (time.time() - t0) > budget_s * 0.55:
            break
        x_old = np.linspace(0, 2, n_best)
        x_new = np.linspace(0, 2, n_try)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_feasible(h_try, n_try)
        h_try = gradient_descent(h_try, n_try, t0, budget_s, frac=0.55, max_iter=3000)
        c_try = evaluate_c5(h_try, n_try)
        if c_try < c_best:
            h_best, n_best, c_best = h_try, n_try, c_try

    # Phase 3: Random restarts
    for _ in range(200):
        if (time.time() - t0) > budget_s * 0.85:
            break
        h_try = h_best + rng.randn(n_best) * 0.15
        h_try = project_feasible(h_try, n_best)
        h_try = gradient_descent(h_try, n_best, t0, budget_s, frac=0.85, max_iter=2000)
        c_try = evaluate_c5(h_try, n_best)
        if c_try < c_best:
            h_best, c_best = h_try, c_try

    # Phase 4: Swap-based local search

    # Phase 5: Binarization + re-optimize (for even n)

        c_bin = evaluate_c5(h_bin, n_best)
        if c_bin < c_best:
            h_best, c_best = h_bin, c_bin

    return h_best, c_best, n_best
