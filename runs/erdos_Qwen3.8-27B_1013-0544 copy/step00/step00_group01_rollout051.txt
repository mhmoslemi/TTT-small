    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    # Shift for numerical stability
    m = np.max(corr)

    h = project_feasible(h.copy(), target)
    best_c5 = true_c5(h)
    best_h = h.copy()
    T_list = [1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0]
    for it in range(n_iter):
        if time.time() > time_limit:
            break
        T = T_list[min(it // max(1, n_iter // len(T_list)), len(T_list) - 1)]
        grad = smooth_gradient(h, T)
        step = 0.05 / (1.0 + 0.002 * it)
        alpha = step
        c5_cur = true_c5(h)
        for _ in range(8):
            h_new = project_feasible(h - alpha * grad, target)
            c5_new = true_c5(h_new)
            if c5_new < c5_cur:
                break
            alpha *= 0.5
        h = project_feasible(h - alpha * grad, target)
        c5_check = true_c5(h)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = h.copy()
    return best_h, best_c5

def pair_local_search(h, n_trials, time_limit, rng_seed=7):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    c5_cur = true_c5(h)
    rng = np.random.default_rng(rng_seed)
    for _ in range(n_trials):
        if time.time() > time_limit:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = rng.uniform(-0.08, 0.08)
        if h[i] + delta < -1e-9 or h[i] + delta > 1 + 1e-9:
            continue
        if h[j] - delta < -1e-9 or h[j] - delta > 1 + 1e-9:
            continue
        h[i] += delta
        h[j] -= delta
        c5_new = true_c5(h)
        if c5_new < c5_cur - 1e-14:
            c5_cur = c5_new
        else:
            h[i] -= delta
            h[j] += delta
    return h, c5_cur

def systematic_sweep(h, time_limit):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    c5_cur = true_c5(h)
    for ss in [0.02, 0.01, 0.005, 0.002, 0.001]:
        for i in range(n):
            if time.time() > time_limit:
                return h, c5_cur
            for j in range(n):
                if i == j:
                    continue
                for sign in [1, -1]:
                    d = sign * ss
                    if h[i] + d < -1e-9 or h[i] + d > 1 + 1e-9:
                        continue
                    if h[j] - d < -1e-9 or h[j] - d > 1 + 1e-9:
                        continue
                    h[i] += d
                    h[j] -= d
                    c5_new = true_c5(h)
                    if c5_new < c5_cur - 1e-14:
                        c5_cur = c5_new
                    else:
                        h[i] -= d
                        h[j] += d
    return h, c5_cur

def gradient_local_search(h, n_iters, time_limit):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = true_c5(h)
    best_h = h.copy()
    for it in range(n_iters):
        if time.time() > time_limit:
            break
        corr = compute_corr(h)
        k_max = int(np.argmax(corr)) - (n - 1)
        grad = subgrad_at(h, k_max)
        grad_c = grad - np.mean(grad)
        d_norm = np.max(np.abs(grad_c))
        if d_norm < 1e-12:
            break
        d = -grad_c / d_norm
        best_alpha = 0.0
        c5_cur = true_c5(h)
        for a in [0.0005, 0.001, 0.002, 0.005, 0.01]:
            h_new = project_feasible(h + a * d, target)
            c5_new = true_c5(h_new)
            if c5_new < c5_cur:
                best_alpha = a
                c5_cur = c5_new
        if best_alpha > 0:
            h = project_feasible(h + best_alpha * d, target)
            c5_now = true_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    time_limit = t_start + budget_s * 0.96

    best_h = None
    best_c5 = float('inf')

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.ones(47) * 0.5

    n0 = len(h0)
    h0 = project_feasible(h0, n0 / 2.0)
    c5_init = true_c5(h0)
    best_c5 = c5_init
    best_h = h0.copy()

    # Phase 1: Smooth gradient optimization
    h_opt, c5_opt = optimize_smoothed(best_h, n_iter=30000, time_limit=time_limit)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt.copy()

    # Phase 2: Gradient local search
    h_opt, c5_opt = gradient_local_search(best_h, n_iters=10000, time_limit=time_limit)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt.copy()

    # Phase 3: Random pair local search
    h_opt, c5_opt = pair_local_search(best_h, n_trials=300000, time_limit=time_limit, rng_seed=seed)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt.copy()

    # Phase 4: Systematic sweep
    h_opt, c5_opt = systematic_sweep(best_h, time_limit=time_limit)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt.copy()

    # Phase 5: More smooth gradient from improved point
    h_opt, c5_opt = optimize_smoothed(best_h, n_iter=20000, time_limit=time_limit)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt.copy()
    h_opt, c5_opt = pair_local_search(best_h, n_trials=200000, time_limit=time_limit, rng_seed=seed+1)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt.copy()

    # Phase 6: Restarts with noise
    for r in range(8):
        if time.time() > time_limit:
            break
        noise_scale = 0.03 / (r + 1)
        h_trial = best_h + np.random.randn(n0) * noise_scale
        h_trial = project_feasible(h_trial, n0 / 2.0)
        h_opt, c5_opt = optimize_smoothed(h_trial, n_iter=8000, time_limit=time_limit)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()
        h_opt, c5_opt = pair_local_search(best_h, n_trials=100000, time_limit=time_limit, rng_seed=seed + r + 10)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()

    # Phase 7: Final systematic sweep + local search
    h_opt, c5_opt = systematic_sweep(best_h, time_limit=time_limit)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt.copy()
    h_opt, c5_opt = pair_local_search(best_h, n_trials=200000, time_limit=time_limit, rng_seed=seed+99)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt.copy()
    h_opt, c5_opt = gradient_local_search(best_h, n_iters=5000, time_limit=time_limit)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt.copy()

    # Final: ensure feasibility and recompute
    best_h = project_feasible(best_h, n0 / 2.0)
    best_c5 = true_c5(best_h)

    return best_h, best_c5, len(best_h)
