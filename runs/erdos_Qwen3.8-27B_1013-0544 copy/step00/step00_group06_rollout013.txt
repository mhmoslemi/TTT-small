```python
def smooth_c5_grad(h, beta, L):
    dx = 2.0 / L
    c = np.correlate(h, 1-h, mode='full')  # length 2L-1
    c_scaled = c * dx
    log_w = beta * c_scaled - logsumexp(beta * c_scaled)
    weights = np.exp(log_w)  # indexed by k, k=0,...,2L-2
    
    grad = np.zeros(L)
    for s in range(-(L-1), L):
        k = L - 1 - s
        w = weights[k] * dx
        # term1: (1-h[j+s]) for valid j+s
        # term2: h[j-s] for valid j-s
        if s >= 0:
            # j+s valid: j+s < L → j < L-s, and j+s >= 0 → j >= -s (always true since j>=0)
            # So j ranges from 0 to L-s-1
            js = np.arange(0, L-s)
            grad[js] += w * (1.0 - h[js + s])
            # j-s valid: j-s >= 0 → j >= s, j-s < L → j < L+s (always true)
            # So j ranges from s to L-1
            js2 = np.arange(s, L)
            grad[js2] -= w * h[js2 - s]
        else:
            ss = -s
            # j+s = j-ss valid: j-ss >= 0 → j >= ss, j-ss < L → j < L+ss (always true)
            # So j ranges from ss to L-1
            js = np.arange(ss, L)
            grad[js] += w * (1.0 - h[js - ss])
            # j-s = j+ss valid: j+ss < L → j < L-ss, j+ss >= 0 (always true)
            # So j ranges from 0 to L-ss-1
            js2 = np.arange(0, L-ss)
            grad[js2] -= w * h[js2 + ss]
    return grad
```
