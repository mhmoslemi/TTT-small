def subgradient_c5(h, n_points):
    """Compute subgradient of C5 at h. Returns (gradient, c5_value)."""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    
    # Active shifts (those achieving the max)
    active_mask = corr >= max_val - 1e-12
    active_indices = np.where(active_mask)[0]
    
    grad = np.zeros(n_points)
    for k in active_indices:
        s = k - (n_points - 1)
        if s == 0:
            grad += (1 - 2 * h)
        elif s > 0:
            # First term: j in [s, n-1], value = 1-h[j-s]
            grad[s:] += (1 - h[:-s])
            # Second term: j in [0, n-s-1], value = -h[j+s]
            grad[:n-s] += (-h[s:])
        else:
            p = -s
            # First term: j in [0, n-p-1], value = 1-h[j+p]
            grad[:n-p] += (1 - h[p:])
            # Second term: j in [p, n-1], value = -h[j-p]
            grad[p:] += (-h[:p])
    
    n_active = len(active_indices)
    grad /= n_active
    return grad, max_val * dx

    lo = -max(h) - 1
    hi = 1 - min(h) + 1

        h_shifted = np.clip(h + mid, 0, 1)

            hi = mid

        if diff > 0:

            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def subgrad(h, n):
    """Subgradient of max_k corr[k] at h; returns (grad, max_corr)"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = float(np.max(corr))
    active = np.where(corr >= mx - 1e-10)[0]
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        if s == 0:
            grad += (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += (1.0 - h[:-s])
            grad[:n - s] += (-h[s:])
        else:
            p = -s
            grad[:n - p] += (1.0 - h[p:])
            grad[p:] += (-h[:p])
    return grad / len(active), mx

def pgd(h0, n, max_iter=50000, lr0=0.5, tlimit=60.0):
    """Projected subgradient descent to minimize C5"""
    t0 = time.time()
    dx = 2.0 / n
    h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    stagnant = 0
    for it in range(max_iter):
        if time.time() - t0 > tlimit:
            break
        g, mx = subgrad(h, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        h_new = project_feasible(h - lr * g * dx, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            stagnant = 0
        else:
            stagnant += 1
        if c5_new <= mx * dx:
            h = h_new
            if stagnant < 500:
                lr = min(lr * 1.08, 2.0)
            else:
                lr = lr * 0.95
        else:
            lr *= 0.6
        if lr < 1e-14:
            break
        if stagnant > 5000:
            break
    return best_h, best_c5

def smooth_pgd(h0, n, max_iter=20000, tau=40.0, lr0=0.3, tlimit=60.0):
    """PGD on smooth max approximation"""
    t0 = time.time()
    dx = 2.0 / n
    h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    for it in range(max_iter):
        if time.time() - t0 > tlimit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        mx = np.max(corr)
        shifted = tau * (corr - mx)
        max_shifted = np.max(shifted)
        weights = np.exp(shifted - max_shifted)
        weights /= np.sum(weights)
        grad = np.zeros(n)
        for k in range(len(corr)):
            if weights[k] < 1e-10:
                continue
            s = k - (n - 1)
            if s == 0:
                g = (1.0 - 2.0 * h)
            elif s > 0:
                g = np.zeros(n)
                g[s:] = (1.0 - h[:-s])
                g[:n - s] += (-h[s:])
            else:
                p = -s
                g = np.zeros(n)
                g[:n - p] = (1.0 - h[p:])
                g[p:] += (-h[:p])
            grad += weights[k] * g
        h_new = project_feasible(h - lr * grad * dx, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        obj_val = mx + np.log(np.sum(weights)) / tau
        if c5_new <= obj_val * 1.01:
            h = h_new
            lr = min(lr * 1.05, 1.0)
        else:
            lr *= 0.7
        if lr < 1e-14:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.full(70, 0.5)

    n = len(h0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: PGD from initial
    tl = min(budget_s * 0.30, 400.0)
    h1, c5_1 = pgd(h0, n, max_iter=100000, lr0=1.0, tlimit=tl)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Smooth PGD refinement
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 20:
        tl2 = min(remaining * 0.25, 150.0)
        h2, c5_2 = smooth_pgd(best_h, n, max_iter=30000, tau=50.0,
                              lr0=0.3, tlimit=tl2)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2
        # PGD polish after smooth
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed
        if remaining > 15:
            h2b, c5_2b = pgd(best_h, n, max_iter=50000, lr0=0.8,
                            tlimit=min(remaining * 0.3, 100.0))
            if c5_2b < best_c5:
                best_c5, best_h = c5_2b, h2b

    # Phase 3: Random restarts
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 30:
        n_restarts = max(3, min(12, int(remaining / 25)))
        for i in range(n_restarts):
            if time.time() - t_start > budget_s - 15:
                break
            frac = i / max(n_restarts, 1)
            scale = 0.2 * (1.0 - frac) + 0.05
            h_pert = project_feasible(best_h + rng.normal(0, scale, n), n)
            tl_r = max(5.0, min(20.0, (budget_s - (time.time() - t_start) - 10) / max(n_restarts - i, 1)))
            h_r, c5_r = pgd(h_pert, n, max_iter=20000, lr0=0.5, tlimit=tl_r)
            if c5_r < best_c5:
                best_c5, best_h = c5_r, h_r

    # Phase 4: Try different n_values
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    n_candidates = [50, 60, 80, 100, 120, 140, 160, 200]
    for n_try in n_candidates:
        if time.time() - t_start > budget_s - 25:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        tl_n = max(5.0, min(20.0, (budget_s - (time.time() - t_start) - 10) / 2.0))
        h_n, c5_n = pgd(h_int, n_try, max_iter=30000, lr0=0.5, tlimit=tl_n)
        if c5_n < best_c5:
            best_c5, best_h, n = c5_n, h_n, n_try

    # Phase 5: Final polish
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 10:
        h_f, c5_f = pgd(best_h, n, max_iter=30000, lr0=0.3,
                        tlimit=min(remaining - 5, 60.0))
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
