import numpy as np
import time

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        h = np.clip(h + (target_sum - s) / len(h), 0.0, 1.0)
    return h

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x_new, x_old, h_old, period=1.0)

def run_sa(h0, n, time_limit, rng, t_init=20.0, alpha=0.999, max_reheats=5):
    h = project(h0, n / 2.0)
    c5_curr = evaluate_c5(h, n)
    h_best = h.copy()
    c5_best = c5_curr
    T = t_init
    t0 = time.time()
    stall = 0
    reheats = 0

    while True:
        if time.time() - t0 > time_limit:
            break
        if T < 0.0005:
            if reheats < max_reheats:
                T = t_init * (0.5 ** reheats)
                h = h_best.copy()
                c5_curr = c5_best
                stall = 0
                reheats += 1
            else:
                break
        if stall > 20000:
            break

        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_d = min(h[src], 1.0 - h[dst])
        if max_d < 1e-12:
            continue

        delta = rng.random() * min(max_d, 0.4)
        h[src] -= delta
        h[dst] += delta
        c5_new = evaluate_c5(h, n)
        dc = c5_new - c5_curr

        if dc <= 0:
            c5_curr = c5_new
            stall = 0
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        else:
            if rng.random() < np.exp(-dc / max(T, 1e-10)):
                c5_curr = c5_new
            else:
                h[src] += delta
                h[dst] -= delta
            stall += 1
        T *= alpha

    return h_best, c5_best

def binary_snap(h, n):
    """Try rounding to binary with one fractional value to maintain sum."""
    target = n / 2.0
    h_bin = np.heaviside(h - 0.5, 0.0)
    s = h_bin.sum()
    if abs(s - target) < 1e-12:
        return h_bin.copy()
    # Need to adjust: find the element closest to 0.5 in the original h
    # to convert to fractional
    if s < target:
        # Need more sum: find a 0 closest to 0.5 in h, make it fractional
        zeros = np.where(h_bin == 0)[0]
        if len(zeros) > 0:
            idx = zeros[np.argmax(h[zeros])]
            h_bin[idx] = target - s
    else:
        # Need less sum: find a 1 closest to 0 in h, make it fractional
        ones = np.where(h_bin == 1)[0]
        if len(ones) > 0:
            idx = ones[np.argmin(h[ones])]
            h_bin[idx] = 1.0 + target - s
    return h_bin
