        o = 0.0
    return _sig(z + o)

def _h2z(h):
    hc = np.clip(h, 1e-8, 1.0 - 1e-8)
    return np.log(hc / (1.0 - hc))

def _optz(z0, n, tl):
    t0 = time.time()
    bh = _proj(_z2h(z0, n), n)
    bc = _c5(bh)
    for b in [8, 15, 30, 60, 120, 300, 600]:
        if time.time() - t0 > tl:
            break
        def obj(z):
            return _smax(_cv(_z2h(z, n)), b)
        try:
            res = minimize(obj, z0, method='L-BFGS-B',
                           options={'maxiter': 500, 'ftol': 1e-15, 'maxls': 50})
            h = _proj(_z2h(res.x, n), n)
            c = _c5(h)
            if c < bc:
                bc = c
                bh = h
                z0 = _h2z(h)
        except:
            pass
    return bh, bc

def _ups(h, nn):
    x = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, nn, endpoint=False)
    return _proj(np.interp(xn, x, h), nn)

def _make_box(n, frac=0.5):
    h = np.zeros(n)
    m = int(n * frac)
    h[:m] = 1.0 / frac
    return _proj(h, n)

def _make_sine(n, freq=1.0, amp=0.3):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.25 + amp * np.sin(2 * np.pi * freq * x)
    return _proj(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    ts = time.time()
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except:
        h0 = np.ones(45) * 0.5
    n0 = len(h0)
    h0 = _proj(h0, n0)
    bh, bc = h0.copy(), _c5(h0)

    # Phase 1: optimize initial
    z0 = _h2z(h0)
    tl = min(budget_s * 0.2, 120)
    h1, c1 = _optz(z0, n0, tl)
    if c1 < bc:
        bh, bc = h1, c1

    # Phase 2: try alternative starts at n0
    alts = [_make_box(n0), _make_sine(n0, 1.0, 0.3), _make_sine(n0, 2.0, 0.2)]
    for ha in alts:
        if time.time() - ts > budget_s * 0.35:
            break
        ha = _proj(ha, n0)
        ca = _c5(ha)
        if ca < bc:
            bh, bc = ha, ca
        za = _h2z(ha)
        rem = budget_s - (time.time() - ts)
        tl = min(rem * 0.08, 40)
        if tl < 5:
            break
        h2, c2 = _optz(za, n0, tl)
        if c2 < bc:
            bh, bc = h2, c2

    # Phase 3: multi-resolution upsample and optimize
    for fac in [2, 3, 4]:
        if time.time() - ts > budget_s * 0.7:
            break
        nn = n0 * fac
        if nn > 500:
            break
        rem = budget_s - (time.time() - ts)
        if rem < 30:
            break
        hu = _ups(bh, nn)
        zu = _h2z(hu)
        tl = min(rem * 0.15, 120)
        hn, cn = _optz(zu, nn, tl)
        if cn < bc:
            bh, bc = hn, cn

    # Phase 4: random restarts at best resolution
    nb = len(bh)
    perts = [0.05, 0.1, 0.15, 0.2]
    for p in perts:
        for _ in range(5):
            if time.time() - ts > budget_s * 0.92:
                break
            hr = _proj(bh + rng.randn(nb) * p, nb)
            rem = budget_s - (time.time() - ts)
            if rem < 10:
                break
            tl = min(rem * 0.04, 25)
            if tl < 5:
                break
            h2, c2 = _optz(_h2z(hr), nb, tl)
            if c2 < bc:
                bh, bc = h2, c2

    # Final polish at best resolution with higher beta
    if time.time() - ts < budget_s * 0.95:
        rem = budget_s - (time.time() - ts)
        if rem > 15:
            zh = _h2z(bh)
            def _polish(z, n, tl):
                t0 = time.time()
                bh2 = _proj(_z2h(z, n), n)
                bc2 = _c5(bh2)
                for b in [500, 1000, 2000]:
                    if time.time() - t0 > tl:
                        break
                    def obj2(zz):
                        return _smax(_cv(_z2h(zz, n)), b)
                    try:
                        res = minimize(obj2, z, method='L-BFGS-B',
                                       options={'maxiter': 300, 'ftol': 1e-16})
                        h3 = _proj(_z2h(res.x, n), n)
                        c3 = _c5(h3)
                        if c3 < bc2:
                            bc2 = c3
                            bh2 = h3
                            z = _h2z(h3)
                    except:
                        pass
                return bh2, bc2
            hp, cp = _polish(zh, nb, rem)
            if cp < bc:
                bh, bc = hp, cp

    # Final validation
    bh = np.clip(bh, 0.0, 1.0)
    nb = len(bh)
    diff = nb * 0.5 - bh.sum()
    if abs(diff) > 1e-10:
        m = (bh > 0) & (bh < 1)
        if m.any():
            bh[m] = np.clip(bh[m] + diff / m.sum(), 0.0, 1.0)
    bc = _c5(bh)
    try:
        cea = evaluate_erdos_solution(bh)
        if cea is not None and cea < bc:
            bc = cea
    except:
        pass
    return bh, bc, nb
