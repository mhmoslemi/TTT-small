For the correlation at a specific lag, the gradient w.r.t. h[j] can be computed analytically.

Let me think about this more carefully.

`np.correlate(h, 1-h, mode="full")` gives output `c` of length 2n-1.

For the "full" mode, the correlation is:
c[m] = sum_i h[i] * (1-h[i + m - (n-1)])

where the sum is over all valid i such that both i and i + m - (n-1) are in [0, n-1].

Let's define lag = m - (n-1), so lag ranges from -(n-1) to n-1.

c[m] = sum_i h[i] * (1-h[i+lag]) for valid i

The gradient w.r.t. h[j]:
dc[m]/dh[j] = sum_i [delta(i,j)*(1-h[i+lag]) + h[i]*(-delta(i+lag,j))]
= (1-h[j+lag]) * I(j+lag in [0,n-1]) - h[j-lag] * I(j-lag in [0,n-1])

So the subgradient of C5 = max_m c[m] * dx is:
grad[j] = dx * [(1-h[j+lag*]) - h[j-lag*]] (with boundary terms being 0)

where lag* corresponds to the argmax.

This is a subgradient (not necessarily the exact gradient when there are ties), but it should work for descent.

```python
import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    
    grad = np.zeros(n)
    # grad[j] = dx * [(1-h[j+lag]) - h[j-lag]] with boundary checks
    for j in range(n):
        val = 0.0
        if 0 <= j + lag <= n - 1:
            val += 1.0 - h[j + lag]
        if 0 <= j - lag <= n - 1:
            val -= h[j - lag]
        grad[j] = dx * val
    return grad, k_star

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        if not m.any():
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize_gradient(h0, n, budget_s, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c = eval_c5(h)
    
    lr = 0.1
    no_improve = 0
    
    while time.time() - start_time < budget_s - 5:
        grad, _ = subgradient(h)
        h_new = h - lr * grad
        h_new = project(h_new, n)
        c_new = eval_c5(h_new)
        
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
            no_improve = 0
            h = h_new
        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.9
                no_improve = 0
            h = project(h - lr * 0.1 * np.random.randn(n), n)
    
    return best_h, best_c

def binary_local_search(h0, n, budget_s, start_time, rng):
    """Optimize a binary (or near-binary) solution using swaps"""
    target = n / 2.0
    h = project(h0, n)
    
    # Threshold to make binary-ish
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    k = int(target)
    frac = target - k
    h_bin[sorted_idx[:k]] = 1.0
    if frac > 1e-10:
        h_bin[sorted_idx[k]] = frac
    
    current_c = eval_c5(h_bin)
    best_h = h_bin.copy()
    best_c = current_c
    
    ones_mask = h_bin >= 0.5
    zeros_mask = h_bin < 0.5
    ones = np.where(ones_mask)[0]
    zeros = np.where(zeros_mask)[0]
    
    no_improve = 0
    
    while time.time() - start_time < budget_s - 5:
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        if h_bin[i] == h_bin[j]:
            continue
        
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        c_new = eval_c5(h_bin)
        
        if c_new < best_c:
            best_h = h_bin.copy()
            best_c = c_new
            no_improve = 0
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            no_improve += 1
        
        if no_improve > 500:
            # Perturb: do a random swap even if worse
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            c_new = eval_c5(h_bin)
            if c_new < best_c:
                best_h = h_bin.copy()
                best_c = c_new
            no_improve = 0
        
        ones = np.where(h_bin >= 0.5)[0]
        zeros = np.where(h_bin < 0.5)[0]
    
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
    else:
        n_init = 48
        h_init = np.ones(n_init) * 0.5
    
    best_h = h_init.copy()
    best_c = eval_c5(h_init)
    
    # Strategy 1: Gradient optimization on the initial n
    elapsed = time.time() - start_time
    if budget_s - elapsed > 100:
        sub_budget = min(200, budget_s - elapsed - 50)
        h_g, c_g = optimize_gradient(h_init, n_init, sub_budget, start_time + elapsed)
        if c_g < best_c:
            best_h = h_g
            best_c = c_g
    
    # Strategy 2: Try binary local search for various even n
    for n in [48, 50, 52, 54, 56, 58, 60, 64, 66, 68, 70, 72, 74, 76, 80]:
        elapsed = time.time() - start_time
        if budget_s - elapsed < 30:
            break
        
        # Initialize from parent (interpolate)
        h_n = np.interp(
            np.linspace(0, 2, n, endpoint=False),
            np.linspace(0, 2, n_init, endpoint=False),
            h_init
        )
        h_n = project(h_n, n)
        
        sub_budget = min(60, budget_s - elapsed - 10)
        h_b, c_b = binary_local_search(h_n, n, sub_budget, start_time + elapsed, rng)
        if c_b < best_c:
            best_h = h_b
            best_c = c_b
    
    # Strategy 3: More aggressive gradient on best found so far
    elapsed = time.time() - start_time
    if budget_s - elapsed > 100:
        sub_budget = budget_s - elapsed - 10
        h_g, c_g = optimize_gradient(best_h, len(best_h), sub_budget, start_time + elapsed)
        if c_g < best_c:
            best_h = h_g
            best_c = c_g
    
    # Final: verify and return
    best_h = project(best_h, len(best_h))
    final_c = eval_c5(best_h)
    
    return best_h, final_c, len(best_h)
```

```python
import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def subgrad_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    # Subgradient: for the active constraint (max lag k_star)
    # corr[k_star] = sum_i h[i] * (1-h[i+lag]) for valid i
    # d corr[k_star] / d h[j] = (1-h[j+lag]) - h[j-lag] (with boundary handling)
    grad = np.zeros(n)
    for j in range(n):
        if 0 <= j + lag < n:
            grad[j] += (1.0 - h[j + lag])
        if 0 <= j - lag < n:
            grad[j] -= h[j - lag]
    return grad * dx, k_star

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        if not m.any():
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_subgrad(h):
    """Average subgradients over top few lags for smoother descent"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Get top lags
    top_k = np.argsort(corr)[-5:]
    grad = np.zeros(n)
    for k in top_k:
        lag = int(k) - (n - 1)
        g = np.zeros(n)
        for j in range(n):
            if 0 <= j + lag < n:
                g[j] += (1.0 - h[j + lag])
            if 0 <= j - lag < n:
                g[j] -= h[j - lag]
        grad += g
    return grad * dx / len(top_k)
```

```python
import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_subgrad(h, k_star):
    n = len(h)
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n-lag] += (1.0 - h[lag:n])
        grad[lag:n] -= h[0:n-lag]
    elif lag < 0:
        l = -lag
        grad[l:n] += (1.0 - h[0:n-l])
        grad[0:n-l] -= h[l:n]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        if not m.any():
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize(h0, n, time_limit, rng):
    """Subgradient descent with perturbations"""
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c = eval_c5(h)
    lr = 0.05
    t_start = time.time()
    
    while time.time() - t_start < time_limit:
        # Compute subgradient
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        grad = compute_subgrad(h, k_star) * dx
        
        # Also get second active lag for smoothing
        corr_sorted = np.sort(corr)[::-1]
        if len(corr_sorted) >= 2 and 1.0 - corr_sorted[1] / corr_sorted[0] < 0.01:
            # Near tie, average subgradients
            top2 = np.argsort(corr)[::-1][:2]
            grad = (compute_subgrad(h, int(top2[0])) + compute_subgrad(h, int(top2[1])) * 0.5) * dx
        
        # Take step
        h_new = project_feasible(h - lr * grad, n)
        c_new = eval_c5(h_new)
        
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
            h = h_new
            # Decrease learning rate slightly
            lr *= 0.999
        else:
            # Perturb
            h = project_feasible(h + 0.05 * rng.standard_normal(n), n)
            lr = max(0.01, lr * 0.95)
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    elapsed = time.time() - start_time

        h_opt, c_opt = optimize(h_parent, len(h_parent), budget_s - elapsed - 100, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    
    # Phase 2: Try other n values with binary search
    elapsed = time.time() - start_time
    n_parent = len(h_parent)
    
    for n in [48, 50, 52, 56, 60, 64, 68, 72, 80, 96, 100, 120]:

        h_n = np.interp(x_new, x_parent, h_parent)
        h_n = project_feasible(h_n, n)

    # Final check
    best_h = project_feasible(best_h, len(best_h))
    final_c = eval_c5(best_h)
    
    return best_h, final_c, len(best_h)
```

```python
def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def subgrad(h, top_k=3):
    """Compute smoothed subgradient using top_k active lags"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top = np.argsort(corr)[::-1][:top_k]
    # Weight by how close each lag is to the max
    weights = np.zeros(len(top))
    for i, k in enumerate(top):
        weights[i] = max(0, (corr[k] - corr[top[0]] * 0.95) / (corr[top[0]] * 0.05 + 1e-10))
    if weights.sum() < 1e-10:
        weights[0] = 1.0
    else:
        weights = np.exp(-5 * (corr[top[0]] - corr[top]) / (corr[top[0]] + 1e-10))
    
    grad = np.zeros(n)
    for w, k in zip(weights, top):
        lag = int(k) - (n - 1)
        g = np.zeros(n)
        if lag > 0:
            g[:n-lag] += (1.0 - h[lag:n])
            g[lag:n] -= h[0:n-lag]
        elif lag < 0:
            g[-lag:n] += (1.0 - h[0:n+lag])
            g[0:n+lag] -= h[-lag:n]
        else:
            g[:] = 1.0 - 2.0 * h
        grad += w * g
    return grad * dx / weights.sum()

def project(h, n):

    target = n / 2.0
    for _ in range(100):

        if not m.any():

        h[m] += d / m.sum()

    return h
```

```python
        g[lag:n] -= h[:n - lag]
    elif lag < 0:

        g[l:n] += (1.0 - h[:n - l])

        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Phase 4: Pair swap polish
    elapsed = time.time() - t_start
    if budget_s - elapsed > 20:
        h_r, c_r = pair_swap_search(best_h, len(best_h), budget_s - elapsed - 5, rng)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Final: one more subgradient pass if time allows
    elapsed = time.time() - t_start
    if budget_s - elapsed > 15:
        h_r, c_r = optimize_subgrad(best_h, budget_s - elapsed - 3, rng)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    best_h = project(best_h, len(best_h))
    final_c = eval_c5(best_h)
    return best_h, final_c, len(best_h)
```
