        rng.shuffle(idx)
        h[idx[:int(s - target_sum)]] = 0.0
    elif s < target_sum:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:int(target_sum - s)]] = 1.0

def hill_climb_random(h, time_limit, rng):
    c5 = compute_c5(h)
    no_improve = 0
    while no_improve < 50000 and time.time() < time_limit:
        ones = np.where(h >= 1.0 - 1e-9)[0]
        zeros = np.where(h <= 1e-9)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        no_improve += 1
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        if new_c5 < c5 - 1e-14:
            c5 = new_c5
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h, c5

def hill_climb_systematic(h, time_limit):

def subgradient_refine(h, time_limit):
    n = len(h)
    target_sum = n / 2.0
    c5 = compute_c5(h)
    lr = 0.3
    stale = 0
    while time.time() < time_limit and stale < 200:
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode="full")
        k_max = int(np.argmax(corr))
        k = k_max - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            g = 0.0
            if 0 <= i - k < n:
                g += (1.0 - h[i - k])
            if 0 <= i + k < n:
                g -= h[i + k]
            grad[i] = g * dx
        h_new = project_constraint(h - lr * grad, target_sum)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5 - 1e-12:
            h = h_new
            c5 = new_c5
            stale = 0
        else:
            lr *= 0.8
            stale += 1
    return h, c5
