def project_onto_constraint(h, n):
    """Project h onto {0≤h≤1, sum(h)=n/2}."""
    target = n / 2.0

        return np.sum(proj) - target

    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if objective(mid) > 0:
            lo = mid
        else:
            hi = mid
