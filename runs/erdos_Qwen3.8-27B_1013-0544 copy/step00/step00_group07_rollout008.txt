import numpy as np
import time

def compute_c5_from_corr(corr, n):
    dx = 2.0 / n
    return np.max(corr) * dx

def objective(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary(n):
    h = np.zeros(n)
    indices = np.random.choice(n, n//2, replace=False)
    h[indices] = 1.0
    return h

def make_binary_from_frac(h, n):
    """Convert fractional h to binary with exactly n//2 ones."""
    # Sort by value, take top n//2
    h_copy = h.copy()
    if len(h_copy) != n:
        # Resize
        from scipy.interpolate import interp1d
        x_old = np.linspace(0, 2, len(h_copy))
        x_new = np.linspace(0, 2, n)
        h_copy = interp1d(x_old, h_copy, kind='linear')(x_new)
    
    # Take top n//2 positions
    threshold_idx = n - n//2
    sorted_idx = np.argsort(h_copy)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[threshold_idx:]] = 1.0
    return h_bin

def local_search(h, n, max_iter=100000):
    """Simple local search with swaps."""
    c5 = objective(h, n)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(max_iter):
        improved = False
        for _ in range(len(ones)):  # Try several random swaps per iteration
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            
            # Try swap
            h[i], h[j] = h[j], h[i]
            new_c5 = objective(h, n)
            if new_c5 < c5:
                c5 = new_c5
                improved = True
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                h[i], h[j] = h[j], h[i]
        
        if not improved:
            break
    
    return h, c5

def simulated_annealing(h, n, start_temp=0.01, end_temp=0.001, iterations=50000):
    """Simulated annealing with swaps."""
    c5 = objective(h, n)
    best_c5 = c5
    best_h = h.copy()
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(iterations):
        temp = start_temp * (end_temp / start_temp) ** (it / iterations)
        
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        h[i], h[j] = h[j], h[i]
        new_c5 = objective(h, n)
        
        delta = new_c5 - c5
        if delta < 0 or np.random.random() < np.exp(-delta / temp):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    best_c5 = float('inf')
    best_n = 0
    
    # Get initial values if available

    if 'initial_h_values' in globals():

        init_n = len(init_h)
    
    n_values = [80, 100, 120, 160, 200, 240, 320]
    if init_h is not None and init_n not in n_values:
        n_values.insert(0, init_n)
    
    for n in n_values:
        if time.time() - start_time > budget_s * 0.85:
            break
        
        time_per_n = (budget_s * 0.85 - (time.time() - start_time)) / max(1, len(n_values) - n_values.index(n))
        
        # Initialize
        if init_h is not None and n == len(init_h):
            h = make_binary_from_frac(init_h, n)
        else:
            h = random_binary(n)
        
        # Run local search
        time_limit = time_per_n * 0.4
        max_iter = int(time_limit * 5000)  # Rough estimate
        h, c5 = local_search(h, n, max_iter=max(1000, min(max_iter, 50000)))
        
        # Run simulated annealing
        if time.time() - start_time < budget_s * 0.85:
            time_limit_sa = (budget_s * 0.85 - (time.time() - start_time)) / max(1, len(n_values) - n_values.index(n))
            max_iter_sa = int(time_limit_sa * 3000)
            h, c5 = simulated_annealing(h, n, iterations=max(1000, min(max_iter_sa, 100000)))
        
        # Verify
        c5 = objective(h, n)
        assert abs(np.sum(h) - n/2) < 1e-10
        assert np.all((h >= 0) & (h <= 1))
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        best_c5 = np.max(corr) * (2.0 / best_n)

import numpy as np
import time

def compute_c5_binary(h, n):
    """Compute C5 for a binary h of length n."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def initial_block(n):
    """Create a block-structured initial solution."""
    h = np.zeros(n)
    # Alternating blocks of 1s and 0s
    block_size = n // 4  # Each block is n/4 long, giving n/2 total 1s
    for i in range(n):
        if (i // block_size) % 2 == 0:
            h[i] = 1.0
    # Adjust to have exactly n//2 ones
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        # Remove excess ones from the end
        excess = n_ones - n // 2
        one_indices = np.where(h == 1.0)[0]
        h[one_indices[-excess:]] = 0.0
    elif n_ones < n // 2:
        deficit = n // 2 - n_ones
        zero_indices = np.where(h == 0.0)[0]
        h[zero_indices[:deficit]] = 1.0
    return h

def greedy_local_search(h, n, max_passes=100):
    """Greedily find the best swap at each step."""
    c5 = compute_c5_binary(h, n)
    best_c5 = c5
    best_h = h.copy()
    
    for passing in range(max_passes):
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        best_delta = 0
        best_swap = None
        
        # Try all pairs (or a subset if too many)
        n_pairs = len(ones) * len(zeros)
        if n_pairs <= 10000:
            # Try all
            for i in ones:
                for j in zeros:
                    h[i], h[j] = h[j], h[i]
                    new_c5 = compute_c5_binary(h, n)
                    delta = best_c5 - new_c5
                    if delta > best_delta:
                        best_delta = delta
                        best_swap = (i, j)
                    h[i], h[j] = h[j], h[i]
        else:
            # Sample
            for _ in range(5000):
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                h[i], h[j] = h[j], h[i]
                new_c5 = compute_c5_binary(h, n)
                delta = best_c5 - new_c5
                if delta > best_delta:
                    best_delta = delta
                    best_swap = (i, j)
                h[i], h[j] = h[j], h[i]
        
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            best_c5 = compute_c5_binary(h, n)
            best_h = h.copy()
            improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

def simulated_annealing(h, n, iterations=20000, t_start=0.005, t_end=0.0001):
    """SA with single swaps."""
    c5 = compute_c5_binary(h, n)
    best_c5 = c5
    best_h = h.copy()
    
    ones = set(np.where(h == 1.0)[0].tolist())
    
    for it in range(iterations):
        t = t_start * (t_end / t_start) ** (it / iterations)
        
        i = ones.pop() if False else np.random.choice(list(ones))
        ones.add(i)  # put it back
        j = np.random.choice(np.setdiff1d(np.arange(n), list(ones)))
        
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5_binary(h, n)
        delta = new_c5 - c5
        
        if delta <= 0 or np.random.random() < np.exp(-delta / t):
            c5 = new_c5
            ones.discard(i)
            ones.add(j)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c5

import time

def c5_of(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def make_binary(h, n):
    """Ensure h is binary with exactly n//2 ones."""
    h = h.copy()
    n_ones = int(np.sum(h > 0.5))
    if n_ones > n // 2:
        idx = np.where(h > 0.5)[0]
        # Remove the weakest ones
        h[idx[np.argsort(h[idx])[:n_ones - n//2]]] = 0
    elif n_ones < n // 2:
        idx = np.where(h <= 0.5)[0]
        # Add the strongest zeros
        h[idx[np.argsort(-h[idx])[:n//2 - n_ones]]] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def random_binary(n):

    h[np.random.choice(n, n//2, replace=False)] = 1.0
    return h

For the full correlation c[k] where k ranges from 0 to 2n-2 (shift s = k - (n-1)):

Let me use a different notation. Let's define:
A[s] = sum_m h[m] * (1 - h[m+s]) for s = -(n-1), ..., n-1

When h[i] changes from 1 to 0:
- For A[s]: 
  - Term m=i: was h[i]*(1-h[i+s]) = 1-h[i+s], now 0. Change: -(1-h[i+s]) if i+s is valid
  - Term m such that m+s=i, i.e., m=i-s: was h[i-s]*(1-h[i]) = 0, now h[i-s]*1 = h[i-s]. Change: +h[i-s] if i-s is valid

When h[j] changes from 0 to 1:
- For A[s]:
  - Term m=j: was 0, now 1*(1-h[j+s]) = 1-h[j+s]. Change: +(1-h[j+s]) if j+s is valid
  - Term m such that m+s=j, i.e., m=j-s: was h[j-s]*(1-0) = h[j-s], now h[j-s]*0 = 0. Change: -h[j-s] if j-s is valid

So the total change for A[s] when swapping i (1→0) and j (0→1):
ΔA[s] = -(1-h[i+s]) + h[i-s] + (1-h[j+s]) - h[j-s]
(with terms being 0 if the index is out of range)
