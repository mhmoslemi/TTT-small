import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(c), c

def _project_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-10:
        return h
    # Bisection for the right shift
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0, 1)
        if h_proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = np.max(c)
    return c_max + np.log(np.sum(np.exp(beta * (c - c_max)))) / beta

            return _smooth_max_obj(h, dx, beta)

            c_max = np.max(c)
            weights = np.exp(beta * (c - c_max))
            weights = weights / weights.sum()
            # Gradient of c w.r.t. h
            N = len(h)
            grad_c = np.zeros((2*N - 1, N))

import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c)), c

def _project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _grad_correlation(h, n):
    J = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        shift = k - (n - 1)
        if shift >= 0:
            m = n - shift
            J[k, shift:] += (1.0 - h[:m])
            J[k, :m] -= h[shift:]
        else:
            s = -shift
            m = n - s
            J[k, :m] += (1.0 - h[s:])
            J[k, s:] -= h[:m]
    return J

def _slsqp_optimize(h0, n, dx, target_sum, beta, maxiter=200):
    def obj(h):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cm = np.max(c)
        return cm + np.log(np.sum(np.exp(beta * (c - cm)))) / beta
    
    def obj_grad(h):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cm = np.max(c)
        w = np.exp(beta * (c - cm))
        w /= w.sum()
        J = _grad_correlation(h, n)
        return (w @ J) * dx
    
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - target_sum,
             'jac': lambda h: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                  constraints=cons, jac=obj_grad,
                  options={'maxiter': maxiter, 'ftol': 1e-14})
    return res

def _local_search(h0, n, dx, target_sum, rng, n_iter=50000, time_limit=None, t0=None):
    best_h = h0.copy()
    best_c5, _ = _compute_c5(h0, n)
    
    for it in range(n_iter):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        # Perturb 1-3 random coordinates
        npert = int(rng.integers(1, 4))
        idx = rng.choice(n, size=npert, replace=False)
        h_trial = best_h.copy()
        for i in idx:

        h_trial = _project_feasible(h_trial, target_sum)
        c5_trial, _ = _compute_c5(h_trial, n)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
    
    return best_h, best_c5
