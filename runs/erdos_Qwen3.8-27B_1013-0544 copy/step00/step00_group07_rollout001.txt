    return m + np.log(np.sum(np.exp(shifted))) / alpha

def project_sum(h, target):
    h = np.clip(h, 0, 1)
    for _ in range(60):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        lo, hi = -1.1, 1.1
        for _ in range(80):
            mid = (lo + hi) / 2
            h_t = np.clip(h + mid, 0, 1)
            if np.sum(h_t) > target:
                hi = mid
            else:
                lo = mid
        h = np.clip(h + (lo + hi) / 2, 0, 1)
    return h

def coord_descent(h, n, max_sweeps=50):
    dx = 2.0 / n
    h = h.copy()
    lags = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            old_val = h[j]
            h_save = h[j]
            h[j] = 0.0
            c_base = np.correlate(h, 1 - h, mode="full") * dx
            h[j] = h_save

            jl = j + lags
            jlm = j - lags
            h_jl = np.where((jl >= 0) & (jl < n), h[np.clip(jl, 0, n - 1)], 0.0)
            h_jlm = np.where((jlm >= 0) & (jlm < n), h[np.clip(jlm, 0, n - 1)], 0.0)
            a = dx * (1.0 - h_jl - h_jlm)

            def g(v):
                return np.max(c_base + v * a)

            lo, hi = 0.0, 1.0
            for _ in range(80):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if g(m1) <= g(m2):
                    hi = m2
                else:
                    lo = m1
            bv = (lo + hi) / 2.0
            cands = [0.0, bv, 1.0, old_val]
            best_v = min(cands, key=lambda v: g(v))
            h[j] = best_v
            if abs(best_v - old_val) > 1e-8:
                improved = True
        if not improved:
            break
    return h

def optimize_slsqp(h_init, n, maxiter=300, alpha=200):
    dx = 2.0 / n
    target = n / 2.0

    def f(h):
        c = np.correlate(h, 1 - h, mode="full") * dx
        return smooth_max(c, alpha)

    def f_jac(h):
        c = np.correlate(h, 1 - h, mode="full") * dx
        m = c.max()
        shifted = alpha * (c - m)
        shifted = np.clip(shifted, -500, 500)
        w = np.exp(shifted)
        w = w / w.sum()
        # Numerical gradient as fallback
        eps = 1e-7
        grad = np.zeros(n)
        for j in range(n):
            h_p = h.copy()
            h_p[j] = min(1.0, h_p[j] + eps)
            c_p = np.correlate(h_p, 1 - h_p, mode="full") * dx
            m_p = c_p.max()
            sh_p = alpha * (c_p - m_p)
            sh_p = np.clip(sh_p, -500, 500)
            f_p = m_p + np.log(np.sum(np.exp(sh_p))) / alpha
            grad[j] = (f_p - f(h)) / eps
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n

    res = minimize(f, h_init, method='SLSQP', bounds=bounds,
                   constraints=cons, jac=f_jac,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return np.clip(res.x, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - start_time) - 5

    # Phase 1: SLSQP from initial
    if time_left() > 60:
        h_opt = optimize_slsqp(best_h, n0, maxiter=300, alpha=200)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    # Phase 2: Coord descent on best
    if time_left() > 30:
        h_cd = coord_descent(best_h, len(best_h), max_sweeps=40)
        c5 = compute_c5(h_cd)
        if c5 < best_c5:
            best_h, best_c5 = h_cd, c5

    # Phase 3: Try other n values
    for n_try in [60, 100, 120, 160]:
        if time_left() < 60:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_sum(h_interp, n_try / 2.0)

        if time_left() > 80:
            h_opt = optimize_slsqp(h_interp, n_try, maxiter=200, alpha=200)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5

        if time_left() > 30:
            h_cd = coord_descent(best_h, len(best_h), max_sweeps=30)
            c5 = compute_c5(h_cd)
            if c5 < best_c5:
                best_h, best_c5 = h_cd, c5

    # Phase 4: Random restarts for small n
    for n_try in [40, 60, 80]:
        for _ in range(3):
            if time_left() < 40:
                break
            h_rand = np.random.uniform(0, 1, n_try)
            h_rand = project_sum(h_rand, n_try / 2.0)
            if time_left() > 50:
                h_opt = optimize_slsqp(h_rand, n_try, maxiter=150, alpha=150)
                h_opt = coord_descent(h_opt, n_try, max_sweeps=20)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_h, best_c5 = h_opt, c5

    # Phase 5: Final polish
    if time_left() > 20:
        h_cd = coord_descent(best_h, len(best_h), max_sweeps=50)
        c5 = compute_c5(h_cd)
        if c5 < best_c5:
            best_h, best_c5 = h_cd, c5

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
