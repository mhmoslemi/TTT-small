import numpy as np
import time

def compute_c5(h):
    """Compute C5 for a given h array."""
    n = len(h)
    dx = 2.0 / n
    overlap = np.correlate(h, 1.0 - h, mode="full")
    return np.max(overlap) * dx

def local_search_binary(h, max_iters=1000):
    """Local search for binary h. Tries all swaps and picks the best improving one."""
    n = len(h)
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        best_gain = 0
        best_pair = None
        best_c5 = compute_c5(h)
        
        for a in ones:
            for b in zeros:
                # Compute new C5 after swap
                h[a], h[b] = h[b], h[a]  # swap
                new_c5 = compute_c5(h)
                h[a], h[b] = h[b], h[a]  # unswap
                gain = best_c5 - new_c5
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (a, b)
        
        if best_pair is not None:
            a, b = best_pair
            h[a], h[b] = h[b], h[a]
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            break
    
    return h
