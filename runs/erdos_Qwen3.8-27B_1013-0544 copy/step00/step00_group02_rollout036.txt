    return float(np.max(corr)) * dx

def project_sum(h, target_sum):
    """Project h to {x: sum(x)=target_sum, 0<=x<=1}"""
    h = h.copy()
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)

        lo = shift

            remaining -= 1.0
        elif remaining > 0:

            remaining = 0
            break
    
    return v

from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0, 1).copy()
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            h[m] += d / max(1, m.sum())
        else:
            m = h > 0.0
            h[m] += d / max(1, m.sum())
        h = np.clip(h, 0, 1)
    return h

def smooth_max(x, beta):
    m = x.max()
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def fw_step(h, n, target, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    shift = k_star - (n - 1)
    # Compute gradient of active component
    grad = np.zeros(n)
    if shift >= 0:
        grad[shift:] += (1.0 - h[:n-shift])
        grad[:n-shift] += (-h[shift:])
    else:
        s = -shift
        grad[:n-s] += (1.0 - h[s:])
        grad[s:] += (-h[:n-s])
    grad *= dx
    # Linear minimization over simplex
    order = np.argsort(-grad)
    v = np.zeros(n)
    rem = target
    for idx in order:
        if rem >= 1.0:
            v[idx] = 1.0
            rem -= 1.0
        elif rem > 0:
            v[idx] = rem
            break
    return v

            else:

                h_try = project_sum(h_try, target)
                if compute_c5(h_try, dx) < c5_best:

                            h = hp

                        else:

                            h[j] += s * step
        if not improved:
            break
    return h, best
