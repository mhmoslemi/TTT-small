    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def steepest_descent_binary(h, max_rounds=100):
    """Repeatedly try all single swaps until no improvement."""
    n = len(h)
    current = h.copy()
    current_c5 = compute_c5(current)
    
    for round_num in range(max_rounds):
        improved = False
        ones = list(np.where(current == 1)[0])
        zeros = list(np.where(current == 0)[0])

        best_move = None

            for j in zeros:
                current[i], current[j] = current[j], current[i]

            current[i], current[j] = current[j], current[i]
            current_c5 += best_delta
            improved = True
        
        if not improved:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def steepest_descent(h, max_rounds=200, time_limit=None, start_time=None):
    n = len(h)
    current = h.copy()
    current_c5 = compute_c5(current)
    
    for r in range(max_rounds):
        if time_limit and start_time and time.time() - start_time > time_limit:
            break
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        
        best_delta = 0
        best_move = None
        
        for i in ones:
            for j in zeros:
                current[i], current[j] = current[j], current[i]
                new_c5 = compute_c5(current)
                delta = new_c5 - current_c5
                if delta < best_delta - 1e-12:
                    best_delta = delta
                    best_move = (i, j)
                current[i], current[j] = current[j], current[i]
        
        if best_move:

            current[i], current[j] = current[j], current[i]
            current_c5 += best_delta
        else:
            break
    
    return current, current_c5

    rng_main = np.random.RandomState(seed)
    for r in range(n_restarts):
        if time.time() >= deadline:
            break
        # Create perturbation
        h_pert = best_h.copy()
        n_pert = rng_main.randint(3, 12)
        for _ in range(n_pert):
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            i = ones[rng_main.randint(len(ones))]
            j = zeros[rng_main.randint(len(zeros))]
            h_pert[i], h_pert[j] = 0, 1
        t_sub = min(20.0, remaining / max(1, n_restarts - r) * 0.8)
        h, c5 = sa_search(h_pert, n_iters=200000, T_start=0.01, T_end=0.00002,
                          seed=seed + 100 * (r + 1), time_limit=t_sub, t0=time.time())
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        # Quick steepest descent
        h2, c52 = steepest_descent(h, max_rounds=100, time_limit=t_sub * 0.4, t0=time.time())
        if c52 < best_c5:
            best_h, best_c5 = h2, c52

    # Phase 5: Greedy constructions
    elapsed = time.time() - t0
    if time.time() < deadline - 20:
        for gs in range(5):
            if time.time() >= deadline - 5:
                break
            h_g = greedy_construct(best_n, seed=seed + 1000 + gs)
            h_g, c5_g = steepest_descent(h_g, max_rounds=200, time_limit=3, t0=time.time())
            if c5_g < best_c5:
                best_h, best_c5 = h_g, c5_g
            h_g, c5_g = sa_search(h_g, n_iters=100000, T_start=0.01, T_end=0.00005,
                                  seed=seed + 2000 + gs, time_limit=5, t0=time.time())
            h_g2, c5_g2 = steepest_descent(h_g, max_rounds=100, time_limit=2, t0=time.time())
            if c5_g2 < best_c5:
                best_h, best_c5 = h_g2, c5_g2

    # Phase 6: Try other n values
    if time.time() < deadline - 10:
        for n_try in [40, 44, 46, 48, 50, 52, 54, 56, 58, 60]:
            if time.time() >= deadline - 5:
                break
            h_try = random_binary_safe(n_try, seed + n_try)
            h_try, c5_try = steepest_descent(h_try, max_rounds=200, time_limit=4, t0=time.time())
            h_try2, c5_try2 = sa_search(h_try, n_iters=100000, T_start=0.01, T_end=0.00005,
                                        seed=seed + 3000 + n_try, time_limit=5, t0=time.time())
            h_try3, c5_try3 = steepest_descent(h_try2, max_rounds=100, time_limit=2, t0=time.time())
            if c5_try3 < best_c5:
                best_h, best_c5, best_n = h_try3, c5_try3, n_try

    return best_h, best_c5, best_n

def random_binary_safe(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n, dtype=int)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1
    return h
