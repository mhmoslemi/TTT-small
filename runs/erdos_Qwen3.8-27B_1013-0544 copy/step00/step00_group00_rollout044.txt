        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_integral(h_interp, n_try)
        for T in [10, 30, 60]:
            if rem() < 30:
                break
            h_opt = run_lbgfs(h_interp, n_try, T, 300.0, 200)
            h_opt = project_integral(h_opt, n_try)
            c5 = compute_c5(h_opt, n_try)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt.copy(), n_try
            h_interp = h_opt

    # Phase 3: Random restarts at n_init
    if rem() > 50:
        for _ in range(30):
            if rem() < 20:
                break
            h_r = h_init + rng.randn(n_init) * 0.12
            h_r = project_integral(np.clip(h_r, 0, 1), n_init)
            h_r = run_lbgfs(h_r, n_init, 20, 300.0, 150)
            h_r = project_integral(h_r, n_init)
            c5 = compute_c5(h_r, n_init)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_r.copy(), n_init

    # Phase 4: Coordinate descent refinement
    if rem() > 40:
        n = best_n
        h_cd = best_h.copy()
        deltas = [0.01, -0.01, 0.03, -0.03, 0.06, -0.06, 0.1, -0.1]
        for _ in range(500):
            if rem() < 15:
                break
            improved = False
            idx = rng.permutation(n)
            for i in idx:
                for d in deltas:
                    h_t = h_cd.copy()
                    h_t[i] = np.clip(h_t[i] + d, 0.0, 1.0)
                    h_t = project_integral(h_t, n)
                    c5_t = compute_c5(h_t, n)
                    if c5_t < best_c5 - 1e-12:
                        h_cd = h_t
                        best_c5 = c5_t
                        improved = True
                        break
            if not improved:
                break
        best_h, best_n = h_cd, n

    # Phase 5: Final L-BFGS polish on best
    if rem() > 20:
        n = best_n
        h_polish = run_lbgfs(best_h.copy(), n, 100, 500.0, 500)
        h_polish = project_integral(h_polish, n)
        c5 = compute_c5(h_polish, n)
        if c5 < best_c5:
            best_c5, best_h = c5, h_polish

    # Fallback
    if best_h is None:
        best_h = project_integral(h_init.copy(), n_init)
        best_n = n_init
        best_c5 = compute_c5(best_h, best_n)

    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h, best_c5, best_n)
