def _smooth_obj_and_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    r = np.correlate(h, g, mode='full') * dx  # smoothed values
    # Softmax weights
    ar = alpha * r
    ar_shifted = ar - np.max(ar)
    p = np.exp(ar_shifted)
    p /= np.sum(p)
    
    # Objective
    f = np.max(ar) / alpha + np.log(np.sum(np.exp(ar_shifted))) / alpha
    # = np.max(r) + np.log(np.sum(np.exp(alpha*(r - np.max(r))))) / alpha
    
    # Gradient
    # s_k = k - (n-1) for k = 0, ..., 2n-2
    s = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    
    grad = np.zeros(n)
    for j in range(n):
        # First term: (1-h[j+s_k]) * I(0≤j+s_k≤n-1)
        valid_plus = (j + s >= 0) & (j + s < n)
        # Second term: -h[j-s_k] * I(0≤j-s_k≤n-1)
        valid_minus = (j - s >= 0) & (j - s < n)
        
        term1 = np.sum(p[valid_plus] * (1.0 - h[j + s[valid_plus]]))
        term2 = np.sum(p[valid_minus] * h[j - s[valid_minus]])
        grad[j] = dx * (term1 - term2)
    
    return f, grad

    deadline = start_time + budget_s - 3
    best_h, best_c5 = _pairwise_cd(best_h, dx, n, 40, deadline)

    return best_h, best_c5, n
