import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(beta * (x - m)))) / beta)

def fix_sum(h, target):
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0, 1)
    return h

def sa_optimize(h, target, rng, t_end, temp_start=0.002, cool=0.9997, temp_min=1e-7):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    temp = temp_start
    best_h = h.copy()
    best_c5 = c5
    while time.time() < t_end:
        npert = rng.randint(1, 4)
        idx = rng.choice(n, size=npert, replace=False)
        delta = rng.uniform(-0.05, 0.05, size=npert)
        new_h = h.copy()
        new_h[idx] += delta
        new_h = np.clip(new_h, 0, 1)
        d = target - new_h.sum()
        if abs(d) > 1e-10:
            mask = (new_h > 0.005) & (new_h < 0.995)
            if mask.sum() > 0:
                new_h[mask] += d / mask.sum()
                new_h = np.clip(new_h, 0, 1)
        if abs(new_h.sum() - target) > 0.05:
            temp = max(temp * cool, temp_min)
            continue
        new_c5 = compute_c5(new_h)
        if new_c5 <= c5 or rng.random() < np.exp(-(new_c5 - c5) / temp):
            h = new_h
            c5 = new_c5
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        temp = max(temp * cool, temp_min)

def fine_tune(h, target, rng, t_end):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    temp = 5e-5
    best_h = h.copy()
    best_c5 = c5

        idx1 = rng.randint(n)
        idx2 = rng.randint(n)
        if idx1 == idx2:
            continue
        delta = rng.uniform(-0.005, 0.005)
        new_h = h.copy()
        new_h[idx1] = np.clip(new_h[idx1] + delta, 0, 1)
        new_h[idx2] = np.clip(new_h[idx2] - delta, 0, 1)
        if abs(new_h.sum() - target) > 0.01:
            continue
        new_c5 = compute_c5(new_h)
        if new_c5 <= c5 or rng.random() < np.exp(-(new_c5 - c5) / temp):
            h = new_h
            c5 = new_c5
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        temp = max(temp * 0.99995, 1e-8)
    return best_h, best_c5

def slsqp_optimize(h, target, n, t_end, betas=(200, 500, 1000)):
    dx = 2.0 / n
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for beta in betas:
        if time.time() > t_end:
            break
        def obj(x, b=beta):
            corr = np.correlate(x, 1 - x, mode="full") * dx
            return smooth_max(corr, b)
        try:
            res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
            if res.x is not None:
                h_new = np.clip(res.x, 0, 1)
                h_new = fix_sum(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def random_bilevel_init(n, rng, target):
    h = np.zeros(n)
    # Create blocks with values near 0 or 1
    n_blocks = rng.randint(6, 15)
    block_sizes = rng.randint(3, max(4, n // n_blocks), size=n_blocks)
    total = block_sizes.sum()
    if total > n:
        block_sizes = block_sizes[:n // max(1, n_blocks)]
        total = block_sizes.sum()
    pos = 0
    ones_needed = target
    for bs in block_sizes:
        if pos + bs > n:
            bs = n - pos
        if bs <= 0:
            break
        val = 1.0 if ones_needed >= bs * 0.5 else 0.0
        h[pos:pos+bs] = val
        ones_needed -= bs if val == 1.0 else 0
        pos += bs
    h = fix_sum(h, target)
    return h
