from scipy.optimize import minimize
import time

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / len(h)
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project h onto {g: 0<=g<=1, sum(g)=target_sum}"""
    h = h.copy()
    h = np.clip(h, 0, 1)
    # Binary search for lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    # Get initial solution
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 50
        h = np.zeros(n)
        h[:n//2] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    n = len(h)
    target_sum = n / 2.0
    
    # Phase 1: Local search with continuous perturbations
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Simulated annealing
    temp = 0.05
    best_so_far = best_c5
    no_improve = 0
    
    while time_left() > 60:
        # Generate a perturbation
        h_pert = best_h.copy()
        n_pert = rng.randint(1, 4)
        idx = rng.choice(n, size=n_pert, replace=False)
        h_pert[idx] += rng.uniform(-0.1, 0.1, n_pert)
        h_pert = project_h(h_pert, target_sum)
        
        c5_new = compute_c5(h_pert)
        delta = c5_new - best_so_far
        
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            best_h = h_pert
            best_so_far = c5_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 1000:
            break
        temp *= 0.999
        if temp < 0.001:
            break
    
    best_h = best_h.copy()
    best_c5 = best_so_far
    
    # Phase 2: Try to improve with coordinate descent
    for _ in range(50 if time_left() > 30 else 5):
        improved = False
        for i in range(n):
            # Try different values for h[i]
            old_val = h[i]
            for dv in [-0.1, -0.05, 0.05, 0.1]:
                h[i] = np.clip(old_val + dv, 0, 1)
                # Adjust another coordinate to maintain sum
                diff = target_sum - np.sum(h)
                if abs(diff) < 1e-10:
                    break
                # Find a coordinate to adjust
                j = (i + 1) % n
                h[j] = np.clip(h[j] + diff, 0, 1)
                if np.sum(h) != target_sum:
                    # Try to fix
                    h = project_h(h, target_sum)
                
                c5_test = compute_c5(h)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h.copy()
                    improved = True
                h = best_h.copy()
        
        if not improved:
            break
    
    return best_h, best_c5, len(best_h)
