import numpy as np


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_h(h, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def grad_c5(h, n, soft=0.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)
    if soft > 0:
        m = np.max(corr)
        w = np.exp((corr - m) / soft)
        w /= np.sum(w)
    else:
        m = np.max(corr)
        w = np.zeros(nc)
        w[corr >= m - 1e-10] = 1.0
        w /= np.sum(w)
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in range(nc):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        j_lm = idx - lag
        j_lp = idx + lag
        m_lm = (j_lm >= 0) & (j_lm < n)
        m_lp = (j_lp >= 0) & (j_lp < n)
        t1 = np.zeros(n)
        t1[m_lm] = 1.0 - h[j_lm[m_lm]]
        t2 = np.zeros(n)
        t2[m_lp] = h[j_lp[m_lp]]
        grad += w[k] * (t1 - t2)
    return grad * dx
