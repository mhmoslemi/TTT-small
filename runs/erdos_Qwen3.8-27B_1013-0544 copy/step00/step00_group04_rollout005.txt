def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    overlap = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(overlap) * dx)

    t0 = time.time()

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]

                h[i] = 0.0

                h[i] = 1.0
