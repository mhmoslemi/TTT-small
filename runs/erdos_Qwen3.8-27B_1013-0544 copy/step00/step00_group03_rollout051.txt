import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def grad_for_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k > 0:
        g[k:] += 1.0 - h[:-k]
        g[:n-k] -= h[k:]
    elif k < 0:
        l = -k
        g[:n-l] += 1.0 - h[l:]
        g[l:] -= h[:n-l]
    else:
        g[:] = 1.0 - 2.0 * h
    return g

def project_feasible(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Binary search for threshold
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fine-tune: adjust a few elements to hit exact sum
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-15:
        # Spread the difference
        n = len(h_proj)
        # Find elements that can absorb the difference
        if diff > 0:
            # Need to increase some elements
            candidates = np.where(h_proj < 1.0)[0]
        else:
            # Need to decrease some elements
            candidates = np.where(h_proj > 0.0)[0]
        if len(candidates) > 0:
            h_proj[candidates] += diff / len(candidates)
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def optimize_from(h0, n_points, budget_s, rng):
    """Optimize from starting point h0"""
    n = n_points
    target_sum = n / 2.0
    dx = 2.0 / n

        corr = np.correlate(h, 1 - h, mode="full")
        max_val = np.max(corr)
        threshold = max_val - 1e-4

            grad += grad_for_lag(h, k)

                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0

                h = project_feasible(h, target_sum)
                no_improve = 0
                step_size = 0.1

            if time.time() - start_time > budget_s - 5:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx, np.max(corr)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Binary search for the right shift
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h_t = np.clip(h - mid, 0.0, 1.0)
        if np.sum(h_t) > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fix sum exactly
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def grad_for_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += 1.0 - h[:n-k]
        g[:n-k] -= h[k:]
    else:
        l = -k
        g[:n-l] += 1.0 - h[l:]
        g[l:] -= h[:n-l]
    return g

def optimize_gradient(h0, n, target_sum, time_limit, rng):
    h = project_feasible(h0.copy(), target_sum)
    dx = 2.0 / n
    best_c5, best_corr = compute_c5(h)
    best_h = h.copy()
    step = 0.05
    stagnant = 0

    while time.time() - t0 < time_limit:
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        # Soft-weighted gradient: weight each lag by its proximity to max
        weights = np.exp(corr / 0.05 - max_corr / 0.05)  # softmax with temperature 0.05
        weights /= np.sum(weights)
        
        grad = np.zeros(n)
        for m in range(len(corr)):
            if weights[m] > 1e-8:
                k = m - (n - 1)
                grad += weights[m] * grad_for_lag(h, k)
        grad *= dx
        
        # Try step
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = project_feasible(h - step * scale * grad, target_sum)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                h = h_new
                best_c5 = c5_new

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        h_t = np.clip(h - mid, 0.0, 1.0)
        if np.sum(h_t) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_gradient(h, temperature):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    w = np.exp((corr - max_corr) / temperature)
    w /= np.sum(w)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-10:
            continue
        k = m - (n - 1)
        if k == 0:
            grad += w[m] * (1.0 - 2.0 * h)
        elif k > 0:
            g = np.zeros(n)
            g[k:] = 1.0 - h[:n - k]
            g[:n - k] -= h[k:]
            grad += w[m] * g
        else:
            l = -k
            g = np.zeros(n)
            g[:n - l] = 1.0 - h[l:]
            g[l:] -= h[:n - l]
            grad += w[m] * g
    return grad * dx

def optimize_gradient(h0, n, target_sum, time_limit, rng, temperature=0.015):
    h = project_feasible(h0.copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.02
    stagnant = 0
    t0 = time.time()
    while time.time() - t0 < time_limit:
        grad = compute_gradient(h, temperature)
        scale_grad = np.linalg.norm(grad)
        if scale_grad < 1e-10:
            break
        grad = grad / scale_grad
        improved = False
        for scale in [step, step * 0.5, step * 0.25, step * 0.1, step * 0.05]:
            h_new = project_feasible(h - scale * grad, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                stagnant = 0
                improved = True
                break
        if not improved:
            stagnant += 1
            step *= 0.92
            if stagnant > 30:
                h = project_feasible(h + rng.normal(0, 0.015, n), target_sum)
                stagnant = 0
                step = 0.02
                temperature *= 0.85
    return best_h, best_c5
