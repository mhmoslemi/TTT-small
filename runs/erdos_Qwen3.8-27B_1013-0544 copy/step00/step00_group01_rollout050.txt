    """Project h onto {x : 0 <= x[i] <= 1, sum(x) = n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = h.min() - target, h.max()  # bounds for lambda
    # Actually: we need sum(clip(h[i] - lam, 0, 1)) = target
    # clip(h[i] - lam, 0, 1) is decreasing in lam
    # At lam = h.min() - 1, all clipped to 1, sum = n
    # At lam = h.max(), all clipped to 0, sum = 0
    lo = h.min() - 1.0
    hi = h.max()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

        [n for n in n_list if n != n_init], key=lambda x: abs(x - n_init))

    remaining = budget_s - (time.time() - t_start) - 8
    n_others = max(1, len(n_ordered) - 1)
    time_parent = remaining * 0.45
    time_each_other = remaining * 0.55 / n_others

    for n in n_ordered:
        elapsed = time.time() - t_start
        if remaining - elapsed < 4:
            break

        if n == n_init:
            t_limit = time_parent
        else:
            t_limit = min(time_each_other, remaining - elapsed - 4)
        if t_limit < 2:
            continue

        # Multiple initializations
        n_starts = 3 if n == n_init else 2
        for s in range(n_starts):
            if n == n_init and s == 0:
                h0 = h_init.copy()
            elif s == 0 or n != n_init:
                h0 = interpolate_h(h_init, n_init, n)
            elif s == 1:
                h0 = np.zeros(n)
                pos = rng.choice(n, n // 2, replace=False)
                h0[pos] = 1.0
            else:
                h0 = project_polytope(h_init[:n_init] if n >= n_init else h_init[:n], n)
                if n > n_init:
                    h0 = project_polytope(interpolate_h(h_init, n_init, n) + 0.3 * rng.standard_normal(n), n)

            h_opt, c5_val = optimize_n(n, h0, t_limit / n_starts, rng)

            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n

    # Final local search on best solution for remaining time
    elapsed = time.time() - t_start
    if elapsed < budget_s - 15:
        h_fs, c5_fs = local_search(best_h, best_n, rng, max_rounds=200)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = h_fs

    # Validate
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_polytope(best_h, best_n)

    return best_h, best_c5, best_n
