            if c5 < best_c5:
                best_h = h_trial
                best_c5 = c5
                h_final = h_trial.copy()

    # Final verification
    best_h = project_to_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
