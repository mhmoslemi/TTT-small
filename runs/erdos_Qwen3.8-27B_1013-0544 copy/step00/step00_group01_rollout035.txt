    m = np.max(corr)
    s = np.sum(np.exp(alpha * (corr - m)))
    return m + np.log(s) / alpha

def softmax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-12:
            continue
        w = k - (n - 1)
        lo = max(0, -w)
        hi = min(n, n - w)  # exclusive upper bound on n_idx
        for ni in range(lo, hi):
            ih = ni + w
            im = ni
            grad[ih] += weights[k] * (1.0 - h[im])
            grad[im] -= weights[k] * h[ih]
    return dx * grad

def project(h, n):

    h = np.clip(h, 0.0, 1.0)

    for _ in range(10):
        if abs(s - target) < 1e-10:

        scale = target / s
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)

        if mask.sum() > 0:
            adj = diff / mask.sum()
            h[mask] += adj

For the gradient phase, I'll use the subgradient of the max:
- Compute the correlation vector
- Find the lag(s) achieving the maximum
- Compute the gradient of the correlation at those lags
- Take a projected gradient step

The subgradient of max(corr_k * dx) with respect to h[i] is:
dx * (1/n_active) * sum_{k in active} dcorr_k/dh[i]

where active = {k : corr_k = max(corr)}.

For a single active lag k (with lag w = k - (n-1)):
dcorr_k/dh[i] = (1 - h[i-w]) * [valid] - h[i+w] * [valid]

Let me implement this:

```python
def subgradient(h, dx):
    """Compute subgradient of max correlation."""
    n = len(h)
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr_raw)
    # Active set: lags where corr is at max (within tolerance)
    tol = 1e-8
    active = np.where(np.abs(corr_raw - max_val) < tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        w = k - (n - 1)  # lag
        # Valid range: max(0, -w) <= j <= min(n-1, n-1-w)
        lo = max(0, -w)
        hi = min(n - 1, n - 1 - w)
        for j in range(lo, hi + 1):
            ih = j + w  # index in h (the "h" part)
            im = j      # index in h (the "1-h" part)
            grad[ih] += (1.0 - h[im])
            grad[im] -= h[ih]
    
    grad /= len(active)
    return dx * grad
```

This is O(n²) per call (for n=47, about 2200 ops per active lag, and there are at most n active lags, so O(n³) worst case = ~100k ops). Still fast.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.array(h, dtype=np.float64).copy()
    target = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        h += diff / n
    return np.clip(h, 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    indices = np.arange(n_new) * n_old / n_new
    i0 = np.floor(indices).astype(int) % n_old
    i1 = (i0 + 1) % n_old
    frac = indices - np.floor(indices)
    h_new = (1.0 - frac) * h[i0] + frac * h[i1]
    return project(h_new, n_new)

def softmax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    wts = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        if wts[k] < 1e-12:
            continue
        w = k - (n - 1)
        lo = max(0, -w)
        hi = min(n, n - w)
        for j in range(lo, hi):
            ih = j + w
            im = j
            grad[ih] += wts[k] * (1.0 - h[im])
            grad[im] -= wts[k] * h[ih]
    return dx * grad

    alpha = 40.0

            if vi < -1e-12 or vi > 1 + 1e-12 or vj < -1e-12 or vj > 1 + 1e-12:
