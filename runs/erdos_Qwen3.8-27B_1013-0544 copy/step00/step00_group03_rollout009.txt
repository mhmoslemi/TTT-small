        improved = False
        # Try flipping a 1 to 0 and a 0 to 1 (maintain sum)
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                val = c5_value(h, dx)
                if val < current - 1e-15:
                    current = val
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    improved = True
                    break
                h[i] = 1
                h[j] = 0
            if improved:
                break
        if not improved:
            break
    return h, current

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except:
        h_init = None

    if h_init is None:
        n_points = 70
        h_init = np.full(n_points, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0

    h = project_onto_constraint(h_init, target_sum)
    best_h = h.copy()
    best_c5 = c5_value(h, dx)

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP with multiple restarts
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]

    n_restarts = 30
    for r in range(n_restarts):
        if time_left() < 60:
            break
        if r == 0:
            h_cur = best_h.copy()
        elif r == 1:
            # Binary rounding of best
            h_bin = (best_h > 0.5).astype(float)
            # Fix sum
            surplus = h_bin.sum() - target_sum
            if surplus > 0:
                idx = np.argsort(h_bin)[::-1]
                flip_count = int(surplus)
                for i in range(flip_count):
                    if h_bin[idx[i]] > 0:
                        h_bin[idx[i]] = 1.0 - h_bin[idx[i]]
            elif surplus < 0:
                idx = np.argsort(h_bin)
                flip_count = int(-surplus)
                for i in range(flip_count):
                    if h_bin[idx[i]] < 1:
                        h_bin[idx[i]] = 1.0 - h_bin[idx[i]]
            h_cur = project_onto_constraint(h_bin, target_sum)
        else:
            noise_scale = 0.05 + 0.1 * (r // 5)
            noise = np.random.randn(n) * noise_scale
            h_cur = project_onto_constraint(best_h + noise, target_sum)

        try:
            res = minimize(
                lambda x: c5_value(x, dx),
                h_cur,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 300, 'ftol': 1e-14}
            )
            if res.fun < best_c5:
                best_h = np.clip(res.x, 0, 1)
                best_h = project_onto_constraint(best_h, target_sum)
                best_c5 = c5_value(best_h, dx)
        except:
            pass

    # Phase 2: Projected gradient with soft-max
    if time_left() > 120:
        temp = 0.02
        alpha = 0.005
        h = best_h.copy()
        cur_val = c5_value(h, dx)
        stalled = 0

        for it in range(20000):
            if time_left() < 60:
                break
            if stalled > 500:
                break

            corr = np.correlate(h, 1.0 - h, mode='full')
            m = np.max(corr)
            exp_vals = np.exp((corr - m) / temp)
            weights = exp_vals / exp_vals.sum()

            grad = compute_weighted_gradient(h, weights, n) * dx

            h_new = h - alpha * grad
            h_new = project_onto_constraint(h_new, target_sum)
            new_val = c5_value(h_new, dx)

            if new_val < cur_val - 1e-15:
                h = h_new
                cur_val = new_val
                stalled = 0
                if cur_val < best_c5:
                    best_h = h.copy()
                    best_c5 = cur_val
            else:
                stalled += 1
                alpha *= 0.95
                if alpha < 1e-6:
                    alpha = 0.005
                    # Nudge
                    h = project_onto_constraint(h + np.random.randn(n) * 0.02, target_sum)
                    cur_val = c5_value(h, dx)

    # Phase 3: Binary local search
    if time_left() > 120:
        h_bin = (best_h > 0.5).astype(float)
        # Ensure exact sum
        surplus = h_bin.sum() - target_sum
        if surplus != 0:
            if surplus > 0:
                order = np.argsort(best_h)[::-1]
                for i in range(int(surplus)):
                    h_bin[order[i]] = 0.0
            else:
                order = np.argsort(best_h)
                for i in range(int(-surplus)):
                    h_bin[order[i]] = 1.0

        for attempt in range(10):
            if time_left() < 60:
                break
            if attempt > 0:
                # Random perturbation of binary
                h_try = h_bin.copy()
                n_flips = np.random.randint(3, 15)
                idx = np.random.choice(n, n_flips, replace=False)
                h_try[idx] = 1.0 - h_try[idx]
                # Fix sum
                s = h_try.sum()
                diff = s - target_sum
                if abs(diff) > 1e-10:
                    if diff > 0:
                        ones_idx = np.where(h_try == 1)[0]
                        for i in range(int(diff)):
                            h_try[ones_idx[i]] = 0
                    else:
                        zeros_idx = np.where(h_try == 0)[0]
                        for i in range(int(-diff)):
                            h_try[zeros_idx[i]] = 1
                h_bin = h_try.copy()

            h_bin, bin_val = binary_local_search(h_bin, dx, target_sum, max_iters=2000)
            if bin_val < best_c5:
                best_h = h_bin.copy()
                best_c5 = bin_val

    # Phase 4: Final SLSQP polish from best
    if time_left() > 30:
        try:
            res = minimize(
                lambda x: c5_value(x, dx),
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-15}
            )
            if res.fun < best_c5:
                best_h = project_onto_constraint(res.x, target_sum)
                best_c5 = c5_value(best_h, dx)
        except:
            pass

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_constraint(best_h, target_sum)
    best_c5 = c5_value(best_h, dx)

    return best_h, best_c5, n
