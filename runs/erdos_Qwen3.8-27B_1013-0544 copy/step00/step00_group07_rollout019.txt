                stall = min(stall + 1, 10)

    # Phase 3: Random restarts at n0
    if elapsed() < budget_s * 0.72:
        for _ in range(100):
            if elapsed() > budget_s * 0.72:
                break
            h_r = np.zeros(n0)
            h_r[np.random.choice(n0, target, replace=False)] = 1.0
            h_c = hill_climb(h_r, max_time=0.8)
            c5 = evaluate_c5(h_c)
            if c5 < best_c5:
                best_h = h_c
                best_c5 = c5

    # Phase 4: Upscale and optimize at higher resolution
    for multiplier in [2, 3, 4]:
        n_new = n0 * multiplier
        if elapsed() > budget_s * 0.93:
            break
        x_old = np.arange(n0) * (2.0 / n0)
        x_new = np.arange(n_new) * (2.0 / n_new)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.where(h_up > 0.5, 1.0, 0.0)
        target_new = n_new // 2
        h_up = fix_sum(h_up, target_new)
        h_up = hill_climb(h_up, max_time=budget_s * 0.04)
        c5_up = evaluate_c5(h_up)
        if c5_up < best_c5:
            best_h = h_up
            best_c5 = c5_up

    # Final check
    if elapsed() > budget_s - 5:
        pass

    return best_h, best_c5, len(best_h)
