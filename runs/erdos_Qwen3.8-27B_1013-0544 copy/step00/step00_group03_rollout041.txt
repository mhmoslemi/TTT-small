import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    return (dx / alpha) * (alpha * max_c + np.log(np.sum(np.exp(alpha * (corr - max_c)))))

def grad_smooth_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    weights = np.exp(alpha * (corr - max_c))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        m = i - n + 1  # shift
        w = weights[i]
        if w < 1e-10:
            continue
        # grad contribution: w * [(1-h[p-m]) * [0<=p-m<n] - h[p+m] * [0<=p+m<n]]
        # For term 1: p such that 0 <= p-m < n, i.e., m <= p < m+n
        #   contribution: (1-h[p-m]) for p in [max(0,m), min(n-1, m+n-1)]
        # For term 2: p such that 0 <= p+m < n, i.e., -m <= p < n-m
        #   contribution: -h[p+m] for p in [max(0,-m), min(n-1, n-m-1)]
        
        p_start1 = max(0, m)
        p_end1 = min(n, m + n)  # exclusive
        if p_start1 < p_end1:
            idx = np.arange(p_start1, p_end1)
            grad[idx] += w * (1.0 - h[idx - m])
        
        p_start2 = max(0, -m)
        p_end2 = min(n, n - m)  # exclusive
        if p_start2 < p_end2:
            idx = np.arange(p_start2, p_end2)
            grad[idx] -= w * h[idx + m]
    
    return dx * grad

def project_onto_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Adjust: add diff/n to all, then clip, iterate
    for _ in range(100):
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        new_diff = target_sum - np.sum(h)
        if abs(new_diff) < 1e-10:
            break
        diff = new_diff
    return h
