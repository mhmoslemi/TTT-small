def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    lo = -5.0
    hi = 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        mask = (result > 1e-14) & (result < 1.0 - 1e-14)
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
    return np.clip(result, 0.0, 1.0)

    n = len(x)
