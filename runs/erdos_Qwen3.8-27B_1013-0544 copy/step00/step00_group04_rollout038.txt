import numpy as np
import time

def eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def local_search(h, time_limit=5.0):
    """Best-improvement local search via single swaps."""
    n = len(h)
    h = h.astype(float).copy()
    score = eval_c5(h)
    t0 = time.time()
    while time.time() - t0 < time_limit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        best_delta = 0.0
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0; h[j] = 1.0
                ns = eval_c5(h)
                d = score - ns
                if d > best_delta:
                    best_delta = d; best_pair = (i, j)
                h[i] = 1.0; h[j] = 0.0
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0.0; h[j] = 1.0
        score -= best_delta
    return h, score

def sa_optimize(h0, seed, t_limit):
    """Simulated annealing."""
    rng = np.random.RandomState(seed)
    n = len(h0)
    h = h0.astype(float).copy()
    score = eval_c5(h)
    best_h = h.copy()
    best_score = score
    T = 0.01
    cool = 0.9998
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    while time.time() < t_limit and T > 1e-9:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0; h[j] = 1.0
        ns = eval_c5(h)
        delta = ns - score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            score = ns
            ones[ones.index(i)] = j
            zeros[zeros.index(j)] = i
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            h[i] = 1.0; h[j] = 0.0
        T *= cool
    return best_h, best_score

def interpolate_to_n(h_src, n_dst):
    """Interpolate a binary source to a new length, then threshold."""
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.arange(n_src, dtype=float)
    x_dst = np.linspace(0, n_src - 1, n_dst)
    vals = np.interp(x_dst, x_src, h_src)
    h = (vals > 0.5).astype(float)
    # Fix count

            h[np.random.choice(idx, cur - target, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, target - cur, replace=False)] = 1
    return h

Actually, ∂C_k/∂h[j] for a specific k is basically a vector with two non-zero "slices":

- -h[j-k] for j in [k, n-1] (for k >= 0)

So the weighted sum is:
∂f/∂h[j] = sum_{k >= 0} w_k * [(1-h[j+k]) * [j <= n-1-k] + (-h[j-k]) * [j >= k]]

- First sum ≈ (1-h) convolved with w (positive lags) + h convolved with w (negative lags)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def local_search(h, time_limit=5.0):
    n = len(h)
    h = h.astype(np.float64).copy()
    score = eval_c5(h)
    t0 = time.time()
    while time.time() - t0 < time_limit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]

        best_p = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0; h[j] = 1.0
                ns = eval_c5(h)
                d = score - ns
                if d > best_d:

        if best_p is None:
            break
        i, j = best_p
        h[i] = 0.0; h[j] = 1.0
        score -= best_d
    return h, score

                grad[idx2] += (-h[idx2 - lag])

            grad[idx2] += (1.0 - h[idx2 - l])
            idx = np.arange(n - l)
            grad[idx] += (-h[idx])
