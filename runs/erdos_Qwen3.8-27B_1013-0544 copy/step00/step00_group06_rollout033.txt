    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = h.copy()
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if m.any():
            h[m] += d / int(m.sum())
        else:
            m = h < 1.0 if d > 0 else h > 0.0
            if not m.any():
                break
            h[m] += d / int(m.sum())
        np.clip(h, 0.0, 1.0, out=h)

    """L-BFGS-B with penalty for sum constraint."""
    def f_and_g(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode="full")

        c = x.sum() - target

            m = n - r
            g[r:] += 1.0 - x[:m]
            g[:m] += -x[r:]
        g += 200.0 * c
        return val, g

        c = xc.sum() - target
        val += 50.0 * c * c
        mi = int(np.argmax(corr))
        lag = mi - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            m = n - lag
            g[:m] += 1.0 - xc[lag:]
            g[lag:] += -xc[:m]
        else:
            r = -lag
            m = n - r
            g[r:] += 1.0 - xc[:m]
            g[:m] += -xc[r:]
        g += 100.0 * c
        return val, g
    res = minimize(fg, h, method='L-BFGS-B',
                  bounds=[(0.0, 1.0)] * n,
                  options={'maxiter': max_iter, 'ftol': 1e-15})
    return fix_sum(np.clip(res.x, 0.0, 1.0), target)

def thue_morse(n):
    return np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)

def make_binary(h, n, target):
    hb = np.zeros(n)
    k = int(np.floor(target))
    idx = np.argsort(-h)[:k]
    hb[idx] = 1.0
    if target != k:
        remaining = np.argsort(-h)[k:k+1]
        if len(remaining) > 0:
            hb[remaining[0]] = target - k
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    target0 = n0 / 2.0
    h0 = fix_sum(h0, target0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS-B from initial
    if rem() > 60:
        h1 = lbfgs_opt(h0, n0, target0, max_iter=3000)
        c5 = compute_c5(h1)
        if c5 < best_c5:
            best_h, best_c5 = h1, c5

    # Phase 2: Binary swap search from best
    if rem() > 30:
        hb = make_binary(best_h, n0, target0)
        tl = min(rem() - 15, 200)
        h2, c52 = bin_swap_search(hb, target0, tl)
        if c52 < best_c5:
            best_h, best_c5 = h2, c52

    # Phase 3: Random binary restarts
    for _ in range(30):
        if rem() < 25:
            break
        hr = np.zeros(n0)
        k = int(np.floor(target0))
        idx = np.random.choice(n0, k, replace=False)
        hr[idx] = 1.0
        hr = fix_sum(hr, target0)
        tl = min(rem() - 10, 40)
        h3, c53 = bin_swap_search(hr, target0, tl)
        if c53 < best_c5:
            best_h, best_c5 = h3, c53

    # Phase 4: Thue-Morse and structured starts
    if rem() > 40:
        tm = thue_morse(n0)
        tm = fix_sum(tm, target0)
        tl = min(rem() - 10, 100)
        h4, c54 = bin_swap_search(tm, target0, tl)
        if c54 < best_c5:
            best_h, best_c5 = h4, c54
    if rem() > 40:
        tm2 = thue_morse(n0)
        tm2 = 1.0 - tm2
        tm2 = fix_sum(tm2, target0)
        tl = min(rem() - 10, 80)
        h5, c55 = bin_swap_search(tm2, target0, tl)
        if c55 < best_c5:
            best_h, best_c5 = h5, c55

    # Phase 5: L-BFGS-B polish
    if rem() > 40:
        h6 = lbfgs_opt(best_h, n0, target0, max_iter=3000)
        c56 = compute_c5(h6)
        if c56 < best_c5:
            best_h, best_c5 = h6, c56

    # Phase 6: Try even resolutions
    for n_try in [96, 128, 192, 256]:
        if rem() < 50:
            break
        tt = n_try / 2.0
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_intr = np.interp(x_new, x_old, best_h)
        h_intr = fix_sum(h_intr, tt)
        hb = make_binary(h_intr, n_try, tt)
        tl = min(rem() - 15, 80)
        h7, c57 = bin_swap_search(hb, tt, tl)
        if c57 < best_c5:
            best_h, best_c5 = h7, c57
        if rem() > 30:
            h8 = lbfgs_opt(h_intr, n_try, tt, max_iter=800)
            c58 = compute_c5(h8)
            if c58 < best_c5:
                best_h, best_c5 = h8, c58

    return best_h, best_c5, len(best_h)
