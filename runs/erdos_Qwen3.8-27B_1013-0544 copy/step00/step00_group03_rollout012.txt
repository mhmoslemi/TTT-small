import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def smooth_max_val(corr, tau):
    m = np.max(corr)
    return m + tau * np.log(np.sum(np.exp((corr - m) / tau)))


def project_feasible(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0, 1)


def smooth_obj(x, tau, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    return smooth_max_val(corr, tau) * dx


def smooth_grad(x, tau, n):
    corr = np.correlate(x, 1 - x, mode='full')
    m = np.max(corr)
    w = np.exp((corr - m) / tau)
    w /= w.sum()

        i_hi = min(n - 1, 2 * n - 2 - k)

    return np.sum(x) - target_sum


def sum_con_jac(x, target_sum):
    return np.ones_like(x)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def smooth_max_val(corr, tau):
    m = np.max(corr)
    return m + tau * np.log(np.sum(np.exp((corr - m) / tau)))


def project_feasible(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0, 1)


def smooth_obj(x, tau, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    return smooth_max_val(corr, tau) * dx


def smooth_grad(x, tau, n):
    corr = np.correlate(x, 1 - x, mode='full')
    m = np.max(corr)
    w = np.exp((corr - m) / tau)
    w /= w.sum()
    n1 = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        i_lo = max(0, n1 - k)
        i_hi = min(n - 1, 2 * n - 2 - k)
        if i_lo > i_hi:
            continue
        j_lo = i_lo + k - n1
        j_hi = i_hi + k - n1
        grad[i_lo:i_hi + 1] += w[k] * (1.0 - x[j_lo:j_hi + 1])
        grad[j_lo:j_hi + 1] -= w[k] * x[i_lo:i_hi + 1]
    return grad * dx


def sum_con_fun(x, target_sum):
    return np.sum(x) - target_sum


def sum_con_jac(x, target_sum):
    return np.ones_like(x)


def run_slsqp(h, tau, n, target_sum, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (target_sum,)}]
    res = minimize(smooth_obj, h, args=(tau, n), jac=smooth_grad,
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return project_feasible(res.x, target_sum)


def projected_gradient(h, tau, n, target_sum, max_iters, t_start, budget_s):
    step = 0.05
    for it in range(max_iters):
        if time.time() - t_start > budget_s - 10:
            break
        g = smooth_grad(h, tau, n)
        h_new = project_feasible(h - step * g, target_sum)
        obj_old = smooth_obj(h, tau, n)
        obj_new = smooth_obj(h_new, tau, n)
        if obj_new < obj_old:
            h = h_new
            step = min(step * 1.3, 0.5)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h_new = project_feasible(h - step * g, target_sum)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    n = 70
    target_sum = n / 2.0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n, 0.5)

    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)

    # Phase 1: SLSQP with tau annealing
    taus = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]
    for tau in taus:
        if time.time() - t_start > budget_s - 60:
            break
        try:
            h_cand = run_slsqp(h, tau, n, target_sum, 500)
            c5_cand = compute_c5(h_cand, n)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_cand.copy()
            h = h_cand
        except Exception:
            pass

    # Phase 2: Projected gradient fine-tuning
    if time.time() - t_start < budget_s - 30:
        h = best_h.copy()
        h = projected_gradient(h, 0.01, n, target_sum, 30000, t_start, budget_s)
        c5_cur = compute_c5(h, n)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()

    # Phase 3: Perturbation-based restarts
    rng = np.random.RandomState(seed)
    if time.time() - t_start < budget_s - 30:
        for pert_scale in [0.15, 0.1, 0.05, 0.03, 0.01]:
            for _ in range(3):
                if time.time() - t_start > budget_s - 20:

                h_pert = best_h + rng.randn(n) * pert_scale
                h_pert = project_feasible(h_pert, target_sum)
                try:
                    h_cand = run_slsqp(h_pert, 0.03, n, target_sum, 300)
                    c5_cand = compute_c5(h_cand, n)
                    if c5_cand < best_c5:
                        best_c5 = c5_cand
                        best_h = h_cand.copy()
                except Exception:
                    pass

    # Phase 4: Final polish with very small tau
