        hp[i] += eps

        hm[i] -= eps

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(v, n, target):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def c5_exact(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_obj(h, dx, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    return (m + np.log(np.sum(np.exp(c - m)))) * dx / alpha

def gradient_smooth(h, dx, alpha, eps=1e-5):
    n = len(h)
    g = np.zeros(n)
    for i in range(n):
        hp = h.copy()
        hp[i] += eps
        hm = h.copy()
        hm[i] -= eps
        g[i] = (smooth_obj(hp, dx, alpha) - smooth_obj(hm, dx, alpha)) / (2.0 * eps)
    return g

def projected_gradient(h, target, dx, alpha, lr, n_iter, rng):
    n = len(h)
    h_best = h.copy()
    c5_best = c5_exact(h_best, dx)
    for it in range(n_iter):
        g = gradient_smooth(h, dx, alpha)
        h = h - lr * g
        h = project_feasible(h, n, target)
        c5 = c5_exact(h, dx)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()
        if lr > 0.002:
            lr *= 0.985
        else:
            h = h_best + 0.03 * rng.randn(n)
            h = project_feasible(h, n, target)
    return h_best, c5_best

def coord_desc(h, target, dx, n_iter, rng):
    n = len(h)
    h_best = h.copy()
    c5_best = c5_exact(h_best, dx)
    T = 0.015
    for it in range(n_iter):
        if it % 8000 == 0:

        i = rng.randint(n)

        delta = 0.06 * rng.randn()
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = project_feasible(h_new, n, target)
        c5_new = c5_exact(h_new, dx)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
            h = h_new
        elif T > 1e-8 and rng.rand() < np.exp(-(c5_new - c5_best) / T):
            h = h_new
    return h_best, c5_best

def deterministic_polish(h, target, dx, rng):
    n = len(h)

    c5_best = c5_exact(h_best, dx)

                    h_best = h_try.copy()
                    improved = True
        if not improved:
