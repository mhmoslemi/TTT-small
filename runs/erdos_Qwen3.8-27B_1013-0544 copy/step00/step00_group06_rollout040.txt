                zeros = np.where(h <= 0.01)[0]
            else:
                h[i], h[j] = old_i, old_j
                patience += 1
        if not improved and patience > max_patience:
            break
    return h, current

def coord_descent(h, time_limit):
    n = len(h)
    h = h.copy()
    current = compute_c5(h)
    start = time.time()
    while time.time() - start < time_limit:
        improved = False
        ones = np.where(h >= 0.99)[0]
        zeros = np.where(h <= 0.01)[0]
        for i in ones:
            if time.time() - start > time_limit:
                break
            best_j = -1
            best_val = current
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                c5 = compute_c5(h)
                h[i], h[j] = h[j], h[i]
                if c5 < best_val - 1e-14:
                    best_val = c5
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = h[best_j], h[i]
                current = best_val
                improved = True
                ones = np.where(h >= 0.99)[0]
                zeros = np.where(h <= 0.01)[0]
        if not improved:
            break
    return h, current

def lbfgsb_optimize(h0, alpha, pen, n, max_iter, time_limit):
    start = time.time()
    def obj_func(x):
        return total_obj(x, alpha, pen, n)
    res = minimize(obj_func, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'maxfun': max_iter * 10,
                            'ftol': 1e-12, 'gtol': 1e-8})
    return res.x

def enforce_sum(h, n):
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        h = h + (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = float('inf')

    def remaining():
        return budget_s - (time.time() - t0)

    h_init = None
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)

    # Phase 1: Optimize from initial_h_values
    if h_init is not None:
        n0 = len(h_init)
        for alpha in [50.0, 100.0, 200.0]:
            if remaining() < 30:
                break
            h_opt = lbfgsb_optimize(h_init, alpha, 500.0, n0, 200, remaining() * 0.4)
            h_opt = enforce_sum(h_opt, n0)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            # Also try binary projection + local search
            h_bin = project_binary(h_opt)
            h_ls, c5_ls = local_search(h_bin, min(20, remaining() * 0.3), rng)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

    # Phase 2: Try structured and random inits at various n
    n_list = [96, 100, 104, 128, 98, 112, 94, 108]
    for n in n_list:
        if remaining() < 40:
            break
        for trial in range(4):
            if remaining() < 25:
                break
            if trial == 0:
                h0 = structured_initial(n, seed + n)
            else:
                h0 = make_binary(n, rng)
            for alpha in [80.0, 150.0]:
                if remaining() < 20:
                    break
                h_opt = lbfgsb_optimize(h0, alpha, 500.0, n, 150, remaining() * 0.3)
                h_opt = enforce_sum(h_opt, n)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                h_bin = project_binary(h_opt)
                h_ls, c5_ls = local_search(h_bin, min(15, remaining() * 0.25), rng)
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls.copy()

    # Phase 3: Systematic coordinate descent on best
    if best_h is not None and remaining() > 20:
        h_bin = project_binary(best_h)
        tl = min(80, remaining() * 0.4)
        h_cd, c5_cd = coord_descent(h_bin, tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: More local search
    if best_h is not None and remaining() > 10:
        h_bin = project_binary(best_h)
        tl = min(40, remaining() * 0.7)
        h_ls, c5_ls = local_search(h_bin, tl, rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 5: Try larger n with init from best
    if best_h is not None and remaining() > 30:
        for n_big in [150, 200, 256]:
            if remaining() < 25:
                break
            # Upsample best_h to n_big
            old_n = len(best_h)
            indices = np.round(np.linspace(0, old_n - 1, n_big)).astype(int)
            h_big = best_h[indices].copy()
            h_big = project_binary(h_big)
            h_ls, c5_ls = local_search(h_big, min(20, remaining() * 0.3), rng)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

    # Final: more systematic coord descent if time
    if best_h is not None and remaining() > 15:
        h_bin = project_binary(best_h)
        tl = min(60, remaining() * 0.8)
        h_cd, c5_cd = coord_descent(h_bin, tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Ensure valid output
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Fix sum if needed
    target = n_final / 2.0
    s = np.sum(best_h)
    if abs(s - target) > 1e-10:
        diff = target - s
        ones = np.where(best_h == 1.0)[0]
        zeros = np.where(best_h == 0.0)[0]
        # Adjust
        if diff > 0 and len(zeros) > 0:
            best_h[zeros[:int(np.ceil(diff))]] = 1.0
        elif diff < 0 and len(ones) > 0:
            best_h[ones[:int(np.ceil(-diff))]] = 0.0
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, n_final
