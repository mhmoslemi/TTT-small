def repair(h, n):
    """Repair h to satisfy constraints."""
    h = np.clip(h, 0, 1)
    # Scale to meet sum constraint
    target_sum = n / 2
    current_sum = h.sum()
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    # If still not meeting constraint, adjust
    diff = target_sum - h.sum()
    if diff > 0:
        # Need to increase some values
        indices = np.where(h < 1)[0]

        if len(indices) > 0:

            h[indices] = np.maximum(0, h[indices] - per)

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def swap_move(h, i, j):
    """Swap h[i] and h[j], return new h."""
    new_h = h.copy()
    new_h[i], new_h[j] = h[j], h[i]
    return new_h

def greedy_local_search(h, dx, max_iter=50000):
    n = len(h)
    current_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    for it in range(max_iter):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        best_gain = 0
        best_pair = None
        
        for i in ones:
            for j in zeros:
                new_h = h.copy()
                new_h[i], new_h[j] = h[j], h[i]
                new_c5 = compute_c5(new_h, dx)
                gain = current_c5 - new_c5
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (i, j)
        
        if best_pair is not None and best_gain > 1e-12:
            i, j = best_pair
            h[i], h[j] = h[j], h[i]
            current_c5 = compute_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            improved = True
        else:
            break
    
    return best_h, best_c5

        for k in range(target_ones - n_ones):
            h_bin[idx[k]] = 1.0

    h_ls, c5_ls = greedy_binary_search(h_bin.copy(), dx, n, max_rounds=100,
                                        deadline=t0 + 30)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()

    # Phase 2: Simulated annealing (main exploration)
    sa_deadline = t0 + budget_s * 0.45
    h_sa, c5_sa = simulated_annealing(best_h.copy(), dx, n, rng, sa_deadline)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Smoothed gradient descent (continuous refinement)
    gd_deadline = t0 + budget_s * 0.75
    h_gd, c5_gd = smoothed_grad_descent(best_h.copy(), dx, n, gd_deadline)
    if c5_gd < best_c5:
        best_c5 = c5_gd
        best_h = h_gd.copy()

    # Phase 4: Another round of SA from the refined solution
    sa2_deadline = t0 + budget_s * 0.85
    h_sa2, c5_sa2 = simulated_annealing((best_h > 0.5).astype(float), dx, n, rng, sa2_deadline)
    if c5_sa2 < best_c5:
        best_c5 = c5_sa2
        best_h = h_sa2.copy()

    # Phase 5: Random perturbation refinement
    rp_deadline = t0 + budget_s * 0.95
    h_rp, c5_rp = random_perturb_refine(best_h.copy(), dx, n, rng, rp_deadline)
    if c5_rp < best_c5:
        best_c5 = c5_rp
        best_h = h_rp.copy()

    # Phase 6: Final greedy polish
    if time.time() < t0 + budget_s * 0.97:
        h_pol = (best_h > 0.5).astype(float)
        n_o = int(round(h_pol.sum()))
        if n_o != n // 2:
            idx1 = np.where(h_pol > 0.5)[0]
            idx0 = np.where(h_pol < 0.5)[0]
            if n_o > n // 2:
                for k in range(n_o - n // 2):
                    h_pol[idx1[k]] = 0.0
            else:
                for k in range(n // 2 - n_o):
                    h_pol[idx0[k]] = 1.0
        h_pol, c5_pol = greedy_binary_search(h_pol, dx, n, max_rounds=50,
                                              deadline=t0 + budget_s * 0.97)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    return best_h, best_c5, n
