Actually, for the cyclic case, if we have a binary sequence of length n with exactly n/2 ones, then:
- The number of 1→0 pairs at shift k equals the number of 0→1 pairs at shift k (for cyclic)
- Total 1-0 pairs at all shifts = (n/2) * (n/2) = n²/4
- So the average over all k is (n²/4) / (n-1) ≈ n/4

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_corr_at_lag(h, tau):
    n = len(h)
    grad = np.zeros(n)
    idx = np.arange(n)
    m1 = idx - tau
    v1 = (m1 >= 0) & (m1 < n)
    grad[v1] += (1.0 - h[m1[v1]])
    m2 = idx + tau
    v2 = (m2 >= 0) & (m2 < n)
    grad[v2] -= h[m2[v2]]
    return grad

def projected_gradient_optimize(h0, target, seed, max_iter, time_limit):
    t_start = time.time()
    n = len(h0)
    h = h0.copy()
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    lr = 0.03
    v = np.zeros(n)
    patience = 300
    no_improve = 0
    rng = np.random.RandomState(seed)

    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(corr)
        tau = idx_max - (n - 1)
        g = grad_corr_at_lag(h, tau) * (2.0 / n)

        # Backtracking line search
        alpha = lr
        for _ in range(20):
            v = 0.4 * v - alpha * g
            new_h = project(h + v, target)
            new_c5 = compute_c5(new_h)
            if new_c5 < cur_c5 - 1e-12:
                break
            alpha *= 0.5
            v = 0.4 * v - alpha * g

        if new_c5 < cur_c5 - 1e-12:
            h = new_h
            cur_c5 = new_c5
            no_improve = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            no_improve += 1
            v *= 0.3

                cur_c5 = compute_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                no_improve = 0
                v = np.zeros(n)

def binary_local_search(n, seed, max_iters, time_limit):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    k = n // 2
    pos = np.zeros(n, dtype=bool)
    pos[rng.choice(n, k, replace=False)] = True
    h = pos.astype(float)
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        ones_idx = np.where(pos)[0]
        zeros_idx = np.where(~pos)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        pos[i] = False
        pos[j] = True
        h[j] = 1.0
        h[i] = 0.0
        c5 = compute_c5(h)
        if c5 <= best_c5:

            best_h = h.copy()
            no_improve = 0
