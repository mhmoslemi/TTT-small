            h_r, c5_r = optimize_from(h_r, n2, rng, seed + 200 + r * 10, dl)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n2

    # Phase 5: Try n=210
    remaining = budget_s - 15 - (time.time() - start)
    if remaining > 40 and best_h is not None:
        n3 = 210
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n3)
        h3_init = np.interp(x_new, x_old, best_h)
        h3 = to_binary(h3_init, n3, rng)
        dl = start + min(remaining - 5, 180)
        h3, c5_3 = optimize_from(h3, n3, rng, seed + 400, dl)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n3

    # Phase 6: Final restarts at best_n
    remaining = budget_s - 8 - (time.time() - start)
    if remaining > 20 and best_h is not None:
        nb = best_n
        for r in range(6):
            t_left = budget_s - 8 - (time.time() - start)
            if t_left < 10:
                break
            h_r = np.zeros(nb)
            pos = rng.choice(nb, nb // 2, replace=False)
            h_r[pos] = 1.0
            dl = start + t_left - 3
            h_r, c5_r = optimize_from(h_r, nb, rng, seed + 600 + r * 7, dl)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = nb

    if best_h is None:
        best_h = h1
        best_n = n1
        best_c5 = eval_c5(h1, n1)

    # Final validation
    assert np.all(best_h >= 0.0) and np.all(best_h <= 1.0)
    assert abs(np.sum(best_h) - best_n // 2) < 1e-9

    return best_h.tolist(), best_c5, best_n
