def compute_gradient(h, weights, n):
    """Compute gradient of weighted sum of correlations.
    weights[k] corresponds to output index k of correlate, i.e., lag k-(n-1).
    """
    G = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k]
        if abs(w) < 1e-12:
            continue
        l = k - (n-1)
        if l >= 0:
            # First factor: j from 0 to n-1-l
            if n - l > 0:
                G[:n-l] += w * (1 - h[l:])
            # Second factor: j from l to n-1
            if l < n:
                G[l:] += w * (-h[:n-l])
        else:
            m = -l
            # First factor: j from m to n-1
            if m < n:
                G[m:] += w * (1 - h[:n-m])
            # Second factor: j from 0 to n-1-m
            if n - m > 0:
                G[:n-m] += w * (-h[m:])
    return G

import numpy as np
import time

def project_onto_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            c = mask.sum()
            if c == 0:
                break
            h[mask] += diff / c
        else:
            mask = h > 1e-10
            c = mask.sum()
            if c == 0:
                break
            h[mask] += diff / c
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmax(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(corr, dx):
    return float(np.max(corr) * dx)

def compute_gradient(h, weights, n):
    G = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        l = k - nm1
        if l >= 0:
            ns = n - l
            if ns > 0:
                G[:ns] += w * (1.0 - h[l:])
                G[l:] += w * (-h[:ns])
        else:
            m = -l
            if m < n:
                ns = n - m
                G[m:] += w * (1.0 - h[:ns])
                G[:ns] += w * (-h[m:])
    return G

def run_optimize(h0, n, dx, end_time, beta0=40.0, alpha0=0.003):
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(compute_corr(h), dx)
    beta = beta0
    alpha = alpha0
    mom = 0.85
    v = np.zeros(n)
    no_improve = 0
    iters = 0
    while time.time() < end_time:
        corr = compute_corr(h)
        max_c = np.max(corr)
        shifted = beta * (corr - max_c)
        exp_s = np.exp(shifted)
        wsum = exp_s.sum()
        if wsum < 1e-300:
            break
        weights = exp_s / wsum
        G = compute_gradient(h, weights, n)
        v = mom * v + alpha * G
        h_new = h - v
        h_new = project_onto_constraint(h_new, target_sum)
        c5_new = compute_c5(compute_corr(h_new), dx)
        iters += 1
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 15:
            alpha *= 0.8
            v *= 0.4
            no_improve = 0
            if alpha < 1e-9:
                break
        if no_improve > 60:
            break
        if iters % 100 == 0 and beta < 400:
            beta *= 1.3
        h = h_new
    return best_h, best_c5

def local_search(h_in, n, dx, max_passes=200, end_time=None):
    target_sum = n / 2.0
    h = h_in.copy()
    best_c5 = compute_c5(compute_corr(h), dx)
    improved = True
    passes = 0
    while improved and passes < max_passes:
        if end_time is not None and time.time() > end_time:
            break
        improved = False
        passes += 1
        for i in range(n):
            if end_time is not None and time.time() > end_time:
                break
            old_val = h[i]
            for new_val in [0.0, 1.0, 0.5, 0.25, 0.75]:
                if abs(new_val - old_val) < 1e-8:
                    continue
                h[i] = new_val
                h = project_onto_constraint(h, target_sum)
                c5_try = compute_c5(compute_corr(h), dx)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    improved = True
                else:
                    h[i] = old_val
                    h = project_onto_constraint(h, target_sum)
    return h, best_c5
