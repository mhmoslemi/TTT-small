    while time.time() - t0 < budget_s - 40:
        step = 0.02 + 0.25 * (T / T0)
        r = np.random.random()

        if r < 0.65:
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            d = np.random.uniform(-step, step)
            if d > 0:
                d = min(d, 1.0 - h[i], h[j])
            else:
                d = max(d, h[i] - 1.0, -h[j])
            if abs(d) < 1e-14:
                continue
            nh = h.copy()
            nh[i] += d
            nh[j] -= d
        elif r < 0.8:
            i, j = np.random.choice(n, 2, replace=False)
            if h[i] == h[j]:
                continue
            nh = h.copy()
            nh[i], nh[j] = nh[j], nh[i]
        elif r < 0.9:
            idx = np.random.choice(n, 4, replace=False)
            d1 = np.random.uniform(-step, step)
            if d1 > 0:
                d1 = min(d1, 1.0 - h[idx[0]], h[idx[1]])
            else:
                d1 = max(d1, h[idx[0]] - 1.0, -h[idx[1]])
            if abs(d1) < 1e-14:
                continue
            nh = h.copy()
            nh[idx[0]] += d1
            nh[idx[1]] -= d1
        else:
            i, j = np.random.choice(n, 2, replace=False)
            d = np.random.uniform(-step * 0.4, step * 0.4)
            if d > 0:
                d = min(d, 1.0 - h[i], h[j])
            else:
                d = max(d, h[i] - 1.0, -h[j])
            if abs(d) < 1e-14:
                continue
            nh = h.copy()
            nh[i] += d
            nh[j] -= d

        nc = compute_c5(nh)

        if nc < cur or (T > Tmin and np.random.random() < np.exp((cur - nc) / T)):
            h = nh
            cur = nc
            if cur < best:
                best = cur
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 20
        else:
            stuck += 1

        T = max(T * cr, Tmin)

        if stuck > 50000:
            stuck = 0
            T = T0 * 0.4
            for _ in range(8):
                ii, jj = np.random.choice(n, 2, replace=False)
                dd = np.random.uniform(-0.08, 0.08)
                if dd > 0:
                    dd = min(dd, 1.0 - h[ii], h[jj])
                else:
                    dd = max(dd, h[ii] - 1.0, -h[jj])
                h[ii] += dd
                h[jj] -= dd
            cur = compute_c5(h)

    # Greedy polish phase
    if time.time() - t0 < budget_s - 5:
        for sweep in range(100):
            if time.time() - t0 > budget_s - 5:
                break
            improved = False
            deltas = (0.01, 0.005, 0.002, 0.001)
            for i in range(n):
                for j in range(i + 1, n):
                    for d in deltas:
                        if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                            nh = h.copy()
                            nh[i] += d
                            nh[j] -= d
                            nc = compute_c5(nh)
                            if nc < cur - 1e-13:
                                h = nh
                                cur = nc
                                improved = True
                                if cur < best:
                                    best = cur
                                    best_h = h.copy()
                        if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                            nh = h.copy()
                            nh[i] -= d
                            nh[j] += d
                            nc = compute_c5(nh)
                            if nc < cur - 1e-13:
                                h = nh
                                cur = nc
                                improved = True
                                if cur < best:
                                    best = cur
                                    best_h = h.copy()
            if not improved:
                break

    # Try finer grid if time remains
    if time.time() - t0 < budget_s - 20:
        n2 = n * 2
        h2 = np.repeat(best_h, 2).copy()
        cur2 = compute_c5(h2)
        T = 0.01
        stuck2 = 0
        while time.time() - t0 < budget_s - 15:
            step2 = 0.01 + 0.1 * (T / 0.01)
            i = np.random.randint(n2)
            j = np.random.randint(n2)
            if i == j:
                continue
            d = np.random.uniform(-step2, step2)
            if d > 0:
                d = min(d, 1.0 - h2[i], h2[j])
            else:
                d = max(d, h2[i] - 1.0, -h2[j])
            if abs(d) < 1e-14:
                continue
            nh2 = h2.copy()
            nh2[i] += d
            nh2[j] -= d
            nc2 = compute_c5(nh2)
            if nc2 < cur2 or (T > Tmin and np.random.random() < np.exp((cur2 - nc2) / T)):
                h2 = nh2
                cur2 = nc2
                if cur2 < best:
                    best = cur2
                    best_h = h2.copy()
                    stuck2 = 0
                else:
                    stuck2 += 10
            else:
                stuck2 += 1
            T = max(T * 0.99999, Tmin)
            if stuck2 > 30000:
                stuck2 = 0
                T = 0.005

    return best_h, best, len(best_h)
