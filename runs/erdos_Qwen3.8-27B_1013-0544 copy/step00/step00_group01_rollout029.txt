            if np.random.random() < np.exp(-(new_c5 - old_c5) / T):
                h = cand
            patience += 1
        T = max(T_min, T * 0.9999)
        if patience >= max_pat:
            patience = 0
            max_pat = int(max_pat * 1.3)
            T = 0.05
            h = best_h + np.random.normal(0, 0.02, n)
            h = fix_sum(h, target)

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 10

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    target = n / 2.0
    h0 = fix_sum(h0, target)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = n

    # Phase 1: continuous SA (50% of budget)
    t1 = start + (budget_s - 10) * 0.35
    h_res, c5_res = sa_continuous(h0, target, min(t1, deadline), T_init=0.03)
    if c5_res < best_c5:
        best_c5 = c5_res
        best_h = h_res.copy()

    # Phase 2: binary-targeted SA from current best
    t2 = start + (budget_s - 10) * 0.60
    h_res2, c5_res2 = sa_binary_binary_on(best_h, target, min(t2, deadline))
    if c5_res2 < best_c5:
        best_c5 = c5_res2
        best_h = h_res2.copy()

    # Phase 3: low-temp fine-tuning (remaining budget)
    t3 = deadline
    h_res3, c5_res3 = sa_continuous(best_h, target, t3, T_init=0.003, T_min=1e-8, T_decay=0.99998)
    if c5_res3 < best_c5:
        best_c5 = c5_res3
        best_h = h_res3.copy()

    # Phase 4: try n=48 binary
    if time.time() < deadline - 30:
        n2 = 48
        target2 = n2 / 2.0
        # Upsample h to n=48
        x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0, 2, n2, endpoint=False) + 1.0 / n2
        h_up = np.interp(x_new, x_old, h0, period=2.0)
        h_up = fix_sum(h_up, target2)
        h_upscore = compute_c5(h_up)
        if h_upscore < best_c5:
            best_c5 = h_upscore
            best_h = h_up.copy()
            best_n = n2
        h_res4, c5_res4 = sa_continuous(h_up, target2, deadline, T_init=0.02)
        if c5_res4 < best_c5:
            best_c5 = c5_res4
            best_h = h_res4.copy()
            best_n = n2

    # Final: ensure exact constraint
    final_target = best_n / 2.0
    best_h = fix_sum(best_h, final_target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
