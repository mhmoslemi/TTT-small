import numpy as np
import time


def evaluate_score(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def steepest_descent(h, n, max_iter=800, deadline=None):
    dx = 2.0 / n
    h = h.astype(np.float64).copy()
    score = evaluate_score(h, n)

    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_improve = 1e-14
        best_swap = None
        for i in ones:
            h[i] = 0
            for j in zeros:
                h[j] = 1
                s = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
                imp = score - s
                if imp > best_improve:
                    best_improve = imp
                    best_swap = (i, j)
                h[j] = 0
            h[i] = 1
        if best_swap is None:
            break
        h[best_swap[0]] = 0
        h[best_swap[1]] = 1
        score -= best_improve
    return h, score


def anneal(h, n, iters=8000, t0=0.004, t1=0.00005, deadline=None):
    h = h.astype(np.float64).copy()
    score = evaluate_score(h, n)
    best_score = score
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    for it in range(iters):
        if deadline and time.time() > deadline:
            break
        temp = t0 * (t1 / t0) ** (it / iters)
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        new_score = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)
        delta = new_score - score
        if delta < 0 or np.random.random() < np.exp(-delta / max(temp, 1e-9)):
            score = new_score
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_score


def kick_optimize(h, n, n_kicks=5, kick_sz=3, deadline=None):
    best_h = h.copy()
    best_score = evaluate_score(h, n)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_kicks):
        if deadline and time.time() > deadline:
            break
        hk = h.copy()
        ok = list(ones)
        zk = list(zeros)
        for _ in range(kick_sz):
            ii = ok[np.random.randint(len(ok))]
            jj = zk[np.random.randint(len(zk))]
            hk[ii] = 0
            hk[jj] = 1
            ok = list(np.where(hk == 1)[0])
            zk = list(np.where(hk == 0)[0])
        ho, so = steepest_descent(hk, n, max_iter=300, deadline=deadline)
        if so < best_score:
            best_score = so
            best_h = ho.copy()
    return best_h, best_score
