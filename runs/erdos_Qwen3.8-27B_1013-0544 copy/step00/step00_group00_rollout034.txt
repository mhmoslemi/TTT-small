```python
import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1}"""
    # Binary search on threshold
    lo, hi = -1.0, 1.0  # threshold range
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0.0, 1.0)
```

```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    return np.clip(h - t, 0.0, 1.0)

def top_k_gradient(h, n, k_top=3, tol=1e-6):
    """Gradient of sum of top-k correlations (approximately)"""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(c)
    # Get indices of top correlations
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:k_top]
    # Only include those within tol of max
    active = active[c[active] >= max_val - tol]
    if len(active) == 0:
        active = sorted_idx[:1]
    
    grad = np.zeros(n)
    for m in active:
        klag = m - (n - 1)
        if klag >= 0:
            i1 = np.arange(klag, n)
            grad[i1] += (1.0 - h[i1 - klag])
            i2 = np.arange(0, n - klag)
            grad[i2] += (-h[i2 + klag])
        else:
            kp = -klag
            i1 = np.arange(0, n - kp)
            grad[i1] += (1.0 - h[i1 + kp])
            i2 = np.arange(kp, n)
            grad[i2] += (-h[i2 - kp])
    grad *= dx / len(active)
    return grad, max_val
```

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, target_sum):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    return np.clip(h - t, 0.0, 1.0)

def grad_at_lag(h, n, m):
    """Gradient of correlation at index m with respect to h"""
    n1 = n - 1
    klag = m - n1
    g = np.zeros(n)
    if klag >= 0:
        i1 = np.arange(klag, n)
        g[i1] += (1.0 - h[i1 - klag])
        i2 = np.arange(0, n - klag)
        g[i2] += (-h[i2 + klag])
    else:
        kp = -klag
        i1 = np.arange(0, n - kp)
        g[i1] += (1.0 - h[i1 + kp])
        i2 = np.arange(kp, n)
        g[i2] += (-h[i2 - kp])
    return g

def smooth_grad(h, n, alpha=30.0):
    """Smoothed gradient using log-sum-exp weights"""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    w = np.exp(alpha * (c - c_max))
    w /= np.sum(w)
    g = np.zeros(n)
    for m in range(len(w)):
        if w[m] < 1e-8:
            continue
        g += w[m] * grad_at_lag(h, n, m)
    g *= dx
    return g, c_max
```
