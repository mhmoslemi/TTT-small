from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def project_sum(h, target):
    def f(mu):
        return np.sum(np.clip(h - mu, 0.0, 1.0)) - target
    lo, hi = float(h.min()) - 2.0, float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    err = target - x.sum()
    if abs(err) > 1e-14:
        if err > 0:
            idx = np.argsort(-(1.0 - x))
            for i in idx:
                a = min(err, 1.0 - x[i])
                x[i] += a
                err -= a
                if abs(err) < 1e-15:
                    break
        else:

                a = min(-err, x[i])
                x[i] -= a
                err += a

                    break
