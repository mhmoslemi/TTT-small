import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_c5(h, dx, t):
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp((vals - m) * t)) / len(vals)) / t


def enforce_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_func(p, dx, t, n):
    h = np.clip(p, 0.0, 1.0)
    c5 = smooth_max_c5(h, dx, t)
    penalty = 100.0 * (np.sum(h) - n / 2.0) ** 2
    return c5 + penalty


def interpolate_h(h_src, n_src, n_dst):
    x_old = np.arange(n_src) * (2.0 / n_src)
    x_new = np.arange(n_dst) * (2.0 / n_dst)
    h_new = np.interp(x_new, x_old, h_src, period=2.0)
    return h_new
