import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def smooth_max_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    c = conv * dx
    c_max = float(np.max(c))
    shifted = T * (c - c_max)
    e = np.exp(shifted)
    sum_e = float(np.sum(e))
    smooth_val = c_max + (1.0 / T) * np.log(sum_e)
    weights = e / sum_e
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        w *= dx
        l = k - (n - 1)
        if l == 0:
            grad += w * (1.0 - 2.0 * h)
        elif l > 0:
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] -= w * h[:n - l]
        else:
            lp = -l
            grad[lp:] += w * (1.0 - h[:-lp])
            grad[:-lp] -= w * h[lp:]
    return smooth_val, grad

def project_feasible(y, target_sum):
    def f(lam):
        return float(np.sum(np.clip(y - lam, 0.0, 1.0))) - target_sum
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def slsqp_optimize(h0, n, T, max_iter):
    target = n / 2.0
    def obj(x):
        return smooth_max_and_grad(x, T)[0]
    def jac(x):
        return smooth_max_and_grad(x, T)[1]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=jac, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    return res.x

def proj_grad(h, T, n_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h_curr = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_curr, dx)
    step = 0.05
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        val, grad = smooth_max_and_grad(h_curr, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-8:
            break
        alpha = step
        improved = False
        for _ in range(30):
            h_new = project_feasible(h_curr - alpha * grad, target)
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < best_c5 - 1e-12:
                h_curr = h_new
                best_c5 = new_c5
                best_h = h_new.copy()
                step = min(alpha * 1.4, 0.3)
                improved = True
                break
            alpha *= 0.5
        if not improved:
            step *= 0.7
            if step < 1e-8:
                break
    return best_h, best_c5

def local_search(h, n, target, dx, n_perturb, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for _ in range(n_perturb):
        if time.time() > deadline:
            break
        idx = np.random.randint(0, n, size=max(1, n // 5))
        h_pert = best_h.copy()
        h_pert[idx] += np.random.uniform(-0.3, 0.3, size=len(idx))
        h_pert = project_feasible(h_pert, target)
        c5_p = compute_c5(h_pert, dx)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pert
    return best_h, best_c5
