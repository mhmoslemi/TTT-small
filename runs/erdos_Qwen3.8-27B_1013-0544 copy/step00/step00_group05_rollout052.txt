    return np.sum(x) - _n / 2.0

def _sum_jac(x):
    return np.ones(_n)

def _obj(x):
    return compute_c5(x)

def _obj_smooth(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return soft_max(corr * (2.0 / _n), _T)

def slsqp_hard(h0, maxiter=2000):
    global _n
    _n = len(h0)
    b = Bounds(np.zeros(_n), np.ones(_n))
    c = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        r = minimize(_obj, h0, method='SLSQP', bounds=b, constraints=c,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return repair(r.x)
    except Exception:
        return h0

def slsqp_soft(h0, T, maxiter=2000):
    global _n, _T
    _n = len(h0)
    _T = T
    b = Bounds(np.zeros(_n), np.ones(_n))
    c = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        r = minimize(_obj_smooth, h0, method='SLSQP', bounds=b, constraints=c,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return repair(r.x)
    except Exception:
        return h0

def subgrad_refine(h, steps=50, lr0=0.02):
    n = len(h)
    h = h.copy()
    best_c = compute_c5(h)
    for s in range(steps):
        lr = lr0 / (1.0 + 0.01 * s)
        corr = np.correlate(h, 1.0 - h, mode="full")
        mi = np.argmax(corr)
        k = mi - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            for j in range(n - k):
                grad[j] += 1.0 - h[j + k]
                grad[j + k] -= h[j]
        else:
            kp = -k
            for j in range(kp, n):
                grad[j] += 1.0 - h[j - kp]
                grad[j - kp] -= h[j]
        h_new = repair(h - lr * grad)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            h = h_new
            best_c = c_new
        else:
            lr0 = lr0 * 0.7
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    global _n
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def _TL():
        return budget_s - (time.time() - t0)

    # Initialize
    if initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = make_binary(70, rng)
    n = len(h_best)
    _n = n
    h_best = repair(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: Optimize from initial
    h_opt = slsqp_hard(h_best, 3000)
    c = compute_c5(h_opt)
    if c < c5_best:
        h_best, c5_best = h_opt, c

    h_opt = slsqp_soft(h_best, T=50, maxiter=2000)
    c = compute_c5(h_opt)
    if c < c5_best:
        h_best, c5_best = h_opt, c

    h_opt, c = subgrad_refine(h_best, steps=80, lr0=0.03)
    if c < c5_best:
        h_best, c5_best = h_opt, c

    # Phase 2: Diverse exploration
    candidates = []
    for _ in range(20):
        candidates.append(make_binary(n, rng))
    for _ in range(8):
        candidates.append(np.clip(h_best + rng.randn(n) * 0.25, 0, 1))
    for bs in [2, 3, 5, 7, 11, 17]:
        h_b = np.zeros(n)
        for i in range(0, n, 2 * bs):
            h_b[i:i + bs] = 1.0
        candidates.append(h_b)
    # Random block structures
    for _ in range(5):
        h_b = np.zeros(n)
        pos = 0
        while pos < n:
            bl = rng.randint(2, 15)
            if rng.random() < 0.5:
                h_b[pos:pos + bl] = 1.0
            pos += bl
        candidates.append(h_b)

    for h0 in candidates:
        if _TL() < 40:
            break
        h0 = repair(h0)
        h_opt = slsqp_hard(h0, 800)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c
            h_opt2 = slsqp_soft(h_best, T=80, maxiter=1500)
            c2 = compute_c5(h_opt2)
            if c2 < c5_best:
                h_best, c5_best = h_opt2, c2

    # Phase 3: Subgradient refinement loop
    if _TL() > 60:
        h_ref, c_ref = subgrad_refine(h_best, steps=100, lr0=0.02)
        c = compute_c5(h_ref)
        if c < c5_best:
            h_best, c5_best = h_ref, c
        h_opt = slsqp_hard(h_best, 2000)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c

    # Phase 4: Higher resolution
    if _TL() > 250:
        n2 = n * 2
        h2 = np.repeat(h_best, 2)
        h_opt = slsqp_hard(h2, 4000)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best, n = h_opt, c, n2
            _n = n
            h_opt = slsqp_soft(h_best, T=100, maxiter=3000)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best, c5_best = h_opt, c

        # Continue at higher resolution
        while _TL() > 80:
            h_p = repair(np.clip(h_best + rng.randn(n) * 0.05, 0, 1))
            h_opt = slsqp_hard(h_p, 2000)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best, c5_best = h_opt, c
            else:
                h_ref, c_ref = subgrad_refine(h_p, steps=50, lr0=0.01)
                if c_ref < c5_best:
                    h_best, c5_best = h_ref, c_ref
                else:
                    break

    # Final polish
    if _TL() > 30:
        h_opt = slsqp_soft(h_best, T=200, maxiter=3000)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c
        h_opt = slsqp_hard(h_best, 2000)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c

    h_best = repair(h_best)
    return h_best, compute_c5(h_best), n
