import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
    return float(np.max(corr) * dx)

def local_search_binary(h, time_limit=60.0, start_time=None):
    if start_time is None:
        start_time = time.time()
    n = len(h)
    h = h.copy().astype(np.float64)
    current_c5 = compute_c5(h)
    sweep_count = 0
    while time.time() - start_time < time_limit and sweep_count < 5000:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = current_c5
        best_move = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                if new_c5 < best_c5 - 1e-14:
                    best_c5 = new_c5
                    best_move = (i, j)
        if best_move is None:
            break
        i, j = best_move
        h[i] = 0.0
        h[j] = 1.0
        current_c5 = best_c5
        sweep_count += 1
    return h, current_c5

def simulated_annealing(h, time_limit=100.0, start_time=None, rng=None):
