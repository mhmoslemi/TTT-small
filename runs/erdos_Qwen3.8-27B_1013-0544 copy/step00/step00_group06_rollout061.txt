import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, n, tau):
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full') * dx
    rmax = np.max(r)
    e = np.exp(tau * (r - rmax))
    w = e / e.sum()
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        a, b = max(0, -k), min(n, n - k)
        if a < b:
            g[a:b] += w[idx] * (1.0 - h[a + k:b + k])
        c, d = max(0, k), min(n, n + k)
        if c < d:
            g[c:d] -= w[idx] * h[c - k:d - k]
    return g * dx

def exact_grad(h, n, k_star):
    dx = 2.0 / n
    k = k_star - (n - 1)
    g = np.zeros(n)
    a, b = max(0, -k), min(n, n - k)
    if a < b:
        g[a:b] = 1.0 - h[a + k:b + k]
    c, d = max(0, k), min(n, n + k)
    if c < d:
        g[c:d] -= h[c - k:d - k]
    return g * dx
