def objective_and_grad(h, dx):
    c = np.correlate(h, 1-h, mode="full")
    val = np.max(c) * dx
    # Numerical gradient (or analytical)
    # For the active lag, compute gradient
    active_k = np.argmax(c)
    offset = active_k - (len(h) - 1)  # the actual lag
    n = len(h)
    grad = np.zeros(n)

        if 0 <= i - offset < n:
            g -= h[i - offset]
        grad[i] = dx * g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * dx

def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if diff > 0:
                adj = min(diff, 1.0 - h[i])
            else:
                adj = max(diff, -h[i])
            h[i] += adj
            diff -= adj
            if abs(diff) < 1e-14:
                break
    return h

def local_search(h, dx, rng, max_iters):
