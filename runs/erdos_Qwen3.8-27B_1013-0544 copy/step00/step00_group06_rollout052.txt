def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-10:
        h *= target_sum / s
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)

            idx = np.argsort(-h)
        else:
