import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    """Project h onto {x: 0 ≤ x_i ≤ 1, sum(x) = target_sum}."""
    n = len(h)
    h = np.clip(h, 0, 1)
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Numerically stable log-sum-exp
    max_c = np.max(corr)
    return max_c + np.log(np.sum(np.exp(corr - max_c))) / beta

def gradient_soft_max_c5(h, beta):
    """Gradient of the soft max approximation."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    
    # Weights for soft max
    max_c = np.max(corr)
    exp_c = np.exp(corr - max_c)
    weights = exp_c / np.sum(exp_c) / beta
    
    # Compute gradient: weighted sum of subgradients
    grad = np.zeros(n)
    for m in range(len(corr)):
        if weights[m] < 1e-12:
            continue
        k = m - (n - 1)
        n_lo = max(0, -k)
        n_hi = min(n - 1, n - 1 - k)
        n_vals = np.arange(n_lo, n_hi + 1)
        j_positions = n_vals + k
        grad[j_positions] += weights[m] * (1.0 - h[n_vals])
        grad[n_vals] += weights[m] * (-h[j_positions])
    
    return grad

    n = len(h)
    h = np.array(h, dtype=float)

        proj = np.clip(h - mid, 0, 1)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def gradient_soft_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    exp_beta = np.exp(beta * (corr - max_c))
    weights = exp_beta / np.sum(exp_beta)
    grad = np.zeros(n)
    for m in range(len(corr)):
        w = weights[m]
        if w < 1e-12:

        k = m - (n - 1)
        n_lo = max(0, -k)
        n_hi = min(n - 1, n - 1 - k)

        j_pos = n_vals + k
        grad[j_pos] += w * (1.0 - h[n_vals])
        grad[n_vals] += w * (-h[j_pos])
