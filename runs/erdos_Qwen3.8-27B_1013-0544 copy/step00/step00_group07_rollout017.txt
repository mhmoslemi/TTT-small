import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(x, target_sum):
    """Project x onto {p : 0 <= p[i] <= 1, sum(p) = target_sum}"""
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        t = (lo + hi) / 2
        p = np.clip(x - t, 0, 1)
        s = np.sum(p)
        if s > target_sum:
            lo = t
        else:
            hi = t
    return np.clip(x - (lo + hi) / 2, 0, 1)

def subgradient(c5_full, h, n):
    """Compute subgradient of the max objective."""
    # Find the lag that achieves the maximum
    max_idx = np.argmax(c5_full)
    d = max_idx - (n - 1)  # the lag
    
    grad = np.zeros(n)
    if d >= 0:
        for i in range(n):
            g = 0.0
            if d <= i <= n - 1:  # h[i] appears as h[n+d] with n=i-d
                g += (1 - h[i - d])
            if 0 <= i <= n - 1 - d:  # h[i] appears as (1-h[n]) with n=i
                g += -h[i + d]
            grad[i] = g * (2.0 / n)
    else:
        m = -d
        for i in range(n):
            g = 0.0
            if 0 <= i <= n - 1 - m:  # h[i] appears as h[n-m] with n=i+m
                g += (1 - h[i + m])
            if m <= i <= n - 1:  # h[i] appears as (1-h[n]) with n=i
                g += -h[i - m]
            grad[i] = g * (2.0 / n)
    return grad

def projected_gradient_step(h, n, alpha, max_iter=5000):
    """Projected gradient descent."""
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        c5_full = corr * dx

        if val < best_val:

        else:
            alpha *= 0.7
        
        # Convergence check

    """Project onto {p: 0<=p[i]<=1, sum(p)=target_sum}"""
    lo, hi = -2.0, 2.0

        if np.sum(p) > target_sum:
            lo = t
        else:
            hi = t
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def grad_at_lag(h, n, d):
    """Gradient of the overlap integral at lag d."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if d >= 0:
        for i in range(n):
            g = 0.0
            if d <= i <= n - 1:
                g += 1.0 - h[i - d]
            if 0 <= i <= n - 1 - d:
                g -= h[i + d]
            grad[i] = g
    else:

        for i in range(n):
            g = 0.0
            if 0 <= i <= n - 1 - m:
                g += 1.0 - h[i + m]
            if m <= i <= n - 1:
                g -= h[i - m]
            grad[i] = g
    return grad * dx
