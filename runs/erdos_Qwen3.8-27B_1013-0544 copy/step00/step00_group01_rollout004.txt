            break
        configs.append((ns, interpolate_h(h0, n0, ns)))

    # Time fractions per config: decreasing
    n_cfg = len(configs)
    fracs = [0.0] * n_cfg
    fracs[0] = 0.45
    for i in range(1, n_cfg):
        fracs[i] = 0.20 / (n_cfg - 1) if n_cfg > 1 else 0.0

    T_init = 0.05
    T_floor = 0.00005

    for ci in range(n_cfg):
        if time_left() < 30.0:
            break
        n_grid, h = configs[ci]
        t_start = time.time()
        t_end = min(t_start + fracs[ci] * budget_s, t0 + budget_s - 25.0)
        t_dur = max(1.0, t_end - t_start)

        c = c5_value(h)
        patience = 0

        while time.time() < t_end and time_left() > 20.0:
            t_frac = min(1.0, (time.time() - t_start) / t_dur)
            T = T_init * (T_floor / T_init) ** t_frac

            i = np.random.randint(n_grid)
            j = np.random.randint(n_grid)
            if i == j:
                continue

            if h[i] >= h[j]:
                src, dst = i, j
            else:
                src, dst = j, i

            d_avail = min(h[src], 1.0 - h[dst])
            if d_avail < 1e-7:
                continue

            r = np.random.random()
            if r < 0.55:
                d = np.random.uniform(1e-4, max(5e-4, d_avail * 0.1))
            elif r < 0.85:
                d = np.random.uniform(5e-4, max(0.01, d_avail * 0.4))
            else:
                d = np.random.uniform(0.01, max(0.05, d_avail * 0.8))
            d = min(d, d_avail)

            h[src] -= d
            h[dst] += d

            c_new = c5_value(h)

            if c_new < c:
                c = c_new
                patience = 0
                if c < best_c5:
                    best_c5 = c
                    best_h = h.copy()
            elif T > 1e-9 and np.random.random() < np.exp((c - c_new) / T):
                c = c_new
            else:
                h[src] += d
                h[dst] -= d
                patience += 1

            if patience > 8000:
                T = max(T_init * 0.4, T_floor)
                patience = 0

    # Fine-tuning phase on best solution
    if time_left() > 20.0:
        h = best_h.copy()
        n_grid = len(h)
        c = c5_value(h)
        t_start = time.time()
        t_end = t_start + time_left() * 0.92

        while time.time() < t_end and time_left() > 5.0:
            t_frac = min(1.0, (time.time() - t_start) / max(1.0, t_end - t_start))
            T = 0.003 * (1e-5 / 0.003) ** t_frac

            i = np.random.randint(n_grid)
            j = np.random.randint(n_grid)
            if i == j:
                continue

            if h[i] >= h[j]:
                src, dst = i, j
            else:
                src, dst = j, i

            d_avail = min(h[src], 1.0 - h[dst])
            if d_avail < 1e-7:
                continue

            d = np.random.uniform(1e-5, max(5e-5, d_avail * 0.08))
            d = min(d, d_avail)

            h[src] -= d
            h[dst] += d

            c_new = c5_value(h)

            if c_new < c:
                c = c_new
                if c < best_c5:
                    best_c5 = c
                    best_h = h.copy()
            elif T > 1e-9 and np.random.random() < np.exp((c - c_new) / T):
                c = c_new
            else:
                h[src] += d
                h[dst] -= d

    # Final projection to ensure exact constraint satisfaction
    n_best = len(best_h)
    for _ in range(20):
        best_h = np.clip(best_h, 0.0, 1.0)
        diff = n_best / 2.0 - best_h.sum()
        if abs(diff) < 1e-12:
            break
        best_h += diff / n_best
    best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = c5_value(best_h)
    return (best_h, final_c5, n_best)
