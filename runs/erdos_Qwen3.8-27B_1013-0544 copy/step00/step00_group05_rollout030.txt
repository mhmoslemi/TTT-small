import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr * dx)

def subgradient(h, top_k=5, eps_rel=0.05):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    c5_vals = corr * dx
    current_max = np.max(c5_vals)
    
    # Active lags: within eps of max
    eps = eps_rel * current_max
    active = np.where(c5_vals >= current_max - eps)[0]
    
    grad = np.zeros(n)
    for k_idx in active:
        k = k_idx - (n - 1)
        if k >= 0:
            if n - k > 0:
                grad[k:] += (1 - h[:n-k])
                grad[:n-k] -= h[k:]
        else:
            m = -k
            if n - m > 0:
                grad[:n-m] += (1 - h[m:])
                grad[m:] -= h[:n-m]
    
    return grad * dx, current_max

def project_h(h, n):
    """Project to {0≤h≤1, sum=h = n/2}"""
    target = n / 2.0
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = np.sum(mask)
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0, 1)

    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        
        # Compute subgradient
        grad, current_max = subgradient(h, top_k=5, eps_rel=0.1)

        if gnorm < 1e-12:

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt == 0:
        return h
    h = h.copy()
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def subgrad_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        nk = n - k
        if nk > 0:
            g[k:] += (1.0 - h[:nk])
            g[:nk] -= h[k:]
    else:
        m = -k
        nm = n - m
        if nm > 0:
            g[:nm] += (1.0 - h[m:])
            g[m:] -= h[:nm]
    return g * dx

def subgrad_descent(h0, n_iters, lr_init, seed, time_limit, eps_frac=0.12, top_k=8):
    n = len(h0)
    h = project_h(h0.copy(), n)
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    lr = lr_init
    stall = 0
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
        mx = corr.max()
        thresh = mx * (1.0 - eps_frac)
        active = np.where(corr >= thresh)[0]
        if len(active) > top_k:
            idx = np.argsort(corr[active])[::-1][:top_k]
            active = active[idx]
        grad = np.zeros(n)
        for ki in active:
            grad += subgrad_lag(h, ki - (n - 1))
        grad /= max(1, len(active))
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        h = h - lr * grad / gn
        h = project_h(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        lr *= 0.9997
        if stall > 2000 and lr > 0.003:
            h = best_h + rng.normal(0, 0.03, n)
            h = project_h(h, n)
            stall = 0
            lr = max(lr, 0.03)
    return best_h, best_c5

def downsample2(h):
    n = len(h)
    if n % 2 != 0:
        return None
    hp = (h[0::2] + h[1::2]) / 2.0
    return project_h(hp, n // 2)
