import numpy as np

def compute_crr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return float(np.max(compute_crr(h)) * 2.0 / n)


def smooth_obj(x, beta):
    corr = compute_crr(x)
    mc = float(np.max(corr))
    return mc + float(np.log(np.sum(np.exp(beta * (corr - mc))))) / beta


def smooth_grad(x, beta):
    n = len(x)
    corr = compute_crr(x)
    mc = float(np.max(corr))
    exp_c = np.exp(beta * (corr - mc))
    w = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        if s >= 0:
            if s == 0:
                grad += wk * (1.0 - 2.0 * x)
            else:
                grad[:n - s] += wk * (1.0 - x[s:])
                grad[s:] -= wk * x[:n - s]
        else:
            sp = -s
            grad[sp:] += wk * (1.0 - x[:n - sp])
            grad[:n - sp] -= wk * x[sp:]
    return grad

    return smooth_obj(x, beta)
