These are different in general (unless h is symmetric). So we can't just look at k >= 0.

But wait, let's think about it differently:
f_k = sum_j h[j] - sum_j h[j]*h[j+k]
f_{-k} = sum_j h[j] - sum_j h[j]*h[j-k] = sum_j h[j] - sum_m h[m-k]*h[m] (substituting m=j+k)
= sum_j h[j] - sum_m h[m]*h[m-k]

So f_k and f_{-k} have the same quadratic part (just with different summation ranges). Actually:
sum_j h[j]*h[j+k] where 0<=j<n and 0<=j+k<n
vs
sum_m h[m]*h[m-k] where 0<=m<n and 0<=m-k<n

These are the same sum! (Just a change of variable.) So the quadratic parts are the same.

But the linear parts (sum_j h[j]) are different because the ranges are different:
For f_k: j ranges over [max(0, -k), min(n-1, n-1-k)]
For f_{-k}: m ranges over [max(0, k), min(n-1, n-1+k)]

The size of the range is n-|k| in both cases, but the actual set of indices is shifted.

So f_k - f_{-k} = S_k - S_{-k} where S_k = sum of h over the valid range for lag k.

This means f_k and f_{-k} differ only in their linear parts. The quadratic part is the same.

This is useful! It means we can write:
f_k = L_k - Q_k where Q_k = sum_{valid} h[j]*h[j+k] (same for ±k) and L_k = sum over valid range of h.

    return np.correlate(h, 1 - h, mode='full')

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(correlation_values(h)) * dx

def c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlation_values(h)
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        if 0 <= j + k < n:
            grad[j] += (1 - h[j + k])
        if 0 <= j - k < n:
            grad[j] -= h[j - k]
    return grad * dx

def project_sum(h, target):
    """Project h to have sum = target while staying in [0,1]."""
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Iteratively adjust
    for _ in range(50):
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        adj = diff / np.sum(mask)
        if (diff > 0 and adj > 1.0 - h[mask].min()) or (diff < 0 and adj < -h[mask].min()):
            h[mask] += adj
            h = np.clip(h, 0, 1)
        else:
            h[mask] += adj
            h = np.clip(h, 0, 1)
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
    return h

        return np.max(np.correlate(x, 1 - x, mode='full')) * dx

        corr = np.correlate(x, 1 - x, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

            if 0 <= j + k < n:
                grad[j] += (1 - x[j + k])
            if 0 <= j - k < n:
                grad[j] -= x[j - k]
        return grad * dx

        return np.sum(x) - n / 2

        return np.ones(n)

                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-9})

    h = h_init.copy()
    best_h = h.copy()
    best_val = c5_value(h)
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_init + (T_final - T_init) * frac
        lr_it = lr / (1 + 0.01 * it)
        
        corr = np.correlate(h, 1 - h, mode='full')
        # Smooth max weights
        corr_shifted = corr - np.max(corr)
        weights = np.exp(T * corr_shifted)
        weights /= np.sum(weights)

        for idx in range(n_lags):
            if weights[idx] < 1e-10:
                continue
            k = idx - (n - 1)
            for j in range(n):
                if 0 <= j + k < n:

                if 0 <= j - k < n:
                    grad[j] -= weights[idx] * h[j - k]

        h = h - lr_it * grad
        h = project_sum(h, n / 2)
        
        val = c5_value(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    return best_h, best_val
