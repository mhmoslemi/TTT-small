Let me also add a projected gradient method as an alternative:

For projected gradient:
1. Compute gradient of the (smoothed) objective
2. Take a step: h_new = h - lr * grad
3. Project onto feasible set {0 ≤ h ≤ 1, sum(h) = n/2}

The projection onto {0 ≤ h ≤ 1, sum(h) = target} can be done as follows:
- We want to find h* = argmin ||h - h_new||^2 s.t. 0 ≤ h* ≤ 1 and sum(h*) = target
- This is equivalent to: h*_i = clip(h_new_i - λ, 0, 1) for some λ
- Find λ such that sum(clip(h_new_i - λ, 0, 1)) = target
- Use bisection on λ

    lo, hi = -1.0, 1.0

        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_temp = 0.05

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    return float(dx * np.max(corr))

def smooth_obj(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    scaled = corr * dx / _temp
    m = np.max(scaled)
    return float(_temp * (m + np.log(np.sum(np.exp(scaled - m)))))

def smooth_grad(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    scaled = corr * dx / _temp
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= w.sum()
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        wi = w[s + n - 1] * dx
        if wi < 1e-12:
            continue
        j1s = max(0, -s)
        j1e = min(n, n - s)
        if j1s < j1e:
            grad[j1s:j1e] += wi * (1.0 - h[j1s + s:j1e + s])
        j2s = max(0, s)
        j2e = min(n, n + s)
        if j2s < j2e:
            grad[j2s:j2e] -= wi * h[j2s - s:j2e - s]
    return grad

def sum_eq(h):
    return float(np.sum(h) - len(h) / 2.0)

def sum_eq_jac(h):
    return np.ones(len(h))

def project_feasible(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = hp.sum() - target
    if abs(diff) > 1e-10:
        free = (hp > 1e-8) & (hp < 1.0 - 1e-8)
        if free.sum() > 0:
            hp[free] -= diff / free.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def exact_subgradient(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    active = int(np.argmax(corr))
    s = active - (n - 1)
    grad = np.zeros(n)
    j1s = max(0, -s)
    j1e = min(n, n - s)
    if j1s < j1e:
        grad[j1s:j1e] += (1.0 - h[j1s + s:j1e + s])
    j2s = max(0, s)
    j2e = min(n, n + s)
    if j2s < j2e:
        grad[j2s:j2e] -= h[j2s - s:j2e - s]
    return dx * grad

def upsample(h, factor=2):
    return np.repeat(h, factor)

    np.random.seed(seed)
