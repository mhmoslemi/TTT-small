def project_box_sum(h, S):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(200):

        s = np.sum(np.clip(h + mid, 0, 1))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

    corr = compute_corr(h, dx)
    m = np.max(corr)

        # For very large alpha, just return max
        return float(m)
    shifted = alpha * (corr - m)
    # Clip to avoid overflow
