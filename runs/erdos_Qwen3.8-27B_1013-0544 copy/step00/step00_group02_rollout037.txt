import time

def compute_c5(h):
    """Compute C5 value for given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def soft_max_obj(h, T):
    """Smooth approximation to max of correlation values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    # Shift so values are reasonable for log-sum-exp
    corr_shifted = corr - np.max(corr)
    return T * np.log(np.sum(np.exp(corr_shifted / T))) + np.max(corr)

def gradient_soft_max(h, T):
    """Gradient of soft-max objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    corr_shifted = corr - np.max(corr)
    weights = np.exp(corr_shifted / T) / np.sum(np.exp(corr_shifted / T))
    # Gradient of each corr[k] w.r.t. h
    # Use finite differences for simplicity in small problems
    grad = np.zeros_like(h)
    eps = 1e-7
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full") * dx
        corr_minus = np.correlate(h_minus, 1 - h_minus, mode="full") * dx
        grad_corr = (corr_plus - corr_minus) / (2 * eps)
        grad[i] = np.sum(weights * grad_corr)
    return grad

def project_onto_constraints(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    # Bisection on Lagrange multiplier
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.clip(h - mid, 0, 1)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def project_onto_box_hyperplane(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    # If already feasible, return as is
    if np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - target_sum) < 1e-10:
        return h.copy()
    # Bisection on Lagrange multiplier
    lo, hi = -1.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    return np.clip(h - mid, 0.0, 1.0)

def soft_max_c5(h, T):
    """Compute soft-max approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Scale for numerical stability
    corr_scaled = corr / T
    max_val = np.max(corr_scaled)
    log_sum = max_val + np.log(np.sum(np.exp(corr_scaled - max_val)))
    return T * log_sum, corr

def gradient_c5(h, T):
    """Compute gradient of soft-max C5 w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Soft-max weights
    corr_scaled = corr / T
    max_val = np.max(corr_scaled)
    exps = np.exp(corr_scaled - max_val)
    weights = exps / np.sum(exps)
    
    # For each lag k (corresponding to index j = k + n - 1 in corr),
    # c_k = sum over valid n of h[n+k]*(1-h[n]) * dx
    # d(c_k)/d(h[i]) = dx * [(1-h[i-k]) * 1[i-k valid as n] + (-h[i+k]) * 1[i valid as n]]
    
    grad = np.zeros(n)
    
    for j in range(2 * n - 1):
        k = j - (n - 1)
        w = weights[j] * dx
        if w < 1e-15:
            continue
        
        # Case 1: h[i] appears as h[n+k] where n = i-k
        # Valid n range: if k>=0: [0, n-1-k]; if k<0: [-k, n-1]
        if k >= 0:
            n_lo, n_hi = 0, n - 1 - k
        else:
            n_lo, n_hi = -k, n - 1
        # i = n + k, so i range: [n_lo + k, n_hi + k]
        i_lo1 = n_lo + k
        i_hi1 = n_hi + k
        i_lo1 = max(0, i_lo1)
        i_hi1 = min(n - 1, i_hi1)
        
        # Case 2: h[i] appears as h[n] where n = i
        # Valid i range: [n_lo, n_hi] (same as valid n range)
        i_lo2 = max(0, n_lo)
        i_hi2 = min(n - 1, n_hi)
        
        if i_lo1 <= i_hi1:
            idx = np.arange(i_lo1, i_hi1 + 1)
            grad[idx] += w * (1.0 - h[idx - k])
        
        if i_lo2 <= i_hi2:
            idx = np.arange(i_lo2, i_hi2 + 1)
            grad[idx] += w * (-h[idx + k])
    
    return grad

def optimize_h(h_init, T_schedule, n_steps, target_sum, verbose=False):
    """Run projected gradient descent with soft-max."""
    h = h_init.copy().astype(float)
    h = project_onto_box_hyperplane(h, target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    n = len(h)
    
    for step in range(n_steps):
        T = T_schedule(step)
        grad = gradient_c5(h, T)
        
        # Backtracking line search
        alpha = 1.0
        f_current, _ = soft_max_c5(h, T)
        for _ in range(50):
            h_new = h - alpha * grad
            h_new = project_onto_box_hyperplane(h_new, target_sum)
            f_new, _ = soft_max_c5(h_new, T)
            if f_new < f_current:
                break
            alpha *= 0.5
        
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

        elapsed = time.time() - t_start
        rem = budget_s - elapsed - 20
        if rem < 30:
            break
        alloc = min(rem, 350)
        h_up = upsample(current, target_n)
        h_up = project_bh(h_up, target_n / 2.0)
        # Try several learning rate settings
        h_opt, c_opt = optimize(h_up, alloc * 0.6, n_steps=100000, lr=0.02, T0=0.02, T1=0.002, mom=0.85)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()
            current = h_opt.copy()
        # Refine if time remains
        elapsed = time.time() - t_start
        rem2 = budget_s - elapsed - 20
        if rem2 > 30 and c_opt < best_c5 + 0.001:
            h_opt2, c_opt2 = optimize(h_opt, min(rem2, 150), n_steps=50000, lr=0.01, T0=0.01, T1=0.001, mom=0.9)
            if c_opt2 < best_c5:
                best_c5 = c_opt2
                best_h = h_opt2.copy()
                current = h_opt2.copy()

    # Phase 3: random restarts at best resolution
    elapsed = time.time() - t_start
    rem = budget_s - elapsed - 5
    if rem > 20 and best_h is not None:
        n_best = len(best_h)
        n_restarts = max(1, min(5, int(rem // 40)))
        per_restart = rem / n_restarts
        for r in range(n_restarts):
            if time.time() - t_start > budget_s - 5:
                break
            # Create diverse starting points
            if r == 0:
                h_rand = project_bh(best_h + rng.normal(0, 0.3, n_best), n_best / 2.0)
            elif r == 1:
                h_rand = project_bh(0.5 * best_h + 0.5 * rng.uniform(0, 1, n_best), n_best / 2.0)
            else:
                # Block pattern perturbation
                block = np.tile(np.array([1.0, 0.0]), n_best // 2)
                if len(block) < n_best:
                    block = np.append(block, np.ones(n_best - len(block)))
                h_rand = project_bh(0.3 * block + 0.7 * best_h, n_best / 2.0)
            h_opt, c_opt = optimize(h_rand, per_restart - 5, n_steps=40000, lr=0.03, T0=0.025, T1=0.002, mom=0.85)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()

    # Phase 4: final polish - try small perturbation around best
    elapsed = time.time() - t_start
    rem = budget_s - elapsed - 2
    if rem > 5 and best_h is not None:
        n_best = len(best_h)
        h_polish, c_polish = optimize(best_h, rem, n_steps=20000, lr=0.01, T0=0.005, T1=0.001, mom=0.9)
        if c_polish < best_c5:
            best_c5 = c_polish
            best_h = h_polish.copy()

    # Final validation
    n = len(best_h)
    best_h = project_bh(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
