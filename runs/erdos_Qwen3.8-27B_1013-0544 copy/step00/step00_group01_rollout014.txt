def interpolate_to_size(h, n, new_n):
    """Interpolate h from size n to size new_n, then project to [0,1] with correct sum."""
    if new_n == n:
        return h.copy()
    # Linear interpolation
    x_old = np.linspace(0, 2, n, endpoint=False) + 1.0/n
    x_new = np.linspace(0, 2, new_n, endpoint=False) + 1.0/new_n
    h_interp = np.interp(x_new, x_old, h)

    current = np.sum(h_interp)
    diff = target - current
    # Distribute the difference
    h_interp += diff / new_n
    h_interp = np.clip(h_interp, 0, 1)

    diff = target - np.sum(h_interp)

import numpy as np
import time
import math

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * dx

def interpolate_h(h, n, new_n, rng):
    if new_n == n:
        return h.copy()
    x_old = np.arange(n) / n * 2.0
    x_new = np.arange(new_n) / new_n * 2.0
    h_interp = np.interp(x_new + 1.0/new_n, x_old + 1.0/n, h)
    h_interp = np.clip(h_interp, 0, 1)
    # Adjust sum
    target = new_n / 2.0
    current = h_interp.sum()
    h_interp += (target - current) / new_n
    h_interp = np.clip(h_interp, 0, 1)
    # Fix remaining error
    diff = target - h_interp.sum()
    idx = rng.choice(new_n, size=min(new_n, max(1, int(abs(diff) * new_n))), replace=False)
    for i in idx:
        adj = min(diff, 1 - h_interp[i]) if diff > 0 else max(diff, -h_interp[i])
        h_interp[i] += adj
        diff -= adj
        if abs(diff) < 1e-12:
            break
    return h_interp

def to_binary(h, n, rng):
    if n % 2 != 0:
        # Odd: need sum = n/2 = (n-1)/2 + 0.5
        # Put 0.5 at the position with median value
        median_idx = np.argmax(h)  # or some heuristic
        h_bin = (h > 0.5).astype(float)
        # Count ones
        n_ones = h_bin.sum()
        target_ones = n // 2
        if n_ones > target_ones:
            # Need to turn some 1s to 0s
            one_positions = np.where(h_bin == 1)[0]
            to_zero = one_positions[n_ones - target_ones:]
            h_bin[to_zero] = 0
        elif n_ones < target_ones:
            zero_positions = np.where(h_bin == 0)[0]
            to_one = zero_positions[:target_ones - n_ones]
            h_bin[to_one] = 1
        # Set one position to 0.5
        h_bin[np.argmax(h)] = 0.5
        # Fix: we need exactly target_ones ones and one 0.5
        # Let me redo
        h_bin = np.zeros(n)
        sorted_idx = np.argsort(h)[::-1]
        h_bin[sorted_idx[:target_ones]] = 1
        h_bin[sorted_idx[target_ones]] = 0.5
        return h_bin
    else:
        target = n // 2
        sorted_idx = np.argsort(h)[::-1]
        h_bin = np.zeros(n)
        h_bin[sorted_idx[:target]] = 1

def sa_binary(h, n, time_limit, rng, T_init=0.1, T_final=1e-6):
    """SA on binary sequence."""
    h_curr = h.copy().astype(float)
    dx = 2.0 / n
    c_curr = np.max(np.correlate(h_curr, 1 - h_curr, mode="full")) * dx
    h_best = h_curr.copy()
    c_best = c_curr
    t_start = time.time()
    
    ones = np.where(h_curr == 1)[0]
    n_ones = len(ones)
    
    it = 0
    no_improve = 0
    while time.time() - t_start < time_limit:
        it += 1
        T = T_init * (T_final / T_init) ** (min(it, 100000) / 100000)
        
        # Pick random swap
        i = ones[rng.randint(n_ones)]
        j = rng.randint(n)
        tries = 0
        while h_curr[j] == 1 and tries < 10:
            j = rng.randint(n)
            tries += 1
        if h_curr[j] == 1:
            continue
        
        # Swap
        h_curr[i] = 0
        h_curr[j] = 1
        
        c_new = np.max(np.correlate(h_curr, 1 - h_curr, mode="full")) * dx
        
        if c_new < c_curr or rng.random() < math.exp(-(c_new - c_curr) / max(T, 1e-10)):
            c_curr = c_new
            ones = np.where(h_curr == 1)[0]
            if c_new < c_best:
                c_best = c_new
                h_best = h_curr.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h_curr[i] = 1
            h_curr[j] = 0
            no_improve += 1
    
    return h_best, c_best
