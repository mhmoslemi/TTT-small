from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def coordinate_descent(h, n_sweeps=100, time_limit=100, constraint_penalty=100.0):
    N = len(h)
    target_sum = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    for sweep in range(n_sweeps):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for i in range(N):
            old_val = h[i]
            # Find optimal h[i]
            def obj(x):
                h[i] = x
                c5_val = np.max(np.correlate(h, 1 - h, mode='full'))
                penalty = constraint_penalty * (np.sum(h) - target_sum) ** 2
                return c5_val + penalty
            
            result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-6})
            h[i] = result.x
            if abs(h[i] - old_val) > 1e-8:
                improved = True
        
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c5

import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def coordinate_descent(h, time_limit, penalty_lambda=50.0):
    N = len(h)
    target_sum = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    while time.time() - start < time_limit:
        improved = False
        for i in range(N):
            old = h[i]
            def obj(x):
                h[i] = x
                c5 = np.max(np.correlate(h, 1.0 - h, mode='full'))
                pen = penalty_lambda * (h.sum() - target_sum) ** 2
                return c5 + pen
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-7})
            h[i] = res.x
            if abs(h[i] - old) > 1e-9:
                improved = True
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c5

def project_sum(h, target_sum):
    """Project h to have sum = target_sum while staying in [0,1]."""
    h = h.copy()
    diff = target_sum - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        # Need to increase sum: add to smallest elements
        order = np.argsort(h)
        for idx in order:
            can_add = 1.0 - h[idx]
            add = min(can_add, diff)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-10:
                break
    else:
        # Need to decrease sum: subtract from largest elements
        order = np.argsort(h)[::-1]
        for idx in order:
            can_sub = h[idx]
            sub = min(can_sub, -diff)
            h[idx] -= sub
            diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_sum(h, target_sum):
    h = h.copy()
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)
        for idx in order:
            add = min(1.0 - h[idx], diff)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        for idx in order:
            sub = min(h[idx], -diff)
            h[idx] -= sub
            diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def get_AB(h, i):

    h[i] = 0.0

    h[i] = 1.0

    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
