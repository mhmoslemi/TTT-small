import numpy as np
import time

def compute_c5(h):
    """Compute C5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n_ones):
    """Convert h to binary with exactly n_ones ones."""
    n = len(h)
    if n % 2 == 1:
        n_ones = n // 2
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:n_ones]] = 1.0
    return result

def upsample(h, new_n):
    """Upsample h to new_n points."""
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)

def local_search_sa(h, time_limit, rng, temperature_start=0.01, temperature_end=0.001):
    """Simulated annealing local search for binary h."""
    n = len(h)
    n_ones = int(np.sum(h))
    current = h.copy()
    best = current.copy()
    best_c5 = compute_c5(best)
    current_c5 = best_c5
    
    ones = np.where(current == 1)[0]
    zeros = np.where(current == 0)[0]
    
    start = time.time()
    t = temperature_start
    
    while time.time() - start < time_limit:
        # An.neal temperature
        progress = (time.time() - start) / time_limit
        t = temperature_start * (temperature_end / temperature_start) ** progress
        
        # Pick random swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        current[i] = 0
        current[j] = 1
        
        new_c5 = compute_c5(current)
        delta = new_c5 - current_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / t):
            current_c5 = new_c5
            # Update sets
            ones = np.where(current == 1)[0]
            zeros = np.where(current == 0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = current.copy()

            current[i] = 1
            current[j] = 0
