The bisection solver for each coordinate:

For coordinate p with current value h0:
- Affected lags: m in [m_lo, m_hi]
- For each affected lag m ≠ n-1: c[m] changes linearly by a[m]*x where x = h[p] and a[m] is the coefficient
- For m = n-1 (if in range): quadratic term
- Unaffected lags give a constant lower bound L_min

Binary search on L:
- For a candidate L, find the feasible interval for x:
  - From linear constraints: x_lo = max over a[m]<0 of (L-c0[m])/a[m], x_hi = min over a[m]>0 of (L-c0[m])/a[m]
  - From quadratic: solve x^2 - x + C >= 0
  - From bounds: 0 <= x <= 1
- If feasible interval is non-empty, L is achievable → L_hi = L
- Else L_lo = L

After convergence, find x in the feasible interval at L = L_hi.

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=1000):

    best_c = compute_c5(h, n)
    for alpha in alphas:

        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})

            new_c = compute_c5(new_h, n)

        except Exception:
            pass

def solve_1d_bisection(c, h, p, n):
    """Solve the 1D subproblem for coordinate p using bisection on level L."""
    h0 = h[p]
    
    m_lo = max(0, n - 1 - p)
    m_hi_f = min(2 * n - 2, n - 1 + p)
    m_lo_b = max(0, p)
    m_hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(m_lo, m_lo_b)
    m_hi = max(m_hi_f, m_hi_b)
    
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    valid_f = (jf >= 0) & (jf < n)
    valid_b = (jb >= 0) & (jb < n)
    
    a = np.zeros(len(ms))
    if np.any(valid_f):
        a[valid_f] += (1.0 - h[jf[valid_f]])
    if np.any(valid_b):
        a[valid_b] -= h[jb[valid_b]]

        # So c[n-1](x) = c_base + x - x^2 - h0 + h0^2
        # The a coefficient for this lag in the linear part should be: (1-h0) - h0 = 1-2*h0
        # But we're using the quadratic form, so we handle it separately
        c_diag = c_base[diag_pos[0]]
        a[diag_pos] = 0.0  # Remove from linear (handled by quadratic)
    
    # Unaffected max
    unaff = 0.0
    if m_lo > 0:
        unaff = max(unaff, float(np.max(c[:m_lo])))

    L_hi = cur_max

    pos_mask = a > 1e-15

        x_hi = 1.0
        feasible = True
        
        if np.any(neg_mask):

        if np.any(pos_mask):
            # a > 0: x <= (L - c_base) / a
            needed = (L - c_base[pos_mask]) / a[pos_mask]
            x_hi = min(x_hi, float(np.min(needed)))
        
        # Check zero-coefficient lags
        zero_mask = np.abs(a) <= 1e-15
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        
        if x_lo > x_hi + 1e-14:
            feasible = False
        
        # Quadratic constraint
        if has_diag and feasible:
            # c_diag + x - x^2 - h0 + h0^2 <= L
            # => x^2 - x + (L - c_diag + h0 - h0^2) >= 0
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                # Infeasible: x in (r1, r2)
                # Need [x_lo, x_hi] ∩ [0,1] to not be entirely in (r1, r2)
                # Feasible if x_lo <= r1 or x_hi >= r2 (after clipping to [0,1])
                x_lo_c = max(x_lo, 0.0)
                x_hi_c = min(x_hi, 1.0)
                if x_lo_c < r1 and x_hi_c > r2:
                    # Overlaps with infeasible region, but might still have feasible points
                    # Feasible if x_lo_c <= r1 or x_hi_c >= r2
                    if not (x_lo_c <= r1 or x_hi_c >= r2):
                        feasible = False
                elif x_lo_c >= r1 and x_hi_c <= r2:
                    # Entirely in infeasible region
                    feasible = False
                # else: entirely outside or just touching → feasible
        
        if feasible:
            L_hi = L
        else:
            L_lo = L

    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    
    if np.any(neg_mask):
        needed = (L - c_base[neg_mask]) / a[neg_mask]
        x_lo = max(x_lo, float(np.max(needed)))
    if np.any(pos_mask):
        needed = (L - c_base[pos_mask]) / a[pos_mask]
        x_hi = min(x_hi, float(np.min(needed)))

    x_hi = min(x_hi, 1.0)

        disc = 1.0 - 4.0 * Cq

            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            # Pick x outside (r1, r2)
            # Prefer the side closer to current h0
            if h0 <= r1:
                x_hi = min(x_hi, r1)
            elif h0 >= r2:
                x_lo = max(x_lo, r2)
            else:

                    x_hi = min(x_hi, r1)
                else:

    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    
    if x_lo > x_hi:
        return h0
    
    x_opt = (x_lo + x_hi) / 2.0
    return float(np.clip(x_opt, 0.0, 1.0))

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=800):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def solve_1d_bisection(c, h, p, n):
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi_f = min(2 * n - 2, n - 1 + p)
    m_lo_b = max(0, p)
    m_hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(m_lo, m_lo_b)
    m_hi = max(m_hi_f, m_hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    valid_f = (jf >= 0) & (jf < n)
    valid_b = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(valid_f):
        a[valid_f] += (1.0 - h[jf[valid_f]])
    if np.any(valid_b):
        a[valid_b] -= h[jb[valid_b]]
    c_base = c[ms]
    diag_pos = np.where(ms == (n - 1))[0]
    has_diag = len(diag_pos) > 0
    if has_diag:
        c_diag = c_base[diag_pos[0]]
        a[diag_pos] = 0.0
    unaff = 0.0
    if m_lo > 0:
        unaff = max(unaff, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff = max(unaff, float(c[m_hi + 1:].max()))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo = unaff
    L_hi = cur_max
    for _ in range(55):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo = 0.0
        x_hi = 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            needed = (L - c_base[neg_mask]) / a[neg_mask]
            x_lo = max(x_lo, float(np.max(needed)))
        if feasible and np.any(pos_mask):
            needed = (L - c_base[pos_mask]) / a[pos_mask]
            x_hi = min(x_hi, float(np.min(needed)))
        if x_lo > x_hi + 1e-14:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo = max(x_lo, 0.0)
                xc_hi = min(x_hi, 1.0)
                if xc_lo < r1 and xc_hi > r2:
                    pass
                elif xc_lo >= r1 and xc_hi <= r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    if np.any(neg_mask):
        needed = (L - c_base[neg_mask]) / a[neg_mask]
        x_lo = max(x_lo, float(np.max(needed)))
    if np.any(pos_mask):
        needed = (L - c_base[pos_mask]) / a[pos_mask]
        x_hi = min(x_hi, float(np.min(needed)))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if xc_lo < r1 and xc_hi > r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_cd_exact(h, n, max_passes, deadline):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_1d_bisection(c, h, p, n)
            if abs(x_new - h0) > 1e-12:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def coord_cd_grid(h, n, max_passes, deadline, grid_size=60):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi_f = min(2 * n - 2, n - 1 + p)
            m_lo_b = max(0, p)
            m_hi_b = min(2 * n - 2, p + n - 1)
            m_lo = min(m_lo, m_lo_b)
            m_hi = max(m_hi_f, m_hi_b)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            best_d = 0.0
            best_val = current_max
            diag_mask = (ms == (n - 1))
            for d in d_grid:
                if abs(d) < 1e-14:
                    continue
                nv = c_aff + d * a
                if np.any(diag_mask):
                    nv[diag_mask] -= d * d
                val = max(unaff_max, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def random_start(n, rng):
    h = rng.uniform(0, 1, n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: SLSQP
    h1, c1 = slsqp_run(best_h, n_init, [1000, 10000, 100000, 1000000], 600)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Grid CD for fast convergence
    h2, c2 = coord_cd_grid(best_h, n_init, 150, deadline, grid_size=50)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Try other n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 600, 300, 700]:
        if time.time() > deadline - 200:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = slsqp_run(h_int, n_try, [2000, 20000, 200000], 400)
        h_opt, c_opt = coord_cd_grid(h_opt, n_try, 80, deadline, grid_size=50)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = np.linspace(0, 2, n_try)
    # Phase 4: Multi-restart
    if time.time() < deadline - 100:
        n = len(best_h)
        remaining = deadline - time.time() - 80
        n_restarts = max(2, int(remaining / 70))
        for r in range(n_restarts):
            if time.time() > deadline - 80:
                break
            if r % 3 == 1:
                hp = random_start(n, rng)
            else:
                s = 0.005 + 0.04 * (r / max(1, n_restarts - 1))
                hp = project_feasible(best_h + rng.normal(0, s, n), n)
            h_o, c_o = slsqp_run(hp, n, [10000, 100000], 350)
            h_o, c_o = coord_cd_grid(h_o, n, 60, deadline, grid_size=40)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 5: High-precision bisection CD
    if time.time() < deadline - 20:
        n = len(best_h)
        remaining = deadline - time.time() - 10
        max_passes = max(10, int(remaining / 2.5))
        hf, cf = coord_cd_exact(best_h, n, max_passes, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Phase 6: Final SLSQP + exact CD
    if time.time() < deadline - 8:
        n = len(best_h)
        hf, cf = slsqp_run(best_h, n, [2000000, 5000000], 1000)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = coord_cd_exact(best_h, n, 15, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
