    pref_n = np.cumsum(w_neg)

    LS['dx'] = 2.0 / n
    LS['target'] = target
    LS['beta'] = float(beta)
    LS['rho'] = float(rho)
    LS['lam'] = 0.0
    mx = int(min(maxiter, max(10, int(rem / 2.5))))
    bounds = [(0.0, 1.0)] * n
    try:
        res = so.minimize(l_fg, h, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxls': 40, 'disp': False})
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)

def s_compute(x):
    st = SS
    n = int(st['n'])
    dx = float(st['dx'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_values(c, float(st['beta']))
    g = smooth_grad_from_w(x, w, n, dx)
    st['cache_x'] = x.copy()
    st['cache_g'] = g
    return f, g

def s_obj(x):
    f, _ = s_compute(x)
    return float(f)

def s_jac(x):
    st = SS
    cx = st.get('cache_x')
    if cx is not None and np.array_equal(x, cx):
        return st.get('cache_g').copy()
    _, g = s_compute(x)
    return g

def s_con_fun(x):
    return float(x.sum() - SS['target'])

def s_con_jac(x):
    return np.ones(int(SS['n']), dtype=float)

def refine_slsqp(h, beta, maxiter, deadline):
    h = np.asarray(h, dtype=float)
    n = len(h)
    target = 0.5 * n
    rem = deadline - time.time()
    if rem < 6.0:
        return h, erdos_c5(h, n)
    SS['n'] = n
    SS['dx'] = 2.0 / n
    SS['target'] = target
    SS['beta'] = float(beta)
    SS['cache_x'] = None
    SS['cache_g'] = None
    mx = int(min(maxiter, max(6, int(rem / 8.0))))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': s_con_fun, 'jac': s_con_jac}]
    try:
        res = so.minimize(s_obj, h, method='SLSQP', jac=s_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': mx, 'iprint': 0, 'ftol': 1e-14})
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)

def pair_line(h, n, i, j, cur, deadline):
    hi0, hj0 = h[i], h[j]
    lo = max(-hi0, hj0 - 1.0)
    hi = min(1.0 - hi0, hj0)
    if hi - lo < 1e-13:
        return h, cur
    best_d, best_c = 0.0, cur
    if time.time() > deadline:
        return h, cur
    def _eval(d):
        a, b = hi0 + d, hj0 - d
        if a < -1e-12 or a > 1.0 + 1e-12 or b < -1e-12 or b > 1.0 + 1e-12:
            return 1e100
        hh = h.copy()
        hh[i], hh[j] = a, b
        return erdos_c5(hh, n)
    c0 = _eval(lo)
    if c0 < best_c:
        best_c, best_d = c0, lo
    c1 = _eval(hi)
    if c1 < best_c:
        best_c, best_d = c1, hi
    try:
        if time.time() < deadline - 0.2:
            rs = so.minimize_scalar(_eval, bounds=(lo, hi), method='bounded',
                                    options={'maxiter': 20, 'xatol': 1e-10})
            c2 = _eval(float(rs.x))
            if c2 < best_c:
                best_c, best_d = c2, float(rs.x)
    except Exception:
        pass
    if best_c < cur - 1e-14:
        hh = h.copy()
        hh[i] += best_d
        hh[j] -= best_d
        return hh, best_c
    return h, cur

def pair_polish(h, n, deadline, seed, max_rounds=5):
    rng = np.random.RandomState(int(seed))
    h = np.asarray(h, dtype=float)
    cur = erdos_c5(h, n)
    m = 30 if n <= 500 else 20
    frac = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if frac.size:
        score = np.minimum(h[frac], 1.0 - h[frac])
        order = frac[np.argsort(score)[::-1]]
    else:
        order = np.array([], dtype=int)
    if order.size >= m:
        top = order[:m]
    else:
        need = min(m - int(order.size), n - int(order.size))
        if need > 0:
            extra = rng.choice(n, size=need, replace=False)
            top = np.unique(np.concatenate((order, extra)))
        else:
            top = order
    if top.size < 2:
        return h, cur
    for _ in range(max_rounds):
        improved = False
        pairs = [(int(top[a]), int(top[b]))
                 for a in range(len(top)) for b in range(a + 1, len(top))]
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline - 3.0:
                return h, erdos_c5(h, n)
            hh, cc = pair_line(h, n, i, j, cur, deadline - 3.0)
            if cc < cur - 1e-14:
                h, cur, improved = hh, cc, True
        if not improved:
            break
    return h, cur

def resample_n(h, n, nh):
    x = np.linspace(0.0, 2.0, int(n))
    y = np.linspace(0.0, 2.0, int(nh))
    hh = np.interp(y, x, np.asarray(h, dtype=float))
    return project_box_sum(hh, 0.5 * nh)

def swap_perturb(h, n, rng, n_swaps):
    h = h.copy()
    for _ in range(n_swaps):
        i, j = rng.choice(n, 2, replace=False)
        h[i], h[j] = h[j], h[i]
    return h

def sine_perturb(h, n, rng, amp):
    f = rng.uniform(0.5, 5.0)
    phi = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 2, n)
    h2 = h + amp * np.sin(np.pi * f * x + phi)
    return project_box_sum(h2, 0.5 * n)

def push_binary(h, n, rng, frac_push=0.3):
    """Push a random fraction of values towards 0 or 1."""
    h = h.copy()
    idx = rng.choice(n, int(n * frac_push), replace=False)
    for i in idx:
        if h[i] < 0.5:
            h[i] = max(0.0, h[i] - 0.4)
        else:
            h[i] = min(1.0, h[i] + 0.4)
    return project_box_sum(h, 0.5 * n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(10.0, float(budget_s) - 5.0)
    deadline = t0 + total
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 400
        h = np.full(n, 0.5, dtype=float)
    if n < 2:
        n = 400
        h = np.full(n, 0.5, dtype=float)
    if n > 800:
        nold = n
        n = 800
        h = np.interp(np.linspace(0.0, 2.0, n), np.linspace(0.0, 2.0, nold), h)
    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_n = n
    best_c = erdos_c5(best_h, best_n)
    rng = np.random.RandomState(int(seed))

    # Phase 1: Initial L-BFGS refinement (20% time)
    end1 = t0 + 0.20 * total
    for beta, mx in [(200, 90), (2000, 80), (10000, 70), (50000, 50)]:
        if time.time() > end1 - 4.0:
            break
        cand, c = refine_lbfgs(best_h, beta, 0.0, mx, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand.copy(), c

    # Phase 2: Iterated local search (47% time)
    end2 = t0 + 0.67 * total
    no_improve = 0
    while time.time() < end2 - 8.0:
        r = rng.random()
        if r < 0.40:
            h_pert = swap_perturb(best_h, best_n, rng, rng.randint(2, 9))
        elif r < 0.70:
            h_pert = sine_perturb(best_h, best_n, rng, rng.uniform(0.02, 0.10))
        elif r < 0.85:
            h_pert = push_binary(best_h, best_n, rng, rng.uniform(0.15, 0.4))
        else:
            h_pert = project_box_sum(
                best_h + rng.uniform(0.01, 0.06) * rng.randn(best_n), 0.5 * best_n)
        beta = rng.choice([2000.0, 5000.0, 10000.0, 30000.0])
        mx = rng.randint(25, 50)
        cand, c = refine_lbfgs(h_pert, beta, 0.0, mx, end2 - 8.0)
        if c < best_c - 1e-14:
            best_h, best_c = cand.copy(), c
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                no_improve = 0
        if time.time() > end2 - 30.0:
            break

    # Phase 3: Multi-resolution (18% time)
    end3 = t0 + 0.85 * total
    for nh in [600, 800, 300]:
        if time.time() > end3 - 25.0 or nh == best_n:
            continue
        hh = resample_n(best_h, best_n, nh)
        hh, c = refine_lbfgs(hh, 5000.0, 0.0, 50, end3 - 10.0)
        if c < best_c + 1e-3:
            hh, c = refine_lbfgs(hh, 100000.0, 0.0, 40, end3 - 5.0)
            if c < best_c - 1e-14:
                best_h, best_n, best_c = hh.copy(), nh, c

    # Phase 4: Final polish (remaining time)
    end4 = deadline - 4.0
    if best_n <= 800 and time.time() < end4 - 30.0:
        best_h, best_c = pair_polish(best_h, best_n, end4 - 25.0, seed, max_rounds=6)
    if time.time() < end4 - 15.0:
        cand, c = refine_lbfgs(best_h, 500000.0, 0.0, 30, end4)
        if c < best_c - 1e-15:
            best_h, best_c = cand.copy(), c
    if time.time() < end4 - 8.0:
        cand, c = refine_slsqp(best_h, 1000000.0, 22, end4)
        if c < best_c - 1e-15:
            best_h, best_c = cand.copy(), c
    if time.time() < end4 - 3.0 and best_n <= 500:
        best_h, best_c = pair_polish(best_h, best_n, end4 - 1.0, seed + 7, max_rounds=3)

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
