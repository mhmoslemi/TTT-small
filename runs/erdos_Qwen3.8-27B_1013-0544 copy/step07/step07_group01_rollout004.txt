def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(150):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def weighted_grad(h, n, w):
    dx = 2.0 / n
    grad = np.zeros(n)
    for li in range(2 * n - 1):
        wi = w[li]
        if wi < 1e-14:
            continue
        k = li - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                grad[:m] += wi * (1.0 - h[k:k+m])
                if k > 0:
                    grad[k:] -= wi * h[:m]
        else:
            ll = -k
            m = n - ll
            if m > 0:
                grad[ll:] += wi * (1.0 - h[:m])
                grad[:m] -= wi * h[ll:ll+m]
    return grad * dx

def wgd(h, n, deadline, alpha_init=50, alpha_max=200000, mu_init=0.82, mu_max=0.95):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    n_lags = 2 * n - 1
    lr = 1.0 / (alpha_init * dx + 1e-10)
    momentum = np.zeros(n)
    stall = 0
    step = 0
    total_steps = 400000
    while time.time() < deadline:
        step += 1
        frac = min(step / max(total_steps, 1), 1.0)
        alpha = alpha_init * (alpha_max / alpha_init) ** frac
        mu = mu_init + (mu_max - mu_init) * frac
        corr = np.correlate(h, 1 - h, mode='full') * dx
        cmax = corr.max()
        sh = alpha * (corr - cmax)
        sh = sh - sh.max()
        w = np.exp(sh)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(n_lags) / n_lags
        else:
            w /= ws
        grad = weighted_grad(h, n, w)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        momentum = mu * momentum - lr * grad
        h_new = project_feasible(h + momentum, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0

        h = h_new
