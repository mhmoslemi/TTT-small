def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y2 = np.clip(x - mid, 0.0, 1.0)
        if float(y2.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(3):
        if abs(diff) < 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
        else:

    return y

    dx = 2.0 / n

    cur = float(np.max(R))

        if time.time() > deadline:
            break
        g = grad_smooth_max(h, R, beta)
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h > 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        gi = g[zeros]
        gj = g[ones]

        top_j = ones[np.argsort(gj)[::-1][:K]]
        best_c = cur
        best_swap = None
        for i in top_i:
            for j in top_j:

                    continue
                if time.time() > deadline:
                    break
                hh = h.copy()
                hh[i] = 1.0
                hh[j] = 0.0
                cc = float(np.max(np.correlate(hh, 1.0 - hh, mode='full')) * dx)

                    best_swap = (i, j)

        h[best_swap[1]] = 0.0
        R = np.correlate(h, 1.0 - h, mode='full') * dx

    return h, cur

        return smooth_max_obj(x, n, beta)

        return smooth_max_grad(x, n, beta)
    
    rem = deadline - time.time()

    try:

        h = project_box_sum(res.x, target)
    except Exception:
        pass
    return h, erdos_c5(h, n)
