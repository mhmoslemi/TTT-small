from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def grad_lse(h, n, alpha):
    """Gradient of LSE(alpha * correlate(h, 1-h) * dx)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    s = alpha * corr * dx
    m = s.max()
    w = np.exp(s - m)
    w = w / w.sum()
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[n - 1 + k] * alpha * dx
        if wk < 1e-18:
            continue
        lo = max(0, k)
        hi = min(n, n + k)
        if lo < hi:
            grad[lo:hi] += wk * (1 - h[lo - k:hi - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 + k:hi2 + k]
    return grad

def proj_grad_descent(h, n, alpha, step_init, max_iter, deadline, ftol=1e-10):
    """Projected gradient descent on LSE objective."""
    h = h.copy()
    step = step_init
    prev_val = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = grad_lse(h, n, alpha)
        # Projected gradient step
        h_new = h - step * g
        h_new = project_feasible(h_new, n)
        val = compute_c5(h_new, n)
        if val < prev_val - ftol:
            h = h_new
            prev_val = val
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.5
            if step < 1e-10:
                break
            h_new = h - step * g
            h_new = project_feasible(h_new, n)
            h = h_new
            prev_val = compute_c5(h, n)
    return h

def binary_sa(h_init, n, deadline, max_iters=50000):
    """Simulated annealing on binary swaps."""
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    
    m_range = np.arange(-(n - 1), n)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    T_start = 0.01
    T_end = 1e-8
    n_total = max_iters
    cool_rate = (T_end / T_start) ** (1.0 / max(n_total, 1))
    T = T_start
    
    rng = np.random.RandomState(42)
    
    for it in range(n_total):
        if time.time() > deadline:
            break
        T *= cool_rate
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        # Compute delta in correlation
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < len(delta):
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < len(delta):
            delta[k2] += 1.0
        
        nc = (corr + delta).max() * dx
        delta_c = nc - cur_c
        
        if delta_c < 0 or (T > 1e-9 and rng.random() < np.exp(-delta_c / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            ones[j], zeros[i] = i, j  # swap in lists
            # Actually need to fix the lists properly
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()

            if s < 1e-14:

    m = s.max()
    w = np.exp(s - m)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(len(w))
        w[n-1] = 1.0
    else:
        w = w / wsum
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[n - 1 + k] * alpha * dx
        if wk < 1e-16:

        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 + k:hi2 + k]
    return grad

    step = 0.05
    prev = compute_c5(h, n)
    stall = 0
    for it in range(max_iter):

            break

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def grad_lse(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    s = alpha * corr * dx
    m = s.max()
    w = np.exp(s - m)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(w)); w[n - 1] = 1.0
    else:
        w = w / ws
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[n - 1 + k] * alpha * dx
        if wk < 1e-16:
            continue
        lo = max(0, k); hi = min(n, n + k)
        if lo < hi:
            grad[lo:hi] += wk * (1.0 - h[lo - k:hi - k])
        lo2 = max(0, -k); hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 + k:hi2 + k]
    return grad

def pgd(h, n, alpha, deadline, max_iter=5000):
    h = h.copy()
    step = 0.05
    prev = compute_c5(h, n)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = grad_lse(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = project_feasible(h - step * g, n)
        val = compute_c5(h_new, n)
        if val < prev - 1e-15:
            h = h_new; prev = val; stall = 0
            if step < 0.15:
                step *= 1.04
        else:
            step *= 0.4
            stall += 1
            if stall > 80 or step < 1e-13:
                break
            h_new = project_feasible(h - step * g, n)
            h = h_new; prev = compute_c5(h, n)
    return h

def swap_delta(h, i, j, n, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < len(delta):
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < len(delta):
        delta[k2] += 1.0
    return delta

def binary_sa(h_init, n, deadline, max_iters=60000, seed=0):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy(); best_c = cur_c
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = 0.004
    cool = (1e-10 / T) ** (1.0 / max(max_iters, 1))
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T *= cool
        ii = rng.randint(len(zeros))
        ij = rng.randint(len(ones))
        i = zeros[ii]; j = ones[ij]
        delta = swap_delta(h, i, j, n, m_range)
        nc = (corr + delta).max() * dx
        dc = nc - cur_c
        if dc < 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            ones[ij] = i; zeros[ii] = j
            if cur_c < best_c:
                best_c = cur_c; best_h = h.copy()
    return best_h, best_c

def lse_slsqp(h, n, alpha, deadline, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(x_new, x_old, h), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng_main = np.random.RandomState(seed)

    try:

        n = len(h0)
    except:
        n = 400
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Multi-resolution binary SA (20%)
    p1 = t0 + 0.20 * total
    # Try lower resolution
    n_low = max(100, n // 2)
    h_low = to_binary(h0, n)
    x_map = np.linspace(0, 2, n)
    h_low = project_feasible(np.interp(np.linspace(0, 2, n_low), x_map, h_low), n_low)
    hb_l, cb_l = binary_sa(h_low, n_low, p1 * 0.4, max_iters=40000, seed=seed)
    if cb_l < best_c:
        h_up = upsample(hb_l, n_low, n)
        cb_up = compute_c5(h_up, n)
        if cb_up < best_c:
            best_c, best_h = cb_up, h_up
    # Binary SA at full resolution
    for att in range(3):
        if time.time() > p1 - 8:
            break
        pert = 0.015 * rng_main.randn(n)
        hb, cb = binary_sa(project_feasible(best_h + pert, n), n, p1 - 8 - att * 4, seed=seed + 10 + att)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    # Phase 2: Projected gradient LSE (30%)
    p2 = t0 + 0.50 * total
    alphas = np.logspace(1.5, 5.5, 25)
    h = best_h.copy()
    for a in alphas:
        if time.time() > p2 - 8:
            break
        h = pgd(h, n, int(a), p2 - 8, max_iter=4000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: SLSQP LSE at high alpha (15%)
    p3 = t0 + 0.65 * total
    for a in [100000, 500000, 2000000, 10000000]:
        if time.time() > p3 - 6:
            break
        h = lse_slsqp(best_h, n, a, p3 - 6, maxiter=2500)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Binarize + aggressive SA (20%)
    p4 = t0 + 0.85 * total
    hb = to_binary(best_h, n)
    for att in range(3):
        if time.time() > p4 - 5:
            break
        hb, cb = binary_sa(hb, n, p4 - 5 - att * 3, max_iters=100000, seed=seed + 200 + att * 7)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    p5 = dl
    for a in [50000, 200000, 1000000]:
        if time.time() > p5 - 10:

        c = compute_c5(h, n)
        if c < best_c:

    if time.time() < p5 - 5:
        h = lse_slsqp(best_h, n, 5000000, p5 - 5, maxiter=2000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
