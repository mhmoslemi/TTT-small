import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_passes=200, deadline=None):
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = compute_corr(h)
        improved = False
        for i in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[i]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            k = np.arange(2 * n - 1) - (n - 1)
            v1 = (i - k >= 0) & (i - k < n)
            v2 = (i + k >= 0) & (i + k < n)
            a = np.zeros(2 * n - 1)
            a[v1] += (1.0 - h[i - k[v1]])
            a[v2] -= h[i + k[v2]]
            lo, hi = -h0, 1.0 - h0
            for _ in range(50):
                mid = (lo + hi) * 0.5
                vals = c + mid * a
                vals[n - 1] -= mid * mid
                mmax = vals.max()
                active = np.abs(vals - mmax) < 1e-9
                derivs = a[active].copy()
                if active[n - 1]:
                    derivs[active[n - 1] == True] = derivs
                    idx = np.where(active)[0]
                    derivs = a[idx]
                    if n - 1 in idx:
                        derivs[idx == n - 1] = a[n - 1] - 2.0 * mid
                sg_lo = derivs.min()
                sg_hi = derivs.max()
                if sg_lo > 1e-12:
                    hi = mid
                elif sg_hi < -1e-12:
                    lo = mid
                else:
                    break
            best_d = (lo + hi) * 0.5
            def fval(d):
                v = c + d * a
                v[n - 1] -= d * d
                return v.max()
            best_v = fval(0.0)
            for d in [lo, hi, best_d, (lo+hi)*0.25, (lo+hi)*0.75]:
                d = float(np.clip(d, lo, hi))
                v = fval(d)
                if v < best_v:
                    best_v = v
                    best_d = d
            if abs(best_d) > 1e-10:
                h[i] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgradient_step(h, n, eta, deadline=None):
    c = compute_corr(h)
    mmax = c.max()
    active = c >= mmax - 1e-8
    n_active = active.sum()
    dx = 2.0 / n
    sg = np.zeros(n)
    for i in range(n):
        k = np.arange(2 * n - 1) - (n - 1)
        v1 = (i - k >= 0) & (i - k < n)
        v2 = (i + k >= 0) & (i + k < n)
        a = np.zeros(2 * n - 1)
        a[v1] += (1.0 - h[i - k[v1]])
        a[v2] -= h[i + k[v2]]
        if a[n-1] != 0 and active[n-1]:
            a[n-1] -= 2.0 * 0  # linear approx, ignore quadratic
        sg[i] = a[active].sum() / n_active
    h_new = h - eta * sg
    h_new = project_feasible(h_new, n)
    return h_new, compute_c5(h_new, n)

def slsqp_smooth(h, n, alpha, maxiter=500):
    def obj(x):
        c = compute_corr(x)
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * (2.0 / n)
    def obj_grad(x):
        c = compute_corr(x)
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        exp_shifted = np.exp(shifted - mx)
        weights = exp_shifted / exp_shifted.sum()
        dc = weights.copy()
        k = np.arange(2 * n - 1) - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            v1 = (i - k >= 0) & (i - k < n)
            v2 = (i + k >= 0) & (i + k < n)
            a = np.zeros(2 * n - 1)
            a[v1] += (1.0 - x[i - k[v1]])
            a[v2] -= x[i + k[v2]]
            grad[i] = (dc * a).sum()
        grad *= (2.0 / n) * (1.0 + alpha / (alpha * 0 + 1.0))
        return grad * (2.0 / n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n), compute_c5(project_feasible(res.x, n), n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def random_binary(n, rng):
    h = np.zeros(n)
    indices = rng.choice(n, size=n // 2, replace=False)
    h[indices] = 1.0
    return project_feasible(h, n)

def structured_start(n, rng, typ=0):
    if typ == 0:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * np.arange(n) / n * rng.randint(2, 8))
    elif typ == 1:

    elif typ == 2:
        h = np.zeros(n)
        block = max(1, n // (2 * rng.randint(3, 8)))
        for i in range(n):
            h[i] = 1.0 if (i % (2 * block)) < block else 0.0
    else:
        h = rng.uniform(0, 1, n)
    return project_feasible(h, n)
