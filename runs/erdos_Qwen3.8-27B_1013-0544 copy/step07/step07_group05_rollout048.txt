import time
import numpy as np
import scipy.optimize as so

LS = {'n': 1, 'dx': 1.0, 'target': 0.5, 'beta': 1.0}


def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(5):
        if abs(diff) < 1e-14:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y


def softmax_vals(c, beta):
    m = float(np.max(c))
    z = beta * (c - m)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    if ws <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    return m + np.log(ws) / beta, ew / ws


def smooth_grad(h, w, n, dx):
    mid = n - 1
    wp = np.zeros(n)
    wn = np.zeros(n)
    wp[1:] = w[mid + 1:]
    wn[1:] = w[:mid][::-1]
    s = wp + wn
    pp = np.cumsum(wp)
    pn = np.cumsum(wn)
    idx = np.arange(n)
    consts = pp + pn[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx


def l_fg(x):
    st = LS
    n = int(st['n'])
    dx = float(st['dx'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    _, w = softmax_vals(c, float(st['beta']))
    return float(np.dot(c, w)), smooth_grad(x, w, n, dx)


def refine_lbfgs(h, beta, maxiter, deadline):
    h = np.asarray(h, dtype=float)
    n = len(h)
    target = 0.5 * n
    rem = deadline - time.time()
    if rem < 3.0:
        return h, erdos_c5(h, n)
    LS['n'], LS['dx'], LS['target'], LS['beta'] = n, 2.0 / n, target, float(beta)
    mx = int(min(maxiter, max(10, int(rem / 4.0))))
    bounds = [(0.0, 1.0)] * n
    try:
        res = so.minimize(l_fg, h, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 30, 'disp': False})
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)


def active_set_step(h, n, deadline, n_iter=200):
    """Projected subgradient on active set with backtracking."""
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    target = 0.5 * n
    dx = 2.0 / n
    c = erdos_c5(h, n)
    step = 0.005
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        mx = float(np.max(corr))
        active = (corr > mx - 1e-6)
        w = active.astype(float)
        w /= w.sum()
        g = smooth_grad(h, w, n, dx)
        h_new = project_box_sum(h - step * g, target)
        nc = erdos_c5(h_new, n)
        if nc < c - 1e-15:
            h = h_new
            c = nc
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.7
            if step < 1e-6:
                break
    return h, c


def pair_eval(h, i, j, delta, n):
    a, b = h[i] + delta, h[j] - delta
    if a < -1e-12 or a > 1.0 + 1e-12 or b < -1e-12 or b > 1.0 + 1e-12:
        return 1e100
    hh = h.copy()
    hh[i], hh[j] = a, b
    return erdos_c5(hh, n)


def pair_polish(h, n, deadline, max_sweeps=8):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    cur = erdos_c5(h, n)
    n_inner = min(n, 300)
    for sw in range(max_sweeps):
        if time.time() > deadline - 5.0:
            break
        improved = False
        idx = np.arange(n_inner)
        for a in range(n_inner):
            if time.time() > deadline - 5.0:
                return h, erdos_c5(h, n)
            for b in range(a + 1, n_inner):
                i, j = int(idx[a]), int(idx[b])
                hi0, hj0 = h[i], h[j]
                lo = max(-hi0, hj0 - 1.0)
                hi = min(1.0 - hi0, hj0)
                if hi - lo < 1e-12:
                    continue
                best_d, best_c = 0.0, cur
                for tc in (lo, hi, 0.5 * (lo + hi)):
                    cval = pair_eval(h, i, j, tc, n)
                    if cval < best_c:
                        best_c, best_d = cval, tc
                if best_c < cur - 1e-14:
                    h[i] += best_d
                    h[j] -= best_d
                    h = np.clip(h, 0.0, 1.0)
                    cur = best_c
                    improved = True
        if not improved:
            break
    return h, erdos_c5(h, n)


def interp_to_n(h, n, nh):
    hh = np.interp(np.linspace(0.0, 2.0, int(nh)), np.linspace(0.0, 2.0, int(n)), h)
    return project_box_sum(hh, 0.5 * nh)
