from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradients(h, n):
    """Compute gradients of all f_k at h. Returns (2n-1, n) array."""
    dx = 2.0 / n
    n_lags = 2 * n - 1
    grads = np.zeros((n_lags, n))
    for k in range(n_lags):
        lag = k - (n - 1)
        if lag >= 0 and lag < n:
            grads[k, :n-lag] = (1 - h[lag:]) * dx
            grads[k, lag:] = -h[:n-lag] * dx
        elif lag < 0 and -lag < n:
            ll = -lag
            grads[k, ll:] = (1 - h[:n-ll]) * dx
            grads[k, :n-ll] = -h[ll:] * dx
    return grads

def lp_cutting_plane(h, n, deadline, max_iters=500):
    """Solve using LP cutting plane method."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        
        # Compute overlaps and gradients
        corr = np.correlate(h, 1 - h, mode='full') * dx
        grads = compute_gradients(h, n)
        intercepts = corr - grads @ h  # shape (2n-1,)
        
        # Solve LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
        constraints.append(grads @ h_var + intercepts <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(h_var.value, 0, 1)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        # Update h (use the LP solution, possibly blended with current)
        h = 0.5 * h_new + 0.5 * h
        h = project_feasible(h, n)
        
        # Convergence check
        if t_var.value is not None and abs(t_var.value - best_c) < 1e-7:
            break
    
    return best_h, best_c

    return np.clip(h, 0.0, 1.0)

def get_grads_and_intercepts(h, n):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    corr = np.correlate(h, 1 - h, mode='full') * dx
    grads = np.zeros((n_lags, n))
    for k in range(n_lags):
        lag = k - (n - 1)
        if lag >= 0 and lag < n:
            grads[k, :n-lag] = (1 - h[lag:]) * dx
            grads[k, lag:] = -h[:n-lag] * dx
        elif lag < 0 and -lag < n:
            ll = -lag
            grads[k, ll:] = (1 - h[:n-ll]) * dx
            grads[k, :n-ll] = -h[ll:] * dx
    intercepts = corr - grads @ h
    return grads, intercepts

def lp_refine(h, n, deadline, max_iters=1000, top_m=None):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    h = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        grads, intercepts = get_grads_and_intercepts(h, n)
        n_lags = 2 * n - 1
        
        if top_m and top_m < n_lags:
            top_idx = np.argpartition(intercepts, -top_m)[-top_m:]
            g_sub = grads[top_idx]
            b_sub = intercepts[top_idx]
        else:
            g_sub = grads
            b_sub = intercepts
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
        constraints.append(g_sub @ h_var + b_sub <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        h_new = np.clip(h_var.value, 0, 1)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        # Blend: use LP solution with some momentum toward current
        h = 0.3 * h_new + 0.7 * h
        h = project_feasible(h, n)
        
        if abs(t_var.value - best_c) < 1e-8:
            break
    
    return best_h, best_c

def project_feasible(h, n):
    target = n / 2.0

    for _ in range(200):

        if abs(diff) < 1e-15:

        if diff > 0:

            if s < 1e-14:
                break

            if s < 1e-14:

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def _corr_and_grads(h, n):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    corr = np.correlate(h, 1 - h, mode='full') * dx
    grads = np.zeros((n_lags, n))
    for k in range(n_lags):
        lag = k - (n - 1)
        if lag >= 0 and lag < n:
            grads[k, :n - lag] = (1.0 - h[lag:]) * dx
            grads[k, lag:] = -h[:n - lag] * dx
        elif lag < 0 and -lag < n:
            ll = -lag
            grads[k, ll:] = (1.0 - h[:n - ll]) * dx
            grads[k, :n - ll] = -h[ll:] * dx
    return corr, grads

_gs = {'n': 400, 'alpha': 1000.0}

def _s_obj(x):
    n = _gs['n']
    alpha = _gs['alpha']
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    n = _gs['n']
    alpha = _gs['alpha']
    dx = 2.0 / n
    n_lags = 2 * n - 1
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    p = np.exp(sh - sm)

    grad = np.zeros(n)
    for k in range(n_lags):
        w = p[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0 and lag < n:
            grad[:n - lag] += w * (1.0 - x[lag:])
            grad[lag:] += -w * x[:n - lag]
        elif lag < 0 and -lag < n:
            ll = -lag
            grad[ll:] += w * (1.0 - x[:n - ll])
            grad[:n - ll] += -w * x[ll:]
    return grad * dx

    return x.sum() - _gs['n'] / 2.0

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:
        return h
