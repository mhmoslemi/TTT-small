    return float(np.max(c) * dx)

def compute_corr(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(x, alpha, penalty):
    n = len(x)
    dx = 2.0 / n
    target = n / 2.0
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    logsum = sm + np.log(np.exp(sh - sm).sum())
    f = m + logsum / alpha
    s = x.sum() - target
    f += 0.5 * penalty * s * s
    p = np.exp(sh - sm)
    p /= p.sum()
    A = np.convolve(p, 1.0 - x, mode='full')[:n]
    B = np.correlate(x, p, mode='full')[2*(n-1):3*(n-1)]
    grad = dx * (A - B) + penalty * s
    return f, grad

def lbfgs_smooth(h, n, alpha, penalty, maxiter, ftol, time_limit=60):

    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

            break

                grad[lag:] -= y[:n-lag]
        else:

            if ll < n:

                grad[:n-ll] -= y[ll:]
