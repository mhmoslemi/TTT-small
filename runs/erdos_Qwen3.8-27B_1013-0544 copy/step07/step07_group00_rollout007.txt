    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

    return np.ones(n)

    best_h = h.copy()

                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[ms[diag_idx]] -= best_d ** 2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True

        if not improved:
            break
        if (p + 1) % 300 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def block_cd(h, n, max_passes=200, deadline=None, grid_size=20):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for idx in range(0, n - 1, 2):
            if deadline is not None and time.time() > deadline:
                break
            idx2 = idx + 1
            h0a, h0b = h[idx], h[idx2]
            if h0a < 1e-10 and h0b < 1e-10:
                continue
            if h0a > 1 - 1e-10 and h0b > 1 - 1e-10:
                continue
            lo_a, hi_a = (-h0a if h0a > 0 else 0.0), (1.0 - h0a if h0a < 1 else 1.0)
            lo_b, hi_b = (-h0b if h0b > 0 else 0.0), (1.0 - h0b if h0b < 1 else 1.0)
            if hi_a - lo_a < 1e-12 or hi_b - lo_b < 1e-12:
                continue
            ga = np.linspace(lo_a, hi_a, grid_size)
            gb = np.linspace(lo_b, hi_b, grid_size)
            best_val = current_c5
            best_da, best_db = 0.0, 0.0
            for da in ga:
                for db in gb:
                    if abs(da) < 1e-12 and abs(db) < 1e-12:
                        continue
                    na, nb = h0a + da, h0b + db
                    if na < -1e-12 or na > 1 + 1e-12 or nb < -1e-12 or nb > 1 + 1e-12:
                        continue
                    c_new = c.copy()
                    for ii, dd in ((idx, da), (idx2, db)):
                        m1 = max(0, min(n-1-ii, ii))
                        m2 = min(2*n-2, max(2*n-2-ii, ii+n-1))
                        ms = np.arange(m1, m2+1)
                        a = np.zeros(len(ms))
                        jh = ii + ms - (n-1)
                        vh = (jh >= 0) & (jh < n)
                        a[vh] += (1.0 - h[jh[vh]])
                        jg = ii - ms + (n-1)
                        vg = (jg >= 0) & (jg < n)
                        a[vg] -= h[jg[vg]]
                        c_new[ms] += dd * a
                        dm = (ms == (n-1))
                        if np.any(dm):
                            c_new[ms[dm]] -= dd * dd
                    vm = float(c_new.max())
                    if vm < best_val - 1e-13:
                        best_val = vm
                        best_da, best_db = da, db
            if abs(best_da) > 1e-10 or abs(best_db) > 1e-10:
                h[idx] = float(np.clip(h0a + best_da, 0.0, 1.0))
                h[idx2] = float(np.clip(h0b + best_db, 0.0, 1.0))
                c = np.correlate(h, 1.0 - h, mode='full')
                current_c5 = float(c.max())
                improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(400, 0.5)
        n = 400

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    if time.time() < deadline - 500:
        h1, c1 = slsqp_optimize(h, n, [200, 2000, 20000, 100000], maxiter=600)
        if c1 < best_c:
            best_h, best_c = h1, c1

    h2, c2 = coord_cd(best_h, len(best_h), max_passes=3000, deadline=deadline - 400, grid_size=100)
    if c2 < best_c:
        best_h, best_c = h2, c2

    h3, c3 = block_cd(best_h, len(best_h), max_passes=100, deadline=deadline - 200, grid_size=15)
    if c3 < best_c:
        best_h, best_c = h3, c3

    h4, c4 = coord_cd(best_h, len(best_h), max_passes=2000, deadline=deadline - 200, grid_size=120)
    if c4 < best_c:
        best_h, best_c = h4, c4

    n_list = [300, 400, 600]
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in n_list:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = slsqp_optimize(h_int, n_try, [5000, 50000], maxiter=400)
        h_opt, c_opt = coord_cd(h_opt, n_try, max_passes=1000, deadline=deadline - 100, grid_size=100)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new

    if time.time() < deadline - 15:
        h5, c5 = coord_cd(best_h, len(best_h), max_passes=5000, deadline=deadline, grid_size=150)
        if c5 < best_c:
            best_h, best_c = h5, c5

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
