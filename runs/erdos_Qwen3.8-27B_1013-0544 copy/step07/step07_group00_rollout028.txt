```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_optimize(h, n, alphas, maxiter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        def obj(x):
            return smooth_obj(x, n, alpha) + 100.0 * (x.sum() - n/2.0)**2
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except:
            pass
    return best_h, best_c

def cd_pass(h, c, n, deadline, grid_sizes=(30, 60, 100)):
    """Single coordinate descent pass with multi-level grid refinement.
    Updates c in place."""
    current_max = float(c.max())
    improved = 0
    
    for p in range(n):
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
            continue
        
        # Determine affected shifts
        # Shift m: affected if 0<=p-m<n or 0<=p+m<n
        # m range: [-(n-1), n-1]
        # 0<=p-m<n => p-n+1 <= m <= p
        # 0<=p+m<n => -p <= m <= n-1-p
        
        m_lo = max(-(n-1), p-n+1, -p)
        m_hi = min(n-1, p, n-1-p)
        
        # Actually the union of affected m's:
        # From first condition: m in [p-n+1, p]
        # From second condition: m in [-p, n-1-p]
        # Union: [min(p-n+1, -p), max(p, n-1-p)]
        # But clamped to [-(n-1), n-1]
        
        m_lo = max(-(n-1), min(p-n+1, -p))
        m_hi = min(n-1, max(p, n-1-p))
        
        if m_lo > m_hi:
            continue
            
        ms = np.arange(m_lo, m_hi + 1)
        ks = ms + (n - 1)  # indices into c array
        affected = len(ms)
        
        # Compute coefficients a[m]
        a = np.zeros(affected)
        # For each m: a[m] = (1-h[p-m])*[0<=p-m<n] - h[p+m]*[0<=p+m<n]
        pm = p - ms  # p - m
        pp = p + ms  # p + m
        valid_back = (pm >= 0) & (pm < n)
        valid_fwd = (pp >= 0) & (pp < n)
        a[valid_back] += 1.0 - h[pm[valid_back]]
        a[valid_fwd] -= h[pp[valid_fwd]]
        
        # Find which shifts have the diagonal term (m=0)
        diag_mask = (ms == 0)
        
        # Unaffected max
        c_aff = c[ks]
        # Create mask for affected indices
        # unaff_max = max of c at non-affected indices
        # For efficiency, just track current_max and check if any affected index has the max
        if float(c_aff.max()) < current_max:
            unaff_max = current_max
        else:
            # Need to compute max of non-affected
            unaff_aff = np.delete(c, ks)
            unaff_max = float(unaff_aff.max()) if len(unaff_aff) > 0 else 0.0
        
        # Multi-level grid search for optimal d
        d_lo = -h0
        d_hi = 1.0 - h0
        best_d = 0.0
        best_val = current_max
        
        for level, npts in enumerate(grid_sizes):
            if level == 0:
                d_vals = np.linspace(d_lo, d_hi, npts)
            else:
                frac = 0.3 / (level + 1)  # shrink to 30%, 15%, 10% ...
                half_w = (d_hi - d_lo) * frac
                d_center = best_d
                d_vals = np.linspace(d_center - half_w, d_center + half_w, npts)
                d_vals = d_vals[(d_vals >= d_lo) & (d_vals <= d_hi)]
                if len(d_vals) == 0:
                    break
            
            for d in d_vals:
                if abs(d) < 1e-15:
                    val = current_max
                else:
                    new_aff = c_aff + d * a
                    if diag_mask.any():
                        new_aff = new_aff.copy()
                        new_aff[diag_mask] -= d * d
                    val = max(unaff_max, float(new_aff.max()))
                if val < best_val - 1e-15:
                    best_val = val
                    best_d = d
        
        if abs(best_d) > 1e-13:
            h[p] = h0 + best_d
            # Update c
            c[ks] = c_aff + best_d * a
            if diag_mask.any():
                c[ks[diag_mask]] -= best_d * best_d
            current_max = max(unaff_max, float(c[ks].max()))
            improved += 1
    
    return h, c, improved

def coord_descent(h, n, max_passes=500, deadline=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    for p in range(max_passes):
        if deadline and time.time() > deadline:
            break
        h, c, improved = cd_pass(h, c, n, deadline)
        if improved == 0:
            break
        # Periodic re-projection (every 10 passes)
        if (p + 1) % 10 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    return h, float(c.max() * 2.0 / n)
```

```python
import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def lbfgs_opt(h, n, alpha, maxiter=400):
    def obj(x):
        return smooth_obj(x, n, alpha) + 200.0 * (x.sum() - n / 2.0) ** 2
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

        if c_int < best_c:
            best_h, best_c = h_int, c_int
            x_orig = x_new

    # Phase 5: Final intensive CD polish with fine grid
    n = len(best_h)
    h_final, c_final = coord_descent(best_h, n, 5000, deadline, grid_sizes=(80, 160, 320))
    if c_final < best_c:
        best_h, best_c = h_final, c_final

    # Phase 6: One more random-order pass
    if time.time() < deadline - 3:
        h_final, c_final = coord_descent_random(best_h, n, 500, deadline, rng=rng, grid_sizes=(80, 160))
        if c_final < best_c:
            best_h, best_c = h_final, c_final

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
