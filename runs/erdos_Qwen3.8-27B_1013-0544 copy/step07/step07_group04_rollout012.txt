            continue
        # Select lag
        if rng.random() < 0.8:
            pos = c[off + 1:] ** lag_power + 1e-10
            neg = c[off - 1::-1] ** lag_power + 1e-10
            w = np.maximum(pos, neg)
            w = w / w.sum()
            d = int(rng.choice(n - 1, p=w)) + 1
        else:
            d = int(rng.randint(1, n))
        i = int(rng.randint(n - d))
        j = i + d
        res = pair_search_golden(h, c, i, j, n)
        if res is not None:
            x, y, fv = res
            h[i] = x
            h[j] = y
            c = compute_c(h, n)
            curc = float(c.max())
            if curc < bestc:
                bestc = curc
                besth = h.copy()
                noimp = 0
            T *= Tcool
        else:
            noimp += 1
            T *= Tcool

    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc


def interpolate_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)


def optimize_n(h, n, seed, deadline):
    h = project_box_sum(h, n)
    bestc = float(compute_c(h, n).max())
    besth = h.copy()
    work = h.copy()

    phases = (
        (5.0, 350),
        (20.0, 300),
        (100.0, 280),
        (500.0, 250),
        (2000.0, 200),
        (8000.0, 180),
        (30000.0, 150),
    )
    for alpha, maxit in phases:
        if time.time() > deadline - 10:
            break
        mu = max(1.0, alpha * 0.1)
        h2 = lbfgsb_phase(work, n, alpha, mu, maxit)
        h2 = project_box_sum(h2, n)
        v2 = float(compute_c(h2, n).max())
        if v2 < bestc:
            bestc = v2
            besth = h2.copy()
        if v2 <= float(compute_c(work, n).max()) + 1e-7:
            work = h2
        else:
            work = besth.copy()

    # Mid pair search
    if time.time() < deadline - 20:
        besth, bestc = pair_local_search(besth, min(time.time() + 30, deadline - 15), seed=seed)
        work = besth.copy()

    # Final SLLSE
    for alpha, maxit in ((10000.0, 120), (50000.0, 100)):
        if time.time() > deadline - 8:
            break
        mu = alpha * 0.05
        h2 = lbfgsb_phase(work, n, alpha, mu, maxit)
        h2 = project_box_sum(h2, n)
        v2 = float(compute_c(h2, n).max())
        if v2 < bestc:
            bestc = v2
            besth = h2.copy()
            work = h2

    return project_box_sum(besth, n), float(compute_c(besth, n).max())


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        hp = np.array(initial_h_values, dtype=float).copy()
        n0 = len(hp)
    except Exception:
        n0 = 400
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        hp = 0.5 + 0.49 * np.sin(3.0 * np.pi * x)
        hp = project_box_sum(hp, n0)

    hp = project_box_sum(hp, n0)
    besth = hp.copy()
    bestc = float(compute_c(hp, n0).max())
    bestn = n0

    # Multi-resolution optimization
    n_list = sorted(set([n0] + [m for m in (320, 384, 448) if 50 <= m <= 600]))
    L = len(n_list)
    time_per = min(250.0, (deadline - t0 - 60) / L)

    for k, nh in enumerate(n_list):
        if time.time() > deadline - 50:
            break
        dl = min(t0 + time_per * (k + 1) + 30, deadline - 50)
        if dl < time.time() + 15:
            dl = min(time.time() + 15, deadline - 50)
        if k == 0:
            hinit = besth.copy()
        else:
            hinit = interpolate_h(besth, bestn, nh)
            hinit = project_box_sum(hinit + 0.003 * rng.randn(nh), nh)
        hopt, copt = optimize_n(hinit, nh, seed + 13 * k, dl)
        if copt < bestc:
            bestc = copt
            besth = hopt.copy()
            bestn = nh

    # Final aggressive pair search on best solution
    if time.time() < deadline - 10:
        besth, bestc = pair_local_search(besth, deadline - 5, seed + 777, lag_power=2.0)

    besth = project_box_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, bestn
