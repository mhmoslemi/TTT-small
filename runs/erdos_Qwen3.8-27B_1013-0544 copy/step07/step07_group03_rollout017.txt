    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def nesterov_subgrad(h, n, deadline, max_iters=50000, top_k=5, gamma=0.05):
    """Nesterov-accelerated subgradient for min max c[k]*dx."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = h.copy()
    y = h.copy()
    t = 1.0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(y, 1 - y, mode='full') * dx
        cur_c = c.max()
        # Subgradient: pick top_k worst shifts
        top_idx = np.argsort(c)[::-1][:top_k]
        # Compute subgradient
        grad = np.zeros(n)
        for idx in top_idx:
            k = idx - (n - 1)
            if k >= 0:
                if k == 0:
                    g = 1.0 - 2.0 * y
                else:
                    g = np.zeros(n)
                    g[k:] += (1.0 - y[:n-k])
                    g[:n-k] -= y[k:]
            else:
                kk = -k
                if kk == 0:
                    g = 1.0 - 2.0 * y
                else:
                    g = np.zeros(n)
                    g[:n-kk] += (1.0 - y[kk:])
                    g[kk:] -= y[:n-kk]
            grad += g
        grad *= dx / top_k
        
        # Nesterov step
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        theta = (t - 1) / t_new
        x_new = y - gamma * grad
        y_new = x_new + theta * (x_new - x)
        y = project_feasible(y_new, n)
        x = x_new
        t = t_new
        
        c_new = np.correlate(y, 1 - y, mode='full') * dx
        if c_new.max() < best_c:
            best_c = c_new.max()
            best_h = y.copy()
    return best_h, best_c
