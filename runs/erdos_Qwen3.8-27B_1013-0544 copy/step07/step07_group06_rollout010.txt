import numpy as np

_CORR_KEY = "hv"
_STOP = 0.0


def stop_callback(x):
    return time.time() > _STOP


def calibrate_corr_mode():
    global _CORR_KEY
    n = 6
    h = np.array([0.0, 1.0, 0.0, 1.0, 0.0, 1.0])
    v = 1.0 - h
    ln = 2 * n - 1
    rv = np.arange(-(n - 1), n, dtype=int)
    ca = np.zeros(ln)
    for p in range(ln):
        l = int(rv[p])
        lo = max(0, -l)
        hi = min(n, n - l)
        ca[p] = float(np.dot(h[lo:hi], v[lo + l:hi + l]))
    cands = {
        "hv": np.correlate(h, v, mode="full"),
        "vh": np.correlate(v, h, mode="full"),
        "hv_r": np.correlate(h, v, mode="full")[::-1],
        "vh_r": np.correlate(v, h, mode="full")[::-1],
    }
    best_key = "hv"
    best_err = 1e100
    for k, arr in cands.items():
        err = float(np.max(np.abs(arr - ca)))
        if err < best_err:
            best_err = err
            best_key = k
    _CORR_KEY = best_key


def get_corr(h, n):
    h = np.asarray(h, dtype=float).ravel()
    v = 1.0 - h
    if _CORR_KEY == "hv":
        return np.correlate(h, v, mode="full")
    if _CORR_KEY == "vh":
        return np.correlate(v, h, mode="full")
    if _CORR_KEY == "hv_r":
        return np.correlate(h, v, mode="full")[::-1]
    return np.correlate(v, h, mode="full")[::-1]


def compute_c5(h, n):
    h = np.asarray(h, dtype=float).ravel()
    dx = 2.0 / float(n)
    c = get_corr(h, n)
    return float(np.max(c) * dx)


def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(max(0, n - a.size))])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo = float(a.min()) - 1.0
    hi = float(a.max()) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(a - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(25):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = float(room.sum())
            if s <= 1e-15:
                break
            h += diff * room / s
        else:
            s = float(h.sum())
            if s <= 1e-15:
                break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h


def top_n_binary(score, n):
    score = np.asarray(score, dtype=float).ravel()
    if score.size != n:
        if score.size < n:
            score = np.concatenate([score, np.zeros(max(0, n - score.size))])
        else:
            score = score[:n]
    score = np.where(np.isfinite(score), score, 0.0)
    out = np.zeros(n, dtype=float)
    target = n // 2
    if target > 0:
        idx = np.argsort(-score)[:target]
        out[idx] = 1.0
    return out


def interp_to(arr, nnew):
    arr = np.asarray(arr, dtype=float).ravel()
    m = arr.size
    if m == nnew:
        return arr.copy()
    if m < 2 or nnew < 2:
        return np.full(nnew, 0.5)
    x0 = np.linspace(0.0, 2.0, m)
    x1 = np.linspace(0.0, 2.0, nnew)
    return np.interp(x1, x0, arr)


def random_score(n, rng, mode):
    x = np.linspace(0.0, 2.0, n)
    if mode == 0:
        z = np.zeros(n)
        km = int(rng.randint(2, 5))
        for _ in range(km):
            f = float(rng.randint(1, 9))
            amp = float(rng.uniform(0.35, 1.0))
            ph = float(rng.uniform(0.0, 2.0 * math.pi))
            z += amp * np.sin(2.0 * math.pi * f * x + ph)
        z += 0.05 * rng.normal(0.0, 1.0, n)
    elif mode == 1:

        kern = np.ones(7)

def make_base(h, n):
    ln = 2 * n - 1
    rv = np.arange(-(n - 1), n, dtype=np.int64)
    h8 = np.asarray(h, dtype=np.int8).ravel()
    base = np.zeros((n, ln), dtype=np.int8)
    for p in range(n):
        f = p + rv
        mf = (f >= 0) & (f < n)
        base[p, mf] = 1 - h8[f[mf]]
        b = p - rv
        mb = (b >= 0) & (b < n)
        base[p, mb] -= h8[b[mb]]
    return base


def pair_delta_row(i, j, base, corr, n):
    ln = 2 * n - 1
    center = n - 1
    row = base[i].astype(np.int64, copy=True) - base[j].astype(np.int64)
    row += corr
    row[center] -= 2
    row[center + (j - i)] += 1
    row[center + (i - j)] += 1
    return row


def apply_binary_swap(h, corr, base, n, i, j):
    h2 = h.copy()
    h2[i] = 1
    h2[j] = 0
    corr2 = pair_delta_row(i, j, base, corr, n).astype(np.int64)
    m2 = int(corr2.max())
    base2 = make_base(h2, n)
    return h2, corr2, m2, base2


def find_best_move(h, n, base, corr32, m, mode, current_cnt, deadline):
    zeros = np.flatnonzero(h == 0)
    ones = np.flatnonzero(h == 1)
    if zeros.size == 0 or ones.size == 0:
        return None
    zi = np.repeat(zeros, ones.size)
    oj = np.tile(ones, zeros.size)
    ln = 2 * n - 1
    center = n - 1
    chunk = max(32, min(1024, 600000 // max(1, ln)))
    for st in range(0, zi.size, chunk):
        if time.time() > deadline:
            break
        en = min(st + chunk, zi.size)
        i = zi[st:en]
        j = oj[st:en]
        arr = base[i].astype(np.int32) - base[j].astype(np.int32)
        arr += corr32[None, :]
        arr[:, center] -= 2
        arr[:, center + (j - i)] += 1
        arr[:, center + (i - j)] += 1
        mx = arr.max(axis=1)
        if mode == 0:
            sel = mx < m
            if sel.any():
                idxs = np.flatnonzero(sel)
                loc = idxs[int(np.argmin(mx[idxs]))]
                return (int(i[loc]), int(j[loc]))
        else:
            imp = mx < m
            if imp.any():
                idxs = np.flatnonzero(imp)
                loc = idxs[int(np.argmin(mx[idxs]))]
                return (int(i[loc]), int(j[loc]))
            same = mx == m
            if same.any():
                idxs = np.flatnonzero(same)
                sub = arr[idxs]
                cnt = (sub == m).sum(axis=1)
                better = cnt < current_cnt
                if better.any():
                    bidx = np.flatnonzero(better)
                    loc = idxs[bidx[int(np.argmin(cnt[bidx]))]]
                    return (int(i[loc]), int(j[loc]))
    return None

def lse_objective_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    if h.size != n:
        h = project_feasible(h, n)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = get_corr(h, n) * dx
    c = np.maximum(c, 0.0)
    mm = float(c.max())
    y = beta * (c - mm)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return 1e100, np.zeros(n)
    p = e / s
    ff = mm + float(np.log(s) / beta)
    r = p[::-1]
    a = np.convolve(r, v, mode="full")[n - 1:2 * n - 1]
    b = np.convolve(p, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (a - b)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n)
    return float(ff), g


def lse_wrap(x, n, beta, pen):
    x = np.asarray(x, dtype=float).ravel()
    if x.size != n:
        x = project_feasible(x, n)
    h = np.clip(x, 0.0, 1.0)
    ff, gg = lse_objective_grad(h, n, beta)
    if pen != 0.0:
        dev = float(h.sum()) - 0.5 * n
        ff += 0.5 * pen * dev * dev
        gg = gg + pen * dev
    if not np.isfinite(ff):
        return 1e100, np.zeros_like(h)
    return float(ff), gg


def polish_continuous(h0, n, betas, pen, stop_deadline, stage_time):
    work = project_feasible(h0, n)
    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        now = time.time()
        if now > stop_deadline - 8.0:
            break
        _STOP = min(stop_deadline - 3.0, now + max(6.0, min(stage_time, 0.45 * max(6.0, stop_deadline - now))))
        try:
            res = minimize(
                lse_wrap, work, method="L-BFGS-B", jac=True, bounds=bounds,
                args=(n, float(beta), float(pen)), callback=stop_callback,
                options={"maxiter": 1000000, "maxls": 40, "ftol": 1e-12, "gtol": 1e-7}
            )
            x = res.x if (res is not None and hasattr(res, "x")) else work
        except Exception:
            x = work
        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc:
            bc = cw
            best = work.copy()
        cands = [work, work[::-1], 1.0 - work[::-1], 0.5 * (work + work[::-1])]
        if n % 2 == 0:
            cands.append(top_n_binary(work, n))
        for cand in cands:
            cand = project_feasible(cand, n)
            cc = compute_c5(cand, n)
            if cc < bc:
                bc = cc
                best = cand.copy()
        if bc < cw - 1e-14:
            work = best.copy()
    return best, bc
