    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lse_omega(x, n, alpha):
    """Compute LSE weights (normalized softmax of alpha*(c - max))."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_max = c.max()
    shifted = np.clip(alpha * (c - c_max), -700, 0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    return w, c_max

def lse_objective(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_max = c.max()
    shifted = np.clip(alpha * (c - c_max), -700, 0)
    logsum = np.log(np.exp(shifted).sum())
    return c_max + logsum / alpha

def lse_gradient(x, n, alpha):
    dx = 2.0 / n
    w, c_max = lse_omega(x, n, alpha)
    s = np.arange(-(n - 1), n)
    one_minus_x = 1.0 - x
    grad = np.zeros(n)
    for j in range(n):
        idx1 = j - s
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j + s
        v2 = (idx2 >= 0) & (idx2 < n)
        A = np.dot(w[v1], one_minus_x[idx1[v1]])
        B = np.dot(w[v2], x[idx2[v2]])
        grad[j] = (A - B) * dx
    return grad
