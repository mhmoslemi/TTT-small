def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0:
        return m
    sh = alpha * (c - m)
    s = float(np.exp(sh).sum())
    if s <= 0.0:
        return m
    return m + np.log(s) / alpha

def lse_weights(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0.0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
        return q
    q = e / s
    r = q.sum()
    if r > 0:
        q /= r
    return q

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    q = lse_weights(c, alpha)
    g = np.zeros(n)
    dx = 2.0 / n

def pair_eval_v2(h, c_base, i, j, xs, n):
    L = 2 * n - 1
    off = n - 1
    dx = 2.0 / n
    S = h[i] + h[j]
    x0 = h[i]
    
    # Compute coef[m] for each lag m
    # r = m - off
    r = np.arange(L) - off  # lags from -(n-1) to (n-1)
    
    # coef[m] = (1-h[i-r]) * I(0<=i-r<n) - h[i+r] * I(0<=i+r<n) - (1-h[j-r]) * I(0<=j-r<n) + h[j+r] * I(0<=j+r<n)
    
    ir = i - r  # = i - (m - off)
    isr = i + r
    jr = j - r
    jsr = j + r
    
    coef = np.zeros(L)
    m1 = (ir >= 0) & (ir < n)
    m2 = (isr >= 0) & (isr < n)
    m3 = (jr >= 0) & (jr < n)
    m4 = (jsr >= 0) & (jsr < n)
    
    coef[m1] += (1.0 - h[ir[m1].astype(int)])
    coef[m2] -= h[isr[m2].astype(int)]
    coef[m3] -= (1.0 - h[jr[m3].astype(int)])
    coef[m4] += h[jsr[m4].astype(int)]
    
    # For each candidate x in xs, t = x - x0
    # New c at lag m = c_base[m] + t * dx * coef[m]
    # We want max_m of this for each x
    
    t = xs - x0  # shape (P,)
    # delta[m, p] = t[p] * dx * coef[m]  -- but we want max over m for each p
    # Equivalent: for each p, max_m (c_base[m] + t[p] * dx * coef[m])
    
    # This is: c_base + outer(t, dx*coef) then max along axis 0
    # But that's P * L memory. For P=64, L=799, that's 51K floats = 400KB. Fine.
    
    vals = c_base[None, :] + t[:, None] * (dx * coef)[None, :]
    return vals.max(axis=1)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        d = tgt - v.sum()
        if abs(d) < 1e-13:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v


def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    s = float(np.exp(sh).sum())
    return m + np.log(s) / alpha if s > 0 else m


def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
    else:
        q = e / s
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    for idx in np.flatnonzero(q > 1e-8):
        k = int(idx) - off
        w = float(q[idx])
        if k > 0:
            g[k:n] += w * (1.0 - x[0:n-k])
            g[0:n-k] -= w * x[k:n]
        elif k < 0:
            s2 = -k
            g[0:n-s2] += w * (1.0 - x[s2:n])
            g[s2:n] -= w * x[0:n-s2]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g


def pen_obj(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_obj(x, n, alpha) + 0.5 * mu * d * d


def pen_grad(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_grad(x, n, alpha) + mu * d


def lbfgsb_fast(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(pen_obj, h, args=(n, alpha, mu), method="L-BFGS-B",
                       jac=pen_grad, bounds=bounds,
                       options={"maxiter": maxiter, "maxfun": 3 * maxiter,
                                "ftol": 1e-15, "gtol": 1e-10})
        if hasattr(res, "x") and res.x is not None:
            return res.x
    except Exception:
        pass
    return h


def sum_fun(x, n):
    return float(x.sum() - 0.5 * n)


def sum_jac(x, n):
    return np.ones(n)


def slsqp_phase(h, n, alpha, maxiter):
    h = project_box_sum(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{"type": "eq", "fun": sum_fun, "args": (n,), "jac": sum_jac}]
        res = minimize(lse_obj, h, args=(n, alpha), method="SLSQP",
                       jac=lse_grad, bounds=bounds, constraints=cons,
                       options={"maxiter": maxiter, "ftol": 1e-14})
        if hasattr(res, "x") and res.x is not None:
            return res.x
    except Exception:
        pass
    return h


def pair_eval_v2(h, c_base, i, j, xs, n):
    L = 2 * n - 1
    off = n - 1
    dx = 2.0 / n
    x0 = float(h[i])
    r = np.arange(L) - off
    ir = i - r
    isr = i + r
    jr = j - r
    jsr = j + r
    m1 = (ir >= 0) & (ir < n)
    m2 = (isr >= 0) & (isr < n)
    m3 = (jr >= 0) & (jr < n)
    m4 = (jsr >= 0) & (jsr < n)
    coef = np.zeros(L)
    if m1.any():
        coef[m1] += (1.0 - h[ir[m1]])
    if m2.any():
        coef[m2] -= h[isr[m2]]
    if m3.any():
        coef[m3] -= (1.0 - h[jr[m3]])
    if m4.any():
        coef[m4] += h[jsr[m4]]
    t = xs - x0
    vals = c_base[None, :] + t[:, None] * (dx * coef)[None, :]
    return vals.max(axis=1)


def pair_search_fast(h, c_base, i, j, n):
    x0 = float(h[i])
    y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-13:
        return None
    xs = np.linspace(lo, hi, 33)
    vals = pair_eval_v2(h, c_base, i, j, xs, n)
    for _ in range(4):
        k = int(np.argmin(vals))
        if k == 0:
            a, b = xs[0], xs[1]
        elif k == len(xs) - 1:
            a, b = xs[-2], xs[-1]
        else:
            a, b = xs[k-1], xs[k+1]
        if b - a < 1e-13:
            break
        xs = np.linspace(a, b, 49)
        vals = pair_eval_v2(h, c_base, i, j, xs, n)
    k = int(np.argmin(vals))
    bv = float(vals[k])
    cur_max = float(c_base.max())
    if bv < cur_max - 1e-12:
        xb = float(np.clip(xs[k], lo, hi))
        yb = float(S - xb)
        if yb < 0.0:
            yb = 0.0
            xb = S
        elif yb > 1.0:
            yb = 1.0
            xb = S - 1.0
        return xb, yb, bv
    return None


def aggressive_pair_search(h, deadline, seed=1):

    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    rng = np.random.RandomState(seed)
    off = n - 1
    noimp = 0
    maxno = 8000

    while time.time() < deadline and noimp < maxno:
        if rng.random() < 0.75:
            pos_part = c[off+1:]
            neg_part = c[off-1::-1]
            w = np.concatenate([pos_part, neg_part]) + 1e-10
            w = w / w.sum()
            idx = int(rng.choice(2 * (n - 1), p=w))
            d = (idx + 1) if idx < (n - 1) else (-(idx - (n - 1)) + 1)
            d = abs(d)
        else:
            d = int(rng.randint(1, n))
        i = int(rng.randint(max(1, n - d)))
        j = i + d
        res = pair_search_fast(h, c, i, j, n)
        if res is not None:
            x, y, bv = res
            h[i] = x
            h[j] = y
            c = compute_c(h, n)
            cur = float(c.max())
            if cur < bestc:
                bestc = cur
                besth = h.copy()
                noimp = 0
            else:
                noimp += 1
        else:
            noimp += 1

    h = project_box_sum(besth, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    return h, bestc


def systematic_sweep(h, deadline, seed=1):
