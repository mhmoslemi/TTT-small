        w = ew / ws
    return maxc + np.log(ws) / beta, w

def smooth_grad(h, w, n, dx):
    mid = n - 1
    if n == 1:
        return (1.0 - 2.0 * h) * w[mid] * dx
    wp = np.zeros(n)
    wn = np.zeros(n)
    wp[1:] = w[mid + 1:]
    wn[1:] = w[:mid][::-1]
    s = wp + wn
    pp = np.cumsum(wp)
    pn = np.cumsum(wn)
    idx = np.arange(n)
    consts = pp + pn[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx

_state = {}

def obj_fn(x):
    st = _state
    n, dx, tgt, beta = st['n'], st['dx'], st['target'], st['beta']
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_w(c, beta)
    g = smooth_grad(x, w, n, dx)
    st['cx'] = x.copy()
    st['cg'] = g
    return float(f), g

def obj_only(x):
    f, _ = obj_fn(x)
    return f

def obj_jac(x):
    cx = _state.get('cx')
    if cx is not None and np.array_equal(x, cx):
        return _state['cg'].copy()
    _, g = obj_fn(x)
    return g

def refine_lbfgs(h, beta, maxiter, deadline):
    n = len(h)
    _state.update({'n': n, 'dx': 2.0/n, 'target': 0.5*n, 'beta': beta, 'cx': None, 'cg': None})
    rem = deadline - time.time()
    if rem < 3:
        return h, erdos_c5(h, n)
    mx = int(min(maxiter, max(10, int(rem / 3.0))))
    bnds = [(0.0, 1.0)] * n
    try:
        r = so.minimize(obj_only, h, method='L-BFGS-B', jac=obj_jac,
                        bounds=bnds, options={'maxiter': mx, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50, 'disp': False})
        cand = project_box_sum(r.x, 0.5 * n)
    except Exception:
        cand = project_box_sum(h, 0.5 * n)
    return cand, erdos_c5(cand, n)

def con_fun(x):
    return float(x.sum() - _state['target'])

def con_jac(x):
    return np.ones(_state['n'])

def refine_slqp(h, beta, maxiter, deadline):
    n = len(h)
    _state.update({'n': n, 'dx': 2.0/n, 'target': 0.5*n, 'beta': beta, 'cx': None, 'cg': None})
    rem = deadline - time.time()
    if rem < 6:
        return h, erdos_c5(h, n)
    mx = int(min(maxiter, max(5, int(rem / 10.0))))
    bnds = [(0.0, 1.0)] * n
    try:
        r = so.minimize(obj_only, h, method='SLSQP', jac=obj_jac, bounds=bnds,
                        constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                        options={'maxiter': mx, 'iprint': 0, 'ftol': 1e-14})
        cand = project_box_sum(r.x, 0.5 * n)
    except Exception:
        cand = project_box_sum(h, 0.5 * n)
    return cand, erdos_c5(cand, n)

def pair_obj(delta, h, i, j, n):
    a = h[i] + delta
    b = h[j] - delta
    if a < -1e-10 or a > 1.0 + 1e-10 or b < -1e-10 or b > 1.0 + 1e-10:
        return 1e100
    hh = h.copy()
    hh[i] = a
    hh[j] = b
    return erdos_c5(hh, n)

def pair_line(h, n, i, j, cur, deadline):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return h, cur
    best_d, best_c = 0.0, cur
    if time.time() > deadline:
        return h, cur
    for d in (lo, hi, 0.0):
        c = pair_obj(d, h, i, j, n)
        if c < best_c:
            best_c, best_d = c, d
    try:
        rs = so.minimize_scalar(pair_obj, args=(h, i, j, n), bounds=(lo, hi),
                                method='bounded', options={'maxiter': 30, 'xatol': 1e-11})
        if rs.x is not None:
            c = pair_obj(float(rs.x), h, i, j, n)
            if c < best_c:
                best_c, best_d = c, float(rs.x)
    except Exception:
        pass
    if best_c < cur - 1e-15:
        hh = h.copy()
        hh[i] += best_d
        hh[j] -= best_d
        return np.clip(hh, 0.0, 1.0), best_c
    return h, cur

def pair_polish(h, n, deadline, seed, n_pairs=50, n_passes=6):
    rng = np.random.RandomState(int(seed))
    cur = erdos_c5(h, n)
    frac = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if frac.size:
        score = np.minimum(h[frac], 1.0 - h[frac])
        order = frac[np.argsort(score)[::-1]]
    else:
        order = np.array([], dtype=int)
    top = order[:min(n_pairs, order.size)] if order.size > 0 else np.array([], dtype=int)
    if top.size < 2:
        top = rng.choice(n, size=min(n_pairs, n), replace=False)
    for _ in range(n_passes):
        improved = False
        pairs = [(int(top[a]), int(top[b])) for a in range(len(top)) for b in range(a+1, len(top))]
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline - 3:
                return h, erdos_c5(h, n)
            hh, cc = pair_line(h, n, i, j, cur, deadline - 3)
            if cc < cur - 1e-15:
                h, cur = hh, cc
                improved = True
        if not improved:
            break
    return h, cur

def block_exchange(h, n, deadline, seed):
    """Try exchanging values between two random blocks of length 2-5."""
    rng = np.random.RandomState(int(seed) + 7)
    cur = erdos_c5(h, n)
    best_h, best_c = h.copy(), cur
    ntry = min(200, max(50, int((deadline - time.time()) / 0.5)))
    for _ in range(ntry):
        if time.time() > deadline:
            break
        bl = rng.randint(2, 6)
        if bl >= n - 1:
            continue
        i = rng.randint(0, n - bl)
        j = rng.randint(0, n - bl)
        if abs(i - j) < bl:
            continue
        hh = h.copy()
        tmp = hh[i:i+bl].copy()
        hh[i:i+bl] = hh[j:j+bl]
        hh[j:j+bl] = tmp
        hh = project_box_sum(hh, 0.5 * n)
        c = erdos_c5(hh, n)
        if c < best_c - 1e-15:
            best_h, best_c = hh.copy(), c
    return best_h, best_c

def resample(h, n, nh):
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, nh)
    hh = np.interp(y, x, h)
    return project_box_sum(hh, 0.5 * nh)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(10.0, float(budget_s) - 5.0)
    deadline = t0 + total
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    if n < 16:
        n = 400
        h = np.full(n, 0.5)
    if n > 800:
        old = n
        n = 800
        h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, old), h)
    h = project_box_sum(h, 0.5 * n)
    best_h, best_n, best_c = h.copy(), n, erdos_c5(h, n)

    e1 = t0 + 0.30 * total
    e2 = t0 + 0.55 * total
    e3 = t0 + 0.78 * total
    e4 = deadline - 4.0

    # Phase 1: L-BFGS with warm-up beta schedule
    for beta, mx in [(200, 80), (1000, 80), (5000, 70), (20000, 60), (100000, 50), (500000, 40)]:
        if time.time() > e1 - 5:
            break
        cand, c = refine_lbfgs(best_h, beta, mx, deadline)
        if c < best_c - 1e-16:
            best_h, best_c = cand.copy(), c

    # Phase 2: SLSQP refinement
    for beta, mx in [(5000, 50), (50000, 40), (500000, 30)]:
        if time.time() > e2 - 5:
            break
        cand, c = refine_slqp(best_h, beta, mx, deadline)
        if c < best_c - 1e-16:
            best_h, best_c = cand.copy(), c

    # Phase 3: Multi-start exploration
    rng = np.random.RandomState(seed)
    starts = [best_h.copy()]
    for sig in (0.03, 0.07, 0.15):
        starts.append(project_box_sum(best_h + sig * rng.randn(best_n), 0.5 * best_n))
    # Sawtooth pattern
    saw = np.zeros(best_n)
    period = max(4, best_n // 20)
    for k in range(0, best_n, 2 * period):
        saw[k:min(k+period, best_n)] = 1.0
    starts.append(project_box_sum(saw, 0.5 * best_n))

    per_start = max(2.0, (e3 - e2) / max(1, len(starts)))
    for si, sh in enumerate(starts):
        if time.time() > e3 - 3:
            break
        sdeadline = min(e3 - 2.0, time.time() + per_start)
        cand, c = refine_lbfgs(sh, 2000.0, 40, sdeadline)
        if time.time() < sdeadline - 1:
            cand, c = refine_lbfgs(cand, 50000.0, 25, sdeadline)
        if c < best_c - 1e-16:
            best_h, best_c = cand.copy(), c

    # Phase 4: Aggressive local polish
    if time.time() < e4 - 30:
        best_h, best_c = pair_polish(best_h, best_n, e4 - 28, seed, n_pairs=60, n_passes=8)

    # Block exchange
    if time.time() < e4 - 15:
        best_h, best_c = block_exchange(best_h, best_n, e4 - 13, seed)

    # Final refinement
    if time.time() < e4 - 10:
        cand, c = refine_lbfgs(best_h, 1000000.0, 25, deadline)
        if c < best_c - 1e-16:
            best_h, best_c = cand.copy(), c
    if time.time() < e4 - 6:
        cand, c = pair_polish(best_h, best_n, e4 - 5, seed + 99, n_pairs=40, n_passes=4)
        if c < best_c - 1e-16:
            best_h, best_c = cand.copy(), c
    if time.time() < e4 - 3:
        cand, c = refine_slqp(best_h, 200000.0, 20, deadline)
        if c < best_c - 1e-16:
            best_h, best_c = cand.copy(), c

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
