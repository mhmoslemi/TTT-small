def lse_grad(x, n, alpha):

    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())

            g[k2:n] -= w * x[:m2]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(20):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    return m + np.log(s) / alpha if s > 0 else m

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0:
        q = np.zeros(len(c))
        q[int(np.argmax(c))] = 1.0
    else:
        q = e / s
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    for idx in np.flatnonzero(q):
        k = idx - off
        w = q[idx]
        if k > 0:
            m2 = n - k
            # FIXED GRADIENT
            g[:m2] += w * (1.0 - x[k:n])   # (1-h[i+k]) for i in [0, n-k-1]
            g[k:n] -= w * x[:m2]             # -h[i-k] for i in [k, n-1]
        elif k < 0:
            k2 = -k
            m2 = n - k2
            # FIXED GRADIENT
            g[k2:n] += w * (1.0 - x[:m2])   # (1-h[i-k]) for i in [k2, n-1]
            g[:m2] -= w * x[k2:n]            # -h[i+k2] for i in [0, n-k2-1]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def pen_obj(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_obj(x, n, alpha) + 0.5 * mu * d * d

def pen_grad(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_grad(x, n, alpha) + mu * d * np.ones(n)

def sum_fun(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones(n)

def pair_search_v2(h, c_base, i, j, n, bg_max):

    Efficient pair search. Only 3 lags are affected: 0, d=j-i, -d.
    bg_max = max of c_base excluding lags 0, d, -d (precomputed)
    """
    if i > j:
        i, j = j, i
    x0 = float(h[i])
    y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-13:
        return None
    
    d = j - i
    off = n - 1
    idx0 = off
    idxp = off - d  # lag -d
    idxm = off + d  # lag +d

    # h[i] = x, h[j] = S - x = y

    # Change: x(1-x) + y(1-y) - x0(1-x0) - y0(1-y0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(20):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_smooth(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    return m + np.log(s) / alpha if s > 0 else m

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0:
        q = np.zeros(len(c)); q[int(np.argmax(c))] = 1.0
    else:
        q = e / s
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    for idx in np.flatnonzero(q):
        k = idx - off
        w = q[idx]
        if k > 0:
            m2 = n - k
            g[:m2] += w * (1.0 - x[k:n])
            g[k:n] -= w * x[:m2]
        elif k < 0:
            k2 = -k
            m2 = n - k2
            g[k2:n] += w * (1.0 - x[:m2])
            g[:m2] -= w * x[k2:n]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def pen_obj(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_smooth(x, n, alpha) + 0.5 * mu * d * d

def pen_grad(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_grad(x, n, alpha) + mu * d * np.ones(n)

def sum_fun(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones(n)

def bg_max_compute(c, idx0, idxp, idxm):
    L = len(c)
    vals = c.copy()
    vals[idx0] = -9999.0
    vals[idxp] = -9999.0
    vals[idxm] = -9999.0
    return float(vals.max())

def pair_search_fast(h, c, i, j, n, bgmx):
    if i > j:
        i, j = j, i
    x0 = float(h[i]); y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0); hi = min(1.0, S)
    if hi - lo < 1e-13:
        return None
    d = j - i
    off = n - 1
    idx0, idxp, idxm = off, off - d, off + d
    xs = np.linspace(lo, hi, 24)
    ys = S - xs
    c0v = c[idx0] + xs * (1 - xs) + ys * (1 - ys) - x0 * (1 - x0) - y0 * (1 - y0)
    dvd = np.full(len(xs), c[idxm])

    return None

def pair_search_refine(h, c, i, j, n, bgmx, xb, yb):

    lo = max(0.0, S - 1.0); hi = min(1.0, S)
    d = abs(j - i)
    off = n - 1

        ii = min(i, j); jj = max(i, j)

        if ii >= d: dvd += -h[ii-d]*(xs-xb)

        k = int(np.argmin(cmax))

            hi = xs[1]

            hi = 1.0 if lo < 1 else lo
            lo = xs[-2]
        else:
            lo = xs[k-1]; hi = xs[k+1]
        if hi - lo < 1e-14:

        xb = float((lo + hi) / 2)
    return xb, S - xb

            d = int(rng.choice(n-1, p=w)) + 1
        else:

def lse_lbfgs(h, n, alpha, mu, maxit, ftol=1e-15):
    h = project_box_sum(h, n)
    try:
        res = minimize(pen_obj, h, args=(n, alpha, mu), method="L-BFGS-B",
                       jac=pen_grad, bounds=[(0.0, 1.0)]*n,
                       options={"maxiter": maxit, "maxfun": 3*maxit, "ftol": ftol, "gtol": 1e-9})
        if res.x is not None:
            return project_box_sum(res.x, n)
    except Exception:
        pass
    return h

def slsqp_opt(h, n, alpha, maxit):
    h = project_box_sum(h, n)
    try:
        cons = [{"type": "eq", "fun": sum_fun, "args": (n,), "jac": sum_jac}]
        res = minimize(lse_smooth, h, args=(n, alpha), method="SLSQP",
                       jac=lse_grad, bounds=[(0.0, 1.0)]*n, constraints=cons,
                       options={"maxiter": maxit, "ftol": 1e-15})
        if res.x is not None:
            return project_box_sum(res.x, n)
    except Exception:
        pass
    return h
