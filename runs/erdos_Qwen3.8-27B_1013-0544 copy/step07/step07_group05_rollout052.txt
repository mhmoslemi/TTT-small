import time
import numpy as np
import scipy.optimize as so

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    diff = float(target - y.sum())
    if abs(diff) < 1e-14:
        return y
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y2 = np.clip(x - mid, 0.0, 1.0)
        if float(y2.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    if abs(diff) > 1e-13:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
    return y

def smooth_max_and_grad(h, beta, dx):
    """Compute smooth max of correlations and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    n_corr = len(corr)
    maxc = float(np.max(corr))
    z = beta * (corr - maxc)
    z = np.clip(z, -700.0, 0.0)
    ew = np.exp(z)
    ws = float(ew.sum())
    if ws < 1e-300:
        f = maxc
        w = np.zeros(n_corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        f = maxc + np.log(ws) / beta
        w = ew / ws
    # Gradient
    g = (np.convolve(1.0 - h, w, 'full')[n-1:2*n-1] - np.correlate(h, w, 'full')[n-1:2*n-1]) * dx
    return f, g
```

Now for the optimization, I'll use a different strategy than the previous code:

1. **Phase 1**: LBFGS with moderate beta (100-1000) to get a good local solution
2. **Phase 2**: LBFGS with high beta (10000+) to sharpen
3. **Phase 3**: Pair polishing (enhanced with more iterations)
4. **Phase 4**: Nelder-Mead on fractional variables
5. **Phase 5**: Try different n and transfer

Let me also think about the Nelder-Mead approach. After gradient-based methods converge, the solution is likely at a "kink" of the max function. Nelder-Mead can handle non-smooth objectives well. I'll apply it to the top-k most fractional variables (those closest to 0.5), since those are the ones most likely to move.

Actually, a better approach for the final polish: **direct search on the active set**.

The max correlation is achieved at one or more lags. If I fix which lag is the max, then I'm optimizing a single quadratic (well, it's h*(1-h) which is not convex). But I can use the fact that the optimal solution likely has h values at 0 or 1 (bang-bang), with a few fractional values.

Let me think about the structure. For n=400, we have 400 variables with sum = 200. If the solution is mostly bang-bang, we have about 200 zeros and 200 ones, with a few fractional values. The pair polishing already exploits this.

My enhanced approach:
1. After gradient convergence, identify the "active" lags (those within, say, 0.01 of max)
2. For each active lag, the constraint sum_i h[i]*(1-h[i+k]) <= C is tight
3. Do coordinate descent / pair moves that specifically reduce the active correlations

Actually, let me just focus on making the pair polishing much more aggressive and adding Nelder-Mead on subsets.

Let me also consider: what if I use a **combination of pair moves and single-variable moves**? A single variable move h[j] → h[j] + δ (breaking the sum constraint, then fixing by adjusting all others proportionally) might escape local optima that pair moves can't.

Or better: **projected line search**. For a given direction d (e.g., d[j] = +1, d[k] = -1, others 0), find the best step α such that h + αd stays in [0,1]^n and sum is preserved.

The pair_polish already does this for 2-variable moves. Let me extend to 3-variable moves and also add a "shift" move where I shift a contiguous block.

```python
import time
import numpy as np
import scipy.optimize as so

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    diff = float(target - y.sum())
    if abs(diff) < 1e-14:
        return y
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y2 = np.clip(x - mid, 0.0, 1.0)
        if float(y2.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    if abs(diff) > 1e-13:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
    return y

def smooth_max_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    maxc = float(np.max(corr))
    z = beta * (corr - maxc)
    z = np.clip(z, -700.0, 0.0)
    ew = np.exp(z)
    ws = float(ew.sum())
    if ws < 1e-300:
        f = maxc
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        f = maxc + np.log(ws) / beta
        w = ew / ws
    g = (np.convolve(1.0 - h, w, 'full')[n-1:2*n-1] - np.correlate(h, w, 'full')[n-1:2*n-1]) * dx
    return f, g

def lbfgs_obj(x, beta, dx):
    f, g = smooth_max_grad(x, beta, dx)
    return f, g

def lbfgs_refine(h, beta, maxiter, deadline):
    n = len(h)
    dx = 2.0 / n
    rem = deadline - time.time()
    if rem < 3.0:
        return h, erdos_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    mx = int(min(maxiter, max(10, int(rem / 4.0))))
    try:
        res = so.minimize(lbfgs_obj, h, args=(beta, dx), method='L-BFGS-B',
                          jac=True, bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 30})
        cand = project_box_sum(res.x, 0.5 * n)
    except Exception:
        cand = project_box_sum(h, 0.5 * n)
    return cand, erdos_c5(cand, n)

def pair_move(h, i, j, delta, n):
    a = h[i] + delta
    b = h[j] - delta
    if a < -1e-12 or a > 1.0 + 1e-12 or b < -1e-12 or b > 1.0 + 1e-12:
        return None
    hh = h.copy()
    hh[i] = a
    hh[j] = b
    return hh

def pair_obj_fast(h, i, j, delta, n):
    hh = h.copy()
    a = h[i] + delta
    b = h[j] - delta
    if a < -1e-12 or a > 1.0 + 1e-12 or b < -1e-12 or b > 1.0 + 1e-12:
        return 1e100
    hh[i] = a
    hh[j] = b
    return erdos_c5(hh, n)

def pair_polish(h, n, deadline, seed, max_rounds=15, top_k=30):
    rng = np.random.RandomState(int(seed))
    h = np.asarray(h, dtype=float).copy()
    cur = erdos_c5(h, n)
    # Identify fractional variables (most likely to improve)
    frac = np.where((h > 0.01) & (h < 0.99))[0]
    if frac.size >= top_k:
        score = np.minimum(h[frac], 1.0 - h[frac])
        order = frac[np.argsort(-score)]
        top = order[:top_k]
    elif frac.size > 0:
        top = np.concatenate([frac, rng.choice(n, size=top_k - frac.size, replace=False)])
        top = np.unique(top)
    else:
        top = rng.choice(n, size=min(top_k, n), replace=False)
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 5.0:
            break
        improved = False
        # Generate all pairs from top
        pairs = [(int(top[a]), int(top[b])) for a in range(len(top)) for b in range(a+1, len(top))]
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline - 5.0:
                break
            # Find best delta
            hi0, hj0 = h[i], h[j]
            lo = max(-hi0, hj0 - 1.0)
            hi = min(1.0 - hi0, hj0)
            if hi - lo < 1e-14:
                continue
            best_d, best_c = 0.0, cur
            # Try endpoints
            for d in [lo, hi, 0.5*(lo+hi)]:
                c = pair_obj_fast(h, i, j, d, n)
                if c < best_c:
                    best_c, best_d = c, d
            # Bounded scalar minimization
            try:
                rs = so.minimize_scalar(lambda d: pair_obj_fast(h, i, j, d, n),
                                        bounds=(lo, hi), method='bounded',
                                        options={'maxiter': 30, 'xatol': 1e-11})
                if rs.fun < best_c:
                    best_c, best_d = float(rs.fun), float(rs.x)
            except Exception:
                pass
            if best_c < cur - 1e-15:
                h[i] += best_d
                h[j] -= best_d
                h = np.clip(h, 0.0, 1.0)
                cur = best_c
                improved = True
        if not improved:
            break
    return h, cur

def nm_polish(h, n, deadline, seed, k=12):
    """Nelder-Mead on top-k fractional variables."""
    rng = np.random.RandomState(int(seed) + 1000)
    h = np.asarray(h, dtype=float).copy()
    cur = erdos_c5(h, n)
    frac = np.where((h > 0.02) & (h < 0.98))[0]
    if frac.size < 2:
        return h, cur
    score = np.minimum(h[frac], 1.0 - h[frac])
    order = frac[np.argsort(-score)]
    idx = order[:min(k, frac.size)]
    if len(idx) < 2:
        return h, cur
    
    def nm_obj(sub):
        """sub: values of h at indices idx, rest fixed. Must preserve sum."""
        rest_sum = float(h.sum()) - float(h[idx].sum())
        new_sum = rest_sum + float(sub.sum())
        if new_sum <= 0 or new_sum >= n:

def triple_polish(h, n, deadline, seed, max_rounds=5, top_k=20):
    """Try 3-variable moves: move delta from i to j and k."""
    rng = np.random.RandomState(int(seed) + 2000)
    h = np.asarray(h, dtype=float).copy()
    cur = erdos_c5(h, n)
    frac = np.where((h > 0.01) & (h < 0.99))[0]
    if frac.size < 3:
        return h, cur
    score = np.minimum(h[frac], 1.0 - h[frac])
    order = frac[np.argsort(-score)]
    top = order[:min(top_k, frac.size)]
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 5.0:
            break
        improved = False
        triples = []
        for a in range(len(top)):
            for b in range(a+1, len(top)):
                for c in range(b+1, len(top)):
                    triples.append((int(top[a]), int(top[b]), int(top[c])))
        rng.shuffle(triples)
        for i, j, k in triples[:200]:  # Limit per round
            if time.time() > deadline - 5.0:
                break
            # Try: move delta from i to j, and delta from i to k
            # i decreases by 2d, j increases by d, k increases by d
            # Or: i increases by d, j decreases by d/2, k decreases by d/2
            hi0, hj0, hk0 = h[i], h[j], h[k]
            # Direction 1: i up, j down, k down (i gains, j and k lose half each)
            # d must satisfy: hi0+d<=1, hj0-d/2>=0, hk0-d/2>=0
            d_lo = 0.0
            d_hi = min(1.0 - hi0, 2.0 * hj0, 2.0 * hk0)
            if d_hi > 1e-14:
                def obj1(d):
                    a, b, c = hi0 + d, hj0 - 0.5*d, hk0 - 0.5*d
                    if a > 1+1e-12 or b < -1e-12 or c < -1e-12:
                        return 1e100
                    hh = h.copy()
                    hh[i], hh[j], hh[k] = a, b, c
                    return erdos_c5(hh, n)
                try:
                    rs = so.minimize_scalar(obj1, bounds=(d_lo, d_hi), method='bounded',
                                           options={'maxiter': 15, 'xatol': 1e-10})
                    if rs.fun < cur - 1e-15:
                        d = float(rs.x)
                        h[i] += d
                        h[j] -= 0.5 * d
                        h[k] -= 0.5 * d
                        h = np.clip(h, 0.0, 1.0)
                        cur = erdos_c5(h, n)
                        improved = True
                except Exception:
                    pass
            # Direction 2: i down, j up, k up
            d_lo2 = 0.0
            d_hi2 = min(hi0, 2.0 * (1.0 - hj0), 2.0 * (1.0 - hk0))
            if d_hi2 > 1e-14:
                def obj2(d):
                    a, b, c = hi0 - d, hj0 + 0.5*d, hk0 + 0.5*d
                    if a < -1e-12 or b > 1+1e-12 or c > 1+1e-12:
                        return 1e100
                    hh = h.copy()
                    hh[i], hh[j], hh[k] = a, b, c
                    return erdos_c5(hh, n)
                try:
                    rs = so.minimize_scalar(obj2, bounds=(d_lo2, d_hi2), method='bounded',
                                           options={'maxiter': 15, 'xatol': 1e-10})
                    if rs.fun < cur - 1e-15:
                        d = float(rs.x)
                        h[i] -= d
                        h[j] += 0.5 * d
                        h[k] += 0.5 * d
                        h = np.clip(h, 0.0, 1.0)
                        cur = erdos_c5(h, n)
                        improved = True
                except Exception:
                    pass
        if not improved:
            break
    return h, cur
