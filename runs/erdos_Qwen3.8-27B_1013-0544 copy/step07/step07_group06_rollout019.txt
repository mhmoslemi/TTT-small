def lse_objective_grad(h,n,beta):
    h=np.clip(np.asarray(h,float).ravel(),0,1)
    dx=2.0/float(n)
    v=1.0-h
    c=np.correlate(h,v,mode='full')*dx
    c=np.maximum(c,0.0)
    m=float(c.max())
    y=beta*(c-m)
    e=np.exp(np.minimum(y,0.0))
    s=float(e.sum())
    if not finite or s<=0:
       p=np.full_like(c,1.0/(2*n-1)); F=m
    else:
       p=e/s; F=m+np.log(s)/beta
    P=p; R=P[::-1]
    A=np.convolve(R,v,mode='full')[n-1:2*n-1]
    B=np.convolve(P,h,mode='full')[n-1:2*n-1]
    g=dx*(A-B)
    if not np.all(np.isfinite(g)): g=np.zeros_like(h)
    return float(F), g

def lbfgs_objective(x,n,beta,pen):
    h=np.clip(np.asarray(x,float).ravel(),0,1)
    F,g=lse_objective_grad(h,n,beta)
    dev=float(h.sum()-0.5*n)
    if pen !=0:
       F +=0.5*pen*dev*dev
       g = g + pen*dev
    if not np.isfinite(F): return 1e100, np.zeros_like(h)
    return float(F), g

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(max(0, n - a.size))])

    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(a - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid

def compute_base_binary(h, n):
    l = np.arange(-(n - 1), n, dtype=int)

    return base

def lse_objective_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    F, g = lse_objective_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g


def candidate_variants(h, n):
    h = project_feasible(h, n)
    out = [
        h,
        h[::-1],
        1.0 - h[::-1],
        0.5 * (h + h[::-1]),
        0.5 * (h + 1.0 - h[::-1]),
    ]
    if n % 2 == 0:
        out.append(top_n_binary(h, n))
    return [project_feasible(x, n) for x in out]


def polish_solution(h, n, betas, pen, deadline, stage_time):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        if time.time() > deadline - 5.0:
            break
        now = time.time()
        stage_dl = min(deadline - 3.0, now + max(2.0, min(stage_time, 0.5 * max(2.0, deadline - now))))
        _stop_time = stage_dl
        options = {"maxiter": 1000000, "maxls": 30, "ftol": 1e-13, "gtol": 1e-6}
        try:
            res = minimize(
                lbfgs_objective, work, method="L-BFGS-B", jac=True,
                bounds=bounds, args=(n, float(beta), float(pen)),
                callback=stop_callback, options=options
            )
            x = res.x if (res is not None and hasattr(res, "x")) else work
        except Exception:
            x = work
        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc:
            bc = cw
            best = work.copy()
        for cand in candidate_variants(work, n):
            cc = compute_c5(cand, n)
            if cc < bc:
                bc = cc
                best = cand.copy()
        if bc < cw - 1e-12:
            work = best.copy()
    return best, bc


def resample_h(h, n_old, n_new):
    if n_old == n_new:
        return np.asarray(h, dtype=float).copy()
    x0 = np.linspace(0.0, 2.0, n_old)
    x1 = np.linspace(0.0, 2.0, n_new)
    return np.interp(x1, x0, np.asarray(h, dtype=float))


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total = float(budget_s)
    except Exception:
        total = 1100.0
    total = max(25.0, total)
    deadline = t0 + max(5.0, total - 25.0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
        if n0 < 2 or not np.all(np.isfinite(h0)):
            raise ValueError("bad initial")
    except Exception:
        n0 = 400
        h0 = ((np.arange(n0) // 50) % 2).astype(float)

    best_n = n0
    best_h = project_feasible(h0, best_n)
    best_c = compute_c5(best_h, best_n)
    for cand in candidate_variants(best_h, best_n):
        cc = compute_c5(cand, best_n)
        if cc < best_c:
            best_c = cc
            best_h = cand.copy()

    first_even = best_n if best_n % 2 == 0 else best_n + 1
    order = [200, 100, 400, first_even, 300]
    n_list = []
    for x in order:
        if 50 <= x <= 800 and x % 2 == 0 and x not in n_list:
            n_list.append(x)

    rem0 = max(5.0, deadline - t0)
    reserve = min(0.40 * rem0, max(15.0, 0.28 * rem0))
    binary_end = t0 + max(5.0, rem0 - reserve)
    if binary_end < t0 + 8.0:
        binary_end = t0 + max(3.0, min(8.0, rem0))

    slice_map = {100: 85.0, 200: 260.0, 300: 115.0, 400: 210.0}
    for n in n_list:
        if time.time() >= binary_end or best_c <= GOAL - 1e-6:
            break
        sl = slice_map.get(n, 110.0)
        n_end = min(binary_end, time.time() + sl)
        seed_h = project_feasible(resample_h(best_h, best_n, n), n)
        target_M = int(np.floor(GOAL * (n / 2.0) + 1e-9))
        target_M = max(1, target_M)
        starts = [top_n_binary(seed_h, n).astype(float)]
        starts += random_binary_starts(n, target_M, 4 if n <= 200 else 2, rng)

        for st in starts:
            if time.time() >= n_end:
                break
            sub_end = min(n_end, time.time() + (35.0 if n <= 200 else 55.0))
            hb, cb = binary_search(st, n, sub_end, rng, target_M)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
                best_n = n
            for cand in candidate_variants(hb, n):
                cc = compute_c5(cand, n)
                if cc < best_c:
                    best_c = cc
                    best_h = cand.copy()
                    best_n = n
            if best_c <= 0.3800:
                n_end = time.time()

    if time.time() < deadline - 20.0:
        betas = (20.0, 80.0, 300.0, 1200.0, 5000.0, 20000.0)
        st_time = 30.0 if best_n > 300 else 22.0
        hp, cp = polish_solution(best_h, best_n, betas, 1000.0, deadline - 15.0, st_time)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()
        for cand in candidate_variants(best_h, best_n):
            cc = compute_c5(cand, best_n)
            if cc < best_c:
                best_c = cc
                best_h = cand.copy()

    if time.time() < deadline - 12.0:
        for scale in (0.004, 0.012):
            if time.time() >= deadline - 8.0:
                break
            nh = project_feasible(best_h + scale * rng.randn(best_n), best_n)
            hp, cp = polish_solution(nh, best_n, (1500.0, 8000.0), 1000.0, deadline - 6.0, 15.0)
            if cp < best_c:
                best_c = cp
                best_h = hp.copy()
            for cand in candidate_variants(best_h, best_n):
                cc = compute_c5(cand, best_n)
                if cc < best_c:
                    best_c = cc
                    best_h = cand.copy()

    if time.time() < deadline - 10.0:
        for nn in (100, 200):
            if time.time() >= deadline - 7.0:
                break
            hr = project_feasible(resample_h(best_h, best_n, nn), nn)
            cr = compute_c5(hr, nn)
            if cr < best_c:
                best_c = cr
                best_h = hr.copy()
                best_n = nn
            sub_end = min(deadline - 4.0, time.time() + 15.0)
            target_M = max(1, int(np.floor(GOAL * (nn / 2.0) + 1e-9)))
            hb, cb = binary_search(hr, nn, sub_end, rng, target_M)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
                best_n = nn

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
