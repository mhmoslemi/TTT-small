E_i[n-1+d] = (1-h[i+d]) - h[i-d] (for valid indices, 0 otherwise)
E_i[n-1-d] = (1-h[i-d]) - h[i+d] (for valid indices, 0 otherwise)

Then for a pair change (h[i] += δ, h[j] -= δ):
dR[k] = δ * (E_i[k] - E_j[k])

And the new R = R + δ * (E_i - E_j)

The objective is max(R) * dx, so I want to minimize max(R + δ*(E_i - E_j)).

This is a 1D optimization problem in δ! I can solve it with golden section search or by finding the point where the max is minimized.

This is much faster than recomputing the full correlation each time.

Let me implement this:

```python
def pair_search_fast(R, h, i, j, n):
    """Fast pair search using precomputed effect vectors."""
    omh = 1.0 - h
    # Compute E_i and E_j
    # E_i[k] for k = 0, ..., 2n-2
    # E_i[n-1] = 1 - 2*h[i]
    # E_i[n-1+d] = omh[i+d] - h[i-d] (if valid)
    # E_i[n-1-d] = omh[i-d] - h[i+d] (if valid)
    
    Ei = np.zeros(2*n-1)
    Ej = np.zeros(2*n-1)
    
    off = n - 1
    Ei[off] = 1.0 - 2.0*h[i]
    Ej[off] = 1.0 - 2.0*h[j]
    
    # For d = 1, 2, ..., n-1:
    # E[off+d]: contribution from position i is omh[i+d] - h[i-d]
    # E[off-d]: contribution from position i is omh[i-d] - h[i+d]
    for d in range(1, n):
        # off + d
        ip = i + d
        im = i - d
        if 0 <= ip < n:
            Ei[off+d] += omh[ip]
        if 0 <= im < n:
            Ei[off+d] -= h[im]
        
        # off - d
        if 0 <= im < n:
            Ei[off-d] += omh[im]
        if 0 <= ip < n:
            Ei[off-d] -= h[ip]
        
        # Same for j
        jp = j + d
        jm = j - d
        if 0 <= jp < n:
            Ej[off+d] += omh[jp]
        if 0 <= jm < n:
            Ej[off+d] -= h[jm]
        if 0 <= jm < n:
            Ej[off-d] += omh[jm]
        if 0 <= jp < n:
            Ej[off-d] -= h[jp]
    
    D = Ei - Ej  # dR = delta * D
    
    # Now minimize max(R + delta * D) over delta
    # Subject to: 0 <= h[i] + delta <= 1 and 0 <= h[j] - delta <= 1
    # i.e., -h[i] <= delta <= 1-h[i] and h[j]-1 <= delta <= h[j]
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    
    if hi - lo < 1e-14:
        return None
    
    # Golden section search on f(delta) = max(R + delta * D)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr*(b-a)
    d1 = a + gr*(b-a)
    fc = float(np.max(R + c1 * D))
    fd = float(np.max(R + d1 * D))
    
    for _ in range(50):
        if b - a < 1e-13:
            break
        if fc < fd:
            b = d1
            d1 = c1
            fd = fc
            c1 = b - gr*(b-a)
            fc = float(np.max(R + c1 * D))
        else:
            a = c1
            c1 = d1
            fc = fd
            d1 = a + gr*(b-a)
            fd = float(np.max(R + d1 * D))
    
    delta_best = 0.5*(a+b)
    new_max = float(np.max(R + delta_best * D))
    old_max = float(np.max(R))
    
    if new_max < old_max - 1e-14:
        return delta_best, new_max
    return None
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-10) & (v < 1.0 - 1e-10)
        c = int(m.sum())
        if c > 0:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    s = float(np.exp(sh).sum())
    if s <= 0:
        return m
    return m + np.log(s) / alpha

def lse_weights(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
        return q
    q = e / s
    r = q.sum()
    if r > 0:
        q /= r
    return q

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    q = lse_weights(c, alpha)
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    omh = 1.0 - x
    for idx in np.nonzero(q > 1e-15)[0]:
        k = int(idx) - off
        w = float(q[idx])
        if k > 0:
            m = n - k
            g[k:n] += w * omh[:m]
            g[:m] -= w * x[k:]
        elif k < 0:
            s = -k
            m = n - s
            g[:m] += w * omh[s:]
            g[s:] -= w * x[:m]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def pen_obj(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_obj(x, n, alpha) + 0.5 * mu * d * d

def pen_grad(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_grad(x, n, alpha) + mu * d

def lbfgsb_opt(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(pen_obj, h, args=(n, alpha, mu), method="L-BFGS-B",
                       jac=pen_grad, bounds=bounds,
                       options={"maxiter": maxiter, "maxfun": 4*maxiter,
                                "ftol": 1e-16, "gtol": 1e-10})
        if hasattr(res, "x") and res.x is not None:
            return res.x
    except Exception:
        pass
    return h

def fast_pair_search(R, h, i, j, n):
    """O(n) pair search using effect vectors."""
    omh = 1.0 - h
    off = n - 1
    L = 2 * n - 1
    
    # Compute D = E_i - E_j (the change in R per unit delta)
    D = np.zeros(L)
    # Diagonal (shift 0)
    D[off] = (1.0 - 2.0*h[i]) - (1.0 - 2.0*h[j])
    # Off-diagonal terms
    # For shift d (positive): R[off+d] involves h[n]*(1-h[n+d])
    # Effect of +delta at position p: adds delta*(1-h[p+d]) to R[off+d]
    #                               subtracts delta*h[p-d] from R[off+d]
    # Effect of +delta at position p: adds delta*(1-h[p-d]) to R[off-d]
    #                               subtracts delta*h[p+d] from R[off-d]
    
    for d in range(1, n):
        # R[off+d]: +delta at i contributes delta*(omh[i+d] - h[i-d])
        #           +delta at j contributes delta*(omh[j+d] - h[j-d])
        vi = 0.0
        vj = 0.0
        ip = i + d
        im = i - d
        jp = j + d
        jm = j - d
        if ip < n:
            vi += omh[ip]
        if im >= 0:
            vi -= h[im]
        if jp < n:
            vj += omh[jp]
        if jm >= 0:
            vj -= h[jm]
        D[off + d] = vi - vj
        
        # R[off-d]: +delta at i contributes delta*(omh[i-d] - h[i+d])
        vi2 = 0.0
        vj2 = 0.0
        if im >= 0:
            vi2 += omh[im]
        if ip < n:
            vi2 -= h[ip]
        if jm >= 0:
            vj2 += omh[jm]
        if jp < n:
            vj2 -= h[jp]
        D[off - d] = vi2 - vj2
    
    # Minimize max(R + delta*D) over feasible delta
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return None
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    d1 = a + gr * (b - a)
    fc = float(np.max(R + c1 * D))
    fd = float(np.max(R + d1 * D))
    
    for _ in range(35):
        if b - a < 1e-13:
            break
        if fc < fd:
            b = d1
            d1 = c1
            fd = fc
            c1 = b - gr * (b - a)
            fc = float(np.max(R + c1 * D))
        else:
            a = c1
            c1 = d1
            fc = fd
            d1 = a + gr * (b - a)
            fd = float(np.max(R + d1 * D))
    
    delta_best = 0.5 * (a + b)
    new_R = R + delta_best * D
    new_max = float(new_R.max())
    old_max = float(R.max())
    
    if new_max < old_max - 1e-14:
        return delta_best, new_R, new_max
    return None

def intense_pair_search(h, n, deadline, seed=1, anneal_temp=0.0):
    """Very fast pair local search with optional annealing."""
    h = project_box_sum(h, n)
    R = compute_c(h, n)
    bestc = float(R.max())
    besth = h.copy()
    if n < 4 or time.time() > deadline:
        return h, bestc
    
    rng = np.random.RandomState(seed)
    noimp = 0
    maxno = 5000
    off = n - 1
    
    while time.time() < deadline and noimp < maxno:
        # Select lag proportional to correlation
        pos = R[off + 1:]
        neg = R[off - 1::-1]
        w = np.maximum(pos, neg) + 1e-12
        w = w / w.sum()
        
        found = False
        for _ in range(5):
            d = int(rng.choice(n, p=w))
            d = max(1, min(d, n - 1))
            i = int(rng.randint(n - d))
            j = i + d
            res = fast_pair_search(R, h, i, j, n)
            if res is not None:
                delta, new_R, new_max = res
                h[i] += delta
                h[j] -= delta
                R = new_R
                if new_max < bestc:
                    bestc = new_max
                    besth = h.copy()
                noimp = 0
                found = True
                break
        if not found:
            noimp += 1
    
    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc

def interpolate_h(h, nold, nnew):
```
