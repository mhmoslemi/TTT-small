    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(3):
        if abs(diff) <= 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y

def subgrad_avg(h, n, topk=5):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    idx = np.argsort(c)[::-1][:topk]
    grad = np.zeros(n)
    dx = 2.0 / n
    mid = n - 1
    for m in idx:
        m = int(m)
        s = m - mid
        if s == 0:
            grad += (1.0 - 2.0 * h) * dx
        elif s > 0:
            grad[:n - s] += (1.0 - h[:n - s]) * dx
            grad[s:] += -h[:n - s] * dx
        else:
            s2 = -s
            grad[s2:] += (1.0 - h[:n - s2]) * dx
            grad[:n - s2] += -h[s2:] * dx
    return grad / topk, c

def proximal_descent(h, n, deadline, topk=5, alpha0=0.02, max_iter=2000):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * n
    best_c = erdos_c5(h, n)
    best_h = h.copy()
    alpha = alpha0
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline - 5.0:
            break
        grad, c = subgrad_avg(h, n, topk)
        cur = float(np.max(c))
        found = False
        for _ls in range(15):
            h_new = project_box_sum(h - alpha * grad, target)
            c_new = erdos_c5(h_new, n)
            if c_new < cur - 1e-13:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    stall = 0
                else:
                    stall += 1
                found = True
                break
            alpha *= 0.6
        if not found:
            alpha *= 0.4
            stall += 1
            if stall > 30:
                alpha = alpha0 * np.sqrt(0.5)
                stall = 0
    return best_h, best_c

def softmax_vals(c, beta):
    maxc = float(np.max(c))
    z = beta * (c - maxc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    f = maxc + np.log(ws) / beta
    if ws <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ew / ws
    return f, w

def _LS():
    return {'n': 1, 'dx': 1.0, 'target': 1.0, 'beta': 1.0}

def lbfgs_obj(x):
    st = _LS()
    n = int(st['n'])
    dx = float(st['dx'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_vals(c, float(st['beta']))
    mid = n - 1
    g = np.zeros(n)
    for m in range(len(w)):
        if w[m] < 1e-15:
            continue
        s = m - mid
        if s == 0:
            g += (1.0 - 2.0 * x) * w[m] * dx
        elif s > 0:
            g[:n - s] += (1.0 - x[:n - s]) * w[m] * dx
            g[s:] += -x[:n - s] * w[m] * dx
        else:
            s2 = -s
            g[s2:] += (1.0 - x[:n - s2]) * w[m] * dx
            g[:n - s2] += -x[s2:] * w[m] * dx
    return f, g

LS_STATE = {'n': 1, 'dx': 1.0, 'beta': 1.0}

def lbfgs_f(x):
    st = LS_STATE
    n = int(st['n'])
    dx = float(st['dx'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_vals(c, float(st['beta']))
    mid = n - 1
    g = np.zeros(n)
    for m in range(len(w)):
        if w[m] < 1e-15:
            continue
        s = m - mid
        if s == 0:
            g += (1.0 - 2.0 * x) * w[m] * dx
        elif s > 0:
            g[:n - s] += (1.0 - x[:n - s]) * w[m] * dx
            g[s:] += -x[:n - s] * w[m] * dx
        else:
            s2 = -s
            g[s2:] += (1.0 - x[:n - s2]) * w[m] * dx
            g[:n - s2] += -x[s2:] * w[m] * dx
    return f, g

def lbfgs_refine(h, beta, maxiter, deadline):
    h = np.asarray(h, dtype=float)
    n = len(h)
    LS_STATE['n'] = n
    LS_STATE['dx'] = 2.0 / n
    LS_STATE['beta'] = float(beta)
    target = 0.5 * n
    rem = deadline - time.time()
    if rem < 3.0:
        return h, erdos_c5(h, n)
    mx = int(min(maxiter, max(10, int(rem / 4.0))))
    bounds = [(0.0, 1.0)] * n
    try:
        res = so.minimize(lbfgs_f, h, method='L-BFGS-B', jac=True,
                          bounds=bounds, options={'maxiter': mx, 'ftol': 1e-16,
                                                   'gtol': 1e-13, 'maxls': 50})
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)

def pair_line(h, n, i, j, cur, deadline):
    hi0 = h[i]
    hj0 = h[j]
    lo = max(-hi0, hj0 - 1.0)
    hi = min(1.0 - hi0, hj0)
    if hi - lo < 1e-13:
        return h, cur
    best_d, best_c = 0.0, cur
    a = hi0 + lo
    b = hj0 + lo
    hh = h.copy(); hh[i] = a; hh[j] = b
    c0 = erdos_c5(hh, n)
    if c0 < best_c:
        best_c, best_d = c0, lo
    a = hi0 + hi; b = hj0 + hi
    hh = h.copy(); hh[i] = a; hh[j] = b
    c1 = erdos_c5(hh, n)
    if c1 < best_c:
        best_c, best_d = c1, hi
    if time.time() < deadline - 0.2:
        try:
            def _obj(d):
                a2 = hi0 + d; b2 = hj0 - d
                if a2 < -1e-12 or a2 > 1.0 + 1e-12 or b2 < -1e-12 or b2 > 1.0 + 1e-12:
                    return 1e100
                hh2 = h.copy(); hh2[i] = a2; hh2[j] = b2
                return erdos_c5(hh2, n)
            rs = so.minimize_scalar(_obj, bounds=(lo, hi), method='bounded',
                                    options={'maxiter': 30, 'xatol': 1e-10})
            c2 = _obj(float(rs.x))
            if c2 < best_c:
                best_c, best_d = c2, float(rs.x)
        except Exception:
            pass
    if best_c < cur - 1e-14:
        hh = h.copy()
        hh[i] += best_d
        hh[j] -= best_d
        return np.clip(hh, 0.0, 1.0), best_c
    return h, cur

def pair_polish(h, n, deadline, seed, nrounds=4):
    rng = np.random.RandomState(int(seed))
    h = np.asarray(h, dtype=float).copy()
    cur = erdos_c5(h, n)
    m = 30 if n <= 500 else 20
    frac = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if frac.size:
        score = np.minimum(h[frac], 1.0 - h[frac])
        order = frac[np.argsort(score)[::-1]]
    else:
        order = np.array([], dtype=int)
    if order.size >= m:
        top = order[:m]
    else:
        extra = rng.choice(n, size=max(2, m - int(order.size)), replace=False)
        top = np.unique(np.concatenate((order, extra)))
    if top.size < 2:
        return h, cur
    for _r in range(nrounds):
        improved = False
        pairs = [(int(top[a]), int(top[b])) for a in range(len(top)) for b in range(a + 1, len(top))]
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline - 3.0:
                return h, erdos_c5(h, n)
            hh, cc = pair_line(h, n, i, j, cur, deadline - 3.0)
            if cc < cur - 1e-14:
                h, cur = hh, cc
                improved = True
        if not improved:
            break
    return h, cur

def resample(h, n_old, n_new):
    x = np.linspace(0.0, 2.0, n_old)
    y = np.linspace(0.0, 2.0, n_new)
    hh = np.interp(y, x, np.asarray(h, dtype=float))
    return project_box_sum(hh, 0.5 * n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(10.0, float(budget_s) - 5.0)
    deadline = t0 + total
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    h0 = project_box_sum(h0, 0.5 * n0)
    best_h = h0.copy()
    best_n = n0
    best_c = erdos_c5(best_h, best_n)

    # Phase 1: L-BFGS with increasing sharpness on n0
    end1 = t0 + 0.30 * total
    for beta, mx in [(500, 100), (5000, 80), (50000, 70), (500000, 50), (5000000, 40)]:
        if time.time() > end1 - 5.0:
            break
        cand, c = lbfgs_refine(best_h, beta, mx, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand.copy(), c

    # Phase 2: Proximal gradient with true subgradient
    end2 = t0 + 0.50 * total
    h2, c2 = proximal_descent(best_h, best_n, end2, topk=5, alpha0=0.03, max_iter=3000)
    if c2 < best_c - 1e-15:
        best_h, best_c = h2.copy(), c2
    h2, c2 = proximal_descent(best_h, best_n, end2, topk=10, alpha0=0.02, max_iter=2000)
    if c2 < best_c - 1e-15:
        best_h, best_c = h2.copy(), c2
    h2, c2 = proximal_descent(best_h, best_n, end2, topk=2, alpha0=0.01, max_iter=2000)
    if c2 < best_c - 1e-15:
        best_h, best_c = h2.copy(), c2

    # Phase 3: Up-resolve to n=800 and optimize
    end3 = t0 + 0.78 * total
    n_hi = 800
    hh = resample(best_h, best_n, n_hi)
    hi_c = erdos_c5(hh, n_hi)
    if hi_c < best_c + 0.005:
        for beta, mx in [(2000, 80), (20000, 70), (200000, 60), (2000000, 40)]:
            if time.time() > end3 - 10.0:
                break
            cand, c = lbfgs_refine(hh, beta, mx, deadline)
            if c < hi_c - 1e-15:
                hh, hi_c = cand.copy(), c
        hh, c = proximal_descent(hh, n_hi, end3 - 5.0, topk=5, alpha0=0.015, max_iter=1500)
        if c < hi_c:
            hh, hi_c = c and hh, c
        if hi_c < best_c - 1e-15:
            best_h, best_n, best_c = hh.copy(), n_hi, hi_c

    # Phase 4: Pair polish
    end4 = deadline - 8.0
    if time.time() < end4 - 20.0:
        best_h, best_c = pair_polish(best_h, best_n, end4 - 15.0, seed, nrounds=5)

    # Phase 5: Final L-BFGS + proximal
    if time.time() < end4 - 10.0:
        cand, c = lbfgs_refine(best_h, 1000000, 30, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand.copy(), c
    if time.time() < end4 - 5.0:
        cand, c = proximal_descent(best_h, best_n, end4, topk=3, alpha0=0.01, max_iter=500)
        if c < best_c - 1e-15:
            best_h, best_c = cand.copy(), c

    # Final polish
    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
