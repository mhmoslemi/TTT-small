```python
def coord_descent_step(h, n, dx, i):
    """Optimize h[i] while keeping others fixed, for the max objective."""
    # Compute all a_s and b_s for the linear functions c_s(x) = a_s * x + b_s
    # where x = h[i]
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    mid = n - 1
    
    # For each lag s, compute a_s = dc_s/dh[i] and b_s = c_s - a_s * h[i]
    a = np.zeros(2*n-1)
    for s_idx in range(2*n-1):
        s = s_idx - mid  # actual shift
        # c_s = sum_j h[j] * (1-h[j+s]) for valid j
        # dc_s/dh[i] = (1-h[i+s]) - h[i-s] (with bounds checking)
        grad_i = 0.0
        if 0 <= i+s < n:
            grad_i += (1.0 - h[i+s])
        if 0 <= i-s < n:
            grad_i -= h[i-s]
        a[s_idx] = grad_i * dx
    
    b = c - a * h[i]
    
    # Now minimize max_s (a_s * x + b_s) over x ∈ [0,1]
    # Find all pairwise intersections
    candidates = [0.0, 1.0]
    for p in range(2*n-1):
        for q in range(p+1, 2*n-1):
            if abs(a[p] - a[q]) < 1e-15:
                continue
            xint = (b[q] - b[p]) / (a[p] - a[q])
            if 0.0 <= xint <= 1.0:
                candidates.append(xint)
    
    # Also check where each line is individually maximized (but since they're linear, 
    # the vertex is always at an intersection or boundary)
    
    best_x = h[i]
    best_val = np.max(c)
    for x in candidates:
        val = np.max(a * x + b)
        if val < best_val:
            best_val = val
            best_x = x
    
    return best_x, best_val
```

    c = np.correlate(h, 1.0 - h, mode='full') * dx
