    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def lbfgsb_optimize(h, n, alpha, maxiter=5000):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        val = m + logsum / alpha
        val += 100.0 * (x.sum() - n / 2.0) ** 2
        return val
    def grad(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        weights = np.exp(shifted - sm)
        weights /= weights.sum()
        g = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            if weights[k] < 1e-10:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                if lag == 0:
                    g[k] += weights[k] * dx * (1.0 - 2.0 * x)
                else:
                    g[k] += weights[k] * dx * np.concatenate([
                        -x[lag:], 1.0 - x[:n - lag]])
            else:
                kk = -lag
                g[k] += weights[k] * dx * np.concatenate([
                        1.0 - x[kk:], -x[:n - kk]])
        g_con = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] < 1e-10:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                if lag == 0:
                    g_con += weights[k] * dx * (1.0 - 2.0 * x)
                else:
                    g_con[lag:] += weights[k] * dx * (1.0 - x[:n - lag])
                    g_con[:n - lag] -= weights[k] * dx * x[lag:]
            else:
                kk = -lag
                g_con[:n - kk] += weights[k] * dx * (1.0 - x[kk:])
                g_con[kk:] -= weights[k] * dx * x[:n - kk]
        g_con += 200.0 * (x.sum() - n / 2.0)
        return g_con
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=[(0, 1)] * n, options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: Quick SLSQP from parent
    p1 = t0 + 0.12 * (dl - t0)
    for alpha in [100000, 1e6, 1e7]:
        if time.time() > p1:
            break
        h = slsqp_optimize(best_h, n, alpha, 1500, 1e-15)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 2: Fast binary search from continuous
    p2 = t0 + 0.35 * (dl - t0)
    hb, cb = fast_binary_search(best_h, n, p2, max_sweeps=60)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 3: Try alternative initial patterns + binary search
    p3 = t0 + 0.65 * (dl - t0)
    patterns = [golden_ratio_binary(n, seed), thue_morse_binary(n),
                golden_ratio_binary(n, seed + 100),
                np.tile([1.0, 1.0, 0.0, 0.0], n // 4 + 1)[:n]]
    for pat in patterns:
        if time.time() > p3:
            break
        pat = project_feasible(pat, n)
        # Quick SLSQP
        hp = slsqp_optimize(pat, n, 1e6, 800, 1e-14)
        cp = compute_c5(hp, n)
        if cp < best_c:
            best_c, best_h = cp, hp.copy()
        # Binary search from this
        hp2, cp2 = fast_binary_search(hp, n, min(time.time() + 30, p3), max_sweeps=30)
        if cp2 < best_c:
            best_c, best_h = cp2, hp2.copy()
    # Phase 4: L-BFGS-B polish on best
    p4 = t0 + 0.80 * (dl - t0)
    h = best_h.copy()
    for alpha in [5e5, 5e6, 5e7]:
        if time.time() > p4:
            break
        h = lbfgsb_optimize(h, n, alpha, 3000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 5: Another binary round on the best continuous
    p5 = t0 + 0.92 * (dl - t0)
    hb2, cb2 = fast_binary_search(best_h, n, p5, max_sweeps=40)
    if cb2 < best_c:
        best_c, best_h = cb2, hb2.copy()
    # Phase 6: Final SLSQP
    if time.time() < dl - 10:
        for alpha in [1e7, 1e8]:
            if time.time() > dl - 5:
                break
            h = slsqp_optimize(best_h, n, alpha, 1000, 1e-16)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
