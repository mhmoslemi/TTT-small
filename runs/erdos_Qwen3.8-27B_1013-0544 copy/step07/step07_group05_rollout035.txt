def softmax_vals(c, beta):
    mc = float(c.max())
    z = np.exp(np.clip(beta * (c - mc), -745, 0))
    ws = z.sum()
    if ws < 1e-300:
        w = np.zeros_like(c)
        w[int(c.argmax())] = 1.0
        return mc, w
    f = mc + np.log(ws) / beta
    return f, z / ws

def c5_grad(h, n, beta):
    """Gradient of smooth max of correlations."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    f, w = softmax_vals(c, beta)
    g = np.zeros(n)
    mid = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        if k <= mid:
            shift = mid - k
            m = n - shift
            if shift == 0:
                g += wk * (1.0 - 2.0 * h) * dx
            else:
                g[:m] += wk * (1.0 - h[shift:]) * dx
                g[shift:] -= wk * h[:m] * dx
        else:
            shift = k - mid
            m = n - shift
            g[shift:] += wk * (1.0 - h[:m]) * dx
            g[:m] -= wk * h[shift:] * dx
    return f, g

    c = np.correlate(x, 1.0 - x, mode='full') * dx

            g[shift:] += wk * (1.0 - x[:m]) * dx
            g[:m] -= wk * x[shift:] * dx
    res = x.sum() - target

    rem = deadline - time.time()
    if rem < 3.0:

    hh = h.copy()
    hh[i] += delta

    """1D line search along direction that moves mass from j to i."""
    hi0 = h[i]
    hj0 = h[j]
    lo = max(-hi0, hj0 - 1.0) + 1e-15
    hi = min(1.0 - hi0, hj0) - 1e-15

    c_ = b - gr * (b - a)
    d_ = a + gr * (b - a)

    """Compute the change in each correlation coefficient when h[i]+=delta, h[j]-=delta."""
    # corr[k] = sum_n h[n+k] * (1-h[n])
    # Index k ranges from -(n-1) to (n-1), mapped to full index k + (n-1)
    # For simplicity, work with the full correlation array
    # Change in corr at full index m:
    # h[i] as first factor (n+k = i, so k = i - n_idx): valid when 0 <= i-n_idx < n, i.e., 0 <= i-m_valid...
    
    # Let me use a different indexing. In the full correlation:
    # full[m] = sum over valid (a,b) pairs with a-b = m-(n-1) of h[a] * (1-h[b])
    # Equivalently: let lag = m - (n-1), then full[m] = sum_{b} h[b+lag] * (1-h[b])
    # where 0 <= b < n and 0 <= b+lag < n
    
    # When h[i] changes by delta:
    # Case 1: i is the first factor: h[i] * (1-h[b]) where i = b + lag, so b = i - lag
    #   For full index m: lag = m-(n-1), b = i - m + (n-1)
    #   Valid when 0 <= b < n, i.e., 0 <= i-m+(n-1) < n
    #   Contribution: delta * (1-h[b]) = delta * (1-h[i-m+(n-1)])
    #

    #   Valid when 0 <= a < n, i.e., 0 <= i+m-(n-1) < n
    #   Contribution: -h[a] * delta = -(1-h[a])*delta + delta... 
    #   Wait: (1-h[i]) → (1-h[i]-delta), so the change is -h[a]*delta
    #   Contribution: -h[i+m-(n-1)] * delta

    # Case 3: j is the first factor: contribution: -delta * (1-h[j-m+(n-1)])
    # Case 4: j is the second factor: contribution: +h[j+m-(n-1)] * delta
    
    # Total change at full index m:
    # delta * [(1-h[i-m+(n-1)])_valid - h[i+m-(n-1)]_valid - (1-h[j-m+(n-1)])_valid + h[j+m-(n-1)]_valid]

    # b2 = j - m + mid  (for case 3: j is first factor)
    # a2 = j + m - mid  (for case 4: j is second factor)
    
    m = np.arange(2 * n - 1)
    b1 = i - m + mid  # needs 0 <= b1 < n
    a1 = i + m - mid  # needs 0 <= a1 < n

def pair_line_fast(corr_vals, h, n, i, j, cur_obj, deadline):
    """
    Fast 1D line search.
    corr_vals: current correlation values (2n-1 array, already * dx)
    Returns (new_h, new_obj) or (h, cur_obj) if no improvement.
    """
    mid = n - 1
    m_arr = np.arange(2 * n - 1)
    b1 = i - m_arr + mid
    a1 = i + m_arr - mid
    b2 = j - m_arr + mid
    a2 = j + m_arr - mid
    v1 = (b1 >= 0) & (b1 < n)
    v2 = (a1 >= 0) & (a1 < n)
    v3 = (b2 >= 0) & (b2 < n)
    v4 = (a2 >= 0) & (a2 < n)
    # Precompute the coefficients
    coef = np.zeros(2 * n - 1)
    coef[v1] += (1.0 - h[b1[v1]])
    coef[v2] -= h[a1[v2]]
    coef[v3] -= (1.0 - h[b2[v3]])
    coef[v4] += h[a2[v4]]
    dx = 2.0 / n
    
    def obj(delta):
        return float(np.max(corr_vals + delta * coef)) * 1.0  # corr_vals already has dx
    
    # Find bounds
    lo = max(-h[i], h[j] - 1.0) + 1e-12
    hi = min(1.0 - h[i], h[j]) - 1e-12
    if hi - lo < 1e-12:
        return h, cur_obj
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a, b = lo, hi
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = obj(c_pt)
    fd = obj(d_pt)
    for _ in range(50):
        if time.time() > deadline:
            break
        if b - a < 1e-9:
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = obj(c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = obj(d_pt)
    
    best_d = 0.5 * (a + b)
    best_obj = obj(best_d)
    if 0 < best_d < hi or 0 > best_d > lo:
        best_obj = min(best_obj, obj(0.0))
        if obj(0.0) < best_obj:
            best_d = 0.0
            best_obj = obj(0.0)
    
    if best_obj < cur_obj - 1e-15:
        new_h = h.copy()
        new_h[i] += best_d
        new_h[j] -= best_d
        return np.clip(new_h, 0.0, 1.0), best_obj
    return h, cur_obj

import time
import numpy as np
import scipy.optimize as so

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_box_sum(x, target):
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) < 1e-15:
        return y
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - y.sum()
    mask = (y > 1e-15) & (y < 1.0 - 1e-15)
    if mask.sum() > 0:
        y[mask] += diff / mask.sum()
    return np.clip(y, 0.0, 1.0)

def softmax_vals(c, beta):
    mc = float(c.max())
    z = np.exp(np.clip(beta * (c - mc), -745.0, 0.0))
    ws = z.sum()
    if ws < 1e-300:
        w = np.zeros_like(c)
        w[int(c.argmax())] = 1.0
        return mc, w
    return mc + np.log(ws) / beta, z / ws

def lbfgs_fg(x, n, beta, target):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_vals(c, beta)
    g = np.zeros(n)
    mid = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        if k <= mid:
            shift = mid - k
            m = n - shift
            if shift == 0:
                g += wk * (1.0 - 2.0 * x) * dx
            else:
                g[:m] += wk * (1.0 - x[shift:]) * dx
                g[shift:] -= wk * x[:m] * dx
        else:
            shift = k - mid
            m = n - shift
            g[shift:] += wk * (1.0 - x[:m]) * dx
            g[:m] -= wk * x[shift:] * dx
    res = x.sum() - target
    f += 0.5 * 1e5 * res * res
    g += 1e5 * res
    return f, g

def refine_lbfgs(h, n, beta, maxiter, deadline):
    target = 0.5 * n
    rem = deadline - time.time()
    if rem < 3.0:
        return h, erdos_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    mx = min(maxiter, max(10, int(rem / 2.5)))
    try:
        res = so.minimize(lbfgs_fg, h, args=(n, beta, target), method='L-BFGS-B',
                          jac=True, bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 40})
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)

    m_arr = np.arange(2 * n - 1)

    a1 = i + m_arr - mid

    v3 = (b2 >= 0) & (b2 < n)

        else:
            a = c_pt; c_pt = d_pt; fc = fd
            d_pt = a + gr * (b - a)
            fd = float(np.max(corr_vals + d_pt * coef))
    best_d = 0.5 * (a + b)
    best_obj = float(np.max(corr_vals + best_d * coef))
    o0 = cur_obj
    if o0 < best_obj:
        best_d = 0.0
        best_obj = o0
    if best_obj < cur_obj - 1e-15:
        nh = h.copy()
        nh[i] += best_d
        nh[j] -= best_d
        return np.clip(nh, 0.0, 1.0), best_obj
    return h, cur_obj

def coord_descent(h, n, deadline):
    target = 0.5 * n
    cur_obj = erdos_c5(h, n)
    corr_vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    offsets = [1, n // 4, n // 2, 3 * n // 4]
    offsets = [o for o in offsets if 0 < o < n]
    max_sweeps = 200
    for sweep in range(max_sweeps):
        if time.time() > deadline - 5.0:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline - 5.0:
                break
            for off in offsets:
                j = (i + off) % n
                if j == i:
                    continue
                h, cur_obj = pair_line_fast(corr_vals, h, n, i, j, cur_obj, deadline)
                if h is not h:  # this never triggers, just for clarity
                    break
        # Recompute correlation periodically
        if sweep % 5 == 4:
            corr_vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        # Check for improvement in the sweep
        cur_check = erdos_c5(h, n)
        if cur_check < cur_obj - 1e-14:
            cur_obj = cur_check
        elif cur_check > cur_obj:
            cur_obj = cur_check
    h = project_box_sum(h, target)
    return h, erdos_c5(h, n)

def coord_descent_v2(h, n, deadline):
    """Alternative: random pair selection with efficiency tracking."""
    target = 0.5 * n
    cur_obj = erdos_c5(h, n)
    corr_vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    rng = np.random.RandomState(12345)
    max_iters = 50000
    for it in range(max_iters):
        if time.time() > deadline - 5.0:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        h, cur_obj = pair_line_fast(corr_vals, h, n, i, j, cur_obj, deadline)
        if it % 200 == 199:
            corr_vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    h = project_box_sum(h, target)
    return h, erdos_c5(h, n)

def resample(h, n, nh):
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, nh)
    hh = np.interp(y, x, h)
    return project_box_sum(hh, 0.5 * nh)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total_time = max(10.0, float(budget_s) - 5.0)
    deadline = t0 + total_time

    # Start from parent
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 300
        h = np.full(n, 0.5)
    if n < 2:
        n = 300
        h = np.full(n, 0.5)
    if n > 700:
        h = np.interp(np.linspace(0, 2, 700), np.linspace(0, 2, n), h)
        n = 700
    h = project_box_sum(h, 0.5 * n)

    best_h = h.copy()
    best_n = n
    best_c = erdos_c5(best_h, best_n)

    # Phase 1: L-BFGS with increasing beta (20%)
    end1 = t0 + 0.20 * total_time
    betas_p1 = [200.0, 1000.0, 5000.0, 20000.0, 100000.0, 500000.0]
    for beta in betas_p1:
        if time.time() > end1 - 5.0:
            break
        cand, c = refine_lbfgs(best_h, best_n, beta, 150, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand.copy(), c

    # Phase 2: Structured coordinate descent (40%)
    end2 = t0 + 0.60 * total_time
    best_h, best_c = coord_descent(best_h, best_n, end2)

    # Phase 3: Random pair coordinate descent (20%)
    end3 = t0 + 0.80 * total_time
    cand_h, cand_c = coord_descent_v2(best_h, best_n, end3)
    if cand_c < best_c - 1e-15:
        best_h, best_c = cand_h.copy(), cand_c

    # Phase 4: Try different resolutions (10%)
    end4 = t0 + 0.90 * total_time
    for nh in [200, 250, 300, 500]:
        if time.time() > end4 - 10.0:
            break
        if nh == best_n:
            continue
        hh = resample(best_h, best_n, nh)
        for beta in [10000.0, 200000.0]:
            hh, c = refine_lbfgs(hh, nh, beta, 40, deadline)
            if c < best_c + 5e-4:
                hh, c = coord_descent(hh, nh, min(deadline - 5, end4))
                if c < best_c - 1e-14:
                    best_h, best_n, best_c = hh.copy(), nh, c
                    break

    # Phase 5: Final polish (10%)
    end5 = deadline - 3.0
    if time.time() < end5 - 10.0:
        cand, c = refine_lbfgs(best_h, best_n, 2000000.0, 30, end5)
        if c < best_c - 1e-15:
            best_h, best_c = cand.copy(), c
    if time.time() < end5 - 5.0:
        best_h, best_c = coord_descent(best_h, best_n, end5)

    # Final stochastic retry
    if time.time() < end5 - 3.0:
        rng = np.random.RandomState(int(seed))
        for sig in [0.01, 0.03]:
            if time.time() > end5 - 3.0:
                break
            hh = project_box_sum(best_h + sig * rng.randn(best_n), 0.5 * best_n)
            hh, c = coord_descent(hh, best_n, min(deadline - 2, time.time() + 20))
            if c < best_c - 1e-15:
                best_h, best_c = hh.copy(), c

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
