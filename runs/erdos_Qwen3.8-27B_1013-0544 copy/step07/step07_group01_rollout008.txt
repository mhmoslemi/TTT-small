def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:

            s = room.sum()
            if s < 1e-14:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:

            h = h + diff / s * h

def compute_smooth_gradient(h, n, alpha):

    dx = 2.0 / n

    if p_sum < 1e-30:
