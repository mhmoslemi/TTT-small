def proj_grad(h, n, deadline, lr, momentum_beta, top_k, max_iters):
    dx = 2.0 / n
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c = compute_c5(h, n)
    pos = h.copy()  # Nesterov position
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Evaluate at Nesterov lookahead
        pos = h + momentum_beta * v
        
        # Compute correlations
        corr = np.correlate(pos, 1 - pos, mode='full') * dx
        
        # Find top-k lags
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        
        # Compute subgradient at pos
        grad = np.zeros(n)
        for idx in top_idx:
            shift = idx - (n - 1)
            ks = abs(shift)
            # dC/dh[j] = (1-h[j-ks])*[j>=ks] - h[j+ks]*[j+ks<n]
            # But we need this at pos, not h
            if ks == 0:
                grad += 1.0 - 2.0 * pos
            else:
                grad[ks:] += (1.0 - pos[:-ks])
                grad[:-ks] -= pos[ks:]
        grad /= top_k
        
        # Update
        v = momentum_beta * v - lr * grad
        h_new = project_feasible(h + v, n)
        
        # Check improvement
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c

def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:

            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgrad_c5(h, n, top_k=3):
    """Compute subgradient of max_k corr_k at h, using top_k lags."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    if top_k >= len(corr):
        top_idx = np.arange(len(corr))
    else:
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    grad = np.zeros(n)

        shift = idx - (n - 1)
        ks = abs(shift)
        if ks == 0:
            grad += 1.0 - 2.0 * h
        else:
            grad[ks:] += (1.0 - h[:-ks])
            grad[:-ks] -= h[ks:]
    return dx * grad / top_k, corr.max() * dx

        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

Let me also add **pairwise swap local search**: try all O(n²) swaps but evaluate them in O(1) each using the precomputed flips. Wait, it's O(n) each (since we need to compute max of the new correlation vector), but with numpy it's fast.

Actually, for a full greedy sweep:
- For each zero position a (n/2 = 200 positions):
  - Compute base = corr + pos_flip[a] (O(m) = O(n))
  - For each one position b (n/2 = 200 positions):
    - new_c = max(base - pos_flip[b]) (O(m) = O(n))
  - Pick the best b, if it improves
- Total per sweep: (n/2) * (n/2) * O(n) = O(n³) but with numpy vectorization, each inner loop is a (n/2, m) operation

With numpy: for each a, compute cands = base[None,:] - pos_flip[ones] which is (200, 799). Then take max along axis 1: (200,). This is ~160K ops, very fast in numpy.

Total per sweep: 200 * 160K = 32M ops → ~0.1s in numpy
After each accepted swap, recompute pos_flip: O(n*m) = 320K ops → ~0.001s

So a full greedy sweep with, say, 50 accepted swaps: 0.1s (evaluations) + 50*0.001s (recomputes) ≈ 0.15s

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def precompute_flip(hb, n):
    m = 2 * n - 1
    pos = np.arange(n)[:, None]
    kf = abs(np.arange(m) - (n - 1))[None, :]
    m1 = (pos >= kf)
    d1 = np.where(m1, 1.0 - hb[pos - kf], 0.0)
    m2 = (pos + kf < n)
    d2 = np.where(m2, -hb[pos + kf], 0.0)
    result = d1 + d2
    result[:, n - 1] = 0.0
    return result

def binary_greedy_sweep(hb, n, deadline, max_sweeps=1000):
    dx = 2.0 / n
    corr = np.correlate(hb, 1 - hb, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = hb.copy()
    best_c = cur_c
    pos_flip = precompute_flip(hb, n)
    zeros = np.where(hb == 0)[0].tolist()
    ones = list(np.where(hb == 1)[0])
    ones_arr = np.array(ones)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        any_improve = False
        # Shuffle zeros for diversity
        rng_local = np.random.RandomState(sweep + 42)
        sz = list(zeros)
        rng_local.shuffle(sz)
        for ai in sz:
            base = corr + pos_flip[ai]
            cands = base[None, :] - pos_flip[ones_arr]
            new_cs = cands.max(axis=1) * dx
            bi = np.argmin(new_cs)
            if new_cs[bi] < cur_c - 1e-14:
                bj = ones[bi]
                corr += pos_flip[ai] - pos_flip[bj]
                hb[ai] = 1.0
                hb[bj] = 0.0
                cur_c = float(new_cs[bi])
                # Update lists
                zeros.remove(ai)
                zeros.append(bj)
                ones_arr = np.delete(ones_arr, bi)
                ones_arr = np.append(ones_arr, ai)
                ones.remove(bj)
                ones.append(ai)
                # Recompute flip effects
                pos_flip = precompute_flip(hb, n)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = hb.copy()
                any_improve = True
        if not any_improve:
            break
    return best_h, best_c

def binary_sa(hb, n, deadline, rng, T0=0.015, cooling=0.999):
    dx = 2.0 / n
    corr = np.correlate(hb, 1 - hb, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = hb.copy()
    best_c = cur_c
    pos_flip = precompute_flip(hb, n)
    zeros = list(np.where(hb == 0)[0])
    ones = list(np.where(hb == 1)[0])
    ones_arr = np.array(ones)
    T = T0
    no_improve = 0
    
    while time.time() < deadline:

        if accept:
            corr += pos_flip[ai] - pos_flip[bj]

                best_h = hb.copy()
                no_improve = 0
            else:

def slsqp_opt(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h

def cont_grad_step(h, n, lr, beta, top_k, max_iters, deadline):
    """Projected gradient with momentum for continuous problem."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Evaluate gradient at Nesterov point
        y = h + beta * v
        y = np.clip(y, 0, 1)
        corr = np.correlate(y, 1 - y, mode='full')
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(n)
        for idx in top_idx:
            s = idx - (n - 1)
            ks = abs(s)
            if ks == 0:
                grad += 1.0 - 2.0 * y
            else:
                grad[ks:] += (1.0 - y[:-ks])
                grad[:-ks] -= y[ks:]
        grad *= dx / top_k
        v = beta * v + grad
        h_new = project_feasible(h - lr * v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c
