def binary_search(h_init, n, deadline, max_sweeps=100):

    h[np.argsort(h_init)[::-1][:n//2]] = 1.0

    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    m_range = np.arange(-(n-1), n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            best_gain = 1e-15
            best_j = -1
            for j in ones:
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n-1 + (j-i)
                if 0 <= k1 < len(delta):
                    delta[k1] += 1.0
                k2 = n-1 + (i-j)
                if 0 <= k2 < len(delta):
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - best_gain:
                    best_gain = cur_c - nc
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                # Recompute delta for the actual swap
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]  # h[i] is now 1, so 1-h=0
                # Actually this is getting complicated. Let me just recompute corr.
                corr = np.correlate(h, 1-h, mode='full').astype(float)
                cur_c = corr.max() * dx
                ones.remove(best_j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(best_j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

            s = room.sum()

            s = h.sum()
            if s < 1e-14:

            h += diff / s * h

def lse_target(x, n, alpha):

    c = np.correlate(x, 1.0 - x, mode='full')
    cmax = c.max()
    sh = np.minimum(alpha * (c - cmax), 0.0)
    return alpha * cmax * dx + np.log(np.exp(sh).sum())

def lse_grad(x, n, alpha):

    t2 = np.convolve(x, p, 'full')[n-1:2*n-1]
