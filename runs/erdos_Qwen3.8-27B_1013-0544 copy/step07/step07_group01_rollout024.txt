def binary_local_search(h, n, deadline, seed):
    # Start from a binary version of h
    threshold = sorted(h)[n//2]
    h_bin = (h >= threshold).astype(float)
    # Adjust to have exactly n/2 ones
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    if len(ones) > n//2:
        # Remove excess ones (those with lowest h values)
        excess = ones[np.argsort(h[ones])[:len(ones)-n//2]]
        h_bin[excess] = 0
    elif len(ones) < n//2:
        # Add needed ones (those with highest h values among zeros)
        needed = zeros[np.argsort(-h[zeros])[:n//2-len(ones)]]
        h_bin[needed] = 1
    
    best_h = h_bin.copy()
    best_c = compute_c5(best_h, n)
    
    rng = np.random.RandomState(seed)
    T = 0.1
    ones_set = set(np.where(h_bin == 1)[0].tolist())
    zeros_set = set(np.where(h_bin == 0)[0].tolist())
    
    while time.time() < deadline:
        # Pick random 1 and 0 to swap
        a = rng.choice(list(ones_set))
        b = rng.choice(list(zeros_set))
        
        h_bin[a] = 0
        h_bin[b] = 1
        
        new_c = compute_c5(h_bin, n)
        
        if new_c <= best_c or rng.rand() < np.exp((best_c - new_c) / T):
            # Accept
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
            ones_set.discard(a)
            ones_set.add(b)
            zeros_set.discard(b)
            zeros_set.add(a)
        else:
            # Reject
            h_bin[a] = 1
            h_bin[b] = 0
        
        # Cool down
        T *= 0.9999
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_sa(h, n, deadline, seed, T_initial=0.05, T_min=0.001, cool=0.99995):
    # Create binary version
    h_sorted = np.sort(h)
    threshold = h_sorted[n // 2]
    h_bin = (h >= threshold).astype(float)
    ones_idx = np.where(h_bin == 1)[0]
    zeros_idx = np.where(h_bin == 0)[0]
    if len(ones_idx) > n // 2:
        excess = ones_idx[np.argsort(h[ones_idx])[:len(ones_idx) - n // 2]]
        h_bin[excess] = 0
        ones_idx = np.setdiff1d(ones_idx, excess)
        zeros_idx = np.union1d(zeros_idx, excess)
    elif len(ones_idx) < n // 2:
        needed = zeros_idx[np.argsort(-h[zeros_idx])[:n // 2 - len(ones_idx)]]
        h_bin[needed] = 1
        zeros_idx = np.setdiff1d(zeros_idx, needed)
        ones_idx = np.union1d(ones_idx, needed)
    
    best_h = h_bin.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(seed)
    T = T_initial
    
    while time.time() < deadline:
        if T < T_min:
            T = T_min
        a = ones_idx[rng.randint(len(ones_idx))]
        b = zeros_idx[rng.randint(len(zeros_idx))]
        if a == b:
            continue
        h_bin[a] = 0
        h_bin[b] = 1
        new_c = compute_c5(h_bin, n)
        delta = new_c - best_c
        if delta <= 0 or rng.rand() < np.exp(-delta / T):
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
            # Update sets
            ones_idx = np.array([x for x in ones_idx if x != a] + [b])
            zeros_idx = np.array([x for x in zeros_idx if x != b] + [a])
        else:
            h_bin[a] = 1
            h_bin[b] = 0
        T *= cool
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > n // 2:
        ones_idx = np.where(h_bin == 1)[0]
        remove = ones_idx[np.argsort(h[ones_idx])[:n_ones - n // 2]]
        h_bin[remove] = 0
    elif n_ones < n // 2:
        zeros_idx = np.where(h_bin == 0)[0]
        add = zeros_idx[np.argsort(-h[zeros_idx])[:n // 2 - n_ones]]
        h_bin[add] = 1
    return h_bin

def slsqp_optimize(h, n, alpha, maxiter, ftol):

        sm = sh.max()
        return float(m + (sm + np.log(np.exp(sh - sm).sum())) / alpha)
    def cons_fun(x):

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
