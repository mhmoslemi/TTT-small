import time
import numpy as np
import scipy.optimize as so

PAIR = {}


def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / float(n)))


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    diff = float(target - y.sum())
    if abs(diff) > 1e-13:
        lo, hi = -1.0, 2.0
        for _ in range(55):
            t = 0.5 * (lo + hi)
            s = float(np.clip(x - t, 0.0, 1.0).sum())
            if s > target:
                lo = t
            else:
                hi = t
        y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
        diff = float(target - y.sum())
        if abs(diff) > 1e-14:
            m = (y > 1e-12) & (y < 1.0 - 1e-12)
            cnt = int(np.count_nonzero(m))
            if cnt:
                y[m] += diff / cnt
            else:
                y += diff / float(n)
            y = np.clip(y, 0.0, 1.0)
    return y


def lag_grad_index(h, r, n, dx):
    g = np.zeros(n, dtype=float)
    mid = n - 1
    l = int(r) - mid
    if l == 0:
        g[:] = 1.0 - 2.0 * h
    elif l > 0:
        q = l
        if q < n:
            g[:n - q] -= h[q:]
            g[q:] += 1.0 - h[:-q]
    else:
        q = -l
        if q < n:
            g[:n - q] += 1.0 - h[q:]
            g[q:] -= h[:-q]
    return g * dx


def select_active(c, count, tol):
    count = int(min(max(1, count), c.size))
    C = float(c.max())
    idx = np.flatnonzero(c >= C - tol)
    if idx.size <= count:
        return idx
    return np.argpartition(c, -count)[-count:]


def active_average_gradient(h, n, count=None, tol=1e-6):
    if count is None:
        count = max(12, n // 5)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = select_active(c, count, tol)
    g = np.zeros(n, dtype=float)
    for r in idx:
        g += lag_grad_index(h, int(r), n, dx)
    if idx.size:
        g /= float(idx.size)
    return g, c


def scale_direction(h, d, step_cap):
    maxd = float(np.max(np.abs(d)))
    if maxd < 1e-14:
        return 0.0
    f = step_cap / maxd
    pos = d > 1e-15
    if np.any(pos):
        f = min(f, float(np.min((1.0 - h[pos]) / (d[pos] + 1e-300))))
    neg = d < -1e-15
    if np.any(neg):
        f = min(f, float(np.min(h[neg] / ((-d[neg]) + 1e-300))))
    if not np.isfinite(f) or f <= 0.0:
        return 0.0
    return f


def line_search_dir(h, d, n, deadline, cur):
    best_c = cur
    best_a = 0.0
    best_h = h

    hh = project_box_sum(best_h, 0.5 * n)

    return hh, best_c


def active_grad_refine(h, n, deadline, max_steps=12, step_cap=0.06, count=None):

    cur = erdos_c5(h, n)

    cur = erdos_c5(h, n)

    h, c = optimize_start(best_h.copy(), best_n, end_parent, seed)
    if c < best_c:
        best_h, best_c, best_n = h, c, n

    end_sym = t0 + 0.42 * total
    hs = project_box_sum(0.5 * (best_h + best_h[::-1]), 0.5 * best_n)
    h, c = optimize_start(hs, best_n, end_sym, seed + 1)
    if c < best_c:
        best_h, best_c, best_n = h, c, best_n

    small = []
    if best_n > 160:
        small.append(2 * (best_n // 4))
    if best_n > 320:
        small.append(300)
    elif best_n > 200:
        small.append(200)
    seen = set()
    small2 = []
    for x in small:
        x = int(x)
        x = (x // 2) * 2
        if 80 <= x < best_n and x not in seen:
            seen.add(x)
            small2.append(x)

    for i, nh in enumerate(small2[:2]):
        end = t0 + (0.52 + 0.09 * i) * total
        h = start_at_n(best_h, best_n, nh)
        h, c = optimize_start(h, nh, end, seed + 20 + i)
        if c < best_c:
            best_h, best_c, best_n = h, c, nh

    if best_n < 700:
        nlarge = min(800, max(600, best_n + 200))
        nlarge = (nlarge // 2) * 2
        if nlarge > best_n:
            end_large = t0 + 0.72 * total
            h = start_at_n(best_h, best_n, nlarge)
            h, c = optimize_start(h, nlarge, end_large, seed + 50)
            if c < best_c:
                best_h, best_c, best_n = h, c, nlarge

    if time.time() < t0 + 0.84 * total - 10:
        end_mut = t0 + 0.86 * total
        rng = np.random.RandomState(int(seed) + 1234)
        hn = project_box_sum(best_h + 0.04 * rng.randn(best_n), 0.5 * best_n)
        hs = project_box_sum(0.5 * (hn + hn[::-1]), 0.5 * best_n)
        if erdos_c5(hs, best_n) < erdos_c5(hn, best_n):
            hn = hs
        h, c = optimize_start(hn, best_n, end_mut, seed + 1234)
        if c < best_c:
            best_h, best_c, best_n = h, c, best_n

    end_final = deadline - 3.0
    h, c = optimize_start(best_h.copy(), best_n, end_final, seed + 9999)
    if c < best_c:
        best_h, best_c, best_n = h, c, best_n

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
