def smgrad(h,w,n,dx):
    mid=n-1
    if n==1:return (1.0-2.0*h)*w[mid]*dx
    wp=np.zeros(n);wn=np.zeros(n)
    wp[1:]=w[mid+1:];wn[1:]=w[:mid][::-1]
    s=wp+wn;pp=np.cumsum(wp);pn=np.cumsum(wn);idx=np.arange(n)
    const=pp+pn[n-1-idx];kern=np.zeros(2*n-1)
    kern[mid+1:]=s[1:];kern[:mid][::-1]=s[1:]
    neigh=np.convolve(h,kern,mode='full')[mid:mid+n]
    return (w[mid]*(1.0-2.0*h)+const-neigh)*dx
