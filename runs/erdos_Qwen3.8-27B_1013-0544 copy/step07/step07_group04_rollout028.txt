def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def project_box_sum(v, n):

    tgt = 0.5 * n
    lo = float(v.min()) - 1.0

def grad_shift(h, n, d):
    dx = 2.0 / n
    g = np.zeros(n)
    if d >= 0:
        g[d:] += (1.0 - h[:n - d]) * dx
        g[:n - d] -= h[d:] * dx
    else:
        dd = -d
        g[:n - dd] += (1.0 - h[dd:]) * dx
        g[dd:] -= h[:n - dd] * dx
    return g


def lse_f(x, n, alpha):
