def binary_pair_search(h0, n, deadline, seed=42):
    """Local search on binary vectors: swap pairs to maintain sum constraint."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    # Create binary from h0
    h = np.zeros(n)
    order = np.argsort(-h0)
    h[order[:n//2]] = 1.0
    
    corr = np.correlate(h, 1.0-h, mode='full') * dx
    best_c = float(corr.max())
    best_h = h.copy()
    
    ones_idx = np.nonzero(h == 1)[0]
    zeros_idx = np.nonzero(h == 0)[0]
    
    it = 0
    no_improve = 0
    while time.time() < deadline and it < 2000000:
        it += 1
        if no_improve > 100000:
            break
        # Pick a random 1 and random 0 to swap
        pi = rng.randint(len(ones_idx))
        qi = rng.randint(len(zeros_idx))
        p = ones_idx[pi]
        q = zeros_idx[qi]
        
        # Perform swap
        h[p] = 0.0
        h[q] = 1.0
        corr = np.correlate(h, 1.0-h, mode='full') * dx
        c = float(corr.max())
        
        if c <= best_c + 1e-14:
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            ones_idx[pi] = q
            zeros_idx[qi] = p
        else:
            # Revert
            h[p] = 1.0
            h[q] = 0.0
            no_improve += 1
    
    return best_h, best_c

def directed_binary_search(h0, n, deadline, seed=42):

    zeros_set = set(np.nonzero(h == 0)[0])
    ones_list = np.array(list(ones_set))
    zeros_list = np.array(list(zeros_set))
    
    no_improve = 0
    it = 0
    while time.time() < deadline and it < 2000000:
        it += 1
        if no_improve > 200000:
            break
        
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        
        # For positive lag: want to swap h[p]=1 with h[q]=0 where q-p relates to lag
        # The overlap at lag is sum of h[i]*(1-h[i+lag]) for valid i
        # To reduce it: if h[i]=1 and h[i+lag]=0, either:
        #   - set h[i]=0 (need to set some other h[j]=1)
        #   - set h[i+lag]=1 (need to set some other h[j]=1)
        # A direct swap: swap h[i] and h[i+lag] directly!
        # But that might not help (it just moves the 1)
        
        # Better: find i where h[i]=1 and h[i+lag]=0, then swap h[i] with some h[j]=0
        # where j is NOT in the "bad" region for lag
        
        if lag >= 0:
            # Bad positions: i where h[i]=1 and h[i+lag]=0, for i in [0, n-1-lag]
            bad_ones = [i for i in range(n - lag) if h[i] == 1 and h[i+lag] == 0]
            if not bad_ones:
                # Try random
                if len(ones_list) < 2:
                    break
                p = ones_list[rng.randint(len(ones_list))]
                q = zeros_list[rng.randint(len(zeros_list))]
            else:
                p = bad_ones[rng.randint(len(bad_ones))]
                # Find a zero to swap with that's not in the bad region
                # Just pick random zero
                q = zeros_list[rng.randint(len(zeros_list))]
                if q == p:
                    continue
                if h[q] == 1:
                    continue
        else:
            ll = -lag
            bad_ones = [i for i in range(ll, n) if h[i] == 1 and h[i-ll] == 0]
            if not bad_ones:
                if len(ones_list) < 2:
                    break
                p = ones_list[rng.randint(len(ones_list))]
                q = zeros_list[rng.randint(len(zeros_list))]
            else:
                p = bad_ones[rng.randint(len(bad_ones))]
                q = zeros_list[rng.randint(len(zeros_list))]
                if q == p:
                    continue
                if h[q] == 1:
                    continue
        
        if h[p] != 1 or h[q] != 0:
            continue
        
        h[p] = 0.0
        h[q] = 1.0
        corr = np.correlate(h, 1.0-h, mode='full') * dx
        c = float(corr.max())
        
        if c <= best_c + 1e-14:
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            ones_set.discard(p)
            ones_set.add(q)
            zeros_set.discard(q)
            zeros_set.add(p)
            ones_list = np.array(sorted(ones_set))
            zeros_list = np.array(sorted(zeros_set))
        else:
            h[p] = 1.0
            h[q] = 0.0
            no_improve += 1
    
    return best_h, best_c

                    continue
                best_h[i] = val
                best_h = project_feasible(best_h, n)
                c = compute_c5(best_h, n)
                if c <= best_c + 1e-14:
                    best_c = c
                    improved = True
                else:
                    best_h[i] = old_val
                    best_h = project_feasible(best_h, n)
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with moderate alpha
    t1 = min(t0 + 120, deadline - 300)
    h1 = slsqp_opt(best_h, n, 2000, 3000, 1e-14)
    c1 = compute_c5(h1, n)
    if c1 < best_c:
        best_c, best_h = c1, h1

    # Phase 2: Binary directed local search
    t2 = min(t0 + 350, deadline - 250)
    bh, bc = bin_search(best_h, n, t2, seed=seed)
    if bc < best_c:
        best_c, best_h = bc, bh

    # Phase 3: SLSQP high alpha from best
    t3 = min(t0 + 550, deadline - 200)
    for a in [10000, 100000, 1000000]:
        if time.time() > t3 - 20:
            break
        h3 = slsqp_opt(best_h, n, a, 3000, 1e-15)
        c3 = compute_c5(h3, n)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: Coordinate descent on continuous solution
    t4 = min(t0 + 700, deadline - 150)
    hc, cc = coord_descent(best_h, n, t4, max_rounds=30)
    if cc < best_c:
        best_c, best_h = cc, hc

    # Phase 5: Binary search again from improved solution
    t5 = min(t0 + 850, deadline - 100)
    bh2, bc2 = bin_search(best_h, n, t5, seed=seed + 7)
    if bc2 < best_c:
        best_c, best_h = bc2, bh2

    # Phase 6: Final SLSQP with very high alpha
    t6 = deadline - 30
    for a in [1000000, 5000000]:
        if time.time() > t6 - 15:
            break
        h6 = slsqp_opt(best_h, n, a, 2000, 1e-15)
        c6 = compute_c5(h6, n)
        if c6 < best_c:
            best_c, best_h = c6, h6

    # Phase 7: Final coordinate descent polish
    if time.time() < deadline - 10:
        hf, cf = coord_descent(best_h, n, deadline - 5, max_rounds=20)
        if cf < best_c:
            best_c, best_h = cf, hf

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
