import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def gcd_matrix(h, n):
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        a = max(0, -k)
        b = min(n, n - k)
        if b > a:
            G[a:b, m] = 1.0 - h[a + k:b + k]
        a = max(0, k)
        b = min(n, n + k)
        if b > a:
            G[a:b, m] -= h[a - k:b - k]
    return G

def lp_optimize(h, n, eps_init, max_iters, deadline):
    best_h = h.copy()
    best_mc = float(np.correlate(h, 1 - h, mode='full').max())
    eps = eps_init
    no_improve = 0
    for i in range(max_iters):
        if time.time() > deadline:
            break
        G = gcd_matrix(h, n)
        C = np.correlate(h, 1.0 - h, mode='full').astype(float)
        delta = cp.Variable(n)
        t_var = cp.Variable()
        cons = [C + G.T @ delta <= t_var,
                cp.abs(delta) <= eps,
                delta >= -h, delta <= 1.0 - h,
                cp.sum(delta) == 0]
        prob = cp.Problem(cp.Minimize(t_var), cons)
        solved = False
        for sol in ['CLARABEL', 'ECOS', 'GLOP']:
            try:
                prob.solve(solver=sol)
                if delta.value is not None:
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break
        d = delta.value
        improved = False
        for lam in [1.0, 0.7, 0.5, 0.3, 0.1]:
            h_try = np.clip(h + lam * d, 0.0, 1.0)
            h_try = project_feasible(h_try, n)
            mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
            if mc_try < best_mc - 1e-13:
                best_mc = mc_try
                best_h = h_try.copy()
                h = h_try
                improved = True
                break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 8:
                eps *= 0.65
                if eps < 0.0005:
                    break
                no_improve = 0
    return best_h, best_mc

def slsqp_polish(h, n, deadline):
    dx = 2.0 / n
    alphas = [2000, 10000, 50000, 200000, 1000000]
    current = h.copy()
    best_mc = float(np.correlate(current, 1 - current, mode='full').max())
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            g = np.correlate(x, 1.0 - x, mode='full') * dx
            m = g.max()
            shifted = alpha * (g - m)
            sm = shifted.max()
            return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

        def obj_grad(x):
            g = np.correlate(x, 1.0 - x, mode='full') * dx
            m = g.max()
            shifted = alpha * (g - m)
            sm = shifted.max()
            w = np.exp(shifted - sm)
            w /= w.sum()
            return gcd_matrix(x, n).T @ w

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, current, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-18})
            current = project_feasible(res.x, n)
            mc = float(np.correlate(current, 1 - current, mode='full').max())
            if mc < best_mc:
                best_mc = mc
        except Exception:
            pass
    return current, best_mc
