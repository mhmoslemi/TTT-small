def project_feasible(h, n):
    target = n / 2.0

        if abs(diff) < 1e-15:

            if s < 1e-15:

            s = h.sum()
            if s < 1e-15:

            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_c5_obj(x, n, alpha):
    """Smooth approximation of C5 using log-sum-exp."""
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    lse = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    return lse

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def c5_obj_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    exp_sh = np.exp(sh - sm)
    lse = sm + np.log(exp_sh.sum())
    val = m + lse / alpha
    p = exp_sh / exp_sh.sum()
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        k = ki - (n - 1)
        w = p[ki]
        if w < 1e-13:
            continue
        if k >= 0 and k < n:
            grad[:n-k] += w * (1.0 - x[k:])
            grad[k:] -= w * x[:n-k]
        elif k < 0 and (-k) < n:
            lk = -k
            grad[lk:] += w * (1.0 - x[:n-lk])
            grad[:n-lk] -= w * x[lk:]
    grad *= dx
    return val, grad
