def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt: lo = mid
        else: hi = mid
    v = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    for _ in range(30):
        d = tgt - v.sum()
        if abs(d) < 1e-14: break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c: v[m] += d / c
        else: v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    return m + np.log(float(np.exp(sh).sum())) / alpha

def lse_grad(x, n, alpha):

    m = float(c.max())
    e = np.exp(alpha * (c - m))
    q = e / e.sum()
    dx = 2.0 / n
    off = n - 1

            g[s:] += w * (1.0 - x[:m_len])

        elif k < 0:

            g[:m_len] += w * (1.0 - x[s:])
            g[s:] -= w * x[:m_len]

def pair_eval(h, c_base, i, j, xs, n, x0, y0, S):

    ax2 = x0 - xs

    si = n - 1 - i
    delta[:, si:si+n] += ax2[:, None] * h[None, :]
    delta[:, j:j+n] += ax2[:, None] * omhrev[None, :]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt: lo = mid
        else: hi = mid
    v = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    for _ in range(30):
        d = tgt - v.sum()
        if abs(d) < 1e-14: break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c: v[m] += d / c
        else: v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    return m + np.log(float(np.exp(sh).sum())) / alpha

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    e = np.exp(alpha * (c - m))
    q = e / e.sum()
    dx = 2.0 / n
    off = n - 1
    g = np.zeros(n)
    for idx in np.flatnonzero(q > 1e-15):
        w = q[idx]
        k = idx - off
        if k > 0:
            g[k:] += w * (1.0 - x[:n-k])
            g[:n-k] -= w * x[k:]
        elif k < 0:
            g[:n+k] += w * (1.0 - x[-k:])
            g[-k:] -= w * x[:n+k]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def pen_obj(x, n, a, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_obj(x, n, a) + 0.5 * mu * d * d

def pen_grad(x, n, a, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_grad(x, n, a) + mu * d

def lbfgsb(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    try:
        res = minimize(pen_obj, h, args=(n, alpha, mu), method="L-BFGS-B",
                       jac=pen_grad, bounds=[(0.0,1.0)]*n,
                       options={"maxiter":maxiter,"maxfun":4*maxiter,"ftol":1e-15,"gtol":1e-8})
        if hasattr(res,"x") and res.x is not None:
            return project_box_sum(res.x, n)
    except Exception: pass
    return h

def pair_eval(h, c_base, i, j, xs, n, x0, y0, S):
    L = 2*n - 1
    off = n - 1
    dx = 2.0 / n
    omh = 1.0 - h
    omhrev = omh[::-1]
    delta = np.zeros((len(xs), L), dtype=float)
    ax1 = xs - x0
    ax2 = x0 - xs
    yv = S - xs
    delta[:, i:i+n] += ax1[:,None] * omhrev[None,:]
    si = n - 1 - i
    delta[:, si:si+n] += ax2[:,None] * h[None,:]
    delta[:, j:j+n] += ax2[:,None] * omhrev[None,:]
    sj = n - 1 - j
    delta[:, sj:sj+n] += ax1[:,None] * h[None,:]
    idx0 = off
    bs = ax1*(1.0-x0) + x0*ax2 + ax2*(1.0-y0) + y0*ax1
    es = xs*(1.0-xs) + yv*(1.0-yv) - (x0*(1.0-x0) + y0*(1.0-y0))
    delta[:, idx0] += es - bs
    d = j - i
    ip = off - d
    bp = ax1*(1.0-y0) + x0*ax1
    ep = xs*(1.0-yv) - x0*(1.0-y0)
    delta[:, ip] += ep - bp
    im = off + d
    bm = ax2*(1.0-x0) + y0*ax2
    em = yv*(1.0-xs) - y0*(1.0-x0)
    delta[:, im] += em - bm
    return np.max(c_base[None,:] + dx*delta, axis=1)

def pair_search(h, c_base, i, j, n):
    if i > j: i, j = j, i
    x0, y0 = float(h[i]), float(h[j])
    S = x0 + y0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt: lo = mid
        else: hi = mid
    v = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    for _ in range(30):
        d = tgt - v.sum()
        if abs(d) < 1e-14: break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c: v[m] += d / c
        else: v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    return m + np.log(float(np.exp(sh).sum())) / alpha

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    e = np.exp(alpha * (c - m))
    q = e / e.sum()
    dx = 2.0 / n
    off = n - 1
    g = np.zeros(n)
    for idx in np.flatnonzero(q > 1e-15):
        w = q[idx]
        k = idx - off
        if k > 0:
            g[k:] += w * (1.0 - x[:n-k])
            g[:n-k] -= w * x[k:]
        elif k < 0:
            g[:n+k] += w * (1.0 - x[-k:])
            g[-k:] -= w * x[:n+k]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def pen_obj(x, n, a, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_obj(x, n, a) + 0.5 * mu * d * d

def pen_grad(x, n, a, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_grad(x, n, a) + mu * d

def lbfgsb(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    try:
        res = minimize(pen_obj, h, args=(n, alpha, mu), method="L-BFGS-B",
                       jac=pen_grad, bounds=[(0.0,1.0)]*n,
                       options={"maxiter":maxiter,"maxfun":4*maxiter,"ftol":1e-15,"gtol":1e-8})
        if hasattr(res,"x") and res.x is not None:
            return project_box_sum(res.x, n)
    except Exception: pass
    return h

def pair_eval(h, c_base, i, j, xs, n, x0, y0, S):
    L = 2*n - 1
    off = n - 1
    dx = 2.0 / n
    omh = 1.0 - h
    omhrev = omh[::-1]
    delta = np.zeros((len(xs), L), dtype=float)
    ax1 = xs - x0
    ax2 = x0 - xs
    yv = S - xs
    delta[:, i:i+n] += ax1[:,None] * omhrev[None,:]
    si = n - 1 - i
    delta[:, si:si+n] += ax2[:,None] * h[None,:]
    delta[:, j:j+n] += ax2[:,None] * omhrev[None,:]
    sj = n - 1 - j
    delta[:, sj:sj+n] += ax1[:,None] * h[None,:]
    idx0 = off
    bs = ax1*(1.0-x0) + x0*ax2 + ax2*(1.0-y0) + y0*ax1
    es = xs*(1.0-xs) + yv*(1.0-yv) - (x0*(1.0-x0) + y0*(1.0-y0))
    delta[:, idx0] += es - bs
    d = j - i
    ip = off - d
    bp = ax1*(1.0-y0) + x0*ax1
    ep = xs*(1.0-yv) - x0*(1.0-y0)
    delta[:, ip] += ep - bp
    im = off + d
    bm = ax2*(1.0-x0) + y0*ax2
    em = yv*(1.0-xs) - y0*(1.0-x0)
    delta[:, im] += em - bm
    return np.max(c_base[None,:] + dx*delta, axis=1)
