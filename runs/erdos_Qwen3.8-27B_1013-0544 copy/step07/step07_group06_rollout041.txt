import time

import numpy as np
from scipy.optimize import minimize

_stop_time = 0.0

def _stop_cb(*args):
    return time.time() > _stop_time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c)) if c.size else 0.0

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        a = np.concatenate([a, np.zeros(n - a.size)]) if a.size < n else a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(a - m, 0.0, 1.0).sum() > target:
            lo = m
        else:
            hi = m
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(25):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-15:
                h += d * room / s
            else:
                h[int(np.argmax(room))] += d
        else:
            s = h.sum()
            if s > 1e-15:
                h += d * h / s
            else:
                h[int(np.argmax(h))] += d
    return np.clip(h, 0.0, 1.0)

def candidate_variants(h, n):
    h = project_feasible(h, n)
    r = h[::-1]
    out = [h, r, 1.0 - r, 0.5 * (h + r), np.clip(2.0 * h - 1.0, 0.0, 1.0)]
    return [project_feasible(x, n) for x in out]

def lse_objective_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode='full') * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    L = 2 * n - 1
    if not np.isfinite(s) or s <= 0.0:
        p = np.full(L, 1.0 / L)
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    pp = p
    A = np.convolve(pp[::-1], v, mode='full')[n - 1:2 * n - 1]
    B = np.convolve(pp, h, mode='full')[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

def lbfgs_objective(x, n, beta, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    F, g = lse_objective_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

def rand_composition(total, parts, rng):
    total = int(total)
    parts = int(parts)
    if parts <= 1:
        return np.array([total], dtype=int)
    if parts > total:
        parts = total
    if parts <= 1:
        return np.array([total], dtype=int)
    cuts = rng.choice(total - 1, size=parts - 1, replace=False)
    cuts.sort()
    return np.diff(np.concatenate(([0], cuts, [total]))).astype(int)

def make_blocky(n, r, rng):
    N = n // 2
    r = min(max(1, int(r)), N)
    rl = rand_composition(N, r, rng)
    zl = rand_composition(N, r, rng)
    rng.shuffle(rl)
    rng.shuffle(zl)
    blocks = []
    if rng.rand() < 0.5:
        for a, b in zip(rl, zl):
            if a:
                blocks.append(np.full(int(a), 1, dtype=np.int8))
            if b:
                blocks.append(np.full(int(b), 0, dtype=np.int8))
    else:
        for a, b in zip(zl, rl):
            if a:
                blocks.append(np.full(int(a), 0, dtype=np.int8))
            if b:
                blocks.append(np.full(int(b), 1, dtype=np.int8))
    h = np.concatenate(blocks)
    if h.size > n:
        h = h[:n]
    elif h.size < n:
        h = np.concatenate([h, np.zeros(n - h.size, dtype=np.int8)])
    cnt = int(h.sum())
    if cnt < N:
        idx = np.nonzero(h == 0)[0]
        if idx.size:
            h[rng.choice(idx, min(N - cnt, idx.size), replace=False)] = 1
    elif cnt > N:
        idx = np.nonzero(h == 1)[0]
        h[rng.choice(idx, min(cnt - N, idx.size), replace=False)] = 0
    if rng.rand() < 0.5:
        h = h[::-1].copy()
    return h

def mutate_bits(h, rng, count):
    h = h.astype(np.int8).copy()
    ones = np.nonzero(h)[0]
    zeros = np.nonzero(1 - h)[0]
    for _ in range(int(count)):
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(ones[rng.randint(ones.size)])
        j = int(zeros[rng.randint(zeros.size)])
        if i == j:
            continue
        h[i] = 0
        h[j] = 1
    return h

def init_binary_state(h, n):
    h = h.astype(np.int8)
    b = 1 - h
    D = np.zeros(2 * n - 1, dtype=np.int32)
    off = n - 1
    for d in range(n):
        L = n - d
        if L > 0:
            D[off + d] = int(np.dot(h[:L], b[d:]))
    for d in range(1, n):
        L = n - d
        D[off - d] = int(np.dot(h[d:], b[:L]))
    hist = np.bincount(D, minlength=n + 1).astype(np.int64)
    cm = int(np.nonzero(hist)[0].max())
    second = cm - 1
    while second >= 0 and hist[second] == 0:
        second -= 1
    return D, hist, cm, second

def collect_candidates(h, hist, cm, off, n, rng, max_cands):
    active = np.nonzero(hist == cm)[0]
    if active.size == 0:
        return []
    if active.size > 30:
        active = rng.choice(active, 30, replace=False)
    else:
        active = active.copy()
    rng.shuffle(active)
    one = (h == 1)
    zero = (h == 0)
    per = max(1, int(max_cands) // max(1, int(active.size)))
    cands = []
    for li in active:
        if len(cands) >= int(max_cands):
            break

        if d >= 0:
            L = n - d
            mask = one[:L] & zero[d:d + L]
            idx = np.nonzero(mask)[0]
            if idx.size:
                if idx.size > per:

                for p in idx:
                    cands.append((int(p), int(p + d)))

            idx = np.nonzero(mask)[0]
            if idx.size:
                if idx.size > per:
                    idx = rng.choice(idx, per, replace=False)
                for p in idx:
                    cands.append((int(p + dd), int(p)))
        if len(cands) > int(max_cands):

    return cands

def swap_effect(i, j, ones, zeros, D, hist, cm, second, n, off, L):
    if i == j:
        return None
    delta = np.zeros(L, dtype=np.int8)
    p1 = ones
    delta[off + (i - p1)] += 1
    delta[off + (j - p1)] -= 1
    p0 = zeros
    delta[off + (p0 - i)] -= 1
    delta[off + (p0 - j)] += 1
    delta[off] -= 2
    delta[off + (j - i)] += 1
    delta[off + (i - j)] += 1
    aff = np.nonzero(delta)[0]
    if aff.size == 0:
        return None
    old = D[aff]
    new = old + delta[aff].astype(np.int32)
    if int(new.max()) > cm:
        return None
    old_hist_cm = int(hist[cm])
    old_cnt = int(np.sum(old == cm))
    new_cnt = int(np.sum(new == cm))
    rem = old_hist_cm - old_cnt + new_cnt
    if rem >= old_hist_cm:
        return None
    affmax = int(new.max())
    true_cm = cm if rem > 0 else max(second, affmax)
    return ((true_cm, rem, affmax), aff, old, new, i, j)

def apply_swap(effect, h, D, hist, cm):
    aff, old, new, i, j = effect[1], effect[2], effect[3], effect[4], effect[5]
    np.add.at(hist, old, -1)
    np.add.at(hist, new, 1)
    D[aff] = new
    h[i] = 0
    h[j] = 1
    if hist[cm] == 0:
        while cm > 0 and hist[cm] == 0:
            cm -= 1
    second = cm - 1
    while second >= 0 and hist[second] == 0:
        second -= 1
    return cm, second

            best_c = cc
            best_h = cand.copy()

    bin_end = t0 + min(250.0, max(60.0, 0.25 * total))
    bin_deadline = min(bin_end, max(t0 + 30.0, end - 220.0))
    if bin_deadline > time.time() + 10.0:
        hb, cb = binary_search(512, bin_deadline, int(seed), target)
        if hb is not None and cb < best_c:
            best_c = float(cb)
            best_h = np.asarray(hb, dtype=float)
            best_n = 512
        if best_c > target and time.time() < bin_end - 15.0:
            hb, cb = binary_search(400, min(bin_end, max(time.time() + 5.0, end - 260.0)), int(seed) + 17, target)
            if hb is not None and cb < best_c:
                best_c = float(cb)
                best_h = np.asarray(hb, dtype=float)
                best_n = 400

    if time.time() < end - 60.0:
        betas = [50.0, 200.0, 800.0, 3000.0, 10000.0, 30000.0]
        hh, cc = polish_solution(best_h, best_n, betas, 5e3, end - 30.0, 50.0)
        if cc < best_c:
            best_c = float(cc)
            best_h = np.asarray(hh, dtype=float)

    if time.time() < end - 90.0:
        for nn in (480, 560, 400, 320):
            if time.time() > end - 45.0:
                break
            if nn == best_n or nn < 80 or nn > 800:
                continue
            x0 = np.linspace(0.0, 2.0, best_n)
            x1 = np.linspace(0.0, 2.0, nn)
            hr = np.interp(x1, x0, best_h)
            hr = project_feasible(hr, nn)
            cr = compute_c5(hr, nn)
            if cr < best_c:
                best_c = float(cr)
                best_h = hr.copy()
                best_n = int(nn)
            if time.time() > end - 35.0:
                break
            hh, cc = polish_solution(hr, nn, [200.0, 800.0, 3000.0, 10000.0], 3e3, end - 15.0, 30.0)
            if cc < best_c:
                best_c = float(cc)
                best_h = np.asarray(hh, dtype=float)
                best_n = int(nn)

    if time.time() < end - 40.0:
        for sc in (0.004, 0.012):
            if time.time() > end - 20.0:
                break
            nh = project_feasible(best_h + sc * rng.randn(best_n), best_n)
            hh, cc = polish_solution(nh, best_n, [5000.0, 20000.0], 5e3, end - 10.0, 20.0)
            if cc < best_c:
                best_c = float(cc)
                best_h = np.asarray(hh, dtype=float)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h.astype(float), float(best_c), int(best_n)
