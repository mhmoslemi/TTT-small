            b = np.zeros(2 * n - 1)
            idx = p - k_vals
            v = (idx >= 0) & (idx < n)
            b[v] += (1.0 - h[idx[v]])
            idx = p + k_vals
            v = (idx >= 0) & (idx < n)
            b[v] -= h[idx[v]]
            idx = q - k_vals
            v = (idx >= 0) & (idx < n)
            b[v] -= (1.0 - h[idx[v]])
            idx = q + k_vals
            v = (idx >= 0) & (idx < n)
            b[v] += h[idx[v]]

            if binary:
                best_d = 0.0
                best_val = current_max
                if h[p] < 1.0 - 1e-10:
                    val = float(np.max(c + b))
                    if val < best_val:
                        best_val = val
                        best_d = 1.0
                if h[q] > 1e-10:
                    val = float(np.max(c - b))
                    if val < best_val:
                        best_val = val
                        best_d = -1.0
                if best_d != 0.0:
                    h[p] += best_d
                    h[q] -= best_d
                    current_max = best_val
                    improved += 1
                elif T > 1e-10:
                    for d_try in [1.0, -1.0]:
                        if d_try > 0 and h[p] >= 1.0 - 1e-10:
                            continue
                        if d_try < 0 and h[q] <= 1e-10:
                            continue
                        val = float(np.max(c + d_try * b))
                        delta = (val - current_max) * n / 2.0
                        if delta < 0 or rng.rand() < np.exp(-delta / T):
                            h[p] += d_try
                            h[q] -= d_try
                            current_max = max(current_max, val)
                            break
            else:
                d_lo = max(-h[p], h[q] - 1.0)
                d_hi = min(1.0 - h[p], h[q])
                if d_lo >= d_hi - 1e-12:
                    continue
                for _ in range(30):
                    m1 = d_lo + (d_hi - d_lo) / 3.0
                    m2 = d_hi - (d_hi - d_lo) / 3.0
                    f1 = float(np.max(c + m1 * b))
                    f2 = float(np.max(c + m2 * b))
                    if f1 < f2:
                        d_hi = m2
                    else:
                        d_lo = m1
                d_opt = (d_lo + d_hi) / 2.0
                new_max = float(np.max(c + d_opt * b))
                if new_max < current_max - 1e-12:
                    h[p] += d_opt
                    h[q] -= d_opt
                    current_max = new_max
                    improved += 1

        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
            stuck_count = 0
        else:
            stuck_count += 1
        if stuck_count > 30 and T0 == 0.0:
            break

    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_phase(h, n, alpha, maxiter, deadline):
    from scipy.optimize import minimize
    if time.time() > deadline:
        return h, compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP on initial
    d1 = deadline - 80
    if time.time() < d1:
        h1, c1 = slsqp_phase(best_h, n, 5000, 500, d1)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Continuous swap descent (main workhorse)
    d2 = deadline - 60
    if time.time() < d2:
        h2, c2 = swap_descent(best_h, n, 300, 1500, d2, rng, binary=False)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Binary swap descent with SA
    h_bin = to_binary(best_h, n)
    d3 = deadline - 50
    if time.time() < d3:
        h3, c3 = swap_descent(h_bin, n, 400, 3000, d3, rng, binary=True, T0=0.5, decay=0.998)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Fine continuous polish on best
    d4 = deadline - 15
    if time.time() < d4:
        h4, c4 = swap_descent(best_h, n, 200, 2000, d4, rng, binary=False)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Try different n
    if time.time() < deadline - 30:
        x_orig = np.linspace(0, 2, len(best_h))
        for n_try in [400, 600, 800]:
            if time.time() > deadline - 20:
                break
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
            ht, ct = swap_descent(h_int, n_try, 150, 1500, deadline - 10, rng, binary=False)
            if ct < best_c:
                best_h, best_c = ht, ct
                x_orig = x_new

    # Phase 6: Final binary attempt on best
    if time.time() < deadline - 5:
        nb = len(best_h)
        h_b2 = to_binary(best_h, nb)
        hb2, cb2 = swap_descent(h_b2, nb, 100, 2000, deadline, rng, binary=True, T0=0.3, decay=0.997)
        if cb2 < best_c:
            best_h, best_c = hb2, cb2

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
