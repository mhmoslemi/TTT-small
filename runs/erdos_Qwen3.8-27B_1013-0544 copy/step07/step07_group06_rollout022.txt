Better: maintain separate lists/arrays for ones and zeros, and update them in-place using a swap-remove technique.

Actually, the simplest efficient approach: maintain two arrays `pos_ones` and `pos_zeros` that are lists of positions. When we swap a and b:
- Remove a from pos_ones (swap with last, pop)
- Add b to pos_ones (append)
- Remove b from pos_zeros (swap with last, pop)
- Add a to pos_zeros (append)

This is O(1) per operation.

```python
class BinaryState:
    def __init__(self, h, n):
        self.h = h.copy()
        self.n = n
        self.ones = np.where(h == 1)[0].tolist()
        self.zeros = np.where(h == 0)[0].tolist()
        self._one_idx = {v: i for i, v in enumerate(self.ones)}
        self._zero_idx = {v: i for i, v in enumerate(self.zeros)}
    
    def swap(self, a, b):
        """Swap h[a]=1 to 0 and h[b]=0 to 1."""
        self.h[a] = 0.0
        self.h[b] = 1.0
        # Remove a from ones
        ia = self._one_idx[a]
        last = self.ones[-1]
        self.ones[ia] = last
        self._one_idx[last] = ia
        self.ones.pop()
        del self._one_idx[a]
        # Add b to ones
        self._one_idx[b] = len(self.ones)
        self.ones.append(b)
        # Remove b from zeros
        ib = self._zero_idx[b]
        last = self.zeros[-1]
        self.zeros[ib] = last
        self._zero_idx[last] = ib
        self.zeros.pop()
        del self._zero_idx[b]
        # Add a to zeros
        self._zero_idx[a] = len(self.zeros)
        self.zeros.append(a)
```

Actually, here's a better idea: instead of plain 2-swaps, use **directed swaps**. At each step:
1. Find the lag k* that achieves the current max.
2. Look at the positions that contribute to the correlation at k*.
3. Propose a swap that specifically reduces the correlation at k*.

This is a form of "greedy" SA that's biased toward reducing the maximum.

For lag k*, the correlation is sum_i h[i]*(1-h[i+k*]). The contributing terms are those with h[i]=1 and h[i+k*]=0. If we can find a position i where h[i]=1, h[i+k*]=0, and a position j where h[j]=0, h[j+k*]=1 (or h[j+k*]=0), and swap h[i] and h[j], we might reduce the correlation at k*.

```python
import numpy as np
from scipy.optimize import minimize
import time

_stop_time = 0.0

def stop_callback(*args):
    return time.time() > _stop_time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        a = np.resize(a, n)
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo, hi = float(np.min(a)) - 1.0, float(np.max(a)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(a - mid, 0.0, 1.0).sum() > target: lo = mid
        else: hi = mid
    h = np.clip(a - 0.5*(lo+hi), 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13: break
        if diff > 0:
            room = 1.0 - h; s = room.sum()
            if s <= 1e-15: break
            h += diff * room / s
        else:
            s = h.sum()
            if s <= 1e-15: break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        idx = int(np.argmax(1.0-h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def top_n_binary(h, n):
    hb = np.zeros(n)
    idx = np.argsort(-h)[:n//2]
    hb[idx] = 1.0
    return hb

def lse_grad(h, n, beta, pen):
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n
    c = np.correlate(h, 1.0-h, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if s <= 0:
        p = np.full(2*n-1, 1.0/(2*n-1))
    else:
        p = e / s
    R = p[::-1]
    A = np.convolve(R, 1.0-h, mode="full")[n-1:2*n-1]
    B = np.convolve(p, h, mode="full")[n-1:2*n-1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)): g = np.zeros(n)
    F = m + np.log(s) / beta if s > 0 else m
    dev = h.sum() - 0.5*n
    if pen: F += 0.5*pen*dev*dev; g = g + pen*dev
    return F, g

def lbfgs_obj(x, n, beta, pen):
    F, g = lse_grad(x, n, beta, pen)
    if not np.isfinite(F): return 1e100, np.zeros(n)
    return F, g

def polish_lse(h, n, betas, pen, deadline, stage_time):
    work = project_feasible(h, n)
    best, bc = work.copy(), compute_c5(work, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        if time.time() > deadline - 6: break
        _stop_time = min(deadline - 3, time.time() + stage_time)
        try:
            res = minimize(lbfgs_obj, work, method="L-BFGS-B", jac=True,
                          bounds=bounds, args=(n, beta, pen), callback=stop_callback,
                          options={"maxiter": 500000, "maxls": 40, "ftol": 1e-14, "gtol": 1e-9})
            work = project_feasible(res.x, n)
        except Exception: pass
        cw = compute_c5(work, n)
        if cw < bc: bc, best = cw, work.copy()
        for v in [best[::-1], 1.0-best[::-1], 0.5*(best+best[::-1])]:
            vv = project_feasible(v, n)
            cv = compute_c5(vv, n)
            if cv < bc: bc, best = cv, vv.copy()
        if bc < cw - 1e-12: work = best.copy()
    return best, bc

def sa_binary(h_init, n, deadline, seed=42):
    h = top_n_binary(h_init, n).astype(np.float64).copy()
    M = float(np.correlate(h, 1.0-h, mode="full").max())
    bestM, besth = M, h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    rng = np.random.RandomState(seed)
    T, T_min, cool = 4.0, 0.03, 0.999995
    best_iter, patience = 0, 200000
    for it in range(10000000):
        if time.time() > deadline: break
        if it - best_iter > patience:
            T = max(T, 2.0); patience = int(patience * 1.3)
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a], h[b] = 0.0, 1.0
        newM = float(np.correlate(h, 1.0-h, mode="full").max())
        if newM <= M or rng.rand() < np.exp(-(newM - M) / T):
            M = newM
            i_a = np.where(ones == a)[0]
            ones[i_a[0]], ones[-1] = ones[-1], ones[i_a[0]]
            ones = ones[:-1]
            ones = np.append(ones, b)
            i_b = np.where(zeros == b)[0]
            zeros[i_b[0]], zeros[-1] = zeros[-1], zeros[i_b[0]]
            zeros = zeros[:-1]
            zeros = np.append(zeros, a)
            if M < bestM - 1e-12:

        else:
            h[a], h[b] = 1.0, 0.0
        T = max(T_min, T * cool)
    return besth, bestM * (2.0 / n)

    """SA biased toward reducing the current max lag."""
    h = top_n_binary(h_init, n).astype(np.float64).copy()
    corr = np.correlate(h, 1.0-h, mode="full").astype(np.float64)
    M = float(corr.max())
    bestM, besth = M, h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    rng = np.random.RandomState(seed)
    T, T_min, cool = 3.0, 0.05, 0.99999
    best_iter, patience = 0, 150000
    for it in range(10000000):

        if it - best_iter > patience:
            T = max(T, 2.0); patience = int(patience * 1.2)
        # Find max lag and bias toward it
        k_star = int(np.argmax(corr)) - (n - 1)
        # With probability 0.4, pick a from positions contributing to max lag
        if rng.rand() < 0.4 and abs(k_star) < n:

                if contrib: a = contrib[rng.randint(len(contrib))]
                else: a = ones[rng.randint(len(ones))]
            else:
                ks = -k_star
                contrib = [i for i in ones if i >= ks and h[i-ks] == 0]
                if contrib: a = contrib[rng.randint(len(contrib))]
                else: a = ones[rng.randint(len(ones))]
        else:

        b = zeros[rng.randint(len(zeros))]

        corr = np.correlate(h, 1.0-h, mode="full").astype(np.float64)

        if newM <= M or rng.rand() < np.exp(-(newM - M) / T):
            M = newM
            i_a = np.where(ones == a)[0]
            ones[i_a[0]], ones[-1] = ones[-1], ones[i_a[0]]; ones = ones[:-1]
            ones = np.append(ones, b)

            zeros[i_b[0]], zeros[-1] = zeros[-1], zeros[i_b[0]]; zeros = zeros[:-1]
            zeros = np.append(zeros, a)
            if M < bestM - 1e-12:

        else:
```
