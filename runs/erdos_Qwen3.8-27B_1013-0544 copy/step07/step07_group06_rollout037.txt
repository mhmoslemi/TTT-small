def lse_objective_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

import numpy as np
from scipy.optimize import minimize
import time

_stop_time = 0.0

def stop_callback(*args):
    return time.time() > _stop_time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(max(0, n - a.size))])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(a - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            h += diff * room / s
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        idx = int(np.argmax(h if diff < 0 else 1.0 - h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def top_n_binary(h, n):
    h = np.asarray(h, dtype=float).ravel()
    target = n // 2
    hb = np.zeros(n)
    if target > 0:
        idx = np.argsort(-h, kind='stable')[:target]
        hb[idx] = 1.0
    return hb

def thue_morse_seq(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def comb_pattern(n, period, width, offset=0):
    h = np.zeros(n)
    for i in range(n):
        if (i + offset) % period < width:
            h[i] = 1.0
    return top_n_binary(h, n)

def compute_base_matrix(h, n):
    l = np.arange(-(n - 1), n, dtype=int)
    base = np.zeros((n, 2 * n - 1), dtype=float)
    for p in range(n):
        f = p + l
        mf = (f >= 0) & (f < n)
        base[p, mf] = 1.0 - h[f[mf]]
        b = p - l
        mb = (b >= 0) & (b < n)
        base[p, mb] -= h[b[mb]]
    return base

def binary_greedy(h, n, deadline, max_sweeps=60, chunk=4096, allow_neutral=True, max_neutral_streak=8):
    if n % 2 != 0:
        hb = top_n_binary(h, n)
        return hb, compute_c5(hb, n)
    h = top_n_binary(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = int(round(float(corr.max())))
    besth = h.copy()
    bestM = M
    neutral_streak = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        if neutral_streak > max_neutral_streak:
            break
        base = compute_base_matrix(h, n)
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        zi = np.repeat(zeros, len(ones))
        oj = np.tile(ones, len(zeros))
        found_improve = False
        found_neutral = False
        for start in range(0, len(zi), chunk):
            if time.time() > deadline:
                return besth, bestM * (2.0 / float(n))
            end = min(start + chunk, len(zi))
            i = zi[start:end]
            j = oj[start:end]
            arr = base[i] - base[j] + corr[None, :]
            arr[:, n - 1] -= 2.0
            d1 = n - 1 + (j - i)
            d2 = n - 1 + (i - j)
            arr[:, d1] += 1.0
            arr[:, d2] += 1.0
            mx = arr.max(axis=1)
            idx = np.nonzero(mx <= M - 1 + 1e-12)[0]
            if len(idx) > 0:
                loc = int(idx[np.argmin(mx[idx])])
                h[i[loc]] = 1.0
                h[j[loc]] = 0.0
                corr = arr[loc].copy()
                M = int(round(float(corr.max())))
                if M < bestM:
                    bestM = M
                    besth = h.copy()
                found_improve = True
                neutral_streak = 0
                break
            if allow_neutral:
                idx = np.nonzero(np.abs(mx - M) < 1e-12)[0]
                if len(idx) > 0:
                    loc = int(idx[np.argmin(mx[idx])])
                    h[i[loc]] = 1.0
                    h[j[loc]] = 0.0
                    corr = arr[loc].copy()
                    M = int(round(float(corr.max())))
                    found_neutral = True
                    neutral_streak += 1
                    break
        if not found_improve and not found_neutral:
            break
    return besth, bestM * (2.0 / float(n))

def lse_objective_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

def lbfgs_objective(x, n, beta, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    F, g = lse_objective_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

def polish(h, n, betas, pen, deadline, stage_time):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        now = time.time()
        if now > deadline - 6.0:
            break
        stage_dl = min(deadline - 3.0, now + max(2.0, min(stage_time, 0.4 * (deadline - now))))
        _stop_time = stage_dl
        try:
            res = minimize(
                lbfgs_objective, work, method="L-BFGS-B", jac=True,
                bounds=bounds, args=(n, float(beta), float(pen)),
                callback=stop_callback,
                options={"maxiter": 500000, "maxls": 30, "ftol": 1e-14, "gtol": 1e-9}
            )
            x = res.x if (res is not None and hasattr(res, "x")) else work
        except Exception:
            x = work
        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc:
            bc = cw
            best = work.copy()
        work = best.copy()
    return best, bc

def two_swap_refine(h, n, deadline, max_iters=200):
    """Try 2-swap moves to escape local optima."""
    if n % 2 != 0:
        return h, compute_c5(h, n)
    h = top_n_binary(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = int(round(float(corr.max())))
    besth = h.copy()
    bestM = M
    base = compute_base_matrix(h, n)
    rng = np.random.RandomState(777)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if len(zeros) < 2 or len(ones) < 2:
            break
        # Pick two random zeros and two random ones
        iz = rng.choice(len(zeros), 2, replace=False)
        io = rng.choice(len(ones), 2, replace=False)
        i1, i2 = zeros[iz[0]], zeros[iz[1]]
        j1, j2 = ones[io[0]], ones[io[1]]
        if i1 == i2 or j1 == j2:
            continue
        # Compute new correlation: remove contributions of j1,j2 (now 0), add i1,i2 (now 1)
        new_corr = corr - base[j1] - base[j2] + base[i1] + base[i2]
        nm = int(round(float(new_corr.max())))
        if nm < M:
            h[i1] = 1.0
            h[i2] = 1.0
            h[j1] = 0.0
            h[j2] = 0.0
            corr = new_corr
            M = nm
            base = compute_base_matrix(h, n)
            if M < bestM:
                bestM = M
                besth = h.copy()
            # Continue from improved state
        elif nm == M:
            # Accept neutral 2-swaps with low probability to explore
            if rng.random() < 0.3:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
                corr = new_corr
                base = compute_base_matrix(h, n)
    return besth, bestM * (2.0 / float(n))
