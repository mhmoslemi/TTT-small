        for pi in range(n_pairs):
            p_idx = int(order[pi])
            q_idx = int(order[n - 1 - pi])
            if p_idx == q_idx:
                continue
            gp = g[p_idx]
            gq = g[q_idx]
            if gp >= gq:
                continue
            # delta > 0: add to p, subtract from q -> decrease c_{k*}
            lo = max(1e-12, -(h[p_idx]))
            hi = min(1.0 - h[p_idx], h[q_idx])
            if hi - lo < 1e-13:
                continue
            # Line search
            best_local = cur
            best_local_d = 0.0
            for t in np.linspace(0.1, 0.9, 9):
                d = lo + t * (hi - lo)
                hh = h.copy()
                hh[p_idx] += d
                hh[q_idx] -= d
                hh = np.clip(hh, 0.0, 1.0)
                c_new = erdos_c5(hh, n)
                if c_new < best_local:
                    best_local = c_new
                    best_local_d = d
            if best_local < best_c:
                best_c = best_local
                best_delta = best_local_d
                best_p, best_q = p_idx, q_idx
        if best_delta == 0.0:
            break
        h[best_p] += best_delta
        h[best_q] -= best_delta
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, target)
        cur = erdos_c5(h, n)
    return h, cur


def upsample(h, n, nh):
    x = np.linspace(0.0, 2.0, int(n))
    y = np.linspace(0.0, 2.0, int(nh))
    hh = np.interp(y, x, np.asarray(h, dtype=float))
    return project_box_sum(hh, 0.5 * nh)


def make_struct_init(n, mode, seed):
    rng = np.random.RandomState(int(seed))
    x = np.linspace(0.0, 2.0, n)
    if mode == 0:
        h = np.full(n, 0.5)
    elif mode == 1:
        h = 0.5 + 0.3 * np.sin(np.pi * x) + 0.15 * np.sin(2 * np.pi * x)
    elif mode == 2:
        h = 0.5 + 0.25 * np.cos(2 * np.pi * x) + 0.2 * np.sin(3 * np.pi * x)
    elif mode == 3:
        h = 0.5 + 0.2 * np.sin(np.pi * x / 2) + 0.15 * np.sin(4 * np.pi * x)
    else:
        h = 0.5 + 0.1 * rng.randn(n)
    return project_box_sum(h, 0.5 * n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(10.0, float(budget_s) - 5.0)
    deadline = t0 + total

    # Get initial solution
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    if n < 2:
        n = 400
        h = np.full(n, 0.5)
    if n > 500:
        nold = n
        n = 500
        h = np.interp(np.linspace(0.0, 2.0, n), np.linspace(0.0, 2.0, nold), h)
    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_n = n
    best_c = erdos_c5(best_h, best_n)

    # Phase 1: Optimize at initial resolution (n=400/500)
    end1 = t0 + 0.35 * total
    betas = [100.0, 500.0, 2000.0, 10000.0, 50000.0, 200000.0, 1000000.0]
    iters = [80, 70, 65, 60, 55, 50, 40]
    for beta, mx in zip(betas, iters):
        if time.time() > end1 - 5.0:
            break
        cand, c = refine_lbfgs(best_h, beta, mx, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    # Greedy pair descent at this resolution
    if time.time() < end1 - 10.0:
        best_h, best_c = greedy_pair_descent(best_h, best_n, end1, max_passes=15)

    # Phase 2: Try multi-start at same resolution
    end2 = t0 + 0.45 * total
    rng = np.random.RandomState(int(seed) + 7)
    for mode in range(5):
        if time.time() > end2 - 15.0:
            break
        hh = make_struct_init(best_n, mode, seed + mode * 13)
        # Add jitter from current best
        hh = 0.4 * hh + 0.6 * best_h + 0.03 * rng.randn(best_n)
        hh = project_box_sum(hh, 0.5 * best_n)
        cand, c = refine_lbfgs(hh, 50000.0, 50, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c
            cand, c = greedy_pair_descent(best_h, best_n, end2 - 5.0, max_passes=8)
            if c < best_c - 1e-15:
                best_h = cand.copy()
                best_c = c

    # Phase 3: Upsample to n=800
    end3 = t0 + 0.80 * total
    if time.time() < end3 - 30.0:
        n2 = 800
        h2 = upsample(best_h, best_n, n2)
        # Optimize at higher resolution
        betas2 = [100.0, 500.0, 2000.0, 10000.0, 50000.0, 200000.0, 800000.0, 3000000.0]
        iters2 = [60, 55, 50, 45, 40, 35, 30, 25]
        for beta, mx in zip(betas2, iters2):
            if time.time() > end3 - 20.0:
                break
            cand, c = refine_lbfgs(h2, beta, mx, deadline)
            if c < best_c - 1e-15:
                best_h = cand.copy()
                best_n = n2
                best_c = c
        # Greedy pair descent at n=800
        if time.time() < end3 - 15.0:
            best_h, best_c = greedy_pair_descent(best_h, best_n, end3, max_passes=20)

    # Phase 4: Final polish
    end4 = deadline - 4.0
    if time.time() < end4 - 30.0:
        # More aggressive greedy descent
        best_h, best_c = greedy_pair_descent(best_h, best_n, end4 - 25.0, max_passes=30)

    if time.time() < end4 - 20.0:
        # Final LBFGS with very high beta
        cand, c = refine_lbfgs(best_h, 5000000.0, 40, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    if time.time() < end4 - 10.0:
        # One more round of greedy
        best_h, best_c = greedy_pair_descent(best_h, best_n, end4 - 5.0, max_passes=15)

    # Final validation
    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
