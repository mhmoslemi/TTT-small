def smooth_grad(x,n,alpha):
    c = np.correlate(x, 1.0-x, 'full')
    z = alpha*c
    z -= z.max()
    e = np.exp(z)
    w = e / e.sum()
    center=n-1
    grad = np.zeros(n)
    for m in range(len(c)):
        wm = w[m]
        if wm < 1e-13: continue
        s = center - m
        if s == 0:
            grad += wm * (1.0 - 2.0*x)
        elif s > 0:
            # first p=0..n-1-s
            grad[:n-s] += wm * (1.0 - x[s:])
            # second p=s..n-1
            grad[s:] -= wm * x[:n-s]
        else:
            r = -s
            # first p=r..n-1, j=p+s
            grad[r:] += wm * (1.0 - x[:n-r])  # x[s:n] length n-r
            # second p=0..n-1-r, i=p-s=p+r
            grad[:n-r] -= wm * x[r:]
    return (2.0/n)*grad

import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if diff > 0.0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0.0:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 0.0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-12:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-12:
        idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def triangle_h(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(1.0 - np.abs(x - 1.0), n)

def basis_h(n, rng, modes=8):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 * np.ones(n)
    scale = rng.uniform(0.18, 0.42)
    t = x - 1.0
    for k in range(1, modes + 1):
        amp = rng.uniform(-scale / k, scale / k)
        h += amp * np.cos(k * np.pi * t)
    return project_feasible(h, n)

def binary_from_h(h):
    n = len(h)
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    h2 = np.zeros(n)
    h2[idx] = 1.0
    return h2

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def sum_con(x, n):
    return float(x.sum()) - 0.5 * n

def sum_jac(x, n):
    return np.ones(n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    z = alpha * c
    zm = float(z.max())
    return ((zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    center = n - 1
    z = alpha * c
    z -= z.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm <= 1e-14:
            continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * x)
        elif s > 0:
            g[:n - s] += wm * (1.0 - x[s:])
            g[s:] -= wm * x[:n - s]
        else:
            r = -s
            g[r:] += wm * (1.0 - x[:n - r])
            g[:n - r] -= wm * x[r:]
    return (2.0 / n) * g

def slsqp(h, n, alphas, maxiter, deadline):
    best_h = project_feasible(h, n)
    best_c = compute_c(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_jac, bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
            x = project_feasible(res.x, n)
            c = compute_c(x, n)
            if c < best_c:
                best_h, best_c = x, c
        except Exception:
            pass
    return best_h, best_c

    else:
        vmax = 0.0
    val = max(unaff, qconst + x * (1.0 - x))
    return max(val, vmax)

        r = min(1.0, x + 0.05)
        if r <= l:
            l, r = 0.0, 1.0
        xg2 = np.linspace(l, r, 32)
        vals2 = coord_vector_eval(xg2, hp, cseg, a, qconst, unaff)
        idx2 = int(np.argmin(vals2))
        x2 = float(xg2[idx2])
        f2 = float(vals2[idx2])
        if f2 < best_f:
            best_f = f2
            best_x = x2
    return best_x, best_f

def cd_unconstrained(h, n, max_passes, deadline, project_every=10):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h = h.copy()
    best_c = (2.0 / n) * cur
    center = n - 1
    L = 2 * n - 1
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            hp = float(h[p])
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            seg = hi - lo + 1
            s = center - np.arange(lo, hi + 1, dtype=int)
            j = p + s
            i = p - s
            a = np.zeros(seg, dtype=float)
            vj = (j >= 0) & (j < n)
            vi = (i >= 0) & (i < n)
            if np.any(vj):
                a[vj] += 1.0 - h[j[vj]]
            if np.any(vi):
                a[vi] -= h[i[vi]]
            off = center - lo
            cseg = np.empty(seg - 1, dtype=float)
            asg = np.empty(seg - 1, dtype=float)
            if off:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center + 1:hi + 1]
                asg[off:] = a[off + 1:]
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(c[hi + 1:].max()))
            bx, bf = optimize_coordinate(hp, cseg, asg, qconst, unaff, 18)
            if bf < cur - 1e-10:
                dxv = bx - hp
                if off:
                    c[lo:center] += asg[:off] * dxv
                if hi > center:
                    c[center + 1:hi + 1] += asg[off:] * dxv
                c[center] = qconst + bx * (1.0 - bx)
                h[p] = bx
                cur = bf
                improved = True
            if (p + 1) % project_every == 0:
                h = project_feasible(h, n)
                c = np.correlate(h, 1.0 - h, mode='full')
                cur = float(c.max())
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        cval = (2.0 / n) * cur
        if cval < best_c - 1e-13:
            best_c = cval
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def optimize_start(h, n, deadline, use_slsqp=True, cd_passes=4, project_every=10,
                   slsqp_alphas=(50, 500, 5000), slsqp_max=100):
    h = project_feasible(h, n)
    c = compute_c(h, n)
    if use_slsqp and n <= 700 and time.time() < deadline - 90:
        h2, c2 = slsqp(h, n, slsqp_alphas, slsqp_max, deadline)
        if c2 < c:
            h, c = h2, c2
    if time.time() < deadline - 20:
        h2, c2 = cd_unconstrained(h, n, cd_passes, deadline, project_every)
        if c2 < c:
            h, c = h2, c2
    if use_slsqp and n <= 700 and time.time() < deadline - 70:
        tail = list(slsqp_alphas)[-2:]
        h3, c3 = slsqp(h, n, tail, max(30, slsqp_max // 2), deadline)
        if c3 < c:
            h, c = h3, c3
    if time.time() < deadline - 15:
        h4, c4 = cd_unconstrained(h, n, 2, deadline, project_every)
        if c4 < c:
            h, c = h4, c4
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 10.0
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = triangle_h(500)
    n_init = len(h_init)
    if n_init % 2:
        n_init += 1
        h_init = resample_h(h_init, n_init)
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c(best_h, n_init)

    best_h, best_c = optimize_start(
        best_h, n_init, deadline,
        use_slsqp=True, cd_passes=8, project_every=8,
        slsqp_alphas=(20, 200, 2000, 20000), slsqp_max=80
    )

    for nt in [n_init, 600, 800, 960, 400, 300, 256]:
        if time.time() > deadline - 220:
            break
        if best_c <= 0.38075:
            break
        h0 = resample_h(best_h, nt)
        pe = 6 if nt <= 500 else (12 if nt <= 800 else 20)
        passes = 5 if nt <= 500 else (4 if nt <= 800 else 3)
        use_s = nt <= 600 and time.time() < deadline - 300
        h1, c1 = optimize_start(
            h0, nt, deadline,
            use_slsqp=use_s, cd_passes=passes, project_every=pe,
            slsqp_alphas=(50, 500, 5000), slsqp_max=70
        )
        if c1 < best_c:
            best_h, best_c = h1, c1

    for r in range(28):
        if time.time() > deadline - 95:
            break
        if best_c <= 0.38070:
            break
        nt = len(best_h) if r % 3 else 500
        base = resample_h(best_h, nt)
        typ = r % 4
        if typ == 0:
            h0 = project_feasible(base + rng.normal(0.0, 0.02, nt), nt)
        elif typ == 1:
            h0 = binary_from_h(base)
        elif typ == 2:
            h0 = np.zeros(nt)
            idx = rng.choice(nt, nt // 2, replace=False)
            h0[idx] = 1.0
        else:
            h0 = basis_h(nt, rng)
        h1, c1 = optimize_start(
            h0, nt, deadline,
            use_slsqp=(nt <= 450 and time.time() < deadline - 180),
            cd_passes=2 if nt > 700 else 3,
            project_every=18 if nt > 800 else 10,
            slsqp_alphas=(100, 1000),
            slsqp_max=50
        )
        if c1 < best_c:
            best_h, best_c = h1, c1

    if time.time() < deadline - 30:
        best_h, best_c = optimize_start(
            best_h, len(best_h), deadline,
            use_slsqp=True, cd_passes=8, project_every=5,
            slsqp_alphas=(100, 1000, 10000), slsqp_max=100
        )

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c(best_h, len(best_h))
    return best_h, best_c, len(best_h)
