import numpy as np
from scipy.optimize import minimize
import time

_CONST_TARGET = 0.0
_SLSQP_N = 1
_SLSQP_ALPHA = 1.0
_SLSQP_DX = 0.2


def slsqp_obj(x):
    x = np.asarray(x, dtype=float)
    g = np.correlate(x, 1.0 - x, mode="full") * _SLSQP_DX
    m = float(np.max(g))
    sh = _SLSQP_ALPHA * (g - m)
    sm = float(np.max(sh))
    return float(m + (sm + np.log(np.exp(sh - sm).sum())) / _SLSQP_ALPHA)


def slsqp_cons(x):
    return np.asarray(x, dtype=float).sum() - _CONST_TARGET


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size != n:
        h = np.resize(h, n)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    diff = target - float(h.sum())
    if diff > 0.0:
        room = 1.0 - h
        s = float(room.sum())
        if s > 0.0:
            h = h + (diff / s) * room
    elif diff < 0.0:
        s = float(h.sum())
        if s > 0.0:
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)


def max_c5(h):
    n = int(h.size)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def eval_g(h):
    n = int(h.size)
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode="full") * dx
    return g, float(np.max(g)), dx


def candidate_from(h, n):
    h = project_feasible(h, n)
    return h, max_c5(h)


def consider_candidate(h, c, n, best_state):
    if c < best_state[1] - 1e-15:

        best_state[1] = float(c)

def resample_to(h, n):
    if h.size == n:
        return h.copy()
    xb = np.linspace(0.0, 2.0, h.size)
    xn = np.linspace(0.0, 2.0, n)
    return np.interp(xn, xb, h)


def top_gradient(h, g, tau, kmax, floor_c):
    n = int(h.size)
    dx = 2.0 / n
    gmax = float(np.max(g))
    thresh = gmax - max(8.0 * tau, 1e-9)
    if floor_c is not None and floor_c < gmax:
        thresh = max(thresh, float(floor_c))
    L = int(g.size)
    if kmax >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(g, -kmax)[-kmax:]
    active = idx[g[idx] >= thresh]
    if active.size == 0:
        active = idx[:1]
    p = (g[active] - gmax) / tau
    p -= float(np.max(p))
    p = np.exp(p)
    s = float(p.sum())
    if s < 1e-300:
        p = np.full(active.size, 1.0 / max(1, active.size))
    else:
        p = p / s
    grad = np.zeros(n, dtype=float)
    for ii, w in zip(active, p):
        if w < 1e-15:
            continue
        lag = int(ii) - (n - 1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                grad[:m] += w * (1.0 - h[lag:])
                if lag >= 0:
                    grad[lag:] -= w * h[:m]
        else:
            d = -lag
            m = n - d
            if m > 0:
                grad[d:] += w * (1.0 - h[:m])
                grad[:m] -= w * h[d:]
    grad *= dx
    return grad

            b[q] = 0.0
        step += 1
    if n <= 128 and time.time() < deadline - 0.3:
        hb, cb = binary_hill_small(best_b, n, deadline, rng)
        if cb < best_c - 1e-12:
            best_b = hb
            best_c = cb
    return best_b, float(best_c)


def slsqp_smooth(h, n, alpha, maxiter, ftol):
    global _SLSQP_N, _SLSQP_ALPHA, _SLSQP_DX, _CONST_TARGET
    h = project_feasible(h, n)
    _SLSQP_N = int(n)
    _SLSQP_ALPHA = float(alpha)
    _SLSQP_DX = 2.0 / float(n)
    _CONST_TARGET = 0.5 * float(n)
    bounds = [(0.0, 1.0)] * int(n)
    cons = [{"type": "eq", "fun": slsqp_cons}]
    try:
        res = minimize(slsqp_obj, h, method="SLSQP", bounds=bounds, constraints=cons,
                       options={"maxiter": int(maxiter), "ftol": float(ftol)})
        x = project_feasible(res.x, int(n))
    except Exception:
        x = h
    return x


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s)
    deadline = t0 + total - 5.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(h0.size)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    h0 = project_feasible(h0, n0)
    c0 = max_c5(h0)
    best = [h0.copy(), c0, n0]

    if deadline <= t0 + 15.0:
        return best[0], best[1], best[2]

    if total > 550:
        low_ns = [32, 64, 128, 192]
        low_frac = 0.36
    elif total > 250:
        low_ns = [64, 128]
        low_frac = 0.30
    else:
        low_ns = [64]
        low_frac = 0.22

    low_deadline = t0 + max(20.0, low_frac * total)
    low_span = max(1.0, low_deadline - t0)
    for i, n in enumerate(low_ns):
        seg_end = min(deadline, t0 + low_span * (i + 1) / float(len(low_ns)))
        if time.time() >= seg_end - 0.3 or n % 2 != 0:
            continue
        hs = project_feasible(resample_to(best[0], n), n)
        hs_c = max_c5(hs)
        bin_dead = min(seg_end, time.time() + max(2.0, 0.55 * (seg_end - time.time())))
        bb = binary_search(hs, n, bin_dead, rng)
        bb_c = binary_maxC(bb)
        consider_candidate(project_feasible(bb, n), bb_c, n, best)
        if bb_c < hs_c:
            start_h = bb
        else:
            start_h = hs
        start_h = project_feasible(start_h, n)
        if time.time() < seg_end - 0.5:
            hh, hc = continuous_refine(start_h, n, seg_end, rng,
                                       kmax=min(24, max(8, n // 8)), lr0=0.045,
                                       floor_c=min(0.3808, best[1]))
            consider_candidate(hh, hc, n, best)
        if n <= 96 and time.time() < seg_end - 8.0:
            hr = project_feasible(0.5 + 0.28 * rng.randn(n), n)
            hh, hc = continuous_refine(hr, n, seg_end, rng,
                                       kmax=min(16, max(6, n // 8)), lr0=0.05,
                                       floor_c=min(0.3808, best[1]))
            consider_candidate(hh, hc, n, best)

    high1_dead = t0 + 0.58 * total
    high2_dead = t0 + 0.82 * total

    if time.time() < high1_dead - 0.5:
        nref = int(best[2])
        if nref < 16 or nref > 512:
            nref = 200
        hh, hc = continuous_refine(resample_to(best[0], nref), nref, high1_dead, rng,
                                   kmax=min(48, max(10, nref // 8)), lr0=0.032,
                                   floor_c=min(0.3807, best[1]))
        consider_candidate(hh, hc, nref, best)

    if time.time() < high2_dead - 0.5:
        n2 = int(n0)
        if int(best[2]) == n2:
            h2 = project_feasible(best[0] + 0.012 * rng.randn(n2), n2)
        else:
            h2 = project_feasible(resample_to(best[0], n2), n2)
        hh, hc = continuous_refine(h2, n2, high2_dead, rng,
                                   kmax=min(48, max(10, n2 // 8)), lr0=0.035,
                                   floor_c=min(0.3806, best[1]))
        consider_candidate(hh, hc, n2, best)

    bin_dead = t0 + 0.93 * total
    if time.time() < bin_dead - 0.5 and int(best[2]) % 2 == 0:
        bb, cb = binary_search(best[0], int(best[2]), bin_dead, rng)
        consider_candidate(project_feasible(bb, int(best[2])), cb, int(best[2]), best)

    nbest = int(best[2])
    if time.time() < deadline - 55.0:
        if nbest <= 160:
            alphas = [150, 1000, 5000]
            maxits = 1200
        else:
            alphas = [3000, 30000]
            maxits = 350
        for a in alphas:
            if time.time() > deadline - 50.0:
                break
            hs = slsqp_smooth(best[0], nbest, a, maxits, 1e-11)
            consider_candidate(project_feasible(hs, nbest), max_c5(project_feasible(hs, nbest)), nbest, best)

    if time.time() < deadline - 2.0:
        hh, hc = continuous_refine(best[0], int(best[2]), deadline, rng,
                                   kmax=min(32, max(8, int(best[2]) // 10)),
                                   lr0=0.022, floor_c=min(0.3805, best[1]))
        consider_candidate(hh, hc, int(best[2]), best)

    best[0] = project_feasible(best[0], int(best[2]))
    best[1] = max_c5(best[0])
    return best[0], float(best[1]), int(best[2])
