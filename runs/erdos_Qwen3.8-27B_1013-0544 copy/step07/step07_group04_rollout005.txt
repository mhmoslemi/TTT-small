    # distribute over values with largest slack iteratively

    idx_order = np.argsort(-(h*(1-h))) # O(n logn) only final maybe okay

import numpy as np
import time
from scipy.optimize import minimize


def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def project_box_sum(v, n=None):
    v = np.nan_to_num(np.asarray(v, dtype=float)).reshape(-1)
    if n is None:
        n = len(v)
    v = v[:n]
    if n == 0:
        return v
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(20):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        cnt = int(m.sum())
        if cnt:
            v[m] += d / cnt
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    d = tgt - v.sum()
    if abs(d) > 1e-15:
        idx = int(np.argmax(v * (1.0 - v)))
        if d > 0:
            v[idx] = min(1.0, v[idx] + d)
        else:
            v[idx] = max(0.0, v[idx] + d)
    return v


def enforce_sum(h, n):
    tgt = 0.5 * n
    d = tgt - float(h.sum())
    if abs(d) < 1e-15 or n <= 0:
        return h
    order = np.argsort(-(h * (1.0 - h)))
    for idx in order:
        if abs(d) < 1e-15:
            break
        if d > 0:
            cap = 1.0 - h[idx]
            if cap <= 0.0:
                continue
            step = min(d, cap)
            h[idx] += step
            d -= step
        else:
            cap = h[idx]
            if cap <= 0.0:
                continue
            step = min(-d, cap)
            h[idx] -= step
            d += step
    d = tgt - float(h.sum())
    if abs(d) > 1e-16:
        idx = int(np.argmax(h * (1.0 - h)))
        if d > 0.0 and (1.0 - h[idx] >= d):
            h[idx] += d
        elif d < 0.0 and (h[idx] >= -d):
            h[idx] += d
    return h


def interpolate_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)


def default_h(n, seed):
    rng = np.random.RandomState(seed)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    v = 0.5 + 0.24 * np.sin(np.pi * x) + 0.10 * np.sin(3.0 * np.pi * x + 0.7)
    v = v + 0.02 * rng.randn(n)
    return project_box_sum(v, n)


def lse_value(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    sh = np.clip(sh, -700.0, 0.0)
    s = float(np.exp(sh).sum())
    if s <= 0.0:
        return m
    return m + np.log(s) / alpha


def lse_obj(x, n, alpha):
    return lse_value(compute_c(x, n), alpha)


def lse_weights(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    sh = np.clip(sh, -700.0, 0.0)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0.0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
        return q
    q = e / s
    q[q < 1e-12] = 0.0
    r = float(q.sum())
    if r <= 0.0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
    else:
        q /= r
    return q


def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    q = lse_weights(c, alpha)
    g = np.zeros(n)
    omh = 1.0 - x
    dx = 2.0 / n
    off = n - 1
    nz = np.nonzero(q > 1e-12)[0]
    for idx in nz:
        k = int(idx) - off
        w = float(q[idx])
        if k == 0:
            g += w * (omh - x)
        elif k > 0:
            m = n - k
            g[k:] += w * omh[:m]
            g[:m] -= x[k:]
        else:
            d = -k
            m = n - d
            g[:m] += omh[d:]
            g[d:] -= x[:m]
    return dx * g


def pen_obj(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_obj(x, n, alpha) + 0.5 * mu * d * d


def pen_grad(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_grad(x, n, alpha) + mu * d


def lbfgsb_phase(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(pen_obj, h, args=(n, alpha, mu), method="L-BFGS-B",
                       jac=pen_grad, bounds=bounds,
                       options={"maxiter": int(maxiter),
                                "maxfun": int(4 * maxiter + 50),
                                "ftol": 1e-15, "gtol": 1e-8})
        if getattr(res, "x", None) is not None:
            return res.x
    except Exception:
        pass
    return h


def add_var_delta(delta, p, coeff, h, omh):
    n = len(h)
    off = n - 1
    if coeff == 0.0:
        return
    delta[off:off + p + 1] += coeff * omh[p::-1]
    m = n - 1 - p
    if m > 0:
        delta[off - m:off] += coeff * omh[n - 1:p - 1:-1]
    m = n - p
    if m > 0:
        delta[off:off + m] -= coeff * h[p:n]
    if p > 0:
        delta[off - p:off] -= coeff * h[0:p]


def pair_eval_values(c, D, l0, off, pos, neg, ts, qscale):
    a = ts - l0
    L = len(D)
    m = len(ts)
    vals = np.empty(m, dtype=float)
    chunk = max(1, 500000 // max(1, L))
    for st in range(0, m, chunk):
        en = min(m, st + chunk)
        aa = a[st:en]

        mat[:, pos] = c[pos] + aa * D[pos] + qscale * aa * aa
        mat[:, neg] = c[neg] + aa * D[neg] + qscale * aa * aa

    return vals

def weighted_index(w, rng):
    size = int(w.size)
    if size <= 0:
        return 0
    arr = np.asarray(w, dtype=float)
    s = float(arr.sum())
    if s <= 1e-300:
        return int(rng.randint(size))
    arr[arr < 0.0] = 0.0
    arr /= float(arr.sum())
    raw = np.cumsum(arr)
    total = float(raw[-1])
    if total <= 0.0:
        return int(rng.randint(size))
    val = float(rng.random() * total)
    idx = int(np.searchsorted(raw, val, side="right"))
    if idx >= size:
        idx = size - 1
    return idx

            if k == 0:
                wself = np.maximum(0.0, h * (1.0 - h))
                if float(wself.sum()) > 0.0:
                    i = weighted_index(wself, rng)
                else:
                    i = int(rng.randint(n))
                if rng.random() < 0.5:
                    j = int(rng.randint(n))
                else:
                    w2 = wself.copy()
                    w2[i] = 0.0
                    if float(w2.sum()) > 0.0:
                        j = weighted_index(w2, rng)
                    else:
                        j = (i + 1) % n
            elif k > 0:
                if 1 <= k < n:
                    contrib = h[k:] * (1.0 - h[:n - k])
                    if float(contrib.sum()) > 0.0:
                        ii = weighted_index(contrib, rng)
                        i = ii
                        j = ii + k
                    else:
                        d = int(rng.randint(1, n))
                        a = int(rng.randint(0, n - d))
                        i = a
                        j = a + d
                else:
                    d = int(rng.randint(1, n))
                    a = int(rng.randint(0, n - d))
                    i = a
                    j = a + d
            else:
                d = -k
                if 1 <= d < n:
                    contrib = h[:n - d] * (1.0 - h[d:])
                    if float(contrib.sum()) > 0.0:
                        ii = weighted_index(contrib, rng)
                        i = ii
                        j = ii + d
                    else:
                        d2 = int(rng.randint(1, n))
                        a = int(rng.randint(0, n - d2))
                        i = a
                        j = a + d2
                else:
                    d2 = int(rng.randint(1, n))
                    a = int(rng.randint(0, n - d2))
                    i = a
                    j = a + d2

        if i == j:
            i = (i + 1) % n

        nv = pair_try(h, c, i, j, n, curc)
        if nv is None:
            noimp += 1
        else:
            curc = nv
            accepts += 1
            if curc < bestc - 1e-15:
                bestc = curc
                besth = h.copy()
                noimp = 0
            else:
                noimp += 1

    besth = project_box_sum(besth, n)
    besth = enforce_sum(besth, n)
    c = compute_c(besth, n)
    bestc = float(c.max())
    return besth, bestc


def make_sizes(n0):
    out = [int(n0)]
    if n0 >= 220:
        small = max(128, int(round(0.75 * n0 / 2.0)) * 2)
        if 128 <= small < n0:
            out.append(small)
    for f in (1.25, 1.60, 2.00, 2.40):
        m = int(round(n0 * f / 2.0)) * 2
        if m > n0 and m <= 1024:
            out.append(m)
    seen = []
    for m in out:
        if m < 8 or m > 1100:
            continue
        if not seen or abs(m - seen[-1]) > max(16, int(0.06 * seen[-1])):
            seen.append(m)
    if len(seen) > 5:
        keep = [seen[0]]
        others = seen[1:]
        for idx in (0, len(others) // 3, 2 * len(others) // 3, max(0, len(others) - 1)):
            m = others[idx]
            if m not in keep:
                keep.append(m)
        seen = keep
    return seen[:5]


def optimize_n(h, n, seed, deadline):
    h = project_box_sum(h, n)
    h = enforce_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    workh = h.copy()
    workc = bestc

    t_start = time.time()
    if n <= 450:
        phases = ((10.0, 50.0, 110), (40.0, 30.0, 85), (150.0, 15.0, 65), (600.0, 8.0, 45))
        lse_dead = min(deadline - 25.0, t_start + 80.0)
    elif n <= 750:
        phases = ((10.0, 30.0, 70), (40.0, 20.0, 55), (150.0, 10.0, 40))
        lse_dead = min(deadline - 25.0, t_start + 55.0)
    else:
        phases = ((10.0, 20.0, 45), (40.0, 10.0, 35), (150.0, 8.0, 25))
        lse_dead = min(deadline - 25.0, t_start + 35.0)

    for alpha, mu, maxit in phases:
        if time.time() > lse_dead:
            break
        h2 = lbfgsb_phase(workh, n, alpha, mu, maxit)
        h2 = project_box_sum(h2, n)
        h2 = enforce_sum(h2, n)
        c2 = compute_c(h2, n)
        v2 = float(c2.max())
        if v2 < bestc:
            bestc = v2
            besth = h2.copy()
        if v2 <= workc + 1e-7:
            workh = h2.copy()
            workc = v2
        else:
            workh = besth.copy()
            workc = bestc

    if time.time() < deadline - 6.0:
        cstart = compute_c(besth, n)
        hp, cp = pair_descent(besth, cstart, deadline - 2.0, seed + 17)
        if cp < bestc:
            bestc = cp
            besth = hp

    if time.time() < deadline - 25.0:
        rng2 = np.random.RandomState(seed + 913)
        h2 = besth.copy() + 0.005 * rng2.randn(n)
        h2 = project_box_sum(h2, n)
        h2 = enforce_sum(h2, n)
        c2 = compute_c(h2, n)
        if float(c2.max()) <= bestc + 0.004:
            hp2, cp2 = pair_descent(h2, c2, deadline - 2.0, seed + 291)
            if cp2 < bestc:
                bestc = cp2
                besth = hp2

    besth = project_box_sum(besth, n)
    besth = enforce_sum(besth, n)
    c = compute_c(besth, n)
    bestc = float(c.max())
    return besth, bestc


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + budget_s - 8.0
    rng = np.random.RandomState(seed)

    try:
        hp = np.array(initial_h_values, dtype=float).copy().reshape(-1)
        n0 = int(len(hp))
    except Exception:
        n0 = 400
        hp = default_h(n0, seed)

    hp = project_box_sum(hp, n0)
    hp = enforce_sum(hp, n0)
    c0 = compute_c(hp, n0)
    bestc = float(c0.max())
    besth = hp.copy()
    bestn = n0

    sizes = make_sizes(n0)
    if len(sizes) == 0:
        sizes = [n0]
    fr = [1.0 / float(len(sizes))] * len(sizes)
    opt_end = deadline - 45.0
    cum = 0.0

    for k, nh in enumerate(sizes):
        cum += fr[k]
        phase_end = t0 + (opt_end - t0) * min(1.0, cum)
        dl = max(time.time() + 8.0, min(phase_end, opt_end))
        if dl <= time.time():
            continue

        if k == 0:
            hinit = besth.copy()
        else:
            hinit = interpolate_h(besth, bestn, nh)
            hinit = hinit + 0.0035 * rng.randn(nh)
            hinit = 0.5 * (hinit + 1.0 - hinit[::-1])
            hinit = hinit + 0.002 * rng.randn(nh)
            hinit = project_box_sum(hinit, nh)
            hinit = enforce_sum(hinit, nh)

        hopt, copt = optimize_n(hinit, nh, seed + 101 + 37 * k, dl)
        if copt < bestc:
            bestc = copt
            besth = hopt.copy()
            bestn = nh

    if time.time() < deadline - 85.0:
        cand_n = min(1024, max(bestn, 800, 2 * n0))
        cand_n = int(round(cand_n / 2.0)) * 2
        if cand_n != bestn and 128 <= cand_n <= 1024:
            htest = interpolate_h(besth, bestn, cand_n)
            htest = htest + 0.003 * rng.randn(cand_n)
            htest = project_box_sum(htest, cand_n)
            htest = enforce_sum(htest, cand_n)
            ctest = compute_c(htest, cand_n)
            if float(ctest.max()) < bestc + 0.006:
                htry, ctypes = optimize_n(htest, cand_n, seed + 777, deadline - 30.0)
                if ctypes < bestc:
                    bestc = ctypes
                    besth = htry.copy()
                    bestn = cand_n

    if time.time() < deadline - 2.0:
        cfinal = compute_c(besth, bestn)
        hfinal, cfinal = pair_descent(besth, cfinal, deadline - 0.5, seed + 1234)
        if cfinal < bestc:
            bestc = cfinal
            besth = hfinal.copy()

    besth = project_box_sum(besth, bestn)
    besth = enforce_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, bestn
