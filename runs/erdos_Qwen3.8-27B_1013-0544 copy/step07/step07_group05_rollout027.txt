    dx = 2.0 / n
    target = 0.5 * n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    active = c > t
    viol = np.maximum(0.0, c - t)
    f = float(np.sum(viol * viol))
    res = float(h.sum() - target)
    f += rho * res * res
    if np.any(active):
        w = np.zeros_like(c)
        w[active] = 2.0 * viol[active]
        g = grad_weighted(h, w, n, dx)
    else:
        g = np.zeros(n)
    g += 2.0 * rho * res
    return f, g

def opt_lbfgs(h, n, obj_fn, mx, deadline):
    target = 0.5 * n
    rem = deadline - time.time()
    if rem < 5.0:
        return h, erdos_c5(h, n)
    mx = int(min(mx, max(10, int(rem / 2.0))))
    bounds = [(0.0, 1.0)] * n
    try:
        res = so.minimize(obj_fn, h, jac=True, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-18,
                                   'gtol': 1e-12, 'maxls': 50, 'disp': False})
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)

def opt_hinge_joint(h, n, t_init, rho, mx, deadline):
    dx = 2.0 / n
    target = 0.5 * n
    rem = deadline - time.time()
    if rem < 10.0:
        return h, t_init, erdos_c5(h, n)
    v0 = np.append(h.copy(), t_init)
    mx = int(min(mx, max(15, int(rem / 3.0))))

    def joint_obj(v):
        hh = v[:n]
        tt = float(v[n])
        dx = 2.0 / n
        c = np.correlate(hh, 1.0 - hh, mode='full') * dx
        active = c > tt
        viol = np.maximum(0.0, c - tt)
        f = float(np.sum(viol * viol))
        res = float(hh.sum() - target)
        f += rho * res * res
        if np.any(active):
            w = np.zeros(2 * n - 1)
            w[active] = 2.0 * viol[active]
            gh = grad_weighted(hh, w, n, dx)
        else:
            gh = np.zeros(n)
        gh += 2.0 * rho * res
        gt = -float(np.sum(2.0 * viol))
        return f, np.append(gh, gt)

    bounds = [(0.0, 1.0)] * n + [(0.01, 1.0)]
    try:
        res = so.minimize(joint_obj, v0, jac=True, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-18,
                                   'gtol': 1e-12, 'maxls': 50, 'disp': False})
        cand = project_box_sum(res.x[:n], target)
        tt = float(res.x[n])
    except Exception:
        cand, tt = h, t_init
    return cand, tt, erdos_c5(cand, n)

def coord_descent(h, n, deadline, max_rounds=5):
    dx = 2.0 / n
    cur = erdos_c5(h, n)
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-9 or h[j] > 1.0 - 1e-9:
                continue
            c = np.correlate(h, 1.0 - h, mode='full') * dx
            g = grad_weighted(h, np.ones(2 * n - 1) / (2 * n - 1), n, dx)
            if g[j] <= 0:
                # try increasing h[j]
                step = min(0.05, 1.0 - h[j])
                hh = h.copy()
                jj = (j + n // 2) % n
                hh[j] += step
                hh[jj] -= step
                hh = np.clip(hh, 0.0, 1.0)
                hh = project_box_sum(hh, 0.5 * n)
                cc = erdos_c5(hh, n)
                if cc < cur - 1e-15:
                    h, cur = hh, cc
                    improved = True
            else:
                step = min(0.05, h[j])
                hh = h.copy()
                jj = (j + n // 2) % n
                hh[j] -= step
                hh[jj] += step
                hh = np.clip(hh, 0.0, 1.0)
                hh = project_box_sum(hh, 0.5 * n)
                cc = erdos_c5(hh, n)
                if cc < cur - 1e-15:
                    h, cur = hh, cc
                    improved = True
        if not improved:
            break
    return h, cur

def pair_polish(h, n, deadline, seed, n_pairs=20):
    rng = np.random.RandomState(seed)
    cur = erdos_c5(h, n)
    frac = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if frac.size == 0:
        return h, cur
    score = np.minimum(h[frac], 1.0 - h[frac])
    order = frac[np.argsort(score)[::-1]]
    top = order[:n_pairs] if order.size >= n_pairs else order
    for _ in range(4):
        if time.time() > deadline - 5.0:
            break
        improved = False
        pairs = [(int(top[a]), int(top[b])) for a in range(len(top)) for b in range(a + 1, len(top))]
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline - 5.0:
                break
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-12:
                continue
            def p_obj(d):
                a, b = h[i] + d, h[j] - d
                if a < -1e-12 or a > 1.0 + 1e-12 or b < -1e-12 or b > 1.0 + 1e-12:
                    return 1e100
                hh = h.copy()
                hh[i], hh[j] = a, b
                return erdos_c5(hh, n)
            try:
                rs = so.minimize_scalar(p_obj, bounds=(lo, hi), method='bounded',
                                        options={'maxiter': 30, 'xatol': 1e-10})
                if rs.x is not None:
                    cc = p_obj(float(rs.x))
                    if cc < cur - 1e-14:
                        hh = h.copy()
                        hh[i] += float(rs.x)
                        hh[j] -= float(rs.x)
                        h, cur = np.clip(hh, 0.0, 1.0), cc
                        improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s) - 5.0)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    if n < 2:
        n = 400
        h = np.full(n, 0.5)
    h = project_box_sum(h, 0.5 * n)
    best_h, best_n, best_c = h.copy(), n, erdos_c5(h, n)

    # Phase 1: quick softmax at n=400
    if n != 400 and n > 400:
        h400 = resample(h, n, 400)
    else:
        h400 = h.copy()
        n400 = n
    d1 = t0 + 0.15 * (deadline - t0)
    for beta in (200.0, 1000.0, 5000.0):
        if time.time() > d1 - 3:
            break
        def o1(x):
            return obj_softmax(x, 400, beta, 1e4)
        c1, c5_1 = opt_lbfgs(h400, 400, o1, 40, deadline)
        if c5_1 < best_c:
            best_h, best_n, best_c, h400 = c1.copy(), 400, c5_1, c1

    # Phase 2: upscale to n=800 and optimize
    n800 = 800
    h800 = resample(best_h, best_n, n800)
    d2 = t0 + 0.45 * (deadline - t0)
    for beta in (500.0, 3000.0, 15000.0, 80000.0):
        if time.time() > d2 - 5:
            break
        def o2(x):
            return obj_softmax(x, n800, beta, 1e4)
        c2, c5_2 = opt_lbfgs(h800, n800, o2, 50, deadline)
        if c5_2 < best_c:
            best_h, best_n, best_c = c2.copy(), n800, c5_2
            h800 = c2.copy()

    # Phase 3: hinge loss at n=800
    d3 = t0 + 0.72 * (deadline - t0)
    t_val = best_c * 0.999
    for _ in range(3):
        if time.time() > d3 - 10:
            break
        h800, t_val, c5_3 = opt_hinge_joint(h800, n800, t_val, 1e3, 60, deadline)
        if c5_3 < best_c:
            best_h, best_n, best_c = h800.copy(), n800, c5_3
            t_val = c5_3 * 0.995

    # Phase 4: also try n=600
    n600 = 600
    h600 = resample(best_h, best_n, n600)
    d4 = t0 + 0.85 * (deadline - t0)
    for beta in (2000.0, 20000.0):
        if time.time() > d4 - 5:
            break
        def o4(x):
            return obj_softmax(x, n600, beta, 1e4)
        c4, c5_4 = opt_lbfgs(h600, n600, o4, 40, deadline)
        if c5_4 < best_c:
            best_h, best_n, best_c = c4.copy(), n600, c5_4

    # Phase 5: pair polish on best
    d5 = deadline - 15.0
    if best_n <= 800 and time.time() < d5 - 30:
        best_h, best_c = pair_polish(best_h, best_n, d5, seed, n_pairs=25)

    # Phase 6: final softmax squeeze
    if time.time() < deadline - 10:
        def o6(x):
            return obj_softmax(x, best_n, 500000.0, 1e5)
        c6, c5_6 = opt_lbfgs(best_h, best_n, o6, 30, deadline)
        if c5_6 < best_c:
            best_h, best_c = c6.copy(), c5_6

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
