    axis_max = sh.max()
    s = np.exp(sh - axis_max).sum()

    m = c.max()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(40):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0:
        return m
    sh = alpha * (c - m)
    am = float(sh.max())
    s = float(np.exp(sh - am).sum())
    return m + (am + np.log(s)) / alpha

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    am = float(sh.max())
    e = np.exp(sh - am)
    s = float(e.sum())
    if s < 1e-300:
        return np.zeros(n)
    q = e / s
    dx = 2.0 / n
    off = n - 1
    grad = np.zeros(n)
    grad += q[off] * (1.0 - 2.0 * x)
    for d in range(1, n):
        qp = q[off + d]
        qm = q[off - d]
        if qp < 1e-16 and qm < 1e-16:
            continue
        if qp >= 1e-16:
            grad[:n-d] += qp * (1.0 - x[d:])
            grad[d:] -= qp * x[d:]
        if qm >= 1e-16:
            grad[d:] += qm * (1.0 - x[:n-d])
            grad[:n-d] -= qm * x[d:]
    return dx * grad

def sc_fun(x, n):
    return float(x.sum() - 0.5 * n)

def sc_jac(x, n):
    return np.ones(n)

def run_lbfgsb(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    try:
        res = minimize(_lbfgsb_obj, h, jac=_lbfgsb_jac,
                       args=(n, alpha, mu), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'maxfun': 3*maxiter, 'ftol': 1e-16})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_box_sum(res.x, n)
    except Exception:
        pass
    return h

def _lbfgsb_obj(x, n, alpha, mu):
    return lse_obj(x, n, alpha) + 0.5 * mu * (x.sum() - 0.5*n)**2

def _lbfgsb_jac(x, n, alpha, mu):
    return lse_grad(x, n, alpha) + mu * (x.sum() - 0.5*n) * np.ones(n)
