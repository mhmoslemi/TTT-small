def coord_descent(h, n, steps=5000, t_end=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(steps):
        if t_end and time.time() > t_end:
            break
        c = np.correlate(h, 1.0-h, mode='full')
        km = int(np.argmax(c))
        k = km - (n-1)
        # Try changing each h[j] to reduce c[k]
        # ∂c_k/∂h[j] = (1-h[j+k])[valid] - h[j-k][valid]
        lo1, hi1 = max(0,-k), min(n, n-k)
        lo2, hi2 = max(0,k), min(n, n+k)
        # Find j where gradient is most negative (decreasing h[j] helps) 
        # or most positive (increasing h[j] helps)
        g = np.zeros(n)
        if hi1 > lo1: g[lo1:hi1] += (1.0-h[lo1+k:hi1+k])
        if hi2 > lo2: g[lo2:hi2] -= h[lo2-k:hi2-k]
        # Try the most negative gradient component (increase h) or most positive (decrease h)
        if g.min() > 0:
            j = int(np.argmin(g))  # most positive → decrease h[j]
            step = min(h[j], 0.1)
            h[j] -= step
        elif g.max() < 0:
            j = int(np.argmin(g))  # most negative → increase h[j]
            step = min(1.0-h[j], 0.1)
            h[j] += step
        else:
            j = int(np.argmin(g))
            if g[j] < 0:
                h[j] = min(1.0, h[j] + 0.05)
            else:
                h[j] = max(0.0, h[j] - 0.05)
        h = project_feasible(h, n)
        mc = compute_c5(h, n)
        if mc >= best_c + 1e-10:
            h[j] += step if g[j] < 0 else -step  # undo
            h = project_feasible(h, n)
    return h, best_c

import numpy as np
import time
from scipy.optimize import linprog

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-13:
        return h
    lo, hi = -1.5, 2.5
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        s = np.clip(h - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - h.sum()
    if abs(r) > 1e-10:
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.sum() > 0:
            h[m] = np.clip(h[m] + r / m.sum(), 0, 1)
    return h

def lp_step(h, n, eps):
    C = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        lo1, hi1 = max(0, -k), min(n, n - k)
        if hi1 > lo1:
            G[m, lo1:hi1] += 1.0 - h[lo1 + k:hi1 + k]
        lo2, hi2 = max(0, k), min(n, n + k)
        if hi2 > lo2:
            G[m, lo2:hi2] -= h[lo2 - k:hi2 - k]
    nv = n + 1
    c_obj = np.zeros(nv)
    c_obj[-1] = 1.0
    nr = (2 * n - 1) + 4 * n
    A = np.zeros((nr, nv))
    b = np.zeros(nr)
    A[:2 * n - 1, :n] = G
    A[:2 * n - 1, n] = -1.0
    b[:2 * n - 1] = -C
    r = 2 * n - 1
    I = np.eye(n)
    A[r:r + n, :n] = I; b[r:r + n] = eps; r += n
    A[r:r + n, :n] = -I; b[r:r + n] = eps; r += n
    A[r:r + n, :n] = I; b[r:r + n] = 1.0 - h; r += n
    A[r:r + n, :n] = -I; b[r:r + n] = h
    Aeq = np.zeros((1, nv)); Aeq[0, :n] = 1.0
    beq = np.zeros(1)
    bnds = [(-1.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq,
                      bounds=bnds, method='highs')
    except Exception:
        return h, compute_c5(h, n)
    if res.status != 0:
        return h, compute_c5(h, n)
    d = res.x[:n]
    best_h, best_c = h.copy(), compute_c5(h, n)
    for lam in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
        ht = project_feasible(h + lam * d, n)
        mc = compute_c5(ht, n)
        if mc < best_c:
            best_c, best_h = mc, ht
    return best_h, best_c

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    w = np.exp(alpha * (c - cmax))
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(c); w[int(np.argmax(c))] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-13:
            continue
        k = m - (n - 1)
        lo1, hi1 = max(0, -k), min(n, n - k)
        if hi1 > lo1:
            grad[lo1:hi1] += wm * (1.0 - h[lo1 + k:hi1 + k])
        lo2, hi2 = max(0, k), min(n, n + k)
        if hi2 > lo2:
            grad[lo2:hi2] -= wm * h[lo2 - k:hi2 - k]
    return grad * dx

def smooth_opt(h, n, t_end, n_iter=5000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 50.0
    lr = 1.0 / n
    y = h.copy()
    t_k = 1.0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        g = smooth_grad(y, n, alpha)
        x = project_feasible(y - lr * g, n)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x + ((t_k - 1.0) / t_new) * (x - h)
        y = np.clip(y, 0.0, 1.0)
        h = x
        t_k = t_new
        c5 = compute_c5(h, n)
        if c5 < best_c:
            best_c = c5; best_h = h.copy()
        if it % 200 == 0:
            alpha = min(alpha * 1.4, 10000.0)
            lr = min(lr * 1.05, 3.0 / n)
            t_k = 1.0
        if it % 1000 == 0:
            lr *= 0.7
            t_k = 1.0
    return best_h, best_c
