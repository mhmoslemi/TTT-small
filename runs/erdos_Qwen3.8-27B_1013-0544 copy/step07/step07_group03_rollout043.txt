def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break

            s = room.sum()
            if s < 1e-14:
                break

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

def slsqp_alpha(h, n, alpha, maxiter, ftol):

        m = c.max()
        if alpha < 1e8:

            sm = shifted.max()

            return m + logsum / alpha
