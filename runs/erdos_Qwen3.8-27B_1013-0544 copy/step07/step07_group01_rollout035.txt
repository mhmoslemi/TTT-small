    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            rsum = room.sum()
            if rsum < 1e-14:
                break
            h = h + diff / rsum * room
        else:
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(x, n, alpha):

    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    sm = sh.max()

    p = np.exp(sh - sm)

def lbgfss(h, n, alpha, maxiter):
    try:
        res = minimize(smooth_obj_grad, h, args=(n, alpha), jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_polish(h, n, deadline, niter=500000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = dx * 0.5
    momentum = np.zeros(n)
    mu = 0.92
    for i in range(niter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0 and lag < n:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        elif lag < 0:
            ll = -lag
            if ll < n:
                grad[ll:] += (1.0 - h[:n - ll])
                grad[:n - ll] -= h[ll:]
        grad *= dx
        momentum = mu * momentum - lr * grad
        h = project_feasible(h + momentum, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            lr = min(lr * 1.005, dx * 5.0)
        else:
            if i % 500 == 0:
                lr = max(lr * 0.95, dx * 0.01)
    return best_h, best_c

def interp_resample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

            hl_hi = lbgfss(hl_hi, n, 100000, 1000)
            ch = compute_c5(hl_hi, n)
            if ch < best_c:
                best_c = ch
                best_h = hl_hi.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
