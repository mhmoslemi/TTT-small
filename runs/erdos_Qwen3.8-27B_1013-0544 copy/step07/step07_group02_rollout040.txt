    """Solve min_t max_k (beta[k] + alpha[k]*t) for t in [lo, hi]."""
    m = len(alpha)
    # Check endpoints
    f_lo = np.max(beta + alpha * lo)
    f_hi = np.max(beta + alpha * hi)
    
    # Bisection on subgradient
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = beta + alpha * mid
        f_mid = vals.max()
        active = np.where(vals >= f_mid - 1e-14)[0]
        min_slope = alpha[active].min()
        max_slope = alpha[active].max()

        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,

def smooth_grad_descent(h, n, alpha_smooth, n_iters, lr):
    """Gradient descent on smoothed max objective."""
    dx = 2.0 / n
    for it in range(n_iters):
        C = np.correlate(h, 1 - h, mode='full') * dx
        m = C.max()
        # Smooth max: log-sum-exp
        shifted = alpha_smooth * (C - m)
        sm = shifted.max()
        weights = np.exp(shifted - sm)
        weights /= weights.sum()
        
        # Gradient of smooth max w.r.t. h
        G = compute_gradient_matrix(h, n)
        grad = (G * weights).sum(axis=1)
        grad *= dx
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        
        # Check if we improved
        C_new = np.correlate(h_new, 1 - h_new, mode='full') * dx

            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = project_feasible(h_new, n)
            h = h_new

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient_matrix(h, n):
    G = np.zeros((n, 2 * n - 1))
    Nm1 = n - 1
    for i in range(n):
        m_lo = max(0, i)
        m_hi = min(2 * n - 2, Nm1 + i)
        if m_hi >= m_lo:
            n_a = Nm1 + i - np.arange(m_lo, m_hi + 1)
            G[i, m_lo:m_hi + 1] += (1.0 - h[n_a])
        m_lo2 = max(0, Nm1 - i)
        m_hi2 = min(2 * n - 2, 2 * Nm1 - i)
        if m_hi2 >= m_lo2:
            n2 = i + np.arange(m_lo2, m_hi2 + 1) - Nm1
            G[i, m_lo2:m_hi2 + 1] -= h[n2]
    return G

        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0

        if mc_try < best_mc - 1e-15:

    t_grid = np.linspace(0.0, 1.0, grid_size)
    Nm1 = n - 1
    for sweep in range(n_sweeps):
        C = np.correlate(h, 1 - h, mode='full').copy()
        for i in range(n):
            old = h[i]
            alpha = np.zeros(2 * n - 1)
            m_lo = max(0, i)
            m_hi = min(2 * n - 2, Nm1 + i)
            if m_hi >= m_lo:
                n_a = Nm1 + i - np.arange(m_lo, m_hi + 1)
                alpha[m_lo:m_hi + 1] += (1.0 - h[n_a])
            m_lo2 = max(0, Nm1 - i)
            m_hi2 = min(2 * n - 2, 2 * Nm1 - i)
            if m_hi2 >= m_lo2:
                n2 = i + np.arange(m_lo2, m_hi2 + 1) - Nm1
                alpha[m_lo2:m_hi2 + 1] -= h[n2]
            beta = C - old * alpha
            vals = beta[None, :] + alpha[None, :] * t_grid[:, None]
            quad_corr = (t_grid - old) - (t_grid ** 2 - old ** 2)
            vals[:, Nm1] += quad_corr
            f = vals.max(axis=1)
            best_idx = f.argmin()
            h[i] = t_grid[best_idx]
        h = project_feasible(h, n)
        C = np.correlate(h, 1 - h, mode='full')
    return h

def smooth_grad_descent(h, n, alpha_s, n_iters, lr):
