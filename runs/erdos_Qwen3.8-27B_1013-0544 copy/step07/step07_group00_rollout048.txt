def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def projected_gradient(h, n, alpha, lr, n_iters, deadline=None):
    h = h.copy()
    dx = 2.0 / n
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        weights = np.exp(alpha * (c - m))
        w_sum = weights.sum()
        
        # Gradient of smooth max with respect to h
        grad = np.zeros(n)
        for k in range(len(c)):
            w = weights[k] / w_sum
            if w < 1e-10:
                continue
            # ∂c_k/∂h[p] for each p
            # First factor: p appears as h[p] when p+k-n+1 in [0,n)
            # Second factor: p appears as h in (1-h) when p-k+n-1 in [0,n)
            k_offset = k - (n-1)
            # First: h[p] * (1-h[p+k_offset]), valid when 0<=p<n and 0<=p+k_offset<n
            # ∂/∂h[p] = (1-h[p+k_offset])
            # Second: h[p-k_offset] * (1-h[p]), valid when 0<=p-k_offset<n and 0<=p<n
            # ∂/∂h[p] = -h[p-k_offset]
            
            # For efficiency, vectorize
            # First factor contributions
            p_range_f = np.arange(0, n)
            idx_f = p_range_f + k_offset
            valid_f = (idx_f >= 0) & (idx_f < n)
            grad[p_range_f[valid_f]] += w * (1.0 - h[idx_f[valid_f]])
            
            # Second factor contributions  
            p_range_b = np.arange(0, n)
            idx_b = p_range_b - k_offset
            valid_b = (idx_b >= 0) & (idx_b < n)
            grad[p_range_b[valid_b]] -= w * h[idx_b[valid_b]]
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Check improvement (with Armijo-like condition)
        c_old = compute_c5(h, n)
        c_new = compute_c5(h_new, n)
        if c_new < c_old:
            h = h_new
        else:
            lr /= 2.0
            if lr < 1e-10:
                break
    
    return h, compute_c5(h, n)

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return project_feasible(res.x, n)

def cd_fine(h, n, max_passes=100, deadline=None, grid_size=300):

        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            # Compute affected correlations
            k_lo = max(0, min(n-1-p, p))
            k_hi = min(2*n-2, max(2*n-2-p, p+n-1))
            ks = np.arange(k_lo, k_hi+1)
            a = np.zeros(len(ks))
            # First factor
            idx_f = p + ks - n + 1
            vf = (idx_f >= 0) & (idx_f < n)
            a[vf] += (1.0 - h[idx_f[vf]])
            # Second factor
            idx_b = p - ks + n - 1
            vb = (idx_b >= 0) & (idx_b < n)
            a[vb] -= h[idx_b[vb]]
            # Diagonal
            diag = (ks == n-1).astype(float)

            unaff = 0.0

            if k_hi < 2*n-2:

            # Grid search
            d_grid = np.linspace(-h0, 1.0-h0, grid_size)
            best_d = 0.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def cd_fine(h, n, max_passes=100, deadline=None, grid_size=400):
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            k_lo = max(0, min(n - 1 - p, p))
            k_hi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
            ks = np.arange(k_lo, k_hi + 1)
            a = np.zeros(len(ks))
            idx_f = p + ks - n + 1
            vf = (idx_f >= 0) & (idx_f < n)
            a[vf] += (1.0 - h[idx_f[vf]])
            idx_b = p - ks + n - 1
            vb = (idx_b >= 0) & (idx_b < n)
            a[vb] -= h[idx_b[vb]]
            diag = (ks == n - 1).astype(float)
            c_aff = c[ks]
            unaff = 0.0
            if k_lo > 0:
                unaff = max(unaff, float(c[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff = max(unaff, float(c[k_hi + 1:].max()))
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :] - diag[None, :] * d_grid[:, None] ** 2
            vals = np.maximum(unaff, nv.max(axis=1))
            idx_best = vals.argmin()
            best_d = d_grid[idx_best]
            best_v = vals[idx_best]
            if abs(best_d) > 1e-12 and best_v < cur_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def interpolate_h(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    return np.interp(x_dst, x_src, h_src)

def optimize_at_n(h_start, n, deadline, min_remaining=60):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alphas = [2000, 20000, 100000, 500000]
    for a in alphas:
        if time.time() > deadline - min_remaining:
            break
        h_new = slsqp_step(best_h, n, a, maxiter=2000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
    if time.time() < deadline - min_remaining:
        h_new, c_new = cd_fine(best_h, n, 150, deadline, grid_size=500)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
    return best_h, best_c
