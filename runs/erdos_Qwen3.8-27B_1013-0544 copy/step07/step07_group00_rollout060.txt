                best_h, best_c = hq, cq

    # Phase 5: Try different resolutions
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [200, 300, 400]:
        if time.time() > deadline - 150:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h1, c1 = coord_cd(h_int, n_try, 60, deadline, grid_size=60, refine=False)
        if time.time() < deadline - 80:
            h2, c2 = slsqp_opt(h1, n_try, [3000, 30000], 1000)
            h2, c2 = coord_cd(h2, n_try, 80, deadline, grid_size=100, refine=True)
        else:
            h2, c2 = h1, c1
        if c2 < best_c:
            best_h, best_c = h2, c2
            n = n_try
            x_orig = x_new

    # Phase 6: High-alpha SLSQP polish
    if time.time() < deadline - 30:
        n = len(best_h)
        hf, cf = slsqp_opt(best_h, n, [100000, 500000], 2000)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Phase 7: Final fine coordinate descent
    if time.time() < deadline - 5:
        n = len(best_h)
        hf, cf = coord_cd(best_h, n, 100, deadline, grid_size=120, refine=True)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
