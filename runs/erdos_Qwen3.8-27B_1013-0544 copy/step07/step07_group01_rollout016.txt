```python
    # df_m/dh[j] = (1-h[j-m])*1_{j>=m} + (-h[j+m])*1_{j<n-m} for m>0
    # df_0/dh[j] = 1-2*h[j]
    
    grad = np.zeros(n)
    # Lag 0
    m0 = p[n-1]
    if m0 > 1e-15:
        grad += m0 * (1.0 - 2.0*h)
    
    # Positive lags: m = 1, ..., n-1 (p index = n-1+m)
    for m in range(1, n):
        w = p[n-1+m]
        if w < 1e-12:
            continue
        grad[m:] += w * (1.0 - h[:n-m])
        grad[:n-m] -= w * h[m:]
        # Negative lags (by symmetry, same contribution):
        w2 = p[n-1-m]
        if w2 < 1e-12:
            continue
        grad[m:] += w2 * (1.0 - h[:n-m])
        grad[:n-m] -= w2 * h[m:]
    
    return grad * dx, corr
```

This is O(n²) per gradient eval. For n=400, that's 160K operations (well, 2*sum_{m=1}^{399} m ≈ 160K). Should be fine.

But wait, the loop over m from 1 to n-1 with numpy operations inside is actually O(n) numpy calls, each O(n) work. So total is O(n²) but with n=400 numpy calls of average length 200. That's 400 * 200 = 80K operations per gradient. Very fast.

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sh = corr - corr.max()
    p = np.exp(alpha * sh)
    ps = p.sum()
    if ps < 1e-30:
        p = np.zeros(2*n-1)
        p[np.argmax(corr)] = 1.0
    else:
        p /= ps
    
    grad = np.zeros(n)
    m0 = p[n-1]
    if m0 > 1e-15:
        grad += m0 * (1.0 - 2.0*h)
    for m in range(1, n):
        wp = p[n-1+m]
        wn = p[n-1-m]
        w = wp + wn
        if w < 1e-12:
            continue
        grad[m:] += w * (1.0 - h[:n-m])
        grad[:n-m] -= w * h[m:]
    return grad * dx, float(corr.max()), p
