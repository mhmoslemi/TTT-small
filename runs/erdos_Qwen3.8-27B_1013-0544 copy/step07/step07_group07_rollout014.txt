def lse_valgrad_full(f, alpha):
    h = 0.5 + f
    c = np.correlate(h, 1.0 - h, mode='full')
    n = len(f)
    maxc = float(np.max(c))
    sh = alpha * (c - maxc)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.full(c.shape, 1.0/float(c.size))
        val = maxc
    else:
        w = e/s
        val = maxc + np.log(s)/alpha
    kern = np.zeros(c.size)
    kern[n-1] = 2.0*w[n-1]
    if n > 1:
        a = w[n:] + w[:n-1][::-1]
        kern[n:] = a
        kern[:n-1] = a[::-1]
    conv = np.convolve(f, kern, mode='full')
    grad = -conv[n-1:2*n-1]
    return val, grad

            b = m

def lse_valgrad_full(f, alpha):
    f = np.asarray(f, dtype=float)
    n = len(f)
    h = 0.5 + f
    c = np.correlate(h, 1.0 - h, mode='full')
    maxc = float(np.max(c))
    sh = alpha * (c - maxc)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
       w = np.full(c.shape, 1.0/c.size)
       val = maxc
    else:
       w = e/s
       val = maxc + float(np.log(s))/alpha
    kern = np.zeros(c.size, dtype=float)
    kern[n-1] = 2.0*w[n-1]
    if n > 1:
       a = w[n:] + w[:n-1][::-1]
       kern[n:] = a
       kern[:n-1] = a[::-1]
    conv = np.convolve(f, kern, mode='full')
    grad = -conv[n-1:2*n-1]
    return val, grad

        if np.clip(v + a, -0.5, 0.5).sum() > 0.0:
            a -= 1.0
        if np.clip(v + b, -0.5, 0.5).sum() < 0.0:
            b += 1.0
    for _ in range(90):
        m = 0.5 * (a + b)
        if np.clip(v + m, -0.5, 0.5).sum() < 0.0:
            a = m
        else:
            b = m
    return np.clip(v + 0.5 * (a + b), -0.5, 0.5)


def sym_to_full(y):
    y = np.asarray(y, dtype=float)
    m = len(y)
    f = np.empty(2 * m, dtype=float)
    f[:m] = y
    f[m:] = y[::-1]
    return f


def lse_valgrad_full(f, alpha):
    f = np.asarray(f, dtype=float)
    n = len(f)
    h = 0.5 + f
    c = np.correlate(h, 1.0 - h, mode='full')
    maxc = float(np.max(c))
    sh = alpha * (c - maxc)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.full(c.shape, 1.0 / float(c.size))
        val = maxc
    else:
        w = e / s
        val = maxc + float(np.log(s)) / alpha
    kern = np.zeros(c.size, dtype=float)
    kern[n - 1] = 2.0 * w[n - 1]
    if n > 1:
        a = w[n:] + w[:n - 1][::-1]
        kern[n:] = a
        kern[:n - 1] = a[::-1]
    conv = np.convolve(f, kern, mode='full')
    grad = -conv[n - 1:2 * n - 1]
    return val, grad


def fun_lse_y(y, alpha):
    return float(lse_valgrad_full(sym_to_full(y), alpha)[0])


def jac_lse_y(y, alpha):
    f = sym_to_full(y)
    _, g = lse_valgrad_full(f, alpha)
    m = len(y)
    return g[:m] + g[2 * m - 1::-1]


def sym_cons(y):
    return float(np.sum(y))


def sym_cons_jac(y):
    return np.ones(len(y), dtype=float)


def fun_lse_f_pen(f, alpha, mu):
    val, _ = lse_valgrad_full(f, alpha)
    s = float(np.sum(f))
    return float(val + 0.5 * mu * s * s)


def jac_lse_f_pen(f, alpha, mu):
    _, g = lse_valgrad_full(f, alpha)
    s = float(np.sum(f))
    return g + mu * s * f


def run_slsqp_y(y, alpha, maxiter, ftol):
    y = np.asarray(y, dtype=float)
    m = len(y)
    bounds = [(-0.5, 0.5)] * m
    cons = [{'type': 'eq', 'fun': sym_cons, 'jac': sym_cons_jac}]
    try:
        res = minimize(
            fun_lse_y, y,
            args=(alpha,),
            jac=jac_lse_y,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': float(ftol)}
        )
        if hasattr(res, 'x') and res.x is not None:
            y2 = proj_zero(np.asarray(res.x, dtype=float))
            if np.all(np.isfinite(y2)):
                y = y2
    except Exception:
        pass
    return y


def run_lbsb_f(f, alpha, mu, maxiter, ftol):
    f = np.asarray(f, dtype=float)
    bounds = [(-0.5, 0.5)] * len(f)
    try:
        res = minimize(
            fun_lse_f_pen, f,
            args=(alpha, mu),
            jac=jac_lse_f_pen,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': int(maxiter), 'ftol': float(ftol),
                     'maxcor': 10, 'maxls': 40}
        )
        if hasattr(res, 'x') and res.x is not None:
            f2 = proj_zero(np.asarray(res.x, dtype=float))
            if np.all(np.isfinite(f2)):
                f = f2
    except Exception:
        pass
    return f


def eval_h(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))


def make_y_starts(h, seed):
    n = len(h)
    m = n // 2
    f = proj_zero(h - 0.5)
    f_sym = 0.5 * (f + f[::-1])
    y0 = proj_zero(f_sym[:m])
    y1 = proj_zero(f[:m])
    x = np.linspace(-1.0, 1.0, m, dtype=float)
    y2 = proj_zero(0.25 * (1.0 - 3.0 * x * x))
    rng = np.random.RandomState(seed)
    y3 = proj_zero(y0 + 0.02 * np.sin(2.0 * np.pi * 0.7 * x) + 0.01 * rng.randn(m))
    y4 = proj_zero(y0 + 0.03 * x * x - 0.01)
    return [y0, y1, y2, y3, y4]


def pg_y(y, alpha, deadline, maxiter=300, step0=0.1):
    y = proj_zero(np.asarray(y, dtype=float))
    m = len(y)
    f = sym_to_full(y)
    val, g = lse_valgrad_full(f, alpha)
    best_y = y.copy()
    best_val = float(val)
    step = step0
    for _ in range(int(maxiter)):
        if time.time() > deadline or not np.isfinite(val):
            break
        if val < best_val:
            best_val = float(val)
            best_y = y.copy()
        gy = g[:m] + g[2 * m - 1::-1]
        gn = float(np.linalg.norm(gy))
        if gn < 1e-12:
            break
        st = step / (1.0 + gn)
        ok = False
        for _ in range(22):
            yn = proj_zero(y - st * gy)
            vn, gn2 = lse_valgrad_full(sym_to_full(yn), alpha)
            if np.isfinite(vn) and vn < val - 1e-12:
                y = yn
                val = float(vn)
                g = gn2
                ok = True
                break
            st *= 0.5
        if not ok:
            st = min(step, 1e-4) / (1.0 + gn)
            yn = proj_zero(y - st * gy)
            vn, gn2 = lse_valgrad_full(sym_to_full(yn), alpha)
            if np.isfinite(vn) and vn < val:
                y = yn
                val = float(vn)
                g = gn2
            else:
                break
        if val < best_val:
            best_val = float(val)
            best_y = y.copy()
    return best_y


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(30.0, float(budget_s) - 12.0)
    dl = t0 + total

    try:
        h = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(400, 0.5, dtype=float)

    h = 0.5 + proj_zero(h - 0.5)
    n = len(h)
    best_h = h.copy()
    best_c = eval_h(best_h)
    best_y = None

    if n % 2 == 0 and n >= 2:
        m = n // 2
        starts = make_y_starts(h, secret_widget if isinstance(secret_widget := None, int) else seed)
        scored = []
        for idx, y in enumerate(starts):
            if time.time() > dl - 2.0:
                break
            y = proj_zero(y)
            hh = 0.5 + sym_to_full(y)
            c = eval_h(hh)
            scored.append((float(c), int(idx), y))
            if c < best_c:
                best_c = float(c)
                best_h = hh.copy()
                best_y = y.copy()
        scored.sort()
        chosen = []
        for c, idx, y in scored:
            if len(chosen) >= 2:
                break
            chosen.append((c, y))
        if len(scored) > 2 and len(chosen) < 3 and scored[2][0] < max(0.45, best_c + 0.025):
            chosen.append((scored[2][0], scored[2][2]))

        alpha_seq = [30.0, 150.0, 800.0, 4000.0, 20000.0, 80000.0, 250000.0]
        maxit_seq = [250, 500, 800, 1200, 1600, 1600, 1200]
        sym_end = t0 + 0.42 * total

        for si, (cstart, ys) in enumerate(chosen):
            ys = proj_zero(ys)
            for ai, alpha in enumerate(alpha_seq):
                if time.time() > sym_end:
                    break
                mi = maxit_seq[ai]
                if si > 0:
                    mi = int(mi * 0.5)
                    if ai > 4:
                        break
                if time.time() > sym_end - 20.0:
                    mi = min(mi, 60)
                if mi < 10:
                    break
                ys = run_slsqp_y(ys, alpha, mi, 1e-12)
                hh = 0.5 + sym_to_full(ys)
                c = eval_h(hh)
                if c < best_c:
                    best_c = float(c)
                    best_h = hh.copy()
                    best_y = ys.copy()
                if cstart > best_c and ai >= 1 and c >= best_c:
                    break

    full_end = t0 + 0.62 * total
    f = proj_zero(best_h - 0.5)
    tasks = [
        (1000.0, 1e4, 800),
        (8000.0, 1e5, 1000),
        (50000.0, 1e5, 1000),
        (200000.0, 1e6, 700)
    ]
    for alpha, mu, mi in tasks:
        if time.time() > full_end:
            break
        if time.time() > full_end - 20.0:
            mi = min(mi, 70)
        if mi < 10:
            break
        f = run_lbsb_f(f, alpha, mu, mi, 1e-14)
        hh = 0.5 + f
        c = eval_h(hh)
        if c < best_c:
            best_c = float(c)
            best_h = hh.copy()

    if time.time() < full_end and len(best_h) < 600:
        rng = np.random.RandomState(int(seed) + 17)
        x = np.linspace(0.0, 1.0, len(best_h), dtype=float)
        fp = proj_zero(f + 0.018 * np.sin(2.0 * np.pi * 1.25 * x) + 0.012 * rng.randn(len(best_h)))
        fp = run_lbsb_f(fp, 20000.0, 1e5, 600, 1e-14)
        hh = 0.5 + fp
        c = eval_h(hh)
        if c < best_c:
            best_c = float(c)
            best_h = hh.copy()

    ups_end = t0 + 0.84 * total
    nh = 800
    if time.time() < ups_end - 60.0 and len(best_h) < 600:
        old_x = np.linspace(0.0, 2.0, len(best_h), dtype=float)
        new_x = np.linspace(0.0, 2.0, nh, dtype=float)
        hup = np.interp(new_x, old_x, best_h)
        fup = proj_zero(hup - 0.5)
        fup = run_lbsb_f(fup, 3000.0, 1e5, 550, 1e-14)
        hup = 0.5 + fup
        cup = eval_h(hup)
        if cup < best_c:
            best_c = float(cup)
            best_h = hup.copy()
        if time.time() < ups_end - 40.0:
            f = proj_zero(best_h - 0.5)
            f = run_lbsb_f(f, 25000.0, 1e6, 500, 1e-15)
            hh = 0.5 + f
            c = eval_h(hh)
            if c < best_c:
                best_c = float(c)
                best_h = hh.copy()

    nbest = len(best_h)
    if time.time() < dl - 35.0 and nbest % 2 == 0:
        if nbest // 2 <= 230:
            if best_y is not None and len(best_y) == nbest // 2:
                yy = proj_zero(best_y)
            else:
                ff = proj_zero(best_h - 0.5)
                yy = proj_zero(0.5 * (ff + ff[::-1])[:nbest // 2])
            for alpha, mi in [(100000.0, 1200), (400000.0, 900)]:
                if time.time() > dl - 28.0:
                    break
                mi = min(mi, 400) if time.time() > dl - 60.0 else mi
                yy = run_slsqp_y(yy, alpha, mi, 1e-13)
                hh = 0.5 + sym_to_full(yy)
                c = eval_h(hh)
                if c < best_c:
                    best_c = float(c)
                    best_h = hh.copy()
                    best_y = yy.copy()
        else:
            ff = proj_zero(best_h - 0.5)
            ff = run_lbsb_f(ff, 300000.0, 1e6, 450, 1e-15)
            hh = 0.5 + ff
            c = eval_h(hh)
            if c < best_c:
                best_c = float(c)
                best_h = hh.copy()

    if time.time() < dl - 14.0 and nbest % 2 == 0 and nbest // 2 <= 230:
        ff = proj_zero(best_h - 0.5)
        yy = proj_zero(0.5 * (ff + ff[::-1])[:nbest // 2])
        yy = pg_y(yy, 120000.0, dl - 9.0, 250, 0.12)
        hh = 0.5 + sym_to_full(yy)
        c = eval_h(hh)
        if c < best_c:
            best_c = float(c)
            best_h = hh.copy()

    best_h = 0.5 + proj_zero(best_h - 0.5)
    best_c = eval_h(best_h)
    return best_h, float(best_c), int(len(best_h))
