def project_feasible(a, n):

            a = a[:n]

    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(a - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    # Fix sum exactly
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:

        else:
            s = h.sum()
            if s < 1e-15:

def lse_objective_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

def lbfgs_objective(x, n, beta, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    F, g = lse_objective_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

def candidate_variants(h, n):
    h = project_feasible(h, n)
    out = [h, h[::-1], 1.0 - h[::-1], 0.5 * (h + h[::-1])]
    if n % 2 == 0:
        out.append(top_n_binary(h, n))
    res = []
    for x in out:
        res.append(project_feasible(x, n))
    return res

def polish_solution(h, n, betas, pen, deadline, stage_time):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        now = time.time()
        if now > deadline - 5.0:
            break
        stage_dl = min(deadline - 3.0, now + max(2.0, min(stage_time, 0.5 * max(2.0, deadline - now))))
        _stop_time = stage_dl
        try:
            res = minimize(
                lbfgs_objective, work, method="L-BFGS-B", jac=True,
                bounds=bounds, args=(n, float(beta), float(pen)),
                callback=stop_callback,
                options={"maxiter": 500000, "maxls": 40, "ftol": 1e-14, "gtol": 1e-9}
            )
            x = res.x if (res is not None and hasattr(res, "x")) else work
        except Exception:
            x = work
        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc:
            bc = cw
            best = work.copy()
        for cand in candidate_variants(work, n):
            cc = compute_c5(cand, n)
            if cc < bc:
                bc = cc
                best = cand.copy()
        if bc < cw - 1e-12:
            work = best.copy()
    return best, bc

def make_comb(n, period):
    """Create a comb pattern with given period and 50% duty cycle."""
    h = np.zeros(n)
    half = period // 2
    for i in range(n):
        if (i % period) < half:
            h[i] = 1.0
    return h

        else: hi = mid

    """One sweep of binary greedy. Returns (h, corr, M, improved)."""
    zeros = np.nonzero(h == 0)[0]
    ones = np.nonzero(h == 1)[0]
    if zeros.size == 0 or ones.size == 0:
        return h, corr, M, False
    
    zi = np.repeat(zeros, ones.size)
    oj = np.tile(ones, zeros.size)
    total = zi.size
    K = np.arange(2 * n - 1)
    improved = False
    
    for start in range(0, total, chunk):
        if time.time() > deadline:
            break
        end = min(start + chunk, total)
        i = zi[start:end]
        j = oj[start:end]
        J = j.reshape(-1, 1)
        I = i.reshape(-1, 1)
        idx_j = J + K - (n - 1)
        idx_i = I + K - (n - 1)
        valid_j = (idx_j >= 0) & (idx_j < n)
        valid_i = (idx_i >= 0) & (idx_i < n)
        a_j = np.where(valid_j, h[idx_j[valid_j]], 0.0)
        a_i = np.where(valid_i, h[idx_i[valid_i]], 0.0)
        new_corr = corr[None, :] + a_j - a_i
        max_vals = new_corr.max(axis=1)

            loc = int(candidates[np.argmin(max_vals[candidates])])

            h[iv] = 1.0
            h[jv] = 0.0
            corr = new_corr[loc].copy()

            improved = True
            break

    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)

    y = beta * (c - m)

    A = np.convolve(R, v, mode="full")[n-1:2*n-1]
    B = np.convolve(P, h, mode="full")[n-1:2*n-1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):

    return float(F), g

def project_feasible(a, n):

def lse_grad(h, n, beta):

    g = dx * (A - B)
