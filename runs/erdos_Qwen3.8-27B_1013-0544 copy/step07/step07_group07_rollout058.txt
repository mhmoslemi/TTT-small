The gradient of 0.5 * Σ (max(c_k - t, 0))^2 with respect to h[j]:
= Σ_{k: c_k > t} (c_k - t) * ∂c_k/∂h[j]

And ∂c_k/∂h[j] for k corresponding to lag l (= k - (n-1)):
= (1-h[j+l]) * I(valid) - h[j-l] * I(valid)

So the gradient is:

import numpy as np
from scipy.optimize import minimize

_state = {}

def _compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-12:
                break
            h += d / s * r
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h += d / s * h
    return np.clip(h, 0.0, 1.0)

def _corr_grad(x, n, weights):
    grad = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = weights[n - 1 + l]
        if wl < 1e-14:
            continue
        if l >= 0:
            grad[:n - l] += wl * (1.0 - x[l:])
            grad[l:] -= wl * x[:n - l]
        else:
            ll = -l
            grad[ll:] += wl * (1.0 - x[:n - ll])
            grad[:n - ll] -= wl * x[ll:]
    return grad

def _lse_obj(x):
    n, alpha = _state['n'], _state['alpha']
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    cm = c.max()
    s = alpha * (c - cm)
    e = np.exp(s)
    return cm + np.log(e.sum()) / alpha

def _lse_jac(x):

    dx = 2.0 / n

    cm = c.max()
    s = alpha * (c - cm)
    e = np.exp(s)
    w = e / e.sum()
    return _corr_grad(x, n, w)
