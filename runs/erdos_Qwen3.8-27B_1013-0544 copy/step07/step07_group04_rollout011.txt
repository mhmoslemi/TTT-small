        xs = np.concatenate((xs, np.linspace(lo, hi, 32)))
    a = max(lo, x0 - 0.03)
    b = min(hi, x0 + 0.03)
    if b - a > 1e-12:
        xs = np.concatenate((xs, np.linspace(a, b, 128)))
    xs = np.unique(xs)
    xs = xs[(xs >= lo - 1e-12) & (xs <= hi + 1e-12)]
    return np.clip(xs, lo, hi)

def pair_eval(h, c_base, i, j, xs, n, x0, y0, S):
    L = 2 * n - 1
    off = n - 1
    dx = 2.0 / n
    omh = 1.0 - h
    omhrev = omh[::-1]
    delta = np.zeros((len(xs), L), dtype=float)
    ax1 = xs - x0
    ax2 = x0 - xs
    yv = S - xs
    delta[:, i:i+n] += ax1[:, None] * omhrev[None, :]
    si = n - 1 - i
    delta[:, si:si+n] += ax2[:, None] * h[None, :]
    delta[:, j:j+n] += ax2[:, None] * omhrev[None, :]
    sj = n - 1 - j
    delta[:, sj:sj+n] += ax1[:, None] * h[None, :]
    idx0 = off
    base_self = ax1*(1.0-x0) + x0*ax2 + ax2*(1.0-y0) + y0*ax1
    exact_self = xs*(1.0-xs) + yv*(1.0-yv) - (x0*(1.0-x0) + y0*(1.0-y0))
    delta[:, idx0] += exact_self - base_self
    d = j - i
    idxp = off - d
    base_p = ax1*(1.0-y0) + x0*ax1
    exact_p = xs*(1.0-yv) - x0*(1.0-y0)
    delta[:, idxp] += exact_p - base_p
    idxm = off + d
    base_m = ax2*(1.0-x0) + y0*ax2
    exact_m = yv*(1.0-xs) - y0*(1.0-x0)
    delta[:, idxm] += exact_m - base_m
    return np.max(c_base[None, :] + dx * delta, axis=1)

def pair_search(h, c, i, j, n):
    if i > j:
        i, j = j, i
    x0 = float(h[i])
    y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-13:
        return None
    xs = pair_xs(lo, hi, x0)
    vals = pair_eval(h, c, i, j, xs, n, x0, y0, S)
    for _ in range(3):
        if len(xs) <= 1:
            break
        k = int(np.argmin(vals))
        if k == 0:
            a, b = xs[0], min(xs[1], hi)
        elif k == len(xs) - 1:
            a, b = max(xs[-2], lo), xs[-1]
        else:
            a, b = xs[k-1], xs[k+1]
        if b - a < 1e-11:
            break
        xs2 = np.linspace(a, b, 80)
        vals = pair_eval(h, c, i, j, xs2, n, x0, y0, S)
        xs = xs2
    k = int(np.argmin(vals))
    bv = float(vals[k])
    if bv < float(c.max()) - 1e-13:
        xb = float(np.clip(xs[k], lo, hi))
        yb = float(S - xb)
        if yb < 0.0:
            yb = 0.0; xb = S - yb
        elif yb > 1.0:
            yb = 1.0; xb = S - yb
        return xb, yb, bv
    return None

def pair_search_intensive(h, deadline, seed=1):
    n = len(h)
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    if n < 2 or time.time() > deadline:
        return besth, bestc
    rng = np.random.RandomState(seed)
    noimp = 0
    maxno = 8000
    off = n - 1
    top_lags = np.array([0])
    
    while time.time() < deadline:
        if noimp > maxno:
            break
        if rng.random() < 0.85:
            pos = c[off+1:]
            neg = c[off-1::-1]
            w = np.maximum(pos, neg) ** 2.5 + 1e-10
            w /= w.sum()
            d = int(rng.choice(n-1, p=w)) + 1
        elif rng.random() < 0.6:
            # Focus on top lags
            top_lags = np.argsort(c)[-5:][::-1]
            tk = int(top_lags[rng.randint(len(top_lags))]) - off
            if tk < 1 or tk >= n:
                d = rkint(1, n)
            else:
                d = abs(tk)
        else:
            d = int(rng.randint(1, n))
        i = int(rng.randint(n - d))
        j = i + d
        res = pair_search(h, c, i, j, n)
        if res is not None:
            h[i], h[j] = res[0], res[1]
            c = compute_c(h, n)
            v = float(c.max())
            if v < bestc:
                bestc = v
                besth = h.copy()
            noimp = 0
        else:
            noimp += 1
    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc

def rkint(a, b):
    return int(np.random.randint(a, b))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h0 = 0.5 + 0.49 * np.sin(3.0 * np.pi * x)
    
    h0 = project_box_sum(h0, n0)
    besth = h0.copy()
    bestc = float(compute_c(h0, n0).max())
    bestn = n0
    total = deadline - t0
    
    # Phase 1: L-BFGS-B warmup (25%)
    dl1 = t0 + total * 0.25
    work = h0.copy()
    for alpha, mu, maxit in ((5, 10, 120), (30, 100, 120), (200, 800, 100), (1000, 5000, 80)):
        if time.time() > dl1 - 3:
            break
        h2 = lbfgsb(work, n0, alpha, mu, maxit)
        h2 = project_box_sum(h2, n0)
        v2 = float(compute_c(h2, n0).max())
        if v2 < bestc:
            bestc = v2
            besth = h2.copy()
        if v2 <= float(compute_c(work, n0).max()) + 1e-7:
            work = h2
        else:
            work = besth.copy()
    
    # Phase 2: Intensive pair search (50%)
    dl2 = t0 + total * 0.75
    besth, bestc = pair_search_intensive(besth, dl2, seed)
    bestn = n0
    
    # Phase 3: Iterated local search with perturbation (15%)
    dl3 = t0 + total * 0.90
    n_restarts = 3
    per_restart = (dl3 - time.time()) / n_restarts
    for r in range(n_restarts):
        if time.time() > dl3 - 5:
            break
        hp = besth.copy()
        npert = max(5, n0 // 15)
        idx = rng.choice(n0, size=npert, replace=False)
        hp[idx] += 0.04 * rng.randn(npert)
        hp = project_box_sum(hp, n0)
        hr, cr = pair_search_intensive(hp, time.time() + per_restart, seed + 1000 + r * 7)
        if cr < bestc:
            bestc = cr
            besth = hr.copy()
    
    # Phase 4: SLSQP + final pair (10%)
    if time.time() < deadline - 12:
        h4 = slsqp(besth, bestn, 3000, 100)
        h4 = project_box_sum(h4, bestn)
        c4 = float(compute_c(h4, bestn).max())
        if c4 < bestc:
            bestc = c4
            besth = h4
    if time.time() < deadline - 2:
        besth, bestc = pair_search_intensive(besth, deadline, seed + 555)
    
    besth = project_box_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, bestn
