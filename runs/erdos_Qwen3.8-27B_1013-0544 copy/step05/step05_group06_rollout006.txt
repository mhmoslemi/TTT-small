    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_descent(h, n, deadline, K=12, pool_size=60):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    g = compute_g(h, n)
    F = compute_F(h, n)
    stalls = 0

    while time.time() < deadline:
        current_max = g.max()
        top_K = np.argsort(g)[::-1][:K]
        g_top = g[top_K]
        stalls_this_round = 0
        max_rounds = 500

        for rnd in range(max_rounds):
            if time.time() > deadline:
                break
            # Score-based pair selection
            scores = F[:, top_K].mean(axis=1)
            i_candidates = np.argsort(scores)[::-1][:pool_size]
            j_candidates = np.argsort(scores)[:pool_size]

            found_improvement = False
            for i in i_candidates:
                if time.time() > deadline:
                    break
                i = int(i)
                for j in j_candidates:
                    if time.time() > deadline:
                        break
                    j = int(j)
                    if i == j:
                        continue
                    s = F[i] - F[j]
                    d_lo = max(-h[i], h[j] - 1.0)
                    d_hi = min(1.0 - h[i], h[j])
                    if d_hi - d_lo < 1e-14:
                        continue
                    s_top = s[top_K]
                    candidates = [d_lo, d_hi]
                    for a in range(K):
                        for b in range(a + 1, K):
                            ds = s_top[b] - s_top[a]
                            if abs(ds) > 1e-15:
                                d_val = (g_top[a] - g_top[b]) / ds
                                if d_lo < d_val < d_hi:
                                    candidates.append(d_val)
                    best_d = 0.0
                    best_val = current_max
                    for d_val in candidates:
                        d_val = float(np.clip(d_val, d_lo, d_hi))
                        val = float(np.max(g + d_val * s))
                        if val < best_val - 1e-13:
                            best_val = val
                            best_d = d_val
                    if best_d != 0.0:
                        g += best_d * s
                        h[i] += best_d
                        h[j] -= best_d
                        update_F_fast(F, i, best_d, n)
                        update_F_fast(F, j, -best_d, n)
                        current_max = float(np.max(g))
                        found_improvement = True
                        break
                if found_improvement:
                    break

            if not found_improvement:
                stalls_this_round += 1
                if stalls_this_round > 3:
                    break

        # Project and check
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        g = compute_g(h, n)
        F = compute_F(h, n)
        if g.max() >= best_c - 1e-10:
            stalls += 1
            if stalls > 5:
                break
        else:
            stalls = 0

    return best_h, best_c

def sa_explore(h, n, deadline, steps=2000):
    """Simulated annealing to escape local optima"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    cur_c = compute_c5(cur_h, n)
    T = 0.001
    cooling = 0.999
    rng = np.random.RandomState(123)

    for _ in range(steps):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d_lo = max(-cur_h[i], cur_h[j] - 1.0)
        d_hi = min(1.0 - cur_h[i], cur_h[j])
        if d_hi - d_lo < 1e-10:
            continue
        d = rng.uniform(d_lo, d_hi)
        trial = cur_h.copy()
        trial[i] += d
        trial[j] -= d
        trial = np.clip(trial, 0, 1)
        trial = project_feasible(trial, n)
        trial_c = compute_c5(trial, n)
        delta = trial_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            cur_h = trial
            cur_c = trial_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        T *= cooling
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP continuation (20% budget)
    p1 = t0 + 0.15 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [200, 500, 2000, 10000, 50000, 200000, 1000000]:
        if time.time() > p1:
            break
        h = slsqp_optimize(h, n, alpha, maxiter=150)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Pair descent (55% budget)
    p2 = t0 + 0.70 * (budget_s - 5)
    h, c = pair_descent(best_h.copy(), n, p2, K=12, pool_size=60)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 3: SA exploration (10% budget)
    p3 = t0 + 0.80 * (budget_s - 5)
    if time.time() < p3:
        h, c = sa_explore(best_h.copy(), n, p3)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 4: More pair descent after SA (7% budget)
    p4 = t0 + 0.87 * (budget_s - 5)
    if time.time() < p4:
        h, c = pair_descent(best_h.copy(), n, p4, K=15, pool_size=40)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 5: Final SLSQP + pair descent (3% budget)
    if time.time() < dl - 30:
        h = slsqp_optimize(best_h.copy(), n, 5000000, maxiter=100)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    if time.time() < dl - 5:
        h, c = pair_descent(best_h.copy(), n, dl - 3, K=15, pool_size=30)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
