import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    g = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = g.max()
    sh = np.clip(g - m, -80.0 / alpha, 0.0)
    e = np.exp(sh * alpha)
    es = e.sum()
    if es < 1e-30:
        return m, np.zeros(n)
    w = e / es
    f = m + np.log(es) / alpha
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        wk = w[k]
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wk * dx * (1.0 - h[lag:])
            grad[lag:] -= wk * dx * h[:n - lag]
        else:
            to = -lag
            grad[to:] += wk * dx * (1.0 - h[:n - to])
            grad[:n - to] -= wk * dx * h[to:]
    return f, grad

def slsqp_opt(h0, n, alpha, deadline, maxiter=2000):
    def obj(x):
        f, _ = smooth_obj_grad(x, n, alpha)
        return f
    def jac(x):
        _, g = smooth_obj_grad(x, n, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def proj_grad(h0, n, alpha, deadline, max_iter=30000):
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        f, grad = smooth_obj_grad(best_h, n, alpha)
        # Project gradient onto the subspace sum=0 (to maintain constraint)
        grad = grad - grad.mean()
        ng = np.linalg.norm(grad)
        if ng < 1e-15:
            break
        grad /= ng
        found = False
        for fct in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            step = lr * fct
            ht = project_feasible(best_h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < f - 1e-13:
                best_h = ht
                if ct < best_c:
                    best_c = ct
                found = True
                break
        if not found:
            lr *= 0.8
            if lr < 1e-12:
                break
    return best_h, best_c

def bin_search(h0, n, deadline, max_sweeps=5000, seed=42):
    target = n // 2
    idx = np.argsort(h0)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0

    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    g = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = g.max()
    sh = np.clip(g - m, -80.0 / alpha, 0.0)
    e = np.exp(sh * alpha)
    es = e.sum()
    if es < 1e-30:
        return m, np.zeros(n)
    w = e / es
    f = m + np.log(es) / alpha
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        wk = w[k]
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wk * dx * (1.0 - h[lag:])
            grad[lag:] -= wk * dx * h[:n - lag]
        else:
            to = -lag
            grad[to:] += wk * dx * (1.0 - h[:n - to])
            grad[:n - to] -= wk * dx * h[to:]
    return f, grad

def slsqp_smooth(h0, n, alpha, deadline, maxiter=2000):
    def obj(x):
        f, _ = smooth_obj_grad(x, n, alpha)
        return f
    def jac(x):
        _, g = smooth_obj_grad(x, n, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def hinge_slsqp(h0, n, t_target, deadline, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        viol = np.maximum(0.0, g - t_target)
        return 0.5 * np.dot(viol, viol)
    def jac(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        viol = np.maximum(0.0, g - t_target)
        grad = np.zeros(n)
        for k in np.where(viol > 0)[0]:
            vk = viol[k]
            lag = k - (n - 1)
            if lag >= 0:
                grad[:n-lag] += vk * dx * (1.0 - x[lag:])
                grad[lag:] -= vk * dx * x[:n-lag]
            else:
                to = -lag
                grad[to:] += vk * dx * (1.0 - x[:n-to])
                grad[:n-to] -= vk * dx * x[to:]
        return grad
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def proj_grad(h0, n, alpha, deadline, max_iter=30000):
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        f, grad = smooth_obj_grad(best_h, n, alpha)
        grad = grad - grad.mean()
        ng = np.linalg.norm(grad)
        if ng < 1e-15:
            break
        grad /= ng
        found = False
        for fct in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            step = lr * fct
            ht = project_feasible(best_h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < f - 1e-13:
                best_h = ht
                if ct < best_c:
                    best_c = ct
                found = True
                break
        if not found:
            lr *= 0.8
