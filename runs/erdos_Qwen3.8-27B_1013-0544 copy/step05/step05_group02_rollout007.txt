def worst_lag_gradient(h, n, k_star):
    s = k_star - (n - 1)
    g = np.zeros(n)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    idx = np.arange(lo, hi + 1)
    g[idx] += (1.0 - h[idx + s])
    lo2 = max(0, lo + s)
    hi2 = min(n - 1, hi + s)
    idx2 = np.arange(lo2, hi2 + 1)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def worst_lag_gradient(h, n, k_star):
    s = k_star - (n - 1)
    g = np.zeros(n)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    idx = np.arange(lo, hi + 1)
    g[idx] += (1.0 - h[idx + s])
    lo2 = max(0, lo + s)
    hi2 = min(n - 1, hi + s)
    idx2 = np.arange(lo2, hi2 + 1)
    g[idx2] -= h[idx2 - s]
    return g

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    out = np.clip(h, 0.0, 1.0)
    # Final exact fix
    residual = target_sum - out.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            mask = out < 1.0
        else:
            mask = out > 0.0
        cnt = mask.sum()
        if cnt > 0:
            out[mask] += residual / cnt
            out = np.clip(out, 0.0, 1.0)
    return out

def _sum_con(x, n):

def slsqp_round(h, n, alpha, maxiter=1500):

        c = np.correlate(hx, 1.0 - hx, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
