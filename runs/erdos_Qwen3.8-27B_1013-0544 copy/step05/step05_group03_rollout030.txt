from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sh -= sh.max()
        return m + (sh.max() + np.log(np.exp(sh).sum())) / alpha
    def con_fun(x):
        return x.sum() - n/2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except:
        return h

def grad_c5(h, n):
    """Gradient of C5 (soft max approximation) w.r.t. h"""
    dx = 2.0 / n
    g = np.correlate(h, 1-h, mode='full') * dx
    m = g.max()
    # Soft max weights
    alpha = 50000
    sh = alpha * (g - m)
    sh -= sh.max()

def project_feasible(h, n):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        wsum = w.sum()
        w = w / wsum
        grad = np.zeros(n)
        t_vec = np.arange(-(n-1), n)
        for k in range(2*n-1):
            if w[k] < 1e-12:
                continue
            t = t_vec[k]
            it = np.arange(n) + t
            mask1 = (it >= 0) & (it < n)
            it2 = np.arange(n) - t
            mask2 = (it2 >= 0) & (it2 < n)
            grad[mask1] += w[k] * (1.0 - x[it[mask1]])
            grad[mask2] -= w[k] * x[it2[mask2]]
        return grad * (1.0 + 1.0/alpha)
    def con_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=obj_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def make_initial(n, seed):
    rng = np.random.RandomState(seed)
    i = np.arange(n)
    candidates = []
    candidates.append(0.5 + 0.35 * np.sin(2*np.pi*3*i/n + 0.5))
    candidates.append(0.5 + 0.25*np.sin(2*np.pi*2*i/n) + 0.2*np.sin(2*np.pi*5*i/n+1.2))
    candidates.append(0.5 + 0.3*np.sin(2*np.pi*4*i/n+0.3) + 0.15*np.cos(2*np.pi*7*i/n))
    candidates.append(rng.rand(n))
    h4 = np.zeros(n)
    for b in range(10):
        start = int(b * n / 10)
        end = int((b+1) * n / 10)
        if b % 3 != 1:
            h4[start:end] = 1.0
    candidates.append(h4)
    best_h = None
    best_c = 999.0
    for h in candidates:
        hp = project_feasible(h, n)
        c = compute_c5(hp, n)
        if c < best_c:
            best_c = c
            best_h = hp.copy()
    return best_h, best_c

def fast_binary_search(h, n, deadline, max_sweeps=300):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    corr = np.correlate(hb, 1 - hb, mode='full').astype(float)
    dx = 2.0 / n
    best_c = float(corr.max() * dx)
    cur_c = best_c
    if time.time() > deadline:
        return hb, best_c
    t_vec = np.arange(-(n-1), n)
    ones = np.where(hb == 1)[0].tolist()
    zeros = np.where(hb == 0)[0].tolist()
    rng = np.random.RandomState(777)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zero_order = zeros[:]
        rng.shuffle(zero_order)
        for i in zero_order:
            one_order = ones[:]
            rng.shuffle(one_order)
            found = False
            for j in one_order:
                it = i + t_vec
                m1 = (it >= 0) & (it < n)
                it2 = i - t_vec
                m2 = (it2 >= 0) & (it2 < n)
                jt = j + t_vec
                m3 = (jt >= 0) & (jt < n)
                jt2 = j - t_vec
                m4 = (jt2 >= 0) & (jt2 < n)
                delta = np.zeros(2*n-1)
                if m1.any():
                    delta[m1] += (1.0 - hb[it[m1]])
                if m2.any():
                    delta[m2] -= hb[it2[m2]]
                if m3.any():
                    delta[m3] -= (1.0 - hb[jt[m3]])
                if m4.any():
                    delta[m4] += hb[jt2[m4]]
                new_corr = corr + delta
                new_c = float(new_corr.max() * dx)
                if new_c < cur_c - 1e-15:
                    hb[i] = 1.0
                    hb[j] = 0.0
                    corr = new_corr
                    cur_c = new_c
                    if new_c < best_c:
                        best_c = new_c
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
                    improved = True
                    found = True
                    break
            if found:
                break
        if not improved:
            break
    return hb, best_c

def simulated_annealing(hb, n, deadline, iters=80000):
    target = n // 2
    dx = 2.0 / n
    best_h = hb.copy()
    best_c = float(np.correlate(hb, 1-hb, mode='full').max() * dx)
    cur_h = hb.copy()
    corr = np.correlate(cur_h, 1-cur_h, mode='full').astype(float)
    cur_c = float(corr.max() * dx)
    ones = np.where(cur_h == 1)[0].tolist()
    zeros = np.where(cur_h == 0)[0].tolist()
    rng = np.random.RandomState(999)
    t_vec = np.arange(-(n-1), n)
    T = 0.008
    T_min = 0.00005
    cooling = (T_min / T) ** (1.0 / max(iters, 1))
    for it in range(iters):
        if time.time() > deadline:
            break
        i = int(rng.randint(len(zeros)))
        j = int(rng.randint(len(ones)))
        ii = zeros[i]
        jj = ones[j]
        it_v = ii + t_vec
        m1 = (it_v >= 0) & (it_v < n)
        it2 = ii - t_vec
        m2 = (it2 >= 0) & (it2 < n)
        jt = jj + t_vec
        m3 = (jt >= 0) & (jt < n)
        jt2 = jj - t_vec
        m4 = (jt2 >= 0) & (jt2 < n)
        delta = np.zeros(2*n-1)
        if m1.any():
            delta[m1] += (1.0 - cur_h[it_v[m1]])
        if m2.any():
            delta[m2] -= cur_h[it2[m2]]
        if m3.any():
            delta[m3] -= (1.0 - cur_h[jt[m3]])
        if m4.any():
            delta[m4] += cur_h[jt2[m4]]
        new_corr = corr + delta
        new_c = float(new_corr.max() * dx)
        d = new_c - cur_c
        if d < 0 or rng.random() < np.exp(-d / T):
            cur_h[ii] = 1.0
            cur_h[jj] = 0.0
            corr = new_corr
            cur_c = new_c
            ones[j] = ii
            zeros[i] = jj
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        T *= cooling
    return best_h, best_c
