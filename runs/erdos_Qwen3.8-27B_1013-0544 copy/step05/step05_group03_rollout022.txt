from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def sa_binary(hb, n, deadline, rng, T0=0.005, cool=0.9997, max_iter=5000000):
    """Simulated annealing on binary solution."""
    best_hb = hb.copy()
    best_c = compute_c5(hb, n)
    cur_c = best_c
    T = T0
    ones = np.where(hb == 1)[0].tolist()
    zeros = np.where(hb == 0)[0].tolist()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Pick random swap
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        if i == j:
            continue
        # Apply swap
        hb[i], hb[j] = 1.0, 0.0
        new_c = compute_c5(hb, n)
        d = new_c - cur_c
        if d < 0 or rng.random() < np.exp(-d / T):
            # Accept
            cur_c = new_c
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)

                best_c = cur_c
                best_hb = hb.copy()
        else:
            # Reject
            hb[i], hb[j] = 0.0, 1.0
        T *= cool
        if T < 1e-8:
            T = 1e-8
    
    return best_hb, best_c
