            if tot < 1e-14:

    g = np.zeros(n)

    hi2 = min(n - 1, hi + s)

            i1, i2 = rng.choice(n, 2, replace=False)
            scale = 6.0 * T + 0.005
            d1 = min(scale, h[i1] * 0.5)
            d2 = min(scale, h[i2] * 0.5)
            if d1 < 1e-9 and d2 < 1e-9:
                T *= decay
                continue
            h[i1] -= d1
            h[i2] -= d2
        else:
            freq = rng.uniform(1.0, 6.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = 6.0 * T + 0.008
            pert = amp * np.sin(freq * x_grid + phase)
            pert -= pert.mean()
            h = h + pert
        
        h_new = project_feasible(h, n)
        new_c = compute_c5(h_new, n)
        
        if new_c <= curr_c:
            h = h_new
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            if rng.random() < np.exp(-(new_c - curr_c) / T):
                h = h_new
                curr_c = new_c
            stuck += 1
        
        if stuck > 5000:
            T = min(T * 3.0, 0.1)
            stuck = 0
        T *= decay
    
    return best_h, best_c

def directed_polish(h, n, deadline, max_stall=400):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    si = 0
    stall = 0
    while time.time() < deadline:
        c = get_corr(h, n)
        k_star = int(np.argmax(c))
        g = worst_lag_gradient(h, n, k_star)
        if g.max() < 1e-12 or -g.min() < 1e-12:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        d = steps[si]
        d = min(d, h[i_src] * 0.99, (1.0 - h[i_dst]) * 0.99)
        if d < 1e-11:
            si = min(si + 1, len(steps) - 1)
            stall += 1
            if stall > max_stall:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = compute_c5(h, n)
        if new_c < best_c - 1e-15:
            best_c = new_c
            best_h = h.copy()
            stall = 0
            si = max(0, si - 1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            si = min(si + 1, len(steps) - 1)
            if stall > max_stall:
                break
    return best_h, best_c

def multi_lag_polish(h, n, deadline, num_lags=4):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.003, 0.001, 0.0005, 0.0002, 0.0001]
    si = 0
    stall = 0
    while time.time() < deadline:
        c = get_corr(h, n)
        top_k = np.argsort(c)[-num_lags:]
        g = np.zeros(n)
        for k in top_k:
            g += worst_lag_gradient(h, n, int(k))
        if g.max() < 1e-12:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        d = steps[si]
        d = min(d, h[i_src] * 0.99, (1.0 - h[i_dst]) * 0.99)
        if d < 1e-11:
            si = min(si + 1, len(steps) - 1)
            stall += 1
            if stall > 200:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = compute_c5(h, n)
        if new_c < best_c - 1e-15:
            best_c = new_c
            best_h = h.copy()
            stall = 0
            si = max(0, si - 1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            si = min(si + 1, len(steps) - 1)
            if stall > 300:
                break
    return best_h, best_c

def optimize_from(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    for a in [200, 1000, 5000, 25000]:
        if time.time() > deadline - 15:
            break
        ht = slsqp_round(best_h, n, a, top_m=5, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
    
    if time.time() < deadline - 25:
        sa_budget = min(50, deadline - time.time() - 25)
        best_h, best_c = sa_optimize(best_h, n, time.time() + sa_budget, seed=777)
    
    if time.time() < deadline - 10:
        best_h, best_c = multi_lag_polish(best_h, n, min(deadline - 10, time.time() + 30), num_lags=4)
    
    if time.time() < deadline - 5:
        best_h, best_c = directed_polish(best_h, n, deadline - 3)
    
    if time.time() < deadline - 3:
        ht = slsqp_round(best_h, n, 100000, top_m=3, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250
    
    best_h = None
    best_c = float('inf')
    
    alloc_a = (deadline - t0) * 0.45
    h_opt, c_opt = optimize_from(h_init, n_init, t0 + alloc_a)
    best_h, best_c = h_opt, c_opt
    
    x_prev = np.linspace(0, 2, n_init)
    for n_try in [200, 250, 300, 350, 400]:
        if n_try == n_init or time.time() > deadline - 100:
            continue
        remaining = deadline - time.time()
        alloc = min(120, remaining * 0.18)
        if alloc < 60:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_prev, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c_opt = optimize_from(h_interp, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            x_prev = x_new
    
    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        remaining = deadline - time.time() - 25
        n_restarts = max(3, int(remaining / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.008 + 0.04 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(65, (deadline - time.time() - 25) * 0.35)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_from(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)
        best_h, best_c = sa_optimize(best_h, n, min(deadline - 8, time.time() + 30), seed=999)
    
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = directed_polish(best_h, n, deadline - 2)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    
    return best_h, best_c, len(best_h)
