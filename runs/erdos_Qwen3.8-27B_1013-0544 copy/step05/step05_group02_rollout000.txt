from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_round(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        h_x = np.clip(x, 0, 1)
        c = np.correlate(h_x, 1 - h_x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    g = np.zeros(n)

    hi = min(n - 1, n - 1 - s)

    idx2 = np.arange(lo2, hi2 + 1)

    return g

    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = h.copy()
    v = np.zeros(n)
    beta = 0.9  # momentum
    lr = 0.01
    t_iter = 0
    
    while time.time() < deadline and t_iter < max_iter:
        c = get_corr(x, n)
        curr_max = float(c.max())
        if curr_max < best_c - 1e-14:
            best_c = curr_max
            best_h = x.copy()
        
        # Find lags at (or near) the maximum
        thresh = curr_max - 0.001
        worst_lags = np.where(c >= thresh)[0]
        
        # Compute subgradient (average of worst lag gradients)
        g = np.zeros(n)
        for k in worst_lags:
            g += worst_lag_gradient(x, n, int(k))
        g /= max(1, len(worst_lags))
        
        # Nesterov step
        v = beta * v + (1 - beta) * g
        x_new = x - lr * v
        x_new = project_feasible(x_new, n)
        x = x_new

            g_norm = np.linalg.norm(g)
            if g_norm < 1e-6:
                lr *= 0.5
            elif g_norm > 1.0:
                lr *= 0.8
    
    return best_h, best_c

One more idea: instead of using `np.correlate` which computes all lags, I can compute just the gradient for the worst lag more efficiently. The gradient computation is O(n) for a single lag, much faster than O(n^2) for the full correlation.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_round(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        h_x = np.clip(x, 0, 1)
        c = np.correlate(h_x, 1 - h_x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient(h, n, k):
    s = k - (n - 1)
    if s == 0:
        return np.zeros(n)
    g = np.zeros(n)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    if lo > hi:
        return g
    idx = np.arange(lo, hi + 1)
    g[idx] += (1.0 - h[idx + s])
    lo2 = max(0, lo + s)
    hi2 = min(n - 1, hi + s)
    if lo2 <= hi2:
        idx2 = np.arange(lo2, hi2 + 1)
        g[idx2] -= h[idx2 - s]
    return g

def nesterov_multi(h, n, deadline, top_k=5, base_lr=0.02):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = h.copy()
    y = h.copy()
    t_nest = 1.0
    lr = base_lr

    for it in range(100000):
        if time.time() > deadline:
            break
        c = get_corr(y, n)
        curr_max = float(c.max())
        top_idx = np.argsort(c)[-top_k:]
        diffs = c[top_idx] - curr_max
        w = np.exp(diffs * 20.0)
        w /= w.sum()
        g = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            g += wi * lag_gradient(y, n, int(ki))
        g /= top_k
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_new
        x_new = x - lr * g
        x_new = project_feasible(x_new, n)
        y_new = x_new + beta * (x_new - x)
        y_new = project_feasible(y_new, n)
        x = x_new
        y = y_new
        t_nest = t_new
        if it % 100 == 0:
            cx = compute_c5(x, n)
            if cx < best_c - 1e-14:
                best_c = cx
                best_h = x.copy()
            gnm = np.linalg.norm(g)
            if gnm < 1e-7:
                lr *= 0.7
            elif gnm > 5.0:
                lr *= 0.8
        if it % 500 == 0:
            lr = max(0.001, min(0.05, base_lr / (1.0 + it * 1e-5)))

    cx = compute_c5(x, n)
    if cx < best_c:
        best_c = cx
        best_h = x.copy()
    return best_h, best_c

def pair_swap_polish(h, n, deadline, max_stall=200):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c = get_corr(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    si = 0
    stall = 0
    top_k = 3
    while time.time() < deadline:
        top_idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for ki in top_idx:
            g += lag_gradient(h, n, int(ki))
        g /= top_k
        if g.max() < 1e-11 or (-g.min()) < 1e-11:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            gidx = np.argsort(g)
            i_dst = int(gidx[0])
            i_src = int(gidx[-1])
            if i_src == i_dst:
                break
        d = steps[si]
        d = min(d, h[i_src] * 0.99, (1.0 - h[i_dst]) * 0.99)
        if d < 1e-11:
            si = min(si + 1, len(steps) - 1)
            stall += 1
            if stall > max_stall:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = get_corr(h, n)
        new_max = float(new_c.max())
        if new_max < best_c - 1e-15:
            best_c = new_max
            best_h = h.copy()
            c = new_c
            stall = 0
            si = max(0, si - 1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            si = min(si + 1, len(steps) - 1)
            if stall > max_stall:
                break
    return best_h, best_c

def optimize_phase(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alphas = [2000, 10000, 50000, 200000, 1000000, 5000000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        ht = slsqp_round(best_h, n, a, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = ht
    if time.time() < deadline - 5:
        best_h, best_c = nesterov_multi(best_h, n, min(deadline - 5, time.time() + 40), top_k=5)
    if time.time() < deadline - 3:
        best_h, best_c = pair_swap_polish(best_h, n, deadline - 2)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def structured_restart(base_h, n, rng, r):
    n = len(base_h)
    x = np.linspace(0, 2, n)
    mode = r % 4
    if mode == 0:
        amp = 0.01 + 0.05 * (r // 4)
        perturber = amp * np.sin(2.0 * np.pi * (1 + r // 4) * x)
    elif mode == 1:
        amp = 0.01 + 0.03 * (r // 4)
        centers = rng choice if False else rng.uniform(0.2, 1.8, 2)
        perturber = amp * np.exp(-((x - centers[0])**2) / 0.02) - amp * np.exp(-((x - centers[1])**2) / 0.02)
    elif mode == 2:
        amp = 0.005 + 0.02 * (r // 4)
        perturber = amp * (1.0 / (1.0 + np.exp(-10.0 * (x - 1.0))) - 0.5)
    else:
        amp = 0.003 + 0.02 * (r // 4)
        perturber = amp * rng.normal(0, 1, n)
    return project_feasible(base_h + perturber, n)
