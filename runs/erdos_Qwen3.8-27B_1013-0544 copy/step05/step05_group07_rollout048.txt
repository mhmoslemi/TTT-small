def coordinate_descent(h, n, deadline, step_sizes):
    """Iterate: for each pair (i,j), try to improve by transferring mass."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for step in step_sizes:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i+1, n):
                    # Try moving mass from i to j
                    for s in (1.0, -1.0):
                        hi = best_h[i] + s * step
                        hj = best_h[j] - s * step
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w = w / w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += w[idx] * (1.0 - h[:n - k])
            grad[:n - k] += w[idx] * (-h[k:])
        else:
            kk = -k
            grad[:n - kk] += w[idx] * (1.0 - h[kk:])
            grad[kk:] += w[idx] * (-h[:n - kk])
    grad *= dx
    return grad

def subgrad_at_max(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    m_star = int(np.argmax(c))
    k = m_star - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] += -h[k:]
    else:
        g[:n + k] += (1.0 - h[-k:])
        g[-k:] += -h[:n + k]
    g *= dx
    return g

def lbfgs_smooth(h, n, alpha, maxiter, deadline):
    target = n / 2.0
    dx = 2.0 / n

    def obj_lbfgs(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e6 * (x.sum() - target) ** 2
        return val

    def grad_lbfgs(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w = w / w.sum()
        g = np.zeros(n)
        for idx in range(2 * n - 1):
            if w[idx] < 1e-12:
                continue
            k = idx - (n - 1)
            if k >= 0:
                g[k:] += w[idx] * (1.0 - x[:n - k])
                g[:n - k] += w[idx] * (-x[k:])
            else:
                kk = -k
                g[:n - kk] += w[idx] * (1.0 - x[kk:])
                g[kk:] += w[idx] * (-x[:n - kk])
        g *= dx
        g += 2e6 * (x.sum() - target)
        return g

    try:
        res = minimize(obj_lbfgs, h, method='L-BFGS-B', jac=grad_lbfgs,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-11})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    target = n / 2.0

    def obj_slsqp(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def cons_slsqp(x):
        return x.sum() - target

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_slsqp}]
    try:
        res = minimize(obj_slsqp, h, method='SLSQP', bounds=bounds,

    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    temp = 0.008
    for it in range(max_iter):
        if time.time() > deadline:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h_bin[i], h_bin[j] = 1.0, 0.0
        c_new = compute_c5(h_bin, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_bin.copy()
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
            temp = max(temp * 0.999, 1e-7)
        else:
            delta = c_new - best_c
            if rng.random() < np.exp(-delta / max(temp, 1e-10)):
                ones.remove(j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(j)
            else:
                h_bin[i], h_bin[j] = 0.0, 1.0

def coordinate_descent(h, n, deadline, steps):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for step in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    for s in (1.0, -1.0):
                        hi = best_h[i] + s * step
                        hj = best_h[j] - s * step
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            best_h[i], best_h[j] = hi, hj
                            c = compute_c5(best_h, n)

                                best_c = c
                                improved = True
                            else:
                                best_h[i], best_h[j] = best_h[i] - s * step, best_h[j] + s * step
    return best_h, best_c

        h_new = lbfgs_smooth(best_h.copy(), n, 500000, 100, deadline - 5)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
