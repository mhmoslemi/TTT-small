    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_h = h.copy()
    
    delta_schedule = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    delta_idx = 0
    stall = 0
    max_stall = 2000
    
    while time.time() < deadline:

        j = rng.randint(n)

            continue

        for s in [1, -1]:

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_objective(x, n, alpha, penalty=1e4):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    s = shifted.max()
    logsum = (s + np.log(np.exp(shifted - s).sum())) / alpha
    val = m + logsum
    val += penalty * (x.sum() - n/2.0)**2
    return val

def smooth_grad(x, n, alpha, penalty=1e4):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m_idx = int(np.argmax(c))
    k = m_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - x[:n-k])
        g[:n-k] += (-x[k:])
    else:
        g[:n+k] += (1.0 - x[-k:])
        g[-k:] += (-x[:n+k])
    g *= dx
    g += 2 * penalty * (x.sum() - n/2.0)
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    s = sh.max()
    return m + (s + np.log(np.exp(sh - s).sum())) / alpha + 1e4 * (x.sum() - n/2.0)**2

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m_idx = int(np.argmax(c))
    k = m_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - x[:n-k])
        g[:n-k] += (-x[k:])
    else:
        g[:n+k] += (1.0 - x[-k:])
        g[-k:] += (-x[:n+k])
    g *= dx
    g += 2e4 * (x.sum() - n/2.0)
    return g

def lbfgs_smooth(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, jac=lambda x: smooth_grad(x, n, alpha),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def fast_sa(h, n, deadline, rng, t0_init=0.1, t_min=0.0001, cooling=0.99999):
    """Fast simulated annealing with O(1) per iteration."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_h = h.copy()
    current_c = best_c
    temp = t0_init
    n_accepted = 0
    n_rejected = 0
    
    while time.time() < deadline and temp > t_min:
        # Choose two random positions
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Choose a random delta
        d = rng.uniform(0.001, 0.1) * rng.choice([1.0, -1.0])
        
        hi = current_h[i] + d
        hj = current_h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            old_hi, old_hj = current_h[i], current_h[j]
            current_h[i], current_h[j] = hi, hj
            new_c = compute_c5(current_h, n)
            delta = new_c - current_c
            if delta < 0 or rng.random() < np.exp(-delta / temp):
                current_c = new_c
                n_accepted += 1
                if new_c < best_c:
                    best_c = new_c
                    best_h = current_h.copy()
            else:
                current_h[i], current_h[j] = old_hi, old_hj
                n_rejected += 1
        else:
            n_rejected += 1
        
        temp *= cooling
    
    return best_h, best_c

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    s = sh.max()
    return m + (s + np.log(np.exp(sh - s).sum())) / alpha + 1e4 * (x.sum() - n/2.0)**2

def smooth_grad(x, n, alpha):
