def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=6000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas, radius=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(600):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    j_lo, j_hi = i + 1, n
                else:
                    j_lo = max(0, i - radius)
                    j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                        continue
                    h[i], h[j] = hi, hj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved = True
                    else:
                        h[i], h[j] = hi - d, hj + d
                    hi = h[i] - d
                    hj = h[j] + d
                    if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                        continue
                    h[i], h[j] = hi, hj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved = True
                    else:
                        h[i], h[j] = hi + d, hj - d
            if not improved:
                break
    return h, best_c

def guided_search(h, n, deadline):
    """Guided local search: target the worst lag specifically."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(20000):
        if time.time() > deadline:
            break
        c_vec = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(c_vec)
        shift = k_max - (n - 1)
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            j = i + shift
            if 0 <= j < n and j != i:
                for d in (0.003, 0.001, 0.0005):
                    hi, hj = h[i] - d, h[j] + d
                    if 0 <= hi and hj <= 1:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i], h[j] = hi + d, hj - d
            j2 = i - shift
            if 0 <= j2 < n and j2 != i:
                for d in (0.003, 0.001, 0.0005):
                    hi, hj = h[i] + d, h[j2] - d
                    if 0 <= hi and hj <= 1:
                        h[i], h[j2] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i], h[j2] = hi - d, hj + d
        if not improved and it > 50:
            break
    return best_h, best_c

def optimize_block(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    alphas = [200, 1000, 5000, 25000, 100000, 500000, 2e6, 1e7, 5e7]
    for a in alphas:
        if time.time() > deadline - 8:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=5000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()

    if time.time() < deadline - 4:
        if n <= 200:
            best_h, best_c = pair_search(best_h, n, deadline - 2,
                                         [0.01, 0.005, 0.002, 0.001])
        else:
            best_h, best_c = pair_search(best_h, n, deadline - 2,
                                         [0.01, 0.005, 0.002], radius=50)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        alloc1 = (deadline - t0) * 0.30
        h_opt, c_opt = optimize_block(h_init, n_init, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt

        x_base = np.linspace(0, 2, n_init)
        for n_try in [150, 200, 100]:
            if n_try == n_init or time.time() > deadline - 45:
                continue
            alloc = min(80, (deadline - time.time()) * 0.22)
            if alloc < 20:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_base, best_h)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_block(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

        if best_h is not None and time.time() < deadline - 40:
            n = len(best_h)
            x_cur = np.linspace(0, 2, n)
            for n_try in [300, 250]:
                if n_try == n or time.time() > deadline - 35:
                    continue
                alloc = min(70, (deadline - time.time()) * 0.2)
                if alloc < 18:
                    break
                x_new = np.linspace(0, 2, n_try)
                h_interp = np.interp(x_new, x_cur, best_h)
                h_interp = project_feasible(h_interp, n_try)
                h_opt, c_opt = optimize_block(h_interp, n_try, time.time() + alloc, rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

        if best_h is not None and time.time() < deadline - 35:
            n = len(best_h)
            remaining = deadline - time.time() - 20
            n_restarts = max(2, int(remaining / 45))
            scales = [0.004, 0.01, 0.02, 0.04, 0.06]
            for r in range(n_restarts):
                if time.time() > deadline - 18:
                    break
                scale = scales[r % len(scales)]
                hp = best_h + rng.normal(0, scale, n)
                hp = project_feasible(hp, n)
                alloc = min(40, (deadline - time.time() - 18) * 0.3)
                if alloc < 12:
                    break
                h_opt, c_opt = optimize_block(hp, n, time.time() + alloc, rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # Guided search to target worst lag
        if time.time() < deadline - 8:
            gh, gc = guided_search(best_h, n, time.time() + min(15, (deadline - time.time()) * 0.3))
            if gc < best_c:
                best_c = gc
                best_h = gh

    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        if n <= 200:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        else:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.001, 0.0005, 0.0002, 0.0001], radius=70)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = 200
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
