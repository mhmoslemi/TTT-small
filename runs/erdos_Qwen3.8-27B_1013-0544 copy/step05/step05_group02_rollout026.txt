def fast_softmax_gradient(h, n, alpha):
    """Compute gradient of softmax-smoothed max correlation using FFT."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_shift = c - c.max()
    p = np.exp(alpha * c_shift)
    p /= p.sum()
    
    # Softmax weights for each shift
    # p[k] is the weight for correlation index k, shift s = k - (n-1)
    
    # Nonlinear part: (h * Q)[j] where Q[s] = p[n-1+s] + p[n-1-s]
    # Q is defined for s in [-(n-1), n-1]
    Q = np.zeros(2*n - 1)
    for s in range(-(n-1), n):
        k1 = n - 1 + s

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

f_p(h) = (sum_k (c_k * dx)^p)^(1/p) for p → ∞ gives the max.

The gradient is:
∇f_p = (1/p) * (sum_k (c_k dx)^p)^{-1/p + 1} * sum_k p * (c_k dx)^{p-1} * dx * ∇c_k
     = (sum_k (c_k dx)^{p})^{1/p - 1} * sum_k (c_k dx)^{p-1} * dx * ∇c_k
     = weighted average of ∇c_k with weights proportional to (c_k dx)^{p-1}

This is similar to the softmax but with polynomial weights instead of exponential. For large p, both concentrate on the maximum.

The advantage of power mean: the gradient is bounded (since c_k dx ≤ 1), so the step size doesn't need to be as small.
