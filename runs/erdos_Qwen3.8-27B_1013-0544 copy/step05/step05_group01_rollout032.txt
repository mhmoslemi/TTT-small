def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    return float(m + np.log(np.sum(np.exp(s - s.max()) + 1e-30)) / alpha)

                    if c_new < c_max - 1e-14:
                        h = h_new
                        if c_new < best_c:
                            best_c = c_new
                            best_h = h.copy()
                        improved = True
                        break
                    step_try *= 0.5
                if improved:
                    break
        if not improved:
            step *= 0.8
            if step < 1e-7:
                break
        else:
            step = min(0.02, step * 1.05)
    return best_h, best_c

def fine_pair_search(h, n, deadline, steps=None, radius=15):
    if steps is None:
        steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    best_c = compute_c5(h, n)
    for d in steps:
        if time.time() > deadline:
            break
        for _pass in range(50):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if j == i:
                        continue
                    for sign in (1, -1):
                        hi = h[i] + sign * d
                        hj = h[j] - sign * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = hi - sign * d, hj + sign * d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: SLSQP with increasing alpha
    alphas = [3000, 15000, 60000, 200000, 800000, 3000000]
    for a in alphas:
        if time.time() > deadline - 250:
            break
        h_new = slsqp_opt(best_h, n_init, a, maxiter=6000)
        c_new = compute_c5(h_new, n_init)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 2: Multi-active descent
    if time.time() < deadline - 120:
        h_new, c_new = multi_active_descent(best_h, n_init, deadline - 120)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 3: Subgradient method
    if time.time() < deadline - 60:
        h_new, c_new = subgradient_method(best_h, n_init, deadline - 60)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 4: SLSQP again on improved solution
    if time.time() < deadline - 50:
        for a in [500000, 2000000]:
            if time.time() > deadline - 30:
                break
            h_new = slsqp_opt(best_h, n_init, a, maxiter=4000)
            c_new = compute_c5(h_new, n_init)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 5: Fine local search
    if time.time() < deadline - 20:
        best_h, best_c = fine_pair_search(best_h, n_init, deadline - 5,
                                          steps=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                          radius=20)

    # Phase 6: Random restarts
    if time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 60))
        scales = [0.003, 0.008, 0.02, 0.04, 0.06]
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            c_try = compute_c5(hp, n)
            if c_try < best_c * 1.01:
                h_new = slsqp_opt(hp, n, 500000, maxiter=4000)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    if time.time() < deadline - 10:
                        best_h, best_c = fine_pair_search(best_h, n, deadline - 5,
                                                          steps=[0.0005, 0.0002, 0.0001, 0.00005],
                                                          radius=15)

    # Final refinement
    if time.time() < deadline - 2:
        n = len(best_h)
        best_h, best_c = fine_pair_search(best_h, n, deadline - 1,
                                          steps=[0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
                                          radius=10)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
