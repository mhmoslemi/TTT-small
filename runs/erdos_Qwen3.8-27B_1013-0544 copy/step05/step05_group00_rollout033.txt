    cur_c = corr.max() * dx
    m_range = np.arange(-(n - 1), n)

                corr = new_corr
                cur_c = nc
                ones.remove(j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(j)
                improved = True
        if not improved:

def make_delta(h, a, b, n, m_range):

    im = a + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = a - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = b + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = b - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + b - a] += 1.0
    delta[n - 1 + a - b] += 1.0
    return delta

def greedy_pass(h, corr, n, m_range, rng, n_sweeps=5, pairs_per_sweep=None):

    zeros = np.where(h == 0)[0].tolist()
    cur_max = corr.max()
    
    for sweep in range(n_sweeps):
        improved = False
        # Shuffle for diversity
        rng.shuffle(zeros)
        rng.shuffle(ones)
        n_pairs = len(zeros) * len(ones) if pairs_per_sweep is None else pairs_per_sweep
        tried = 0
        for i_idx in range(len(zeros)):
            if tried >= n_pairs:
                break
            i = zeros[i_idx]
            for j_idx in range(len(ones)):
                if tried >= n_pairs:

                j = ones[j_idx]
                tried += 1
                delta = make_delta(h, i, j, n, m_range)

                    h[i], h[j] = 1.0, 0.0
                    corr = corr + delta
                    cur_max = new_max
                    ones[j_idx] = i
                    zeros[i_idx] = j
                    improved = True
                    break
        if not improved:
