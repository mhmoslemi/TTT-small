def update_profiles_after_swap(profiles, n, pos_out, pos_in):
    """Update profiles after h[pos_out]:1->0 and h[pos_in]:0->1.
    v[pos_out] changes by +2, v[pos_in] changes by -2.
    profiles[p, k] is affected if k = n-1+p+pos_out or k = n-1+p+pos_in."""
    for p in range(n):
        ko = n - 1 + p + pos_out
        if 0 <= ko < 2*n - 1:

    profiles = build_profiles(hb, n)
    zeros = set(np.where(hb == 0)[0].tolist())
    ones = set(np.where(hb == 1)[0].tolist())
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if not zeros or not ones:
            break

        ones_arr = np.array(sorted(ones))
        C_max = C.max()
        M = C_max - C
        
        best_imp = 0.0
        best_i = -1
        best_j = -1
        
        for i in zeros_arr:
            threshold = profiles[i] - M
            valid = np.all(profiles[ones_arr] > threshold[None, :], axis=1)
            if not np.any(valid):
                continue
            vo = ones_arr[valid]
            new_maxes = (C[None, :] + profiles[i][None, :] - profiles[vo]).max(axis=1)

            if imps[bi] > best_imp:

                best_i = i
                best_j = vo[bi]
        
        if best_i < 0:
            break
        
        # Apply
        C += profiles[best_i] - profiles[best_j]
        update_profiles_after_swap(profiles, n, best_j, best_i)
        zeros.discard(best_i)
        ones.discard(best_j)
        zeros.add(best_j)
        ones.add(best_i)
    
    c_final = C.max() * (2.0 / n)
    return hb, c_final

        h0_200 = h0_200 * target_sum / s
    h0_200 = np.clip(h0_200, 0, 1)
    # Project to exact sum
    for _ in range(1000):
        diff = target_sum - h0_200.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h0_200
            rs = room.sum()
            if rs < 1e-14:
                break
            h0_200 += diff / rs * room
        else:
            ss = h0_200.sum()
            if ss < 1e-14:
                break
            h0_200 += diff / ss * h0_200
    h0_200 = np.clip(h0_200, 0, 1)
    hb200 = to_binary(h0_200, n2)
    try_solution(hb200, n2)
    hb200d, c200d = steepest_descent(hb200, n2, p5)
    try_solution(hb200d, n2)
    hb200i, c200i = ilds(hb200d, n2, p5 + 50, max_rounds=30, kick_size=2)
    try_solution(hb200i, n2)

    # Phase 6: Final SA polish on best (70s)
    p6 = t0 + 900
    if best_n == n:
        hbf, cf = sa_refine(best_h, n, p6, t_start=0.008, t_end=0.0002, max_iters=200000)
        try_solution(hbf, n)

    # Phase 7: Quick steepest descent polish (30s)
    p7 = dl
    if best_h is not None:
        hbpol, cpol = steepest_descent(best_h, best_n, p7, max_iters=500)
        try_solution(hbpol, best_n)

    if best_h is None:
        best_h = hb0.copy()
        best_c = compute_c5(best_h, n)
        best_n = n

    return best_h, best_c, best_n
