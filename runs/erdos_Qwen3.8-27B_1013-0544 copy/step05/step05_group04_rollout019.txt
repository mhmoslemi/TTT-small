import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(corr.max())

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            mask = h > 1e-15
            tot = h[mask].sum()
            if tot < 1e-15:
                break
            h[mask] += diff / tot
    return np.clip(h, 0.0, 1.0)

def solve_subproblem(h0, n, weights, maxiter=250):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        return float(np.dot(weights, c))

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

    best_c = compute_c5(h, n)
    best_h = h.copy()
    beta = beta_start
    stagnation = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = get_corr(h, n)
        c_max = corr.max()

        scaled = beta * (corr - c_max) / max(c_max, 1e-10)
        scaled = np.clip(scaled, -60.0, 0.0)
        exp_s = np.exp(scaled)
        w_sum = exp_s.sum()
        if w_sum < 1e-300:
            weights = np.ones(len(corr)) / len(corr)
        else:
            weights = exp_s / w_sum

        h_new = solve_subproblem(h, n, weights, maxiter=200)
        c_new = compute_c5(h_new, n)

        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation >= 5:
                beta = beta_start
                stagnation = 0

        beta *= beta_rate
        if beta > 200.0:
            beta = beta_start

    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    if radius is None:
        radius = n
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return project_feasible(h, n), best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 - 1e-12 and h[j] - d >= 1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d

                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return project_feasible(h, n), best_c
