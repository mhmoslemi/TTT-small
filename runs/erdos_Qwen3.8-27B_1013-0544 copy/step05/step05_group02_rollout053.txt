def lag_gradient(h, n, k_star):
    """Gradient of c[k_star] with respect to h."""
    # c[k] = sum_i h[i] * (1-h[i+k]) * dx for valid indices
    # In numpy correlate with mode='full', the lag indexing is:
    # corr[m] = sum_i h[i] * (1-h[i + (m - (n-1))]) * dx
    # where valid i means 0 <= i < n and 0 <= i + (m-(n-1)) < n
    # So for lag m, s = m - (n-1), and we need 0 <= i < n and 0 <= i+s < n
    # i.e., max(0,-s) <= i <= min(n-1, n-1-s)
    s = k_star - (n - 1)
    g = np.zeros(n)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    idx = np.arange(lo, hi + 1)
    # c = dx * sum_{i in [lo,hi]} h[i] * (1 - h[i+s])
    # dc/dh[i] for i in [lo,hi]: dx * (1 - h[i+s])
    # dc/dh[i+s] for i+s in [lo+s, hi+s]: dx * (-h[i])
    g[idx] += (1.0 - h[idx + s])
    lo2 = max(0, lo + s)
    hi2 = min(n - 1, hi + s)
    idx2 = np.arange(lo2, hi2 + 1)
    g[idx2] -= h[idx2 - s]
    return g

    g = np.zeros(n)
    for w, k in zip(weights, top_k):

            if tot < 1e-15:

            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

    return g

        i_src, i_dst = int(np.argmax(g_sm)), int(np.argmin(g_sm))
        if i_src != i_dst and h[i_src] > 1e-8 and (1 - h[i_dst]) > 1e-8:
            h_new, c_new = ternary_pair(h, n, i_src, i_dst, c_curr, deadline, iters_used)
            if c_new < c_curr - 1e-13:
                h, c_curr = h_new, c_new
                improved = True

        if improved:
            stuck = 0
            continue

        # Strategy 3: multi-lag pair
        top3 = np.argsort(c)[-3:]
        g_ml = np.zeros(n)
        for kk in top3:
            g_ml += lag_gradient(h, n, int(kk))
        i_src, i_dst = int(np.argmax(g_ml)), int(np.argmin(g_ml))
        if i_src != i_dst and h[i_src] > 1e-8 and (1 - h[i_dst]) > 1e-8:
            h_new, c_new = ternary_pair(h, n, i_src, i_dst, c_curr, deadline, iters_used)
            if c_new < c_curr - 1e-13:
                h, c_curr = h_new, c_new
                improved = True

        if improved:
            stuck = 0
            continue

        # Strategy 4: random pairs
        for _ in range(8):
            a, b = int(rng.randint(n)), int(rng.randint(n))
            if a == b or h[a] < 0.005 or h[b] > 0.995:
                continue
            h_new, c_new = ternary_pair(h, n, a, b, c_curr, deadline, 25)
            if c_new < c_curr - 1e-13:
                h, c_curr = h_new, c_new
                improved = True
                break

        if improved:
            stuck = 0
            continue

        # Strategy 5: perturbation + reoptimize
        stuck += 1
        scale = min(0.25, 0.005 * (1 + stuck * 0.5))
        h_pert = project_feasible(h + rng.normal(0, scale, n), n)
        c_pert, c_pert_full = compute_c5(h_pert, n)
        k_p = int(np.argmax(c_pert_full))
        g_p = lag_gradient(h_pert, n, k_p)
        h_new, c_new = ternary_pair(h_pert, n, int(np.argmax(g_p)), int(np.argmin(g_p)), c_pert, deadline, iters_used)
        if c_new < c_curr:
            h, c_curr = h_new, c_new
        elif c_pert < c_curr:
            h, c_curr = h_pert, c_pert
        else:
            h, c_curr = h_pert, c_pert
        stuck = 0
        iters_used = min(45, iters_used + 5)

    h = project_feasible(h, n)
    fc = compute_c5(h, n)[0]
    if fc < best_c:
        best_h, best_c = h.copy(), fc
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h, best_c = h_init.copy(), compute_c5(h_init, n_init)[0]

    # Phase 1: main optimization at parent resolution
    alloc1 = (deadline - t0) * 0.50
    h_opt, c_opt = optimize(h_init, n_init, t0 + alloc1)
    if c_opt < best_c:
        best_h, best_c = h_opt, c_opt

    # Phase 2: try nearby resolutions
    x_ref = np.linspace(0, 2, len(best_h))
    for n_try in [200, 300, 400]:
        if time.time() > deadline - 50:
            break
        alloc = min(130, (deadline - time.time()) * 0.25)
        if alloc < 25:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_ref, best_h), n_try)
        h_opt, c_opt = optimize(h_interp, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 3: random restarts from best
    if time.time() < deadline - 25:
        n = len(best_h)
        while time.time() < deadline - 20:
            scale = 0.02 + 0.1 * rng.random()
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(35, max(12, (deadline - time.time() - 20) * 0.4))
            if alloc < 10:
                break
            h_opt, c_opt = optimize(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

    # Phase 4: final polish
    if time.time() < deadline - 2:
        n = len(best_h)
        h_opt, c_opt = optimize(best_h, n, deadline - 1)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)[0]

    return best_h, best_c, len(best_h)
