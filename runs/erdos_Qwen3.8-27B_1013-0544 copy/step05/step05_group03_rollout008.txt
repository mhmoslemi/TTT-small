def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sh -= sh.max()
        return m + np.log(np.exp(sh).sum()) / alpha
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': 300, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    hb = to_binary(h0, n)
    best_c = compute_c5(hb, n)
    best_h = hb.copy()
    # Phase 1: quick SLSQP to get good continuous (8%)
    p1 = t0 + 0.08 * (budget_s - 15)
    hc = slsqp_quick(hb.copy(), n, p1, alpha=200000)
    hb_c = to_binary(hc, n)
    c_c = compute_c5(hb_c, n)
    if c_c < best_c:
        best_c = c_c
        best_h = hb_c.copy()
    # Phase 2: greedy on best binary (4%)
    p2 = t0 + 0.12 * (budget_s - 15)
    hb_g, c_g = greedy_binary(best_h.copy(), n, p2, rng, max_sweeps=20)
    if c_g < best_c:
        best_c = c_g
        best_h = hb_g.copy()
    # Phase 3: multi-restart SA (68%)
    p3 = t0 + 0.80 * (budget_s - 15)
    best_h, best_c = multi_restart_sa(best_h.copy(), n, p3, rng, n_restarts=7)
    # Phase 4: greedy polish (7%)
    p4 = t0 + 0.87 * (budget_s - 15)
    hb_p, c_p = greedy_binary(best_h.copy(), n, p4, rng, max_sweeps=25)
    if c_p < best_c:
        best_c = c_p
        best_h = hb_p.copy()
    # Phase 5: try alternative initializations (7%)
    p5 = dl - 5
    for trial in range(4):
        if time.time() > p5:
            break
        if trial == 0:
            hp = project_feasible(best_h + 0.15 * rng.randn(n), n)
        elif trial == 1:
            hp = project_feasible(np.roll(best_h, n // 4) + 0.3 * rng.randn(n), n)
        elif trial == 2:
            hp = project_feasible(1.0 - best_h + 0.2 * rng.randn(n), n)
        else:
            hp = project_feasible(0.5 + 0.3 * rng.randn(n), n)
        hb_a = to_binary(hp, n)
        c_a = compute_c5(hb_a, n)
        if c_a < best_c:
            c_a2, best_c = c_a, c_a
            best_h = hb_a.copy()
        hb_a, c_a = greedy_binary(hb_a, n, min(p5, time.time() + 40), rng, max_sweeps=15)
        if c_a < best_c:
            best_c = c_a
            best_h = hb_a.copy()
    # Final greedy
    if time.time() < dl - 3:
        hb_f, c_f = greedy_binary(best_h.copy(), n, dl - 2, rng, max_sweeps=20)
        if c_f < best_c:
            best_c = c_f
            best_h = hb_f.copy()
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
