import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_dG(best_h, n, i, j):
    """Compute the linearized gradient of G for changing h[i]+=δ, h[j]-=δ."""
    dG = np.zeros(2 * n - 1)
    v = 1.0 - best_h
    
    # h[i] += δ: contributes v[i + m - (n-1)] for valid m
    lo = max(0, n - 1 - i)
    hi = min(2 * n - 2, 2 * n - 2 - i)
    if lo <= hi:
        dG[lo:hi+1] += v[lo - (n-1-i) : hi - (n-1-i) + 1]
    
    # h[j] -= δ: contributes -v[j + m - (n-1)] for valid m
    lo = max(0, n - 1 - j)
    hi = min(2 * n - 2, 2 * n - 2 - j)
    if lo <= hi:
        dG[lo:hi+1] -= v[lo - (n-1-j) : hi - (n-1-j) + 1]
    
    # (1-h)[i] -= δ: contributes -h[i - (m-(n-1))] = -h[i-m+(n-1)]
    lo = max(0, i - (n-1))
    hi = min(2 * n - 2, i)
    if lo <= hi:
        dG[lo:hi+1] -= best_h[lo - i + (n-1) : hi - i + (n-1) + 1]

    lo = max(0, j - (n-1))

    if lo <= hi:

def find_opt_delta(G, dG, delta_lo, delta_hi):
    """Find δ that minimizes max_m (G[m] + dG[m]*δ) using binary search."""
    # Binary search on V
    lo_V = G.min()
    hi_V = G.max()
    
    for _ in range(50):
        if hi_V - lo_V < 1e-10:
            break
        mid_V = (lo_V + hi_V) / 2.0
        
        # Check: exists δ in [delta_lo, delta_hi] s.t. G[m] + dG[m]*δ <= mid_V for all m?
        d_lo = delta_lo
        d_hi = delta_hi
        
        pos = dG > 1e-14
        neg = dG < -1e-14
        
        if np.any(pos):
            ub = (mid_V - G[pos]) / dG[pos]
            d_hi = min(d_hi, ub.min())
        if np.any(neg):
            lb = (mid_V - G[neg]) / dG[neg]
            d_lo = max(d_lo, lb.max())
        if np.any(~pos & ~neg):
            if G[~pos & ~neg].max() > mid_V + 1e-12:
                lo_V = mid_V
                continue
        
        if d_lo <= d_hi + 1e-12:
            hi_V = mid_V
        else:
            lo_V = mid_V
    
    # Find δ at the optimal V
    target_V = hi_V
    d_lo = delta_lo
    d_hi = delta_hi
    pos = dG > 1e-14
    neg = dG < -1e-14

    if d_lo > d_hi:

def pair_descent(h, n, deadline, max_sweeps=200, top_k=50):
    """Pair coordinate descent."""
    best_h = h.copy()
    G = np.correlate(best_h, 1.0 - best_h, mode='full')  # unscaled
    best_c = G.max()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        
        # Select candidate indices
        high_idx = np.where(best_h > 0.4)[0]
        low_idx = np.where(best_h < 0.6)[0]
        
        if len(high_idx) > top_k:
            high_idx = high_idx[np.random.choice(len(high_idx), top_k, replace=False)]
        if len(low_idx) > top_k:
            low_idx = low_idx[np.random.choice(len(low_idx), top_k, replace=False)]
        
        for i in high_idx:
            if time.time() > deadline:
                break
            for j in low_idx:
                if abs(i - j) < 2:
                    continue
                dG = compute_dG(best_h, n, int(i), int(j))
                
                delta_lo = max(-best_h[i], best_h[j] - 1.0)
                delta_hi = min(1.0 - best_h[i], best_h[j])
                if delta_hi - delta_lo < 1e-10:
                    continue
                
                opt_delta = find_opt_delta(G, dG, delta_lo, delta_hi)
                
                # Evaluate true objective
                trial = best_h.copy()
                trial[int(i)] += opt_delta
                trial[int(j)] -= opt_delta
                trial = np.clip(trial, 0, 1)
                
                new_G = np.correlate(trial, 1.0 - trial, mode='full')
                new_max = new_G.max()
                
                if new_max < best_c - 1e-12:
                    best_h = trial
                    G = new_G
                    best_c = new_max
                    improved = True
        
        if not improved:
            break
    
    return best_h, best_c * (2.0 / n)

def nesterov_subgrad(h, n, deadline, max_iter=50000):
    """Nesterov-accelerated subgradient descent."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    y = best_h.copy()
    v = np.zeros(n)
    t = 1.0
    lr0 = 0.5
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        alpha = (t - 1.0) / t_new
        
        y_new = project_feasible(y + alpha * (best_h - y) - (lr0 / t_new) * v, n)

        g = compute_all_g(y_new, n) / dx
        m = g.max()
        k = int(np.argmax(g))
        lag = k - (n - 1)
        
        grad = np.zeros(n)
        if lag >= 0:
            valid_x = slice(max(0, lag), n)
