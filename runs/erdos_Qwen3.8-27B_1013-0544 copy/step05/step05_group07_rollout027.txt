def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    val = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    val += penalty * (x.sum() - n / 2.0) ** 2
    return val

        res = minimize(smooth_obj, h, jac=smooth_grad, method='L-BFGS-B',
                       bounds=bounds, args=(n, alpha, pen),
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxls': 30, 'maxfun': maxiter * 4})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_refine(h, n, deadline, rng):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.003
    mom = 0.9
    v = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode='full')
        m_idx = int(np.argmax(c))
        k = m_idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            ov = n - k
            g[:ov] += (1.0 - h[k:k+ov])
            g[k:k+ov] -= h[:ov]
        else:
            kk = -k
            ov = n - kk
            g[kk:kk+ov] += (1.0 - h[:ov])
            g[:ov] -= h[kk:kk+ov]
        g *= dx
        g -= g.sum() / n
        v = mom * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.003, 0.2)
        else:
            stall += 1
            lr *= 0.97
            v *= 0.5
            if stall > 400:
                h = project_feasible(best_h + 0.015 * rng.randn(n), n)
                v = np.zeros(n)
                lr = 0.002
                stall = 0
        if lr < 1e-10:
            break
    return best_h, best_c

def binary_search(h, n, deadline, rng, max_iter=80000):
    target = n // 2
    h_bin = np.zeros(n)
    h_bin[np.argsort(-h)[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    ones_arr = np.where(h_bin == 1)[0].copy()
    zeros_arr = np.where(h_bin == 0)[0].copy()
    temp = 0.002
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        i = zeros_arr[rng.randint(len(zeros_arr))]
        j = ones_arr[rng.randint(len(ones_arr))]
        h_bin[i], h_bin[j] = 1.0, 0.0
        c_new = compute_c5(h_bin, n)
        accept = c_new < best_c or rng.random() < np.exp(-(c_new - best_c) / temp)
        if accept:
            mask_j = ones_arr == j
            ones_arr[mask_j] = i
            mask_i = zeros_arr == i
            zeros_arr[mask_i] = j
            if c_new < best_c:
                best_c = c_new
                best_h = h_bin.copy()
                temp = max(temp * 0.998, 1e-7)
        else:
            h_bin[i], h_bin[j] = 0.0, 1.0
        temp = max(temp * 0.9999, 1e-7)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except (NameError, Exception):
        n = 250
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)

    # Multi-start: smooth and random patterns
    starts = [best_h]
    x = np.linspace(0, 2, n)
    for freq in [1, 2, 3]:
        hs = 0.5 + 0.3 * np.cos(np.pi * freq * x)
        hs = project_feasible(hs, n)
        starts.append(hs)
    for _ in range(3):
        idx = rng.permutation(n)
        hr = np.zeros(n)
        hr[idx[:n//2]] = 1.0
        starts.append(hr)
    for hs in starts:
        cs = compute_c5(hs, n)
        if cs < best_c:
            best_c, best_h = cs, hs.copy()

    # Phase 1: L-BFGS-B coarse to medium (45% time)
    p1 = t0 + 0.45 * (budget_s - 3)
    alphas1 = [500, 2000, 10000, 50000, 200000]
    for alpha in alphas1:
        if time.time() > p1 - 5:
            break
        pen = 5000.0
        h_new = run_lbfgs(best_h.copy(), n, alpha, pen, 600)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()

    # Phase 2: L-BFGS-B fine (20% time)
    p2 = t0 + 0.65 * (budget_s - 3)
    alphas2 = [500000, 2000000, 5000000]
    for alpha in alphas2:
        if time.time() > p2 - 5:
            break
        h_new = run_lbfgs(best_h.copy(), n, alpha, 8000.0, 500)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()

    # Phase 3: Subgradient + L-BFGS-B (20% time)
    p3 = t0 + 0.85 * (budget_s - 3)
    sg_time = min(120.0, p3 - time.time() - 30)
    if sg_time > 30:
        h_new, c_new = subgrad_refine(best_h.copy(), n, time.time() + sg_time, rng)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
    if time.time() < p3 - 10:
        h_new = run_lbfgs(best_h.copy(), n, 20000000, 10000.0, 400)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()

    # Phase 4: Binary search + final polish (15% time)
    hb, cb = binary_search(best_h.copy(), n, deadline - 25, rng, max_iter=60000)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    if time.time() < deadline - 10:
        h_new = run_lbfgs(best_h.copy(), n, 50000000, 10000.0, 300)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
    if time.time() < deadline - 5:
        h_new = run_lbfgs(best_h.copy(), n, 200000000, 20000.0, 200)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
