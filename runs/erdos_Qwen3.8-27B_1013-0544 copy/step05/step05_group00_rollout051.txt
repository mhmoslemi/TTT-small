    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def softmax_grad_descend(h, n, alpha, max_iters, eta0, decay, momentum, deadline):
    """Softmax-smoothed projected gradient descent with momentum."""
    dx = 2.0 / n

    best_c = compute_c5(h, n)
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        L = 1.0 - h
        c = np.correlate(h, L, mode='full') * dx
        cmax = c.max()
        ec = np.exp(np.clip(alpha * (c - cmax), -700, 700))
        p = ec / ec.sum()

            si = np.arange(s_lo, s_hi + 1)

            ia = j + si
            ib = j - si
            grad[j] = ps @ L[ia] - ps @ h[ib]
        grad *= dx

        v = momentum * v + (1 - momentum) * grad
        eta = eta0 / (1.0 + decay * it)

        c_now = compute_c5(h, n)
        if c_now < best_c:

            best = h.copy()

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h < 1.0 - 1e-15) if diff > 0 else (h > 1e-15)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_softmax(h, p):
    n = len(h)
    dx = 2.0 / n
    L = 1.0 - h
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(-(n-1), -j)
        s_hi = min(n-1, n-1-j)
        if s_lo <= s_hi:
            si = np.arange(s_lo, s_hi+1)
            ps = p[si + n - 1]
            A = ps @ L[j + si]
        else:
            A = 0.0
        b_lo = max(-(n-1), j - (n-1))
        b_hi = min(n-1, j)
        if b_lo <= b_hi:
            bi = np.arange(b_lo, b_hi+1)
            bp = p[bi + n - 1]
            B = bp @ h[j - bi]
        else:
            B = 0.0
        grad[j] = A - B
    return grad * dx

def softmax_pgd(h, n, alpha, max_iters, eta0, decay, mom, deadline):
    best = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        L = 1.0 - h
        c = np.correlate(h, L, mode='full')
        cmax = c.max()
        ec = np.exp(np.clip(alpha * (c - cmax), -600, 600))
        p = ec / ec.sum()
        g = grad_softmax(h, p)
        v = mom * v + (1.0 - mom) * g
        eta = eta0 / (1.0 + decay * it)
        h_new = project_feasible(h - eta * v, n)
        # Backtracking
        f_new = compute_c5(h_new, n)
        f_old = compute_c5(h, n)
        k = 0
        while f_new > f_old * 0.999 + 1e-12 and k < 20:
            eta *= 0.5
            h_new = project_feasible(h - eta * v, n)
            f_new = compute_c5(h_new, n)
            k += 1
        h = h_new
        if f_new < best_c:
            best_c = f_new
            best = h.copy()
    return best, best_c

        cur = (cur > 0.5).astype(float)
        target = n // 2
        oi = np.where(cur == 1)[0]

        corr = np.correlate(cur, 1.0 - cur, mode='full').copy()
        cur_c = corr.max() * dx

                jm = j + m_range
                jmn = j - m_range

                msk = (imn >= 0) & (imn < n)

                    delta[k2] += 1.0

                    corr += delta
                    cur_c = nc
                    improved = True
                    stall = 0

                    zeros = np.where(cur == 0)[0]

            if not improved:
                stall += 1
        if cur_c < best_c:
