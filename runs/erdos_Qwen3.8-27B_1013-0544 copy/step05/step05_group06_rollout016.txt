def slsqp_optimize(h, n, alpha, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n/2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except:
        return h

        best_i = -1
        best_j = -1

        for zi in range(len(zeros)):
            i = zeros[zi]
            for oj in range(len(ones)):
                j = ones[oj]
                s = F[i] - F[j]
                val = float(np.max(g + s))
                if val < best_swap_val - 1e-14:

                    best_i = i
                    best_j = j
        
        if best_i < 0:

        h_bin[best_j] = 0.0
        # Update sets

        zeros = zeros[zeros != best_i]
        ones = np.append(ones, best_i)
        zeros = np.append(zeros, best_j)
        
        c = compute_c5(h_bin, n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def compute_F_vec(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)
    ii = np.arange(n)[:, None]
    s_i = ii + dm[None, :]
    d_i = ii - dm[None, :]
    v1 = (s_i >= 0) & (s_i < n)
    v2 = (d_i >= 0) & (d_i < n)
    F = np.zeros((n, 2 * n - 1))
    F[v1] = 1.0 - h[np.clip(s_i, 0, n - 1)[v1]]
    F[v2] -= h[np.clip(d_i, 0, n - 1)[v2]]

    best_c = compute_c5(best_h, n)
    # SLSQP
    p1 = t0 + 0.25 * frac * (budget_s - 5)
    for alpha in [2000, 10000, 50000, 200000]:
        if time.time() > p1:
            break
        h = slsqp_opt(h, n, alpha, 150)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Stochastic pair descent
    p2 = t0 + 0.75 * frac * (budget_s - 5)
    psweep = min(300, n * (n - 1) // 2)
    h, c = stochastic_pair_descent(best_h.copy(), n, p2, 10, psweep, 3000, rng)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    # Full pair sweeps
    p3 = t0 + 0.9 * frac * (budget_s - 5)
    for _ in range(20):
        if time.time() > p3:
            break
        best_h = full_pair_sweep(best_h.copy(), n, p3, 10)
        c = compute_c5(best_h, n)
        if c < best_c:
            best_c = c
    # Binary
    if time.time() < dl - 20:
        hb, cb = binary_swap_search(best_h.copy(), n, dl - 10)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Final continuous polish
    if time.time() < dl - 10:
        h, c = stochastic_pair_descent(best_h.copy(), n, dl - 5, 12, 500, 1000, rng)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    # Try n0 (primary)
    h_r, c_r, nr = try_n(h0, n0, t0, budget_s, seed, 0.55)
    if c_r < best_c:
        best_c = c_r
        best_h = h_r
    # Try smaller n
    n_small = n0 - 50
    if n_small >= 100 and time.time() < t0 + 0.75 * budget_s:
        h_r2, c_r2, nr2 = try_n(h0, n_small, t0, budget_s, seed + 1, 0.20)
        if c_r2 < best_c:
            best_c = c_r2
            best_h = h_r2
    # Try larger n
    n_large = n0 + 50
    if time.time() < t0 + 0.85 * budget_s:
        h_r3, c_r3, nr3 = try_n(h0, n_large, t0, budget_s, seed + 2, 0.15)
        if c_r3 < best_c:
            best_c = c_r3
            best_h = h_r3
    # Final: restart + polish on best
    dl = t0 + budget_s - 5
    if time.time() < dl - 30:
        rng = np.random.RandomState(seed + 99)
        for r in range(5):
            if time.time() > dl - 10:
                break
            hp = best_h.copy()
            for _ in range(rng.randint(3, 8)):
                a, b = rng.choice(len(hp), 2, replace=False)
                d = min(hp[a], 1.0 - hp[b]) * rng.uniform(0.2, 1.0)
                hp[a] += d
                hp[b] -= d
            hp = project_feasible(hp, len(hp))
            h2, c2 = stochastic_pair_descent(hp, len(hp), dl - 10, 12, 400, 500, rng)
            if c2 < best_c:
                best_c = c2
                best_h = h2
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
