def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_raw_overlap(h):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h

def find_optimal_x(raw_overlap, h_current, a):

def coordinate_descent_pass(h, n, order, raw_overlap):
    """One pass of coordinate descent with exact line search per coordinate."""
    shifts = np.arange(-(n - 1), n)
    for i in order:
        H_d = np.zeros(2 * n - 1)
        vd = (i + shifts >= 0) & (i + shifts < n)
        H_d[vd] = h[i + shifts[vd]]
        H_md = np.zeros(2 * n - 1)
        vmd = (i - shifts >= 0) & (i - shifts < n)
        H_md[vmd] = h[i - shifts[vmd]]
        a = 1.0 - H_d - H_md

        if abs(x_star - h[i]) > 1e-15:
            h[i] = x_star
            raw_overlap = compute_raw_overlap(h)
    return h, raw_overlap
