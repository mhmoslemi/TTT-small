    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, n, alpha):
    """Compute LSE-smoothed objective and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    expo = alpha * (corr - max_c)
    S = np.exp(expo).sum()
    obj = max_c + np.log(S) / alpha
    w = np.exp(expo) / S
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-16:
            continue
        k = idx - (n - 1)
        i_lo = max(0, -k)
        i_hi = min(n - 1, n - 1 - k)
        if i_hi < i_lo:
            continue
        # First term: (1-h[i+k]) for i in [i_lo, i_hi]
        idxs = np.arange(i_lo, i_hi + 1)
        grad[idxs] += wk * (1.0 - h[idxs + k])
        # Second term: -h[i-k] for j=i-k in [i_lo-k, i_hi-k] ∩ [0, n-1]
        j_lo = max(0, i_lo - k)
        j_hi = min(n - 1, i_hi - k)
        if j_hi >= j_lo:
            grad[j_lo:j_hi+1] -= wk * h[j_lo+k:j_hi+k+1]
    
    return obj, grad

def proj_grad(h, n, alpha, max_iter=5000, deadline=None):
    """Projected gradient descent on LSE-smoothed objective."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 2.0 / (alpha * n)  # Initial step size
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        obj, grad = lse_obj_grad(h, n, alpha)
        # Projected gradient step
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        # Backtracking line search
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if c_new <= obj + 0.5 * step * np.sum(grad * (h_new - h)):
            h = h_new
            step = min(step * 1.5, 1.0)
        else:
            step *= 0.5
            if step < 1e-10:
                break
    return project_feasible(best_h, n), best_c

def pair_search(h, n, deadline, deltas):

    dx = 2.0 / n
    best_c = compute_c5(h, n)

def compute_corrs_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single_lag(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    i_lo = max(0, -k)
    i_hi = min(n - 1, n - 1 - k)
    if i_hi >= i_lo:
        idxs = np.arange(i_lo, i_hi + 1)
        grad[idxs] += (1.0 - h[idxs + k])
        j_lo = max(0, i_lo - k)
        j_hi = min(n - 1, i_hi - k)
        if j_hi >= j_lo:
            grad[j_lo:j_hi + 1] -= h[j_lo + k:j_hi + k + 1]
    return grad * dx

def lse_obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    expo = alpha * (corr - max_c)
    S = np.exp(expo).sum()
    obj = max_c + np.log(S) / alpha
    w = np.exp(expo) / S
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-16:
            continue
        k = idx - (n - 1)
        i_lo = max(0, -k)
        i_hi = min(n - 1, n - 1 - k)
        if i_hi < i_lo:
            continue
        idxs = np.arange(i_lo, i_hi + 1)
        grad[idxs] += wk * (1.0 - h[idxs + k])
        j_lo = max(0, i_lo - k)
        j_hi = min(n - 1, i_hi - k)
        if j_hi >= j_lo:
            grad[j_lo:j_hi + 1] -= wk * h[j_lo + k:j_hi + k + 1]
    return obj, grad

def proj_grad_optimize(h, n, alpha, deadline, max_iter=3000):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 4.0 / (alpha * n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        obj, grad = lse_obj_grad(h, n, alpha)
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if c_new <= obj + 0.5 * step * np.sum(grad * (h_new - h)) + 1e-14:
            h = h_new
            step = min(step * 1.3, 2.0 / (alpha * n) * 10)
        else:
            step *= 0.5
            if step < 1e-12:
                break
    return project_feasible(best_h, n), best_c

def directed_pair_search(h, n, deadline, deltas):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    corr = compute_corrs_full(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            k_star = int(np.argmax(corr)) - (n - 1)
            grad_k = grad_single_lag(h, n, k_star)
            active_mask = corr > best_c - 0.005
            if active_mask.sum() > 1:
                n_active = active_mask.sum()
                grad_multi = np.zeros(n)
                for idx in np.where(active_mask)[0]:
                    k = idx - (n - 1)
                    grad_multi += grad_single_lag(h, n, k) / n_active
                grad_use = 0.7 * grad_k + 0.3 * grad_multi
            else:
                grad_use = grad_k
            n_cand = min(40, n // 3)
            order = np.argsort(grad_use)
            inc_set = order[:n_cand]
            dec_set = order[-n_cand:][::-1]
            for i in inc_set:
                if time.time() > deadline:
                    return h, best_c
                for j in dec_set:
                    if i == j:
                        continue
                    if h[j] - d < -1e-12 or h[i] + d > 1.0 + 1e-12:
                        continue
                    if grad_use[j] - grad_use[i] <= 0:
                        continue
                    h[i] += d
                    h[j] -= d
                    new_corr = compute_corrs_full(h, n)
                    new_c = new_corr.max()
                    if new_c < best_c - 1e-14:
                        best_c = new_c
                        corr = new_corr
                        improved = True
                        k_star = int(np.argmax(corr)) - (n - 1)
                        grad_k = grad_single_lag(h, n, k_star)
                        active_mask = corr > best_c - 0.005
                        if active_mask.sum() > 1:
                            n_active = active_mask.sum()
                            grad_multi = np.zeros(n)
                            for idx2 in np.where(active_mask)[0]:
                                k2 = idx2 - (n - 1)
                                grad_multi += grad_single_lag(h, n, k2) / n_active
                            grad_use = 0.7 * grad_k + 0.3 * grad_multi
                        else:
                            grad_use = grad_k
                        order = np.argsort(grad_use)
                        inc_set = order[:n_cand]
                        dec_set = order[-n_cand:][::-1]
                    else:
                        h[i] -= d
                        h[j] += d
            if not improved:
                break
    return h, best_c

def pair_search_fine(h, n, deadline, deltas, radius=None):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is not None:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                else:
                    lo = i + 1
                    hi = n
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n = 250

    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Projected gradient with ramping alpha
    alphas = [500, 2000, 5000, 15000, 50000, 150000]
    for a in alphas:
        if time.time() > deadline - 150:
            break
        hd = project_feasible(h, n)
        h_opt, c_opt = proj_grad_optimize(hd, n, a, deadline=deadline - 130, max_iter=1500)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        h = h_opt

    # Phase 2: Directed pair search with moderate steps
    if time.time() < deadline - 80:
        h = project_feasible(h, n)
        h, c = directed_pair_search(h, n, deadline - 60, [0.02, 0.01, 0.005, 0.002])
        if c < best_c:
            best_c = c
            best_h = h

    # Phase 3: Fine directed pair search
    if time.time() < deadline - 30:
        h = project_feasible(best_h, n)
        h, c = directed_pair_search(h, n, deadline - 10, [0.001, 0.0005, 0.0002])
        if c < best_c:
            best_c = c
            best_h = h

    # Phase 4: Full fine pair search
    if time.time() < deadline - 8:
        h = project_feasible(best_h, n)
        h, c = pair_search_fine(h, n, deadline - 3, [0.001, 0.0005, 0.0002, 0.0001])
        if c < best_c:
            best_c = c
            best_h = h

    # Phase 5: Restart with perturbation if time remains
    if time.time() < deadline - 10:
        for _ in range(3):
            if time.time() > deadline - 8:
                break
            scale = 0.015
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            h_opt, c_opt = proj_grad_optimize(hp, n, 30000, deadline=deadline - 6, max_iter=800)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            if time.time() < deadline - 4:
                h2, c2 = directed_pair_search(h_opt, n, deadline - 2, [0.001, 0.0005, 0.0001])
                if c2 < best_c:
                    best_c = c2
                    best_h = h2

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
