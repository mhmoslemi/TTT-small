import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def binarize(h, n):
    """Convert to binary with exactly n/2 ones."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def fast_binary_local_search(h, n, deadline, max_sweeps=10000, seed=42):
    """Fast local search on binary h."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    ones = np.where(best_h == 1)[0]
    zeros = np.where(best_h == 0)[0]
    rng = np.random.RandomState(seed)
    
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        rng.shuffle(zeros)
        rng.shuffle(ones)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                best_h[i] = 1.0
                best_h[j] = 0.0
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    # Update ones and zeros
                    ones = np.where(best_h == 1)[0]
                    zeros = np.where(best_h == 0)[0]
                    improved = True
                    break
                else:
                    best_h[i] = 0.0
                    best_h[j] = 1.0
            if improved:
                break
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0
    return best_h, best_c

def iterate_local_search(h, n, deadline, n_iter=50, perturb_size=3, seed=100):
    """Iterated local search."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(seed)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Perturb
        cur_h = best_h.copy()
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        for _ in range(perturb_size):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
        
        # Local search from perturbed state
        cur_h2, cur_c = fast_binary_local_search(cur_h, n, deadline, max_sweeps=5000, seed=seed+it)
        if cur_c < best_c:
            best_h = cur_h2.copy()
            best_c = cur_c
    
    return best_h, best_c

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h, n):

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]

    target = n / 2.0

    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:

            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

Another idea: **alternating minimization**. Fix all but one variable, optimize that variable, sweep through all variables. This is like coordinate descent.

For a fixed set of all variables except h[i], the objective is:
C₅ = max_k corr[k] where corr[k] is linear in h[i] (for each k).

So for a given i, C₅ as a function of h[i] is the max of linear functions, which is convex. We can minimize it by finding the point where the maximum of the relevant lines is minimized (this is a 1D convex optimization).

The relevant lags for position i are those k where position i appears in the sum for corr[k]. That's all k (as I showed earlier). But the coefficient of h[i] in corr[k] is:
- (1-h[i+k]) if i+k is valid (from the first index term)
- -h[i-k] if i-k is valid (from the second index term, since (1-h[i]) = 1-h[i])

So corr[k] = a_k * h[i] + b_k where:
a_k = (1-h[i+k]) - h[i-k] (sum of applicable terms)
b_k = corr[k] - a_k * h[i] (the part not involving h[i])

C₅(h[i]) = max_k (a_k * h[i] + b_k)

This is a piecewise linear convex function of h[i] ∈ [0,1]. We can find its minimum in O(n) time (by finding the breakpoints).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def thue_morse_binary(n):
    t = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    # Ensure exactly n/2 ones
    n_ones = int(t.sum())
    target = n // 2
    if n_ones > target:
        idx = np.where(t == 1)[0]
        t[idx[:n_ones-target]] = 0
    elif n_ones < target:
        idx = np.where(t == 0)[0]
        t[idx[:target-n_ones]] = 1
    return t

def local_search(h, n, deadline, max_sweeps=50000, seed=0):
    """Local search: try all improving swaps, repeat until convergence."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        zeros = zeros[rng.permutation(len(zeros))]
        for i in zeros:
            if time.time() > deadline:
                break
            ones_tmp = ones[rng.permutation(len(ones))]
            for j in ones_tmp:
                best_h[i] = 1.0
                best_h[j] = 0.0
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    break
                best_h[i] = 0.0
                best_h[j] = 1.0
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def iterated_ls(h, n, deadline, n_iters=100, perturb=3, seed=0):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(seed)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        cur = best_h.copy()
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        for _ in range(perturb):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            cur[i], cur[j] = 1.0, 0.0
        cur, c = local_search(cur, n, deadline, max_sweeps=5000, seed=seed+it)
        if c < best_c - 1e-15:
            best_h = cur
            best_c = c
    return best_h, best_c
