        if T < 1e-8:
            break

        m = g.max()

        sh -= sh.max()  # numerical stability
        w = np.exp(sh)
        w /= w.sum()

        # ∇g_k[j] = dx * ((1-h[j-k]) if valid - h[j+k] if valid)
        grad = np.zeros(n)

            wk = w[k_idx]
            if wk < 1e-15:
                continue

                grad[:n-s] -= wk * dx * y[s:]

                # j-s = j+ss in [0,n-1] means j in [0, n-1-ss]
                # j+s = j-ss in [0,n-1] means j in [ss, n-1]
                grad[:n-ss] += wk * dx * (1-y[ss:])

        y = x_new + ((t_old - 1) / t) * (x_new - x)
