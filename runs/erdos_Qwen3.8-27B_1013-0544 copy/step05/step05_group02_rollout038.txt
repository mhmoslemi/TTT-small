    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n, max_iter=5000):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_softmax(h, n, alpha, maxiter=5000):
    dx = 2.0 / n
    def obj(x):
        h_x = np.clip(x, 0, 1)
        c = np.correlate(h_x, 1 - h_x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})

def directed_polish(h, n, deadline, top_k=3):

    best_h = h.copy()
    base_step = 0.02
    stall = 0

    try:
        res = minimize(obj, h_sub, method='SLSQP', bounds=bounds, constraints=cons,
                      jac=obj_grad, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = h_fixed.copy()
        h_new[lo:hi] = np.clip(res.x, 0, 1)
        return project_feasible(h_new, n)
    except Exception:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n, max_iter=5000):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_softmax(h, n, alpha, maxiter=5000):
    dx = 2.0 / n
    def obj(x):
        h_x = np.clip(x, 0, 1)
        c = np.correlate(h_x, 1 - h_x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_lag(h, n, k_star, maxiter=3000):
    dx = 2.0 / n
    s = k_star - (n - 1)
    def obj(x):
        h_x = np.clip(x, 0, 1)
        if s >= 0:
            return float(np.sum(h_x[s:] * (1.0 - h_x[:n-s])) * dx)
        else:
            sp = -s
            return float(np.sum(h_x[:n-sp] * (1.0 - h_x[sp:])) * dx)
    def obj_grad(x):
        h_x = np.clip(x, 0, 1)
        g = np.zeros(n)
        if s >= 0:
            g[s:] += (1.0 - h_x[:n-s]) * dx
            g[:n-s] -= h_x[s:] * dx
        else:
            sp = -s
            g[:n-sp] += (1.0 - h_x[sp:]) * dx
            g[sp:] -= h_x[:n-sp] * dx
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      jac=obj_grad, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_grad_vec(h, n, k):
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[s:] += (1.0 - h[:n-s])
        g[:n-s] -= h[s:]
    else:
        sp = -s
        g[:n-sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n-sp]
    return g

def local_subsys(h, n, lo, hi, k_star, maxiter=1000):
    h_fixed = h.copy()
    h_sub = h[lo:hi].copy()
    target_sub = n / 2.0 - (h_fixed.sum() - h_sub.sum())
    s = k_star - (n - 1)
    dx = 2.0 / n
    def obj(x):
        h_full = h_fixed.copy()
        h_full[lo:hi] = np.clip(x, 0, 1)
        if s >= 0:
            return float(np.sum(h_full[s:] * (1.0 - h_full[:n-s])) * dx)
        else:
            sp = -s
            return float(np.sum(h_full[:n-sp] * (1.0 - h_full[sp:])) * dx)
    def obj_grad(x):
        h_full = h_fixed.copy()
        h_full[lo:hi] = np.clip(x, 0, 1)
        g_full = np.zeros(n)
        if s >= 0:
            g_full[s:] += (1.0 - h_full[:n-s]) * dx
            g_full[:n-s] -= h_full[s:] * dx
        else:
            sp = -s
            g_full[:n-sp] += (1.0 - h_full[sp:]) * dx
            g_full[sp:] -= h_full[:n-sp] * dx
        return g_full[lo:hi]
    bounds = [(0.0, 1.0)] * (hi - lo)
    m_size = hi - lo
    def sub_sum(x): return x.sum() - target_sub
    def sub_sum_jac(x): return np.ones(m_size)
    cons = [{'type': 'eq', 'fun': sub_sum, 'jac': sub_sum_jac}]
    try:
        res = minimize(obj, h_sub, method='SLSQP', bounds=bounds, constraints=cons,
                      jac=obj_grad, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = h_fixed.copy()
        h_new[lo:hi] = np.clip(res.x, 0, 1)
        return project_feasible(h_new, n)
    except Exception:
        return h.copy()

def cycle_lags(h, n, deadline, top_k=5, max_rounds=30):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = get_corr(best_h, n)
        top_idx = np.argsort(c)[-top_k:]
        improved = False
        for k in sorted(top_idx, key=lambda x: -c[x]):
            if time.time() > deadline:
                break
            ht = slsqp_lag(best_h, n, int(k), maxiter=2000)
            nc = compute_c5(ht, n)
            if nc < best_c - 1e-14:
                best_c = nc
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c

def subsystem_polish(h, n, deadline, block_size=15, n_blocks=8):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c = get_corr(best_h, n)
    k_star = int(np.argmax(c))
    g = lag_grad_vec(best_h, n, k_star)
    centers = np.argsort(np.abs(g))[-n_blocks:]
    rng = np.random.RandomState(77)
    order = rng.permutation(centers)
    for ci in order:
        if time.time() > deadline:
            break
        center = int(centers[ci])
        lo = max(0, center - block_size // 2)
        hi = min(n, lo + block_size)
        lo = max(0, hi - block_size)
        c_now = get_corr(best_h, n)
        k_now = int(np.argmax(c_now))
        ht = local_subsys(best_h, n, lo, hi, k_now, maxiter=1500)
        nc = compute_c5(ht, n)
        if nc < best_c - 1e-14:
            best_c = nc
            best_h = ht
    return best_h, best_c

def directed_polish(h, n, deadline, top_k=3):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    base_step = 0.015
    stall = 0
    while time.time() < deadline:
        c = get_corr(h, n)
        curr = float(c.max())
        if curr < best_c - 1e-15:
            best_c = curr
            best_h = h.copy()
        top_idx = np.argsort(c)[-top_k:]
        w = np.exp(20.0 * (c[top_idx] - c[top_idx].max()))
        w /= w.sum()
        g = np.zeros(n)
        for i, k in enumerate(top_idx):
            g += w[i] * lag_grad_vec(h, n, int(k))
        g -= g.mean()
        gm = np.abs(g).max()
        if gm < 1e-13:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        d = min(base_step, h[i_src] * 0.8, (1.0 - h[i_dst]) * 0.8)
        if d < 1e-13:
            break
        h[i_src] -= d
        h[i_dst] += d
        new_c = compute_c5(h, n)
        if new_c < best_c - 1e-14:
            stall = 0
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            base_step *= 0.85
            if stall > 600:
                break
    return best_h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alphas = [100, 500, 2000, 10000, 50000, 200000]
    for a in alphas:

    if time.time() < deadline - 40:

    if time.time() < deadline - 25:

    if time.time() < deadline - 15:
        best_h, best_c = directed_polish(best_h, n, min(deadline - 15, time.time() + 45), top_k=3)
    if time.time() < deadline - 5:

        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    if time.time() < deadline - 2:
        best_h, best_c = directed_polish(best_h, n, deadline - 1, top_k=1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
