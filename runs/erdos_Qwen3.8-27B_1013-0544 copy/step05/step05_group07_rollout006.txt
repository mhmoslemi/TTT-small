        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lag_grad_arr(h, n, lags):
    dx = 2.0 / n
    g = np.zeros(n)
    for lag in lags:
        if lag >= 0:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] += -h[lag:]
        else:
            g[:n + lag] += (1.0 - h[-lag:])
            g[-lag:] += -h[:n + lag]
    return g * dx

def subgradient_topk(h, n, deadline, rng, top_k=2, max_iter=80000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.005
    v = np.zeros(n)
    mom = 0.88
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        c = all_c5(h, n)
        top_idx = np.argsort(c)[::-1][:top_k]
        lags = (top_idx - (n - 1)).astype(int)
        grad = lag_grad_arr(h, n, lags) / top_k
        v = mom * v - lr * grad
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.015, 0.05)
        else:
            h = project_feasible(h - 0.4 * v, n)
            v *= 0.4
            lr *= 0.96
            stall += 1
            if stall > 150:
                h = project_feasible(best_h + 0.03 * rng.randn(n), n)
                v = np.zeros(n)
                lr = 0.003
                stall = 0
        if lr < 1e-9:
            break
    return best_h, best_c

def smooth_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    val += 5e4 * (x.sum() - n / 2.0) ** 2
    w = np.exp(sh - sm)
    w /= w.sum()
    lags = (np.arange(2 * n - 1) - (n - 1)).astype(int)
    g = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-12:
            continue
        lag = lags[i]
        if lag >= 0:
            g[lag:] += wi * (1.0 - x[:n - lag])
            g[:n - lag] += -wi * x[lag:]
        else:
            g[:n + lag] += wi * (1.0 - x[-lag:])
            g[-lag:] += -wi * x[:n + lag]
    g *= dx
    g += 1e5 * (x.sum() - n / 2.0)
    return val, g

def lbfgs_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 5e4 * (x.sum() - n / 2.0) ** 2
        return val
    def grad(x):
        _, g = smooth_obj_and_grad(x, n, alpha)
        return g
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': 5000})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def binary_sa(h, n, deadline, rng, max_iter=100000):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    temp = 0.003
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h_bin[i], h_bin[j] = 1.0, 0.0
        c_new = compute_c5(h_bin, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_bin.copy()
            ones.remove(j)
            ones.append(i)
            zeros = [x for x in range(n) if h_bin[x] == 0]
            temp = max(temp * 0.999, 1e-8)
        else:
            delta = c_new - best_c
            if rng.random() < np.exp(-delta / max(temp, 1e-10)):
                ones.remove(j)
                ones.append(i)
                zeros = [x for x in range(n) if h_bin[x] == 0]
            else:
                h_bin[i], h_bin[j] = 0.0, 1.0
        temp *= 0.9998
        if temp < 1e-9:
            temp = 1e-9
    return best_h, best_c

def coord_descent(h, n, deadline, rng, max_rounds=80000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        pool = [j for j in range(n) if j != i]
        j = pool[rng.randint(len(pool))]
        improved = False
        for s in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001,
                  -0.04, -0.02, -0.01, -0.005, -0.002, -0.001]:
            hi, hj = best_h[i] + s, best_h[j] - s
            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                test = best_h.copy()
                test[i] = hi
                test[j] = hj
                c = compute_c5(test, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = test.copy()
                    improved = True
                    break
        if not improved and rng.random() < 0.005:
            s = rng.uniform(-0.15, 0.15)
            hi, hj = best_h[i] + s, best_h[j] - s
            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                test = best_h.copy()
                test[i] = hi
                test[j] = hj
                c = compute_c5(test, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = test.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Subgradient top-k (25%)
    p1 = t0 + 0.25 * (budget_s - 3)
    for k in [1, 2, 3, 5]:
        if time.time() > p1 - 5:
            break
        hn, cn = subgradient_topk(best_h.copy(), n, p1, rng, top_k=k, max_iter=40000)
        if cn < best_c:
            best_c, best_h = cn, hn

    # Phase 2: L-BFGS polishing with increasing alpha (35%)
    p2 = t0 + 0.60 * (budget_s - 3)
    alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    for a in alphas:
        if time.time() > p2 - 15:
            break
        hn = lbfgs_smooth(best_h.copy(), n, a, 800, p2)
        cn = compute_c5(hn, n)
        if cn < best_c:
            best_c, best_h = cn, hn

    # Phase 3: Binary SA (15%)
    p3 = t0 + 0.75 * (budget_s - 3)
    hb, cb = binary_sa(best_h.copy(), n, p3, rng, max_iter=80000)
    if cb < best_c:
        best_c, best_h = cb, hb

    # Phase 4: Coordinate descent (15%)
    p4 = t0 + 0.90 * (budget_s - 3)
    hn, cn = coord_descent(best_h.copy(), n, p4, rng, max_rounds=60000)
    if cn < best_c:
        best_c, best_h = cn, hn

    # Phase 5: Final polish (10%)
    if time.time() < deadline - 10:
        for a in [5000000, 20000000, 100000000]:
            if time.time() > deadline - 5:
                break
            hn = lbfgs_smooth(best_h.copy(), n, a, 500, deadline - 2)
            cn = compute_c5(hn, n)
            if cn < best_c:
                best_c, best_h = cn, hn
        hn, cn = coord_descent(best_h.copy(), n, deadline - 1, rng, max_rounds=30000)
        if cn < best_c:
            best_c, best_h = cn, hn

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
