def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_gradients(h, n, shifts):

        b_ub = np.zeros(n_rows)
        row = 0
        for i, s in enumerate(active_shifts):
            g = compute_grad_single(h, n, s, dx)
            A_ub[row, :n] = g
            A_ub[row, n] = -1.0
            b_ub[row] = -active_c[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = h[i]
            row += 1
            A_ub[row, i] = 1.0
            b_ub[row] = 1.0 - h[i]
            row += 1
        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        res = linprog(c_obj, A_ub=A_ub[:row, :], b_ub=b_ub[:row],
                      A_eq=A_eq, b_eq=b_eq, method='highs')
        if not res.success:
            # Fallback: subgradient step
            g, _, _ = subgradient_step(h, n, eps, max_active)
            for alpha in [0.5, 0.2, 0.1, 0.05]:
                h_try = project_feasible(h - alpha * g, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-12:
                    best_c = c_try
                    best_h = h_try.copy()
                    h = h_try
                    no_imp = 0
                    break
            else:
                no_imp += 1
                if no_imp > 30:
                    break
                h = project_feasible(h + 0.003 * rng.randn(n), n)
                no_imp = 0
            continue
        d = res.x[:n]
        improved = False
        for damp in [1.0, 0.7, 0.5, 0.3, 0.1]:
            h_try = np.clip(h + damp * d, 0.0, 1.0)
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-12:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try
                improved = True
                no_imp = 0
                eps = max(eps * 0.9, 1e-5)
                break
        if not improved:
            no_imp += 1
            if no_imp > 20:
                eps = min(eps * 1.5, 0.03)
                h = project_feasible(h + 0.004 * rng.randn(n), n)
                no_imp = 0
            elif no_imp > 5:
                eps = max(eps * 0.7, 1e-5)
    return best_h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
            if not improved:
                break
    return h, best_c

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    for a in [2000, 10000, 50000]:
        if time.time() > deadline - 20:
            break
        ht = slsqp_smooth(h, n, a, maxiter=800)
        c = compute_c5(ht, n)
        if c < best_c - 1e-12:
            best_c = c
            h = ht
    remaining = deadline - time.time() - 10
    if remaining > 20:
        h, best_c = cut_plane_loop(h, n, time.time() + remaining, rng)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100
    best_h = None
    best_c = float('inf')
    # Phase 1: Optimize initial construction
    alloc1 = min(250, (deadline - t0) * 0.25)
    h1, c1 = optimize_n(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h1, c1
    # Phase 2: Try other n values
    x_src = np.linspace(0, 2, len(best_h))
    for n_try in [100, 150, 200, 300]:
        if time.time() > deadline - 150:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_src, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(150, (deadline - time.time()) * 0.2)
        if alloc < 40:
            break
        ht, ct = optimize_n(h_interp, n_try, time.time() + alloc, rng)
        if ct < best_c:
            best_c = ct
            best_h = ht
    # Phase 3: Restart with perturbation on best
    n_best = len(best_h)
    if time.time() < deadline - 120:
        remaining = deadline - time.time() - 60
        n_restarts = max(2, int(remaining / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            scale = 0.05 + 0.05 * r
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(70, (deadline - time.time() - 30) * 0.3)
            if alloc < 20:
                break
            ht, ct = optimize_n(hp, n_best, time.time() + alloc, rng)
            if ct < best_c:
                best_c = ct
                best_h = ht
    # Phase 4: Pair search
    n_b = len(best_h)
    if time.time() < deadline - 15:
        if n_b <= 120:
            best_h, best_c = pair_search(best_h, n_b, deadline - 5,
                                         [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        else:
            best_h, best_c = near_pair_search(best_h, n_b, deadline - 5,
                                              [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                                              radius=60)
    # Phase 5: Final cutting plane + pair
    if time.time() < deadline - 8:
        ht, ct = cut_plane_loop(best_h, len(best_h), deadline - 3, rng, max_active=30)
        if ct < best_c:
            best_c = ct
            best_h = ht
    if time.time() < deadline - 3:
        n_b = len(best_h)
        if n_b <= 150:
            best_h, best_c = pair_search(best_h, n_b, deadline - 1,
                                         [0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        else:
            best_h, best_c = near_pair_search(best_h, n_b, deadline - 1,
                                              [0.001, 0.0005, 0.0002, 0.0001],
                                              radius=40)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
