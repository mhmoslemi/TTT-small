def correlation_gradient(h, n, s):
    """Compute d/dh[i] of c[k] where s = k - (n-1)."""
    grad = np.ones(n)
    # Subtract h[i+s] for valid i
    if s >= 0:
        grad[:n-s] -= h[s:]
    else:
        grad[-s:] -= h[:n+s]
    # Subtract h[i-s] for valid i
    if s >= 0:
        grad[s:] -= h[:n-s]
    else:
        grad[:n+s] -= h[-s:]
    return grad

        if alloc < 35:

        x_new = np.linspace(0, 2, nt)
        x_ref = np.linspace(0, 2, ref_n)
        hi = np.interp(x_new, x_ref, ref_h)
        hi = project_feasible(hi, nt)
        h_opt, c_opt = full_optimize(hi, nt, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            ref_h = best_h.copy()
            ref_n = nt
    # Phase 3: diverse restarts
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        n_rest = max(3, int((deadline - time.time() - 20) / 50))
        for r in range(n_rest):
            if time.time() > deadline - 25:
                break
            scale = 0.003 + 0.05 * (r / max(1, n_rest - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 25) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = full_optimize(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
    # Phase 4: final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = block_polish(best_h, n, min(deadline - 8, time.time() + 20), rng)
        best_h, best_c = coord_polish(best_h, n, deadline - 3)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(250, 0.5), 250)
        best_c = compute_c5(best_h, 250)
    return best_h, best_c, len(best_h)
