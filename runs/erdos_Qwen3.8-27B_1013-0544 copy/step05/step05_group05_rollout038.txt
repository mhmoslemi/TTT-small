def compute_corr(h):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

    m = g.max()
    sh = g - m
    return (m + np.log(np.exp(sh - sh.max()).sum()) / alpha) * (2.0 / n)

def thue_morse_binary(n):
    """Generate first n bits of Thue-Morse sequence."""
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    # Convert to binary h with n/2 ones
    # Thue-Morse has roughly n/2 ones, but not exactly
    # Adjust by swapping
    ones_needed = n // 2
    num_ones = t.sum()
    if num_ones > ones_needed:
        # Remove some ones (preferentially at the end)
        one_positions = np.where(t == 1)[0]
        for pos in one_positions[ones_needed:]:
            t[pos] = 0
    elif num_ones < ones_needed:
        zero_positions = np.where(t == 0)[0]
        for pos in zero_positions[:ones_needed - num_ones]:
            t[pos] = 1
    return t.astype(float)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

            if s < 1e-14:
                break

But the `guided_binary_search` is much faster per pass since it only tries moves that target the active constraint. The tradeoff: guided search might miss moves that don't directly target the active shift but still improve the overall max. But in practice, the first improvement (reducing the active shift) is usually the best.
