import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def c5_gradient_weighted(h, n, p):
    """Gradient of sum_m p[m] * c[m] * dx where c = correlate(h, 1-h, 'full')"""
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(2*n - 1):
        if p[m] < 1e-10: continue
        k = m - (n-1)
        if k >= 0:
            g[:n-k] += p[m] * (1.0 - h[k:])
            g[k:] += p[m] * (-h[:n-k])
        else:
            ka = -k
            g[ka:] += p[m] * (1.0 - h[:n-ka])
            g[:n-ka] += p[m] * (-h[ka:])
    g *= dx
    return g

def lbfgs_softmax(h, n, alpha, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e4 * (x.sum() - n/2.0)**2
        return val
    def grad(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        probs = np.exp(sh - sm)
        probs /= probs.sum()
        g = c5_gradient_weighted(x, n, probs)
        g += 2e4 * (x.sum() - n/2.0)
        return g
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0,1.0)]*n,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except:
        return h

                        improved = True

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def c5_grad_weighted(h, n, weights):
    """Gradient of sum_m weights[m] * c[m] * dx"""
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += w * (1.0 - h[k:])
            g[k:] += w * (-h[:n - k])
        else:
            ka = -k
            g[ka:] += w * (1.0 - h[:n - ka])
            g[:n - ka] += w * (-h[ka:])
    g *= dx
    return g

def lbfgs_run(h, n, alpha, maxiter, topk=None):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e4 * (x.sum() - n / 2.0) ** 2
        return val
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        e = np.exp(sh - sm)
        probs = e / e.sum()
        if topk is not None and topk < len(probs):
            # Keep only top-k probabilities
            idx = np.argsort(probs)[::-1][:topk]
            probs2 = np.zeros_like(probs)
            probs2[idx] = probs[idx]
            probs = probs2 / probs2.sum()
        g = c5_grad_weighted(x, n, probs)
        g += 2e4 * (x.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
        return project_feasible(res.x, n)
    except Exception:
        return h

    """Coordinate descent trying one coordinate at a time"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]

                    best_c = c
                    best_h = h.copy()
                else:
                    h[i] = hi - delta
                    h = project_feasible(h, n)
    return best_h, best_c

def binary_ls(h, n, deadline, rng):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    best_c = compute_c5(hb, n)

    ones = list(np.where(hb == 1)[0])

            ones.append(i)
            zeros[i] = zeros[-1]
            zeros.pop()
            zeros.append(j)
        else:

def continuous_sa(h, n, deadline, rng, T0=0.01, T1=0.0001):
    """Simulated annealing on continuous values"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = T0
    h_cur = h.copy()
    c_cur = best_c
    n_steps = 50000
    for t in range(n_steps):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (t / n_steps)
        # Pick 2 random coordinates and swap a bit
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        amount = rng.uniform(-0.05, 0.05)
        hi = h_cur[i] + amount
        hj = h_cur[j] - amount
        if 0 <= hi <= 1 and 0 <= hj <= 1:
            h_cur[i], h_cur[j] = hi, hj
            c_new = compute_c5(h_cur, n)
            if c_new < c_cur or rng.random() < np.exp(-(c_new - c_cur) / T):
                c_cur = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_cur.copy()
            else:
                h_cur[i], h_cur[j] = h_cur[i] - amount, h_cur[j] + amount
    return best_h, best_c

    def grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        e = np.exp(sh - sm)
        probs = e / e.sum()
        if topk is not None and topk < len(probs):
            idx = np.argsort(probs)[::-1][:topk]
            p2 = np.zeros_like(probs)
            p2[idx] = probs[idx]
            probs = p2 / p2.sum()
        g = c5_grad_weighted(x, n, probs)
        g += 2e4 * (x.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for step in steps:
        if time.time() > deadline:
            break
        no_improve = 0
        while time.time() < deadline and no_improve < 3 * n:
            no_improve = 0
            for i in rng.permutation(n):
                if time.time() > deadline:
                    break
                for sign in (1, -1):
                    delta = sign * step
                    hi = h[i] + delta
                    if hi < 1e-10 or hi > 1 - 1e-10:
                        continue
                    j = rng.randint(n)
                    if j == i:
                        continue
                    hj = h[j] - delta
                    if hj < 1e-10 or hj > 1 - 1e-10:
                        continue
                    h[i], h[j] = hi, hj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_h = h.copy()
                        no_improve = 0
                    else:
                        h[i], h[j] = hi - delta, hj + delta
                        no_improve += 1
    return best_h, best_c

def binary_ls(h, n, deadline, rng):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    best_c = compute_c5(hb, n)
    best_hb = hb.copy()
    ones = list(np.where(hb == 1)[0])
    zeros = list(np.where(hb == 0)[0])
    for _ in range(200000):
        if time.time() > deadline:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        hb[i], hb[j] = 1.0, 0.0
        c = compute_c5(hb, n)
        if c < best_c:
            best_c = c
            best_hb = hb.copy()
            ones[j] = ones[-1]; ones.pop(); ones.append(i)
            zeros[i] = zeros[-1]; zeros.pop(); zeros.append(j)
        else:
            hb[i], hb[j] = 0.0, 1.0
    return best_hb, best_c
