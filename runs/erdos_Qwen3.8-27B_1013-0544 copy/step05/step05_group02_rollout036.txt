def active_set_slsqp(h, n, deadline, max_active=20):
    """Active set method: minimize max of top K lags using SLSQP."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        c = get_corr(best_h, n)
        curr_max = float(c.max())

        threshold = curr_max - 0.005
        active = np.where(c >= threshold)[0]
        if len(active) > max_active:
            active = np.argsort(c)[-max_active:]

            h_x = np.clip(x, 0, 1)
            c_x = get_corr(h_x, n)
            return float(c_x[active].max())

            h_x = np.clip(x, 0, 1)
            c_x = get_corr(h_x, n)
            max_val = c_x[active].max()

            g = np.zeros(n)

                g += lag_gradient(h_x, n, int(active[idx]))
            g /= len(active_max)
            return g

        cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]

                best_h = new_h
            else:
                break
        except Exception:
            break
    
    return best_h, best_c

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        h_x = np.clip(x, 0, 1)
        c = np.correlate(h_x, 1 - h_x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient(h, n, k_star):
    s = k_star - (n - 1)
    g = np.zeros(n)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    idx = np.arange(lo, hi + 1)
    g[idx] += (1.0 - h[idx + s])
    lo2 = max(0, lo + s)
    hi2 = min(n - 1, hi + s)
    idx2 = np.arange(lo2, hi2 + 1)
    g[idx2] -= h[idx2 - s]
    return g

            if tot < 1e-15:
