```python
import cvxpy as cp

def outer_approximation_step(h, n, max_iter=100):
    h_curr = h.copy()
    for it in range(max_iter):
        # Compute all lags and gradients
        c = compute_all_lags(h_curr, n)
        dx = 2.0 / n
        
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [cp.sum(h_var) == n/2, h_var >= 0, h_var <= 1]
        
        for k in range(2*n - 1):
            # Compute gradient of f_k at h_curr
            # f_k = c[k] (before scaling, c[k] is the raw sum)
            # Actually c[k] = sum_i a[i] * v[i] where a = h, v = 1-h
            # For mode='full', index k corresponds to lag (k - (n-1))
            lag = k - (n - 1)
            
            # Compute linearization
            if lag >= 0:
                # f = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
                # grad[j] = (1-h[j+lag]) if j <= n-1-lag, plus (-h[j-lag]) if j >= lag
                grad = np.zeros(n)
                for j in range(n - lag):
                    grad[j] += (1 - h_curr[j + lag])
                for j in range(lag, n):
                    grad[j] -= h_curr[j - lag]
            else:
                # lag < 0, let lag2 = -lag
                lag2 = -lag
                # f = sum_{i=lag2}^{n-1} h[i]*(1-h[i-lag2])
                # Same as f for positive lag2 by symmetry of correlate
                # Actually correlate(h, 1-h, 'full')[n-1-lag2] = sum_i h[i]*(1-h[i-lag2]) for i>=lag2
                # This is the same as sum_i h[i+lag2]*(1-h[i]) for i=0..n-1-lag2
                # Which is f_{lag2} with h and (1-h) swapped... not quite.
                # Let me just compute it directly.
                grad = np.zeros(n)
                for j in range(lag2, n):
                    # h[j] is first factor: gradient = (1-h[j-lag2])
                    grad[j] += (1 - h_curr[j - lag2])
                    # h[j-lag2] is in second factor: -h[j-lag2], so gradient w.r.t. h[j-lag2] is -1*h[j]
                for j in range(n - lag2):
                    # h[j] is second factor (as 1-h[j]), gradient = -h[j+lag2]
                    grad[j] -= h_curr[j + lag2]
            
            f_val = c[k]  # already includes dx
            # Linear constraint: f_val + dx * grad . (h_var - h_curr) <= t_var
            constraints.append(f_val + dx * (grad @ (h_var - h_curr)) <= t_var)
        
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.GLOP)  # or CPLEX, GUROBI
        
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        
        h_new = h_var.value
        if h_new is None:
            break
        
        h_new = project_feasible(h_new, n)
        
        if np.max(np.abs(h_new - h_curr)) < 1e-8:
            h_curr = h_new
            break
        h_curr = h_new
    
    return h_curr
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _active_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _compute_lag_gradient(h_curr, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n or m == 0:
        if m == 0:
            grad[:] = 1.0 - 2.0 * h_curr
        return grad
    if lag > 0:
        grad[:n - m] -= h_curr[m:]
        grad[m:] += (1.0 - h_curr[:n - m])
    else:
        grad[:n - m] += (1.0 - h_curr[m:])
        grad[m:] -= h_curr[:n - m]
    return grad

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _compute_lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=100):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.15
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-8:
            break
        h_curr = h_new
        trust = max(0.005, trust * 0.88)
    return best_h, best_c

def active_set_optimize(h, n, deadline, top_m=5, max_rounds=8):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    for rnd in range(max_rounds):
        if time.time() > deadline - 10:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        for alpha in [5000, 50000]:
            if time.time() > deadline - 10:
                break
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
            try:
                res = minimize(_active_smooth_obj, h_curr,
                              args=(n, top_idx, alpha),
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 3000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    best_h = h_new.copy()
                    h_curr = h_new
            except Exception:
                pass
    return best_h, best_c

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j]:
                    delta = min(d, (h[i] - h[j]) * 0.5, h[i], 1.0 - h[j])
                    if delta < 1e-8:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i]:
                    delta = min(d, (h[j] - h[i]) * 0.5, h[j], 1.0 - h[i])
                    if delta < 1e-8:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if time.time() < deadline - 30:
        oo_end = time.time() + min(80, (deadline - time.time()) * 0.35)
        h_oo, c_oo = outer_approx(best_h, n, oo_end)
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo
    if time.time() < deadline - 15:
        for alpha in [10000, 100000]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_optimize(best_h, n, alpha, maxiter=3000, ftol=1e-15)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    if time.time() < deadline - 20:
        as_end = time.time() + min(50, (deadline - time.time()) * 0.4)
        h_as, c_as = active_set_optimize(best_h, n, as_end)
        if c_as < best_c:
            best_c = c_as
            best_h = h_as
    if time.time() < deadline - 10:
        h_new = slsqp_optimize(best_h, n, 500000, maxiter=4000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 3:
        best_h, best_c = local_search(best_h, n, min(deadline - 1, time.time() + 25))
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc = min(400, (deadline - t0) * 0.3)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    n_candidates = [100, 150, 200, 300, 400]
    for n_try in n_candidates:
        if time.time() > deadline - 70:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 40
        if remaining < 50:
            break
        alloc = min(90, remaining * 0.25)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.015 * (1.0 + 0.4 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 25) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_optimize(best_h, n, 1000000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h, best_c = local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
```
