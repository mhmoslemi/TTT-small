- At each iteration, find h* = argmin Q(h, p)

def minimax_iterate(h_init, n, alpha_inner, max_outer, qp_iters, lr_qp, deadline):
    """Iterative minimax algorithm."""
    dx = 2.0 / n
    n_full = 2 * n - 1
    h = h_init.copy()
    p = np.ones(n_full) / n_full
    
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        
        # Step 1: Build B matrix and solve QP
        # c_k(h) = dx * sum_i h[i] * (1-h[i+shift_k]) where shift_k = k-(n-1)
        # sum_k p_k * c_k(h) = dx * [sum_i h[i] - sum_{i,k} p_k * h[i] * h[i+shift_k]]
        # = dx * [sum(h) - h^T B h]
        # where B[i,j] = sum_k p_k * dx * 1_{j = i+shift_k} = sum_k p_k * dx * 1_{j-i = k-(n-1)}
        # = dx * p[(j-i)+(n-1)] if |j-i| <= n-1
        
        # Build B (it's Toeplitz)
        B = np.zeros((n, n))
        for diff in range(-(n-1), n):
            k = diff + n - 1
            val = p[k] * dx
            if val < 1e-15:
                continue
            for i in range(n):
                j = i + diff
                if 0 <= j < n:
                    B[i, j] = val
        
        BT = B + B.T  # For gradient: grad of -h^T B h is -(B+B^T)@h
        
        # Solve QP: min sum(h) - h^T B h s.t. 0<=h<=1, sum(h)=n/2
        h_qp = np.full(n, 0.5)
        for it in range(qp_iters):
            grad = np.ones(n) - BT @ h_qp  # gradient of sum(h) - h^T B h
            h_qp = project_feasible(h_qp - lr_qp * grad, n)
        
        h = h_qp
        
        # Step 2: Update p
        corr = np.correlate(h, 1-h, mode='full') * dx
        p = np.exp(alpha_inner * (corr - corr.max()))
        p /= p.sum()
    
    return h, compute_c5(h, n)
