    for p in range(max_passes):

            break

            c = np.correlate(h, 1 - h, mode='full') * dx
            m = c.shape[0]  # 2n-1
            # c[m] corresponds to lag l = m - (n-1)
            # Center offset
            center = n - 1
            
            # b(l) = (1 - h[j-l] - h[j+l]) * dx
            b = np.zeros(m)
            for idx in range(m):
                l = idx - center  # lag
                # h[j-l]: valid if 0 <= j-l < n
                val1 = h[j - l] if 0 <= j - l < n else 0.0
                # h[j+l]: valid if 0 <= j+l < n
                val2 = h[j + l] if 0 <= j + l < n else 0.0
                b[idx] = (1.0 - val1 - val2) * dx
            
            # a(l) = c(l) - b(l) * h[j]
            a = c - b * h[j]

            # This is a 1D convex optimization (min of max of linear functions = convex)
            # Use bisection on the subgradient
            # Or: the optimal x is where the "upper envelope" is minimized

            # Find minimum of U on [0, 1]

            # Simple approach: ternary search on [0, 1]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new, n) < compute_c5(h, n) + 1e-15:
            return h_new
    except:
        pass
    return h

def coord_descent(h, n, max_passes=200, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    center = n - 1
    m = 2 * n - 1
    
    for p in range(max_passes):
        if deadline and time.time() > deadline:
            break
        any_improve = False
        for j in range(n):
            # Compute current correlations
            c = np.correlate(h, 1 - h, mode='full') * dx
            
            # Compute b(l) for coordinate j
            # b[k] = (1 - h[j-l] - h[j+l]) * dx where l = k - center
            b = np.ones(m) * dx  # start with 1*dx
            for k in range(m):
                l = k - center
                jl = j - l
                jr = j + l
                if 0 <= jl < n:
                    b[k] -= h[jl] * dx
                if 0 <= jr < n:
                    b[k] -= h[jr] * dx
            
            # a = c - b * h[j]
            a = c - b * h[j]
            
            # Ternary search for x* = argmin_x max(a + b*x) on [0,1]
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                U1 = np.max(a + b * m1)
                U2 = np.max(a + b * m2)
                if U1 <= U2:
                    hi = m2
                else:
                    lo = m1
            x_star = (lo + hi) / 2.0
            
            if abs(x_star - h[j]) > 1e-12:
                h[j] = x_star
                any_improve = True
        
        if not any_improve:
            break
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def pair_search(h, n, deadline):

    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]

                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:

                        else:

                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
                if time.time() > deadline:

            if not improved:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new, n) <= compute_c5(h, n) + 1e-15:
            return h_new
    except Exception:
        pass
    return h

def coord_descent(h, n, max_passes=300, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    center = n - 1
    m = 2 * n - 1
    for p in range(max_passes):
        if deadline and time.time() > deadline:
            break
        any_improve = False
        for j in range(n):
            c = np.correlate(h, 1 - h, mode='full') * dx
            b = np.full(m, dx)
            for k_idx in range(m):
                l = k_idx - center
                jl = j - l
                jr = j + l
                if 0 <= jl < n:
                    b[k_idx] -= h[jl] * dx
                if 0 <= jr < n:
                    b[k_idx] -= h[jr] * dx
            a = c - b * h[j]
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(a + b * m1) <= np.max(a + b * m2):
                    hi = m2
                else:
                    lo = m1
            x_star = (lo + hi) / 2.0
            if abs(x_star - h[j]) > 1e-12:
                h[j] = x_star
                any_improve = True
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
        if not any_improve:
            break
    return h, best_c

def pair_search(h, n, deadline):
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(3):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
                if time.time() > deadline:
                    return h, best_c
            if not improved:
                break
    return h, best_c
