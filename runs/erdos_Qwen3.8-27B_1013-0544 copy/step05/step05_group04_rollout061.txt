def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    # Smooth max weights
    m = corr.max()
    shifted = alpha * (corr - m)
    shift = shifted.max()
    W = np.exp(shifted - shift) / np.sum(np.exp(shifted - shift))
    
    n_corr = len(corr)  # 2n-1
    grad = np.zeros(n)
    lags = np.arange(n_corr) - (n-1)  # lags from -(n-1) to n-1
    
    for j in range(n):
        # First term: sum_k W[k] * (1-h[j+lag_k]) where 0 <= j+lag_k < n
        valid1 = (j + lags >= 0) & (j + lags < n)
        idx1 = j + lags[valid1]
        grad[j] += np.sum(W[valid1] * (1 - h[idx1]))
        
        # Second term: -sum_k W[k] * h[j-lag_k] where 0 <= j-lag_k < n
        valid2 = (j - lags >= 0) & (j - lags < n)
        idx2 = j - lags[valid2]
        grad[j] -= np.sum(W[valid2] * h[idx2])
    
    return dx * grad

    # Phase 1: Coordinate descent from initial
    cd_dl = min(deadline - 200, t0 + 500)
    h1, c1 = coordinate_descent(best_h, n, cd_dl)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Pair search refinement
    if time.time() < deadline - 100:
        best_h, best_c = pair_search(best_h, n, deadline - 100,
                                      [0.01, 0.005, 0.002, 0.001, 0.0005])

    # Phase 3: Coordinate descent again
    if time.time() < deadline - 60:
        h3, c3 = coordinate_descent(best_h, n, min(deadline - 40, time.time() + 120))
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()

    # Phase 4: Fine pair search
    if time.time() < deadline - 20:
        best_h, best_c = pair_search(best_h, n, deadline - 20,
                                      [0.001, 0.0005, 0.0002, 0.0001])

    # Phase 5: Try perturbed restarts
    if time.time() < deadline - 40:
        for r in range(5):
            if time.time() > deadline - 20:
                break
            scale = 0.02 + 0.03 * r
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, deadline - time.time() - 15)
            if alloc < 15:
                break
            hr, cr = coordinate_descent(hp, n, time.time() + alloc)
            if cr < best_c:
                best_c = cr
                best_h = hr.copy()
                # Quick pair search on the new best
                if time.time() < deadline - 10:
                    best_h, best_c = pair_search(best_h, n, deadline - 10,
                                                  [0.005, 0.002, 0.001, 0.0005, 0.0002])

    # Phase 6: Final coordinate descent + pair search
    if time.time() < deadline - 10:
        h6, c6 = coordinate_descent(best_h, n, min(deadline - 5, time.time() + 60))
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()

    if time.time() < deadline - 2:
        best_h, best_c = pair_search(best_h, n, deadline - 1,
                                      [0.0005, 0.0002, 0.0001, 0.00005])

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
