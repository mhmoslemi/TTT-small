            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def smooth_obj(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        soft = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        soft = m + np.log(np.sum(np.exp(sh))) / alpha
    s = x.sum() - n / 2.0
    return float(soft + lam * s * s)

    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = np.logspace(2, np.log10(200000), max_alphas)
    lams = [100, 1000, 10000]

        for a in alphas:

    """Pair search guided by gradient of smooth objective."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    
    # Compute gradient numerically
    alpha = 1000
    eps = 1e-5
    grad = np.zeros(n)
    for i in range(n):
        if time.time() > deadline:
            break
        h_plus = h.copy()
        h_plus[i] = min(1, h[i] + eps)
        # Project to maintain sum
        h_plus = project_feasible(h_plus, n)
        h_minus = h.copy()
        h_minus[i] = max(0, h[i] - eps)
        h_minus = project_feasible(h_minus, n)
        grad[i] = (smooth_obj(h_plus, n, alpha, 1000) - smooth_obj(h_minus, n, alpha, 1000)) / (2*eps)
    
    # Sort by gradient
    order = np.argsort(grad)

                        best_c = c
                        improved = True
                    else:
                        h[i] += d
                        h[j] -= d

    best_h, best_c = pair_search(best_h, n, t4 - 2,
                                  [0.001, 0.0005, 0.0002, 0.0001, 0.00005], rng, sa_temp=0.0)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
