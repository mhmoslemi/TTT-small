def compute_F(h, n):
    dx = 2.0 / n
    dm = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    i = np.arange(n)  # shape (n,)
    # i + dm has shape (n, 2n-1)
    i_plus = i[:, None] + dm[None, :]  # (n, 2n-1)
    i_minus = i[:, None] - dm[None, :]  # (n, 2n-1)
    valid_plus = (i_plus >= 0) & (i_plus < n)
    valid_minus = (i_minus >= 0) & (i_minus < n)
    # h[i_plus] where valid
    h_plus = np.where(valid_plus, h[i_plus[valid_plus]], 0.0).reshape(n, 2*n-1)
    h_minus = np.where(valid_minus, h[i_minus[valid_minus]], 0.0).reshape(n, 2*n-1)
    F = dx * ((1.0 - h_plus) * valid_plus - h_minus * valid_minus)
    return F

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            room = np.where(mask, 1.0 - h, 0.0)
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            mask = h > 0.0
            s = h[mask].sum()
            if s < 1e-15:
                break
            h = np.where(mask, h * (1.0 + diff / s), h)
    return np.clip(h, 0.0, 1.0)

def smooth_max(x, n, alpha):
    g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = g.max()
    s = np.clip(alpha * (g - m), -50, 0)
    return m + np.logaddexp.reduce(s) / alpha

def slsqp_optimize(h, n, alpha, maxiter=300):
    def obj(x):
        return smooth_max(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_F_vectorized(h, n):
    dx = 2.0 / n
    dm = np.arange(2 * n - 1) - (n - 1)
    i = np.arange(n)
    ip = i[:, None] + dm[None, :]
    im = i[:, None] - dm[None, :]
    vp = (ip >= 0) & (ip < n)
    vm = (im >= 0) & (im < n)
    F = np.zeros((n, 2 * n - 1))
    F[vp] = (1.0 - h[ip[vp]])
    F[vm] -= h[im[vm]]
    return F * dx

def pair_descent(h, n, deadline, top_M=20, K=8, max_sweeps=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        cur_max = float(g.max())
        F = compute_F_vectorized(h, n)
        
        # Gradient-guided pair selection
        alpha_g = max(1000.0, 50.0 * len(g) / (cur_max + 1e-10))
        w = np.exp(np.clip(alpha_g * (g - cur_max), -50, 0))
        w /= w.sum()
        grad = F @ w
        
        donors = np.argsort(grad)[::-1][:top_M]
        recipients = np.argsort(grad)[:top_M]
        top_idx = np.argsort(g)[::-1][:K]
        
        improved = 0
        for j in donors:
            for i in recipients:
                if i == j:
                    continue
                s = F[i] - F[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                cands = [d_lo, d_hi]
                for a in range(K):
                    for b in range(K):
                        if top_idx[a] == top_idx[b]:
                            continue
                        dd = s[top_idx[b]] - s[top_idx[a]]
                        if abs(dd) > 1e-15:
                            dc = (g[top_idx[a]] - g[top_idx[b]]) / dd
                            if d_lo < dc < d_hi:
                                cands.append(dc)
                bd = 0.0
                bv = cur_max
                for d in cands:
                    d = float(np.clip(d, d_lo, d_hi))
                    if d == 0.0:
                        continue
                    v = float(np.max(g + d * s))
                    if v < bv - 1e-13:
                        bv = v
                        bd = d
                if bd != 0.0:
                    h[i] += bd
                    h[j] -= bd
                    g += bd * s
                    cur_max = bv
                    top_idx = np.argsort(g)[::-1][:K]

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            mask = h > 0.0
            s = h[mask].sum()
            if s < 1e-15:
                break
            h = np.where(mask, h * (1.0 + diff / s), h)
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = g.max()
    s = np.clip(alpha * (g - m), -50, 0)
    sm = s.max()
    return m + (sm + np.log(np.exp(s - sm).sum())) / alpha

                       constraints=cons,

    F[vm] -= h[im[vm]]
    return F * dx
