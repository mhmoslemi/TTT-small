import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx), c * dx

def project_feasible(h, n):
    """Euclidean projection onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    # Bisection on tau
    lo, hi = -2.0, 2.0
    for _ in range(60):
        tau = (lo + hi) / 2.0
        proj = np.clip(h - tau, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = tau
        else:
            lo = tau
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine adjustment
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        # Distribute among non-bound variables
        free = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if free.sum() > 0:
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def softmax_gradient(h, n, alpha):
    """Compute gradient of soft-max of correlations"""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Softmax weights
    c_shifted = c - c.max()
    w = np.exp(alpha * c_shifted)
    w /= w.sum()
    
    # Compute weighted gradient
    grad = np.zeros(n)
    n_full = len(c)  # 2n-1
    center = n - 1
    for k_idx in range(n_full):
        if w[k_idx] < 1e-10:
            continue
        k = k_idx - center
        if k >= 0:
            m = n - k
            grad[:m] += w[k_idx] * (1.0 - h[k:k+m])
            grad[k:k+m] += w[k_idx] * (-h[:m])
        else:
            m = n + k  # k is negative, so m = n - |k|
            grad[-k:] += w[k_idx] * (1.0 - h[:m])
            grad[:m] += w[k_idx] * (-h[-k:])
    grad *= dx
    return grad, c.max()

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)[0]
    y = h.copy()
    t = 1.0
    lr_base = 0.05
    
    while time.time() < deadline:
        # Compute c values
        cdx = np.correlate(h, 1 - h, mode='full') * (2.0/n)
        m = cdx.max()
        
        if m < best_c:
            best_c = m
            best_h = h.copy()
        
        # Top-m active set
        top_idx = np.argsort(cdx)[-top_m:]
        top_c = cdx[top_idx]
        # Soft weights among top-m
        w = np.exp(100 * (top_c - top_c.max()))
        w /= w.sum()
        
        # Weighted gradient
        dx = 2.0 / n
        center = n - 1
        grad = np.zeros(n)
        for j, k_idx in enumerate(top_idx):
            if w[j] < 1e-10:
                continue
            k = k_idx - center
            if k >= 0:
                mlen = n - k
                grad[:mlen] += w[j] * (1.0 - h[k:k+mlen])
                grad[k:k+mlen] += w[j] * (-h[:mlen])
            else:
                mlen = n + k

        grad *= dx

        y = project_feasible(h + theta * (h - y), n)

def lbfgs_softmax(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e6 * (x.sum() - n/2.0)**2
        return val
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
        center = n - 1
        g = np.zeros(n)
        n_full = len(c)

                g[:mlen] += w[k_idx] * (-x[-k:])
        g *= dx
        g += 2e6 * (x.sum() - n/2.0)

        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 50})

    """Local search: try small perturbations of pairs"""
    best_h = h.copy()
    best_c = compute_c5(h, n)[0]
    dx = 2.0 / n
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

            improved = False
            for i in range(n):

                    return best_h, best_c
                for j in range(n-1, i, -1):
                    for s in (1.0, -1.0):
                        hi, hj = h[i] + s*d, h[j] - s*d

                                improved = True

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(55):
        tau = (lo + hi) / 2.0
        if np.clip(h - tau, 0.0, 1.0).sum() > target:
            hi = tau
        else:
            lo = tau
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-14:
        free = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = free.sum()
        if cnt > 0:
            proj[free] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def grad_c_k(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        g[:m] += 1.0 - h[k:k + m]
        g[k:k + m] -= h[:m]
    else:
        m = n + k
        g[-k:] += 1.0 - h[:m]
        g[:m] -= h[-k:]
    return g * dx

def nesterov_opt(h, n, deadline, rng, top_m=10, lr0=0.04):

    best_h, best_c = h.copy(), compute_c5(h, n)
    y, prev = h.copy(), h.copy()
    t, lr = 1.0, lr0
    stall = 0
    while time.time() < deadline:
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        t = t_new
        y = project_feasible(h + beta * (h - prev), n)
        prev = h.copy()
        cdx = compute_all_c(y, n)
        m = cdx.max()
        if m < best_c - 1e-13:
            best_c, best_h, stall = m, y.copy(), 0
        else:
            stall += 1
        top = np.argsort(cdx)[-top_m:]
        tv = cdx[top]
        w = np.exp(200.0 * (tv - tv.max()))
        w /= w.sum()
        g = np.zeros(n)
        center = n - 1
        for j in range(len(top)):
            if w[j] < 1e-10:
                continue
            g += w[j] * grad_c_k(y, n, top[j] - center)
        step = lr / np.sqrt(t)
        h_new = project_feasible(y - step * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < m:
            h = h_new
            lr = min(lr * 1.004, 0.12)
        else:

            if compute_c5(h2, n) < m:
                h = h2
            else:
                h = y.copy()
            lr *= 0.94
            if stall > 200:
                h = project_feasible(best_h + 0.02 * rng.randn(n), n)
                y, prev = h.copy(), h.copy()
                lr = lr0
                stall = 0
        if lr < 1e-6:
            break
    return best_h, best_c

def lbfgs_sm(h, n, alpha, maxiter=300):
    dx = 2.0 / n
    center = n - 1
    nfull = 2 * n - 1

    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        cm = c.max()
        sh = alpha * (c - cm)
        sm = sh.max()
        val = cm + (sm + np.log(np.exp(sh - sm).sum() + 1e-300)) / alpha
        val += 1e8 * (x.sum() - n / 2.0) ** 2
        return val

    def jac(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        cm = c.max()
        sh = alpha * (c - cm)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
        g = np.zeros(n)
        for ki in range(nfull):
            if w[ki] < 1e-12:
                continue
            g += w[ki] * grad_c_k(x, n, ki - center)
        g += 2e8 * (x.sum() - n / 2.0)
        return g

    try:
        r = minimize(obj, h, method='L-BFGS-B', jac=jac,
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxls': 40})
        return project_feasible(r.x, n)
    except Exception:
        return h
