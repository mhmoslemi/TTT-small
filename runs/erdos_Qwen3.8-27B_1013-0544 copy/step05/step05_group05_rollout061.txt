import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def binarize(h, n):
    """Convert h to binary with exactly n//2 ones."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    gm = corr.max()
    shifted = alpha * (corr - gm)
    shifted -= shifted.max()
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = p[k_idx]
        if w < 1e-15:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] += -w * h[:n - lag]
        else:
            lp = -lag
            grad[lp:] += w * (1.0 - h[:n - lp])
            grad[:n - lp] += -w * h[lp:]
    
    fval = gm + np.log(exp_s.sum()) / alpha
    return fval, grad

def proj_grad_descent(h0, n, alpha, deadline, max_iter=50000):
    h = h0.copy()
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.5
    for it in range(max_iter):
        if time.time() > deadline:
            break
        fval, grad = smooth_max_and_grad(h, alpha)
        h_new = project_box_sum(h - step * grad, target)
        c_new = compute_c5(h_new, n)
        if c_new < best_c + 1e-12:
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()

For each ki, lag = ki - (n-1):

        if w < 1e-14:
            continue
        lag = ki - (n - 1)
        if lag >= 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n - lag]
        else:
            lp = -lag
            grad[lp:] += w * (1.0 - h[:n - lp])
            grad[:n - lp] -= w * h[lp:]
    return gm + np.log(e.sum()) / alpha, grad

def proj_grad(h0, n, alpha, deadline):
    h = h0.copy()
    tgt = n / 2.0
    bc = compute_c5(h, n)
    bh = h.copy()
    step = 0.4
    stall = 0
    while time.time() < deadline:
        fv, grad = smooth_max_grad(h, alpha)
        hn = project_box_sum(h - step * grad, tgt)
        c = compute_c5(hn, n)
        if c < bc - 1e-15:
            bc, bh = c, hn.copy()
            h = hn
            step = min(step * 1.08, 8.0)
            stall = 0
        else:
            stall += 1
            step *= 0.6
            if step < 1e-11 or stall > 200:
                break
    return bh, bc

def bin_local_search(hb, n, deadline):
    best_c = compute_c5(hb, n)
    improved = True
    no_improve = 0
    while improved and time.time() < deadline:
        improved = False
        zeros = list(np.where(hb == 0)[0])
        ones = list(np.where(hb == 1)[0])
        for i in zeros:
            if time.time() > deadline or not ones:
                break
            for j in ones:
                hb[i], hb[j] = 1.0, 0.0
                c = compute_c5(hb, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    ones.remove(j)
                    zeros.remove(i)
                    break
                hb[i], hb[j] = 0.0, 1.0
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    return hb, best_c

def sa_binary(hb, n, deadline, rng, t_start=0.02, t_end=0.001):
    best_c = compute_c5(hb, n)
    best_hb = hb.copy()
    cur_c = best_c
    cur_hb = hb.copy()
    n_iter = 300000
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (it / n_iter)
        zeros = np.where(cur_hb == 0)[0]
        ones = np.where(cur_hb == 1)[0]
        i = int(rng.choice(zeros))
        j = int(rng.choice(ones))
        cur_hb[i], cur_hb[j] = 1.0, 0.0
        nc = compute_c5(cur_hb, n)
        if nc < cur_c or rng.random() < np.exp(-(nc - cur_c) / max(T, 1e-10)):
            cur_c = nc
            if nc < best_c:
                best_c, best_hb = nc, cur_hb.copy()
        else:
            cur_hb[i], cur_hb[j] = 0.0, 1.0
    return best_hb, best_c

def try_grid(h_init, n, deadline, rng):
    h_init = project_box_sum(h_init, n / 2.0)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)
    # Continuous phase
    frac = 0.35
    pdl = time.time() + frac * (deadline - time.time())
    for alpha in [30, 100, 500, 2000, 20000, 200000]:
        if time.time() > pdl:
            break
        h, c = proj_grad(best_h, n, alpha, pdl)
        if c < best_c:
            best_c, best_h = c, h
    # Binary phase
    bdl = time.time() + 0.40 * (deadline - pdl)
    hb = binarize(best_h, n)
    hb, cb = bin_local_search(hb, n
, bdl)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # SA
    if time.time() < deadline - 60:
        hb2, cb2 = sa_binary(hb, n, deadline - 60, rng)
        if cb2 < best_c:
            best_c, best_h = cb2, hb2.copy()
        hb2 = binarize(best_h, n)
        hb2, cb2 = bin_local_search(hb2, n, deadline - 30)
        if cb2 < best_c:
            best_c, best_h = cb2, hb2.copy()
    # Final continuous
    if time.time() < deadline - 5:
        h, c = proj_grad(best_h, n, 500000, deadline - 3)
        if c < best_c:
            best_c, best_h = c, h
    return project_box_sum(best_h, n / 2.0), compute_c5(project_box_sum(best_h, n / 2.0), n), n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total_dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        n_parent = 250
        h_parent = np.full(n_parent, 0.5)
    h_parent = project_box_sum(h_parent, n_parent / 2.0)
    best_h = h_parent.copy()
    best_c = compute_c5(best_h, n_parent)
    best_n = n_parent
    # Primary: optimize on parent grid
    dl1 = t0 + 0.55 * (budget_s - 5)
    h1, c1, n1 = try_grid(h_parent, n_parent, dl1, rng)
    if c1 < best_c:
        best_c, best_h, best_n = c1, h1, n1
    # Secondary: try n=200
    n2 = 200
    if time.time() < total_dl - 20:
        xo = np.linspace(0, 2, n_parent)
        xn = np.linspace(0, 2, n2)
        h2_init = np.interp(xn, xo, h_parent, left=0, right=0)
        dl2 = min(total_dl - 20, time.time() + 0.25 * (budget_s - 5))
        h2, c2, n2r = try_grid(h2_init, n2, dl2, rng)
        if c2 < best_c:
            best_c, best_h, best_n = c2, h2, n2r
    # Tertiary: try slightly perturbed on parent grid
    if time.time() < total_dl - 10:
        h3_init = project_box_sum(h_parent + 0.1 * rng.randn(n_parent), n_parent / 2.0)
        dl3 = total_dl - 5
        h3, c3, n3 = try_grid(h3_init, n_parent, dl3, rng)
        if c3 < best_c:
            best_c, best_h, best_n = c3, h3, n3
    # Final polish on best
    if time.time() < total_dl - 2:
        h_f, c_f = proj_grad(best_h, best_n, 5000000, total_dl - 1)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
