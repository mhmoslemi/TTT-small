```python
            #   As first factor: appears in c_m when i is in range, contributing (1-h[i+k_m])

            # By symmetry of correlation... actually it's not symmetric because we use h and 1-h.
            # c_{-k} = np.correlate(h,1-h)[m] where m-(n-1)=-k, i.e., m=n-1-k

            #   d(c_{-k})/dh[i] = (1-h[i-k])*I(i>=k) + (-h[i+k])*I(i<n-k)

                    term2 = -h[i-kk] if (i-kk >= 0) else 0.0

                    # For kk >= 0: i' ranges from 0 to n-1-kk

                    # term2: i' = i-kk, valid if 0 <= i-kk < n-kk, i.e., kk <= i < n
                    t1 = (1 - h[i+kk]) if (0 <= i and i < n-kk and i+kk < n) else 0.0
                    t2 = -h[i-kk] if (kk <= i and i < n and i-kk >= 0) else 0.0
                    d[d_idx] = t1 + t2
                else:
                    kk2 = -kk  # positive
                    d_idx = kk + (n-1)  # = n-1-kk2
                    # c at this index = sum_{i'=kk2}^{n-1} h[i']*(1-h[i'-kk2])
                    # d/dh[i]: term1 = (1-h[i-kk2]) if kk2 <= i < n
                    #          term2 = -h[i+kk2] if 0 <= i < n-kk2
                    t1 = (1 - h[i-kk2]) if (kk2 <= i < n and i-kk2 >= 0) else 0.0
                    t2 = -h[i+kk2] if (0 <= i < n-kk2 and i+kk2 < n) else 0.0
                    d[d_idx] = t1 + t2
            
            # Similarly for coordinate j (but with opposite sign since h[j] decreases)
            d_j = np.zeros(2*n-1)
            # d_j = -d_i but with i replaced by j
            # Actually d_j is the same formula but with j instead of i
            # And the total effect is: c_new = c + delta * (d_i - d_j)
            
            # Compute d for j
            for kk in range(-(n-1), n):
                if kk >= 0:
                    t1 = (1 - h[j+kk]) if (0 <= j and j < n-kk and j+kk < n) else 0.0
                    t2 = -h[j-kk] if (kk <= j and j < n and j-kk >= 0) else 0.0
                    d_j[kk+(n-1)] = t1 + t2
                else:
                    kk2 = -kk
                    t1 = (1 - h[j-kk2]) if (kk2 <= j < n and j-kk2 >= 0) else 0.0
                    t2 = -h[j+kk2] if (0 <= j < n-kk2 and j+kk2 < n) else 0.0
                    d_j[kk+(n-1)] = t1 + t2
            
            dd = d - d_j  # change in c per unit delta
            
            # Now minimize max(c + delta * dd) subject to 0 <= h[i]+delta <= 1 and 0 <= h[j]-delta <= 1
            # Delta range: max(-h[i], h[j]-1) <= delta <= min(1-h[i], h[j])
            dmin = max(-h[i], h[j]-1)
            dmax = min(1-h[i], h[j])
            
            if dmin >= dmax:
                continue
            
            # c + delta*dd is a set of lines. We want to minimize the maximum.
            # This is a linear programming problem in 1D: minimize t subject to t >= c_k + delta*dd_k for all k
            # The optimal delta is where the upper envelope of these lines is minimized.
            # This occurs at an intersection of two lines, or at a boundary.
            
            # The upper envelope is piecewise linear and convex (max of lines is convex).
            # So we can use ternary search or find the intersection points.
            
            # For efficiency, let's use scipy minimize_scalar
            def f1(delta):
                val = c + delta * dd
                return val.max()
            
            # Use golden section search on [dmin, dmax]
            from scipy.optimize import minimize_scalar
            res = minimize_scalar(f1, bounds=(dmin, dmax), method='bounded',
                                  options={'xatol': 1e-10})
            delta_opt = res.x
            c_new_val = res.fun
            
            if c_new_val * dx < best_c - 1e-12:
                h[i] += delta_opt
                h[j] -= delta_opt
                c = c + delta_opt * dd
                best_c = c_new_val * dx
    
    return h, best_c
```

```python
import time

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full')

    k = m_star - (n - 1)
    g = np.zeros(n)
    if k >= 0:

        g[:n-k] += -x[k:]

        g[-k:] += -x[:n+k]

                if new_c5 < best_c:
                    best_c = new_c5
        
        if not improved:
            break
    
    return h, best_c
```

```python
            s = room.sum()
            if s < 1e-14:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smoothed_grad(x, n, alpha, top_k=10):
    """Compute smoothed gradient as weighted average of top lag subgradients"""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    exp_sh = np.exp(sh - sm)
    Z = exp_sh.sum()
    weights = exp_sh / Z
    
    # Get top-k lags by weight
    top_idx = np.argsort(weights)[::-1][:top_k]
    
    g = np.zeros(n)
    for idx in top_idx:
        w = weights[idx]
        if w < 1e-10:
            break
        k = idx - (n - 1)
        if k >= 0:
            g[k:] += w * (1.0 - x[:n-k])
            g[:n-k] += -w * x[k:]
        else:
            g[:n+k] += w * (1.0 - x[-k:])
            g[-k:] += -w * x[:n+k]
    g *= dx
    return g, float(m * dx)

def lbfgs_smooth(h, n, alpha, maxiter, deadline, top_k=10):

        val += 500.0 * (x.sum() - n/2.0)**2
        return float(val)
    def grad(x):
        g, _ = smoothed_grad(x, n, alpha, top_k)
        g += 1000.0 * (x.sum() - n/2.0)
        return g
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)]*n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except:
        return h

def binary_sa(h0, n, deadline, rng, max_iter=100000):
    """Simulated annealing on binary sequences"""
    h = h0.copy()
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_bin[idx[:n//2]] = 1.0
    
    def c5_bin(hb):
        dx = 2.0 / n
        c = np.correlate(hb, 1-hb, mode='full')
        return float(c.max() * dx)
    
    best_h = h_bin.copy()
    best_c = c5_bin(h_bin)
    cur_c = best_c
    cur_h = h_bin.copy()
    
    ones = set(np.where(h_bin == 1)[0])
    zeros = set(np.where(h_bin == 0)[0])
    
    t0 = time.time()
    elapsed_total = deadline - t0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        i = list(zeros)[rng.randint(len(zeros))]
        j = list(ones)[rng.randint(len(ones))]
        
        cur_h[i] = 1.0
        cur_h[j] = 0.0
        new_c = c5_bin(cur_h)
        
        if new_c <= cur_c or rng.random() < np.exp(-(new_c - cur_c) / (0.01 * (1 - it/max_iter) + 0.001)):
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            ones.remove(j)
            ones.add(i)
            zeros.discard(i)
            zeros.add(j)
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 0.0
            cur_h[j] = 1.0
    
    return best_h, best_c
```

4. Update h, go to 2

```python
import cvxpy as cp

def cutting_plane(h0, n, deadline, max_iters=100):
    """Cutting plane method for the min-max problem"""
    h = h0.copy()
    t = None
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Compute current correlation and gradients
        c_vals = np.correlate(h, 1 - h, mode='full')
        t_cur = float(c_vals.max() * (2.0/n))
        
        # Build linearized constraints
        # c_k(h) ≈ c_k(h) + grad_c_k(h) · (h_new - h)
        # We need grad_c_k for each k
        # c_k = sum h[i](1-h[i+k]) 
        # dc_k/dh[j] = (1-h[j+k])*I(valid1) - h[j-k]*I(valid2)
        
        H = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [H >= 0, H <= 1, cp.sum(H) == n/2.0, t_var >= 0]
        
        for k in range(-(n-1), n):
            # Compute c_k at current h
            idx = k + (n-1)
            ck_cur = c_vals[idx]
            
            # Compute gradient dc_k/dh at h
            gk = np.zeros(n)
            if k >= 0:
                gk[:n-k] += (1.0 - h[k:n])  # from h[i] factor, i in [0, n-k)
                gk[k:] += -h[:n-k]  # from -h[i]*h[i+k], j=i+k, so j in [k, n)
            else:
                kk = -k
                gk[kk:] += (1.0 - h[:-kk])  # from h[i] factor, i in [kk, n)
                gk[:n-kk] += -h[kk:]  # from -h[i-kk]*h[i], j=i, i in [kk, n)
            
            # Linearized: ck_cur + gk · (H - h) <= t_var * (n/2)  [scale by dx]
            # Actually: c_k * dx <= t, so c_k <= t/dx = t*n/2
            # Linearized: ck_cur + gk·(H-h) <= t_var * n / 2
            constraints.append(ck_cur + gk @ (H - h) <= t_var * n / 2.0)
        
        objective = cp.Minimize(t_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLOP)
        except:
            try:
                prob.solve(solver=cp.CBC)
            except:
                try:
                    prob.solve(solver=cp.SCIP)
                except:
                    break
        
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        
        h_new = H.value
        # Project to ensure feasibility
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        c5_new = compute_c5(h_new, n)
        if c5_new < t_cur - 1e-8:
            h = h_new.copy()
        else:
            # No improvement, try small perturbation
            h = project_feasible(h_new + 0.01 * np.random.randn(n), n)
            if compute_c5(h, n) > t_cur:
                break
    
    return h, compute_c5(h, n)
```

import numpy as np
from scipy.optimize import minimize
from scipy.optimize import minimize_scalar
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_c_at_k(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            g[:n-k] += (1.0 - h[k:n])
            g[k:n] += -h[:n-k]
    else:
        kk = -k
        if n - kk > 0:
            g[kk:] += (1.0 - h[:n-kk])
            g[:n-kk] += -h[kk:]
    return g

def all_grads(h, n):
    """Compute gradients for all lags, shape (2n-1, n)"""
    grads = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        idx = k + (n-1)
        grads[idx] = grad_c_at_k(h, n, k)
    return grads

def cutting_plane(h0, n, deadline, max_iters=200, verbose=False):
    """Cutting plane / Kelley's method"""
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Evaluate at current point
        c_vals = np.correlate(h, 1 - h, mode='full')
        t_cur = float(c_vals.max())
        
        # Compute all gradients
        G = all_grads(h, n)  # shape (2n-1, n)
        
        # Solve LP:
        # min t
        # s.t. c_vals[i] + G[i] · (H - h) <= t  for all i
        #      0 <= H <= 1
        #      sum(H) = n/2
        
        H = cp.Variable(n)
        t = cp.Variable()

            H >= 0,
            H <= 1,
            cp.sum(H) == n / 2.0,
            c_vals + G @ (H - h) <= t
        ]
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        solved = False
        for solver in [cp.GLOP, cp.CBC, cp.SCIP]:

                prob.solve(solver=solver)
                solved = True
                break
            except:
                continue

            else:
