    n_corr = 2 * n - 1

        temp *= 0.9997
        if temp < 1e-6:
            temp = 1e-6
    
    return best_h, best_c5

def lbfgs_smooth(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e6 * (x.sum() - n / 2.0) ** 2
        return val
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full')
        m_star = int(np.argmax(c))
        k = m_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - x[:n - k])
            g[:n - k] += -x[k:]
        else:
            g[:n + k] += (1.0 - x[-k:])
            g[-k:] += -x[:n + k]
        g *= dx
        g += 2e6 * (x.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def shake_and_bake(h, n, rng, kick_size=3):
    """Randomly flip a few bits maintaining feasibility."""
    h_new = h.copy()
    ones = list(np.where(h_new == 1)[0])
    zeros = list(np.where(h_new == 0)[0])
    n_flip = min(kick_size, len(ones), len(zeros))
    for _ in range(n_flip):
        j = ones.pop(rng.randint(len(ones)))
        i = zeros.pop(rng.randint(len(zeros)))
        h_new[i], h_new[j] = 1.0, 0.0
        ones.append(i)
        zeros.append(j)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    
    # Round to binary
    h_bin = np.zeros(n)
    idx_sorted = np.argsort(h0)[::-1]
    h_bin[idx_sorted[:n // 2]] = 1.0
    best_h = h_bin.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Best-improvement from initial binary (30%)
    p1 = t0 + 0.30 * (budget_s - 3)
    h1, c1 = binary_best_improvement(best_h, n, p1)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()
    
    # Phase 2: Shake and bake with multiple restarts (35%)
    p2 = t0 + 0.65 * (budget_s - 3)
    n_restarts = 30
    per_restart = (p2 - time.time()) / max(n_restarts, 1)
    for r in range(n_restarts):
        if time.time() > p2 - 5:
            break
        # Shake the current best
        h_shake = shake_and_bake(best_h, n, rng, kick_size=3 + (r % 3))
        sub_dl = min(time.time() + per_restart, p2 - 2)
        h2, c2 = binary_best_improvement(h_shake, n, sub_dl, max_rounds=500)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()
        # Also try fully random
        if r % 5 == 0:
            h_rand = np.zeros(n)
            h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
            h3, c3 = binary_best_improvement(h_rand, n, min(time.time() + per_restart * 0.5, p2 - 2), max_rounds=300)
            if c3 < best_c:
                best_c, best_h = c3, h3.copy()
    
    # Phase 3: Simulated annealing from best (15%)
    p3 = t0 + 0.80 * (budget_s - 3)
    if time.time() < p3:
        h4, c4 = binary_sa(best_h, n, p3, rng, n_iter=50000)
        if c4 < best_c:
            best_c, best_h = c4, h4.copy()
        # Polish after SA
        h4b, c4b = binary_best_improvement(best_h, n, p3 + 10, max_rounds=200)
        if c4b < best_c:
            best_c, best_h = c4b, h4b.copy()
    
    # Phase 4: More shake-and-bake refinements (10%)
    p4 = t0 + 0.90 * (budget_s - 3)
    for r in range(15):
        if time.time() > p4:
            break
        h_shake = shake_and_bake(best_h, n, rng, kick_size=2 + (r % 2))
        h5, c5 = binary_best_improvement(h_shake, n, min(time.time() + 15, p4), max_rounds=200)
        if c5 < best_c:
            best_c, best_h = c5, h5.copy()
    
    # Phase 5: Continuous L-BFGS polish (final 10%)
    p5 = deadline
    if time.time() < p5 - 20:
        # Try from binary solution
        for alpha in [50000, 200000, 1000000]:
            if time.time() > p5 - 10:
                break
            h6 = lbfgs_smooth(best_h.copy(), n, alpha, 150)
            c6 = compute_c5(h6, n)
            if c6 < best_c:
                best_c, best_h = c6, h6.copy()
        # Also try from a slightly perturbed version
        h_pert = best_h + 0.1 * rng.randn(n)
        h_pert = project_feasible(h_pert, n)
        h7 = lbfgs_smooth(h_pert, n, 200000, 100)
        c7 = compute_c5(h7, n)
        if c7 < best_c:
            best_c, best_h = c7, h7.copy()
    
    # Final: ensure feasibility and recompute
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
