    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

            break

            if s < 1e-14:

            h = h + diff / s * h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def greedy_search(h, n, deadline, rng):
    cur_h = h.copy()
    cur_c = compute_c5(cur_h, n)
    best_c = cur_c
    stall = 0
    while time.time() < deadline and stall < 40:

        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in zeros:

                return cur_h, best_c
            for j in ones:
                cur_h[i], cur_h[j] = 1.0, 0.0

                    if c < best_c:

                    improved = True
                    break
                else:

            if improved:
                break
        if not improved:
            stall += 1
    return cur_h, best_c

                best_h = cur_h.copy()
        else:

        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def con_fun(x):
