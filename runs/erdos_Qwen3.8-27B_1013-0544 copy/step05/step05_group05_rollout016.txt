```python
        fi = b - m + (n-1)
        if 0 <= fi < n:
            corr[m] += h[fi]
    
    return corr
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    """Project to feasible set: 0≤h≤1, sum=h_target."""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-12: break
            h += diff / s * h
    return np.clip(h, 0, 1)

            for j in ones:
                # Try swap: h[i] 0→1, h[j] 1→0
                best_h[i] = 1
                best_h[j] = 0
                new_corr = compute_corr(best_h, n)
                c = float(np.max(new_corr) * 2.0 / n)
                if c < best_c - 1e-15:
                    best_c = c
                    corr = new_corr
                    improved = True
                    break
                else:
                    best_h[i] = 0
                    best_h[j] = 1
            if improved:
                break
        if not improved:
            break
    
    return best_h, best_c
```
