def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_delta(h, i, j, n):
    """Compute correlation change when h[i] goes 0->1 and h[j] goes 1->0."""
    delta = np.zeros(2 * n - 1)
    # Direct h[i]: 0->1, add (1-h[i+k]) for valid k
    lo = max(0, n - 1 - i)
    hi = min(2 * n - 2, 2 * n - 2 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        idx_g = m - (n - 1) + i
        delta[m] += (1.0 - h[idx_g])
    # Indirect g[i]: 1->0, subtract h[j'] for j'+k=i => m=n-1+i-j'
    m = (n - 1 + i - np.arange(n))
    valid = (m >= 0) & (m <= 2 * n - 2)
    delta[m[valid]] -= h[np.arange(n)[valid]]
    # Direct h[j]: 1->0, subtract (1-h[j+k]) for valid k
    lo = max(0, n - 1 - j)
    hi = min(2 * n - 2, 2 * n - 2 - j)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        idx_g = m - (n - 1) + j
        delta[m] -= (1.0 - h[idx_g])
    # Indirect g[j]: 0->1, add h[j'] for j'+k=j => m=n-1+j-j'
    m = (n - 1 + j - np.arange(n))

    return delta

def local_search(h, n, deadline, max_sweeps=500, seed=42, temperature=0.0):
    """Binary local search with optional simulated annealing."""
    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                delta = swap_delta(h, int(i), int(j), n)
                new_corr = corr + delta
                nc = float(new_corr.max()) * dx
                accept = nc < cur_c - 1e-15
                if not accept and temperature > 0:
                    d = nc - cur_c
                    if d < 1.0:  # only accept small worsening
                        accept = rng.random() < np.exp(-d * n / (2.0 * temperature))
                if accept:
                    h[i], h[j] = 1.0, 0.0
                    corr = new_corr
                    cur_c = nc
                    improved = True
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                    break
            if improved:
                break
        if not improved:
            break
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=2000):
    dx = 2.0 / n

        m = g.max()

        # = (1-x[i+k]) - x[i] (from x[i] term) + sum of terms where j+k=i
        # This is complex; skip and let SLSQP compute numerically
        return None
    def cons_fun(x):
        return x.sum() - n / 2.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    lo = max(0, n - 1 - i)
    hi = min(2 * n - 2, 2 * n - 2 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] += 1.0 - h[m - (n - 1) + i]
    m2 = np.arange(max(0, n - 1 + i - (n - 1)), min(2 * n - 2, n - 1 + i) + 1)
    jprime = n - 1 + i - m2
    delta[m2] -= h[jprime]
    lo = max(0, n - 1 - j)
    hi = min(2 * n - 2, 2 * n - 2 - j)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] -= 1.0 - h[m - (n - 1) + j]
    m2 = np.arange(max(0, n - 1 + j - (n - 1)), min(2 * n - 2, n - 1 + j) + 1)
    jprime = n - 1 + j - m2
    delta[m2] += h[jprime]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)

    return delta

def local_search(h, n, deadline, max_sweeps=500, seed=42, temp=0.0):

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                d = swap_delta(h, int(i), int(j), n)
                nc = float((corr + d).max()) * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + d
                    cur_c = nc
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                    improved = True
                    break
                elif temp > 0.0:
                    delta_c = nc - cur_c
                    if delta_c < 0.5 * dx and rng.random() < np.exp(-delta_c / temp):
                        h[i], h[j] = 1.0, 0.0
                        corr = corr + d
                        cur_c = nc
                        improved = True
                        break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def sa_local_search(h, n, deadline, max_sweeps=300, seed=42, t0=0.02, t1=0.001):

                best_h = h.copy()

def slsqp_smooth(h, n, alpha, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
