```python
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000):
    dx = dx_of(n)
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()

        return h.copy()

    # k is the index into the full correlation array (0 to 2n-2)
    # lag = k - (n-1)

        if 0 <= ts < n:
            g[t] += (1.0 - h[ts])
            g[ts] -= h[t]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def dx_of(n):
    return 2.0 / n

def compute_c5(h, n):
    c = np.correlate(h, 1 - h, mode='full') * dx_of(n)
    return float(np.max(c))

def get_corr(h, n):
    return np.correlate(h, 1 - h, mode='full') * dx_of(n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sc(x, n):
    return float(np.sum(x) - n / 2.0)

def _sj(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=5000):
    dx = dx_of(n)
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0,1)]*n,
                      constraints=[{'type':'eq','fun':_sc,'jac':_sj,'args':(n,)}],
                      options={'maxiter':maxiter,'ftol':1e-18})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def lag_grad(h, n, k):
    s = k - (n-1)
    g = np.zeros(n)
    lo, hi = max(0,-s), min(n-1, n-1-s)
    if lo > hi: return g
    g[lo:hi+1] += (1.0 - h[lo+s:hi+1+s])
    lo2, hi2 = max(0, lo+s), min(n-1, hi+s)
    if lo2 <= hi2:
        g[lo2:hi2+1] -= h[lo2-s:hi2+1-s]
    return g

def cd_ls(h, n, deadline, top_lags=3, max_stall=500):
    best_h, best_c = h.copy(), compute_c5(h, n)
    stall = 0
    while time.time() < deadline - 1:
        c = get_corr(h, n)
        cur = float(c.max())
        if cur < best_c - 1e-15:
            best_c, best_h, stall = cur, h.copy(), 0
        else:
            stall += 1
            if stall > max_stall: break
        tk = np.argsort(c)[-top_lags:]
        g = np.zeros(n)
        for k in tk: g += lag_grad(h, n, int(k))
        if g.max() - g.min() < 1e-12: break
        gs = np.argsort(g)
        improved = False
        for sr in range(4):
            for dr in range(4):
                i, j = int(gs[n-1-sr]), int(gs[dr])
                if i == j: continue
                md = min(h[i]*0.999, (1-h[j])*0.999)
                if md < 1e-10: continue
                bd, be = 0, cur
                for fr in [0.5, 0.3, 0.7, 0.1, 0.9, 0.2, 0.4, 0.6, 0.8, 0.05, 0.15, 0.35]:
                    d = md * fr
                    h[i] -= d; h[j] += d
                    nc = compute_c5(h, n)
                    if nc < be: be, bd = nc, d
                    h[i] += d; h[j] -= d
                if bd >= 1e-10:
                    h[i] -= bd; h[j] += bd
                    improved = True
                    break
            if improved: break
        if not improved and stall > max_stall: break
    return best_h, best_c

def full_optimize(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h, best_c = h.copy(), compute_c5(h, n)
    for a in [200, 1000, 5000, 20000, 100000]:
        if time.time() > deadline - 20: break
        ht = slsqp(best_h, n, a, maxiter=5000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:

        if time.time() < deadline - 30:
            ht2, c2 = cd_ls(best_h, n, min(deadline-20, time.time()+20), top_lags=3)
            if c2 < best_c:

    if time.time() < deadline - 5:

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def dx_of(n):
    return 2.0 / n

def compute_c5(h, n):
    c = np.correlate(h, 1 - h, mode='full') * dx_of(n)
    return float(np.max(c))

def get_corr(h, n):
    return np.correlate(h, 1 - h, mode='full') * dx_of(n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sc(x, n):
    return float(np.sum(x) - n / 2.0)

def _sj(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=5000):
    dx = dx_of(n)
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': _sc, 'jac': _sj, 'args': (n,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_grad(h, n, k):
    s = k - (n - 1)
    g = np.zeros(n)
    lo, hi = max(0, -s), min(n - 1, n - 1 - s)
    if lo > hi:
        return g
    g[lo:hi+1] += (1.0 - h[lo+s:hi+1+s])
    lo2, hi2 = max(0, lo+s), min(n - 1, hi+s)
    if lo2 <= hi2:
        g[lo2:hi2+1] -= h[lo2-s:hi2+1-s]
    return g

def cd_ls(h, n, deadline, top_lags=3, max_stall=400):
    best_h, best_c = h.copy(), compute_c5(h, n)
    stall = 0
    d_fracs = [0.5, 0.3, 0.7, 0.1, 0.9, 0.2, 0.4, 0.6, 0.8, 0.05, 0.15, 0.35, 0.45, 0.55, 0.65]
    while time.time() < deadline - 1:
        c = get_corr(h, n)
        cur = float(c.max())
        if cur < best_c - 1e-15:
            best_c, best_h, stall = cur, h.copy(), 0
        else:
            stall += 1
            if stall > max_stall:
                break
        tk = np.argsort(c)[-top_lags:]
        g = np.zeros(n)
        for k in tk:
            g += lag_grad(h, n, int(k))
        if g.max() - g.min() < 1e-12:
            break
        gs = np.argsort(g)
        improved = False
        for sr in range(5):
            for dr in range(5):
                i, j = int(gs[n-1-sr]), int(gs[dr])
                if i == j:
                    continue
                md = min(h[i] * 0.999, (1.0 - h[j]) * 0.999)
                if md < 1e-10:
                    continue
                bd, be = 0.0, cur
                for fr in d_fracs:
                    d = md * fr
                    h[i] -= d
                    h[j] += d
                    nc = compute_c5(h, n)
                    if nc < be:
                        be, bd = nc, d
                    h[i] += d
                    h[j] -= d
                if bd >= 1e-10:
                    h[i] -= bd
                    h[j] += bd
                    improved = True
                    break
            if improved:
                break
        if not improved:
            stall += 2
            if stall > max_stall:
                break
    return best_h, best_c

def subgrad_step(h, n, step_size, top_lags=5):
    c = get_corr(h, n)
    tk = np.argsort(c)[-top_lags:]
    g = np.zeros(n)
    for k in tk:
        g += lag_grad(h, n, int(k))
    g /= len(tk)
    h_new = h - step_size * g
    return project_feasible(h_new, n)

def full_optimize(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h, best_c = h.copy(), compute_c5(h, n)
    alphas = [100, 500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 25:
            break
        ht = slsqp(best_h, n, a, maxiter=4000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c, best_h = c, ht
        if time.time() < deadline - 35:
            t_alloc = min(25, deadline - time.time() - 25)
            if t_alloc > 5:
                ht2, c2 = cd_ls(best_h, n, time.time() + t_alloc, top_lags=3, max_stall=300)
                if c2 < best_c:
                    best_c, best_h = c2, ht2
    if time.time() < deadline - 5:
        best_h, best_c = cd_ls(best_h, n, deadline - 3, top_lags=4, max_stall=3000)
    if time.time() < deadline - 3:
        alpha = max(1.0, (compute_c5(best_h, n) - 0.3) * 0.5)
        for _ in range(50):
            if time.time() > deadline - 3:
                break
            ht = subgrad_step(best_h, n, alpha, top_lags=5)
            c = compute_c5(ht, n)
            if c < best_c:
                best_h, best_c = ht, c
                alpha = max(0.1, alpha * 1.2)
            else:
                alpha = max(0.01, alpha * 0.5)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init, n_init = np.full(250, 0.5), 250

    best_h, best_c = h_init.copy(), float('inf')

    # Phase 1: Main optimization at initial resolution
    alloc1 = min(budget_s * 0.30, deadline - t0 - 80)
    if alloc1 > 30:
        best_h, best_c = full_optimize(h_init, n_init, t0 + alloc1)

    # Phase 2: Try alternate resolutions
    x_best = np.linspace(0, 2, len(best_h))
    for nt in [200, 300, 220, 280, 180]:
        if nt == n_init or time.time() > deadline - 100:
            continue
        h_interp = project_feasible(np.interp(np.linspace(0, 2, nt), x_best, best_h), nt)
        alloc = min(80, (deadline - time.time() - 30) * 0.25)
        if alloc < 30:
            break
        ht, ct = full_optimize(h_interp, nt, time.time() + alloc)
        if ct < best_c:
            best_c, best_h = ct, ht
            x_best = np.linspace(0, 2, nt)

    # Phase 3: Local restarts at best resolution
    n = len(best_h)
    if time.time() < deadline - 50:
        remaining = deadline - time.time() - 30
        n_rest = max(4, int(remaining / 45))
        for r in range(n_rest):
            if time.time() > deadline - 30:
                break
            scale = 0.002 + 0.04 * (r / max(1, n_rest - 1))
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(50, (deadline - time.time() - 20) * 0.4)
            if alloc < 20:
                break
            ht, ct = full_optimize(hp, n, time.time() + alloc)
            if ct < best_c:
                best_c, best_h = ct, ht
                n = len(best_h)

    # Phase 4: Final aggressive polish
    if time.time() < deadline - 3:
        n = len(best_h)
        best_h, best_c = cd_ls(best_h, n, deadline - 1, top_lags=5, max_stall=10000)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(250, 0.5), 250)
        best_c = compute_c5(best_h, 250)

    return best_h, best_c, len(best_h)
```
