def linearize_and_solve(h, n, active_lags, solver='highs'):
    """Solve the linearized subproblem."""
    g = np.correlate(h, 1 - h, mode='full') * (2.0/n)
    # Linearize each active lag
    # g[k](h) ≈ g[k] + grad_k · (h - h)
    # where grad_k[j] = (1-h[j+k-(n-1)]) - h[j-k+(n-1)] (with bounds)
    
    # Build the constraint matrix
    # For each active lag k, the constraint is:
    # sum_j grad_k[j] * h[j] ≤ t - g[k] + sum_j grad_k[j] * h[j] = t
    # i.e., sum_j grad_k[j] * h[j] - t ≤ 0
    
    m = len(active_lags)
    # A is m x (n+1), constraint: A @ [h; t] ≤ b
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    
    for idx, k in enumerate(active_lags):
        # grad_k[j] = (1-h[j+k-(n-1)])_{valid} - h[j-k+(n-1)]_{valid}
        grad = np.zeros(n)
        for j in range(n):
            # First term: 1-h[j+k-(n-1)] if 0 <= j+k-(n-1) < n
            idx1 = j + k - (n-1)
            if 0 <= idx1 < n:
                grad[j] += 1.0 - h[idx1]
            # Second term: -h[j-k+(n-1)] if 0 <= j-k+(n-1) < n
            idx2 = j - k + (n-1)
            if 0 <= idx2 < n:
                grad[j] -= h[idx2]
        A[idx, :n] = grad
        A[idx, n] = -1.0
        b[idx] = -g[k] + np.dot(grad, h)  # = -g[k] + g[k] = 0? No...
        # Actually: g[k](h) ≈ g[k](h₀) + grad · (h - h₀)
        # We want: g[k](h₀) + grad · (h - h₀) ≤ t
        # => grad · h ≤ t - g[k](h₀) + grad · h₀
        # => grad · h - t ≤ -g[k](h₀) + grad · h₀
        b[idx] = -g[k] + np.dot(grad, h)
    
    # Add integral constraint: sum(h) = n/2
    # This is an equality, handle separately
    
    # Add bounds: 0 ≤ h ≤ 1, t ≥ 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    # Solve LP: min t subject to A @ x ≤ b, sum(h) = n/2, bounds
    # Use scipy.optimize.linprog
    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                  method='highs')
    if res.success:
        h_new = res.x[:n]
        return np.clip(h_new, 0, 1)
    return h

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def grad_lag(h, n, k):
    """Gradient of the k-th correlation value with respect to h."""
    grad = np.zeros(n)
    n1 = n - 1
    for j in range(n):
        idx1 = j + k - n1
        if 0 <= idx1 < n:
            grad[j] += 1.0 - h[idx1]
        idx2 = j - k + n1
        if 0 <= idx2 < n:
            grad[j] -= h[idx2]
    return grad

def vectorized_grads(h, n, lags):
    """Compute gradients for multiple lags at once."""
    n1 = n - 1
    grads = np.zeros((len(lags), n))
    for idx, k in enumerate(lags):
        g = np.zeros(n)
        for j in range(n):
            idx1 = j + k - n1
            if 0 <= idx1 < n:
                g[j] += 1.0 - h[idx1]
            idx2 = j - k + n1
            if 0 <= idx2 < n:
                g[j] -= h[idx2]
        grads[idx] = g
    return grads

def lp_solve(h, n, active_lags):
    """Solve the linearized LP subproblem."""
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    n1 = n - 1
    m = len(active_lags)
    
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    
    for idx, k in enumerate(active_lags):
        grad = np.zeros(n)
        for j in range(n):
            idx1 = j + k - n1
            if 0 <= idx1 < n:
                grad[j] += 1.0 - h[idx1]
            idx2 = j - k + n1
            if 0 <= idx2 < n:
                grad[j] -= h[idx2]
        A[idx, :n] = grad
        A[idx, n] = -1.0
        b[idx] = -g[idx] + np.dot(grad, h)  # should be ≈ 0
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return np.clip(res.x[:n], 0, 1)
    return h

def cutting_plane(h, n, deadline, max_iter=200):
    """Iterative cutting plane / outer approximation."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, mode='full') * (2.0/n)
        gm = g.max()
        # Active set: lags where g[k] > gm - tolerance
        tol = 1e-6 / n  # relative tolerance
        active = np.where(g > gm - max(tol, 5*(2.0/n)))[0]
        
        if len(active) > 100:
            # Too many active, take top 100
            top_idx = np.argsort(g)[::-1][:100]
            active = top_idx
        
        h_new = lp_solve(h, n, active)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        # Check convergence
        if abs(c_new - best_c) < 1e-10:
            break
        
        h = h_new
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    n1 = n - 1
    lo = max(0, n1 - i)
    hi = min(2 * n1, n1 + 2*n1 - i)  # 2n-2 - i
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] += 1.0 - h[m - n1 + i]
    m2_lo = max(0, n1 + i - n1)  # = i
    m2_hi = min(2*n1, n1 + i)
    m2 = np.arange(m2_lo, m2_hi + 1)
    delta[m2] -= h[n1 + i - m2]
    lo = max(0, n1 - j)
    hi = min(2*n1, 2*n1 - j)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] -= 1.0 - h[m - n1 + j]
    m2_lo = max(0, n1 + j - n1)
    m2_hi = min(2*n1, n1 + j)
    m2 = np.arange(m2_lo, m2_hi + 1)
    delta[m2] += h[n1 + j - m2]
    delta[n1] -= 2.0
    k1 = n1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def local_search(h, n, deadline, max_sweeps=500, seed=42):
    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                d = swap_delta(h, int(i), int(j), n)
                nc = float((corr + d).max()) * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + d
                    cur_c = nc
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

Actually, let me reconsider the gradient computation. For n=250 and 50 active lags, computing the gradient naively takes 50*250 = 12500 operations. But I can vectorize it:

For a given k, the gradient at position j is:
grad[j] = (1-h[j+k-n1]) * (0 ≤ j+k-n1 < n) - h[j-k+n1] * (0 ≤ j-k+n1 < n)

This can be computed as:
grad = (1-h) shifted by k - h shifted by -k (with zero-padding)

Using numpy:
```python
def compute_grad(h, n, k):
    n1 = n - 1
    grad = np.zeros(n)
    # First term: 1-h[j+k-n1] for valid j
    # j+k-n1 in [0, n-1] => j in [n1-k, n1-k+n-1] = [n1-k, 2n1-k]
    # Also j in [0, n-1]
    lo = max(0, n1-k)
    hi = min(n-1, 2*n1-k)
    if lo <= hi:
        grad[lo:hi+1] += 1 - h[lo+k-n1 : hi+1+k-n1]
    # Second term: -h[j-k+n1] for valid j
    # j-k+n1 in [0, n-1] => j in [k-n1, k-n1+n-1] = [k-n1, k]
    # Also j in [0, n-1]
    lo2 = max(0, k-n1)
    hi2 = min(n-1, k)
    if lo2 <= hi2:
        grad[lo2:hi2+1] -= h[lo2-k+n1 : hi2+1-k+n1]
    return grad
```

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def compute_grad(h, n, k):
    n1 = n - 1
    grad = np.zeros(n)
    lo = max(0, n1 - k)
    hi = min(n - 1, 2 * n1 - k)
    if lo <= hi:
        grad[lo:hi+1] += 1.0 - h[lo + k - n1 : hi + 1 + k - n1]
    lo2 = max(0, k - n1)
    hi2 = min(n - 1, k)
    if lo2 <= hi2:
        grad[lo2:hi2+1] -= h[lo2 - k + n1 : hi2 + 1 - k + n1]
    return grad

def lp_subproblem(h, n, active_lags):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = len(active_lags)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    for idx, k in enumerate(active_lags):
        grad = compute_grad(h, n, k)
        A[idx, :n] = grad
        A[idx, n] = -1.0
        b[idx] = -g[k] + np.dot(grad, h)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(0, None)]
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return np.clip(res.x[:n], 0, 1)
    return h

def cutting_plane(h, n, deadline, max_iter=1000, tol_lags=8):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, mode='full') * dx
        gm = g.max()
        tol = tol_lags * dx
        active = np.where(g >= gm - tol)[0]
        if len(active) > 80:
            active = np.argsort(g)[::-1][:80]
        h_lp = lp_subproblem(h, n, active)
        h_lp = project_feasible(h_lp, n)
        c_lp = compute_c5(h_lp, n)
        # Damping to avoid oscillation
        if c_lp < best_c:
            h = h_lp
            if c_lp < best_c:
                best_c = c_lp
                best_h = h.copy()
        elif c_lp < gm + 2 * dx:
            # Blend
            h = 0.5 * h + 0.5 * h_lp
            h = project_feasible(h, n)
            c_blend = compute_c5(h, n)
            if c_blend < best_c:
                best_c = c_blend
                best_h = h.copy()
        else:
            # Stuck, try smaller active set

            h_lp2 = lp_subproject(h, n, active2) if False else h_lp
            h = h_lp2
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
            else:

    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def swap_delta(h, i, j, n):
    n1 = n - 1
    delta = np.zeros(2 * n - 1)
    lo = max(0, n1 - i)
    hi = min(2 * n1, n1 + n1 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] += 1.0 - h[m - n1 + i]
    m2 = np.arange(max(0, i), min(2 * n1, n1 + i) + 1)
    delta[m2] -= h[n1 + i - m2]
    lo = max(0, n1 - j)
    hi = min(2 * n1, n1 + n1 - j)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] -= 1.0 - h[m - n1 + j]
    m2 = np.arange(max(0, j), min(2 * n1, n1 + j) + 1)
    delta[m2] += h[n1 + j - m2]
    delta[n1] -= 2.0
    k1 = n1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def local_search(h, n, deadline, max_sweeps=2000, seed=42):
    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                d = swap_delta(h, int(i), int(j), n)
                nc = float((corr + d).max()) * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + d
                    cur_c = nc
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c
