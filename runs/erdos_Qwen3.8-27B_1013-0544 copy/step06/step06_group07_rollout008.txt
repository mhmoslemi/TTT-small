    """Find delta in [delta_lo, delta_hi] that minimizes max(c + delta*D)."""

    # => delta <= (t - c[m]) / D[m] for D[m] > 0
    # => delta >= (t - c[m]) / D[m] for D[m] < 0
    # => c[m] <= t for D[m] = 0

    hi = delta_hi

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def get_delta_vector(h, n, i, j):

    idx2 = i - s_range  # positions of i-s
    mask2 = (idx2 >= 0) & (idx2 < n)
    contrib2 = np.zeros(corr_len)
    contrib2[mask2] = -h[idx2[mask2]]
    
    idx3 = j + s_range
    mask3 = (idx3 >= 0) & (idx3 < n)
    contrib3 = np.zeros(corr_len)
    contrib3[mask3] = -(1.0 - h[idx3[mask3]])
    
    idx4 = j - s_range
    mask4 = (idx4 >= 0) & (idx4 < n)
    contrib4 = np.zeros(corr_len)
    contrib4[mask4] = h[idx4[mask4]]
    
    D = contrib1 + contrib2 + contrib3 + contrib4
    D[center] = 0.0  # For binary case, s=0 change is always 0
    return D

def binary_sa(h_bin, n, deadline, rng, t_start=0.03, t_end=0.00001, patience=50000):
    dx = 2.0 / n
    h = h_bin.copy()
    corr_len = 2 * n - 1
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 200000.0)
    T = t_start
    s_range = np.arange(-(n-1), n)
    center = n - 1
    step = 0
    
    while time.time() < deadline:
        # Move selection: sometimes target bottleneck shifts
        if rng.random() < 0.3:
            # Targeted: find i where h[i]=0 and h[i+s_max]=1 for a max shift
            max_shifts = np.argsort(corr)[::-1][:5]
            s_idx = max_shifts[rng.randint(5)]
            s_val = s_idx - center
            # Find valid i where h[i]=0 and h[i+s_val]=1
            if s_val >= 0:
                valid_i = np.arange(0, n - s_val)
            else:
                valid_i = np.arange(-s_val, n)
            candidates = valid_i[(h[valid_i] == 0) & (h[valid_i + s_val] == 1)]
            if len(candidates) == 0:
                i = zeros[rng.randint(len(zeros))]
                j = ones[rng.randint(len(ones))]
            else:
                i = candidates[rng.randint(len(candidates))]
                j = i + s_val
                if h[j] != 1:
                    i = zeros[rng.randint(len(zeros))]
                    j = ones[rng.randint(len(ones))]
        else:
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
        
        D = get_delta_vector(h, n, i, j)
        new_corr = corr + D
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        
        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        step += 1
        if step % 100 == 0:
            T *= cool_rate ** 100
            if T < t_end:
                T = t_end
        
        if no_improve > patience:
            # Restart with perturbation
            nk = 3 + rng.randint(8)
            for _ in range(nk):
                ii = zeros[rng.randint(len(zeros))]
                jj = ones[rng.randint(len(ones))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start * 2, max(T * 5, 0.01))
    
    return best_h, best_c

def continuous_pair_opt(h, n, deadline, rng, max_sweeps=200):
    """Allow continuous values by doing optimal partial transfers."""
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    center = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try random pairs
        for _ in range(2000):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] >= h[j]:
                continue
            # delta in [0, h[j]-0] but also h[i]+delta <= 1
            delta_max = min(1.0 - h[i], h[j])
            if delta_max < 1e-10:
                continue
            
            D = get_delta_vector(h, n, i, j)
            # For continuous, s=0 is not exactly 0, but let's approximate
            # Find optimal delta by ternary search
            lo, hi = 0.0, delta_max
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(corr + m1 * D) <= np.max(corr + m2 * D):
                    hi = m2
                else:
                    lo = m1
            delta_opt = (lo + hi) / 2.0
            new_corr = corr + delta_opt * D
            nc = new_corr.max() * dx
            
            if nc < cur_c - 1e-12:
                h[i] += delta_opt
                h[j] -= delta_opt
                h = np.clip(h, 0, 1)
                # Re-project to maintain sum
                h = project_feasible(h, n)
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if not improved:
            break
    
    return best_h, best_c
