    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy().astype(float)
    cur_c = float(corr.max()) * dx
    
    # Precompute C matrix: C[i, k] = 1 - h[i + k - (n-1)] if valid
    C = np.zeros((n, 2*n-1))
    for i in range(n):
        lo = max(0, n-1-i)
        hi = min(2*n-2, 2*n-2-i)
        if lo <= hi:
            C[i, lo:hi+1] = 1 - h[i + np.arange(lo, hi+1) - (n-1)]
    
    best_c = cur_c

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy().astype(float)

    best_h = h.copy()

C[a, k] = 1 - h[a + k - (n-1)] if 0 ≤ a+k-(n-1) ≤ n-1, else 0.

When h[i] changes from 0 to 1:
- For all (a, k) such that a + k - (n-1) = i, C[a, k] changes from 1-0=1 to 1-1=0.
- So C[a, k] decreases by 1.
- The condition a + k - (n-1) = i means k = i - a + (n-1).
- For a given a, k = i - a + (n-1).
- We need 0 ≤ k ≤ 2n-2, i.e., 0 ≤ i - a + (n-1) ≤ 2n-2.
- This is i + (n-1) - a ≤ 2n-2 → a ≥ i - (n-1), and i + (n-1) - a ≥ 0 → a ≤ i + (n-1).
- Since a is in [0, n-1], the valid range is a ∈ [max(0, i-(n-1)), min(n-1, i+(n-1))].
- Since 0 ≤ i ≤ n-1, this simplifies to a ∈ [0, n-1] (always valid since i-(n-1) ≤ 0 and i+(n-1) ≥ n-1).

Wait: i + (n-1) - a ≥ 0 → a ≤ i + (n-1). Since a ≤ n-1 and i + (n-1) ≥ n-1 (because i ≥ 0), this is always satisfied.
And i + (n-1) - a ≤ 2n-2 → a ≥ i + (n-1) - (2n-2) = i - n + 1. Since a ≥ 0 and i - n + 1 ≤ 0 (because i ≤ n-1), this is always satisfied.

So for ALL a in [0, n-1], k = i - a + (n-1) is valid (in [0, 2n-2]).

Similarly for j: for all a, k = j - a + (n-1) is valid.

So the update is simply:
- For all a: C[a, i - a + (n-1)] -= 1
- For all a: C[a, j - a + (n-1)] += 1

This is O(n) per swap, which is fast.

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def build_C(h, n):
    idx = np.arange(n)[:, None] + np.arange(2 * n - 1)[None, :] - (n - 1)
    mask = (idx >= 0) & (idx < n)
    C = np.where(mask, 1.0 - h[np.clip(idx, 0, n - 1)], 0.0)
    return C

def optimize_ls(h, n, deadline, max_stalls=200, seed=42):
    h = binarize(h, n).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    C = build_C(h, n)
    m = np.arange(n)
    stalls = 0
    prev_best_c = best_c
    impatient = 0

    while time.time() < deadline:
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h >= 0.5)[0]
        nz, no = len(zeros), len(ones)
        if nz == 0 or no == 0:
            break

        vals = corr[None, None, :] + C[zeros][:, None, :] - C[ones][None, :, :]
        new_max = vals.max(axis=2)
        flat_min = int(np.argmin(new_max))
        bi, bj = np.unravel_index(flat_min, new_max.shape)
        best_nc = float(new_max[bi, bj]) * dx

        if best_nc >= cur_c - 1e-14:
            stalls += 1
            if stalls > max_stalls:
                n_pert = rng.randint(3, min(12, nz, no))
                for _ in range(n_pert):
                    zi = np.where(h < 0.5)[0]
                    oi = np.where(h >= 0.5)[0]
                    if len(zi) == 0 or len(oi) == 0:
                        break
                    a, b_ = int(rng.choice(zi)), int(rng.choice(oi))
                    h[a], h[b_] = h[b_], h[a]
                corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
                C = build_C(h, n)
                cur_c = float(corr.max()) * dx
                stalls = 0
                impatient += 1
                if impatient > 8:
                    break
                continue
        else:
            if cur_c < prev_best_c:
                impatient = 0
            prev_best_c = cur_c
            stalls = 0

        i, j = int(zeros[bi]), int(ones[bj])
        h[i] = 1.0
        h[j] = 0.0
        corr += C[i] - C[j]
        C[n - 1 - m, i + m] -= 1.0
        C[n - 1 - m, j + m] += 1.0
        cur_c = best_nc
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

    return best_h, best_c

def gen_diverse(n, seed):
    rng = np.random.RandomState(seed)
    inits = []
    b = np.zeros(n)
    b[rng.choice(n, n // 2, replace=False)] = 1.0
    inits.append(b.copy())
    b2 = np.zeros(n)
    step = max(2, n // 60)
    for i in range(0, n, step):
        b2[i:min(i + step // 2, n)] = 1.0
    if b2.sum() > n // 2:
        idx = np.where(b2 == 1)[0]
        b2[rng.choice(idx, int(b2.sum() - n // 2), replace=False)] = 0
    elif b2.sum() < n // 2:
        idx = np.where(b2 == 0)[0]
        b2[rng.choice(idx, int(n // 2 - b2.sum()), replace=False)] = 1
    inits.append(b2.copy())
    b3 = np.zeros(n)
    phi = (1.0 + 5.0 ** 0.5) / 2.0
    for i in range(n):
        b3[i] = 1.0 if (i * phi) % 1.0 < 0.5 else 0.0
    if b3.sum() != n // 2:
        idx = np.argsort(b3)[::-1]
        b3 = np.zeros(n)
        b3[idx[:n // 2]] = 1.0
    inits.append(b3.copy())
    b4 = np.zeros(n)
    b4[rng.permutation(n)[:n // 2]] = 1.0
    inits.append(b4.copy())
    return inits
