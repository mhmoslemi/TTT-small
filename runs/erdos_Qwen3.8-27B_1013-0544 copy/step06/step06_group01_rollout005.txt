import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_cd(h, n, max_passes=500, deadline=None, top_k=15):
    h = h.copy()
    c = np.correlate(h, 1-h, mode='full')
    current_max = float(np.max(c))
    dx = 2.0/n
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            t = h[p]
            if t < 1e-10 or t > 1-1e-10:
                continue
            k_lo = n - 1 - p
            k_hi = 2*n - 2 - p
            um = 0.0
            if k_lo > 0:
                um = float(np.max(c[:k_lo]))
            if k_hi < 2*n - 2:
                um = max(um, float(np.max(c[k_hi+1:])))
            ks = np.arange(k_lo, k_hi+1)
            slopes = np.zeros(len(ks))
            for idx in range(len(ks)):
                s = ks[idx] - (n-1)
                if s == 0:
                    slopes[idx] = 1 - 2*t
                elif s > 0:
                    slopes[idx] = 1 - h[p+s]
                else:
                    slopes[idx] = -h[p+s]
            vals = c[ks]
            # Get top correlations
            top_idx = np.argsort(vals)[-top_k:]
            candidates = [0.0, -t, 1-t, (1-2*t)/2.0]
            for i in range(len(top_idx)):
                for j in range(i+1, len(top_idx)):
                    ai, aj = top_idx[i], top_idx[j]
                    ds = slopes[ai] - slopes[aj]
                    if abs(ds) > 1e-12:
                        d_ij = (vals[aj] - vals[ai]) / ds
                        if -t+1e-12 < d_ij < 1-t-1e-12:
                            candidates.append(d_ij)
                if abs(slopes[top_idx[i]]) > 1e-12:
                    d_um = (um - vals[top_idx[i]]) / slopes[top_idx[i]]
                    if -t+1e-12 < d_um < 1-t-1e-12:
                        candidates.append(d_um)
            best_d = 0.0
            best_val = current_max
            for d in candidates:
                d = np.clip(d, -t, 1-t)
                if abs(d) < 1e-13:
                    continue
                new_c = vals + d * slopes
                di = np.where(ks == (n-1))[0]
                if len(di) > 0:
                    new_c[di[0]] = c[n-1] + d*(1-2*t) - d*d
                val = max(um, float(np.max(new_c)))
                if val < best_val - 1e-14:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                for idx in range(len(ks)):
                    s = ks[idx] - (n-1)
                    if s == 0:
                        c[ks[idx]] += best_d*(1-2*t) - best_d*best_d
                    elif s > 0:
                        c[ks[idx]] += best_d*(1-h[p+s])
                    else:
                        c[ks[idx]] -= best_d*h[p+s]
                h[p] = t + best_d
                current_max = max(um, float(np.max(c[k_lo:k_hi+1])))
                improved = True
        if not improved:
            break
    return h, current_max * dx

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * dx)
    return float((m + np.log(np.sum(np.exp(sh))) / alpha) * dx)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha=100000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def optimize_candidate(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    
    # Phase 1: Exact coordinate descent
    alloc = min(200, (deadline - time.time()) * 0.5)
    if alloc > 30:
        h_cd, c_cd = exact_cd(h, n, max_passes=1000, deadline=time.time()+alloc, top_k=15)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            h, best_c = h_cd, c_cd
    
    # Phase 2: SLSQP
    if time.time() < deadline - 60:
        for a in [50000, 200000, 500000]:
            if time.time() > deadline - 40:
                break
            h_s = slsqp(h, n, alpha=a, maxiter=2000)
            c_s = compute_c5(h_s, n)
