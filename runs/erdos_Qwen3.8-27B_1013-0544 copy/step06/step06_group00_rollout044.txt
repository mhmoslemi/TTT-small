def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    shift = corr - corr.max()
    p = np.exp(alpha * shift)
    p /= p.sum()
    # p has length 2n-1, indexed from -(n-1) to (n-1)
    # gradient[j] = dx * [(1 - sum_l p[l] * h[j+l]) - sum_l p[l] * h[j-l]]
    # where h is 0 outside [0, n)

    # l ranges from -(n-1) to (n-1)
    # h[j+l] is valid when 0 <= j+l < n, i.e., -j <= l < n-j
    
    S_plus = np.zeros(n)
    S_minus = np.zeros(n)
    for l_idx in range(2*n-1):
        l = l_idx - (n-1)
        if abs(l) >= n or p[l_idx] < 1e-15:

        if l >= 0:

            S_plus[:n-l] += p[l_idx] * h[l:]

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):

            break

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1-h, mode='full') * dx
    shift = corr - corr.max()

        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha

        return x.sum() - n / 2.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = dx * 2.0
    momentum = np.zeros(n)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    stall = 0
    for step in range(2000000):
        if time.time() > deadline:
            break
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            if lag < n:
                grad[:n - lag] += (1.0 - h[lag:])
                grad[lag:] -= h[:n - lag]
        else:
            ll = -lag
            if ll < n:
                grad[ll:] += (1.0 - h[:n - ll])
                grad[:n - ll] -= h[ll:]
        grad *= dx
        momentum = mu * momentum - lr * grad
        h_new = project_feasible(h + momentum, n)
        corr = np.correlate(h_new, 1 - h_new, mode='full') * dx
        c = float(corr.max())
        if c < best_c - 1e-10:
            best_c = c
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.02, dx * 20.0)

def smooth_grad_descent(h, n, alpha, deadline, mu=0.85):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = 1.0 / (alpha * dx)

        sh = corr - corr.max()
        p = np.exp(alpha * sh)
        ps = p.sum()
        if ps < 1e-30:

        p /= ps
        S_plus = np.zeros(n)

            if time.time() > t_end - 10:
                break
            h6 = slsqp_smooth(h6, n, a, 3000, 1e-17)
            c6 = compute_c5(h6, n)
            if c6 < best_c:
                best_c, best_h = c6, h6.copy()

    # Phase 7: Try interpolated lower resolution and back
    if time.time() < deadline - 30 and n > 200:
        nl = n // 2
        xo = np.linspace(0, 2, n)
        xl = np.linspace(0, 2, nl)
        hl = np.interp(xl, xo, best_h)
        hl = project_feasible(hl, nl)
        hl = slsqp_smooth(hl, nl, 100000, 1000, 1e-15)
        hl = slsqp_smooth(hl, nl, 2000000, 1000, 1e-15)
        hl_hi = np.interp(xo, xl, hl)
        hl_hi = project_feasible(hl_hi, n)
        hl_hi = slsqp_smooth(hl_hi, n, 1000000, 1000, 1e-15)
        ch = compute_c5(hl_hi, n)
        if ch < best_c:
            best_c, best_h = ch, hl_hi.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
