def projected_subgradient(h, n, n_iters=10000, eta0=0.01, top_k=5):
    dx = 2.0 / n
    h = h.copy().astype(float)
    for t in range(n_iters):
        corr = np.correlate(h, 1-h, mode='full')
        # Get top-k lags
        top = np.argsort(corr)[-top_k:]
        # Average subgradient over top lags
        grad = np.zeros(n)
        for k in top:
            k = int(k)  # lag index in full correlation
            # In full correlation, index k corresponds to lag k - (n-1)
            lag = k - (n-1)
            # C(lag) = sum_i h[i+lag](1-h[i]) for valid i
            # grad_j = sum over contributions
            if lag >= 0:
                # h[i+lag] for i in [0, n-lag-1]
                # For position j: appears as h[i+lag] when i = j-lag, i.e., j in [lag, n-1]
                #   contribution: (1-h[j-lag])
                # appears as h[i] when i = j, j in [0, n-lag-1]
                #   contribution: -(h[j+lag])

                for j in range(n-lag):
                    grad[j] -= h[j+lag]

                #   contribution: (1-h[j+lag])
                # appears as h[i+lag] when i+lag=j, i=j-lag, j in [lag, n-1]

                    grad[j] += (1 - h[j+lag])
                for j in range(lag, n):
                    grad[j] -= h[j-lag]
        
        grad /= len(top)

        h = h - eta * grad
        h = np.clip(h, 0, 1)

        h = project_feasible(h, n)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    # Project h onto {0<=h<=1, sum(h)=n/2}
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, eta, top_k=5):
    """One step of projected subgradient method."""
    corr = np.correlate(h, 1-h, mode='full')
    top = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for m in top:
        lag = int(m) - (n-1)
        # Term 1: (1-h[j+lag]) for valid j
        if lag >= 0:
            grad[:n-lag] += (1 - h[lag:])
        else:
            grad[-lag:] += (1 - h[:n+lag])
        # Term 2: -h[j-lag] for valid j
        if lag >= 0:
            grad[lag:] -= h[:n-lag]
        else:
            grad[:n+lag] -= h[-lag:]
    grad /= len(top)
    h = h - eta * grad
    return np.clip(h, 0, 1), corr

def subgradient_polish(h, n, n_iters=5000, eta0=0.05, top_k=5, deadline=None):
    """Polish h using projected subgradient method."""
