```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single_lag(h, n, k):
    """Gradient of C(k) wrt h, vectorized."""
    dx = 2.0 / n
    nvalid = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nvalid] += dx * (1.0 - h[k:])
        grad[k:] += -dx * h[:nvalid]
    else:
        k_abs = -k
        grad[k_abs:] += dx * (1.0 - h[:nvalid])
        grad[:nvalid] += -dx * h[k_abs:]
    return grad

def subgradient_descent(h, n, deadline, lr_init=0.005, top_m=5):
    """Subgradient method with exact max, focusing on top-m lags."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    stale = 0
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        # Find top-m lags
        top_indices = np.argsort(c)[-top_m:][::-1]
        top_max = c[top_indices[0]]
        # Subgradient: average gradient of top lags weighted by how close they are to max
        diffs = top_max - c[top_indices]
        weights = np.exp(diffs * 10.0)  # sharp weighting
        weights /= weights.sum()
        g = np.zeros(n)
        for idx, w in zip(top_indices, weights):
            k = idx - (n - 1)
            g += w * grad_single_lag(h, n, k)
        # Also add a small projection gradient for constraint
        constraint_violation = h.sum() - n / 2.0
        g += 0.1 * constraint_violation
        # Gradient step with backtracking
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        improved = False
        for attempt in range(30):
            h_new = h - lr * g
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            lr *= 0.7
        if not improved:
            lr = max(lr, lr_init * 0.1)
            stale += 1
            if stale > 200:
                break
        else:
            stale = 0
            lr = min(lr * 1.05, 0.05)
    return best_h, best_c
```

```python
def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    dx = 2.0 / n
    c_shift = alpha * c
    mx = c_shift.max()
    exp_c = np.exp(c_shift - mx)

    return grad

    mx = sh.max()
    if mx > 500:

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2

            if best_local != old_val:
                h[i] = best_local
                h = project_feasible(h, n)
                if compute_c5(h, n) < best_c - 1e-14:
                    best_c = compute_c5(h, n)
                    best_h = h.copy()
                    improved_any = True
                else:
                    h[i] = old_val
        if not improved_any:
            # Try finer candidates
            for i in order:
                if time.time() > deadline:
                    break
                old_val = h[i]
                for delta in [0.01, -0.01, 0.005, -0.005]:
                    cv = old_val + delta
                    if cv < 0 or cv > 1:
                        continue
                    h[i] = cv
                    h_tmp = project_feasible(h, n)
                    c_new = compute_c5(h_tmp, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        improved_any = True
                    else:
                        h[i] = old_val
            if not improved_any:
                break
    return best_h, best_c

def pair_search(h, n, deadline, steps=None):
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    for d in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i+1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _grad_single(h, n, k):
    dx = 2.0 / n
    nv = n - abs(k)
    g = np.zeros(n)
    if k >= 0:
        g[:nv] += dx * (1.0 - h[k:])
        g[k:] += -dx * h[:nv]
    else:
        ka = -k
        g[ka:] += dx * (1.0 - h[:nv])
        g[:nv] += -dx * h[ka:]
    return g

def _grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    cs = alpha * c
    mx = cs.max()
    ec = np.exp(cs - mx)
    w = ec / ec.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-13:
            continue
        k = m - (n - 1)
        grad += wm * _grad_single(h, n, k)
    return grad

def _obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _jobj(x, n, alpha, mu):
    g = _grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [500, 5000, 50000, 200000, 1000000, 3000000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_obj, best_h, jac=_jobj, args=(n, alpha, 5.0),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 400, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 4000, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def subgrad_phase(h, n, deadline, top_m=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.008
    stale = 0
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        top_idx = np.argsort(c)[-top_m:][::-1]
        top_max = c[top_idx[0]]
        diffs = top_max - c[top_idx]
        wts = np.exp(diffs * 80.0)
        wts /= wts.sum()
        g = np.zeros(n)
        for idx, wt in zip(top_idx, wts):
            k = idx - (n - 1)
            g += wt * _grad_single(h, n, k)
        cv = h.sum() - n / 2.0
        g += 0.3 * cv
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        improved = False
        for _ in range(50):
            h_new = project_feasible(h - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            lr *= 0.6
        if improved:
            stale = 0
            lr = min(lr * 1.08, 0.04)
        else:
            stale += 1
            lr = max(lr, 0.002)
            if stale > 80:

    return best_h, best_c

def pair_search(h, n, deadline, steps=None, radius=None):

                 0.00005, 0.00002, 0.00001]

                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)

                        else:

                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def grad_pair_search(h, n, deadline, step=0.003):
    best_c = compute_c5(h, n)
    for _ in range(200):

            for j in range(n):

                gj = g[j]

                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:

                        else:
                            h[i] -= step; h[j] += step
```
