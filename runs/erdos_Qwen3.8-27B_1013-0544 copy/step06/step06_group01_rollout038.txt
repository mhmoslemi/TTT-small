import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def full_values(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_feasible(y):
    y = np.asarray(y, dtype=float)
    if not np.all(np.isfinite(y)):
        y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    n = y.size
    m = 0.5 * n

    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        z = np.clip(y - mid, 0.0, 1.0)
        s = float(z.sum())
        if s > m:
            lo = mid
        else:
            hi = mid

    z = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(25):
        d = m - float(z.sum())
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            room = 1.0 - z
            tot = float(room.sum())
            if tot < 1e-15:
                break
            z += (d / tot) * room
        else:
            tot = float(z.sum())
            if tot < 1e-15:
                break
            z += (d / tot) * z
    return np.clip(z, 0.0, 1.0)


def resample(h, n_to):
    n_from = len(h)
    if n_to == n_from:
        return project_feasible(h)
    x0 = np.linspace(0.0, 2.0, n_from, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n_to, endpoint=False)
    y = np.interp(x1, x0, h)
    return project_feasible(y)


def _top_obj(x, args):
    n, qdata, dx, scale = args
    g = 1.0 - x
    total = 0.0
    for _, L, sgn, w in qdata:
        if w == 0.0:
            continue
        if L == 0:
            total += w * float(np.dot(x, g))
        elif sgn > 0:
            total += w * float(np.dot(x[L:], g[:-L]))
        else:
            total += w * float(np.dot(x[:-L], g[L:]))
    if scale <= 1e-300:
        return 0.0
    return (total * dx) / scale


def _top_jac(x, args):
    n, qdata, dx, scale = args
    grad = np.zeros(n, dtype=float)
    for _, L, sgn, w in qdata:
        if w == 0.0:
            continue
        if L == 0:
            grad += w * (1.0 - 2.0 * x)
        elif sgn > 0:
            grad[L:] += w * (1.0 - x[:-L])
            grad[:-L] -= w * x[L:]
        else:
            grad[:-L] += w * (1.0 - x[L:])
            grad[L:] -= w * x[:-L]
    if scale <= 1e-300:
        return grad
    return grad * (dx / scale)


def _cons_fun(x, args):
    n = args[0]
    return x.sum() - 0.5 * n


def _cons_jac(x, args):
    return np.ones_like(x, dtype=float)


def make_top(vals, K, n):
    L = len(vals)
    k = int(min(max(1, K), L))
    if k >= L:
        idx = list(range(L))
    else:
        part = np.argpartition(vals, L - k)[-k:]
        idx = part.tolist()

    s = set(idx)
    maxv = float(vals.max())
    thresh = maxv - 0.0015
    cap = min(L, k + 64)
    extra = np.flatnonzero(vals >= thresh)
    for q in extra:
        if len(s) >= cap:
            break
        s.add(int(q))

    idx = sorted(s)
    qdata = tuple(
        (int(q), abs(int(q) - (n - 1)), 1 if int(q) >= n - 1 else -1, 1.0)
        for q in idx
    )
    scale = float(len(idx))
    if scale > 32.0:
        scale = 32.0
    return qdata, scale


def make_soft(vals, n, beta, cap=120):
    L = len(vals)
    m = float(vals.max())
    sh = beta * (vals - m)
    sh -= float(sh.max())
    w = np.exp(sh)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 1e-300:
        order = np.argsort(-vals)
        idx = [int(order[i]) for i in range(max(4, min(cap, L)))]
        qdata = tuple(
            (int(q), abs(int(q) - (n - 1)), 1 if int(q) >= n - 1 else -1, 1.0)
            for q in idx
        )
        scale = float(min(len(idx), 32))
        return qdata, scale

    w = w / s
    order = np.argsort(w)[::-1]
    idx = []
    cum = 0.0
    for q in order:
        qi = int(q)
        if len(idx) >= cap or cum >= 0.95:
            break
        wq = float(w[qi])
        if wq > 1e-6:
            idx.append(qi)
            cum += wq

    if len(idx) == 0:
        idx = [int(order[i]) for i in range(max(4, min(cap, L)))]
        qdata = tuple(
            (int(q), abs(int(q) - (n - 1)), 1 if int(q) >= n - 1 else -1, 1.0)
            for q in idx
        )
        scale = float(min(len(idx), 32))
        return qdata, scale

    scale = 0.0
    for q in idx:
        scale += float(w[q])
    if scale <= 1e-300:
        scale = 1.0 / float(len(idx))

def slsqp_call(h, n, qdata, scale, maxiter):
    if len(qdata) == 0 or maxiter <= 0:
        return h
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}]
    args = (n, qdata, 2.0 / float(n), scale)
    try:
        res = minimize(
            _top_obj,
            h,
            args=args,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': 1e-10, 'disp': False},
        )
        x = res.x
        if not np.all(np.isfinite(x)):
            return h
        return project_feasible(x)
    except Exception:
        return h

def pair_polish(h, n, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(98765)
    deltas = (0.01, 0.005, 0.002, 0.001, 0.0005)
    max_trials = min(1500, max(80, int(max(0.0, deadline - time.time()) * 25)))
    trials = 0
    while trials < max_trials and time.time() < deadline - 2:
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        bi = h[i]
        bj = h[j]
        accepted = False
        for d in deltas:
            if accepted:
                break
            for sgn in (1, -1):
                if accepted:
                    break
                dd = sgn * d
                vi = bi + dd
                vj = bj - dd
                if vi >= -1e-12 and vi <= 1.0 + 1e-12 and vj >= -1e-12 and vj <= 1.0 + 1e-12:
                    vi = float(np.clip(vi, 0.0, 1.0))
                    vj = float(np.clip(vj, 0.0, 1.0))
                    h[i] = vi
                    h[j] = vj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-13:
                        best_c = c
                        best_h = h.copy()
                        accepted = True
                    else:
                        h[i] = bi
                        h[j] = bj
        trials += 1
    return best_h, float(best_c)

            h_t = resample(h_m, n_base)
            t_t = min(t_cand, time.time() + 50.0)
            h_t, c_t = improve(h_t, n_base, t_t, profile='quick')
            if c_t < c_best:
                h_best, c_best, n_best = h_t, c_t, n_base

    if time.time() < t_cand - 40:
        if n_best != n_base:
            h_source = resample(h_best, n_base)
        else:
            h_source = project_feasible(h_best)

        starts = candidate_starts(h_source, n_base, rng, c_best)
        for si, hs in enumerate(starts):
            if time.time() > t_cand - 20:
                break
            left = t_cand - time.time()
            if left < 25:
                break
            per = left / float(max(1, len(starts) - si))
            per = max(18.0, min(45.0, per))
            hs, cs = improve(hs, n_base, time.time() + per, profile='quick')
            if cs < c_best:
                h_best, c_best, n_best = hs, cs, n_base

    if time.time() < deadline - 20:
        profile_final = 'normal' if n_best >= 120 else 'quick'
        h_f, c_f = improve(h_best, n_best, deadline - 8.0, profile=profile_final)
        if c_f < c_best:
            h_best, c_best, n_best = h_f, c_f, n_best

    if time.time() < deadline - 25:
        h_p, c_p = pair_polish(h_best, n_best, deadline - 4.0)
        if c_p < c_best:
            h_best, c_best = h_p, c_p

    h_best = project_feasible(h_best)
    c_best = compute_c5(h_best, n_best)
    return h_best, float(c_best), int(n_best)
