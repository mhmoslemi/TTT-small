import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n-m] -= h[m:]
        grad[m:] += (1.0 - h[:n-m])
    else:
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] -= h[m:]
    return grad

def sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def sum_constraint_jac(x, n):
    return np.ones(n)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_polish(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_descent_polish(h, n, deadline, top_k=1):
    """Fine gradient descent on the top active lags."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]
    
    for ss in step_sizes:
        for _ in range(500):
            if time.time() > deadline:
                return best_h, best_c
            c = compute_all_lags(h, n)
            idx = np.argsort(c)[-top_k:]
            grad = np.zeros(n)
            for i in idx:
                lag = i - (n - 1)
                grad += lag_gradient(h, n, lag)
            grad /= len(idx)
            
            gn = np.linalg.norm(grad)
            if gn < 1e-18:
                break
            
            h_trial = h - ss * grad / gn
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-16:
                best_c = c_trial
                best_h = h_trial.copy()
                h = h_trial
            elif c_trial < best_c + 1e-14:
                pass  #neutral, continue
            else:
                # Try smaller step
                h_trial2 = h - ss * 0.1 * grad / gn
                h_trial2 = project_feasible(h_trial2, n)
                c_trial2 = compute_c5(h_trial2, n)
                if c_trial2 < best_c - 1e-16:
                    best_c = c_trial2
                    best_h = h_trial2.copy()
                    h = h_trial2
                else:
                    break
    
    return best_h, best_c

def active_lag_pairwise(h, n, deadline):
    """Targeted pairwise moves on the active lag."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for delta_base in [0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        for _ in range(20):
            if time.time() > deadline:
                return best_h, best_c
            c = compute_all_lags(h, n)
            max_idx = int(np.argmax(c))
            lag = max_idx - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                break
            
            improved = False
            delta = delta_base
            while delta > 1e-8:
                if time.time() > deadline:
                    return best_h, best_c
                any_improved = False
                if lag > 0:
                    for i in range(0, n - m, max(1, (n-m)//500)):
                        if time.time() > deadline:
                            return best_h, best_c
                        di = min(delta, (h[i] - h[i+m]) * 0.5)
                        if di > 1e-10:
                            h[i] -= di
                            h[i+m] += di
                            cn = compute_c5(h, n)
                            if cn < best_c - 1e-16:
                                best_c = cn
                                best_h = h.copy()
                                any_improved = True
                            else:
                                h[i] += di
                                h[i+m] -= di
                else:
                    for i in range(0, n - m, max(1, (n-m)//500)):
                        if time.time() > deadline:
                            return best_h, best_c
                        di = min(delta, (h[i+m] - h[i]) * 0.5)
                        if di > 1e-10:
                            h[i] += di
                            h[i+m] -= di
                            cn = compute_c5(h, n)
                            if cn < best_c - 1e-16:
                                best_c = cn
                                best_h = h.copy()
                                any_improved = True
                            else:
                                h[i] -= di
                                h[i+m] += di
                if not any_improved:
                    delta *= 0.5
                    if delta > 1e-8:
                        improved = True
            if not improved:
                break
    
    return best_h, best_c

def multi_start_refine(h_best, n, deadline, rng):
    """Try tiny perturbations and refine."""
    best_c = compute_c5(h_best, n)
    best_h = h_best.copy()
    
    for scale in [0.002, 0.001, 0.0005]:
        attempts = 0
        while time.time() < deadline - 10:
            attempts += 1
            if attempts > 200:
                break
            hp = h_best + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            c0 = compute_c5(hp, n)
            if c0 < best_c + 0.001:  # Only refine if not too far
                h_r, c_r = gradient_descent_polish(hp, n, deadline - 10)
                if c_r < best_c:
                    best_c = c_r
                    best_h = h_r.copy()
                    h_best = h_r.copy()
    
    return best_h, best_c

    if time.time() < deadline - 30:

    # Phase 6: Final SLSQP with very high alpha

    if time.time() < deadline - 3:

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n

Let me think about what would actually help:

1. The SLSQP with log-sum-exp is the main workhorse. The alpha parameter controls how "peaked" the approximation is. Higher alpha → closer to the true max, but harder to optimize.

2. The key insight: at the optimum, the active lag has a specific structure. The gradient of the active lag at the optimum should be "flat" over the free variables (those not at 0 or 1 bound). This is the KKT condition.

3. So a good polishing strategy is:
   a. Identify the active lag
   b. Check if the gradient is flat over free variables
   c. If not, move in the direction that makes it flatter (i.e., reduce the variance of the gradient over free variables)

Actually, the KKT condition says: at the optimum, ∇c_k* · d = 0 for all feasible directions d (i.e., d that maintains the constraint sum(h)=n/2 and 0≤h≤1).

For free variables (0<h[j]<1), the constraint is sum_j∈free d[j] = 0. So we need ∇c_k*[j] = λ for all j in free (where λ is the Lagrange multiplier).

This means: the gradient of the active lag should be constant over the free variables at the optimum.

So a good local optimization is:
1. Compute the gradient of the active lag
2. Over the free variables, compute the mean and variance of this gradient
3. Move in a direction that reduces the variance (while maintaining the sum constraint)

The direction that reduces variance while maintaining sum=0 is: d[j] = -(grad[j] - mean_grad) for free j, scaled appropriately.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def _sum_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_polish(h, n, deadline, top_k=1):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for ss in step_sizes:
        for _ in range(800):
            if time.time() > deadline:
                return best_h, best_c
            c = compute_all_lags(h, n)
            idx = np.argsort(c)[-top_k:]
            grad = np.zeros(n)
            for i in idx:
                lag = i - (n - 1)
                grad += lag_gradient(h, n, lag)
            grad /= len(idx)
            gn = np.linalg.norm(grad)
            if gn < 1e-18:
                break
            h_trial = project_feasible(h - ss * grad / gn, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-16:
                best_c = c_trial
                best_h = h_trial.copy()
                h = h_trial
            else:
                h_trial2 = project_feasible(h - ss * 0.1 * grad / gn, n)
                c_trial2 = compute_c5(h_trial2, n)
                if c_trial2 < best_c - 1e-16:
                    best_c = c_trial2
                    best_h = h_trial2.copy()
                    h = h_trial2
                else:
                    break
    return best_h, best_c

def gradient_flatten(h, n, deadline):
    """KKT-inspired: flatten gradient of active lag over free variables."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for _ in range(300):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        max_idx = int(np.argmax(c))
        lag = max_idx - (n - 1)
        grad = lag_gradient(h, n, lag)
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() < 2:
            break
        gf = grad[free]
        mean_g = gf.mean()
        var_g = gf.var()
        if var_g < 1e-20:
            break
        direction = np.zeros(n)
        direction[free] = -(gf - mean_g)
        dn = np.linalg.norm(direction)
        if dn < 1e-18:
            break
        direction /= dn
        for ss in [0.001, 0.0005, 0.0001, 0.00005]:
            h_trial = project_feasible(h + ss * direction, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-16:
                best_c = c_trial
                best_h = h_trial.copy()
                h = h_trial
                break
    return best_h, best_c

def active_pairwise(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for delta_base in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
        for _ in range(15):
            if time.time() > deadline:
                return best_h, best_c
            c = compute_all_lags(h, n)
            max_idx = int(np.argmax(c))
            lag = max_idx - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                break
            improved = False
            delta = delta_base
            while delta > 1e-9:
                if time.time() > deadline:
                    return best_h, best_c
                any_imp = False
                if lag > 0:
                    for i in range(n - m):
                        di = min(delta, (h[i] - h[i + m]) * 0.5)
                        if di > 1e-12:
                            h[i] -= di
                            h[i + m] += di
                            cn = compute_c5(h, n)
                            if cn < best_c - 1e-16:
                                best_c = cn
                                best_h = h.copy()
                                any_imp = True
                            else:
                                h[i] += di
                                h[i + m] -= di
                else:
                    for i in range(n - m):
                        di = min(delta, (h[i + m] - h[i]) * 0.5)
                        if di > 1e-12:
                            h[i] += di
                            h[i + m] -= di
                            cn = compute_c5(h, n)
                            if cn < best_c - 1e-16:
                                best_c = cn
                                best_h = h.copy()
                                any_imp = True
                            else:
                                h[i] -= di
                                h[i + m] += di
                if not any_imp:
                    delta *= 0.5
                else:
                    improved = True
            if not improved:
                break
    return best_h, best_c

def multi_start(h_best, n, deadline, rng):
    best_c = compute_c5(h_best, n)
    best_h = h_best.copy()
    for scale in [0.003, 0.001, 0.0005]:
        attempts = 0
        while time.time() < deadline - 10 and attempts < 100:
            attempts += 1
            hp = h_best + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            c0 = compute_c5(hp, n)
            if c0 < best_c + 0.002:
                h_r, c_r = gradient_polish(hp, n, deadline - 10, top_k=2)
                if c_r < best_c:
                    best_c = c_r
                    best_h = h_r.copy()
                    h_best = h_r.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()

    for a in [3000, 10000, 50000, 200000, 500000]:
        if time.time() > deadline - 100:

        remaining = deadline - time.time() - 50

        alloc = min(40, remaining * 0.2)
        if alloc < 15:

        h_new = slsqp_optimize(best_h, n, a, maxiter=4000)
        c_new = compute_c5(h_new, n)

    if time.time() < deadline - 70:
        best_h, best_c = gradient_polish(best_h, n, min(time.time() + 70, deadline - 40), top_k=1)

    # Phase 3: Gradient flattening (KKT-inspired)
    if time.time() < deadline - 50:

    if time.time() < deadline - 30:
        best_h, best_c = gradient_polish(best_h, n, min(time.time() + 25, deadline - 15), top_k=2)

    # Phase 6: Multi-start with tiny perturbations
    if time.time() < deadline - 25:

    # Phase 7: Final SLSQP polish
    if time.time() < deadline - 15:

        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 8: Final gradient polish
    if time.time() < deadline - 3:
        best_h, best_c = gradient_polish(best_h, n, deadline - 1, top_k=1)

    best_h = project_feasible(best_h, n)
