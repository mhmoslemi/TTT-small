```python
                            h = project_feasible(h, n)

            # overlap_m = sum_{i=0}^{n-m-1} h[i]*(1-h[i+m])

                    if d < 1e-8:

                        c = c_check

                j = i_best + m  # this is the "source" (h[j] is large, h[j-m] is small)

                        c = c_check
                    else:
                        h[j] += d
                        h[i] -= d
    
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def _obj_full(x, n, alpha):
    c = compute_all_lags(x, n)
    return smooth_max(c, alpha)

def _obj_top(x, n, alpha, top_idx):
    c = compute_all_lags(x, n)[top_idx]
    return smooth_max(c, alpha)

def _cons_eq(x, n):
    return x.sum() - n / 2.0

def _cons_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_full, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return h.copy()

            lag = k_star - (n - 1)

                # Find position with h closest to 0.5 and push it

                if abs(h[mid_idx] - 0.5) < 0.01:

                # Push towards 0 or 1
                if h[mid_idx] < 0.5:
                    d = min(delta, h[mid_idx])
                    if d < 1e-8:

                    high_idx = np.argmax(h)
```

```python
def binary_push(h, n, deadline):
    """Try to push h values towards 0 or 1."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for _ in range(10000):
        if time.time() > deadline:
            break
        # Find position closest to 0.5
        mid = np.argmin(np.abs(h - 0.5))
        if h[mid] < 0.15 or h[mid] > 0.85:
            break  # Already quite binary
        
        # Try pushing to 0 or 1
        target = 0.0 if h[mid] < 0.5 else 1.0
        d = h[mid] - target  # amount to move h[mid]
        
        # Find compensation position
        # If pushing mid to 0, we need to add d somewhere (find a low position)
        # If pushing mid to 1, we need to remove d somewhere (find a high position)
        if target == 0.0:
            comp = np.argmin(h)
            if comp == mid:
                comp = np.argmin(np.where(h == h[mid], 1, h))
            if h[comp] + d > 1.0:
                d = 1.0 - h[comp]
            if d < 1e-8:
                break
            h[mid] -= d
            h[comp] += d
        else:
            comp = np.argmax(h)
            if comp == mid:
                comp = np.argmax(np.where(h == h[mid], 0, h))
            if h[comp] - d < 0.0:
                d = h[comp]
            if d < 1e-8:
                break
            h[mid] += d
            h[comp] -= d
        
        h = np.clip(h, 0, 1)
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
        else:
            break  # Can't improve further with binary push
    
    return h, best_c
```

```python
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

    c = compute_all_lags(x, n)
```
