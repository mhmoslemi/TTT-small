def fast_local_search(h, n, deadline, max_sweeps=50000):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.int32)
    max_corr = int(corr.max())
    best_corr = max_corr
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        # Find critical lags (within 2 of max)
        critical_mask = corr >= max_corr - 2
        critical_idx = np.where(critical_mask)[0]
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        found = False
        for ia in ones:
            if found or time.time() > deadline:
                break
            # Precompute contribution of removing ia
            # For each critical lag k (in the full correlation index space):
            #   lag = k - (n-1)
            #   delta_from_ia[k] = -(1-h[ia+lag]) - h[ia-lag] (with bounds)
            
            for jb in zeros:
                if found or time.time() > deadline:
                    break
                # Check if this swap improves all critical lags
                feasible = True
                for k_idx in critical_idx:
                    lag = k_idx - (n - 1)
                    d = 0
                    # ia becomes 0
                    if 0 <= ia + lag < n:
                        d -= (1 - h[ia + lag])
                    if 0 <= ia - lag < n:
                        d -= h[ia - lag]
                    # jb becomes 1
                    if 0 <= jb + lag < n:
                        d += (1 - h[jb + lag])
                    if 0 <= jb - lag < n:
                        d -= h[jb - lag]
                    
                    if corr[k_idx] + d >= max_corr:
                        feasible = False
                        break
                
                if feasible:
                    # Apply the swap
                    h[ia] = 0
                    h[jb] = 1
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    max_corr = int(corr.max())
                    if max_corr < best_corr:
                        best_corr = max_corr
                        best_h = h.copy()
                    found = True
                    break
        
        if not found:
            # Try harder: allow temporary degradation
            # Or break and restart
            break
    
    return best_h, best_corr * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-11:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

            h = np.clip(h + diff / s * room, 0.0, 1.0)

            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)

def sa_binary_search(h_init, n, deadline, max_iter=500000, seed=42):
    """Simulated annealing for binary solutions."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    
    T_start = 3.0
    T_end = 0.01
    n_iter = max_iter
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        # Cooling schedule
        T = T_start * (T_end / T_start) ** (it / n_iter)
        
        # Random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        ia = ones[rng.randint(len(ones))]
        jb = zeros[rng.randint(len(zeros))]
        
        # Compute delta for all lags
        # delta[k] for lag = k - (n-1)
        # Remove ia (1->0): contributes -(1-h[ia+lag]) - h[ia-lag] to corr[lag]
        # Add jb (0->1): contributes (1-h[jb+lag]) - h[jb-lag] to corr[lag]
        
        # Vectorized computation
        lags = np.arange(2*n - 1) - (n - 1)  # range: -(n-1) to n-1

        ia_plus = ia + lags
        ia_minus = ia - lags
        valid_ia_plus = (ia_plus >= 0) & (ia_plus < n)
        valid_ia_minus = (ia_minus >= 0) & (ia_minus < n)
        
        delta_ia = np.zeros(2*n - 1)
        # h[ia](1-h[ia+lag]) was 1*(1-h[ia+lag]), now 0 → change: -(1-h[ia+lag])
        delta_ia[valid_ia_plus] = -(1 - h[ia_plus[valid_ia_plus]])
        # h[ia-lag](1-h[ia]) was h[ia-lag]*1, now h[ia-lag]*0 → change: -h[ia-lag]
        delta_ia[valid_ia_minus] -= h[ia_minus[valid_ia_minus]]
        
        # For jb: positions jb+lag and jb-lag

        delta_jb[valid_jb_plus] = (1 - h[jb_plus[valid_jb_plus]])

This is related to the theory of "low autocorrelation" binary sequences, which is a well-studied problem in signal processing. Nottingham sequences, Golay sequences, and m-sequences are all examples.

For our specific problem (minimizing the maximum of the cross-correlation between h and 1-h), this is equivalent to minimizing the maximum of the "aperiodic autocorrelation" of the sequence (2h-1), which takes values in {-1, +1}.

Let me define s[i] = 2*h[i] - 1, so s[i] ∈ {-1, +1}.

The aperiodic autocorrelation of s at lag k is:

And: R(k) = (number of matches) - (number of mismatches)
= (|{i: s[i]=s[i+k]}|) - (|{i: s[i]≠s[i+k]}|)
= (n-|k|) - 2 * (|{i: s[i]≠s[i+k]}|)

And |{i: s[i]=+1, s[i+k]=-1}| = overlap(k) (our quantity of interest)
And |{i: s[i]=-1, s[i+k]=+1}| = overlap(-k) (by symmetry of the correlation)

Actually, for the aperiodic autocorrelation:

And the number of valid pairs is n - |k|.

Number of mismatches = (n-|k|) - (number of matches)

By symmetry of the problem (we can reverse the sequence), overlap(k) and overlap(-k) aren't necessarily equal. But their sum is determined by R(k).

To minimize max(overlap(k), overlap(-k)), we want to minimize the mismatches for the lags with the most mismatches, AND make overlap(k) ≈ overlap(-k).

This is a well-defined optimization problem on ±1 sequences.

    h = h.copy()

    lags = np.arange(-(n-1), n)  # length 2n-1
    offset = n - 1  # to convert to index

    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    
    # Precompute index arrays
    # For position p, the lags where p+k and p-k are valid
    # p+k valid: -(p) <= k <= n-1-p
    # p-k valid: p-n+1 <= k <= p
    
    T_start = 2.0
    T_end = 0.005
    cooldown = max_iter  # cool over this many iterations
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        T = T_start * (T_end/T_start) ** (it / cooldown)

        ones = np.where(h == 1)[0]

        # (terms out of bounds are 0)
        
        k_vals = lags
        
        # ia + k valid: 0 <= ia+k < n → -ia <= k < n-ia
        mask_ia_plus = (ia + k_vals >= 0) & (ia + k_vals < n)

        mask_ia_minus = (ia - k_vals >= 0) & (ia - k_vals < n)
        # jb + k valid
        mask_jb_plus = (jb + k_vals >= 0) & (jb + k_vals < n)

        mask_jb_minus = (jb - k_vals >= 0) & (jb - k_vals < n)

        delta[mask_ia_minus] -= h[ia - k_vals[mask_ia_minus]]

            h[jb] = 1

            cur_max = new_max

                best_h = h.copy()

    cycles = reheat
    iter_per = max_iter // cycles
    
    for c in range(cycles):
        T_s = T_start * (0.6 ** c)
        T_e = max(T_end * (0.5 ** c), 0.001)
        
        for it in range(iter_per):
            if time.time() > deadline:
                break
            
            T = T_s * (T_e / T_s) ** (it / iter_per)
            
            # Random swap
            ia = rng.randint(n)
            while h[ia] != 1:
                ia = rng.randint(n)
            jb = rng.randint(n)
            while h[jb] != 0:
                jb = rng.randint(n)
            
            if ia == jb:
                continue
            
            # Precomputed slice starts
            sa = n - 1 - ia
            sb = ia
            sc = n - 1 - jb
            sd = jb
            
            # Apply delta
            corr[sa:sa+n] -= h_inv
            corr[sb:sb+n] -= h_rev
            corr[sc:sc+n] += h_inv
            corr[sd:sd+n] -= h_rev
            
            new_max = float(corr.max())
            
            # Accept
            if new_max <= cur_max or (T > 1e-12 and rng.random() < np.exp(-(new_max - cur_max) / T)):
                h[ia] = 0.0; h[jb] = 1.0
                h_inv[ia] = 1.0; h_inv[jb] = 0.0
                h_rev[n-1-ia] = 0.0; h_rev[n-1-jb] = 1.0
                cur_max = new_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
            else:
                # Undo
                corr[sa:sa+n] += h_inv
                corr[sb:sb+n] += h_rev
                corr[sc:sc+n] -= h_inv
                corr[sd:sd+n] += h_rev
        if time.time() > deadline:
            break
    
    return best_h, best_max * dx

def slsqp_polish(h, n, deadline, maxiter=2000):
    """SLSQP continuous polish."""
    dx = 2.0 / n
    alpha = 50000.0
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        mx = shifted.max()
        return m + (mx + np.log(np.exp(shifted - mx).sum())) / alpha
    
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        mx = shifted.max()
        weights = np.exp(shifted - mx)
        weights /= weights.sum()
        # d/dx of g[k] = d/dx of sum_i x[i](1-x[i+lag[k]]) * dx
        # = dx * sum_i (1 - 2*x[i+lag[k]] or 1-2*x[i-lag[k]] depending on which term)
        # This is complex; use numerical gradient
        return None  # Let SLSQP use numerical
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h
