def build_grad_matrix(h,n, include_zero=True):
    h = np.asarray(h,float)
    m = 2*n - 1
    B = np.zeros((m,n), dtype=float)
    for l in range(m):

            B[l, t:] -= h[:n-t]
            B[l, :t] -= h[t:]
    if not include_zero:

```python
import time
import numpy as np
from scipy.optimize import minimize, linprog

_OPT = {'n': 250, 'alpha': 50.0}
_CACHE = {'g': None}


def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    target = 0.5 * n
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    s = 0.5 * (lo + hi)
    for _ in range(60):
        s = 0.5 * (lo + hi)
        ss = float(np.clip(h - s, 0.0, 1.0).sum())
        if ss > target:
            lo = s
        else:
            hi = s
    return np.clip(h - s, 0.0, 1.0)


def corr_counts(h, n):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode='full')


def c5(h, n):
    return float(np.max(corr_counts(h, n)) * (2.0 / n))


def binarize(h, n):
    out = np.zeros(n, dtype=float)
    k = int(n // 2)
    if k > 0:
        out[np.argpartition(h, n - k)[-k:]] = 1.0
    return out


def tm_start(n, seed=42):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    k = n // 2
    ones = np.where(t == 1)[0]
    rng = np.random.RandomState(seed)
    if len(ones) > k:
        rng.shuffle(ones)
        t[ones[:len(ones) - k]] = 0
    elif len(ones) < k:
        z = np.where(t == 0)[0]
        rng.shuffle(z)
        t[z[:k - len(ones)]] = 1
    return t.astype(float)


def upsample(h, n, m):
    h = np.asarray(h, dtype=float).copy()
    if m == n:
        return project_feasible(h, m)
    if m % n == 0:
        h = np.repeat(h, m // n)
    else:
        a = np.linspace(0.0, 1.0, n, endpoint=False)
        b = np.linspace(0.0, 1.0, m, endpoint=False)
        h = np.interp(b, a, h)
    return project_feasible(h, m)


def build_g(h, n, zero=True):
    h = np.asarray(h, dtype=float)
    m = 2 * n - 1
    B = np.zeros((m, n), dtype=float)
    for l in range(m):
        s = n - 1 - l
        if s > 0:
            B[l, :n - s] = 1.0 - h[s:]
            B[l, s:] -= h[:n - s]
        elif s < 0:
            t = -s
            B[l, t:] = 1.0 - h[:n - t]
            B[l, :t] -= h[t:]
        elif zero:
            B[l, :] = 1.0 - 2.0 * h
    if not zero:
        B[n - 1, :] = 0.0
    return B


def lse_val_grad(h, n, a):
    h = np.asarray(h, dtype=float)
    f = corr_counts(h, n)
    M = float(f.max())
    sh = a * (f - M)
    sm = float(sh.max())
    e = np.exp(sh - sm)
    Z = float(e.sum())
    p = e / Z
    val = M + float(np.log(Z)) / a
    g = np.zeros(n, dtype=float)
    act = np.where(p > 1e-10)[0]
    for l in act:
        w = float(p[l])
        s = n - 1 - int(l)
        if s > 0:
            g[:n - s] += w * (1.0 - h[s:])
            g[s:] -= w * h[:n - s]
        elif s < 0:
            t = -s
            g[t:] += w * (1.0 - h[:n - t])
            g[:t] -= w * h[t:]
        else:
            g += w * (1.0 - 2.0 * h)
    return float(val), g


def _obj(x):
    x = np.asarray(x, dtype=float)
    val, g = lse_val_grad(x, int(_OPT['n']), float(_OPT['alpha']))
    _CACHE['g'] = g
    return val


def _jac(x):
    g = _CACHE['g']
    if g is None:
        _, g = lse_val_grad(np.asarray(x, dtype=float), int(_OPT['n']), float(_OPT['alpha']))
    return g


def _con(x):
    return float(np.sum(x) - 0.5 * int(_OPT['n']))


_CONS = [{'type': 'eq', 'fun': _con}]


def slsqp_run(h, n, alpha, maxiter):
    h = np.asarray(h, dtype=float).copy()
    _OPT['n'] = int(n)
    _OPT['alpha'] = float(alpha)
    _CACHE['g'] = None
    bounds = [(0.0, 1.0)] * int(n)
    try:
        res = minimize(_obj, h, jac=_jac, method='SLSQP', bounds=bounds,
                       constraints=_CONS,
                       options={'maxiter': int(max(1, maxiter)), 'ftol': 1e-8, 'disp': False})
        if hasattr(res, 'x') and res.x is not None:
            return np.asarray(res.x, dtype=float)
    except Exception:
        pass
    return h


def best_swap(corr, cols, zeros, ones):
    if len(zeros) == 0 or len(ones) == 0:
        return float(corr.max()), -1, -1
    best_val = float(corr.max())
    best_i = int(zeros[0])
    best_j = int(ones[0])
    o = len(ones)
    m = corr.size
    if o * m:
        chunk = max(1, min(len(zeros), 1000000 // (o * m)))
    else:
        chunk = 1
    co = cols[ones]
    for a in range(0, len(zeros), chunk):
        zb = zeros[a:a + chunk]
        vals = (corr[None, None, :] + cols[zb, None, :] - co[None, :, :]).max(axis=2)
        flat = int(np.argmin(vals))
        r = flat // o
        c = flat % o
        v = float(vals[r, c])
        if v < best_val:
            best_val = v
            best_i = int(zb[r])
            best_j = int(ones[c])
    return best_val, best_i, best_j


def binary_local(h, n, deadline):
    if n % 2 != 0:
        h = project_feasible(h, n)
        return h, c5(h, n)
    h = binarize(h, n)
    corr = corr_counts(h, n).astype(float)
    cur = float(corr.max())
    best_h = h.copy()
    best_c = cur
    cols = build_g(h, n, False)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    while time.time() < deadline - 2 and len(ones) and len(zeros):
        val, bi, bj = best_swap(corr, cols, zeros, ones)
        if bi < 0 or val >= cur - 1e-12:
            break
        corr = corr + cols[bi] - cols[bj]
        cur = float(corr.max())
        h[bi] = 1.0
        h[bj] = 0.0
        cols = build_g(h, n, False)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if cur < best_c - 1e-12:
            best_c = cur
            best_h = h.copy()
    return best_h, best_c


def binary_iterated(h, n, deadline, seed):
    if n % 2 != 0:
        h = project_feasible(h, n)
        return h, c5(h, n)
    rng = np.random.RandomState(seed)
    h = binarize(h, n)
    d1 = min(deadline, time.time() + max(10.0, 0.4 * (deadline - time.time())))
    best_h, best_c = binary_local(h, n, d1)
    state_h = best_h.copy()
    for _ in range(120):
        if time.time() > deadline - 25:
            break
        k = int(rng.choice([2, 2, 3, 3, 4]))
        ones = np.where(state_h == 1.0)[0]
        zeros = np.where(state_h == 0.0)[0]
        kk = min(k, len(ones), len(zeros))
        if kk <= 0:
            break
        iz = rng.choice(len(zeros), kk, replace=False)
        oj = rng.choice(len(ones), kk, replace=False)
        hp = state_h.copy()
        for a in range(kk):
            hp[int(iz[a])] = 1.0
            hp[int(oj[a])] = 0.0
        d2 = min(deadline - 10, time.time() + max(5.0, 0.12 * (deadline - time.time())))
        h2, c2 = binary_local(hp, n, d2)
        if c2 < best_c - 1e-12:
            best_c = c2
            best_h = h2.copy()
            state_h = h2.copy()
    return best_h, best_c


def lp_polish(h, n, deadline, max_lps=6, step=0.2):
    if time.time() >= deadline:
        return h, c5(h, n)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = c5(h, n)
    for _ in range(int(max_lps)):
        if time.time() > deadline - 3:
            break
        f = corr_counts(h, n)
        B = build_g(h, n, True)
        m = 2 * n - 1
        cnt = 1 + 2 * n
        cc = np.full(cnt, 1e-9, dtype=float)
        cc[0] = 1.0
        A = np.empty((m, cnt), dtype=float)
        A[:, 0] = -1.0
        A[:, 1:1 + n] = B
        A[:, 1 + n:] = -B
        b = -f
        Eq = np.zeros((1, cnt), dtype=float)
        Eq[0, 1:1 + n] = 1.0
        Eq[0, 1 + n:] = -1.0
        bounds = [(None, None)]
        for v in h:
            u = min(1.0 - float(v), float(step))
            bounds.append((0.0, max(0.0, u)))
        for v in h:
            u = min(float(v), float(step))
            bounds.append((0.0, max(0.0, u)))
        try:
            r = linprog(cc, A, b, A_eq=Eq, b_eq=np.zeros(1), bounds=bounds, method='highs')
        except Exception:
            break
        if r is None or r.x is None:
            break
        z = r.x[1:1 + n] - r.x[1 + n:]
        if float(np.abs(z).sum()) < 1e-10:
            break
        ok = False
        for sc in (1.0, 0.5, 0.25, 0.1, 0.05):
            hh = project_feasible(h + sc * z, n)
            ccv = c5(hh, n)
            if ccv < best_c - 1e-12:
                best_h = hh
                best_c = ccv
                h = hh
                ok = True
                break
        if not ok:
            break
    return best_h, best_c


def continuous_polish(h, n, deadline, maxit=None):
    if time.time() >= deadline:
        return project_feasible(h, n), c5(h, n)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = c5(h, n)
    rem0 = max(1.0, deadline - time.time())
    if maxit is None:
        if n <= 300:
            maxit = int(max(8, min(35, rem0 / 8.0)))
        else:
            maxit = int(max(8, min(22, rem0 / 12.0)))
    for a in (20.0, 50.0, 120.0, 300.0):
        if time.time() > deadline - 30:
            break
        hh = slsqp_run(h, n, a, maxit)
        hh = project_feasible(hh, n)
        cc = c5(hh, n)
        if cc < best_c - 1e-12:
            best_c = cc
            best_h = hh.copy()
            h = hh.copy()
        elif cc <= best_c + 1e-8:
            h = hh.copy()
        else:
            h = best_h.copy()
    for st in (0.05, 0.2, 0.5):
        if time.time() > deadline - 15:
            break
        hh, cc = lp_polish(h, n, deadline - 10, max_lps=5, step=st)
        if cc < best_c - 1e-12:
            best_c = cc
            best_h = hh.copy()
            h = hh.copy()
    if time.time() < deadline - 20:
        hh = slsqp_run(h, n, 500.0, max(6, maxit // 2))
        hh = project_feasible(hh, n)
        cc = c5(hh, n)
        if cc < best_c - 1e-12:
            best_c = cc
            best_h = hh.copy()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = max(60.0, float(budget_s) - 10.0)
    overall = t0 + float(budget_s) - 5.0
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 250
        h0 = tm_start(n0, seed)
    if n0 < 20:
        n0 = 250
        h0 = tm_start(n0, seed)
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_n = n0
    best_c = c5(h0, n0)

    d1 = min(overall, time.time() + 0.30 * B)
    h1, c1 = continuous_polish(h0, n0, d1, None)
    if c1 < best_c:
        best_h = h1.copy()
        best_n = n0
        best_c = c1

    if n0 % 2 == 0:
        d2 = min(overall, time.time() + 0.20 * B)
        h2, c2 = binary_iterated(h1 if c1 <= best_c + 1e-12 else h0, n0, d2, int(seed) + 1)
        if c2 < best_c:
            best_h = h2.copy()
            best_n = n0
            best_c = c2

    d3 = min(overall, time.time() + 0.15 * B)
    h3, c3 = continuous_polish(best_h if best_n == n0 else h1, n0, d3, None)
    if c3 < best_c:
        best_h = h3.copy()
        best_n = n0
        best_c = c3

    if best_n != 500 and overall - time.time() > 300.0:
        n5 = 500
        h5 = upsample(best_h, best_n, n5)
        rem = max(0.0, overall - time.time())
        h5a, c5a = continuous_polish(h5, n5, min(overall, time.time() + 0.45 * rem), None)
        if c5a < best_c:
            best_h = h5a.copy()
            best_n = n5
            best_c = c5a
        if overall - time.time() > 120.0:
            rem2 = max(0.0, overall - time.time())
            h5b, c5b = binary_iterated(h5a, n5, min(overall, time.time() + 0.35 * rem2), int(seed) + 101)
            if c5b < best_c:
                best_h = h5b.copy()
                best_n = n5
                best_c = c5b
        if overall - time.time() > 60.0:
            h5c, c5c = continuous_polish(best_h if best_n == n5 else h5a, n5, overall - 3.0, None)
            if c5c < best_c:
                best_h = h5c.copy()
                best_n = n5
                best_c = c5c

    if overall - time.time() > 25.0:
        hf, cf = continuous_polish(best_h, best_n, overall - 2.0, None)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
```
