    n=len(y); target=0.5*n
    lo=float(np.min(y))-1.0; hi=float(np.max(y))+1.0

```python
import numpy as np
import time
from scipy.optimize import minimize


def compute_all_lags(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def compute_c5(h):
    return float(np.max(compute_all_lags(h)))


def project_box_sum(y):
    y = np.asarray(y, dtype=float)
    n = len(y)
    target = 0.5 * n
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = (1.0 - h) > 1e-12
        else:
            mask = h > 1e-12
        if mask.any():
            h[mask] += diff / float(mask.sum())
        h = np.clip(h, 0.0, 1.0)
    return h


def value_single_lag(h, lag):
    li = int(lag)
    n = len(h)
    dx = 2.0 / n
    if li == 0:
        return dx * (float(h.sum()) - float(np.dot(h, h)))
    m = abs(li)
    if li > 0:
        a = h[:-m]
        b = h[m:]
    else:
        a = h[m:]
        b = h[:-m]
    return dx * (float(a.sum()) - float(np.dot(a, b)))


def lag_gradient_single(h, lag):
    li = int(lag)
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if li == 0:
        return dx * (1.0 - 2.0 * h)
    m = abs(li)
    if li > 0:
        g[:-m] += dx * (1.0 - h[m:])
        g[m:] -= dx * h[:-m]
    else:
        g[:-m] -= dx * h[m:]
        g[m:] += dx * (1.0 - h[:-m])
    return g


def values_for_lags(h, lags):
    n = len(h)
    dx = 2.0 / n
    out = np.empty(len(lags))
    sh = float(h.sum())
    sh2 = float(np.dot(h, h))
    for i, lg in enumerate(lags):
        li = int(lg)
        if li == 0:
            out[i] = dx * (sh - sh2)
            continue
        m = abs(li)
        if li > 0:
            a = h[:-m]
            b = h[m:]
        else:
            a = h[m:]
            b = h[:-m]
        out[i] = dx * (float(a.sum()) - float(np.dot(a, b)))
    return out


def weighted_gradient(h, lags, weights):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for i, lg in enumerate(lags):
        w = float(weights[i])
        if w <= 1e-14:
            continue
        li = int(lg)
        if li == 0:
            g += w * dx * (1.0 - 2.0 * h)
            continue
        m = abs(li)
        if li > 0:
            g[:-m] += w * dx * (1.0 - h[m:])
            g[m:] -= w * dx * h[:-m]
        else:
            g[:-m] -= w * dx * h[m:]
            g[m:] += w * dx * (1.0 - h[:-m])
    return g


def smooth_value(h, lags, beta):
    vals = values_for_lags(h, lags)
    m = float(np.max(vals))
    z = np.exp(beta * (vals - m))
    s = float(np.sum(z))
    if s <= 0.0:
        return m
    return m + float(np.log(s)) / beta


def smooth_round(h, deadline, r, beta, step0=15.0):
    n = len(h)
    h_x = project_box_sum(h)
    best_h = h_x.copy()
    best_c = compute_c5(h_x)
    lag_table = np.arange(2 * n - 1) - (n - 1)
    step = float(step0)
    stall = 0

    for _ in range(20000):
        if time.time() > deadline:
            break

        c = compute_all_lags(h_x)
        T = float(c.max())
        if T < best_c - 1e-13:
            best_h = h_x.copy()
            best_c = T
            stall = 0
        else:
            stall += 1
            if stall > 1000:
                break

        idx_top = np.argsort(c)[-r:]
        near = np.where(c >= T - max(5e-4, 0.01 * T))[0]
        idx = np.unique(np.concatenate((idx_top, near)))
        if len(idx) > r:
            idx = idx[np.argsort(c[idx])[-r:]]
        lags = lag_table[idx].astype(int)

        vals = values_for_lags(h_x, lags)
        m = float(np.max(vals))
        z = np.exp(beta * (vals - m))
        s = float(np.sum(z))
        obj = m + float(np.log(s)) / beta
        w = z / s
        g = weighted_gradient(h_x, lags, w)

        if float(np.linalg.norm(g)) < 1e-12:
            break

        step_ls = step
        accepted = False
        for __ in range(9):
            zt = project_box_sum(h_x - step_ls * g)
            oz = smooth_value(zt, lags, beta)
            if oz <= obj + 1e-12:
                h_x = zt
                accepted = True
                break
            step_ls *= 0.5

        if not accepted:
            step *= 0.5
            if step < 1e-7:
                break
            continue

        if step_ls >= 0.99 * step:
            step = min(5000.0, step * 1.05)
        else:
            step = max(1e-7, step_ls)

    c = compute_all_lags(h_x)
    T = float(c.max())
    if T < best_c:
        best_h = h_x.copy()
        best_c = T
    return best_h, best_c


def _slsqp_obj(x):
    return x[-1]


def _slsqp_obj_jac(x):
    j = np.zeros(len(x))
    j[-1] = 1.0
    return j


def _slsqp_sum(x):
    n = len(x) - 1
    return x[:n].sum() - 0.5 * n


def _slsqp_sum_jac(x):
    n = len(x) - 1
    j = np.zeros(len(x))
    j[:n] = 1.0
    return j


def _slsqp_lag_fun(x, lag):
    n = len(x) - 1
    return x[n] - value_single_lag(x[:n], lag)


def _slsqp_lag_jac(x, lag):
    n = len(x) - 1
    g = lag_gradient_single(x[:n], lag)
    j = np.zeros(len(x))
    j[:n] = -g
    j[n] = 1.0
    return j


def slsqp_active(h, deadline, max_rounds=3, max_active=10, maxiter=50):
    n = len(h)
    h_x = project_box_sum(h)
    best_h = h_x.copy()
    best_c = compute_c5(h_x)
    no_improve = 0
    lag_table = np.arange(2 * n - 1) - (n - 1)

    for _ in range(max_rounds):
        if time.time() > deadline:
            break

        c = compute_all_lags(h_x)
        T = float(c.max())
        if T < best_c - 1e-13:
            best_h = h_x.copy()
            best_c = T
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_rounds:
                break

        idx_top = np.argsort(c)[-max_active:]
        near = np.where(c >= T - max(2e-4, 0.005 * T))[0]
        idx = np.unique(np.concatenate((idx_top, near)))
        cap = max_active + 6
        if len(idx) > cap:
            idx = idx[np.argsort(c[idx])[-cap:]]
        lags = lag_table[idx].astype(int)

        cons = [{"type": "eq", "fun": _slsqp_sum, "jac": _slsqp_sum_jac}]
        for lg in lags:
            cons.append({
                "type": "ineq",
                "fun": _slsqp_lag_fun,
                "jac": _slsqp_lag_jac,
                "args": (int(lg),)
            })

        x0 = np.empty(n + 1)
        x0[:n] = h_x
        x0[n] = T + 0.004
        t_bound = max(2.0, T + 0.1)
        bounds = [(0.0, 1.0)] * n + [(0.0, t_bound)]

        try:
            res = minimize(
                _slsqp_obj,
                x0,
                jac=_slsqp_obj_jac,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": maxiter, "ftol": 1e-12},
            )
        except Exception:
            break

        if not hasattr(res, "x") or res.x is None:
            break
        if not np.all(np.isfinite(res.x)):
            break

        h_new = project_box_sum(res.x[:n])
        T_new = compute_c5(h_new)
        if T_new < best_c - 1e-12:
            best_h = h_new.copy()
            best_c = T_new
            h_x = h_new
            no_improve = 0
        elif T_new <= T + 1e-9:
            h_x = h_new

    return best_h, best_c


def pair_local(h, deadline, top_k=8):
    n = len(h)
    h = project_box_sum(h)
    best = compute_c5(h)
    no_improve = 0

    for _ in range(200):
        if time.time() > deadline:
            break

        c = compute_all_lags(h)
        T = float(c.max())
        if T < best - 1e-13:
            best = T
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                break
            if no_improve > 3:
                break

        idx = np.argsort(c)[-top_k:]
        cands = []
        for k in idx:
            li = int(k - (n - 1))
            if li == 0:
                continue
            m = abs(li)
            if li > 0:
                diff = h[:-m] - h[m:]
            else:
                diff = h[m:] - h[:-m]
            order = np.argsort(-diff)[:6]
            for off in order:
                i = int(off)
                j = i + m
                if li > 0:
                    if h[i] > h[j] + 1e-5:
                        cands.append((i, j))
                else:
                    if h[j] > h[i] + 1e-5:
                        cands.append((i, j))

        seen = set()
        improved = False
        for i, j in cands:
            if (i, j) in seen or (j, i) in seen:
                continue
            seen.add((i, j))
            seen.add((j, i))

            if h[i] > h[j]:
                hi = i
                lo = j
            else:
                hi = j
                lo = i

            d0 = 0.5 * (h[hi] - h[lo])
            if d0 <= 1e-8:
                continue

            hi_v = float(h[hi])
            lo_v = float(h[lo])
            for fr in (1.0, 0.5, 0.25, 0.125):
                d = fr * d0
                if d <= 1e-10:
                    break
                h[hi] = hi_v - d
                h[lo] = lo_v + d
                T2 = compute_c5(h)
                if T2 < best - 1e-13:
                    best = T2
                    improved = True
                    break
                h[hi] = hi_v
                h[lo] = lo_v

            if improved:
                break

        if not improved:
            break

    return project_box_sum(h), best


def resample_h(h, new_n):
    n = len(h)
    if new_n == n:
        return project_box_sum(h)
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return project_box_sum(np.interp(xn, xo, h))


def default_h(n, seed):
    k = n // 2
    r = np.arange(max(1, k))
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    pos = np.floor(((r * phi) % 1.0) * n).astype(int) % n
    h = np.zeros(n)
    h[pos] = 1.0
    if h.sum() < k:
        miss = np.where(h == 0)[0]
        h[miss[:int(k - h.sum())]] = 1.0
    elif h.sum() > k:
        order = np.argsort(h)
        h[order[:int(h.sum()) - k]] = 0.0
    return project_box_sum(h)


def binary_round(h):
    n = len(h)
    k = n // 2
    if k <= 0 or k >= n:
        return project_box_sum(np.full(n, 0.5))
    idx = np.argsort(h)[-k:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return project_box_sum(hb)


def optimize_n(h, deadline):
    h_best = project_box_sum(h)
    c_best = compute_c5(h_best)
    n = len(h)

    r = min(80, max(24, int(24000.0 / max(64, n))))
    if n <= 450:
        betas = [80, 400, 2000, 8000]
        step0 = 15.0
    else:
        betas = [200, 1000, 4000]
        step0 = 8.0

    for beta in betas:
        rem = deadline - time.time()
        if rem < 25:
            break
        end = time.time() + min(60.0, max(20.0, rem * 0.5))
        if end > deadline:
            end = deadline
        h2, c2 = smooth_round(h_best, end, r, beta, step0)
        if c2 < c_best - 1e-14:
            h_best, c_best = h2, c2

    rem = deadline - time.time()
    if rem > 80:
        maxiter = 50 if n <= 500 else 35
        end = time.time() + min(100.0, rem * 0.6)
        if end > deadline:
            end = deadline
        h2, c2 = slsqp_active(h_best, end, 3, 10, maxiter)
        if c2 < c_best - 1e-14:
            h_best, c_best = h2, c2

    rem = deadline - time.time()
    if rem > 35:
        end = time.time() + min(35.0, rem * 0.7)
        if end > deadline:
            end = deadline
        h2, c2 = pair_local(h_best, end, 8)
        if c2 < c_best - 1e-14:
            h_best, c_best = h2, c2

    h_best = project_box_sum(h_best)
    c_best = compute_c5(h_best)
    return h_best, c_best


def run(seed=42, budget_s=1100, **kwargs):
    try:
        B = float(budget_s)
    except Exception:
        B = 1100.0

    t0 = time.time()
    hard = t0 + max(1.0, B - 5.0)
    rng = np.random.RandomState(seed)

    try:
        iv = initial_h_values
    except NameError:
        iv = None

    if iv is not None:
        h0 = np.array(iv, dtype=float).copy()
        n_base = len(h0)
    else:
        n_base = 300
        h0 = default_h(n_base, seed)

    best_h = project_box_sum(h0)
    best_c = compute_c5(best_h)

    rem = hard - t0
    first_end = t0 + min(350.0, max(15.0, rem * 0.35))
    if first_end > hard:
        first_end = hard
    if first_end > t0:
        h1, c1 = optimize_n(best_h, first_end)
        if c1 < best_c - 1e-14:
            best_h, best_c = h1, c1

    processed = {len(best_h)}
    for mult in (1.2, 1.5, 2.0):
        nn = int(round(n_base * mult))
        if nn % 2:
            nn += 1
        if nn < 20 or nn >= 1000 or nn in processed:
            continue
        rem = hard - time.time()
        if rem < 180:
            break
        end = time.time() + min(300.0, rem * 0.45)
        if end > hard:
            end = hard
        hst = resample_h(best_h, nn)
        h2, c2 = optimize_n(hst, end)
        if c2 < best_c - 1e-14:
            best_h, best_c = h2, c2
        processed.add(nn)

    rem = hard - time.time()
    if rem > 150:
        hb = binary_round(best_h)
        cb = compute_c5(hb)
        if cb < best_c + 0.02:
            end = time.time() + min(140.0, rem * 0.45)
            if end > hard:
                end = hard
            h2, c2 = optimize_n(hb, end)
            if c2 < best_c - 1e-14:
                best_h, best_c = h2, c2

    rem = hard - time.time()
    if rem > 160:
        n = len(best_h)
        hp = project_box_sum(best_h + rng.normal(0.0, 0.025, n))
        end = time.time() + min(140.0, rem * 0.45)
        if end > hard:
            end = hard
        h2, c2 = optimize_n(hp, end)
        if c2 < best_c - 1e-14:
            best_h, best_c = h2, c2

    rem = hard - time.time()
    if rem > 45:
        h2, c2 = optimize_n(best_h, hard - 0.1)
        if c2 < best_c - 1e-14:
            best_h, best_c = h2, c2

    best_h = project_box_sum(best_h)
    best_c = compute_c5(best_h)
    return best_h, float(best_c), int(len(best_h))
```
