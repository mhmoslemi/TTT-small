def solve_epigraph(h,n,target,K,maxiter,deadline):
   global _ACT_S,_TARGET
   if time.time() > deadline: return h, compute_c5(h,n)
   slist,c5 = select_active(h,n,K,tol=0.03)

   t0=float(C.max()) + 0.02
   v0=np.empty(n+1); v0[:-1]=h0; v0[-1]=t0
   _ACT_S=slist; _TARGET=float(target)

def project_box_sum(h, target):
    y = np.asarray(h, dtype=float)
    low = float(y.min()) - 1.0
    high = float(y.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            low = mid
        else:
            high = mid
    y = np.clip(y - 0.5 * (low + high), 0.0, 1.0)
    diff = target - float(y.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = y < 1.0 - 1e-12
            cnt = int(mask.sum())
            if cnt > 0:
                y[mask] += diff / cnt
        else:
            mask = y > 1e-12
            cnt = int(mask.sum())
            if cnt > 0:
                y[mask] += diff / cnt
    y = np.clip(y, 0.0, 1.0)
    diff = target - float(y.sum())
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = y < 1.0
            if mask.any():
                i = int(np.argmin(1.0 - y))
                y[i] = min(1.0, y[i] + diff)
        else:
            mask = y > 0.0
            if mask.any():
                i = int(np.argmax(y))
                y[i] = max(0.0, y[i] + diff)
    return y

def select_active(h,n,cap,tol=0.1):
    C=np.correlate(h,1.0-h,'full')
    m=float(C.max())
    c5=m*(2.0/n)
    off=n-1
    order=np.argsort(C, kind='stable')[::-1]
    top=order[:int(cap)]
    near=np.where(C >= m - tol)[0]
    idx=np.unique(np.concatenate((top,near)))
    maxK=int(min(2*cap, len(C)))
    if len(idx) > maxK:
        idx=idx[np.argsort(C[idx], kind='stable')[::-1][:maxK]]
    s=off - idx if _S_MODE == 1 else idx - off
    return s.astype(int), c5

def resample_h(h,m):
   n=len(h)
   if m==n: return h.copy()
   if m % n ==0:
      return np.repeat(h, m//n)
   x_old=(np.arange(n)+0.5)/n
   x_new=(np.arange(m)+0.5)/m
   idx=np.searchsorted(x_old,x_new,side='right')-1
   idx=np.clip(idx,0,n-1)

            m = y > 0.0
            if m.any():
                y[m] += d / float(m.sum())
    y = np.clip(y, 0.0, 1.0)
    d = target - float(y.sum())
    if abs(d) > 1e-14:
        if d > 0:
            i = int(np.argmin(1.0 - y))
            y[i] = min(1.0, y[i] + d)
        else:
            i = int(np.argmax(y))
            y[i] = max(0.0, y[i] + d)
    return y

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * (2.0 / n)

def active_cvals(x):
    s_list = _ACT_S
    om = 1.0 - x
    n = len(x)
    vals = np.empty(len(s_list), dtype=float)
    for i in range(len(s_list)):
        s = int(s_list[i])
        if s == 0:
            vals[i] = float(np.dot(x, om))
        elif s > 0:
            vals[i] = float(np.dot(x[:n - s], om[s:]))
        else:
            t = -s
            vals[i] = float(np.dot(x[t:], om[:n - t]))
    return vals

def add_shift_grad(G, x, om, s, w):
    n = len(x)
    if s == 0:
        G += w * (om - x)
    elif s > 0:
        G[:n - s] += w * om[s:]
        G[s:] -= w * x[:n - s]
    else:
        t = -s
        G[t:] += w * om[:n - t]
        G[:n - t] -= w * x[t:]

def soft_value(vals, alpha):
    vals = np.asarray(vals, dtype=float)
    m = float(vals.max())
    a = float(alpha)
    if a <= 1e-8:
        return m
    e = np.exp(a * (vals - m))
    s = float(e.sum())
    if s <= 0.0:
        return m
    return m + float(np.log(s)) / a

def select_active(h, n, cap, tol=0.10):
    C = np.correlate(h, 1.0 - h, mode='full')
    m = float(C.max())
    off = n - 1
    order = np.argsort(C, kind='stable')[::-1]
    top = order[:int(cap)]
    near = np.where(C >= m - tol)[0]
    idx = np.unique(np.concatenate((top, near)))
    mk = int(min(2 * cap, len(C)))
    if len(idx) > mk:
        idx = idx[np.argsort(C[idx], kind='stable')[::-1][:mk]]
    s = off - idx
    return s.astype(int), m * (2.0 / n)

def epi_obj(v):
    return float(v[-1])

def epi_ineq(v):
    x = v[:-1]
    return float(v[-1]) - active_cvals(x)

def epi_ineq_jac(v):
    x = v[:-1]
    n = int(x.size)
    K = len(_ACT_S)
    J = np.ones((K, n + 1), dtype=float)
    om = 1.0 - x
    for r in range(K):
        G = np.zeros(n, dtype=float)
        add_shift_grad(G, x, om, int(_ACT_S[r]), 1.0)
        J[r, :n] = -G
    return J

def epi_sum(v):
    return float(v[:-1].sum() - _TARGET)

def epi_sum_jac(v):
    n = int(v.size - 1)
    J = np.zeros(n + 1, dtype=float)
    J[:n] = 1.0
    return J

def solve_epigraph(h, n, target, K, max_it, deadline):
    global _ACT_S, _TARGET
    if time.time() > deadline:
        return h, compute_c5(h, n)
    s_list, c5 = select_active(h, n, K)
    if s_list.size == 0:
        return h, c5
    h0 = project_box_sum(h, target)
    raw = c5 * n / 2.0
    t0 = raw + 0.02
    v0 = np.empty(n + 1, dtype=float)
    v0[:-1] = h0
    v0[-1] = t0
    _ACT_S = s_list
    _TARGET = float(target)
    bounds = [(0.0, 1.0)] * n + [(0.0, float(max(n, 1.2 * t0)))]
    cons = [
        {'type': 'eq', 'fun': epi_sum, 'jac': epi_sum_jac},
        {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac}
    ]
    try:
        res = minimize(epi_obj, v0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': int(max_it), 'ftol': 1e-11})
        x = project_box_sum(np.asarray(res.x, dtype=float)[:-1], target)
    except Exception:
        x = h0
    return x, compute_c5(x, n)

def active_pg(h, n, target, alpha, cap, max_iter, deadline, step=0.05):
    global _ACT_S, _ACT_ALPHA
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    cur_step = float(step)
    it = 0
    while it < max_iter and time.time() < deadline:
        s_list, c_now = select_active(h, n, cap)
        if c_now < cur_c:
            cur_c = c_now
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()
        if s_list.size == 0:
            break
        _ACT_S = s_list
        _ACT_ALPHA = float(alpha)
        vals = active_cvals(h)
        F = soft_value(vals, float(alpha))
        m = float(vals.max())
        e = np.exp(float(alpha) * (vals - m))
        ss = float(e.sum())
        if ss > 0.0:
            w = e / ss
        else:
            w = np.zeros(len(vals)); w[int(np.argmax(vals))] = 1.0
        G = np.zeros(n, dtype=float)
        om = 1.0 - h
        for i in range(len(s_list)):
            wk = float(w[i])
            if wk > 1e-12:
                add_shift_grad(G, h, om, int(s_list[i]), wk)
        if float(np.max(np.abs(G))) < 1e-9:
            break
        moved = None
        for _ in range(6):
            hnew = project_box_sum(h - cur_step * G, target)
            Fnew = soft_value(active_cvals(hnew), float(alpha))
            if Fnew < F - 1e-9:
                cnew = compute_c5(hnew, n)
                if cnew < cur_c + 1e-7:
                    moved = (hnew, cnew)
                    break
            cur_step *= 0.5
        if moved is not None:
            h, cnew = moved
            cur_c = cnew
            if cnew < best_c:
                best_c = cnew
                best_h = h.copy()
            cur_step = min(1.0, cur_step * 1.15)
            it += 1
        else:
            if cur_step < 1e-5:
                break
            break
    return best_h, best_c

def binarize(h, n, target):
    b = np.zeros(n, dtype=float)
    k = int(np.floor(target))
    if 0 < k < n:
        idx = np.argsort(h, kind='stable')[::-1][:k]
        b[idx] = 1.0
    rem = target - k
    if rem > 1e-12:
        m = b < 1.0
        if m.any():
            c = np.where(m)[0]
            b[c[int(np.argmax(h[c]))]] = min(1.0, rem)
    return b

def resample_h(h, m):
    n = len(h)
    if m == n:
        return h.copy()
    if m % n == 0:
        return np.repeat(h, m // n)
    x0 = (np.arange(n) + 0.5) / n
    x1 = (np.arange(m) + 0.5) / m
    idx = np.searchsorted(x0, x1, side='right') - 1
    idx = np.clip(idx, 0, n - 1)
    return h[idx].copy()

def optimize_epigraph(h, n, cur_c, best_h, best_c, target, cap, end_phase, deadline, seed):
    rng = np.random.RandomState(seed)
    cur_h = h.copy()
    stall = 0
    if n <= 300:
        Ks = (4, 8, 16)
    elif n <= 450:
        Ks = (4, 8, 12)
    else:
        Ks = (4, 8)
    passn = 0
    while time.time() < end_phase and time.time() < deadline - 20:
        prev = cur_c
        for K in Ks:
            if time.time() >= end_phase or time.time() >= deadline - 20:
                break
            rem = end_phase - time.time()
            mi = int(min(70 if n <= 300 else 45, max(15, rem * 0.5)))
            hb, cb = solve_epigraph(cur_h, n, target, K, mi, deadline)
            if cb < cur_c - 1e-12:
                cur_h, cur_c = hb, cb
                if cb < best_c:
                    best_c, best_h = cb, hb.copy()
        if time.time() < end_phase - 15 and time.time() < deadline - 20:
            hp = project_box_sum(cur_h + 0.015 * rng.standard_normal(n), target)
            mi = int(min(40, max(10, (end_phase - time.time()) * 0.25)))
            hb, cb = solve_epigraph(hp, n, target, 8, mi, deadline)
            if cb < cur_c - 1e-12:
                cur_h, cur_c = hb, cb
                if cb < best_c:
                    best_c, best_h = cb, hb.copy()
        if passn % 2 == 1 and time.time() < end_phase - 15 and time.time() < deadline - 20:
            hp = project_box_sum(binarize(cur_h, n, target) + 0.02 * rng.standard_normal(n), target)
            mi = int(min(30, max(8, (end_phase - time.time()) * 0.20)))
            hb, cb = solve_epigraph(hp, n, target, 8, mi, deadline)
            if cb < cur_c - 1e-12:
                cur_h, cur_c = hb, cb
                if cb < best_c:
                    best_c, best_h = cb, hb.copy()
        stall = stall + 1 if abs(cur_c - prev) < 1e-12 else 0
        if stall >= 3:
            break
        passn += 1
    return cur_h, cur_c, best_h, best_c

def optimize_pg(h, n, cur_c, best_h, best_c, target, cap, end_phase, deadline, seed):
    rng = np.random.RandomState(seed)
    cur_h = h.copy()
    for alpha in (20, 80, 300):
        if time.time() >= end_phase:
            break
        seg = min(end_phase, time.time() + 60.0)
        hb, cb = active_pg(cur_h, n, target, alpha, cap, 300, seg, 0.05)
        if cb < cur_c - 1e-12:
            cur_h, cur_c = hb, cb
            if cb < best_c:
                best_c, best_h = cb, hb.copy()
    if time.time() < end_phase - 15 and time.time() < deadline - 15:
        hp = project_box_sum(cur_h + 0.015 * rng.standard_normal(n), target)
        seg = min(end_phase, time.time() + 50.0)
        hb, cb = active_pg(hp, n, target, 80, cap, 200, seg, 0.04)
        if cb < cur_c - 1e-12:
            cur_h, cur_c = hb, cb
            if cb < best_c:
                best_c, best_h = cb, hb.copy()
    return cur_h, cur_c, best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 3.0
    total = max(0.0, float(budget_s) - 10.0)
    rng_seed = int(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        r0 = np.random.RandomState(rng_seed)
        h0 = np.zeros(n, dtype=float)
        h0[r0.choice(n, n // 2, replace=False)] = 1.0
    target = n / 2.0
    h0 = project_box_sum(h0, target)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    cur_h = h0.copy()
    cur_c = best_c

    if time.time() >= deadline - 5:
        best_h = project_box_sum(best_h, len(best_h) / 2.0)
        return best_h, float(compute_c5(best_h, len(best_h))), int(len(best_h))

    end1 = min(deadline - 30.0, t0 + 0.20 * total) if total > 30.0 else t0
    if end1 > t0 + 10:
        cap1 = 64 if n <= 800 else 48
        if n <= 600:
            cur_h, cur_c, best_h, best_c = optimize_epigraph(
                cur_h, n, cur_c, best_h, best_c, target, cap1, end1, deadline, rng_seed)
        else:
            cur_h, cur_c, best_h, best_c = optimize_pg(
                cur_h, n, cur_c, best_h, best_c, target, cap1, end1, deadline, rng_seed)

    if time.time() < deadline - 60 and len(best_h) < 1000:
        nb = len(best_h)
        if nb < 500:
            n2 = min(1000, max(500, nb * 2))
        else:
            n2 = 1000
        if n2 > nb and total > 80.0:
            h2 = resample_h(best_h, int(n2))
            target2 = int(n2) / 2.0
            h2 = project_box_sum(h2, target2)
            c2 = compute_c5(h2, int(n2))
            cur_h = h2
            cur_c = c2
            if c2 < best_c:
                best_c, best_h = c2, h2.copy()
            end2 = min(deadline - 30.0, t0 + 0.68 * total) if total > 60.0 else t0
            cap2 = 64 if int(n2) <= 800 else 48
            if end2 > time.time() + 10:
                if int(n2) <= 600:
                    cur_h, cur_c, best_h, best_c = optimize_epigraph(
                        cur_h, int(n2), cur_c, best_h, best_c, target2, cap2, end2, deadline, rng_seed + 1)
                else:
                    if time.time() < end2 - 120 and deadline - time.time() > 180:
                        for K in (4, 8):
                            if time.time() < end2 - 80 and deadline - time.time() > 120:
                                hb, cb = solve_epigraph(cur_h, int(n2), target2, K, 20, deadline - 60)
                                if cb < cur_c - 1e-12:
                                    cur_h, cur_c = hb, cb
                                    if cb < best_c:
                                        best_c, best_h = cb, hb.copy()
                    cur_h, cur_c, best_h, best_c = optimize_pg(
                        cur_h, int(n2), cur_c, best_h, best_c, target2, cap2,
                        min(end2, deadline - 20), deadline, rng_seed + 2)

    if time.time() < deadline - 5:
        fb = len(best_h)
        ft = fb / 2.0
        capf = 64 if fb <= 800 else 48
        if fb <= 600 and time.time() < deadline - 25:
            mi = int(min(35, max(10, (deadline - time.time()) * 0.35)))
            hb, cb = solve_epigraph(best_h, fb, ft, 8, mi, deadline - 3)
            if cb < best_c:
                best_c, best_h = cb, hb.copy()
        if time.time() < deadline - 5:
            hb, cb = active_pg(best_h, fb, ft, 100, capf, 300, deadline - 2, 0.02)
            if cb < best_c:
                best_c, best_h = cb, hb.copy()

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c = compute_c5(best_h, len(best_h))
    return best_h, float(best_c), int(len(best_h))
