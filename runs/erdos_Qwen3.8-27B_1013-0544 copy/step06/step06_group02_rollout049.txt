```python
import numpy as np
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topk_subgrad(h, n, k=10):
    """Compute subgradient using top-k lags."""
    c = compute_c5_vec(h, n)
    c_max = c.max()
    # Find indices where c is close to max
    thresh = c_max - 0.01  # within 0.01 of max
    idx = np.where(c >= thresh)[0]
    if len(idx) == 0:
        idx = np.array([np.argmax(c)])
    elif len(idx) > k:
        # Keep only top k
        c_idx = c[idx]
        top = np.argsort(c_idx)[-k:]
        idx = idx[top]
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in idx:
        wm = c[m]  # weight by the value itself (or just 1)
        wm = 1.0  # equal weight for top lags
        kk = m - (n - 1)
        wm_dx = wm * dx
        if kk >= 0:
            nv = n - kk
            grad[0:nv] += wm_dx * (1.0 - h[kk:n])
            grad[kk:n] += wm_dx * (-h[0:nv])
        else:
            kk = -kk
            nv = n - kk
            grad[kk:n] += wm_dx * (1.0 - h[0:nv])
            grad[0:nv] += wm_dx * (-h[kk:n])
    return grad, c_max

def projected_subgrad(h, n, deadline, topk=10, lr_init=0.05, patience=200):
    """Projected subgradient descent with backtracking."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    stagnation = 0
    
    for it in range(50000):
        if time.time() > deadline:
            break
        g, c_cur = topk_subgrad(h, n, topk)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Backtracking line search
        step = lr
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                stagnation = 0
                break
            step *= 0.5
        if not improved:
            stagnation += 1
            lr *= 0.8
            if stagnation > patience:
                break
    
    return best_h, best_c

def fast_pair_search(h, n, deadline, topk=5, max_rounds=5000):
    """Gradient-guided pair search: only try top candidates."""
    best_c = compute_c5(h, n)
    for round_idx in range(max_rounds):
        if time.time() > deadline:
            break
        g, _ = topk_subgrad(h, n, topk)
        # Sort coordinates by gradient (most negative first for increases, most positive for decreases)
        # We want to increase h where grad is negative and decrease where grad is positive
        order = np.argsort(g)
        # Try top pairs: increase h[order[i]], decrease h[order[j]]
        improved = False
        deltas = [0.01, 0.005, 0.002, 0.001]
        for d in deltas:
            if time.time() > deadline:
                return h, best_c
            # Try top 20 increases and top 20 decreases
            n_inc = 20
            n_dec = 20
            inc_idx = order[:n_inc]  # most negative gradient -> increase
            dec_idx = order[-n_dec:]  # most positive gradient -> decrease
            for i in inc_idx:
                for j in dec_idx:
                    if i == j:

                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:

                            improved = True
                        else:

            if improved:

        if not improved:

            if not improved:

    return h, best_c
```

```python
def solve_1d_convex(a, b, t_lo=0.0, t_hi=1.0):
    """Find t in [t_lo, t_hi] minimizing max_m (a[m]*t + b[m])."""
    n = len(a)
    # Sort by slope
    order = np.argsort(a)
    a_sorted = a[order]
    b_sorted = b[order]
    
    # Build upper envelope
    # Lines are sorted by slope. The upper envelope consists of a subset.
    # Use the standard convex hull trick.
    hull = []  # list of (a, b) on the upper envelope
    
    for i in range(n):
        ai, bi = a_sorted[i], b_sorted[i]
        # Remove lines from the back that are dominated
        while len(hull) >= 2:
            a1, b1 = hull[-2]
            a2, b2 = hull[-1]
            # Check if line (a2,b2) is never the max
            # Intersection of (a1,b1) and (a2,b2): t = (b1-b2)/(a2-a1)
            # Intersection of (a2,b2) and (ai,bi): t = (b2-bi)/(ai-a2)
            if a2 == a1:
                # Parallel lines, keep the one with larger intercept
                if bi > b2:
                    hull.pop()
                else:
                    break
            else:
                t1 = (b1 - b2) / (a2 - a1)
                t2 = (b2 - bi) / (ai - a2) if ai != a2 else np.inf
                if t2 <= t1:
                    hull.pop()
                else:
                    break
        if len(hull) == 0 or a_sorted[i] != hull[-1][0]:
            hull.append((ai, bi))
        else:
            # Same slope, keep larger intercept
            if bi > hull[-1][1]:
                hull[-1] = (ai, bi)
    
    # Now find the minimum of the upper envelope over [t_lo, t_hi]
    # The upper envelope is convex, so the minimum is where slope crosses 0
    best_t = t_lo
    best_val = np.max(a * t_lo + b)
    
    for i in range(len(hull)):
        ai, bi = hull[i]
        # This line is active for t in some interval
        # Find the interval where this line is on the envelope
        if i == 0:
            t_start = t_lo
        else:
            a_prev, b_prev = hull[i-1]
            if ai == a_prev:
                t_start = t_lo
            else:
                t_start = (b_prev - bi) / (ai - a_prev)
        
        if i == len(hull) - 1:
            t_end = t_hi
        else:
            a_next, b_next = hull[i+1]
            if ai == a_next:
                t_end = t_hi
            else:
                t_end = (bi - b_next) / (a_next - ai)
        
        t_start = max(t_start, t_lo)
        t_end = min(t_end, t_hi)
        
        if t_start > t_end:
            continue
        
        # On this interval, the function is linear: ai*t + bi
        # Minimum is at one of the endpoints
        for t in [t_start, t_end]:
            t = np.clip(t, t_lo, t_hi)
            val = ai * t + bi
            if val < best_val:
                best_val = val
                best_t = t
    
    return best_t, best_val
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topk_subgrad(h, n, k=8):
    c = compute_c5_vec(h, n)
    c_max = c.max()
    thresh = c_max - 0.005
    idx = np.where(c >= thresh)[0]
    if len(idx) == 0:
        idx = np.array([np.argmax(c)])
    elif len(idx) > k:
        c_idx = c[idx]
        top = np.argsort(c_idx)[-k:]
        idx = idx[top]
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in idx:
        kk = m - (n - 1)
        wm_dx = dx
        if kk >= 0:
            nv = n - kk
            grad[0:nv] += wm_dx * (1.0 - h[kk:n])
            grad[kk:n] += wm_dx * (-h[0:nv])
        else:
            kk2 = -kk
            nv = n - kk2
            grad[kk2:n] += wm_dx * (1.0 - h[0:nv])
            grad[0:nv] += wm_dx * (-h[kk2:n])
    return grad, c_max

def nesterov_subgrad(h, n, deadline, v0=100, momentum=0.9):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    lr = 0.1
    lr_min = 1e-8
    stall = 0
    for it in range(200000):
        if time.time() > deadline:
            break
        g, _ = topk_subgrad(h, n, k=8)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            h = project_feasible(h, n)
            stall += 1
            if stall % 5 == 0:
                lr *= 0.9
            if stall > v0:
                v *= 0.5
                lr = max(lr * 0.7, lr_min)
                if lr < lr_min:
                    break
    return best_h, best_c

def pair_exact_1d(h, n, i, j, n_samples=80):
    """Try moving mass from j to i (or vice versa) via 1D search."""
    h0 = h.copy()
    max_d = min(h[j], 1.0 - h[i])
    if max_d < 1e-12:
        return h0, None
    ds = np.linspace(-max_d, max_d, n_samples)
    best_c = compute_c5(h0, n)
    best_d = 0.0
    for d in ds:
        if abs(d) < 1e-14:
            continue
        if d > 0:
            if h[i] + d > 1.0 + 1e-12 or h[j] - d < -1e-12:
                continue
        else:
            if h[j] + d > 1.0 + 1e-12 or h[i] - d < -1e-12:
                continue
        h0[i] += d
        h0[j] -= d
        c = compute_c5(h0, n)
        if c < best_c - 1e-15:
            best_c = c
            best_d = d
    h0 = h.copy()
    if best_d != 0.0:
        h0[i] += best_d
        h0[j] -= best_d
    h0 = project_feasible(h0, n)
    return h0, best_d

def pair_descent(h, n, deadline, top_pairs=30, n_samples=60):
    best_c = compute_c5(h, n)
    for rnd in range(500):
        if time.time() > deadline:
            break
        g, _ = topk_subgrad(h, n, k=10)
        order = np.argsort(g)
        inc_idx = order[:top_pairs]
        dec_idx = order[-top_pairs:]
        improved = False
        for i in inc_idx:
            if time.time() > deadline + 10:
                break
            for j in dec_idx:
                if i == j:
                    continue
                h_try, d = pair_exact_1d(h, n, i, j, n_samples=n_samples)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-14:
                    h = h_try
                    best_c = c_try
                    improved = True
                    if d is not None and abs(d) > 0.05:
                        break
            if improved and time.time() > deadline - 2:
                break
        if not improved:
            break
    return h, best_c

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 20000]
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        def obj(x):
            c = compute_c5_vec(x, n)
            m = c.max()
            sh = alpha * (c - m)
            mx = sh.max()
            if mx > 500:
                return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
            return m + np.log(np.sum(np.exp(sh))) / alpha
        def grad(x):
            g, _ = topk_subgrad(x, n, k=15)
            return g
        try:
            t_sub = min(80, deadline - time.time() - 3)
            res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18, 'gtol': 1e-12,
                                   'maxfun': max_iters * 20, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

        h4, c4 = nesterov_subgrad(best_h, n, t4, v0=100, momentum=0.9)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Random restarts with perturbation
    if time.time() < deadline - 100:
        for _ in range(5):
            if time.time() > deadline - 60:
                break
            h_pert = best_h + rng.randn(n) * 0.02
            h_pert = project_feasible(h_pert, n)
            c_pert = compute_c5(h_pert, n)
            if c_pert < best_c + 0.01:
                t_r = min(time.time() + 40, deadline - 30)
                h_r, c_r = nesterov_subgrad(h_pert, n, t_r, v0=100, momentum=0.85)
                if c_r < best_c:
                    best_c = c_r
                    best_h = h_r

    # Phase 6: Try different n values
    if time.time() < deadline - 80:
        for n_try in [200, 250, 350, 400]:
            if time.time() > deadline - 40:
                break
            if n_try == n:
                continue
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(50, deadline - time.time() - 30)
            if alloc < 20:
                break
            h_t, c_t = nesterov_subgrad(h_interp, n_try, time.time() + alloc * 0.6, v0=100, momentum=0.85)
            if time.time() < deadline - 15:
                h_t, c_t = pair_descent(h_t, n_try, time.time() + min(20, deadline - time.time() - 10),
                                        top_pairs=20, n_samples=40)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n = n_try

    # Phase 7: Final polish
    if time.time() < deadline - 5:
        nf = len(best_h)
        if time.time() < deadline - 3:
            best_h, best_c = nesterov_subgrad(best_h, nf, deadline - 3, v0=80, momentum=0.9)
        if time.time() < deadline - 1:
            best_h, best_c = pair_descent(best_h, nf, deadline - 1, top_pairs=15, n_samples=30)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
