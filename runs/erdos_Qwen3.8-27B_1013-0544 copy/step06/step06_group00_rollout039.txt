def compute_c5(h,n):
    dx=2.0/float(n)
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))*dx)

def project_feasible(h,n):
    target=n/2.0
    h=np.clip(h,0.0,1.0)
    for _ in range(80):
        diff=target-h.sum()
        if abs(diff)<1e-13: break
        if diff>0:
            room=1.0-h
            s=room.sum()
            if s<1e-14: break
            h = h + diff/s*room
        else:
            s=h.sum()
            if s<1e-14: break
            h = h + diff/s*h
    return np.clip(h,0.0,1.0)

def project_polytope(x,n):
    target=n/2.0
    lo=float(np.min(x))-1.0; hi=float(np.max(x))+1.0
    for _ in range(40):
       mid=(lo+hi)/2.0
       s=np.clip(x-mid,0.0,1.0).sum()
       if s>target: lo=mid
       else: hi=mid
    y=np.clip(x-(lo+hi)*0.5,0.0,1.0)
    return project_feasible(y,n)

def smooth_fg(x,n,alpha,cut=50.0):

def compute_c5(h,n):
    dx=2.0/float(n)
    return float(np.max(np.correlate(h,1.0-h,mode='full'))*dx)

def project_feasible(h,n):
    target=n/2.0
    h=np.clip(h,0.0,1.0)
    for _ in range(100):
        diff=target-h.sum()
        if abs(diff)<1e-13:
            break
        if diff>0:
            room=1.0-h
            s=room.sum()
            if s<1e-14: break
            h=h+diff/s*room
        else:
            s=h.sum()
            if s<1e-14: break
            h=h+diff/s*h
    return np.clip(h,0.0,1.0)

def to_binary(h,n):
    b=np.zeros(n)
    b[np.argsort(h)[::-1][:n//2]]=1.0
    return b

def resample(h,n2):
    n=len(h)
    if n==n2:
        return h.copy()
    xo=(np.arange(n)+0.5)*2.0/float(n)
    xn=(np.arange(n2)+0.5)*2.0/float(n2)
    return project_feasible(np.interp(xn,xo,h),n2)

def smooth_fg(x,n,alpha,cut=40.0):
    dx=2.0/float(n)
    G=np.correlate(x,1.0-x,mode='full')*dx
    m=float(G.max())
    mask=G >= m - cut/alpha
    if not np.any(mask):
        mask=np.zeros(G.shape,bool); mask[int(G.argmax())]=True
    vals=G[mask]
    e=np.exp(alpha*(vals-m))
    se=e.sum()
    if se <= 0.0:
        e=np.ones_like(e); se=1.0
    F=m+float(np.log(se))/alpha
    grad=np.zeros(n)
    lags=np.arange(-(n-1),n)[mask]
    for e_i,k in zip(e,lags):
        w=float(e_i)*dx/se
        if k==0:
            grad += w*(1.0-2.0*x)
        elif k>0:
            s=int(k)
            grad[s:] += w*(1.0-x[:n-s])
            grad[:n-s] -= w*x[s:]
        else:
            s=int(-k)
            grad[:n-s] += w*(1.0-x[s:])
            grad[s:] -= w*x[:n-s]
    return F,grad

import time
import numpy as np
from scipy.optimize import minimize, linprog

def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    b = np.zeros(n)
    b[np.argsort(h)[::-1][:n//2]] = 1.0
    return b

def resample(h, n2):
    n = len(h)
    if n == n2:
        return h.copy()
    xo = (np.arange(n) + 0.5) * 2.0 / float(n)
    xn = (np.arange(n2) + 0.5) * 2.0 / float(n2)
    return project_feasible(np.interp(xn, xo, h), n2)

def smooth_fg(x, n, alpha, cut=40.0):
    dx = 2.0 / float(n)
    G = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(G.max())
    mask = G >= m - cut / alpha
    if not np.any(mask):
        mask = np.zeros(G.shape, bool)
        mask[int(G.argmax())] = True
    vals = G[mask]
    e = np.exp(alpha * (vals - m))
    se = float(e.sum())
    if se <= 0.0:
        e = np.ones_like(e)
        se = 1.0
    F = m + float(np.log(se)) / alpha
    grad = np.zeros(n)
    lags = np.arange(-(n - 1), n)[mask]
    for ei, k in zip(e, lags):
        w = float(ei) * dx / se
        if k == 0:
            grad += w * (1.0 - 2.0 * x)
        elif k > 0:
            s = int(k)
            grad[s:] += w * (1.0 - x[:n - s])
            grad[:n - s] -= w * x[s:]
        else:
            s = int(-k)
            grad[:n - s] += w * (1.0 - x[s:])
            grad[s:] -= w * x[:n - s]
    return F, grad

def smooth_fg_pen(x, n, alpha, pen):
    F, g = smooth_fg(x, n, alpha)
    if pen > 0.0:
        s = x.sum() - n / 2.0
        F = F + pen * s * s
        g = g + 2.0 * pen * s
    return F, g

def obj_fun(x, n, alpha, pen):
    return smooth_fg_pen(x, n, alpha, pen)[0]

def obj_jac(x, n, alpha, pen):
    return smooth_fg_pen(x, n, alpha, pen)[1]

def lag_gradient(h, n, k):
    dx = 2.0 / float(n)
    g = np.zeros(n)
    if k == 0:
        return dx * (1.0 - 2.0 * h)
    if k > 0:
        s = int(k)
        g[s:] += dx * (1.0 - h[:n - s])
        g[:n - s] -= dx * h[s:]
    else:
        s = int(-k)
        g[:n - s] += dx * (1.0 - h[s:])
        g[s:] -= dx * h[:n - s]
    return g

def solve_lp(A, b, bounds, cobj, Aeq):
    try:
        return linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=0.0, bounds=bounds, method='highs')
    except Exception:
        try:
            return linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=0.0, bounds=bounds)
        except Exception:
            return None

def binary_search(h, n, deadline, seed=42):
    h = to_binary(h, n)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur = float(c.max() * dx)
    best = cur
    best_h = h.copy()
    zeros = np.where(h == 0.0)[0]
    ones = np.where(h == 1.0)[0]
    if zeros.size == 0 or ones.size == 0 or time.time() > deadline - 1:
        return best_h, best
    omh = 1.0 - h
    rng = np.random.RandomState(seed)
    T = 0.008
    no_improve = 0
    while time.time() < deadline - 1.0:
        ai = int(rng.randint(zeros.size))
        bi = int(rng.randint(ones.size))
        i = int(zeros[ai])
        j = int(ones[bi])
        d = np.zeros(2 * n - 1)
        st_i = i - 1 if i > 0 else None
        st_j = j - 1 if j > 0 else None
        d[n - 1 + i : st_i : -1] += omh
        d[n - 1 : n - 1 + (n - i)] -= h[i:]
        d[n - 1 + j : st_j : -1] -= omh
        d[n - 1 : n - 1 + (n - j)] += h[j:]
        d[n - 1] -= 2.0
        d[n - 1 + i - j] += 1.0
        d[n - 1 + j - i] += 1.0
        nc = float((c + d).max() * dx)
        ex = (cur - nc) / T
        accept = False
        if ex >= 0.0:
            accept = True
        elif ex > -700.0 and float(rng.random()) < float(np.exp(ex)):
            accept = True
        if accept:
            h[i] = 1.0
            h[j] = 0.0
            c += d
            cur = nc
            zeros[ai] = j
            ones[bi] = i
            omh = 1.0 - h
            no_improve = 0
            if cur < best:
                best = cur
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 3000:
                h = to_binary(project_feasible(best_h + 0.05 * rng.randn(n), n))
                c = np.correlate(h, 1.0 - h, mode='full').astype(float)
                cur = float(c.max() * dx)
                zeros = np.where(h == 0.0)[0]
                ones = np.where(h == 1.0)[0]
                omh = 1.0 - h
                no_improve = 0
                T = 0.010
        T *= 0.99998
    return best_h, best

import time
import numpy as np
from scipy.optimize import minimize, linprog


def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)


def to_binary(h, n):
    b = np.zeros(n)
    b[np.argsort(h)[::-1][:n // 2]] = 1.0
    return b


def resample(h, n2):
    n = len(h)
    if n == n2:
        return h.copy()
    xo = (np.arange(n) + 0.5) * 2.0 / float(n)
    xn = (np.arange(n2) + 0.5) * 2.0 / float(n2)
    return project_feasible(np.interp(xn, xo, h), n2)


def smooth_fg(x, n, alpha, cut=40.0):
    dx = 2.0 / float(n)
    G = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(G.max())
    mask = G >= m - cut / alpha
    if not np.any(mask):
        mask = np.zeros(G.shape, bool)
        mask[int(G.argmax())] = True
    vals = G[mask]
    e = np.exp(alpha * (vals - m))
    se = float(e.sum())
    if se <= 0.0:
        e = np.ones_like(e)
        se = 1.0
    F = m + float(np.log(se)) / alpha
    grad = np.zeros(n)
    lags = np.arange(-(n - 1), n)[mask]
    for ei, k in zip(e, lags):
        w = float(ei) * dx / se
        if k == 0:
            grad += w * (1.0 - 2.0 * x)
        elif k > 0:
            s = int(k)
            grad[s:] += w * (1.0 - x[:n - s])
            grad[:n - s] -= w * x[s:]
        else:
            s = int(-k)
            grad[:n - s] += w * (1.0 - x[s:])
            grad[s:] -= w * x[:n - s]
    return F, grad


def smooth_fg_pen(x, n, alpha, pen):
    F, g = smooth_fg(x, n, alpha)
    if pen > 0.0:
        s = x.sum() - n / 2.0
        F = F + pen * s * s
        g = g + 2.0 * pen * s
    return F, g


def obj_fun(x, n, alpha, pen):
    return smooth_fg_pen(x, n, alpha, pen)[0]


def obj_jac(x, n, alpha, pen):
    return smooth_fg_pen(x, n, alpha, pen)[1]


def lag_gradient(h, n, k):
    dx = 2.0 / float(n)
    g = np.zeros(n)
    if k == 0:
        return dx * (1.0 - 2.0 * h)
    if k > 0:
        s = int(k)
        g[s:] += dx * (1.0 - h[:n - s])
        g[:n - s] -= dx * h[s:]
    else:
        s = int(-k)
        g[:n - s] += dx * (1.0 - h[s:])
        g[s:] -= dx * h[:n - s]
    return g


def solve_lp(A, b, bounds, cobj, Aeq):
    try:
        return linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=0.0,
                       bounds=bounds, method='highs')
    except Exception:
        try:
            return linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=0.0,
                           bounds=bounds)
        except Exception:
            return None


def lp_polish(h, n, deadline, seed=42, K=24):
    h = project_feasible(h, n).copy()
    best = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(seed)

    for tr in (0.25, 0.12, 0.06, 0.03):
        if time.time() > deadline - 3.0:
            break
        stuck = 0
        while time.time() < deadline - 3.0:
            G = np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))
            mx = float(G.max())
            if mx < best:
                best = mx
                best_h = h.copy()

            tol = max(1e-5, 0.25 / float(n))
            act = np.where(G >= mx - tol)[0]
            if act.size == 0:
                act = np.array([int(G.argmax())])
            if act.size > K:
                act = act[np.argsort(G[act])[::-1][:K]]

            m = act.size
            gr = np.empty((m, n))
            for a, kk in enumerate(act):
                gr[a] = lag_gradient(h, n, int(kk))

            A = np.hstack((gr, -np.eye(m)))
            b = -G[act]

            cobj = np.zeros(n + 1)
            cobj[n] = 1.0
            cobj[:n] = 1e-7 * (rng.rand(n) - 0.5)

            Aeq = np.ones(n + 1)
            Aeq[n] = 0.0

            bounds = []
            for i in range(n):
                lo = max(-h[i], -tr)
                hi = min(1.0 - h[i], tr)
                bounds.append((lo, hi))
            bounds.append((0.0, None))

            res = solve_lp(A, b, bounds, cobj, Aeq)
            if res is not None and getattr(res, 'success', False) and np.all(np.isfinite(res.x)):
                d = res.x[:n]
                d = np.clip(d, -h + 1e-12, 1.0 - h - 1e-12)
                improved = False
                for f in (1.0, 0.6, 0.3, 0.1):
                    h2 = project_feasible(h + f * d, n)
                    c2 = compute_c5(h2, n)
                    if c2 < best - 1e-13:
                        h = h2
                        best = c2
                        best_h = h2.copy()
                        improved = True
                        break
                if improved:
                    stuck = 0
                else:
                    stuck += 1
                    if stuck > 50:
                        break
                    if stuck % 10 == 0:
                        h = project_feasible(h + 0.01 * rng.randn(n), n)
            else:
                stuck += 1
                if stuck > 50:
                    break
                if stuck % 10 == 0:
                    h = project_feasible(h + 0.01 * rng.randn(n), n)

    return best_h, best


def lbfgsb_polish(h, n, deadline, seed=42):
    h = project_feasible(h, n).copy()
    best = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

    for alpha in (4000, 20000, 100000):
        if time.time() > deadline - 15.0:
            break
        for pen in (1.0, 10.0, 100.0):
            if time.time() > deadline - 10.0:
                break
            try:
                res = minimize(obj_fun, h, method='L-BFGS-B', bounds=bounds,
                               args=(n, alpha, pen), jac=obj_jac,
                               options={'maxiter': 150, 'ftol': 1e-12,
                                        'maxfun': 700})
                h2 = project_feasible(res.x, n)
                c2 = compute_c5(h2, n)
                if c2 < best:
                    best = c2
                    best_h = h2.copy()
                    h = h2
            except Exception:
                pass

    return best_h, best


def binary_search(h, n, deadline, seed=42):
    h = to_binary(h, n)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur = float(c.max() * dx)
    best = cur
    best_h = h.copy()

    zeros = np.where(h == 0.0)[0]
    ones = np.where(h == 1.0)[0]
    if zeros.size == 0 or ones.size == 0 or time.time() > deadline - 1:
        return best_h, best

    omh = 1.0 - h
    rng = np.random.RandomState(seed)
    T = 0.008
    no_improve = 0

    while time.time() < deadline - 1.0:
        ai = int(rng.randint(zeros.size))
        bi = int(rng.randint(ones.size))
        i = int(zeros[ai])
        j = int(ones[bi])

        d = np.zeros(2 * n - 1)
        st_i = i - 1 if i > 0 else None
        st_j = j - 1 if j > 0 else None

        d[n - 1 + i : st_i : -1] += omh
        d[n - 1 : n - 1 + (n - i)] -= h[i:]
        d[n - 1 + j : st_j : -1] -= omh
        d[n - 1 : n - 1 + (n - j)] += h[j:]

        d[n - 1] -= 2.0
        d[n - 1 + i - j] += 1.0
        d[n - 1 + j - i] += 1.0

        nc = float((c + d).max() * dx)
        ex = (cur - nc) / T
        accept = False
        if ex >= 0.0:
            accept = True
        elif ex > -700.0 and float(rng.random()) < float(np.exp(ex)):
            accept = True

        if accept:
            h[i] = 1.0
            h[j] = 0.0
            c += d
            cur = nc
            zeros[ai] = j
            ones[bi] = i
            omh = 1.0 - h
            no_improve = 0
            if cur < best:
                best = cur
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 3000:
                h = to_binary(project_feasible(best_h + 0.05 * rng.randn(n), n))
                c = np.correlate(h, 1.0 - h, mode='full').astype(float)
                cur = float(c.max() * dx)
                zeros = np.where(h == 0.0)[0]
                ones = np.where(h == 1.0)[0]
                omh = 1.0 - h
                no_improve = 0
                T = 0.010

        T *= 0.99998

    return best_h, best
