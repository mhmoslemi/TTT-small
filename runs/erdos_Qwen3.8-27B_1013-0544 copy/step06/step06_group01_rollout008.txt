def slsqp_opt(h, n, alpha=100000, maxiter=2000):
    """Smooth max approximation with SLSQP."""
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        # Log-sum-exp approximation
        shifted = alpha * (c - m)
        max_shifted = shifted.max()
        if max_shifted > 500:
            log_sum = (m + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / alpha)
        else:
            log_sum = (m + np.log(np.sum(np.exp(shifted))) / alpha)
        return float(log_sum * (2.0/n))
    
    def con_fun(x):
        return x.sum() - n/2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

def slsqp_opt(h, n, alpha, maxiter=2000):

    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(lambda x: _slsqp_obj(x, n, alpha)[0], h,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h.copy(), n)

def pair_search(h, n, deadline, steps, radius=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in steps:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-14:
                            best_c = cv
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-14:
                            best_c = cv
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def sa_burst(h, n, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(c.max())
    best_h = h.copy()
    best_c = cur_max
    nc = 2 * n - 1
    temp = 0.02
    while time.time() < deadline:
        p = rng.randint(n)
        hp = h[p]
        if hp < 1e-6 or hp > 1.0 - 1e-6:
            continue
        d_lo = -hp
        d_hi = 1.0 - hp
        k_lo = max(0, min(p, n - 1 - p))
        k_hi = min(nc - 1, max(p + n - 1, nc - 1 - p))
        um = 0.0
        if k_lo > 0:
            um = float(c[:k_lo].max())
        if k_hi < nc - 1:
            um = max(um, float(c[k_hi + 1:].max()))
        ks = np.arange(k_lo, k_hi + 1)
        fwd = (ks >= p) & (ks <= p + n - 1)
        bwd = (ks >= n - 1 - p) & (ks <= nc - 1 - p)
        s = np.zeros(len(ks))
        if fwd.any():
            s[fwd] = 1.0 - h[p - ks[fwd] + n - 1]
        if bwd.any():
            s[bwd] -= h[p + ks[bwd] - (n - 1)]
        ci = (ks == n - 1)
        d = rng.uniform(d_lo, d_hi)
        if abs(d) < 1e-14:
            continue
        nv = c[ks] + d * s
        if ci.any():
            nv[ci] = c[n - 1] + d * (1.0 - 2.0 * hp) - d * d
        new_max = max(um, float(nv.max()))
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            if fwd.any():
                c[ks[fwd]] += d * (1.0 - h[p - ks[fwd] + n - 1])
            if bwd.any():
                c[ks[bwd]] -= d * h[p + ks[bwd] - (n - 1)]
            c[n - 1] -= d * d
            h[p] = np.clip(h[p] + d, 0.0, 1.0)
            cur_max = max(um, float(c[ks].max()))
            if cur_max < best_c:
                best_c = cur_max
                best_h = h.copy()
        temp *= 0.995
        if temp < 1e-7:
            temp = 1e-7
    return best_h, best_c * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        h0 = np.full(200, 0.5)
        n = 200

    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SA burst for global exploration (short)
    if time.time() < deadline - 800:
        h_sa, c_sa = sa_burst(h0, n, time.time() + 40, rng)
        h_sa = project_feasible(h_sa, n)
        c_sa = compute_c5(h_sa, n)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # Phase 2: Medium coordinate descent
    if time.time() < deadline - 500:
        h_m, c_m = coord_descent(best_h, n, 20000, 60, time.time() + 120, rng, penalty=5.0, d_scale=0.5)
        h_m = project_feasible(h_m, n)
        c_m = compute_c5(h_m, n)
        if c_m < best_c:
            best_h, best_c = h_m, c_m

    # Phase 3: SLSQP with multiple alphas
    if time.time() < deadline - 300:
        for a in [50000, 200000, 500000]:
            if time.time() > deadline - 280:
                break
            hs = slsqp_opt(best_h, n, a, maxiter=2000)
            cs = compute_c5(hs, n)
            if cs < best_c:
                best_h, best_c = hs, cs

    # Phase 4: Fine coordinate descent
    if time.time() < deadline - 150:
        h_f, c_f = coord_descent(best_h, n, 50000, 100, time.time() + 200, rng, penalty=0.1, d_scale=0.1)
        h_f = project_feasible(h_f, n)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Phase 5: SLSQP high alpha
    if time.time() < deadline - 100:
        hs = slsqp_opt(best_h, n, 1000000, maxiter=3000)
        cs = compute_c5(hs, n)
        if cs < best_c:
            best_h, best_c = hs, cs

    # Phase 6: Very fine coordinate descent
    if time.time() < deadline - 40:
        h_vf, c_vf = coord_descent(best_h, n, 50000, 150, time.time() + 120, rng, penalty=0.0, d_scale=0.03)
        h_vf = project_feasible(h_vf, n)
        c_vf = compute_c5(h_vf, n)
        if c_vf < best_c:
            best_h, best_c = h_vf, c_vf

    # Phase 7: Pair search for final nudge
    if time.time() < deadline - 15:
        best_h, best_c = pair_search(best_h, n, time.time() + 50,
                                     [0.005, 0.002, 0.001, 0.0005, 0.0002], radius=40)

    # Phase 8: Final ultra-fine CD
    if time.time() < deadline - 5:
        h_uf, c_uf = coord_descent(best_h, n, 50000, 200, deadline - 2, rng, penalty=0.0, d_scale=0.01)
        h_uf = project_feasible(h_uf, n)
        c_uf = compute_c5(h_uf, n)
        if c_uf < best_c:
            best_h, best_c = h_uf, c_uf

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
