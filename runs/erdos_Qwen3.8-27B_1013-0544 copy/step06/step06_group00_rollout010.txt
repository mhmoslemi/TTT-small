```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Binary search for threshold tau such that sum(clip(h-tau, 0, 1)) = target
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    proj = np.clip(h - tau, 0.0, 1.0)
    # Fine-tune
    for _ in range(100):
        diff = target - proj.sum()
        if abs(diff) < 1e-15:
            break
        mask = (proj > 1e-15) & (proj < 1.0 - 1e-15)
        if not mask.any():
            break
        proj[mask] += diff / mask.sum()
        proj = np.clip(proj, 0.0, 1.0)
    return proj

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full').astype(float)
    c = c_raw * dx
    c_max = c.max()
    c_shift = c - c_max
    exp_val = np.exp(alpha * c_shift)
    w_sum = exp_val.sum()
    obj = c_max + np.log(w_sum) / alpha
    w = exp_val / w_sum
    
    # Compute gradient
    # df/dh[i] = (1/Z) * sum_k exp(alpha*c_k) * dc_k/dh[i]
    # dc_k/dh[i] = dx * [(1-h[i+k-n+1]) - h[i-k+n-1]] (with h=0 out of bounds)

    # S2[i] = sum_k w[k] * h[i-k+n-1]
    
    S1 = np.full(n, w.sum())  # sum_k w[k] * 1 = w.sum()
    S2 = np.zeros(n)
    
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-18:

        # S1: index i+k-n+1 must be in [0, n-1]
        # 0 <= i+k-n+1 < n → n-1-k <= i < n-1-k+n → i in [max(0,n-1-k), min(n-1, 2n-2-k)]
        lo1 = max(0, n - 1 - k)
```

**Yao's method** (also known as the "generalized gradient method" for min-max problems):
- Maintain a weight vector w_k for each constraint
- Update: h ← project(h - step * sum_k w_k * grad(c_k))
- Update weights: w_k ← w_k * exp(step * c_k) (normalized)

This is essentially what the smooth approximation does, but with explicit weight tracking. The advantage is that we don't need to recompute the exponential every iteration.

```python
            if wk < 1e-18:
                continue
            lo1 = max(0, n - 1 - k)
            hi1 = min(n, 2 * n - 1 - k)
            if lo1 < hi1:
                idx = np.arange(lo1, hi1) + k - n + 1
                S1[lo1:hi1] -= wk * h[idx]
            lo2 = max(0, k - n + 1)
            hi2 = min(n, k)
            if lo2 < hi2:
                idx2 = np.arange(lo2, hi2) - k + n - 1
                S2[lo2:hi2] += wk * h[idx2]
        grad = dx * (S1 - S2)
        
        # Update
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        
        # Compute new c
        c_raw_new = np.correlate(h_new, 1 - h_new, mode='full').astype(float)
        c_new = c_raw_new * dx
        max_new = c_new.max()
        max_old = c.max()
        
        if max_new <= max_old:
            h = h_new
            c = c_new
            # Update weights
            w *= np.exp(step * c)
            w /= w.sum()
            if max_new < best_c5:
                best_c5 = max_new
                best_h = h.copy()
            step = min(step * 1.02, 0.2)
        else:
            step *= 0.7
            h_new = h - step * grad
            h_new = project_feasible(h_new, n)
            c_raw_new = np.correlate(h_new, 1 - h_new, mode='full').astype(float)
            c_new = c_raw_new * dx
            if c_new.max() < c.max():
                h = h_new
                c = c_new
                w *= np.exp(step * c)
                w /= w.sum()
                if c.max() < best_c5:
                    best_c5 = c.max()
                    best_h = h.copy()
    
    return best_h, best_c5
```

This is essentially the same as the smooth gradient method but with explicit weight tracking, which avoids the numerical issues of computing exp(alpha * c) for very large alpha.

The key advantage: the weights w naturally concentrate on the "active" constraints (those with large c_k), and the method automatically adapts which constraints to focus on.
