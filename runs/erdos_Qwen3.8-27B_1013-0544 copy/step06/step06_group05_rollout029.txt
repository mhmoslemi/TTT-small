def project_feasible(h, n):

def compute_neighbors(h, n):
    """Compute D_j(k) = h[j+k] + h[j-k] for all j, k (lags -n+1 to n-1)."""
    # Lags from -(n-1) to (n-1), total 2n-1
    # For each j and lag l: D[j, l] = h[j+l] + h[j-l] (0 if out of range)
    lags = np.arange(-(n-1), n)
    D = np.zeros((n, 2*n-1))
    for li, l in enumerate(lags):
        idx1 = np.arange(n) + l
        idx2 = np.arange(n) - l
        v1 = (idx1 >= 0) & (idx1 < n)
        v2 = (idx2 >= 0) & (idx2 < n)
        D[v1, li] += h[idx1[v1]]
        D[v2, li] += h[idx2[v2]]
    return D, lags
