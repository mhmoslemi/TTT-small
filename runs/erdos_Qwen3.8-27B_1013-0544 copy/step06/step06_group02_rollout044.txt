But still, the search might not be finding the best moves because it's limited to nearby pairs.

My improvement: **Use the gradient to select which pairs to try**. Specifically:
1. Compute the gradient g
2. For each i, the best j to transfer to is the one with the smallest g[j] (if g[i] > 0) or largest g[j] (if g[i] < 0)
3. Try only the top few candidates for each i

This reduces the number of evaluations dramatically while still finding good moves.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_max = c.max()
    sh = alpha * (c - c_max)
    mx = sh.max()
    if mx < -600:
        w = np.zeros(len(c))
        w[sh.argmax()] = 1.0
    else:
        w = np.exp(sh - mx)
        w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def slsqp_con(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        remaining = deadline - time.time()
        if remaining < 3:
            break
        mu = alpha * 0.01
        maxit = min(max_iters, max(10, int(remaining * 30)))
        try:
            res = minimize(penalty_obj, h, jac=penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': maxit * 10, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_opt(h, n, deadline, alpha=1000000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_descent(h, n, deadline, lr_init=0.005, alpha=2000.0):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    for _ in range(30000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        step = lr
        improved = False
        for _ in range(25):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-14:
                break
    return best_h, best_c

def guided_pair_search(h, n, deadline, base_step=0.002, alpha=2000.0):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in [500, 2000, 10000, 50000]:
        stale_count = 0
        while time.time() < deadline:
            g = grad_smooth_max(h, n, alpha)
            improved = False
            # For each i, find best j to transfer to based on gradient
            g_sorted_idx = np.argsort(g)
            n_eval = 0
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                gi = g[i]
                # Try transferring from i to j where g[j] < g[i]
                # Pick top 5 candidates with lowest g
                for pos in range(n):
                    j = g_sorted_idx[pos]
                    if j == i:
                        continue
                    gj = g[j]
                    if gj >= gi:
                        continue
                    diff = gi - gj
                    if diff < 1e-12:
                        continue
                    step = min(base_step, h[i], 1.0 - h[j])
                    step = min(step, 0.01)
                    if step < 1e-8:
                        continue
                    h[i] -= step
                    h[j] += step
                    c = compute_c5(h, n)
                    n_eval += 1
                    if c < best_c - 1e-14:
                        best_c = c
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += step
                        h[j] -= step
                    if n_eval > 500:
                        break
                if n_eval > 500:
                    break
            if not improved:
                stale_count += 1
                if stale_count > 3:
                    break
            else:
                stale_count = 0
    return best_h, best_c

def local_search_fine(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - 80)
                hi = min(n, i + 81)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:

        h = np.full(250, 0.5)

    best_h = h.copy()

                                                [0.001, 0.0005, 0.0002, 0.0001])

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
