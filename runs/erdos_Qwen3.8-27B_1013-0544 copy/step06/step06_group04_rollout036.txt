```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    lo = max(0, n - 1 - i)
    hi = min(2 * n - 2, 2 * n - 2 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] += 1.0 - h[m - (n - 1) + i]
    m2 = np.arange(max(0, i), min(2 * n - 1, n + i))
    jp = n - 1 + i - m2
    delta[m2] -= h[jp]
    lo = max(0, n - 1 - j)
    hi = min(2 * n - 2, 2 * n - 2 - j)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] -= 1.0 - h[m - (n - 1) + j]
    m2 = np.arange(max(0, j), min(2 * n - 1, n + j))
    jp = n - 1 + j - m2
    delta[m2] += h[jp]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def best_improvement_search(h, n, deadline, max_sweeps=200):
    """Check all possible single swaps, pick the best improving one each iteration."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta_val = 0
        best_i, best_j = -1, -1
        best_d = None
        for i in zeros:
            for j in ones:
                d = swap_delta(h, i, j, n)
                nc = float((corr + d).max()) * dx
                if nc < cur_c - best_delta_val:
                    best_delta_val = cur_c - nc
                    best_i, best_j = i, j
                    best_d = d
        if best_i >= 0:
            h[best_i], h[best_j] = 1.0, 0.0
            corr += best_d
            cur_c -= best_delta_val
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            break
    return best_h, best_c
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_delta(h, i, j, n):
    # (copied from previous code, verified to work)
    delta = np.zeros(2 * n - 1)
    lo = max(0, n - 1 - i)
    hi = min(2 * n - 2, 2 * n - 2 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] += 1.0 - h[m - (n - 1) + i]
    m2 = np.arange(max(0, i), min(2 * n - 1, n + i))
    jp = n - 1 + i - m2
    delta[m2] -= h[jp]
    lo = max(0, n - 1 - j)
    hi = min(2 * n - 2, 2 * n - 2 - j)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] -= 1.0 - h[m - (n - 1) + j]
    m2 = np.arange(max(0, j), min(2 * n - 1, n + j))
    jp = n - 1 + j - m2
    delta[m2] += h[jp]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
```

```python
def swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    lo = max(0, n - 1 - i)
    hi = min(2 * n - 2, 2 * n - 2 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] += 1.0 - h[m - (n - 1) + i]
    m2 = np.arange(max(0, i), min(2 * n - 1, n + i))
    jp = n - 1 + i - m2
    delta[m2] -= h[jp]
    lo = max(0, n - 1 - j)
    hi = min(2 * n - 2, 2 * n - 2 - j)
    if lo <= hi:

    return delta
```

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    lo = max(0, n - 1 - i)
    hi = min(2 * n - 2, 2 * n - 2 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] += 1.0 - h[m - (n - 1) + i]
    m2 = np.arange(max(0, i), min(2 * n - 1, n + i))
    jp = n - 1 + i - m2
    delta[m2] -= h[jp]
    lo = max(0, n - 1 - j)
    hi = min(2 * n - 2, 2 * n - 2 - j)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] -= 1.0 - h[m - (n - 1) + j]
    m2 = np.arange(max(0, j), min(2 * n - 1, n + j))
    jp = n - 1 + j - m2
    delta[m2] += h[jp]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def two_swap_delta(h, i1, j1, i2, j2, n):
    d1 = swap_delta(h, i1, j1, n)
    h_t = h.copy()
    h_t[i1] = 1.0
    h_t[j1] = 0.0
    d2 = swap_delta(h_t, i2, j2, n)
    return d1 + d2

        if ss < 1:
            break
        ii = rng.randint(0, nz, size=ss)
        jj = rng.randint(0, no, size=ss)
        best_imp = 0.0
        bi, bj, bd = -1, -1, None
```
