def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] -= h[m:]
    else:
        grad[:n-m] += (1.0 - h[m:])

def _soft_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    e = np.exp(alpha * (c - m))
    se = e.sum()
    f = (m + np.log(se)) / alpha
    w = alpha * e / se
    grad = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        grad += wk * _lag_gradient(h, n, lag)
    return f, grad * dx

def lbn_solve(h, n, alpha, rho0=1.0, max_outer=15, max_inner=1000, deadline=None):
    h_curr = h.copy()
    lam = 0.0
    rho = rho0
    for outer in range(max_outer):
        if deadline and time.time() > deadline - 3:
            break
        def omg(x):
            f, g = _soft_max_and_grad(x, n, alpha)
            s = x.sum() - n / 2.0
            f += lam * s + 0.5 * rho * s * s
            g = g + (lam + rho * s) * np.ones(n)
            return f, g
        bounds = [(0.0, 1.0)] * n
        res = minimize(omg, h_curr, jac=True, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': max_inner, 'ftol': 1e-15, 'gtol': 1e-10})
        h_curr = np.clip(res.x, 0.0, 1.0)
        s = h_curr.sum() - n / 2.0
        if abs(s) < 1e-7:
            lam = 0.0
            break
        lam += rho * s
        rho = min(rho * 2.0, 1e6)
    return project_feasible(h_curr, n)

def projected_subgrad(h, n, n_iter, epsilon, deadline):
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    best_h = h_curr.copy()
    lr = 0.01
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        c_max = c.max()
        if c_max < best_c:
            best_c = c_max
            best_h = h_curr.copy()
        active = np.where(c > c_max - epsilon)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        grad = np.zeros(n)
        for k in active:
            lag = k - (n - 1)
            grad += _lag_gradient(h_curr, n, lag)
        grad /= len(active)
        dx = 2.0 / n
        grad *= dx
        h_new = project_feasible(h_curr - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < c_max:

            lr = min(lr * 1.2, 0.1)
        else:

def pairwise_swap(h, n, deadline, max_rounds=50):

    t_var = cp.Variable()

    h_new = h_var.value

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def _soft_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    e = np.exp(alpha * (c - m))
    se = e.sum()
    f = (m + np.log(se)) / alpha
    w = alpha * e / se
    grad = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        grad += wk * _lag_gradient(h, n, lag)
    return f, grad * dx

def lbn_solve(h, n, alpha, max_outer=12, max_inner=2000, deadline=None):
    h_curr = h.copy()
    lam = 0.0
    rho = 1.0
    for outer in range(max_outer):
        if deadline and time.time() > deadline - 3:
            break

        def omg(x):
            f, g = _soft_max_grad(x, n, alpha)
            s = x.sum() - n / 2.0
            f += lam * s + 0.5 * rho * s * s
            g = g + (lam + rho * s) * np.ones(n)
            return f, g

        bounds = [(0.0, 1.0)] * n
        res = minimize(omg, h_curr, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': max_inner, 'ftol': 1e-15, 'gtol': 1e-10})
        h_curr = np.clip(res.x, 0.0, 1.0)
        s = h_curr.sum() - n / 2.0
        if abs(s) < 1e-7:
            break
        lam += rho * s
        rho = min(rho * 2.0, 1e6)
    return project_feasible(h_curr, n)

def subgrad_search(h, n, n_iter, epsilon, deadline):
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    best_h = h_curr.copy()
    lr = 0.05
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        c_max = c.max()
        if c_max < best_c - 1e-13:
            best_c = c_max
            best_h = h_curr.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break
        active = np.where(c > c_max - epsilon)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        grad = np.zeros(n)
        for k in active:
            lag = k - (n - 1)
            grad += _lag_gradient(h_curr, n, lag)
        grad /= len(active)
        dx = 2.0 / n
        grad *= dx
        found = False
        for factor in [1.0, 0.5, 0.25, 0.125]:
            h_try = project_feasible(h_curr - lr * factor * grad, n)
            c_try = compute_c5(h_try, n)
            if c_try < c_max - 1e-14:
                h_curr = h_try
                found = True
                break
        if not found:
            lr *= 0.7
            if lr < 1e-7:
                break
        else:
            lr = min(lr * 1.1, 0.2)
    return best_h, best_c

def pairwise_local(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for rnd in range(60):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        improved = False
        for i in range(n - m):
            if time.time() > deadline:
                break
            j = i + m
            if h[i] > h[j] + 1e-8:
                d = (h[i] - h[j]) * 0.5
                h[i] -= d
                h[j] += d
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    improved = True
                else:
                    h[i] += d
                    h[j] -= d
            elif h[j] > h[i] + 1e-8:
                d = (h[j] - h[i]) * 0.5
                h[i] += d
                h[j] -= d
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    improved = True
                else:
                    h[i] -= d
                    h[j] += d
        if not improved:
            break
    return h, best_c

def outer_lp(h_curr, n, trust, deadline):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        g = _lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (g @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    hv = h_var.value
    if hv is None:
        return h_curr.copy()
    return project_feasible(hv, n)
