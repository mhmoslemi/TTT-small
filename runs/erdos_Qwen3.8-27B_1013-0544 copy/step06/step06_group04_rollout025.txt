```python
def objective_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    g = c * dx  # length 2n-1
    
    m = g.max()
    sh = alpha * (g - m)
    sh = np.clip(sh, -500, 500)
    exp_sh = np.exp(sh)
    weights = exp_sh / exp_sh.sum()
    
    F = m + np.log(exp_sh.sum()) / alpha
    
    # Gradient: dF/dh[j] = sum_k weights[k] * dc[k]/dh[j] * dx
    # dc[k]/dh[j] = (1-h[j-k']) * I(0<=j-k'<n) - h[j+k'] * I(0<=j+k'<n)
    # where k' = k-(n-1)
    
    grad = np.zeros(n)
    for j in range(n):
        for k in range(2*n-1):
            kp = k - (n-1)
            if 0 <= j-kp < n:
                grad[j] += weights[k] * (1 - h[j-kp])
            if 0 <= j+kp < n:
                grad[j] -= weights[k] * h[j+kp]
    grad *= dx
    
    return F, grad
```

def swap_delta(h, i, j, n):
    """Compute delta in correlation when h[i]: 0->1 and h[j]: 1->0"""
    delta = np.zeros(2*n-1)
    for k in range(2*n-1):
        kp = k - (n-1)
        # Contribution from h[i] going 0->1
        idx1 = i - kp
        if 0 <= idx1 < n:

            delta[k] -= (1.0 - h[idx3])
        idx4 = j + kp
        if 0 <= idx4 < n:
            delta[k] += h[idx4]
