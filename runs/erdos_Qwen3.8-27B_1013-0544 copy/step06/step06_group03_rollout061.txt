```python
import numpy as np
from scipy.optimize import minimize, linprog
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient_matrix(h, n):
    """Compute the Jacobian of all correlations w.r.t. h.
    Returns G of shape (2n-1, n) where G[k, i] = d(c_k)/d(h_i)"""
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    one_minus_h = 1.0 - h
    for k in range(2 * n - 1):
        L = k - (n - 1)  # lag
        if L == 0:
            G[k, :] = dx * (1.0 - 2.0 * h)
        elif L > 0:
            # c_L = dx * sum_{m=0}^{n-1-L} h[m] * (1 - h[m+L])
            # d/dh[i] = dx * [(1-h[i+L])*1_{i+L < n} - h[i-L]*1_{i-L >= 0}]
            # First term: i in [0, n-1-L], contributes (1-h[i+L])
            G[k, :n-L] += dx * one_minus_h[L:]
            # Second term: i in [L, n-1], contributes -h[i-L]
            G[k, L:] -= dx * h[:n-L]
        else:
            L_neg = -L
            # c_{-L_neg} = dx * sum_{m=L_neg}^{n-1} h[m] * (1 - h[m-L_neg])
            # d/dh[i] = dx * [(1-h[i-L_neg])*1_{i-L_neg >= 0} - h[i+L_neg]*1_{i+L_neg < n}]
            G[k, L_neg:] += dx * one_minus_h[:n-L_neg]
            G[k, :n-L_neg] -= dx * h[L_neg:]
    return G
```

    c0 = compute_all_lags(h0, n)
    G = compute_gradient_matrix(h0, n)
    
    # Weights: emphasize lags near the max
    m = c0.max()
    shifted = alpha * (c0 - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w = w / w.sum()
    
    # Build QP: min (h-h0)^T (G^T W G) (h-h0) + (G^T W c0)^T (h-h0)

    # = d^T (G^T W G) d + 2 (G^T W c0)^T d + const

    try:

    except:
        pass
    
    if h_var.value is not None:
        h_new = h_var.value
        return project_feasible(h_new, n)
    return h0.copy()

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_matrix(h, n):
    """Jacobian of correlations w.r.t. h. Shape (2n-1, n)."""
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for k in range(2 * n - 1):
        L = k - (n - 1)
        if L == 0:
            G[k] = dx * (1.0 - 2.0 * h)
        elif L > 0:
            G[k, :n-L] = dx * omh[L:]
            G[k, L:] += -dx * h[:n-L]
        else:
            La = -L
            G[k, La:] = dx * omh[:n-La]
            G[k, :n-La] += -dx * h[La:]
    return G

def gn_step(h0, n, alpha, reg=1e-8):
    """One Gauss-Newton step with line search."""
    c0 = compute_all_lags(h0, n)
    G = compute_grad_matrix(h0, n)
    m = c0.max()
    sh = alpha * (c0 - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    
    A = G.T @ (w[:, None] * G) + reg * np.eye(n)
    b = G.T @ (w * c0)
    
    h_var = cp.Variable(n)
    d = h_var - h0
    obj = cp.quad_form(d, cp.psd_wrap(A)) + 2 * b @ d
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
    prob = cp.Problem(cp.Minimize(obj), cons)
    
    solved = False
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=10000)
            if h_var.value is not None:
                solved = True
                break
        except Exception:
            continue
    if not solved:
        return h0.copy(), compute_c5(h0, n)
    
    h_full = project_feasible(h_var.value, n)
    c_full = compute_c5(h_full, n)
    c_best = compute_c5(h0, n)
    
    if c_full < c_best:
        return h_full, c_full
    
    # Line search
    for ls in range(20):
        t = 2.0 ** (-ls)
        if t < 1e-5: break
        h_try = project_feasible((1 - t) * h0 + t * h_full, n)
        c_try = compute_c5(h_try, n)
        if c_try < c_best:
            return h_try, c_try
    return h0.copy(), c_best

The key is step 4: for a given pair (i,j), find delta that minimizes max_L (c_L + delta * (G[L,j] - G[L,i])).

This is a 1D minimization of the max of linear functions, which can be solved analytically:
- For each L, the constraint c_L + delta * G_L <= T is:

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_matrix(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for k in range(2 * n - 1):
        L = k - (n - 1)
        if L == 0:
            G[k] = dx * (1.0 - 2.0 * h)
        elif L > 0:
            G[k, :n - L] = dx * omh[L:]
            G[k, L:] -= dx * h[:n - L]
        else:
            La = -L
            G[k, La:] = dx * omh[:n - La]
            G[k, :n - La] -= dx * h[La:]
    return G

def gn_step(h0, n, alpha, reg=1e-6):

    m = c0.max()

    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    A = G.T @ (w[:, None] * G) + reg * np.eye(n)
    b = G.T @ (w * c0)
    h_var = cp.Variable(n)
    d = h_var - h0
    obj = cp.quad_form(d, cp.psd_wrap(A)) + 2 * b @ d
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
    prob = cp.Problem(cp.Minimize(obj), cons)
    solved = False
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if h_var.value is not None and not np.any(np.isnan(h_var.value)):
                solved = True
                break
        except Exception:
            continue
    if not solved:
        return h0.copy(), compute_c5(h0, n), False
    h_full = project_feasible(h_var.value, n)

def slsqp_polish(h, n, alpha, maxiter=5000):
    dx = 2.0 / n
    def _obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        cm = c.max()
        s = alpha * (c - cm)
        sm = s.max()
        return float(cm + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha)
    def _con(x):
        return np.sum(x) - n / 2.0
    def _con_j(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
