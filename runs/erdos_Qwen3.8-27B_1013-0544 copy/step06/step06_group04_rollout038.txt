def swap_delta(i, j, h, n):
    """Compute delta in correlation for binary swap i(0→1), j(1→0)."""
    d = np.zeros(2*n-1)
    # i contributes
    lo_i = max(0, n - 1 - i)
    hi_i = min(2*n-2, 2*n-2-i)
    if lo_i <= hi_i:
        ks = np.arange(lo_i, hi_i+1)
        others = i + ks - (n-1)
        d[ks] = 1.0 - h[others]
        # Fix the case where other == j
        mask = others == j
        if mask.any():
            d[ks[mask]] = 1.0
    # j contributes
    lo_j = max(0, n - 1 - j)
    hi_j = min(2*n-2, 2*n-2-j)

        ks = np.arange(lo_j, hi_j+1)
        others = j + ks - (n-1)
        d[ks] = -(1.0 - h[others])
        mask = others == i
        if mask.any():
            d[ks[mask]] = -1.0
    return d

from scipy.optimize import minimize
import time

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1.0 - 1e-10
            cnt = free.sum()
            if cnt == 0:
                break
            h[free] = np.minimum(h[free] + diff / cnt, 1.0)
        else:
            free = h > 1e-10
            cnt = free.sum()
            if cnt == 0:
                break
            h[free] = np.maximum(h[free] + diff / cnt, 0.0)
    return h

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def grad_c_k(h, k, n):
    """Gradient of c[k] (the k-th element of full correlation) w.r.t. h."""
    # c[k] = sum_i h[i] * (1-h[i+k-n+1]) for valid i
    # ∂c[k]/∂h[m] = (1-h[m+k-n+1])[valid] + (-h[m-k+n-1])[valid]
    g = np.zeros(n)
    j = k - n + 1  # the effective shift
    # m as first index: m + j valid means 0 ≤ m+j < n
    if j >= 0:
        # m ranges from 0 to n-1-j
        ms = np.arange(0, n - j)
        g[ms] += (1 - h[ms + j])
        # m as second index: m-j valid (as first index) means 0 ≤ m-j < n and 0 ≤ m < n
        # i = m-j, so i ranges from 0 to n-1, m = i+j ranges from j to j+n-1
        # But m must be in [0, n-1], so m ranges from j to n-1
        ms2 = np.arange(j, n)
        g[ms2] += (-h[ms2 - j])
    else:
        # j < 0, let j' = -j > 0
        jp = -j
        # m as first index: m + j valid means 0 ≤ m-jp < n, so m ranges from jp to n-1
        ms = np.arange(jp, n)
        g[ms] += (1 - h[ms - jp])
        # m as second index: m-j = m+jp valid as first index, 0 ≤ m+jp < n, m from 0 to n-1-jp
        ms2 = np.arange(0, n - jp)
        g[ms2] += (-h[ms2 + jp])
    return g

def soft_max_grad(c, alpha):
    """Compute softmax weights and gradient of soft-max objective."""
    shifted = c - c.max()
    p = np.exp(alpha * shifted)
    p /= p.sum()
    return p

def adam_continuous(h, n, deadline, alpha=100.0, lr=0.01, max_iter=50000):
    """Adam optimization of soft-max approximation of C5."""
    target = n / 2.0
    dx = 2.0 / n
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        c = compute_corr(h, n)
        p = soft_max_grad(c, alpha)
        
        # Gradient of soft-max: (dx) * sum_k p[k] * grad_c[k]
        # But computing all grads is O(n^2). Instead, use the structure:
        # grad = sum_k p[k] * grad_c_k(h, k, n)
        # This is expensive. Let's use a simpler approach:
        # Just use the gradient of the max (subgradient) with a few top-k terms.
        
        # Alternative: use the gradient of c[k*] where k* = argmax
        k_star = int(np.argmax(c))
        g = grad_c_k(h, k_star, n) * dx
        
        # Actually, let's use a weighted combination of top few
        top_k = np.argsort(c)[::-1][:5]
        g = np.zeros(n)
        for kk in top_k:
            g += p[int(kk)] * grad_c_k(h, int(kk), n)
        g *= dx
        
        # Adam update
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g**2
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, target)
        
        # Check every 100 iterations
        if it % 100 == 0:
            c_now = compute_c5(h, n)
            if c_now < best_c:
                best_c = c_now
                best_h = h.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            free = h < 1.0 - 1e-12
            cnt = free.sum()
            if cnt == 0:
                break
            h[free] = np.minimum(h[free] + diff / cnt, 1.0)
        else:
            free = h > 1e-12
            cnt = free.sum()
            if cnt == 0:
                break
            h[free] = np.maximum(h[free] + diff / cnt, 0.0)
    return h

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def grad_c_at_k(h, k, n):
    """Gradient of c[k] w.r.t. h, where k is index into full correlation."""
    g = np.zeros(n)
    j = k - (n - 1)  # effective shift
    if j >= 0:
        # h[m] * (1 - h[m+j]) for m = 0..n-1-j
        ms = np.arange(0, n - j)
        g[ms] += (1 - h[ms + j])
        # h[m-j] * (1 - h[m]) → -h[m-j] contribution to ∂/∂h[m] for m = j..n-1
        ms2 = np.arange(j, n)
        g[ms2] += (-h[ms2 - j])
    else:
        jp = -j
        # h[m] * (1 - h[m-jp]) for m = jp..n-1
        ms = np.arange(jp, n)
        g[ms] += (1 - h[ms - jp])
        # h[m+jp] * (1 - h[m]) → -h[m+jp] for m = 0..n-1-jp
        ms2 = np.arange(0, n - jp)
        g[ms2] += (-h[ms2 + jp])
    return g

def continuous_optimize(h0, n, deadline, alpha_init=50, alpha_max=500, lr=0.005):
    """Adam optimization of soft-max C5."""
    target = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        t += 1
        alpha = min(alpha_init + (alpha_max - alpha_init) * t / 10000, alpha_max)
        
        c = compute_corr(h, n)
        shifted = c - c.max()
        p = np.exp(alpha * shifted)
        p /= p.sum()
        
        # Gradient: weighted sum of top terms
        top_indices = np.argsort(p)[::-1]
        # Take terms with p > 1e-6 or top 10
        n_terms = min(10, int(np.sum(p > 1e-6)))
        n_terms = max(n_terms, 3)
        top_indices = top_indices[:n_terms]
        
        g = np.zeros(n)
        for kk in top_indices:
            g += p[int(kk)] * grad_c_at_k(h, int(kk), n)
        g *= dx
        
        # Adam update
        m_adam = b1 * m_adam + (1 - b1) * g
        v_adam = b2 * v_adam + (1 - b2) * g * g
        m_hat = m_adam / (1 - b1 ** t)
        v_hat = v_adam / (1 - b2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, target)
        
        if t % 200 == 0:
            c_now = compute_c5(h, n)
            if c_now < best_c:
                best_c = c_now
                best_h = h.copy()
    
    return best_h, best_c

def local_search_binary(h, n, deadline, seed=42):
    """Fast first-improvement local search for binary h."""
    h = binarize(h, n)
    dx = 2.0 / n
    corr = compute_corr(h, n).copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    sweep = 0
    max_sweeps = 5000
    
    while time.time() < deadline and sweep < max_sweeps:
        sweep += 1
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        # Only check a random subset per sweep for speed
        n_check = min(len(zeros), 20)
        n_check_ones = min(len(ones), 20)
        zeros_sub = rng.choice(zeros, n_check, replace=False)
        ones_sub = rng.choice(ones, n_check_ones, replace=False)
        for i in zeros_sub:
            if time.time() > deadline:
                break
            for j in ones_sub:
                i, j = int(i), int(j)
                # Compute swap delta
                d = np.zeros(2 * n - 1)
                lo_i = max(0, n - 1 - i)
                hi_i = min(2 * n - 2, 2 * n - 2 - i)
                if lo_i <= hi_i:
                    ks = np.arange(lo_i, hi_i + 1)
                    others = i + ks - (n - 1)
                    d[ks] = 1.0 - h[others]
                    mask = (others == j)
                    if mask.any():
                        d[ks[mask]] = 1.0
                lo_j = max(0, n - 1 - j)
                hi_j = min(2 * n - 2, 2 * n - 2 - j)
                if lo_j <= hi_j:
                    ks = np.arange(lo_j, hi_j + 1)
                    others = j + ks - (n - 1)
                    d[ks] = -(1.0 - h[others])
                    mask = (others == i)
                    if mask.any():
                        d[ks[mask]] = -1.0
                
                new_max = float(np.max(corr + d))
                if new_max * dx < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + d
                    cur_c = new_max * dx
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return best_h, best_c
