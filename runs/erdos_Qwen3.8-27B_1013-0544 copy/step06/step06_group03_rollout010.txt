def active_set_slsqp(h, n, top_m=5, maxiter=2000):
    """SLSQP on active set constraints."""
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    
    def obj(vars_):
        return vars_[n]  # t
    
    def obj_grad(vars_):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    # Constraints: c_k(h) - t <= 0 for each top lag
    def make_con(k_idx):
        def con(vars_):
            h_ = vars_[:n]
            t_ = vars_[n]
            lag = k_idx - (n - 1)
            # Compute c_lag
            m = abs(lag)
            if m >= n:
                return 1.0  # always satisfied
            if m == 0:
                return float(np.sum(h_ * (1 - h_)) * (2.0/n) - t_)
            if lag > 0:
                return float(np.sum(h_[:n-m] * (1 - h_[m:])) * (2.0/n) - t_)
            else:
                return float(np.sum(h_[m:] * (1 - h_[:n-m])) * (2.0/n) - t_)
        def con_grad(vars_):
            h_ = vars_[:n]
            g = np.zeros(n + 1)
            g[n] = -1.0
            lag = k_idx - (n - 1)
            m = abs(lag)
            if m < n:
                g[:n] = (2.0/n) * lag_grad(h_, n, lag)
            return g
        return {'type': 'ineq', 'fun': con, 'jac': con_grad}
    
    cons = [make_con(k) for k in top_idx]
    cons.append({'type': 'eq', 'fun': lambda v: v[:n].sum() - n/2,

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    t = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        d = t - h.sum()
        if abs(d) < 1e-14: break
        if d > 0:

            if s < 1e-14: break

            s = h.sum()
            if s < 1e-14: break
            h += (d / s) * h
