import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def build_A(h, n):
    offset = n - 1
    p = np.arange(n)[:, None]
    lags = np.arange(2 * n - 1) - offset
    p1 = p - lags[None, :]
    p2 = p + lags[None, :]
    A = np.zeros((n, 2 * n - 1))
    v1 = (p1 >= 0) & (p1 < n)
    v2 = (p2 >= 0) & (p2 < n)
    A[v1] += h[p1[v1]]
    A[v2] += h[p2[v2]]
    return A

def binarize(h, n, seed=0):
    rng = np.random.RandomState(seed)
    sorted_h = np.sort(h)[::-1]
    thresh = sorted_h[n // 2]
    h_bin = (h >= thresh).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > n // 2:
        excess = np.where(h_bin == 1)[0]
        rng.shuffle(excess)
        for idx in excess[:n_ones - n // 2]:
            h_bin[idx] = 0.0
    elif n_ones < n // 2:
        deficit = np.where(h_bin == 0)[0]
        rng.shuffle(deficit)
        for idx in deficit[:n // 2 - n_ones]:
            h_bin[idx] = 1.0
    return h_bin

def vectorized_binary_search(cur, n, deadline, n_restarts=20, max_sweeps_per_restart=100000, seed=42):
    dx = 2.0 / n
    best_count = np.inf
    best_h = cur.copy()
    rng = np.random.RandomState(seed)

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart == 0:
            c = cur.copy()
        else:
            c = best_h.copy()
            n_pert = rng.randint(3, 12 + restart)
            for _ in range(n_pert):
                oi = np.where(c == 1)[0]
                zi = np.where(c == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    c[rng.choice(oi)] = 0.0
                    c[rng.choice(zi)] = 1.0

        corr = np.correlate(c, 1.0 - c, mode='full').astype(np.float64)
        cur_max = corr.max()
        if cur_max < best_count:
            best_count = cur_max
            best_h = c.copy()

        stuck = 0
        for sweep in range(max_sweeps_per_restart):
            if time.time() > deadline:
                break
            zeros = np.where(c == 0)[0]
            ones = np.where(c == 1)[0]
            if len(zeros) == 0 or len(ones) == 0:
                break

            A = build_A(c, n)
            T = cur_max - corr

            A_z = A[zeros, :]
            A_o = A[ones, :]
            diff_mat = A_o[None, :, :] - A_z[:, None, :] - T[None, None, :]
            violations = diff_mat.max(axis=2)
            min_v = float(violations.min())

            if min_v < -1e-10:
                pi, qi = np.unravel_index(violations.argmin(), violations.shape)
                c[zeros[pi]] = 1.0
                c[ones[qi]] = 0.0
                corr = np.correlate(c, 1.0 - c, mode='full').astype(np.float64)
                cur_max = corr.max()
                stuck = 0
                if cur_max < best_count:
                    best_count = cur_max
                    best_h = c.copy()
            else:
                stuck += 1
                if stuck > 3:
                    break

    return best_h, float(best_count * dx)

def slsqp_polish(h, n, deadline, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        alpha = 100000.0
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_topk(h, n, deadline, maxiter=3000, topk=5):

    g0 = np.correlate(h, 1.0 - h, mode='full') * dx
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(g)[::-1][:topk]
        vals = g[idx]
        m = vals.max()
        alpha = 500.0
        sh = alpha * (vals - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h
