            if s < 1e-14:

    corr = np.correlate(h, 1 - h, mode='full')
    t = float(corr.max())

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub_list = []
    b_ub_list = []

            g[v2] -= h[v2 - k]

        b_ub_list.append(rhs)
