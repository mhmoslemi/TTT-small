This is LINEAR in h (and t)! So I can use a standard LP/QP solver.

1. Start with current h
2. Linearize all constraints at h
3. Solve the resulting LP
4. Update h
5. Repeat

def slp_optimize(h_init, n, max_iter, deadline):
    """Successive Linear Programming for the Erdős problem."""
    h = h_init.copy()
    dx = 2.0 / n
    S = n / 2.0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        
        # Linearize each f_k at current h
        # f_k(h) = sum_i h[i](1-h[i+k]) (for valid i)
        # grad f_k[i] = (1-h[i+k]) - h[i-k] (with boundary conditions)
        
        # Build LP:
        # min t
        # s.t. g_k + grad_k^T (x - h) ≤ t  for all k
        #      sum x = n/2
        #      0 ≤ x ≤ 1
        
        # This is: min t
        # s.t. grad_k^T x ≤ t - g_k + grad_k^T h  for all k
        #      sum x = n/2
        #      0 ≤ x ≤ 1
        
        # Use cvxpy for this
        try:
            import cvxpy as cp
        except:
            return h
        
        x = cp.Variable(n)
        t = cp.Variable()
        constraints = [x.sum() == S, x >= 0, x <= 1]
        
        # Only add constraints for the top lags (for speed)
        top_k = 20
        top_indices = np.argsort(g)[-top_k:]
        
        for k_idx in top_indices:
            lag = k_idx - (n - 1)
            # grad f_k[i] for valid i
            grad = np.zeros(n)
            for i in range(n):
                # f_k[i] contribution: h[i] * (1-h[i+lag]) if i+lag < n
                # d/dh[i] of h[i]*(1-h[i+lag]) = (1-h[i+lag])
                # d/dh[i] of h[i-lag]*(1-h[i]) = -h[i-lag] (if i-lag >= 0)
                c = 0.0
                if 0 <= i + lag < n:
                    c += (1.0 - h[i + lag])
                if 0 <= i - lag < n:
                    c -= h[i - lag]
                grad[i] = c
            constraints.append(grad @ x <= t - g[k_idx] + grad @ h)
        
        # Also add overall max constraint
        constraints.append(t >= 0)
        
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLOP)  # or ECOS, or CLARABEL
        except:
            try:
                prob.solve(solver=cp.ECOS)
            except:
                break
        
        if x.value is None:
            break
        new_h = np.clip(x.value, 0, 1)
        new_h = project_feasible(new_h, n)
        
        new_c = compute_c5(new_h, n)
        old_c = compute_c5(h, n)
        
        if new_c < old_c + 1e-10:  # accept if not worse
            h = new_h

            alpha_step = 0.5
            new_h = (1 - alpha_step) * h + alpha_step * np.clip(x.value, 0, 1)
            new_h = project_feasible(new_h, n)
            h = new_h
    
    return h
