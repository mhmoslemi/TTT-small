def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:

            s = room.sum()
            if s < 1e-14:

            if s < 1e-14:

        flat = new_maxes.argmin()
        ii = flat // len(ones)
        jj = flat % len(ones)
        best_val = new_maxes[ii, jj]
        i = zeros[ii]
        j = ones[jj]
        new_c = best_val * dx

        if new_c < cur_c - 1e-14:
            g, h, corr, zeros, ones = do_swap(g, h, corr, i, j, n, zeros, ones, ii, jj)
            cur_c = corr.max() * dx
            stall = 0
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            stall += 1
            no_improve += 1
            if stall > 20 or (no_improve > 150 and kick_count < 30):
                kick_count += 1
                nk = min(2, len(zeros), len(ones))
                ri = rng.choice(len(zeros), nk, replace=False)
                rj = rng.choice(len(ones), nk, replace=False)
                for a in range(nk):
                    iz = np.where(zeros == zeros[ri[a]])[0]
                    jo = np.where(ones == ones[rj[a]])[0]
                    if len(iz) > 0 and len(jo) > 0:
                        g, h, corr, zeros, ones = do_swap(
                            g, h, corr, zeros[ri[a]], ones[rj[a]], n, zeros, ones, iz[0], jo[0])
                cur_c = corr.max() * dx
                stall = 0

    return best_h, best_c

def compute_gradient(h, p, n):
    grad = np.ones(n)
    for i in range(n):
        lo1 = max(0, i)
        hi1 = min(2 * n - 2, i + n - 1)
        if lo1 <= hi1:
            h_idx = i - np.arange(lo1, hi1 + 1) + (n - 1)
            grad[i] -= np.dot(p[lo1:hi1 + 1], h[h_idx])
        lo2 = max(0, n - 1 - i)
        hi2 = min(2 * n - 2, 2 * n - 2 - i)
        if lo2 <= hi2:
            h_idx = i + np.arange(lo2, hi2 + 1) - (n - 1)
            grad[i] -= np.dot(p[lo2:hi2 + 1], h[h_idx])
    return grad

def gradient_polish(h_start, n, deadline, alpha_init=20000):
    dx = 2.0 / n
    h = h_start.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.8
    beta1, beta2, eps = 0.8, 0.999, 1e-10
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    no_improve = 0
    alpha = alpha_init

    while time.time() < deadline:
        t += 1
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sh -= sh.max()
        p = np.exp(sh)
        ps = p.sum()
        if ps < 1e-30:
            break
        p = p / ps

        grad = compute_gradient(h, p, n)

        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad ** 2
        m_hat = m_v / (1 - beta1 ** t)
        v_hat = v_v / (1 - beta2 ** t)

        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)

        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        h = h_new
        lr *= 0.9997

        if no_improve > 100:
            alpha = min(alpha * 1.3, 500000)
            lr = max(0.05, lr * 0.6)
            m_v = np.zeros(n)
            v_v = np.zeros(n)
            t = 0
            no_improve = 0

        if t > 50000 and lr < 0.02:
            break

    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def round_and_refine(h, n, deadline):
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    c = compute_c5(h_bin, n)
    if time.time() < deadline - 5:
        hb, cb = fast_binary_search(h_bin, n, deadline, top_k=8, seed=999)
        if cb < c:
            h_bin, c = hb, cb
    return h_bin, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.zeros(n)
        h0[np.arange(0, n, 2)[:n // 2]] = 1.0

    best_h = np.array(h0, dtype=float)
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary swap optimization (35% time)
    p1 = t0 + 0.35 * (budget_s - 5)
    if time.time() < p1:
        hb, cb = fast_binary_search(h0, n, p1, top_k=12, seed=seed)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 2: Continuous gradient polish (30% time)
    p2 = t0 + 0.65 * (budget_s - 5)
    if time.time() < p2:
        h_cont = project_feasible(best_h + 0.05 * rng.randn(n), n)
        hb, cb = gradient_polish(h_cont, n, p2 - 10, alpha_init=20000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        # Round and refine
        if time.time() < p2 - 5:
            hr, cr = round_and_refine(best_h, n, p2)
            if cr < best_c:
                best_c = cr
                best_h = hr.copy()

    # Phase 3: SLSQP with multiple starts (20% time)
    p3 = t0 + 0.85 * (budget_s - 5)
    if time.time() < p3:
        for attempt in range(3):
            if time.time() > p3 - 10:
                break
            perturbed = project_feasible(best_h + 0.03 * rng.randn(n), n)
            alpha = int(50000 * (attempt + 1) ** 2)
            max_it = 200 + attempt * 100
            hf = slsqp_smooth(perturbed, n, alpha, max_it, p3 - 5)
            c = compute_c5(hf, n)
            if c < best_c:
                best_c = c
                best_h = hf.copy()
            # Also try rounding the SLSQP result
            if time.time() < p3 - 5:
                hr, cr = round_and_refine(hf, n, p3 - 3)
                if cr < best_c:
                    best_c = cr
                    best_h = hr.copy()

    # Phase 4: Final binary polish (remaining time)
    if time.time() < dl - 3:
        hb, cb = fast_binary_search(best_h, n, dl - 2, top_k=10, seed=seed + 500)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Final: ensure feasibility and compute exact value
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
