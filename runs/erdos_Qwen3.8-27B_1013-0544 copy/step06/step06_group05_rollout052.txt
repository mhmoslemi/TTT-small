    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-12:

        if diff > 0:

            s = room.sum()

            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha, top_k=None):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if top_k is not None and top_k < len(c):

        c = c[idx]
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

            x[i] -= 2*eps

        # If 0->1: (1-cur[p-Lk] if 0<=p-Lk<n) - (cur[p+Lk] if 0<=p+Lk<n)

        # The second term: p+Lk valid means 0<=p+Lk<n, i.e., -Lk<=p<n-Lk
        
        # Term 1: 1-cur[p-Lk] for p in [max(0,Lk), min(n, n+Lk))
        p1_min = max(0, Lk)

            deltas[idx, k] += 1.0 - cur[idx - Lk]

        p2_min = max(0, -Lk)
        p2_max = min(n, n - Lk)
        if p2_min < p2_max:

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + (diff / s) * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + (diff / s) * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def precompute_deltas(cur, n):
    m = 2 * n - 1
    L = np.arange(-(n - 1), n)
    deltas = np.zeros((n, m))
    for k in range(m):
        Lk = int(L[k])
        p1_min = max(0, Lk)
        p1_max = min(n, n + Lk)
        if p1_min < p1_max:
            idx = np.arange(p1_min, p1_max)
            deltas[idx, k] += 1.0 - cur[idx - Lk]
        p2_min = max(0, -Lk)
        p2_max = min(n, n - Lk)
        if p2_min < p2_max:
            idx2 = np.arange(p2_min, p2_max)
            deltas[idx2, k] -= cur[idx2 + Lk]
    return deltas

def slsqp_opt(h, n, alpha, maxiter, deadline):
