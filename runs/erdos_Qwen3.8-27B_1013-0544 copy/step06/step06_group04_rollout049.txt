import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgradient_nesterov(h_init, n, deadline, alpha0=1.0):
    h = h_init.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x_prev = h.copy()
    
    t = 0
    while time.time() < deadline:
        if t > 0:
            beta = (t - 1.0) / (t + 2.0)
            y = h + beta * (h - x_prev)
        else:
            y = h.copy()
        
        y_p = project_feasible(y, n)
        
        corr = np.correlate(y_p, 1.0 - y_p, mode='full')
        cur_c = float(corr.max()) * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = y_p.copy()
        
        idx_star = int(np.argmax(corr))
        d = idx_star - (n - 1)
        
        g = np.zeros(n)
        if d >= 0:
            if d < n:
                g[:n-d] += (1.0 - y_p[d:n])
                g[d:n] -= y_p[:n-d]
        else:
            d2 = -d
            if d2 < n:
                g[d2:n] += (1.0 - y_p[:n-d2])
                g[:n-d2] -= y_p[d2:n]
        
        alpha = alpha0 / (t + 1.0)
        x_prev = h.copy()
        h = h - alpha * g
        h = project_feasible(h, n)
        t += 1
    
    return best_h, best_c

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def slsqp_smooth(h, n, alpha, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    cons_fun = lambda x: x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def local_search_binary(h, n, deadline, max_sweeps=2000, seed=42):
    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    # Precompute swap deltas more efficiently
    def get_delta(h, i, j, n):
        d = np.zeros(2 * n - 1)
        # Swap: h[i] 0->1, h[j] 1->0
        # Change in corr[k] for each k:
        # corr[k] = sum_m h[m] * (1-h[m+k-d0]) where d0 = n-1
        # When h[i] changes from 0 to 1:
        #   corr[k] increases by (1-h[i+d]) where d = k-(n-1), if valid
        # When h[j] changes from 1 to 0:
        #   corr[k] decreases by (1-h[j+d]) where d = k-(n-1), if valid
        #   AND corr[k] increases by h[j+d] where d = k-(n-1) [from the (1-h[...]) term]
        
        # For h[i]: 0->1
        d0 = n - 1
        lo = max(0, -(i - (n-1)))  # d such that i+d >= 0, d >= -(i)
        # Actually let me use the direct formula
        # corr[idx] = sum_m h[m] * (1-h[m + idx - (n-1)])
        # Contribution of h[i] to corr[idx]: h[i] * (1-h[i + idx - (n-1)]) if 0 <= i + idx - (n-1) < n
        # So for h[i] going 0->1, corr[idx] increases by (1-h[i+idx-(n-1)]) if valid
        # i + idx - (n-1) is in [0, n-1] means idx is in [(n-1)-i, (n-1)-i+n-1] = [n-1-i, 2n-2-i]
        
        a = max(0, n - 1 - i)
        b = min(2 * n - 2, 2 * n - 2 - i)
        if a <= b:
            # corr[a:b+1] += (1 - h[i + (a:b+1) - (n-1)])
            idxs = np.arange(a, b + 1)
            h_idx = i + idxs - (n - 1)  # these are in [0, n-1]
            d[a:b+1] += (1.0 - h[h_idx])
        
        # Also, h[i] appears in the (1-h[...]) part:
        # 1-h[m + idx - (n-1)] where m + idx - (n-1) = i, i.e., idx = (n-1) + i - m
        # For each valid m: idx = (n-1) + i - m, which ranges from (n-1)+i-(n-1) to (n-1)+i-0 = i to n-1+i
        # So m ranges from 0 to n-1, and idx = n-1+i-m
        # But we need 0 <= m < n (always) and 0 <= idx < 2n-1
        # idx = n-1+i-m, for m=0: idx=n-1+i; for m=n-1: idx=i
        # So idx ranges from i to n-1+i
        a2 = max(0, i)
        b2 = min(2 * n - 2, n - 1 + i)
        if a2 <= b2:
            idxs2 = np.arange(a2, b2 + 1)
            m_vals = (n - 1 + i) - idxs2  # m = n-1+i-idx
            d[a2:b2+1] -= h[m_vals]
        
        # For h[j]: 1->0 (negative of the above with i replaced by j)
        a = max(0, n - 1 - j)
        b = min(2 * n - 2, 2 * n - 2 - j)
        if a <= b:
            idxs = np.arange(a, b + 1)
            h_idx = j + idxs - (n - 1)
            d[a:b+1] -= (1.0 - h[h_idx])
        
        a2 = max(0, j)
        b2 = min(2 * n - 2, n - 1 + j)
        if a2 <= b2:
            idxs2 = np.arange(a2, b2 + 1)
            m_vals = (n - 1 + j) - idxs2
            d[a2:b2+1] += h[m_vals]
        
        return d
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                delta = get_delta(h, int(i), int(j), n)
                nc = float((corr + delta).max()) * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + delta
                    cur_c = nc
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return best_h, best_c

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Use bisection to find the right scaling
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-12:
            return h

def project_feasible(h, n):
    target = n / 2.0
    # Bisection on the offset
    lo = -max(h.max(), 0.0) - 1.0
    hi = 1.0 - min(h.min(), 1.0) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_t = np.clip(h + mid, 0.0, 1.0)
        s = h_t.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction for exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 0:
                h += diff / s * room
        else:
            s = h.sum()
            if s > 0:
                h += diff / s * h
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo = -abs(h.max()) - 2.0
    hi = 2.0 - abs(h.min())
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h = h + diff / s * room
        else:
            s = h.sum()
            if s > 1e-14:
                h = h + diff / s * h
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def compute_swap_delta(h, i, j, n):
    d = np.zeros(2 * n - 1)
    d0 = n - 1
    a = max(0, d0 - i)
    b = min(2 * n - 2, 2 * n - 2 - i)
    if a <= b:
        idxs = np.arange(a, b + 1)
        d[a:b+1] += (1.0 - h[i + idxs - d0])
    a2 = max(0, i)
    b2 = min(2 * n - 2, d0 + i)
    if a2 <= b2:
        idxs2 = np.arange(a2, b2 + 1)
        d[a2:b2+1] -= h[d0 + i - idxs2]
    a = max(0, d0 - j)
    b = min(2 * n - 2, 2 * n - 2 - j)
    if a <= b:
        idxs = np.arange(a, b + 1)
        d[a:b+1] -= (1.0 - h[j + idxs - d0])
    a2 = max(0, j)
    b2 = min(2 * n - 2, d0 + j)
    if a2 <= b2:
        idxs2 = np.arange(a2, b2 + 1)
        d[a2:b2+1] += h[d0 + j - idxs2]
    return d

def local_search_binary(h, n, deadline, max_sweeps=5000, seed=42):
    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                delta = compute_swap_delta(h, int(i), int(j), n)
                nc = float((corr + delta).max()) * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + delta
                    cur_c = nc
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def subgradient_nesterov(h_init, n, deadline, alpha0=1.0):
    h = h_init.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x_prev = h.copy()
    d0 = n - 1
    t = 0
    while time.time() < deadline:
        if t > 0:
            beta = (t - 1.0) / (t + 2.0)
            y = h + beta * (h - x_prev)
        else:
            y = h.copy()
        y_p = project_feasible(y, n)
        corr = np.correlate(y_p, 1.0 - y_p, mode='full')
        cur_c = float(corr.max()) * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = y_p.copy()
        idx_star = int(np.argmax(corr))
        d = idx_star - d0
        g = np.zeros(n)
        if d >= 0:
            if d < n:
                g[:n-d] += (1.0 - y_p[d:n])
                g[d:n] -= y_p[:n-d]
        else:
            d2 = -d
            if d2 < n:
                g[d2:n] += (1.0 - y_p[:n-d2])
                g[:n-d2] -= y_p[d2:n]
        alpha = alpha0 / float(t + 1)
        x_prev = h.copy()
        h = h - alpha * g
        h = project_feasible(h, n)
        t += 1
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
