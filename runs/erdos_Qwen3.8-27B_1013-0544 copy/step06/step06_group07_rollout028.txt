def compute_soft_gradient(h, weights, n):
    """Compute gradient of soft-max objective."""
    n_lags = 2*n - 1
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for k in range(n_lags):
        wk = weights[k]
        if wk < 1e-8:
            continue
        lag = lags[k]
        # ∂c[k]/∂h[i] = (1-h[i-lag]) * [0≤i-lag<n] - h[i+lag] * [0≤i+lag<n]
        # For valid i-lag:
        i_range = np.arange(max(0, lag), min(n, n+lag))  # i such that 0≤i-lag<n
        valid1 = (i_range - lag >= 0) & (i_range - lag < n)
        # Actually i_range already ensures this
        # Term 1: (1-h[i-lag]) for i in i_range
        # i-lag ranges from 0 to min(n-1, n-1+lag) if lag>0, or from -lag to n-1-lag if lag<0
        if lag >= 0:
            # i ranges from lag to n-1, i-lag ranges from 0 to n-1-lag
            idx1 = np.arange(lag, n)
            term1 = (1 - h[idx1 - lag])
            # i+lag ranges from 2*lag to n-1+lag, valid when < n, so i < n-lag
            idx2 = np.arange(0, n - lag) if lag > 0 else np.arange(0, n)
            term2 = h[idx2 + lag] if lag > 0 else h[idx2]
        else:
            # i ranges from 0 to n-1+lag (= n-1+|lag|... wait)
            # i-lag = i+|lag|, valid when i+|lag| < n, so i < n-|lag|
            idx1 = np.arange(0, n + lag)  # n+lag = n-|lag|
            term1 = (1 - h[idx1 - lag])
            # i+lag = i-|lag|, valid when i>=|lag|, so i from |lag| to n-1
            idx2 = np.arange(-lag, n)
            term2 = h[idx2 + lag]
        
        grad[idx1] += wk * term1
        grad[idx2] -= wk * term2
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full').astype(float)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = c - m
    log_sum = np.log(np.sum(np.exp(alpha * shifted)))
    return m + log_sum / alpha

def soft_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    weights = exp_c / exp_c.sum()
    
    # Compute gradient
    n_lags = 2 * n - 1
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    
    for k in range(n_lags):
        wk = weights[k]
        if wk < 1e-10:
            continue
        lag = lags[k]
        if lag >= 0:
            idx1 = np.arange(lag, n)
            term1 = (1.0 - h[idx1 - lag])
            if lag > 0:
                idx2 = np.arange(0, n - lag)
                term2 = h[idx2 + lag]
            else:
                idx2 = np.arange(0, n)
                term2 = h[idx2]
        else:
            abs_lag = -lag
            idx1 = np.arange(0, n - abs_lag)
            term1 = (1.0 - h[idx1 + abs_lag])
            idx2 = np.arange(abs_lag, n)
            term2 = h[idx2 - abs_lag]
        
        grad[idx1] += wk * term1
        grad[idx2] -= wk * term2
    
    return grad * dx

def projected_gradient_descent(h0, n, deadline, alpha_start=50, alpha_end=500000, 
                                total_iters=5000, seed=42):
    h = h0.copy()
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(total_iters):
        if time.time() > deadline:
            break
        # Adaptive alpha
        progress = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        # Compute objective and gradient
        f = soft_max_obj(h, n, alpha)
        g = soft_max_grad(h, n, alpha)
        
        # Project gradient to handle the equality constraint
        # The constraint is sum(h) = n/2
        # Project: g_proj = g - mean(g) (project out the constant direction)
        g_proj = g - np.mean(g)
        
        # Line search
        best_step = 1.0
        best_f = f
        for step in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
            h_trial = project_feasible(h - step * g_proj, n)
            f_trial = soft_max_obj(h_trial, n, alpha)
            if f_trial < best_f:
                best_f = f_trial
                best_step = step
                h_trial_best = h_trial
            else:
                break
        
        h = h_trial_best
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def slsqp_refine(h0, n, deadline, alphas=None):
    if alphas is None:
        alphas = [500, 2000, 10000, 50000, 200000, 500000, 1000000]
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            c = np.correlate(x, 1-x, mode='full') * (2.0/n)
            m = c.max()
            sh = alpha * (c - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh-sm).sum())) / alpha
        
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-16})
            h = project_feasible(res.x, n)
        except Exception:
            pass
    
    return h

def soft_max_grad_fast(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    weights = exp_c / exp_c.sum()
    
    # For large alpha, only a few lags have significant weight
    # Prune to top lags
    n_lags = 2 * n - 1
    grad = np.zeros(n)
    
    # Sort by weight and keep top ones
    thresh = 1e-8
    for k in range(n_lags):
        wk = weights[k]
        if wk < thresh:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            idx1 = np.arange(lag, n)
            grad[idx1] += wk * (1.0 - h[idx1 - lag])
            if lag > 0:
                idx2 = np.arange(0, n - lag)
                grad[idx2] -= wk * h[idx2 + lag]
            else:
                grad -= wk * h
        else:
            al = -lag
            idx1 = np.arange(0, n - al)
            grad[idx1] += wk * (1.0 - h[idx1 + al])
            idx2 = np.arange(al, n)
            grad[idx2] -= wk * h[idx2 - al]
    
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = c - m
    exp_c = np.exp(alpha * shifted)
    log_sum = np.log(exp_c.sum())
    f = m + log_sum / alpha
    weights = exp_c / exp_c.sum()
    
    n_lags = 2 * n - 1
    grad = np.zeros(n)
    thresh = 1e-10
    for k in range(n_lags):
        wk = weights[k]
        if wk < thresh:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            idx1 = np.arange(lag, n)
            grad[idx1] += wk * (1.0 - h[idx1 - lag])
            if lag > 0:
                idx2 = np.arange(0, n - lag)
                grad[idx2] -= wk * h[idx2 + lag]
            else:
                grad -= wk * h
        else:
            al = -lag
            idx1 = np.arange(0, n - al)
            grad[idx1] += wk * (1.0 - h[idx1 + al])
            idx2 = np.arange(al, n)
            grad[idx2] -= wk * h[idx2 - al]
    return f, grad * dx

        progress = it / n_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        f, g = soft_max_obj_and_grad(h, n, alpha)
        
        # Project gradient onto equality constraint (remove mean)
        g = g - np.mean(g)
        
        # Backtracking line search
        best_step = 0.001
        best_f = f

        c_val = compute_c5(h, n)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
    
    return best_h, best_c

def slsqp_opt(h0, n, deadline, alphas):
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            c = np.correlate(x, 1-x, mode='full') * (2.0/n)
            m = c.max()
            sh = alpha * (c - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh-sm).sum())) / alpha
        try:

                          constraints=con, options={'maxiter': 5000, 'ftol': 1e-16})
            h = project_feasible(res.x, n)
        except Exception:
            pass
    return h
