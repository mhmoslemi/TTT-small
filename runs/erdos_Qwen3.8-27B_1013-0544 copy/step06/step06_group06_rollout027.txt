def project_feasible(y):
    n = len(y)
    target = 0.5 * n
    y = np.asarray(y, dtype=float).copy()
    y = np.clip(y, 0.0, 1.0)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    # final correction
    for _ in range(20):
        diff = target - x.sum()
        if abs(diff) < 1e-13: break
        if diff > 0:
            room = 1.0 - x
            sr = room.sum()
            if sr < 1e-15: break
            x = x + (diff / sr) * room
        else:
            sx = x.sum()
            if sx < 1e-15: break

    n = len(h)

def project_fast(y):
    n = len(y)
    target = 0.5 * n
    y = np.clip(np.asarray(y, dtype=float), 0.0, 1.0).copy()
    for _ in range(30):
        diff = target - y.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - y
            sr = room.sum()
            if sr < 1e-15:
                break
            y = y + (diff / sr) * room
        else:
            sy = y.sum()
            if sy < 1e-15:
                break
            y = y + (diff / sy) * y
    return np.clip(y, 0.0, 1.0)

def project_exact(y):
    y = np.clip(np.asarray(y, dtype=float), 0.0, 1.0).copy()

    hi = float(y.max()) + 1.0
    for _ in range(50):

       s=np.sum(np.clip(y-mid,0,1))
       if s > target: lo=mid
       else: hi=mid
    x=np.clip(y-0.5*(lo+hi),0,1)

def thue_binary(n, target, rng):
    t=np.zeros(n, dtype=np.uint8)
    for i in range(1,n):
       t[i] = t[i >> 1] ^ (i & 1)
    ones=np.where(t==1)[0]

import numpy as np
import time
from scipy.optimize import minimize

def corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5(h):
    return float(np.max(corr(h)) * (2.0 / len(h)))

def project(y):
    y = np.clip(np.asarray(y, float), 0.0, 1.0)
    t = 0.5 * len(y)
    for _ in range(60):
        d = t - y.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - y
            s = r.sum()
            if s < 1e-15:
                break
            y = y + (d / s) * r
        else:
            s = y.sum()
            if s < 1e-15:
                break
            y = y + (d / s) * y
    return np.clip(y, 0.0, 1.0)

def soft_info(h, alpha):
    n = len(h)
    dx = 2.0 / n
    D = corr(h)
    m = float(D.max())
    z = alpha * (D - m)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        w = np.zeros_like(D)
        w[D >= m - 1e-12] = 1.0
        if w.sum() == 0.0:
            w[0] = 1.0
        w /= w.sum()
    else:
        w = e / s
    return D, w, dx * (m + float(np.log(s)) / alpha), m * dx

def gradw(h, w):
    n = len(h)
    dx = 2.0 / n
    c = n - 1
    g = np.zeros(n)
    g += w[c] * (1.0 - 2.0 * h)
    if n > 1:
        p = w[c + 1:]
        q = w[c - 1::-1]
        for l in range(1, n):
            a = p[l - 1]
            b = q[l - 1]
            if a or b:
                s = a + b
                g[:-l] += a - s * h[l:]
                g[l:] += b - s * h[:-l]
    return dx * g

def pg(h0, alpha, dl, lr, rng, ns):
    h = project(h0.copy())
    D, w, F, c = soft_info(h, alpha)
    bh = h.copy()
    bc = c
    base = lr
    na = 0
    ni = 0
    while time.time() < dl:
        g = gradw(h, w)
        ok = False
        lr0 = base
        for _ in range(12):
            h2 = project(h - lr0 * g)
            if np.array_equal(h2, h):
                lr0 *= 0.5
                if lr0 < 1e-9:

def pen_fun(x, alpha, pen):
    D, w, F, c = soft_info(x, alpha)
    return F + pen * (x.sum() - 0.5 * len(x)) ** 2

def pen_jac(x, alpha, pen):
    D, w, F, c = soft_info(x, alpha)
    return gradw(x, w) + 2.0 * pen * (x.sum() - 0.5 * len(x)) * np.ones_like(x)

def lbfgs(h, alpha, pen, it):
    h = project(h.copy())
    bh = h.copy()
    bc = c5(h)
    try:
        res = minimize(pen_fun, h, method='L-BFGS-B', bounds=[(0.0, 1.0)] * len(h),
                       jac=pen_jac, args=(alpha, pen),
                       options={'maxiter': int(it), 'ftol': 1e-15, 'gtol': 1e-8, 'maxls': 25})
        if hasattr(res, 'x') and res.x is not None:
            h2 = project(res.x)
            c2 = c5(h2)
            if c2 < bc:
                bh, bc = h2, c2
    except Exception:
        pass
    return bh, bc

def exactpol(h0, dl, rng):

def softseq(h0, dl, alphas, rng):

def binb(hf, n, tg):
    b = np.zeros(n)
    if tg <= 0:
        return b
    if tg >= n:
        b.fill(1.0)
        return b
    b[np.argsort(hf)[::-1][:tg]] = 1.0

def randb(n, tg, rng):
    b = np.zeros(n)
    if 0 < tg < n:
        b[rng.choice(n, tg, replace=False)] = 1.0
    elif tg == n:
        b.fill(1.0)
    return b

def thue(n, tg, rng):
    t = np.zeros(n, dtype=np.uint8)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    on = np.where(t == 1)[0]
    off = np.where(t == 0)[0]
    if len(on) > tg:
        t[on[rng.choice(len(on), len(on) - tg, replace=False)]] = 0
    elif len(on) < tg:
        t[off[rng.choice(len(off), tg - len(on), replace=False)]] = 1
    return t.astype(float)

def delta01(pos, h, act):
    n = len(h)
    c = n - 1
    m = len(pos)
    out = np.zeros((m, len(act)))
    if m == 0 or len(act) == 0:
        return out
    p = np.asarray(pos, int)
    for ai, r in enumerate(act):
        d = abs(int(r) - c)
        val = np.zeros(m)
        if d:
            if r > c:
                mk = p + d < n
                val[mk] += 1.0 - h[p[mk] + d]
                mk = p >= d
                val[mk] -= h[p[mk] - d]
            else:
                mk = p >= d
                val[mk] += 1.0 - h[p[mk] - d]
                mk = p + d < n
                val[mk] -= h[p[mk] + d]
        out[:, ai] = val
    return out

def bipol(hb, dl, A, seed, test=8):

    tg = int(round(n / 2.0))

    bh = h.copy()

    ni = 0
    kc = 0
    L = 2 * n - 1
    while time.time() < dl:
        D = corr(h)
        md = float(np.max(D))
        uc = md * (2.0 / n)
        if uc < bc - 1e-14:
            bc = uc
            bh = h.copy()
        z = np.where(h < 0.5)[0]
        o = np.where(h > 0.5)[0]
        if len(z) == 0 or len(o) == 0:
            break
        ae = min(A, L)
        if ae < L:
            act = np.argpartition(D, L - ae)[-ae:]
        else:
            act = np.arange(L)
        nr = np.where(D >= md - 1.0 - 1e-12)[0]

        od = delta01(o, h, act)
        da = D[act]
        nm = (da[None, None, :] + zd[:, None, :] - od[None, :, :]).max(axis=2)
        fl = nm.ravel()
        nt = min(test, fl.size)
        if nt < fl.size:
            idxs = np.argpartition(fl, nt - 1)[:nt]
        else:
            idxs = np.argsort(fl)[:nt]
        found = False
        aa = len(act)
        O = len(o)
        for ix in idxs:
            if fl[ix] > md - 1e-12:
                continue
            pr = ix // aa
            zi = pr // O
            oi = pr % O
            h2 = h.copy()
            h2[z[zi]] = 1.0
            h2[o[oi]] = 0.0
            c2 = float(np.max(corr(h2)) * (2.0 / n))
            if c2 < uc - 1e-14:
                h = h2
                D = corr(h)
                uc = c2
                ni = 0
                found = True
                break

            ni += 1
            if ni > 20:
                ni = 0
                kc += 1
                temp = max(0.002, 0.012 * (0.95 ** kc))
                for _ in range(30):
                    i = int(z[int(rng.randint(len(z)))])
                    j = int(o[int(rng.randint(len(o)))])
                    h2 = h.copy()
                    h2[i] = 1.0
                    h2[j] = 0.0
                    c2 = float(np.max(corr(h2)) * (2.0 / n))
                    if c2 < uc or rng.rand() < np.exp(-(c2 - uc) / temp):
                        h = h2
                        D = corr(h)
                        uc = c2

def mixes(hs, rng, dl):

from scipy.optimize import minimize

def corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5(h):
    return float(np.max(corr(h)) * (2.0 / len(h)))

def project(y):
    y = np.clip(np.asarray(y, float), 0.0, 1.0)
    t = 0.5 * len(y)
    for _ in range(60):
        d = t - y.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - y
            s = r.sum()
            if s < 1e-15:
                break
            y = y + (d / s) * r
        else:
            s = y.sum()
            if s < 1e-15:
                break
            y = y + (d / s) * y
    y = np.clip(y, 0.0, 1.0)
    d = t - y.sum()
    if 1e-12 < d < 0.5:
        idx = int(np.argmax(1.0 - y))
        y[idx] = min(1.0, y[idx] + d)
    elif -0.5 < d < -1e-12:
        idx = int(np.argmin(y))
        y[idx] = max(0.0, y[idx] + d)
    return np.clip(y, 0.0, 1.0)

def soft_info(h, alpha):
    n = len(h)
    dx = 2.0 / n
    D = corr(h)
    m = float(D.max())
    z = alpha * (D - m)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        w = np.zeros_like(D)
        w[D >= m - 1e-12] = 1.0
        w[0] = 1.0
        w /= w.sum()
    else:
        w = e / s
    return D, w, dx * (m + float(np.log(s)) / alpha), m * dx

def gradw(h, w):
    n = len(h)
    dx = 2.0 / n
    c = n - 1
    g = np.zeros(n)
    g += w[c] * (1.0 - 2.0 * h)
    if n > 1:
        p = w[c + 1:]
        q = w[c - 1::-1]
        for l in range(1, n):
            a = p[l - 1]
            b = q[l - 1]
            if a or b:
                s = a + b
                g[:-l] += a - s * h[l:]
                g[l:] += b - s * h[:-l]
    return dx * g

def pg(h0, alpha, dl, lr, rng, ns):

                ok = True

            base = max(1e-7, base * 0.5)
            if na > 40:
                h = project(h + ns * rng.standard_normal(len(h)))
                D, w, F, c = soft_info(h, alpha)
                na = 0
                ni = 0
                if ns > 0.001:
                    ns *= 0.6
        else:
            na = 0
            if c < bc - 1e-14:
                bc = c
                bh = h.copy()
                ni = 0
            else:
                ni += 1
                if ni > 120 and ns > 0.0:
                    h = project(h + ns * rng.standard_normal(len(h)))
                    D, w, F, c = soft_info(h, alpha)
                    ni = 0
                    if ns > 0.001:
                        ns *= 0.6
    return bh, bc

def softseq(h0, dl, alphas, rng):
    h = project(h0.copy())
    bh = h.copy()
    bc = c5(h)
    seg = max(0.0, dl - time.time()) / max(1, len(alphas))
    for i, alpha in enumerate(alphas):
        if time.time() >= dl:
            break
        se = min(dl, time.time() + seg + 1.0)
        if time.time() + 5.0 < se:
            h2, c2 = pg(h, alpha, se, 0.25 if i == 0 else 0.10, rng, 0.02 if i == 0 else 0.01)
            if c2 < bc:
                bc = c2
                bh = h2.copy()
        h = bh.copy()
    return bh, bc

def pen_fun(x, alpha, pen):
    D, w, F, c = soft_info(x, alpha)
    return F + pen * (x.sum() - 0.5 * len(x)) ** 2

def pen_jac(x, alpha, pen):
    D, w, F, c = soft_info(x, alpha)
    return gradw(x, w) + 2.0 * pen * (x.sum() - 0.5 * len(x)) * np.ones_like(x)

def lbfgs(h, alpha, pen, it):
    h = project(h.copy())
    bh = h.copy()
    bc = c5(h)
    try:
        res = minimize(pen_fun, h, method='L-BFGS-B', bounds=[(0.0, 1.0)] * len(h), jac=pen_jac, args=(alpha, pen), options={'maxiter': int(it), 'ftol': 1e-15, 'gtol': 1e-8, 'maxls': 25})
        if hasattr(res, 'x') and res.x is not None:
            h2 = project(res.x)
            c2 = c5(h2)
            if c2 < bc:
                bh, bc = h2, c2
    except Exception:
        pass
    return bh, bc

def exactpol(h0, dl, rng):
    h = project(h0.copy())
    D = corr(h)
    bc = float(np.max(D) * (2.0 / len(h)))
    bh = h.copy()
    uc = bc
    base = 0.1
    na = 0
    ci = 0
    counts = (5, 20, 80)
    while time.time() < dl:
        n = len(h)
        L = len(D)
        a = min(counts[ci % 3], L)
        ci += 1
        if a < L:
            top = np.argpartition(D, L - a)[-a:]
        else:
            top = np.arange(L)
        nr = np.where(D >= D.max() - 0.5)[0]
        if len(nr):
            top = np.unique(np.concatenate((top, nr)))
            if len(top) > 80:
                top = top[np.argsort(D[top])[-80:]]
        w = np.zeros(L)
        w[top] = 1.0 / len(top)
        g = gradw(h, w)
        ok = False
        lr = base
        for _ in range(12):
            h2 = project(h - lr * g)
            c2 = c5(h2)
            if c2 < uc - 1e-14:
                h = h2
                D = corr(h)
                uc = c2
                ok = True
                base = min(1.0, base * 1.05)
                na = 0
                break
            lr *= 0.5
            if lr < 1e-8:
                break
        if not ok:
            na += 1
            base = max(1e-8, base * 0.5)
            if na > 40:
                h = project(h + 0.01 * rng.standard_normal(n))
                D = corr(h)
                uc = float(np.max(D) * (2.0 / n))
                na = 0
        if uc < bc - 1e-14:
            bc = uc
            bh = h.copy()
    return bh, bc

def binb(hf, n, tg):
    b = np.zeros(n)
    if tg <= 0:
        return b
    if tg >= n:
        b.fill(1.0)
        return b
    b[np.argsort(hf)[::-1][:tg]] = 1.0
    return b

def thue(n, tg, rng):
    t = np.zeros(n, dtype=np.uint8)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    on = np.where(t == 1)[0]
    off = np.where(t == 0)[0]
    if len(on) > tg:
        t[on[rng.choice(len(on), len(on) - tg, replace=False)]] = 0
    elif len(on) < tg:
        t[off[rng.choice(len(off), tg - len(on), replace=False)]] = 1
    return t.astype(float)

def delta01(pos, h, act):
    n = len(h)
    c = n - 1
    m = len(pos)
    out = np.zeros((m, len(act)))
    if m == 0 or len(act) == 0:
        return out
    p = np.asarray(pos, int)
    for ai, r in enumerate(act):
        d = abs(int(r) - c)
        val = np.zeros(m)
        if d:
            if r > c:
                mk = p + d < n
                val[mk] += 1.0 - h[p[mk] + d]
                mk = p >= d
                val[mk] -= h[p[mk] - d]
            else:
                mk = p >= d
                val[mk] += 1.0 - h[p[mk] - d]
                mk = p + d < n
                val[mk] -= h[p[mk] + d]
        out[:, ai] = val
    return out

def bipol(hb, dl, A, seed, test=8):
    n = len(hb)
    tg = int(round(n / 2.0))
    rng = np.random.RandomState(seed)
    h = binb(hb, n, tg)
    D = corr(h)
    uc = float(np.max(D) * (2.0 / n))
    bh = h.copy()
    bc = uc
    ni = 0
    kc = 0
    L = 2 * n - 1
    while time.time() < dl:
        D = corr(h)
        md = float(np.max(D))
        uc = md * (2.0 / n)
        if uc < bc - 1e-14:
            bc = uc
            bh = h.copy()
        z = np.where(h < 0.5)[0]
        o = np.where(h > 0.5)[0]
        if len(z) == 0 or len(o) == 0:
            break
        ae = min(A, L)
        if ae < L:
            act = np.argpartition(D, L - ae)[-ae:]
        else:
            act = np.arange(L)
        nr = np.where(D >= md - 1.0 - 1e-12)[0]
        if len(nr):
            act = np.unique(np.concatenate((act, nr)))
            if len(act) > A:
                act = act[np.argsort(D[act])[-A:]]
        zd = delta01(z, h, act)
        od = delta01(o, h, act)
        da = D[act]
        nm = (da[None, None, :] + zd[:, None, :] - od[None, :, :]).max(axis=2)
        fl = nm.ravel()
        nt = min(test, fl.size)
        if nt < fl.size:
            idxs = np.argpartition(fl, nt - 1)[:nt]
        else:
            idxs = np.argsort(fl)[:nt]
        found = False
        aa = len(act)
        O = len(o)
        for ix in idxs:
            if fl[ix] > md - 1e-12:
                continue
            pr = ix // aa
            zi = pr // O
            oi = pr % O
            h2 = h.copy()
            h2[z[zi]] = 1.0
            h2[o[oi]] = 0.0
            c2 = float(np.max(corr(h2)) * (2.0 / n))
            if c2 < uc - 1e-14:
                h = h2
                D = corr(h)
                uc = c2
                ni = 0
                found = True
                break
        if not found:
            ni += 1
            if ni > 20:
                ni = 0
                kc += 1
                temp = max(0.002, 0.012 * (0.95 ** kc))
                for _ in range(30):
                    i = int(z[int(rng.randint(len(z)))])
                    j = int(o[int(rng.randint(len(o)))])
                    h2 = h.copy()
                    h2[i] = 1.0
                    h2[j] = 0.0
                    c2 = float(np.max(corr(h2)) * (2.0 / n))
                    if c2 < uc or rng.rand() < np.exp(-(c2 - uc) / temp):
                        h = h2
                        D = corr(h)
                        uc = c2
                        break
                if kc % 5 == 0:
                    z = np.where(h < 0.5)[0]
                    o = np.where(h > 0.5)[0]
                    if len(z) and len(o):
                        kk = int(rng.randint(2, 6))
                        h2 = h.copy()
                        z2 = z.copy()
                        o2 = o.copy()
                        for _ in range(min(kk, len(z2), len(o2))):
                            iz = int(rng.randint(len(z2)))
                            jo = int(rng.randint(len(o2)))
                            h2[z2[iz]] = 1.0
                            h2[o2[jo]] = 0.0
                            z2 = np.delete(z2, iz)
                            o2 = np.delete(o2, jo)
                        c2 = float(np.max(corr(h2)) * (2.0 / n))
                        if c2 < bc + 0.05:
                            h = h2
                            D = corr(h)
                            uc = c2
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    tot = max(10.0, float(budget_s))
    dur = tot - 4.0
    dl = t0 + dur
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    h0 = project(h0)
    bh = h0.copy()
    bc = c5(h0)
    tg = int(round(n / 2.0))

    t1 = t0 + 0.22 * dur
    if time.time() < t1:
        hb, cb = bipol(binb(h0, n, tg), min(t1, dl), 160, seed + 1, 10)
        if cb < bc:
            bc, bh = cb, hb.copy()

    t2 = t0 + 0.38 * dur
    if time.time() < t2:
        hb, cb = bipol(thue(n, tg, rng), min(t2, dl), 160, seed + 7, 8)
        if cb < bc:
            bc, bh = cb, hb.copy()

    t_soft = t0 + 0.72 * dur
    if time.time() < t_soft:
        h2, c2 = softseq(bh, min(t_soft, dl), (5.0, 20.0, 100.0, 400.0), rng)
        if c2 < bc:
            bc, bh = c2, h2.copy()

    t_lb = t0 + 0.88 * dur
    if time.time() < t_lb and t_lb - time.time() > 50:
        rem = t_lb - time.time()
        it = int(min(45, max(15, rem * 0.35)))
        h2, c2 = lbfgs(bh, 300.0, 10.0, it)
        if c2 < bc:
            bc, bh = c2, h2.copy()
        if t_lb - time.time() > 40:
            h2, c2 = lbfgs(bh, 1000.0, 20.0, int(max(12, it * 0.6)))
            if c2 < bc:
                bc, bh = c2, h2.copy()

    t_ex = t0 + 0.95 * dur
    if time.time() < t_ex:
        h2, c2 = exactpol(bh, min(t_ex, dl), rng)
        if c2 < bc:
            bc, bh = c2, h2.copy()

    if time.time() < dl - 2:
        h2, c2 = pg(bh, 2000.0, dl - 1, 0.05, rng, 0.005)
        if c2 < bc:
            bc, bh = c2, h2.copy()

    bh = project(bh)
    bc = c5(bh)
    return bh, bc, n
