```python
def project_feasible(v, n):
    """Project v onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        tau = (lo + hi) / 2.0
        h = np.clip(v - tau, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    h = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # Fine-tune
    diff = target - h.sum()
    # Distribute remaining to non-saturated entries
    non_sat = (h > 1e-10) & (h < 1 - 1e-10)
    if non_sat.sum() > 0 and abs(diff) > 1e-12:
        h[non_sat] += diff / non_sat.sum()
        h = np.clip(h, 0.0, 1.0)
    return h
```

    for m_val in range(1, n):
        wp = w[n - 1 + m_val]
        wn = w[n - 1 - m_val]
        if wp < 1e-12 and wn < 1e-12:
            continue
        if wp > 1e-12:
            grad[:n - m_val] += wp * dx * (1.0 - h[m_val:])
            grad[m_val:] -= wp * dx * h[:n - m_val]
        if wn > 1e-12:
            grad[m_val:] += wn * dx * (1.0 - h[:n - m_val])
            grad[:n - m_val] -= wn * dx * h[m_val:]
    return val, grad

def proj_grad_descent(h, n, deadline, alpha_start=500, alpha_end=50000):
    h_c = h.copy()
    best_h = h_c.copy()
    best_c = compute_c5(h_c, n)
    n_iter = 0
    max_iter = 3000
    for alpha in np.geomspace(alpha_start, alpha_end, 8):
        s = 0.1
        for it in range(max_iter):
            n_iter += 1
            if time.time() > deadline:
                return best_h, best_c
            val, grad = grad_smooth_max(h_c, n, alpha)
            # Backtracking line search
            for _ in range(15):
                h_t = project_feasible(h_c - s * grad, n)
                c_t = compute_c5(h_t, n)
                if c_t < best_c or c_t < val - 0.1 * s * (grad @ (h_t - h_c)):
                    break
                s *= 0.5
            h_c = project_feasible(h_c - s * grad, n)
            c_c = compute_c5(h_c, n)
            if c_c < best_c - 1e-14:
                best_c = c_c
                best_h = h_c.copy()
            s = max(s * 0.95, 0.001)
    return best_h, best_c

def outer_approx_step(h, n, trust):
    dx = 2.0 / n
    c = get_lags(h, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.norm(h_var - h, np.inf) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        m_val = abs(lag)
        g = np.zeros(n)
        if m_val == 0:
            g[:] = 1.0 - 2.0 * h
        elif m_val < n:
            if lag > 0:
                g[:n - m_val] += (1.0 - h[m_val:])
                g[m_val:] -= h[:n - m_val]
            else:
                g[m_val:] += (1.0 - h[:n - m_val])
                g[:n - m_val] -= h[m_val:]
        cons.append(c[k] + dx * (g @ (h_var - h)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h.copy()
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return h.copy()
    if h_var.value is None:
        return h.copy()
    return project_feasible(h_var.value, n)

def outer_loop(h, n, deadline, max_iters=30):
    h_c = h.copy()
    best_h = h_c.copy()
    best_c = compute_c5(h_c, n)
    trust = 0.2
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        h_new = outer_approx_step(h_c, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_c))
        if step < 1e-9:
            break
        h_c = h_new
        trust = max(0.01, trust * 0.85)
    return best_h, best_c

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    delays = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in delays:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = get_lags(h, n)
            # Try top few lags
            top_lags = np.argsort(c)[-5:]
            for k_idx in top_lags:
                if time.time() > deadline:
                    break
                lag = k_idx - (n - 1)
                m_val = abs(lag)
                if m_val == 0 or m_val >= n:
                    continue
                for i in range(n - m_val):
                    if time.time() > deadline:
                        break
                    j = i + m_val
                    if h[i] > h[j] + 1e-10:
                        delta = min(d, (h[i] - h[j]) * 0.4, h[i] * 0.5, (1.0 - h[j]) * 0.5)
                        if delta < 1e-9:
                            continue
                        h[i] -= delta
