**Alternative: Use the dual problem / LP relaxation**

from scipy.optimize import minimize
import time

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def build_v(h, n):
    v = np.zeros((n, 2 * n - 1))
    for i in range(n):
        k_lo = max(0, i + 1 - n)
        k_hi = min(2 * n - 2, i)
        js = i + n - 1 - np.arange(k_lo, k_hi + 1)
        v[i, k_lo:k_hi + 1] = 1.0 - h[js]
    return v

def swap_search(h_init, n, deadline, top_k=150, seed=42, kick_size=5, kick_interval=100):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    target = n // 2
    h = project_binary(h_init, n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = corr.max()
    best_c = cur_max * dx
    best_h = h.copy()
    
    v = build_v(h, n)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    top_k = min(top_k, 2 * n - 1)
    arange_n = np.arange(n)
    
    stall = 0
    temp = 0.015
    no_improve = 0
    kicks_left = 200
    
    while time.time() < deadline:
        top_idx = np.argsort(corr)[-top_k:]
        corr_top = corr[top_idx]
        vz = v[zeros][:, top_idx]
        vo = v[ones][:, top_idx]
        D = corr_top[None, None, :] + vz[:, None, :] - vo[None, :, :]
        new_maxes = D.max(axis=2)
        flat = new_maxes.argmin()
        bi = int(flat // len(ones))
        bj = int(flat % len(ones))
        ii = zeros[bi]
        jj = ones[bj]
        test_corr = corr + v[ii] - v[jj]
        test_max = test_corr.max()
        
        if test_max < cur_max - 1e-15:
            h[ii], h[jj] = 1.0, 0.0
            corr = test_corr
            cur_max = test_max
            v[arange_n, arange_n + n - 1 - ii] -= 1.0
            v[arange_n, arange_n + n - 1 - jj] += 1.0
            zeros = np.delete(zeros, bi)
            ones = np.delete(ones, bj)
            zeros = np.append(zeros, jj)
            ones = np.append(ones, ii)

                no_improve = 0

                ri = rng.choice(len(zeros), nk, replace=False)
                rj = rng.choice(len(ones), nk, replace=False)
                tc = corr.copy()
                for a in range(nk):
                    tc += v[zeros[ri[a]]] - v[ones[rj[a]]]
                tc_max = tc.max()

```python
def compute_gradient(h, n, alpha):
    corr = np.correlate(h, 1 - h, mode='full')
    g = corr * (2.0 / n)  # this is c_k * dx
    # w_k = softmax(alpha * g_k)
    max_g = g.max()
    exp_g = np.exp(alpha * (g - max_g))
    w = exp_g / exp_g.sum()
    
    grad = np.zeros(n)
    for m in range(n):
        # First term: sum over k where 0 <= m+n-1-k <= n-1
        k_lo = max(0, m + 1 - n)
        k_hi = min(2*n-2, m)
        k_range = np.arange(k_lo, k_hi + 1)
        j_vals = m + n - 1 - k_range  # = m+n-1-k
        # Second term: sum over k where 0 <= m+k-n+1 <= n-1
        k_lo2 = max(0, n - 1 - m)
        k_hi2 = min(2*n-2, n - 1 + m)
        k_range2 = np.arange(k_lo2, k_hi2 + 1)
        i_vals = m + k_range2 - n + 1  # = m+k-n+1
        
        term1 = w[k_range].sum() - (w[k_range] * h[j_vals]).sum()
        term2 = (w[k_range2] * h[i_vals]).sum()
        grad[m] = term1 - term2
    
    return grad
```

This is O(n²) per gradient evaluation. For n=250, that's 62,500 operations per gradient. Very fast.

The projection onto {h: 0≤h≤1, ∑h=n/2} is done by water-filling:
1. Clip to [0,1]
2. If sum > n/2, reduce values (starting from the largest) until sum = n/2
3. If sum < n/2, increase values (starting from the smallest) until sum = n/2

        # = (1/alpha) * (alpha * max_g + log(sum(exp(alpha*(g - max_g)))))
        # = max_g + (1/alpha) * log(sum(exp(alpha*(g - max_g))))
        
        # Actually, let me use a simpler smoothing:
        # f = max_g + (1/alpha) * log(1 + sum_{k != k*} exp(alpha*(g_k - g_k*)))
        # where k* is argmax

        # f = (1/alpha) * log(sum(exp(alpha * g)))
