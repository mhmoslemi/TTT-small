Actually, L-BFGS-B already handles bounds [0,1]. But it doesn't handle the equality constraint sum(h) = n/2. I can either:

b) Use SLSQP which handles both bounds and equality constraints

Option (a) is simplest: add penalty * (sum(h) - n/2)^2 to the objective. With a large enough penalty, the constraint will be approximately satisfied. Then project at the end.

Let me write the full algorithm:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(alpha * (c - m))
    w = e / e.sum()
    obj = (m + np.log(e.sum()) / alpha) * dx
    # Add penalty
    s = x.sum() - n / 2.0
    obj += 0.5 * penalty * s * s
    
    u = 1.0 - x
    first = np.correlate(w, u, 'full')[:n]
    conv = np.convolve(x, w, 'full')
    second = conv[n-1:3*n-2][:n]
    grad = dx * (first - second)
    grad += penalty * s * np.ones(n)
    return obj, grad

def lbfqb(h, n, alpha, penalty, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj_and_grad, h, method='L-BFGS-B',
                      bounds=bounds, args=(n, alpha, penalty),
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def coordinate_descent(h, n, max_passes=200, grid=80, deadline=None, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            # Find affected correlation indices
            k_lo = max(0, n - 1 - p)
            k_hi = min(2 * n - 2, p + n - 1)
            # The affected c values are in [min(k_lo2, k_lo), max(k_hi2, k_hi)]
            k_lo2 = p
            k_hi2 = min(p + n - 1, 2 * n - 2)
            k_start = min(k_lo, k_lo2)
            k_end = max(k_hi, k_hi2)
            um = 0.0
            if k_start > 0:
                um = float(np.max(c[:k_start]))
            if k_end < 2 * n - 2:
                um = max(um, float(np.max(c[k_end + 1:])))
            ks = np.arange(k_start, k_end + 1)
            # Compute the linear coefficient s for each k
            s = np.zeros(len(ks))
            for ki, k in enumerate(ks):
                # First part: p in S(k)?
                if k_lo <= k <= k_hi:
                    s[ki] += 1.0 - h[p + k - n + 1]
                # Second part: p-k+n-1 in S(k)?
                j = p - k + n - 1
                if k_lo2 <= k <= k_hi2:
                    s[ki] -= h[j]
            # Grid search for best d
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = current_max
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                new_c = c[ks] + d * s
                # Self-correlation at k=n-1
                ci = (ks == (n - 1))
                if np.any(ci):
                    new_c[ci] = c[n-1] + d * (1.0 - 2.0 * h[p]) - d * d
                val = max(um, float(np.max(new_c)))
                if val < best_val - 1e-14:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                d = best_d
                for ki, k in enumerate(ks):
                    if s[ki] != 0:
                        c[k] += d * s[ki]
                # Fix self-correlation
                c[n-1] -= d * d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                current_max = max(um, float(np.max(c[ks])))
                improved = True
        if not improved:
            break
    return h, current_max * (2.0 / n)
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(alpha * (c - m))
    ssum = e.sum()
    obj = (m + np.log(ssum) / alpha) * dx
    s = x.sum() - n / 2.0
    obj += 0.5 * penalty * s * s
    u = 1.0 - x
    first = np.correlate(e / ssum, u, 'full')[:n]
    conv = np.convolve(x, e / ssum, 'full')
    second = conv[n - 1:3 * n - 2][:n]
    grad = dx * (first - second) + penalty * s
    return obj, grad

def lbfqb(h, n, alpha, penalty, maxiter=800):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, args=(n, alpha, penalty),
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def coord_descent(h, n, max_passes=300, grid=100, deadline=None, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            k_lo1 = max(0, n - 1 - p)
            k_hi1 = min(2 * n - 2, p + n - 1)
            k_lo2 = p
            k_hi2 = min(p + n - 1, 2 * n - 2)
            k_start = min(k_lo1, k_lo2)
            k_end = max(k_hi1, k_hi2)
            um = 0.0
            if k_start > 0:
                um = float(np.max(c[:k_start]))
            if k_end < 2 * n - 2:
                um = max(um, float(np.max(c[k_end + 1:])))
            ks = np.arange(k_start, k_end + 1)
            s_arr = np.zeros(len(ks))
            for ki, k in enumerate(ks):
                if k_lo1 <= k <= k_hi1:
                    s_arr[ki] += 1.0 - h[p + k - n + 1]
                if k_lo2 <= k <= k_hi2:
                    s_arr[ki] -= h[p - k + n - 1]
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = current_max
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                new_c = c[ks] + d * s_arr
                ci = (ks == (n - 1))
                if np.any(ci):
                    new_c[ci] = c[n - 1] + d * (1.0 - 2.0 * h[p]) - d * d
                val = max(um, float(np.max(new_c)))
                if val < best_val - 1e-15:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-13:
                d = best_d
                c[ks] += d * s_arr
                c[n - 1] -= d * d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                current_max = max(um, float(np.max(c[k_start:k_end + 1])))
                improved = True
        if not improved:
            break
    return h, current_max * (2.0 / n)

def pair_refine(h, n, deadline, steps, radius=50):
    best_c = compute_c5(h, n)
    for d in steps:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-15:
                            best_c = cv
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-15:
                            best_c = cv
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)
