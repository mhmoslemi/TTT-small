def alternating_lp(h_init, n, max_iter=1000, deadline=None):

    best_c = compute_c5(h, n)
    g = 1.0 - h

        if deadline is not None and time.time() > deadline:
            break
        
        # Build constraint matrix A
        # A has (2n-1) rows, n columns
        # Row k_idx corresponds to lag k = k_idx - (n-1)
        A = np.zeros((2*n-1, n))
        for k_idx in range(2*n-1):
            k = k_idx - (n-1)
            if k >= 0:
                # A[k_idx, j] = g[j-k] for j in [k, n-1]
                for j in range(k, n):
                    A[k_idx, j] = g[j-k]
            else:
                m = -k
                # A[k_idx, j] = g[j+m] for j in [0, n-1-m]
                for j in range(0, n-m):
                    A[k_idx, j] = g[j+m]
        
        # Solve LP: min t s.t. A @ h <= t, sum(h) = n/2, 0 <= h <= 1
        # Variables: h[0], ..., h[n-1], t
        # c = [0, ..., 0, 1] (minimize t)
        from scipy.optimize import linprog
        c_obj = np.zeros(n+1)
        c_obj[-1] = 1.0
        
        # Inequality constraints: A @ h - t <= 0 → [A, -1] @ [h; t] <= 0
        A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
        b_ub = np.zeros(2*n-1)
        
        # Equality: sum(h) = n/2
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n/2.0])
        
        bounds = [(0, 1)] * n + [(0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.success:
            h_new = res.x[:n]
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                h = h_new
                best_c = c_new
                g = 1.0 - h
        
        if time.time() > (deadline or (1e18)) - 1:
            break
    
    return h, best_c

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= (target / tot)
    return np.clip(h, 0.0, 1.0)

def fast_cd(h, n, max_passes=5000, deadline=None, penalty=5.0, grid_coarse=32, grid_fine=16):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    sumh = float(h.sum())
    dx = 2.0 / n
    nm1 = n - 1
    
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            
            k_lo = min(p, nm1 - p)
            k_hi = min(2 * nm1, max(p + nm1, nm1 + p))
            
            um = 0.0
            if k_lo > 0:
                um = float(np.max(c[:k_lo]))
            if k_hi < 2 * nm1:
                um = max(um, float(np.max(c[k_hi + 1:])))
            
            ks = np.arange(k_lo, k_hi + 1)
            s = np.zeros(len(ks))
            
            j_vals = p + nm1 - ks
            valid1 = (j_vals >= 0) & (j_vals <= nm1)
            if np.any(valid1):
                s[valid1] += (1.0 - h[j_vals[valid1]])
            
            idx2 = ks - nm1 + p
            valid2 = (idx2 >= 0) & (idx2 <= nm1)
            if np.any(valid2):
                s[valid2] -= h[idx2[valid2]]
            
            self_idx = nm1 - k_lo if (k_lo <= nm1 <= k_hi) else -1
            
            # Coarse grid
            deltas = np.linspace(d_lo, d_hi, grid_coarse)
            best_d = 0.0
            best_val = current_max + penalty * (sumh - n/2.0)**2
            
            for d in deltas:
                if abs(d) < 1e-15:
                    val = max(um, float(np.max(c[ks])))
                    if penalty > 0:
                        val += penalty * (sumh - n/2.0)**2
                else:
                    new_c = c[ks] + d * s
                    if self_idx >= 0:
                        new_c = new_c.copy()
                        new_c[self_idx] -= d * d
                    val = max(um, float(np.max(new_c)))
                    if penalty > 0:
                        val += penalty * (sumh + d - n/2.0)**2
                if val < best_val - 1e-15:
                    best_val = val
                    best_d = d
            
            # Fine grid
            if abs(best_d) > 1e-12:
                margin = max((d_hi - d_lo) / grid_coarse, 1e-4)
                f_lo = max(d_lo, best_d - 2*margin)
                f_hi = min(d_hi, best_d + 2*margin)
                deltas_fine = np.linspace(f_lo, f_hi, grid_fine)
                for d in deltas_fine:
                    if abs(d) < 1e-15:
                        continue
                    new_c = c[ks] + d * s
                    if self_idx >= 0:
                        new_c = new_c.copy()
                        new_c[self_idx] -= d * d
                    val = max(um, float(np.max(new_c)))
                    if penalty > 0:
                        val += penalty * (sumh + d - n/2.0)**2
                    if val < best_val - 1e-15:
                        best_val = val
                        best_d = d
            
            if abs(best_d) > 1e-12:
                d = best_d
                c[ks] += d * s
                if self_idx >= 0:
                    c[nm1] -= d * d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                sumh += d
                current_max = max(um, float(np.max(c[ks])))
                improved = True
        
        if not improved:
            break
    
    return h, current_max * dx

def alternating_lp(h_init, n, max_iter=200, deadline=None):
    h = project_feasible(h_init.copy(), n)
    best_c = compute_c5(h, n)
    nm1 = n - 1
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        g = 1.0 - h
        
        # Build A matrix (2n-1) x n
        A = np.zeros((2*n-1, n))
        for k_idx in range(2*n-1):
            k = k_idx - nm1
            if k >= 0:
                A[k_idx, k:] = g[:n-k]
            else:
                m = -k
                A[k_idx, :n-m] = g[m:]
        
        # LP: min t s.t. A@h - t <= 0, sum(h) = n/2, 0<=h<=1
        c_obj = np.zeros(n+1)
        c_obj[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
        b_ub = np.zeros(2*n-1)
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n/2.0])
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.success:
            h_new = project_feasible(res.x[:n], n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                h = h_new
                best_c = c_new
            elif it > 5:
                # No improvement in several iterations, try continuing
                h = h_new
                # Update but don't update best
        else:
            break
    
    return h, best_c

def pair_search(h, n, deadline, deltas, radius=30):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= (target / tot)
    return np.clip(h, 0.0, 1.0)

def fast_cd(h, n, max_passes=5000, deadline=None, penalty=5.0):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    sumh = float(h.sum())
    dx = 2.0 / n
    nm1 = n - 1
    target = n / 2.0

    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            k_lo = min(p, nm1 - p)
            k_hi = min(2 * nm1, max(p + nm1, nm1 + p))
            um = 0.0
            if k_lo > 0:
                um = float(np.max(c[:k_lo]))
            if k_hi < 2 * nm1:
                um = max(um, float(np.max(c[k_hi + 1:])))
            ks = np.arange(k_lo, k_hi + 1)
            s = np.zeros(len(ks))
            j_vals = p + nm1 - ks
            valid1 = (j_vals >= 0) & (j_vals <= nm1)
            if np.any(valid1):
                s[valid1] += (1.0 - h[j_vals[valid1]])
            idx2 = ks - nm1 + p
            valid2 = (idx2 >= 0) & (idx2 <= nm1)
            if np.any(valid2):
                s[valid2] -= h[idx2[valid2]]
            self_idx = nm1 - k_lo if (k_lo <= nm1 <= k_hi) else -1

            # Coarse + fine grid
            deltas = np.concatenate([
                [d_lo, d_hi, (d_lo + d_hi) / 2, 0.0],
                np.linspace(d_lo, d_hi, 24)[1:-1]
            ])
            best_d = 0.0
            best_val = current_max + penalty * (sumh - target) ** 2
            for d in deltas:
                if abs(d) < 1e-15:
                    val = max(um, float(np.max(c[ks])))
                    if penalty > 0:
                        val += penalty * (sumh - target) ** 2
                else:
                    new_c = c[ks] + d * s
                    if self_idx >= 0:
                        new_c = new_c.copy()
                        new_c[self_idx] -= d * d
                    val = max(um, float(np.max(new_c)))
                    if penalty > 0:
                        val += penalty * (sumh + d - target) ** 2
                if val < best_val - 1e-15:
                    best_val = val
                    best_d = d
            # Fine refinement
            if abs(best_d) > 1e-12:
                margin = max((d_hi - d_lo) / 24.0, 1e-4)
                f_lo = max(d_lo, best_d - 2 * margin)
                f_hi = min(d_hi, best_d + 2 * margin)
                for d in np.linspace(f_lo, f_hi, 12):
                    if abs(d) < 1e-15:
                        continue
                    new_c = c[ks] + d * s
                    if self_idx >= 0:
                        new_c = new_c.copy()
                        new_c[self_idx] -= d * d
                    val = max(um, float(np.max(new_c)))
                    if penalty > 0:
                        val += penalty * (sumh + d - target) ** 2
                    if val < best_val - 1e-15:
                        best_val = val
                        best_d = d
            if abs(best_d) > 1e-12:
                d = best_d
                c[ks] += d * s
                if self_idx >= 0:
                    c[nm1] -= d * d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                sumh += d
                current_max = max(um, float(np.max(c[ks])))
                improved = True
        if not improved:
            break
    return h, current_max * dx

def alternating_lp(h_init, n, max_iter=100, deadline=None):
    h = project_feasible(h_init.copy(), n)
    best_c = compute_c5(h, n)
    nm1 = n - 1
    target = n / 2.0
    no_improve = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        g = 1.0 - h
        A = np.zeros((2 * n - 1, n))
        for k_idx in range(2 * n - 1):
            k = k_idx - nm1
            if k >= 0:
                A[k_idx, k:n] = g[0:n - k]
            else:
                m = -k
                A[k_idx, 0:n - m] = g[m:n]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            break
        h_new = project_feasible(res.x[:n], n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            h = h_new
            best_c = c_new
            no_improve = 0
        else:
            h = h_new
            no_improve += 1
            if no_improve > 15:
                break
    return h, best_c

def pair_search(h, n, deadline, deltas, radius=25):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-14:
                            best_c = cv; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-14:
                            best_c = cv; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c
