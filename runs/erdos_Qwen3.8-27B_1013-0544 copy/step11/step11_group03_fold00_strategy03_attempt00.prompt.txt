def grad_c_at_lag(h, s):

    else:

    return grad

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * dx

def slsqp_opt(h, n, alphas, maxiter=400):

    bounds = [(0.0, 1.0)] * n

            return smooth_obj(x, n, alpha)

            return x.sum() - n / 2.0

            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
