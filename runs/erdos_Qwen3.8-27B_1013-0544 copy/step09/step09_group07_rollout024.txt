import numpy as np
from scipy.optimize import minimize, golden, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [float(np.clip(lo, lo, hi)), float(np.clip(hi, lo, hi)), 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    best_d = 0.0
    best_val = obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        dc = float(np.clip(d, lo, hi))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15:
            best_val = v
            best_d = dc
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.12, abs(best_d) * 0.4 + span * 0.04)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:

                res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=80)
                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    return best_d

    h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:

            m_f_lo = max(0, n - 1 - idx)

            m_b_lo = max(0, idx)

def smooth_obj_logsumexp(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_obj_power(x, n, p):
    c = np.correlate(x, 1.0 - x, mode='full')
    return float(np.sum(c**p)) ** (1.0/p) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_logsumexp(h, n, alphas, maxiter_each=600, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj_logsumexp, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def slsqp_power(h, n, ps, maxiter_each=400, deadline=None):

import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [float(np.clip(lo, lo, hi)), float(np.clip(hi, lo, hi)), 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    best_d = 0.0
    best_val = obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:

    span = hi - lo

                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    return best_d

def coord_cd(h, n, max_passes=300, deadline=None, top_k=40):
    h = h.copy()
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break

        current_max = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:

            m_b_lo = max(0, idx)
            m_b_hi = min(2 * n - 2, idx + n - 1)
            m_lo = min(m_f_lo, m_b_lo)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def power_obj(x, n, p):
    c = np.correlate(x, 1.0 - x, mode='full')
    return float(np.sum(c ** p)) ** (1.0 / p) * (2.0 / n)

def slsqp_lse(h, n, alphas, maxiter_each=500, deadline=None):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-17})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def slsqp_power(h, n, ps, maxiter_each=400, deadline=None):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for p in ps:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(power_obj, best_h, args=(n, p), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-17})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:

        except Exception:
            pass

def perturb_top(h, n, rng, n_perturb=8, magnitude=0.03):
    """Perturb the coordinates with highest sensitivity to the max correlation."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m_star = int(np.argmax(c))
    dx = 2.0 / n
    # Compute sensitivity: for each i, how much does h[i] affect c[m_star]?
    sens = np.zeros(n)
    m = m_star - (n - 1)  # actual lag
    if m >= 0:
        for i in range(n - m):
            sens[i] += (1.0 - h[i]) * dx
            sens[i + m] -= h[i] * dx
    else:
        mm = -m
        for i in range(n - mm):
            sens[i + mm] += (1.0 - h[i]) * dx
            sens[i] -= h[i + mm] * dx
    # Also consider nearby top lags
    top_lags = np.argsort(c)[-3:]
    for ml in top_lags:
        if ml == m_star:
            continue
        m2 = ml - (n - 1)
        if m2 >= 0:
            for i in range(n - m2):
                sens[i] += (1.0 - h[i]) * dx
                sens[i + m2] -= h[i] * dx
        else:
            mm = -m2
            for i in range(n - mm):
                sens[i + mm] += (1.0 - h[i]) * dx
                sens[i] -= h[i + mm] * dx
    free = (h > 1e-4) & (h < 1.0 - 1e-4)
    free_idx = np.where(free)[0]
    if len(free_idx) == 0:
        return h
    # Pick top sensitivity free coordinates
    s_vals = np.abs(sens[free_idx])
    order = np.argsort(s_vals)[-n_perturb:]
    chosen = free_idx[order]
    h_new = h.copy()
    for i in chosen:
        h_new[i] = np.clip(h[i] + rng.choice([-1, 1]) * magnitude, 0.0, 1.0)
    return project_feasible(h_new, n)
