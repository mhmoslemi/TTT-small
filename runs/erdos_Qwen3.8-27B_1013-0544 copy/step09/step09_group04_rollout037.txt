```python
import numpy as np
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_lp_h(h_current, n, solver='GLPK'):
    """Solve LP to update h given q=1-h_current fixed."""
    q = 1.0 - h_current
    h_var = cp.Variable(n)
    L_var = cp.Variable()
    constraints = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
    # Build correlation constraints
    c = np.correlate(h_current, q, mode='full')  # just for reference
    for m in range(2 * n - 1):
        k = m - (n - 1)
        # c[m] = sum_n h[n+k] * q[n] for valid n
        # n ranges over [max(0,-k), min(n, n-k)]
        n_lo = max(0, -k)
        n_hi = min(n, n - k)
        if n_lo >= n_hi:
            continue
        # h[n+k] for n in [n_lo, n_hi]: h[n_lo+k : n_hi+k]
        # q[n] for n in [n_lo, n_hi]: q[n_lo : n_hi]
        constraint = cp.sum(h_var[n_lo+k:n_hi+k] * q[n_lo:n_hi]) <= L_var
        constraints.append(constraint)
    objective = cp.Minimize(L_var)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver)
        if h_var.value is not None:
            return np.array(h_var.value).flatten(), L_var.value
    except:
        pass
    return h_current.copy(), None

def solve_lp_q(h_current, n, solver='GLPK'):
    """Solve LP to update q given h_current fixed."""
    h = h_current.copy()
    q_var = cp.Variable(n)
    L_var = cp.Variable()
    constraints = [cp.sum(q_var) == 3.0 * n / 2.0, q_var >= 0, q_var <= 1]
    for m in range(2 * n - 1):
        k = m - (n - 1)
        n_lo = max(0, -k)
        n_hi = min(n, n - k)
        if n_lo >= n_hi:
            continue
        constraint = cp.sum(h[n_lo+k:n_hi+k] * q_var[n_lo:n_hi]) <= L_var
        constraints.append(constraint)
    objective = cp.Minimize(L_var)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver)
        if q_var.value is not None:
            return np.array(q_var.value).flatten(), L_var.value
    except:
        pass
    return 1.0 - h_current.copy(), None
```

Let me estimate: for n=700, we have 701 variables and ~1400 linear inequality constraints + 1 equality + 1400 bounds. With GLPK or GLOP, this should solve in well under 1 second.

But we need to do many iterations. If each LP takes 0.1s and we do 50 iterations, that's 5 seconds. Plus the exact CD. Total should fit in budget.

Actually, I realize there might be an issue with the LP formulation. The constraint c[m] ≤ L is:
sum of h[i] * q[j] ≤ L where (i,j) are the pairs that contribute to c[m].

For a given m, the pairs are: (n+k, n) for n in [n_lo, n_hi). So the constraint is:
sum_{n=n_lo}^{n_hi-1} h[n+k] * q[n] ≤ L

This is a single linear constraint with (n_hi - n_lo) nonzero coefficients in h.

The total number of nonzeros across all constraints is the sum of (n_hi - n_lo) over all m, which is n² (each h[i] appears in exactly n constraints). So the constraint matrix is n*(2n-1) in size with n² nonzeros. For n=700, that's about 490,000 nonzeros. This should be fine for an LP solver.

Let me also think about whether the alternating LP will actually converge. The function F(h,q) = max_m sum_n h[n+k]q[n] is the max of bilinear functions. The alternating minimization of a max of bilinear functions is known to converge to a local optimum (it's a special case of the "alternating direction method" for non-convex problems). It might cycle, but in practice it usually converges or at least makes progress.

To prevent cycling, I can:
1. Keep the best solution seen so far
2. If we detect cycling (same h repeated), add a small perturbation
3. Limit the number of iterations

Now, let me also think about combining the LP approach with the exact CD:
- Use the LP to make large steps (navigate the landscape)
- Use the exact CD to polish (find the precise local optimum)

This hybrid should be more effective than either alone.

```python
def coord_cd(h, n, max_passes, deadline):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline: break
        c = np.correlate(h, 1.0-h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline: break
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0-1e-13: continue
            x_new = solve_1d(c, h, p, n)
            if abs(x_new - h0) > 1e-12:
                h[p] = x_new
                improved = True
        if not improved: break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def alt_lp_cd(h, n, max_iters, deadline, solver='GLPK'):

    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    q = 1.0 - h
    for it in range(max_iters):
```

```python
def subgradient_step(h, n, k=10, lr=0.1, beta=0.9):
    """One subgradient step with momentum."""
    c = np.correlate(h, 1.0-h, mode='full')
    # Top-k indices
    top_idx = np.argsort(c)[-k:]
    # Compute subgradient
    grad = np.zeros(n)
    for m in top_idx:
        lag = m - (n-1)
        # dc[m]/dh[i] = (1-h[i-lag])*[0<=i-lag<n] - h[i+lag]*[0<=i+lag<n]
        i1 = np.arange(n) - lag
        v1 = (i1 >= 0) & (i1 < n)
        contrib1 = np.zeros(n)
        contrib1[v1] = 1.0 - h[i1[v1]]
        i2 = np.arange(n) + lag
        v2 = (i2 >= 0) & (i2 < n)
        contrib2 = np.zeros(n)
        contrib2[v2] = -h[i2[v2]]
        grad += contrib1 + contrib2
    return grad, c.max()
```

Wait, but the subgradient of max is not the sum of gradients of the top-k. It's the gradient of one of the maximizing functions (or a convex combination if we use a smooth approximation).

For the non-smooth max, the subdifferential is the convex hull of gradients of all functions achieving the max. For a practical method, we can:
1. Pick the argmax and use its gradient
2. Or use a random convex combination of the top-k gradients
3. Or use the gradient of the smooth max (log-sum-exp)

Option 3 is the most principled. The gradient of m + (1/alpha) * log(sum(exp(alpha*c))) is:
softmax(alpha*c) * alpha * grad(c[m]) averaged... 

Actually, let's think about it differently. Let f(h) = max_m c[m]. The subgradient at h is any element of:
∂f(h) = conv{∇_h c[m] : m ∈ argmax c[m]}

For a practical method, I'll use:
- The gradient of the smooth approximation: (1/alpha) * sum_m exp(alpha*c[m]) * ∇_h c[m] / sum_m exp(alpha*c[m])
  = sum_m softmax_m * ∇_h c[m]
  where softmax_m = exp(alpha*c[m]) / sum exp(alpha*c[m])

This is a weighted average of the gradients, weighted by a softmax. For large alpha, it concentrates on the max.

The computational cost: for each m in the support of the softmax (which is effectively the top few for large alpha), compute ∇_h c[m]. This is O(n) per m, so O(k*n) total where k is the effective support size.

```python
def smooth_subgradient(h, n, alpha=100.0, k=20):
    """Compute smooth subgradient of max correlation."""
    c = np.correlate(h, 1.0-h, mode='full')
    # Softmax weights (top-k only for efficiency)
    top_k_idx = np.argsort(c)[-k:]
    c_top = c[top_k_idx]
    weights = np.exp(alpha * (c_top - c_top.max()))
    weights /= weights.sum()
    # Compute gradient
    grad = np.zeros(n)
    for w, m in zip(weights, top_k_idx):
        if w < 1e-10: continue
        lag = m - (n-1)
        i1 = np.arange(n) - lag
        v1 = (i1 >= 0) & (i1 < n)
        grad[v1] += w * (1.0 - h[i1[v1]])
        i2 = np.arange(n) + lag
        v2 = (i2 >= 0) & (i2 < n)
        grad[v2] += w * (-h[i2[v2]])
    return grad, c
```

```python
def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except:
        h_init = np.full(500, 0.5)
        n_init = 500
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Initial CD to ensure local optimum
    h1, c1 = coord_cd(best_h, n_init, 5, t0 + 60)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Subgradient exploration
    t_sg_end = min(deadline - 120, t0 + 200)
    if t0 + 60 < t_sg_end:
        h2, c2 = subgrad_phase(best_h, n_init, 2000, t_sg_end, alpha=50, k=15)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h3, c3 = coord_cd(best_h, n_init, 10, min(deadline - 60, t_sg_end + 60))
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 3: Alternating LP
    t_lp_end = min(deadline - 80, time.time() + 200)
    if time.time() < t_lp_end - 10:
        h4, c4 = alternating_lp(best_h, n_init, 15, t_lp_end, solver='GLPK')
        if c4 < best_c:
            best_h, best_c = h4, c4
        h5, c5 = coord_cd(best_h, n_init, 10, min(deadline - 30, t_lp_end + 40))
        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 4: More subgradient with different params
    if time.time() < deadline - 60:
        t_sub2 = min(deadline - 40, time.time() + 150)
        # Perturb and explore
        n = len(best_h)
        for trial in range(4):
            if time.time() > t_sub2:
                break
            perturb = 0.05 + 0.05 * trial
            hp = project_feasible(best_h + rng.normal(0, perturb, n), n)
            h_exp, c_exp = subgrad_phase(hp, n, 500, min(t_sub2, time.time() + 60), alpha=30, k=10)
            h_pol, c_pol = coord_cd(h_exp, n, 10, min(t_sub2, time.time() + 30))
            if c_pol < best_c:
                best_h, best_c = h_pol, c_pol

    # Phase 5: Try different resolutions
    for n_try in [500, 600, 800, 1000]:
        if time.time() > deadline - 80:
            break
        if n_try == len(best_h):
            continue
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_cd(h_int, n_try, 5, min(deadline - 30, time.time() + 30))
        h_opt2, c_opt2 = subgrad_phase(h_opt, n_try, 300, min(deadline - 30, time.time() + 80), alpha=40, k=12)
        h_opt3, c_opt3 = coord_cd(h_opt2, n_try, 10, deadline - 10)
        if c_opt3 < best_c:
            best_h, best_c = h_opt3, c_opt3

    # Phase 6: Final polish
    n = len(best_h)
    if time.time() < deadline - 5:
        hf, cf = coord_cd(best_h, n, 20, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
