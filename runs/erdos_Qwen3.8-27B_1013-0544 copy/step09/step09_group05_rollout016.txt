    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

            x_orig = x_new
        
        # SLSQP on this n
        if time.time() < deadline - 80:
            h_s, c_s = slsqp_polish(best_h if len(best_h) == n_try else h_int, n_try, deadline)
            if c_s < best_c:
                best_h, best_c = h_s, c_s
            h_opt = polish(h_s, n_try, 100, deadline, rng)
            c_opt = compute_c5(h_opt, n_try)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
                x_orig = x_new

    # Phase 4: Basin hopping with diverse starts
    n_best = len(best_h)
    remaining = deadline - time.time()
    n_hops = max(2, int(remaining / 40))
    
    for hop in range(n_hops):
        if time.time() > deadline - 30:
            break
        # Diverse perturbation
        typ = hop % 6
        if typ < 3:
            scale = 0.02 + 0.03 * (hop / max(1, n_hops))
            hp = best_h + rng.normal(0, scale, n_best)
        else:
            hp = make_initial(n_best, seed + hop * 1000, typ - 3)
        hp = project_feasible(hp, n_best)
        h_hop = polish(hp, n_best, 80, deadline, rng)
        c_hop = compute_c5(h_hop, n_best)
        if c_hop < best_c:
            best_h, best_c = h_hop, c_hop

    # Phase 5: Final polish
    if time.time() < deadline - 5:
        max_p = max(100, int((deadline - time.time() - 2) / 0.15))
        best_h = polish(best_h, n_best, max_p, deadline, rng)
        best_c = compute_c5(best_h, n_best)

    # Phase 6: Try smaller n for potentially better discrete C5
    for n_sm in [200, 250, 350]:
        if time.time() > deadline - 20:
            break
        if n_sm == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_sm)
        h_sm = project_feasible(np.interp(x_new, np.linspace(0, 2, len(best_h)), best_h), n_sm)
        h_sm = polish(h_sm, n_sm, 100, deadline, rng)
        c_sm = compute_c5(h_sm, n_sm)
        if c_sm < best_c:
            best_h, best_c = h_sm, c_sm

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
