def project_feasible(h,n):
    h=np.asarray(h,dtype=float).copy()
    h=np.clip(h,0.0,1.0)
    target=0.5*n
    # bisection

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    lo, hi = -1.0, 1.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            idxs = np.argsort(1.0 - h)[::-1]
        else:
            idxs = np.argsort(h)
        for idx in idxs:
            room = (1.0 - h[idx]) if diff > 0 else h[idx]
            if room <= 1e-15:
                continue
            step = diff if abs(diff) <= room else (room if diff > 0 else -room)
            h[idx] += step
            diff -= step
            if abs(diff) <= 1e-13:
                break
    return h


def mlo_mhi(n):
    n1 = n - 1
    p = np.arange(n, dtype=int)
    mlo = np.minimum(p, n1 - p)
    mhi = np.maximum(n1 + p, 2 * n1 - p)
    return mlo.astype(int), mhi.astype(int)

def sum_con(x, n):
    return x.sum() - 0.5 * n


def sum_con_jac(x, n):
    return np.ones_like(x)


def smooth_value(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(c))
    y = alpha * (c - m)
    esum = float(np.sum(np.exp(y)))
    return (2.0 / n) * (m + float(np.log(esum)) / alpha)


def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(c))
    y = alpha * (c - m)
    e = np.exp(y)
    w = e / float(np.sum(e))
    return (2.0 / n) * grad_weights(w, x, n)


def grad_weights(w, h, n):
    g = np.zeros(n, dtype=float)
    n1 = n - 1
    for m, wt in enumerate(w):
        if wt <= 1e-14:
            continue
        r = n1 - m
        if r >= 0:
            g[:n - r] += wt * (1.0 - h[r:])
            g[r:] -= wt * h[:-r]
        else:
            q = -r
            g[q:] += wt * (1.0 - h[:n - q])
            g[:n - q] -= wt * h[q:]
    return g


def slsqp_polish(h, n, alphas, limit, maxiter=35):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > limit:
            break
        try:
            res = minimize(
                smooth_value, best_h, args=(n, alpha), method='SLSQP',
                jac=smooth_grad, bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': 0}
            )
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-14:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    return project_feasible(np.interp(xn, xo, h), n_new)

            lim = min(deadline - 15, time.time() + (55 if nt <= 700 else 45))
            hr, cr = cd_optimize(hi, nt, passes, lim, 0.10, rng)
            if cr < best_c:
                best_h, best_c, n = hr, cr, nt

    if deadline - time.time() > 170:
        for size in [0.02, 0.05, 0.10, 0.20, 0.35]:
            if time.time() > deadline - 55:
                break
            hp = project_feasible(best_h + rng.normal(0.0, size, n), n)
            lim = min(deadline - 15, time.time() + 35)
            hx, cx = cd_optimize(hp, n, 90, lim, 0.06, rng)
            if cx < best_c:
                best_h, best_c = hx, cx

    if deadline - time.time() > 100 and n <= 900:
        mi = 25 if n <= 700 else 18
        hf, cf = slsqp_polish(best_h, n, [8000, 30000], min(deadline - 20, time.time() + 70), mi)
        if cf < best_c:
            best_h, best_c = hf, cf

    if deadline - time.time() > 55:
        hf, cf = cd_optimize(best_h, n, 350, min(deadline - 5, time.time() + 45), 0.018, rng)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, float(best_c), int(n)
