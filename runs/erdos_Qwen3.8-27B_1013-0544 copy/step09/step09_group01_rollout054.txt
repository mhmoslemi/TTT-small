def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def correlation_gradient(h, m, n):
    """Compute gradient of c[m] with respect to h."""
    grad = np.zeros(n)
    # c[m] = sum_i h[i] * (1 - h[i+m-n+1])
    # d(c[m])/d(h[p]) = (1-h[p+m-n+1]) if valid - h[p-m+n-1] if valid
    for p in range(n):
        j1 = p + m - (n - 1)  # first factor index for second term
        if 0 <= j1 < n:
            grad[p] += (1.0 - h[j1])
        j2 = p - m + (n - 1)  # second factor index for first term
        if 0 <= j2 < n:
            grad[p] -= h[j2]
    return grad

    j1 = np.arange(n) + m - (n - 1)
    valid1 = (j1 >= 0) & (j1 < n)

    j2 = np.arange(n) - m + (n - 1)
    valid2 = (j2 >= 0) & (j2 < n)

def cutting_plane_solve(h, n, max_iters=100, tol=1e-8):
    """Solve min max_k c_k(h) s.t. 0<=h<=1, sum(h)=n/2 using cutting plane."""
    h = h.copy()
    c = compute_correlations(h)
    M = c.max()
    
    # Build LP: min t s.t. c_k(h_0) + grad_k · (h - h_0) <= t, 0<=h<=1, sum(h)=n/2
    # Variables: h[0],...,h[n-1], t (total n+1 variables)
    # Objective: minimize t → c_obj = [0,...,0, 1]
    
    for it in range(max_iters):
        c = compute_correlations(h)
        M = c.max()
        active = c >= M - 0.01 * M  # active set
        
        # Build A_ub, b_ub for the active constraints
        # Each active constraint: grad_k · h <= t + (grad_k · h_0 - c_k(h_0))
        # Rearranged: grad_k · h - t <= grad_k · h_0 - c_k(h_0)
        
        A_rows = []
        b_vals = []
        for m in np.where(active)[0]:
            g = correlation_gradient(h, m, n)
            row = np.concatenate([g, [-1.0]])
            rhs = g @ h - c[m]
            A_rows.append(row)
            b_vals.append(rhs)
        
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
        
        # Box constraints: 0 <= h[i] <= 1 → already handled by bounds
        # Sum constraint: sum(h) = n/2
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0  # minimize t
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0  # sum(h) = n/2
        b_eq = np.array([n / 2.0])
        
        bounds = [(0, 1)] * n + [(M - 0.1, None)]  # t >= M - 0.1
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if not res.success:
            break
        
        h_new = res.x[:n]
        t_new = res.x[-1]
        
        if abs(t_new - M) < tol:
            break
        
        h = h_new
    
    return h, compute_c5(h, n)

    return result


def coord_descent(h, n, max_passes, deadline, gs_iters=40):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    h1, c1 = lp_method(best_h, n, min(t0 + 200, deadline))
    if c1 < best_c:
        best_h, best_c = h1, c1

    alphas = [500, 2000, 10000, 50000, 200000]
    h1, c1 = slsqp_optimize(best_h, n, alphas, 400)
    if c1 < best_c:
        best_h, best_c = h1, c1

    h1, c1 = lp_method(best_h, n, min(t0 + 350, deadline))
    if c1 < best_c:
        best_h, best_c = h1, c1

    h1, c1 = coord_descent(best_h, n, 40, min(t0 + 700, deadline), gs_iters=35)
    if c1 < best_c:
        best_h, best_c = h1, c1

    multi_alphas = [2000, 20000, 200000, 1000000]
    h1, c1 = slsqp_optimize(best_h, n, multi_alphas, 300)
    if c1 < best_c:
        best_h, best_c = h1, c1

    if time.time() < deadline - 20:
        h1, c1 = coord_descent(best_h, n, 80, deadline, gs_iters=55)
        if c1 < best_c:
            best_h, best_c = h1, c1

    if time.time() < deadline - 5:
        h1, c1 = slsqp_optimize(best_h, n, [5000000], 600)
        if c1 < best_c:
            best_h, best_c = h1, c1

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
