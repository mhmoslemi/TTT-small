    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(700, 0.5)
        n = 700

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP warmup
    h1, c1 = slsqp_refine(best_h, n, deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Main sequential CD
    h2, c2 = coord_cd(best_h, n, 300, deadline, rng, 'sequential')
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Top-room ordering CD
    if time.time() < deadline - 20:
        h3, c3 = coord_cd(best_h, n, 150, deadline, rng, 'top')
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Random ordering CD
    if time.time() < deadline - 20:
        h4, c4 = coord_cd(best_h, n, 100, deadline, rng, 'random')
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Try different n values
    for dn in [50, -50, 100, -100, 150]:
        if time.time() > deadline - 80:
            break
        nt = len(best_h) + dn
        if nt < 300:
            continue
        x_o = np.linspace(0, 2, len(best_h))
        x_n = np.linspace(0, 2, nt)
        hi = project_feasible(np.interp(x_n, x_o, best_h), nt)
        ho, co = coord_cd(hi, nt, 80, deadline, rng, 'sequential')
        if co < best_c:
            best_h, best_c = ho, co

    # Phase 6: Perturbation restarts
    if time.time() < deadline - 40:
        n_cur = len(best_h)
        for sigma in [0.003, 0.008, 0.015]:
            if time.time() > deadline - 25:
                break
            hp = project_feasible(best_h + rng.normal(0, sigma, n_cur), n_cur)
            ho, co = coord_cd(hp, n_cur, 80, deadline, rng, 'random')
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 7: Final polish with best ordering
    if time.time() < deadline - 3:
        n_cur = len(best_h)
        hf, cf = coord_cd(best_h, n_cur, 200, deadline, rng, 'sequential')
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
