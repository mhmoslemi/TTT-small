import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [float(np.clip(lo, lo, hi)), float(np.clip(hi, lo, hi)), 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    best_d = 0.0
    best_val = obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        dc = float(np.clip(d, lo, hi))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15:
            best_val = v
            best_d = dc
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.12, abs(best_d) * 0.4 + span * 0.04)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=80)
                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    return best_d

    h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            m_f_lo = max(0, n - 1 - idx)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - idx)
            m_b_lo = max(0, idx)
            m_b_hi = min(2 * n - 2, idx + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_idx = np.argsort(vals)[-top_k:]
                top_idx = np.sort(top_idx)
                ms = ms_all[top_idx]
                ms_set = set(ms.tolist())
                skipped = [m for m in ms_all if m not in ms_set]
                unaff_extra = float(c[skipped].max()) if len(skipped) > 0 else 0.0
            else:
                ms = ms_all
                unaff_extra = 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff_max = unaff_extra
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n - 1):
                    diag_local = ii
                    break
            lo = -h0
            hi_ = 1.0 - h0
            best_d = solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi_)
            if abs(best_d) > 1e-12:

                if new_val < current_max - 1e-14:

                    improved = True
                    current_max = min(current_max, new_val)
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

- Effect of j flip: +(1-h[j+m-(n-1)]) - h[j-m+(n-1)] (for valid indices)

def solve_1d_vectorized(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(c_aff)
    if k == 0:
        return 0.0
    
    # Generate all pairwise intersection candidates
    i_idx, j_idx = np.triu_indices(k, 1)
    da = a_coeff[i_idx] - a_coeff[j_idx]
    dc = c_aff[j_idx] - c_aff[i_idx]
    dd = np.zeros(len(da))
    dd[i_idx == diag_local] -= 1.0
    dd[j_idx == diag_local] += 1.0
    
    # For dd == 0: linear, d = dc/da
    # For dd != 0: quadratic, d = (-da ± sqrt(da² + 4*dd*dc)) / (2*dd)
    
    mask_lin = np.abs(dd) < 1e-12
    mask_quad = ~mask_lin
    
    candidates = []
    
    if np.any(mask_lin):
        valid = mask_lin & (np.abs(da) > 1e-12)
        if np.any(valid):
            d_lin = dc[valid] / da[valid]
            d_lin = d_lin[(d_lin > lo) & (d_lin < hi)]
            candidates.append(d_lin)
    
    if np.any(mask_quad):
        disc = da[mask_quad]**2 + 4.0 * dd[mask_quad] * dc[mask_quad]
        valid = disc >= 0
        if np.any(valid):
            sq = np.sqrt(disc[valid])
            d1 = (-da[mask_quad][valid] + sq) / (2.0 * dd[mask_quad][valid])
            d2 = (-da[mask_quad][valid] - sq) / (2.0 * dd[mask_quad][valid])
            d1 = d1[(d1 > lo) & (d1 < hi)]
            d2 = d2[(d2 > lo) & (d2 < hi)]
            candidates.append(d1)
            candidates.append(d2)
    
    if candidates:
        all_candidates = np.concatenate(candidates)
        all_candidates = np.unique(all_candidates)
    else:
        all_candidates = np.array([])
    
    # Add boundary candidates
    all_candidates = np.concatenate([all_candidates, [lo, hi, 0.0]])
    all_candidates = all_candidates[(all_candidates >= lo) & (all_candidates <= hi)]
    
    # Evaluate objective at all candidates
    if len(all_candidates) == 0:
        all_candidates = np.array([(lo + hi) / 2.0])
    
    # c_aff + d * a_coeff for each candidate d
    # Shape: (n_candidates, k)
    vals = c_aff[None, :] + all_candidates[:, None] * a_coeff[None, :]
    if diag_local is not None:
        vals[:, diag_local] -= all_candidates**2
    
    obj_vals = np.maximum(vals.max(axis=1), unaff_max)
    
    best_idx = np.argmin(obj_vals)
    return float(all_candidates[best_idx])

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

    for p in range(max_passes):

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-15:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    best_h = project_feasible(best_h, n_fin)
    best_c = compute_c5(best_h, n_fin)

    # Try 1-h flip
    h_flip = project_feasible(1.0 - best_h, n_fin)
    c_flip = compute_c5(h_flip, n_fin)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
