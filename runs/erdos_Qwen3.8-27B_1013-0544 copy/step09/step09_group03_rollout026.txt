def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)

def subgradient_step(h, n, lr, momentum=0.9):
    """One step of subgradient descent on max correlation."""
    c = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(c)
    s = max_idx - (n - 1)  # shift
    
    # Gradient of c[n-1+s] w.r.t. h:
    # c[n-1+s] = sum_j h[j]*(1-h[j-s])
    # d/dh[p] = (1-h[p-s]) if p-s in range, minus h[p+s] if p+s in range
    g = np.zeros(n)
    for p in range(n):
        gp = 0.0

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x.copy(), n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def block_cd(h, n, block_size=3, max_passes=200, deadline=None, seed=0):
    """Coordinate descent optimizing blocks of coordinates."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        # Select random blocks
        perm = rng.permutation(n)
        improved = 0
        for start in range(0, n - block_size + 1, block_size):
            idx = perm[start:start+block_size]
            # For small blocks, do a local grid search
            if block_size == 1:
                p = idx[0]
                if 1e-10 < h[p] < 1.0 - 1e-10:
                    new_val = exact_coord_min(c, p, h, n)
                    if abs(new_val - h[p]) > 1e-10:
                        h[p] = new_val
                        improved += 1
            elif block_size == 2:
                p, q = idx[0], idx[1]
                if 1e-10 < h[p] < 1.0-1e-10 or 1e-10 < h[q] < 1.0-1e-10:
                    xp, xq = pair_coord_min(c, p, q, h, n)

                        h[p], h[q] = xp, xq
                        improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:

    return h, best_c

def structured_perturb(h, n, rng):
    """Apply structured perturbations to escape local optima."""
    h_new = h.copy()
    method = rng.randint(4)
    if method == 0:
        # Flip a contiguous block
        bw = rng.randint(5, max(6, n//20))
        start = rng.randint(0, n - bw)
        h_new[start:start+bw] = 1.0 - h_new[start:start+bw]
    elif method == 1:
        # Add a Fourier mode perturbation
        k = rng.randint(1, n//4 + 1)
        amp = 0.1 * rng.random()
        x = np.arange(n)
        h_new += amp * np.sin(2 * np.pi * k * x / n)
    elif method == 2:
        # Move a transition boundary
        trans = np.where(np.abs(np.diff(h)) > 0.3)[0]
        if len(trans) > 0:
            t = trans[rng.randint(len(trans))]
            shift = rng.randint(-5, 6)
            h_new[t:t+3] = np.roll(h_new[t:t+3], shift)
    else:
        # Gaussian noise
        h_new += 0.05 * rng.randn(n)
    return project_feasible(h_new, n)

    mx = shifted.max()

    # Phase 2: L-BFGS-B with penalty (fast gradient-based)
    if time.time() < deadline - 250:
        for mu in [5000.0, 50000.0]:
            if time.time() > deadline - 220:
                break
            for alpha in [5000.0, 50000.0]:
                h2, c2 = lbfgs_penalized(best_h, n, alpha, mu, 300, deadline - 210)
                h2, c2 = coord_descent(h2, n, 300, deadline - 200, seed + 50)
                if c2 < best_c:
                    best_h, best_c = h2, c2

    # Phase 3: SLSQP with exact constraint
    if time.time() < deadline - 150:
        h3, c3 = slsqp_smooth(best_h, n, [20000.0, 200000.0], 400, deadline - 130)
        if c3 < best_c:
            best_h, best_c = h3, c3
        h3, c3 = coord_descent(best_h, n, 500, deadline - 110, seed + 100)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Structured restarts
    n_cur = len(best_h)
    for r in range(10):
        if time.time() > deadline - 80:
            break
        hp = structured_perturb(best_h, n_cur, rng)
        h_o, c_o = coord_descent(hp, n_cur, 500, deadline - 60, seed + 200 + r * 11)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            n_cur = len(best_h)

    # Phase 5: Multi-resolution
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 450, 500, 700, 800, 1000, 1200]:
        if time.time() > deadline - 45:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = np.interp(x_new, x_orig, best_h)
        h_int = project_feasible(h_int, n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, 600, deadline - 35, seed + n_try)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new

    # Phase 6: Final intensive CD
    n_final = len(best_h)
    if time.time() < deadline - 4:
        hf, cf = coord_descent(best_h, n_final, 8000, deadline - 2, seed + 9999)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Flip symmetry check
    h_flip = project_feasible(1.0 - best_h, n_final)
    c_flip = compute_c5(h_flip, n_final)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
