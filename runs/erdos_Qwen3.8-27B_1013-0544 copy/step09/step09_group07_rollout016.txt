    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

        return 0.0

            dc = c_aff[j] - c_aff[i]
            dd = 0.0

                dd -= 1.0

                        if lo < d < hi:

        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15:
            best_val = v
            best_d = dc
    span = hi - lo

                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    return best_d

    for p in range(max_passes):

            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())

                break

            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue

            m_f_hi = min(2 * n - 2, 2 * n - 2 - idx)
            m_b_lo = max(0, idx)
            m_b_hi = min(2 * n - 2, idx + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_idx = np.argsort(vals)[-top_k:]
                top_idx = np.sort(top_idx)
                ms = ms_all[top_idx]
                ms_set = set(ms.tolist())
                skipped = [m for m in ms_all if m not in ms_set]
                unaff_extra = float(c[skipped].max()) if len(skipped) > 0 else 0.0
            else:
                ms = ms_all
                unaff_extra = 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff_max = unaff_extra
            if m_lo > 0:

            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n - 1):
                    diag_local = ii
                    break
            lo = -h0
            hi = 1.0 - h0
            best_d = solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi)
            if abs(best_d) > 1e-12:

                    improved = True
                    current_max = min(current_max, new_val)
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [float(np.clip(lo, lo, hi)), float(np.clip(hi, lo, hi)), 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i: dd -= 1.0
            if diag_local == j: dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    best_d = 0.0
    best_val = obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        dc2 = float(np.clip(d, lo, hi))
        v = obj_1d(dc2, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15:
            best_val = v
            best_d = dc2
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.12, abs(best_d) * 0.4 + span * 0.04)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=80)
                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
            except Exception:
                pass
    return best_d

def coord_cd(h, n, max_passes=500, deadline=None, top_k=40):
    h = h.copy()
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            m_lo = min(max(0, n - 1 - idx), max(0, idx))
            m_hi = max(min(2*n-2, 2*n-2-idx), min(2*n-2, idx+n-1))
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_idx = np.sort(np.argsort(vals)[-top_k:])
                ms = ms_all[top_idx]
                ms_set = set(ms.tolist())
                skipped = np.array([m for m in ms_all if m not in ms_set])
                unaff_extra = float(c[skipped].max()) if len(skipped) > 0 else 0.0
            else:
                ms = ms_all
                unaff_extra = 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf): a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb): a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff_max = unaff_extra
            if m_lo > 0: unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n-2: unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n-1):
                    diag_local = ii
                    break
            best_d = solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, -h0, 1.0-h0)
            if abs(best_d) > 1e-12:
                new_val = obj_1d(best_d, c_aff, a_coeff, diag_local, unaff_max)
                if new_val < current_max - 1e-14:
                    h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                    improved = True
                    current_max = min(current_max, new_val)
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def softmax_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    p = np.exp(s)
    p /= p.sum()
    grad = np.zeros(n)
    for k in range(2*n - 1):
        mk = k - (n - 1)
        pk = p[k]
        if pk < 1e-12: continue
        lo = max(0, -mk)
        hi = min(n, n - mk)
        if lo < hi: grad[lo:hi] += pk * (1.0 - h[lo+mk:hi+mk])
        lo2 = max(0, mk)
        hi2 = min(n, n + mk)
        if lo2 < hi2: grad[lo2:hi2] -= pk * h[lo2-mk:hi2-mk]
    return alpha * grad, c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=600, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    alpha = 50.0
    lr = 0.015
    stall = 0
    for it in range(max_iter):

            break

        if grad_norm < 1e-14:
            break
        h_new = project_feasible(h - lr * grad / grad_norm * np.linalg.norm(h - 0.5), n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            h = h_new
            best_c = c_new
            stall = 0
            lr = min(lr * 1.08, 0.06)
        else:
            lr *= 0.7
            stall += 1
            if stall > 80 or lr < 1e-12:
                break
    return h, best_c

S = m + log(Σ_k exp(α(c_k - m)))

dS/dh[i] = α * Σ_k p_k * dc_k/dh[i]

where p_k = exp(α(c_k - m)) / Σ_j exp(α(c_j - m))

And dc_k/dh[i] = (1-h)[i+m_k] * 1[valid] - h[i-m_k] * 1[valid]

where m_k = k - (n-1).

So:
dS/dh[i] = α * Σ_k p_k * [(1-h)[i+m_k] * 1[0≤i+m_k<n] - h[i-m_k] * 1[0≤i-m_k<n]]
