import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()

            if tot < 1e-15:

def multi_cd(h, n, max_passes, deadline, top_k, grow_active=True):
    """Multi-peak coordinate descent with growing active set"""
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float).copy()
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            # Compute coefficients for each active peak
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            # Peak values as function of d: c_peak[idx] + d * a_coeff[idx] [+ d² if diag]
            # Find d that minimizes max of these
            lo = -h0
            hi = 1.0 - h0
            # Bisection + candidate evaluation
            best_d = 0.0
            best_val = current_max
            # Try bisection for about 40 iterations
            for _ in range(40):
                mid = (lo + hi) / 2
                vals = c_peak + mid * a_coeff
                if diag_idx is not None:
                    vals[diag_idx] -= mid * mid
                v = float(vals.max())

                    best_d = mid

                    hi = mid

                    lo = mid

def solve_1d_minimax(c_peak, a_coeff, diag_idx, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            # c[diag] - d^2 = c[i] + a*i*d
            # c[diag] - c[i] = a[i]*d + d^2
            # d^2 + a[i]*d + (c[i] - c[diag]) = 0
            B = a_coeff[i]
            C = c_peak[i] - c_peak[diag_idx]
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
    best_d = lo
    best_val = float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        vals = c_peak + d * a_coeff
        if diag_idx is not None:
            vals[diag_idx] -= d * d
        v = float(vals.max())
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    c5_best = compute_c5(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float).copy()
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            lo = -h0
            hi = 1.0 - h0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, lo, hi)
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c5_new = compute_c5(h, n)
        if c5_new < c5_best:
            c5_best = c5_new
    return h, c5_best

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c5_new = compute_c5(h, n)
        if c5_new < c5_best:
            c5_best = c5_new
    return h, c5_best

def zero_one_search(n, deadline, seed_val, max_time):
    """Greedy 0-1 local search"""
    rng = np.random.RandomState(seed_val)
    best_h = None
    best_c = float('inf')
    
    # Try multiple initial patterns
    patterns = []
    # Pattern 1: first half
    h = np.zeros(n); h[:n//2] = 1.0; patterns.append(h.copy())
    # Pattern 2: alternating pairs
    h = np.zeros(n)
    for i in range(n):
        if (i // 2) % 2 == 0:
            h[i] = 1.0
    patterns.append(h.copy())
    # Pattern 3: random
    h = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    patterns.append(h.copy())
    # Pattern 4: periodic 3
    h = np.zeros(n)
    for i in range(n):
        if i % 3 != 2:
            h[i] = 1.0
    # Fix count
    while h.sum() > n/2:
        idx = np.where(h == 1)[0]
        h[idx[rng.randint(len(idx))]] = 0
    while h.sum() < n/2:
        idx = np.where(h == 0)[0]
        h[idx[rng.randint(len(idx))]] = 1
    patterns.append(h.copy())
    
    for pat in patterns:
        if time.time() > deadline:
            break
        h = pat.copy()
        # Ensure exactly n/2 ones
        while h.sum() > n/2:
            idx = np.where(h == 1)[0]
            h[idx[rng.randint(len(idx))]] = 0
        while h.sum() < n/2:
            idx = np.where(h == 0)[0]
            h[idx[rng.randint(len(idx))]] = 1
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # Local search
        iterations = 0
        while time.time() < deadline and iterations < 5000:
            iterations += 1
            improved = False
            # Get current correlation
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_val = corr.max()
            max_idx = np.argmax(corr)
            
            # Try swaps that might reduce the max peak
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            # Try up to 200 random swaps
            tried = set()
            for _ in range(200):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                key = (min(i,j), max(i,j))
                if key in tried:
                    continue
                tried.add(key)
                
                # Compute change in correlation
                # Swap: h[i] 1->0, h[j] 0->1
                h[i] = 0; h[j] = 1
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                else:
                    h[i] = 1; h[j] = 0
            
            if not improved:
                # Try more targeted: focus on positions near the max peak
                m_max = int(max_idx)
                for _ in range(50):
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    # Prefer positions near the peak
                    h[i] = 0; h[j] = 1
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                        break
                    else:
                        h[i] = 1; h[j] = 0
                
                if not improved:
                    break
        
        if time.time() > deadline:
            break
    
    return best_h, best_c
