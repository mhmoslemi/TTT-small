import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def fast_corr_full(a, b):
    n, m = len(a), len(b)
    L = 1
    while L < n + m - 1:
        L *= 2
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    r = np.fft.irfft(fa * np.conj(fb), L)
    return np.concatenate([r[n-1:], r[:n-1][::-1]])

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _cd_evaluate(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0) + d * d
    r = float(np.max(vals))
    if unaff_max > r:
        r = unaff_max
    return r

def _cd_slope(c_aff, a, diag_idx, has_diag, h0, d):
    """Compute the slope (derivative) of f(d) = max_k(c_aff[k] + d*a[k] + quad_term)."""
    idx = int(np.argmax(c_aff + d * a))
    if has_diag and idx == diag_idx:
        return a[idx] + 2.0 * d
    return a[idx]

def coord_descent(h, n, max_passes, deadline, use_fft=False, tol=1e-12):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(np.max(c))
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        max_drop = 0.0
        for p in range(n):
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            a[valid_f] += (1.0 - h[jf[valid_f]])
            a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            if has_diag:
                # Quadratic: find vertex if in range
                a_diag = a[diag_idx]
                if abs(a_diag) > 1e-15:
                    dv = -a_diag / 2.0
                else:
                    dv = 0.0
                if d_lo < dv < d_hi:
                    f_v = _cd_evaluate(c_aff, a, diag_idx, True, unaff_max, h0, dv)
                    f_cv = _cd_evaluate(c_aff, a, diag_idx, True, unaff_max, h0, 0.0)
                    if f_v < f_cv - 1e-14:
                        h[p] = h0 + dv
                        c = np.correlate(h, 1.0 - h, mode='full')
                        if float(np.max(c)) < best_c:
                            best_c = float(np.max(c))
                        improved = True
                        max_drop = max(max_drop, f_cv - f_v)
                        continue
            # Binary search on slope for piecewise-linear part
            s_lo = _cd_slope(c_aff, a, diag_idx, has_diag, h0, d_lo)
            s_hi = _cd_slope(c_aff, a, diag_idx, has_diag, h0, d_hi)
            if s_lo >= -1e-14:
                d_opt = d_lo
            elif s_hi <= 1e-14:
                d_opt = d_hi
            else:
                for _ in range(40):
                    if d_hi - d_lo < tol:
                        break
                    d_mid = 0.5 * (d_lo + d_hi)
                    s_mid = _cd_slope(c_aff, a, diag_idx, has_diag, h0, d_mid)
                    if s_mid < 0:
                        d_lo = d_mid
                    else:
                        d_hi = d_mid
                d_opt = 0.5 * (d_lo + d_hi)
            f_opt = _cd_evaluate(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_evaluate(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                c = np.correlate(h, 1.0 - h, mode='full')
                if float(np.max(c)) < best_c:
                    best_c = float(np.max(c))
                improved = True
                max_drop = max(max_drop, f_curr - f_opt)
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        if float(np.max(c)) < best_c:
            best_c = float(np.max(c))
    return h, compute_c5(h, n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for alpha in alphas:
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            shifted = alpha * (c - m)
            mx = shifted.max()
            lse = mx + np.log(np.sum(np.exp(shifted - mx)))
            return (m + lse / alpha) * dx
        def jac(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            shifted = alpha * (c - m)
            mx = shifted.max()
            weights = np.exp(shifted - mx)
            weights /= weights.sum()
            grad = np.zeros(n)
            for k in range(len(c)):
                w = weights[k] * dx
                if w < 1e-15:
                    continue
                i_start = max(0, k - (n - 1))
                i_end = min(n, k + 1)
                for i in range(i_start, i_end):
                    j_f = i + k - (n - 1)
                    j_b = i - k + (n - 1)
                    if 0 <= j_f < n:
                        grad[i] += w * (1.0 - x[j_f])
                    if 0 <= j_b < n:
                        grad[i] -= w * x[j_b]
            return grad
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, best_h, jac=jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
