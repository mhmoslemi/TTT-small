def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if time.time() > deadline:
            return h, compute_c5(h, n)
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def try_symmetry(h, n):
    h_flip = project_feasible(1.0 - h, n)
    c_flip = compute_c5(h_flip, n)
    c_orig = compute_c5(h, n)
    if c_flip < c_orig:

    return h, c_orig

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_minimax(c_vec, d_vec, lo, hi):
    K = len(c_vec)
    candidates = [lo, hi]
    for i in range(K):
        for j in range(i + 1, K):
            da = d_vec[i] - d_vec[j]
            if abs(da) > 1e-15:
                b = (c_vec[j] - c_vec[i]) / da
                if lo <= b <= hi:
                    candidates.append(b)
    best_b = lo
    best_val = float('inf')
    for b in candidates:
        b = float(np.clip(b, lo, hi))
        vals = c_vec + d_vec * b
        v = float(vals.max())
        if v < best_val:
            best_val = v
            best_b = b
    return best_b, best_val

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                L = m - (n - 1)
                af = (1.0 - h[p + L]) if 0 <= p + L < n else 0.0
                ab = (-h[p - L]) if 0 <= p - L < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo, hi = -h0, 1.0 - h0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, lo, hi)
            if diag_idx is not None:
                v2 = max(best_val, unaff_max)
                for bd in [lo, hi]:
                    c_ext = c_peak + a_coeff * bd
                    c_ext[diag_idx] -= bd * bd
                    v = max(float(c_ext.max()), unaff_max)
                    if v < v2:
                        v2 = v
                        best_val = v
                        best_d = bd
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def block_cd_2d(h, n, max_passes, deadline, top_k, rng):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[top_idx] = False
        unaff_max = float(c[mask].max()) if mask.any() else 0.0
        improved = False
        n_pairs = 60
        for _ in range(n_pairs):
            if time.time() > deadline:
                break
            p = rng.randint(n)
            q = rng.randint(n)
            if p == q:
                continue
            h0p, h0q = h[p], h[q]
            if h0p < 1e-10 or h0p > 1 - 1e-10 or h0q < 1e-10 or h0q > 1 - 1e-10:
                continue
            Am = np.zeros(k)
            Bm = np.zeros(k)
            Pm = np.zeros(k)
            Qm = np.zeros(k)
            Jm = np.zeros(k)
            for idx in range(k):
                m = int(top_idx[idx])
                L = m - (n - 1)
                if 0 <= p + L < n:
                    Am[idx] += (1.0 - h[p + L])
                if 0 <= p - L < n:
                    Am[idx] += (-h[p - L])
                if L == 0:
                    Pm[idx] = -1.0
                if 0 <= q + L < n:
                    Bm[idx] += (1.0 - h[q + L])
                if 0 <= q - L < n:
                    Bm[idx] += (-h[q - L])
                if L == 0:
                    Qm[idx] = -1.0
                if p + L == q:
                    Jm[idx] += -1.0
                if q + L == p:
                    Jm[idx] += -1.0
            lo_a, hi_a = -h0p, 1.0 - h0p
            lo_b, hi_b = -h0q, 1.0 - h0q
            best_val = float('inf')
            best_a, best_b = 0.0, 0.0
            na = 12
            nb = 18
            a_vals = np.linspace(lo_a, hi_a, na)
            b_vals = np.linspace(lo_b, hi_b, nb)
            for a in a_vals:
                c_base = c_peak + a * Am + a * a * Pm
                d_b = Bm + a * Jm
                for b in b_vals:
                    vals = c_base + b * d_b + b * b * Qm
                    v = max(float(vals.max()), unaff_max)
                    if v < best_val:
                        best_val = v
                        best_a, best_b = a, b
            if best_val < current_max - 1e-13:
                h[p] = float(np.clip(h0p + best_a, 0.0, 1.0))
                h[q] = float(np.clip(h0q + best_b, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def try_symmetry(h, n):
    h_flip = project_feasible(1.0 - h, n)
    c_flip = compute_c5(h_flip, n)
    c_orig = compute_c5(h, n)
    if c_flip < c_orig:
        return h_flip, c_flip
    return h, c_orig

def subgrad_polish(h, n, deadline, top_k=8):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 0.008
    for it in range(3000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        grad = np.zeros(n)
        for idx in range(k):
            m = int(top_idx[idx])
            L = m - (n - 1)
            pv = np.arange(n)
            fwd = pv + L
            bwd = pv - L
            g1 = np.where((fwd >= 0) & (fwd < n), 1.0 - h[np.clip(fwd, 0, n-1)], 0.0)
            g2 = np.where((bwd >= 0) & (bwd < n), -h[np.clip(bwd, 0, n-1)], 0.0)
            grad += g1 + g2
        grad /= k
        h = project_feasible(h - alpha * grad, n)
        cc = compute_c5(h, n)
        if cc < best_c:
            best_h = h.copy()
            best_c = cc
        alpha = max(0.0005, alpha * 0.997)
    return best_h, best_c
