            res = minimize(_smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-20})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def make_sinusoidal(n, freq):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.3 * np.sin(np.pi * freq * x)
    return project_feasible(h, n)

def make_comb(n, period):
    h = np.zeros(n)
    duty = max(1, period // 3)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(600, 0.5)
        n = 600

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Coordinate descent
    d1 = min(deadline, t0 + budget_s * 0.28)
    h1, c1 = coord_cd(best_h, n, max_passes=300, deadline=d1, top_k=40)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: SLSQP with increasing alphas
    d2 = min(deadline, t0 + budget_s * 0.55)
    alphas_mid = [500, 2000, 10000, 50000]
    h2, c2 = slsqp_polish(best_h, n, alphas_mid, 350, d2)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: More coordinate descent
    d3 = min(deadline, t0 + budget_s * 0.72)
    h3, c3 = coord_cd(best_h, n, max_passes=400, deadline=d3, top_k=50)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: High-precision SLSQP
    d4 = min(deadline, t0 + budget_s * 0.85)
    alphas_hi = [100000, 300000, 1000000]
    h4, c4 = slsqp_polish(best_h, n, alphas_hi, 500, d4)
    if c4 < best_c:
        best_h, best_c = h4, c4

    # Phase 5: Final coordinate descent
    h5, c5v = coord_cd(best_h, n, max_passes=500, deadline=deadline, top_k=50)
    if c5v < best_c:
        best_h, best_c = h5, c5v

    # Phase 6 (if time): Try different resolution
    n_cur = len(best_h)
    if time.time() < deadline - 40:
        x_orig = np.linspace(0, 2, n_cur)
        for n_try in [400, 500, 800]:
            if time.time() > deadline - 30:
                break
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
            hn, cn = coord_cd(h_int, n_try, max_passes=150, deadline=deadline - 25, top_k=40)
            if cn < best_c:
                best_h, best_c = hn, cn
                n_cur = n_try
                x_orig = x_new

    # Phase 7: Multi-restart
    if time.time() < deadline - 50:
        for r in range(6):
            if time.time() > deadline - 40:
                break
            s = 0.002 + 0.005 * r
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            ho, co = coord_cd(hp, n_cur, max_passes=150, deadline=deadline - 35, top_k=45)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 8: Try structured initializations
    if time.time() < deadline - 60:
        for freq in [2.0, 3.0, 5.0]:
            if time.time() > deadline - 50:
                break
            hs = make_sinusoidal(n_cur, freq)
            ho, co = coord_cd(hs, n_cur, max_passes=120, deadline=deadline - 45, top_k=40)
            if co < best_c:
                best_h, best_c = ho, co
        for period in [10, 15, 20]:
            if time.time() > deadline - 40:
                break
            hc = make_comb(n_cur, period)
            ho, co = coord_cd(hc, n_cur, max_passes=100, deadline=deadline - 38, top_k=40)
            if co < best_c:
                best_h, best_c = ho, co

    # Final: try 1-h flip
    n_fin = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_fin)
    c_flip = compute_c5(h_flip, n_fin)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    # Final SLSQP polish if time remains
    if time.time() < deadline - 15:
        h_fin, c_fin = slsqp_polish(best_h, n_fin, [500000, 2000000], 300, deadline - 5)
        if c_fin < best_c:
            best_h, best_c = h_fin, c_fin

    best_h = project_feasible(best_h, n_fin)
    best_c = compute_c5(best_h, n_fin)
    return best_h, best_c, n_fin
