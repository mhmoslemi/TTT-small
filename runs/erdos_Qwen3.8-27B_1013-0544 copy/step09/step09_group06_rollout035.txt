import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    dx = 2.0 / n
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo = -h0
            hi = 1.0 - h0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if best_val < current_max - 1e-15:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d = lo
    best_val = float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        v = eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

def eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max):
    vals = c_peak + d * a_coeff
    if diag_idx is not None:
        vals[diag_idx] -= d * d
    return max(float(vals.max()), unaff_max)

Specifically:
1. Find the top m correlations
2. Find a direction in h-space that reduces all of them simultaneously
3. Take a step in that direction

The direction can be found by solving:
minimize Σ_{k in top_m} C[k]² (or max C[k])
subject to: ||Δh||² ≤ ε, 0 ≤ h + Δh ≤ 1

This is a trust-region subproblem. For small ε, the solution is approximately the negative gradient of the active constraints.

    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[::-1][:top_m]
    # Gradient of c[k] w.r.t. h[i] for k in top_idx
    grad = np.zeros(n)
    for idx in top_idx:
        m_shift = int(idx)
        shift = m_shift - (n - 1)
        if shift >= 0:
            for i in range(n - shift):
                grad[i + shift] += (1 - h[i])
                grad[i] -= h[i + shift]
        else:

                grad[i] -= h[i - s]
                grad[i - s] += (1 - h[i])

    # Take step

    h_new = project_feasible(h_new, n)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max):
    vals = c_peak + d * a_coeff
    if diag_idx is not None:
        vals[diag_idx] -= d * d
    return max(float(vals.max()), unaff_max)

def solve_1d(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d, best_val = lo, float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        v = eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

def cd_pass(h, n, top_k):
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(c.max())
    k = min(top_k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    c_peak = c[top_idx].astype(float)
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[top_idx] = False
    unaff_max = float(c[mask].max()) if mask.any() else 0.0
    improved = False
    for p in range(n):
        h0 = h[p]
        if h0 < 1e-10 or h0 > 1.0 - 1e-10:
            continue
        a_coeff = np.zeros(k)
        diag_idx = None
        for idx in range(k):
            m = int(top_idx[idx])
            jf = p + m - (n - 1)
            jb = p - m + (n - 1)
            af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
            ab = (-h[jb]) if 0 <= jb < n else 0.0
            a_coeff[idx] = af + ab
            if m == n - 1:
                diag_idx = idx
        lo, hi = -h0, 1.0 - h0
        best_d, best_val = solve_1d(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
        if best_val < current_max - 1e-15:
            h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
            improved = True
            current_max = best_val
    return improved

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        if not cd_pass(h, n, top_k):
            break

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = alpha * (c - m)
    mx = np.max(s)
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def fourier_smooth(h, n, K):
    x = np.linspace(0, 2, n)
    h_centered = h - 0.5
    h_new = np.full(n, 0.5)
    for k in range(1, K + 1):
        a_k = (2.0 / n) * np.sum(h_centered * np.cos(np.pi * k * x))
        b_k = (2.0 / n) * np.sum(h_centered * np.sin(np.pi * k * x))
        h_new += a_k * np.cos(np.pi * k * x) + b_k * np.sin(np.pi * k * x)
    return project_feasible(h_new, n)

    for step in range(n_steps):

            break

        grad = np.zeros(n)
        for idx in top_idx:
            m_shift = int(idx)
            shift = m_shift - (n - 1)
            if shift >= 0:
                for i in range(n - shift):
                    grad[i + shift] += (1.0 - h[i])
                    grad[i] -= h[i + shift]
            else:
                s = -shift
                for i in range(s, n):
                    grad[i] -= h[i - s]
                    grad[i - s] += (1.0 - h[i])
        lr = lr_base / (1.0 + 0.1 * step)
        h_new = project_feasible(h - lr * grad / top_m, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            h, best_c = h_new, c_new
        else:
            lr_base *= 0.7
    return h, best_c
