def compute_subgradient(h, n, top_k=20):
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    # Get top-k shifts
    sorted_idx = np.argsort(c)[::-1][:top_k]
    # Weight by how close to max
    max_c = c.max()
    min_c_top = c[sorted_idx].min()
    weights = (c[sorted_idx] - min_c_top) / (max_c - min_c_top + 1e-15)
    if max_c == min_c_top:
        weights = np.ones(top_k) / top_k
    
    grad = np.zeros(n)
    for w, m_idx in zip(weights, sorted_idx):
        s = m_idx - (n - 1)
        if s == 0:
            # c = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
            # d/dh[j] = 1 - 2*h[j]

        else:

def lp_equalize(h, n, top_k=40):
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c_val = c_raw * dx
    sorted_idx = np.argsort(c_val)[::-1][:top_k]
    active_m = sorted_idx
    grads = compute_grads(h, n, active_m)
    intercept = c_val[active_m] - np.dot(grads, h)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.hstack([grads, -np.ones((top_k, 1))])
    b_ub = -intercept
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return project_feasible(res.x[:n], n)
    except Exception:
        pass
    return h

def lp_cd_cycle(h, n, deadline, top_k=40, seed=0):
    h = h.copy()
    best_c = compute_c5(h, n)
    it = 0
    while time.time() < deadline:
        h_lp = lp_equalize(h, n, top_k)
        c_lp = compute_c5(h_lp, n)
        if c_lp < best_c:
            h, best_c = h_lp, c_lp
        cd_deadline = min(deadline - 1, time.time() + 8)
        h_cd, c_cd = coord_descent(h, n, 100, cd_deadline, seed + it)
        if c_cd < best_c:
            h, best_c = h_cd, c_cd
        if time.time() > deadline - 3:
            break
        it += 1
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Initial CD
    if time.time() < deadline - 500:
        h1, c1 = coord_descent(best_h, n, 800, deadline - 480, seed)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: LP equalization + CD at multiple top_k
    for tk in [15, 30, 50, 80, 120]:
        if time.time() > deadline - 250:
            break
        h2, c2 = lp_cd_cycle(best_h, n, time.time() + 80, top_k=tk, seed=seed + tk * 10)
        if c2 < best_c:
            best_h, best_c = h2, c2
        if time.time() < deadline - 240:
            h3, c3 = coord_descent(best_h, n, 300, deadline - 230, seed + 500 + tk)
            if c3 < best_c:
                best_h, best_c = h3, c3

    # Phase 3: Multi-resolution with LP
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 500, 700, 800, 1000]:
        if time.time() > deadline - 150:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, 400, deadline - 140, seed + n_try)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new
        if c_opt < best_c + 5e-4 and time.time() < deadline - 130:
            h_lp, c_lp = lp_cd_cycle(h_opt, n_try, time.time() + 40, top_k=50, seed=seed + 1000 + n_try)
            if c_lp < best_c:
                best_h, best_c = h_lp, c_lp
                x_orig = np.linspace(0, 2, n_try)

    # Phase 4: Structured perturbations (Fourier + binary push)
    n_cur = len(best_h)
    x = np.linspace(0, 2, n_cur)
    fourier_configs = [(3, 0.0), (5, 1.0), (7, np.pi / 2), (4, 0.5), (6, 1.5)]
    for fi, (freq, ph) in enumerate(fourier_configs):
        if time.time() > deadline - 80:
            break
        h_four = best_h + 0.03 * np.sin(np.pi * freq * x + ph)
        h_four = project_feasible(h_four, n_cur)
        h_o, c_o = coord_descent(h_four, n_cur, 300, deadline - 70, seed + 2000 + fi)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
        if time.time() < deadline - 75:
            h_lp, c_lp = lp_cd_cycle(h_o, n_cur, time.time() + 10, top_k=40, seed=seed + 2100 + fi)
            if c_lp < best_c:
                best_h, best_c = h_lp, c_lp

    # Binary push
    if time.time() < deadline - 60:
        thresh = np.where(best_h > 0.5, 1.0, 0.0)
        hb = project_feasible(thresh, n_cur)
        h_o, c_o = coord_descent(hb, n_cur, 300, deadline - 50, seed + 2500)
        if c_o < best_c:
            best_h, best_c = h_o, c_o

    # Phase 5: Final LP + CD polish
    n_cur = len(best_h)
    if time.time() < deadline - 30:
        h_f, c_f = lp_cd_cycle(best_h, n_cur, deadline - 15, top_k=60, seed=seed + 3000)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Final CD
    if time.time() < deadline - 3:
        h_fin, c_fin = coord_descent(best_h, len(best_h), 1000, deadline - 1, seed + 9999)
        if c_fin < best_c:
            best_h, best_c = h_fin, c_fin

    # Check 1-h symmetry
    n_final = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_final)
    c_flip = compute_c5(h_flip, n_final)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
