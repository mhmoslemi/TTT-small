def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def eval_obj(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d_fast(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [lo, hi, 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i: dd -= 1.0
            if diag_local == j: dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi: candidates.append(d)
            else:
                disc = da*da + 4.0*dd*dc

                    sq = np.sqrt(disc)
                    for d in [(-da+sq)/(2*dd), (-da-sq)/(2*dd)]:
                        if lo < d < hi: candidates.append(d)
    best_d, best_val = 0.0, eval_obj(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        dc = np.clip(d, lo, hi)
        v = eval_obj(float(dc), c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15:
            best_val, best_d = v, float(dc)
    # Golden section

def slsqp_smooth(h, n, alphas, deadline=None):
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full')
        m = c.max()
        sh = alphas[0] * (c - m)
        return (m + sh.max() + np.log(np.sum(np.exp(sh-sh.max()))) / alphas[0]) * (2.0/n)
    def con_eq(x):
        return x.sum() - n/2.0
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if deadline and time.time() > deadline: break
        def obj_a(x):
            c = np.correlate(x, 1.0-x, mode='full')
            m = c.max()
            sh = alpha * (c - m)
            return (m + sh.max() + np.log(np.sum(np.exp(sh-sh.max()))) / alpha) * (2.0/n)
        try:
            res = minimize(obj_a, best_h, method='SLSQP',
                          bounds=[(0,1)]*n,
                          constraints=[{'type':'eq','fun':con_eq}],
                          options={'maxiter':200,'ftol':1e-18})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c: best_h, best_c = nh, nc
        except: pass
    return best_h, best_c

                continue
            m_f_lo = max(0, n - 1 - idx)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - idx)
            m_b_lo = max(0, idx)
            m_b_hi = min(2 * n - 2, idx + n - 1)
            m_lo, m_hi = min(m_f_lo, m_b_lo), max(m_f_hi, m_b_hi)
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_idx = np.sort(np.argsort(vals)[-top_k:])
                ms = ms_all[top_idx]
                ms_set = set(ms.tolist())
                skipped = [m for m in ms_all if m not in ms_set]
                unaff_extra = float(c[skipped].max()) if skipped else 0.0
            else:
                ms, unaff_extra = ms_all, 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf, vb = (jf >= 0) & (jf < n), (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms].copy()
            unaff = unaff_extra
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == n - 1:
                    diag_local = ii
                    break
            best_d = solve_1d_fast(c_aff, a_coeff, diag_local, unaff, -h0, 1.0 - h0)
            if abs(best_d) > 1e-12:
                new_val = eval_obj(best_d, c_aff, a_coeff, diag_local, unaff)
                if new_val < current_max - 1e-14:
                    h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                    improved = True
                    current_max = min(current_max, new_val)
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def binary_push(h, n, deadline=None):
    """Push intermediate values toward 0 or 1, keep if C5 improves."""
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(3):
        if deadline and time.time() > deadline:
            break
        changed = False
        for idx in range(n):
            if deadline and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 0.1 or h0 > 0.9:
                continue
            for target in [0.0, 1.0]:
                h[idx] = target
                h = project_feasible(h, n)
                c = compute_c5(h, n)
                if c < best_c - 1e-10:
                    best_c = c
                    changed = True
                    break
            else:
                h[idx] = h0
                h = project_feasible(h, n)
        if not changed:
            break
    return h, best_c

def freq_sweep(h, n, deadline=None, max_freq=40):
    """Try sinusoidal perturbations at various frequencies."""
    h = h.copy()
    best_c = compute_c5(h, n)
    x = np.arange(n, dtype=float)
    for f in range(1, max_freq + 1):
        if deadline and time.time() > deadline:
            break
        for amp in [0.02, 0.05, 0.1, 0.15]:
            for phase in [0.0, np.pi / 2]:
                pert = amp * np.sin(2.0 * np.pi * f * x / n + phase)
                candidate = project_feasible(h + pert, n)
                c = compute_c5(candidate, n)
                if c < best_c - 1e-9:
                    h, best_c = candidate, c
    return h, best_c

def slsqp_smooth(h, n, alphas, deadline=None, maxiter=300):
    def con_eq(x):
        return x.sum() - n / 2.0
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if deadline and time.time() > deadline:
            break
        def obj_a(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            sh = alpha * (c - m)
            return (m + sh.max() + np.log(np.sum(np.exp(sh - sh.max()))) / alpha) * (2.0 / n)
        try:
            res = minimize(obj_a, best_h, method='SLSQP',
                           bounds=[(0, 1)] * n,
                           constraints=[{'type': 'eq', 'fun': con_eq}],
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def downsample(h, n_from, n_to):
    x1 = np.linspace(0, 2, n_from)
    x2 = np.linspace(0, 2, n_to)
    h2 = np.interp(x2, x1, h)
    return project_feasible(h2, n_to)

def upsample(h, n_from, n_to):
    x1 = np.linspace(0, 2, n_from)
    x2 = np.linspace(0, 2, n_to)
    h2 = np.interp(x2, x1, h)
    return project_feasible(h2, n_to)

def make_rand_binary(n, seed=0):
    rng = np.random.RandomState(seed)
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return project_feasible(h, n)

def make_m_seq_like(n, seed=1):
    """Create a pseudo-random binary with better autocorrelation structure."""
    rng = np.random.RandomState(seed)
    h = (rng.rand(n) > 0.5).astype(float)
    h[:n // 2] = 1.0
    h[n // 2:] = 0.0
    rng.shuffle(h)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_full = len(h_init)
    except Exception:
        h_init = np.full(600, 0.5)
        n_full = 600

    best_h = project_feasible(h_init, n_full)
    best_c = compute_c5(best_h, n_full)

    # Phase 1: Multi-resolution from coarse to fine
    resolutions = [100, 150, 200, 300, 400, n_full]
    cur_n = n_full
    cur_h = best_h.copy()

    for rn in resolutions:
        if time.time() > deadline - 60:
            break
        if rn < cur_n:
            cur_h = downsample(cur_h, cur_n, rn)
        elif rn > cur_n:
            cur_h = upsample(cur_h, cur_n, rn)
        cur_n = rn

        # Coordinate descent
        tl = min(deadline - 30, time.time() + 120)
        h_cd, c_cd = coord_descent(cur_h, rn, max_passes=150, deadline=tl, top_k=20)
        if c_cd < best_c:
            best_h, best_c = upsample(h_cd, rn, n_full), compute_c5(upsample(h_cd, rn, n_full), n_full)

        # Binary push
        if time.time() < deadline - 50:
            h_bp, c_bp = binary_push(h_cd, rn, deadline=deadline - 40)
            if c_bp < best_c:
                best_h = upsample(h_bp, rn, n_full)
                best_c = compute_c5(best_h, n_full)

        cur_h = h_cd if c_cd <= c_bp else h_bp
        cur_n = rn

    # Phase 2: Full resolution optimization
    best_h = project_feasible(best_h, n_full)
    best_c = compute_c5(best_h, n_full)

    if time.time() < deadline - 200:
        h2, c2 = coord_descent(best_h, n_full, max_passes=300, deadline=deadline - 180, top_k=30)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Frequency sweep at full resolution
    if time.time() < deadline - 150:
        h3, c3 = freq_sweep(best_h, n_full, deadline=deadline - 130, max_freq=30)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: SLSQP with smooth max
    if time.time() < deadline - 100:
        h4, c4 = slsqp_smooth(best_h, n_full, [200, 800, 3000], deadline=deadline - 80, maxiter=200)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Binary push at full resolution
    if time.time() < deadline - 80:
        h5, c5 = binary_push(best_h, n_full, deadline=deadline - 60)
        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 6: More coordinate descent
    if time.time() < deadline - 60:
        h6, c6 = coord_descent(best_h, n_full, max_passes=400, deadline=deadline - 40, top_k=35)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Phase 7: Try different n values
    for n_try in [500, 700, 800, 900]:
        if time.time() > deadline - 40:
            break
        h_t = upsample(best_h, n_full, n_try)
        h_t = coord_descent(h_t, n_try, max_passes=100, deadline=deadline - 30, top_k=25)[0]
        h_t = project_feasible(h_t, n_try)
        c_t = compute_c5(h_t, n_try)
        if c_t < best_c:
            best_h, best_c, n_full = h_t, c_t, n_try

    # Phase 8: Multi-restart from best with noise
    n_cur = len(best_h)
    if time.time() < deadline - 30:
        for r in range(8):
            if time.time() > deadline - 20:
                break
            s = 0.005 + 0.01 * r
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            ho, co = coord_descent(hp, n_cur, max_passes=120, deadline=deadline - 15, top_k=30)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 9: Final SLSQP
    if time.time() < deadline - 15:
        h9, c9 = slsqp_smooth(best_h, n_cur, [5000, 20000], deadline=deadline - 8, maxiter=300)
        if c9 < best_c:
            best_h, best_c = h9, c9

    # Phase 10: Final coordinate descent
    if time.time() < deadline - 5:
        h10, c10 = coord_descent(best_h, n_cur, max_passes=500, deadline=deadline - 2, top_k=40)
        if c10 < best_c:
            best_h, best_c = h10, c10

    # Try 1-h symmetry
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, n_cur)
    best_c = compute_c5(best_h, n_cur)
    return best_h, best_c, n_cur
