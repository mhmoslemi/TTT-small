    nx = 2 * n - 1
    for idx in range(n):
        h0 = h[idx]
        if h0 < 1e-10 or h0 > 1.0 - 1e-10:
            continue
        lo_m = max(0, n - 1 - idx)
        hi_m = min(nx - 1, n - 1 + idx)
        ms = np.arange(lo_m, hi_m + 1)
        if len(ms) > top_k:
            order = np.argsort(c[ms])
            keep = np.sort(order[-top_k:])
            ms = ms[keep]
            unaff = float(c[order[:-top_k]].max())
        else:
            unaff = 0.0
        if lo_m > 0:
            unaff = max(unaff, float(c[:lo_m].max()))
        if hi_m < nx - 1:
            unaff = max(unaff, float(c[hi_m + 1:].max()))
        a = np.zeros(len(ms))
        jf = idx + ms - (n - 1)
        vf = (jf >= 0) & (jf < n)
        a[vf] += (1.0 - h[jf[vf]])
        jb = idx - ms + (n - 1)
        vb = (jb >= 0) & (jb < n)
        a[vb] -= h[jb[vb]]
        cv = c[ms]
        di = np.where(ms == (n - 1))[0]
        has_d = len(di) > 0
        lo, hi = -h0, 1.0 - h0
        if hi - lo < 1e-14:
            continue
        def f(d):
            v = cv + d * a
            if has_d:
                v[di[0]] -= d * d
            return max(float(v.max()), unaff)
        # Candidate points
        bds = [lo, hi]
        k2 = min(len(ms), 25)
        for i in range(k2):
            for j in range(i + 1, k2):
                da = a[i] - a[j]
                if abs(da) > 1e-13:
                    d = (cv[j] - cv[i]) / da
                    if lo < d < hi:
                        bds.append(d)
        bv = f(0.0)
        bd = 0.0
        for d in bds:
            d = float(np.clip(d, lo, hi))
            v = f(d)
            if v < bv:
                bv, bd = v, d
        # Golden section
        margin = min((hi - lo) * 0.12, abs(bd) * 0.3 + (hi - lo) * 0.03)
        gl, gh = max(lo, bd - margin), min(hi, bd + margin)
        if gh - gl > 1e-14:
            gr = (5.0 ** 0.5 - 1.0) / 2.0
            c1, c2 = gh - gr * (gh - gl), gl + gr * (gh - gl)
            f1, f2 = f(c1), f(c2)
            for _ in range(35):
                if f1 < f2:
                    gh, c2, f2 = c2, c1, f1
                    c1 = gh - gr * (gh - gl)
                    f1 = f(c1)
                else:
                    gl, c1, f1 = c1, c2, f2
                    c2 = gl + gr * (gh - gl)
                    f2 = f(c2)
            d_mid = (gl + gh) * 0.5
            v_mid = f(d_mid)
            if v_mid < bv:
                bv, bd = v_mid, d_mid
        if bv < f(0.0) - 1e-14:
            h[idx] = float(np.clip(h0 + bd, 0.0, 1.0))
    return h

def coord_descent(h, n, max_rounds=500, deadline=None, top_k=50):
    h = h.copy()
    for r in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        old_c = compute_c5(h, n)
        h = cd_round(h, n, top_k)
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c >= old_c - 1e-15:
            break
    return h, compute_c5(h, n)

def targeted_swaps(h, n, max_iters=50000, deadline=None):
    """Try swaps targeting the worst shift."""
    h = h.copy()
    best_c = compute_c5(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        am = int(np.argmax(c))
        t = am - (n - 1)
        # Find positions where h[i]=1 and h[i+t]=0 (or vice versa)
        if t > 0:
            ii = np.arange(min(n - t))
            bad = (h[ii] > 0.3) & (h[ii + t] < 0.7)
        elif t < 0:
            tt = -t
            ii = np.arange(n - tt)
            bad = (h[ii + tt] > 0.3) & (h[ii] < 0.7)
        else:
            bad = np.zeros(n, dtype=bool)
        if not np.any(bad):
            break
        candidates = ii[bad]
        # Pick a random candidate
        idx = candidates[np.random.randint(len(candidates))]
        if t > 0:
            pair = (idx, idx + t)
        elif t < 0:
            pair = (idx, idx - t)
        else:
            break
        i, j = pair[0], pair[1]
        if abs(h[i] - h[j]) < 1e-8:
            continue
        # Try swap
        h[i], h[j] = h[j], h[i]
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
            c = np.correlate(h, 1.0 - h, mode='full')
        else:
            h[i], h[j] = h[j], h[i]
            h = project_feasible(h, n)
    return h, best_c

def slsqp_polish(h, n, alphas=None, deadline=None):
    if alphas is None:
        alphas = [200, 1000, 5000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        # Smooth max approximation
        alpha = alphas[0]  # Will be overridden
        return _smooth_c5(x, n, alpha)
    
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        def _obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = float(c.max())
            shifted = alpha * (c - m)
            mx = shifted.max()
            lse = mx + np.log(np.sum(np.exp(shifted - mx)))
            return (m + lse / alpha) * (2.0 / n)
        
        def _sum_con(x):
            return x.sum() - n / 2.0
        
        def _sum_jac(x):
            return np.ones(len(x))
        
        try:
            res = minimize(_obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac}],
                          options={'maxiter': 300, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def make_construction(n, seed=0):
    """Create various initial constructions."""
    rng = np.random.RandomState(seed)
    # Option 1: random binary
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return project_feasible(h, n)
