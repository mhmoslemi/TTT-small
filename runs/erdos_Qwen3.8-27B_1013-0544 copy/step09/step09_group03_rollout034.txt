def binary_local_search(h, n, max_iters=10000, deadline=None, seed=0):

    rng = np.random.RandomState(seed)
    # First binarize

    # Fix sum constraint
    target = n // 2
    ones = np.where(h_bin == 1.0)[0]

            h_bin[i] = 0.0

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_and_grad(h, n, alpha, mu):
    """Return (objective, gradient) for smooth approximation with penalty."""
    c = np.correlate(h, 1.0 - h, mode='full')
    c_shift = c - c.max()
    exp_c = np.exp(alpha * c_shift)
    p = exp_c / exp_c.sum()
    # Gradient of smooth max
    grad_c = np.correlate(p, 1.0 - 2.0 * h, mode='full')[n-1 : 2*n-1]
    # Full gradient with penalty
    sum_h = h.sum()
    obj = (2.0/n) * (c.max() + np.log(exp_c.sum()) / alpha) + mu * (sum_h - n/2.0)**2
    grad = (2.0/n) * grad_c + 2.0 * mu * (sum_h - n/2.0)
    return obj, grad

def lbfgs_smooth(h, n, alpha, mu, maxiter=500):
    """L-BFGS-B with smooth objective."""
    def obj_and_grad(x):
        return smooth_c5_and_grad(x, n, alpha, mu)
    
    res = minimize(obj_and_grad, h, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    return res.x

def exact_coord_min(c, p, h, n):
    """Exact 1D minimization for coordinate p given correlation c."""
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(50):

        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:

        active = vals >= (am - 1e-13)
        if not active.any():

        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:

        else:

        v = max(v, M_unaff, c0_base + x - x * x)

            best_x = float(x)

def coord_descent(h, n, max_passes=500, deadline=None, seed=0):

    return h, best_c

def binary_sa(h, n, max_iters=20000, deadline=None, seed=0):
    """Simulated annealing on binary swaps."""
    rng = np.random.RandomState(seed)
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    target = n // 2
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]

        h_bin[idx] = 0.0
    elif len(ones) < target:
        idx = rng.choice(zeros, target - len(ones), replace=False)
        h_bin[idx] = 1.0

    c = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    best_m = c.max()
    T_init = best_m * 0.1

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c_new = np.correlate(h_bin, 1.0 - h_bin, mode='full')

                pass

            h_bin[i] = 1.0

            h_bin[j] = 1.0
            c = c_new
            if m_new < best_m:
                best_m = m_new
        # else: reject, h_bin unchanged

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)

def smooth_obj_grad(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode='full')
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    p = exp_c / exp_c.sum()

    return obj, grad

            if h[p] < 1e-10 or h[p] > 1.0 - 1e-10:
                continue
            nv = exact_coord_min(c, p, h, n)
            if abs(nv - h[p]) > 1e-10:
                h[p] = nv
                imp += 1
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < best_c:
            best_c = nc
        elif imp == 0:
            break
    return h, best_c

def binary_hill_climb(h, n, max_iters, deadline, seed):
    rng = np.random.RandomState(seed)
    hb = np.where(h > 0.5, 1.0, 0.0)
    targ = n // 2
    ones = np.where(hb == 1.0)[0]
    zeros = np.where(hb == 0.0)[0]
    if len(ones) > targ:
        hb[rng.choice(ones, len(ones)-targ, replace=False)] = 0.0
    elif len(ones) < targ:
        hb[rng.choice(zeros, targ-len(ones), replace=False)] = 1.0
    dx = 2.0 / n
    c = np.correlate(hb, 1.0 - hb, mode='full').astype(float)
    best_m = float(c.max())
    stall = 0
    for _ in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        if i == j:
            continue
        d = np.arange(-(n-1), n)
        ipd = i + d; ipd_v = (ipd >= 0) & (ipd < n)
        ipm = i - d; ipm_v = (ipm >= 0) & (ipm < n)
        jpd = j + d; jpd_v = (jpd >= 0) & (jpd < n)
        jpm = j - d; jpm_v = (jpm >= 0) & (jpm < n)
        delta = np.zeros(2*n-1)
        if ipd_v.any(): delta[ipd_v] += hb[ipd[ipd_v]]
        if ipm_v.any(): delta[ipm_v] += hb[ipm[ipm_v]]
        if jpd_v.any(): delta[jpd_v] -= hb[jpd[jpd_v]]
        if jpm_v.any(): delta[jpm_v] -= hb[jpm[jpm_v]]
        c_new = c + delta
        m_new = float(c_new.max())
        if m_new < best_m:
            hb[i] = 0.0; hb[j] = 1.0
            c = c_new; best_m = m_new
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break
    return hb, best_m * dx

def random_binary_hc(n, max_iters, deadline, seed):
    rng = np.random.RandomState(seed)
    hb = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    hb[idx] = 1.0
    return binary_hill_climb(hb, n, max_iters, deadline, seed)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_init = 400
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: L-BFGS-B smooth optimization at initial resolution
    if time.time() < deadline - 300:
        h_cur = best_h.copy()
        for alpha in [50, 200, 1000, 5000, 20000]:
            if time.time() > deadline - 300:
                break
            try:
                h_new = lbfgs_optimize(h_cur, n_init, alpha, 5.0, 300)
                h_new = project_feasible(h_new, n_init)
                c_new = compute_c5(h_new, n_init)
                if c_new < best_c:
                    best_h, best_c = h_new, c_new
                h_cur = h_new
            except Exception:
                pass
    # Phase 2: Binary hill climb + CD polish
    if time.time() < deadline - 280:
        hb, cb = binary_hill_climb(best_h, n_init, 80000, deadline - 270, seed)
        if cb < best_c:
            h_pol = project_feasible(hb, n_init)
            h_pol, cp = coord_descent(h_pol, n_init, 400, deadline - 265, seed)
            if cp < best_c:
                best_h, best_c = h_pol, cp
        else:
            h_cd, cd = coord_descent(best_h, n_init, 500, deadline - 265, seed)
            if cd < best_c:
                best_h, best_c = h_cd, cd
    # Phase 3: Multi-n exploration
    n_trials = [200, 250, 300, 350, 400, 450, 500, 550, 650, 700, 800]
    n_trials = [nt for nt in n_trials if nt != n_init]
    n_trials.sort(key=lambda x: abs(x - n_init))
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in n_trials[:8]:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        hb, cb = binary_hill_climb(h_int, n_try, 30000, deadline - 115, seed + n_try)
        h_cand = project_feasible(hb, n_try) if cb < compute_c5(h_int, n_try) else h_int
        h_cd, cd = coord_descent(h_cand, n_try, 400, deadline - 110, seed + n_try)
        if cd < best_c:
            best_h, best_c, x_orig = h_cd, cd, x_new
    # Phase 4: Random binary restarts at best n
    n_cur = len(best_h)
    for r in range(5):
        if time.time() > deadline - 80:
            break
        hb_r, cb_r = random_binary_hc(n_cur, 30000, deadline - 75, seed + 100 + r * 13)
        if cb_r < best_c:
            h_pol = project_feasible(hb_r, n_cur)
            h_pol, cp = coord_descent(h_pol, n_cur, 300, deadline - 70, seed + 200 + r)
            if cp < best_c:
                best_h, best_c = h_pol, cp
    # Phase 5: L-BFGS-B polish on best
    if time.time() < deadline - 50:
        for alpha in [5000, 20000, 50000]:
            if time.time() > deadline - 50:
                break
            try:
                h_new = lbfgs_optimize(best_h, n_cur, alpha, 10.0, 200)
                h_new = project_feasible(h_new, n_cur)
                c_new = compute_c5(h_new, n_cur)
                if c_new < best_c:
                    best_h, best_c = h_new, c_new
            except Exception:
                pass
    # Phase 6: Final CD
    if time.time() < deadline - 5:
        h_f, cf = coord_descent(best_h, len(best_h), 800, deadline - 3, seed + 999)
        if cf < best_c:
            best_h, best_c = h_f, cf
    # Symmetry and final check
    hf = project_feasible(1.0 - best_h, len(best_h))
    cf = compute_c5(hf, len(best_h))
    if cf < best_c:
        best_h, best_c = hf, cf
    n_fin = len(best_h)
    best_h = project_feasible(best_h, n_fin)
    best_c = compute_c5(best_h, n_fin)
    return best_h, best_c, n_fin
