            x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xl2 = max(x_lo, 0.0)
            xh2 = min(x_hi, 1.0)
            if xl2 < r1:
                x_hi = min(x_hi, r1)
            elif xh2 > r2:
                x_lo = max(x_lo, r2)
            else:
                if xl2 >= r1 - 1e-14 and xh2 <= r2 + 1e-14:
                    if h0 - r1 <= r2 - h0:
                        x_hi = min(x_hi, r1 + 1e-9)
                    else:
                        x_lo = max(x_lo, r2 - 1e-9)
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n) if (random_order and rng is not None) else range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def binary_local_search(h, n, max_passes, deadline, rng):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    dx = 2.0 / n
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            old_val = h[p]
            if old_val < 0.01 and old_val > 0.99:
                continue
            best_trial = old_val
            best_val = best_c5
            for trial in (0.0, 1.0):
                if abs(trial - old_val) < 1e-12:
                    continue
                h[p] = trial
                new_c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
                if new_c5 < best_val - 1e-14:
                    best_val = new_c5
                    best_trial = trial
            h[p] = best_trial
            if best_trial != old_val:
                best_c5 = best_val
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    best_c5 = compute_c5(h, n)
    return h, best_c5

def block_init(n, n_blocks):
    half = n_blocks // 2
    best_h = np.full(n, 0.5)
    best_c = compute_c5(best_h, n)
    block_size = n / n_blocks
    for combo in combinations(range(n_blocks), half):
        h = np.zeros(n)
        for b in combo:
            lo = int(b * block_size)
            hi = int((b + 1) * block_size)
            h[lo:hi] = 1.0
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h
    return best_h, best_c

def soft_threshold(h, n, temp):
    centered = h - 0.5
    scaled = centered / max(temp, 1e-6)
    pushed = 0.5 + 0.5 * np.tanh(scaled)
    return project_feasible(pushed, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: CD convergence
    h1, c1 = coord_descent(best_h, n, 150, deadline, rng, True)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Binary local search
    h2, c2 = binary_local_search(best_h, n, 40, deadline, rng)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: CD after binary
    h3, c3 = coord_descent(best_h, n, 150, deadline, rng, True)
    if c3 < best_c:
        best_h, best_c = h3, c3
    # Phase 4: Binary again
    h4, c4 = binary_local_search(best_h, n, 40, deadline, rng)
    if c4 < best_c:
        best_h, best_c = h4, c4
    # Phase 5: Soft thresholding with decreasing temperature
    for temp in (0.25, 0.15, 0.1, 0.07, 0.05, 0.03):
        if time.time() > deadline - 40:
            break
        ht = soft_threshold(best_h, n, temp)
        ht, ct = coord_descent(ht, n, 80, deadline, rng, True)
        if ct < best_c:
            best_h, best_c = ht, ct
        ht, ct = binary_local_search(ht, n, 15, deadline, rng)
        if ct < best_c:
            best_h, best_c = ht, ct
    # Phase 6: Block-based initializations at various resolutions
    for nb in (8, 10, 12, 14, 16):
        if time.time() > deadline - 60:
            break
        hb, cb = block_init(n, nb)
        hb, cb2 = coord_descent(hb, n, 80, deadline, rng, True)
        if cb2 < best_c:
            best_h, best_c = hb, cb2
        hb, cb3 = binary_local_search(hb, n, 15, deadline, rng)
        if cb3 < best_c:
            best_h, best_c = hb, cb3
    # Phase 7: Try different n values
    n_cands = [350, 400, 450, 500, 600, 750, 800]
    for n_try in n_cands:
        if time.time() > deadline - 80:
            break
        if n_try == n:
            continue
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_int, c_int = coord_descent(h_int, n_try, 100, deadline, rng, True)
        h_int, c_int = binary_local_search(h_int, n_try, 20, deadline, rng)
        if c_int < best_c:
            best_h, best_c = h_int, c_int
    # Phase 8: Random restarts near best
    n = len(best_h)
    n_restarts = max(2, int((deadline - time.time() - 40) / 80))
    for r in range(n_restarts):
        if time.time() > deadline - 30:
            break
        s = 0.02 + 0.04 * r
        hp = best_h + rng.normal(0, s, n)
        hp = project_feasible(hp, n)
        hp, cp = coord_descent(hp, n, 60, deadline, rng, True)
        hp, cp = binary_local_search(hp, n, 10, deadline, rng)
        if cp < best_c:
            best_h, best_c = hp, cp
    # Phase 9: Final polish
    n = len(best_h)
    hf, cf = coord_descent(best_h, n, 200, deadline, rng, False)
    if cf < best_c:
        best_h, best_c = hf, cf
    hf, cf = binary_local_search(best_h, n, 20, deadline, rng)
    if cf < best_c:
        best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
