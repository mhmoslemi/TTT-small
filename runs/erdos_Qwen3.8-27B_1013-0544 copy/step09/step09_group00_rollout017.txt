    lse = mx + np.log(np.sum(np.exp(shifted - mx)))

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def _cf(x):
            return x.sum() - n / 2.0
        def _cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
        try:

                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})

            new_c = compute_c5(new_h, n)
            if new_c < best_c:

        except Exception:
            pass
    return best_h, best_c

def subgrad_step(h, n, scale):

    active = c > c.max() - 0.05 * c.max()
    grad = np.zeros(n)

        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib = np.zeros(len(ms))
        if np.any(vf):
            contrib[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):

        contrib[ms == (n - 1)] = (1.0 - 2.0 * h[p])
        mask = active[ms]
        if mask.any():
            grad[p] = np.sum(contrib[mask]) / mask.sum()
    h = np.clip(h - scale * grad, 0.0, 1.0)
    return project_feasible(h, n)

        bounds = [(0.0, 1.0)] * n
        def _cf(x):
            return x.sum() - n / 2.0
        def _cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgrad_step(h, n, scale):
    c = np.correlate(h, 1.0 - h, mode='full')
    active = c > c.max() - 0.05 * c.max()
    grad = np.zeros(n)
    for p in range(n):
        m_min, m_max = n - 1 - p, 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib = np.zeros(len(ms))
        if np.any(vf):
            contrib[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            contrib[vb] -= h[jb[vb]]
        contrib[ms == (n - 1)] = (1.0 - 2.0 * h[p])
        mask = active[ms]
        if mask.any():
            grad[p] = np.sum(contrib[mask]) / mask.sum()
    h = np.clip(h - scale * grad, 0.0, 1.0)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: SLSQP polish
    h1, c1 = slsqp_opt(h_best, n, [5000, 20000, 100000], 150, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: CD with full randomization (fixed bug: use c2 not c1)
    h2, c2 = exact_cd(h_best, n, 200, deadline, rng)
    if c2 < best_c:
        h_best, best_c = h2, c2

    # Phase 3: Shake and settle with adaptive perturbation
    n = len(h_best)
    for r in range(15):
        if time.time() > deadline - 250:
            break
        s = 0.01 + 0.008 * r
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        h_o, c_o = slsqp_opt(hp, n, [10000, 50000], 60, deadline)
        h_o, c_o = exact_cd(h_o, n, 100, deadline, rng)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 4: Subgradient diversification
    if time.time() < deadline - 150:
        for _ in range(8):
            if time.time() > deadline - 100:
                break
            h_sg = subgrad_step(h_best, n, 0.012)
            h_sg = subgrad_step(h_sg, n, 0.008)
            h_o, c_o = exact_cd(h_sg, n, 100, deadline, rng)
            if c_o < best_c:
                h_best, best_c = h_o, c_o

    # Phase 5: Try different n (multi-resolution)
    if time.time() < deadline - 100:
        for n_try in [400, 500, 600, 800, 1000, 450]:
            if time.time() > deadline - 50:
                break
            if n_try == len(h_best):
                continue
            x_old = np.linspace(0, 2, len(h_best))
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
            h_o, c_o = slsqp_opt(h_int, n_try, [5000, 20000], 50, deadline)
            h_o, c_o = exact_cd(h_o, n_try, 80, deadline, rng)
            if c_o < best_c - 1e-7:
                h_best, best_c = h_o, c_o

    # Phase 6: Final CD polish with many passes
    if time.time() < deadline - 5:
        n = len(h_best)
        h_f, c_f = exact_cd(h_best, n, 300, deadline, rng)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
