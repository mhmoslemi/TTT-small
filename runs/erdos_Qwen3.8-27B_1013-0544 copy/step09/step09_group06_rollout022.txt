import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cd_step(h, n, top_k, deadline):
    """Single pass of coordinate descent, updating one coordinate at a time."""
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(c.max())
    k = min(top_k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    c_peak = c[top_idx].astype(float)
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[top_idx] = False
    unaff_max = float(c[mask].max()) if mask.any() else 0.0
    improved = False
    for p in range(n):
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-10 or h0 > 1.0 - 1e-10:
            continue
        a_coeff = np.zeros(k)
        diag_idx = None
        for idx in range(k):
            m = int(top_idx[idx])
            jf = p + m - (n - 1)
            jb = p - m + (n - 1)
            af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
            ab = (-h[jb]) if 0 <= jb < n else 0.0
            a_coeff[idx] = af + ab
            if m == n - 1:
                diag_idx = idx
        # Solve 1D minimax
        lo = -h0
        hi = 1.0 - h0
        # Find best d
        best_d = 0.0
        best_val = current_max
        # Candidate d values
        candidates = [lo, hi, 0.0]
        for i in range(k):
            for j in range(i + 1, k):
                if i == diag_idx or j == diag_idx:
                    continue
                da = a_coeff[i] - a_coeff[j]
                if abs(da) > 1e-15:
                    d = (c_peak[j] - c_peak[i]) / da
                    if lo <= d <= hi:
                        candidates.append(d)
        if diag_idx is not None:
            for i in range(k):
                if i == diag_idx:
                    continue
                B = -(a_coeff[diag_idx] - a_coeff[i])
                C = -(c_peak[diag_idx] - c_peak[i])
                disc = B * B - 4.0 * C
                if disc >= 0:
                    sq = np.sqrt(disc)
                    d1 = (-B + sq) / 2.0
                    d2 = (-B - sq) / 2.0
                    if lo <= d1 <= hi:
                        candidates.append(d1)
                    if lo <= d2 <= hi:
                        candidates.append(d2)
        for i in range(k):
            if abs(a_coeff[i]) > 1e-15:
                d = (unaff_max - c_peak[i]) / a_coeff[i]
                if lo <= d <= hi:
                    candidates.append(d)
        for d in candidates:
            d = float(np.clip(d, lo, hi))
            vals = c_peak + d * a_coeff
            if diag_idx is not None:
                vals[diag_idx] -= d * d
            v = max(float(vals.max()), unaff_max)
            if v < best_val:
                best_val = v
                best_d = d
        if best_val < current_max - 1e-15:
            h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
            improved = True
            current_max = best_val
    return h, improved, current_max

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        h, improved, _ = cd_step(h, n, top_k, deadline)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgradient_method(h, n, deadline, lr_init=0.01, momentum=0.9):
    h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(10000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max()) * (2.0 / n)
        if current_max < best_c:
            best_c = current_max
            best_h = h.copy()
        # Find top few lags for subgradient
        top_idx = np.argsort(c)[::-1][:5]
        # Compute subgradient (averaged over top lags)
        grad = np.zeros(n)
        for m in top_idx:
            m = int(m)
            for p in range(n):
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                gf = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                gb = (-h[jb]) if 0 <= jb < n else 0.0
                grad[p] += gf + gb
        grad /= len(top_idx)
        grad *= (2.0 / n)  # scale by dx
        v = momentum * v - lr * grad
        h = project_feasible(h + v, n)
        lr *= 0.999
    return h, best_c, best_h

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cd_pass(h, n, top_k, deadline):
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(c.max())
    k = min(top_k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    c_peak = c[top_idx].astype(float)
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[top_idx] = False
    unaff_max = float(c[mask].max()) if mask.any() else 0.0
    improved = False
    for p in range(n):
        if time.time() > deadline:
            return h, False
        h0 = h[p]
        if h0 < 1e-10 or h0 > 1.0 - 1e-10:
            continue
        a_coeff = np.zeros(k)
        diag_idx = None
        for idx in range(k):
            m = int(top_idx[idx])
            jf = p + m - (n - 1)
            jb = p - m + (n - 1)
            af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
            ab = (-h[jb]) if 0 <= jb < n else 0.0
            a_coeff[idx] = af + ab
            if m == n - 1:
                diag_idx = idx
        lo = -h0
        hi = 1.0 - h0
        best_d = 0.0
        best_val = current_max
        candidates = [lo, hi, 0.0]
        for i in range(k):
            for j in range(i + 1, k):
                if i == diag_idx or j == diag_idx:
                    continue
                da = a_coeff[i] - a_coeff[j]
                if abs(da) > 1e-15:
                    d = (c_peak[j] - c_peak[i]) / da
                    if lo <= d <= hi:
                        candidates.append(d)
        if diag_idx is not None:
            for i in range(k):
                if i == diag_idx:
                    continue
                B = -(a_coeff[diag_idx] - a_coeff[i])
                C = -(c_peak[diag_idx] - c_peak[i])
                disc = B * B - 4.0 * C
                if disc >= 0:
                    sq = np.sqrt(disc)
                    for d in [(-B + sq) / 2.0, (-B - sq) / 2.0]:
                        if lo <= d <= hi:
                            candidates.append(d)
        for i in range(k):
            if abs(a_coeff[i]) > 1e-15:
                d = (unaff_max - c_peak[i]) / a_coeff[i]
                if lo <= d <= hi:
                    candidates.append(d)
        for d in candidates:
            d = float(np.clip(d, lo, hi))
            vals = c_peak + d * a_coeff
            if diag_idx is not None:
                vals[diag_idx] -= d * d
            v = max(float(vals.max()), unaff_max)
            if v < best_val:
                best_val = v
                best_d = d
        if best_val < current_max - 1e-15:
            h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
            improved = True
            current_max = best_val
    return h, improved

def multi_cd(h, n, max_passes, deadline, top_k):

    for _ in range(max_passes):
        if time.time() > deadline:
            break
        h, improved = cd_pass(h, n, top_k, deadline)
        if not improved:
            break
        h = project_feasible(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def try_symmetry(h, n):

def subgrad_step(h, n, top_k=5):
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[::-1][:top_k]
    grad = np.zeros(n)
    for m in top_idx:
        m = int(m)
        s = m - (n - 1)
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:

        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] += (-h[lo2-s:hi2-s])
    grad /= top_k
    return grad

    n_cur = len(best_h)
    if time.time() < deadline - 10:
        h_f, c_f = multi_cd(best_h, n_cur, 2000, deadline, 40)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Final symmetry and feasibility check
    best_h, best_c = try_symmetry(best_h, len(best_h))
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
