    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):

            if h0 < 1e-12 or h0 > 1.0 - 1e-12:

                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)

                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))

def smooth_minsum(x, n, alpha):
    """Smooth approximation of max(c) using log-sum-exp."""
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def gradient_sliding(h, n, steps, deadline, base_scale=0.02):
    """Subgradient-based method focusing on active constraints."""
    h = h.copy()
    for s in range(steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = c.max()
        # Active set: lags where c is close to max
        threshold = cmax - 0.02 * max(cmax, 1e-10)
        active = c >= threshold
        n_active = max(active.sum(), 1)
        grad = np.zeros(n)
        for p in range(n):
            m_min, m_max = n - 1 - p, 2 * n - 2 - p
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            contrib = np.zeros(len(ms))
            if np.any(vf):
                contrib[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                contrib[vb] -= h[jb[vb]]
            diag_pos = np.where(ms == (n - 1))[0]
            if len(diag_pos) > 0:
                contrib[diag_pos[0]] = (1.0 - 2.0 * h[p])
            mask = active[ms]
            if mask.any():
                grad[p] = np.sum(contrib[mask]) / mask.sum()
        scale = base_scale / (1.0 + 0.1 * s)
        h = np.clip(h - scale * grad, 0.0, 1.0)
        h = project_feasible(h, n)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = float(c_base[diag_idx[0]])
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(45):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    prev_max = float(np.max(c))
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
            prev_max = float(np.max(c))
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_step(h, n, alpha, maxiter, deadline):
    if time.time() > deadline:
        return h, compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n_=n: x.sum() - n_/2.0,
             'jac': lambda x, n_=n: np.ones(n_)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    shifted = alpha * (c - m)
    mx = float(shifted.max())
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_step(h, n, alpha, maxiter, deadline):
    if time.time() > deadline:
        return h, compute_c5(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq',
                 'fun': lambda x, n_=n: x.sum() - n_ / 2.0,
                 'jac': lambda x, n_=n: np.ones(n_)}]
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(350, 0.5)
        n = 350

    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Exact CD polish (fixed: use c2 not c1)
    h1, c1 = exact_cd(h_best, n, 200, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: SLSQP escape + CD
    if time.time() < deadline - 200:
        for alpha in [30000, 100000]:
            if time.time() > deadline - 200:
                break
            h_s, c_s = slsqp_step(h_best, len(h_best), alpha, 200, deadline)
            h_s, c_s = exact_cd(h_s, len(h_s), 80, deadline)
            if c_s < best_c:
                h_best, best_c = h_s, c_s

    # Phase 3: Multi-n exploration
    n_list = [400, 500, 600, 800, 300, 450, 550, 700]
    for n_try in n_list:
        if time.time() > deadline - 300:
            break
        if n_try == len(h_best):
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best, clip=0.0), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 100, deadline)
        if c_o < best_c - 1e-7:
            h_best, best_c = h_o, c_o

    # Phase 4: Diverse random restarts with CD
    n = len(h_best)
    for r in range(35):
        if time.time() > deadline - 100:
            break
        s = 0.015 + 0.025 * ((r * 7) % 5) / 4.0
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        h_o, c_o = exact_cd(hp, n, 60, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o
            n = len(h_best)

    # Phase 5: Structured perturbations
    n = len(h_best)
    if time.time() < deadline - 80:
        h_ant = project_feasible(1.0 - h_best[::-1], n)
        h_o, c_o = exact_cd(h_ant, n, 80, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o
    if time.time() < deadline - 60:
        h_ph = project_feasible(np.roll(h_best, n // 4), n)
        h_o, c_o = exact_cd(h_ph, n, 80, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o
    if time.time() < deadline - 50:
        h_ph2 = project_feasible(np.roll(h_best, n // 3), n)
        h_o, c_o = exact_cd(h_ph2, n, 60, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 6: One more SLSQP escape on the best
    if time.time() < deadline - 40:
        h_s, c_s = slsqp_step(h_best, len(h_best), 80000, 150, deadline)
        h_s, c_s = exact_cd(h_s, len(h_s), 60, deadline)
        if c_s < best_c:
            h_best, best_c = h_s, c_s

    # Phase 7: Final CD polish
    if time.time() < deadline - 3:
        h_f, c_f = exact_cd(h_best, len(h_best), 150, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
